const STACK_FRAME_RE = /^\s*at .*(?::\d+:\d+\)?|\((?:native|<anonymous>)\))\s*$/;
function stackFrames(stack: string, header: string): string {
  const body = stack.startsWith(header) ? stack.slice(header.length) : stack;
  return body.split("\n").filter((l) => STACK_FRAME_RE.test(l)).map((l) => l.trim()).join(" | ");
}

/** I4-07-R1 hand-off (I3-2012): the runtime shape of `OutboxMessage`. Nullable
 * columns accept an absent field (read as null by the consumer). */
function outboxBodyShaped(b: Partial<OutboxMessage> | null): boolean {
  if (!b) return false;
  const strOrNull = (v: unknown) => v == null || typeof v === "string";
  const intOrNull = (v: unknown) => v == null || Number.isSafeInteger(v);
  return (
    Number.isSafeInteger(b.seq) &&
    typeof b.passport_id === "string" &&
    typeof b.event_type === "string" &&
    typeof b.occurred_day === "string" &&
    strOrNull(b.fan_key) &&
    strOrNull(b.scope_kind) &&
    strOrNull(b.scope_id) &&
    strOrNull(b.dims) &&
    intOrNull(b.fan_key_v) &&
    intOrNull(b.amount_cents)
  );
}

/** I4-05-R8 hand-off (I3-0898): the one-shot legacy top-fans purge. The done
 * marker is written only after a complete pass, so a failed pass retries on
 * the next sweep. A psp is "boarded" if its id appears anywhere in the
 * discovery artifact (a superset, so the purge can only under-empty). */
export const TOPFANS_LEGACY_PURGE_MARKER = "rankings/topfans-legacy-purge.done";
/** I4-RG2-RANK lead: the last key a bounded pass reached. A pass that hits the
 * page bound saves it and the next sweep lists `startAfter` it, so a bucket over
 * TOPFANS_LEGACY_PURGE_PAGES pages is finished across sweeps instead of re-scanning
 * its first pages forever. Deleted with the marker's write. */
export const TOPFANS_LEGACY_PURGE_PROGRESS = "rankings/topfans-legacy-purge.progress";
export const TOPFANS_LEGACY_PURGE_PAGES = 50;
/** I14-FX2 (QUEUE #912, I14-CAPS HANDOFF-2): the R2 calls one pass may make — its three reads, its list pages, its
 * puts and the closing write(s); a pass that reaches it saves its place in TOPFANS_LEGACY_PURGE_PROGRESS. Its share
 * of ARTIFACTS_LEGS_AFTER_PURGE. */
export const TOPFANS_LEGACY_PURGE_BUDGET = 20;
export async function purgeLegacyTopFans(bucket: R2Bucket): Promise<number> {
  if (await bucket.get(TOPFANS_LEGACY_PURGE_MARKER)) return 0;
  const discovery = await bucket.get("rankings/discovery.json");
  if (!discovery) return 0; // nothing built yet: no boarded set to compare against
  const boarded = new Set((await discovery.text()).match(/psp_[0-9a-f]{16}/g) ?? []);
  const generatedAt = new Date().toISOString();
  const saved = await bucket.get(TOPFANS_LEGACY_PURGE_PROGRESS);
  const resumeAfter = saved ? (await saved.text()) || undefined : undefined;
  let emptied = 0;
  let cursor: string | undefined;
  let lastKey = resumeAfter;
  let spent = 3; // the marker, discovery and progress reads
  for (let page = 0; page < TOPFANS_LEGACY_PURGE_PAGES; page++) {
    if (spent + 1 + 2 > TOPFANS_LEGACY_PURGE_BUDGET) break; // a list, then room to close
    const listed = await bucket.list(
      cursor ? { prefix: "rankings/topfans/", cursor } : resumeAfter ? { prefix: "rankings/topfans/", startAfter: resumeAfter } : { prefix: "rankings/topfans/" },
    );
    spent++;
    for (const o of listed.objects) {
      if (spent + 1 + 2 > TOPFANS_LEGACY_PURGE_BUDGET) {
        if (lastKey !== undefined) await bucket.put(TOPFANS_LEGACY_PURGE_PROGRESS, lastKey);
        return emptied;
      }
      lastKey = o.key;
      const psp = /^rankings\/topfans\/(psp_[0-9a-f]{16})\.json$/.exec(o.key)?.[1];
      if (!psp || boarded.has(psp)) continue;
      await bucket.put(o.key, JSON.stringify({ passport_id: psp, generated_at: generatedAt, fans: [] }));
      spent++;
      emptied++;
    }
    if (!listed.truncated) {
      await bucket.put(TOPFANS_LEGACY_PURGE_MARKER, generatedAt);
      if (saved) await bucket.delete(TOPFANS_LEGACY_PURGE_PROGRESS);
      return emptied;
    }
    cursor = (listed as { cursor?: string }).cursor;
  }
  // page bound hit: no marker; save the place so the next sweep continues from it
  if (lastKey !== undefined) await bucket.put(TOPFANS_LEGACY_PURGE_PROGRESS, lastKey);
  return emptied;
}

/** I4-03-R8 hand-off (I3-1975): the face-triage gate as one testable predicate —
 * the operator setting AND the binding AND a usable internal secret. Any
 * missing term leaves the sweep on the human triage path.
 * D-0911(4B) (I3-0027): the exact-photo check needs no face model, so it stays
 * automatic and the gate no longer waits on one. The image service decides the
 * stages: `/internal/triage-find` runs the embedding stage only when its own
 * embedder reports a real, wired model. `embedderWired` is that service's
 * readiness answer (`checks.embedder_wired === true`, services/image/src/http.ts
 * `/health?ready=1`), read on every call and false on a throw, a timeout, a
 * non-2xx, an unparseable body or a missing/non-boolean field; the sweep only
 * logs it. */
export async function faceTriageArmed(
  setting: string | null | undefined,
  env: Pick<Env, "IMAGE_SERVICE" | "FACE_TRIAGE_SECRET">,
  probeTimeoutMs = 5_000,
): Promise<{ svc: NonNullable<Env["IMAGE_SERVICE"]>; secret: string; embedderWired: boolean } | null> {
  if (setting !== "1" || !env.IMAGE_SERVICE) return null;
  const secret = faceTriageSecretFromEnv(env); // >= 16 characters, not blank, no TODO (INV-091; admin.ts)
  if (secret === null) return null;
  const svc = env.IMAGE_SERVICE;
  const ctrl = new AbortController();
  const timer = setTimeout(() => ctrl.abort(), probeTimeoutMs);
  const probe = (async (): Promise<boolean> => {
    const res = await svc.fetch("https://image.internal/health?ready=1", { method: "GET", redirect: "manual", signal: ctrl.signal }); // #611 (D-0930(3))
    if (!res.ok) return false;
    const body = (await res.json()) as { checks?: { embedder_wired?: unknown } } | null;
    return body?.checks?.embedder_wired === true;
  })();
  // The race bounds the body read too, and a double that ignores the signal.
  const timedOut = new Promise<boolean>((resolve) => ctrl.signal.addEventListener("abort", () => resolve(false)));
  let wired = false;
  try {
    wired = await Promise.race([probe, timedOut]);
  } catch {
    wired = false;
  } finally {
    clearTimeout(timer);
    probe.catch(() => {});
  }
  return { svc, secret, embedderWired: wired };
}

/** #609a: the face-triage sweep's call to the image service. Every failure is
 * still `null` (stage unavailable — the human path), but the two kinds are now
 * told apart in the log: a non-2xx answer is `image_triage_status` with the status
 * (503 unconfigured, 4xx unprocessable), a throw or the 10 s abort is
 * `image_triage_error`, and an answer with no boolean `matched` is
 * `image_triage_bad_body`. */
export function faceTriageMatcher(svc: NonNullable<Env["IMAGE_SERVICE"]>, secret: string, cid: string): FaceMatcher {
  return async (bytes, contentType) => {
    // Bound the internal call so a hung image service can't stall the
    // sweep item indefinitely (review hardening #3).
    const ctrl = new AbortController();
    const timer = setTimeout(() => ctrl.abort(), 10_000);
    try {
      const res = await svc.fetch("https://image.internal/internal/triage-find", {
        method: "POST",
        headers: { "X-Internal-Auth": secret, "Content-Type": contentType ?? "application/octet-stream" },
        body: bytes,
        // #611 (D-0930(3)): never follow — a followed 3xx would carry X-Internal-Auth to wherever it points.
        redirect: "manual",
        signal: ctrl.signal,
      });
      if (!res.ok) {
        logError({ cid, hop: "cron", msg: "image_triage_status", status: res.status });
        return null;
      }
      const body = (await res.json().catch(() => null)) as FaceMatch | null;
      if (body && typeof body.matched === "boolean") return body;
      logError({ cid, hop: "cron", msg: "image_triage_bad_body", status: res.status });
      return null;
    } catch (e) {
      logError({ cid, hop: "cron", msg: "image_triage_error", err: e instanceof Error ? e.name : typeof e });
      return null;
    } finally {
      clearTimeout(timer);
    }
  };
}

export async function drainClosedCreatorFaceErasures(db: DbLike, cid: string, inBudget: () => boolean): Promise<{ calls: number; owed: number; more: boolean }> {
  const { oweClosedCreatorFaceErasures } = await import("./creator-close.js");
  let calls = 0;
  let owed = 0;
  for (;;) {
    const r = await oweClosedCreatorFaceErasures(db);
    calls++;
    owed += r.owed;

    if (!r.more || r.owed === 0 || !inBudget()) return { calls, owed, more: r.more };
  }
}

/**

 * alarm markers this instance buffered are mailed to the operators (OPERATOR_EMAILS), once per marker per UTC hour, under
 * ctx.waitUntil. Nothing buffered: no env read, no DB touch. Never throws into the handler.
 * I34-FX2-ALARM (I34-RG-W2M TC:16): when the handler's work handed promises to waitUntil (`deferred`, operator-alarm.ts
 * trackWaitUntil), the flush runs after they settle, so a marker that deferred work logs is mailed by this invocation.
 */
export function flushOperatorAlarmsAfter(env: Env, ctx?: { waitUntil(p: Promise<unknown>): void }, deferred?: Pick<DeferredWork<unknown>, "handed" | "settled">): void {
  const flush = (): Promise<unknown> => {
    try {
      if (pendingOperatorAlarms().length === 0) return Promise.resolve();
      const e = env as unknown as Record<string, unknown>;
      return flushOperatorAlarms(
        d1(env.TRUST_DB),
        { send: httpEmailFromEnv(e), mailer: mailerFromEnv(e), html: (t) => wrapHtml(textToHtml(t), "") },
        accessConfigFromEnv(e)?.operatorEmails ?? [],
      );
    } catch {
      return Promise.resolve();
    }
  };
  try {
    if (ctx && deferred && deferred.handed() > 0) {
      ctx.waitUntil(deferred.settled().then(flush));
      return;
    }
    if (pendingOperatorAlarms().length === 0) return;
    ctx?.waitUntil(flush());
  } catch {
    // the record of an alarm must never fail the handler it rode on
  }
}

     * the duplicate push and duplicate mail the lease exists to prevent.

     * A holder that has lost its lease has no business continuing, so it is
     * asked between the expensive legs. One holder- and expiry-guarded
     * statement (it renews, D-0844(9) below), and it fails SAFE: an unreadable answer stops this run rather than letting two run.
     */
    // D-0844(9) (rg2c-03): each fence RENEWS rather than reads — renewCronLease is
    // one holder- and expiry-guarded UPDATE, so "still mine" and "extended" are the
    // same statement — and, as withCronLease does, a timer renews every TTL/3 while
    // a single leg is in flight. Both stop in the `finally` below, before release.
    // RGOPS-04 (D-1154(2)): a renewal the timer below saw refused or thrown marks the lease lost for the rest of the
    // run, and every later fence answers false without asking again.
    let leaseLost = false;
    // I41-LEASE (I40FX2L-02): a renewal a FENCE saw refused (no row changed; it cannot later succeed) is recorded too, for
    // the retracted lap, which asks no fence of its own; the fence's own answer is unchanged.
    let fenceLost = false;
    // I43-LEASE (D-1248(3), W0-LS-02): a renewal a fence saw THROW is a lost lease: it latches `leaseLost`, as the timer's
    // throw does, so every later fence answers false without asking again (and the retracted lap, which reads
    // `leaseLost || fenceLost`, stops); logged once, by error kind.
    const fenceThrew = (e: unknown): false => {
      leaseLost = true;
      logError({ cid: lease.holder, hop: "cron", msg: "scheduled_lease_renewal_threw", err: e instanceof Error ? e.name : typeof e });
      return false;
    };
    // I46-FX2 (I43-RJ-08/FINDINGS.jsonl:2): on a manual run the fence renews the head lease too, as the timer does, so a
    // head lease another live holder took ends the run at the next fence; a throw from either renewal is fenceThrew's.
    const stillLeased = async (): Promise<boolean> => {
      if (leaseLost) return false;
      try {
        const ok = (await renewCronLease(db, lease, Date.now(), leaseTtlMs)) && (headLease === null || (await renewCronLease(db, headLease, Date.now(), CRON_LEASE_TTL_MS)));
        if (!ok) fenceLost = true;
        return ok;
      } catch (e) {
        return fenceThrew(e);
      }
    };
    let leaseStopped = false;
    let leaseTimer: ReturnType<typeof setTimeout> | undefined;
    const scheduleLeaseRenew = (): void => {
      if (leaseStopped) return;
      leaseTimer = setTimeout(() => {
        Promise.all([renewCronLease(db, lease, Date.now(), leaseTtlMs), headLease === null ? true : renewCronLease(db, headLease, Date.now(), CRON_LEASE_TTL_MS)])
          .then(
            ([own, head]) => {
              if (!own || !head) leaseLost = true;
            },
            () => {
              leaseLost = true;
            },
          )
          .finally(scheduleLeaseRenew);
      }, leaseRenewMs);
    };
    scheduleLeaseRenew();
    try {
    /**
     * ONE ISOLATED STEP. Each item below is a single bounded statement or two
     * against an independent table, so a failure in one says nothing about the
     * next — and before this the whole run was one unguarded sequence, which is
     * how a single throw took fourteen prunes with it. Logged by name, never
     * fatal, and the names stay literal at the call sites so the cron-
     * reachability guard still sees every sweep it is meant to see.
     */
    const runStep = async (name: string, fn: () => Promise<unknown>): Promise<void> => {
      try {
        await fn();
      } catch (e) {
        logError({ cid: lease.holder, hop: "cron", msg: "hygiene_step_failed", detail: name, err: e instanceof Error ? e.name : typeof e });
      }
    };
    /**
     * I3-0182 — THE HYGIENE BLOCK IS A RESUMABLE ROTATION, not a fixed run from
     * the top. While `hygieneSteps` is non-null, `step` only COLLECTS; the
     * rotation (`runHygieneRotation`, after the last hygiene site) runs them from
     * the `sweep_cursor` row HYGIENE_ROTATION_CURSOR, wrapping round, writing
     * cursor = i+1 BEFORE step i (an invocation killed inside step i resumes at
     * i+1, so one step that never finishes cannot starve the rest), stopping at
     * HYGIENE_BUDGET_MS so the legs below keep their share of the invocation, and
     * resetting the cursor to 0 once every step ran (canonical order again).
     * Later `step` sites (after `hygieneSteps = null`) run immediately.
     *
     * I12-RG-PRIVACY-06 — A STEP THAT KILLS THE INVOCATION IS HELD, NOT RETRIED
     * FIRST. Resuming at i+1 still reached the killer every tick (last, after the
     * wrap) and before the legs below, so those legs never ran again. The cursor
     * write before step i also records the step's NAME (cursor_id) and the
     * rotation clears it when it ends (finished or deferred); a tick that finds a
     * name left behind knows that step's invocation died inside it, logs
     * `hygiene_step_killed`, and holds that step for HYGIENE_SUSPECT_HOLD_TICKS
     * ticks (row HYGIENE_SUSPECT_CURSOR) before retrying it. Keyed by name, so a
     * conditional step that shifts the indices cannot misdirect the hold.
     */
    let hygieneSteps: { name: string; fn: () => Promise<unknown> }[] | null = [];
    /** Q974-B15: when the rotation began, so a step can spend what is left of HYGIENE_BUDGET_MS. 0 = no rotation running. */
    let hygieneBegan = 0;
    const step = async (name: string, fn: () => Promise<unknown>): Promise<void> => {
      if (hygieneSteps) hygieneSteps.push({ name, fn });
      else await runStep(name, fn);
    };
    const runHygieneRotation = async (): Promise<void> => {
      const steps = hygieneSteps ?? [];
      hygieneSteps = null;
      const n = steps.length;
      if (n === 0) return;
      // I43-FX-LEASE (D-1254(2), D-1255 D2): every hygiene cursor write is this ONE statement. I47-CODE (I47-B02/B05/B30,
      // D-1315 R2): it writes only while the rotation's head-lease row is still THIS run's (name and holder), lapsed or not;
      // a successor's acquire overwrites that row (cron-lease.ts:69-77) and its release deletes it (:87), so a stale run's
      // write is refused in both cases, and a lapsed lease with no successor keeps its row and still writes. I43-FX2-LEASE
      // (D-1257(3)): the head lease only (HEAD_LEASE on a manual run and on the head cron); `scheduled` alone does not block,
      // since the five-minute delivery tick takes it and never runs the rotation.
      const head = headLease ?? lease;
      const writeRow = async (name: string, seq: number, id: string): Promise<void> => {
        try {
          const nowIso = new Date().toISOString();
          await db
            .prepare("INSERT INTO sweep_cursor (name, cursor_seq, cursor_id, updated_at) SELECT ?, ?, ?, ? WHERE EXISTS (SELECT 1 FROM cron_leases WHERE name = ? AND holder = ?) ON CONFLICT(name) DO UPDATE SET cursor_seq = excluded.cursor_seq, cursor_id = excluded.cursor_id, updated_at = excluded.updated_at")
            .bind(name, seq, id, nowIso, head.name, head.holder)
            .run();
        } catch (e) {
          logError({ cid: lease.holder, hop: "cron", msg: "hygiene_rotation_cursor", err: e instanceof Error ? e.name : typeof e });
        }
      };
      /** `inFlight` is the step about to run (its name), or "" once the rotation has ended. */
      const setCursor = (next: number, inFlight: string): Promise<void> => writeRow(HYGIENE_ROTATION_CURSOR, next, inFlight);
      let start = 0;
      let held = "";
      try {
        const { results } = await db
          .prepare("SELECT name, cursor_seq, cursor_id FROM sweep_cursor WHERE name IN (?, ?)")
          .bind(HYGIENE_ROTATION_CURSOR, HYGIENE_SUSPECT_CURSOR)
          .all<{ name: string; cursor_seq: number; cursor_id: string | null }>();
        const rot = results.find((r) => r.name === HYGIENE_ROTATION_CURSOR);
        const sus = results.find((r) => r.name === HYGIENE_SUSPECT_CURSOR);
        const c = Number(rot?.cursor_seq ?? 0);
        start = Number.isInteger(c) && c >= 0 ? c % n : 0;
        let suspect = sus?.cursor_id ?? "";
        let holds = Number(sus?.cursor_seq ?? 0);
        const died = rot?.cursor_id ?? "";
        if (died !== "") {
          logError({ cid: lease.holder, hop: "cron", msg: "hygiene_step_killed", detail: died });
          suspect = died;
          holds = HYGIENE_SUSPECT_HOLD_TICKS;
        }
        if (suspect !== "" && Number.isInteger(holds) && holds > 0) {
          held = suspect;
          await writeRow(HYGIENE_SUSPECT_CURSOR, holds - 1, suspect);
        }
      } catch (e) {
        logError({ cid: lease.holder, hop: "cron", msg: "hygiene_rotation_cursor", err: e instanceof Error ? e.name : typeof e });
      }
      const began = Date.now();
      hygieneBegan = began;
      for (let k = 0; k < n; k++) {
        if (k > 0 && Date.now() - began >= HYGIENE_BUDGET_MS) {
          logError({ cid: lease.holder, hop: "cron", msg: "hygiene_rotation_deferred", count: n - k, detail: steps[(start + k) % n]!.name });
          await setCursor((start + k) % n, "");
          return;
        }
        const i = (start + k) % n;
        if (steps[i]!.name === held) {
          logError({ cid: lease.holder, hop: "cron", msg: "hygiene_step_held", detail: held });
          continue;
        }
        // RGOPS-04 (W0 amendment 6): no step starts once the lease is lost; the cursor resumes at this step. I43-FX-LEASE
        // (D-1254(2)): writeRow writes nothing once the HEAD lease row is not this run's (I47-CODE; I43-FX2-LEASE, D-1257(3)), so a
        // lapsed lease with no successor, or a `scheduled` lease the delivery tick took while the head lease stays this run's,
        // clears the in-flight name here (the step before this one finished).
        if (!(await stillLeased())) {
          logError({ cid: lease.holder, hop: "cron", msg: "hygiene_rotation_lease_lost", count: n - k, detail: steps[i]!.name });
          await setCursor(i, "");
          return;
        }
        await setCursor((i + 1) % n, steps[i]!.name);
        await runStep(steps[i]!.name, steps[i]!.fn);
      }
      await setCursor(0, "");
    };
    /**
     * I13-CRON (QUEUE #900) — ONE TAIL LEG. It runs only when its group runs
     * (`tailOnly` null is a manual run: every group). The leg's name is written to
     * its group's in-flight row BEFORE it runs and cleared after it, so a tick that
     * finds a name left behind knows that leg's invocation died inside it: it logs
     * `tail_leg_killed` and holds the leg for HYGIENE_SUSPECT_HOLD_TICKS ticks (the
     * I12-RG-PRIVACY-06 rule of the hygiene rotation, keyed by name), and the legs
     * after it run on those ticks. A thrown leg is logged here and costs nothing else.
     */
    const runs = (g: TailGroup): boolean => tailOnly === null || tailOnly === g;
    // I13-FX1 (I13-RG-OPS-04, D-0889(3)): the artifacts trigger's legs run on METERED handles
    // (every D1 statement and R2 operation charged before it is sent, refused past
    // TAIL_ARTIFACTS_LEG_BUDGET), so the invocation keeps TAIL_OVERHEAD_SUBREQUESTS for its
    // own scaffolding under the ceiling. A manual run (no cron) is unmetered, as before.
    const art = tailOnly === "artifacts" ? tailMeter(TAIL_ARTIFACTS_LEG_BUDGET, db, env) : { metered: false as const, db, env, store, exhausted: () => false, spent: () => 0, share: (_c?: number): void => {}, left: () => TAIL_ARTIFACTS_LEG_BUDGET };
    let artCut = false;
    let artCarry: string | undefined;
    let artFirst: string | undefined;
    // I47-FXR-CODE (F7, D-1319 C1): the tail legs' and the chain verdict's cursors under the hygiene rotation's guard lease.
    const writeCursorRow = (name: string, seq: number, id: string): Promise<void> => writeSweepCursorRow(db, lease.holder, name, seq, id, headLease ?? lease);
    const tailHeld = new Map<TailGroup, string>();
    // I13-FX1 (I13-RG-OPS-01, D-0889(2)): the hold RELEASES after its ticks whatever the
    // group's size. A fresh kill clears the in-flight row here, at detection, instead of
    // waiting for a later leg of the group to run (a one-leg group has none). And an
    // in-flight name equal to the suspect, last written NO LATER than the suspect row, is
    // the kill already detected (its clear failed): it never re-arms the hold. A real
    // re-kill rewrote the in-flight row when the leg ran again, after the hold was spent.
    const tailHoldOf = async (g: TailGroup): Promise<string> => {
      try {
        const { results } = await db
          .prepare("SELECT name, cursor_seq, cursor_id, updated_at FROM sweep_cursor WHERE name IN (?, ?)")
          .bind(tailInflightCursor(g), tailSuspectCursor(g))
          .all<{ name: string; cursor_seq: number; cursor_id: string | null; updated_at: string | null }>();
        const inflight = results.find((r) => r.name === tailInflightCursor(g));
        const died = inflight?.cursor_id ?? "";
        const sus = results.find((r) => r.name === tailSuspectCursor(g));
        let suspect = sus?.cursor_id ?? "";
        let holds = Number(sus?.cursor_seq ?? 0);
        const stale = died !== "" && died === suspect && Date.parse(inflight?.updated_at ?? "") <= Date.parse(sus?.updated_at ?? "");
        if (died !== "" && !stale) {
          logError({ cid: lease.holder, hop: "cron", msg: "tail_leg_killed", detail: `${g}:${died}` });
          suspect = died;
          holds = HYGIENE_SUSPECT_HOLD_TICKS;
        }
        if (died !== "") await writeCursorRow(tailInflightCursor(g), 0, "");
        if (suspect !== "" && Number.isInteger(holds) && holds > 0) {
          await writeCursorRow(tailSuspectCursor(g), holds - 1, suspect);
          return suspect;
        }
      } catch (e) {
        logError({ cid: lease.holder, hop: "cron", msg: "tail_cursor", detail: g, err: e instanceof Error ? e.name : typeof e });
      }
      return "";
    };
    const tailLeg = async (g: TailGroup, name: string, fn: () => Promise<unknown>): Promise<void> => {
      if (!runs(g)) return;
      // I13-FX1 (I13-RG-OPS-04): a METERED group (artifacts, on its own trigger) stops at its
      // budget and carries the rest: the run after starts at the carried leg.
      // I13-FX7 (QUEUE #515, D-0653): a PINNED leg (the retraction purge) is never carried and
      // never cut: it runs on every metered tick, on the budget buildSweep's share left it.
      const pinned = g === "artifacts" && ARTIFACT_PINNED_LEGS.includes(name);
      if (art.metered && g === "artifacts" && !pinned) {
        if (artCut) {
          logError({ cid: lease.holder, hop: "cron", msg: "tail_leg_carried", detail: `${g}:${name}` });
          return;
        }
        if (artCarry === undefined) artCarry = await readTailCarry(db, g);
        if (artCarry !== "" && artCarry !== name) {
          logError({ cid: lease.holder, hop: "cron", msg: "tail_leg_carried", detail: `${g}:${name}` });
          return;
        }
        if (artCarry === name) {
          artCarry = "";
          await writeCursorRow(tailCarryCursor(g), 0, "");
        }
        artFirst ??= name;
      }
      let held = tailHeld.get(g);
      if (held === undefined) {
        held = await tailHoldOf(g);
        tailHeld.set(g, held);
      }
      if (held === name) {
        logError({ cid: lease.holder, hop: "cron", msg: "tail_leg_held", detail: `${g}:${name}` });
        return;
      }
      await writeCursorRow(tailInflightCursor(g), 0, name);
      try {
        await fn();
      } catch (e) {
        logError({ cid: lease.holder, hop: "cron", msg: "tail_leg_failed", detail: `${g}:${name}`, err: e instanceof Error ? e.name : typeof e });
      }
      await writeCursorRow(tailInflightCursor(g), 0, "");
      if (art.metered && g === "artifacts" && !artCut && art.exhausted()) {
        // The leg the budget cut is retried first next run — unless it was this run's
        // FIRST leg (it alone outspends the budget), then the next leg is, so the group turns.
        artCut = true;
        // A pinned leg is never the carry: the next unpinned leg after it is.
        const i = ARTIFACT_TAIL_LEGS.indexOf(name);
        const carry = !pinned && name !== artFirst ? name : nextUnpinnedArtifactLeg(i + 1);
        await writeCursorRow(tailCarryCursor(g), 0, carry);
        logError({ cid: lease.holder, hop: "cron", msg: "tail_budget_carried", detail: `${g}:${name}->${carry}`, spent: art.spent() });
      }
    };
    /** I13-CRON: the head records its chain verdict for the tail triggers (see TAIL_CHAIN_VERDICT_CURSOR). */
    const writeChainVerdict = (v: "pending" | "ok" | "broken"): Promise<void> => writeCursorRow(TAIL_CHAIN_VERDICT_CURSOR, 0, v);
    const chainVerdictFresh = (): Promise<boolean> => tailChainGateOpen(db, env, Date.now());
    // Captured at the head's verdict-removing sweeps (below); a tail-only run uses its own start.
    let nowMs = Date.now();
    // I13-CRON: THE HEAD — everything down to the hygiene rotation. A tail-only run skips it.
    if (tailOnly === null) {
    // I13-FX1 (I13-RG-OPS-02, D-0889(2)): 'pending' FIRST, so a head that throws before
    // its verdict (a failed chain read, a failed 'broken' write) leaves no 'ok' behind for
    // the tails of this cycle; tailChainGateOpen also refuses an 'ok' from an earlier cycle.
    await writeChainVerdict("pending");
    // 1. Internal chain integrity. Corrupt log → do NOT rebuild or re-anchor.
    const chain = await verifyChainTick(db); // I3-2157 + QUEUE #903: bounded per tick; the rolling full walk (`VERIFY_CHAIN_WINDOW_TICKS` ticks) resumes across ticks (checkpoint.ts)
    if (!chain.ok) {
      logError({ cid: lease.holder, hop: "cron", msg: "ledger_chain_corrupt", ...chainLogFields(chain) });
      await markLedgerBroken(env, { breach: "chain", at: new Date().toISOString(), detail: chain }, lease.holder);
      await writeChainVerdict("broken");
      throw new Error(`ledger chain corrupt at seq ${chain.brokenAtSeq}: ${chain.reason}`);
    }

    // 2. Anchor check. Proven tamper → hard stop. Missing anchor → chain is
    //    sound, so proceed + re-export (a bad anchor file can't freeze rebuilds).
    //    Item 12: `keys` is the PUBLISHED verification set. Absent ⇒ the
    //    signature axis reports "disabled" and this behaves exactly as it did
    //    before signing existed.
    //
    //    WHAT STOPS THE WORLD, decided explicitly (see checkpoint.ts BreachAxis).
    //    "broken" is the right VERDICT for every one of these — the console and
    //    /v1/info are unchanged and still say broken — but halting the safety
    //    machinery on all of them is a SEPARATE decision, and it was previously
    //    being made by accident:
    //
    //      breach "chain"  / "ledger" → STOP. The log does not recompute, or an
    //        anchor's pin disagrees with it, or nothing signed vouches for it at
    //        all. Everything below publishes something DERIVED from the log —
    //        the registry buckets fans read, the rankings, the digests — and
    //        publishing from a log that may have been rewritten launders the
    //        rewrite. Refusing to publish is the whole point of the anchor.
    //
    //      breach "store" → ALARM, and RUN. The log recomputes, a signature
    //        under a published key vouches for it, every signed anchor agrees —
    //        and some other object under `checkpoints/` does not verify. That is
    //        a fact about R2, not about D1: one bad restore, one partial write,
    //        one key re-issued under a reused key_id. Nothing about the log
    //        became less true, so the registry rebuild, the alert/email
    //        delivery, the expiry sweep and the payout-tag promotion have the
    //        same inputs they had a minute ago. Stopping them instead traded a
    //        storage fault for a fan-safety outage — fans pinned to a stale
    //        bucket and newly published warnings never reaching them, with no
    //        recovery path in code — and handed anyone who can write ONE junk
    //        object into the checkpoint prefix a permanent kill switch on the
    //        safety pipeline, at strictly less access than writing D1.
    //
    //    Recovery for BOTH stopping axes is an operator act, not an automatic
    //    one, and it is written down: DEPLOY.md §3e (quarantine the object,
    //    never delete it — the bytes are the evidence).
    //
    //    A READ FAILURE IS NOT A BREACH — the catch policy, decided explicitly.
    //    This whole leg reaches R2, and R2 can fail for reasons that say nothing
    //    whatever about the log: a listing error, a GET that times out, a
    //    transient 5xx. Until this change the call was UNCAUGHT and sat ABOVE the
    //    registry rebuild, so any one of those threw the cron and took the
    //    rebuild, the delivery legs, the expiry sweep and fourteen retention
    //    prunes with it — fans pinned to a stale bucket and newly published
    //    warnings never reaching them. That is the same reasoning DEPLOY.md §3e
    //    already applies to `breach === "store"` (a fact about R2 is not a fact
    //    about D1), so it gets the same answer: LOG LOUDLY AND CONTINUE. What is
    //    NOT swallowed is a genuine verdict — `breach` "chain"/"ledger" still
    //    throws below and still stops the world, because that IS a fact about
    //    the log. The chain itself was already verified from D1 above, so
    //    continuing past a failed anchor read never publishes from an
    //    unverified log.
    const cfg = env as unknown as Record<string, unknown>;
    const keys = checkpointKeysFromEnv(cfg);
    const anchorCursor = (await db.prepare("SELECT cursor_seq FROM sweep_cursor WHERE name = 'checkpoint_anchors'").first<{ cursor_seq: number }>())?.cursor_seq ?? 0;
    let verdict = null as Awaited<ReturnType<typeof verifyAgainstCheckpoint>> | null;
    try {
      // `chain` is handed in rather than re-derived: `verifyAgainstCheckpoint`
      // called `verifyChain(db)` itself, so every cron run walked the entire
      // ledger TWICE — one D1 query per 1000 rows plus a SHA-256 per row, each
      // time — and the first walk's result was used only for an early-out the
      // second one re-computed. At 2M rows that was 4,000 queries and 4M hashes
      // spent before the sweep did anything useful.
      verdict = await verifyAgainstCheckpoint(db, store, keys, { chain, tailCursor: anchorCursor, deploymentId: checkpointDeploymentIdFromEnv(cfg) });
    } catch (e) {
      logError({ cid: lease.holder, hop: "cron", msg: "CHECKPOINT VERIFICATION UNAVAILABLE", err: e instanceof Error ? e.name : typeof e });
    }
    if (verdict) {
      if (verdict.anchored && !verdict.ok && verdict.breach !== "store") {
        logError({ cid: lease.holder, hop: "cron", msg: "ledger_tamper_detected", ...anchorLogFields(verdict) });
        await markLedgerBroken(env, { breach: verdict.breach ?? "anchor", at: new Date().toISOString(), detail: verdict }, lease.holder);
        await writeChainVerdict("broken");
        throw new Error(`ledger tamper: ${verdict.anchorReason ?? verdict.signatureReason}`);
      }
      if (verdict.breach === "store") {
        // Loud, and every sweep below still runs. See DEPLOY.md §3e.
        logError({ cid: lease.holder, hop: "cron", msg: "checkpoint_object_unverifiable", ...anchorLogFields(verdict) });
      }
      if (!verdict.anchored) {
        logError({ cid: lease.holder, hop: "cron", msg: "checkpoint_anchor_missing", ...anchorLogFields(verdict) });
      }
      // Rotate the tail window so the anchors this run did NOT read are the ones
      // the next run does. Best-effort: a cursor that fails to persist costs a
      // repeated window, never a missed verdict.
      try {
        // I47-FXR-CODE (F7): only while this run's guard lease row (`headLease ?? lease`, name and holder) exists.
        const anchorLease = headLease ?? lease;
        await db
          .prepare("INSERT INTO sweep_cursor (name, cursor_seq, cursor_id, updated_at) SELECT 'checkpoint_anchors', ?, '', ? WHERE EXISTS (SELECT 1 FROM cron_leases WHERE name = ? AND holder = ?) ON CONFLICT(name) DO UPDATE SET cursor_seq = excluded.cursor_seq, updated_at = excluded.updated_at")
          .bind(verdict.nextTailCursor ?? 0, new Date().toISOString(), anchorLease.name, anchorLease.holder)
          .run();
      } catch (e) {
        logError({ cid: lease.holder, hop: "cron", msg: "checkpoint_anchor_cursor", err: e instanceof Error ? e.name : typeof e });
      }
    }

    // I3-0010 / I3-0275 b / I3-1403 a (D-0982(4), I22-FX-HUB): the newest hourly anchor against its copy outside D1.
    // A mismatch stops the world exactly as a signed-checkpoint disagreement does above.
    if (await anchorCopyMismatch(db, env, lease.holder)) {
      await writeChainVerdict("broken");
      throw new Error("ledger tamper: D1 disagrees with the newest anchor held outside D1");
    }

    // I15-ANCHOR (I3-0010, D-0911(5)): the witnessed seals against the ledger. A
    // difference is a rewrite under a seal the main keys cannot reach: ALERT. Whether it
    // also stops the sweep (markLedgerBroken) is not decided here (open question).
    // I15-FX1 (RGI-ANCHOR-HEAD-RESEAL): the verdict is KEPT, because a mismatch also
    // withholds the 'ok' below and the witness seal further down — sealTick's rule.
    const verdictWitness = sealWitnessFromEnv(cfg);
    let headWitnessVerdict: WitnessVerdict | null = null;
    if (verdictWitness) {
      try {
        const w = await verifyAgainstWitness(db, verdictWitness, Date.now(), SEAL_HEAD_VERIFY_READS);
        headWitnessVerdict = w;
        if (w.mismatches.length > 0) logError({ cid: lease.holder, hop: "cron", msg: "ledger_witness_mismatch", ...witnessLogFields(w) });
        else if (w.nothing_compared && !w.ok) logError({ cid: lease.holder, hop: "cron", msg: "seal_witness_nothing_compared", ...witnessLogFields(w) });
        else if (w.unreadable > 0 || w.truncated) logError({ cid: lease.holder, hop: "cron", msg: "seal_witness_unreadable", ...witnessLogFields(w) });
      } catch (e) {
        logError({ cid: lease.holder, hop: "cron", msg: "seal_witness_unreadable", err: e instanceof Error ? e.name : typeof e });
      }
    }

    // I13-CRON: the chain passed and no anchor proved tamper (an unreadable anchor
    // is not a breach, above), so the tail triggers may publish from this log.
    // I15-FX1: NOT over a witness mismatch — the 'pending' written at the top of this
    // run stands, so tailChainGateOpen keeps the tails (and the five-minute seal) shut.
    // I3-0010 b (I23-W1-HUB): ONE predicate for both doors that must not act over a disputed head —
    // the 'ok' verdict here and the re-signed export below.
    const witnessDisputesHead = headWitnessVerdict !== null && headWitnessVerdict.mismatches.length > 0;
    // I22-Q12e (D-1030: "stop the whole sweep and alert"): a witness mismatch fails closed exactly as the signed-checkpoint
    // breach above does: the durable broken-ledger marker, a 'broken' verdict (the tails stay shut) and a throw before
    // any later step. Nothing is signed or exported over the disputed head.
    if (witnessDisputesHead) {
      logError({ cid: lease.holder, hop: "cron", msg: "checkpoint_export_skipped_witness_mismatch" });
      await markLedgerBroken(env, { breach: "seal_witness", at: new Date().toISOString(), detail: headWitnessVerdict }, lease.holder);
      await writeChainVerdict("broken");
      throw new Error("ledger tamper: the seal witness disputes the ledger head");
    }
    await writeChainVerdict("ok");

    // 3. Re-anchor, signed by a key this Worker does not hold. Signing is a
    //    NETWORK HOP and the signer can be down: a failed signature yields an
    //    UNSIGNED checkpoint (logged, and read back as "unverified"), never a
    //    thrown cron and never a fake "signed" state. Nothing on the ledger
    //    APPEND path calls any of this.
    //
    //    ONLY WHEN THERE IS SOMETHING NEW TO ANCHOR. The history key is
    //    `taken_at`, so this used to write a new object every run whether or not
    //    the chain had moved — four a day, forever, on a prefix nothing prunes
    //    and `allAnchors()` reads. Same head + already signed ⇒ skip (including
    //    the signer's network hop); unsigned at the current head ⇒ retry, which
    //    is the self-healing the module's header promises.
    //
    //    ISOLATED for the same reason the verification above is: writing the
    //    anchor is an R2 WRITE, and an R2 fault is not a reason to stop
    //    publishing warnings to fans. A run that cannot write its anchor simply
    //    has not anchored; the next one anchors a higher seq and covers the gap,
    //    which is the self-healing this module's header already relies on.
    // I3-0973: a run that lost its lease while verifying the chain must not write
    // the anchor (or seal it on the witness) on the new holder's watch. One fence,
    // the same renew-and-check every later fence uses.
    if (!(await stillLeased())) {
      logError({ cid: lease.holder, hop: "cron", msg: "checkpoint export SKIPPED - lease lost" });
    } else try {
      const signerCfg = checkpointSignerFromEnv(cfg);
      const signer = signerCfg && keys ? remoteCheckpointSigner(signerCfg, keys, outFetch, checkpointDeploymentIdFromEnv(cfg)) : null;
      // `keys` is the 5th argument, added by W3-04 I3-1620/I3-2063: exportCheckpointIfChanged
      // used to skip re-anchoring on the mere PRESENCE of a sig block on latest.json, so an
      // unverifiable signature suppressed re-signing. It now verifies and only skips on
      // state === "verified" -- but only when it is given the keys. Without this argument the fix
      // is dormant and the old behaviour stands, which is how it shipped from the owning room
      // (the call site is outside that room's ownership list, so it flagged it instead).
      const { record: written, written: didWrite } = await exportCheckpointIfChanged(db, store, undefined, signer, keys);
      if (didWrite && written && signer && !written.sig) {
        logError({ cid: lease.holder, hop: "cron", msg: "checkpoint_written_unsigned", seq: written.seq });
      }
      // I15-ANCHOR (D-0911(5)): every anchor is also a seal on the witness. AFTER the anchor
      // write, and publishSeal never throws, so the witness can never cost the anchor.
      // I15-FX1 (RGI-ANCHOR-HEAD-RESEAL): only over a witness compare that was `ok`, the
      // rule sealTick applies — never a seal over a mismatch, an empty window after a
      // seal was published, or a compare that did not run.
      const headWitness = sealWitnessFromEnv(cfg);
      if (headWitness && written) {
        if (headWitnessVerdict === null) {
          logError({ cid: lease.holder, hop: "cron", msg: "seal_witness_publish_failed", detail: `compare_unavailable:${written.seq}` });
        } else if (!headWitnessVerdict.ok) {
          logError({ cid: lease.holder, hop: "cron", msg: "seal_witness_refused", detail: `${written.seq}` });
        } else {
          const published = await publishSeal(headWitness, written);
          if (published === "failed" || published === "conflict") logError({ cid: lease.holder, hop: "cron", msg: "seal_witness_publish_failed", detail: `${published}:${written.seq}` });
          await noteSealPublished(db, headWitnessVerdict, published).catch((e: unknown) => logError({ cid: lease.holder, hop: "cron", msg: "seal_witness_publish_failed", err: e instanceof Error ? e.name : typeof e }));
        }
      }
    } catch (e) {
      logError({ cid: lease.holder, hop: "cron", msg: "checkpoint_not_written", err: e instanceof Error ? e.name : typeof e });
    }
    // Retention on the anchor prefix — the other half of the read bound. See
    // checkpoint.ts for what is kept and why deleting old anchors does not
    // weaken the pin (the chain is what carries history; an anchor at seq M
    // commits to everything under it). Isolated: a retention failure must never
    // cost the sweeps below.
    // I40-FX2-LEASE (RGOPS-04, I40TER-P5): the anchor fence's shape; a lost lease skips this step only, never returns.
    if (!(await stillLeased())) {
      logError({ cid: lease.holder, hop: "cron", msg: "checkpoint retention SKIPPED - lease lost" });
    } else try {
      // I3-1279: with the published keys, only a VERIFIED anchor counts as a bucket's signed survivor.
      const pruned = await pruneCheckpoints(store, Date.now(), undefined, keys);
      if (pruned.undeletable > 0) {
        logError({ cid: lease.holder, hop: "cron", msg: "checkpoint_retention_inert", undeletable: pruned.undeletable, scanned: pruned.scanned });
      }
    } catch (e) {
      logError({ cid: lease.holder, hop: "cron", msg: "checkpoint_retention", err: e instanceof Error ? e.name : typeof e });
    }
    // SLICE 25 — the published directory, rebuilt AFTER the write and the
    // retention sweep so it describes the prefix as it now stands rather than as
    // it stood at the start of the tick. Built from object NAMES only (`seq` and
    // `taken_at` are both in the key), so it costs one listing and zero gets.
    //
    // Isolated like every leg around it: a directory that fails to rebuild is a
    // stale directory, and a stale directory still points at real anchors that
    // are still individually fetchable and still individually verifiable. It
    // must never cost the registry rebuild below it.
    //
    // Written whether or not signing is configured. The ROUTES fail closed on an
    // absent key set; the artifact is not the gate, and building it
    // unconditionally means turning signing on publishes history immediately
    // instead of six hours later.
    // I40-FX2-LEASE (RGOPS-04, I40TER-P5): the anchor fence's shape; a lost lease skips this step only, never returns.
    if (!(await stillLeased())) {
      logError({ cid: lease.holder, hop: "cron", msg: "checkpoint index SKIPPED - lease lost" });
    } else try {
      await exportCheckpointIndex(store);
    } catch (e) {
      logError({ cid: lease.holder, hop: "cron", msg: "checkpoint_index", err: e instanceof Error ? e.name : typeof e });
    }
    // I3-1010: THE TWO VERDICT-REMOVING SWEEPS RUN BEFORE THE REGISTRY PLAN.
    // A plan freezes the published member set for the next six hours; planned
    // before the expiry and lapse sweeps, it froze a passport this same tick was
    // about to de-verify, and the extension served it as verified until the next
    // plan. Both are isolated (try/catch), so neither can cost the rebuild.
    nowMs = Date.now();
    // Enforce `verification_granted.expires_at`. Appends `verification_expired`
    // for grants past their stated deadline; the fold moves the passport
    // verified → claimed, so the registry verdict, the badge, the card, payout
    // presentation and every verified-only write path change together. A grant
    // with an absent/empty/unparseable expiry is NEVER expired — see
    // verification-sweep.ts. Isolated: a sweep failure must not take the hygiene
    // sweeps below with it.
    // I40-FX2-LEASE (RGOPS-04, I40TER-P5): the anchor fence's shape; a lost lease skips this step only, never returns.
    if (!(await stillLeased())) {
      logError({ cid: lease.holder, hop: "cron", msg: "verification expiry SKIPPED - lease lost" });
    } else try {
      const vsweep = await sweepExpiredVerifications(db, nowMs);
      // F-01: DEPLOY.md §7 tells the operator to monitor the ONE cron that
      // de-badges creators with no human in the loop by
      // `wrangler tail --search "VERIFICATION EXPIRY"` (DEPLOY.md:3484). That
      // string was printed nowhere in the tree, and the only line this leg DID
      // write fired solely when `unreadable_expiry > 0` — so on an ordinary
      // tick the documented monitoring surface was empty. Printed EVERY tick,
      // counters only (INV-061: no passport id, no handle, no free text).
      // I3-0257-b: all four DEPLOY.md counters (log.ts ALLOWED_KEYS carries expired/warned since I16-W07A).
      logError({ cid: lease.holder, hop: "cron", msg: "VERIFICATION EXPIRY", scanned: vsweep.scanned, expired: vsweep.expired, warned: vsweep.warned, unreadable_expiry: vsweep.unreadable_expiry, no_grant: vsweep.no_grant });
    } catch (e) {
      logError({ cid: lease.holder, hop: "cron", msg: "verification_expiry_sweep", err: e instanceof Error ? e.name : typeof e });
    }

    // THE CHARACTER LAPSE SWEEP — the other way a Verified badge stops being
    // true. The verification sweep above enforces a DEADLINE; this one enforces
    // the accountability rule after the claim, which slice 14 could only enforce
    // at the write. Isolated for the same reason and in the same shape: one
    // sweep failing must not take the others with it.
    //
    // AFTER the expiry sweep on purpose. A character whose grant has ALSO run
    // out should lapse for the reason with a date on it — `verification_expired`
    // carries the deadline it enforced, which an operator can dispute; a lapse
    // event carries only when we noticed. Both walk verified → claimed, so
    // whichever lands first makes the other a no-op for that passport.
    // BEHIND A LEASE FENCE. The lease is acquired before both sweeps, so a
    // simultaneous start is already excluded — but a predecessor that blew the
    // TTL is not, and this is the one sweep on the tick that REMOVES a public
    // verdict. Two overlapping runs take independent scan snapshots, and the
    // loser's rolled-back batch leaves a watch row the winner has already acted
    // on. Cheap to check; the alternative is a de-badge decided by a run that no
    // longer holds the lease.
    // I3-1010: moved ABOVE the registry plan, so its lease check can no longer
    // RETURN (that would cost the registry plan, which asks no fence and still
    // runs on a lost lease); a lost lease skips THIS sweep only, the hygiene
    // rotation below asks its own fence before each step (runHygieneRotation),
    // and the returning fence it used to be is kept below, in front of the artifact legs.
    if (!(await stillLeased())) {
      logError({ cid: lease.holder, hop: "cron", msg: "character lapse SKIPPED - lease lost" });
    } else {
    try {
      const lsweep = await sweepLapsedOperators(db, nowMs, undefined, undefined, undefined, undefined, lease.holder);
      if (lsweep.lapsed > 0 || lsweep.opened > 0 || lsweep.failed > 0) {
        logError({
          cid: lease.holder,
          hop: "cron",
          msg: "character_lapse_sweep_summary",
          opened: lsweep.opened,
          notified: lsweep.notified,
          recovered: lsweep.recovered,
          voided: lsweep.voided,
          in_grace: lsweep.in_grace,
          deferred: lsweep.deferred,
          lapsed: lsweep.lapsed,
          refused: lsweep.refused,
          failed: lsweep.failed,
        });
      }
    } catch (e) {
      logError({ cid: lease.holder, hop: "cron", msg: "character_lapse_sweep", err: e instanceof Error ? e.name : typeof e });
    }
    }

    // Registry rebuild. REGISTRY_DECOY_SEED is a REQUIRED secret: the decoy
    // padding is derived from it, and a build padded with fresh random decoys
    // lets an observer intersect two consecutive builds down to the real member
    // set. Absent or placeholder => SKIP the rebuild — the last-published
    // buckets keep serving (stale but consistent) — rather than publish an
    // unsafe one. Skipping, not throwing: an unset secret must not take the
    // rest of the scheduled sweep down with it. See DEPLOY.md §3d.
    //
    // ISOLATED. The rebuild is the most safety-critical leg here and it is also
    // an unbounded series of R2 writes, so it is the likeliest to fail on a
    // storage fault — and it used to be UNCAUGHT, which meant every sweep below
    // it (now including the retention prunes and the expiry sweep) died with it.
    // A failed rebuild leaves the last-published buckets serving, exactly as the
    // absent-secret path does; it must not also cost the hygiene work.
    const decoySeed = registryDecoySeedFromEnv(cfg);
    // The registry build flips its pointer with a CONDITIONAL put, so it takes
    // the adapter that has one (ADR-0012 §2) rather than the plain `store` every
    // other sweep in this handler shares. Same bucket, same objects.
    const registryStore = registryR2Store(env.REGISTRY_BUCKETS);
    if (decoySeed) {
      try {
        // ADR-0012: this tick PLANS a generation — it does not build one. A plan
        // freezes every global quantity and the per-bucket steps run across this
        // and the following five-minute ticks, so a build that outgrows one
        // invocation finishes instead of restarting from zero. A generation is
        // planned only when none is open, which is what keeps "starts every 6
        // hours" true; `advanceRegistryBuild` then spends what is left of THIS
        // invocation on steps rather than leaving the first one five minutes away.
        //
        // The publication signer, when one is provisioned. Absent keys or an
        // absent endpoint => the build publishes an UNSIGNED root and says so;
        // it is never skipped for want of a signature, because an unsigned
        // registry still serves every client and no registry serves none.
        const advanced = await advanceRegistryBuild(db, registryStore, {
          decoySeed,
          signer: registrySignerFor(cfg, outFetch),
          plan: true,
          cid: lease.holder,
        });
        const built = advanced.result;
        if (built) {
          // THE DELISTING BACKLOG. Every bucket sheds a fixed number of published
          // hashes per build so that a build which delisted somebody is shaped
          // like one that did not (registry-build.ts, THE CHURN QUOTA), which caps
          // real departures at that quota per bucket. It is not reached in
          // ordinary operation; a non-zero number here means removals are queued
          // behind it and the published SLA (LISTING-POLICY.md 5.1) is being
          // spent. Log it so it is not invisible — the quota must NEVER be raised
          // in response, because a churn that answers removal pressure is a churn
          // that reports it.
          if (built.spilledOut > 0) {
            logError({ cid: lease.holder, hop: "cron", msg: "registry_delisting_backlog", spilled: built.spilledOut, quota: built.churn });
          }
          // AND THE RETRACTION LANE, WHICH IS A DIFFERENT NUMBER. Retractions have
          // their own lane, ahead of removals and EXEMPT from the real-churn cap,
          // so this is 0 by construction today. It is logged anyway, separately,
          // because the alternative is reporting a stuck retraction — an accusation
          // we have withdrawn but are still publishing — inside a counter an
          // operator reads as "people waiting to be delisted".
          if (built.spilledOutRetractions > 0) {
            logError({ cid: lease.holder, hop: "cron", msg: "registry_retraction_backlog", spilled: built.spilledOutRetractions });
          }
          // OR-01: a safety entry the reserved lane could not fit is pending, and
          // steps the rung at the next plan rather than evicting anybody.
          if (built.safetyPending > 0) {
            logError({ cid: lease.holder, hop: "cron", msg: "registry_safety_backlog", pending: built.safetyPending, seq: built.seq });
          }
        }
        // "Live within 12 hours, or an alarm fires" — the served generation's age
        // is checked here as well as on the five-minute tick, so a deployment
        // whose step driver never runs still raises it.
        await registryPublicationAlarm(registryStore, Date.now(), lease.holder);
      } catch (e) {
        logError({ cid: lease.holder, hop: "cron", msg: "registry_rebuild_failed", err: e instanceof Error ? e.name : typeof e });
      }
    } else {
      logError({ cid: lease.holder, hop: "cron", msg: "registry_decoy_seed_missing_skip" });
    }

    // ─────────────────────────────────────────────────────────────────────────
    // HYGIENE FIRST, ARTIFACTS AND DELIVERY SECOND.
    // ─────────────────────────────────────────────────────────────────────────
    // Everything from here to the end of the prune block is CHEAP (a bounded D1
    // statement or two each) and touches no network. It used to sit at the very
    // BOTTOM of this handler, below the alert fan-out — which was itself the one
    // expensive leg with no try/catch around it — so a single throw up there
    // silently took fourteen retention prunes and the verification-expiry sweep
    // with it, every tick, forever. Table growth and an unenforced `expires_at`
    // are exactly the failures that do not announce themselves.
    //
    // The order is now: things that must not be starved, then things that can
    // afford to be. Each expensive leg below is individually isolated, so no
    // rearrangement is load-bearing for correctness — but ordering means the
    // cheap work is already DONE when an expensive leg exhausts the invocation.


    // Billing hygiene: downgrade subscriptions that lapsed at their period end
    // with no cancel webhook (keeps the cached accounts.tier honest), and prune
    // old webhook-dedup rows so billing_events can't grow without bound.
    await step("sweepExpiredSubscriptions", () => sweepExpiredSubscriptions(db, nowMs));
    // I3-1691: credit checkouts the webhook never resolved (stripe.ts). D-0843(1): a
    // paid but ungranted attempt leaves the open set as `reconciled` (operator review).
    await step("sweepCreditCheckoutPending", async () => {
      const cfg = stripeConfigFromEnv(env as unknown as Record<string, unknown>);
      if (!cfg) return;
      const r = await sweepCreditCheckoutPending(db, cfg, nowMs, outFetch);
      // Counts only (log.ts allowlist): the rows themselves are the operator's
      // worklist — `state = 'reconciled'` (D-0843(1)).
      if (r.paid_ungranted > 0) logError({ cid: lease.holder, hop: "cron", msg: "credit_checkout_paid_ungranted", count: r.paid_ungranted });
      if (r.checked > 0) logInfo({ cid: lease.holder, hop: "cron", msg: "credit_checkout_reconcile", scanned: r.checked, voided: r.expired, paid: r.paid_ungranted });
    });
    // S-2 (I23-RG-MONEY, D-0995(11); I23-FX-TC hand-off, landed by I23-FX-INT): the Stripe collection steps still owed,
    // retried from their own audit rows (closures.ts retryCompromiseStripePauses), in four passes: the PAUSE of a
    // compromise suspension (failed, never asked, or skipped while Stripe was unconfigured — JA-05); the RESUME after a
    // lift (failed, or skipped — D-1005(b), RG-05); and the termination CANCEL and the erasure CANCEL skipped while
    // Stripe was unconfigured (D-1010(6), JN-02 item 3). Each pass takes up to STRIPE_PAUSE_RETRY_ACCOUNTS rows; never
    // throws per account; with no Stripe config the step returns before any pass runs. Counts only in the log
    // (log.ts allowlist).
    await step("retryCompromiseStripePauses", async () => {
      const cfg = stripeConfigFromEnv(env as unknown as Record<string, unknown>);
      if (!cfg) return;
      const r = await retryCompromiseStripePauses(db, { stripe: cfg, fetchImpl: outFetch });
      if (r.failed > 0) logError({ cid: lease.holder, hop: "cron", msg: "stripe_pause_retry_failed", count: r.failed, scanned: r.accounts });
      // I21-Q49: an erasure that kept the Stripe customer because a cancel failed deletes it once that retry above has
      // cancelled every failed subscription (account_data.ts settleErasedStripeCustomers; its second row is the record).
      const cu = await settleErasedStripeCustomers(db, { stripe: cfg, fetchImpl: outFetch });
      if (cu.failed > 0) logError({ cid: lease.holder, hop: "cron", msg: "erased_stripe_customer_delete_failed", count: cu.failed });
    });
    // BILLING-05 (D-1037; I29-FX-BILLING H1, wired by I29-FX-INT): the creator twin — a FAILED Creator Pro pause of a
    // still-suspended creator is retried from its own audit rows (creator-close.ts retryCreatorStripePauses, <=
    // STRIPE_PAUSE_RETRY_ACCOUNTS episodes, never throws per creator; its rows are the record). No Stripe config: not run.
    // The same call also retries a failed Creator Pro cancel at termination (D-1051(3)) and, OPEN-ROWS B3 (I31), a failed
    // Creator Pro RESUME after a lift (<= STRIPE_PAUSE_RETRY_ACCOUNTS lifted episodes a pass).
    await step("retryCreatorStripePauses", async () => {
      const cfg = stripeConfigFromEnv(env as unknown as Record<string, unknown>);
      if (cfg) await (await import("./creator-close.js")).retryCreatorStripePauses(db, { stripe: cfg, fetchImpl: outFetch });
    });
    // I32 (I3-0040, D-1082(10); I32-FX1-DATA H1): a creator closure notice whose mail failed is re-sent from its own
    // audit rows until sent or dead-lettered (creator-close.ts retryCreatorClosureNotices: <= STRIPE_PAUSE_RETRY_ACCOUNTS
    // episodes a run, never throws per episode; CREATOR_CLOSURE_NOTICE_DEAD is the operator alert). No mail rail: nothing sent.
    await step("retryCreatorClosureNotices", async () => {
      const r = await (await import("./creator-close.js")).retryCreatorClosureNotices(db, { env: env as unknown as Record<string, unknown>, fetchImpl: outFetch });
      if (r.failed > 0) logError({ cid: lease.holder, hop: "cron", msg: "creator_closure_notice_retry", count: r.failed, scanned: r.scanned });
    });
    // QUEUE #418 (D-0776): rows an erasure race left naming an account that no
    // longer exists — bounded per table per run (collections.ts ORPHAN_SWEEPS).
    await step("sweepOrphanAccountRows", () => sweepOrphanAccountRows(db));
    // I29-MONEYKEEP H2 (I21-Q26/I20-Q3, I21-Q75, I24-Q1; landed by I29-INT): money records of erased accounts past their
    // retention (<= MONEY_SWEEP_GROUPS_PER_RUN 50 groups), lifted closures past 6 years (<= ACCOUNT_CLOSURE_PRUNE_LIMIT 500)
    // and supporter day snapshots past 90 days (one DELETE) — money-retention.ts, supporter-status.ts.
    await step("sweepMoneyRecords", () => sweepMoneyRecords(db, nowMs));
    await step("pruneAccountClosures", () => pruneAccountClosures(db, nowMs));
    await step("pruneSupporterSnapshots", () => pruneSupporterSnapshots(db, nowMs));
    // I29-CREATORERASE H1 (landed by I29-INT): creator wipes past their due instant (creator-erasure.ts, CREATOR_WIPE_LIMIT).
    // D-1113(3) / D-1112(5) (I34-FX1-MONEY H1): the payout dependencies, so a residual is paid once before the wipe.
    // D-1127(5), D-1126(6): the residual's Connect account read is handed the tick's fetch (its transfer is not: see outFetch).
    await step("sweepCreatorWipes", () => sweepCreatorWipes(db, nowMs, CREATOR_WIPE_LIMIT, { env: env as unknown as Record<string, unknown>, accountFetch: outFetch }));
    // I29-OPS H-INT (I21-Q82; landed by I29-INT): closed agencies' member emails and domains 30 days after closure,
    // at most 4 batches of AGENCY_CLOSURE_SWEEP_BATCH per run, stopping when one reports nothing more.
    await step("sweepClosedAgencies", async () => {
      for (let i = 0; i < 4; i++) if (!(await sweepClosedAgencies(db, nowMs)).more) break;
    });
    // I39-OPS (D-1158 answer 4): move the one-signature account transfers that are due (console.ts
    // runDueScheduledTransfers: at most 50 a run, oldest due first; a failed row stays scheduled for the next run).
    await step("runDueScheduledTransfers", async () => {
      const r = await runDueScheduledTransfers(db, mailerFromEnv(env as unknown as Record<string, unknown>, outFetch), nowMs, { env: env as unknown as Record<string, unknown> });
      if (r.failed > 0) logError({ cid: lease.holder, hop: "cron", msg: "scheduled_transfers_failed", count: r.failed });
    });
    // I20-W1D (QUEUE #946 (a)), landed by I20-L01: fill passports.display_name_skeleton for rows written before
    // migration 0207. Budget: one SELECT (<= DISPLAY_NAME_SKELETON_BACKFILL_LIMIT = 200 rows) + one batch per run;
    // idempotent, a no-op once every live row is filled (population: passports rows predating 0207, finite and shrinking).

    // and two R2 GETs — I13-CRON measured), and the candidate list was `SELECT DISTINCT subject
    // FROM trust_events WHERE event_type = 'war_opt_in'` with no ceiling at all.
    // Nothing here can be PAGED the way the payout run is: the table is written
    // create-if-absent and frozen for the ISO week so an opened pack's
    // odds_hash stays verifiable, which means the whole set has to be priced in
    // ONE tick or not at all. So the bound is a cap on the candidates, ordered
    // by FIRST OPT-IN — a rule that can be stated to a creator ("the first
    // WAR_ODDS_MAX_CREATORS to opt in are in this week's packs") rather than an
    // arbitrary slice of a hash order — and the cap is LOGGED LOUDLY when it
    // binds, because a creator silently missing from a week's odds table is
    // exactly the kind of exclusion nobody discovers from a green cron.
    await tailLeg("odds", "packOdds", async () => {
      try {
        const sel = await selectOddsCandidates(db, WAR_ODDS_MAX_CREATORS, Date.now());
        if (sel.total > WAR_ODDS_MAX_CREATORS) {
          // Still logged when it binds — a creator missing from a week's table is
          // exactly the kind of exclusion nobody discovers from a green cron. But
          // the sentence has changed with the rule: the window ROTATES, so this is
          // now "who is in this week", not "who is permanently out".
          logError({
            cid: lease.holder,
            hop: "cron",
            msg: "pack_odds_cap_binding",
            opted_in: sel.total,
            cap: WAR_ODDS_MAX_CREATORS,
            subjects: sel.subjects.length,
            window_offset: sel.offset,
          });
        }
        const pk = await buildPackOdds(store, db, sel.subjects, Date.now());
        if (pk.written) logError({ cid: lease.holder, hop: "cron", msg: "pack_odds_published", opted_in: sel.total, window_offset: sel.offset });
      } catch (e) {
        logError({ cid: lease.holder, hop: "cron", msg: "pack_odds_sweep", err: e instanceof Error ? e.name : typeof e });
      }
    });

    // S-CW6c: the WEEKLY creator payout run. Safe to call every tick — it
    // self-limits to one payout per creator per ISO week (UNIQUE(passport,
    // period)) and is idempotent on the payout id (the Stripe transfer key), so
    // a retry or overrun cannot double-pay. Absent Connect config → no-op (the
    // config read fails closed). PAGED, transfer-capped and deadline-bounded —
    // see creator-payout.ts THE BOUNDS for the arithmetic and for what stopping
    // early costs.
    await tailLeg("money", "weeklyPayouts", async () => {
      try {
        const connectCfg = connectConfigFromEnv(env as unknown as Record<string, unknown>);
        if (connectCfg) {
          // D-1127(5), D-1126(6): the Connect account read and the Connect transfer on the tick's fetch (see outFetch).
          const pr = await runWeeklyPayouts(
            db,
            connectCfg,
            nowMs,
            () => "pay_" + crypto.randomUUID().replace(/-/g, "").slice(0, 20),
            undefined,
            undefined,
            { fetchImpl: outFetch },
            env.ANALYTICS_DB && ctx ? { prod: db, analytics: d1(env.ANALYTICS_DB), queue: env.ANALYTICS_QUEUE } : undefined,
            ctx,
          );
          // Docket #43 (a): `held` stays the standing PREDICATE's decline count,
          // unchanged. `retry_truncated` and `deadline_abandoned` are logged
          // alongside it, named so a 3 a.m. reader can tell "we declined these"
          // (held) from "we ran out of budget before reading these" (retry_
          // truncated) from "we ran out of time on these" (deadline_abandoned)
          // — the next operator action differs for each.
          //
          // Docket #58 (b): `held` is COUNTED CURSOR-RELATIVE (`creator-payout.ts`
          // `pendWindow`) — it is the decline count for THIS TICK'S window
          // (cursor forward to the page end, or to the end of the table on the
          // last page), never a table-wide total. Left as PAYOUT23 (S-032) found
          // it: not mine to widen, and docket #43 (a) already rejected that
          // reading twice. What changes here is only the log line's claim about
          // itself — `wrapped` (already computed, already free) is logged
          // alongside `held` so a reader is not left inferring scope from
          // silence: `wrapped: false` means more of the connected set remains
          // for a later tick and this number is a partial count; `wrapped: true`
          // means this window closed out the table, but STILL only counts what
          // was forward of the cursor when the tick started — a `held` row
          // sitting BEHIND that cursor (already passed in a prior tick, as an
          // empty-page tick's window is) is invisible to it. `held: 0` is
          // therefore "nothing held in this window", never "nothing held" —
          // that reading needs a full lap (cursor `''` to `''`) with `held`
          // summed across every tick in it, which no single line here reports.
          //
          // I3-0062 / W3-09a item 5: `stalled` is ALSO logged, and is ALSO a
          // reason to emit the line at all. It counts the rows this tick refused
          // for at least `PAYOUT_STALL_ATTEMPTS` re-attempts (`creator-payout.ts`
          // `runWeeklyPayouts`) — money that is owed and has stopped moving,
          // which is the one number here whose next operator action is "go and
          // look at this creator". It was computed and dropped: until this line
          // the field had no reader anywhere in the tree. Added to the CONDITION
          // as well as the body deliberately — a tick that stalls rows is never
          // a tick that paid or failed none of them today, but the condition is
          // what decides whether a human ever sees the number, and a count that
          // can only be logged when some OTHER count is non-zero is not a
          // reported count. Tick-shaped: a row whose backoff has grown to a week
          // appears in at most one tick in seven, which is why the standing
          // population read is the other half of this and belongs on the
          // operator console. THAT HALF IS NOT WIRED: `creator-payout.ts`
          // `stalledPayouts` still has NO CALLER — this line is a comment, not
          // one, and a caller-count grep on the symbol must not read it as one.
          // Its door is `/ops/payouts` in `console.ts`, outside W3-09a-FR3's
          // ownership list; the exact edit is in that room's report.
          const summary = payoutSummaryFields(pr, lease.holder);
          if (summary) logError({ ...summary });
          // A ceiling that binds is NOT a clean run. Without this the leg reports
          // nothing while paying a shrinking fraction of what it owes, which is
          // the failure mode that does not announce itself.
          if (pr.stopped !== null) {
            logError({ cid: lease.holder, hop: "cron", msg: "payout_run_stopped", stopped: pr.stopped, transfers: pr.transfers });
          }
        } else {
          // I3-1161: no Connect config means no creator is paid this week. Said
          // every run, and recorded ONCE per UTC day as a `payout:run_skipped`
          // audit row, which GET /ops/payouts counts (`skipped_no_connect_7d`).
          logError({ cid: lease.holder, hop: "cron", msg: "payout_run_skipped_no_connect" });
          const skipped = `no_connect day=${new Date(nowMs).toISOString().slice(0, 10)}`;
          await db
            .prepare("INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT 'system', 'payout:run_skipped', ?, ?, NULL, 'system' WHERE NOT EXISTS (SELECT 1 FROM audit_log WHERE action = 'payout:run_skipped' AND detail = ?)")
            .bind(skipped, new Date(nowMs).toISOString(), skipped)
            .run();
        }
      } catch (e) {
        logError({ cid: lease.holder, hop: "cron", msg: "creator_payout_run", err: e instanceof Error ? e.name : typeof e });
      }
      // I29-STANDING H3 (I20-Q2, D-1031; landed by I29-INT): a closed creator's unrecoverable debt, AFTER the payout run and
      // in its own try, on the money group's ONE leg. At most 4 calls of CREATOR_DEBT_SWEEP_LIMIT (50) per tick, stopping
      // when one reports nothing more; the sweep counts each creator once per ISO week itself.
      try {
        for (let i = 0; i < 4; i++) if (!(await sweepUnrecoverableCreatorDebt(db, nowMs)).more) break;
      } catch (e) {
        logError({ cid: lease.holder, hop: "cron", msg: "creator_debt_sweep", err: e instanceof Error ? e.name : typeof e });
      }
      // #489a (D-0920, I16-W06C H-489-wire): the cash-outflow register's identity against the ledger, once per money
      // tick, AFTER the payout run and in its own try (a fault here costs only this check). It rides the money group's
      // ONE leg rather than adding a second: the group's kill-hold is pinned as a one-leg group (i13-fx1-tails.test.ts).
      // A mismatch is an ALARM line for an operator to investigate before the next repayment: `count` is the number of
      // mismatched rows and `scanned` the rows examined (log.ts ALLOWED_KEYS; no ref or account id leaves the worker).
      try {
        const rc = await reconcileCashOutflows(db);
        if (!rc.ok) {
          logError({
            cid: lease.holder,
            hop: "cron",
            msg: "cash_outflow_reconcile_mismatch",
            // + the payout arm (D-0928(2); I16-L09 H4, landed by I16-L10): paid payouts vs kind 'payout' register rows;
            // + the voided arm (I17-J01 H3, D-0938(1a)): voided payouts vs the register row voidPayout writes.
            count: rc.ledger_without_outflow.length + rc.outflow_without_ledger.length + rc.cents_mismatch.length + rc.payouts.paid_without_outflow.length + rc.payouts.voided_without_outflow.length + rc.payouts.outflow_without_payout.length + rc.payouts.cents_mismatch.length,
            scanned: rc.checked.ledger_repayments + rc.checked.register_rows + rc.payouts.checked.paid_payouts + rc.payouts.checked.voided_payouts + rc.payouts.checked.register_rows,
          });
        }
      } catch (e) {
        logError({ cid: lease.holder, hop: "cron", msg: "cash_outflow_reconcile_failed", err: e instanceof Error ? e.name : typeof e });
      }
      // I53-RECONCILE (D-1383(3)): the money books' conservation identities, read-only, AFTER the cash-outflow reconcile
      // and in its own try on the same ONE leg; a drift is ONE `money_reconcile_drift` line (an operator alarm, counts only).
      try {
        await (await import("./money-reconcile.js")).runMoneyReconcile(db, nowMs, lease.holder);
      } catch (e) {
        logError({ cid: lease.holder, hop: "cron", msg: "money_reconcile_failed", err: e instanceof Error ? e.name : typeof e });
      }
    });

    } finally {
      // The ordinary path (including a thrown sweep) gives the lease back at
      // once. A Worker evicted mid-sweep runs no `finally` — that case is what
      // the lease's TTL is for, and why it must be bounded.
      leaseStopped = true;
      if (leaseTimer !== undefined) clearTimeout(leaseTimer);
      // I14-FX2 (RG-CRON-S2): each lease is released on its own — a throw releasing one never keeps the other.
      try {
        await releaseCronLease(db, lease);
      } finally {
        if (headLease !== null) await releaseCronLease(db, headLease);
      }
    }
      })();
    } finally {
      flushOperatorAlarmsAfter(env, ctx0, deferred); // I21-Q27
    }
  },
};

ON CONFLICT(name) DO UPDATE SET cursor_seq = excluded.cursor_seq, cursor_id = excluded.cursor_id, updated_at = excluded.updated_at")

class TailBudgetExhausted extends Error {}
/**
 * I13-FX1 (I13-RG-OPS-04): metered handles for one tail group. Every D1 statement (a batch
 * per statement), every raw-D1 statement on ANALYTICS_DB and every R2 call on REGISTRY_BUCKETS
 * is charged BEFORE it is sent; one that would cross `budget` is refused with a throw, which
 * the leg's own catch logs — the platform's own "too many subrequests" behaviour, at a line
 * that leaves the scaffolding (in-flight clear, carry write, lease release) its headroom.
 */
function tailMeter(
  budget: number,
  db: DbLike,
  env: Env,
): { metered: true; db: DbLike; env: Env; store: ReturnType<typeof r2Store>; exhausted: () => boolean; spent: () => number; share: (c?: number) => void; left: () => number } {
  let n = 0;
  let out = false;
  // I13-FX7: one leg may run under a lower ceiling (`share`); `share()` restores the budget. A
  // refusal under a share stops that leg only: `exhausted` (the group's cut) is the BUDGET's.
  let ceiling = budget;
  const charge = (k: number): void => {
    if (n + k > ceiling) {
      if (ceiling >= budget) out = true;
      throw new TailBudgetExhausted("tail subrequest budget spent");
    }
    n += k;
  };
  const targets = new WeakMap<object, unknown>();
  const counted = (o: object, chargeOn: ReadonlySet<PropertyKey>, rewrap: ReadonlySet<PropertyKey> = new Set()): unknown => {
    const p = new Proxy(o, {
      get(t, k) {
        const v = Reflect.get(t, k) as unknown;
        if (typeof v !== "function") return v;
        if (k === "batch") return (sts: unknown[]) => (charge(sts.length), (v as (x: unknown[]) => unknown).call(t, sts.map((x) => (typeof x === "object" && x !== null ? (targets.get(x) ?? x) : x))));
        if (rewrap.has(k)) return (...a: unknown[]) => counted((v as (...x: unknown[]) => object).apply(t, a), STATEMENT_CHARGED, STATEMENT_REWRAP);
        if (chargeOn.has(k) || chargeOn.has("*"))
          return (...a: unknown[]) => {
            charge(1);
            return (v as (...x: unknown[]) => unknown).apply(t, a);
          };
        return (v as (...x: unknown[]) => unknown).bind(t);
      },
    });
    targets.set(p, o);
    return p;
  };
  const mdb = counted(db as unknown as object, new Set(), new Set(["prepare"])) as DbLike;
  const menv = new Proxy(env, {
    get(t, k) {
      const v = Reflect.get(t, k) as unknown;
      if (k === "REGISTRY_BUCKETS" && v) return counted(v as object, new Set(["*"]));
      if (k === "ANALYTICS_DB" && v) return counted(v as object, new Set(), new Set(["prepare"]));
      return v;
    },
  }) as Env;
  return {

/**
 * I14-CRON (QUEUE #916, D-0904(1)): ONE digest-generation pass, as the digests tail ran it since I13-FX1 —
 * the owed periods first (DIGEST_OWED_CURSOR), then the daily periods with no row at all in the last
 * `backfillDays` days (oldest first; the tail walks DAILY_BACKFILL_MAX_DAYS, the five-minute tick 1), then the
 * week — every generateDigests call charged against ONE `budget`, and a period the budget cut short carried as
 * owed. `weeklyOnlyIfPending` (the tick) asks `periodHasPendingCreator` first, so a finished week costs one read,
 * not the weekly call's proof scan and warning context every five minutes. Probe, owed read and owed write sit
 * outside `budget` (the caller's scaffolding: tail 14 + 2, tick 1 + 1 + 2).
 * I14-FX2 (RG-CRON-L3, -S1): THE CURSOR IS WRITTEN AHEAD. A pass that adds a period the cursor does not carry
 * writes the whole list BEFORE it generates (a throw there: nothing generated); otherwise it writes the shrunk
 * list after. Still one owed write a pass. So a failed write, a throw out of generateDigests or an evicted
 * invocation leaves a superset of the unfinished periods recorded, never a subset; a period added and finished
 * in one pass stays listed one pass longer (a no-op call). A period a creator FAILED in (not truncated) stays
 * owed and is retried with backoff (I14-FX6, D-0907(2): `next_at`, DIGEST_FAILED_BACKOFF_MS doubling to
 * DIGEST_FAILED_BACKOFF_MAX_MS) until DIGEST_FAILED_RETRY_WINDOW_MS (24 h) after its first failure and at least
 * DIGEST_FAILED_RETRIES retries, then is dropped with a `digest_creator_failed` error line.
 * I14-FX5 (QUEUE #916, D-0906(1)): a pass that wrote ahead holds 1 of `budget` back and writes the list AGAIN after
 * generating when a retry advanced or a period was dropped, so the counter advances and a drop is saved (and logged)
 * once on every pass; the total stays `budget` + the one scaffolding write. A dropped period stays in the list as a
 * marker (`dropped`; a pre-I14-FX6 marker is retries > DIGEST_FAILED_RETRIES with no first_failed_at), never attempted, while this pass would re-add it (the current week, a day in
 * the backfill walk), so the tail cannot re-add a dropped weekly with its counter reset.
 * I38-TICK (L39-C1-L2): every owed write is COMPARE-AND-SET on the cursor_id this pass read or last wrote (the upsert's
 * DO UPDATE ... WHERE sweep_cursor.cursor_id = it, or DO NOTHING when no row was read). A pass whose lease lapsed and was re-taken finds the
 * successor's list there and does not overwrite it (a `tail_cursor` warn line; the write reports false). `opts.lease`
 * (withCronLease's guard): once `lost`, the pass makes no further write — not the write-ahead, no generateDigests
 * call, not the final write.
 */
export async function generateDigestPass(
  db: DbLike,
  store: Parameters<typeof generateDigests>[1],
  budget: number,
  cid: string,
  nowMs: number,
  opts: { backfillDays?: number; weeklyOnlyIfPending?: boolean; lease?: CronLeaseGuard } = {},
): Promise<{ spent: number; generated: number; owed: number }> {
  const weekly = weeklyPeriod(nowMs);
  const seen: { raw?: string } = {};
  // A call, not a property read: the guard's `lost` flips between awaits, which TypeScript's narrowing cannot see.
  const leaseLost = (): boolean => opts.lease?.lost === true;
  const recorded = await readOwedDigestPeriods(db, seen);
  if (recorded === null) {
    logError({ cid, hop: "cron", msg: "tail_cursor", detail: `${DIGEST_OWED_CURSOR} unreadable; pass skipped` });
    return { spent: 0, generated: 0, owed: 0 };
  }
  const periods: OwedDigestPeriod[] = [...recorded];
  const add = (p: OwedDigestPeriod): void => {
    if (!periods.some((q) => q.kind === p.kind && q.key === p.key)) periods.push(p);
  };
  // I4-08-R7 hand-off: every daily period a >24 h outage skipped, oldest first (dailyPeriodsToBackfill bounds the walk).
  const backfill = await dailyPeriodsToBackfill(db, nowMs, opts.backfillDays);
  for (const p of backfill) add({ kind: "daily", ...p });
  if (!periods.some((q) => q.kind === "weekly" && q.key === weekly.key) && (opts.weeklyOnlyIfPending !== true || (await periodHasPendingCreator(db, "weekly", weekly.key)))) add({ kind: "weekly", ...weekly });
  // I14-FX5: a dropped period's marker is kept only while this pass would re-add the period, and is never attempted.
  const isDropped = (p: OwedDigestPeriod): boolean => p.dropped === true || (p.first_failed_at === undefined && (p.retries ?? 0) > DIGEST_FAILED_RETRIES);
  const readdable = new Set([`weekly:${weekly.key}`, ...backfill.map((p) => `daily:${p.key}`)]);
  const dropped = periods.filter((p) => isDropped(p) && readdable.has(`${p.kind}:${p.key}`));
  // Only what the cursor can carry is attempted; a row-less day past it is found again by the backfill walk.
  const attempt = periods.filter((p) => !isDropped(p)).slice(0, DIGEST_OWED_MAX);
  const writeOwed = async (list: OwedDigestPeriod[]): Promise<boolean> => {
    const next = JSON.stringify(list);
    try {
      const res =
        seen.raw === undefined
          ? await db.prepare("INSERT INTO sweep_cursor (name, cursor_seq, cursor_id, updated_at) VALUES (?, ?, ?, ?) ON CONFLICT(name) DO NOTHING").bind(DIGEST_OWED_CURSOR, list.length, next, new Date().toISOString()).run()
          : await db.prepare("INSERT INTO sweep_cursor (name, cursor_seq, cursor_id, updated_at) VALUES (?, ?, ?, ?) ON CONFLICT(name) DO UPDATE SET cursor_seq = excluded.cursor_seq, cursor_id = excluded.cursor_id, updated_at = excluded.updated_at WHERE sweep_cursor.cursor_id = ?").bind(DIGEST_OWED_CURSOR, list.length, next, new Date().toISOString(), seen.raw).run();
      if (res.changes !== 1) {
        logWarn({ cid, hop: "cron", msg: "tail_cursor", detail: `${DIGEST_OWED_CURSOR} moved by another pass; not overwritten` });
        return false;
      }
      seen.raw = next;
      return true;
    } catch (e) {
      logError({ cid, hop: "cron", msg: "tail_cursor", detail: DIGEST_OWED_CURSOR, err: e instanceof Error ? e.name : typeof e });
      return false;
    }
  };
  const writtenAhead = attempt.some((p) => !recorded.some((q) => q.kind === p.kind && q.key === p.key));
  if (leaseLost()) return { spent: 0, generated: 0, owed: attempt.length };
  if (writtenAhead && !(await writeOwed([...attempt, ...dropped]))) return { spent: 0, generated: 0, owed: attempt.length };
  const genBudget = writtenAhead ? budget - 1 : budget;
  let left = genBudget;
  let generated = 0;
  let changed = false;
  const stillOwed: OwedDigestPeriod[] = [];
  for (const p of attempt) {
    if (leaseLost()) return { spent: genBudget - left + (writtenAhead ? 1 : 0), generated, owed: attempt.length };
    // I14-FX6: a failed period waits out its backoff; it stays owed and costs nothing this pass.
    if (left <= 0 || (p.next_at !== undefined && nowMs < Date.parse(p.next_at))) {
      stillOwed.push(p);
      continue;
    }
    const r = await generateDigests(db, store, p.kind, p.key, p.startIso, p.endIso, { nowMs, budget: left });
    left -= r.spent ?? 0;
    generated += r.generated;
    if (r.truncated === true) stillOwed.push(p);
    else if ((r.failed ?? 0) > 0) {
      const retries = (p.retries ?? 0) + 1;
      changed = true;
      const first = p.first_failed_at !== undefined && Number.isFinite(Date.parse(p.first_failed_at)) ? p.first_failed_at : new Date(nowMs).toISOString();
      if (retries <= DIGEST_FAILED_RETRIES || nowMs - Date.parse(first) < DIGEST_FAILED_RETRY_WINDOW_MS) {
        const wait = Math.min(DIGEST_FAILED_BACKOFF_MS * 2 ** Math.min(retries - 1, 20), DIGEST_FAILED_BACKOFF_MAX_MS);
        stillOwed.push({ ...p, retries, first_failed_at: first, next_at: new Date(nowMs + wait).toISOString() });
      } else {
        logError({ cid, hop: "cron", msg: "digest_creator_failed", detail: `${p.kind}:${p.key}`, failed: r.failed });
        if (readdable.has(`${p.kind}:${p.key}`)) dropped.push({ ...p, retries, first_failed_at: first, dropped: true });
      }
    }
  }
  const rewrite = !writtenAhead || changed;
  if (leaseLost()) return { spent: genBudget - left + (writtenAhead ? 1 : 0), generated, owed: attempt.length };
  if (rewrite) await writeOwed([...stillOwed, ...dropped]);
  if (stillOwed.length > 0) logInfo({ cid, hop: "cron", msg: "digest_generation_carried", detail: stillOwed.map((p) => `${p.kind}:${p.key}`).join(",") });
  return { spent: genBudget - left + (writtenAhead && rewrite ? 1 : 0), generated, owed: stillOwed.length };
}
