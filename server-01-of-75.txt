import { build } from "esbuild";
import { mkdirSync, readFileSync, writeFileSync } from "node:fs";
import { dirname, join, resolve } from "node:path";
import { fileURLToPath } from "node:url";

/**
 * Bundle the operator dashboard into a single self-contained module
 * (dist/app.js) + index.html. Served behind Cloudflare Access; it calls the
 * trust-core `/ops/*` API on OPS_BASE with the Access cookie (credentials:
 * include). OPS_BASE is injected + VALIDATED as a plain https URL so a bad value
 * can't break out of the define string or point the app at a hostile origin.
 */
const root = dirname(fileURLToPath(import.meta.url));
/**
 * THE CHOKEPOINT. Under a test runner, building into `dist/` is REFUSED
 * outright — not warned about, not tolerated. `pnpm test` used to run four of
 * these builds with no env on every invocation, so a real
 * `OPS_BASE=… node build.mjs` followed by the suite left the deploy artifact
 * rebuilt from the `dhwaja.example` defaults (exit 0, no warning), and the
 * console's `connect-src` named a domain we do not own. A test that wants a
 * build must say where to put it.
 */
if (process.env.VITEST && !process.env.BUILD_OUT_DIR) {
  throw new Error(
    "refusing to build into dist/ under a test runner: dist/ is the deploy artifact. " +
      "Set BUILD_OUT_DIR to a temp directory.",
  );
}

/**
 * OUTPUT DIRECTORY. `dist/` is the DEPLOY ARTIFACT — the checklist is
 * build-then-deploy and `wrangler pages deploy` ships exactly what is in it —
 * so nothing may write there implicitly. A caller that is not deploying MUST
 * pass BUILD_OUT_DIR and gets a directory of its own.
 *
 * This exists because `pnpm test` used to silently rebuild `dist/` from the
 * `dhwaja.example` defaults: a real `OPS_BASE=... node build.mjs` followed by
 * the test suite left the shipped bundle pointed at a domain we do not own,
 * exit 0, no warning. tests/invariants/build-artifacts.test.ts pins that a
 * test can never write into an app's dist/ again.
 */
const out = process.env.BUILD_OUT_DIR
  ? resolve(process.env.BUILD_OUT_DIR)
  : join(root, "dist");

const opsBase = process.env.OPS_BASE ?? "https://api.dhwaja.example";
const URL_RE = /^https:\/\/[a-z0-9.-]+(:\d+)?(\/[^\s"'`]*)?$/i;
if (!URL_RE.test(opsBase)) throw new Error(`OPS_BASE must be a plain https URL, got: ${opsBase}`);

/**
 * PUBLIC_BASE is NOT read here (I23-FX-INT, D-0995(4)): the Terms / Privacy pages render in-app from @dhwaja/legal
 * (src/legal.ts), so this console needs no fan-app origin, and a dist/ build no longer requires one. Its old refusal
 * ("the Terms / Privacy links would be silently absent") described links that no longer exist.
 */

/**
 * THE SECOND CHOKEPOINT: a PLACEHOLDER build may never land in `dist/`.
 *
 * `dhwaja.example` is a reserved example domain nobody owns. A bundle built
 * against it cannot reach the API or the registry, so on a real deploy EVERY
 * check it makes fails — and a failed check is the state a user reads the least
 * clearly.
 *
 * The VITEST chokepoint above stops a TEST from overwriting the deploy artifact.
 * This one stops a developer from filling it with an unusable one: `node
 * build.mjs` with no env is a perfectly reasonable thing to run, it just may not
 * produce the thing DEPLOY.md ships. Anywhere else — a preview, a test, a local
 * sandbox — `BUILD_OUT_DIR` gets its own directory and the defaults are fine,
 * which is why this is scoped to `dist/`.
 *
 * IT RUNS BEFORE `mkdirSync`, DELIBERATELY. Refusing after the directory exists
 * leaves an EMPTY `dist/` behind — which reds the test suite and, worse, is a
 * directory `wrangler pages deploy` will happily publish as an empty site
 * (audit v10, B5). The refusal must write nothing at all.
 *
 * This block used to exist only in `apps/web` while five sibling builds carried
 * the same comment describing the same hazard and none of the code (audit v10,
 * B4). `tests/invariants/build-artifacts.test.ts` now decides which apps owe a
 * refusal by READING THE BUILT BYTES for the placeholder host, so an app cannot
 * opt out by naming its variables differently.
 */
const PLACEHOLDER_BASES = [["OPS_BASE", opsBase]];
if (!process.env.BUILD_OUT_DIR) {
  const placeholders = PLACEHOLDER_BASES.filter(([, val]) => /(^|\.)dhwaja\.example(\/|$|:)/i.test(val)).map(([name]) => name);
  if (placeholders.length > 0) {
    throw new Error(
      `refusing to build dist/ from the placeholder defaults (${placeholders.join(", ")} still point at dhwaja.example). ` +
        "dist/ is what `wrangler pages deploy` ships, and a bundle pointed at a domain nobody owns fails every check it makes. " +
        "Pass the real origins, or set BUILD_OUT_DIR to build a throwaway copy.",
    );
  }
}
mkdirSync(out, { recursive: true });

await build({
  entryPoints: { app: join(root, "src/main.ts") },
  outdir: out,
  bundle: true,
  format: "esm",
  target: "es2022",
  minify: true,
  legalComments: "none",
  define: { __OPS_BASE__: JSON.stringify(opsBase) },
});

// The shipped CSP names the ops origin explicitly. The checked-in index.html
// carries `connect-src 'self'` so that an un-built or mis-built file fails
// CLOSED — the console can reach nothing rather than anything — and the one
// origin this app is allowed to talk to is appended here from the OPS_BASE that
// was validated above. It used to ship `connect-src *`, the loosest policy in
// the repo, on the console that can erase an account.
const CONNECT_SRC = "connect-src 'self'";
const html = readFileSync(join(root, "index.html"), "utf8");
if (!html.includes(CONNECT_SRC)) {
  throw new Error(`index.html must carry "${CONNECT_SRC}" for the ops origin to be appended to it`);
}
if (!html.includes("form-action 'none'")) {
  throw new Error("index.html must carry form-action 'none' — it does not fall back to default-src");
}
const opsOrigin = new URL(opsBase).origin;
writeFileSync(join(out, "index.html"), html.replace(CONNECT_SRC, `${CONNECT_SRC} ${opsOrigin}`));
// `frame-ancestors` is IGNORED inside a <meta http-equiv> CSP — the spec makes
// it header-only, so every browser drops it and the copy in index.html has
// never protected anything. This app ships as static assets, so the only place
// a REAL header can come from is a Cloudflare Pages `_headers` file at the root
// of the output directory. The meta keeps the directives that do work there;
// this adds the one that does not. Two policies are enforced as an
// INTERSECTION, so this can only tighten, never loosen, the meta policy.
const HEADERS_FILE = "/*\n  Content-Security-Policy: frame-ancestors 'none'\n";
writeFileSync(join(out, "_headers"), HEADERS_FILE);

// eslint-disable-next-line no-console
console.log(`built admin → ${out}/app.js + index.html (OPS_BASE=${opsBase}, connect-src ${opsOrigin})`);

import { esc, fmtTs } from "./html.js";
import { integrityBanner, type LedgerIntegrity } from "./integrity.js";
import { seatBanner, type OpsRole } from "./role.js";

/**
 * ABUSE & INTEGRITY — two halves, deliberately unequal.
 *
 * The top is the LOAD-BEARING CLAIM: the ledger is append-only and
 * tamper-evident, and this is where a violation of that promise becomes
 * visible. It is the one screen allowed full red (see integrity.ts).
 *
 * The bottom is SIGNAL. Four tables of shapes in the data. Every one of them has
 * at least one boring explanation and at least one bad one, and none of them can
 * tell the two apart — so each table prints both, no row carries a status pill,
 * and the only row action is NAVIGATION. There is no button in this half that
 * punishes anyone, throttles anyone, or changes anyone's state, because an
 * operator must not be able to mistake a row for a verdict.
 */

/**
 * SLICE 22 — recognition-grant volume, from `GET /ops/recognition-volume`.
 *
 * A `support`-role read that has existed with no caller since S90: creator-level
 * counts, no fan ids. It belongs on THIS half of THIS screen and nowhere else,
 * and the placement is what constrains the rendering — the standing rule three
 * paragraphs up is "the only row action is NAVIGATION", so this table gets no
 * status pill, no threshold colour and no ranking by suspicion, exactly like the
 * four beside it.
 */
export interface RecognitionVolumeRow {
  passport_id: string;
  granted: number;
  active: number;
  retracted: number;
  spotlights_active: number;
}

export interface AbuseSignals {
  redemption_attempts: { redeemed: number; dead: number; unknown: number };
  leaked_credentials: { credential: string; dead_attempts: number; distinct_accounts: number }[];
  noisy_accounts: { account: string; dead_attempts: number }[];
  /** I3-1676 (D-0829): `unqualified_referees` = live referees with no qualification
   * row (admin.ts). Optional: an older worker does not send them, and the cell says "—". */
  referral_clusters: { referrer: string; referees: number; qualified_referees?: number; unqualified_referees?: number; rewards_granted: number }[];
  /** I3-1676 (I22-L03 landing PLATFORM hand-off): the referrers BELOW the 3-referee cluster floor, as totals
   * (admin.ts referral_fragments). Optional: an older worker does not send it, and the line is not drawn. */
  referral_fragments?: { referrers: number; referees: number; unqualified_referees: number; rewards_granted: number };
  /** `refused` (QUEUE #483): the server says this bucket's ceiling FIRED this
   * window. Optional — an older worker does not send it. */
  throttle_pressure: { key: string; window_start: string; count: number; refused?: boolean }[];
}

/** The standing frame over the whole bottom half. Split into a lead and a body
 * (N-20) rather than one string that a `.replace()` call subtracts a prefix
 * from: that construction silently no-ops on any re-punctuation, sentence-case
 * edit or translation and prints the lead twice. Composed the way
 * `pv.marks_note` / `pv.marks_note_strong` are in packages/i18n. */
export const PATTERNS_NOTE_LEAD = "Patterns to investigate, not findings.";
export const PATTERNS_NOTE_BODY =
  "Nothing below this line punishes anyone, throttles anyone, or changes anyone's state — there is no control in this half that does. Every table is a shape in the data with at least one boring explanation and at least one bad one, and the table cannot tell them apart. A cluster is a question, not a verdict. If a row turns out to be real abuse, the action lives somewhere else and is taken by a person: a warning goes through the warnings funnel with its cooling-off and second look; a credential is revoked by its creator; an account is handled in Accounts. Nothing here is evidence on its own.";
/** Kept for callers (and tests) that still read the combined sentence. */
export const PATTERNS_NOTE = `${PATTERNS_NOTE_LEAD} ${PATTERNS_NOTE_BODY}`;

/** A table with an explicit empty sentence. Emptiness is the expected state for
 * every table on this page and gets a plain line — never an illustration and
 * never congratulation, because making emptiness an achievement would make a
 * populated table feel like an emergency. */
/** I3-1676 — one line under the clusters table for the sub-floor referrers; "" when the worker did not send the
 * totals or a field is not a finite count (never a guessed 0). */
export function fragmentsLine(f: AbuseSignals["referral_fragments"]): string {
  if (!f) return "";
  const vals = [f.referrers, f.referees, f.unqualified_referees];
  if (!vals.every((v) => typeof v === "number" && Number.isFinite(v))) return "";
  return `<p class="panenote" data-referral-fragments>Below the 3-referee floor: ${esc(String(f.referrers))} referrers, ${esc(String(f.referees))} referees, ${esc(String(f.unqualified_referees))} never qualified.</p>`;
}

function table(headers: string[], rows: string, empty: string): string {
  const numCols = headers.map((h) => (h.startsWith("#") ? ` class="num"` : ""));
  return `<table class="ds-table"><thead><tr>${headers.map((h, i) => `<th scope="col"${numCols[i]}>${esc(h.replace(/^#/, "") || "Action")}</th>`).join("")}</tr></thead>
    <tbody>${rows || `<tr><td colspan="${headers.length}" class="muted">${esc(empty)}</td></tr>`}</tbody></table>`;
}

/**
 * The edition-set anchors table. Each row is one `edition_set_anchored` ledger
 * event. The last column is the honest join: an anchor row is only as strong as
 * the external checkpoint that covers its seq, so a row ABOVE the highest
 * checkpoint renders "not yet covered" in slate — true, unalarming, and not
 * dressed up as a tick.
 */
export function anchorsTable(anchors: LedgerIntegrity["recent_anchors"], anchorSeq: number | null): string {
  const rows = anchors.map((a) => {
    const covered = anchorSeq !== null && a.seq <= anchorSeq;
    const cell = anchorSeq === null
      ? `<span class="ds-chip chip">no checkpoint to compare</span>`
      : covered
        ? `<span class="ds-chip chip ok">seq ${esc(String(anchorSeq))} ✓</span>`
        : `<span class="ds-chip chip">not yet covered</span>`;
    return `<tr><td class="mono">${esc(a.subject)}</td><td class="mono num">${esc(String(a.seq))}</td><td class="mono">${esc(fmtTs(a.created_at))}</td><td>${cell}</td></tr>`;
  }).join("");
  return `<h2>Recent edition-set anchors</h2>
    <p class="hsub">The <b>tamper-evidence artifacts</b>: once a week ends, the sweep hashes that week's whole edition set and appends one <code>edition_set_anchored</code> event, whose subject is the set hash itself. Silently replacing any stored edition would break the anchored set hash. Weeks appear here <b>only after they have ended</b> — an in-flight week has no anchor because its set can still grow.</p>
    ${table(["Set hash (subject)", "#Seq", "Anchored at", "Covered by checkpoint"], rows, "No edition-set anchors yet.")}
    <p class="panenote">The last column is a join, not a verdict: an anchor is only as strong as the external checkpoint that covers its seq.</p>`;
}

/**

 * four above it.
 *

 * `granted` is a creator with a big audience who thanks it often; it is also a

 * a creator correcting mistakes; it is also one testing what sticks. The counts

 * was recognised, which is deliberate.
 *
 * NAVIGATION IS THE ONLY ACTION, so the row links to the creator dossier. That

 * have been a sentence with no navigation behind it.
 *
 * A NULL read renders as a stated failure, never as an empty table. Every other
 * table on this page follows the same rule for the same reason: emptiness here
 * is the expected healthy state, so a failed read that rendered as empty would
 * be indistinguishable from good news.
 */
export function recognitionSection(rec: RecognitionVolumeRow[] | null): string {

  if (rec === null) {
    return `<h3>Recognition volume</h3>${sub}
      <div class="ds-banner banner warn"><span class="g">!</span><div class="body"><span class="lead">The recognition-volume read did not come back.</span> <b>This is an absent read, not a quiet ladder</b> — nothing about grant volume can be concluded from this load.</div></div>`;
  }

      <td class="mono">${esc(r.passport_id)}</td>
      <td class="num">${esc(String(r.granted))}</td>
      <td class="num">${esc(String(r.active))}</td>

}

/**

/** The appeal column, which is the one an operator misreads if it is a date
 * alone: "no appeal" and "an appeal we have not decided" are opposite facts. */
function appealCell(r: ClosureRow, nowMs: number): string {
  if (r.appeal_decided_at) return `<span class="muted">${esc(r.appeal_outcome ?? "decided")} · ${esc(fmtTs(r.appeal_decided_at))}</span>`;
  if (r.appeal_opened_at) return `<strong>open, undecided</strong> · filed ${esc(fmtTs(r.appeal_opened_at))}`;
  const left = Math.ceil((Date.parse(r.appeal_deadline) - nowMs) / 86_400_000);
  return Number.isFinite(left) && left > 0 ? `<span class="muted">none · ${left} day${left === 1 ? "" : "s"} left</span>` : `<span class="muted">none · window closed</span>`;
}

function rowControls(r: ClosureRow, role: OpsRole, nowMs: number): string {
  // QUEUE #419: termination cancels the account's Stripe subscriptions; a cancel
  // the provider refused is an `account:terminate_stripe_cancel failed` audit row,
  // and this re-runs it (only the subs with no `ok` row; one full operator).
  // #487a / #663: a terminated account's kept balance is recorded as repaid here
  // (dual control, server-side), after the money has been sent out of band.
  if (r.state === "terminated")
    return `${writeControl({ label: "Retry Stripe cancel", role, attrs: { "data-act": "closure-stripe-retry", "data-id": r.account_id }, lock: false })} ${writeControl({ label: "Record repayment…", role, tone: "danger", attrs: { "data-act": "closure-repay", "data-id": r.account_id }, lock: false })}`;
  if (r.state !== "suspended") return "";
  const windowOpen = Date.parse(r.appeal_deadline) > nowMs;
  const undecided = Boolean(r.appeal_opened_at) && !r.appeal_decided_at;
  // WHY IT IS DISABLED, ON THE CONTROL. The server refuses both of these, and a
  // button that could only ever 409 under copy naming no reason is exactly the
  // failure the agency console's F-G4 finding is about.
  const blocked = undecided ? `an appeal is open and undecided` : windowOpen ? `the appeal window runs to ${fmtTs(r.appeal_deadline)}` : "";

    ? `<span class="muted" title="${esc(blocked)}" style="font-size:12px">Terminate — not yet: ${esc(blocked)}</span>`

}

/**
 * I3-0567 a, admin half (D-0522: "Not just the action, the CLOCK") — THE DATA-SUBJECT-REQUEST REGISTER, READ-ONLY.
 *
 * `GET /ops/dsr` (services/trust-core/src/console.ts, full-operator) lists the statutory-deadline register; migration
 * 0210 derives `deadline` from `arrival_at`. This screen shows each request's arrival, due date and whether it is
 * OVERDUE: an OPEN request (`received` / `in_progress`) whose deadline is before now. It writes nothing — recording,
 * extending and closing stay on their own doors.
 *
 * THE LIMIT IS NAMED (FAMILY-003): the route answers at most `DSR_PAGE_MAX` rows, OPEN ones first and then by deadline
 * (I26-W1-CONTENT), so a full page says that more may exist beyond it instead of reading as the whole register, and
 * the route's `open_total` makes an open request beyond the page a stated count rather than a silent drop.
 */
export interface DsrRow {
  dsr_id: string;
  request_type: string;
  subject: string;
  state: string;
  arrival_at: string;
  deadline: string;
  extended_at?: string | null;
  extended_by?: string | null;
  extension_reason?: string | null;
  closed_at?: string | null;
}

/** console.ts `GET /ops/dsr` `LIMIT 200`. */
export const DSR_PAGE_MAX = 200;
const OPEN_STATES = new Set(["received", "in_progress"]);
const DAY_MS = 86_400_000;

/** Where a request stands against its clock, at `nowMs`. An unparseable deadline is NOT on time: it says so. */
export function dsrClock(r: Pick<DsrRow, "state" | "deadline" | "closed_at">, nowMs: number): { kind: "overdue" | "due" | "closed" | "closed-late" | "unknown"; text: string } {
  const due = Date.parse(r.deadline);
  if (!OPEN_STATES.has(r.state)) {
    const closed = typeof r.closed_at === "string" ? Date.parse(r.closed_at) : Number.NaN;
    if (Number.isFinite(due) && Number.isFinite(closed) && closed > due) return { kind: "closed-late", text: `Closed ${Math.ceil((closed - due) / DAY_MS)} day(s) after its deadline` };
    return { kind: "closed", text: "Closed" };
  }
  if (!Number.isFinite(due)) return { kind: "unknown", text: "Deadline unreadable — treat as urgent" };
  if (due < nowMs) return { kind: "overdue", text: `OVERDUE by ${Math.ceil((nowMs - due) / DAY_MS)} day(s)` };
  return { kind: "due", text: `Due in ${Math.ceil((due - nowMs) / DAY_MS)} day(s)` };
}

function dsrRowHtml(r: DsrRow, nowMs: number): string {
  const c = dsrClock(r, nowMs);
  const chip = c.kind === "overdue" || c.kind === "unknown" ? "danger" : c.kind === "closed-late" ? "warn" : c.kind === "due" ? "" : "ok";
  const ext = r.extended_at ? `<div class="muted">Extended ${esc(fmtTs(r.extended_at))}${r.extended_by ? ` by ${esc(r.extended_by)}` : ""}${r.extension_reason ? ` — ${escClamp(r.extension_reason, 200)}` : ""}</div>` : "";
  return `<tr data-dsr="${esc(r.dsr_id)}" data-clock="${c.kind}"><td><code>${esc(r.request_type)}</code></td><td>${escClamp(r.subject, 200)}</td><td>${esc(r.state)}${r.closed_at ? `<div class="muted">${esc(fmtTs(r.closed_at))}</div>` : ""}</td><td>${esc(fmtTs(r.arrival_at))}</td><td>${esc(fmtTs(r.deadline))}${ext}</td><td><span class="ds-chip${chip ? ` ${chip}` : ""}">${esc(c.text)}</span></td></tr>`;
}

 * 3. SUPPRESS AND DISMISS ARE DRAWN AS DIFFERENT OBJECTS. Dismiss is the soft
 *    close: the item can come back on a fresh reporter. Suppress is the terminal
 *    one: the item never reopens and stops feeding Caution. Two closes with two
 *    different permanences must not look alike, so they sit in separate boxes
 *    with different weight. BOTH confirm, because both purge the item's
 *    evidence: suppress through `suppressConfirm` below, dismiss through
 *    `openDismiss` (apps/admin/src/main.ts, #152), which also requires a reason.
 *
 *    NEITHER OF THEM REMOVES THE SUBJECT'S LISTING, and this copy used to say
 *    the opposite. Four strings on the terminal control read "The listing stops
 *    being served" / "The listing is not being served" — but `suppressReport`
 *    (services/trust-core/src/reports.ts) only sets `fan_report_items.state` to
 *    `'suppressed'`; it never writes `request_suppressions`, never touches
 *    `passports`, and never reaches the registry. "Listing" means the registry
 *    entry everywhere else in this product — including on the adjacent Removals
 *    screen, where that exact sentence describes the control that really does
 *    it — so an operator reaching for "make this stop being served" pressed
 *    this, got no public change at all, and believed they had delisted someone.
 *    The words now say what the code does, and point at Removals for the act
 *    they used to promise. What BOTH closes do change for fans is the Caution
 *    class: caution.ts builds its lookalike population from OPEN
 *    `fan_report_items` only, so closing the item drops that input at the next
 *    registry rebuild (I3-1846) — the copy says that too, rather than "nothing
 *    fans are served changes". `SUPPRESS_FACTS` / `SUPPRESS_LISTING_NOTE` are the
 *    single source of those sentences and are pinned against the server's
 *    `suppressReport` by apps/admin/test/evidence-review.test.ts, the same way
 *    apps/web/src/report.ts pins its caps against evidence.ts.
 */

function fmtBytes(n: number): string {
  if (!Number.isFinite(n) || n < 0) return "—";
  if (n < 1024) return `${n} B`;
  if (n < 1024 * 1024) return `${(n / 1024).toFixed(0)} KB`;
  return `${(n / (1024 * 1024)).toFixed(1)} MB`;
}

/** Triage facts for one item. It renders the server's own ordering sentences
 * rather than composing a summary of them — the one place those words live is
 * `queueOrderReasons`, and this pane is not going to become a second one. */
function triageFacts(it: FanReportRow): string {
  const notes = (it.notes ?? []).map((n) => `<li>${esc(n)}</li>`).join("");
  const ev = it.evidence ?? {};
  // WHAT IT RESEMBLES, not just how. The pane used to render `sim.kind` alone —
  // the bare word `contains` with no indication of what it contains — so an
  // operator could not check the machine's reasoning, and an item whose match
  // pointed several ways looked identical to one that pointed at nobody. Since
  // slice 11 an ambiguous match deliberately withholds the pointer, which makes
  // showing the matched handle and the ambiguity the difference between a lead
  // and a shrug.
  const sim = (ev.handle_similarity ?? ev.domain_similarity) as
    | { kind?: unknown; matched_handle?: unknown; matched_platform?: unknown; matched_domain?: unknown; distance?: unknown; ambiguous?: unknown }
    | undefined;
  const matched =
    typeof sim?.matched_handle === "string"
      ? `${typeof sim.matched_platform === "string" ? `${sim.matched_platform} · ` : ""}${sim.matched_handle}`
      : typeof sim?.matched_domain === "string"
        ? sim.matched_domain
        : "";
  const simLine =
    sim && typeof sim.kind === "string"
      ? `<div class="line">handle/domain similarity: <b>${esc(sim.kind)}</b>${typeof sim.distance === "number" ? ` <span class="muted">(${esc(String(sim.distance))} edit${sim.distance === 1 ? "" : "s"})</span>` : ""}${matched ? ` to <code>${esc(matched)}</code>` : ""}${sim.ambiguous === true ? ` — <b>ambiguous</b>: several creators match equally well, so the machine named none of them` : ""}</div>`
      : "";
  const puny = ev.domain_punycode ? `<div class="line">host carries a <b>punycode label</b> — the classic confusable typosquat shape</div>` : "";
  // I3-0706: the face stage (services/trust-core/src/face-triage.ts) writes
  // `evidence.face` and can open a warning DRAFT on a high match, and this pane
  // is the human gate for that draft — so the machine's evidence is shown here.
  // `image_url` is the reported profile's og:image, attacker-chosen: text only,
  // never an <a> or an <img>.
  const face = ev.face as
    | { stage?: unknown; near_duplicate?: unknown; matched_passport_id?: unknown; image_url?: unknown; confidence?: unknown; checked_at?: unknown }
    | undefined;
  const faceLine =
    face && typeof face === "object"
      ? `<div class="line">face match (machine): <b>${esc(typeof face.confidence === "string" ? face.confidence : "unknown")}</b> confidence${typeof face.stage === "string" ? ` · stage <code>${esc(face.stage)}</code>` : ""}${face.near_duplicate === true ? " · <b>near-duplicate of the enrolled photo</b>" : face.confidence === "medium" ? " · same face, different photo — a lookalike or twin is possible" : ""}${typeof face.matched_passport_id === "string" ? ` · matched <code>${esc(face.matched_passport_id)}</code>` : ""}${typeof face.image_url === "string" && face.image_url !== "" ? ` · profile image <code>${esc(face.image_url)}</code> <span class="muted">(the reported account's own URL — not opened here)</span>` : ""}${typeof face.checked_at === "string" ? ` · checked ${esc(fmtTs(face.checked_at))}` : ""}. <span class="muted">A machine match, not a finding: publication stays with a human.</span></div>`
      : "";
  const faceState = typeof it.face_state === "string" && it.face_state !== "" ? `<div class="line">face stage: <code>${esc(it.face_state)}</code></div>` : "";
  // I3-1363: the server returns at most three notes and `notes_total` beside
  // them; when it is larger, say how many are not on screen.
  const shownNotes = (it.notes ?? []).length;
  const totalNotes = typeof it.notes_total === "number" && Number.isFinite(it.notes_total) ? it.notes_total : shownNotes;
  const hidden = totalNotes > shownNotes ? totalNotes - shownNotes : 0;
  const notesHead = hidden > 0
    ? ` — <b>showing ${esc(String(shownNotes))} of ${esc(String(totalNotes))}</b>, the most recent. ${esc(String(hidden))} older note${hidden === 1 ? " is" : "s are"} retained on this item. <button class="ds-btn" type="button" data-act="notes-all" data-id="${esc(it.item_id)}">Show all ${esc(String(totalNotes))} notes</button>`
    : "";
  return `<div class="facts">
    <div class="k">Triage</div>
    <div class="line">subject <code>${esc(it.canonical)}</code>${it.suspected_passport_id ? ` — resembles <code>${esc(it.suspected_passport_id)}</code>` : ""}</div>
    <div class="line"><b>${esc(String(it.distinct_reporters))}</b> distinct reporter(s) · state <code>${esc(it.state)}</code>${it.reopen_count > 0 ? ` · reopened <b>${esc(String(it.reopen_count))}×</b>` : ""}</div>
    ${simLine}${puny}${faceLine}${faceState}
    <div class="line">categories: ${it.categories.length ? it.categories.map((c) => `<span class="ds-chip chip">${esc(c)}</span>`).join(" ") : '<span class="muted">none filed</span>'}</div>
    <div class="line">first seen ${esc(fmtTs(it.created_at))}${it.updated_at ? ` · last change ${esc(fmtTs(it.updated_at))}` : ""}</div>
    ${erasedNote(it) ? `<div class="line" data-erased-evidence><b>${esc(erasedNote(it)!)}</b></div>` : ""}
    ${notes ? `<div class="line">reporter notes
 <span class="muted">(markup stripped at intake, operator-only, never public)</span>${notesHead}<ul class="notes">${notes}</ul></div>` : ""}
    <div class="foot"><b>Band labels, honestly.</b> A reporter's band is EARNED — account age plus past reports an operator upheld, or a claimed/verified passport. It is not a purchase, not a subscription perk, and not a claim about the person. The ordering sentences under the queue row above say which part of an item's position was earned and which part was not; they are the server's words and this pane does not paraphrase them.</div>
  </div>`;
}

/**
 * The single-object viewer. `objectUrl` is a blob URL the console created from
 * the Access-gated fetch — it is never a link to storage and never leaves the
 * document.
 *
 * A failed load is AMBER, never red: a failed fetch is an unknown, not an
 * accusation, and red on this screen is spent on purge and suppress only. It
 * also never renders as "probably fine" — an object that did not come back is
 * UNSEEN, and the copy says so.
 */
export function evidenceViewer(itemId: string, obj: EvidenceObject, status: ViewerStatus, objectUrl: string | null, role: OpsRole): string {
  const bar = `<div class="vbar">
    <span class="id mono">${esc(obj.evidence_id)}</span>
    <span class="meta">${esc(obj.content_type)} · ${esc(fmtBytes(obj.byte_size))} · uploaded ${esc(fmtTs(obj.created_at))} · reporter ${esc(obj.account_id)}</span>
  </div>`;

  let canvas: string;
  if (status === "ok" && objectUrl) {
    canvas = `<div class="canvas"><img alt="report attachment ${esc(obj.evidence_id)}" src="${esc(objectUrl)}"></div>`;
  } else if (status === "loading") {
    canvas = `<div class="canvas"><p class="muted">Loading the object…</p></div>`;
  } else {
    const why = status === "unavailable"
      ? "<b>The evidence store binding is absent in this environment</b>, so the route fail-closed with a 503. That is configuration being off on purpose, not a fault and not a missing feature — the row exists and the bytes were never reachable from here."
      : status === "missing"
        ? "<b>The row exists; the bytes did not come back.</b> Either the object is already gone, or the item/object pair does not match — the pair is enforced in the query, and a mismatch is a 404 rather than a way to read any object from any item's URL."
        : "<b>The row exists; the bytes did not come back.</b> The fetch failed for a reason this console cannot name.";
    canvas = `<div class="failbox">
      <div class="hl2"><span class="badge">!</span>Couldn't load this object</div>
      <p>${why} <b>Treat it as unseen</b> — never as "probably fine". Any other attachments on this item are unaffected.</p>
      <div class="row"><button class="ds-btn" data-act="retry-evidence" data-evid="${esc(obj.evidence_id)}">Retry</button><a class="btnlink" href="#/config">Check config health</a></div>
      <p class="faint">Amber, <b>never red</b> — a failed fetch is an unknown, not an accusation.</p>
    </div>`;
  }

  return `<div class="viewer">${bar}${canvas}${handlingBlock()}</div>${purgeZone(itemId, obj, role)}`;
}

/**
 * The two closes, drawn as two different objects.
 *
 * Dismiss: plain box, quiet button — the item can come back on a fresh
 * reporter. Its click opens a confirm (main.ts `openDismiss`, #152) because it
 * deletes the evidence too, and that confirm requires a reason.
 * Suppress: its own red-hairline box, extra room, a confirm — the close is
 * permanent and the item stops feeding Caution.
 *
 * NEITHER removes the subject's listing; both say so, in the same words, from
 * `SUPPRESS_LISTING_NOTE` — which also says the one served thing both DO
 * change: an open item can feed Caution (caution.ts), a closed one cannot.
 */
function dispositions(it: FanReportRow, role: OpsRole): string {
  const open = it.state === "open";
  const suppressible = it.state === "open" || it.state === "dismissed" || it.state === "retracted";
  const suppressed = it.state === "suppressed";

  if (suppressed) {
    return `<div class="disp">
      <div class="dbox terminal">
        <span class="tag">Terminal close · in force</span>
        <div class="t">Suppressed</div>
        <p>This report is closed indefinitely: fresh reports on this subject do not reopen it and do not feed Caution — only an explicit, audited unsuppress can. Its evidence was deleted when it was suppressed and does not come back.

/** HTML escaping, shared by every view in the operator console. Every value that
 * reaches innerHTML goes through it — the console renders operator-supplied and
 * fan-supplied strings (handles, notes, reasons) and holds no sanitizer.
 *
 * It escapes the FIVE characters that matter, including the single quote, so one
 * function is correct in element text, in a double-quoted attribute AND in a
 * single-quoted one. It used to leave `'` alone on the theory that a structural
 * test proved no single-quoted attribute exists; that test matched only an
 * attribute name preceded by whitespace, on one line, so a nested template —
 * ``${`title='${esc(x)}'`}`` — walked straight past it with the whole admin
 * suite green. Correctness at the sink must not depend on a lint holding: it has
 * to hold whatever the surrounding markup turns out to be.
 *
 * This is one of ELEVEN escapers in the tree, and that population is DERIVED,
 * never listed from memory (I3-1407):
 *   grep -rnE 'replace\(/&/g, "&amp;"\)|replace\(/\[[&<>"'\'']{5}\]/g' --include=*.ts \
 *     apps packages services | grep -v node_modules | grep -v test
 * One line per escaper. The first alternative is the chained form (five of
 * them); the second is a single character-class replace in any order of the
 * five characters (six: `[&<>"']` five times, `[<>&"']` once). The command
 * printed here until I32 used `\|` and a bare `[&<>"']`, which grep reads as a
 * ONE-character bracket: it found no class-form escaper at all and counted two
 * SQL quote-doublers in alerts.ts instead. All eleven escape all five
 * characters (`&`, `<`, `>`, `"`, `'`); the one that once lacked the `'` limb,
 * `services/trust-core/src/email.ts`'s `esc`, has it now. Line numbers drift,
 * so the census grep above, not a line number here, is the authority, and
 * `apps/admin/test/i32-ops-escaper-census.test.ts` runs it against this count. */
export function esc(s: unknown): string {
  return String(s).replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;").replace(/'/g, "&#39;");
}

/** An ISO timestamp in the console's one reading format: `2026-08-18 04:12Z`.

 * It lives here, beside `esc`, because it had been defined FOUR times with

 * formatter, so the copies sat there unchecked;

/**
 * A SERVER REFUSAL, IN THE SERVER'S OWN WORDS.
 *
 * Every `/ops/*` route answers a refusal with `{error: "…"}` and a specific,
 * written sentence — *"disposition must be one of: resolved_threat, …"*,
 * *"verification item needs a parseable expires_at before it can be approved"*,
 * *"transfer_account requires contested_account_key"*, *"case not open (state
 * approved)"*. The console parsed the body and then threw it away at every one
 * of ~14 call sites, showing `Failed (400).` instead. An operator hitting a
 * validation refusal had no way to learn what was missing, so the most
 * available diagnosis was "the console is broken".
 *
 * BOUNDED ON PURPOSE, copied from the agency console, which already got this
 * right: a body longer than a sentence is not a message the server wrote, it is
 * a proxy's error page, and pasting one into the operator's status line is how
 * a failure mode becomes a rendering surface.
 *
 * `did` names the action in the operator's language ("Couldn't approve this
 * item"), so the line reads as one sentence with the server's reason attached.
 */
export function failText(status: number, data: unknown, did: string): string {
  const b = data as { error?: unknown; detail?: unknown; warning?: unknown } | null;
  // I25 (JN-02, D-0996(2)): a refusal's own sentence (`refusal(error, message)`, `errorBody`) is printed in place of
  // the machine `error` when the server wrote one; without it, `error` as before (D-1004(1): no new wording).
  const msg = refusalMessage(data);
  const said = msg ?? (typeof b?.error === "string" && b.error.length > 0 && b.error.length <= 200 ? b.error : "");
  const detail = typeof b?.detail === "string" && b.detail.length > 0 && b.detail.length <= 300 ? ` — ${b.detail}` : "";
  // RGC-09 (I15-FX3): status 0 is "no answer arrived", which is also what a
  // connection lost AFTER the server ran the request looks like — so it never
  // claims the request did not land, or that nothing changed.
  if (status === 0) return `${did}: no answer arrived from the server, so this console can't tell whether it went through. Reload and check before trying again.`;
  // I3-1325 (I15-W3L): a 2xx reaches here only when its body could not be read
  // (main.ts `unreadableReply`). `did` assumes a refusal ("nothing changed"),
  // which that does not establish, so a 2xx gets its own sentence and no `did`.
  if (status >= 200 && status < 300) return `The server answered (${status}) but its reply couldn't be read — reload before acting again.`;
  return said ? `${did} (${status}): ${said}${detail}` : `${did} (${status}).`;
}

/** I25 (JN-02): a refusal body's `message` — the sentence trust-core's `refusal(error, message?)` / `errorBody` writes
 * for a person (route-helpers.ts) — or null: a non-empty string of at most 300 characters (web api.ts refusalMessage). */
export function refusalMessage(data: unknown): string | null {
  const m = data !== null && typeof data === "object" ? (data as { message?: unknown }).message : undefined;
  return typeof m === "string" && m.trim() !== "" && m.length <= 300 ? m : null;
}

/** Truncate FIRST, then escape — never the other way round.
 *
 * `esc(s).slice(0, n)` reads naturally and is wrong: it measures the escaped
 * string, so the budget shrinks with every `&` in the input, and a cut can land
 * inside an entity and emit `&am`. It cannot reintroduce a `<` (there is no `<`
 * left to reintroduce), so this is a rendering bug rather than an injection one
 * — but it is the kind of ordering that stops being cosmetic the moment the
 * escaper changes, and it changed here: `&#39;` is five characters wide now.
 * Doing it in one place means no call site has to get the order right.
 *
 * AL-12: `max` still counts UTF-16 code units (this budget is small and
 * console-only, not worth a grapheme library), but the cut itself must not
 * land inside a surrogate pair — that turns one displayable character into
 * two invalid ones instead of just shortening the string. If the character
 * straddling the cut is a high surrogate followed by its low surrogate, the
 * budget backs up by one unit so the pair stays whole. */
export function escClamp(s: unknown, max: number): string {
  const t = String(s ?? "");
  if (t.length <= max) return esc(t);
  let cut = max;
  const before = t.charCodeAt(cut - 1);
  const after = t.charCodeAt(cut);
  if (before >= 0xd800 && before <= 0xdbff && after >= 0xdc00 && after <= 0xdfff) cut -= 1;
  return `${esc(t.slice(0, cut))}…`;
}

 * FAIL CLOSED ON THE COPY: a missing or unrecognised `state` renders as
 * "unverified" at best and never as "verified"; an unrecognised `signature`
 * renders its own sentence rather than `undefined`; and the word "signed" is
 * printed only when a signature actually verified, under a key id we can name.
 */

/** The server's folded verdict. Mirrors `IntegrityState` in checkpoint.ts. */
export type IntegrityState = "verified" | "unverified" | "broken";

/** Why `state` is what it is, on the signature axis. Mirrors `SignatureState`. */
export type SignatureState = "verified" | "absent" | "unverifiable" | "forged" | "disabled" | "wrong_deployment";

export interface LedgerIntegrity {
  ok: boolean;
  events: number;
  brokenAtSeq?: number;
  reason?: string;
  anchored: boolean;
  anchorSeq?: number;
  anchorReason?: string;
  head: { seq: number; event_hash: string } | null;
  recent_anchors: { seq: number; subject: string; created_at: string }[];
  /** The field to switch on. OPTIONAL on the wire: an older deployment, or a
   * trimmed response, simply does not carry it — and that reads as "unverified",
   * never as a pass. */
  state?: IntegrityState;
  signature?: SignatureState;
  signatureReason?: string;
  signatureKeyId?: string;
  /** I3-0275 (I16-W01C): the operator audit-log seal's coverage, mirrored from
   * services/trust-core/src/admin.ts `AuditLogSealCoverage`. OPTIONAL on the
   * wire: a deployment before the seal does not carry it. */
  audit_log?: {
    chained: boolean;
    ok: boolean;
    seals: number;
    sealed_through: number | null;
    unsealed_rows: number;
    verified_from: number | null;
    broken_seal?: number;
    reason?: string;
    note: string;
  };
}

/** What this component actually draws: the three server states plus the absence. */
export type IntegrityRenderState = IntegrityState | "unknown";

const SIGNATURE_STATES: readonly string[] = ["verified", "absent", "unverifiable", "forged", "disabled", "wrong_deployment"];

/** The signature state, or null when the server sent nothing we recognise. */
function signatureOf(d: LedgerIntegrity): SignatureState | null {
  return typeof d.signature === "string" && SIGNATURE_STATES.includes(d.signature) ? (d.signature as SignatureState) : null;
}

/** The published-key-id grammar, mirrored from `KEY_ID_RE` in checkpoint.ts.
 * Only ids in this shape are ever published, so an id outside it did not come
 * from the key set — whatever else it is, it is not something to make a claim
 * about. */
const KEY_ID_RE = /^[a-z0-9][a-z0-9._-]{0,63}$/;

/** The raw key id the response carried, for the BROKEN banner, where the id is
 * evidence rather than a claim. Escaped at every render site. */
function rawKeyId(d: LedgerIntegrity): string | null {
  const k = typeof d.signatureKeyId === "string" ? d.signatureKeyId.trim() : "";
  return k === "" ? null : k;
}

/**
 * The key id, only when there is a real one to name. A blank, absent or
 * out-of-grammar key id is the difference between "signed by ck_2026a" and
 * "signed by undefined" — or worse, "signed by" something the key set never
 * published. Neither sentence may exist, so an id that fails here downgrades the
 * whole render to `unverified` rather than decorating a claim.
 */
export function signatureKeyId(d: LedgerIntegrity): string | null {
  const k = rawKeyId(d);
  return k !== null && KEY_ID_RE.test(k) ? k : null;
}

/**
 * THE ONE RESOLUTION FUNCTION. Everything on both surfaces switches on this and
 * nothing switches on `ok && anchored` any more.
 *
 * It fails closed in both directions: anything that smells broken is broken, and
 * "verified" requires ALL FOUR of the things that claim rests on — a clean
 * chain, a compared anchor, a signature the server says verified, and a key id
 * to name. Any one missing and the screen falls back to the narrower sentence.
 */
export function integrityState(d: LedgerIntegrity | null): IntegrityRenderState {
  if (d === null) return "unknown";
  if (d.ok !== true || d.state === "broken" || signatureOf(d) === "forged") return "broken";
  if (d.state === "verified" && signatureOf(d) === "verified" && d.anchored === true && signatureKeyId(d) !== null) return "verified";
  return "unverified";
}

/**
 * THE SIGNATURE CHIP, one sentence per cause. `forged` is the only one that is
 * ever drawn in red, and it is drawn only inside the broken banner.
 */
export const SIGNATURE_NOTE: Record<Exclude<SignatureState, "verified">, string> = {
  disabled: "checkpoint signing is not configured on this deployment",
  absent: "the signing authority was unreachable when this anchor was taken — the next signed anchor covers the gap",
  unverifiable: "the key that signed this anchor is not published",
  forged: "a checkpoint object carries a signature that does not verify under a published key — someone wrote an anchor they could not sign",
  // I3-1101 (I23): trust-core checkpoint.ts `wrong_deployment` — a v2 signature that verifies, over a preimage naming
  // ANOTHER deployment. Reported, never a ledger breach: amber, with the server's `signatureReason`.
  wrong_deployment: "another deployment's anchor is in this store — it verifies, but it was signed for a different deployment",
};

/** What we say when the server sent nothing, or something we do not recognise —
 * or claims "verified" while failing one of the other three conditions. Never
 * `undefined`, and never a claim. */
export const SIGNATURE_UNREPORTED = "this anchor's signature was not reported — treat it as unsigned";

/** The chip sentence for a NON-verified render. Never returns the forged line
 * here: forged renders as the alarm, not as a slate chip. */
export function signatureNote(d: LedgerIntegrity): string {
  const s = signatureOf(d);
  if (s === null || s === "verified" || s === "forged") return SIGNATURE_UNREPORTED;
  return SIGNATURE_NOTE[s];
}

/**
 * WHY THIS IS NOT AMBER — the lead clause, per cause. The reasoning after it is
 * verbatim what the no-anchor case has always said, because it has always been
 * the same argument: a narrower claim is not a fault, and amber spent on a
 * narrower claim is amber that gets ignored on the day it means something.
 */
function notAmberLead(d: LedgerIntegrity): string {
  if (!d.anchored) {
    return "A missing anchor is not a fault, it is a younger deployment — or an environment with no object-store binding, which is config being absent on purpose rather than anything going wrong.";
  }
  switch (signatureOf(d)) {
    case "disabled":
      return "A deployment with no published key set is not a fault — checkpoint signing is config that is absent on purpose, and the anchor comparison above still ran and still matched.";
    case "absent":
      return "An anchor taken while the signing authority was unreachable is an outage that has already ended, not a rewrite — and the next signed anchor covers the gap.";
    case "unverifiable":
      return "A key that is no longer published makes an old anchor unverifiable rather than wrong — that is what key rotation looks like from here.";
    default:
      return "A signature status this console did not receive is a narrower read, not a fault — it is treated as unsigned until one arrives.";
  }
}

export function brokenSeq(d: LedgerIntegrity): number | null {
  if (d.brokenAtSeq !== undefined) return d.brokenAtSeq;
  if (d.anchorSeq !== undefined) return d.anchorSeq;
  return null;
}

/**
 * The reason, preferring the chain's own over the anchor's over the signature's.
 *
 * The forged case is why the third fallback exists: it arrives with no `reason`
 * and often no `anchorReason` either — the chain recomputed, the anchor matched,
 * and the ONLY thing wrong is that a checkpoint carries a signature nobody with
 * a published key could have made. Without this, the worst finding on the whole
 * screen would print "reason not reported by the check".
 */
export function brokenReason(d: LedgerIntegrity): string {
  const first = [d.reason, d.anchorReason, signatureOf(d) === "forged" ? d.signatureReason : undefined]
    .find((r) => typeof r === "string" && r.trim() !== "");
  return first ?? "reason not reported by the check";
}

/**
 * THE HEADLINE. This exact sentence appears on the Overview strip and at the top
 * of the Abuse screen, unmodified, so the two can never drift.
 */
export function integrityHeadline(d: LedgerIntegrity | null): string {
  if (d === null) return "Ledger integrity could not be read — treat as UNVERIFIED, not as intact";
  const state = integrityState(d);
  if (state === "broken") {
    const seq = brokenSeq(d);
    return `CHAIN BROKEN${seq === null ? "" : ` at seq ${seq}`} — ${brokenReason(d)}`;
  }
  const n = Number.isFinite(d.events) ? String(d.events) : "an unreported number of";
  // The ONLY sentence in the product that says an anchor was signed, and it
  // names the key that signed it. `integrityState` has already established that
  // the signature verified AND that there is a key id to name.
  if (state === "verified") return `Chain intact — ${n} events, external anchor matched and signed (key ${signatureKeyId(d)})`;
  if (d.anchored) return `Chain intact — ${n} events, external anchor matched`;
  return `Chain intact — ${n} events, internal recompute (no external anchor compared)`;
}

/** One line, for the Overview attention strip. Hushed unless it is the alarm. */
export function integrityAttentionRow(d: LedgerIntegrity | null, checkedLabel: string): string {
  const state = integrityState(d);
  const head = integrityHeadline(d);
  if (state === "broken") {
    return `<div class="arow"><span class="ic red">⚠</span>
      <b class="alarm">${esc(head)}</b>
      <span class="right"><span class="stamp">${esc(checkedLabel)}</span><a href="#/abuse">Integrity</a></span></div>`;
  }
  // I3-1101: a wrong-deployment anchor is amber here too, never hushed under the green line.
  if (state === "unverified" && d !== null && signatureOf(d) === "wrong_deployment") {
    return `<div class="arow"><span class="ic amb">!</span>
      <b>${esc(head)}</b>
      <span class="sub">${esc(wrongDeploymentText(d))}</span>
      <span class="right"><span class="stamp">${esc(checkedLabel)}</span><a href="#/abuse">Integrity</a></span></div>`;
  }
  if (state === "unknown") {
    return `<div class="arow"><span class="ic amb">!</span>
      <b>${esc(head)}</b>
      <span class="sub">the check did not answer — that is not the same as a pass</span>
      <span class="right"><span class="stamp">${esc(checkedLabel)}</span><a href="#/abuse">Integrity</a></span></div>`;
  }
  return `<div class="arow hush"><span class="ic ok">✓</span>
    <b><span class="lit">${esc(head)}</span></b>
    <span class="right"><span class="stamp">${esc(checkedLabel)}</span><a href="#/abuse">Integrity</a></span></div>`;
}

const CHIP = (cls: string, text: string): string => `<span class="ichip ${cls}">${esc(text)}</span>`;

/** I3-1101: the amber line for a `wrong_deployment` signature — the server's own reason when it sent one. */
export function wrongDeploymentText(d: LedgerIntegrity): string {
  const r = typeof d.signatureReason === "string" && d.signatureReason.trim() !== "" ? d.signatureReason.trim() : "";
  return r ? `${SIGNATURE_NOTE.wrong_deployment} (${r})` : SIGNATURE_NOTE.wrong_deployment;
}

function fact(k: string, value: string, note = ""): string {
  return `<div class="ifact"><span class="k">${esc(k)}</span><span class="num">${esc(value)}</span>${note ? `<span>${note}</span>` : ""}</div>`;
}

/**
 * THE FOLD REPAIR ZONE — `POST /ops/fold/rebuild`, the operator's only repair
 * tool for the projection, which had no button until now.
 *
 * `passports` / `passport_accounts` are a FOLD of `trust_events`, not a source of
 * truth. They drift for ordinary reasons — most often a migration that adds a
 * fold column with no backfill — and `fold/verify` next door is what finds it.
 * Finding drift with nothing to press is how a deploy ships a half-populated
 * projection and serves it.
 *
 * WHY IT SITS BEHIND THE VERIFY, AND ONLY OPENS WHEN THE CHAIN IS SOUND. The
 * rebuild REFUSES on a broken chain by design, and step 2 of the broken runbook
 * above says do not work around that: a rebuild over a break would launder the
 * break into projected state and destroy the discrepancy the investigation
 * needs. So on a broken chain this zone renders the refusal instead of the
 * control — the button is not merely disabled, it is replaced by the reason.
 *
 * IT IS NOT DRAWN AS DESTRUCTIVE, and that is a claim about the operation rather
 * than a softening. A rebuild writes nothing to the ledger, is idempotent, is
 * FK-safe (UPSERT, never a truncate), and self-verifies before it reports
 * success. What it cannot do is invent state the log does not contain — so a
 * rebuild that "fixes" a number you expected to see differently has told you
 * something about the log, not about the rebuild.
 */
export function foldRepairZone(d: LedgerIntegrity | null, role: OpsRole): string {
  const state = integrityState(d);
  if (state === "broken") {
    return `<div class="zone destructive">
      <div class="k">⚠ Fold rebuild — refused while the chain is broken</div>
      <p><b>There is no button here right now, and that is the correct behaviour.</b> The rebuild refuses on a broken chain by design and the runbook above says not to work around it: replaying a chain that does not recompute would write the break into <code>passports</code> and <code>passport_accounts</code> and destroy the discrepancy the investigation needs.</p>
      <p><b>Fix the chain first.</b> Preserve the table, work the runbook, and come back to this when a recompute passes. A fold that disagrees with a broken log is not the problem you have.</p>
    </div>`;
  }
  return `<div class="zone">
    <div class="k">Fold rebuild</div>
    <p><code>passports</code> and <code>passport_accounts</code> are a <b>projection of the ledger</b>, not a second source of truth. When <b>fold verify</b> reports drift, this replays the whole log and rewrites the projection to match it. Run it as a maintenance step <b>before serving</b> a populated deploy of a new fold column.</p>
    <p>It writes <b>nothing to the ledger</b>, is idempotent, never truncates the FK-referenced passport rows, self-verifies before reporting success, and refuses outright on a broken chain. It is audited under your Access identity. <b>It cannot invent state the log does not contain</b> — if a rebuild changes a number you did not expect to move, that is a fact about the log.</p>
    <div class="row spaced">
      ${writeControl({ label: "Verify first (read-only)", role, attrs: { "data-act": "fold-verify" }, lock: false })}
      ${writeControl({ label: "Rebuild the fold…", role, tone: "danger", attrs: { "data-act": "fold-rebuild-open" }, lock: false })}
      ${role === "support" ? lockChip() : ""}
    </div>
    <p class="panenote"><b>Verify before you rebuild, every time.</b> The drift report names the first divergent keys, and that is the only chance to see WHAT was wrong before the evidence of it is overwritten by the correct values.</p>
  </div>`;
}

  if (role !== "full") return "";
  return `<div class="scrim open" id="m-fold-rebuild">

/**

// --- 1. Which seat is this? --------------------------------------------------

/** What the console believes about the operator sitting in front of it.
 * `confirmed` is the load-bearing field: it is the difference between "the
 * server told us this is a support seat" and "we never got an answer". */
export interface Seat {
  role: OpsRole;
  who: string;
  /** True only when `/ops/whoami` returned an answer we could actually read. */
  confirmed: boolean;
}

/**
 * The seat the console holds before — and unless — the server says otherwise.
 *
 * It defaults to SUPPORT. It used to default to `full`, and `boot()` only ever
 * downgraded: `api()` returns null on a network failure, a 5xx, an Access
 * redirect that lands on HTML, or anything else non-OK, and every one of those
 * left a support operator holding the entire full console — live purge and erase
 * confirms and a live Export button. That is invariant 6 exactly inverted
 * (absent config must disable a feature, never open it), on the one surface
 * whose own copy tells the operator "the server refuses it too".
 */
export const UNCONFIRMED_SEAT: Seat = { role: "support", who: "", confirmed: false };

/**
 * Resolve `/ops/whoami` into a seat. UPGRADES ONLY on a successful, well-formed
 * read; every other shape lands on `UNCONFIRMED_SEAT`.
 *
 * "Well-formed" means the role is exactly one of the two known tokens. An
 * unrecognised role — a new server tier, a truncated body, a proxy's JSON error
 * envelope that happens to parse — is not a reason to guess upward.
 *
 * THIS IS ONLY THE CLIENT HALF OF THE FIX, AND IT IS NOT SUFFICIENT ALONE. The
 * console has never been the security boundary: the server 403s a support seat
 * on every non-GET under `/ops`, and a peer is separately closing the hole where
 * a support seat could actually call the DSAR export (a GET, so the blanket
 * non-GET rule never covered it). Neither half is sufficient without the other —
 * this one stops the console from *presenting* powers the seat may not have and
 * from lying to the operator about it; only the server can stop the request.
 */
export function resolveSeat(raw: unknown): Seat {
  if (typeof raw !== "object" || raw === null) return UNCONFIRMED_SEAT;
  const { role, email } = raw as { role?: unknown; email?: unknown };
  if (role !== "full" && role !== "support") return UNCONFIRMED_SEAT;
  return { role, who: typeof email === "string" ? email : "", confirmed: true };
}

// --- 2. Did they type the id exactly? ---------------------------------------

/**
 * The typed-confirm gate in front of the two irreversible acts (erase an
 * account, purge an evidence object): the operator must retype the exact id.
 *
 * Exact, case-sensitive, and no normalisation beyond trimming what was typed —
 * an id that differs only in case is a DIFFERENT account, and "close enough" is
 * not a property this control may ever have.
 *
 * The empty-expected guard is the one addition over the inline comparison it
 * replaces. `typed !== id` treated an empty id as satisfied by an empty box: a
 * dossier that somehow rendered with a blank id would hand out a one-click
 * confirm on a still-armed destructive control. Nothing is confirmable against
 * an id we do not have.
 */
export function typedConfirmMatches(typed: unknown, expected: unknown): boolean {
  if (typeof expected !== "string" || expected.trim() === "") return false;
  if (typeof typed !== "string") return false;
  return typed.trim() === expected;
}

// --- 3. Exactly one live blob URL, and it dies with the screen ---------------

/** The injected half of the slot, so a test can watch the lifecycle without a
 * DOM. Defaults to the real `URL` in the browser. */
export interface ObjectUrlIo {
  create(blob: Blob): string;
  revoke(url: string): void;
}

/**
 * A slot holding AT MOST ONE live object URL, revoking the previous one on every
 * change and on release.
 *
 * Two payloads on this console need this and only one had it. The evidence
 * viewer got it right ("operator-eyes-only bytes do not outlive the screen that
 * showed them" — `route()` revokes on every navigation). The DSAR export did
 * not: `dl.href = URL.createObjectURL(...)` was never tracked and never revoked,
 * so a named stranger's COMPLETE personal data — the single most sensitive
 * payload the console can assemble — stayed live and fetchable for the rest of
 * the session, one more per press, unbounded, in the same document that renders
 * attacker-supplied handles. Sharing one implementation is what stops the next
 * blob from being the third case that forgets.
 */
export interface ObjectUrlSlot {
  /** The live URL, or null. */
  readonly current: string | null;
  /** Take a blob, revoke whatever was here, return the new URL. */
  set(blob: Blob): string;
  /** Adopt an already-created URL (or null to just release). */
  adopt(url: string | null): void;
  /** Open a new generation and return its token. An `adoptIf` carrying an
   * older token is a superseded fetch: its URL is revoked and the slot is
   * left alone. */
  begin(): number;
  /** Adopt only if `token` is still the live generation. Returns whether it
   * was adopted; a false return means the caller must not render either. */
  adoptIf(token: number, url: string | null): boolean;
  /** Revoke and empty. Safe to call any number of times, in any state. */
  release(): void;
}

export function objectUrlSlot(io: ObjectUrlIo = defaultObjectUrlIo()): ObjectUrlSlot {
  let live: string | null = null;
  let gen = 0;
  const release = (): void => {
    if (live !== null) io.revoke(live);
    live = null;
  };
  return {
    get current(): string | null {
      return live;
    },
    set(blob: Blob): string {
      const url = io.create(blob);
      release();
      live = url;
      return url;
    },
    adopt(url: string | null): void {
      release();
      live = url;
    },
    begin(): number {
      return ++gen;
    },
    adoptIf(token: number, url: string | null): boolean {
      if (token !== gen) {
        if (url !== null) io.revoke(url);
        return false;
      }
      release();
      live = url;
      return true;
    },
    release,
  };
}

function defaultObjectUrlIo(): ObjectUrlIo {
  return {
    create: (blob) => URL.createObjectURL(blob),
    revoke: (url) => URL.revokeObjectURL(url),
  };
}

/**
 * THE LEGAL FOOTER — Terms and Privacy, rendered IN THIS CONSOLE (I3-1646, D-0995(4), I23-FX-APPS).
 *
 * The text is the one `@dhwaja/legal` source the fan app and trust-core's `/terms` / `/privacy` print, so this
 * console needs no other origin to show it: the rail links the in-app `#/terms` and `#/privacy`, and `route()` in

 *
 * I21-Q146 (D-1073(1)): the footer also links About — trust-core's server-rendered GET /about (about-page.ts) — on

 * URL). `opsBase` defaults to that build define; a value that is not a bare https origin (or an unbuilt module, where
 * the define is absent) renders no About link, never a dead or hostile one.
 */
declare const __OPS_BASE__: string | undefined;
const BARE_HTTPS_ORIGIN = /^https:\/\/[a-z0-9.-]+(:\d+)?$/i;
export function legalFooter(opsBase: unknown = typeof __OPS_BASE__ === "string" ? __OPS_BASE__ : undefined): string {
  const link = (hash: "#/terms" | "#/privacy", label: string): string => `<a href="${esc(hash)}">${esc(label)}</a>`;

  return `<footer class="foot"><nav>${link("#/terms", "Terms")}${link("#/privacy", "Privacy")}${about}</nav></footer>`;
}
