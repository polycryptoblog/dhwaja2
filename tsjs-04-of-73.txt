/**
 * I3-0422 a — THE IN-PAGE CONFIRM AND PROMPT (creator console and agency console).
 *
 * `window.confirm`/`window.prompt` are the browser's own boxes: no title, no focus return, one wall of text, dismissed
 * by muscle memory, and the same look for "publish a reply" as for "remove every payout destination". This asks the
 * same question inside the page, with the cost sentences as their own paragraphs, in the DOM shape admin's
 * `confirmDialog`/`reasonDialog` use (apps/admin/src/main.ts) — `role="alertdialog"`, a button whose data-act is `confirm-go`
 * / `confirm-cancel`, a `-retype` input under `label code`, `reason-go` / `reason-cancel` with a textarea — so one test
 * driver answers all three consoles.
 *
 * RETYPE-TO-CONFIRM is for money and for acts the person cannot undo themselves (the I3-0422 a fix shape): the caller
 * passes `retype`, and the confirm button does nothing until the box holds exactly that text (`retypeMatches`: trimmed,
 * case-sensitive, and an empty expected text never matches — admin `typedConfirmMatches`, apps/admin/src/interlocks.ts).
 *
 * `true` only on the confirm button. Cancel, Escape and the backdrop are all `false`, so nothing proceeds on a
 * dismissal. One dialog at a time: a second ask waits for the first to settle.
 *
 * apps/agency/src/dialog.ts is a BYTE COPY of this file (no shared package: D-0922(2)); the copy is pinned identical by
 * apps/creator-console/test/i22-fx-consoles.test.ts.
 */
export interface AskOptions {
  /** The dialog heading. Defaults to "Are you sure?". */
  title?: string;
  /** The confirm button's label. Defaults to "Continue". */
  confirmLabel?: string;
  /** When set, the person must type exactly this before the confirm button acts. */
  retype?: string;
}

/** Exact, trimmed, case-sensitive; nothing is confirmable against an empty expected text. */
export function retypeMatches(typed: unknown, expected: unknown): boolean {
  if (typeof expected !== "string" || expected.trim() === "") return false;
  if (typeof typed !== "string") return false;
  return typed.trim() === expected;
}

let seq = 0;
let queue: Promise<unknown> = Promise.resolve();

function enqueue<T>(run: () => Promise<T>): Promise<T> {
  const next = queue.then(run, run);
  queue = next.catch(() => undefined);
  return next;
}

/** The message's blank-line-separated paragraphs, each set as TEXT (never markup). */
function paragraphs(doc: Document, message: string): HTMLElement[] {
  return message
    .split(/\n\s*\n/)
    .map((p) => p.trim())
    .filter((p) => p.length > 0)
    .map((p) => {
      const el = doc.createElement("p");
      el.textContent = p;
      return el;
    });
}

function mount<T>(build: (box: HTMLElement, id: string, settle: (v: T) => void) => HTMLElement, cancelValue: T): Promise<T> {
  return enqueue(
    () =>
      new Promise<T>((resolve) => {
        const doc = document;
        const id = `app-dialog-${++seq}`;
        const holder = doc.createElement("div");
        holder.className = "dlg-scrim";
        holder.setAttribute("data-app-dialog", "");
        const box = doc.createElement("div");
        box.className = "dlg";
        box.setAttribute("role", "alertdialog");
        box.setAttribute("aria-modal", "true");
        box.setAttribute("aria-labelledby", `${id}-h`);
        holder.append(box);
        const trigger = doc.activeElement as HTMLElement | null;
        let done = false;
        const settle = (v: T): void => {
          if (done) return;
          done = true;
          doc.removeEventListener("keydown", onKey, true);
          holder.remove();
          if (trigger && typeof trigger.focus === "function" && trigger.isConnected) trigger.focus();
          resolve(v);
        };
        const onKey = (e: KeyboardEvent): void => {
          if (e.key === "Escape") {
            e.preventDefault();
            settle(cancelValue);
            return;
          }
          if (e.key !== "Tab") return;
          const f = [...box.querySelectorAll<HTMLElement>("button, input, textarea")];
          if (f.length === 0) return;
          const first = f[0]!;
          const last = f[f.length - 1]!;
          if (e.shiftKey && doc.activeElement === first) { e.preventDefault(); last.focus(); }
          else if (!e.shiftKey && doc.activeElement === last) { e.preventDefault(); first.focus(); }
        };
        const focusFirst = build(box, id, settle);
        holder.addEventListener("click", (e) => { if (e.target === holder) settle(cancelValue); });
        doc.addEventListener("keydown", onKey, true);
        doc.body.append(holder);
        focusFirst.focus();
      }),
  );
}

function heading(doc: Document, id: string, title: string): HTMLElement {
  const h = doc.createElement("h2");
  h.id = `${id}-h`;
  h.textContent = title;
  return h;
}

function button(doc: Document, act: string, label: string, primary: boolean): HTMLButtonElement {
  const b = doc.createElement("button");
  b.type = "button";
  b.className = primary ? "ds-btn primary" : "ds-btn";
  b.setAttribute("data-act", act);
  b.textContent = label;
  return b;
}

/** In-page replacement for `window.confirm(message)`: resolves `true` only on the confirm button. */
export function askConfirm(message: string, opts: AskOptions = {}): Promise<boolean> {
  return mount<boolean>((box, id, settle) => {
    const doc = box.ownerDocument;
    box.append(heading(doc, id, opts.title ?? "Are you sure?"), ...paragraphs(doc, message));
    let field: HTMLInputElement | null = null;
    const err = doc.createElement("p");
    err.className = "dlg-err";
    err.setAttribute("role", "alert");
    if (opts.retype !== undefined) {
      const p = doc.createElement("p");
      const label = doc.createElement("label");
      const code = doc.createElement("code");
      code.textContent = opts.retype;
      field = doc.createElement("input");
      field.id = `${id}-retype`;
      field.className = "dlg-field";
      field.autocomplete = "off";
      field.spellcheck = false;
      label.append("Type ", code, " to confirm ", field);
      p.append(label);
      box.append(p, err);
    }
    const row = doc.createElement("div");
    row.className = "dlg-row";
    const cancel = button(doc, "confirm-cancel", "Cancel", false);
    const go = button(doc, "confirm-go", opts.confirmLabel ?? "Continue", true);
    row.append(cancel, go);
    box.append(row);
    cancel.addEventListener("click", () => settle(false));
    go.addEventListener("click", () => {
      if (field && !retypeMatches(field.value, opts.retype)) {
        err.textContent = "That doesn't match. Type it exactly to confirm — nothing has happened yet.";
        field.focus();
        return;
      }
      settle(true);
    });
    return field ?? cancel;
  }, false);
}

/** In-page replacement for `window.prompt(message)`: the trimmed text on Continue, `null` on Cancel/Escape. */
export function askText(message: string, opts: { title?: string; placeholder?: string; maxLength?: number } = {}): Promise<string | null> {
  return mount<string | null>((box, id, settle) => {
    const doc = box.ownerDocument;
    box.append(heading(doc, id, opts.title ?? "Add a note"), ...paragraphs(doc, message));
    const ta = doc.createElement("textarea");
    ta.id = `${id}-reason`;
    ta.className = "dlg-field";
    ta.maxLength = opts.maxLength ?? 500;
    if (opts.placeholder) ta.placeholder = opts.placeholder;
    ta.setAttribute("aria-labelledby", `${id}-h`);
    const p = doc.createElement("p");
    p.append(ta);
    const row = doc.createElement("div");
    row.className = "dlg-row";
    const cancel = button(doc, "reason-cancel", "Cancel", false);
    const go = button(doc, "reason-go", "Continue", true);
    row.append(cancel, go);
    box.append(p, row);
    cancel.addEventListener("click", () => settle(null));
    go.addEventListener("click", () => settle(ta.value.trim()));
    return ta;
  }, null);
}

/**
 * AU-02 — A BROWSER THAT WILL NOT WRITE IS A STATE, NOT A SILENT FAILURE.
 *
 * `setToken` swallowed every `localStorage.setItem` exception and returned
 * nothing, and both request helpers re-read the bearer FROM STORAGE. So on an
 * origin whose storage is unavailable or full — private modes, site data
 * blocked, quota exhausted — a perfectly good sign-in was accepted by the
 * server, discarded locally, and the very next read went out with an empty
 * bearer and came back 401. The console then said "Your session expired or was
 * revoked", which is false about a session issued one second earlier, and
 * signing in again repeats it forever with no way out and nothing naming the
 * real cause.
 *
 * `memToken` is the fallback: the session lives for this TAB. That is strictly
 * LESS persistent than the previous intent, never more — nothing new is written
 * anywhere — and it is the difference between a usable console and a loop. The
 * key and the provider token are still never stored, in memory or otherwise.
 */
let memToken = "";
/** Whether this device is holding the session in memory only, because storage
 * refused it. Drives the copy: the person needs to know a tab close ends it. */
let tokenIsTabOnly = false;
/**
 * I3-1238 — the memory copy is consulted ONLY while storage refused the write
 * (`tokenIsTabOnly`). It used to be the fallback for ANY empty read, so a
 * session another tab had signed out of (storage emptied) stayed usable here
 * from memory. Storage, when it works, is the one source of truth.
 */
/** I25-W1-COOKIE: the handle lives in memory only; the durable half is the HttpOnly cookie, which the browser keeps
 * (or not) and GET /session/csrf reports on, so AU-02's "storage refused it" state can no longer arise: `setToken`
 * always reports durable and `tokenIsTabOnly` stays false. */
function getToken(): string {
  return memToken;
}
function setToken(t: string): boolean {
  memToken = t;
  tokenIsTabOnly = false;
  return true;
}
function clearToken(): void { memToken = ""; tokenIsTabOnly = false; }
/** The request headers for the held handle (none when signed out). */
function authHeader(handle = getToken()): Record<string, string> {
  return handle ? { authorization: `Dhwaja-Csrf ${handle}` } : {};
}
/** I25-W1-COOKIE: the bootstrap and cross-tab sync — the handle of the agency cookie, or "" (401). Status 0 / 5xx: null. */
async function fetchHandle(): Promise<string | null> {
  try {
    const r = await fetch(`${API}/session/csrf?kind=agency`, { credentials: "include", cache: "no-store", headers: { accept: "application/json" } });
    if (r.status === 401) return "";
    if (!r.ok) return null;
    const d = (await r.json()) as { csrf?: unknown };
    return typeof d.csrf === "string" && /^[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+$/.test(d.csrf) ? d.csrf : null;
  } catch { return null; }
}

/** I25-W1-COOKIE: expire this browser's agency cookie (POST /session/signout under the handle that was held). */

function esc(s: unknown): string {
  // All FIVE characters, matching every other escaper in the repo (see
  // apps/admin/src/html.ts for why the four-character version is a latent
  // hazard: a single-quoted attribute slips past a structural lint and the
  // quote walks straight out of it).
  return String(s).replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;").replace(/'/g, "&#39;");
}

async function api<T>(path: string): Promise<{ ok: boolean; status: number; data: T | null }> {
  try {
    const gen = sessionGen;
    const sent = getToken();
    const r = await fetch(`${API}${path}`, { credentials: "include", headers: { accept: "application/json", ...authHeader(sent) } });
    reExpireIfStale(gen, sent, r.ok);
    if (r.status === 401) { onUnauthorized(gen); return { ok: false, status: 401, data: null }; }
    let data: T | null = null;
    try { data = (await r.json()) as T; } catch { /* no body */ }
    return { ok: r.ok, status: r.status, data };
  } catch { return { ok: false, status: 0, data: null }; }
}

async function apiPost<T>(path: string, body: unknown, handle = getToken()): Promise<{ ok: boolean; status: number; data: T | null }> {
  try {
    const gen = sessionGen;
    const r = await fetch(`${API}${path}`, { method: "POST", credentials: "include", headers: { "content-type": "application/json", accept: "application/json", ...authHeader(handle) }, body: JSON.stringify(body) });
    reExpireIfStale(gen, handle, r.ok);
    if (r.status === 401) { onUnauthorized(gen); return { ok: false, status: 401, data: null }; }
    let data: T | null = null;
    // JN-W2A / I3-1325 (I15-FX3), as the admin console's apiPost (I15-W2C): every
    // agency POST answers its 2xx with JSON, so a 2xx whose body does not parse is
    // an answer we could not READ — never a success. `data` stays null, which is
    // what `issueFailure`'s "the answer never arrived" arm keys on, and `failText`
    // prints a 2xx as "answered, unreadable", never as a refusal.
    try { data = (await r.json()) as T; } catch { if (r.ok) return { ok: false, status: r.status, data: null }; }
    return { ok: r.ok, status: r.status, data };
  } catch { return { ok: false, status: 0, data: null }; }
}

let signedOut = false;

/**
 * UI-AUDIT F-G1 — THE DEAD CONSOLE THAT LOOKED LIKE A WORKING ONE.
 *
 * `api()` and `apiPost()` call `onUnauthorized()` on a 401, which clears the
 * token and paints the login screen — and then RETURN `{ok:false}` to a caller
 * that goes straight on to `root.innerHTML = chrome(…)`. So the login screen
 * this function had just rendered was immediately overwritten by the full agency
 * chrome: the agency's name, an `active` status chip, the seat-role chip, and
 * Extend / Sign out / Team buttons — every one of them dead, because the token
 * was gone. The only sign anything was wrong was one inner sentence
 * ("Couldn't load your creators"), which reads as a transient hiccup, not as
 * "you are signed out".
 *
 * Every render path did this: the roster, the dashboard and the creator detail
 * all repaint on `!d.ok` without asking WHY it was not ok. `sessionEnded()` is
 * that question, and every one of them bails on it now, leaving the login screen
 * standing — which is both true and the thing that gets the agency back in.
 */
function sessionEnded(): boolean {
  return signedOut;
}

/**
 * WHICH SESSION a request belongs to. Bumped by a successful sign-in and by
 * nothing else.
 *
 * `onUnauthorized` used to act on ANY 401, whenever it landed. A review drove
 * the consequence: a slow read from the OLD session (an intelligence fetch that
 * takes half a second) answers 401 AFTER the operator has signed in again — and
 * it deleted the brand-new valid session from storage, re-armed `signedOut`, and
 * told them "your session expired or was revoked", which was false about the
 * session they were holding. A 401 is only ever news about the session that
 * asked.
 */
let sessionGen = 0;

function onUnauthorized(gen: number): void {
  if (gen !== sessionGen) return; // a dead session's answer, arriving late
  if (signedOut) return;
  signedOut = true;
  clearToken();
  renderLogin("Your session expired or was revoked. Sign in again with your agency key.");
}

// I26 (I25-FX-COOKIE residual): no `storage` listener. Nothing in this build writes TOKEN_KEY, so the only event for it
// is another tab's start-up purge (above), which the I3-1238 listener read as a sign-out and signed this tab out on.
// I25-W1-COOKIE: with the session in a cookie, another tab's sign-out reaches this one as a 401 from GET /session/csrf,
// asked when the tab regains focus. A changed handle (another sign-in on this browser) is adopted.
window.addEventListener("focus", () => {
  if (signedOut || !getToken()) return;
  const gen = sessionGen;
  void fetchHandle().then((h) => {
    if (gen !== sessionGen || signedOut || h === null) return;
    if (h !== "") { setToken(h); return; }
    signedOut = true;
    sessionGen++;
    clearToken();
    renderLogin("You signed out in another tab. Sign in again with your agency key.");
  });
});

// --- Provider SDK loaders (Google Identity Services / Sign in with Apple JS) ---
const scriptPromises = new Map<string, Promise<void>>();
function loadScript(src: string): Promise<void> {
  let p = scriptPromises.get(src);
  if (!p) {
    p = new Promise<void>((resolve, reject) => {
      const s = document.createElement("script");
      s.src = src; s.async = true;
      s.onload = () => resolve();
      s.onerror = () => reject(new Error(`failed to load ${src}`));
      document.head.appendChild(s);
    });
    scriptPromises.set(src, p);
  }
  return p;
}

interface GoogleId { accounts: { id: {
  initialize(cfg: { client_id: string; nonce?: string; callback: (r: { credential?: string }) => void; ux_mode?: string; auto_select?: boolean; cancel_on_tap_outside?: boolean }): void;
  prompt(listener?: (n: { isNotDisplayed?: () => boolean; isSkippedMoment?: () => boolean }) => void): void;
} } }
interface AppleAuth { auth: {

  signIn(): Promise<{ authorization?: { id_token?: string } }>;

/** Google Identity Services → an ID token whose `nonce` claim equals ours
 * (Google embeds the nonce verbatim; the server checks it). */
async function googleIdToken(nonce: string): Promise<string | null> {
  await loadScript("https://accounts.google.com/gsi/client");
  const g = (window as unknown as { google?: GoogleId }).google;
  if (!g) return null;
  return new Promise<string | null>((resolve) => {
    let settled = false;
    const done = (v: string | null) => { if (!settled) { settled = true; resolve(v); } };
    try {
      g.accounts.id.initialize({ client_id: GOOGLE_CLIENT_ID, nonce, ux_mode: "popup", cancel_on_tap_outside: false, callback: (r) => done(r.credential ?? null) });
      g.accounts.id.prompt((n) => { if (n.isNotDisplayed?.() || n.isSkippedMoment?.()) done(null); });
    } catch { done(null); }
    setTimeout(() => done(null), 90_000);
  });
}

/** Sign in with Apple JS (web popup) → the returned id_token embeds our nonce
 * verbatim for the JS SDK (unlike native iOS, which expects a hashed nonce). */
async function appleIdToken(nonce: string): Promise<string | null> {
  await loadScript("https://appleid.cdn-apple.com/appleauth/static/jsapi/appleid/1/en_US/appleid.auth.js");
  const a = (window as unknown as { AppleID?: AppleAuth }).AppleID;
  if (!a) return null;
  a.auth.init({ clientId: APPLE_CLIENT_ID, scope: "email", redirectURI: APPLE_REDIRECT_URI, nonce, usePopup: true });
  try {
    const res = await a.auth.signIn();
    return res.authorization?.id_token ?? null;
  } catch { return null; }
}

/**
 * Obtain a provider id_token for the given nonce via the provider SDK.
 *
 * I3-0194 — `configured` separates the two ways there is no token. A build with
 * NO client id for this provider is a developer build, and only it offers the
 * manual token paste. A CONFIGURED provider whose SDK was blocked (ad-blocker,
 * corporate network) or dismissed is a real invitee, who has no way to obtain a
 * raw ID token — they get plain-language guidance instead, never the paste box.
 */
async function obtainIdToken(provider: "google" | "apple", nonce: string): Promise<{ token: string | null; configured: boolean }> {
  const configured = provider === "google" ? !!GOOGLE_CLIENT_ID : !!(APPLE_CLIENT_ID && APPLE_REDIRECT_URI);
  if (!configured) return { token: null, configured };
  try {
    return { token: provider === "google" ? await googleIdToken(nonce) : await appleIdToken(nonce), configured };
  } catch { return { token: null, configured }; }
}

/** Exchange key + a verified provider identity for a session. The key IS one
 * factor; the provider id_token (with single-use nonce) is the second. */
async function doLogin(key: string, provider: "google" | "apple", id_token: string, nonce: string): Promise<{ ok: boolean; status: number; data: LoginResp | null }> {
  try {
    const r = await fetch(`${API}/agency/login`, { method: "POST", credentials: "include", headers: { "content-type": "application/json", accept: "application/json" }, body: JSON.stringify({ key, provider, id_token, nonce }) });
    let data: LoginResp | null = null;
    try { data = (await r.json()) as LoginResp; } catch { /* no body */ }
    return { ok: r.ok, status: r.status, data };
  } catch { return { ok: false, status: 0, data: null }; }
}

/**
 * I3-1778 (D-1080(5)): the login nonce is ISSUED by the server (`POST /agency/login/nonce`), fresh per attempt, and
 * the login door spends only a nonce it issued for this door — the console no longer makes one up (I3-1777's
 * predictable-fallback concern now sits with the server's CSPRNG). `nonce` is null when none was issued, with the
 * HTTP status (0 for a transport failure) and the server's own sentence when it sent one (a rate limit).
 */
async function issuedNonce(): Promise<{ nonce: string | null; status: number; message: string | null }> {
  try {
    const r = await fetch(`${API}/agency/login/nonce`, { method: "POST", credentials: "include", headers: { "content-type": "application/json", accept: "application/json" }, body: "{}" });
    let data: { nonce?: unknown; message?: unknown } | null = null;
    try { data = (await r.json()) as { nonce?: unknown; message?: unknown }; } catch { /* no body */ }
    // An empty string is returned as it came and refused by the caller's `!issued.nonce`, the one emptiness check.
    if (r.ok && typeof data?.nonce === "string") return { nonce: data.nonce, status: r.status, message: null };
    const said = typeof data?.message === "string" && data.message.length > 0 && data.message.length <= 300 ? data.message : null;
    return { nonce: null, status: r.status, message: said };
  } catch { return { nonce: null, status: 0, message: null }; }
}

/**
 * UI-AUDIT F-G3 — A MALFORMED LIST USED TO TAKE THE WHOLE CONSOLE WITH IT.
 *
 * Every render here does `d.data.creators.map(...)` / `c.accounts.map(...)` /
 * `c.releases.map(...)` / `d.data.members.map(...)` straight off an `as`-cast
 * response. The casts are types, not checks: a 200 whose body is a partial
 * object, or whose list field arrived as `null`, `{}` or a string — a partial

 * `.map is not a function` INSIDE the render. These renders are the whole app
 * (`root.innerHTML = chrome(...)` is the last statement), so the throw escapes
 * before anything is assigned and the agency is left on whatever was on screen,
 * or on nothing, with no error and no way forward but a reload.
 *
 * `list()` is the same rule `apps/web/src/views.ts` states as the fail-closed

 * one. Here the absence is rendered as the empty state, which is honest enough
 * for a roster (the sections say when a scope is the reason), and — critically —
 * the app keeps standing.
 */
function list<T>(x: unknown): T[] {
  return Array.isArray(x) ? (x as T[]) : [];
}

export const ACTION_UNREACHED = "We couldn't reach the server, so we can't tell whether that went through — it may or may not have happened. Reload this page and check before trying again.";
export const SIGNIN_UNREACHED = "Sign-in didn't finish: we couldn't reach the server, so this device isn't signed in. Check your connection and try again.";

/** A server refusal, in words when the body carried any (bounded — a proxy's
 * error page must not become the message), else just the status. `read` marks a
 * load (a GET changes nothing), whose status-0 line keeps `did`. */
function failText(status: number, data: unknown, did: string, read = false): string {
  const b = data as { error?: unknown; detail?: unknown } | null;
  const said = typeof b?.error === "string" && b.error.length > 0 && b.error.length <= 200 ? b.error : "";
  const detail = typeof b?.detail === "string" && b.detail.length > 0 && b.detail.length <= 300 ? ` — ${b.detail}` : "";
  // A 2xx reaches here only when `apiPost` could not read its body: `did` assumes
  // a refusal, which that does not establish, so it gets its own sentence.
  if (status >= 200 && status < 300) return `The server answered (${status}) but its reply couldn't be read, so we can't tell whether that went through — reload and check before trying again.`;
  // QUEUE #940/6 (I34-W2-CLIENTS; DRAFT wording): status 0 is apiGet/apiPost's thrown fetch — no answer, so no refusal either.
  if (status === 0) return read ? `${did}: we couldn't reach the server. Check your connection and reload.` : ACTION_UNREACHED;
  return said ? `${did} (${status}): ${said}${detail}` : `${did} (${status}).`;
}

/** I3-1598 -- header-level feedback (Extend) is an inline status, like every
 * other message in this app, never a blocking native alert. */
function chromeNote(text: string): void {
  const el = root.querySelector<HTMLElement>("#chromemsg");
  if (el) el.textContent = text;
}
function wireChrome(): void {
  root.querySelector<HTMLButtonElement>('button[data-act="extend"]')?.addEventListener("click", async () => {
    // AG-05: which session asked. A refresh that completes after the person
    // signed out, or after somebody else signed in on this device, must not put
    // its token anywhere.
    const gen = sessionGen;
    const r = await apiPost<{ session: string }>("/agency/refresh", {});
    if (gen !== sessionGen || signedOut) return; // an ended session's answer, arriving late
    if (r.ok && r.data) {
      // AU-02: do not announce a stored extension that was not stored.
      const saved = setToken(r.data.session);
      chromeNote(saved ? "Session extended." : `Session extended for this tab. ${TAB_ONLY_NOTE}`);
    } else if (r.status !== 401) chromeNote(failText(r.status, r.data, "Couldn't extend the session"));
  });
  /**
   * I3-0411 — TWO BUTTONS, NOT ONE TOOLTIP, DECIDING WHICH SIGN-OUT HAPPENS.
   *
   * UI-AUDIT F-G2 found `POST /agency/logout` — the agency-WIDE kill switch,
   * which bumps the epoch and ends every member's session — fired from a
   * SINGLE "Sign out" button for every seat, with only a tooltip telling a
   * `manager`, `campaign_operator`, `analytics_only` or `performer` seat (none
   * of whom hold the `security` cap the route requires) that their press would
   * be refused. The first fix made the refusal honest; this one removes the
   * dead call for the four seats that can never make it: "Sign out on this
   * device" (`data-act="signout"`) is local-only — it clears this device's
   * token and never reaches the server — and is the only button a non-security
   * seat sees. "Sign out everywhere" (`data-act="signout-all"`), rendered only
   * for a security-capable seat, is the one that calls `/agency/logout`; the
   * fan app draws the same line at `apps/web/src/main.ts` (`signout` /
   * `signout-all`).
   *
   * The local sign-out still happens on every seat, because discarding the
   * token always works and is what the server itself recommends.
   */
  root.querySelector<HTMLButtonElement>('button[data-act="signout"]')?.addEventListener("click", () => {
    // Local-only: no server call, so there is nothing here for the server to
    // refuse and nothing for AG-05's stale-response race to race against —
    // but signedOut/sessionGen are still bumped first, exactly as below, so a
    // reply already in flight for this device is still recognisably stale.
    signedOut = true;
    sessionGen++;
    const held = getToken();
    clearToken();
    expireCookie(held); // I25-W1-COOKIE: this browser's cookie expires too
    renderLogin("Signed out on this device. Other sessions for this agency are unaffected.");
  });
  /**
   * UI-AUDIT F-G2 (the agency-wide half) — "SIGNED OUT EVERYWHERE", SAID TO A
   * SEAT THE SERVER HAD JUST REFUSED.
   *
   * This button is rendered only for a security-capable seat (chrome(),
   * above), so the 403 branch below is the seat's OWN grant being revoked or
   * downgraded mid-session, not the routine refusal every other seat used to
   * hit silently — the response is still read, never discarded, either way.
   *
   * It is the same defect `apps/mobile-check/test/source-mask.ts` was extracted
   * over: a failed revoke rendered as a completed one. It matters most in the
   * one situation the button exists for — a laptop lost, a seat compromised —
   * where the person walks away believing the other sessions are dead. They
   * are not, and nothing told them.
   */
  root.querySelector<HTMLButtonElement>('button[data-act="signout-all"]')?.addEventListener("click", async () => {
    // AG-05 — END THE SESSION *BEFORE* THE REQUEST, NOT AFTER IT.
    //
    // `sessionGen` and `signedOut` were the console's answer to a LATE 401, and
    // explicit sign-out set neither. So every successful response still in
    // flight belonged, as far as this client was concerned, to a live session:
    // a `/agency/refresh` started a moment earlier stored its replacement token
    // AFTER the token had been cleared — putting a valid credential back on a
    // device the person had just signed out of. Pending dashboard, team and
    // domain reads likewise passed their `sessionEnded()` guard and painted
    // private rosters over the login screen.
    //
    // Bumping the generation and arming the flag here makes every one of those
    // completions recognisably stale — successes included, which is the half
    // that was missing — and it happens before the network call, because the
    // local sign-out does not depend on the server's answer.
    signedOut = true;
    sessionGen++;
    // I25-W1-COOKIE: the handle is captured BEFORE it is cleared and the logout is sent under it. At i25-base the
    // token was cleared first and `/agency/logout` went out with an EMPTY bearer, so it could only 401.
    const held = getToken();
    clearToken();
    const r = await apiPost<{ ok?: boolean }>("/agency/logout", {}, held);
    clearToken();
    expireCookie(held);
    if (r.ok) { renderLogin("Signed out everywhere. Every session for this agency has ended."); return; }
    if (r.status >= 200 && r.status < 300) {
      renderLogin("Signed out on this device. The server answered but its reply couldn't be read, so we can't confirm the agency's other sessions ended — sign in again and retry if any of them need killing.");
      return;
    }
    if (r.status === 403) {
      renderLogin("Signed out on this device. Ending EVERY session for this agency needs an admin or security seat — yours can't, so any other sessions are still live. Ask an admin if one needs killing.");
      return;
    }
    // A transport failure or any other refusal: the token is gone here, and we
    // do not know what happened at the other end. Say exactly that.
    renderLogin("Signed out on this device — but we couldn't reach the server to end the agency's other sessions. Sign in again and retry if any of them need killing.");
  });
}

/**
 * I3-1460 — every back link is a real link: an `href` makes it focusable and
 * gives it a destination (the app root, which boots to this dashboard), and a
 * plain click stays in the SPA. It runs over whatever is on screen, so the
 * roster's link (rendered by views.ts) is covered by the same rule.
 */
function wireBack(): void {
  root.querySelectorAll<HTMLAnchorElement>('a[data-act="back"]').forEach((a) => {
    if (!a.getAttribute("href")) a.setAttribute("href", "./");
    a.onclick = (e) => {
      if (e.button !== 0 || e.metaKey || e.ctrlKey || e.shiftKey || e.altKey) return; // a new tab/window is the browser's
      e.preventDefault();
      void renderDashboard();
    };
  });
}

}

 * `claim_state` CHECK in migrations/0001_ledger.sql). A value outside it is
 * shown escaped in a neutral chip rather than guessed at.
 */
function claimChip(state: string): string {
  if (state === "verified") return `<span class="ds-chip chip ok" title="This creator's identity is verified">✓ Verified</span>`;
  if (state === "claimed") return `<span class="ds-chip chip warn" title="Claimed by the creator, but not yet verified">Claimed — not yet verified</span>`;

  return `<span class="ds-chip chip">${esc(state)}</span>`;
}

  };

      // still live. Report the refusal beside it, not over it.
      if (active) { const m = root.querySelector<HTMLElement>("#dmsg"); if (m) m.textContent = failText(r.status, r.data, "Already in progress — the record above is still valid"); return; }

    }
    showChallenge(r.data);

// --- Seat keys (QUEUE #777, D-0860(3); members cap = admin) ---
/**
 * List, mint and revoke per-seat agency keys. A minted key reaches this code
 * once, in the POST answer, and is painted from that local variable into
 * `#seatkey` and nowhere else: no storage, no module state, so hiding it,
 * leaving the screen or reloading loses it for good.
 */
/** A freshly minted seat key: held only by the mint handler and the renders it
 * triggers; `shown` goes false when the operator hides it, so a later re-read
 * can't bring it back. */
interface MintedSeat { seat_id: string; label: string; key: string; shown: boolean }

/** Paint the one-time key into `host` and wire its copy and hide controls. */
function showSeatKey(host: HTMLElement, minted: MintedSeat): void {
  host.innerHTML = renderSeatKeyOnce(esc, minted);
  const kmsg = host.querySelector<HTMLElement>("#seatkeymsg")!;
  host.querySelector<HTMLButtonElement>('button[data-act="seatcopy"]')!.onclick = async () => {
    kmsg.textContent = (await copyText(minted.key, "seatkeyval")) ? "Copied." : "Couldn't copy — select the key and copy it by hand.";
  };
  host.querySelector<HTMLButtonElement>('button[data-act="seatkeydone"]')!.onclick = () => {
    minted.shown = false;
    host.innerHTML = "";
    const m = root.querySelector<HTMLElement>("#smsg");
    if (m) m.textContent = "Key hidden. It can't be shown again.";
  };
}

/** The seat list failed to load. A key minted a moment ago is live and can't
 * be shown again: keep it on screen beside the plain error instead of losing
 * it with the list. */
function paintSeatsLoadError(status: number, data: unknown, minted?: MintedSeat): void {
  const why = status === 403 ? `<p class="err">Seat keys need an admin seat — your role can't manage them.</p>` : `<p class="err">${esc(failText(status, data, "Couldn't load the seat keys", true))}</p>`;
  paintLoadError(minted?.shown ? `<div id="seatkey"></div>${why}<p class="muted">The seat was created; only the list failed to reload.</p>` : why);
  const kh = minted?.shown ? root.querySelector<HTMLElement>("#seatkey") : null;
  if (minted && kh) showSeatKey(kh, minted);
}

  root.querySelector<HTMLButtonElement>('button[data-act="seatmint"]')!.onclick = async () => {

    // the re-read below answers 401 and onUnauthorized() shows sign-in.
    await renderSeats(`Seat revoked: ${label}.`);
  }; });
}

/**
 * Gate markup for the S86–S88 dashboards. Three independent conditions, each
 * named honestly instead of a dead button: the CREATOR must be verified, the
 * GRANT must carry a sales scope (reporting follows the sales scopes — the
 * server's scopeAllowsReports), and the SEAT must hold the analytics cap. The
 * server re-checks all three (plus suspension) on every read.
 */
/**
 * UI-AUDIT F-G4 — A BUTTON THAT COULD ONLY EVER 403, UNDER COPY THAT NAMED THE
 * WRONG REASON, THREE LINES ABOVE A BLOCK THAT NAMED THE RIGHT ONE.
 *
 * `GET /agency/intelligence/:id` goes through `resolveAgencyGrantedRead(c,
 * "analytics")`, which asks two questions: does the SEAT hold the `analytics`
 * cap, and does the GRANT allow reports (`scopeAllowsReports` — sales scopes
 * only). The button was gated on neither. It was gated on `verified`, which is
 * a fact about the CREATOR.
 *
 * So a `performer`, `security` or `campaign_operator` seat, or any seat on a
 * `manage_accounts` grant, saw a live "Load intelligence" button that could
 * only fail — and when it failed the panel said *"Intelligence unavailable
 * (403). It unlocks at verified status and needs an active grant."*, both of
 * which were already true. Meanwhile `bizGate`, rendered directly beneath, was
 * explaining the scope rule correctly for the sibling dashboards. The page
 * contradicted itself and the operator was left believing the feature was
 * broken rather than not theirs.
 *
 * The gate is the server's own two questions, in the server's own order, and
 * every refusal states the reason that actually applies.
 */
function intelBlock(scope: string, verified: boolean): string {
  if (!verified) return `<p class="muted">Aggregates unlock once this creator reaches verified status.</p>`;
  if (!hasCap(AGENCY.role, "analytics")) {
    return `<p class="muted">Your seat's role (${esc(AGENCY.role ?? "unknown")}) can't view audience intelligence — an admin, manager or analytics_only seat can. This is your seat, not this creator.</p>`;
  }
  if (!scopeAllowsReports(scope)) {
    return `<p class="muted">This creator's grant to you (${esc(scope)}) doesn't cover reporting. Audience intelligence follows the sales scopes (manage_sales / manage_all) — a roster-management grant deliberately doesn't include performance data.</p>`;
  }
  return `<div class="row"><button class="ds-btn" data-act="intel">Load intelligence</button></div><div id="intelbox"></div>`;
}

  return `${head}<p class="muted" style="font-size:12px">${esc(name)} gave you this grant for one thing: to enter them into Creator War, the card game where cards of them are sold, or to take them out of it. Each choice is filed on their ledger as an agency-signed action, so fans can tell it wasn't personally ${esc(name)}. If ${esc(name)} took themselves out, only they can opt back in.</p>
    <div class="row"><button class="ds-btn" data-war="in">Enter ${esc(name)} into Creator War</button><button class="ds-btn" data-war="out">Take ${esc(name)} out of Creator War</button></div>
    <p id="wmsg" class="muted" aria-live="polite"></p>`;

function bizGate(scope: string, verified: boolean): string {
  if (!verified) return `<p class="muted">Results, attribution and back-catalog unlock once this creator reaches verified status.</p>`;
  if (!scopeAllowsReports(scope)) return `<p class="muted">The grant scope (${esc(scope)}) doesn't authorize reports — ask the creator for a sales scope (manage_sales / manage_all).</p>`;
  if (!hasCap(AGENCY.role, "analytics")) return `<p class="muted">Your seat's role (${esc(AGENCY.role ?? "unknown")}) can't view analytics — an admin, manager or analytics_only seat can.</p>`;
  return `<div class="row"><button class="ds-btn" data-act="roi">Results</button><button class="ds-btn" data-act="attribution">Attribution</button><button class="ds-btn" data-act="backcat">Back catalog</button></div><div id="bizbox"></div>`;
}

/** I3-1509 (D-0920 option A, copy only): the frozen agency reads — intelligence, Results (roi) and Attribution — are
 * served from the daily sweep, which builds the day before the one it runs on (analytics-build.ts buildSweep
 * `shiftDay(…, -1)`), so the view says which day it is, instead of reading as live. Back catalog is a live read
 * (agency-analytics-routes.ts backCatalog) and carries no such note. */
// D-0931(9) (RG-CLIENT:7, I16-FXE): roi.ts readFrozenBundle starts its lookback at today's key, so the served day is
// yesterday or earlier — never promise the day before yesterday.
const AS_OF_NOTE = "As of yesterday or earlier (UTC): the newest day that has been built. This view is built once a day and does not update live.";
function asOfNote(): string {
  return `<p class="muted as-of" style="font-size:12px;margin:6px 0">${esc(AS_OF_NOTE)}</p>`;
}

async function loadDashboard(passportId: string, kind: "roi" | "attribution" | "back-catalog"): Promise<void> {
  const box = document.getElementById("bizbox");
  if (!box) return;
  box.innerHTML = `<p class="muted">Loading…</p>`;
  const d = await api<unknown>(`/agency/${kind}/${encodeURIComponent(passportId)}`);
  if (!d.ok || !d.data) { box.innerHTML = `<p class="err">${esc(failText(d.status, d.data, "Dashboard unavailable", true))}</p>`; return; }
  // I3-0754 — the casts are types, not checks: a 200 missing a section threw
  // inside the renderer and left this panel on "Loading…" for good.
  try {
    if (kind === "roi") box.innerHTML = asOfNote() + renderRoiView(esc, d.data as RoiView);
    else if (kind === "attribution") box.innerHTML = asOfNote() + renderAttributionView(esc, d.data as AttributionView);
    else box.innerHTML = renderBackCatalogView(esc, d.data as BackCatalogView);
  } catch {
    box.innerHTML = `<p class="err">This dashboard's data arrived incomplete, so it can't be shown. Try again in a moment — nothing on your side changed.</p>`;
  }
}

/** I3-1136 / #799b: does `url` point AT this account? The SAME two steps the
 * ledger door runs (services/trust-core/src/events.ts appendEventsOnce step 1):
 * the canonical subject, then @dhwaja/schema `urlNamesAccount` (the platform's
 * own host, the id at its profile position). ADVISORY: an account the
 * canonicalizer cannot map is left to the server's own refusal. */
function linkNamesAccount(platform: string, idKind: string, id: string, url: string): boolean {
  let subj: { platform: string; id: string };
  try { subj = subjectForAccount(platform, idKind, id); } catch { return true; }
  return urlNamesAccount(subj.platform, subj.id, url);
}

/**
 * Why issuing failed, in the agency's own terms. 403 IS THREE DIFFERENT ANSWERS
 * on this route and a bare "Failed (403)." collapses them into one dead end:
 *
 *   - `verified creators only` — the CREATOR is not currently verified. This one
 *     is not about the agency at all, and saying "Failed" invites the seat to
 *     go hunting for a problem with their own grant or their own key. An agency
 *     acts on behalf of a creator and cannot hold more authority than the
 *     creator it represents, so when that badge lapses this ability lapses with
 *     it — and comes back when the creator's does.
 *   - `no sales grant for this release` — the GRANT is the problem, and the seat
 *     can act on that (ask the creator to broaden it).
 *   - `role_forbidden` — the SEAT is the problem, and an admin can act on that.
 *
 * A 403 whose body did not survive (a proxy's error page) claims none of the
 * three. Fail closed: say what is known, which is the status.
 */
function issueFailure(status: number, error: unknown, bodyArrived: boolean): string {
  const said = typeof error === "string" && error.length > 0 && error.length <= 200 ? error : "";
  // AG-06 — "NO CODES WERE ISSUED" IS A CLAIM ABOUT THE SERVER, AND THIS CLIENT
  // CANNOT ALWAYS MAKE IT.
  //
  // `apiPost` returns status 0 on a transport rejection, and can return an
  // HTTP-SUCCESS response whose body did not parse. In both cases the mint may
  // already have run: `credentials.ts` atomically consumes limited edition
  // supply and mints one-time secrets before the response is written. Telling
  // the person nothing happened invites a retry that burns the supply a second
  // time, for codes nobody will ever see. The Summary is the authority on what
  // was actually issued, so send them there instead of guessing.
  if (status === 0 || (status >= 200 && status < 300 && !bodyArrived)) {
    return "We don't know whether codes were issued — the answer never arrived. Codes may already have been minted against this edition's supply, and if they were, they are gone: they are shown once and we keep no copy. Check the Summary below for the issued count BEFORE issuing again.";
  }
  if (status === 403 && said === "verified creators only") {
    return "No codes were issued. This creator isn't currently verified, and codes can only be issued for a verified creator — including by an agency acting for them. Your grant is unaffected; issuing works again when their verification does.";
  }
  if (!said) return `No codes were issued (${status}).`;
  return `No codes were issued (${status}): ${said}`;
}

// Best-effort clipboard with a synchronous selection fallback for browsers/
// contexts where navigator.clipboard is unavailable. Never throws.
async function copyText(text: string, preId = "codespre"): Promise<boolean> {
  try {
    if (navigator.clipboard?.writeText) { await navigator.clipboard.writeText(text); return true; }
  } catch { /* fall through to selection */ }
  try {
    const pre = document.getElementById(preId);
    if (pre) {
      const sel = window.getSelection();
      const range = document.createRange();
      range.selectNodeContents(pre);
      sel?.removeAllRanges();
      sel?.addRange(range);
      const ok = document.execCommand("copy");
      sel?.removeAllRanges();
      return ok;
    }
  } catch { /* ignore */ }
  return false;
}

/** Returns whether the download was actually handed to the browser. */
function downloadText(filename: string, text: string): boolean {
  try {
    const blob = new Blob([text], { type: "text/plain" });
    const url = URL.createObjectURL(blob);
    const a = document.createElement("a");
    a.href = url;
    a.download = filename;
    document.body.appendChild(a);
    a.click();
    a.remove();
    setTimeout(() => URL.revokeObjectURL(url), 0);
    return true;
  } catch { return false; /* download is best-effort; Copy remains */ }
}

/**
 * THE LEGAL FOOTER — Terms and Privacy on the FAN app, opened in a new tab.
 *
 * The pages live on the public origin, which this app otherwise never speaks
 * to, so it arrives as the build-time `__PUBLIC_BASE__` (validated in build.mjs
 * as a bare https origin) and is re-checked here to the same shape: anything
 * else — absent, empty, http, a path, a fragment — renders NO anchor and a
 * one-line note. A missing origin fails closed to the note, never to a
 * placeholder host.
 */
/**
 * I3-2204 a (I23-W1-AUTH H6; the button, I23-L05): what the agency revoke door's answer means, in one line.
 * `revoked: false` is the server's no-oracle answer (credentials.ts revokeCredential): the code is not one THIS agency
 * issued on this release, or it was already redeemed or revoked — the line does not say which, because the server
 * does not. Refusals carry the server's own sentence through `fail`.
 */
export function revokeOutcome(r: { ok: boolean; status: number; data: { revoked?: unknown } | null }, fail: (did: string) => string): string {
  if (r.ok && r.data?.revoked === true) return "Revoked — that code can no longer be redeemed.";
  if (r.ok && r.data?.revoked === false) return "Nothing was revoked: that code is not an unredeemed code this agency issued for this release.";
  return fail("Couldn't revoke the code — nothing changed");
}

export function renderLegalFooter(esc: Esc, publicBase: unknown, apiBase?: unknown): string {
  const base = typeof publicBase === "string" && /^https:\/\/[a-z0-9.-]+(:\d+)?$/i.test(publicBase) ? publicBase : null;
  // CONSOLES H3b (I3-1646 b, I23-L05): with a bare-origin API_BASE, Terms/Privacy open trust-core's server-rendered
  // /terms and /privacy (path form); otherwise the fan app's #/terms, #/privacy. Both render the one @dhwaja/legal text.
  const api = typeof apiBase === "string" && /^https:\/\/[a-z0-9.-]+(:\d+)?$/i.test(apiBase) ? apiBase : null;
  // I21-Q146 (D-1073(1)): About is trust-core's server-rendered GET /about (about-page.ts) on the API origin, as the
  // creator console's footer links it; no usable API origin renders no About link, never a dead one.
  const about = api ? `<a href="${esc(`${api}/about`)}" target="_blank" rel="noopener noreferrer">${esc("About")}</a>` : "";
  if (!base) return `<footer class="foot"><nav><span class="muted">Legal pages not configured.</span>${about ? ` · ${about}` : ""}</nav></footer>`;
  const link = (path: "/terms" | "/privacy", label: string): string => `<a href="${esc(api ? `${api}${path}` : `${base}/#${path}`)}" target="_blank" rel="noopener noreferrer">${esc(label)}</a>`;
  return `<footer class="foot"><nav>${link("/terms", "Terms")} · ${link("/privacy", "Privacy")}${about ? ` · ${about}` : ""}</nav></footer>`;
}

/** A count that may be k-floored to null server-side: null → an honest em-dash
 * (the suppression note explains it), anything else through esc. */
function num(esc: Esc, v: unknown): string {
  return v == null ? "—" : esc(v);
}

/** A 0..1 ratio → "NN.N%", or "—" when null/absent. A non-number (a hostile or
 * malformed body) is escaped verbatim rather than arithmetic'd into NaN. */
function pct(esc: Esc, v: unknown): string {
  if (v == null) return "—";
  if (typeof v !== "number" || !Number.isFinite(v)) return esc(v);
  return esc((v * 100).toFixed(1)) + "%";
}

function kpi(esc: Esc, value: string, label: string): string {
  return `<div class="card"><div class="kpi">${value}<div class="lab">${esc(label)}</div></div></div>`;
}

/**
 * WHAT A WITHHELD ARCHIVE-DEMAND COUNT MAY SAY ON THIS SCREEN.
 *
 * The same contract the creator console holds (`apps/creator-console/src/views.ts`
 * `demandWithheldLine`), and it is the same sentence deliberately: an agency
 * member reading for a creator sees exactly what the creator sees, so the two
 * screens cannot be differenced against each other.
 *
 * `request_count` is null for every population below the server's floor, and
 * that INCLUDES zero. If "nobody asked" and "four fans asked" rendered
 * differently here, the difference between two screenshots would be the per-fan
 * inference the floor exists to remove, and a server-side control would have
 * been undone by client copywriting. So:
 *
 *   - no number, and `demand.floor` is NEVER rendered — the server sends it for
 *     inspection, not for copy. "Fewer than N" is still a statement about how
 *     many fans asked;
 *   - no claim that anybody asked and none that nobody did. A bare em-dash was
 *     the old render and it is not enough on its own: an unexplained dash beside
 *     a "Demand" heading reads as zero;
 *   - identical bytes for every population from nobody up to floor-1.
 *
 * What is left to say truthfully is the CONDITION and the CLOCK, neither of
 * which carries fan data. `through` is derived from the clock alone, so it is
 * the same string for every population and may be shown.
 */
function demandThrough(esc: Esc, through: unknown): string {
  return typeof through === "string" && /^\d{4}-\d{2}-\d{2}$/.test(through) ? ` Counted through ${esc(through)}; today's requests aren't in yet.` : "";
}

}

import { constants as zlibConstants, inflateSync } from "node:zlib";

const { Z_SYNC_FLUSH } = zlibConstants;
import { PNG } from "pngjs";
import jpeg from "jpeg-js";

/**
 * Real image perceptual fingerprinting for the Archive Agent — pure JS, no
 * native decoders, headless-testable. Pipeline: decode (PNG/JPEG) → grayscale
 * luminance → box-resample to 9×8 → dHash (64-bit, 16 hex chars).
 *
 * The dHash bit/format definition is PINNED to the platform's canonical
 * implementation in services/image/src/phash.ts (the cascade's near-duplicate
 * stage) — a parity test asserts both implementations agree bit-for-bit, so an
 * agent-produced fingerprint is directly comparable to server-side hashes.
 *
 * Bounds: decode work is capped (input bytes + pixel count) so a malicious
 * archive file can't balloon memory — a creator's own machine, but agents get
 * pointed at messy folders with unknown provenance.
 */

export const DHASH_COLS = 9;
export const DHASH_ROWS = 8;
const MAX_INPUT_BYTES = 128 * 1024 * 1024; // 128 MB encoded
export const MAX_PIXELS = 64 * 1024 * 1024; // 64 MP decoded

/** dHash over a 9×8 row-major grayscale matrix → 16 hex chars. MUST match
 * services/image/src/phash.ts exactly (parity-tested). */
export function dHash(gray: Uint8Array | number[]): string {
  if (gray.length !== DHASH_COLS * DHASH_ROWS) {
    throw new Error(`dHash expects ${DHASH_COLS * DHASH_ROWS} grayscale values (9x8), got ${gray.length}`);
  }
  const bits: number[] = [];
  for (let row = 0; row < DHASH_ROWS; row++) {
    for (let col = 0; col < DHASH_COLS - 1; col++) {
      const a = gray[row * DHASH_COLS + col]!;
      const b = gray[row * DHASH_COLS + col + 1]!;
      bits.push(a < b ? 1 : 0);
    }
  }
  let hex = "";
  for (let i = 0; i < 64; i += 4) {
    const nibble = (bits[i]! << 3) | (bits[i + 1]! << 2) | (bits[i + 2]! << 1) | bits[i + 3]!;
    hex += nibble.toString(16);
  }
  return hex;
}

export interface DecodedImage {
  width: number;
  height: number;
  rgba: Uint8Array; // RGBA, row-major
}

/** Read a PNG's declared dimensions straight from the IHDR (bytes 16..24)
 * WITHOUT decoding — pngjs has no pre-decode size cap, so a few-hundred-byte
 * file can declare a 20000×20000 image and force a ~1.6 GB allocation (an
 * interlaced bomb is unbounded). We gate on the header before ever decoding. */
export function pngIhdrPixels(bytes: Uint8Array): number | null {
  // 8-byte signature, then a length(4)+"IHDR"(4) chunk header, then W(4) H(4).
  if (bytes.length < 24 || bytes[12] !== 0x49 || bytes[13] !== 0x48 || bytes[14] !== 0x44 || bytes[15] !== 0x52) return null;
  const w = (bytes[16]! << 24) | (bytes[17]! << 16) | (bytes[18]! << 8) | bytes[19]!;
  const h = (bytes[20]! << 24) | (bytes[21]! << 16) | (bytes[22]! << 8) | bytes[23]!;
  if (w <= 0 || h <= 0) return null;
  return w * h; // may overflow to a huge/negative-ish number → caught by the > MAX_PIXELS gate
}

/** Channels per pixel for each PNG colour type, or null for an undefined one. */
function pngChannels(colourType: number): number | null {
  return colourType === 0 ? 1 : colourType === 2 ? 3 : colourType === 3 ? 1 : colourType === 4 ? 2 : colourType === 6 ? 4 : null;
}

/** The bit depths each PNG colour type is allowed to carry. Rejecting an illegal
 * pair (colour 3 at depth 16, say) BEFORE the inflate below matters: pngjs bails
 * on those in about a millisecond, and without this check the gate would do
 * hundreds of milliseconds of decompression on the way to the same answer. */
const PNG_LEGAL_DEPTHS: Readonly<Record<number, readonly number[]>> = {
  0: [1, 2, 4, 8, 16], // greyscale
  2: [8, 16], // truecolour
  3: [1, 2, 4, 8], // palette
  4: [8, 16], // greyscale + alpha
  6: [8, 16], // truecolour + alpha
};

// Adam7 pass geometry: seven passes, each with its own origin and stride.
const ADAM7_X_START = [0, 4, 0, 2, 0, 1, 0];
const ADAM7_Y_START = [0, 0, 4, 0, 2, 0, 1];
const ADAM7_X_STEP = [8, 8, 4, 4, 2, 2, 1];
const ADAM7_Y_STEP = [8, 8, 8, 4, 4, 2, 2];

/**
 * How many raw (post-inflate) bytes a complete image of this shape must carry:
 * one filter byte per scanline plus the packed samples, summed over every Adam7
 * pass when the image is interlaced.
 *
 * INTERLACED USED TO BE "CANNOT TELL" HERE, and that hole was a live
 * decompression bomb. The gate returned early for interlaced images, so the
 * bounded inflate never ran, and `pngjs`'s interlaced branch uses **stock
 * `zlib.inflateSync` with no output limit** (`parser-sync.js`) — a 1x1
 * interlaced PNG (one pixel: every pixel cap waves it through) carrying a 1 GB
 * payload measured 3.1 s and +1046 MB of RSS, from a 1019 KB file. On the
 * container, which accepts 16 MB uploads, one request bought ~16 GB. Found by
 * the F20 regression review, in the F20 fix, one round after the same class of
 * bug was fixed on the non-interlaced path. See MANIFEST-F20.md §5.
 *
 * Adam7 sizing is exactly computable, so there is no need to punt: pass p covers
 * `ceil((w - xStart) / xStep)` columns and `ceil((h - yStart) / yStep)` rows,
 * and a pass with no columns or no rows contributes nothing at all (which is
 * what happens for images narrower or shorter than 8 px).
 */
function pngRawBytesNeeded(w: number, h: number, channels: number, depth: number, interlace: number): number {
  const rowBytes = (px: number): number => 1 + Math.ceil((px * channels * depth) / 8);
  if (interlace === 0) return h * rowBytes(w);
  let total = 0;
  for (let p = 0; p < 7; p++) {
    const passW = w > ADAM7_X_START[p]! ? Math.ceil((w - ADAM7_X_START[p]!) / ADAM7_X_STEP[p]!) : 0;
    const passH = h > ADAM7_Y_START[p]! ? Math.ceil((h - ADAM7_Y_START[p]!) / ADAM7_Y_STEP[p]!) : 0;
    if (passW === 0 || passH === 0) continue;
    total += passH * rowBytes(passW);
  }
  return total;
}

/**
 * IS THE IMAGE DATA ACTUALLY ALL THERE? — and the bug this exists for.
 *
 * `PNG.sync.read` on a PNG whose CRCs are valid but whose DEFLATE STREAM IS
 * SHORT reads past the end of the inflated data into an uninitialised buffer.
 * The fuzz pass measured it: the SAME bytes, in the same process, microseconds
 * apart, decoded successfully about 63% of the time and threw
 * `Unrecognised filter type - N` the rest — **with a different N almost every
 * time** (9, 8, 255, 222, 112, 60, 136, 58, 226, …). That varying N is the
 * uninitialised memory being read as a filter byte, and when it happens to land
 * on 0–4 the decode "succeeds" with pixel rows made of whatever was in that
 * memory.
 *
 * Two consequences, and the first is the one that matters most for what this
 * module IS. A perceptual fingerprint has to be REPRODUCIBLE: scanning the same
 * folder twice must give the same answer, or near-duplicate matching against the
 * platform is a coin flip. It was not. And second, a "successful" decode of such
 * a file computes a dHash over process memory — a 64-bit function of bytes this
 * program never meant to look at, on a creator's own machine, from a file whose
 * provenance they do not know.
 *
 * We cannot fix `pngjs`, so the boundary checks the precondition it lacks:
 * concatenate the IDAT payloads, inflate them, and require exactly the bytes a
 * complete image needs (`pngRawBytesNeeded` — Adam7 included). That is EXACTLY
 * the condition pngjs mishandles.
 *
 * Three things this function is careful about, each of which was got wrong once:
 *
 *   1. THE INFLATE IS BOUNDED, at `needed`. An unbounded one made this gate a
 *      worse bomb than the decoder it protects (see `pngRawBytesNeeded` and
 *      MANIFEST-F20.md §4-5). Hitting the cap answers the question by itself.
 *   2. MORE data than needed is REFUSED, not accepted. pngjs refuses those
 *      itself, so it costs nothing — and saying "complete" is what hands a file
 *      on to pngjs's *interlaced* inflate, which has no bound at all.
 *   3. `Z_SYNC_FLUSH`, because pngjs uses it. A stream that carries every pixel
 *      but never terminates tidily is not an incomplete image, and refusing
 *      those cost honest users their fingerprints for one review round.
 *
 * The IHDR is validated before any of that: an undefined colour type, bit depth,
 * colour/depth pair, compression method, filter method or interlace method is
 * refused without decompressing anything, because pngjs rejects those for free
 * and the inflate is the expensive part of this function.
 */
function pngDataIsComplete(bytes: Uint8Array): boolean {
  if (bytes.length < 33) return false;
  const depth = bytes[24]!;
  const colour = bytes[25]!;
  const compression = bytes[26]!;
  const filter = bytes[27]!;
  const interlace = bytes[28]!;
  // Reject what the spec does not define BEFORE doing any decompression: pngjs
  // refuses these in about a millisecond, and the inflate below is the expensive
  // part of this function.
  if (compression !== 0 || filter !== 0 || interlace > 1) return false;
  const channels = pngChannels(colour);
  if (channels === null || !PNG_LEGAL_DEPTHS[colour]!.includes(depth)) return false;
  const w = ((bytes[16]! << 24) | (bytes[17]! << 16) | (bytes[18]! << 8) | bytes[19]!) >>> 0;
  const h = ((bytes[20]! << 24) | (bytes[21]! << 16) | (bytes[22]! << 8) | bytes[23]!) >>> 0;
  if (w <= 0 || h <= 0) return false;
  if (w * h > MAX_PIXELS) return false; // keeps `needed` (and the inflate cap below) bounded standalone

  // Concatenate every IDAT payload, in order, bounded by the file itself.
  //
  // ME-02 (S-068). THE SAME WALK NOW COUNTS IHDR CHUNKS, because the bytes that
  // ADMIT this file must be the bytes that DECODE it.
  //
  // Everything above reads the header off a FIXED OFFSET — bytes 12..28, the
  // first chunk after the signature — and computes the pixel cap and `needed`
  // from it. `pngjs@7.0.0` does not agree that that header is authoritative:
  // `lib/parser.js:73-82` checks only that the FIRST chunk is IHDR, and
  // `_parseIHDR` (`:119-178`) is dispatched again for EVERY later IHDR chunk,
  // calling `this.metadata(...)` again each time. `lib/parser-sync.js:28-29`
  // keeps the LAST one, and `:82-88` derives the non-interlaced inflate's
  // `chunkSize`/`maxLength` from `metaData.width * bpp * depth` and
  // `metaData.height` — the REPLACEMENT header.
  //
  // Measured on this tree before this line existed: a 4x4 PNG carrying a second
  // IHDR declaring 8000x8000 passed the declared-pixel cap and this
  // completeness gate as a 4x4, and `PNG.sync.read` then returned an 8000x8000
  // image — 64 MP, ~256 MB of RGBA allocated — with the pixel cap firing only
  // AFTERWARDS, from the post-decode check. The cap that exists to run before
  // the allocation ran after it.
  //
  // A spec-valid PNG has exactly one IHDR and it is the first chunk, so
  // requiring that costs nothing and closes the disagreement at its root: there
  // is one header, the one this function and the caps above already read.
  let ihdrCount = 0;
  const parts: Uint8Array[] = [];
  let off = 8;
  let total = 0;
  while (off + 12 <= bytes.length) {
    const len = ((bytes[off]! << 24) | (bytes[off + 1]! << 16) | (bytes[off + 2]! << 8) | bytes[off + 3]!) >>> 0;
    const dataStart = off + 8;
    const dataEnd = dataStart + len;
    if (len > bytes.length || dataEnd + 4 > bytes.length) return false; // chunk table lies
    if (bytes[off + 4] === 0x49 && bytes[off + 5] === 0x48 && bytes[off + 6] === 0x44 && bytes[off + 7] === 0x52) {
      // ME-02: a second IHDR anywhere makes "the header" ambiguous, and the
      // decoder resolves that ambiguity differently from this gate. Refused —
      // and refused HERE, before `PNG.sync.read` is ever called.
      if (++ihdrCount > 1 || off !== 8) return false;
    }
    if (bytes[off + 4] === 0x49 && bytes[off + 5] === 0x44 && bytes[off + 6] === 0x41 && bytes[off + 7] === 0x54) {
      parts.push(bytes.subarray(dataStart, dataEnd));
      total += len;
      if (total > MAX_INPUT_BYTES) return false;
    }
    off = dataEnd + 4;
  }
  if (ihdrCount !== 1) return false; // exactly one, and it was the first chunk
  if (parts.length === 0) return false;

  const idat = new Uint8Array(total);
  let at = 0;
  for (const p of parts) {
    idat.set(p, at);
    at += p.length;
  }
  // BOUND THE INFLATE, or this gate is itself the bomb. `pngjs`'s NON-interlaced
  // inflate is bounded (it caps output at the image size), so an unbounded
  // `inflateSync` here would make a hardened path strictly more bombable than the
  // one it replaced: a 1x1 PNG with a 1029:1 IDAT measured 1 GB of RSS and 1.9 s,
  // against a 7 ms throw from pngjs alone.
  //
  // The only question this function asks is "at least `needed` bytes?", so the
  // inflate is capped AT `needed` and the cap being hit is itself the answer.
  // `needed` is bounded by the pixel cap re-checked above.
  //
  // `Z_SYNC_FLUSH` because a stream that carries every pixel but never cleanly
  // terminates — no adler32, or a flush an embedded encoder never finished — is
  // NOT an incomplete image, and pngjs's own inflate accepts it. Measured: three
  // such files decoded correctly and reproducibly before this gate existed and
  // were refused by the first version of it, which conflated "the zlib stream
  // did not end tidily" with "the pixels are missing". A genuinely short stream
  // still produces fewer than `needed` bytes and is still refused.
  const needed = pngRawBytesNeeded(w, h, channels, depth, interlace);
  if (needed <= 0) return false;
  try {
    return inflateSync(Buffer.from(idat), { maxOutputLength: needed, finishFlush: Z_SYNC_FLUSH }).length >= needed;
  } catch (e) {
    // MORE output than the image needs. `pngjs` refuses those itself — its
    // reader throws on leftover data as well as on a shortfall — so refusing
    // here costs nothing and buys the thing that actually matters: the inflate
    // never runs past `needed` bytes, on EITHER interlace setting. That is what
    // closes the interlaced bomb, because the gate saying "complete" is what
    // used to hand the file on to pngjs's own UNBOUNDED interlaced inflate.
    // (Both arms answer `false`; they are spelled separately because they are
    // different facts — "too much data" and "not decompressible at all" — and a
    // future change may want to tell them apart.)
    if ((e as NodeJS.ErrnoException).code === "ERR_BUFFER_TOO_LARGE") return false;
    return false; // a stream that will not inflate at all is one pngjs will read past
  }
}

/** Decode PNG or JPEG bytes (sniffed by magic, not extension). Returns null
 * for unsupported/corrupt input — the caller falls back to exact-hash-only. */
export function decodeImage(bytes: Uint8Array): DecodedImage | null {
  if (bytes.length === 0 || bytes.length > MAX_INPUT_BYTES) return null;
  try {
    // PNG magic: 89 50 4E 47
    if (bytes[0] === 0x89 && bytes[1] === 0x50 && bytes[2] === 0x4e && bytes[3] === 0x47) {
      // Reject oversized dimensions from the IHDR BEFORE pngjs allocates.
      const declared = pngIhdrPixels(bytes);
      if (declared === null || declared > MAX_PIXELS || declared < 0) return null;
      // THE PRECONDITION pngjs DOES NOT CHECK — see `pngDataIsComplete`. Without
      // it, a short deflate stream makes the non-interlaced decode read
      // uninitialised memory: non-deterministic between calls on identical
      // bytes, and when it "succeeds" the fingerprint is computed over process
      // memory. The gate answers for interlaced images too (Adam7 is exactly
      // sizeable), which is also what keeps THEIR inflate bounded.
      if (!pngDataIsComplete(bytes)) return null;
      const png = PNG.sync.read(Buffer.from(bytes));
      if (png.width * png.height > MAX_PIXELS) return null;
      return { width: png.width, height: png.height, rgba: new Uint8Array(png.data) };
    }
    // JPEG magic: FF D8
    if (bytes[0] === 0xff && bytes[1] === 0xd8) {
      const img = jpeg.decode(Buffer.from(bytes), { useTArray: true, maxMemoryUsageInMB: 512, maxResolutionInMP: 64 });
      if (img.width * img.height > MAX_PIXELS) return null;
      return { width: img.width, height: img.height, rgba: new Uint8Array(img.data) };
    }
  } catch {
    return null; // corrupt/unsupported → exact-hash-only
  }
  return null;
}

/** Rec.601 luminance of RGBA alpha-composited over opaque white (I3-0836). */
export function toGrayscale(img: DecodedImage): Uint8Array {
  const out = new Uint8Array(img.width * img.height);
  for (let i = 0; i < out.length; i++) {
    const o = i * 4;
    const y = 0.299 * img.rgba[o]! + 0.587 * img.rgba[o + 1]! + 0.114 * img.rgba[o + 2]!;
    const a = img.rgba[o + 3]!;
    // I3-0836: composite over opaque white BEFORE hashing — the matrix is of the
    // image a viewer sees, so alpha-only content is structure and RGB under
    // alpha 0 is not. Luminance is linear, so compositing it equals compositing
    // each channel. Opaque pixels take the untouched path.
    out[i] = a === 255 ? Math.round(y) : Math.round((y * a + 255 * (255 - a)) / 255);
  }
  return out;
}

/**
 * Area-average (box) resample of a grayscale image to targetW×targetH.
 * Averaging over source regions (not point sampling) is what makes the
 * downstream dHash robust to resize/recompress.
 */
export function boxResize(gray: Uint8Array, w: number, h: number, targetW: number, targetH: number): Uint8Array {
  if (w < 1 || h < 1 || gray.length !== w * h) throw new Error("boxResize: bad source dimensions");
  const out = new Uint8Array(targetW * targetH);
  for (let ty = 0; ty < targetH; ty++) {
    const y0 = Math.floor((ty * h) / targetH);
    const y1 = Math.max(y0 + 1, Math.floor(((ty + 1) * h) / targetH));
    for (let tx = 0; tx < targetW; tx++) {
      const x0 = Math.floor((tx * w) / targetW);
      const x1 = Math.max(x0 + 1, Math.floor(((tx + 1) * w) / targetW));
      let sum = 0;
      for (let y = y0; y < y1; y++) {
        for (let x = x0; x < x1; x++) sum += gray[y * w + x]!;
      }
      out[ty * targetW + tx] = Math.round(sum / ((y1 - y0) * (x1 - x0)));
    }
  }
  return out;
}

export interface ImageFingerprint {
  perceptual: string; // 16-hex dHash (platform-comparable)
  width: number;
  height: number;
}

/** Full pipeline: bytes → perceptual fingerprint + dimensions, or null when
 * the format isn't decodable (caller keeps the exact hash alone). */
export function imageFingerprint(bytes: Uint8Array): ImageFingerprint | null {
  const img = decodeImage(bytes);
  if (!img) return null;
  const gray = toGrayscale(img);
  const small = boxResize(gray, img.width, img.height, DHASH_COLS, DHASH_ROWS);
  return { perceptual: dHash(small), width: img.width, height: img.height };
}

const POPCOUNT = new Uint8Array(256);
for (let i = 0; i < 256; i++) POPCOUNT[i] = (i & 1) + POPCOUNT[i >> 1]!;

/** Hamming distance between two 16-hex dHashes (0..64) — for local grouping. */
export function hamming(a: string, b: string): number {
  if (a.length !== 16 || b.length !== 16 || !/^[0-9a-f]+$/.test(a) || !/^[0-9a-f]+$/.test(b)) {
    throw new Error("hamming expects two 16-hex-char dHashes");
  }
  let dist = 0;
  for (let i = 0; i < 16; i++) dist += POPCOUNT[parseInt(a[i]!, 16) ^ parseInt(b[i]!, 16)]!;
  return dist;
}

/**

 * walks local folders / NAS mounts, fingerprints media LOCALLY, groups exact

 * reviewable manifest the creator edits and hands to the Dhwaja app to register.

 *

 * never reaches for its own crypto (single-hasher invariant holds in apps/ too).
 *

 * image/video by extension. Perceptual image hashing and robust video/audio
 * fingerprinting require a media decoder and are left as the documented seam in
 * README.md (`fingerprintMedia`). A fuller agent is NOT a drop-in: the release-

export const MAX_MEDIA_BYTES = 2 * 1024 * 1024 * 1024; // 2 GB

const IMAGE_EXT = new Set([".jpg", ".jpeg", ".png", ".gif", ".webp", ".heic", ".tif", ".tiff", ".bmp"]);

export type FingerprintSkip =
  | "not_an_image" // magic bytes are neither PNG nor JPEG (or the file is empty)
  | "too_large" // more pixels than this agent will decode
  | "incomplete_pixel_data" // header promises more pixels than the file carries
  | "unreadable" // a decoder we do have refused it for some other reason
  | "video_unreadable" // ffmpeg is missing from PATH, or refused the file
  | "source_changed"; // AR-02: the file was replaced mid-scan; the two halves would describe different versions

/** A file or folder the scan never fingerprinted at all - as opposed to
 * `FingerprintSkip`, which is for media the scan DID read but could not
 * decode. Recorded so the draft's `scanned` count can be reconciled against
 * what is actually on disk (I3-0712): before this, an unreadable directory,
 * an oversized file, or a file that failed to read left no trace anywhere. */
export interface ScanIssue {
  rel: string;
  reason: "unreadable_directory" | "oversized_media" | "read_failed";
}

/**
 * Why `imageFingerprint` said no, in terms a creator can act on.
 *
 * Deliberately re-derived from the bytes rather than reported by the decoder:
 * `decodeImage`'s contract is null-or-image and nothing downstream should start
 * depending on WHICH null it was. This is presentation, and it is allowed to be
 * approximate — but the distinction it draws is the one that matters, between
 * "this is not a picture" and "this picture is missing part of itself".
 */
export function fingerprintSkipReason(bytes: Uint8Array): FingerprintSkip {
  const isPng = bytes.length > 8 && bytes[0] === 0x89 && bytes[1] === 0x50 && bytes[2] === 0x4e && bytes[3] === 0x47;
  const isJpeg = bytes.length > 3 && bytes[0] === 0xff && bytes[1] === 0xd8;
  if (!isPng && !isJpeg) return "not_an_image";
  if (isPng) {
    const declared = pngIhdrPixels(bytes);
    if (declared !== null && declared > MAX_PIXELS) return "too_large";
    if (declared !== null) return "incomplete_pixel_data";
  }
  return "unreadable";
}

function hex(b: Uint8Array): string {
  let s = "";
  for (const x of b) s += x.toString(16).padStart(2, "0");
  return s;
}

async function* walk(root: string, dir: string, issues?: ScanIssue[]): AsyncGenerator<string> {
  let entries: import("node:fs").Dirent[];
  try {
    entries = await readdir(dir, { withFileTypes: true });
  } catch {
    // I3-0712: this used to be silent - a folder the creator can't read
    // (permissions, a broken NAS mount, a folder that vanished mid-walk) just
    // disappeared from the count with nothing to show for it.
    issues?.push({ rel: relative(root, dir) || ".", reason: "unreadable_directory" });
    return;
  }
  for (const e of entries) {
    if (e.name.startsWith(".")) continue;
    const full = join(dir, e.name);
    if (e.isDirectory()) {
      yield* walk(root, full, issues);
      continue;
    }
    if (e.isFile()) {
      yield full;
      continue;
    }
    if (e.isSymbolicLink()) {
      // I3-2140: resolve one hop with stat() (follows the link) rather than
      // opening the target. stat() cannot block the way reading a FIFO or a
      // character device can; readFile() on one would hang the scan forever.
      const target = await statOrNull(full);
      if (target?.isFile()) yield full;
      continue;
    }
    // FIFOs, sockets, char/block devices: never opened. A named pipe or a
    // symlink to a special device with a media-looking name would otherwise
    // hang readFile() waiting for a writer that will never come (I3-2140).
  }
}

/** Invoked after each file is scanned (PS-086): lets a caller persist partial
 * progress so a crash mid-run does not lose every file scanned before it. */
export type ScanCheckpoint = (soFar: ScannedFile[]) => void | Promise<void>;

/** Scan a root folder; fingerprint each media file (exact hash) and suggest a
 * release grouping from its first-level subfolder.
 *
 * PS-086: `checkpoint` (optional) fires after each file so a caller can
 * persist progress incrementally — a multi-hour scan then loses only the
 * unflushed tail on a crash, not the whole run.
 *
 * I3-1706: `resumeFrom` (optional), keyed by `rel`, lets a caller skip
 * re-fingerprinting a file it already has a record for — an entry is reused
 * only when the file's (ino, mtime, size) on disk still match the recorded
 * ones, so a file replaced since the last checkpoint is re-scanned, never
 * silently assumed unchanged.
 *
 * I3-0712: `issues` (optional), when passed, is appended to with every
 * folder, oversized file, or read failure the scan could not account for as
 * an ordinary scanned-or-skipped-with-a-reason media file. */
async function statOrNull(path: string): Promise<Stats | null> {
  try {
    return await stat(path);
  } catch {
    return null; // vanished mid-scan: that is a change, and `sameSource` says so
  }
}

/**
 * Did these two stats observe the same file version? (AR-02)
 *
 * NOT A SECURITY CHECK, and the distinction matters. This catches the ordinary
 * cause — a cloud sync replacing a file, a download finishing, an export
 * overwriting its own output — where the writer is not trying to hide. It is
 * deliberately NOT size-only, because a re-encode or an edit can land on the
 * same byte count; identity here is (inode, size, mtime), and a change in any of
 * them means the bytes we hashed are not necessarily the bytes ffmpeg read.
 *
 * It can be defeated by a writer that restores the inode and timestamps, and an
 * agent running on the creator's own machine over the creator's own archive is
 * not the place to defend against that. What it must never do is the opposite
 * error — call two different versions the same one — and for every failure mode
 * this scanner actually meets, it does not.
 */
export function sameSource(before: Stats, after: Stats | null): boolean {
  if (!after) return false;
  return before.ino === after.ino && before.size === after.size && before.mtimeMs === after.mtimeMs;
}

export async function scanRoot(
  root: string,
  checkpoint?: ScanCheckpoint,
  resumeFrom?: Map<string, ScannedFile>,
  issues?: ScanIssue[],
): Promise<ScannedFile[]> {
  const out: ScannedFile[] = [];
  for await (const path of walk(root, root, issues)) {
    const ext = extname(path).toLowerCase();
    const kind = IMAGE_EXT.has(ext) ? "image" : VIDEO_EXT.has(ext) ? "video" : null;
    if (!kind) continue;
    const rel = relative(root, path);
    // Size-gate BEFORE reading the whole file into memory (a multi-GB file in
    // a scanned folder would otherwise OOM the agent). Oversized media is
    // skipped entirely rather than partially hashed.
    let before: Stats;
    try {
      before = await stat(path);
    } catch {
      // I3-0712: a file that vanished between the walk listing it and the
      // stat() here used to just disappear from the count.
      issues?.push({ rel, reason: "read_failed" });
      continue;
    }
    if (before.size > MAX_MEDIA_BYTES) {
      issues?.push({ rel, reason: "oversized_media" });
      continue;
    }
    const prev = resumeFrom?.get(rel);
    if (prev && prev.ino === before.ino && prev.mtime_ms === before.mtimeMs && prev.size === before.size) {
      // I3-1706: the same file, unchanged since the last checkpoint — reuse
      // its record instead of re-hashing it.
      out.push(prev);
      if (checkpoint) await checkpoint(out);
      continue;
    }
    // PS-087: past the size gate, the read itself can still fail for reasons
    // that are not "corrupt media" — a symlink named with a media extension
    // can point at a directory (stat() follows it and reports the directory's
    // own tiny "size", then readFile() throws EISDIR), the file can vanish, or
    // it can be swapped/grown between stat() and readFile() (stat->readFile
    // TOCTOU). Before this fix that exception was unhandled and aborted the
    // ENTIRE scanRoot() call — contradicting the promise two lines below that
    // "a corrupt file never fails the scan" (that promise only covered decode
    // failures, not the read). One bad entry now skips; the run continues.
    try {
      // I8 #175: the Buffer readFile returns IS a Uint8Array; wrapping it in
      // `new Uint8Array(...)` made a second full-size copy of up to 2 GB.
      const bytes: Uint8Array = await readFile(path);
      const parts = rel.split(/[\\/]/);
      const group = parts.length > 1 ? parts[0]! : basename(dirname(path)) || "unsorted";
      const entry: ScannedFile = {
        path,
        rel,
        group,
        media_kind: kind,
        exact_hash: hex(await sha256Bytes(bytes)),
        size: bytes.length,
        ino: before.ino,
        mtime_ms: before.mtimeMs,
      };
      // S28: real perceptual fingerprints, computed locally. Failures degrade to
      // exact-hash-only — a corrupt file never fails the scan.
      if (kind === "image") {
        const fp = imageFingerprint(bytes);
        if (fp) {
          entry.perceptual = fp.perceptual;
          entry.width = fp.width;
          entry.height = fp.height;
        } else {
          // SAY SO. The degrade is deliberate — a file the decoder will not touch
          // must never fail a whole folder scan — but until F20 it was also
          // INVISIBLE: the entry landed in the draft looking like any other, and a
          // half-downloaded fragment was registered as release provenance with no
          // hint that the creator was vouching for a broken file. `pnpm scan`
          // printed one line, a count of everything.
          //
          // The decoder answers null for one of three reasons and the creator can
          // act on the difference: the file is not a format we read, it declares
          // more pixels than we will decode, or it does not carry the pixel data
          // its own header promises (a truncated download, a partial cloud sync).
          entry.fingerprint_skipped = fingerprintSkipReason(bytes);
        }
      } else {
        // AR-02: TWO READS OF ONE PATH, and until this fix nothing tied them
        // together. The exact hash above is of the BYTES WE READ; `videoFingerprint`
        // takes the PATHNAME and lets ffmpeg open the file again, independently.
        // An ordinary cloud-sync replacement or a download completing between the
        // two produces an exact hash of version A and a perfectly successful
        // temporal/segment fingerprint of version B — and both stages report
        // success, so no skip reason fires and the draft stores them together as
        // one coherent record. Images are not affected: they fingerprint from the
        // same in-memory `bytes` the hash was taken over.
        const vfp = await videoFingerprint(path); // best-effort (needs ffmpeg on PATH)
        if (!vfp) {
          entry.fingerprint_skipped = "video_unreadable"; // usually: ffmpeg is not on PATH
        } else if (!sameSource(before, await statOrNull(path))) {
          // Do not emit a record whose halves came from different files. Keep the
          // exact hash — it is a true hash of bytes we really read — and mark the
          // entry retryable rather than inventing a combined identity. A rescan
          // over a settled file gets the fingerprint.
          entry.fingerprint_skipped = "source_changed";
        } else {
          entry.perceptual = vfp.perceptual;
          entry.segment_fps = vfp.segment_fps;
          entry.duration_ms = vfp.duration_ms;
        }
      }
      out.push(entry);
    } catch {
      // I3-0712: a read failure past the size gate (EISDIR, vanished, TOCTOU)
      // used to just `continue` with nothing recorded anywhere.
      issues?.push({ rel, reason: "read_failed" });
      continue;
    }
    if (checkpoint) await checkpoint(out);
  }
  return out;
}

/** Group scanned files into suggested releases (one per top-level group),
 * deduping exact re-copies (same hash). The creator reviews/edits before it
 * becomes a real ReleaseInput in the app. */
export interface SuggestedMedia {
  media_kind: "image" | "video";
  exact_hash: string;
  perceptual?: string;
  width?: number;
  height?: number;
  duration_ms?: number;
  segment_fps?: string[];
}

export function suggestReleases(files: ScannedFile[]): { title: string; media: SuggestedMedia[]; sources: string[] }[] {
  const byGroup = new Map<string, ScannedFile[]>();
  for (const f of files) (byGroup.get(f.group) ?? byGroup.set(f.group, []).get(f.group)!).push(f);
  const releases: { title: string; media: SuggestedMedia[]; sources: string[] }[] = [];
  for (const [group, gfiles] of byGroup) {
    const seen = new Set<string>();
    const media: SuggestedMedia[] = [];
    const sources: string[] = [];
    for (const f of gfiles) {
      if (seen.has(f.exact_hash)) continue; // exact duplicate / re-copy
      seen.add(f.exact_hash);
      media.push({
        media_kind: f.media_kind,
        exact_hash: f.exact_hash,
        ...(f.perceptual ? { perceptual: f.perceptual } : {}),
        ...(f.width ? { width: f.width, height: f.height } : {}),
        ...(f.duration_ms ? { duration_ms: f.duration_ms } : {}),
        ...(f.segment_fps ? { segment_fps: f.segment_fps } : {}),
      });
      sources.push(f.rel);
    }
    if (media.length > 0) releases.push({ title: group, media, sources });
  }
  return releases;
}

/** CLI: `tsx src/scan.ts <root> [out.json]` → writes a reviewable draft manifest
 * (fingerprints + metadata only; NO media). */
export async function main(argv: string[]): Promise<void> {
  const root = argv[2];
  if (!root) {
    // eslint-disable-next-line no-console
    console.error("usage: tsx src/scan.ts <root-folder> [out.json]");
    return;
  }
  const out = argv[3] ?? "dhwaja-archive-draft.json";
  const info = await stat(root);
  if (!info.isDirectory()) throw new Error("root must be a folder");
  // PS-086: a scan over a large NAS mount can run for hours. Checkpoint the
  // in-progress draft to `<out>.partial.json` every 25 files (not every file —
  // that would thrash the disk on a folder of thousands) so a crash partway
  // through leaves a resumable draft of what was already fingerprinted,
  // instead of losing the entire run. The checkpoint itself must never abort
  // the scan (best-effort: a write failure here is not the finding).
  const partial = `${out}.partial.json`;

  // I3-1706: an ACTUAL resume, not a checkpoint nothing ever reads back. If a
  // previous, crashed run of this exact `out` path left a partial checkpoint,
  // load its per-file records; scanRoot() below reuses any whose stat
  // identity still matches the file on disk instead of re-hashing it. A
  // corrupt or foreign partial file is treated the same as none at all — the
  // scan still runs, just without the head start.
  let resumeFrom: Map<string, ScannedFile> | undefined;
  try {
    const prev = JSON.parse(await readFile(partial, "utf8")) as { files?: Array<Omit<ScannedFile, "path">> };
    if (Array.isArray(prev.files)) {
      resumeFrom = new Map(prev.files.map((f) => [f.rel, { ...f, path: join(root, f.rel) } as ScannedFile]));
    }
  } catch {
    // no partial file from a previous run, or it doesn't parse: start clean.
  }

  let sinceCheckpoint = 0;
  const issues: ScanIssue[] = [];
  const files = await scanRoot(
    root,
    async (soFar) => {
      if (++sinceCheckpoint < 25) return;
      sinceCheckpoint = 0;
      try {
        // I3-1706: the checkpoint now carries the full per-file records (with
        // their stat identity) so a resumed run can skip re-hashing them —
        // not just a `suggestReleases()` summary nothing could resume from.
        // I3-0513: no absolute `path` on disk here either — `rel` plus the
        // root passed on the next invocation is enough to reconstruct it.
        const withoutPath = soFar.map(({ path: _path, ...rest }) => rest);
        await writeFile(partial, JSON.stringify({ scanned: soFar.length, files: withoutPath, partial: true }, null, 2));
      } catch {
        // best-effort checkpoint; a write failure here must not abort the scan
      }

    issues,
  );

  // read back, wrongly, by the next unrelated scan writing to this same `out`).

    // to clean up.

  const skipped = files.filter((f) => f.fingerprint_skipped);

  // a terminal that scrolls away.
  //

  // media entries mirror the `FingerprintInput` shape the server validates and
  // the creator eventually signs; a diagnostic is about THIS SCAN, not about the

  // the releases rather than inside them is what lets the draft stay honest
  // without widening what gets vouched for.
  //
  // EVERY skipped file, not a sample — `scanned - fingerprinted` reconciles
  // against this array's length, so the summary and the list cannot disagree.
  const diagnostics = skipped.map((f) => ({ rel: f.rel, reason: f.fingerprint_skipped! }));

    out,

    // "fingerprints + metadata only"; the creator's absolute local folder path
    // is neither, and combined with the relative paths already in

    // local path of every scanned file. Nothing downstream reads `root` back
    // (grepped: no consumer of `draft.root`), so dropping it costs nothing.
    JSON.stringify(

      null,

  if (skipped.length > 0) {

    // still identified by their exact hash — but they carry no perceptual
    // fingerprint, and `incomplete_pixel_data` in particular means the file on

    // release provenance would want to know before they do.

    const summary = [...byReason.entries()].map(([r, n]) => `${n} ${r}`).join(", ");
    // eslint-disable-next-line no-console

    for (const f of skipped.slice(0, 10)) {

      // eslint-disable-next-line no-console
      console.log(`    …and ${skipped.length - 10} more`);
    }
  }
  if (issues.length > 0) {

    // files past the size cap are no longer invisible — they're counted and
    // named, so the draft's `scanned` figure can be reconciled against what's
    // actually on disk instead of silently undercounting it.
    const byReason = new Map<string, number>();
    for (const i of issues) byReason.set(i.reason, (byReason.get(i.reason) ?? 0) + 1);
    const summary = [...byReason.entries()].map(([r, n]) => `${n} ${r}`).join(", ");
    // eslint-disable-next-line no-console
    console.log(`  ${issues.length} path(s) could not be scanned at all (${summary}) — listed in ${out} under "scan_issues"`);
  }
}

    // eslint-disable-next-line no-console

 *    rawvideo on stdout — the media itself still never leaves the machine and
 *    nothing is written to disk. Absent ffmpeg → null, and the scanner keeps
 *    the exact hash alone (fail-open to LESS data, never to an error).
 */

export const FRAME_CADENCE_SECONDS = 2;
export const FRAMES_PER_SEGMENT = 15; // 15 frames × 2s = 30-second segments
const FRAME_LEN = DHASH_COLS * DHASH_ROWS; // 72 bytes per sampled frame
const MAX_FRAMES = 4 * 60 * 60 / FRAME_CADENCE_SECONDS; // cap at 4h of video

/** Per-frame hashes from concatenated 9×8 grayscale frame samples. */
export function frameHashes(raw: Uint8Array): string[] {
  const n = Math.min(Math.floor(raw.length / FRAME_LEN), MAX_FRAMES);
  const out: string[] = [];
  for (let i = 0; i < n; i++) out.push(dHash(raw.subarray(i * FRAME_LEN, (i + 1) * FRAME_LEN)));
  return out;
}

/** Majority bit across a set of 64-bit hex hashes → one 16-hex fingerprint.
 * Ties (exactly half) resolve to 1, deterministically. */
export function majorityHash(hashes: string[]): string {
  if (hashes.length === 0) throw new Error("majorityHash needs at least one hash");
  const counts = new Array<number>(64).fill(0);
  for (const h of hashes) {
    for (let i = 0; i < 16; i++) {
      const nibble = parseInt(h[i]!, 16);
      for (let b = 0; b < 4; b++) if (nibble & (1 << (3 - b))) counts[i * 4 + b]! += 1;
    }
  }
  let hex = "";
  for (let i = 0; i < 64; i += 4) {
    let nibble = 0;
    for (let b = 0; b < 4; b++) if (counts[i + b]! * 2 >= hashes.length) nibble |= 1 << (3 - b);
    hex += nibble.toString(16);
  }
  return hex;
}

export interface VideoFingerprint {
  perceptual: string; // overall temporal fingerprint (16 hex)
  segment_fps: string[]; // one per FRAMES_PER_SEGMENT window (clip matching)
  duration_ms: number; // derived from sampled frame count × cadence
}

/** Fold per-frame hashes into the server's fingerprint fields. */
export function combineFrames(hashes: string[]): VideoFingerprint | null {
  if (hashes.length === 0) return null;
  const segments: string[] = [];
  for (let i = 0; i < hashes.length; i += FRAMES_PER_SEGMENT) {
    segments.push(majorityHash(hashes.slice(i, i + FRAMES_PER_SEGMENT)));
  }
  return {
    perceptual: majorityHash(hashes),
    segment_fps: segments,
    duration_ms: hashes.length * FRAME_CADENCE_SECONDS * 1000,
  };
}

/**
 * Sample a local video via ffmpeg (if installed) straight to 9×8 grayscale
 * rawvideo — no temp files, no media leaving the machine. Returns null when
 * ffmpeg is missing or fails (the scanner then keeps exact-hash-only).
 *
 * SECURITY (review HIGH): execFile blocks SHELL injection, but ffmpeg treats
 * -i's argument as an input URL and its demuxers follow references embedded in
 * the FILE CONTENT (HLS/m3u8 playlists, ffconcat scripts, containers with
 * external http/file/subfile/concat refs). An unknown-provenance file in a
 * scanned NAS folder could thus make the "trusted" agent issue network
 * requests or read arbitrary local files. Defenses:
 *   - `-protocol_whitelist file` (no http/tcp/…): the file protocol only, so
 *     no outbound request and no non-file reference is followed;
 *   - the path is RESOLVED to an absolute path (a leading '-' can't be parsed
 *     as an ffmpeg option), and a path that still begins with '-' is refused;
 *   - `-nostdin` so ffmpeg never blocks on / consumes our stdin.
 */
/** Resolve a scan path to a safe ffmpeg input argument, or null if it can't be
 * made safe. Absolute-resolving neutralizes a leading '-' (which ffmpeg would
 * read as an option); a still-'-'-leading result (only reachable on exotic
 * platforms) is refused. Exported for testing the guard without ffmpeg. */
export function safeInputPath(path: string): string | null {
  if (typeof path !== "string" || path.length === 0) return null;
  const abs = isAbsolute(path) ? path : resolve(path);
  if (abs.startsWith("-")) return null;
  return abs;
}

export async function ffmpegFrameSamples(path: string): Promise<Uint8Array | null> {
  const abs = safeInputPath(path);
  if (abs === null) return null; // never let the input parse as an option
  try {
    const { stdout } = await execFileAsync(
      "ffmpeg",
      [
        "-v", "error",
        "-nostdin",
        "-protocol_whitelist", "file",
        "-i", abs,
        "-vf", `fps=1/${FRAME_CADENCE_SECONDS},scale=${DHASH_COLS}:${DHASH_ROWS}`,
        "-pix_fmt", "gray",
        "-f", "rawvideo",
        "pipe:1",
      ],
      { encoding: "buffer", maxBuffer: MAX_FRAMES * FRAME_LEN + FRAME_LEN, timeout: 10 * 60 * 1000 },
    );
    return new Uint8Array(stdout as unknown as Buffer);
  } catch {
    return null; // no ffmpeg / decode failure / oversized → exact-hash-only
  }
}

/** Full best-effort pipeline for a local video file. */
export async function videoFingerprint(path: string): Promise<VideoFingerprint | null> {
  const raw = await ffmpegFrameSamples(path);
  if (!raw) return null;
  return combineFrames(frameHashes(raw));
}
