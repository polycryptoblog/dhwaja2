import type { DbLike } from "./db.js";
import { FOUNDER_FOUNDING_RANK, FOUNDER_JOIN_NO, RESERVED_SENTINEL, founderIds } from "./founder.js";
import { CLAIM_STATE_AT_SQL, grantLiveAtSql, withInstantLiteral } from "./standing.js";

/**
 * Registry ORDINALS: the two scarce, monotonic, never-reused numbers a creator
 * can hold.
 *
 *  1. FOUNDING BADGE (`founding`) — a fixed-capacity award for the first
 *     cohort of creators an operator VETTED.
 *  2. JOIN NUMBER (`join_numbers`) — the permanent registry number a creator
 *     is given when they CLAIM their passport.
 *
 * Both are allocated here so the "#1 is the founder's, reserved from migration
 * time" mechanism has exactly one implementation.
 *
 * THERE IS A THIRD SCARCE ORDINAL, AND IT LIVES IN `requests.ts`: the FAN-side
 * Founding Fan rank (`founding_fan_ranks`, 0020). It was written to this file's
 * model deliberately (audit #19) — numbers are ROWS, allocated once, never
 * recomputed, never re-minted into a slot somebody vacated — and it copies
 * WINDOW ITEM 10 below: the creator badge stopped accepting merely-`claimed`
 * passports because being EARLY was absorbing ranks that should have gone to
 * being VETTED, and the fan rank now requires a QUALIFIED, seven-day-old
 * account for exactly the same reason. It is not allocated here only because
 * its ordering key is the frozen `creator_requests` set, which is that
 * module's.
 *
 * Integrity, DB-enforced for both:
 *  - Scarce / dense: rank and join_no are UNIQUE, so two creators can never
 *    hold the same number. Concurrent allocations race on the UNIQUE
 *    constraint — the loser retries the next number; nobody double-books #7.
 *  - Immutable + idempotent: passport_id is the PK, so a creator is numbered
 *    at most once and their number never changes.
 *  - #1 is RESERVED, not raced: `0011_founder.sql` inserts a sentinel row holding
 *    number 1 in BOTH tables before any allocation can run. Non-founders
 *    allocate from 2 upward (the expression cannot even produce 1) and are
 *    blocked by UNIQUE besides; the founder takes #1 by swapping the sentinel
 *    out in ONE atomic batch, which is idempotent under re-runs.
 */

export const FOUNDING_CAPACITY = 500;

/**
 * WINDOW ITEM 10: `verified` ONLY. `claimed` used to qualify, which meant the
 * planned bulk-claim cohort could absorb the first ranks by being early rather
 * than by being vetted. The badge now means an operator looked at you, which is
 * what the trust ladder says and what makes the badge worth holding.
 */
const QUALIFYING: readonly string[] = ["verified"];

const PSP_RE = /^psp_[0-9a-f]{16}$/;

export interface FoundingResult {
  awarded: boolean;
  rank?: number;
  reason?: "already_awarded" | "not_qualified" | "capacity_full" | "retracted";
}

async function currentRank(db: DbLike, passportId: string): Promise<number | null> {
  const r = await db.prepare("SELECT rank FROM founding WHERE passport_id = ?").bind(passportId).first<{ rank: number }>();
  return r?.rank ?? null;
}

/**
 * The Founding rank a passport may be SHOWN with, or null.
 *
 * I3-0910 (DECIDED, I4-08-R3): the rank is served only while the verification
 * that earned it is in force — the passport is live AND `verified`. A creator
 * walked back to `claimed` (downgrade / expiry / operator lapse) keeps the ROW
 * (the number stays burned: never freed, never reused, never re-counted against
 * capacity) but every surface reading through here stops showing it, and a
 * re-verification restores the same number. Withheld, not deleted, because a
 * deletion is irreversible and would free a scarce number `MAX(rank)+1` could
 * then hand to somebody else.
 *
 * I3-1768 (D-1085(1), I34-EXPIRY-A): "verified" is read at `nowIso` — a `verified` row whose latest grant or renewal
 * has lapsed by then (standing.ts `grantLiveAtSql`) shows no rank, before the expiry sweep folds it back.
 */
export async function foundingRank(db: DbLike, passportId: string, nowIso: string = new Date().toISOString()): Promise<number | null> {
  if (!PSP_RE.test(passportId)) return null; // the reservation sentinel is never addressable
  const r = await db
    .prepare(
      `SELECT f.rank AS rank FROM founding f JOIN passports p ON p.passport_id = f.passport_id WHERE f.passport_id = ? AND p.retracted = 0 AND p.claim_state = 'verified' AND ${grantLiveAtSql("p.passport_id")}`,
    )
    .bind(passportId, nowIso)
    .first<{ rank: number }>();
  return r?.rank ?? null;
}

/**
 * Take a RESERVED number by swapping the sentinel out, atomically. Both
 * statements ride ONE batch, so either the sentinel is gone AND the founder
 * holds the number, or neither happened. The DELETE is guarded on the sentinel
 * still holding that exact number, so a re-run (or a race with a second sweep)
 * changes nothing and the caller re-reads the existing number.
 *
 * The two SQL strings are a closed, hard-coded vocabulary selected by this
 * function's own literal-union parameter — never caller data, never built by
 * concatenation.
 */
async function takeReservedNumber(db: DbLike, table: "founding" | "join_numbers", passportId: string, reserved: number, ts: string): Promise<boolean> {
  const del = table === "founding" ? "DELETE FROM founding WHERE passport_id = ? AND rank = ?" : "DELETE FROM join_numbers WHERE passport_id = ? AND join_no = ?";
  const ins =
    table === "founding"
      ? "INSERT INTO founding (passport_id, rank, awarded_at) VALUES (?, ?, ?)"
      : "INSERT INTO join_numbers (passport_id, join_no, allocated_at) VALUES (?, ?, ?)";
  try {
    await db.batch([db.prepare(del).bind(RESERVED_SENTINEL, reserved), db.prepare(ins).bind(passportId, reserved, ts)]);
    return true;
  } catch (e) {
    // I3-1507: only a CONSUMED reservation is a "no". If the sentinel still
    // holds the number, the swap failed for some other reason (a transient D1
    // fault) — surface it so the caller retries instead of telling the founder
    // "not qualified". A throwing re-read propagates for the same reason.
    const held = table === "founding" ? "SELECT 1 AS held FROM founding WHERE passport_id = ? AND rank = ?" : "SELECT 1 AS held FROM join_numbers WHERE passport_id = ? AND join_no = ?";
    if ((await db.prepare(held).bind(RESERVED_SENTINEL, reserved).first<{ held: number }>()) !== null) throw e;
    return false; // the reservation was already consumed — caller re-reads
  }
}

/**
 * Award the next Founding rank to a passport if eligible. Idempotent (returns
 * the existing rank), atomic (UNIQUE(rank) serializes concurrent awards),
 * capacity-bounded, and gated on the passport being live + VERIFIED.
 *
 * The FOUNDER is awarded the reserved rank #1. DECISION (window item 10): a
 * Founding badge is an AWARD attached to one creator, not a board fans read to
 * find creators, so the "the founder is off every board" rule deliberately does
 * not reach it — see FOUNDER_EXEMPTIONS.founding_rank_1. The founder must still
 * be VERIFIED to receive it; until then the slot stays reserved and unusable by
 * anyone else, and #1 is the only rank the founder can ever hold.
 */
export async function awardFoundingIfEligible(
  db: DbLike,
  passportId: string,
  now: () => string = () => new Date().toISOString(),
): Promise<FoundingResult> {
  if (!PSP_RE.test(passportId)) return { awarded: false, reason: "not_qualified" };

  const existing = await currentRank(db, passportId);
  if (existing !== null) return { awarded: true, rank: existing, reason: "already_awarded" };

  // EXPA-17 (D-1110(2), I34-W2-EXPIRY-B): the claim state AT `now()` (standing.ts CLAIM_STATE_AT_SQL) — a `verified`
  // row whose latest grant or renewal has lapsed by then reads `claimed` and burns no number before the sweep folds it.
  const pass = await db
    .prepare(`SELECT ${CLAIM_STATE_AT_SQL} AS claim_state, p.retracted AS retracted FROM passports p WHERE p.passport_id = ?`)
    .bind(now(), passportId)
    .first<{ claim_state: string; retracted: number }>();
  if (!pass || pass.retracted === 1) return { awarded: false, reason: "retracted" };
  if (!QUALIFYING.includes(pass.claim_state)) return { awarded: false, reason: "not_qualified" };

  const founder = await founderIds(db);
  if (founder && founder.passport_id === passportId) {
    if (await takeReservedNumber(db, "founding", passportId, FOUNDER_FOUNDING_RANK, now())) {
      return { awarded: true, rank: FOUNDER_FOUNDING_RANK };
    }
    const nowRank = await currentRank(db, passportId);
    if (nowRank !== null) return { awarded: true, rank: nowRank, reason: "already_awarded" };
    // The reservation is gone and the founder does not hold it. FAIL CLOSED:
    // the founder is #1 or nothing, never a consolation rank.
    return { awarded: false, reason: "not_qualified" };
  }

  // Try to claim the next rank. Concurrent awards collide on UNIQUE(rank); the
  // loser retries with the recomputed next rank. Bounded by CAPACITY. The
  // reserved-#1 sentinel keeps MAX(rank) >= 1, and the floor below makes the
  // non-founder path structurally unable to produce 1 even if it did not.
  for (let attempt = 0; attempt < FOUNDING_CAPACITY + 1; attempt++) {
    const row = await db.prepare("SELECT COUNT(*) AS n, MAX(rank) AS maxr FROM founding").bind().first<{ n: number; maxr: number | null }>();
    const count = row?.n ?? 0;
    if (count >= FOUNDING_CAPACITY) return { awarded: false, reason: "capacity_full" };
    const nextRank = Math.max(FOUNDER_FOUNDING_RANK + 1, (row?.maxr ?? 0) + 1);
    try {
      await db.prepare("INSERT INTO founding (passport_id, rank, awarded_at) VALUES (?, ?, ?)").bind(passportId, nextRank, now()).run();
      return { awarded: true, rank: nextRank };
    } catch (e) {
      // Lost the race on either rank (someone took nextRank) or passport
      // (a concurrent award of THIS passport) — re-read and resolve.
      const nowRank = await currentRank(db, passportId);
      if (nowRank !== null) return { awarded: true, rank: nowRank, reason: "already_awarded" };
      // Only a UNIQUE-constraint clash (a rank taken by a concurrent award) is
      // retryable; any other constraint/error is a real fault — surface it.
      if (e instanceof Error && /UNIQUE constraint/i.test(e.message)) continue;
      throw e;
    }
  }
  return { awarded: false, reason: "capacity_full" };
}

/**
 * Sweep: award Founding to any qualifying, not-yet-awarded passports (cron).
 *
 * WINDOW ITEM 10: the order is DETERMINISTIC. This used to be
 * `ORDER BY p.created_seq` over anything claimed-or-better — i.e. the rank of
 * an award called "Founding" was decided by when Dhwaja happened to LIST you,
 * and the badge itself by whatever order the query returned. Rank order is now
 * VERIFICATION order: the `verification_granted` event's own timestamp, tied
 * broken by that event's ledger seq (a strict total order, so two grants in the
 * same millisecond still resolve) and then by passport_id (so even a
 * hand-assembled ledger cannot produce a coin flip).
 *
 * U-10 FIX: a passport can carry MORE than one `verification_granted` — a
 * downgrade/expiry/operator-lapse walks it back to `claimed`, and it can be
 * re-verified later, appending a second grant. `MIN(e.created_at)` over every
 * grant let a WITHDRAWN grant's timestamp still supply the ordering key,
 * putting a passport ahead of others on trust the ledger no longer holds.
 * `MAX` is correct, not just "less wrong": `verification_granted` only ever
 * fires from `claim_state = 'claimed'`, and every walk-back
 * (`downgrade_verification` / `verification_expired` / `operator_lapsed`)
 * only ever fires from `verified` back to `claimed` — so a passport whose
 * CURRENT state is `verified` (the WHERE clause below) necessarily has its
 * highest-seq `verification_granted` as the grant presently in force. No
 * `supersedes`/correction walk needed — the state machine already proves it.
 *
 * Re-running the sweep cannot reorder anything already awarded: a rank is a PK
 * per passport and UNIQUE per number, so the sweep only ever appends.
 */
export async function sweepFounding(db: DbLike, now: () => string = () => new Date().toISOString(), limit = 200): Promise<number> {
  const tick = await openDeltaTick(db, FOUNDING_CURSOR);
  if (tick === null) return 0;
  // I3-1533: BOTH ordering keys come from ONE event — the highest-seq grant `g`
  // (the grant in force, per the U-10 argument above). Two independent MAX()
  // aggregates could pair one grant's timestamp with another grant's seq.
  const { results } = await db
    .prepare(
      `SELECT p.passport_id AS passport_id, g.created_at AS verified_at, g.seq AS verified_seq, d.touch_seq AS touch_seq
         FROM (SELECT subject, MAX(seq) AS touch_seq FROM trust_events NOT INDEXED WHERE seq > ? GROUP BY subject) d
         JOIN passports p ON p.passport_id = d.subject
         JOIN trust_events g ON g.seq = (SELECT MAX(e.seq) FROM trust_events e WHERE e.subject = p.passport_id AND +e.event_type = 'verification_granted')
        WHERE p.retracted = 0 AND p.claim_state = 'verified' AND ${withInstantLiteral(grantLiveAtSql("p.passport_id"), Date.parse(now()))}
          AND NOT EXISTS (SELECT 1 FROM founding f WHERE f.passport_id = p.passport_id)
        ORDER BY verified_at ASC, verified_seq ASC, p.passport_id ASC
        LIMIT ?`,
    )
    .bind(tick.cursor, limit) // EXPA-17: a grant lapsed at the sweep's instant is not a candidate (inlined: the delta plan's binds are pinned)
    .all<{ passport_id: string; touch_seq: number }>();
  let awarded = 0;
  for (const p of results) {
    const r = await awardFoundingIfEligible(db, p.passport_id, now);
    if (r.awarded && r.reason !== "already_awarded") awarded += 1;
    if (r.reason === "capacity_full") break; // terminal: ranks are never freed, so nothing later can be awarded either
  }
  await closeDeltaTick(db, FOUNDING_CURSOR, tick, results.length >= limit, [], now);
  return awarded;
}

/**
 * The creator's permanent registry join number, or null if they have none yet.
 * A number exists only for a CLAIMED (or verified) passport: a seeded,
 * unclaimed listing about someone who never signed up is not a member and does
 * not take a number.
 */
export async function joinNumber(db: DbLike, passportId: string): Promise<number | null> {
  if (!PSP_RE.test(passportId)) return null; // the reservation sentinel is never addressable
  const r = await db.prepare("SELECT join_no FROM join_numbers WHERE passport_id = ?").bind(passportId).first<{ join_no: number }>();
  return r?.join_no ?? null;
}

/**
 * Allocate this passport's join number if it qualifies. Idempotent — an
 * already-numbered passport gets its existing number back, never a new one.
 *
 * WHY #1 CANNOT BE TAKEN BY ORDERING LUCK:
 *  - the sentinel row holding join_no = 1 is inserted by `0011_founder.sql`,
 *    before any passport can be claimed, so #1 is never free;
 *  - `join_no` is UNIQUE, so the DB refuses a second holder;
 *  - the non-founder allocator computes `max(2, MAX(join_no) + 1)`, so it
 *    cannot even ATTEMPT 1 — not on a fresh table, not after a manual delete,
 *    not on a re-run;
 *  - only the configured founder reaches the swap path, and that swap is one
 *    atomic batch, so a re-run or a concurrent sweep lands it exactly once.
 */
export async function allocateJoinNumber(db: DbLike, passportId: string, now: () => string = () => new Date().toISOString()): Promise<number | null> {
  if (!PSP_RE.test(passportId)) return null;

  const existing = await joinNumber(db, passportId);
  if (existing !== null) return existing;

  const pass = await db.prepare("SELECT claim_state, retracted FROM passports WHERE passport_id = ?").bind(passportId).first<{ claim_state: string; retracted: number }>();
  if (!pass || pass.retracted === 1) return null;
  // Numbered by CLAIM, not by listing: unclaimed shells never take a number.
  if (pass.claim_state !== "claimed" && pass.claim_state !== "verified") return null;

  const founder = await founderIds(db);
  if (founder && founder.passport_id === passportId) {
    if (await takeReservedNumber(db, "join_numbers", passportId, FOUNDER_JOIN_NO, now())) return FOUNDER_JOIN_NO;
    // Either a concurrent run already took it for the founder (fine — re-read)
    // or the reservation is gone. #1 or nothing; never a consolation number.
    return joinNumber(db, passportId);
  }

  for (let attempt = 0; attempt < 64; attempt++) {
    const row = await db.prepare("SELECT MAX(join_no) AS maxn FROM join_numbers").bind().first<{ maxn: number | null }>();
    const next = Math.max(FOUNDER_JOIN_NO + 1, (row?.maxn ?? 0) + 1);
    try {
      await db.prepare("INSERT INTO join_numbers (passport_id, join_no, allocated_at) VALUES (?, ?, ?)").bind(passportId, next, now()).run();
      return next;
    } catch (e) {
      const raced = await joinNumber(db, passportId);
      if (raced !== null) return raced;
      if (e instanceof Error && /UNIQUE constraint/i.test(e.message)) continue;
      throw e;
    }
  }
  return null;
}

/**
 * Sweep: number every claimed passport that has none yet (cron), in CLAIM
 * order — the ledger seq of the passport's first `claim_proven` (or of
 * `verification_granted`, for a passport an operator verified without a
 * separate claim event). Ledger seq is a strict total order, so the sweep is
 * fully deterministic and a re-run appends rather than reorders.
 */
export async function sweepJoinNumbers(db: DbLike, now: () => string = () => new Date().toISOString(), limit = 200): Promise<number> {
  const tick = await openDeltaTick(db, JOIN_NUMBERS_CURSOR);
  if (tick === null) return 0;
  const { results } = await db
    .prepare(
      `SELECT p.passport_id AS passport_id, d.touch_seq AS touch_seq,
              (SELECT MIN(e.seq) FROM trust_events e WHERE e.subject = p.passport_id AND +e.event_type IN ('claim_proven','verification_granted')) AS claim_seq
         FROM (SELECT subject, MAX(seq) AS touch_seq FROM trust_events NOT INDEXED WHERE seq > ? GROUP BY subject) d
         JOIN passports p ON p.passport_id = d.subject
        WHERE p.retracted = 0 AND p.claim_state IN ('claimed','verified')
          AND claim_seq IS NOT NULL
          AND NOT EXISTS (SELECT 1 FROM join_numbers j WHERE j.passport_id = p.passport_id)
        ORDER BY claim_seq ASC, p.passport_id ASC
        LIMIT ?`,
    )
    .bind(tick.cursor, limit)
    .all<{ passport_id: string; touch_seq: number }>();
  let allocated = 0;
  const unsettled: number[] = [];
  for (const p of results) {
    if ((await allocateJoinNumber(db, p.passport_id, now)) !== null) allocated += 1;
    else unsettled.push(p.touch_seq); // e.g. contention exhausted the retries — keep it inside the next tick's delta
  }
  await closeDeltaTick(db, JOIN_NUMBERS_CURSOR, tick, results.length >= limit, unsettled, now);
  return allocated;
}

/**
 * Both sweeps used to re-derive their candidate set from the WHOLE qualifying
 * event history every tick (`idx_trust_events_type` + GROUP BY), so a tick with
 * nothing to do cost O(every claim/grant ever written). They now keep a ledger
 * watermark in `sweep_cursor` (0026 — no new table) and look only at passports
 * that are the SUBJECT of an event appended after it: a tick with no new events
 * reads the ledger head and the cursor row and stops; any other tick reads a
 * rowid range proportional to what was appended since. `NOT INDEXED` is
 * load-bearing: without it the planner serves the GROUP BY from
 * `idx_trust_events_subject` and walks EVERY event (measured by EXPLAIN QUERY
 * PLAN, test/i4-08-r3-founding.test.ts); rowid range lookup survives it. The
 * unary `+e.event_type` is load-bearing the same way: it keeps the per-candidate
 * lookup on `idx_trust_events_subject` instead of `idx_trust_events_type`.
 *
 * WHY NOTHING ELIGIBLE CAN HIDE BEHIND THE WATERMARK: the eligibility predicate
 * reads `passports.claim_state`, `passports.retracted` and the passport's own
 * events. Every live write of `claim_state` is a fold effect riding the SAME
 * atomic batch as the event whose subject is that passport (events.ts
 * `appendEvents` -> one `db.batch`), and `retracted` only ever goes 0 -> 1. So a
 * passport that becomes eligible does so in a batch that also appends an event
 * with a seq above the head this tick read, and the NEXT tick's delta holds it.
 * The watermark advances only when the page was not full and every candidate
 * settled; an unsettled one pins it just below that candidate's newest event.
 * `awardFoundingIfEligible` / `allocateJoinNumber` THROWING (a transient fault,
 * I3-1507) aborts the tick before the cursor moves.
 *
 * RESIDUAL: a fold rebuild (`fold-rebuild.ts`) writes passports from the log
 * without appending. It only ever makes the fold AGREE with the log, but if the
 * live fold had drifted, a passport it makes eligible would not be seen until
 * its next event. Operator remedy after a rebuild: delete the two cursor rows
 * (`DELETE FROM sweep_cursor WHERE name IN ('founding_sweep','join_numbers_sweep')`),
 * which makes the next tick a full pass.
 */
const FOUNDING_CURSOR = "founding_sweep";
const JOIN_NUMBERS_CURSOR = "join_numbers_sweep";

interface DeltaTick {
  cursor: number;
  head: number;
}

async function openDeltaTick(db: DbLike, name: string): Promise<DeltaTick | null> {
  const h = await db.prepare("SELECT COALESCE(MAX(seq), 0) AS head FROM trust_events").bind().first<{ head: number }>();
  const c = await db.prepare("SELECT cursor_seq FROM sweep_cursor WHERE name = ?").bind(name).first<{ cursor_seq: number }>();
  const head = h?.head ?? 0;
  const cursor = c?.cursor_seq ?? 0;
  return head > cursor ? { cursor, head } : null;
}

async function closeDeltaTick(db: DbLike, name: string, tick: DeltaTick, pageFull: boolean, unsettledTouchSeqs: number[], now: () => string): Promise<void> {
  if (pageFull) return; // more candidates may sit beyond the page: re-read the same delta next tick
  const next = unsettledTouchSeqs.length > 0 ? Math.min(...unsettledTouchSeqs) - 1 : tick.head;
  if (next <= tick.cursor) return;
  await db
    .prepare("INSERT INTO sweep_cursor (name, cursor_seq, cursor_id, updated_at) VALUES (?, ?, '', ?) ON CONFLICT(name) DO UPDATE SET cursor_seq = excluded.cursor_seq, updated_at = excluded.updated_at")
    .bind(name, next, now())
    .run();
}

/**
 * Gifting — ONE transfer primitive for collectibles, ownership history,
 * consented gifting (F3, 2026-09-03). (`collectibles.ts` is the creator-side
 * collectible MINT ceremony; this module is what happens to a collectible after
 * a fan holds it.)
 *
 * TWO KINDS, ONE SHAPE. A Creator War card (`card_instances`, provenance in
 * `card_instance_log`) and a credential-redeemed acquisition (`acquisitions`,
 * provenance in `acquisition_log` since 0060). Both move the same way: a log
 * row INSERT predicated on the PRE-state (the sender still owns it, the
 * recipient exists and has room) and an UPDATE with the same predicate, in one
 * batch — so a move without a log row, or a log row without a move, cannot
 * exist. Whether a move happened is read from THIS call's log row, never from
 * post-state ownership (the `war-instances.ts` discipline).
 *
 * CONSENT IS A ROW. `war-instances.ts:195` is the contract: a bare gift
 * endpoint is not consent — a stranger could fill a victim's shelf. So a gift
 * is an OFFER the sender writes and the recipient ACCEPTS; only the accept
 * moves the item, and the accept is the recipient's own authenticated call.
 * Offers are cheap, so they are capped in the INSERT (open per sender, per
 * day, per pair), expire after `GIFT_OFFER_TTL_MS`, are one-per-item (partial
 * UNIQUE), and carry NO free text. The recipient is named by PUBLIC handle —
 * an acc_ id is never accepted from a client and a private handle reads as
 * "no such collector" (the same answer as a handle that does not exist, so an
 * offer is not a visibility oracle).
 *
 * WHAT A GIFT IS NOT. Not money (no journal row, no `sales` row, nothing for
 * the refund engine); not a marketplace (no price, no counter-offer); not
 * surplus-gated (a gift may move the last copy — that is the point). Bound
 * (starter) cards and escrowed (staked) cards cannot be offered.
 *
 * RECOVERY. `recovered` is the one move nobody accepts: an operator returns an
 * item to the owner an account takeover took it from. It rides the same
 * primitive with an audit line; the console route runs it through dual control.
 *
 * HISTORY. `collectibleHistory` is OWNER-ONLY and names no one: counterparties
 * render as `you` / `another collector`. A history that named handles would be
 * a social graph the collector never agreed to publish.
 */

export class CollectibleError extends Error {
  override name = "CollectibleError";

/** ONE refusal for every "this cannot be offered / accepted": not yours, not
 * transferable, on offer already, recipient unknown or private, caps. Which of
 * those it was is not for the caller to learn. */
const REFUSAL = "this can't be gifted right now";

function checkItem(kind: CollectibleKind, itemId: string): void {
  if (kind !== "card" && kind !== "acquisition") throw new CollectibleError("bad kind");
  if (typeof itemId !== "string" || !(kind === "card" ? CI_RE : ACQ_RE).test(itemId)) throw new CollectibleError("bad item id");
}

/**
 * The transfer primitive. Two statements, one batch, both predicated on the
 * PRE-state. `event` is `gifted` or `recovered`; `detail` the offer id or the
 * audit ref. Cards delegate to `transferInstanceStatements` (mode `any`, the
 * same escrow/bound/shelf guards); acquisitions get the twin shape here.
 */
export function transferCollectibleStatements(
  db: DbLike,
  args: { kind: CollectibleKind; itemId: string; fromOwner: string; toOwner: string; event: "gifted" | "recovered"; detail: string; logId: string; ts: string; extraGuard?: { sql: string; binds: unknown[] } },
): DbStatement[] {
  checkItem(args.kind, args.itemId);
  if (!OWNER_RE.test(args.fromOwner) || !ACCOUNT_RE.test(args.toOwner)) throw new CollectibleError("bad owner id", 500);
  if (args.fromOwner === args.toOwner) throw new CollectibleError("transfer to self", 500);
  if (args.event !== "gifted" && args.event !== "recovered") throw new CollectibleError("bad event", 500);
  if (typeof args.detail !== "string" || args.detail.length === 0 || args.detail.length > MAX_DETAIL_LEN) throw new CollectibleError("bad detail", 500);
  if (args.kind === "card") {
    if (!CIL_RE.test(args.logId)) throw new CollectibleError("bad log id", 500);
    return transferInstanceStatements(db, {
      instanceId: args.itemId,
      fromOwner: args.fromOwner,
      toOwner: args.toOwner,
      reason: args.detail,
      logId: args.logId,
      ts: args.ts,
      mode: "any",
      event: args.event,
      ...(args.extraGuard ? { extraGuard: args.extraGuard } : {}),
    });
  }
  if (!AQL_RE.test(args.logId)) throw new CollectibleError("bad log id", 500);
  if (args.extraGuard && !/^ AND /.test(args.extraGuard.sql)) throw new CollectibleError("extraGuard must begin with ' AND '", 500);
  const extraSql = args.extraGuard?.sql ?? "";
  const extraBinds = args.extraGuard?.binds ?? [];
  return [
    db
      .prepare(
        `INSERT INTO acquisition_log (log_id, acquisition_id, event, from_owner, to_owner, detail, at)
         SELECT ?, ?, ?, ?, ?, ?, ?
          WHERE EXISTS (SELECT 1 FROM acquisitions WHERE acquisition_id = ? AND account_id = ?)
            AND EXISTS (SELECT 1 FROM accounts WHERE account_id = ?)${extraSql}`,
      )
      .bind(args.logId, args.itemId, args.event, args.fromOwner, args.toOwner, args.detail, args.ts, args.itemId, args.fromOwner, args.toOwner, ...extraBinds),
    // The holder changes; the REDEEMER does not (0061) — a moved row always
    // carries who redeemed it, so evidence readers can tell property from proof.
    db
      .prepare(
        `UPDATE acquisitions SET account_id = ?, redeemed_by = COALESCE(redeemed_by, ?)
          WHERE acquisition_id = ? AND account_id = ?
            AND EXISTS (SELECT 1 FROM accounts WHERE account_id = ?)${extraSql}`,
      )
      .bind(args.toOwner, args.fromOwner, args.itemId, args.fromOwner, args.toOwner, ...extraBinds),
    // A featured slot pointing at the moved acquisition is the SENDER's shelf
    // decoration, not the item — it goes with the move (predicated on THIS
    // call's log row, so a no-op move deletes nothing). `setFeatured` writes
    // acquisitions as kind 'acquisition' (kind 'card' rows carry a passport).
    db
      .prepare(
        `DELETE FROM featured_items WHERE account_id = ? AND kind = 'acquisition' AND ref_id = ?
            AND EXISTS (SELECT 1 FROM acquisition_log WHERE log_id = ?)`,
      )
      .bind(args.fromOwner, args.itemId, args.logId),
  ];
}

/** Did THIS call's transfer land? Read from its log row. */
export async function transferLanded(db: DbLike, kind: CollectibleKind, logId: string): Promise<boolean> {
  const table = kind === "card" ? "card_instance_log" : "acquisition_log";
  const row = await db.prepare(`SELECT 1 AS x FROM ${table} WHERE log_id = ?`).bind(logId).first();
  return row !== null;
}

/** Owns it, and it can move: living, unbound, not staked (cards); simply owned
 * (acquisitions). This read is the early, generic refusal at OFFER time; the
 * accept re-asserts every condition inside its batch (the transfer predicate),
 * so an item that stopped being movable in between voids the offer instead. */
async function transferable(db: DbLike, kind: CollectibleKind, itemId: string, owner: string, nowMs: number): Promise<boolean> {
  const held = openHoldGuard(kind, itemId, new Date(nowMs).toISOString());
  const row =
    kind === "card"
      ? await db.prepare(`SELECT 1 AS x FROM card_instances WHERE instance_id = ? AND owner_id = ? AND bound = 0 AND consumed_at IS NULL AND escrowed_by IS NULL${held.sql}`).bind(itemId, owner, ...held.binds).first()
      : await db.prepare(`SELECT 1 AS x FROM acquisitions WHERE acquisition_id = ? AND account_id = ?${held.sql}`).bind(itemId, owner, ...held.binds).first();
  return row !== null;
}

/** I3-0457: an item under an OPEN, unexpired recovery hold does not move by gift.
 * The hold exists so a disputed/stolen item survives until it is returned
 * (war-bond.ts and war-instances.ts refuse burn and stake on the same row); a
 * gift was the one movement verb that ignored it. ONE predicate, used at offer
 * time (`transferable`) and inside the accept batch (`acceptGift`), keyed on the
 * bound item id so it reads the same in the log INSERT and the owner UPDATE.
 * `recoverCollectible` does not carry it: recovery is the verb that resolves a hold. */
function openHoldGuard(kind: CollectibleKind, itemId: string, ts: string): { sql: string; binds: unknown[] } {
  return { sql: " AND NOT EXISTS (SELECT 1 FROM collectible_holds h WHERE h.kind = ? AND h.item_id = ? AND h.state = 'open' AND h.expires_at > ?)", binds: [kind, itemId, ts] };
}

/** The recipient, by PUBLIC handle only. Unknown and private are the same null. */
async function publicAccountByHandle(db: DbLike, handle: unknown): Promise<string | null> {
  const h = typeof handle === "string" ? handle.trim().toLowerCase() : "";
  if (!HANDLE_RE.test(h)) return null;
  const row = await db.prepare("SELECT account_id FROM user_profiles WHERE handle = ? AND visibility = 'public'").bind(h).first<{ account_id: string }>();
  return row?.account_id ?? null;
}

export interface OfferView {
  offer_id: string;
  kind: CollectibleKind;
  /** The item — for the SENDER's own offers, and for an incoming offer once
   * accepted. An OPEN incoming offer carries none: an acquisition id is a
   * capability (it verifies the item to anyone who has it), and a recipient
   * who declines must not walk away holding it (F3 review, Negative-space). */
  item_id: string | null;
  /** The other party's public handle — null if they have since gone private
   * or erased (the offer still stands until it expires). */
  counterparty: string | null;
  direction: "incoming" | "outgoing";
  state: OfferState;
  created_at: string;
  expires_at: string;
}

/** `gift_sent`'s landed fact: this call's own offer row (binds offer_id, kind, item_id, from_account, created_at). */
const GIFT_OFFER_LANDED = "EXISTS (SELECT 1 FROM gift_offers WHERE offer_id = ? AND kind = ? AND item_id = ? AND from_account = ? AND created_at = ?)";

/**
 * Write an offer. Caps live INSIDE the insert; the partial UNIQUE on
 * (kind, item_id) WHERE open refuses a second offer on one item. Every
 * refusal is the same 409.
 */
export async function offerGift(
  db: DbLike,
  fromAccount: string,
  input: { kind: unknown; item_id: unknown; to_handle: unknown },
  offerId: string,
  nowMs: number,
  pubEnv?: AnalyticsTickEnv,
  pubCtx?: { waitUntil(p: Promise<unknown>): void },
): Promise<OfferView> {
  if (!ACCOUNT_RE.test(fromAccount)) throw new CollectibleError("bad account id", 500);
  if (!GFT_RE.test(offerId)) throw new CollectibleError("bad offer id", 500);
  const kind = input.kind as CollectibleKind;
  const itemId = input.item_id as string;
  checkItem(kind, itemId);
  const to = await publicAccountByHandle(db, input.to_handle);
  if (!to || to === fromAccount) throw new CollectibleError(REFUSAL, 409);
  if (!(await transferable(db, kind, itemId, fromAccount, nowMs))) throw new CollectibleError(REFUSAL, 409);
  const ts = new Date(nowMs).toISOString();
  const dayStart = new Date(nowMs - 24 * 60 * 60 * 1000).toISOString();
  const declineCutoff = new Date(nowMs - DECLINE_COOLDOWN_MS).toISOString();
  const expiresAt = new Date(nowMs + GIFT_OFFER_TTL_MS).toISOString();
  // A stale OPEN offer written by a PREVIOUS owner (the item since changed
  // hands by a match ante or a recovery) must not lock the current owner out
  // of gifting for a week: void any open offer on this item whose sender no
  // longer holds it, before the partial UNIQUE is consulted.
  await voidOrphanedOffersStatement(db, kind, itemId, ts).run();
  // I3-1769, SECOND DOOR. The caps above count only UNEXPIRED open offers, but
  // the partial UNIQUE on (kind, item_id) WHERE open does not read `expires_at`
  // at all, so the sender's OWN dead offer still locks their item until the
  // cron sweep runs (`index.ts` -> `sweepExpiredOffers`). This is that sweep's
  // own predicate, narrowed to this item: idempotent with it, same state word,
  // same `resolved_at`. The statement above cannot serve — it is keyed to a
  // sender who no longer HOLDS the item, which is the opposite case.
  await db
    .prepare("UPDATE gift_offers SET state = 'expired', resolved_at = ? WHERE kind = ? AND item_id = ? AND state = 'open' AND expires_at <= ?")
    .bind(ts, kind, itemId, ts)
    .run();
  // THE CREATOR THIS GIFT IS OF. `gift_sent` is creator-scoped and the offer
  // row names no passport, so the row is read FROM the item's own table by an
  // `INSERT … SELECT` in the same batch (ADR-0006 §2) — `card_instances` holds
  // the passport directly, an acquisition holds it one join away on `releases`.
  // The `EXISTS` on the offer is what makes the event exist iff the offer does:
  // every cap above lives inside the INSERT, so a capped-out call writes no
  // offer row and must write no event either. It is `extraWhere` (INV-051 U-1)
  // and names THIS call's row — id, item, sender and stamp — so an offer id
  // already on file does not stand in for a refused one (S-088 G291).
  const giftEvent = db
    .prepare(
      kind === "card"
        ? outboxSelectSql("gift_sent", { table: "card_instances", where: "instance_id = ?", extraWhere: GIFT_OFFER_LANDED })
        : outboxSelectSql("gift_sent", { table: "releases", where: "release_id IN (SELECT release_id FROM acquisitions WHERE acquisition_id = ?)", extraWhere: GIFT_OFFER_LANDED }),
    )
    .bind(ts, itemId, offerId, kind, itemId, fromAccount, ts);
  let seqs: number[] = [];
  let inserted: unknown[];
  try {
    // Every cap is INSIDE the insert. Sender side: open, per day, per pair.
    // Recipient side: open incoming across all senders (alts cannot flood one
    // inbox), a decline in the last DECLINE_COOLDOWN_MS closes the pair. An
    // IGNORE or a BLOCK (`gift_blocks`, 0106) refuses nothing — D-0811(2), D-0812(1):
    // the new offer lands like any other and PAIR_QUIET hides it from every
    // recipient door until its own expires_at, so the sender cannot tell ignore,
    // block and silence apart. The recipient-wide incoming cap therefore counts
    // hidden offers too (D-0814(3)): excluding them made the free slots an
    // Ignore/Block oracle to a sender with alts. And
    // the SENDER must hold a PUBLIC profile — an offer that names nobody is
    // not an offer the recipient can evaluate. An offer id ALREADY ON FILE is
    // never refused by the caps (S-090 REG-G1): the trailing `OR EXISTS` makes
    // the row be attempted, the PRIMARY KEY aborts the whole batch, and the
    // gift_sent emit (whose EXISTS would match the earlier row) never runs.
    const span = outboxSpan(db);
    const spanRes = await db.batch([
      span.open,
      db
      .prepare(
        `INSERT INTO gift_offers (offer_id, kind, item_id, from_account, to_account, state, created_at, expires_at)
         SELECT ?, ?, ?, ?, ?, 'open', ?, ?
          WHERE ((SELECT COUNT(*) FROM gift_offers WHERE from_account = ? AND state = 'open' AND expires_at > ?) < ?
            AND (SELECT COUNT(*) FROM gift_offers WHERE from_account = ? AND created_at >= ?) < ?
            AND (SELECT COUNT(*) FROM gift_offers WHERE from_account = ? AND to_account = ? AND state = 'open' AND expires_at > ?) < ?
            AND (SELECT COUNT(*) FROM gift_offers WHERE to_account = ? AND state = 'open' AND expires_at > ?) < ?
            AND NOT EXISTS (SELECT 1 FROM gift_offers WHERE from_account = ? AND to_account = ? AND state = 'declined' AND resolved_at >= ?)
            AND EXISTS (SELECT 1 FROM user_profiles WHERE account_id = ? AND visibility = 'public')
            AND EXISTS (SELECT 1 FROM accounts WHERE account_id = ?))
             OR EXISTS (SELECT 1 FROM gift_offers WHERE offer_id = ?)
         RETURNING offer_id`,
      )
      .bind(offerId, kind, itemId, fromAccount, to, ts, expiresAt, fromAccount, ts, MAX_OPEN_OFFERS_PER_SENDER, fromAccount, dayStart, OFFERS_PER_SENDER_PER_DAY, fromAccount, to, ts, MAX_OPEN_OFFERS_PER_PAIR, to, ts, MAX_OPEN_INCOMING_PER_RECIPIENT, fromAccount, to, declineCutoff, fromAccount, to, offerId),
      giftEvent,
      span.close,
    ]);
    seqs = outboxSpanSeqs(spanRes[0], spanRes[spanRes.length - 1]);
    const res = spanRes.slice(1, -1);
    inserted = (res[0] as { results?: unknown[] } | undefined)?.results ?? [];
  } catch (e) {
    const msg = e instanceof Error ? e.message : String(e);
    if (/unique/i.test(msg)) throw new CollectibleError(REFUSAL, 409); // already on offer, or this offer id is already on file
    throw e;
  }
  // THIS call's insert decides (S-090 #313b): the capped INSERT's own
  // `RETURNING offer_id` row, enumerated (db.ts: enumerate, never count). A read
  // by offer id would answer success for a refused offer whose id is on file.
  if (inserted.length !== 1) throw new CollectibleError(REFUSAL, 409);
  if (pubEnv && pubCtx) publishAfterCommit(pubEnv, pubCtx, seqs);
  return { offer_id: offerId, kind, item_id: itemId, counterparty: String(input.to_handle).trim().toLowerCase(), direction: "outgoing", state: "open", created_at: ts, expires_at: expiresAt };
}

/**
 * QUEUE #172 / AS-112: void every OPEN offer on this item whose sender no
 * longer holds it. ONE statement for every door an item leaves its owner by —
 * `offerGift` (a new owner offering), `recoverCollectible`, and match
 * settlement (`war-match.ts` resolveAndSettle) — so the predicate is never
 * re-typed. Batched AFTER the move, it voids nothing when the move no-ops.
 */
export function voidOrphanedOffersStatement(db: DbLike, kind: CollectibleKind, itemId: string, ts: string): DbStatement {
  checkItem(kind, itemId);
  const held =
    kind === "card"
      ? "card_instances i WHERE i.instance_id = gift_offers.item_id AND i.owner_id = gift_offers.from_account AND i.consumed_at IS NULL"
      : "acquisitions i WHERE i.acquisition_id = gift_offers.item_id AND i.account_id = gift_offers.from_account";
  return db
    .prepare(`UPDATE gift_offers SET state = 'void', resolved_at = ? WHERE kind = ? AND item_id = ? AND state = 'open' AND NOT EXISTS (SELECT 1 FROM ${held})`)
    .bind(ts, kind, itemId);
}

/**
 * QUEUE #172 / D-0810 (11), D-0812(1): the recipient has quietly closed this
 * offer's pair (Ignore / Block, `gift_blocks` 0106) and the row is live, OR the
 * offer was made before the row's end (an offer made inside an Ignore is never
 * delivered, even after the Ignore ends). Correlated to the `gift_offers` row in
 * scope; one bind, the instant. Only RECIPIENT-side doors read it — the sender's
 * doors never do, so to them the offer stays pending.
 */
const PAIR_QUIET =
  "EXISTS (SELECT 1 FROM gift_blocks b WHERE b.account_id = gift_offers.to_account AND b.blocked_account = gift_offers.from_account AND (b.expires_at IS NULL OR b.expires_at > ? OR b.expires_at > gift_offers.created_at))";

interface OfferRow {
  offer_id: string;
  kind: CollectibleKind;
  item_id: string;
  from_account: string;
  to_account: string;
  state: OfferState;
  created_at: string;
  expires_at: string;
}

async function openOfferFor(db: DbLike, offerId: string, side: "to" | "from", account: string, nowMs: number): Promise<OfferRow> {
  if (!GFT_RE.test(offerId)) throw new CollectibleError("no such offer", 404);
  const row = await db
    .prepare(`SELECT offer_id, kind, item_id, from_account, to_account, state, created_at, expires_at, ${PAIR_QUIET} AS quiet FROM gift_offers WHERE offer_id = ? AND ${side === "to" ? "to_account" : "from_account"} = ?`)
    .bind(new Date(nowMs).toISOString(), offerId, account)
    .first<OfferRow & { quiet: number }>();
  // Not yours and does not exist read the same — an offer id is a capability
  // the two parties hold, not a public fact.
  if (!row) throw new CollectibleError("no such offer", 404);
  if (row.state !== "open") throw new CollectibleError(`this offer was ${row.state}`, 409);
  // The recipient already closed it quietly: it is theirs no longer to act on.
  // The SENDER's side never branches here — to them it is still pending.
  if (side === "to" && row.quiet) throw new CollectibleError("this offer was already resolved", 409);
  if (Date.parse(row.expires_at) <= nowMs) {
    await db.prepare("UPDATE gift_offers SET state = 'expired', resolved_at = ? WHERE offer_id = ? AND state = 'open'").bind(new Date(nowMs).toISOString(), offerId).run();
    throw new CollectibleError("this offer has expired", 409);
  }
  return row;
}

/**
 * The recipient's accept — THE consent. One batch: the transfer statements +
 * the offer flip predicated on the transfer's log row; a second UPDATE marks
 * the offer `void` if the item could not move (the sender staked, burned or
 * lost it meanwhile), so a dead offer never stays open.
 */
export async function acceptGift(db: DbLike, toAccount: string, offerId: string, logId: string, nowMs: number): Promise<{ kind: CollectibleKind; item_id: string }> {
  if (!ACCOUNT_RE.test(toAccount)) throw new CollectibleError("bad account id", 500);
  const offer = await openOfferFor(db, offerId, "to", toAccount, nowMs);
  const ts = new Date(nowMs).toISOString();
  const table = offer.kind === "card" ? "card_instance_log" : "acquisition_log";
  // ORDER IS THE FIX (F3 review, Concurrency / Scarcity): the offer is flipped
  // FIRST, predicated on being open and unexpired for THIS recipient, and the
  // transfer is predicated on that flip having happened (`resolved_at = ts`
  // is this call's mark). A cancel, decline or sweep that lands between the
  // read above and this batch makes the flip a no-op, which makes the move a
  // no-op — nothing moves under a resolved offer. If the flip landed but the
  // item could not move (staked / burned / lost meanwhile), the last statement
  // turns the accepted row into `void`, so a dead offer never reads accepted.
  const notHeld = openHoldGuard(offer.kind, offer.item_id, ts);
  const stillOpen = { sql: ` AND EXISTS (SELECT 1 FROM gift_offers WHERE offer_id = ? AND state = 'accepted' AND resolved_at = ?)${notHeld.sql}`, binds: [offerId, ts, ...notHeld.binds] };
  await db.batch([
    // F18 / #32 — A CLOSED SENDER'S COLLECTIBLE DOES NOT LEAVE THEIR SHELF.
    //
    // The recipient's session is checked; the SENDER's is not, because they are
    // not the caller. So an offer created before a suspension was still
    // acceptable afterwards, and a review moved a card out of a TERMINATED
    // account — against copy in both `TERMS.md §7.3` and `APPEALS.md` saying
    // collectibles "stay recorded as yours … non-transferable".
    //
    // In the FLIP's own WHERE, not as a read above it, for the reason the
    // ordering comment already gives: a suspension landing between a read and
    // this batch must make the flip a no-op, which makes the move a no-op.
    // `transferLanded` below then answers the recipient honestly.
    // I3-0345/I3-0346: the SENDER's restrictions are evaluated here,
    // independently of the recipient's permission to resolve the offer (that
    // permission is `openOfferFor` above) -- because the sender is not the
    // caller, they get neither check for free from the session fold the way
    // the recipient does (fan-auth.ts). Closure (suspended/terminated) was
    // the only half asked; a chargeback-banned sender (`accounts.banned_at`,
    // the same fact `accountBanned()` reads) was not, so a previously offered
    // gift retained outdated sender authority once its sender was banned
    // after the offer was created. Both halves of the required set are now
    // in the same guarded WHERE, atomic with the flip, for the reason the
    // ordering comment above already gives: a ban landing between a read and
    // this batch must make the flip a no-op.
    db
      .prepare(
        `UPDATE gift_offers SET state = 'accepted', resolved_at = ? WHERE offer_id = ? AND to_account = ? AND state = 'open' AND expires_at > ?
          AND NOT EXISTS (SELECT 1 FROM account_closures WHERE account_id = gift_offers.from_account AND state IN ${FROZEN_CLOSURE_STATES_SQL})
          AND NOT EXISTS (SELECT 1 FROM accounts WHERE account_id = gift_offers.from_account AND banned_at IS NOT NULL)
          AND NOT ${PAIR_QUIET}`,
      )
      .bind(ts, offerId, toAccount, ts, ts),
    ...transferCollectibleStatements(db, { kind: offer.kind, itemId: offer.item_id, fromOwner: offer.from_account, toOwner: toAccount, event: "gifted", detail: offer.offer_id, logId, ts, extraGuard: stillOpen }),
    // CO-10 — THE VOID ASKS ABOUT THE OFFER, NOT ABOUT THIS CALL'S LOG ROW.
    //
    // `resolved_at` is a millisecond, and it was doing duty as this operation's
    // ownership marker. Two overlapping accepts by the SAME recipient (one
    // double-tap; the same shape `gifting.test.ts:314` already tests for the
    // transfer) can read one open offer and compute the same `ts`. The winner
    // flips and moves the item. The loser's flip no-ops and its transfer
    // no-ops — correctly, the item is no longer the sender's — but the old
    // predicate then matched the WINNER's row (same offer, same `resolved_at`)
    // and found the loser's own fresh `logId` absent, so the loser marked the
    // COMPLETED gift `void`. Ownership and provenance said transferred; the
    // offer said void; and because the inbox selects only `open` and
    // `accepted`, the recipient's record of what they received vanished.
    //
    // The question this statement exists to ask is "did the item fail to move
    // under this offer", and the log row carries the offer id as its `detail`
    // (see `transferCollectibleStatements`), so ask that directly: void only
    // when NO `gifted` row exists for this offer at all. A losing call now
    // cannot void the winner's offer, and a genuinely dead offer (staked,
    // burned, lost meanwhile — no transfer landed from any call) still voids.
    db
      .prepare(`UPDATE gift_offers SET state = 'void' WHERE offer_id = ? AND state = 'accepted' AND resolved_at = ? AND NOT EXISTS (SELECT 1 FROM ${table} WHERE detail = ? AND event = 'gifted')`)
      .bind(offerId, ts, offerId),
  ]);
  if (!(await transferLanded(db, offer.kind, logId))) throw new CollectibleError("this offer is no longer valid — the item can't be moved", 409);
  return { kind: offer.kind, item_id: offer.item_id };
}

export async function declineGift(db: DbLike, toAccount: string, offerId: string, nowMs: number): Promise<void> {
  if (!ACCOUNT_RE.test(toAccount)) throw new CollectibleError("bad account id", 500);
  await openOfferFor(db, offerId, "to", toAccount, nowMs);
  const ts = new Date(nowMs).toISOString();
  const r = await db.prepare(`UPDATE gift_offers SET state = 'declined', resolved_at = ? WHERE offer_id = ? AND to_account = ? AND state = 'open' AND NOT ${PAIR_QUIET}`).bind(ts, offerId, toAccount, ts).run();
  if (r.changes === 0) throw new CollectibleError("this offer was already resolved", 409);
}

export async function cancelGift(db: DbLike, fromAccount: string, offerId: string, nowMs: number): Promise<void> {
  if (!ACCOUNT_RE.test(fromAccount)) throw new CollectibleError("bad account id", 500);
  await openOfferFor(db, offerId, "from", fromAccount, nowMs);
  // `changes === 0` = an accept (or a sweep) won the race: the item MOVED and
  // the SPA's "never left your shelf" must not be said. 409, not ok.
  const r = await db.prepare("UPDATE gift_offers SET state = 'cancelled', resolved_at = ? WHERE offer_id = ? AND from_account = ? AND state = 'open'").bind(new Date(nowMs).toISOString(), offerId, fromAccount).run();
  if (r.changes === 0) throw new CollectibleError("this offer was already resolved", 409);
}

/**
 * QUEUE #172 / O-12, D-0810 (11) — the recipient's QUIET exits. `ignoreGift`
 * closes the pair for DECLINE_COOLDOWN_MS; `blockGiftSender` closes it with no
 * end (`gift_blocks`, migration 0106). NO OFFER ROW CHANGES: the offer, and
 * every other open offer of the pair, stay `open` in `gift_offers`, so every
 * door the SENDER reads (listGifts outgoing, cancelGift, the per-item UNIQUE
 * and the caps, the export) shows it pending until its own `expires_at`, then
 * `expired` by the ordinary expiry — exactly what silence shows. Only the
 * recipient's doors read the block (`PAIR_QUIET`) and stop showing or acting
 * on it. ONE statement: the block upsert is predicated on THIS offer being
 * open and unexpired for this recipient, and its own RETURNING row is the
 * verdict (INV-058) — a call that answers 409 wrote nothing.
 */
export async function ignoreGift(db: DbLike, toAccount: string, offerId: string, nowMs: number): Promise<void> {
  await quietlyClose(db, toAccount, offerId, nowMs, new Date(nowMs + DECLINE_COOLDOWN_MS).toISOString());
}

export async function blockGiftSender(db: DbLike, toAccount: string, offerId: string, nowMs: number): Promise<void> {
  await quietlyClose(db, toAccount, offerId, nowMs, null);
}

async function quietlyClose(db: DbLike, toAccount: string, offerId: string, nowMs: number, blockUntil: string | null): Promise<void> {
  if (!ACCOUNT_RE.test(toAccount)) throw new CollectibleError("bad account id", 500);
  const offer = await openOfferFor(db, offerId, "to", toAccount, nowMs);
  const ts = new Date(nowMs).toISOString();
  // An indefinite block is never shortened by a later Ignore.
  const res = await db.batch([
    db
      .prepare(
        `INSERT INTO gift_blocks (account_id, blocked_account, created_at, expires_at)
         SELECT ?, ?, ?, ? WHERE EXISTS (SELECT 1 FROM gift_offers WHERE offer_id = ? AND to_account = ? AND from_account = ? AND state = 'open' AND expires_at > ?)
         ON CONFLICT(account_id, blocked_account) DO UPDATE SET created_at = excluded.created_at,
           expires_at = CASE WHEN gift_blocks.expires_at IS NULL OR excluded.expires_at IS NULL THEN NULL ELSE excluded.expires_at END
         RETURNING account_id`,
      )
      .bind(toAccount, offer.from_account, ts, blockUntil, offerId, toAccount, offer.from_account, ts),
  ]);
  const landed = (res[0] as { results?: unknown[] } | undefined)?.results ?? [];
  if (landed.length !== 1) throw new CollectibleError("this offer was already resolved", 409);
}

/**
 * Own offers. OUTGOING: every state (the sender's own record). INCOMING: only
 * `open` (to act on) and `accepted` (what you received) — a recipient is never
 * shown an offer that was cancelled, expired or voided without their act, so
 * offer→cancel churn leaves nothing behind in their inbox; an open offer they
 * quietly closed (Ignore / Block, `PAIR_QUIET`) is gone from it too, while the
 * sender's OUTGOING row stays exactly as silence leaves it. The counterparty's
 * handle is resolved ONLY for open offers: a resolved row is not a permanent
 * handle-tracking hook, and a rename cannot re-label history. Open incoming
 * offers carry no item id (see OfferView). Expired-on-read. Per-direction limit.
 */
export async function listGifts(db: DbLike, account: string, nowMs: number): Promise<{ incoming: OfferView[]; outgoing: OfferView[] }> {
  if (!ACCOUNT_RE.test(account)) throw new CollectibleError("bad account id", 500);
  const nowIso = new Date(nowMs).toISOString();
  const read = async (side: "to" | "from"): Promise<(OfferRow & { counterparty: string | null })[]> =>
    (
      await db
        .prepare(
          `SELECT o.offer_id, o.kind, o.item_id, o.from_account, o.to_account, o.state, o.created_at, o.expires_at,
                  CASE WHEN o.state = 'open' THEN (SELECT handle FROM user_profiles p WHERE p.account_id = ${side === "to" ? "o.from_account" : "o.to_account"} AND p.visibility = 'public') ELSE NULL END AS counterparty
             FROM gift_offers o
            WHERE o.${side === "to" ? "to_account" : "from_account"} = ?${side === "to" ? ` AND o.state IN ('open','accepted') AND NOT (o.state = 'open' AND ${PAIR_QUIET.replaceAll("gift_offers.", "o.")})` : ""}
            ORDER BY CASE WHEN o.state = 'open' THEN 0 ELSE 1 END, o.created_at DESC
            LIMIT ${LIST_LIMIT}`,
        )
        .bind(account, ...(side === "to" ? [nowIso] : []))
        .all<OfferRow & { counterparty: string | null }>()
    ).results;
  const view = (r: OfferRow & { counterparty: string | null }, direction: "incoming" | "outgoing"): OfferView => {
    const state: OfferState = r.state === "open" && r.expires_at <= nowIso ? "expired" : r.state;
    return { offer_id: r.offer_id, kind: r.kind, item_id: direction === "incoming" && state !== "accepted" ? null : r.item_id, counterparty: state === "open" ? r.counterparty : null, direction, state, created_at: r.created_at, expires_at: r.expires_at };
  };
  const [inc, out] = await Promise.all([read("to"), read("from")]);
  return { incoming: inc.map((r) => view(r, "incoming")).filter((v) => v.state !== "expired"), outgoing: out.map((r) => view(r, "outgoing")) };
}

export interface HistoryEntry {
  event: string;
  /** `you` / `another collector` / `creator` (a mint or redemption) — never an id. */
  from: "you" | "another collector" | "creator" | null;
  to: "you" | "another collector" | null;
  at: string;
}

/** OWNER-ONLY ownership history; counterparties are unnamed. */
export async function collectibleHistory(db: DbLike, owner: string, kind: CollectibleKind, itemId: string): Promise<HistoryEntry[] | null> {
  if (!ACCOUNT_RE.test(owner)) throw new CollectibleError("bad account id", 500);
  checkItem(kind, itemId);
  const who = (v: string | null): "you" | "another collector" | null => (v === null ? null : v === owner ? "you" : "another collector");
  if (kind === "card") {
    const owns = await db.prepare("SELECT 1 AS x FROM card_instances WHERE instance_id = ? AND owner_id = ?").bind(itemId, owner).first();
    if (!owns) return null;
    const { results } = await db
      .prepare(`SELECT event, from_owner, to_owner, at FROM card_instance_log WHERE instance_id = ? ORDER BY at ASC, log_id ASC LIMIT ${HISTORY_LIMIT}`)
      .bind(itemId)
      .all<{ event: string; from_owner: string | null; to_owner: string | null; at: string }>();
    return results.map((r) => ({ event: r.event, from: r.event === "minted" ? "creator" : who(r.from_owner), to: who(r.to_owner), at: r.at }));
  }
  const acq = await db.prepare("SELECT acquired_at FROM acquisitions WHERE acquisition_id = ? AND account_id = ?").bind(itemId, owner).first<{ acquired_at: string }>();
  if (!acq) return null;
  const { results } = await db
    .prepare(`SELECT event, from_owner, to_owner, at FROM acquisition_log WHERE acquisition_id = ? ORDER BY at ASC, log_id ASC LIMIT ${HISTORY_LIMIT}`)
    .bind(itemId)
    .all<{ event: string; from_owner: string; to_owner: string; at: string }>();
  const first: HistoryEntry = { event: "redeemed", from: "creator", to: results.length === 0 ? "you" : who(results[0]!.from_owner), at: acq.acquired_at };
  return [first, ...results.map((r) => ({ event: r.event, from: who(r.from_owner), to: who(r.to_owner), at: r.at }))];
}

/**
 * Operator recovery — the one unaccepted move, with an audit line in the batch.
 *
 * ESCROW-BREAK (F4b, Jason 2026-09-03). A card staked in a WAITING war-queue
 * entry is escrowed, and `transferInstanceStatements` refuses to move an
 * escrowed card — so a taker could park a stolen card in the queue and the
 * recovery would report `recovered:false` forever. With `break_escrow` the
 * batch first cancels that waiting entry (exactly `leaveQueue`'s two
 * statements, predicated on the entry still waiting), which frees the stake,
 * and the transfer then proceeds. The operators sign for the break explicitly
 * (it is in the dual-control key): it cancels a queue entry someone else
 * joined. A MATCHED entry (settle in flight) is never touched — the move
 * simply does not land, as before. Any open HOLD on the item is closed by the
 * landed move (predicated on this call's log row).
 *
 * THE BREAK AND THE MOVE LAND TOGETHER OR NOT AT ALL (F4b review — all four
 * personas): the cancel carries the move's own preconditions (the destination
 * exists and has shelf room), so a dead or full destination cancels nobody's
 * queue entry; the destination is also checked before the batch (404) so the
 * console refuses it before a sign-off is banked. The `escrow_break` audit
 * line is predicated on the CANCEL (this call's `updated_at`), never on the
 * move, and `escrow_broken` is read from that same fact — the trail records
 * exactly what happened to the queue entry.
 */
export async function recoverCollectible(
  db: DbLike,
  operator: string,
  input: { kind: CollectibleKind; item_id: string; to_account: string; ref: string; break_escrow?: boolean },
  logId: string,
  now: () => string = () => new Date().toISOString(),
): Promise<{ recovered: boolean; escrow_broken: boolean }> {
  checkItem(input.kind, input.item_id);
  if (!ACCOUNT_RE.test(input.to_account)) throw new CollectibleError("bad account id");
  if (typeof input.ref !== "string" || !RECOVERY_REF_RE.test(input.ref)) throw new CollectibleError("bad ref");
  if (input.break_escrow !== undefined && typeof input.break_escrow !== "boolean") throw new CollectibleError("break_escrow must be a boolean");
  const current =
    input.kind === "card"
      ? await db.prepare("SELECT owner_id AS o, escrowed_by AS e FROM card_instances WHERE instance_id = ? AND consumed_at IS NULL").bind(input.item_id).first<{ o: string; e: string | null }>()
      : await db.prepare("SELECT account_id AS o, NULL AS e FROM acquisitions WHERE acquisition_id = ?").bind(input.item_id).first<{ o: string; e: string | null }>();
  if (!current) throw new CollectibleError("no such item", 404);
  if (current.o === input.to_account) return { recovered: false, escrow_broken: false };
  if ((await db.prepare("SELECT 1 AS x FROM accounts WHERE account_id = ?").bind(input.to_account).first()) === null) throw new CollectibleError("no such account", 404);
  const ts = now();
  const table = input.kind === "card" ? "card_instance_log" : "acquisition_log";
  const breaking = input.kind === "card" && input.break_escrow === true && current.e !== null && WQ_RE.test(current.e);
  // I3-1805 — THE BREAK MUST REACH THE ESCROW CLASS NOTHING ELSE CAN.
  //
  // Three classes of stake, not two. `waiting` is cancelled below. `matched` is
  // never touched (settle in flight), as before. The THIRD is an ante escrowed
  // to an entry that is neither: no `war_queue` row at all, or one already
  // cancelled or voided whose paired release no-op'd. That class is the one no
  // queue transition will ever free -- there is no row left to transition --
  // and no queue path frees it (QUEUE #236 removed `joinQueue`'s orphan
  // release), so an operator cannot make a takeover taker call one. The cron
  // leg `sweepOrphanEscrow` (war-instances.ts, I10-FX5; widened by I10-FX13,
  // D-0873(1)) frees this whole class -- row-less, and a row that survives as
  // cancelled or voided -- but only on its next scheduled run. So without this
  // arm the signed break reports `recovered:false` on exactly the stake it
  // exists to free, until that run.
  //
  // The class is decided by a read, but the GUARD is in the statement, for the
  // reason the accept's ordering comment gives: an entry that becomes waiting
  // or matched between this read and the batch makes the release a no-op. The
  // release carries the same two move preconditions the cancel does (the
  // destination exists and has shelf room), so a dead or full destination frees
  // nobody's stake either -- the F4b rule that the break and the move land
  // together or not at all.
  const entryState = breaking ? ((await db.prepare("SELECT state FROM war_queue WHERE entry_id = ?").bind(current.e).first<{ state: string }>())?.state ?? null) : null;
  const releasing = breaking && entryState !== "waiting" && entryState !== "matched";
  const escrowBreak: DbStatement[] = !breaking
    ? []
    : releasing
      ? [
          db
            .prepare(
              `UPDATE card_instances SET escrowed_by = NULL
                WHERE instance_id = ? AND escrowed_by = ?
                  AND NOT EXISTS (SELECT 1 FROM war_queue WHERE entry_id = ? AND state IN ('waiting','matched'))
                  AND EXISTS (SELECT 1 FROM accounts WHERE account_id = ?)
                  AND (SELECT COUNT(*) FROM card_instances d WHERE d.owner_id = ? AND d.consumed_at IS NULL) < ?`,
            )
            .bind(input.item_id, current.e, current.e, input.to_account, input.to_account, MAX_INSTANCES_PER_ACCOUNT),
        ]
      : [
          db
            .prepare(
              `UPDATE war_queue SET state = 'cancelled', updated_at = ?
              WHERE entry_id = ? AND state = 'waiting'
                AND EXISTS (SELECT 1 FROM accounts WHERE account_id = ?)
                AND (SELECT COUNT(*) FROM card_instances d WHERE d.owner_id = ? AND d.consumed_at IS NULL) < ?`,
            )
            .bind(ts, current.e, input.to_account, input.to_account, MAX_INSTANCES_PER_ACCOUNT),
          db
            .prepare("UPDATE card_instances SET escrowed_by = NULL WHERE escrowed_by = ? AND EXISTS (SELECT 1 FROM war_queue WHERE entry_id = ? AND state = 'cancelled' AND updated_at = ?)")
            .bind(current.e, current.e, ts),
        ];
  const cancelLanded = "EXISTS (SELECT 1 FROM war_queue WHERE entry_id = ? AND state = 'cancelled' AND updated_at = ?)";
  // Each arm reports its OWN landed fact, never the move's: the cancel arm the
  // cancelled entry, the release arm the stake actually being gone (`breaking`
  // established it was present before the batch). `escrow_broken` and the
  // audit line read the same predicate, so the trail records what happened to
  // the escrow and nothing else.
  const releaseLanded = "NOT EXISTS (SELECT 1 FROM card_instances WHERE instance_id = ? AND escrowed_by = ?)";
  const brokeSql = releasing ? releaseLanded : cancelLanded;
  const brokeBinds: unknown[] = releasing ? [input.item_id, current.e] : [current.e, ts];
  await db.batch([
    ...escrowBreak,
    ...transferCollectibleStatements(db, { kind: input.kind, itemId: input.item_id, fromOwner: current.o, toOwner: input.to_account, event: "recovered", detail: input.ref, logId, ts }),
    // QUEUE #172: an offer the taker left open on the item dies with the move.
    voidOrphanedOffersStatement(db, input.kind, input.item_id, ts),
    // A recovered card must not stay staked in a queue: transferInstanceStatements
    // refuses an escrowed card, so a staked one simply does not move (unless
    // the escrow was broken above) — and the audit line is written ONLY when
    // the move landed (predicated on this call's log row), so the operator
    // trail never claims a move that did not happen. The route reports
    // `recovered: false` either way.
    // D-0813(3): the destination ACCOUNT stays out of the audit detail (erasure
    // rewrites item_id, never detail); the transfer row above records it.
    auditStmtIf(db, operator, "collectible:recover", `${input.kind} ${input.item_id} -> account-withheld (${input.ref})`, ts, `EXISTS (SELECT 1 FROM ${table} WHERE log_id = ?)`, [logId]),
    ...(breaking
      ? [auditStmtIf(db, operator, "collectible:escrow_break", `${input.item_id} left war-queue entry ${current.e} for recovery (${input.ref})`, ts, brokeSql, brokeBinds)]
      : []),
    // The landed move closes the case's hold on this item.
    db
      .prepare(
        `UPDATE collectible_holds SET state = 'closed', closed_at = ?, closed_by = 'recovery', closed_reason = ?
          WHERE kind = ? AND item_id = ? AND state = 'open' AND EXISTS (SELECT 1 FROM ${table} WHERE log_id = ?)`,
      )
      .bind(ts, input.ref, input.kind, input.item_id, logId),
  ]);
  const recovered = await transferLanded(db, input.kind, logId);
  const broken = breaking && (await db.prepare(`SELECT 1 AS x WHERE ${brokeSql}`).bind(...brokeBinds).first()) !== null;
  return { recovered, escrow_broken: broken };
}

const WQ_RE = /^wq_[0-9a-f]{20}$/;

// ---------------------------------------------------------------------------
// RECOVERY HOLDS (F4b). "A taker who erases destroys the item": erasure deletes
// the erased fan's shelf, so a stolen card or acquisition vanished with the
// thief's account before the operator could return it. A hold is the
// operator's record that an ITEM is the subject of an open recovery case; while
// it is open, `deleteAccount` refuses (409) for whichever account holds the
// item — the hold follows the item, not the person. A hold is never
// indefinite: it carries an expiry, the recovery closes it when the move lands,
// an operator can close it, and the sweep expires it.
// ---------------------------------------------------------------------------

export const HOLD_TTL_MS = 30 * 24 * 60 * 60 * 1000;
export const HOLD_MAX_TTL_MS = 90 * 24 * 60 * 60 * 1000;
/** The deferral is bounded per ITEM and per ACCOUNT (I3-0878, see `openHold`), not just per hold (F4b review, Sybil
 * Farmer): holds on one item may not add up to more than `HOLD_MAX_TTL_MS` of
 * scheduled time inside any `HOLD_LOOKBACK_MS` window, so chaining holds
 * cannot make "never indefinite" false. PRIVACY §7.1 states the same bound. */
export const HOLD_LOOKBACK_MS = 365 * 24 * 60 * 60 * 1000;
/** Rows `openHold`'s per-account bound reads before it refuses (I3-0878). */
const ACCOUNT_HOLD_SCAN = 1000;
/** Total length of the union of [start, end) spans (ms). Unparseable spans
 * count as a full HOLD_MAX_TTL_MS, so a bad row can only make the bound stricter. */
function heldUnionMs(spans: [number, number][]): number {
  const s = spans.map(([a, b]) => (Number.isFinite(a) && Number.isFinite(b) ? [a, Math.max(a, b)] : [0, HOLD_MAX_TTL_MS + 1])).sort((x, y) => x[0]! - y[0]!);
  let total = 0;
  let curA = -Infinity;
  let curB = -Infinity;
  for (const [a, b] of s as [number, number][]) {
    if (a > curB) {
      if (curB > curA) total += curB - curA;
      curA = a;
      curB = b;
    } else if (b > curB) curB = b;
  }
  if (curB > curA) total += curB - curA;
  return total;
}
const CHD_RE = /^chd_[0-9a-f]{20}$/;
const OP_RE = /^op_[0-9a-f]{20}$/;

export interface HoldView {
  hold_id: string;
  kind: CollectibleKind;
  item_id: string;
  ref: string;
  state: "open" | "closed" | "expired";
  opened_by: string;
  opened_at: string;
  expires_at: string;
  closed_at: string | null;
  closed_by: string | null;
  closed_reason: string | null;
  /** Whether the item still exists (a consumed card / erased shelf reads false). */
  item_present: boolean;
}

/** Open a hold on an item. One open hold per item (409 on a second). */
export async function openHold(
  db: DbLike,
  operator: string,
  input: { kind: CollectibleKind; item_id: string; ref: string; ttl_ms?: number },
  holdId: string,
  nowMs: number,
): Promise<HoldView> {
  checkItem(input.kind, input.item_id);
  if (typeof input.ref !== "string" || !RECOVERY_REF_RE.test(input.ref)) throw new CollectibleError("bad ref");
  if (!OP_RE.test(operator)) throw new CollectibleError("bad operator", 500);
  if (!CHD_RE.test(holdId)) throw new CollectibleError("bad hold id", 500);
  const ttl = input.ttl_ms === undefined ? HOLD_TTL_MS : input.ttl_ms;
  if (!Number.isInteger(ttl) || ttl < 60_000 || ttl > HOLD_MAX_TTL_MS) throw new CollectibleError("ttl_ms out of range");
  const present =
    input.kind === "card"
      ? await db.prepare("SELECT owner_id AS holder FROM card_instances WHERE instance_id = ? AND consumed_at IS NULL").bind(input.item_id).first<{ holder: string }>()
      : await db.prepare("SELECT account_id AS holder FROM acquisitions WHERE acquisition_id = ?").bind(input.item_id).first<{ holder: string }>();
  if (!present) throw new CollectibleError("no such item", 404);
  const ts = new Date(nowMs).toISOString();
  const exp = new Date(nowMs + ttl).toISOString();
  // An expired hold the sweep has not reached yet must not hold the slot (it
  // defers nothing — erasure reads `expires_at` live).
  await db.prepare("UPDATE collectible_holds SET state = 'expired', closed_at = ?, closed_by = 'sweep' WHERE kind = ? AND item_id = ? AND state = 'open' AND expires_at <= ?").bind(ts, input.kind, input.item_id, ts).run();
  // Cumulative bound per item: the scheduled time of every hold opened in the
  // lookback window, plus this one, may not exceed HOLD_MAX_TTL_MS.
  const since = new Date(nowMs - HOLD_LOOKBACK_MS).toISOString();
  const used = await db
    .prepare("SELECT COALESCE(SUM((julianday(expires_at) - julianday(opened_at)) * 86400000.0), 0) AS ms FROM collectible_holds WHERE kind = ? AND item_id = ? AND opened_at >= ?")
    .bind(input.kind, input.item_id, since)
    .first<{ ms: number }>();
  if ((used?.ms ?? 0) + ttl > HOLD_MAX_TTL_MS + 1000) throw new CollectibleError("holds on this item would exceed the 90-day bound", 409);
  // I3-0878: the bound is ALSO per ACCOUNT, because the erasure gate
  // (ERASURE_HOLD_SQL) fires per account — any open hold on any item the account
  // holds defers its erasure. Per item alone, a fan with K items could be held for
  // K x 90 days by moving the next hold to the next item. So the time the
  // holder's erasure is deferred — the UNION of the hold intervals on items they
  // hold, opened inside the lookback window, each ending at its expiry or its
  // earlier close, plus this new one — may not exceed HOLD_MAX_TTL_MS.
  // Concurrent holds on several items overlap and cost the budget once; chaining
  // them does not reset it. Over ACCOUNT_HOLD_SCAN rows it refuses rather than
  // conclude anything about the rows it did not read (fail closed).
  // D-1073(4) (I21-Q133): an item the account GAVE AWAY after a hold on it opened still counts — every ownership move
  // writes its provenance row in the same batch (`card_instance_log` war-instances.ts transfer, `acquisition_log`
  // gifting.ts move), so `from_owner = holder AND at >= opened_at` finds it. Without this, giving an item away freed the
  // budget and a second 90-day hold opened inside the year.
  const onAccount = await db
    .prepare(
      `SELECT h.opened_at AS opened_at, CASE WHEN h.closed_at IS NOT NULL AND h.closed_at < h.expires_at THEN h.closed_at ELSE h.expires_at END AS ends_at
         FROM collectible_holds h
        WHERE h.opened_at >= ?
          AND ((h.kind = 'card' AND (EXISTS (SELECT 1 FROM card_instances ci WHERE ci.instance_id = h.item_id AND ci.owner_id = ?)
                OR EXISTS (SELECT 1 FROM card_instance_log cl WHERE cl.instance_id = h.item_id AND cl.from_owner = ? AND cl.at >= h.opened_at)))
            OR (h.kind = 'acquisition' AND (EXISTS (SELECT 1 FROM acquisitions a WHERE a.acquisition_id = h.item_id AND a.account_id = ?)
                OR EXISTS (SELECT 1 FROM acquisition_log al WHERE al.acquisition_id = h.item_id AND al.from_owner = ? AND al.at >= h.opened_at))))
        LIMIT ?`,
    )
    .bind(since, present.holder, present.holder, present.holder, present.holder, ACCOUNT_HOLD_SCAN + 1)
    .all<{ opened_at: string; ends_at: string }>();
  if (onAccount.results.length > ACCOUNT_HOLD_SCAN) throw new CollectibleError("too many holds on this account's items to bound its erasure deferral", 409);
  const spans = onAccount.results.map((r) => [Date.parse(r.opened_at), Date.parse(r.ends_at)] as [number, number]);
  spans.push([nowMs, nowMs + ttl]);
  if (heldUnionMs(spans) > HOLD_MAX_TTL_MS + 1000) {
    throw new CollectibleError("holds on this account's items would defer its erasure past the 90-day bound", 409);
  }
  try {
    await db.batch([
      db
        .prepare("INSERT INTO collectible_holds (hold_id, kind, item_id, ref, state, opened_by, opened_at, expires_at) VALUES (?, ?, ?, ?, 'open', ?, ?, ?)")
        .bind(holdId, input.kind, input.item_id, input.ref, operator, ts, exp),
      auditStmtIf(db, operator, "collectible:hold", `${input.kind} ${input.item_id} held until ${exp} (${input.ref})`, ts, "EXISTS (SELECT 1 FROM collectible_holds WHERE hold_id = ?)", [holdId]),
    ]);
  } catch (e) {
    if (/unique/i.test(e instanceof Error ? e.message : String(e))) throw new CollectibleError("an open hold already names this item", 409);
    throw e;
  }
  const v = await readHold(db, holdId);
  if (!v) throw new CollectibleError("hold did not land", 500);
  return v;
}

/** Close an open hold (operator). 409 when it is not open. */
export async function closeHold(db: DbLike, operator: string, holdId: string, reason: string | null, nowMs: number): Promise<HoldView> {
  if (!CHD_RE.test(holdId)) throw new CollectibleError("bad hold id");
  if (!OP_RE.test(operator)) throw new CollectibleError("bad operator", 500);
  if (reason !== null && (typeof reason !== "string" || reason.length > 200)) throw new CollectibleError("reason must be ≤ 200 chars");
  const ts = new Date(nowMs).toISOString();
  // Flip + audit in ONE batch, the audit predicated on this call's flip.
  await db.batch([
    db
      .prepare("UPDATE collectible_holds SET state = 'closed', closed_at = ?, closed_by = ?, closed_reason = ? WHERE hold_id = ? AND state = 'open'")
      .bind(ts, operator, reason, holdId),
    auditStmtIf(db, operator, "collectible:hold_close", `${holdId}${reason ? " (" + reason + ")" : ""}`, ts, "EXISTS (SELECT 1 FROM collectible_holds WHERE hold_id = ? AND state = 'closed' AND closed_at = ? AND closed_by = ?)", [holdId, ts, operator]),
  ]);
  const v = await readHold(db, holdId);
  if (!v) throw new CollectibleError("hold vanished", 500);
  if (v.state !== "closed" || v.closed_at !== ts) throw new CollectibleError("hold is not open", 409);
  return v;
}

export async function listHolds(db: DbLike, state: "open" | "all" = "open"): Promise<HoldView[]> {
  const { results } = await db
    .prepare(`${HOLD_SELECT} ${state === "open" ? "WHERE h.state = 'open'" : ""} ORDER BY h.opened_at DESC LIMIT 200`)
    .bind()
    .all<HoldRow>();
  return results.map(viewHold);
}

/** Cron: an open hold past its expiry becomes `expired` (erasure reads
 * `expires_at` too, so a missed sweep never extends a hold). */
export async function sweepExpiredHolds(db: DbLike, nowMs: number): Promise<number> {

/**
 * The erasure gate, two halves. `holdBlockingErasure` is the friendly read
 * (`deleteAccount` turns it into a 409 that says when the hold lifts);
 * `erasureHoldValidator` is the in-batch statement that ABORTS the erasure
 * batch when a live hold names an item this account holds at commit time — a
 * NOT NULL violation on purpose, the repo's force-rollback shape — so a hold
 * opened between the read and the batch still wins.
 */
export const ERASURE_HOLD_SQL = `EXISTS (
  SELECT 1 FROM collectible_holds h
   WHERE h.state = 'open' AND h.expires_at > ?
     AND ((h.kind = 'card' AND EXISTS (SELECT 1 FROM card_instances ci WHERE ci.instance_id = h.item_id AND ci.owner_id = ? AND ci.consumed_at IS NULL))
       OR (h.kind = 'acquisition' AND EXISTS (SELECT 1 FROM acquisitions a WHERE a.acquisition_id = h.item_id AND a.account_id = ?))))`;

export async function holdBlockingErasure(db: DbLike, accountId: string, nowMs: number): Promise<{ expires_at: string } | null> {
  const nowIso = new Date(nowMs).toISOString();
  return db
    .prepare(
      `SELECT h.expires_at AS expires_at FROM collectible_holds h
        WHERE h.state = 'open' AND h.expires_at > ?
          AND ((h.kind = 'card' AND EXISTS (SELECT 1 FROM card_instances ci WHERE ci.instance_id = h.item_id AND ci.owner_id = ? AND ci.consumed_at IS NULL))
            OR (h.kind = 'acquisition' AND EXISTS (SELECT 1 FROM acquisitions a WHERE a.acquisition_id = h.item_id AND a.account_id = ?)))
        ORDER BY h.expires_at DESC LIMIT 1`,
    )
    .bind(nowIso, accountId, accountId)
    .first<{ expires_at: string }>();
}

export function erasureHoldValidator(db: DbLike, accountId: string, nowMs: number): DbStatement {
  const nowIso = new Date(nowMs).toISOString();
  // `hold_id` is NOT NULL: the SELECT yields a NULL id only when a live hold
  // names an item this account holds, and that violation rolls the batch back.
  return db
    .prepare(`INSERT INTO collectible_holds (hold_id, kind, item_id, ref, state, opened_by, opened_at, expires_at) SELECT NULL, 'card', '', '', 'open', '', '', '' WHERE ${ERASURE_HOLD_SQL}`)
    .bind(nowIso, accountId, accountId);
}

interface HoldRow {
  hold_id: string;
  kind: CollectibleKind;
  item_id: string;
  ref: string;
  state: "open" | "closed" | "expired";
  opened_by: string;
  opened_at: string;
  expires_at: string;
  closed_at: string | null;
  closed_by: string | null;
  closed_reason: string | null;
  present: number;
}
const HOLD_SELECT = `SELECT h.hold_id, h.kind, h.item_id, h.ref, h.state, h.opened_by, h.opened_at, h.expires_at, h.closed_at, h.closed_by, h.closed_reason,
  CASE WHEN h.kind = 'card' THEN EXISTS (SELECT 1 FROM card_instances ci WHERE ci.instance_id = h.item_id AND ci.consumed_at IS NULL)
       ELSE EXISTS (SELECT 1 FROM acquisitions a WHERE a.acquisition_id = h.item_id) END AS present
  FROM collectible_holds h`;
function viewHold(r: HoldRow): HoldView {
  return { hold_id: r.hold_id, kind: r.kind, item_id: r.item_id, ref: r.ref, state: r.state, opened_by: r.opened_by, opened_at: r.opened_at, expires_at: r.expires_at, closed_at: r.closed_at, closed_by: r.closed_by, closed_reason: r.closed_reason, item_present: r.present === 1 };
}
async function readHold(db: DbLike, holdId: string): Promise<HoldView | null> {
  const r = await db.prepare(`${HOLD_SELECT} WHERE h.hold_id = ?`).bind(holdId).first<HoldRow>();
  return r ? viewHold(r) : null;
}

/** Offer CREATES per fan per hour — bounds the cheap refused work (unknown
 * handles, items not yours); the real caps live inside the INSERT. */
const GIFT_OFFERS_PER_HOUR = 30;

/**
 * F3 — CONSENTED GIFTING AND OWNERSHIP HISTORY (fan half).
 *
 * BOUNDARY. Seven fan-session routes over `collectibles.ts`: write an offer,
 * list own offers, accept / decline / ignore / block (recipient), cancel
 * (sender), and the
 * owner-only history of one item. No money moves here and nothing is public:
 * an offer id is a capability its two parties hold; the history names nobody.
 */
export function makeGiftsRoutes(): Hono<{ Bindings: Env }> {
  const app = new Hono<{ Bindings: Env }>();

  const mapErr = (c: { json: (b: unknown, s: 400 | 403 | 404 | 409 | 500) => Response }, e: unknown): Response => {
    if (e instanceof CollectibleError) return c.json({ error: e.message }, e.status);
    if (e instanceof WarInstanceError) return c.json({ error: e.message }, e.status as 400 | 409 | 500);
    throw e;
  };

  // Offer one of your collectibles to a PUBLIC handle. The recipient accepts.
  app.post("/account/gifts", anonBodyLimit, async (c) => {
    const id = await fanId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const body = (await c.req.json().catch(() => null)) as { kind?: unknown; item_id?: unknown; to_handle?: unknown } | null;
    if (!body || typeof body.kind !== "string" || typeof body.item_id !== "string" || typeof body.to_handle !== "string") return c.json({ error: "kind, item_id and to_handle required" }, 400);
    const db = d1(c.env.TRUST_DB);
    // A chargeback-banned fan cannot OFFER (moving disputed inventory to a
    // clean alt is the ban-evasion the review named); accept/decline/cancel
    // stay open so a victim can still resolve what was sent to them.
    if (await accountBanned(db, id)) return c.json({ error: "account_suspended" }, 403);
    if (!(await consumeRateLimit(db, `gift:${id}`, Date.now(), GIFT_OFFERS_PER_HOUR, HOUR_MS))) return c.json(refusal("rate limit exceeded"), 429);
    try {
      return c.json({ ok: true, offer: await offerGift(db, id, { kind: body.kind, item_id: body.item_id, to_handle: body.to_handle }, mintId("gft", 32), Date.now(), ...analyticsPublishArgs(c)) });
    } catch (e) {
      return mapErr(c, e);
    }
  });

  app.get("/account/gifts", async (c) => {
    const id = await fanId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    try {
      return c.json(await listGifts(d1(c.env.TRUST_DB), id, Date.now()));
    } catch (e) {
      return mapErr(c, e);
    }
  });

  app.post("/account/gifts/:offer_id/accept", async (c) => {
    const id = await fanId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const db = d1(c.env.TRUST_DB);
    try {
      // The log id's prefix follows the KIND, which we learn from the offer —

/**
 * HANDSHAKE — a fan with standing asks a structured question; the creator
 * answers with their own passkey, or does not answer at all.
 * F7, 2026-09-04 (master list #43).
 *
 * ---------------------------------------------------------------------------
 * THE ONE IDEA, AND EVERY LOCK IT KEEPS
 * ---------------------------------------------------------------------------
 * A HANDSHAKE NEVER INTRODUCES A FACT THE PASSPORT DOES NOT ALREADY CARRY. It
 * re-attests, freshly and under the creator's own passkey, something the fan
 * can already verify — or it produces nothing at all.
 *
 * That single rule is what lets a FAN-INITIATED channel exist inside every lock
 * this product has:
 *
 *  - NOT A SECOND OWNERSHIP CHANNEL. `identity` refuses to build a draft unless
 *    the account key is ALREADY an active, unshielded row on that passport. The
 *    proof model (bio-code / OAuth / operator) stays the only way an account
 *    gets onto a passport; a Handshake can only say "yes, that one is mine,
 *    signed, today" about a row a fan can already see.
 *  - NEVER A NEW PLACE TO SEND MONEY. `payment` refuses unless the destination
 *    matches a currently-RENDERING payout tag (`readLivePayoutTags`, which
 *    carries the currently-verified gate inside its own statement). A creator
 *    cannot sign a destination that is not already published on their passport,
 *    so the highest-value scam this feature could otherwise enable — getting a
 *    creator to sign off on the attacker's wallet — is structurally unreachable
 *    rather than policed.
 *  - NEVER NAMES A FAKE. A `declined` answer is PRIVATE to the asking fan and
 *    non-accusatory: it says the creator did not confirm, never that anything
 *    is an impersonation. `DIRECTION.md`'s PSA lock ("points to the REAL
 *    account only — never names the fake") holds because impersonation goes to
 *    the report path, which has operator review and due process. This is not an
 *    accusation channel and must not grow into one.
 *  - NOT A VERDICT INPUT. `handshake_answered` is derive-from-log, exactly like
 *    `recognition_granted`: no fold effect, no registry answer, nothing in the
 *    check path. This module is not in the verdict closure and must not enter
 *    it.
 *
 * ---------------------------------------------------------------------------
 * INSIDE THE S100 LOCK
 * ---------------------------------------------------------------------------
 * `DIRECTION.md:441`: *the client never chooses what is signed*. The draft is
 * built HERE, server-side, from the STORED handshake row — never from anything
 * the answering client sends. `buildAnswerDraft` takes a handshake id and a
 * disposition and reads everything else out of the database, so the two legs of
 * the ceremony cannot disagree and a hostile client cannot substitute a
 * different scope, a different subject or somebody else's handshake.
 *
 * And the fan cannot influence the payload AT ALL: it is
 * `{handshake_id, scope, disposition}`. The note, the account key, the payout
 * string and the fan's own `acc_` id stay in the row, which erasure can reach.
 * That is the S90 window-item-1 decision made on day one instead of after the
 * incident.
 *
 * ---------------------------------------------------------------------------
 * NON-RESPONSE IS THE DEFAULT, AND IT MEANS NOTHING

export function newHandshakeId(): string {
  return `hsk_${crypto.randomUUID().replace(/-/g, "").slice(0, 16)}`;
}

/** Scopes that admit a fan note. `identity` and `payment` say everything in
 * their structured field; free text there would be a channel with no purpose. */
export const NOTE_SCOPES: ReadonlySet<string> = new Set<string>(["offer", "call"]);

/**
 * STANDING. The fan must clear the seven-day floor AND any one of: an active
 * save (follow), a Bond at level 1+, a Founding Fan slot or the Early Scout
 * badge for this creator, or a Supporter tier.
 *
 * Expressed as an `eligibility.ts` RULE rather than as four hand-written
 * queries, so there is one predicate language in the product and a creator can
 * be given "raise the bar on my own handshakes" later without new evaluation
 * code. `account_age` carries the floor, so the whole gate is one rule.
 */
export const HANDSHAKE_STANDING: EligibilityRule = {
  kind: "all",
  rules: [
    { kind: "account_age", min_days: FOUNDING_FAN_MIN_AGE_DAYS },
    {
      kind: "any",
      rules: [{ kind: "save" }, { kind: "bond", min_level: 1 }, { kind: "founding_fan" }, { kind: "early_scout" }, { kind: "supporter", tier: "active" }],
    },
  ],
};

/**
 * Is this creator accepting handshakes at all?
 *
 * THE DEFAULT IS DERIVED, NOT STORED: a passport with no settings row accepts
 * iff it is VERIFIED. An unclaimed or claimed passport has nobody who can sign,
 * so accepting the question would only ever produce silence — and a fan who
 * asked and heard nothing would read that silence as an answer, which is
 * exactly what this feature promises never to manufacture. The table therefore
 * only ever holds a deliberate change.
 */
export async function acceptingHandshakes(db: DbLike, passportId: string): Promise<boolean> {
  const row = await db
    .prepare(
      `SELECT p.claim_state AS claim_state, (SELECT accepting FROM handshake_settings h WHERE h.passport_id = p.passport_id) AS accepting
         FROM passports p WHERE p.passport_id = ? AND p.retracted = 0`,
    )
    .bind(passportId)
    .first<{ claim_state: string; accepting: number | null }>();
  if (!row) return false;
  if (row.claim_state !== "verified") return false;
  return row.accepting === null ? true : row.accepting === 1;
}

/**
 * The creator's own switch. Idempotent.
 *
 * I3-1094 / I3-1264: switching OFF also lapses every handshake the switch had
 * already admitted, in the SAME batch — the off switch is a block on everyone,
 * and a handshake admitted under the old setting must not stay answerable
 * after it. `expired` is the no-signal state (no event, no notice). Switching
 * back ON revives nothing.
 */
export async function setAcceptingHandshakes(db: DbLike, passportId: string, accepting: boolean, now: () => string = () => new Date().toISOString()): Promise<void> {
  const upsert = db
    .prepare(
      `INSERT INTO handshake_settings (passport_id, accepting, updated_at) VALUES (?, ?, ?)
       ON CONFLICT(passport_id) DO UPDATE SET accepting = excluded.accepting, updated_at = excluded.updated_at`,
    )
    .bind(passportId, accepting ? 1 : 0, now());
  if (accepting) {
    await upsert.run();
    return;
  }
  await db.batch([upsert, db.prepare("UPDATE handshakes SET state = 'expired' WHERE passport_id = ? AND state = 'open'").bind(passportId)]);
}

/**
 * Block the fan behind ONE handshake, named by the handshake — never by their
 * account id.
 *
 * THIS IS THE ONLY BLOCK DOOR A CREATOR HAS, and it is shaped this way because
 * the first version was unusable: `InboxItem` deliberately carries only the
 * fan's PUBLIC handle and never their `acc_` id (a creator has no business
 * learning the internal identifier of somebody who asked them a question), so a
 * block route keyed on `account_id` had no value the console could ever supply.
 * The button existed and could not fire, which left a harassed creator with one
 * working remedy — turning the whole channel off — and that silences every
 * legitimate fan to stop one.
 *
 * A handshake id IS the capability: the creator holds it because the question
 * was addressed to them, this resolves it against their own passport, and the
 * fan's id never leaves the server.
 */
export async function blockFanByHandshake(db: DbLike, passportId: string, handshakeId: string, now: () => string = () => new Date().toISOString()): Promise<void> {
  const row = await getHandshake(db, handshakeId);
  if (!row || row.passport_id !== passportId) throw new HandshakeError("no such handshake", 404);
  await blockFan(db, passportId, row.account_id, now);
}

/** Block one fan from asking this creator anything. Idempotent. */
export async function blockFan(db: DbLike, passportId: string, accountId: string, now: () => string = () => new Date().toISOString()): Promise<void> {
  if (!ACCOUNT_RE.test(accountId)) throw new HandshakeError("bad account id");
  // Blocking also closes anything already open from that fan — a block that
  // left the question sitting in the inbox would be a block in name only.
  // ONE batch (I3-1845): a fault between the two cannot leave the block set
  // and the question open, which a re-press (`DO NOTHING`) would not repair.
  await db.batch([
    db.prepare("INSERT INTO handshake_blocks (passport_id, account_id, created_at) VALUES (?, ?, ?) ON CONFLICT(passport_id, account_id) DO NOTHING").bind(passportId, accountId, now()),
    // I3-0892 (D-0920 A): the same block shuts the questions door (question_blocks).
    db.prepare("INSERT INTO question_blocks (passport_id, account_id, created_at) VALUES (?, ?, ?) ON CONFLICT(passport_id, account_id) DO NOTHING").bind(passportId, accountId, now()),
    db.prepare("UPDATE handshakes SET state = 'expired' WHERE passport_id = ? AND account_id = ? AND state = 'open'").bind(passportId, accountId),
  ]);
}

export async function unblockFan(db: DbLike, passportId: string, accountId: string): Promise<void> {
  // I3-0892 (D-0920 A): either unblock frees both channels, in one batch.
  await db.batch([
    db.prepare("DELETE FROM handshake_blocks WHERE passport_id = ? AND account_id = ?").bind(passportId, accountId),
    db.prepare("DELETE FROM question_blocks WHERE passport_id = ? AND account_id = ?").bind(passportId, accountId),
  ]);
}

/**
 * THE PER-SCOPE REFUSAL — the heart of the safety argument.
 *
 * Returns the CANONICAL subject string to store, or throws. For `identity` and
 * `payment` this asks the database whether the passport ALREADY carries the
 * fact, and refuses otherwise; for `offer` and `call` there is no fact to
 * carry, so it validates shape only.
 *
 * REFUSING AT ASK TIME, rather than only at answer time, is deliberate: a
 * handshake that could never be answered truthfully should not reach a
 * creator's inbox at all, because it would be an unanswerable question a fan
 * then reads silence into. `buildAnswerDraft` re-checks anyway (state can move
 * in fourteen days), so this is the fast, kind refusal and not the enforcement.
 */
export async function canonicalSubject(db: DbLike, passportId: string, scope: HandshakeScope, subject: string, nowMs: number): Promise<string> {
  const trimmed = typeof subject === "string" ? subject.trim() : "";
  if (trimmed.length === 0 || trimmed.length > SUBJECT_MAX || hasBadText(trimmed)) {
    throw new HandshakeError(`subject must be 1..${SUBJECT_MAX} printable characters without < or > or invisible formatting characters such as direction marks`);
  }
  switch (scope) {
    case "identity": {
      // The account key must ALREADY be an active row on this passport, and not
      // shielded — a shielded account is deliberately off every public roster
      // (P1, `account-visibility.ts`), and answering a handshake about one would
      // put it back on a surface the creator took it off.
      // Read it through `getPassportView`, which IS the public roster — so this
      // asks the exact question the safety argument needs answered: "can the fan
      // already see this account on this passport?" The view has done the work
      // and doing it again by hand would fork the rules. It drops SHIELDED
      // accounts (P1 — a creator who took an account off every public roster
      // must not have it put back by a handshake) and drops ASSERTED-but-
      // unproven rows (S101 — otherwise a handshake would launder an unproven
      // famous handle into a signed attestation, which is the single worst thing
      // this feature could do), and it returns null for a suppressed or
      // retracted subject.
      const view = await getPassportView(db, passportId, nowMs);
      // PLATFORM-BOUND, AND CONFIRMED. Two separate holes were here.
      //
      //  (a) Matching a bare handle across every platform row meant a fan who
      //      encountered `instagram.com/realcreator` — a fake — could type
      //      `realcreator`, match the passport's `x:realcreator`, and be handed
      //      a signed "yes, that is mine" with the platform nowhere on the
      //      screen. A handle is not an identity; a handle ON A PLATFORM is.
      //      So the input must name the platform (or be the exact URL), and the
      //      stored subject is the MATCHED ROW's `platform:handle` — which is
      //      what "canonical" was always supposed to mean here.
      //  (b) `getPassportView` drops SELF-ASSERTED unproven rows (S101) but
      //      KEEPS operator/system-listed ones nobody has proved, marked
      //      `confirmed: false` — the OSINT coverage tier. Signing "yes, that
      //      account is mine, today" for a row the passport itself renders as
      //      not confirmed is precisely the second ownership channel this
      //      feature refuses to be, so `confirmed` is required too.
      const row = view?.accounts.find((a) => a.status === "active" && a.confirmed && (a.url === trimmed || `${a.platform}:${a.handle}` === trimmed));
      if (!row) {
        // Deliberately ONE refusal for unknown, revoked and shielded alike: the
        // differences between those three are exactly what an attacker probing a
        // roster wants, and a fan does not need them.
        throw new HandshakeError("there is nothing on this creator's passport to confirm about that account", 409);
      }
      return `${row.platform}:${row.handle}`;
    }
    case "payment": {
      // Must match a CURRENTLY-RENDERING payout tag. `readLivePayoutTags` is the
      // read-only form and it carries the currently-verified gate inside its own
      // statement, so there is no ungated way to ask this question.
      const tags = await readLivePayoutTags(db, passportId, nowMs);
      if (!tags.some((t) => t.value === trimmed)) {
        throw new HandshakeError("there is nothing on this creator's passport to confirm about that destination", 409);
      }
      return trimmed;
    }
    case "offer": {
      // There is no fact on the passport to check an offer against — it is a
      // question about the FUTURE — so the check is STRUCTURAL instead: a kind
      // from a closed vocabulary and an integer number of credits, and nothing
      // else. That is what stops the subject being the free-text channel the
      // note is bounded to prevent. The signed answer is an acknowledgement
      // that CREATES NO OBLIGATION AND MOVES NO MONEY; the ≤120-char note is
      // where a fan says what they actually want.
      const m = OFFER_RE.exec(trimmed);
      const kind = m?.[1];
      const credits = m ? Number(m[2]) : NaN;
      if (!kind || !(OFFER_KINDS as readonly string[]).includes(kind) || !Number.isInteger(credits) || credits < 1 || credits > MAX_OFFER_CREDITS) {
        throw new HandshakeError(`an offer is "<kind>:<credits>" where kind is one of ${OFFER_KINDS.join("|")} and credits is 1..${MAX_OFFER_CREDITS}`);
      }
      return `${kind}:${credits}`;
    }
    case "call": {
      // ONE calendar day, and deliberately not a time, a range or a sentence.
      // The fan is asking "would you"; the scheduling that would follow a yes is
      // a conversation, and a conversation is the thing this product does not
      // host.
      // STRICT (I3-1879): `Date.parse` rolls `2026-02-30` over to March 2nd.
      if (!CALL_RE.test(trimmed) || !isStrictDate(trimmed)) {
        throw new HandshakeError("a call proposes one day, as YYYY-MM-DD");
      }
      return trimmed;
    }
    default:
      throw new HandshakeError("unknown scope");
  }
}

export interface AskInput {
  scope: HandshakeScope;
  subject: string;
  note?: string | null;
}

/**
 * Ask. Returns the new handshake.
 *
 * ORDER, AND WHY:
 *  1. THE PASSPORT GATE, first and through `livePassport`, so suppressed /
 *     retracted / never-listed are one indistinguishable 404 — F6's lesson.
 *  2. THE DOOR. "Not accepting", "blocked" and "not verified" answer
 *     IDENTICALLY, on the same statement. "You are blocked" is information a
 *     harasser acts on; this door does not give it.
 *  3. STANDING, then the note/subject validation, then the INSERT.
 *  4. EVERY CAP RIDES INSIDE THE INSERT. F6's review lesson applied on day one:
 *     a SELECT above an INSERT is not a ceiling, because N concurrent callers
 *     all read the same pre-state. The one-open-per-pair rule is additionally
 *     structural (`idx_hsk_open_pair`), so the INSERT's own counts are not the
 *     only thing between a fan and a queue.
 */
export async function askHandshake(
  db: DbLike,
  accountId: string,
  passportId: string,
  input: AskInput,
  nowMs: number,
  now: () => string = () => new Date().toISOString(),
  /** `ANALYTICS_FAN_SECRET`. Undefined is refused by `fanKey` (ADR-0006 §2). */
  fanSecret?: string,
  /** `AnalyticsTickEnv` for `publishAfterCommit` (S-003 X4/A-4); undefined
   * (a caller with no route context — a test, a cron leg with no waitUntil)
   * skips the request-path publish and leaves the row for the cron drain. */
  pubEnv?: AnalyticsTickEnv,
  /** The route's `c.executionCtx`-shaped waitUntil, threaded from the caller
   * (ADR §2 request-path publish). */
  pubCtx?: { waitUntil(p: Promise<unknown>): void },
): Promise<HandshakeRow> {
  if (!ACCOUNT_RE.test(accountId)) throw new HandshakeError("no such account", 401);
  const acct = await db.prepare("SELECT 1 AS x FROM accounts WHERE account_id = ?").bind(accountId).first();
  if (!acct) throw new HandshakeError("no such account", 401);
  if (!PSP_RE.test(passportId)) throw new HandshakeError("no such creator", 404);
  if (!(HANDSHAKE_SCOPES as readonly string[]).includes(input.scope)) throw new HandshakeError("unknown scope");

  const p = await livePassport(db, passportId);
  if (!p) throw new HandshakeError("no such creator", 404);

  // ONE STATEMENT for both switches, and ONE refusal for all three reasons.
  const door = await db
    .prepare(
      `SELECT (SELECT accepting FROM handshake_settings h WHERE h.passport_id = ?) AS accepting,
              (SELECT 1 FROM handshake_blocks b WHERE b.passport_id = ? AND b.account_id = ?) AS blocked`,
    )
    .bind(passportId, passportId, accountId)
    .first<{ accepting: number | null; blocked: number | null }>();
  const switchedOn = door?.accepting === null || door?.accepting === undefined ? true : door.accepting === 1;
  if (!(p.claim_state === "verified" && switchedOn && !door?.blocked)) {
    throw new HandshakeError("this creator is not taking handshakes", 403);
  }

  if (!(await evaluate(db, accountId, passportId, HANDSHAKE_STANDING, nowMs))) {
    throw new HandshakeError("handshakes are open to fans who already follow or support this creator", 403);
  }

  const subject = await canonicalSubject(db, passportId, input.scope, input.subject, nowMs);
  const rawNote = typeof input.note === "string" ? input.note.trim() : "";
  if (rawNote.length > 0 && !NOTE_SCOPES.has(input.scope)) throw new HandshakeError("a note is only allowed on an offer or a call");
  if (rawNote.length > NOTE_MAX || hasBadText(rawNote)) throw new HandshakeError(`the note must be at most ${NOTE_MAX} printable characters without < or > or invisible formatting characters such as direction marks`);
  const note = rawNote.length > 0 ? rawNote : null;

  const id = newHandshakeId();
  const createdAt = now();
  const expiresAt = new Date(nowMs + HANDSHAKE_TTL_DAYS * DAY_MS).toISOString();
  const windowStart = new Date(nowMs - PER_CREATOR_WINDOW_DAYS * DAY_MS).toISOString();

  // THE ANALYTICS ROW RIDES THIS BATCH (ADR-0006 §2) AND CARRIES THE INSERT'S
  // OWN VERDICT. Every cap lives INSIDE the INSERT (point 4 above), so the
  // event is predicated on the row existing rather than on anything read before
  // it — the fresh `handshake_id` makes that an exact test.
  const key = await fanKey(fanSecret, accountId, passportId);
  const span = outboxSpan(db);
  const spanRes = await db
    .batch([
      span.open,
      db
        .prepare(
          `INSERT INTO handshakes (handshake_id, passport_id, account_id, scope, subject, note, state, created_at, expires_at)
       SELECT ?, ?, ?, ?, ?, ?, 'open', ?, ?
        WHERE NOT EXISTS (SELECT 1 FROM handshake_blocks b WHERE b.passport_id = ? AND b.account_id = ?)
          AND COALESCE((SELECT s.accepting FROM handshake_settings s WHERE s.passport_id = ?), 1) = 1
          AND (SELECT COUNT(*) FROM handshakes WHERE account_id = ? AND state = 'open' AND expires_at > ?) < ?
          AND (SELECT COUNT(*) FROM handshakes WHERE account_id = ? AND passport_id = ? AND created_at >= ?) < ?
          AND (SELECT COUNT(*) FROM handshakes WHERE passport_id = ? AND state = 'open' AND expires_at > ?) < ?`,
        )
        .bind(
          id,
          passportId,
          accountId,
          input.scope,
          subject,
          note,
          createdAt,
          expiresAt,
          passportId,
          accountId,
          passportId,
          accountId,
          createdAt,
          MAX_OPEN_PER_FAN,
          accountId,
          passportId,
          windowStart,
          MAX_PER_FAN_PER_CREATOR,
          passportId,
          createdAt,
          MAX_OPEN_PER_CREATOR,
        ),
      ...outboxStatements(
        db,
        { kind: "handshake_requested", passportId, occurredDay: occurredDay(createdAt), fanKey: key },
        { accountId, extraWhere: { sql: "EXISTS (SELECT 1 FROM handshakes WHERE handshake_id = ?)", params: [id] } },
      ),
      span.close,
    ])
    .catch((e: unknown) => {
      // `idx_hsk_open_pair` — a second OPEN question at the same creator. The
      // structural half of the per-pair cap.
      if (e instanceof Error && /UNIQUE constraint/i.test(e.message)) throw new HandshakeError("you already have an open handshake with this creator", 409);
      throw e;
    });
  const seqs = outboxSpanSeqs(spanRes[0], spanRes[spanRes.length - 1]);
  if (pubEnv && pubCtx) publishAfterCommit(pubEnv, pubCtx, seqs);
  // `DbLike.batch` returns `unknown[]` — no `changes` (db.ts:20) — so the
  // refusal is read off the TABLE by the id this call alone minted.
  const landed = await db.prepare("SELECT 1 AS x FROM handshakes WHERE handshake_id = ?").bind(id).first();
  if (!landed) {
    // I3-1844 / I3-1264: the door is re-asked by the INSERT itself (a block or
    // an opt-out can land after the read above), and answers with the door's
    // own single refusal.
    const shut = await db
      .prepare(
        `SELECT (SELECT 1 FROM handshake_blocks b WHERE b.passport_id = ? AND b.account_id = ?) AS blocked,
                (SELECT accepting FROM handshake_settings s WHERE s.passport_id = ?) AS accepting`,
      )
      .bind(passportId, accountId, passportId)
      .first<{ blocked: number | null; accepting: number | null }>();
    if (shut?.blocked || shut?.accepting === 0) throw new HandshakeError("this creator is not taking handshakes", 403);
    // Tell the fan WHICH ceiling stopped them. All three are facts about their
    // own activity or about an inbox size, so none discloses anything about the
    // creator a fan could not already infer.
    const mine = await db.prepare("SELECT COUNT(*) AS n FROM handshakes WHERE account_id = ? AND state = 'open' AND expires_at > ?").bind(accountId, createdAt).first<{ n: number }>();
    if ((mine?.n ?? 0) >= MAX_OPEN_PER_FAN) throw new HandshakeError(`you already have ${MAX_OPEN_PER_FAN} handshakes waiting for an answer`, 429);
    const recent = await db
      .prepare("SELECT COUNT(*) AS n FROM handshakes WHERE account_id = ? AND passport_id = ? AND created_at >= ?")
      .bind(accountId, passportId, windowStart)
      .first<{ n: number }>();
    if ((recent?.n ?? 0) >= MAX_PER_FAN_PER_CREATOR) throw new HandshakeError(`you have already asked this creator in the last ${PER_CREATOR_WINDOW_DAYS} days`, 429);
    throw new HandshakeError("this creator's handshake inbox is full — try again later", 429);
  }
  return { handshake_id: id, passport_id: passportId, account_id: accountId, scope: input.scope, subject, note, state: "open", created_at: createdAt, expires_at: expiresAt, answered_at: null };
}

/** One handshake by id, or null. */
export async function getHandshake(db: DbLike, handshakeId: string): Promise<HandshakeRow | null> {
  if (!HANDSHAKE_ID_RE.test(handshakeId)) return null;
  const r = await db
    .prepare("SELECT handshake_id, passport_id, account_id, scope, subject, note, state, created_at, expires_at, answered_at FROM handshakes WHERE handshake_id = ?")
    .bind(handshakeId)
    .first<HandshakeRow>();
  return r ?? null;
}

/**
 * THE DRAFT, BUILT SERVER-SIDE FROM THE STORED ROW (the S100 lock).
 *
 * The caller supplies a handshake id and a disposition and NOTHING else; the
 * scope comes out of the database. So the /start and /finish legs cannot
 * disagree about what is being signed, and a hostile client cannot substitute a
 * different scope, a different subject or somebody else's handshake.
 *
 * It re-runs the per-scope refusal, because fourteen days is long enough for an
 * account to be revoked or a payout tag to be replaced, and a creator must not
 * be able to sign "yes, that's mine" about a row that stopped being true while
 * the question sat in their inbox.
 */
export async function buildAnswerDraft(
  db: DbLike,
  passportId: string,
  handshakeId: string,
  disposition: HandshakeDisposition,
  nowMs: number,
): Promise<{ row: HandshakeRow; draft: DraftEvent }> {
  if (!(HANDSHAKE_DISPOSITIONS as readonly string[]).includes(disposition)) throw new HandshakeError("disposition must be confirmed or declined");
  const row = await getHandshake(db, handshakeId);
  if (!row || row.passport_id !== passportId) throw new HandshakeError("no such handshake", 404);
  if (row.state !== "open") throw new HandshakeError("this handshake has already been answered or has lapsed", 409);
  if (Date.parse(row.expires_at) <= nowMs) throw new HandshakeError("this handshake has already been answered or has lapsed", 409);
  // CONFIRMING re-checks the fact. DECLINING does not: a decline asserts
  // nothing about the passport, and refusing a decline because the underlying
  // row changed would leave a creator unable to close a question.
  if (disposition === "confirmed") await canonicalSubject(db, passportId, row.scope, row.subject, nowMs);
  const draft: DraftEvent = {
    event_type: "handshake_answered",
    subject: passportId,
    actor_kind: "creator_passkey",
    actor_id: "passkey",
    payload: { handshake_id: row.handshake_id, scope: row.scope, disposition },
  };
  return { row, draft };
}

/**
 * The projection write that rides the SAME signed batch as the append — atomic
 * with it, so a `confirmed` row with no event behind it is not reachable, and
 * neither is an event with no row.
 *
 * The state guard is INSIDE the UPDATE, and the admission gate below turns
 * "the UPDATE matched nothing" into "the whole batch fails" — the
 * `recognition-routes.ts` idiom. Without it a second answer that overtook the
 * first would append a signed event over a row that was already closed.
 */
export function buildAnswerStatements(db: DbLike, handshakeId: string, disposition: HandshakeDisposition, nowIso: string, passportId: string): DbStatement[] {
  // THE NONCE IS THE GATE, and matching on the disposition was not.
  //
  // The first version's gate read "does the row now say `confirmed`?" — a
  // question a COMPETING batch can already have made true. Two answers racing
  // with the SAME disposition: the loser's UPDATE matches nothing, the subquery
  // still finds the row, NOT NULL is satisfied, and a second signed
  // `handshake_answered` lands over a closed handshake. Not exotic either:
  // `appendEvents` re-runs these same statements when it loses a ledger-head
  // race, against freshly committed state.
  //
  // A per-attempt nonce written into `answer_event` makes the gate read back a
  // row THIS batch alone created — the `recognitionQuotaGate` idiom
  // (`standing.ts`), which is the one that is actually sound.
  const nonce = `ans_${crypto.randomUUID().replace(/-/g, "")}`;
  const fresh = freshProofBounds(Date.parse(nowIso));
  return [
    // LW-03's PAYMENT LIMB, CLOSED AT COMMIT TIME. `buildAnswerDraft` rechecks
    // `canonicalSubject` with a plain read, well before this batch executes —
    // a passkey assertion is verified in between (`appendCreatorSigned`). A
    // payout tag can be retired or replaced in that window, and the read-then-
    // write shape would let a `confirmed` land for a destination that already
    // stopped rendering, asserting the fact as current when it no longer is.
    // Declines assert nothing about the passport (`buildAnswerDraft`'s own
    // rule), so this only ever narrows a `confirmed`. `scope` and `subject`
    // are columns on the ROW BEING UPDATED, so this is one self-contained
    // predicate — no second lookup, no TOCTOU of its own.
    //
    // THE IDENTITY LIMB, CLOSED THE SAME WAY (I3-1386): a `confirmed` identity
    // answer lands only while `canonicalSubject`'s question is still yes AT
    // COMMIT — an ACTIVE `passport_accounts` row on this passport whose
    // `platform:id` is the stored subject, not shielded
    // (`shieldedSqlPredicate`, account-visibility.ts), with a CURRENT proof of
    // this passport's own (`deriveStanding`'s `proven_current`:
    // `claim_challenges` used/non-contest or `oauth_challenges` proven, fresh by
    // explicit `valid_until`, else `proved_at` within `PROOF_TTL_MS` —
    // `freshProofBounds`).
    //
    // THE DESTINATION SET IS THE PUBLIC READ'S AT THIS INSTANT (B25-633):
    // `pickRendering` (payout.ts) — a DUE staged generation, the empty removal
    // marker included, replaces the live set before `promoteDueTags` retires
    // it, so a live row confirms only while no due staged row exists. The
    // standing half of that read is `verifiedMintGate` below, in this batch.
    db
      .prepare(

            SET state = ?, answered_at = ?, answer_event = ?
          WHERE handshake_id = ? AND state = 'open' AND passport_id = ?

                   SELECT 1 FROM payout_tags t
                    WHERE t.passport_id = ?

                            (t.status = 'staged' AND t.effective_at <= ?)

                                  SELECT 1 FROM payout_tags d
                                   WHERE d.passport_id = t.passport_id
                                     AND d.status = 'staged' AND d.effective_at <= ?))
                          )
                 ))
            )`,
      )
      .bind(disposition, nowIso, nonce, handshakeId, passportId, disposition, passportId, fresh.nowIso, fresh.provedSince, fresh.nowIso, fresh.provedSince, passportId, nowIso, nowIso),
    // THE ADMISSION GATE — the `write_admissions` mechanism (0022) in a table of
    // its own. The value going into a NOT NULL column is a SUBQUERY that returns
    // NULL exactly when the UPDATE above matched nothing; NULL then fails the
    // constraint and the WHOLE signed batch rolls back. So a
    // `handshake_answered` event can never land over a handshake that was
    // already answered, already lapsed, or belongs to somebody else.
    //
    // A table of its own rather than a new `write_admissions` gate value,
    // because that column's CHECK is a closed vocabulary written in an applied
    // migration and widening a CHECK means rebuilding the table — the wrong
    // trade for one more gate. Same mechanism, same NOT NULL, same audit row.
    db
      .prepare(
        `INSERT INTO handshake_admissions (handshake_id, admitted_at)
         VALUES ((SELECT handshake_id FROM handshakes WHERE handshake_id = ? AND answer_event = ?), ?)
         ON CONFLICT(handshake_id) DO UPDATE SET admitted_at = excluded.admitted_at`,
      )
      .bind(handshakeId, nonce, nowIso),
    // THE STANDING GATE, at COMMIT TIME. `events.ts` refuses a
    // `handshake_answered` from a creator who is not verified, but that check
    // runs in `foldEffects` — BEFORE the batch — so it is the same read-then-
    // write shape one layer lower, and it never looks at suppression at all.
    // Every other signed mint path in this tree rides `verifiedMintGate`; an
    // attestation a fan acts on to decide where to send money is not the one to
    // leave off it. A downgrade, a retraction or a suppression landing between
    // the ceremony's two legs now fails the whole batch.
    verifiedMintGate(db, passportId, handshakeId, () => nowIso),
    // CONDITIONED ON THE FACT (INV-051 U-1, B25-633): the UPDATE above is a
    // conditional write (its payment limb carries `AND NOT EXISTS`), so the
    // emit carries the landed fact — this batch's own `answer_event` nonce on
    // the row — as `extraWhere`, and the event exists iff the answer does.
    // The admission gate above already fails the WHOLE batch when the UPDATE
    // matched nothing (ADR-0006 §2); this makes the emit say so itself. The dimension
    // is the disposition THIS BATCH WRITES — the same value bound to the UPDATE
    // — not a re-read of the row, so `handshakes_authenticated` and
    // `handshakes_denied` can never disagree with the state that landed. The
    // vocabulary is `HANDSHAKE_DISPOSITIONS`, which is the closed enum
    // `analytics-events.ts` checks the dim against.
    ...outboxStatements(
      db,
      { kind: "handshake_answered", passportId, occurredDay: occurredDay(nowIso), dims: { disposition } },
      { extraWhere: { sql: "EXISTS (SELECT 1 FROM handshakes WHERE handshake_id = ? AND answer_event = ?)", params: [handshakeId, nonce] } },
    ),
  ];
}

/**

 * suppressed subject the way `listOwnRequests` does — a fan's own list is not a
 * back door around somebody's delisting.

 * An open row past `expires_at` lists as `expired` before the sweep reaches it

export async function listOwnHandshakes(db: DbLike, accountId: string, nowIso: string = new Date().toISOString()): Promise<HandshakeView[]> {

    .prepare(
      `SELECT h.handshake_id AS handshake_id, h.passport_id AS passport_id, p.display_name AS display_name,
              h.scope AS scope, h.subject AS subject, h.note AS note,
              CASE WHEN h.state = 'open' AND h.expires_at <= ? THEN 'expired' ELSE h.state END AS state,
              h.created_at AS created_at, h.expires_at AS expires_at, h.answered_at AS answered_at
         FROM handshakes h JOIN passports p ON p.passport_id = h.passport_id
        WHERE h.account_id = ? AND p.retracted = 0
          AND NOT EXISTS (SELECT 1 FROM request_suppressions x WHERE x.passport_id = h.passport_id)
        ORDER BY h.created_at DESC LIMIT ?`,
    )
    .bind(nowIso, accountId, LIST_LIMIT)
    .all<HandshakeView>();

  handshake_id: string;
  scope: HandshakeScope;
  subject: string;
  note: string | null;
  state: HandshakeState;
  created_at: string;
  expires_at: string;

   * extent the product already publishes it. */
  from_handle: string | null;
}

/**
 * The creator's inbox. OPEN, unexpired items only: an answered or lapsed
 * handshake is not work, and an inbox that accumulates every question a creator
 * chose not to answer is a guilt surface, not a tool.
 */
export async function handshakeInbox(db: DbLike, passportId: string, nowMs: number): Promise<InboxItem[]> {
  const nowIso = new Date(nowMs).toISOString();
  const { results } = await db
    .prepare(
      `SELECT h.handshake_id AS handshake_id, h.scope AS scope, h.subject AS subject, h.note AS note,
              h.state AS state, h.created_at AS created_at, h.expires_at AS expires_at,
              (SELECT u.handle FROM user_profiles u WHERE u.account_id = h.account_id AND u.visibility = 'public') AS from_handle
         FROM handshakes h
        WHERE h.passport_id = ? AND h.state = 'open' AND h.expires_at > ?
        ORDER BY h.created_at ASC LIMIT ?`,
    )
    .bind(passportId, nowIso, INBOX_LIMIT)
    .all<InboxItem>();
  return results;
}

/**
 * The expiry sweep (cron). Lapses every open handshake past its TTL.
 *
 * It writes NO ledger event and emits NO notification, on purpose: an expiry is
 * the ABSENCE of an act, and minting a record of somebody not doing something
 * is how "did not answer" becomes a number that an operator, a fan or a future
 * ranking later reads as a score. The row simply stops being open.
 */
export async function sweepExpiredHandshakes(db: DbLike, nowMs: number, limit = 500): Promise<number> {
  const nowIso = new Date(nowMs).toISOString();
  const res = await db
    .prepare("UPDATE handshakes SET state = 'expired' WHERE handshake_id IN (SELECT handshake_id FROM handshakes WHERE state = 'open' AND expires_at <= ? ORDER BY expires_at LIMIT ?)")
    .bind(nowIso, limit)
    .run();
  return res.changes ?? 0;
}

 * `purchaseState`/`obfuscatedExternalAccountId`/`orderId`, RTDN
 * `voidedPurchaseNotification`) follow Apple/Google's documented shapes and are
 * exercised by the tests with synthetic payloads; confirm against the live store
 * consoles before launch, same as any store integration.
 */

const placeholder = (v: string | undefined): boolean => !v || String(v).trim() === "" || /TODO/i.test(v);

/** productId → credits granted. Server-authoritative; a SKU we did not configure
 * grants nothing. */
export type CreditSkus = ReadonlyMap<string, number>;

/** Parse `IAP_CREDIT_SKUS` (JSON object productId→credits). Null (rail disabled)
 * if absent/placeholder/malformed. Every value must be a positive integer. */
export function creditSkusFromEnv(env: Record<string, unknown>): CreditSkus | null {
  const raw = env.IAP_CREDIT_SKUS as string | undefined;
  if (placeholder(raw)) return null;
  let parsed: unknown;
  try {
    parsed = JSON.parse(raw!);
  } catch {
    return null;
  }
  if (!parsed || typeof parsed !== "object" || Array.isArray(parsed)) return null;
  const map = new Map<string, number>();
  for (const [k, v] of Object.entries(parsed as Record<string, unknown>)) {
    if (typeof k !== "string" || k.length === 0 || k.length > 128) return null;
    if (typeof v !== "number" || !Number.isInteger(v) || v <= 0 || v > 10_000_000) return null;
    map.set(k, v);
  }
  return map.size > 0 ? map : null;
}

/**
 * I3-0162 — THE 14-DAY WITHDRAWAL WAIVER, AND WHY IT IS A GATE IN THIS FILE.
 *
 * EU Consumer Rights Directive art. 16(m) / UK CCR 2013 reg. 37: a consumer has
 * 14 days to withdraw from a distance contract for digital content, AND that
 * right is lost only if the consumer gave PRIOR EXPRESS CONSENT to immediate
 * performance and ACKNOWLEDGED the loss. Credits are delivered instantly and
 * are spendable at once, so without a captured waiver the trader owes a refund
 * for 14 days on every purchase — and nothing anywhere in the credit-purchase
 * path captured one: `grep -rn "withdrawal|waiver" services/trust-core/src`
 * answered nothing before this change.
 *
 * WHAT IS OURS AND WHAT IS NOT. The CAPTURE (a version, an instant, and a
 * refusal when either is missing) is code. The WORDING of the consent and the
 * acknowledgement, and which policy version is in force, are counsel's — so
 * this ships with `version: null` and refuses nothing until that ruling lands.
 * Setting a version turns the gate on deployment-wide; no code change is needed
 * at that point, which is the whole shape of it.
 *
 * DIRECTION OF FAILURE, once a version exists: a missing, mismatched or
 * post-dated waiver REFUSES the grant (`null`, this file's existing fail-closed
 * answer), because "we could not tell whether they waived" must never grant an
 * instantly-spendable, dollar-backed credit under a right the buyer still holds.
 *
 * WAITING ON: counsel's wording + policy version (see this room's report, I3-0162).
 */
export interface WithdrawalWaiver {
  /** The consent text version the buyer was shown and accepted. */
  policy_version: string;
  /** When they accepted it (ISO 8601). Must not be AFTER the purchase. */
  acknowledged_at: string;
}

export interface WithdrawalPolicy {
  /** The version in force, or null while unruled (gate inert). */
  version: string | null;
}

/** THE POLICY, DELIBERATELY UNSET. Filled by counsel's ruling, not by a builder. */
export const CREDIT_WITHDRAWAL_POLICY: WithdrawalPolicy = { version: null };

/**
 * REG-07 — THE ARMING PRECONDITION: THE CAPTURE BEFORE THE VERSION.
 *
 * Setting `version` turns the gate on deployment-wide, which is the right
 * shape — and it is exactly why the ordering has to be code. Measured against
 * the unwired tree: setting `version` turns 5 of 12 `test/iap-credits*.test.ts`
 * cases red, and in production would refuse EVERY credit purchase on both
 * rails with "no withdrawal waiver was captured for this purchase". The buyer
 * did nothing wrong; nobody had built the place to capture the waiver.
 *
 * So the source is declared beside the policy. The routes now READ a
 * `withdrawal_waiver` from the purchase body (`index.ts`, both rails), so the
 * server half is wired — but no client posts one and nothing persists one, so
 * the source is NOT live and arming the version is a DEPLOYMENT fault that
 * answers 503 `credit_gate_misconfigured`, not 400 per buyer.
 */
export type CreditWaiverSource = "live" | "none";

export interface CreditWaiverSources {
  /**
   * NOT LIVE. What has to exist before this is flipped:
   *   1. a consent + acknowledgement screen in the checkout flow of BOTH store
   *      clients, showing counsel's wording at the version in force;
   *   2. that client posting `withdrawal_waiver: { policy_version, acknowledged_at }`
   *      on `/account/war/credits/iap/{apple,google}` — the routes already read it;
   *   3. a persisted record of the acknowledgement (a table; there is none
   *      today), because art. 16(m) has to be provable after the fact, not
   *      only at the instant of the grant.
   * Flip this in the SAME edit that lands (1)-(3), never in the edit that sets
   * `CREDIT_WITHDRAWAL_POLICY.version`.
   */
  waiver: CreditWaiverSource;
}

export const CREDIT_WITHDRAWAL_SOURCES: CreditWaiverSources = { waiver: "none" };

/** The deployment fault, or null. Defaults to the SHIPPED policy and the
 * SHIPPED sources: this asks whether THIS DEPLOYMENT armed the gate ahead of
 * its capture, not whether a caller passed an odd policy. */
export function creditPolicyArmingFault(
  sources: CreditWaiverSources = CREDIT_WITHDRAWAL_SOURCES,
  policy: WithdrawalPolicy = CREDIT_WITHDRAWAL_POLICY,
): string | null {
  if (policy.version !== null && sources.waiver !== "live") {
    return "CREDIT_WITHDRAWAL_POLICY.version is in force but no checkout captures a withdrawal waiver";
  }
  return null;
}

/** Parse a `withdrawal_waiver` out of a purchase body. Undefined on anything
 * that is not both fields as strings — the predicate below then refuses, which
 * is the correct fail-closed answer for a buyer who acknowledged nothing. */
export function parseWithdrawalWaiver(v: unknown): WithdrawalWaiver | undefined {
  if (!v || typeof v !== "object") return undefined;
  const o = v as Record<string, unknown>;
  if (typeof o.policy_version !== "string" || typeof o.acknowledged_at !== "string") return undefined;
  if (o.policy_version.length > 64 || o.acknowledged_at.length > 64) return undefined;
  return { policy_version: o.policy_version, acknowledged_at: o.acknowledged_at };
}

/** The refusal reason, or null when this purchase may be granted. Pure.
 * `purchaseMs` is the instant of the purchase being verified; a waiver
 * acknowledged after it is not PRIOR consent and does not discharge art. 16(m). */
export function creditWithdrawalRefusal(
  waiver?: WithdrawalWaiver,
  policy: WithdrawalPolicy = CREDIT_WITHDRAWAL_POLICY,
  purchaseMs?: number,
): string | null {
  if (policy.version === null) return null; // unruled: this rail is unchanged
  if (!waiver || typeof waiver.policy_version !== "string" || typeof waiver.acknowledged_at !== "string") return "no withdrawal waiver was captured for this purchase";
  if (waiver.policy_version !== policy.version) return "the withdrawal waiver is for a superseded policy version";
  const ackMs = Date.parse(waiver.acknowledged_at);
  if (!Number.isFinite(ackMs)) return "the withdrawal waiver has no usable timestamp";
  if (typeof purchaseMs === "number" && Number.isFinite(purchaseMs) && ackMs > purchaseMs) return "the withdrawal waiver post-dates the purchase";
  return null;
}

export interface CreditPurchaseVerified {
  credits: number;
  externalTxnId: string;
  productId: string;
}

/** I3-1254: Apple could not answer (iap.ts storeStatusTransient — 408/425/429/5xx —
 * or a thrown infra error) — a TEMPORARY failure, never a refusal. Thrown so the
 * route answers 5xx (onError, or its own 503 mapping) instead of the 400 a bad
 * purchase gets; the native clients treat any non-2xx as a failure (JA-03), so the
 * distinction is the server's record, not a client retry. */
export class AppleVerifyUnavailable extends Error {
  constructor(detail: string) {
    super(`apple verification unavailable: ${detail}`);
    this.name = "AppleVerifyUnavailable";
  }
}

/** I3-1138 (I16-W07A), the Google twin of `AppleVerifyUnavailable`: Play could
 * not answer (no access token, 401/403, or iap.ts storeStatusTransient — 408/425/
 * 429/5xx — or a thrown fetch/parse). TEMPORARY, never a refusal: the route answers
 * 503, not the refusal status (the native clients do not retry by themselves, JA-03). */
export class GoogleCreditVerifyUnavailable extends Error {
  constructor(detail: string) {
    super(`google verification unavailable: ${detail}`);
    this.name = "GoogleCreditVerifyUnavailable";
  }
}

/** I3-1138 (I19-W1E, D-0954(6)): a store answering 401/403 refused OUR credentials (an operator fault, not the
 * buyer's refusal): logged here — each door's catch rethrows its Unavailable unlogged — and answered as TEMPORARY. */
function storeAuthRefused(rail: "apple" | "google", status: number): AppleVerifyUnavailable | GoogleCreditVerifyUnavailable {
  logError({ cid: "unknown", hop: "http", msg: "iap_credit_store_auth_refused", rail, err: `status ${status}` });
  return rail === "apple" ? new AppleVerifyUnavailable(`status ${status}`) : new GoogleCreditVerifyUnavailable(`status ${status}`);
}

/**
 * Verify an Apple StoreKit CONSUMABLE transaction for a credit SKU against the
 * App Store Server API, bound to `expectedToken` (the account's appAccountToken).
 * Null on any mismatch (wrong app/product/token, not a credit SKU, revoked, bad
 * shape) — never grants unbound. THROWS `AppleVerifyUnavailable` when Apple
 * cannot answer (I3-1254) or refuses our credentials (401/403, I3-1138): that is not a mismatch.
 */
export async function verifyAppleCreditPurchase(
  cfg: AppleIapConfig,
  skus: CreditSkus,
  transactionId: string,
  expectedToken: string,
  nowSec: number,
  fetchImpl: typeof fetch = fetch,
  /** I3-0162: the waiver captured at checkout, as the route knows it. */
  waiver?: WithdrawalWaiver,
  policy: WithdrawalPolicy = CREDIT_WITHDRAWAL_POLICY,
): Promise<CreditPurchaseVerified | null> {
  if (!/^[0-9]{1,32}$/.test(transactionId)) return null; // Apple transaction ids are numeric
  if (!expectedToken) return null;
  // I3-0162, DOOR 1 OF 2. Asked before the store call so an ungrantable
  // purchase costs no outbound request, and logged because a refusal that
  // returns the same bare `null` as a fraud attempt is the exact signal loss
  // I3-1138 fixed below.
  {
    const refusal = creditWithdrawalRefusal(waiver, policy, nowSec * 1000);
    if (refusal) {
      logError({ cid: "unknown", hop: "http", msg: "iap_credit_withdrawal_refused", rail: "apple", err: refusal });
      return null;
    }
  }
  // I17-FXA RGM-11: stripe.ts's deadline idiom (iap.ts STORE_FETCH_TIMEOUT_MS); an abort throws -> AppleVerifyUnavailable below.
  const ctl = new AbortController();
  const timer = setTimeout(() => ctl.abort(), STORE_FETCH_TIMEOUT_MS);
  try {
    const token = await appleApiToken(cfg, nowSec);
    const res = await fetchImpl(`${cfg.apiBase}/inApps/v1/transactions/${transactionId}`, { redirect: "manual", headers: { authorization: `Bearer ${token}` }, signal: ctl.signal });
    if (!res.ok) {
      if (res.status === 401 || res.status === 403) throw storeAuthRefused("apple", res.status); // I3-1138 (I19-W1E)
      if (storeStatusTransient(res.status)) throw new AppleVerifyUnavailable(`status ${res.status}`); // JA-03 (I20-W1E)
      return null;
    }
    const body = (await res.json()) as { signedTransactionInfo?: string };
    if (!body.signedTransactionInfo) return null;
    // Review F1: the credit-grant ROUTE requires cfg.rootCaSha256 (fail closed),
    // so in production this always takes the cryptographically-verified path — a
    // credit is permanent + dollar-backed and the refund/claw path needs the pin,
    // so we never grant from a response we could not later reverse. The decode
    // fallback exists only for the unpinned case the route already refuses.
    const info = cfg.rootCaSha256 ? await verifyAppleJws(body.signedTransactionInfo, cfg.rootCaSha256, nowSec * 1000) : decodeJwsPayload(body.signedTransactionInfo);
    if (!info) return null;
    if (info.bundleId !== cfg.bundleId) return null; // wrong app
    // #600(b), mirroring iap.ts:254-262 (appleGrantFromInfo): the signature proves Apple signed it, never which
    // store. A payload naming the environment we do not sell in is refused; one with NO environment is recorded
    // and proceeds, exactly as iap.ts does.
    const payloadEnv = typeof info.environment === "string" ? info.environment : "";
    if (payloadEnv && payloadEnv !== cfg.environment) {
      logError({ cid: "unknown", hop: "http", msg: "apple_iap_environment_mismatch", rail: "apple" });
      return null;
    }
    if (!payloadEnv) logError({ cid: "unknown", hop: "http", msg: "apple_iap_environment_absent", rail: "apple" });
    if (info.revocationDate) return null; // already refunded/revoked → grant nothing
    const productId = typeof info.productId === "string" ? info.productId : "";
    const credits = skus.get(productId);
    if (!credits) return null; // not a credit SKU we sell
    const tokenOnPurchase = String(info.appAccountToken ?? "");
    if (!tokenOnPurchase || tokenOnPurchase.toLowerCase() !== expectedToken.toLowerCase()) return null; // not this account's purchase
    const extTxn = typeof info.transactionId === "string" ? info.transactionId : transactionId;
    if (!/^[0-9]{1,32}$/.test(extTxn)) return null;
    return { credits, externalTxnId: extTxn, productId };
  } catch (err) {
    // I3-1138: an infra failure here (store-API timeout, malformed JSON, a
    // thrown from verifyAppleJws) used to collapse into the exact same
    // `null` as a fraudulent/invalid purchase, with zero signal for an
    // operator to tell "the fan tried to cheat" from "Apple's API hiccupped
    // and a real purchase got a 400". Logging is what makes the difference
    // visible to an operator; nothing is ever granted from this branch.
    // I3-1254: …and a thrown infra error is now answered as TEMPORARY, not as
    // the refusal `null` means (the log above is unchanged).
    if (err instanceof AppleVerifyUnavailable) throw err;
    logError({ cid: "unknown", hop: "http", msg: "iap_credit_verify_error", rail: "apple", err: err instanceof Error ? err.name : typeof err });
    throw new AppleVerifyUnavailable(err instanceof Error ? err.message : String(err));
  } finally {
    clearTimeout(timer);
  }
}

/**
 * Verify a Google Play CONSUMABLE product purchase for a credit SKU against the
 * Developer API, bound to `expectedToken` (obfuscatedExternalAccountId). Null on
 * any mismatch (not purchased, wrong account, not a credit SKU). THROWS
 * `GoogleCreditVerifyUnavailable` when Play cannot answer (I3-1138).
 *
 * WHERE THE PRODUCT ID COMES FROM, stated because it differs from the Apple
 * twin and the difference is not obvious. Apple's signed transaction carries
 * `productId`, so that rail reads the SKU out of the receipt and the client's
 * claim is never consulted. Google's ProductPurchase resource carries NO product
 * id, so the only binding between a purchase token and a SKU is the REQUEST
 * PATH: `.../purchases/products/{productId}/tokens/{purchaseToken}`. Google
 * resolves that pair and errors when the token does not belong to that product,
 * which is what makes the client-supplied id safe here — the fan cannot submit a
 * 500-credit token under the 5000-credit id, because Google will not answer 200
 * for the mismatched pair.
 *
 * That is a property of Google's API, not of this function, so it is written
 * down rather than asserted: if Google ever starts answering 200 for a
 * mismatched pair, the credit amount becomes client-chosen and this comment is
 * the place that says so. The amount itself is never client-supplied — it comes
 * from `skus`, server-side, keyed by the id Google agreed to.
 */
export async function verifyGoogleCreditPurchase(
  cfg: GooglePlayConfig,
  skus: CreditSkus,
  productId: string,
  purchaseToken: string,
  expectedToken: string,
  nowSec: number,
  fetchImpl: typeof fetch = fetch,
  /** I3-0162: the waiver captured at checkout, as the route knows it. */
  waiver?: WithdrawalWaiver,
  policy: WithdrawalPolicy = CREDIT_WITHDRAWAL_POLICY,
): Promise<CreditPurchaseVerified | null> {
  if (typeof productId !== "string" || !skus.has(productId)) return null;
  // I3-0162, DOOR 2 OF 2. The Google rail is a separate entry point with its own
  // callers; the Apple door does not cover it (FAMILY-004: the control at ONE
  // caller is the trap). Same predicate, same policy object.
  {
    const refusal = creditWithdrawalRefusal(waiver, policy, nowSec * 1000);
    if (refusal) {
      logError({ cid: "unknown", hop: "http", msg: "iap_credit_withdrawal_refused", rail: "google", err: refusal });
      return null;
    }
  }
  if (typeof purchaseToken !== "string" || purchaseToken.length === 0 || purchaseToken.length > 1024) return null;
  if (!expectedToken) return null;
  // I17-FXA RGM-11: stripe.ts's deadline idiom over the product request; an abort throws -> GoogleCreditVerifyUnavailable.
  const ctl = new AbortController();
  const timer = setTimeout(() => ctl.abort(), STORE_FETCH_TIMEOUT_MS);
  try {
    const token = await googlePlayAccessToken(cfg, nowSec, fetchImpl);
    if (!token) throw new GoogleCreditVerifyUnavailable("no Play access token");
    const url = `https://androidpublisher.googleapis.com/androidpublisher/v3/applications/${encodeURIComponent(cfg.packageName)}/purchases/products/${encodeURIComponent(productId)}/tokens/${encodeURIComponent(purchaseToken)}`;
    const res = await fetchImpl(url, { redirect: "manual", headers: { authorization: `Bearer ${token}` }, signal: ctl.signal });
    if (!res.ok) {
      if (res.status === 401 || res.status === 403) throw storeAuthRefused("google", res.status); // I3-1138 (I19-W1E)
      if (storeStatusTransient(res.status)) throw new GoogleCreditVerifyUnavailable(`status ${res.status}`); // JA-03 (I20-W1E)
      return null;
    }
    const body = (await res.json()) as { purchaseState?: number; obfuscatedExternalAccountId?: string; orderId?: string; purchaseType?: number };
    if (body.purchaseState !== 0) return null; // 0 = Purchased (1 = Canceled, 2 = Pending)
    // AF-09: `purchaseType` is Google's OWN signal that this was NOT a normal
    // paid purchase — 0 = licence-test (Play Console tester list / internal-
    // closed track), 1 = promo code, 2 = rewarded-ad. All three carry
    // `purchaseState: 0` exactly like a real purchase, so the state check
    // above cannot tell them apart; this field is the only one that can. A
    // credit here is dollar-backed and payable out to a creator (tips.ts,
    // creator-payout.ts), so any non-production purchase type is refused,
    // same fail-closed stance as an unmapped SKU or an unbound token. The
    // field is absent (undefined) on a genuine paid purchase.
    if (body.purchaseType !== undefined && body.purchaseType !== null) return null;
    const bound = body.obfuscatedExternalAccountId ?? "";
    if (!bound || bound.toLowerCase() !== expectedToken.toLowerCase()) return null;
    const credits = skus.get(productId)!;
    // UI-AUDIT F-M7 — the id we key the grant on MUST be an id the claw path can
    // name. `credit_purchases` is keyed on `externalTxnId`, `reverseGrant` looks
    // a reversal up by it, and the ONLY id a Google voided-purchase RTDN ever
    // carries is `orderId` (see `googleCreditVoid`, which returns nothing else).
    // This used to fall back to `purchaseToken.slice(0, 64)`, which made such a
    // grant permanent by construction: the fan refunds through Play, the void
    // arrives naming an order id that matches no row, and the credits stay spent.
    //
    // So it fails closed, the same stance the Apple side already takes (Review
    // F1: never grant from a response we could not later reverse). A completed
    // Play purchase (purchaseState 0) always carries an order id; a response
    // without one is a response we do not understand well enough to take money
    // for. Refusing costs the fan a retry; granting costs us a credit we could
    // never claw back.
    const orderId = body.orderId;
    if (typeof orderId !== "string" || orderId.length === 0 || orderId.length > 128) return null;
    return { credits, externalTxnId: orderId, productId };
  } catch (err) {
    // I3-1138 (Google twin of the Apple catch above): same collapse, same fix —
    // logged, then answered as TEMPORARY rather than as the refusal `null` means.
    if (err instanceof GoogleCreditVerifyUnavailable) throw err;
    logError({ cid: "unknown", hop: "http", msg: "iap_credit_verify_error", rail: "google", err: err instanceof Error ? err.name : typeof err });
    throw new GoogleCreditVerifyUnavailable(err instanceof Error ? err.message : String(err));
  } finally {
    clearTimeout(timer);
  }
}

/**
 * Detect an Apple REFUND of a credit-SKU consumable in an App Store Server
 * Notification. Returns the transaction id to reverse (the credit_purchases
 * external_txn_id), or null if it is not a credit refund. Signature-verified
 * (pinned root required — the notification is the only auth).
 */
export async function appleCreditRefund(cfg: AppleIapConfig, skus: CreditSkus, signedPayload: string, nowSec: number): Promise<{ externalTxnId: string } | null> {
  return appleCreditNotice(cfg, signedPayload, nowSec, "REFUND");
}

/**
 * I17 #140 PE-03, option A (D-0941(2e)) — Detect an Apple REFUND_REVERSED of a
 * credit-SKU consumable: the store undid a refund it had granted, so the credits
 * `reverseGrant` clawed back are owed to the fan again. Same signature chain and
 * product-identity routing as `appleCreditRefund`. Nothing is reinstated
 * automatically: the caller records it (`recordCreditRefundReversed`) and an
 * operator re-grants by hand.
 */
export async function appleCreditRefundReversed(cfg: AppleIapConfig, signedPayload: string, nowSec: number): Promise<{ externalTxnId: string } | null> {
  return appleCreditNotice(cfg, signedPayload, nowSec, "REFUND_REVERSED");
}

/** `audit_log.action` of the one row a credit refund reversal writes. */
export const CREDIT_REFUND_REVERSED_ACTION = "credit:store_refund_reversed";

/**
 * I17 #140 PE-03, option A: the record of a store-reversed credit refund — ONE
 * `audit_log` row (actor `rail:<rail>`, kind system, item_id = the transaction)
 * naming the transaction and the operator's manual re-grant. Idempotent on
 * redelivery: written only WHERE NOT EXISTS the same action, rail and
 * transaction. Touches no credit table. Returns whether a row was written; a
 * database error THROWS, so the caller answers 5xx and the store retries
 * (record-then-ack).
 */
export async function recordCreditRefundReversed(db: DbLike, rail: "apple" | "google", externalTxnId: string, at: string): Promise<boolean> {
  const actor = `rail:${rail}`;
  // I17-FXA RGM-14 (D-0942(1)): the clawback claim is made only when `refunds.ts reverseGrant` left a reversal
  // record (`credit_reversals`, rail + external_txn_id) for THIS transaction — decided inside the INSERT, so no
  // read-then-write window. Otherwise nothing was taken back and there is nothing for an operator to re-grant.
  const clawed = JSON.stringify({ rail, externalTxnId, reinstated: false, clawed_back: true, action_required: "the store reversed this refund; the credits clawed back for it were NOT reinstated — an operator must re-grant them by hand" });
  const notClawed = JSON.stringify({ rail, externalTxnId, reinstated: false, clawed_back: false, note: "the store reversed a refund for which no credits were taken back here; nothing is owed to the fan" });
  const r = await db
    .prepare(
      "INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) " +
        "SELECT ?, ?, CASE WHEN EXISTS (SELECT 1 FROM credit_reversals WHERE rail = ? AND external_txn_id = ?) THEN ? ELSE ? END, ?, ?, 'system' " +
        "WHERE NOT EXISTS (SELECT 1 FROM audit_log WHERE action = ? AND actor_id = ? AND item_id = ?)",
    )
    .bind(actor, CREDIT_REFUND_REVERSED_ACTION, rail, externalTxnId, clawed, notClawed, at, externalTxnId, CREDIT_REFUND_REVERSED_ACTION, actor, externalTxnId)
    .run();
  return r.changes === 1;
}

async function appleCreditNotice(cfg: AppleIapConfig, signedPayload: string, nowSec: number, type: "REFUND" | "REFUND_REVERSED"): Promise<{ externalTxnId: string } | null> {
  if (!cfg.rootCaSha256) return null;
  if (typeof signedPayload !== "string" || signedPayload.length === 0) return null;
  try {
    const nowMs = nowSec * 1000;
    const payload = await verifyAppleJws(signedPayload, cfg.rootCaSha256, nowMs);
    if (!payload || payload.notificationType !== type) return null;
    const data = payload.data as { signedTransactionInfo?: unknown } | undefined;
    if (!data || typeof data.signedTransactionInfo !== "string") return null;
    const info = await verifyAppleJws(data.signedTransactionInfo, cfg.rootCaSha256, nowMs);
    if (!info || info.bundleId !== cfg.bundleId) return null;
    const productId = typeof info.productId === "string" ? info.productId : "";
    // SC-05: route on IDENTITY (is this the subscription product?), never on
    // the CURRENT live catalog. `skus` is today's sellable SKUs; a credit SKU
    // retired from IAP_CREDIT_SKUS (or the whole rail paused) must still let
    // an already-granted purchase's REFUND reach `reverseGrant`, which looks
    // the transaction up in `credit_purchases` by its own external_txn_id —
    // independent of what is on sale today. Only the subscription product
    // (`cfg.productId`) must fall through to `verifyAppleNotification` /
    // `applySubscriptionEvent`; every other product this app's bundle can
    // notify about is a credit consumable, live or retired, and belongs here.
    // `skus` (appleCreditRefund's) is kept as the parameter callers already pass; it is no
    // longer the gate (a still-current SKU is handled identically either way,
    // see the "control" cases in sc05-break.test.ts).
    if (productId === cfg.productId) return null; // the subscription product — not ours
    const txn = typeof info.transactionId === "string" ? info.transactionId : "";
    return /^[0-9]{1,32}$/.test(txn) ? { externalTxnId: txn } : null;
  } catch (err) {
    // I3-1138: the refund-detection path swallowed infra failures identically —

/**

 * with the matched VoidedPurchase's `voidedReason` — Play's integer code, or null when the listed entry carries none
 * (or the check is not `confirmed`). `googleVoidKind` maps it to the reversal kind.

export async function confirmGoogleCreditVoidDetail(
  cfg: GooglePlayConfig,
  orderId: string,
  nowSec: number,
  fetchImpl: typeof fetch = fetch,
): Promise<{ check: GoogleVoidCheck; voidedReason: number | null }> {
  const none = (check: GoogleVoidCheck) => ({ check, voidedReason: null });
  // I17-FXA RGM-11: ONE deadline over every page (stripe.ts's idiom); an abort throws -> "transient", never "absent".
  const ctl = new AbortController();
  const timer = setTimeout(() => ctl.abort(), STORE_FETCH_TIMEOUT_MS);
  try {
    const token = await googlePlayAccessToken(cfg, nowSec, fetchImpl);
    if (!token) return none("transient");
    const startTime = nowSec * 1000 - GOOGLE_VOID_WINDOW_MS;
    let pageToken: string | undefined;
    for (let page = 0; page < GOOGLE_VOID_PAGES_MAX; page++) {
      const url =
        `https://androidpublisher.googleapis.com/androidpublisher/v3/applications/${encodeURIComponent(cfg.packageName)}/purchases/voidedpurchases` +
        `?startTime=${startTime}&type=0&maxResults=1000${pageToken ? `&token=${encodeURIComponent(pageToken)}` : ""}`;

    }

  } catch {
    return none("transient");
  } finally {
    clearTimeout(timer);
  }
}

/**
 * Detect a Google voided-purchase (refund/chargeback) of a one-time product in a
 * decoded RTDN body. Returns the orderId to reverse, or null. The caller has
 * already authenticated the RTDN (shared-secret) and base64-decoded the data.

export function googleCreditVoid(decoded: Record<string, unknown>): { externalTxnId: string } | null {
  const v = decoded.voidedPurchaseNotification as { orderId?: unknown; productType?: unknown } | undefined;
  if (!v || typeof v.orderId !== "string" || v.orderId.length === 0) return null;
  // productType 1 = one-time product (our consumable credit SKU); 2 = subscription.
  //
  // REQUIRED, not merely "not 2". This used to read
  // `productType !== undefined && productType !== 1`, so an RTDN that OMITS the
  // field was claimed as a credit void — and the caller's credit branch
  // (index.ts POST /billing/google/notifications) RETURNS before

/**
 * I3-0726 (I21-Q156 (a), D-1078(30)): the Google-signed token an AUTHENTICATED Pub/Sub push carries in
 * `Authorization: Bearer <jwt>`. Accepted only as RS256 under a key Google publishes at GOOGLE_PUSH_JWKS_URL, issued by
 * accounts.google.com, for the configured audience, unexpired, and naming the configured service account as a verified
 * email. `unavailable`: no key could be read (the door answers 503 so Pub/Sub redelivers); `refused`: anything else.
 * Pure but for `jwksFor`, which the door binds to the module-scope JWKS cache.
 */
export const GOOGLE_PUSH_JWKS_URL = "https://www.googleapis.com/oauth2/v3/certs";
const GOOGLE_PUSH_ISSUERS: readonly string[] = ["https://accounts.google.com", "accounts.google.com"];
const GOOGLE_PUSH_SKEW_SECONDS = 60;
export type GooglePushVerdict = "ok" | "refused" | "unavailable";
export async function verifyGooglePushToken(
  authorization: string | undefined,
  auth: GoogleRtdnPushAuth,
  jwksFor: (kid: string) => Promise<readonly Jwk[]>,
  nowSec: number,
): Promise<GooglePushVerdict> {
  const m = /^Bearer\s+([A-Za-z0-9_-]+)\.([A-Za-z0-9_-]+)\.([A-Za-z0-9_-]+)$/.exec(authorization ?? "");
  if (!m) return "refused";
  const [, h, p, s] = m as unknown as [string, string, string, string];
  let header: unknown;
  let claims: unknown;
  try {
    header = JSON.parse(b64urlToString(h));
    claims = JSON.parse(b64urlToString(p));
  } catch {
    return "refused";
  }
  if (header === null || typeof header !== "object" || claims === null || typeof claims !== "object") return "refused";
  const { alg, kid } = header as { alg?: unknown; kid?: unknown };
  if (alg !== "RS256" || typeof kid !== "string" || kid === "") return "refused";
  const jwks = await jwksFor(kid);
  if (jwks.length === 0) return "unavailable";
  const jwk = jwks.find((k) => k.kid === kid);
  if (!jwk) return "refused";
  let signed = false;
  try {
    signed = await verifyRs256(jwk, `${h}.${p}`, b64urlToBytes(s));
  } catch {
    signed = false;
  }
  if (!signed) return "refused";
  const c = claims as { iss?: unknown; aud?: unknown; exp?: unknown; iat?: unknown; email?: unknown; email_verified?: unknown };
  if (typeof c.iss !== "string" || !GOOGLE_PUSH_ISSUERS.includes(c.iss)) return "refused";
  const auds = Array.isArray(c.aud) ? c.aud : [c.aud];
  if (!auds.includes(auth.audience)) return "refused";
  if (typeof c.exp !== "number" || !Number.isFinite(c.exp) || c.exp + GOOGLE_PUSH_SKEW_SECONDS < nowSec) return "refused";
  if (typeof c.iat === "number" && (!Number.isFinite(c.iat) || c.iat - GOOGLE_PUSH_SKEW_SECONDS > nowSec)) return "refused";
  if (typeof c.email !== "string" || c.email.toLowerCase() !== auth.serviceAccount) return "refused";
  if (c.email_verified !== true) return "refused";
  return "ok";
}

 * it before purchase and attaches it, and validation requires the store's
 * receipt to echo it. Race-safe: the conditional UPDATE + re-read means

 */
/** Resolve the account that owns an IAP binding token (appAccountToken /
 * obfuscatedExternalAccountId), or null. Used by the store-notification paths to
 * map a store event back to the right account. */
export async function accountByIapToken(db: DbLike, token: string): Promise<string | null> {
  if (!token) return null;

export async function getOrCreateIapToken(db: DbLike, accountId: string, mint: () => string = () => crypto.randomUUID(), now: () => string = () => new Date().toISOString()): Promise<string> {
  void now;

  if (row?.iap_token) return row.iap_token;

  await db.prepare("UPDATE accounts SET iap_token = ? WHERE account_id = ? AND iap_token IS NULL").bind(token, accountId).run();

/**
 * In-app purchase (IAP) validation — selling Premium through Apple StoreKit and
 * Google Play, the store-compliant path for digital subscriptions. The security
 * boundary is SERVER-SIDE validation: a client-submitted transaction/purchase is
 * never trusted; we confirm it directly with the store's authenticated API, then
 * feed the result into the SAME subscription state machine as web checkout
 * (`applySubscriptionEvent`), so tier has one source of truth.
 *
 * Both providers authenticate with a signed JWT (Apple: ES256 App Store Connect
 * key; Google: service-account RS256 → OAuth2 access token), reusing the exact
 * signing used for APNs/FCM. Fail-closed: absent/placeholder config → the
 * provider is disabled (route answers 503), never an open grant path.
 */

/** A normalized grant, ready to become a SubscriptionEvent for a fan account. */
export interface IapGrant {
  ok: boolean;
  /** Idempotency key — see `stateChangeKey` / `notificationKey`. NOT the bare
   * transaction id: that made every store event after the first a no-op. */
  event_id: string;
  status: SubStatus;
  current_period_end: string | null; // RFC3339
  event_time: string; // RFC3339 (ordering)
  /** I3-1300 — RFC3339 tie-break WITHIN one `event_time`. Apple's `event_time` is
   * the transaction's `purchaseDate` (PV-01), which is the SAME for every state
   * change of one transaction, so the ordering guard ties and whichever
   * notification ARRIVES last wins, not whichever HAPPENED last. This is the
   * instant Apple signed the notification (its own `signedDate`; the transaction
   * info's on the client path), which is when the state change was reported.
   * Consumed by billing.ts's ordering key (see the I4-04-R2 WRONG OWNER patch);
   * absent on rails that do not supply one. */
  event_tiebreak?: string;
  customer_id: string | null; // store subscription identifier (support/reference)
}

/**
 * THE DEDUP KEY IS A STATE CHANGE, NOT A TRANSACTION.
 *
 * WHAT WENT WRONG, measured. `event_id` was `iap:apple:<transactionId>` /
 * `iap:google:<orderId>` — one key per PURCHASE. `applySubscriptionEvent`
 * dedups on it, so only the FIRST state change a transaction ever produced was
 * applied and every later one was silently dropped: an Apple REFUND on the
 * transaction that granted the subscription returned `{applied:false,
 * tier:"premium"}`, and a Google SUBSCRIPTION_REVOKED on the same order did the
 * same. A refunded fan kept Premium until the period end — up to a year on an
 * annual plan — because the one message that would have taken it away collided
 * with the purchase that gave it.
 *
 * Dedup is meant to make a REDELIVERY harmless, not to make a REVERSAL
 * impossible. Both stores retry until they get a 2xx, so the same notification
 * arrives many times; that is what must collapse. So:
 *
 *  - Where the store names the notification (Apple's `notificationUUID`, and
 *    Pub/Sub's `messageId` on the Google rail), that name IS the key. It is
 *    stable across the store's retries and unique per event, which is exactly
 *    the contract dedup wants.
 *  - Otherwise — the client-resubmit path, where the app re-presents a receipt
 *    on launch and there is no notification to name — the key is the resulting
 *    STATE: transaction + status + period end. A resubmit that says nothing new
 *    still collapses (so `billing_events` cannot be grown by a client looping
 *    the endpoint), and a resubmit that carries a refund or an expiry gets
 *    through.
 *
 * KNOWN GAP, written down rather than papered over: on the client-resubmit path
 * a state that flips BACK to one already applied (refund → refund reversed, same
 * period end) collides with the earlier key and is dropped. The notification
 * rail carries reversals under their own notification id, so this is a backstop
 * losing a race it does not normally run; it is not a hole in the store→server
 * path.
 *
 * THAT LAST SENTENCE IS ONLY TRUE IF BOTH NOTIFICATION ROUTES ACTUALLY PASS THE
 * NAME, and for a while the Google one did not. Apple reads `notificationUUID`
 * inside verifyAppleNotification, so it cannot be forgotten by a caller; Google's
 * name lives on the Pub/Sub ENVELOPE, not in the RTDN payload, so it can only
 * come from the route — and the route was decoding `message.data` and never
 * reading `message.messageId`. Every Google RTDN therefore keyed on
 * (order, status, period end), and every Play transition returning to a
 * previously-applied pair was dropped: ON_HOLD → RECOVERED, PAUSED → RESTARTED,
 * CANCELED → RESTARTED, grace-period flapping. Measured before the fix: purchase
 * applied, hold applied (free), RECOVERED {applied:false, tier:"free"} — a paying
 * fan held on free until the expiry itself moved. If you add a third rail, the
 * store's own name for the notification is not optional.
 */
function stateChangeKey(store: "apple" | "google", txId: string, status: SubStatus, periodEnd: string | null): string {
  return `iap:${store}:${txId}:${status}:${periodEnd ?? "none"}`;
}

  return `iap:${store}:n:${notificationId}`;
}

/** Decode a JWS payload (Apple returns transaction info as a signed JWS). The

/**
 * THE PAID-AHEAD CEILING, ON THE STORE RAIL TOO.
 *
 * The Stripe rail has refused an implausible `current_period_end` since M7
 * (`MAX_PAID_AHEAD_MS`, ~400 days: the longest plan we sell plus a trial and a
 * leap day). The store rails had no equivalent, so ONE event could buy an
 * unbounded grant — measured, straight through the Apple notification route:
 * `expiresDate: 4070908800000` wrote `current_period_end = '2099-01-01'` with
 * `tier = 'premium'`, and because `sweepExpiredSubscriptions` only downgrades
 * a subscription that is PAST its period end, nothing would ever have taken it
 * away.
 *
 * REFUSE, do not clamp — the same stance and the same reasoning as the Stripe
 * rail: a period end we cannot believe means the payload is not what we think
 * it is, and inventing a plausible grant out of an implausible body is how a
 * parsing bug becomes free product. Refusing costs a `fail` grant on the
 * client path and a non-ACK on the notification path, both of which are
 * visible; granting quietly costs the product.
 *
 * The ceiling is applied to the PERIOD END, which means a revocation carrying
 * an out-of-range `expiresDate` is refused along with it. That is correct
 * rather than unfortunate: the grant that expiry describes was itself refused,
 * so there is no entitlement for the revocation to fail to take away.
 */
function withinPaidAheadCeiling(periodEndMs: number, nowMs: number): boolean {
  return periodEndMs <= nowMs + MAX_PAID_AHEAD_MS;
}

  // `billing.ts` `tierFor` answers "free" for any status that is not `active`, and
  // the 3-day `SUBSCRIPTION_GRACE_MS` window only covers rows still marked
  // `active` — so a failed renewal mapped to `expired` takes Premium away the
  // instant the notification lands, while Apple is still retrying the card and the
  // subscriber is still entitled. The Google rail does not do this:
  // SUBSCRIPTION_STATE_IN_GRACE_PERIOD maps to `active` below, so a failed renewal
  // on Play keeps access and a failed renewal on Apple did not.
  // `gracePeriodExpiresDate` is Apple's own answer for how long that entitlement
  // runs — the App Store Connect-configured grace, up to 16 days — and it rides on
  // the RENEWAL info, which is why it is passed in rather than read from `info`.
  // The same ceiling applies to it as to any other period end: refused, not
  // clamped (see `withinPaidAheadCeiling`).
  const graceMs = typeof renewal?.gracePeriodExpiresDate === "number" ? renewal.gracePeriodExpiresDate : NaN;
  if (Number.isFinite(graceMs) && !withinPaidAheadCeiling(graceMs, nowMs)) return null;
  const inGrace = !revoked && !active && Number.isFinite(graceMs) && graceMs > nowMs;
  const status: SubStatus = revoked ? "canceled" : active || inGrace ? "active" : "expired";
  const periodEnd = inGrace ? new Date(graceMs).toISOString() : Number.isFinite(expiresMs) ? new Date(expiresMs).toISOString() : null;
  return {
    appAccountToken: String(info.appAccountToken ?? ""),
    grant: {
      ok: true,
      // A REFUND arrives on the transaction that granted the subscription, so
      // keying on the transaction alone dropped it. See stateChangeKey.
      event_id: stateChangeKey("apple", transactionId, status, periodEnd),
      status,
      current_period_end: periodEnd,
      event_time: new Date(eventMs).toISOString(),
      ...(typeof info.signedDate === "number" && Number.isFinite(info.signedDate) ? { event_tiebreak: new Date(info.signedDate).toISOString() } : {}),
      customer_id: `apple:${origTx}`,
    },
  };
}

/**
 * D-0955(9) (I19-W1E H3, built by I19-L01): Apple could not answer the SUBSCRIPTION rail (storeStatusTransient) or refused
 * OUR credentials (401/403 — an operator fault, logged here): TEMPORARY, never the buyer's refusal. The route
 * answers 503 (unavailable), not 402; the native clients treat any non-2xx as a failed purchase and do not retry
 * by themselves (ApiClient.kt, ApiClient.swift — JA-03), so the buyer retries. Its own class rather than iap-credits.ts's `AppleVerifyUnavailable`,
 * because iap-credits.ts imports this module (no cycle).
 */
export class AppleSubscriptionUnavailable extends Error {
  override name = "AppleSubscriptionUnavailable";
}
/**
 * JA-03 (I20-W1E, QUEUE #946 (h)): the ONE predicate every store door uses for "the store gave no answer about the
 * purchase" — 408 (request timeout), 425 (too early), 429 (rate limited) and every 5xx. TEMPORARY (the route's 503),
 * never the buyer's refusal (402). 401/403 are OUR credentials and each door logs them in its own arm; every other
 * status (400, 404, 410, ...) is the store's answer and stays a refusal. Callers: appleStoreUnanswered and
 * GOOGLE_TRANSIENT_STATUS here, verifyAppleCreditPurchase and verifyGoogleCreditPurchase in iap-credits.ts.
 */
export const storeStatusTransient = (status: number): boolean => status === 408 || status === 425 || status === 429 || status >= 500;

/** The credit door's status split (iap-credits.ts verifyAppleCreditPurchase): 401/403 logged, storeStatusTransient thrown, else null (a refusal). */
function appleStoreUnanswered(status: number): AppleSubscriptionUnavailable | null {
  if (status === 401 || status === 403) {
    logError({ cid: "unknown", hop: "http", msg: "iap_apple_store_auth_refused", rail: "apple", err: `status ${status}` });
    return new AppleSubscriptionUnavailable(`status ${status}`);
  }
  if (storeStatusTransient(status)) return new AppleSubscriptionUnavailable(`status ${status}`);
  return null;
}

/**
 * Validate an Apple StoreKit transaction id against the App Store Server API and
 * return a normalized grant. Rejects a product mismatch or a revoked/expired
 * transaction. Returns a non-ok grant on any validation failure. THROWS
 * `AppleSubscriptionUnavailable` when Apple cannot answer (storeStatusTransient: 408/425/429/5xx) or refuses our
 * credentials (401/403, logged) — D-0955(9), the I3-1138 shape the credit rail has —
 * and (I19-FXA RG-MONEY-03) when the call itself throws or hits its deadline (logged).
 */
export async function verifyAppleTransaction(cfg: AppleIapConfig, transactionId: string, expectedToken: string, nowSec: number, fetchImpl: typeof fetch = fetch): Promise<IapGrant> {
  const fail: IapGrant = { ok: false, event_id: "", status: "none", current_period_end: null, event_time: new Date(nowSec * 1000).toISOString(), customer_id: null };
  if (!/^[0-9]{1,32}$/.test(transactionId)) return fail; // Apple transaction ids are numeric
  if (!expectedToken) return fail; // no binding token → refuse (never grant unbound)
  // I17-FXA RGM-11: ONE deadline over this call's store requests (the transaction and, when lapsed, the renewal
  // status) — stripe.ts's idiom; an abort is a thrown fetch error, answered `fail` by the catch below.
  const ctl = new AbortController();
  const timer = setTimeout(() => ctl.abort(), STORE_FETCH_TIMEOUT_MS);
  try {
    const token = await appleApiToken(cfg, nowSec);
    const res = await fetchImpl(`${cfg.apiBase}/inApps/v1/transactions/${transactionId}`, { redirect: "manual", headers: { authorization: `Bearer ${token}` }, signal: ctl.signal });
    if (!res.ok) {
      const unanswered = appleStoreUnanswered(res.status);
      if (unanswered) throw unanswered;
      return fail;
    }
    const body = (await res.json()) as { signedTransactionInfo?: string };
    if (!body.signedTransactionInfo) return fail;
    // When a root-CA fingerprint is pinned, VERIFY the JWS cryptographically
    // (Apple's x5c chain → pinned root + leaf-signs-payload), so a forged
    // response can't grant even if the TLS channel were compromised. Absent a
    // pin, fall back to decoding the payload from the authenticated API call.
    const info = cfg.rootCaSha256 ? await verifyAppleJws(body.signedTransactionInfo, cfg.rootCaSha256, nowSec * 1000) : decodeJwsPayload(body.signedTransactionInfo);
    if (!info) return fail;
    // QUEUE #600: the transaction endpoint carries no renewal info, so a LAPSED,
    // unrevoked subscription asks the status endpoint for it before it may read
    // `expired` — otherwise a subscriber in Apple's billing grace who resubmits is
    // downgraded (the notification rail passes renewal at `verifyAppleNotification`).
    // A lapse this rail cannot confirm returns `fail`: applyIapGrant writes nothing.
    let renewal: Record<string, unknown> | null = null;
    const expiresMs = typeof info.expiresDate === "number" ? info.expiresDate : NaN;
    if (typeof info.revocationDate !== "number" && Number.isFinite(expiresMs) && expiresMs <= nowSec * 1000) {
      renewal = await appleRenewalInfo(cfg, token, info, transactionId, nowSec, fetchImpl, ctl.signal);
      if (!renewal) return fail;
    }
    const built = appleGrantFromInfo(info, cfg, transactionId, nowSec, renewal);
    if (!built) return fail;
    // Bind to the account: the purchase must carry THIS account's token.
    if (built.appAccountToken.toLowerCase() !== expectedToken.toLowerCase()) return fail;
    return built.grant;
  } catch (err) {
    // D-0955(9): a store that did not answer is TEMPORARY — rethrown for the route's 503, never a refusal.
    if (err instanceof AppleSubscriptionUnavailable) throw err;
    // #609d (I16): an infra fault (token signing, fetch, JSON, a throw inside
    // verifyAppleJws) is RECORDED first, as iap-credits.ts's credit-rail twin does (I3-1138).
    logError({ cid: "unknown", hop: "http", msg: "iap_apple_verify_error", rail: "apple", err: err instanceof Error ? err.name : typeof err });
    // I19-FXA (RG-MONEY-03, D-0956(1)): and it is TEMPORARY, never the buyer's refusal — a store that never
    // answered (a thrown fetch, the deadline's abort) said nothing about the purchase. The route answers 503
    // (index.ts AppleSubscriptionUnavailable arm), the credit door's mapping (iap-credits.ts:284-286).
    throw new AppleSubscriptionUnavailable(err instanceof Error ? err.name : typeof err);
  } finally {
    clearTimeout(timer);
  }
}

/**
 * The renewal info for `info`'s subscription, from Apple's status endpoint
 * (`/inApps/v1/subscriptions/{originalTransactionId}`), verified like the
 * transaction and bound to the SAME originalTransactionId. Null when it cannot be
 * confirmed. Used only by `verifyAppleTransaction` for a lapsed transaction.
 */
async function appleRenewalInfo(cfg: AppleIapConfig, token: string, info: Record<string, unknown>, transactionId: string, nowSec: number, fetchImpl: typeof fetch, signal: AbortSignal): Promise<Record<string, unknown> | null> {
  const origTx = typeof info.originalTransactionId === "string" ? info.originalTransactionId : transactionId;
  if (!/^[0-9]{1,32}$/.test(origTx)) return null;
  const res = await fetchImpl(`${cfg.apiBase}/inApps/v1/subscriptions/${origTx}`, { redirect: "manual", headers: { authorization: `Bearer ${token}` }, signal });
  if (!res.ok) {
    // D-0955(9): the SAME door's second store request (FAMILY-004 e) — unanswered is temporary here too.
    const unanswered = appleStoreUnanswered(res.status);
    if (unanswered) throw unanswered;
    return null;
  }
  const body = (await res.json()) as { data?: { lastTransactions?: { originalTransactionId?: unknown; signedRenewalInfo?: unknown }[] }[] };
  const last = (Array.isArray(body.data) ? body.data : []).flatMap((g) => (Array.isArray(g?.lastTransactions) ? g.lastTransactions : [])).find((t) => t?.originalTransactionId === origTx);
  if (!last || typeof last.signedRenewalInfo !== "string") return null;
  const renewal = cfg.rootCaSha256 ? await verifyAppleJws(last.signedRenewalInfo, cfg.rootCaSha256, nowSec * 1000) : decodeJwsPayload(last.signedRenewalInfo);
  if (!renewal || renewal.originalTransactionId !== origTx) return null;
  return renewal;
}

/**
 * Verify an App Store Server Notification V2 (`{ signedPayload }`) and return
 * the grant + the purchase's appAccountToken (so the caller can resolve which
 * account it belongs to). The signature IS the only authentication for this
 * server-to-server webhook, so a pinned root CA is REQUIRED — absent it, null.
 * Verifies BOTH the outer notification JWS and the inner signedTransactionInfo
 * JWS against Apple's chain → pinned root. Null on any failure; never throws.
 */
/**
 * I15-FX1 (RGM-06): `onRefuse`, when given, is told WHY a null came back: `signed` is
 * true when the OUTER notification JWS verified against the pinned root — the refusal
 * is then about the content (another product, an environment, an inner JWS, a
 * ceiling), not a forged request, and index.ts does not spend the failed-signature
 * budget on it. (A callback, not a second export: the route's paywall allowlist,
 * tests/invariants/paywall.test.ts, names this function.)
 */
export async function verifyAppleNotification(cfg: AppleIapConfig, signedPayload: string, nowSec: number, onRefuse?: (r: { signed: boolean }) => void): Promise<{ grant: IapGrant; appAccountToken: string } | null> {
  const v = await verifyAppleNotificationDetailed(cfg, signedPayload, nowSec);
  if (v.ok) return { grant: v.grant, appAccountToken: v.appAccountToken };
  onRefuse?.({ signed: v.signed });
  return null;
}

async function verifyAppleNotificationDetailed(
  cfg: AppleIapConfig,
  signedPayload: string,
  nowSec: number,
): Promise<{ ok: true; grant: IapGrant; appAccountToken: string } | { ok: false; signed: boolean }> {
  const unsigned = { ok: false as const, signed: false };
  if (!cfg.rootCaSha256) return unsigned; // no anchor → can't trust an unauthenticated POST
  if (typeof signedPayload !== "string" || signedPayload.length === 0) return unsigned;
  let payload: Record<string, unknown> | null;
  try {
    payload = await verifyAppleJws(signedPayload, cfg.rootCaSha256, nowSec * 1000);
  } catch {
    return unsigned;
  }
  if (!payload) return unsigned;
  const built = await appleNotificationGrant(cfg, payload, nowSec);
  return built ? { ok: true, ...built } : { ok: false, signed: true };
}

/** The grant a VERIFIED notification payload carries, or null (content refused). */
async function appleNotificationGrant(cfg: AppleIapConfig, payload: Record<string, unknown>, nowSec: number): Promise<{ grant: IapGrant; appAccountToken: string } | null> {
  if (!cfg.rootCaSha256) return null;
  try {
    const nowMs = nowSec * 1000;
    const data = payload.data as { signedTransactionInfo?: unknown; signedRenewalInfo?: unknown } | undefined;
    if (!data || typeof data.signedTransactionInfo !== "string") return null;
    const info = await verifyAppleJws(data.signedTransactionInfo, cfg.rootCaSha256, nowMs);
    if (!info) return null;
    // The configured Billing Grace Period rides on the RENEWAL info, not on the
    // transaction — see `appleGrantFromInfo`. Verified through the same chain and
    // the same pinned root as everything else on this route; absent or
    // unverifiable, the grant is built exactly as it was before.
    const renewal = typeof data.signedRenewalInfo === "string" ? await verifyAppleJws(data.signedRenewalInfo, cfg.rootCaSha256, nowMs) : null;
    const transactionId = typeof info.transactionId === "string" ? info.transactionId : String(info.originalTransactionId ?? "");
    if (!transactionId) return null;
    const built = appleGrantFromInfo(info, cfg, transactionId, nowSec, renewal);
    if (!built) return null;
    // Apple names every notification and REPEATS that name on every retry of it
    // — `notificationUUID` is the store's own idempotency key, so it is the one
    // to dedup on. Without it a REFUND collides with the purchase it reverses
    // (see stateChangeKey); with it, retries collapse and reversals land.
    // Absent or out of grammar, fall back to the state-change key rather than
    // trusting whatever came off the wire.
    // I3-1300: the NOTIFICATION's own signing instant orders two state changes of
    // one transaction (same purchaseDate) by when Apple reported them.
    if (typeof payload.signedDate === "number" && Number.isFinite(payload.signedDate)) {
      built.grant.event_tiebreak = new Date(payload.signedDate).toISOString();
    }
    const uuid = payload.notificationUUID;
    if (typeof uuid === "string" && NOTIFICATION_ID_RE.test(uuid)) {
      built.grant.event_id = notificationKey("apple", uuid);
    }
    return built;
  } catch {
    return null;
  }
}

/** I17-FXA RGM-11 (QUEUE #352b siblings): the deadline on every App Store / Play request in this file and in
 * iap-credits.ts — stripe.ts's TRANSFER_TIMEOUT_MS value, kept as this rail's own constant. */
export const STORE_FETCH_TIMEOUT_MS = 10_000;

/** Service-account access token for the androidpublisher scope (RS256 JWT-bearer
 * grant, same mechanism as FCM). Returns "" on failure. */
export async function googlePlayAccessToken(cfg: GooglePlayConfig, nowSec: number, fetchImpl: typeof fetch): Promise<string> {
  const assertion = await signJwt(cfg.privateKeyPem, "RS256", {}, {
    iss: cfg.clientEmail,
    scope: "https://www.googleapis.com/auth/androidpublisher",
    aud: cfg.tokenUri,
    iat: nowSec,
    exp: nowSec + 3600,
  });
  // I17-FXA RGM-11: stripe.ts's deadline idiom; an abort throws, as any failed fetch here always has (callers map it).
  const ctl = new AbortController();
  const timer = setTimeout(() => ctl.abort(), STORE_FETCH_TIMEOUT_MS);
  try {
    const res = await fetchImpl(cfg.tokenUri, { redirect: "manual",
      method: "POST",
      headers: { "content-type": "application/x-www-form-urlencoded" },
      body: `grant_type=${encodeURIComponent("urn:ietf:params:oauth:grant-type:jwt-bearer")}&assertion=${encodeURIComponent(assertion)}`,
      signal: ctl.signal,
    });
    if (!res.ok) {
      // I19-FXA (RG-MONEY-04): the token exchange refusing OUR service account (400 invalid_grant, 401, 403) or
      // failing (429/5xx) is an operator fault no other line records: both callers turn "" into an unavailable
      // error (GoogleTransientError here, GoogleCreditVerifyUnavailable in iap-credits.ts) and log nothing.
      logError({ cid: "unknown", hop: "http", msg: "iap_google_token_refused", rail: "google", err: `status ${res.status}` });
      return "";
    }
    const j = (await res.json()) as { access_token?: string };
    return j.access_token ?? "";
  } finally {
    clearTimeout(timer);
  }
}

/** Fetch + normalize a Google subscription from the Developer API. Returns the
 * grant + the purchase's obfuscatedExternalAccountId (`boundToken`) so the
 * caller can bind it (match the submitter, or resolve the account in the RTDN

/**
 * I3-0725: a Google answer that says nothing about the purchase — no access
 * token (our credentials or Google's token endpoint), 401/403 (credentials),
 * 429 or 5xx (Google) — as opposed to a definite miss. Thrown by

export class GoogleTransientError extends Error {
  override name = "GoogleTransientError";
}

/**
 * I19-FXA (RG-MONEY-02, D-0956(1)): the Google SUBSCRIPTION door could not get an answer about the purchase —
 * Play 401/403 (our credentials), storeStatusTransient (408/425/429/5xx), no access token, a thrown fetch, the deadline's
 * abort, an unreadable body. TEMPORARY, never the buyer's refusal: the route answers 503, not 402 (the native clients
 * treat both as a failed purchase and do not retry by themselves — JA-03), as the credit door
 * (`GoogleCreditVerifyUnavailable`) and the Apple subscription door (`AppleSubscriptionUnavailable`) do.
 */
export class GoogleSubscriptionUnavailable extends Error {
  override name = "GoogleSubscriptionUnavailable";
}

const GOOGLE_TRANSIENT_STATUS = (status: number): boolean => status === 401 || status === 403 || storeStatusTransient(status);

async function fetchGoogleGrant(cfg: GooglePlayConfig, purchaseToken: string, nowSec: number, fetchImpl: typeof fetch): Promise<{ grant: IapGrant; boundToken: string } | null> {
  if (typeof purchaseToken !== "string" || purchaseToken.length === 0 || purchaseToken.length > 1024) return null;
  const token = await googlePlayAccessToken(cfg, nowSec, fetchImpl);
  if (!token) throw new GoogleTransientError("no Play access token");
  const url = `https://androidpublisher.googleapis.com/androidpublisher/v3/applications/${encodeURIComponent(cfg.packageName)}/purchases/subscriptionsv2/tokens/${encodeURIComponent(purchaseToken)}`;
  // I17-FXA RGM-11: stripe.ts's deadline idiom over the request and its body; an abort throws (callers: transient/fail).
  const ctl = new AbortController();
  const timer = setTimeout(() => ctl.abort(), STORE_FETCH_TIMEOUT_MS);
  let res: Response;
  let raw: unknown;
  try {
    res = await fetchImpl(url, { redirect: "manual", headers: { authorization: `Bearer ${token}` }, signal: ctl.signal });
    raw = res.ok ? await res.json() : null;
  } finally {
    clearTimeout(timer);
  }
  if (!res.ok) {
    if (GOOGLE_TRANSIENT_STATUS(res.status)) throw new GoogleTransientError(`Play API ${res.status}`);
    return null;
  }
  const body = raw as {
    subscriptionState?: string;
    latestOrderId?: string;
    startTime?: string;
    testPurchase?: Record<string, unknown>;
    externalAccountIdentifiers?: { obfuscatedExternalAccountId?: string };
    lineItems?: Array<{ productId?: string; expiryTime?: string }>;
  };
  // A Play licence-tester subscription is FREE and is reported with every field a
  // paid one has — state, order id, product, expiry, binding token. `testPurchase`
  // is the only thing in this response that tells them apart, and nothing read it,
  // so a tester account granted production Premium. Fail closed, the stance this
  // file already takes on absent config and on an implausible period end.
  if (body.testPurchase) {
    logError({ cid: "unknown", hop: "http", msg: "google_iap_test_purchase_refused", rail: "google" });
    return null;
  }
  const items = body.lineItems ?? [];
  if (!items.some((li) => li.productId === cfg.productId)) {
    // See the Apple arm: a drift here is a paid renewal dropped, and the RTDN
    // route answers 200 on the null, so without this line there is no record at all.
    logError({ cid: "unknown", hop: "http", msg: "google_iap_product_mismatch", rail: "google" });
    return null; // not our product
  }
  let expiryMs = 0;
  for (const li of items) {
    const t = li.expiryTime ? Date.parse(li.expiryTime) : NaN;
    if (Number.isFinite(t)) expiryMs = Math.max(expiryMs, t);
  }
  // Same ceiling as Apple and as Stripe — one number, refused not clamped.
  // See `withinPaidAheadCeiling`.
  if (expiryMs > 0 && !withinPaidAheadCeiling(expiryMs, nowSec * 1000)) return null;
  const state = body.subscriptionState ?? "";
  // CANCELED in Play Billing means auto-renew was turned OFF — the user KEEPS
  // access until expiryTime. So CANCELED (like ACTIVE/IN_GRACE) still grants
  // "active", and `tierFor` downgrades only once the period end passes. Only
  // ON_HOLD / PAUSED / EXPIRED / PENDING actually revoke access now.
  const grantsAccess = state === "SUBSCRIPTION_STATE_ACTIVE" || state === "SUBSCRIPTION_STATE_IN_GRACE_PERIOD" || state === "SUBSCRIPTION_STATE_CANCELED";
  const orderId = body.latestOrderId ?? purchaseToken.slice(0, 40);
  // PV-01 ON THIS RAIL TOO (the Apple arm argues it at length above): ORDER on the
  // subscription's OWN anchor, never on our clock. Stamped with `Date.now()`, a
  // purchase token that a newer purchase has already superseded — replayed by the
  // app on launch, or by anyone holding an old receipt — always satisfied
  // billing.ts's ordering guard (SUB_UPSERT `WHERE excluded.last_event_at >=
  // subscriptions.last_event_at`) and overwrote the LIVE subscription with the dead
  // token's expired state. `startTime` is stable per subscription and strictly
  // earlier for the superseded token, so the guard now refuses it; a renewal or a
  // revocation on the SAME subscription carries the same `startTime`, ties on `>=`,
  // and still applies. Falls back to the clock when Play omits it.
  const startMs = body.startTime ? Date.parse(body.startTime) : NaN;
  const eventMs = Number.isFinite(startMs) ? startMs : nowSec * 1000;
  const status: SubStatus = grantsAccess ? "active" : "expired";
  const periodEnd = expiryMs > 0 ? new Date(expiryMs).toISOString() : null;
  return {
    boundToken: body.externalAccountIdentifiers?.obfuscatedExternalAccountId ?? "",
    grant: {
      ok: true,
      // SUBSCRIPTION_REVOKED reports the SAME latestOrderId as the purchase, so
      // keying on the order alone dropped it. See stateChangeKey.
      event_id: stateChangeKey("google", orderId, status, periodEnd),
      status,
      current_period_end: periodEnd,
      event_time: new Date(eventMs).toISOString(),
      customer_id: `google:${orderId}`,
    },
  };
}

/**
 * Validate a Google Play purchase token against the Developer API and return a
 * normalized grant, bound to the submitting account. Rejects a product mismatch
 * or a binding mismatch with a non-ok grant. THROWS `GoogleSubscriptionUnavailable`
 * (logged) when Play gave no answer about the purchase (I19-FXA RG-MONEY-02).
 */
export async function verifyGoogleSubscription(cfg: GooglePlayConfig, purchaseToken: string, expectedToken: string, nowSec: number, fetchImpl: typeof fetch = fetch): Promise<IapGrant> {
  const fail: IapGrant = { ok: false, event_id: "", status: "none", current_period_end: null, event_time: new Date(nowSec * 1000).toISOString(), customer_id: null };
  if (!expectedToken) return fail; // no binding token → refuse (never grant unbound)
  try {
    const r = await fetchGoogleGrant(cfg, purchaseToken, nowSec, fetchImpl);
    if (!r) return fail;
    if (r.boundToken !== expectedToken) return fail; // bind to the account
    return r.grant;
  } catch (err) {
    // #609d (I16): the Google twin of the Apple catch above — recorded first.
    logError({ cid: "unknown", hop: "http", msg: "iap_google_verify_error", rail: "google", err: err instanceof Error ? err.name : typeof err });
    // I19-FXA (RG-MONEY-02): every throw here is Play not answering (fetchGoogleGrant returns null for a definite
    // miss), so it is unavailable, never `fail` (402): iap-credits.ts:388-390's mapping.
    throw new GoogleSubscriptionUnavailable(err instanceof Error ? err.name : typeof err);
  }
}

/** RTDN path: re-validate a purchase token from a Real-time Developer
 * Notification against the authenticated Developer API (the notification itself
 * is just a trigger — the API is the source of truth). Returns the grant + the
 * account binding token so the caller resolves the account.
 *
 * I3-0725: TAGGED, never a bare null. `miss` is a definite answer (bad token,
 * not our product, a test purchase, an implausible period, Google says no such
 * purchase) and the route acks it; `transient` (no access token, a network
 * throw, 401/403 or storeStatusTransient, an unreadable body) is not an answer, and the route
 * answers 503 so Pub/Sub redelivers instead of the fan staying on free. */
export type GoogleNotificationVerdict = { kind: "ok"; grant: IapGrant; boundToken: string } | { kind: "miss" } | { kind: "transient"; detail: string };

export async function verifyGoogleNotification(
  cfg: GooglePlayConfig,
  purchaseToken: string,
  nowSec: number,
  fetchImpl: typeof fetch = fetch,
  /** Pub/Sub's `message.messageId` — stable across Pub/Sub's redeliveries of the
   * same RTDN and unique per notification, so it is the right dedup key when the
   * route has it. Absent, the grant falls back to the state-change key, which
   * still lets a REVOKED through where the bare order id did not. */
  notificationId?: string,
): Promise<GoogleNotificationVerdict> {
  try {
    const r = await fetchGoogleGrant(cfg, purchaseToken, nowSec, fetchImpl);
    if (!r) return { kind: "miss" };
    if (typeof notificationId === "string" && NOTIFICATION_ID_RE.test(notificationId)) {
      r.grant.event_id = notificationKey("google", notificationId);
    }
    return { kind: "ok", grant: r.grant, boundToken: r.boundToken };
  } catch (e) {
    return { kind: "transient", detail: e instanceof GoogleTransientError ? e.message : "fetch or parse threw" };
  }
}
