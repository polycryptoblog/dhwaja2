/**
 * WHAT MAY BE SAID ABOUT A THROW ON AN ERASURE PATH, AND NOTHING MORE.
 *
 * I3-0233 removed the `passport_id` FIELD from the two `logError` sites in this
 * file, because `ALLOWED_KEYS` (`log.ts:64-77`) does not carry it and a creator
 * who asked to be gone must not be named in a platform log erasure cannot reach.
 * The other half of the same object literal re-admitted it: `err` IS on
 * `ALLOWED_KEYS` (`log.ts:69`) and `redact` (`log.ts:146-147`) passes a string
 * through verbatim apart from a 200-character clamp — so `e.message` carried
 * whatever the thrower chose to say. On THIS path the thrower is the object
 * store, whose keys are `frozen/day/<passport_id>/<day>.json` by construction
 * (`:262`, `:341-345`), so a binding that names the key it failed on names the
 * creator.
 *
 * The class is the whole of what is loggable here: it is drawn from the error's
 * TYPE, not from any string the thrower composed, so no message — present,
 * future, or from a binding this repo does not own — can carry an identifier
 * through it. `log.ts` is not this room's to narrow; this closes the channel at
 * the only end that is.
 */
function errClass(e: unknown): string {
  return e instanceof Error ? (e.name || "Error") : typeof e;
}

/**
 * WHAT IS LEFT OF THE F12 BUILDER, AND WHY.
 *
 * F12 answered "the snapshot instant must belong to the system, not to whoever
 * opens the page" with a CRON that computed every interested creator's day at
 * an instant the schedule picked. ADR-0006 answers the same question at the
 * DATA MODEL instead: a day object contains exactly the cells that have SEALED,
 * so its bytes are a function of (creator, UTC day) and it does not matter WHO
 * computes it or WHEN, as long as the build grace has passed. Two builds at two
 * instants emit identical bytes; a read cannot move a number because there is
 * no number left that moves.
 *
 * So the whole scheduling apparatus retires (ADR section 1, MECHANISM):
 * `FROZEN_BATCH`, `FROZEN_SLOTS`, `slotOf`, `sweepIndexOf`, `rotateWindow`, the
 * FIRED / claim / DONE / rewrite markers, `FROZEN_CEILING_MARKER`,
 * `FROZEN_BUILD_REPORT_MARKER`, `analytics_interest` + `noteAnalyticsInterest`,
 * and `buildFrozenAnalytics` itself. Every one of them existed to ration or to
 * arbitrate a compute whose RESULT depended on when it ran. Nothing rations a
 * function of the day.
 *
 * What stays is the part that was never about the instant (CONTRACT):
 * `sweepFrozenSnapshots` (retention), `purgeFrozenAnalytics` (erasure, now
 * reaching the analytics store too), and ONE bounded pre-build - `buildSweep` -
 * whose only job is that an AGENCY, which may not build, still finds an object
 * to read.
 */

  list?(prefix: string): Promise<string[]>;

interface BuildSweepReport {
  /** `T - 1`: the day the sweep pre-builds. */
  day: string;
  /** Creators holding a live, report-capable agency grant, verified and listed. */
  granted: number;
  /** How many of them this sweep looked at (bounded by `limit`). */
  examined: number;
  written: number;
  failed: number;
  /** Granted creators this sweep REACHED and did not build because they were no
   * longer eligible at the write - retracted or suppressed after the claim
   * below was taken. Distinct from `failed` (the build threw) and from
   * `written`: a reader of this report must not read a drop as a build. */
  dropped: number;
  /** `granted - examined`: creators holding a live report-capable grant that
   * this tick's `limit` did NOT reach. A reader may conclude nothing else about
   * them — not that their day was built, and not that it was not; only that this
   * sweep did not look. Named rather than left derivable: the cron only logs
   * this object, and `buildSweep` records it for the console (D-0709). */
  unreached: number;
  /** "lease-lost" when this run no longer owns the cron tick. */
  skipped?: string;
  /** I13-FX7 (D-0891(1)): "budget" when the caller's subrequest budget stopped the loop before
   * `limit`; the report is still recorded, and `unreached` counts who it did not reach. */
  stopped?: "budget";
  /** I14-FX1 (RG-CAPS-08, D-0905(1)): creators whose build did not fit what was left of the cap
   * this tick (counted in `failed` too). Each is rotated BEHIND every creator examined that day. */
  over_budget?: number;
  /** I14-FX6 (JA-01, D-0907(1)): the longest run of consecutive over-budget TURNS any one live creator is on
   * (its own examined ticks, not the sweep's ticks — the rotation spaces a heavy creator's turns by the pool size);
   * BUILD_SWEEP_OVER_BUDGET_ALERT raises the alarm on the turn that reaches it. */
  over_budget_streak?: number;
  /** #925 B (D-0927(2)): creators whose build landed part of its served state and stopped short
   * of the object (analytics-read.ts `allowance`). NOT rotated and NOT over-budget: their cursor
   * stays before `generatedAtFor(day)`, so the next tick resumes them where this one stopped. */
  resumed?: number;
}

/** I14-FX6 (JA-01): what one tick saw per creator, for the per-creator over-budget streak (buildSweep). */
interface SweepTurns {
  /** The live granted cohort this tick, or undefined when the run stopped before reading it. */
  live?: Set<string>;
  /** Examined this tick and refused by the budget. */
  over: Set<string>;
  /** Examined this tick and NOT refused by the budget (built, found, dropped, or failed otherwise). */
  other: Set<string>;
}

/**
 * WHICH CREATORS AN AGENCY MAY READ - the live grant per (creator, agency).
 *
 * I3-2167 (D-0840(5)): read from `agency_report_grants` (0139), the projection
 * `events.ts foldEffects` writes in the batch of every `agency_authorized` /
 * `agency_revoked` append (later seq wins, scope NULL once revoked), so the
 * sweep reads one row per live pair instead of folding every grant event ever
 * appended. `scopeAllowsReports` is the SAME predicate the agency read gate
 * applies: a narrow `manage_accounts` grant does not authorize reports, so it
 * must not cause a pre-build either - otherwise the sweep would spend its
 * budget building dashboards nobody is allowed to open.
 */
async function agencyGrantedCreators(prod: DbLike): Promise<string[]> {
  const { results } = await prod
    .prepare("SELECT passport_id, scope FROM agency_report_grants WHERE scope IS NOT NULL")
    .bind()
    .all<{ passport_id: string; scope: string }>();
  const out = new Set<string>();
  for (const r of results) {
    if (!PSP_RE.test(r.passport_id) || !scopeAllowsReports(r.scope)) continue;
    out.add(r.passport_id);
  }
  return [...out].sort();
}

/**
 * WHO MAY HAVE A DAY BUILT FOR THEM - ONE PREDICATE, INTERPOLATED AT BOTH DOORS.
 *
 * The same SQL text is the cohort claim (`live`) and the per-creator re-check
 * immediately before the write. It is a const rather than two typings of the
 * same clause because a predicate applied at one door and re-typed at the other
 * is two predicates that drift (FAMILY-004 P1/(c)): the read gate asks exactly
 * this - verified, not retracted, not suppressed - so the build door must ask it
 * unchanged, or the sweep writes a bucket object nobody may open.
 */
const BUILD_ELIGIBLE_WHERE = `p.claim_state = 'verified' AND p.retracted = 0
            AND NOT EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = p.passport_id)`;

/**
 * PRE-BUILD UP TO `limit` MISSING (creator, T-1) OBJECTS, ROUND-ROBIN.
 *
 * WHY IT EXISTS AT ALL, given that the creator's own read builds. An AGENCY read
 * never builds (ADR section 1, O-18/C5): a grant authorizes reports, not the
 * right to spend the creator's compute or to choose an instant on their behalf.
 * Without this sweep the agency mirror would be dead for any creator who has not
 * opened their own dashboard since the day turned.
 *
 * ROUND-ROBIN, AND NEVER THE SAME PREFIX TWICE (A6). The order is
 * `frozen_builds.last_built_at` ASCENDING WITH NULLS FIRST - a creator who has
 * never been built goes first, then the least recently built. `buildDayIfMissing`
 * updates that cursor whether it BUILT or found the object already there, so a
 * creator examined this tick sorts to the back and the next tick reaches the
 * ones behind them. Ordering by passport id instead would have re-built the same
 * alphabetical prefix every six hours forever.
 *
 * THE BOUND IS ON CREATORS EXAMINED, not on builds attempted: a tick costs at
 * most `limit` skip-reads and `limit` builds, whatever the mix.
 *
 * AND IT ASKS WHETHER IT STILL OWNS THE TICK. `stillOwned` is the cron lease,
 * handed down from `scheduled()`; a false answer stops the sweep where it is.
 * Nothing needs repairing when it does - a day object is a function of the day,
 * so a successor rebuilds exactly the same bytes.
 */
async function runBuildSweep(
  prod: DbLike,
  analytics: DbLike,
  store: ConditionalSnapshotStore,
  nowMs: number,
  limit: number,
  stillOwned: () => Promise<boolean>,
  withinBudget: () => boolean,
  /** I14-FX1: the handle the over-budget rotation stamp goes through (counted, outside the refusal). */
  rotAnalytics: DbLike = analytics,
  /** I14-FX1: true when a build's throw was a budget refusal (buildSweep's meter, or the caller's share). */
  overBudget: (e: unknown) => boolean = () => false,
  /** I14-FX6 (JA-01): filled with this tick's per-creator outcomes. */
  turns: SweepTurns = { over: new Set(), other: new Set() },
  /** #925 B: the statements a build may still send (buildSweep's meter); absent = unbounded. */
  allowance?: () => number,
): Promise<BuildSweepReport> {
  const day = shiftDay(new Date(nowMs).toISOString().slice(0, 10), -1);
  const granted = await agencyGrantedCreators(prod);
  const report: BuildSweepReport = { day, granted: 0, examined: 0, written: 0, failed: 0, dropped: 0, unreached: 0 };
  // ONE exit, so the shortfall cannot be right on three return paths and absent
  // on the fourth.
  const done = (): BuildSweepReport => {
    report.unreached = Math.max(0, report.granted - report.examined);
    return report;
  };
  if (granted.length === 0) return done();

  // The READ is verified-gated and a delisted creator's dashboards stop being
  // served to them AND to their agency, so building one would be writing a fan
  // population into the bucket that nobody may open.
  const live: string[] = [];
  for (const chunk of inChunks(granted)) {
    const { results } = await prod
      .prepare(
        `SELECT p.passport_id AS passport_id FROM passports p
          WHERE p.passport_id IN (${chunk.map(() => "?").join(",")})
            AND ${BUILD_ELIGIBLE_WHERE}`,
      )
      .bind(...chunk)
      .all<{ passport_id: string }>();
    for (const r of results) live.push(r.passport_id);
  }
  report.granted = live.length;
  turns.live = new Set(live);
  if (live.length === 0) return done();

  const built = new Map<string, string>();
  for (const chunk of inChunks(live)) {
    const { results } = await analytics
      .prepare(`SELECT passport_id, last_built_at FROM frozen_builds WHERE passport_id IN (${chunk.map(() => "?").join(",")})`)
      .bind(...chunk)
      .all<{ passport_id: string; last_built_at: string }>();
    for (const r of results) built.set(r.passport_id, r.last_built_at);
  }
  const order = [...live].sort((a, b) => {
    const x = built.get(a);
    const y = built.get(b);
    if (x === y || (x === undefined && y === undefined)) return a < b ? -1 : a > b ? 1 : 0;
    if (x === undefined) return -1; // never built sorts first
    if (y === undefined) return 1;
    return x < y ? -1 : x > y ? 1 : a < b ? -1 : 1;
  });

  // I3-1958 (D-0837(2)): the SCHEDULED sweep repairs a (creator, day) object
  // proven not whole; a read's build never sets `repair` (analytics-read.ts).
  // #925 B (D-0927(2)): with the meter's allowance a heavy creator's day lands in turns.
  let partial = false;
  const onPartial = (): void => {
    partial = true;
  };
  const ctx: DayBuildContext = { analytics, prod, store, repair: true, ...(allowance === undefined ? {} : { allowance, onPartial }) };
  for (const psp of order.slice(0, Math.max(0, limit))) {
    // I13-FX7 (D-0891(1)): asked BEFORE the lease fence, which is itself a charged statement.
    if (!withinBudget()) {
      report.stopped = "budget";
      return done();
    }
    if (!(await stillOwned())) {
      report.skipped = "lease-lost";
      return done();
    }
    report.examined++;
    // THE CLAIM IS RE-ASKED AT THE ACT (FAMILY-008; INV-004 clause 3). `live`
    // above is ONE query over the whole cohort, taken before a loop that writes
    // one creator at a time, so every build after the first acts on a claim as
    // old as the loop: a creator who retracted, or whom an operator suppressed,
    // while the sweep was mid-cohort still had a day object of fan-derived
    // aggregates WRITTEN for them (`analytics-read.ts:757`). Re-asking the same
    // predicate here binds the write to the value it claims rather than to the
    // shape the cohort had minutes ago.
    //
    // IT NARROWS THE WINDOW; THE BUILD CLOSES IT (I3-2183, D-0653). The
    // object-store write is not in the D1 transaction, so a retraction
    // committed between this SELECT and `putIfAbsent` still lands an object;
    // `buildDayIfMissing` re-reads `retracted` AFTER that write and, with no open
    // legal hold, deletes the object there (`analytics-read.ts purgeIfRetracted`),
    // after re-filing the creator's erasure order. A retraction committing after
    // that re-read files its own order, whose drain runs after the object exists.
    const { results: eligible } = await prod
      .prepare(`SELECT p.passport_id AS passport_id FROM passports p WHERE p.passport_id = ? AND ${BUILD_ELIGIBLE_WHERE}`)
      .bind(psp)
      .all<{ passport_id: string }>();
    if (eligible.length === 0) {
      report.dropped++;
      turns.other.add(psp);
      continue;
    }
    turns.other.add(psp);
    partial = false;
    try {
      if (await buildDayIfMissing(ctx, psp, day)) report.written++;
      else if (partial) report.resumed = (report.resumed ?? 0) + 1;
    } catch (e) {
      report.failed++;
      // I14-FX1 (RG-CAPS-08, D-0905(1)): a build the cap (or the caller's share) refused is
      // SKIPPED for the tick — the loop goes on to the next creator while the budget still admits
      // one — and rotated BEHIND everyone examined today: its cursor takes this tick's instant, later
      // than every `generatedAtFor(day)` a build of `day` writes, so the next tick reaches the
      // creators behind it first instead of spending the tick on it again. Later days' stamps
      // overtake it, so it is retried, never wedged (alarm: buildSweep's over_budget_streak).
      if (overBudget(e)) {
        turns.other.delete(psp);
        turns.over.add(psp);
        report.over_budget = (report.over_budget ?? 0) + 1;
        await rotateBehind(rotAnalytics, psp, nowMs);
        logError({ cid: "unknown", hop: "cron", msg: "analytics_build_day_prebuild_over_budget", day, failed: report.failed });
        continue;
      }
      // THROUGH `log.ts`, NEVER `console` DIRECTLY. The passport id used to ride
      // this line as a field; `ALLOWED_KEYS` does not carry `passport_id`, so it
      // is dropped here rather than written into a platform log erasure has no
      // reach into. `failed` is the count a reader acts on, and `day` says which
      // day's pre-build is short. `err` is the error's CLASS and never its
      // message — see `errClass` for the channel that closes.
      logError({ cid: "unknown", hop: "cron", msg: "analytics_build_day_prebuild_failed", day, failed: report.failed, err: errClass(e) });
    }
  }
  return done();
}

/**

 * artifacts trigger (index.ts TAIL_ARTIFACTS_LEG_BUDGET), so the leg ends on a creator boundary
 * instead of being refused mid-creator by the trigger's meter; `unreached` carries the rest to the

 * the artifacts legs' caps sum under the trigger budget with ARTIFACTS_PURGE_RESERVE intact.
 */
export const BUILD_SWEEP_SUBREQUEST_CAP = 270;
/** The most one examined creator costs (the lease fence, its eligibility read, the day build).
 * D-0922(7) (I16-W05A): 44 measured for a FRESH build but 52 for a LATER day's build of an
 * agency-granted creator (it reads the prior day's carry and served state) — the production
 * state, which the old 50 never measured. 56 = 52 + the #842 in-build standing re-check (one
 * read, plus its purge on the rare retracted creator) + slack. test/i14-build-sweep-cap.test.ts
 * measures both and fails if any creator spends as much as this. */
export const BUILD_SWEEP_CREATOR_MAX = 56;
/** I14-FX1 (RG-CAPS-01, D-0905(1)): what the meter keeps back under `cap` for the calls a run owes
 * after its builds — the over-budget creator's rotation stamp, the prior report read (the alarm's
 * streak) and the report write. */
export const BUILD_SWEEP_RESERVE = 3;
/** I14-FX1 (RG-CAPS-08), I14-FX6 (JA-01): consecutive over-budget turns of ONE creator before the alarm. */
export const BUILD_SWEEP_OVER_BUDGET_ALERT = 3;

/** I14-FX1: a counted call buildSweep's meter refused because it would cross `cap - BUILD_SWEEP_RESERVE`. */
class SweepBudgetRefused extends Error {}

/** I14-FX1: the round-robin cursor of a creator whose build the budget refused, set to the tick's
 * instant so it sorts behind every creator built or found for `day` (analytics-read.ts noteBuilt). */
async function rotateBehind(analytics: DbLike, passportId: string, nowMs: number): Promise<void> {
  try {
    await analytics
      .prepare("INSERT OR REPLACE INTO frozen_builds (passport_id, last_built_at) VALUES (?, ?)")
      .bind(passportId, new Date(nowMs).toISOString())
      .run();
  } catch (e) {
    logError({ cid: "unknown", hop: "cron", msg: "analytics_build_sweep_rotate_failed", err: errClass(e) });
  }
}

/** I14-CAPS: counts what buildSweep sends — every D1 statement executed (a batch per statement),
 * every store call, and one per lease-fence call. */
function sweepMeter(prod: DbLike, analytics: DbLike, store: ConditionalSnapshotStore, stillOwned: () => Promise<boolean>, ceiling: number) {
  let n = 0;
  // I14-FX1 (RG-CAPS-01): REFUSED before it is sent, so what one creator costs can never carry the run past `ceiling`.
  const take = (k: number): void => {
    if (n + k > ceiling) throw new SweepBudgetRefused("buildSweep subrequest cap");
    n += k;
  };
  const real = new WeakMap<object, unknown>();
  const stmt = (s: object): unknown => {
    const p = new Proxy(s, {
      get(t, k) {
        const v = Reflect.get(t, k) as unknown;
        if (typeof v !== "function") return v;
        if (k === "bind") return (...a: unknown[]) => stmt((v as (...x: unknown[]) => object).apply(t, a));
        if (k === "first" || k === "all" || k === "run" || k === "raw") return (...a: unknown[]) => (take(1), (v as (...x: unknown[]) => unknown).apply(t, a));
        return (v as (...x: unknown[]) => unknown).bind(t);
      },
    });
    real.set(p, s);
    return p;
  };
  const db = (d: DbLike): DbLike => ({
    prepare: (sql: string) => stmt(d.prepare(sql) as unknown as object) as ReturnType<DbLike["prepare"]>,
    batch: (sts) => {
      take(sts.length);
      return d.batch(sts.map((x) => (real.get(x as unknown as object) ?? x) as typeof x));
    },
  });
  const counted = new Proxy(store, {
    get(t, k) {
      const v = Reflect.get(t, k) as unknown;
      return typeof v === "function" ? (...a: unknown[]) => (take(1), (v as (...x: unknown[]) => unknown).apply(t, a)) : v;
    },
  });
  return { prod: db(prod), analytics: db(analytics), store: counted, stillOwned: () => (take(1), stillOwned()), spent: () => n, add: (k: number): void => {
      n += k;
    },
  };
}

/** The `sweep_cursor` row `buildSweep` records its last report under
 * (QUEUE #161/#167 R-11, D-0709); `admin.ts overviewStats` reads it. */
export const BUILD_SWEEP_REPORT_ROW = "analytics_build_sweep";
/** The `sweep_cursor` row a THROWING `buildSweep` records (D-0817(3)); `admin.ts overviewStats` reads it. */
export const BUILD_SWEEP_FAILURE_ROW = "analytics_build_sweep_failed";

/**
 * `runBuildSweep`, then its report RECORDED where an operator reads it
 * (D-0709: the backlog reaches an operator-visible signal; the limit is not
 * raised). The cron only logged this object, so a backlog at `limit` reached no
 * reader. One `sweep_cursor` row (0026), overwritten per tick: `cursor_seq` is
 * `unreached`, `cursor_id` the report as JSON with the `limit` it ran under.
 * Best-effort: a failed write is logged and never changes the sweep's result. A
 * lease-lost run records nothing - its successor owns the tick's report.
 */
export async function buildSweep(
  prod: DbLike,
  analytics: DbLike,
  store: ConditionalSnapshotStore,
  nowMs: number = Date.now(),
  limit: number = BUILD_SWEEP_LIMIT,
  stillOwned: () => Promise<boolean> = async () => true,
  /** I13-FX7 (D-0891(1)): false stops the sweep before the next creator (the cron's metered artifacts budget). */
  withinBudget: () => boolean = () => true,
  /** I14-CAPS (QUEUE #912): this run's own subrequest cap, report write included. I67 O5: `lease`, the artifacts leg's (index.ts), binds the report write (CursorLease). */
  cap: number = BUILD_SWEEP_SUBREQUEST_CAP, lease?: CursorLease,
): Promise<BuildSweepReport> {
  let report: BuildSweepReport;
  // I14-CAPS: a creator starts only while a whole creator (BUILD_SWEEP_CREATOR_MAX) and the report
  // write (1) still fit under `cap`; the stop is the existing `stopped: "budget"`.
  // I14-FX1 (RG-CAPS-01): the meter also REFUSES any call past `cap - BUILD_SWEEP_RESERVE`, so the
  // estimate only decides whether to start; the leg's real spend never passes `cap`.
  const m = sweepMeter(prod, analytics, store, stillOwned, cap - BUILD_SWEEP_RESERVE);
  // The rotation stamp is one of the two reserved calls: counted, never refused by our own meter.
  const rotAnalytics: DbLike = { prepare: (sql: string) => (m.add(1), analytics.prepare(sql)), batch: (sts) => analytics.batch(sts) };
  const turns: SweepTurns = { over: new Set(), other: new Set() };
  try {
    report = await runBuildSweep(m.prod, m.analytics, m.store, nowMs, limit, m.stillOwned, () => withinBudget() && m.spent() + BUILD_SWEEP_CREATOR_MAX + 1 <= cap, rotAnalytics, (e) => e instanceof SweepBudgetRefused || !withinBudget(), turns, () => cap - BUILD_SWEEP_RESERVE - m.spent());
  } catch (e) {
    // D-0817(3) (RG-OPS-161-r11-throw): a THROWING sweep is recorded — its time
    // and the error's CLASS only (`errClass`: never the message, which can carry
    // row data) — so the overview shows the failure instead of silently the last
    // good report. Best-effort, and the throw still propagates to the cron step.
    try {
      await prod
        .prepare("INSERT INTO sweep_cursor (name, cursor_seq, cursor_id, updated_at) VALUES (?, 0, ?, ?) ON CONFLICT(name) DO UPDATE SET cursor_seq = excluded.cursor_seq, cursor_id = excluded.cursor_id, updated_at = excluded.updated_at")
        .bind(BUILD_SWEEP_FAILURE_ROW, JSON.stringify({ err: errClass(e) }), new Date(nowMs).toISOString())
        .run();
    } catch (w) {
      logError({ cid: "unknown", hop: "cron", msg: "analytics_build_sweep_failure_unrecorded", err: errClass(w) });
    }
    throw e;
  }
  if (report.skipped === undefined) {
    // I14-FX1 (RG-CAPS-08, D-0905(1)), I14-FX6 (JA-01, D-0907(1)): the alarm after repeated failures —
    // ONE creator's consecutive over-budget TURNS (ticks that examined it), carried per creator in the report
    // row the overview already reads (`over_budget_turns`; same read, same write). Counting the SWEEP's ticks
    // never fired in a real pool: the rotation puts a refused creator behind everyone else, so its next turn
    // comes about one pool-length of ticks later and every tick between reset the count. A creator's count
    // grows only on its own refused turn, is removed on any other turn it is examined, and is dropped once it
    // leaves the live cohort (retracted, suppressed, grant revoked, erased).
    let prior: Record<string, number> = {};
    try {
      const row = await prod.prepare("SELECT cursor_id FROM sweep_cursor WHERE name = ?").bind(BUILD_SWEEP_REPORT_ROW).first<{ cursor_id: string | null }>();
      const parsed = JSON.parse(row?.cursor_id ?? "{}") as { over_budget_turns?: unknown };
      const t = parsed.over_budget_turns;
      if (t !== null && typeof t === "object" && !Array.isArray(t)) {
        for (const [k, v] of Object.entries(t as Record<string, unknown>)) if (typeof v === "number" && Number.isInteger(v) && v > 0) prior[k] = v;
      }
    } catch {
      prior = {};
    }
    const streaks: Record<string, number> = {};
    for (const [psp, n] of Object.entries(prior)) if (!turns.other.has(psp) && !turns.over.has(psp) && (turns.live === undefined || turns.live.has(psp))) streaks[psp] = n;
    let alarmed = 0;
    for (const psp of turns.over) {
      streaks[psp] = (prior[psp] ?? 0) + 1;
      if (streaks[psp] >= BUILD_SWEEP_OVER_BUDGET_ALERT) alarmed++;
    }
    report.over_budget_streak = Math.max(0, ...Object.values(streaks));
    if (alarmed > 0) {
      logError({ cid: "unknown", hop: "cron", msg: "analytics_build_sweep_over_budget", count: report.over_budget_streak, failed: alarmed, day: report.day });
    }
    try {
      await (lease // I67 O5: with the leg's lease, written only while that lease row (name, holder) is this run's
        ? prod.prepare("INSERT INTO sweep_cursor (name, cursor_seq, cursor_id, updated_at) SELECT ?, ?, ?, ? WHERE EXISTS (SELECT 1 FROM cron_leases WHERE name = ? AND holder = ?) ON CONFLICT(name) DO UPDATE SET cursor_seq = excluded.cursor_seq, cursor_id = excluded.cursor_id, updated_at = excluded.updated_at")
        : prod.prepare("INSERT INTO sweep_cursor (name, cursor_seq, cursor_id, updated_at) VALUES (?, ?, ?, ?) ON CONFLICT(name) DO UPDATE SET cursor_seq = excluded.cursor_seq, cursor_id = excluded.cursor_id, updated_at = excluded.updated_at")
      ).bind(BUILD_SWEEP_REPORT_ROW, report.unreached, JSON.stringify({ ...report, limit, over_budget_turns: streaks }), new Date(nowMs).toISOString(), ...(lease ? [lease.name, lease.holder] : [])).run();
    } catch (e) {
      logError({ cid: "unknown", hop: "cron", msg: "analytics_build_sweep_report_unrecorded", err: errClass(e) });
    }
  }
  return report;
}

/**
 * Everything this feature holds about ONE creator, removed.
 *
 * The presence or absence of a day object across the retention window is a
 * day-resolution record of which creators used analytics on which days - finer
 * grained and longer lived than any row the migrations describe - so erasure has
 * to reach the bucket. It now also reaches the SECOND STORE (ADR section 1): a
 * retracted creator's rollups, membership rows, tombstones and sentinels are the
 * same fan-derived aggregates under a different binding, and leaving them would
 * mean the copy outlived the original (INV-004 clause 3).
 */
export async function purgeFrozenAnalytics(db: DbLike, store: ConditionalSnapshotStore, passportId: string, analytics?: DbLike, leaseLost: () => boolean = () => false): Promise<number> {
  if (!PSP_RE.test(passportId)) return 0;
  // I40-FX2-LEASE (RGOPS-03, I40TER-P8): the caller's lease, read before each delete below; lost -> no further delete.
  if (leaseLost()) return 0;
  // I29-FX10 (D-1047(1)): both callers skip a passport an open legal hold names before calling (the drain's read below,
  // sweepFrozenSnapshots' lap query) — the analytics DB and the bucket hold no legal_holds, and the lap's per-creator cost
  // is pinned (i13-fx1-artifacts-budget.test.ts), so no read is added here; this prod statement re-asks it.
  // The retired interest table: nothing writes it, and what a deployed database
  // still holds for this creator goes with the rest of their record.
  await db.prepare(`DELETE FROM analytics_interest WHERE passport_id = ? AND NOT ${PASSPORT_LEGAL_HOLD_SQL}`).bind(passportId, passportId).run();
  if (leaseLost()) return 0;
  if (analytics !== undefined) await purgeCreatorAnalytics(analytics, passportId);
  if (typeof store.list !== "function" || typeof store.delete !== "function") {
    // FINDING-0019: this used to return 0 indistinguishable from "this creator
    // left nothing behind." A capability gap on the store is an erasure
    // FAILURE, not an empty result, and it must say so — the object the
    // caller believed was gone is still there.
    // FINDING-0019's requirement is that this SAYS so, and it still does. The
    // passport id is gone from it: a store with no `list`/`delete` has none for
    // EVERY passport on the tick, so the identifier named no population the
    // message does not, and it is the one field an erased creator's record must
    // not leave behind in a log (`log.ts ALLOWED_KEYS` carries no `passport_id`).
    logError({ cid: "unknown", hop: "cron", msg: "purge_frozen_analytics_store_capability_missing" });
    return 0;
  }
  let removed = 0;
  // `frozen/day` FIRST - that is where a creator's fan-population aggregates
  // live. The per-kind prefixes, `intel` and `regions` are LEGACY: nothing writes
  // them any more, and they stay in this loop until retention has cleared
  // whatever a deployed bucket already holds. `<prefix>/<psp>/` reaches the
  // object because the passport is the second segment of its key (`bundleKey`).
  for (const prefix of OBJECT_PREFIXES) {
    for (const k of await store.list(`${prefix}/${passportId}/`)) {
      if (leaseLost()) return removed;
      await store.delete(k);
      removed++;
    }
  }
  // THE LEGACY MARKERS TOO. They carry no fan data, but
  // `frozen/built/<day>/<psp>.json` present across the retention window is the
  // same day-resolution record of which creators used analytics on which days,
  // and it names the passport in the key. The loop above cannot reach them: its
  // shape is `<prefix>/<psp>/` and the passport segment of these is last.
  for (const prefix of LEGACY_MARKER_PREFIXES) {
    for (const k of await store.list(prefix)) {
      if (!k.endsWith(`/${passportId}.json`)) continue;
      if (leaseLost()) return removed;
      await store.delete(k);
      removed++;
    }
  }
  return removed;
}

/**
 * #166 (D-0925(4), D-0933(3); I16-FXD H2's leg body): drain ONE per-creator purge work order — the oldest
 * `analytics_creator_erasures` row (written in the retraction's own batch, events.ts retract_passport) — through
 * `purgeFrozenAnalytics`, and delete the order only AFTER the purge returns, so a throw leaves it for the next tick.
 * The caller (the analytics tick) decides whether its budget allows a call; this does at most one creator. Returns the
 * creator drained (null when there was no order) and the objects removed.
 * #166(b) (I19-W1D, JA-14): a purge that THROWS moves its order's `requested_at` to `now()` before rethrowing, so the
 * next-oldest order is the next tick's and the failing creator is retried behind it — never starving the queue. The
 * UPDATE is fenced on the `requested_at` this drain READ (a re-file during the purge wins) and moves forward only.
 */
export async function drainCreatorErasureOrder(
  prod: DbLike,
  store: ConditionalSnapshotStore,
  analytics?: DbLike,
  now: () => string = () => new Date().toISOString(),
  /** I40-DIGEST (RGOPS-03): the caller's lease, read before every write below; lost -> no further write, the order stays. */
  leaseLost: () => boolean = () => false,
): Promise<{ passportId: string | null; removed: number }> {
  const order = await prod
    // I29-FX10 (D-1047(1)): an order whose passport an open legal hold names waits (it is not read, so it neither runs nor
    // blocks the next one) and is drained on the first tick after release; every delete below re-asks the same predicate.
    .prepare(`SELECT passport_id, requested_at FROM analytics_creator_erasures WHERE NOT ${legalHoldOpenSql("passport", "analytics_creator_erasures.passport_id")} ORDER BY requested_at, passport_id LIMIT 1`)
    .first<{ passport_id: string; requested_at: string }>();
  if (order === null) return { passportId: null, removed: 0 };
  let removed: number;
  try {
    if (leaseLost()) return { passportId: null, removed: 0 };
    // I3-0859 b (I22): the creator's likeness prompts (`likeness_jobs`, keyed by passport only — no account owns
    // them) go with the retraction. Here, once per order (every retraction files one, events.ts retract_passport),
    // not in `purgeFrozenAnalytics`, whose per-creator statements the retracted lap's reserve is sized on; no job is
    // minted for a retracted passport (likeness.ts refuses it), and migration 0208 cleared those retracted before.
    await prod.prepare(`DELETE FROM likeness_jobs WHERE passport_id = ? AND NOT ${PASSPORT_LEGAL_HOLD_SQL}`).bind(order.passport_id, order.passport_id).run();
    if (leaseLost()) return { passportId: null, removed: 0 };
    // I3-0220 (I23-W1-PROFILE H6): the creator's bio/links/avatar, their hide state, change history and the anonymous
    // reports against them (0214, creator-authored text) go with the retraction, in ONE batch, once per order.
    await prod.batch(
      ["creator_profiles", "creator_profile_moderation", "creator_profile_history", "creator_profile_reports"].map((t) =>
        prod.prepare(`DELETE FROM ${t} WHERE passport_id = ? AND NOT ${PASSPORT_LEGAL_HOLD_SQL}`).bind(order.passport_id, order.passport_id),
      ),
    );
    // I29-FX10 (D-1047(1)): a hold opened after the order was read stops the analytics-DB and bucket purge too (neither
    // holds legal_holds); the order stays (its delete below re-asks the hold) and is drained after release.
    if (leaseLost()) return { passportId: null, removed: 0 };
    removed = (await passportUnderLegalHold(prod, order.passport_id)) ? 0 : await purgeFrozenAnalytics(prod, store, order.passport_id, analytics, leaseLost);
  } catch (e) {
    const at = now();
    try {
      if (!leaseLost()) await prod
        .prepare("UPDATE analytics_creator_erasures SET requested_at = ? WHERE passport_id = ? AND requested_at = ? AND requested_at < ?")
        .bind(at, order.passport_id, order.requested_at, at)
        .run();
    } catch {
      // the purge's own error is the one the tick logs; an order left in place is retried as before
    }
    throw e;
  }
  // FAMILY-003: a store that cannot list/delete purged nothing (purgeFrozenAnalytics logs it) — the order stays.
  if (typeof store.list !== "function" || typeof store.delete !== "function") return { passportId: order.passport_id, removed };
  if (leaseLost()) return { passportId: null, removed };
  // FAMILY-008: bound to the order this drain READ (its requested_at), so an order re-filed during the purge
  // (analytics-read.ts purgeIfRetracted moves requested_at forward) survives for the next tick.
  // I29-FX10: a hold opened during the purge keeps the order, so the purge it stopped runs after release.
  await prod
    .prepare(`DELETE FROM analytics_creator_erasures WHERE passport_id = ? AND requested_at = ? AND NOT ${PASSPORT_LEGAL_HOLD_SQL}`)
    .bind(order.passport_id, order.requested_at, order.passport_id)
    .run();
  return { passportId: order.passport_id, removed };
}

/**
 * I13-FX10 (D-0892(1)): the `sweep_cursor` row naming the last retracted creator the purge in
 * `sweepFrozenSnapshots` finished (`cursor_id`). The purge runs under a share of the metered
 * artifacts budget that covers only some creators a tick, so each tick starts AFTER this creator
 * and wraps: every retracted creator is reached within ceil(retracted / per-tick) + 1 ticks.
 */
export const FROZEN_PURGE_CURSOR = "frozen_purge_cursor";

/** I67 O5 (QUEUE #1020, I66-ERASE-P5-SIB): the cron lease a sweep_cursor write is bound to (cron-lease.ts CronLease's shape).
 * buildSweep's report write and sweepFrozenSnapshots' cursor write, given one, land only while that lease row (name,
 * holder) is this run's, lapsed or not: a successor's acquire rewrites the holder and its release deletes the row, so a
 * stale pass writes nothing (index.ts writeSweepCursorRow's guard). */
export type CursorLease = { readonly name: string; readonly holder: string };
/** I67 O5: sweepFrozenSnapshots' rotation cursor write (cursor_seq kept on conflict), bound to `lease` when one is given.
 * Outside the sweep's body so i13-cron-reachability does not read the guard's `cron_leases` as a sweep of that table. */
function writeFrozenPurgeCursor(db: DbLike, psp: string, at: string, lease?: CursorLease): Promise<unknown> {
  return (lease
    ? db.prepare("INSERT INTO sweep_cursor (name, cursor_seq, cursor_id, updated_at) SELECT ?, 0, ?, ? WHERE EXISTS (SELECT 1 FROM cron_leases WHERE name = ? AND holder = ?) ON CONFLICT(name) DO UPDATE SET cursor_id = excluded.cursor_id, updated_at = excluded.updated_at")
    : db.prepare("INSERT INTO sweep_cursor (name, cursor_seq, cursor_id, updated_at) VALUES (?, 0, ?, ?) ON CONFLICT(name) DO UPDATE SET cursor_id = excluded.cursor_id, updated_at = excluded.updated_at")
  ).bind(FROZEN_PURGE_CURSOR, psp, at, ...(lease ? [lease.name, lease.holder] : [])).run();
}

/** I23 (I3-1021): the retracted-passport population above which `sweepFrozenSnapshots` alarms (census: ~5,000). */
export const RETRACTED_LAP_ALARM = 5000;
export const RETRACTED_LAP_ALARM_MARKER = "frozen_purge_lap_over_threshold";

/**
 * Drop day objects past the retention window.
 *
 * THE ENUMERATION MOVED OFF `analytics_interest` (ADR section 1). The retracted
 * and delisted creators used to be found by joining the interest table, which
 * only ever held creators who had READ - so a creator whose objects were
 * pre-built for an agency and who then retracted was invisible to this sweep.
 * The predicate is now the two tables that actually say "gone", which is also
 * what the read gate asks.
 */
export async function sweepFrozenSnapshots(
  db: DbLike,
  store: ConditionalSnapshotStore,
  nowMs: number = Date.now(),
  analytics?: DbLike,
  /**
   * Called once per retracted creator whose purge THREW, with that creator's
   * passport id. Optional so every existing caller is unchanged, and the return
   * stays the `removed` count the retention tests read. The one production door is
   * the artifacts leg in `index.ts` (`recordUnerasedSubject` per id it collects).
   */
  onSubjectFailed?: (passportId: string) => void,
  /**
   * I13-FX11 (FX10-lead1): true for a throw that is the caller's BUDGET refusing a statement (index.ts passes
   * `e instanceof TailBudgetExhausted` from the metered artifacts leg). Such a throw is not a failure: the lap
   * stops, the creator it hit and every one after it are deferred to the next tick (the cursor still names the
   * last creator finished); the retention phase still runs (I13-FX12, `onRetention`), and a budget refusal inside
   * it ends retention there. Absent: every throw is a failure, as before.
   */
  budgetStop?: (e: unknown) => boolean,
  /**
   * I13-FX12 (D-0894(3)): called once just before the retention phase. The artifacts leg in `index.ts` moves its
   * budget share to the retention reserve here, so retention runs every tick whatever the retracted backlog.
   */
  onRetention?: () => void,
  /** I40-FX2-LEASE (I40TER-P8): the running sweep's lease, handed to every `purgeFrozenAnalytics` of the lap. */
  leaseLost: () => boolean = () => false,
  /** I67 O5: the artifacts leg's lease (index.ts); with it the rotation cursor write is bound to it (CursorLease). */
  lease?: CursorLease,
): Promise<number> {
  const cutoff = new Date(nowMs - SNAPSHOT_RETENTION_DAYS * DAY_MS).toISOString().slice(0, 10);
  // RETRACTED PASSPORTS FIRST, whatever their age: their objects are per-creator
  // fan-population aggregates, and "wait thirty days" is not an answer to
  // somebody who asked to be gone.
  //
  // `request_suppressions` IS NOT IN THIS SET, AND THAT IS THE POINT. It used to
  // be a second UNION arm here, which put a REVERSIBLE operator toggle in front
  // of an IRREVERSIBLE consequence: `setRequestSuppression` INSERTs the row
  // (`requests.ts:164`) and DELETEs it again (`:173`), and `UNSUPPRESS_WARNING`
  // (`requests.ts:123`) is the declared ceremony for lifting a removal — but
  // `purgeFrozenAnalytics` below `store.delete`s the objects, so the lift
  // restored the listing and could never restore the history. `retracted` is the
  // monotone half: nothing in `src/` sets it back to 0 (`events.ts:1822` is the
  // only writer), so its consequence may be irreversible too.
  //
  // DELETION IS NOT WHAT KEEPS A SUPPRESSED CREATOR'S ANALYTICS UNSERVED. Every
  // door onto these objects re-asks `request_suppressions` LIVE, per request,
  // through `passportStanding` (`standing.ts:252`): the nine creator-side doors
  // (`intelligence-routes.ts:152,186,298,340,364,399,424,439`) and the agency
  // gate (`agency-analytics-routes.ts:104`), both via `isVerifiedPassport`
  // (`intelligence.ts:336`). The build door asks it too, in this file
  // (`buildSweep`'s `live` query below). Dropping this arm removes an
  // irreversible side effect and removes no guarantee.
  // ROTATION (I13-FX10, D-0892(1)): creators after the cursor first, then the wrap. Unreadable -> ''.
  let after = "";
  try {
    after = (await db.prepare("SELECT cursor_id FROM sweep_cursor WHERE name = ?").bind(FROZEN_PURGE_CURSOR).first<{ cursor_id: string }>())?.cursor_id ?? "";
  } catch {
    after = "";
  }
  const { results: gone } = await db
    // I29-FX10 (D-1047(1)): a passport an open legal hold names is not in the lap; the first lap after release purges it.
    .prepare(`SELECT passport_id FROM passports WHERE retracted = 1 AND NOT ${legalHoldOpenSql("passport", "passports.passport_id")} ORDER BY passport_id <= ?, passport_id`)
    .bind(after)
    .all<{ passport_id: string }>();
  // I23 (I3-1021, D-0993(1)): the lap re-purges EVERY retracted passport each tick — kept as the safety net for
  // late objects (i13-fx10/fx11/fx1, D-0653, D-0892(1)) — so its cost grows with history. Past the threshold the
  // census pick revisits a purged marker; until then the population is an alarm, level-triggered every tick.
  if (gone.length > RETRACTED_LAP_ALARM) logError({ cid: "unknown", hop: "cron", msg: RETRACTED_LAP_ALARM_MARKER, count: gone.length });
  let purged = 0;
  let purgeFailed = 0;
  let deferred = 0;
  let stoppedByBudget = false;
  // PER PASSPORT, NOT PER SWEEP. This loop is the only erasure reach into the
  // bucket, and erasure is the irreversible direction: an unhandled throw on one
  // passport used to abandon every retracted creator BEHIND it, leaving their
  // day objects in place with nothing anywhere saying they were skipped. The
  // failure is recorded and the sweep continues to the next subject.
  for (const [i, { passport_id: psp }] of gone.entries()) {
    try {
      purged += await purgeFrozenAnalytics(db, store, psp, analytics, leaseLost);
    } catch (e) {
      // I13-FX11: the budget, not the creator — this creator and the rest of the lap are next tick's.
      if (budgetStop?.(e) === true) {
        deferred = gone.length - i;
        stoppedByBudget = true;
        break;
      }
      purgeFailed++;
      // `err` is the error's CLASS, never its message (`errClass`): the store's
      // keys name the creator, and this is the ERASURE path.
      logError({ cid: "unknown", hop: "cron", msg: "purge_frozen_analytics_failed", failed: purgeFailed, err: errClass(e) });
      // AND THE SUBJECT GOES TO THE CALLER, NOT TO THE LOG. The comment above
      // this loop promises the failure is "recorded"; `purgeFailed` is a COUNT,
      // and a count does not tell an operator WHICH retracted creator still has
      // day objects in the bucket, so erasure could not be re-run for them.
      // The id cannot go where the count goes — that is the whole of I3-0233 and
      // of `errClass` — so it goes to the caller in process, inside the
      // erasure perimeter, and the caller decides what may hold it.
      onSubjectFailed?.(psp);
      continue;
    }
    // I41-LEASE (I40FX2L-03): a lease lost during this creator's purge left it unfinished: no cursor write, and the lap ends.
    if (leaseLost()) break;
    // The cursor moves only past a FINISHED creator; a failed one is revisited next lap.
    try {
      await writeFrozenPurgeCursor(db, psp, new Date(nowMs).toISOString(), lease);
    } catch (e) {
      // I13-FX11: a budget refusal of the cursor write is the end of the lap; this creator is redone next tick.
      if (budgetStop?.(e) === true) {
        deferred = gone.length - i - 1;
        stoppedByBudget = true;
        break;
      }
      logError({ cid: "unknown", hop: "cron", msg: "frozen_purge_cursor_unrecorded", err: errClass(e) });
    }
  }
  // THE ERASURE PHASE'S OWN NUMBERS, AND THEY REACH BOTH RETURN PATHS. `purged`
  // and `purgeFailed` used to end here: this function returns the RETENTION
  // count, so a tick that erased four hundred objects for a retracted creator
  // and a tick that erased none were the same integer to the only caller
  // (`index.ts:4174`). Emitted after the loop and before either `return`, so
  // neither path drops it.
  //
  // COUNTERS ONLY. `subjects`, `count` and `failed` are on `log.ts ALLOWED_KEYS`
  // (`log.ts:64-127`) and `passport_id` is not - this is the erasure path, and
  // an erased creator must not be named in a platform log erasure cannot reach
  // (I3-0233, and see `errClass`).
  // I13-FX11: a lap the budget stopped says so (`stopped`, `deferred` = creators not reached); I13-FX12: retention still runs below.
  if (gone.length > 0) logInfo({ cid: "unknown", hop: "cron", msg: "sweep_frozen_snapshots_purged", subjects: gone.length, count: purged, failed: purgeFailed, ...(stoppedByBudget ? { stopped: "budget", deferred } : {}) });
  // I13-FX12 (D-0894(3)): a budget-stopped lap no longer skips retention. Retention runs on its own reserve
  // (`onRetention`); a budget refusal inside it ends it there and keeps what it already deleted (`done`).
  onRetention?.();
  const done = { removed: 0 };
  try {
    return await sweepRetention(db, store, nowMs, cutoff, done, { subjects: gone.length, count: purged, failed: purgeFailed });
  } catch (e) {
    if (budgetStop?.(e) !== true) throw e;
    logInfo({ cid: "unknown", hop: "cron", msg: "sweep_frozen_snapshots_retention_stopped", stopped: "budget", count: done.removed });
    return done.removed;
  }
}

/** The retention phase of `sweepFrozenSnapshots`: the retired interest table, then every object and legacy marker past `cutoff`. */
async function sweepRetention(db: DbLike, store: ConditionalSnapshotStore, nowMs: number, cutoff: string, done: { removed: number }, erasure: { subjects: number; count: number; failed: number }): Promise<number> {
  // The retired interest table, emptied by age. I34-FX2-ALARM (legal-hold census): retention keeps what an open legal hold
  // on a passport names (its interest rows here, its per-creator objects below), as the erasure purge does (D-1047(1));
  // both go on the first tick after the hold is released.
  await db
    .prepare(`DELETE FROM analytics_interest WHERE last_read_at < ? AND NOT ${legalHoldOpenSql("passport", "analytics_interest.passport_id")}`)
    .bind(new Date(nowMs - INTEREST_WINDOW_DAYS * DAY_MS).toISOString())

/**
 * I23 (I3-2181 a, D-0993(1)): how many declared relations the day's RAW numbers
 * break — a partition whose residual (`other`, or `parent − Σ children` when it
 * has none) is negative or is not `parent − Σ children`, a subset whose inner
 * exceeds its outer. A relation naming a cell in `skip` (an unsealed window,
 * carried as 0 — `analytics-read.ts unsealedWindowIds`) or a cell the day does
 * not carry is not checked. A COUNT for a metric, never a throw: `serveDay`
 * rests on these relations holding, so a break is an operator's to see.
 */
export function relationMismatches(cells: readonly Cell[], graph: CellGraph, skip: ReadonlySet<string> = new Set()): number {
  const vals = new Map(cells.map((c) => [c.id, c.value]));
  const checkable = (ids: readonly string[]): boolean => ids.every((id) => vals.has(id) && !skip.has(id));
  const v = (id: string): number => vals.get(id) as number;
  let n = 0;
  for (const r of graph.relations) {
    if (r.type === "subset") {
      if (checkable([r.outer, r.inner]) && v(r.inner) > v(r.outer)) n++;
      continue;
    }
    if (!checkable([r.parent, ...r.children, ...(r.other === undefined ? [] : [r.other])])) continue;
    const base = v(r.parent) - r.children.reduce((s, c) => s + v(c), 0);
    if (base < 0 || (r.other !== undefined && v(r.other) !== base)) n++;
  }
  return n;
}

/**
 * The day object's cells and EXACTLY the relations ADR §0 P1 declares, plus the
 * ±1-counter partition of §6 row 2 (see `EVENT_KIND_DAY_COUNTER`).
 *
 * Every partition is written so that `parent = Σ children + other` holds on the
 * true values; `serveDay` rests on it and `test/s002-servday.test.ts` asserts it
 * over the built graph.
 */
export function buildDayGraph(input: DayInput): { cells: Cell[]; graph: CellGraph } {
  const cells: Cell[] = [];
  const relations: Relation[] = [];
  const pop = (id: string, value: number): void => void cells.push({ id, kind: "population", value });
  const money = (id: string, value: number): void => void cells.push({ id, kind: "money", value });
  const virt = (id: string, value: number): void => void cells.push({ id, kind: "virtual", value });

  // ---- ±1 counters and stocks.
  // ACTION counts are VIRTUAL (`ACTION_METRICS`): K actions by one fan are a
  // population of one.
  for (const col of Object.keys(input.metrics).sort(byIdAsc) as Array<keyof DayMetrics>)
    (ACTION_METRICS.has(col) ? virt : pop)(cellId.metric(col), input.metrics[col]);
  for (const col of Object.keys(input.stocks).sort(byIdAsc) as Array<keyof DayStocks>) pop(cellId.stock(col), input.stocks[col]);
  // Cumulative link buckets nest (ADR §0 P1).
  relations.push({ type: "subset", outer: cellId.stock("linked_2plus"), inner: cellId.stock("linked_3plus") });

  // ---- money. Named by no relation, so it can neither be withheld nor
  // suppress anything (O-10/O-15).
  for (const col of MONEY_COLUMNS) money(cellId.revenue(col), input.revenue[col]);
  pop(cellId.revenue("buyers_new"), input.revenue.buyers_new);

  // ---- geo. `active_fans(day) := Σ_country fan_days` (ADR §2, O-17), so the
  // countries PARTITION it rather than being a second population beside it.
  const countryNames = Object.keys(input.countries).sort(byIdAsc);
  let activeFans = 0;
  for (const cc of countryNames) activeFans += input.countries[cc] as number;
  pop(cellId.activeFans, activeFans);
  if (countryNames.length > 0) {
    for (const cc of countryNames) pop(cellId.country(cc), input.countries[cc] as number);
    pop(cellId.geoOther, 0); // every country is enumerated, so the base is 0
    const children = countryNames.map((cc) => cellId.country(cc));
    relations.push(
      activeFans < GEO_DAY_MIN_ACTIVE
        ? { type: "partition", parent: cellId.activeFans, children, other: cellId.geoOther, blocked: true }
        : { type: "partition", parent: cellId.activeFans, children, other: cellId.geoOther },
    );
  }

  // ---- cohorts. Sources partition the day's `new_supporters`; the windows
  // nest inside the cohort's size.
  const dayCohorts = [...input.cohorts].sort((a, b) => byIdAsc(`${a.cohort_day} ${a.source}`, `${b.cohort_day} ${b.source}`));
  const sourceChildren: string[] = [];
  let sourceSum = 0;
  for (const c of dayCohorts) {
    const size = cellId.cohort(c.cohort_day, c.source, "size");
    const d30 = cellId.cohort(c.cohort_day, c.source, "d30");
    const d7 = cellId.cohort(c.cohort_day, c.source, "d7");
    const d1 = cellId.cohort(c.cohort_day, c.source, "d1");
    pop(size, c.size);
    pop(d30, c.d30);
    pop(d7, c.d7);
    pop(d1, c.d1);
    relations.push(
      { type: "subset", outer: size, inner: d30 },
      { type: "subset", outer: d30, inner: d7 },
      { type: "subset", outer: d7, inner: d1 },
    );
    if (c.cohort_day === input.day) {
      sourceChildren.push(size);
      sourceSum += c.size;
    }
  }
  if (sourceChildren.length > 0) {
    pop(cellId.cohortOther, input.metrics.new_supporters - sourceSum);
    relations.push({ type: "partition", parent: cellId.metric("new_supporters"), children: sourceChildren, other: cellId.cohortOther });
  }

  // ---- events. An event is a cohort keyed by event; its rows are keyed by
  // JOIN DAY and its totals are Σ over those days (ADR §4, pass-3 BREAK 6).
  const grouped = new Map<string, DayEventRow[]>();
  for (const r of input.events) {
    const key = `${r.kind} ${r.id}`;
    const rows = grouped.get(key);
    if (rows === undefined) grouped.set(key, [r]);
    else rows.push(r);
  }
  const daySavers: string[] = [];
  let daySaversSum = 0;
  const counterChildren = new Map<keyof DayMetrics, string[]>();
  const counterSum = new Map<keyof DayMetrics, number>();
  for (const key of [...grouped.keys()].sort(byIdAsc)) {
    const rows = (grouped.get(key) as DayEventRow[]).slice().sort((a, b) => byIdAsc(a.join_day, b.join_day));
    const first = rows[0] as DayEventRow;
    const kind = first.kind;
    const id = first.id;
    const totals = { participants: 0, new_fans: 0, acquired_d30: 0, acquisitions: 0, dead_attempts: 0, acquirers: 0 };
    for (const r of rows) {
      totals.participants += r.participants;
      totals.new_fans += r.new_fans;
      totals.acquired_d30 += r.acquired_d30;
      totals.acquisitions += r.acquisitions ?? 0;
      totals.dead_attempts += r.dead_attempts ?? 0;
      totals.acquirers += r.acquirers ?? 0;
    }
    for (const col of ["participants", "new_fans", "acquired_d30"] as const) {
      pop(cellId.event(kind, id, col), totals[col]);
      for (const r of rows) pop(cellId.eventDay(kind, id, r.join_day, col), r[col]);
      relations.push({
        type: "partition",
        parent: cellId.event(kind, id, col),
        children: rows.map((r) => cellId.eventDay(kind, id, r.join_day, col)),
      });
    }
    if (kind === "release") {
      // S-1's per-release cells: an acquisition count and the integrity block's
      // distinct dead-attempt accounts. Neither partitions nor nests inside
      // anything ADR §0 P1 declares.
      // I3-0356 (D-0829): an ACTION count summarises the people who acted, so
      // it is floored on them — five acquisitions by one fan is a population of
      // one; the consumer keeps DISTINCT `acquirers` beside it for that floor.
      // RG-AN-03 (D-0843(8)): the count is also CUMULATIVE (`isCumulativeCell`),
      // so object(T+1) − object(T) is the interval's actions, and no population
      // in any object floors the people behind that interval (`acquirers`
      // counts first acquisitions only). The cell is VIRTUAL: never served, in
      // any object, and its explain names that rather than a false "below K".
      virt(cellId.event(kind, id, "acquisitions"), totals.acquisitions);
      pop(cellId.event(kind, id, "dead_attempts"), totals.dead_attempts);
    }
    // acquired_d30 ⊆ new_fans ⊆ participants, per event AND per join day: both
    // levels are served cells, so both nestings are real.
    relations.push(
      { type: "subset", outer: cellId.event(kind, id, "participants"), inner: cellId.event(kind, id, "new_fans") },
      { type: "subset", outer: cellId.event(kind, id, "new_fans"), inner: cellId.event(kind, id, "acquired_d30") },
    );
    for (const r of rows) {
      relations.push(
        { type: "subset", outer: cellId.eventDay(kind, id, r.join_day, "participants"), inner: cellId.eventDay(kind, id, r.join_day, "new_fans") },
        { type: "subset", outer: cellId.eventDay(kind, id, r.join_day, "new_fans"), inner: cellId.eventDay(kind, id, r.join_day, "acquired_d30") },
      );
    }
    // The event's returning participants: virtual, so `participants` and
    // `new_fans` are never both served across a sub-floor difference.
    virt(cellId.eventReturning(kind, id), totals.participants - totals.new_fans);
    relations.push({
      type: "partition",
      parent: cellId.event(kind, id, "participants"),
      children: [cellId.event(kind, id, "new_fans")],
      other: cellId.eventReturning(kind, id),
    });
    // I21-Q164 (D-0820(3)): a war join fires NO `new_supporters` (analytics-consumer.ts, collectible_acquired), so
    // this event's new fans are NOT a subset of the day's new supporters. The part of `new_supporters` that came
    // through this event is X: its new fans of THIS day whose save the day counted, a membership-derived counter
    // (DayEventRow.new_supporters), virtual like the residual beside it.
    const todays = rows.find((r) => r.join_day === input.day);
    if (todays !== undefined) {
      const x = cellId.eventDay(kind, id, input.day, "new_supporters");
      virt(x, todays.new_supporters ?? 0);
      daySavers.push(x);
      daySaversSum += todays.new_supporters ?? 0;
      const counter = EVENT_KIND_DAY_COUNTER[kind];
      if (counter !== null) {
        const kids = counterChildren.get(counter);
        const p = cellId.eventDay(kind, id, input.day, "participants");
        if (kids === undefined) counterChildren.set(counter, [p]);
        else kids.push(p);
        counterSum.set(counter, (counterSum.get(counter) ?? 0) + todays.participants);
      }
    }
  }
  if (daySavers.length > 0) {
    virt(cellId.nonEventNew, input.metrics.new_supporters - daySaversSum);
    relations.push({ type: "partition", parent: cellId.metric("new_supporters"), children: daySavers, other: cellId.nonEventNew });
  }
  for (const counter of [...counterChildren.keys()].sort(byIdAsc)) {
    pop(cellId.dayCounterOther(counter), input.metrics[counter] - (counterSum.get(counter) ?? 0));
    relations.push({
      type: "partition",
      parent: cellId.metric(counter),
      children: counterChildren.get(counter) as string[],
      other: cellId.dayCounterOther(counter),
    });
  }

  // ---- reverse-code pools: two ±1 counters per pool. `redeemed ⊆ claimed` is
  // a population relation (a fan redeems a code they claimed), so the pair is
  // declared and `serveDay(:266)` floors the gap between them like any other
  // nesting — docket #17 B-2.
  for (const p of [...input.pools].sort((a, b) => byIdAsc(a.pool_id, b.pool_id))) {
    pop(cellId.pool(p.pool_id, "claimed"), p.claimed);
    pop(cellId.pool(p.pool_id, "redeemed"), p.redeemed);
    relations.push({ type: "subset", outer: cellId.pool(p.pool_id, "claimed"), inner: cellId.pool(p.pool_id, "redeemed") });
  }

  return { cells, graph: { relations } };
}

/**
 * I25 (#841): the flow-to-total relations a day object's cells carry. Each total is a sum
 * over sealed days `<= D` and each flow is day `D`'s cell (`analytics-read.ts readDayInput`,
 * `readLegacyCells`), so `total(T) − total(T-1)` is exactly the flows. Declared only where
 * every flow cell exists in the object: a pool with no row on `D` added nothing.
 * `stock.supporters` is clamped at 0 (`readDayInput`), so there the relation can be inexact;
 * `serveDay` (3d) treats a mismatch as "not what T-1 served".
 */
export function flowRelations(ids: ReadonlySet<string>): FlowRelation[] {
  const out: FlowRelation[] = [];
  const supporters = cellId.stock("supporters");
  const joins = cellId.metric("new_supporters");
  const leaves = cellId.metric("unfollows");
  if (ids.has(supporters) && ids.has(joins) && ids.has(leaves)) out.push({ total: supporters, inflow: [joins], outflow: [leaves] });
  const byCol: Record<"claimed" | "redeemed", string[]> = { claimed: [], redeemed: [] };
  for (const id of [...ids].sort(byIdAsc)) {
    const m = /^legacy\.pool\.([^.]+)\.(claimed|redeemed)$/.exec(id);
    if (m === null) continue;
    const col = m[2] as "claimed" | "redeemed";
    const flow = cellId.pool(m[1] as string, col);
    if (ids.has(flow)) out.push({ total: id, inflow: [flow], outflow: [] });
  }
  for (const id of [...ids].sort(byIdAsc)) {
    const m = /^pool\.([^.]+)\.(claimed|redeemed)$/.exec(id);
    if (m !== null) byCol[m[2] as "claimed" | "redeemed"].push(id);
  }
  for (const [total, col] of [["legacy.stock.code_claims", "claimed"], ["legacy.stock.redemptions", "redeemed"]] as const) {
    if (ids.has(total) && byCol[col].length > 0) out.push({ total, inflow: byCol[col], outflow: [] });
  }
  // I26 (I25-AN-05): the buyer stock is net of tombstones (`readLegacyCells`), so `collectors(T) − collectors(T-1)`
  // is D's `buyers_new` less the earlier days' buyers erased on D, which no object serves: a slack `s >= 0`
  // (`analytics_tombstones` rows only accrue until the creator's purge).
  const buyers = cellId.revenue("buyers_new");
  if (ids.has("legacy.stock.collectors") && ids.has(buyers)) out.push({ total: "legacy.stock.collectors", inflow: [buyers], outflow: [], slack: true });
  return out;
}

/**

/** One row of the (3c) reader's system: `Σ coef·x (= | <=) rhs`, plus the served ids folded into `rhs`. */
export interface JointRow {

  readonly coef: number[];

/**
 * min `obj·x` over `rows`, `x >= 0`: two-phase dense simplex, Bland's rule.
 * `null` = unbounded below; `NaN` = infeasible. Pure; (3c)'s systems are one
 * connected component of a day object, tens of terms.
 */
const SIMPLEX_EPS = 1e-9;

function simplexPivot(T: number[][], basis: number[], m: number, ncol: number, row: number, col: number): void {
  const pr = T[row] as number[];
  const pv = pr[col] as number;
  for (let j = 0; j <= ncol; j++) pr[j] = (pr[j] as number) / pv;
  for (let i = 0; i < m; i++) {
    if (i === row) continue;
    const ri = T[i] as number[];
    const f = ri[col] as number;
    if (f === 0) continue;
    for (let j = 0; j <= ncol; j++) ri[j] = (ri[j] as number) - f * (pr[j] as number);
  }
  basis[row] = col;
}

/**
 * (3c)'s simplex work budget per serveDay call, in tableau cell updates
 * (I13-FX2, D-0889(5)): deterministic, so the served set never depends on the
 * machine. A target still unsettled when it is spent is withheld. A 1,983-cell
 * day object spends about 1.3e8 (`test/i13-joint-derivation.test.ts`, cost case).
 */
export const JOINT_WORK_BUDGET = 400_000_000;

/** The solver stopped before an answer (iteration cap or budget spent). Thrown, so no caller can read it as "not pinned". */
export class SolverCap extends Error {}

/**
 * Phase 2's iteration cap when set (default `50 × (m + ncol) + 100`). Bland's rule settles (3c)'s tableaux far inside
 * the default, and phase 1 outlasts phase 2 on them, so the phase-2 cap arm is reached only with this lowered:
 * exported for that test alone (JA-05); production never writes it.
 */
export const SIMPLEX_PHASE2_CAP: { iterations: number | undefined } = { iterations: undefined };

/** Bland's rule to optimality: true = optimal, false = unbounded; throws `SolverCap` at the iteration cap or when the budget is spent. */
function simplexRun(T: number[][], basis: number[], m: number, ncol: number, cost: readonly number[], allowed: (j: number) => boolean, budget: { left: number }, cap = 50 * (m + ncol) + 100): boolean {
  for (let iter = 0; iter < cap; iter++) {
    budget.left -= 2 * m * (ncol + 1);
    if (budget.left < 0) throw new SolverCap("analytics-cells: (3c) work budget spent");
    let enter = -1;
    for (let j = 0; j < ncol && enter < 0; j++) {
      if (!allowed(j)) continue;
      let rc = cost[j] as number;
      for (let i = 0; i < m; i++) rc -= (cost[basis[i] as number] as number) * ((T[i] as number[])[j] as number);
      if (rc < -SIMPLEX_EPS) enter = j;
    }
    if (enter < 0) return true;
    let leave = -1;
    let best = Infinity; // a bare global: env-inventory reads `X.UPPER_CASE` as an env key
    for (let i = 0; i < m; i++) {
      const a = (T[i] as number[])[enter] as number;
      if (a <= SIMPLEX_EPS) continue;
      const ratio = ((T[i] as number[])[ncol] as number) / a;
      if (ratio < best - SIMPLEX_EPS || (Math.abs(ratio - best) <= SIMPLEX_EPS && (basis[i] as number) < (basis[leave] as number))) {
        best = ratio;
        leave = i;
      }
    }
    if (leave < 0) return false;
    simplexPivot(T, basis, m, ncol, leave, enter);
  }
  throw new SolverCap("analytics-cells: (3c) simplex iteration cap");
}

export interface LpState {
  readonly n: number;
  readonly m: number;
  readonly ncol: number;
  readonly firstArt: number;
  readonly T: number[][];
  readonly basis: number[];
}

/** The simplex after phase 1 (a feasible basis), or `undefined` when the rows are infeasible; throws `SolverCap`. */
export function lpFeasible(n: number, rows: ReadonlyArray<{ idx: readonly number[]; coef: readonly number[]; rhs: number; eq: boolean }>, budget: { left: number }): LpState | undefined {
  const EPS = 1e-9;
  const m = rows.length;
  let nSlack = 0;
  let nArt = 0;
  for (const r of rows) {
    if (!r.eq) nSlack++;
    if (r.eq || r.rhs < 0) nArt++;
  }
  const ncol = n + nSlack + nArt;
  const T: number[][] = [];
  const basis: number[] = [];
  let slackAt = n;
  let artAt = n + nSlack;
  for (const r of rows) {
    const sgn = r.rhs < 0 ? -1 : 1;
    const t = new Array<number>(ncol + 1).fill(0);
    for (let k = 0; k < r.idx.length; k++) t[r.idx[k] as number] = (t[r.idx[k] as number] as number) + sgn * (r.coef[k] as number);
    t[ncol] = sgn * r.rhs;
    let b = -1;
    if (!r.eq) {
      t[slackAt] = sgn;
      if (sgn > 0) b = slackAt;
      slackAt++;
    }
    if (b < 0) {
      t[artAt] = 1;
      b = artAt;
      artAt++;
    }
    T.push(t);
    basis.push(b);
  }
  const value = (cost: readonly number[]): number => {
    let v = 0;
    for (let i = 0; i < m; i++) v += (cost[basis[i] as number] as number) * ((T[i] as number[])[ncol] as number);
    return v;
  };
  const isArt = (j: number): boolean => j >= n + nSlack;
  const c1 = Array.from({ length: ncol }, (_, j) => (isArt(j) ? 1 : 0));
  simplexRun(T, basis, m, ncol, c1, () => true, budget); // phase 1 is bounded below by 0: it cannot be unbounded
  if (value(c1) > 1e-7) return undefined;
  for (let i = 0; i < m; i++) {
    if (!isArt(basis[i] as number)) continue;
    for (let j = 0; j < n + nSlack; j++) {
      if (Math.abs((T[i] as number[])[j] as number) > EPS) {
        simplexPivot(T, basis, m, ncol, i, j);
        break;
      }
    }
  }
  return { n, m, ncol, firstArt: n + nSlack, T, basis };
}

/** min `obj·x` from a phase-1 state (copied, so one state serves every objective); `null` = unbounded; throws `SolverCap`. */
export function lpMinimise(state: LpState, obj: readonly number[], budget: { left: number }, point?: (x: readonly number[]) => void): number | null {
  const { n, m, ncol, firstArt } = state;
  const c2 = new Array<number>(ncol).fill(0);
  for (let j = 0; j < n; j++) c2[j] = obj[j] as number;
  // Copy the shared phase-1 tableau only when this objective needs a pivot.
  let T = state.T;
  let basis = state.basis;
  let optimal = true;
  for (let j = 0; j < firstArt && optimal; j++) {
    let rc = c2[j] as number;
    for (let i = 0; i < m; i++) rc -= (c2[basis[i] as number] as number) * ((T[i] as number[])[j] as number);
    if (rc < -SIMPLEX_EPS) optimal = false;
  }
  if (!optimal) {
    T = state.T.map((r) => [...r]);
    basis = [...state.basis];
    if (!simplexRun(T, basis, m, ncol, c2, (j) => j < firstArt, budget, SIMPLEX_PHASE2_CAP.iterations)) return null;
  }
  let v = 0;
  for (let i = 0; i < m; i++) v += (c2[basis[i] as number] as number) * ((T[i] as number[])[ncol] as number);
  if (point !== undefined) {
    const x = new Array<number>(n).fill(0);
    for (let i = 0; i < m; i++) if ((basis[i] as number) < n) x[basis[i] as number] = (T[i] as number[])[ncol] as number;
    point(x);
  }
  return v;
}

  readonly complementary?: ReadonlyMap<string, string>;

/**
 * I3-0309 (D-0829): a cell whose value is a RUNNING TOTAL over days, so the
 * value a reader holds from one day object and the value served in a later one
 * are two terms of the same cumulative quantity. Stocks (`stock.*`,
 * `legacy.stock.*`), cohort windows (`cohort.<day>.<source>.d1|d7|d30`), the
 * per-pool legacy totals (`legacy.pool.<id>.claimed|redeemed`) and an event's
 * all-join-days totals (`event.<kind>.<id>.<col>` — four segments).
 * I23 (I3-0606, I3-0255; D-0993(1)): a SEALED cohort's size
 * (`cohort.<day>.<source>.size`) and an event's per-join-day cells
 * (`event.<kind>.<id>.<YYYY-MM-DD>.<col>` — five segments) are re-served under
 * the same id in every later object, so an erasure that lowers them is a
 * cross-object delta too. Everything else is a daily flow and is evaluated per
 * day, since no cross-day equation exists for it.
 */
export function isCumulativeCell(id: string): boolean {
  return (
    id.startsWith("stock.") ||
    id.startsWith("legacy.stock.") ||
    /^legacy\.pool\.[^.]+\.(claimed|redeemed)$/.test(id) ||
    /^cohort\.[^.]+\.[^.]+\.(size|d1|d7|d30)$/.test(id) ||
    /^event\.(release|war|qna|drop)\.[^.]+\.(\d{4}-\d{2}-\d{2}\.)?(participants|new_fans|acquired_d30|acquisitions|dead_attempts)$/.test(id)
  );
}

/**
 * ADR §6 geo row: a day with fewer than ten active fans serves no country rows
 * at all. Distinct from `K`: `K` floors a cell, this suppresses the whole
 * breakdown before any cell is looked at.
 */
export const GEO_DAY_MIN_ACTIVE = 10;

export const isPartition = (r: Relation): r is PartitionRelation => r.type === "partition";
export const byIdAsc = (a: string, b: string): number => (a < b ? -1 : a > b ? 1 : 0);

/**
 * A ratio is served only when BOTH operands are (ADR §6 row 3, INV-004 clause
 * 4). It is not a cell: a ratio has no population of its own, and floors are
 * computed against populations.
 */
export function ratio(set: ServedSet, numerator: string, denominator: string): number | null {
  const a = set.served.get(numerator);
  const b = set.served.get(denominator);
  if (a === undefined || b === undefined || b === 0) return null;
  return a / b;
}

/**
 * THE CELL GRAPH AND THE SERVED-SET FUNCTION (ADR-0006 §0 P1, §4, §6).

 * ---------------------------------------------------------------------------
 * WHY ONE FUNCTION OVER ONE GRAPH, AND NOT A FLOOR PER WIDGET
 * ---------------------------------------------------------------------------
 * v0.2-v0.4 of the ADR floored each widget with its own rule: a partition rule
 * for geo, a nesting rule for cohorts, a per-cell floor for the ±1 counters.

 * ran one event on a quiet day read `participants − new_supporters` and got the
 * sub-floor population the nesting rule had just withheld. The repair is not a
 * better rule; it is ONE rule over the WHOLE day object. Everything a
 * creator-private surface can serve is declared here as a cell with a
 * population kind and its relations to other cells, and `serveDay` is the only
 * thing that decides what leaves.
 *
 * The guarantee (P1, INV-004 clause 4): no withheld cell is exactly
 * recoverable from the served cells through the declared relations, and the
 * interval any relation leaves open for it is at least `K` wide.
 *

 * WHAT A RELATION MEANS
 * ---------------------------------------------------------------------------
 *  - `partition`: `parent = Σ children + other`. `other` is the partition's
 *    RESIDUAL TERM and it is dynamic: a child that gets withheld is POOLED into
 *    it (the geo widget's "cells below the floor pool into `other`", ADR §6).

 *    non-event new supporters — is a residual that is never served at all; it
 *    exists so the algebra closes and so the arithmetic that would otherwise
 *    happen in the reader's head happens here instead.
 *  - `subset`: `outer ⊇ inner`. Not an equation: it is the statement that
 *    `outer − inner` is a population, which is exactly what step (2) floors.
 *

 * counts are never floored and never erased (ADR §6, O-10/O-15, INV-033), so a

 * withheld nor be used to suppress anything else.
 *
 * ---------------------------------------------------------------------------

 * ---------------------------------------------------------------------------
 * No I/O, no clock, no database handle, no `env`. `analytics-read.ts` (B7),
 * `regions.ts` (B8) and `creator-ai.ts` (B11) all import this module, and the

 * itself is `K_ANONYMITY_FLOOR` from `intelligence.ts` — imported, never

/**
 * P1 steps (1)-(4) over the whole day object at once.
 *
 * (1) withhold every population cell below `K`;
 * (2) for every `subset` pair with both served and `outer − inner < K`,
 *     withhold the inner — over the TRANSITIVE closure, so a cell is floored
 *     against its nearest SERVED ancestor and not merely against its immediate
 *     parent ("from the coarsest cell inward"; the two readings coincide on a
 *     monotone chain, pass-3 V3c);
 * (3) complementary suppression, per partition: the residual mass the reader
 *     cannot see must be at least `K`, and the equation must leave at least two
 *     unknowns — otherwise withhold one more term (the smallest served child by
 *     value, ties by id, else the parent);
 * (3b) INV-004 clause 4 EXISTENCE (docket #17 B-1; NARROWED by Jason 2026-09-07,
 *     S-003 S3-16 — the clause's parenthetical is the whole obligation: no
 *     served value may admit a single integer solution for a withheld child):
 *     a reader has INTEGER BOUNDS — a floor-withheld cell lies in `[0, K−1]`
 *     and they are told so — and (3) reasons without them, so a residual of
 *     `n(K−1)` over `n` withheld children has ONE integer solution (`5,4,4`
 *     beside a total of `13` recovers `4` and `4`). For the ONE equation this
 *     partition gives the reader, bound every unknown — `[0, K−1]` if the floor
 *     withheld it, `[0, mass]` otherwise, and the base not at all when the
 *     breakdown enumerates its parent — and withhold the served value that
 *     carries the equation when some unknown is pinned to a SUB-FLOOR point,
 *     `0` included: a pin at `0` says the cell holds no one, which is itself a
 *     sub-K fact, so only a pin at `K` or above is let out, as the floor lets
 *     out `K` people (`pinsAnUnknown`: `min === max && min < K`). Two unknowns
 *     that only bound each other's SUM leave the total served — the geo
 *     `active_fans` over one sub-K country beside a served pool, or
 *     `participants` over one sub-K `new_fans` and a virtual `returning`, stay
 *     served. This is ADR-0006 §0 P1's interval sentence as an executable rule:
 *     no served value leaves any withheld cell an interval narrower than a
 *     point it could not already be read off;
 * (4) recompute every served partition total as `Σ served children + served
 *     other`.
 *
 * (2) and (3) repeat to a fixed point. Withholding only ever grows, so the loop
 * terminates in at most one pass per cell.
 */
export function serveDay(
  cells: readonly Cell[],
  graph: CellGraph,
  K: number = K_ANONYMITY_FLOOR,
  carried?: ReadonlyMap<string, string>,
  priorServed?: ReadonlyMap<string, readonly number[]>,
  forced?: ReadonlySet<string>,
  jointBudget: number = JOINT_WORK_BUDGET,
  priorDay?: ReadonlyMap<string, number | null>,
): ServedSet {
  const byId = new Map<string, Cell>();
  for (const c of [...cells].sort((a, b) => byIdAsc(a.id, b.id))) {
    if (byId.has(c.id)) throw new Error(`analytics-cells: duplicate cell '${c.id}'`);
    byId.set(c.id, c);
  }
  const cellOf = (id: string): Cell => {
    const c = byId.get(id);
    if (c === undefined) throw new Error(`analytics-cells: relation names unknown cell '${id}'`);
    return c;
  };
  const val = (id: string): number => cellOf(id).value;

  // Relations in a fixed order: the output must not depend on how the caller
  // happened to build the array (the determinism test shuffles it).
  const partitions = graph.relations
    .filter(isPartition)
    .map((p) => {
      p.children.forEach((c) => { cellOf(c); });
      cellOf(p.parent);
      if (p.other !== undefined) cellOf(p.other);
      return p;
    })
    .sort((a, b) => byIdAsc(`${a.parent} ${a.children.join(",")}`, `${b.parent} ${b.children.join(",")}`));
  const flows: readonly FlowRelation[] = [...(graph.flows ?? [])]
    .map((f) => {
      for (const id of [f.total, ...f.inflow, ...f.outflow]) cellOf(id);
      return f;
    })
    .sort((a, b) => byIdAsc(a.total, b.total));
  const subsets = graph.relations
    .filter((r): r is SubsetRelation => r.type === "subset")
    .map((s) => {
      cellOf(s.outer);
      cellOf(s.inner);
      return s;
    });

  const hidden = new Set<string>();
  const why = new Map<string, string>();
  /** I3-0309: complementary victim -> the parent of the partition that forced it. */
  const forcedBy = new Map<string, string>();
  const hide = (id: string, reason: string): boolean => {
    if (hidden.has(id)) return false;
    hidden.add(id);
    why.set(id, reason);
    return true;
  };

  // A partition's residual term is not a cell with a fixed served value — its
  // value is the pool — so it is exempt from step (1) and decided in step (3).
  const residualIds = new Set<string>();
  for (const p of partitions) if (p.other !== undefined) residualIds.add(p.other);

  // ---- step (0a): virtual cells are never served, whatever their value.
  for (const c of byId.values()) {
    if (c.kind === "virtual") hide(c.id, "virtual: never served; it exists to close a relation (ADR §0 P1)");
  }

  // ---- step (0b): the geo day rule. A blocked partition leaves the algebra.
  const active: PartitionRelation[] = [];
  for (const p of partitions) {
    if (p.blocked === true) {
      for (const c of p.children) hide(c, `blocked: no rows for this breakdown on this day (ADR §6 geo row, active < ${GEO_DAY_MIN_ACTIVE})`);
      if (p.other !== undefined) hide(p.other, "blocked: no rows for this breakdown on this day (ADR §6 geo row)");
    } else {
      active.push(p);
    }
  }

  // ---- step (1): the floor.
  // A REASON NAMES THE RULE AND THE OPERAND IDS, NEVER A VALUE (RUN-0012
  // B-3). `explain` is exposed beside the served set (`analytics-read.ts:721`,
  // `DayCells.explain`; QUEUE #895 census: no projection reads it at I13, and if
  // one ever does, "floor" versus any other reason is itself a bound the (3c)
  // reader does not model), so a reason carrying `${c.value}`, the nesting gap or
  // the residual mass would hand back by string exactly the integer step
  // (1)-(3) withheld; the operand IDs themselves (`:209`, `:320`, `:336`, `:345`, `:354`, `:356`) are
  // not values and are deliberately named. NOTE (RUN-0012 B-2, OPEN): the
  // sentence is itself a served value — a RULE name can still disclose a
  // bound; see docs/INVARIANTS.md INV-004 clause 4 and UNSTATED-0010.
  for (const c of byId.values()) {
    if (c.kind !== "population") continue; // money is never withheld (O-10/O-15)
    if (residualIds.has(c.id)) continue;
    if (c.value < K) hide(c.id, "floor: below K");
  }

  // The transitive closure of `subset`. Declaring only adjacent pairs would let
  // a served cell sit a sub-floor distance from a served NON-adjacent ancestor
  // once the cell between them was withheld.
  const closure: Array<{ outer: string; inner: string }> = [];
  {
    const edges = new Map<string, string[]>();
    for (const s of subsets) {
      const to = edges.get(s.outer);
      if (to === undefined) edges.set(s.outer, [s.inner]);
      else to.push(s.inner);
    }
    for (const outer of [...edges.keys()].sort(byIdAsc)) {
      const seen = new Set<string>();
      const stack = [...(edges.get(outer) ?? [])];
      while (stack.length > 0) {
        const next = stack.pop() as string;
        if (next === outer || seen.has(next)) continue;
        seen.add(next);
        for (const deeper of edges.get(next) ?? []) stack.push(deeper);
      }
      for (const inner of [...seen].sort(byIdAsc)) closure.push({ outer, inner });
    }
  }

  /** base = the residual the children do not enumerate; `parent − Σ children`. */
  const baseOf = (p: PartitionRelation): number => {
    if (p.other !== undefined) return val(p.other);
    let sum = 0;
    for (const c of p.children) sum += val(c);
    return val(p.parent) - sum;
  };
  /** The mass the reader cannot see: the base plus every withheld child. */
  const residualOf = (p: PartitionRelation): number => {
    let r = baseOf(p);
    for (const c of p.children) if (hidden.has(c)) r += val(c);
    return r;
  };
  const servedChildrenOf = (p: PartitionRelation): string[] => p.children.filter((c) => !hidden.has(c));
  /**
   * The residual is EMITTED (and so is a known term rather than an unknown)
   * only when it is a real cell, the parent is served, some child is served,
   * the pool itself clears the floor, and step (3b) has not withheld it.
   */
  const residualEmitted = (p: PartitionRelation): boolean =>
    p.other !== undefined &&
    cellOf(p.other).kind !== "virtual" &&
    !hidden.has(p.other) &&
    !hidden.has(p.parent) &&
    residualOf(p) >= K &&
    servedChildrenOf(p).length > 0;
  /**
   * (3b) The integer-bounds check for ONE equation `mass = Σ unknowns`. Each
   * unknown is `[0, K−1]` when the floor withheld it (the reader is told "below
   * K"), else `[0, mass]`. Returns true when some unknown is pinned to a single
   * value BELOW the floor. A pin at `K` or above names a population the floor
   * already lets out, which is why a virtual residual of `10` beside a served
   * `new_fans` is not a pin (`test/s002-servday.test.ts:299`, `:812`).
   */
  const pinsAnUnknown = (mass: number, unknownIds: readonly string[], baseUnknown: boolean): boolean => {
    const hi: number[] = unknownIds.map((id) => (why.get(id) === "floor: below K" ? K - 1 : mass));
    if (baseUnknown) hi.push(mass);
    if (hi.length === 0) return false;
    const sumHi = hi.reduce((a, b) => a + b, 0);
    for (let i = 0; i < hi.length; i++) {
      const min = Math.max(0, mass - (sumHi - (hi[i] as number)));
      const max = Math.min(hi[i] as number, mass);
      if (min === max && min < K) return true;
    }
    return false;
  };
  /**
   * The residual term of the TOTAL equation. A non-virtual `other` whose base
   * is zero is NOT an unknown — the breakdown enumerates its parent, so the
   * reader has `parent − Σ served` exactly; a virtual residual is an unserved
   * population and stays a variable (`test/s002-servday.test.ts:143-155`).
   */
  const baseIsUnknown = (p: PartitionRelation): boolean =>
    p.other !== undefined && (cellOf(p.other).kind === "virtual" || baseOf(p) > 0);

  // ---- step (1b): I3-0309 CARRIED complementary suppression (D-0829,
  // D-0843(8)). A cumulative cell that step (3) withheld on an earlier day to
  // protect a sub-K term is re-withheld on EVERY later day it exists. Any later
  // served value of a running total is an upper bound on its earlier value, so
  // releasing it bounds the earlier day's floored child from below (RG-AN-01:
  // P_D=23, b_later=20 puts a_D in [3,4]) whatever today's partition looks
  // like. Daily flows are never carried (`isCumulativeCell`). The choice reads
  // ids only, never a value. (2)/(3) below then run as usual.
  if (carried !== undefined) {
    for (const [id, parent] of [...carried].sort((a, b) => byIdAsc(a[0], b[0]))) {
      if (!byId.has(id) || !isCumulativeCell(id)) continue;
      if (hide(id, `complementary (carried): withheld since ${parent} forced it on an earlier day`)) forcedBy.set(id, parent);
    }
  }

  // ---- step (1c): QUEUE #734 (D-0843(8)) the RUNNING-TOTAL DELTA. A reader
  // who holds a cumulative cell from another day object subtracts: a change of
  // 0 < |delta| < K since a SERVED value names a sub-K population. `priorServed`
  // carries, per cell, EVERY value this creator's other day objects SERVED that
  // is still retained (up to SERVED_STATE_KEEP per cell, before and after this
  // day — `analytics-read.ts readServedState`). A withheld day does not
  // move them, so the delta grows until it reaches K. Before the fixed point, so
  // (2)/(3) treat the withheld total as a term like any other. Money is never
  // withheld (O-10/O-15); the reason names the rule, never a value.
  if (priorServed !== undefined) {
    for (const [id, values] of [...priorServed].sort((a, b) => byIdAsc(a[0], b[0]))) {
      const c = byId.get(id);
      if (c === undefined || c.kind !== "population" || !isCumulativeCell(id)) continue;
      if (values.some((v) => Math.abs(c.value - v) > 0 && Math.abs(c.value - v) < K)) {
        hide(id, "delta below K: this running total moved by fewer than K since a value served on another day");
      }
    }
  }

  // ---- step (1d): QUEUE #734 / I8RG-PRIV-05. A running total this day's
  // served-state row already records as WITHHELD (an overlapping build of a
  // neighbouring day made this build re-evaluate, `analytics-read.ts
  // settleServedState`) stays withheld: a row another build may have read as
  // "no neighbour" never becomes a served value. Not a complementary victim, so
  // never carried to later days. Population cells only, as (1c).
  if (forced !== undefined) {
    for (const id of [...forced].sort(byIdAsc)) {
      const c = byId.get(id);
      if (c === undefined || c.kind !== "population" || !isCumulativeCell(id)) continue;
      hide(id, "withheld: this running total is recorded as withheld for this day");
    }
  }

  // ---- step (3c): QUEUE #895 (I3-0325 general limb) THE JOINT SYSTEM. Steps
  // (3)/(3b) read ONE equation at a time; a reader reads every served relation
  // at once. Seeded search (`test/i13-joint-derivation.test.ts`): a release's
  // new fans total 18 with day 29 served at 6, day 28 under a served 6
  // participants and day 30 under a served `new_supporters` of 6 — no single
  // relation pins anything, together they pin day 30's returning participants
  // at 2. Over the reader's system — served values as constants, every other
  // cell a non-negative unknown (a non-virtual residual at 0 known, as
  // `baseIsUnknown`), every active partition an equation, an emitted pool
  // `other + Σ withheld children = pool`, every subset `inner <= outer` — a
  // withheld cell or a nesting gap below K whose range holds ONE integer is a
  // pin, and the id-first served population cell its rows name goes. Ids only,
  // never a value (docket #22(a)). Then (2)/(3)/(3b) run again.
  const blockedIds = new Set<string>();
  for (const p of partitions) {
    if (p.blocked !== true) continue;
    for (const c of p.children) blockedIds.add(c);
    if (p.other !== undefined) blockedIds.add(p.other);
  }
  const partitionOfOther = new Map<string, PartitionRelation>();
  for (const p of active) if (p.other !== undefined) partitionOfOther.set(p.other, p);
  /** What the reader holds for a cell: its served value, or undefined. A residual is held as its POOL. */
  const servedView = (id: string): number | undefined => {
    const c = cellOf(id);
    if (c.kind === "virtual" || hidden.has(id)) return undefined;
    const p = partitionOfOther.get(id);
    if (p !== undefined) return residualEmitted(p) ? residualOf(p) : undefined;
    return c.value;
  };
  // I13-FX2 (RGD-895-cost, D-0889(5)): ONE simplex work budget per serveDay
  // call, in tableau cell updates, shared by every round; a target still open
  // when it is spent is withheld, never served. `settledFree` caches targets
  // proven not pinned: withholding only grows, so the reader's feasible set only
  // grows and a target once free stays free in every later round.
  const budget = { left: jointBudget };
  const settledFree = new Set<string>();
  const jointVictim = (): Array<{ victim: string; target: string }> => {
    const varIx = new Map<string, number>();
    const varIds: string[] = [];
    const view = new Map<string, number | undefined>();
    const viewOf = (id: string): number | undefined => {
      if (!view.has(id)) view.set(id, servedView(id));
      return view.get(id);
    };
    /** A term the reader knows: a served value; a residual's BASE only when it is a non-virtual 0. */
    const known = (id: string): number | undefined => {
      if (partitionOfOther.has(id)) {
        const c = cellOf(id);
        return c.kind !== "virtual" && c.value === 0 ? 0 : undefined;
      }
      return viewOf(id);
    };
    const varOf = (id: string): number => {
      let i = varIx.get(id);
      if (i === undefined) {
        i = varIds.length;
        varIx.set(id, i);
        varIds.push(id);
      }
      return i;
    };
    const rows: JointRow[] = [];
    const add = (terms: ReadonlyArray<readonly [string, number]>, rhs: number, eq: boolean, consts: string[] = []): void => {
      const r: JointRow = { idx: [], coef: [], rhs, eq, consts };
      for (const [id, a] of terms) {
        const k = known(id);
        if (k !== undefined) {
          r.rhs -= a * k;
          r.consts.push(id);
        } else {
          r.idx.push(varOf(id));
          r.coef.push(a);
        }
      }
      if (r.idx.length > 0) rows.push(r);
    };
    for (const p of active) {
      const terms: Array<readonly [string, number]> = [[p.parent, 1], ...p.children.map((c) => [c, -1] as const)];
      if (p.other !== undefined) terms.push([p.other, -1]);
      add(terms, 0, true);
      const pool = p.other === undefined ? undefined : viewOf(p.other);
      if (pool !== undefined) {
        const hiddenKids = p.children.filter((c) => viewOf(c) === undefined);
        add([[p.other as string, 1], ...hiddenKids.map((c) => [c, 1] as const)], pool, true, [p.other as string]);
      }
    }
    for (const s of subsets) {
      if (blockedIds.has(s.inner) || blockedIds.has(s.outer)) continue;
      add([[s.inner, 1], [s.outer, -1]], 0, false);
    }
    // Targets: every unknown below K that is not a served pool's base, and
    // every nesting gap below K the reader cannot read off directly.
    const targets: Array<{ id: string; terms: Array<readonly [string, number]> }> = [];
    for (const id of [...varIds].sort(byIdAsc)) {
      const c = cellOf(id);
      if (c.kind === "money" || c.value >= K) continue;
      if (partitionOfOther.has(id) && viewOf(id) !== undefined) continue;
      targets.push({ id, terms: [[id, 1]] });
    }
    for (const { outer, inner } of closure) {
      if (blockedIds.has(outer) || blockedIds.has(inner)) continue;
      if (known(outer) !== undefined && known(inner) !== undefined) continue;
      if (val(outer) - val(inner) >= K) continue;
      targets.push({ id: `${outer} - ${inner}`, terms: [[outer, 1], [inner, -1]] });
    }
    if (targets.length === 0) return [];
    // Components over the unknowns, so each range is solved on the rows it touches.
    const root = varIds.map((_, i) => i);
    const find = (i: number): number => {
      let r = i;
      while (root[r] !== r) r = root[r] as number;
      while (root[i] !== r) {
        const next = root[i] as number;
        root[i] = r;
        i = next;
      }
      return r;
    };
    for (const r of rows) for (let k = 1; k < r.idx.length; k++) root[find(r.idx[k] as number)] = find(r.idx[0] as number);
    const compRowsOf = new Map<number, JointRow[]>();
    for (const r of rows) {
      const c = find(r.idx[0] as number);
      const list = compRowsOf.get(c);
      if (list === undefined) compRowsOf.set(c, [r]);
      else list.push(r);
    }
    /** A component's unknowns, numbered from 0 — built only for a component that reaches an LP. */
    const localOf = new Map<number, Map<number, number>>();
    const localFor = (c: number): Map<number, number> => {
      let l = localOf.get(c);
      if (l === undefined) {
        l = new Map<number, number>();
        for (let i = 0; i < varIds.length; i++) if (find(i) === c) l.set(i, l.size);
        localOf.set(c, l);
      }
      return l;
    };
    const lpOf = new Map<number, LpState | undefined>();
    // A cheap witness before any LP: the true point moved by ±1 on a term and
    // ∓1 on a partner in one of its rows. If every row still holds and the
    // target moved, the target is not pinned (the moved point is a world the
    // reader cannot rule out). Sound; the LP decides whatever it does not.
    const x0 = varIds.map((id) => val(id));
    const rowsOfVar: JointRow[][] = varIds.map(() => []);
    for (const r of rows) for (const i of r.idx) (rowsOfVar[i] as JointRow[]).push(r);
    const holds = (r: JointRow): boolean => {
      let sum = 0;
      for (let k = 0; k < r.idx.length; k++) sum += (r.coef[k] as number) * (x0[r.idx[k] as number] as number);
      return r.eq ? sum === r.rhs : sum <= r.rhs;
    };
    const moves = (tv: ReadonlyArray<readonly [number, number]>): boolean => {
      const objOf = new Map<number, number>(tv);
      for (const [v] of tv) {
        for (const r of rowsOfVar[v] as JointRow[]) {
          const kv = r.idx.indexOf(v);
          for (let k = 0; k < r.idx.length; k++) {
            const y = r.idx[k] as number;
            if (y === v) continue;
            for (const d of [1, -1]) {
              const dy = (-d * (r.coef[kv] as number)) / (r.coef[k] as number);
              if ((objOf.get(v) ?? 0) * d + (objOf.get(y) ?? 0) * dy === 0) continue;
              if ((x0[v] as number) + d < 0 || (x0[y] as number) + dy < 0) continue;
              x0[v] = (x0[v] as number) + d;
              x0[y] = (x0[y] as number) + dy;
              const ok = (rowsOfVar[v] as JointRow[]).every(holds) && (rowsOfVar[y] as JointRow[]).every(holds);
              x0[v] = (x0[v] as number) - d;
              x0[y] = (x0[y] as number) - dy;
              if (ok) return true;
            }
          }
        }
      }
      return false;
    };
    /**
     * The `size` unknowns nearest `seed` in the row graph, where crossing a row
     * costs its number of unknowns: a two-term nesting is near, a breakdown over
     * every day of an event is farther, and a day's hub row (every event's new
     * fans on the day) is farthest.
     */
    const ballOf = (seed: readonly number[], size: number): number[] => {
      const dist = new Map<number, number>();
      const buckets: number[][] = [[...seed]];
      for (const v of seed) dist.set(v, 0);
      const inBall: number[] = [];
      const done = new Set<number>();
      for (let d = 0; d < buckets.length && inBall.length < size; d++) {
        const q = buckets[d];
        if (q === undefined) continue;
        for (let qi = 0; qi < q.length && inBall.length < size; qi++) {
          const v = q[qi] as number;
          if (done.has(v) || dist.get(v) !== d) continue;
          done.add(v);
          inBall.push(v);
          for (const row of rowsOfVar[v] as JointRow[]) {
            const nd = d + row.idx.length;
            for (const y of row.idx) {
              if (done.has(y) || (dist.get(y) ?? Infinity) <= nd) continue;
              dist.set(y, nd);
              const bucket = buckets[nd];
              if (bucket === undefined) buckets[nd] = [y];
              else bucket.push(y);
            }
          }
        }
      }
      return inBall;
    };
    // Every LP optimum is a point of the reader's system; kept per component (as
    // its difference from the truth) so a later target it already moves by >= 1
    // is settled without an LP of its own.
    const witnesses = new Map<number, Array<Map<number, number>>>();
    const witnessFree = (comp: number, tv: ReadonlyArray<readonly [number, number]>, truth: number): boolean => {
      for (const w of witnesses.get(comp) ?? []) {
        let v = 0;
        for (const [i, a] of tv) v += a * (w.get(i) ?? (x0[i] as number));
        if (v <= truth - 1 + 1e-7 || v >= truth + 1 - 1e-7) return true;
      }
      return false;
    };
    /** "free": the LP range holds an integer other than the truth; "pinned": it holds one; "cap": the solver stopped first (ONE catch, below). */
    const decide = (lpFor: () => LpState | undefined, obj: readonly number[], truth: number, comp: number, globalOf: readonly number[]): "free" | "pinned" | "cap" => {
      try {
        return decideLp(lpFor(), obj, truth, comp, globalOf);
      } catch (e) {
        if (e instanceof SolverCap) return "cap";
        throw e;
      }
    };
    const decideLp = (lp: LpState | undefined, obj: readonly number[], truth: number, comp: number, globalOf: readonly number[]): "free" | "pinned" => {
      if (lp === undefined) return "pinned"; // infeasible cannot happen (the truth satisfies every row): fail closed
      const keep = (x: readonly number[]): void => {
        const w = new Map<number, number>();
        for (let j = 0; j < x.length; j++) {
          const g = globalOf[j] as number;
          if (Math.abs((x[j] as number) - (x0[g] as number)) > 1e-9) w.set(g, x[j] as number);
        }
        const list = witnesses.get(comp);
        if (w.size === 0) return;
        if (list === undefined) witnesses.set(comp, [w]);
        else if (list.length < 512) list.push(w);
      };
      const lo = lpMinimise(lp, obj, budget, keep);
      if (lo !== null && Math.ceil(lo - 1e-7) < truth) return "free";
      const hiNeg = lpMinimise(lp, obj.map((x) => -x), budget, keep);
      if (lo === null || hiNeg === null) return "free"; // unbounded: not one value
      return Math.ceil(lo - 1e-7) === Math.floor(-hiNeg + 1e-7) ? "pinned" : "free";
    };
    const out: Array<{ victim: string; target: string }> = [];
    const spentComps = new Set<number>();
    for (const t of targets) {
      if (settledFree.has(t.id)) continue;
      const tv = t.terms.filter(([id]) => known(id) === undefined).map(([id, a]) => [varIx.get(id) as number, a] as const);
      if (tv.length === 0) continue;
      const comp = find((tv[0] as readonly [number, number])[0]);
      const compRows = compRowsOf.get(comp) ?? [];
      // One equation is (3)/(3b)'s case; a component with no served constant is
      // a cone, which scales any non-zero term off its value.
      let truth = 0;
      for (const [i, a] of tv) truth += a * val(varIds[i] as string);
      if (compRows.length < 2 || (truth !== 0 && compRows.every((r) => r.rhs === 0))) continue;
      if (moves(tv) || witnessFree(comp, tv, truth)) {
        settledFree.add(t.id);
        continue;
      }
      // LOCAL LP first: the ball around the target with every unknown outside it
      // FIXED at its true value and every row touching it kept. A point of that
      // LP is a point of the reader's whole system (rows outside the ball hold at
      // the truth), so a local range reaching truth ± 1 settles "free" exactly as
      // the component LP would; anything else grows the ball, and the component
      // LP decides last.
      // A spent budget settles nothing more: every target still open is withheld.
      let status: "free" | "pinned" | "cap" | undefined = budget.left > 0 ? undefined : "cap";
      const local = localFor(comp);
      const seed = tv.map(([i]) => i);
      for (let size = 12; budget.left > 0 && size < local.size; size *= 3) {
        const B = ballOf(seed, size);
        const lix = new Map<number, number>();
        for (const v of B) lix.set(v, lix.size);
        const seen = new Set<JointRow>();
        const lrows: Array<{ idx: number[]; coef: number[]; rhs: number; eq: boolean }> = [];
        for (const v of B)
          for (const row of rowsOfVar[v] as JointRow[]) {
            if (seen.has(row)) continue;
            seen.add(row);
            const lr = { idx: [] as number[], coef: [] as number[], rhs: row.rhs, eq: row.eq };
            for (let k = 0; k < row.idx.length; k++) {
              const y = row.idx[k] as number;
              const j = lix.get(y);
              if (j === undefined) lr.rhs -= (row.coef[k] as number) * (x0[y] as number);
              else {
                lr.idx.push(j);
                lr.coef.push(row.coef[k] as number);
              }
            }
            lrows.push(lr);
          }
        const lobj = new Array<number>(B.length).fill(0);
        for (const [i, a] of tv) lobj[lix.get(i) as number] = (lobj[lix.get(i) as number] as number) + a;
        if (decide(() => lpFeasible(B.length, lrows, budget), lobj, truth, comp, B) === "free") {
          status = "free";
          break;
        }
      }
      if (status === undefined) {
        const obj = new Array<number>(local.size).fill(0);
        for (const [i, a] of tv) obj[local.get(i) as number] = (obj[local.get(i) as number] as number) + a;
        const globalOfComp: number[] = [];
        for (const [g, l] of local) globalOfComp[l] = g;
        const lpFor = (): LpState | undefined => {
          if (!lpOf.has(comp)) lpOf.set(comp, lpFeasible(local.size, compRows.map((r) => ({ idx: r.idx.map((i) => local.get(i) as number), coef: r.coef, rhs: r.rhs, eq: r.eq })), budget));
          return lpOf.get(comp);
        };
        status = decide(lpFor, obj, truth, comp, globalOfComp);
      }
      if (status === "free") {
        settledFree.add(t.id);
        continue;
      }
      // "pinned", or "cap" (the solver's iteration cap, or the call's budget
      // spent): withheld either way (RGD-895-failopen, D-0889(5)).
      const pick = (rs: readonly JointRow[]): string | undefined => {
        let v: string | undefined;
        for (const r of rs)
          for (const id of r.consts)
            if (cellOf(id).kind === "population" && viewOf(id) !== undefined && (v === undefined || byIdAsc(id, v) < 0)) v = id;
        return v;
      };
      const tset = new Set(tv.map(([i]) => i));
      const victim = pick(compRows.filter((r) => r.idx.some((i) => tset.has(i)))) ?? pick(compRows);
      if (victim === undefined) {
        // No served population cell enters this component. A served MONEY
        // constant is never withheld (O-10/O-15), so the object is refused
        // rather than served with the pin. With no served value at all in its
        // rows the pin holds in every world over this graph: nothing served
        // discloses it, and there is nothing to withhold.
        const money = compRows.flatMap((r) => r.consts).filter((id) => cellOf(id).kind === "money" && viewOf(id) !== undefined);
        if (money.length > 0) throw new Error(`analytics-cells: (3c) target '${t.id}' is pinned by served money cells only; refusing the day object`);
        settledFree.add(t.id);
        continue;
      }
      const target = t.terms[0]?.[0] as string;
      if (budget.left > 0) return [{ victim, target }];
      // Budget spent: withhold every served population cell of the component at
      // once, so the remaining rounds find nothing served to settle there.
      if (!spentComps.has(comp)) {
        spentComps.add(comp);
        for (const r of compRows) for (const id of r.consts) if (cellOf(id).kind === "population" && viewOf(id) !== undefined) out.push({ victim: id, target });
      }
    }
    return out;
  };

  /**
   * (3d)(c) I25-FX-DATA (AN-03/AN-04): the integer-bounds check for ONE flow equation
   * `total(T) − total(T−1) − Σ in + Σ out = 0` once some term is withheld. Unknowns: every withheld flow, total(T)
   * when withheld, and total(T−1) unless T−1 served it (`held` a number); a T−1 NOT on record (`undefined`) counts
   * as held when it is 0 — the start of a running total, which a reader knows without any object; D-1016(2): so does a
   * T−1 recorded withheld (`null`) when no value of the total was ever served (`firstTotal`). Each unknown is
   * `[0, K−1]` when the floor withheld it, else `[0, ∞)` — `pinsAnUnknown`'s model with signed terms. True when some
   * unknown is pinned to one value below K. I26 (I25-AN-05): a `slack` relation adds `+ s`, `s ∈ [0, ∞)`, never
   * tested as a pin, and its total(T−1) is known only as T−1's served value.
   */
  /** D-1016(2): no served value of this running total on record — T−1 recorded it withheld (`null`) with no served
   * value before, so the creator's first served total; the reader's start-at-0 guess is then the prior. */
  const firstTotal = (id: string): boolean => (priorServed?.get(id)?.length ?? 0) === 0;
  const flowPins = (f: FlowRelation, held: number | null | undefined): boolean => {
    // Σ sign·x over unknowns = c, where c moves every known term across. total(T−1) is total(T) − net (the true
    // value; a served value is the true one): the reader's equation is the true one.
    let net = 0;
    for (const id of f.inflow) net += val(id);
    for (const id of f.outflow) net -= val(id);
    const slack = f.slack === true;
    const prev = slack ? (typeof held === "number" ? held : 0) : val(f.total) - net;
    const terms: Array<{ sign: number; id: string | null; known: boolean }> = [
      { sign: 1, id: f.total, known: !hidden.has(f.total) },
      { sign: -1, id: null, known: typeof held === "number" || (!slack && prev === 0 && (held === undefined || (held === null && firstTotal(f.total)))) },
      ...f.inflow.map((id) => ({ sign: -1, id, known: !hidden.has(id) })),
      ...f.outflow.map((id) => ({ sign: 1, id, known: !hidden.has(id) })),
    ];
    let c = 0;
    const unknowns: Array<{ sign: number; hi: number }> = [];
    for (const t of terms) {
      const v = t.id === null ? prev : val(t.id);
      if (t.known) c -= t.sign * v;
      else unknowns.push({ sign: t.sign, hi: t.id !== null && why.get(t.id) === "floor: below K" ? K - 1 : Infinity });
    }
    const asked = unknowns.length; // the slack below is an unknown of the equation, never a pin
    if (slack) unknowns.push({ sign: 1, hi: Infinity });
    for (let i = 0; i < asked; i++) {
      let lo = 0;
      let hi = 0; // range of Σ_{j≠i} sign_j·x_j
      for (let j = 0; j < unknowns.length; j++) {
        if (j === i) continue;
        const u = unknowns[j] as { sign: number; hi: number };
        if (u.sign > 0) hi += u.hi;
        else lo -= u.hi;
      }
      const me = unknowns[i] as { sign: number; hi: number };
      // sign_i·x_i = c − rest, rest ∈ [lo, hi]
      const a = me.sign > 0 ? c - hi : -(c - lo);
      const b = me.sign > 0 ? c - lo : -(c - hi);
      const min = Math.max(0, a);
      const max = Math.min(me.hi, b);
      if (min === max && min < K) return true;
    }
    return false;
  };

  // ---- steps (2) and (3) to a fixed point, then (3c); repeat until (3c) finds no pin.
  for (let round = 0; round <= byId.size; round++) {
  for (let pass = 0; pass < byId.size + 2; pass++) {
    let changed = false;

    // (2) nestings: a served finer cell must sit at least K below every served
    // coarser cell it is contained in.
    for (const { outer, inner } of closure) {
      if (hidden.has(outer) || hidden.has(inner)) continue;
      const gap = val(outer) - val(inner);
      if (gap < K) {
        changed = hide(inner, `subset: ${inner} sits under ${outer} by a gap below K`) || changed;
      }
    }

    // (3) complementary suppression. An equation the reader can form must
    // leave every withheld cell in it a choice: two unknown terms, or one
    // unknown whose whole mass is at least K.
    for (const p of active) {
      const withheldChildren = p.children.filter((c) => hidden.has(c));
      const base = baseOf(p);
      const R = residualOf(p);
      let need = "";
      if (hidden.has(p.parent)) {
        // The reader cannot use this equation against a child — but they can
        // still recompute the PARENT from it, which would undo whatever
        // withheld the parent in the first place.
        if (withheldChildren.length === 0 && base === 0) need = `${p.parent} is Σ of its served children`;
      } else if (R < K && (R > 0 || withheldChildren.length > 0)) {
        // The unseen mass of this equation is below the floor, so whatever is
        // in it is pinned to an interval narrower than K.
        need = "residual below K";
      } else if (withheldChildren.length === 1 && base === 0) {
        // One withheld child and no residual beside it: `parent − Σ served` IS
        // that child. This is pass-3 V3c-2 — the day's new supporters over one
        // event's new fans, with no non-event remainder to hide them in.
        need = `${p.parent} leaves ${withheldChildren[0] as string} as its only unknown term`;
      }
      if (need !== "") {
        // WHICH cell is suppressed is itself a served fact, so the choice reads
        // NOTHING but the id: first by `byIdAsc` among the eligible served
        // children (docket #22(a), 2026-09-07). A value-dependent choice —
        // "the smallest, id breaking ties" — published an order over the
        // values, and with the equation beside it that order pinned a sub-K
        // child: 11 of them on the pool shape and a country of 3 on
        // `geo.country.*` (RUN-0012 B-1). Same rule at (3b) below.
        const servedKids = servedChildrenOf(p);
        let victim: string | undefined;
        for (const c of servedKids) {
          if (victim === undefined || byIdAsc(c, victim) < 0) victim = c;
        }
        if (victim !== undefined) {
          if (hide(victim, `complementary: ${need} (first served child of ${p.parent} by id)`)) {
            forcedBy.set(victim, p.parent);
            changed = true;
          }
        } else if (hide(p.parent, `complementary: ${need} (no served child left, so the total goes)`)) {
          forcedBy.set(p.parent, p.parent);
          changed = true;
        }
      }

      // (3b) integer bounds (B-1, narrowed). AFTER (3), so a term (3) withholds
      // is one this rule then sees. The reader forms ONE of two equations, and
      // the served value that carries it is the one that goes: an EMITTED pool
      // is `base + Σ withheld children` over a known base, so the pool goes;
      // otherwise the reader has `total − Σ served children`, whose unknowns
      // are the withheld children plus the base when `baseIsUnknown`, so the
      // total goes. Both are the pinned solver's own model
      // (`test/s002-servday.test.ts:256` and `:143-155`).
      const unknownKids = p.children.filter((c) => hidden.has(c));
      if (residualEmitted(p)) {
        if (unknownKids.length > 0 && pinsAnUnknown(residualOf(p) - baseOf(p), unknownKids, false)) {
          changed = hide(p.other as string, "residual not served: the pool pins a withheld term to one value") || changed;
        }
      }
      if (!hidden.has(p.parent) && !residualEmitted(p)) {
        const baseUnknown = baseIsUnknown(p);
        if ((unknownKids.length > 0 || baseUnknown) && pinsAnUnknown(residualOf(p), unknownKids, baseUnknown)) {
          // The victim is (3)'s: the id-first served child, else the total. It
          // is NOT the total while a child can go — withholding the total on a
          // pin and nothing else makes "the total left" the equation, and the
          // reader recovers the children from the suppression itself (the
          // RUN-0010 B-floors indistinguishability search finds them).
          let victim: string | undefined;
          for (const c of servedChildrenOf(p)) {
            if (victim === undefined || byIdAsc(c, victim) < 0) victim = c;
          }
          if (victim !== undefined) {
            if (hide(victim, "complementary: this breakdown pins a withheld term to one value (first served child by id)")) {
              forcedBy.set(victim, p.parent);
              changed = true;
            }
          } else if (hide(p.parent, "total not served: this breakdown pins a withheld term to one value")) {
            forcedBy.set(p.parent, p.parent);
            changed = true;
          }
        }
      }
    }

    // (3d) I25 (#841, I3-0158 b / I3-0611): FLOW-TO-TOTAL. A reader holding the
    // object before this one has `total(T) = total(T-1) + Σ in − Σ out` with every
    // flow served: (a) a served total hands over `total(T-1)`, which is a breach
    // unless it IS what T-1 served (`priorDay`): T-1 withheld it, or, not on record,
    // it sits 0 < |d| < K from a retained served value (as (1c)) or 0 < v < K (as
    // (1)); (b) a withheld total comes back as T-1's served value plus the flows.
    // The id-first flow cell goes; ids only in the choice.
    // (c) I25-FX-DATA (AN-03/AN-04): a flow cell already WITHHELD is an unknown of the same equation, not a reason
    // to skip it — `flowPins` below; the id-first SERVED flow goes, else the total.
    // (d) I26 (I25-RJ-02 all-hidden day; D-1013(4)): a total withheld on T beside T−1's SERVED value withholds EVERY
    // flow of the relation whatever its value — this subsumes (b) — so no withheld flow there is bounded below K by
    // its withholding, and the day where total, joins and leaves all sit below K after T−1 served 2K−2 leaves
    // each a choice. (e) I26 (I25-AN-05): a `slack` relation has no exact `total(T−1)`, so it is always judged by
    // `flowPins`; with every flow served, T−1's withheld total gains a bound, which (a) treats as a breach.
    for (const f of flows) {
      const flowIds = [...f.inflow, ...f.outflow];
      if (flowIds.length === 0 || flowIds.some((id) => cellOf(id).kind !== "population")) continue;
      const held = priorDay?.get(f.total);
      if (hidden.has(f.total) && typeof held === "number") {
        for (const id of flowIds) {
          if (hide(id, "complementary (flow): a running total withheld after the object before it served it withholds every flow beside it")) {
            forcedBy.set(id, f.total);
            changed = true;
          }
        }
        continue;
      }
      let breach: boolean;
      const someHidden = flowIds.some((id) => hidden.has(id));
      if (someHidden || f.slack === true) {
        breach = someHidden ? flowPins(f, held) : !hidden.has(f.total) && held === null;
        let victim: string | undefined;
        for (const id of flowIds) if (!hidden.has(id) && (victim === undefined || byIdAsc(id, victim) < 0)) victim = id;
        if (!breach) continue;
        const target = victim ?? (hidden.has(f.total) ? undefined : f.total);
        if (target !== undefined && hide(target, "complementary (flow): the day's served flows and running totals pin a withheld flow (first served flow cell by id, else the total)")) {
          forcedBy.set(target, f.total);
          changed = true;
        }
        continue;
      }
      if (hidden.has(f.total)) {
        // (f) I26-FX-DATA (JN-03 lead, D-1015(6)): a total withheld beside EVERY flow served is still one equation —
        // T−1 not on record at 0 (the start of the running total, the creator's first object) makes it known, so
        // `flowPins` judges it; on a pin the id-first flow goes (below).
        if (!flowPins(f, held)) continue;
        breach = true;
      } else {
        let net = 0;
        for (const id of f.inflow) net += val(id);
        for (const id of f.outflow) net -= val(id);
        const before = val(f.total) - net;
        breach =
          held === null ||
          (held !== before &&
            ((before > 0 && before < K) || (priorServed?.get(f.total) ?? []).some((v) => Math.abs(before - v) > 0 && Math.abs(before - v) < K)));
      }
      if (!breach) continue;
      let victim: string | undefined;
      for (const id of flowIds) if (victim === undefined || byIdAsc(id, victim) < 0) victim = id;
      if (victim !== undefined && hide(victim, "complementary (flow): the day's flows and a running total recover another day object's total (first flow cell by id)")) {
        forcedBy.set(victim, f.total);
        changed = true;
      }
    }

    if (!changed) break;
  }
  const joint = jointVictim();
  if (joint.length === 0) break;
  for (const j of joint) {
    if (hide(j.victim, "complementary (joint): the served relations together pin a withheld term to one value (first served cell by id)")) forcedBy.set(j.victim, j.target);
  }
  }

  // ---- the residual cells that are not emitted are withheld.
  const pool = new Map<string, number>();
  for (const p of active) {
    if (p.other === undefined) continue;
    if (residualEmitted(p)) pool.set(p.other, residualOf(p));
    else if (cellOf(p.other).kind !== "virtual") {
      hide(
        p.other,
        `residual not served: ${servedChildrenOf(p).length === 0 ? "no served child, so the total IS the residual" : "pool below K"}`,
      );
    }
  }

  // ---- step (4): totals over served children.
  const served = new Map<string, number>();
  for (const c of byId.values()) {
    if (c.kind === "virtual" || hidden.has(c.id)) continue;
    served.set(c.id, residualIds.has(c.id) ? (pool.get(c.id) as number) : c.value);
  }
  for (const p of active) {
    if (!served.has(p.parent)) continue;
    const terms: string[] = [...servedChildrenOf(p)];
    if (p.other !== undefined && served.has(p.other)) terms.push(p.other);
    if (terms.length === 0) {
      // ADR §6 geo row: a partition whose `other` is below the floor serves
      // ONLY its total, and that total is then the TRUE total — there is
      // nothing served beneath it for the total to be checked against, and a
      // sum over an empty set would be a lie rather than a suppression.
      why.set(p.parent, `served ${val(p.parent)}: true total; every term of its breakdown is withheld (ADR §6)`);
      continue;
    }
    let sum = 0;
    for (const t of terms) sum += served.get(t) as number;
    if (sum === val(p.parent)) {
      // The pool absorbed every withheld child, so Σ served children + other IS
      // the true total: step (4) closes the reader's arithmetic exactly.
      served.set(p.parent, sum);
    }
    // Otherwise the breakdown does not enumerate the parent (an event's join
    // days beside a pool that is not emitted): the total stays true and the
    // difference is the residual, which step (3) has already held at ≥ K.
  }

  const withheld = new Set<string>();
  for (const c of byId.values()) {
    if (c.kind === "virtual") continue;
    if (!served.has(c.id)) withheld.add(c.id);
  }

  const complementary = new Map<string, string>();
  for (const [id, parent] of [...forcedBy].sort((a, b) => byIdAsc(a[0], b[0]))) if (withheld.has(id)) complementary.set(id, parent);

  return {
    served,
    withheld,
    complementary,
    explain(id: string): string {
      const c = byId.get(id);
      if (c === undefined) return `${id}: not a cell of this day object`;
      if (c.kind === "virtual") return `${id}: ${why.get(id) ?? "virtual"}`;
      const s = served.get(id);
      if (s !== undefined) return why.get(id) ?? `served ${s}`;
      return `withheld — ${why.get(id) ?? "not served"}`;
    },
  };
}

/**
 * THE QUEUE CONSUMER (ADR-0006 §2/§3). One outbox row in, two to four analytics
 * rows out, per message, in ONE atomic `ANALYTICS_DB.batch`.
 *
 * ---------------------------------------------------------------------------
 * WHY THIS IS A PURE FUNCTION OVER `DbLike`
 * ---------------------------------------------------------------------------
 * `consumeBatch` never touches a queue. It takes the message BODIES and returns
 * which `event_key`s to ack and which to retry; `index.ts`'s `queue()` does the
 * ack/retry. Everything worth testing here — idempotency, commutativity, the
 * per-cell seal rules, the erasure sentinel — is then testable without a queue
 * runtime, and the one thing that is not (the ack call itself) is three lines.
 *
 * ---------------------------------------------------------------------------
 * THE SHAPE OF A MESSAGE'S BATCH, AND WHY THAT ORDER
 * ---------------------------------------------------------------------------
 *   [ INSERT INTO processed_events, <the kind's upserts> ]
 *
 * The ledger row is FIRST because it is the idempotency control: `event_key` is
 * the table's primary key, so a second delivery of the same row violates it,
 * the whole batch throws, and NOTHING in it applies. A duplicate is therefore
 * not "detected and skipped" by a read — it is impossible to apply, which is a
 * different and stronger property, because a read-then-write would race a
 * concurrent invocation holding the same message.
 *
 * A throw from that PK means "already processed" and is ACKed. Any OTHER throw
 * is retried: the batch is atomic, so a retry re-runs a batch none of whose
 * statements landed.
 *
 * Statement ORDER INSIDE the batch is load-bearing a second time. D1 runs a
 * batch sequentially, so a statement can ask "was this fan already a member?"
 * only while the membership INSERT that answers it has not run yet. Every
 * first-time-only counter (`creator_cohort_daily.size`, an event's
 * `participants`/`new_fans`, `buyers_new`, `dead_attempts`) is therefore
 * written as `INSERT … SELECT … WHERE NOT EXISTS (<the membership row>)` placed
 * BEFORE that row's `INSERT OR IGNORE`. The alternative — reading `changes`
 * back off the membership insert — is not available: `DbLike.batch` returns
 * `unknown[]` (db.ts:20) and D1's per-statement `meta` is not part of that
 * contract.
 *
 * ---------------------------------------------------------------------------
 * LATENESS IS A PROPERTY OF THE TARGET CELL, NOT OF THE MESSAGE (ADR §3)
 * ---------------------------------------------------------------------------
 * The consumer's clock is `opts.now()`, read ONCE per message at
 * statement-build time, and `today` is its UTC day. Then, per target cell:
 *
 *   - a day cell of day D is unsealed iff `D >= today`;
 *   - a window cell `d<w>` of cohort day C is unsealed iff `C + w + 1 > today`;
 *   - a scheduled-clear cell on `clears_day` is unsealed iff `clears_day >= today`.
 *
 * A COUNTER whose day cell has sealed lands on `today`'s row in the SAME column
 * (the arrival day) and marks `platform_incomplete_days(today)`. A PER-FAN
 * CLAIM whose cell has sealed is DROPPED and marks the same day: shifting a
 * per-(fan, creator, day) claim onto another day would count that fan twice.
 *
 * Incompleteness is platform-wide (A2-vi/O-3). There is no per-creator column
 * for it and there must never be one — a per-creator late count is a
 * fan-derived number on a creator surface.
 *
 * ---------------------------------------------------------------------------
 * THE RESURRECTION SENTINEL IS ON EVERY FAN-KEYED UPSERT (A4)
 * ---------------------------------------------------------------------------
 * `AND NOT EXISTS (SELECT 1 FROM analytics_erased WHERE passport_id = ? AND
 * fan_key = ?)` guards every membership insert, every membership-derived
 * counter and every retention flag. A message that was in flight when the
 * erasure drain ran — or replayed from `analytics_outbox_dead`, which is kept
 * 30 days against the sentinel's 90 — cannot re-create the fan.
 *
 * The ±1 day counters are NOT guarded, deliberately: they are "one account per
 * increment" facts, and erasure reaches them through the creator-scoped
 * decrement events of ADR §3 (`supporter_erased`, `card_erased`,
 * `achievement_erased`) on the arrival day, not by refusing an
 * increment that already happened.
 *
 * ---------------------------------------------------------------------------
 * WHY THE COHORT PARTITION CLOSES (S-15)
 * ---------------------------------------------------------------------------
 * ADR §0 P1 declares `cohort sources → new_supporters` a PARTITION, and a
 * partition is a claim about WRITERS, not a sum somebody checks afterwards.
 * Exactly ONE path in this file writes a `creator_cohort_daily.size`:
 * `fan_followed`, which writes `new_supporters` in the same statement list, on
 * the same day, from the same message. No other path writes either.
 *
 * The path that could have broken it is `war_joined`. A fan first seen through
 * an event becomes a CREATOR-scope member — otherwise a second event would call
 * them new again — and that membership carries `source = NULL`: it is counted
 * into no cohort row, and no `new_supporters` fires for it either. B6 chose
 * NULL over inventing a source precisely so that both sides of the relation
 * stay empty rather than one side gaining a term the other lacks. `source IS
 * NOT NULL` on the retention fold's cohort increment is the same decision
 * carried forward: a `d<w>` on a cohort whose `size` never counted that fan
 * would make `d<w> ⊆ size` false.
 *
 * What the relation gives, exactly: every `size` increment is accompanied by a
 * `new_supporters` increment on its own day, so `Σ_source size(d) <=
 * new_supporters(d)` always, with equality unless one fan follows the same
 * creator twice in one day — `new_supporters` is a ±1 event counter and counts
 * both follows, `size` is membership-derived and counts the fan once (A3's
 * same-day re-follow is two events by design, the nonce `event_key` making
 * them distinct). The inequality is in the safe direction for `serveDay()`
 * step (4), which recomputes a total as Σ over its served children.
 */

/**
 * Is this string a REAL UTC day, not merely day-SHAPED?
 *
 * Shape is not enough for a CELL KEY: `9999-99-99` matches
 * `^\d{4}-\d{2}-\d{2}$` and is not a date. `strictDateUtcMs` (dates.ts) checks
 * shape, month and day-of-month (leap-aware) and returns null rather than
 * throwing: a validator that can throw is a second unusable-message path.
 */
function isCalendarDay(d: string): boolean {
  // I3-1750: `strictDateUtcMs` (dates.ts) is the one strict parser; it also
  // does not remap years 0..99 to 19xx the way a bare `Date.UTC` does.
  return strictDateUtcMs(d) !== null;
}
const DAY_MS = 86_400_000;

/**
 * UTC day arithmetic on 'YYYY-MM-DD' strings. Via `strictDateUtcMs`, so it is calendar
 * arithmetic (month lengths, leap days) rather than string slicing, and it
 * never touches local time — the whole design is keyed by UTC days.
 *
 * Lexicographic comparison of two such strings IS chronological comparison, so
 * every seal predicate below is a `<`/`>=` on the strings themselves.
 */
export function addDays(day: string, n: number): string {
  const base = strictDateUtcMs(day);
  if (base === null) throw new Error(`addDays: '${String(day)}' is not a UTC day`);
  return new Date(base + n * DAY_MS).toISOString().slice(0, 10);
}

/**
 * Which branch a batch takes (S-6). The SAME worker is both consumers, so
 * `batch.queue` is what distinguishes them at runtime. The explicit name wins:
 * `index.ts`'s `queue()` passes `env.ANALYTICS_DLQ_NAME`, which every
 * `wrangler.jsonc` env sets to its own `dead_letter_queue` (AC-07). The
 * `-dlq` suffix is the fallback for a deployment that omits that var.
 */
export function isDeadLetterQueue(queueName: string | undefined, dlqName?: string): boolean {
  if (!queueName) return false;
  if (dlqName && queueName === dlqName) return true;
  return queueName.endsWith("-dlq");
}

/** Statement collector: `add(sql, ...params)` so a call site reads as one line
 * of SQL and its bindings, and the batch is a plain array at the end. */
class Stmts {
  readonly out: DbStatement[] = [];
  constructor(private readonly db: DbLike) {}
  add(sql: string, ...params: unknown[]): void {
    this.out.push(this.db.prepare(sql).bind(...params));
  }
}

/** `AND NOT EXISTS (…)` on BOTH sentinel tables, spelled once. A fan's erasure
 * is written to two rows (`analytics-erasure.ts eraseScope`) because a creator's
 * retraction deletes one of them and not the other (docket #23 C-2, S4-1); a
 * guard that read only the deletable one would start passing for a fan who asked
 * to be forgotten, so it reads both and either row refuses the write.
 *
 * TWO SEPARATE `NOT EXISTS`, not one over a UNION: analytics/0001's header
 * permits exactly one shape of read on these tables — ONE `(passport_id,
 * fan_key)` at a time — and a subquery that selected the tables and filtered
 * outside would bind neither, which is the cross-creator read that rule refuses.
 * Its FOUR parameters are `(passport_id, fan_key)` twice, one pair per table,
 * appended by the caller in the same order. */
const NOT_ERASED =
  "NOT EXISTS (SELECT 1 FROM analytics_erased WHERE passport_id = ? AND fan_key = ?) " +
  "AND NOT EXISTS (SELECT 1 FROM analytics_erasure_sentinels WHERE passport_id = ? AND fan_key = ?)";
/** `NOT EXISTS (<this membership row>)` — "this fan has not been counted in this
 * scope yet", which is what makes a membership-derived counter fire once. */
const NOT_MEMBER = "NOT EXISTS (SELECT 1 FROM analytics_membership WHERE passport_id = ? AND scope_kind = ? AND scope_id = ? AND fan_key = ?)";

/** A ±1 day counter on `creator_metrics_daily`. Commutative by construction:
 * `col = col + ?`, never `col = ?`. */
function metric(s: Stmts, psp: string, day: string, col: string, delta = 1): void {
  if (!METRIC_COLUMNS.has(col)) throw new Error(`analytics-consumer: '${col}' is not a creator_metrics_daily counter`);
  s.add(
    `INSERT INTO creator_metrics_daily (passport_id, day, ${col}) VALUES (?, ?, ?) ` +
      `ON CONFLICT(passport_id, day) DO UPDATE SET ${col} = ${col} + ?`,
    psp,
    day,
    delta,
    delta,
  );
}

/**
 * I3-0356: the follow transitions count each FAN once per landing day, never each event.
 * `analytics_follow_net` (analytics/0009) sums the fan's `fan_followed` (F) and `fan_unfollowed` (U) for the
 * (creator, landing day) and records what the fan last contributed (`cn`, `cu`). Per fan:
 *   new_supporters = [F > U] + [F = U >= 1 AND f],  unfollows = [U > F] + [F = U >= 1 AND f]
 * where f = "this fan's FIRST-EVER creator-scope membership landed on this day" (the row `cohortSize`
 * counted; written before this runs). So follow/unfollow/follow is one follow; sum(new_supporters -
 * unfollows) is the day's net change; a fan the day's cohort counted is always in new_supporters, so the
 * cohort sizes still partition it (analytics-cells cohort.sources_other >= 0); and both counters are a
 * function of commutative sums and of f, which only this fan's own follow can set — so any delivery order
 * ends on the same rows. Each counter moves by (target - what this fan contributed), then the row records
 * the target. Rows of days before `today` are pruned here.
 * AN ERASED FAN (either sentinel) gets NO row — the row would re-store the pseudonym erasure removed — and
 * keeps the pre-I3-0356 per-event counter instead: the ±1 counters are unguarded by design (ADR §3), because
 * `supporter_erased` has already taken the fan out of the stock, so dropping a follow that lands after the
 * erasure would leave sum(new_supporters - unfollows) one short (I22-FX-TAIL round 2, erasure-race probe).
 */
function followNet(s: Stmts, psp: string, day: string, today: string, fanKey: string, to: 0 | 1): void {
  const key = "passport_id = ? AND day = ? AND fan_key = ?";
  const k = [psp, day, fanKey];
  const f = "EXISTS (SELECT 1 FROM analytics_membership WHERE passport_id = ? AND scope_kind = 'creator' AND scope_id = ? AND fan_key = ? AND cell_day = ?)";
  const fp = [psp, psp, fanKey, day];
  const both = `(follows = unfollows AND follows >= 1 AND ${f})`;
  const newT = `((follows > unfollows) + ${both})`;
  const unfT = `((unfollows > follows) + ${both})`;
  s.add(
    "INSERT INTO analytics_follow_net (passport_id, day, fan_key, follows, unfollows) " +
      `SELECT ?, ?, ?, ?, ? WHERE ${NOT_ERASED} ` +
      "ON CONFLICT(passport_id, day, fan_key) DO UPDATE SET follows = follows + excluded.follows, unfollows = unfollows + excluded.unfollows",
    ...k,
    to,
    1 - to,
    psp,
    fanKey,
    psp,
    fanKey,
  );
  for (const [col, target, mine] of [["new_supporters", newT, "cn"], ["unfollows", unfT, "cu"]] as const) {
    s.add(
      `INSERT INTO creator_metrics_daily (passport_id, day, ${col}) ` +
        `SELECT ?, ?, d FROM (SELECT ${target} - ${mine} AS d FROM analytics_follow_net WHERE ${key}) WHERE d <> 0 ` +
        `ON CONFLICT(passport_id, day) DO UPDATE SET ${col} = ${col} + excluded.${col}`,
      psp,
      day,
      ...fp,
      ...k,
    );
  }
  // I21-Q164 (D-1030 (a), D-0820(3)): X(e, day) = creator_event_performance.new_supporters (analytics/0011) moves by
  // EXACTLY this fan's new_supporters delta when this fan is a war-NEW fan of event e on this day (the event-scope
  // membership row the war join wrote: acquired = 1, cell_day = day). Placed BEFORE the `cn` update, which it reads.
  // No follow_net row (an erased fan) moves nothing: that fan's fallback +1 below reaches new_supporters only.
  s.add(
    `UPDATE creator_event_performance SET new_supporters = new_supporters + (SELECT ${newT} - cn FROM analytics_follow_net WHERE ${key}) ` +
      `WHERE passport_id = ? AND event_kind = 'war' AND join_day = ? AND EXISTS (SELECT 1 FROM analytics_follow_net WHERE ${key}) ` +
      "AND event_id IN (SELECT scope_id FROM analytics_membership WHERE passport_id = ? AND scope_kind = 'event' AND fan_key = ? AND cell_day = ? AND acquired = 1)",
    ...fp,
    ...k,
    psp,
    day,
    ...k,
    psp,
    fanKey,
    day,
  );
  s.add(`UPDATE analytics_follow_net SET cn = ${newT}, cu = ${unfT} WHERE ${key}`, ...fp, ...fp, ...k);
  const col = to === 1 ? "new_supporters" : "unfollows";
  s.add(
    `INSERT INTO creator_metrics_daily (passport_id, day, ${col}) SELECT ?, ?, 1 WHERE NOT (${NOT_ERASED}) ` +
      `ON CONFLICT(passport_id, day) DO UPDATE SET ${col} = ${col} + 1`,
    psp,
    day,
    psp,
    fanKey,
    psp,
    fanKey,
  );
  s.add("DELETE FROM analytics_follow_net WHERE passport_id = ? AND day < ?", psp, today);
}

interface Member {
  readonly psp: string;
  readonly scopeKind: "creator" | "event" | "buyer" | "release_attempt";
  readonly scopeId: string;
  readonly fanKey: string;
  readonly fanKeyV: number;
  /** The day key of THE CELL THE ACCOMPANYING COUNTER INCREMENTED (BREAK 5). */
  readonly cellDay: string;
  /** The cohort source, or NULL when this membership was not counted into a
   * `creator_cohort_daily` row (see `war_joined` below). */
  readonly source: string | null;
  /**
   * S-14, EVENT SCOPE ONLY: SQL that yields 1 when this event ACQUIRED the fan.
   * It is an expression rather than a value because the only thing that can
   * answer it is the database, in this batch, BEFORE the creator-scope
   * membership this same message is about to write.
   */
  readonly acquiredExpr?: { readonly sql: string; readonly params: readonly unknown[] };
}

/** The membership row. `INSERT OR IGNORE` (never-decremented, idempotent) and
 * sentinel-guarded, so a replay after erasure inserts nothing. */
function membership(s: Stmts, m: Member): void {
  s.add(
    "INSERT OR IGNORE INTO analytics_membership (passport_id, scope_kind, scope_id, fan_key, fan_key_v, cell_day, source, acquired) " +
      `SELECT ?, ?, ?, ?, ?, ?, ?, ${m.acquiredExpr ? `CASE WHEN ${m.acquiredExpr.sql} THEN 1 ELSE 0 END` : "0"} WHERE ${NOT_ERASED}`,
    m.psp,
    m.scopeKind,
    m.scopeId,
    m.fanKey,
    m.fanKeyV,
    m.cellDay,
    m.source,
    ...(m.acquiredExpr ? m.acquiredExpr.params : []),
    m.psp,
    m.fanKey,
    m.psp,
    m.fanKey,
  );
}

/**
 * A `creator_event_performance` counter that fires once per fan per scope.
 * `alsoAbsent` is a SECOND membership row that must also be missing — that is
 * how `new_fans` says "and this fan is not already a member of the creator's
 * audience" without a second round-trip.
 */
function eventFirstTime(
  s: Stmts,
  args: {
    psp: string;
    eventKind: string;
    eventId: string;
    joinDay: string;
    col: "participants" | "new_fans" | "dead_attempts" | "acquirers";
    fanKey: string;
    scopeKind: string;
    scopeId: string;
    alsoAbsent?: { scopeKind: string; scopeId: string };
  },
): void {
  const extra = args.alsoAbsent ? ` AND ${NOT_MEMBER}` : "";
  const extraParams = args.alsoAbsent ? [args.psp, args.alsoAbsent.scopeKind, args.alsoAbsent.scopeId, args.fanKey] : [];
  s.add(
    `INSERT INTO creator_event_performance (passport_id, event_kind, event_id, join_day, ${args.col}) ` +
      `SELECT ?, ?, ?, ?, 1 WHERE ${NOT_ERASED} AND ${NOT_MEMBER}${extra} ` +
      `ON CONFLICT(passport_id, event_kind, event_id, join_day) DO UPDATE SET ${args.col} = ${args.col} + 1`,
    args.psp,
    args.eventKind,
    args.eventId,
    args.joinDay,
    args.psp,
    args.fanKey,
    args.psp,
    args.fanKey,
    args.psp,
    args.scopeKind,
    args.scopeId,
    args.fanKey,
    ...extraParams,
  );
}

/**

 * I3-1241), none with an `acq:` prefix — so
 * the retention fold (which joins `e.scope_id = event_id AND e.acquired = 1`;
 * these rows carry `acquired = 0`), the `NOT_MEMBER` guards of real events and

 * and `analytics-read.ts readDayInput` subtracts it under the same key.
 */
export const releaseAcquirerScope = (releaseId: string): string => `acq:${releaseId}`;

/**
 * THE RETENTION FOLD (ADR §3/O-6). A `fan_active` for a fan who is already a
 * creator-scope member sets the member's `r<w>` flag and increments the cohort's
 * `d<w>`, once per window.
 *
 * The window's seal predicate is `cohort_day + w + 1 > today`, and the cohort
 * day is IN THE ROW rather than in the message — so the predicate is rewritten
 * as its bound on `cell_day`: `cell_day >= today − w`. That is computable at
 * statement-build time from the consumer's own clock, which is what keeps the
 * decision on the consumer's side of ADR §3 rather than the database's.
 *
 * `source IS NOT NULL` on the cohort increment: a membership created by a path
 * that wrote no `creator_cohort_daily.size` (see `war_joined`) carries a NULL
 * source, and incrementing `d<w>` on a cohort whose `size` that fan was never
 * counted into would make `d<w> ⊆ size` false. The FLAG is still set, because
 * it is the member's own state.
 */
function retention(s: Stmts, psp: string, fanKey: string, today: string): void {
  for (const w of [1, 7, 30] as const) {
    const col = `d${w}` as const;
    const flag = `r${w}` as const;
    // `cohort_day + w + 1 > today`  ⟺  `cohort_day >= today − w`.
    const minCohortDay = addDays(today, -w);
    // I3-0717 / I3-1240: the window counts activity strictly AFTER the cohort
    // day (`cohort_day <= today − 1`). A membership with `cell_day < today` was
    // committed on an earlier processing day, so which of a same-day
    // follow/join and `fan_active` the queue delivers first cannot change the
    // result — at `<= today` the active-first order silently lost the fact.
    const maxCohortDay = addDays(today, -1);
    s.add(
      `INSERT INTO creator_cohort_daily (passport_id, cohort_day, source, ${col}) ` +
        `SELECT m.passport_id, m.cell_day, m.source, 1 FROM analytics_membership m ` +
        "WHERE m.passport_id = ? AND m.scope_kind = 'creator' AND m.scope_id = ? AND m.fan_key = ? " +
        `AND m.${flag} = 0 AND m.source IS NOT NULL AND m.cell_day >= ? AND m.cell_day <= ? AND ${NOT_ERASED} ` +
        `ON CONFLICT(passport_id, cohort_day, source) DO UPDATE SET ${col} = ${col} + 1`,
      psp,
      psp,
      fanKey,
      minCohortDay,
      maxCohortDay,
      psp,
      fanKey,
      psp,
      fanKey,
    );
    // S-14. `acquired_d30` is a fact about the EVENT, decided by a flag on the
    // CREATOR-scope row — so it is attributed here, in the one statement that
    // knows the flag is about to flip, and BEFORE the flip, which is what makes
    // it fire exactly once per acquired fan. Its two `EXISTS` are the two halves
    // of the attribution: (a) an event-scope row for this key that CLAIMED the
    // acquisition (`acquired = 1`, written by `war_joined` only when no
    // creator-scope row existed), and (b) that creator-scope row still at
    // `r30 = 0` inside the window — the same predicate the flip below runs on,
    // so the two cannot disagree. The window bound is the event's own
    // (`join_day + 31 > today`), which for an acquiring event is the same day as
    // the cohort's, because the two rows were written by one message.
    if (w === 30) {
      s.add(
        "UPDATE creator_event_performance SET acquired_d30 = acquired_d30 + 1 " +
          "WHERE passport_id = ? AND join_day >= ? AND join_day <= ? " +
          "AND EXISTS (SELECT 1 FROM analytics_membership e WHERE e.passport_id = ? AND e.scope_kind = 'event' " +
          "AND e.scope_id = creator_event_performance.event_id AND e.cell_day = creator_event_performance.join_day " +
          "AND e.fan_key = ? AND e.acquired = 1) " +
          "AND EXISTS (SELECT 1 FROM analytics_membership c WHERE c.passport_id = ? AND c.scope_kind = 'creator' " +
          `AND c.scope_id = ? AND c.fan_key = ? AND c.r30 = 0 AND c.cell_day >= ? AND c.cell_day <= ?) AND ${NOT_ERASED}`,
        psp,
        minCohortDay,
        maxCohortDay,
        psp,
        fanKey,
        psp,
        psp,
        fanKey,
        minCohortDay,
        maxCohortDay,
        psp,
        fanKey,
        psp,
        fanKey,
      );
    }
    s.add(
      `UPDATE analytics_membership SET ${flag} = 1 ` +
        "WHERE passport_id = ? AND scope_kind = 'creator' AND scope_id = ? AND fan_key = ? " +
        `AND ${flag} = 0 AND cell_day >= ? AND cell_day <= ? AND ${NOT_ERASED}`,
      psp,
      psp,
      fanKey,
      minCohortDay,
      maxCohortDay,
      psp,
      fanKey,
      psp,
      fanKey,
    );
  }
}

/** I3-0719 / I3-2011: `planUpserts`' third answer — the message names a value

const SKEW = "skew" as const;
type Plan = boolean | typeof SKEW;

/** Kinds the vocabulary RETIRED (S-090 #314, QUEUE #348). No later deploy will

const RETIRED_KINDS: ReadonlySet<string> = new Set(["link_erased", "account_linked", "account_unlinked"]);

/**
 * `null` means MALFORMED (FINDING-0010-A-8), and is not the same answer as
 * `{}`. `{}` is "this message carried no dimensions", which is correct for
 * every kind that declares none; the old spelling gave the same answer for
 * unparseable bytes and for a JSON scalar/array, so a corrupted `dims` column
 * became "no dims" and every kind whose dims are OPTIONAL wrote its cell as if
 * the producer had sent none. The caller turns `null` into the refusal the
 * arrival-day flag already exists for.
 */
function parseDims(raw: string | null): Record<string, unknown> | null {
  if (!raw) return {};
  try {
    const v: unknown = JSON.parse(raw);
    return typeof v === "object" && v !== null && !Array.isArray(v) ? (v as Record<string, unknown>) : null;
  } catch {
    return null;
  }
}

/** Kinds that are deliberately not written in Phase 1. Each ACKs with no cell

 * still advance, because a message the consumer refuses to write is still a
 * message it has SEEN — leaving the watermark behind it would stall the
 * contiguous advance forever and keep the outbox prune from ever running. */
const NO_CELL: ReadonlySet<string> = new Set([
  // S-10: the link vocabulary is out of Phase 1 (`linked_2plus_delta` /
  // `linked_3plus_delta` have no writer). `link_erased` (S-090 #314) and
  // `account_linked`/`account_unlinked` (QUEUE #348) left the vocabulary: no
  // producer ever emitted them.
  // S-11: platform-level, no creator.
  "credit_purchase",
  // Creator-scoped FACTS with no cell: B7 reads them from production.
  "creator_claimed",
  "release_published",
  "release_withdrawn",
]);

/**
 * Build one message's statements. Returns `null` when the message is malformed
 * beyond use (a kind that needs a scope and has none, a fan-keyed kind with no
 * key): such a message can never succeed, so it is ACKed with the ledger row,
 * the watermark and the incompleteness flag — never retried into the DLQ for a
 * defect a redelivery cannot fix.
 */
function planUpserts(s: Stmts, msg: OutboxMessage, today: string): Plan {
  const psp = msg.passport_id;
  const kind = msg.event_type;

  // ARRIVAL-SIDE VALIDATION (FINDING-0010-MERGED-A-7/8). `assertEvent`
  // (`analytics-events.ts:249`) runs in the OTHER database, against the
  // producer's own object; it is not this consumer's guard. The queue carries
  // whatever it carries, and `occurred_day` is a CELL KEY — `msg.occurred_day <
  // today` is a lexicographic compare that accepts "9999-99-99" and
  // "2099-01-01" as happily as a real day, and either one becomes a
  // `creator_metrics_daily` row nothing downstream can distinguish from a
  // sealed cell. A day this consumer cannot parse can never be fixed by a
  // redelivery, so it takes the same disposition as any other unusable
  // message: `false`, which the caller turns into the arrival-day
  // incompleteness flag plus the ack.
  //
  // UPDATE (I3-1436, D-0829): the "WHAT THIS DOES NOT DO" note that used to
  // sit here was overtaken by the ruling below — this consumer now DOES bound
  // how far ahead a day may be. `occurred_day >= today` is still the
  // definition of an unsealed cell, so tomorrow (today + 1) stays accepted as
  // the documented UTC-boundary case; the bound the next check adds is the
  // upper one, not a replacement for this one.
  if (!isCalendarDay(msg.occurred_day)) return false;
  // I3-1436 (D-0829): tomorrow (today + 1) stays allowed for the documented
  // UTC-boundary case above; anything further ahead is a clock or forgery
  // error and must not seal into a future day object. D-0845(6) (the adopted
  // I3-1436 ruling): it is DEAD-LETTERED, not acked — SKEW writes no batch, so
  // no `processed_events` row burns its key, and the DLQ leg (`consumeDead`)
  // writes the replayable `analytics_outbox_dead` row and the arrival-day flag.
  if (msg.occurred_day > addDays(today, 1)) return SKEW;
  const dims = parseDims(msg.dims);
  if (dims === null) return false;

  if (NO_CELL.has(kind)) return true;

  // A day cell of D is unsealed iff D >= today; a sealed COUNTER lands on the
  // arrival day in the same column, and that shift is what makes the day
  // incomplete.
  const late = msg.occurred_day < today;
  const day = late ? today : msg.occurred_day;
  if (late) incomplete(s, today);

  const fanKey = msg.fan_key;
  const fanKeyV = msg.fan_key_v ?? 1;
  const scopeId = msg.scope_id;
  const needsFan = (): boolean => typeof fanKey === "string" && fanKey.length > 0;
  const needsScope = (): boolean => typeof scopeId === "string" && scopeId.length > 0;

  switch (kind) {
    // ---- fan-keyed ---------------------------------------------------------
    case "fan_followed": {
      if (!needsFan()) return false;
      if (dims.source !== undefined && typeof dims.source !== "string") return false;
      const src = typeof dims.source === "string" ? dims.source : DEFAULT_FOLLOW_SOURCE;
      if (!COHORT_SOURCES.has(src)) return SKEW;
      const m: Member & { source: string } = { psp, scopeKind: "creator", scopeId: psp, fanKey: fanKey as string, fanKeyV, cellDay: day, source: src };
      cohortSize(s, m); // BEFORE the membership row: it asks whether that row is absent.
      membership(s, m);
      followNet(s, psp, day, today, fanKey as string, 1); // AFTER it: reads whether that row landed today.
      return true;
    }
    case "fan_unfollowed":
      if (!needsFan()) return false;
      followNet(s, psp, day, today, fanKey as string, 0);
      return true;
    case "fan_active": {
      if (!needsFan()) return false;
      // A PER-FAN CLAIM, so a late one is DROPPED rather than shifted (ADR §3):
      // moving "this fan was active on D" onto the arrival day would count the
      // fan on a day they were not seen, and the window arithmetic below reads
      // `today` — which for a late message is not the day the claim describes.
      if (late) return true; // `incomplete(today)` was already recorded above.
      retention(s, psp, fanKey as string, today);
      return true;
    }
    case "scout_created":
      metric(s, psp, day, "scouts");
      return true;
    case "question_submitted":
      metric(s, psp, day, "questions");
      return true;
    case "handshake_requested":
      metric(s, psp, day, "handshakes_requested");
      return true;
    case "collectible_acquired": {
      if (!needsFan() || !needsScope()) return false;
      metric(s, psp, day, "collectibles_acquired");
      eventCounter(s, psp, "release", scopeId as string, day, "acquisitions");
      // I3-0356: the DISTINCT acquirers of this release, first acquisition only
      // (the same shape as `dead_attempts`), so the acquisitions cell can be
      // floored on people rather than on actions. BEFORE its membership row.
      eventFirstTime(s, {
        psp,
        eventKind: "release",
        eventId: scopeId as string,
        joinDay: day,
        col: "acquirers",
        fanKey: fanKey as string,
        scopeKind: "event",
        scopeId: releaseAcquirerScope(scopeId as string),
      });
      membership(s, { psp, scopeKind: "event", scopeId: releaseAcquirerScope(scopeId as string), fanKey: fanKey as string, fanKeyV, cellDay: day, source: null });
      buyersNew(s, psp, day, fanKey as string); // BEFORE the buyer membership row.
      membership(s, { psp, scopeKind: "buyer", scopeId: psp, fanKey: fanKey as string, fanKeyV, cellDay: day, source: null });
      return true;
    }
    case "pack_opened": {
      if (!needsFan()) return false;
      metric(s, psp, day, "packs_opened");
      // ADR §2: `buyers_new` is written from the fan-keyed `collectible_acquired`
      // / `pack_opened` membership. Both doors, one scope, so a fan who arrives
      // through a pack is not counted a second time when they later buy.
      buyersNew(s, psp, day, fanKey as string);
      membership(s, { psp, scopeKind: "buyer", scopeId: psp, fanKey: fanKey as string, fanKeyV, cellDay: day, source: null });
      return true;
    }
    case "war_joined": {
      if (!needsFan() || !needsScope()) return false;
      // D-0840(2): a PRACTICE war is acknowledged and counted in no cell. A row
      // with no `mode` predates the dim and was emitted as a ranked count.
      if (dims.mode === "practice") return true;
      if (dims.mode !== undefined && dims.mode !== "ranked") return false;
      const key = fanKey as string;
      const eventId = scopeId as string;
      metric(s, psp, day, "war_joins");
      // Both counters are membership-derived (ADR §4: "each a never-decremented
      // membership counter"), so both are placed before the two membership rows
      // they interrogate. `new_fans` also requires the CREATOR-scope row to be
      // absent, which is what "new to this creator" means.
      eventFirstTime(s, { psp, eventKind: "war", eventId, joinDay: day, col: "participants", fanKey: key, scopeKind: "event", scopeId: eventId });
      eventFirstTime(s, {
        psp,
        eventKind: "war",
        eventId,
        joinDay: day,
        col: "new_fans",
        fanKey: key,
        scopeKind: "event",
        scopeId: eventId,
        alsoAbsent: { scopeKind: "creator", scopeId: psp },
      });
      // First-seen through the event ⇒ the fan becomes a creator-scope member,
      // so a second event does not call them new again. `source` is NULL: this
      // path writes no `creator_cohort_daily.size`, and `analytics_membership.source`
      // is nullable precisely so a membership that was not counted into a cohort
      // can say so. Giving it a source would put `d<w>` on a cohort row whose
      // `size` this fan is not in; incrementing `size` here instead would break
      // `Σ_source size = new_supporters` (ADR §0 P1's partition), since no
      // `new_supporters` counter fires for a war join.
      // THE EVENT ROW IS WRITTEN FIRST, and the order is the whole mechanism
      // (S-14). `acquired` asks "did the creator have this fan before this
      // event", and the creator-scope row below is this same message's answer
      // to that question — so the question has to be put while the answer is
      // still absent. The predicate is character-for-character the one
      // `new_fans` fired on, so `acquired_d30 <= new_fans` holds by
      // construction rather than by arithmetic.
      membership(s, {
        psp,
        scopeKind: "event",
        scopeId: eventId,
        fanKey: key,
        fanKeyV,
        cellDay: day,
        source: null,
        acquiredExpr: { sql: NOT_MEMBER, params: [psp, "creator", psp, key] },
      });
      membership(s, { psp, scopeKind: "creator", scopeId: psp, fanKey: key, fanKeyV, cellDay: day, source: null });
      return true;
    }
    case "redemption_attempt": {
      if (!needsFan() || !needsScope()) return false;
      // 'live' is accounted for by the `collectible_acquired` that credentials.ts
      // emits separately in the same batch; counting it here too would double it.
      if (dims.outcome !== "dead") return true;
      const key = fanKey as string;
      const releaseId = scopeId as string;
      // DISTINCT accounts whose attempt died (S-1's integrity block), so it is
      // membership-derived and fires once per fan per release.
      eventFirstTime(s, {
        psp,
        eventKind: "release",
        eventId: releaseId,
        joinDay: day,
        col: "dead_attempts",
        fanKey: key,
        scopeKind: "release_attempt",
        scopeId: releaseId,
      });
      membership(s, { psp, scopeKind: "release_attempt", scopeId: releaseId, fanKey: key, fanKeyV, cellDay: day, source: null });
      return true;
    }

    case "card_collected":
      metric(s, psp, day, "cards_collected");
      return true;
    case "card_uncollected":
      metric(s, psp, day, "cards_uncollected");
      return true;
    case "achievement_earned":
      metric(s, psp, day, "achievements");
      return true;

    // ---- creator-scoped ----------------------------------------------------
    case "gift_sent":
      metric(s, psp, day, "gifts_sent");
      return true;
    case "war_completed":
      if (dims.mode === "practice") return true; // D-0840(2), as war_joined
      if (dims.mode !== undefined && dims.mode !== "ranked") return false;
      metric(s, psp, day, "wars_completed");
      return true;
    case "question_answered":
      metric(s, psp, day, "answers");
      return true;
    case "handshake_answered": {
      const d = dims.disposition;
      if (d === "confirmed") metric(s, psp, day, "handshakes_authenticated");
      else if (d === "declined") metric(s, psp, day, "handshakes_denied");
      else if (typeof d === "string") return SKEW;
      else return false;
      return true;
    }
    case "supporter_erased":
      // ADR §3: the stock's adjustment is itself a day counter, on the arrival
      // day, so `supporters(T) = Σ(new_supporters − unfollows)` leaves the
      // erased fan at the next UTC boundary and no served value moves.
      metric(s, psp, day, "unfollows");
      return true;
    case "card_erased":
      // The erasure decrement for `card_collections`, on the SAME column the
      // fan's own removal writes: a collection that left is a collection that
      // left, and the door it used is not a fact the creator's surface owes
      // anybody. Same forward-counting direction as `supporter_erased`.
      metric(s, psp, day, "cards_uncollected");
      return true;
    case "achievement_erased":
      // Its OWN column, NOT `achievements − 1`. A day counter is never revised
      // downwards once its day sealed (ADR §3), so the departure is counted
      // forward on its arrival day and the stock is Σ(achievements −
      // achievements_erased) over sealed days.
      metric(s, psp, day, "achievements_erased");
      return true;
    case "impersonation_report_confirmed":
      metric(s, psp, day, "warnings_published");
      return true;
    case "market_sale": {
      if (msg.amount_cents === null) return false;
      // S-12. THE ROW'S THREE MONEY FIGURES ARE ONE JOURNAL ROW'S FIGURES.
      // `amount_cents` is the GROSS the fan paid (`sales.gross_backing_cents`);
      // the split of it rides as dimensions, because INV-051 refuses a SECOND
      // amount column but allows an amount "by reference to a journal row", and
      // net and cut are that row's own numbers. Booking gross without the split
      // would serve a `gross_credits` that no `creator_net_cents +
      // platform_cut_cents` adds up to, so a message missing either is refused
      // whole (the day is marked incomplete by the caller) rather than written
      // half.
      const net = typeof dims.net_cents === "number" ? dims.net_cents : null;
      const cut = typeof dims.platform_cut_cents === "number" ? dims.platform_cut_cents : null;
      // I3-2031: a CALENDAR day, the same test `occurred_day` gets above.
      const clears = typeof dims.clears_day === "string" && isCalendarDay(dims.clears_day) ? dims.clears_day : null;
      // `clears_day` IS A DECLARED DIM OF THIS KIND (`analytics-events.ts:126`)
      // AND IS REFUSED WITH THE OTHER TWO (FINDING-0010-A-8). It was optional
      // here, so a message that dropped it wrote gross, net, cut and `sales`
      // and silently did not write `scheduled_clear_cents` — INV-033's
      // scheduled series short by that sale, with no flag and no way to tell
      // the loss from a day on which nothing was scheduled. All three
      // producers send it (`credits.ts:303`, `packs.ts:576`, `tips.ts:160`),
      // so a message without it is corrupt, and a corrupt money message is
      // refused WHOLE rather than written in part — the same rule the net/cut
      // refusal above already states.
      if (net === null || cut === null || clears === null) return false;
      revenue(s, psp, day, "gross_credits", msg.amount_cents);
      revenue(s, psp, day, "creator_net_cents", net);
      revenue(s, psp, day, "platform_cut_cents", cut);
      revenue(s, psp, day, "sales", 1);
      {
        // THE SCHEDULED SERIES IS THE CREATOR'S OWN MONEY, so it carries the
        // NET, not the gross: `scheduled_clear_cents` is "what is scheduled to
        // reach you on that day", and the platform's cut never was.
        // Normally unsealed by construction (a sale clears in the future). A
        // clear day that has already sealed takes the same arrival-day rule as
        // any other sealed counter target.
        const target = clears >= today ? clears : today;
        if (target !== clears) incomplete(s, today);
        revenue(s, psp, target, "scheduled_clear_cents", net);
      }
      return true;
    }
    case "sale_reversed": {
      if (msg.amount_cents === null) return false;
      // TWO BRANCHES, SPLIT COLUMNS (ADR §3/O-11). The gross inverse always
      // lands on the arrival day; the scheduled inverse decrements the
      // `clears_day` cell while that cell is still open, and otherwise becomes
      // its own arrival-day column — so a sealed day is never rewritten and the
      // two facts stay distinguishable at read.
      //
      // S-12: EACH INVERSE IS THE INVERSE OF THE SERIES IT REVERSES.
      // `reversals_gross_cents` is against `gross_credits`, so it takes the
      // GROSS the row carries; the scheduled inverse is against
      // `scheduled_clear_cents`, which the sale wrote as the NET, so it takes
      // `net_cents`. Using one number for both would leave the scheduled series
      // short by the platform's cut on every reversal.
      const netBack = typeof dims.net_cents === "number" ? dims.net_cents : null;
      if (netBack === null) return false;
      revenue(s, psp, day, "reversals_gross_cents", msg.amount_cents);
      const clears = typeof dims.clears_day === "string" ? dims.clears_day : null;
      // I3-2031: a PRESENT clears_day that is not a calendar day is corrupt, and a
      // corrupt money message is refused whole (as `market_sale` above).
      if (clears !== null && !isCalendarDay(clears)) return false;
      if (clears) {
        if (clears >= today) revenue(s, psp, clears, "scheduled_clear_cents", -netBack);
        else {
          revenue(s, psp, today, "scheduled_clear_reversals_cents", netBack);
          incomplete(s, today);
        }
      }
      return true;
    }
    case "creator_earning":
      if (msg.amount_cents === null) return false;
      revenue(s, psp, day, "paid_cents", msg.amount_cents);
      return true;
    case "pool_code_claimed":
      if (!needsScope()) return false;
      pool(s, psp, scopeId as string, day, "claimed");
      return true;
    case "pool_code_redeemed":
      if (!needsScope()) return false;
      pool(s, psp, scopeId as string, day, "redeemed");
      return true;
    default:
      // AN UNKNOWN KIND — and, since S-13, ONLY an unknown kind. Every member of
      // `analytics-events.ts`'s closed vocabulary is either handled above or
      // named in `NO_CELL`; `test/s002-consumer.test.ts` enumerates
      // `EVENT_KINDS` and fails if one falls through to here. That guard is what
      // makes the flag below MEAN something: a kind the platform emits on
      // purpose and this consumer deliberately does not write is NOT an
      // incomplete day (B10 OQ3 — `card_erased` reaching this branch was
      // marking every erasure day incomplete platform-wide).
      //
      // The vocabulary is closed at the producer, so a kind this consumer does
      // not know is a deploy-order skew. I3-0719 / I3-2011: it is RETRIED, not
      // acked — acking wrote the `processed_events` row that made a later
      // deploy's replay a duplicate, and left no replayable record. After
      // `max_retries` the DLQ leg writes the dead row and flags the day.
      if (RETIRED_KINDS.has(kind)) {
        incomplete(s, today);
        return true;
      }
      return SKEW;
  }
}

/** Was this throw the idempotency ledger's primary key? That is the ONE error
 * that means "already processed" and is acked; everything else is retried. */
function isAlreadyProcessed(err: unknown): boolean {
  const msg = err instanceof Error ? err.message : String(err);
  return /constraint failed[^\n]*processed_events/i.test(msg);
}

/**
 * Consume one queue batch, one message at a time.
 *
 * Per message, ONE atomic `ANALYTICS_DB.batch`. Never a batch spanning two
 * messages: a single bad message would then take its batch-mates down with it,
 * and the point of explicit per-message ack/retry (A6) is that it cannot.
 */
export async function consumeBatch(analytics: DbLike, messages: readonly OutboxMessage[], opts: ConsumeOptions): Promise<ConsumeResult> {
  const ack: string[] = [];
  const retry: string[] = [];

  for (const msg of messages) {
    const nowIso = opts.now();
    const today = nowIso.slice(0, 10);
    const s = new Stmts(analytics);
    // FIRST: the idempotency ledger. Its PK is what makes a redelivery apply
    // nothing at all rather than apply twice.
    s.add("INSERT INTO processed_events (event_key, processed_at) VALUES (?, ?)", msg.event_key, nowIso);

    // The kind's upserts are planned into a SEPARATE collector and merged only
    // on success: a plan that throws half-built would otherwise contribute its
    // first statements to a batch the rest of the plan never finished.
    const planned = new Stmts(analytics);
    let usable: Plan;
    try {
      usable = planUpserts(planned, msg, today);
    } catch {
      // A statement this consumer could not even BUILD (an unknown column, a
      // malformed day) will not build on redelivery either.
      usable = false;
    }
    if (usable === SKEW) {
      // No batch at all: the ledger row is exactly what must not be written.
      retry.push(msg.event_key);
      continue;
    }
    if (usable) s.out.push(...planned.out);
    else incomplete(s, today);

    // NO WATERMARK WRITE (docket #17 A-2). This batch's `processed_events` row
    // IS this consumer's whole report: the watermark is recomputed from that
    // ledger by `recomputeWatermark` (`analytics-outbox.ts`), so a message that
    // arrives above the gap is recorded as applied at once and is counted the
    // moment the gap closes. The `? <= seq + 1` UPDATE that used to sit here
    // made the value a function of ARRIVAL ORDER: it could never revisit a seq
    // that arrived early, and a redelivery could not fix it either, because the
    // PK below aborts the whole batch before any UPDATE in it runs.

    try {
      await analytics.batch(s.out);
      ack.push(msg.event_key);
    } catch (err) {
      if (isAlreadyProcessed(err)) ack.push(msg.event_key);
      else retry.push(msg.event_key);
    }
  }
  return { ack, retry };
}

/**
 * THE DEAD-LETTER BRANCH (ADR §2, pass-3 BREAK 4).
 *
 * A message the queue gave up on is not lost: its row is written to
 * `analytics_outbox_dead` IN PRODUCTION with `reason = 'dlq'` — the dead row is
 * then that event's record, so the outbox prune below the watermark is safe and
 * the republish repair does not chase a row already accounted for — and the
 * arrival day is marked platform-incomplete.
 *
 * THE WATERMARK IS NOT WRITTEN HERE (docket #17 A-2). It is recomputed from the
 * ledger and this table by `recomputeWatermark` (`analytics-outbox.ts`), which
 * crosses a seq only once its dead row EXISTS — so "the record is written
 * before the watermark that declares it recorded" is now a property of the
 * recompute's own input rather than an ordering this function has to keep.
 * Both writes are idempotent (`INSERT OR IGNORE`, and a pure recompute), so a
 * retry repeats safely.
 *
 * A FAILING PRODUCTION WRITE IS ACKED, NOT RETRIED (FINDING-0010-A-9). This
 * consumer runs on `dhwaja-analytics-<env>-dlq`, which `wrangler.jsonc`
 * declares `"max_retries": 0` with NO `dead_letter_queue` of its own at all
 * three levels — so `retry()` here is not a redelivery, it is a discard, and
 * discarding was the one disposition that left no record anywhere. The ack
 * plus the arrival-day flag is a record; and the WATERMARK IS THE PART THAT
 * MUST NOT MOVE, because the live `analytics_outbox` row is what remains of
 * this event: `pruneOutbox` deletes only BELOW the watermark, so holding it
 * back is exactly what keeps that row on disk for the republish rule to reach.
 * An unrecorded message cannot move it: `recomputeWatermark` crosses a seq only
 * on a dead row or a ledger row, and this branch wrote neither.
 *
 * THE INCOMPLETE-DAY FLAG WRITE IS ALSO ACKED, NOT RETRIED (AC-09/E-43). It
 * used to keep the retry a transient D1 error deserves, reasoning that this
 * leg (unlike the production write) could still be fixed by a redelivery —
 * but a redelivery is not what `max_retries: 0` gives this queue. `[EXEC-AC]`
 * confirmed a `retry()` return here is dropped after one attempt, same as a
 * throw. `DLQ_INCOMPLETE_UNRECORDED_MARKER` is the resulting record.
 */
export async function consumeDead(analytics: DbLike, prod: DbLike, messages: readonly OutboxMessage[], opts: ConsumeOptions): Promise<ConsumeResult> {
  const ack: string[] = [];
  const retry: string[] = [];

  for (const msg of messages) {
    const nowIso = opts.now();
    const today = nowIso.slice(0, 10);
    try {
      await prod.batch([
        prod
          .prepare(
            "INSERT OR IGNORE INTO analytics_outbox_dead " +
              "(seq, event_key, passport_id, event_type, occurred_day, fan_key, fan_key_v, scope_kind, scope_id, dims, amount_cents, attempts, published_at, dead_at, reason) " +
              // I3-1713: `published_at` is the live outbox row's own stamp (the
              // publisher's send-then-mark), matched on seq AND event_key; NULL
              // only when that row is gone. `attempts` is D-0731's DLQ count.
              "VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, (SELECT published_at FROM analytics_outbox WHERE seq = ? AND event_key = ?), ?, 'dlq')",
          )
          .bind(
            msg.seq,
            msg.event_key,
            msg.passport_id,
            msg.event_type,
            msg.occurred_day,
            msg.fan_key,
            msg.fan_key_v,
            msg.scope_kind,
            msg.scope_id,
            msg.dims,
            msg.amount_cents,
            msg.attempts ?? 0,
            msg.seq,
            msg.event_key,
            nowIso,
          ),
      ]);
    } catch (e) {
      // I3-1807: through log.ts's allowlist; `event_key` is not on it and is not logged.
      logError({ cid: "analytics-dlq", hop: "queue", msg: DLQ_UNRECORDED_MARKER, marker: DLQ_UNRECORDED_MARKER, seq: msg.seq, day: today, err: e instanceof Error ? e.name : typeof e });
    }
    try {
      const s = new Stmts(analytics);
      incomplete(s, today);
      await analytics.batch(s.out);
    } catch (e) {
      // AC-09/E-43 (this consumer runs on `dhwaja-analytics-<env>-dlq`, which
      // `wrangler.jsonc` declares `max_retries: 0` with NO dead-letter queue of
      // its own at any of the three levels): a `retry` returned from THIS
      // function is not a redelivery, it is `index.ts`'s `m.retry()` handing
      // the runtime a message it will drop after this one attempt. That is
      // strictly worse than acking — an ack plus a missed incomplete-day flag
      // still leaves the production dead row (written above) as the event's
      // record; a retry leaves the same missing flag AND destroys the message,
      // with only this log line anywhere. So this branch never retries: the
      // production write already happened (or was logged as unrecorded above),
      // and the flag write's failure is logged, not resurrected by a redelivery
      // this queue cannot honor.
      logError({ cid: "analytics-dlq", hop: "queue", msg: DLQ_INCOMPLETE_UNRECORDED_MARKER, marker: DLQ_INCOMPLETE_UNRECORDED_MARKER, seq: msg.seq, day: today, err: e instanceof Error ? e.name : typeof e });
    }
    ack.push(msg.event_key);
  }
  return { ack, retry };
}

/**
 * ERASURE ACROSS TWO DATABASES (ADR-0006 §5).
 *
 * `deleteAccount` cannot reach the analytics store: two D1 bindings never share
 * a batch, and an erasure that spanned them would be two half-commits. So the
 * batch that deletes the account writes one durable work order —
 * `analytics_erasures` — and this drain does the cross-database half afterwards,
 * on the analytics cron, under its own lease.
 *
 * ---------------------------------------------------------------------------
 * WHY NOT A QUEUE EVENT
 * ---------------------------------------------------------------------------
 * Sending the erasure through `analytics_outbox` would put the undo on the same
 * at-least-once path as everything it has to undo: behind the same backlog,
 * subject to the same dead-letter table, with a person's departure sitting in a
 * queue. The work order is a ROW that commits WITH the deletion, and every step
 * below is replay-safe, so a Worker death costs a retry and never a missed
 * erasure.
 *
 * ---------------------------------------------------------------------------
 * WHAT REPLAY-SAFE MEANS HERE, STEP BY STEP
 * ---------------------------------------------------------------------------
 * A crash may land between any two statements, and the tick that follows starts
 * this row again from (a). Each step therefore has to be a no-op the second
 * time for the RIGHT reason, not by accident:
 *
 *   (a) reading the scopes writes nothing.
 *   (b) the per-scope batch is atomic and, after it, `analytics_membership` has
 *       no row for that `(passport_id, fan_key)` — so a re-run's
 *       `INSERT INTO analytics_tombstones … SELECT … FROM analytics_membership`
 *       selects nothing and writes NO second tombstone. That is what makes
 *       "exactly one tombstone per membership row" hold across a crash: the
 *       tombstone is derived from the rows it destroys, in the same
 *       transaction, rather than counted and then written.
 *   (c) the re-read applies a scope the first pass missed; a scope it already
 *       did is idempotent by (b) and by `INSERT OR IGNORE` on the sentinel.
 *   (d) deleting the scopes and the ledger row is the last thing, so a crash
 *       before it costs a repeat of work that is already idempotent, and a
 *       crash after it has nothing left to do.
 *
 * ---------------------------------------------------------------------------
 * WHY THE SCOPES ARE READ TWICE
 * ---------------------------------------------------------------------------
 * A producer's `WHERE EXISTS (SELECT 1 FROM accounts …)` guard evaluated a
 * moment before `deleteAccount` committed still writes its `analytics_fan_scopes`
 * row (A4). D1 serialises write transactions and `DELETE FROM accounts` is the
 * LAST statement of the batch that wrote the ledger row, so such a row is
 * already visible to the first read — the second read is provably a no-op
 * (pass-3 V3e) and is done anyway, because "provably" rests on a serialisation
 * property of a database this code does not implement, and the cost of being
 * wrong is a fan left inside a creator's analytics forever.
 *
 * The scopes are NOT deleted inside `deleteAccount`'s batch for that same race:
 * a delete there would miss the row and the drain would then have nothing to
 * enumerate.
 *
 * ---------------------------------------------------------------------------
 * NO KEYS ARE STORED ANYWHERE; THIS MODULE COMPUTES THEM
 * ---------------------------------------------------------------------------
 * `analytics_erasures` carries the account and the day, `analytics_fan_scopes`
 * the (account, creator) pairs. Neither holds a `fan_key`, because a stored key
 * beside an account id is the translation table the per-creator pseudonym
 * exists to prevent — written down, durably, at the moment somebody asked to be
 * forgotten (A3, O-5). The keys are recomputed here from
 * `ANALYTICS_FAN_SECRET`, held in memory for one scope, and never written.
 *
 * ---------------------------------------------------------------------------
 * THE READER RULE (migration analytics/0001, restating 0075)
 * ---------------------------------------------------------------------------
 * This module holds ONE of the two statements permitted to read
 * `analytics_membership`, and it is the tombstone-then-delete pair: one
 * `passport_id`, one `fan_key`, no projection of a row to any caller. Nothing
 * here returns a membership row, a fan key or a count of either.
 */

import { FAN_KEY_V, fanKey } from "./analytics-events.js";
import type { DbLike, DbStatement } from "./db.js";
import { logError } from "./log.js";
import { TOMBSTONE_COMMIT_DAY } from "./tombstone-day.js";

interface LedgerRow {
  erasure_id: string;
  account_id: string;
  requested_day: string;
}

/**
 * The tombstones, the sentinel and the delete for ONE (account, creator), for
 * every live key version — one atomic batch per version.
 *
 * The sentinel carries `fan_key_v` (D-4): the version of ANALYTICS_FAN_SECRET
 * the key beside it was minted under.
 *
 * TWO SENTINEL ROWS, ONE FACT (docket #23 C-2, S4-1). `analytics_erased` is
 * named in `CREATOR_SCOPED_DELETES` below, so a creator's retraction deletes it;
 * `analytics_erasure_sentinels` (analytics/0003) is named in no creator-scoped
 * statement, so it does not. Both are written HERE, in this batch, because the
 * consumer's guard (`analytics-consumer.ts NOT_ERASED`) requires BOTH to be
 * absent before a fan-keyed upsert may write — a batch that wrote one and not
 * the other would be a half-erasure, and the batch is atomic so it cannot be.
 *
 * ORDER IS THE CONTROL. The sentinels go FIRST: an event for this fan that is
 * already in the queue when this runs will be consumed AFTER them, and every
 * fan-keyed upsert carries `AND NOT EXISTS (… analytics_erased …) AND NOT
 * EXISTS (… analytics_erasure_sentinels …)` — so between the sentinels and the
 * delete a late event writes nothing rather than re-creating the membership row
 * this batch is about to tombstone.
 */
/**
 * ONE LIVE VERSION OF `ANALYTICS_FAN_SECRET`: the secret and the `fan_key_v` its
 * keys were minted under. I3-0256: erasure takes the whole LIVE SET, because a
 * rotation leaves one fan's rows under two keys and an erasure that computed
 * only the current key left the older rows unfindable once its scopes row was
 * deleted. A bare string is the single-version case, `FAN_KEY_V`.
 */
export interface FanSecretVersion {
  readonly v: number;
  readonly secret: string;
}

/** The live set, normalised, refusing the two inputs that would make erasure
 * silently find nothing: an empty set, and two entries claiming one version
 * (whose rows would then be tombstoned under whichever key came first). */
export function liveFanSecrets(secret: string | readonly FanSecretVersion[]): readonly FanSecretVersion[] {
  const set = typeof secret === "string" ? [{ v: FAN_KEY_V, secret }] : secret;
  if (set.length === 0) throw new Error("drainErasures: no live ANALYTICS_FAN_SECRET version");
  const seen = new Set<number>();
  for (const s of set) {
    if (!Number.isInteger(s.v) || s.v < 1) throw new Error(`drainErasures: fan_key_v must be a positive integer, got ${String(s.v)}`);
    if (seen.has(s.v)) throw new Error(`drainErasures: fan_key_v ${s.v} listed twice`);
    seen.add(s.v);
  }
  if (!seen.has(FAN_KEY_V)) throw new Error(`drainErasures: the live set omits the CURRENT version ${FAN_KEY_V}`);
  return set;
}

async function eraseScope(analytics: DbLike, secrets: readonly FanSecretVersion[], accountId: string, passportId: string): Promise<void> {
  for (const { v, secret } of secrets) await eraseScopeVersion(analytics, v, secret, accountId, passportId);
}

/** RG-PRIVACY:3 (I16-FXC, D-0926(2)): the erasure day is the day of the batch's FIRST statement's own clock
 * (TOMBSTONE_COMMIT_DAY, SQLite `now`, fixed for one statement), stamped into `analytics_erased`; the sentinel and every
 * tombstone COPY that row's day, so the three cannot straddle a midnight that falls between two statements of the batch.
 * It was a JS clock read once per ledger row before any awaited scope batch, so a drain spanning midnight stamped
 * yesterday on scopes it erased today. */
async function eraseScopeVersion(analytics: DbLike, keyV: number, secret: string, accountId: string, passportId: string): Promise<void> {
  // ONE KEY PER LIVE VERSION (ADR §5: a sentinel for EVERY live version).
  // `fanKey` is an HMAC over the secret it is handed, so the key for version
  // `keyV` is `fanKey(<that version's secret>, …)`; `eraseScope` calls this once
  // per entry of the live set, and `liveFanSecrets` refuses a set that lists a
  // version twice, so two versions can never collapse onto one computed key.
  const key = await fanKey(secret, accountId, passportId);
  const stmts: DbStatement[] = [
    // `fan_key_v` is provenance, not part of the match: the consumer's guard
    // (`analytics-consumer.ts:258-260`) still matches on `(passport_id, fan_key)`
    // alone, in each of its two tables, and the PK is still those two columns
    // (analytics/0002_fan_key_v.sql; analytics/0003_erasure_sentinels.sql).
    // RG-PRIVACY:3: the sentinel is written FIRST and takes the commit statement's day; `analytics_erased` copies it,
    // and the tombstones below copy `analytics_erased`'s, so the three rows carry ONE day whatever the clock does
    // between the batch's statements.
    analytics
      .prepare(`INSERT OR IGNORE INTO analytics_erasure_sentinels (passport_id, fan_key, fan_key_v, erased_day) VALUES (?, ?, ?, ${TOMBSTONE_COMMIT_DAY})`)
      .bind(passportId, key, keyV),
    // The SAME fact in the table no creator-scoped statement names (S4-1). Same
    // key, same day, same version, so the two rows cannot drift; the column
    // ORDER is analytics/0003's, which is not analytics_erased's.
    analytics
      .prepare("INSERT OR IGNORE INTO analytics_erased (passport_id, fan_key, erased_day, fan_key_v) SELECT passport_id, fan_key, erased_day, ? FROM analytics_erasure_sentinels WHERE passport_id = ? AND fan_key = ?")
      .bind(keyV, passportId, key),
    // ONE TOMBSTONE PER MEMBERSHIP ROW, derived from the rows themselves in the
    // transaction that destroys them. `cell_day` is copied across unchanged
    // (pass-3 BREAK 5): the served denominator subtracts on the day key of the
    // CELL THE COUNTER INCREMENTED, so a tombstone carrying any other day would
    // subtract from a cell the fan was never counted in and leave the one they
    // were.
    analytics
      .prepare(
        // I3-1122: each tombstone carries WHICH nested cells its fan was counted
        // in (analytics/0007), so the reader subtracts it only from those —
        // creator scope: d_w iff r_w; event scope: new_fans iff acquired,
        // acquired_d30 iff acquired AND the creator row's r30 flipped
        // (analytics-consumer.ts, the S-14 acquired_d30 UPDATE).
        `INSERT INTO analytics_tombstones (passport_id, scope_kind, scope_id, cell_day, source, erased_day, in_r1, in_r7, in_r30, in_new, in_acq30)
         SELECT m.passport_id, m.scope_kind, m.scope_id, m.cell_day, m.source, (SELECT d.erased_day FROM analytics_erased d WHERE d.passport_id = m.passport_id AND d.fan_key = m.fan_key), m.r1, m.r7, m.r30, m.acquired,
                CASE WHEN m.scope_kind = 'event' AND m.acquired = 1 AND EXISTS (SELECT 1 FROM analytics_membership c WHERE c.passport_id = m.passport_id
                  AND c.scope_kind = 'creator' AND c.fan_key = m.fan_key AND c.r30 = 1) THEN 1 ELSE 0 END
         FROM analytics_membership m WHERE m.passport_id = ? AND m.fan_key = ?`,
      )
      .bind(passportId, key),
    analytics.prepare("DELETE FROM analytics_membership WHERE passport_id = ? AND fan_key = ?").bind(passportId, key),
    // I3-0356 (D-0984): the per-day follow row carries the same pseudonym (analytics/0009).
    analytics.prepare("DELETE FROM analytics_follow_net WHERE passport_id = ? AND fan_key = ?").bind(passportId, key),
  ];
  await analytics.batch(stmts);
}

/** I3-0218: `limit` bounds ONE read so a single fan with many creator scopes
 * cannot make one tick do unbounded work (see SCOPE_TICK_LIMIT below).
 * `ORDER BY passport_id` makes successive bounded reads make PROGRESS rather
 * than fetch the same head of the list forever: a scope is removed from this
 * table (by the caller, right after it is erased — see `drainErasures`) as
 * soon as it is done, so a later call with the same LIMIT necessarily returns
 * different, still-pending rows. */
async function readScopes(prod: DbLike, accountId: string, limit: number): Promise<string[]> {
  const rows = await prod
    .prepare("SELECT passport_id FROM analytics_fan_scopes WHERE account_id = ? ORDER BY passport_id LIMIT ?")
    .bind(accountId, limit)
    .all<{ passport_id: string }>();
  return rows.results.map((r) => r.passport_id);
}

/**
 * Drain up to `limit` erasure work orders, oldest first.
 *
 * `limit` is per tick and small (ADR §8: 20 accounts) because the work is
 * unbounded in the number of creators one fan touched, and the tick's
 * subrequest budget is not.
 *
 * THE ERASURE DAY IS THE DAY THIS DRAIN RAN, not the day the fan asked (docket
 * #17 C-1(e); INV-005 WRITER SET). `erased_day` is read at
 * `analytics-read.ts:279` as `erased_day < T`, so a tombstone stamped with the
 * REQUEST's day subtracts from every object for a day after it — including days
 * whose objects were already built and served while the work order sat in the
 * ledger. The drain's own day is the only stamp that lands in an object still
 * open: the fan is counted through the whole UTC day the erasure was APPLIED and
 * drops at the next boundary, so no served number moves inside a day even though
 * this runs every five minutes. `row.requested_day` is therefore no longer read
 * for the day at all. RG-PRIVACY:3 (D-0926(2)): the day is the scope batch's own
 * commit-statement day (`eraseScopeVersion`), SQLite's `strftime('%Y-%m-%d','now')`,
 * which is a well-formed day by construction (the old `DAY_RE` guard on a JS clock
 * value is gone with the JS clock).
 */
/** I3-0218: erasure drain bounds accounts per tick (ADR §8: 20) but, until
 * now, not scopes per account — a fan with many creator scopes made this
 * loop's inner `for` unbounded, so ONE such fan could exceed the tick's
 * subrequest budget and throw, aborting `drainErasures` before it ever
 * reached the other 19 (up to) accounts the outer LIMIT selected — and
 * because the ledger's `ORDER BY created_at` always resurfaces the SAME
 * oldest, still-undrained account first, every following tick hit the exact
 * same wall before touching anyone behind it in the queue. Bounding scopes
 * per account per tick fixes both: this account's own work is capped (no
 * throw), and the tick still reaches every other account queued behind it. */
const SCOPE_TICK_LIMIT = 200;

export async function drainErasures(prod: DbLike, analytics: DbLike, secret: string | readonly FanSecretVersion[], _now: () => string, limit = 20): Promise<{ drained: number }> {
  const secrets = liveFanSecrets(secret);
  const ledger = await prod
    .prepare("SELECT erasure_id, account_id, requested_day FROM analytics_erasures ORDER BY created_at LIMIT ?")
    .bind(limit)
    .all<LedgerRow>();

  let drained = 0;
  for (const row of ledger.results) {
    // RG-PRIVACY:3: no day is read here; each scope's batch stamps its own commit day (eraseScopeVersion). `_now` stays
    // in the signature for the callers (index.ts) and is not read.
    // Bounded (a)+(b), THEN bounded (c) re-read — same two-pass shape as
    // before (A4), just capped per pass at SCOPE_TICK_LIMIT. `eraseScope` is a
    // no-op the second time a scope reaches it (idempotent by construction,
    // see module docs), and the per-scope delete right after it is what lets
    // a LIMITed read make forward progress across calls within this tick and
    // across ticks (see `readScopes`).
    const eraseBatch = async (): Promise<void> => {
      for (const psp of await readScopes(prod, row.account_id, SCOPE_TICK_LIMIT)) {
        await eraseScope(analytics, secrets, row.account_id, psp);
        // Removes this ONE scope from the work remaining. Not atomic with the
        // analytics batch above (cross-database, same reason `eraseScope`
        // itself never spans both stores) — safe because a crash between the
        // two just means a later pass re-erases an already-erased scope,
        // which is a no-op, and re-deletes an already-gone row, also a no-op.
        await prod.prepare("DELETE FROM analytics_fan_scopes WHERE account_id = ? AND passport_id = ?").bind(row.account_id, psp).run();
      }
    };
    await eraseBatch();
    await eraseBatch();

    // Anything still queued past SCOPE_TICK_LIMIT * 2 for this account? Leave
    // its ledger row in place — do NOT delete it — and move on to the next
    // account in this tick's batch. The next tick's `ORDER BY created_at`
    // will pick this account up again and continue where the per-scope
    // deletes left off, instead of this account blocking everyone behind it.
    const stillPending = await readScopes(prod, row.account_id, 1);
    if (stillPending.length > 0) continue;

    // (d) The production ledgers hold the raw account id only until here (P2).
    // ONE batch: a scopes delete that committed without the ledger delete would
    // leave a work order whose enumeration is gone, and the next tick would
    // then "complete" it having erased nothing. (The per-scope deletes above
    // ordinarily leave nothing here to delete; this is the safety net for any
    // row a crash left behind.)
    await prod.batch([
      prod.prepare("DELETE FROM analytics_fan_scopes WHERE account_id = ?").bind(row.account_id),
      prod.prepare("DELETE FROM analytics_erasures WHERE erasure_id = ?").bind(row.erasure_id),
    ]);
    drained++;
  }

  return { drained };
}

/* ────────────────────────────────────────────────────────────────────────────
 * ERASED ACCOUNTS' EVIDENCE BYTES (QUEUE #564, D-0686, D-0848(3)).
 *
 * `deleteAccount`'s batch copies every report-evidence object key of the erased
 * account into `erasure_evidence_purges` (migration 0146) and deletes the rows;

/** What the drain needs of the evidence bucket (structurally an `EvidenceStore`). */
export interface ErasedObjectStore {
  delete(key: string): Promise<void>;
}

export const ERASED_EVIDENCE_BACKOFF_BASE_MS = 5 * 60 * 1000;
/**
 * #792 (D-0859): an upload's PRE-PUT INTENT (`erasure_id` = `upload:<evidence id>`,
 * written in `storeEvidence`'s pointer batch) is left alone until it is this old,
 * because until then its `putBytes` may still be in flight and deleting the key now
 * would let the put land after the delete. One hour is far beyond any request's
 * life; the intent is the record the drain acts on after it.
 */
export const UPLOAD_INTENT_GRACE_MS = 60 * 60 * 1000;
export const ERASED_EVIDENCE_BACKOFF_MAX_MS = 24 * 60 * 60 * 1000;

/** Delay before the next try after the `attempts`-th failure (1-based). */
export function erasedEvidenceBackoffMs(attempts: number): number {
  return Math.min(ERASED_EVIDENCE_BACKOFF_MAX_MS, ERASED_EVIDENCE_BACKOFF_BASE_MS * 2 ** Math.max(0, attempts - 1));
}

export async function drainErasedEvidence(
  prod: DbLike,

  now: () => string,
  limit = 50,

    const waiting = row?.n ?? 0;

    return { deleted: 0, failed: 0, waiting };
  }
  const nowIso = now();
  const nowMs = Date.parse(nowIso);
  // #792: a young upload intent is not selected at all (it cannot fill the LIMIT
  // window); an old one whose pointer row still exists is a finished upload whose
  // intent delete was lost, so the row goes and the object stays (`live`).

/**
 * ANALYTICS_FAN_SECRET is missing or too short to sign anything (ADR-0006 §2;
 * A-10 / QUEUE #189). Distinct from a plain `EventError`: every OTHER
 * `EventError` describes a malformed EVENT — the caller's fault, a 400. This
 * one describes the SERVER's own configuration — nothing the caller sent was
 * wrong, and treating it as a 400 (the fate of every `EventError` a route
 * cannot otherwise name — see `usableFanSecret` below) tells fans and
 * operators alike that a launch-critical outage is a client mistake. A route
 * that wants to answer 503 for exactly this condition can now write
 * `instanceof FanSecretUnavailable` — `EventError` alone can't be pattern-
 * matched to a status the way `AccountError`/`SocialError` carry their own
 * `.status`, because most `EventError`s truly are 400s and always will be.

export class FanSecretUnavailable extends EventError {
  override name = "FanSecretUnavailable";

/**
 * The rule for every string that becomes a column value: opaque, bounded, not
 * an identifier of a person. Whitespace is the tell for free text (a title, a
 * note, a display name); `@` for an email; `acc_` for the account id the whole
 * design exists to keep out of the second store; a colon-separated clock for a
 * sub-day timestamp.
 */
function assertOpaque(what: string, v: unknown): asserts v is string {
  if (typeof v !== "string" || v.length === 0) throw new EventError(`${what} must be a non-empty string`);
  if (v.length > MAX_VALUE_CHARS) throw new EventError(`${what} is longer than ${MAX_VALUE_CHARS} characters — free text does not enter an analytics row (INV-051)`);
  if (/\s/.test(v)) throw new EventError(`${what} contains whitespace — free text does not enter an analytics row (INV-051)`);
  if (v.includes("@")) throw new EventError(`${what} looks like an email address (INV-051)`);
  if (ACCOUNT_ID_RE.test(v)) throw new EventError(`${what} is an account id — no analytics row is derived from one (ADR-0006 §0 P2)`);
  if (SUB_DAY_RE.test(v)) throw new EventError(`${what} carries a sub-day timestamp (INV-051)`);
}

/**
 * The full contract, run before any statement is built. Throws — it never
 * returns a boolean, because a caller that forgets to check a boolean emits the
 * row anyway.
 */
export function assertEvent(e: unknown): asserts e is AnalyticsEvent {
  if (typeof e !== "object" || e === null) throw new EventError("event must be an object");
  const ev = e as Record<string, unknown>;

  for (const key of Object.keys(ev)) {
    if (EVENT_FIELDS.has(key)) continue;
    // The named case first: the field somebody adds is a clock.
    if (/_at$|_ts$|^ts$|time|stamp/i.test(key)) throw new EventError(`unknown field '${key}': no instant below the UTC day crosses the queue (INV-051)`);
    throw new EventError(`unknown field '${key}': the analytics event shape is closed (INV-051)`);
  }

  const spec = (EVENT_KINDS as Record<string, KindSpec | undefined>)[ev.kind as string];
  if (!spec) throw new EventError(`unknown event kind '${String(ev.kind)}' — the vocabulary is closed (INV-051)`);

  if (typeof ev.passportId !== "string" || !PSP_RE.test(ev.passportId)) throw new EventError("passportId must be a psp_ id");
  if (typeof ev.occurredDay !== "string" || !DAY_RE.test(ev.occurredDay)) throw new EventError("occurredDay must be 'YYYY-MM-DD' — no sub-day timestamp crosses the queue (INV-051)");

  // INV-051 / ADR §0 P2, checked before the per-carrier branches so the message
  // is the invariant's own and not a consequence of one branch.
  if (ev.fanKey !== undefined && ev.amountCents !== undefined) {
    throw new EventError("an event carries a fan pseudonym OR an amount, never both (INV-051, migration 0076 CHECK)");
  }

  if (spec.carries === "fan_key") {
    if (typeof ev.fanKey !== "string" || !FAN_KEY_RE.test(ev.fanKey)) throw new EventError(`${String(ev.kind)} is fan-keyed: fanKey must be 64 hex from fanKey()`);
  } else if (ev.fanKey !== undefined) {
    throw new EventError(`${String(ev.kind)} is creator-scoped and carries no fanKey`);
  }

  if (spec.carries === "amount") {
    if (!Number.isSafeInteger(ev.amountCents)) throw new EventError(`${String(ev.kind)} must carry an integer amountCents`);
  } else if (ev.amountCents !== undefined) {
    throw new EventError(`${String(ev.kind)} carries no amount`);
  }

  if (spec.scopeKind === null) {
    if (ev.scopeId !== undefined) throw new EventError(`${String(ev.kind)} carries no scope`);
  } else {
    assertOpaque(`${String(ev.kind)} scopeId`, ev.scopeId);
  }

  if (ev.dims !== undefined) {
    if (typeof ev.dims !== "object" || ev.dims === null || Array.isArray(ev.dims)) throw new EventError("dims must be an object");
    for (const [k, v] of Object.entries(ev.dims as Record<string, unknown>)) {
      if (!(spec.dims as readonly string[]).includes(k)) throw new EventError(`dim '${k}' is not declared by ${String(ev.kind)} (INV-051)`);
      if (typeof v === "string") assertOpaque(`dim '${k}'`, v);
      else if (typeof v !== "number") throw new EventError(`dim '${k}' must be a string or a number`);
      if (!DIM_VALUES[k as DimKey](v as string | number)) throw new EventError(`dim '${k}' value '${String(v)}' is not in its closed enum (INV-051)`);
    }
  }
}

/** The UTC day a fact is attributed to: the first ten characters of the ISO

  if (!DAY_RE.test(day)) throw new EventError(`occurredDay: '${String(nowIso)}' does not start with a UTC date`);
  return day;
}

 * PER CREATOR is the whole point: two creators' rows for one human do not join,
 * so no set of analytics rows describes a person across the platform. The
 * concatenation is unambiguous because an account id is fixed-shape
 * (`acc_` + hex, `ids.ts`), so no two different pairs concatenate alike.
 *
 * Async, and there is deliberately NO sync variant: a producer awaits the key
 * BEFORE it builds its batch, so no statement is ever built from a promise.
 */
export async function fanKey(secret: string | undefined, accountId: string, passportId: string): Promise<FanKey> {
  if (!usableFanSecret(secret)) throw new FanSecretUnavailable(`fanKey: ANALYTICS_FAN_SECRET is refused: ${sharedSecretRefusal(secret) ?? "unusable"} (a usable secret is at least 16 characters, not blank, with no TODO in any case)`);
  if (typeof accountId !== "string" || accountId.length === 0) throw new EventError("fanKey: accountId required");
  if (!PSP_RE.test(passportId)) throw new EventError("fanKey: passportId must be a psp_ id");
  const enc = new TextEncoder();
  const key = await crypto.subtle.importKey("raw", enc.encode(secret) as unknown as ArrayBuffer, { name: "HMAC", hash: "SHA-256" }, false, ["sign"]);
  const mac = await crypto.subtle.sign("HMAC", key, enc.encode(`${accountId}${passportId}`) as unknown as ArrayBuffer);
  let out = "";
  for (const b of new Uint8Array(mac)) out += b.toString(16).padStart(2, "0");
  return out as FanKey;
}

/**
 * I60-CONFIG C2: why usableSharedSecret refuses `v` (its own four tests, in its order), or null when it accepts it.
 * The reason never carries anything derived from the value; fanKey above names it in its refusal.
 */
export function sharedSecretRefusal(v: unknown, min = 16): string | null {
  if (typeof v !== "string" || v.length === 0) return "missing";
  if (v.length < min) return `shorter than ${min} characters`;
  if (v.trim() === "") return "blank (whitespace only)";
  if (/TODO/i.test(v)) return "contains TODO (in any case)";
  return null;
}

/** `dims` as canonical JSON (keys sorted), or NULL. Canonical so two emitters
 * of the same dimensions produce the same bytes and a diff of the column is a
 * diff of the dimensions. */
function canonicalDims(dims: EventDims | undefined): string | null {
  if (!dims) return null;
  const keys = Object.keys(dims).sort();
  if (keys.length === 0) return null;
  const obj: Record<string, string | number> = {};
  for (const k of keys) obj[k] = dims[k as DimKey] as string | number;
  return JSON.stringify(obj);
}

const OUTBOX_COLUMNS = "event_key, passport_id, event_type, occurred_day, fan_key, fan_key_v, scope_kind, scope_id, dims, amount_cents, created_at";

/**
 * THE FAN'S PAUSE CONTROL, AS A PREDICATE ON THE ROW (I3-0157; Jason 2026-09-22).
 *
 * `codex_settings.logging` is the ONE control the product gives a fan
 * ("Pause logging"), written by `codex.ts setCodexLogging`. It used to be read
 * in exactly one producer — `codex.ts:145`, guarding `recordEncounter` — so it
 * stopped `fan_active` and nothing else: the other fourteen fan-keyed kinds
 * kept writing a pseudonym row while the fan believed logging was off.
 *
 * It is spelled HERE, once, for the same reason the `accounts` guard is: this
 * function is the only door a fan-keyed kind can leave by (`outboxSelectSql`
 * refuses `carries === "fan_key"` at :563, and the one-emitter scan in
 * `test/s002-events.test.ts` keeps the INSERT in this file alone), so a
 * predicate here gates the WHOLE fan-keyed population — including a kind added
 * tomorrow — which a per-kind enumeration would cover only vacuously.
 *
 * NOT-EXISTS, NOT EXISTS-AND-ONE. Logging is ON by default and the ABSENCE of a
 * settings row IS consent (`codex.ts:60`), so only an explicit `logging = 0`
 * blocks. `EXISTS (… AND logging = 1)` would silently refuse every fan who has
 * never touched the control, which is most of them.
 */
// #680 (I16-W04C): EXPORTED so the save door and the catch-up sweep (accounts.ts) set `saves.counted_at` under the
// SAME predicate that gates the fan_followed row — a count the outbox never emitted must not be taken back later.
export const LOGGING_NOT_PAUSED = "NOT EXISTS (SELECT 1 FROM codex_settings WHERE account_id = ? AND logging = 0)";

/** What the producer knows and the event does not carry. */
export interface OutboxContext {
  /** The fan whose key the row carries. REQUIRED for a fan-keyed kind and
   * refused for a creator-scoped one: it is bound only to the `EXISTS`
   * guards, never to a column. */
  readonly accountId?: string;
  /** Test seam only: the nonce is minted here otherwise. */
  readonly eventKey?: string;
  /**
   * ONE MORE PREDICATE ON THE ROW, ANDed onto BOTH guarded SELECTs.
   *
   * The `accounts` guard answers "may this fan's key be written at all"; it
   * cannot answer "did the fact this row describes actually land", and for a
   * producer whose authoritative write is itself conditional — the encounter
   * path's day claim (ADR-0006 §2), a capped `INSERT … WHERE (counts) < ?` —
   * those are different questions. Without this the batch would commit an event
   * for a fact that was refused in the same batch, which is INV-051's "it exists
   * iff its fact does" read backwards.
   *
   * EVERY KIND (U-1, INV-051), not only the fan-keyed ones: a creator-scoped
   * authoritative write can be conditional too — `creator-payout.ts` STEP 0's
   * retry UPDATE is (S3-11). Carrying one turns the creator-scoped emit into
   * `INSERT … SELECT … WHERE` (:400); without one it stays `INSERT … VALUES`.
   */
  readonly extraWhere?: { readonly sql: string; readonly params: readonly unknown[] };
}

/**
 * The statements that put one event in the outbox, to be APPENDED TO THE
 * AUTHORITATIVE BATCH (ADR §2) — never run on their own. The event exists iff
 * its fact does.
 *
 * Fan-keyed kinds get two statements and both are guarded by
 * `WHERE EXISTS (SELECT 1 FROM accounts WHERE account_id = ?)`, binding the id
 * of THE FAN WHOSE KEY THE ROW CARRIES: after `deleteAccount` commits, no
 * fan-keyed event for that account can be emitted at all, and the guard fails
 * the ROW rather than the batch (0076: analytics never vetoes the fact it
 * describes). The `analytics_fan_scopes` row is what tells the erasure drain
 * which creators this account has a pseudonym under.
 *
 * AND BY `LOGGING_NOT_PAUSED` (:429), which is the fan's own "Pause logging"
 * control asked of every fan-keyed kind rather than of one producer (I3-0157).
 * A paused fan writes NEITHER statement, for EVERY kind in the vocabulary; a
 * creator-scoped kind is not gated by it and never was, because it carries no
 * pseudonym for anyone to pause.
 */
export function outboxStatements(db: DbLike, ev: AnalyticsEvent, ctx: OutboxContext = {}): DbStatement[] {
  assertEvent(ev);
  const spec = EVENT_KINDS[ev.kind];
  const fanKeyed = spec.carries === "fan_key";
  if (fanKeyed && !ctx.accountId) throw new EventError(`${ev.kind} is fan-keyed: ctx.accountId is required for the post-erasure guard (ADR-0006 §2)`);
  if (!fanKeyed && ctx.accountId) throw new EventError(`${ev.kind} is creator-scoped: no account id takes part in its emission`);

  const key = ctx.eventKey ?? eventKey();
  const row = [
    key,
    ev.passportId,
    ev.kind,
    ev.occurredDay,
    ev.fanKey ?? null,
    ev.fanKey === undefined ? null : FAN_KEY_V,
    spec.scopeKind,
    ev.scopeId ?? null,
    canonicalDims(ev.dims),
    ev.amountCents ?? null,
    // I3-0890: the commit instant on the WORKER clock — the clock `selectOwed`
    // and `pruneOutbox` compare `created_at` to — never the D1 column default.
    new Date().toISOString(),
  ];

  if (!fanKeyed) {
    // The VALUES form is kept byte-identical when there is nothing to guard, so
    // an unconditional creator-scoped emit is the statement it always was.
    if (!ctx.extraWhere) return [db.prepare(`INSERT INTO analytics_outbox (${OUTBOX_COLUMNS}) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`).bind(...row)];
    return [
      db
        .prepare(`INSERT INTO analytics_outbox (${OUTBOX_COLUMNS}) SELECT ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ? WHERE (${ctx.extraWhere.sql})`)
        .bind(...row, ...ctx.extraWhere.params),
    ];
  }
  // BOTH statements or neither. The scopes row is what the erasure drain
  // enumerates, so a predicate that gated only the outbox row would leave a
  // scope for a creator this account has no row under — and one that gated only
  // the scopes row would leave a fan-keyed row the drain cannot reach.
  const extra = ctx.extraWhere ? ` AND (${ctx.extraWhere.sql})` : "";
  const extraParams = ctx.extraWhere ? ctx.extraWhere.params : [];
  return [
    db
      .prepare(
        `INSERT INTO analytics_outbox (${OUTBOX_COLUMNS}) SELECT ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ? WHERE EXISTS (SELECT 1 FROM accounts WHERE account_id = ?) AND ${LOGGING_NOT_PAUSED}${extra}`,
      )
      .bind(...row, ctx.accountId, ctx.accountId, ...extraParams),
    db
      .prepare(
        `INSERT OR IGNORE INTO analytics_fan_scopes (account_id, passport_id) SELECT ?, ? WHERE EXISTS (SELECT 1 FROM accounts WHERE account_id = ?) AND ${LOGGING_NOT_PAUSED}${extra}`,
      )
      .bind(ctx.accountId, ev.passportId, ctx.accountId, ctx.accountId, ...extraParams),
  ];
}

/** The parts of a fan-out the caller owns. Every expression is SQL over
 * `table`'s own columns. */
export interface OutboxSelectSpec {
  /** The table the rows are read from. Unless `passportExpr` says otherwise it
   * must have a `passport_id` column: that column, and only that column, is
   * copied onto the outbox row. */
  readonly table: string;
  /**
   * SQL for `passport_id`, when the table does not have that column.
   *
   * `achievements` (0005) is keyed `(account_id, achievement_id)` and the
   * creator is INSIDE the id — `completion:<psp>`, `founding_fan:<psp>`,
   * `milestone:<psp>:<n>` — which `social.ts`'s own `PSP_IN_ACHIEVEMENT` is
   * already the reader of on the emit side. Without this the erasure decrement
   * for those badges could only be spelled by putting the account id in the
   * row, or not spelled at all.
   *
   * It is held to the same three rules as every other expression here: no
   * `account_id`, no placeholders, SQL over the table's own columns. The caller
   * is responsible for a WHERE that admits only rows the expression yields a
   * real passport for — an expression that can return rubbish would put an
   * unbounded value in the row (INV-051), so the predicate is where that is
   * decided, not here.
   */
  readonly passportExpr?: string;
  /** The predicate, with `?` placeholders. Bound AFTER the day. */
  readonly where: string;
  /**
   * THE LANDED-FACT CONDITION (INV-051 U-1 on this door, D-0431): the second
   * door's form of `OutboxContext.extraWhere`. SQL with `?` placeholders, ANDed
   * onto `where` in the SAME `INSERT … SELECT`, bound AFTER `where`'s params. It
   * names this call's own witness of the authoritative write (a stamp, a flag,
   * the row that write created), so a refused write selects no row.
   * `where` says WHICH rows; this says the fact they describe has landed.
   * `tests/invariants/conditional-emit-scan.test.ts` requires it on every call
   * of this door not adjudicated as conditional by construction.
   */
  readonly extraWhere?: string;
  /** SQL for `scope_id`. Required exactly when the kind declares a scope. */
  readonly scopeIdExpr?: string;
  /** SQL producing the closed-enum `dims` JSON, or omitted for NULL. */
  readonly dimsExpr?: string;
  /** SQL for `amount_cents`. Required exactly for an amount-bearing kind. */
  readonly amountExpr?: string;
}

const IDENT_RE = /^[a-z_][a-z0-9_]*$/;
/** `account_id` anywhere in a SELECTed expression — the one thing a fan-out
 * could copy into the second store that nothing downstream would notice. */
const ACCOUNT_COLUMN_RE = /\baccount_id\b/i;

/**
 * The `INSERT … SELECT` fan-out (ADR §2 multi-row statements; §3 erasure
 * decrements): one creator-scoped row per matching table row, inside the
 * authoritative batch, with the nonce minted IN SQL.
 *
 * CREATOR-SCOPED ONLY, and not as a simplification: a `fan_key` is an HMAC
 * under a secret the database does not hold, so SQL cannot compute one. A
 * fan-keyed fan-out could only be spelled by putting the account id in the row,
 * which is the thing this design exists to prevent — so the kind is refused
 * here rather than served by a weaker row.
 *
 * The caller binds `[nowIso, ...whereParams, ...extraWhereParams]` (`nowIso` is
 * also the row's `created_at`, I3-0890 — so it must be a full instant): the day is
 * the first placeholder and the expressions may carry none of their own, so the
 * order is fixed by the shape rather than by the caller counting question marks.
 * Without `extraWhere` the statement is byte-identical to the unconditioned form.
 */
export function outboxSelectSql(kind: EventKind, spec: OutboxSelectSpec): string {
  const k = EVENT_KINDS[kind];
  if (!k) throw new EventError(`unknown event kind '${String(kind)}' — the vocabulary is closed (INV-051)`);
  if (k.carries === "fan_key") throw new EventError(`${kind} is fan-keyed: no fan_key can be computed in SQL, so it has no fan-out form (ADR-0006 §2)`);
  if (!IDENT_RE.test(spec.table)) throw new EventError("outboxSelectSql: table must be a plain identifier");
  if (!spec.where.trim()) throw new EventError("outboxSelectSql: an unpredicated fan-out would emit one row per table row");
  if (spec.extraWhere !== undefined && !spec.extraWhere.trim()) throw new EventError("outboxSelectSql: a blank extraWhere names no landed fact (INV-051 U-1)");

  const scopeIdExpr = spec.scopeIdExpr ?? null;
  if (k.scopeKind === null && scopeIdExpr) throw new EventError(`${kind} carries no scope`);
  if (k.scopeKind !== null && !scopeIdExpr) throw new EventError(`${kind} requires a scopeIdExpr`);
  const amountExpr = spec.amountExpr ?? null;
  if (k.carries === "amount" && !amountExpr) throw new EventError(`${kind} requires an amountExpr`);
  if (k.carries !== "amount" && amountExpr) throw new EventError(`${kind} carries no amount`);

  for (const expr of [scopeIdExpr, spec.dimsExpr ?? null, amountExpr, spec.passportExpr ?? null]) {
    if (expr === null) continue;
    if (ACCOUNT_COLUMN_RE.test(expr)) throw new EventError("outboxSelectSql: no account_id is selected into an analytics row (ADR-0006 §0 P2)");
    if (expr.includes("?")) throw new EventError("outboxSelectSql: expressions are SQL over the table's columns and carry no placeholders");
  }

  return (
    // I3-0890: the caller's instant is the row's created_at as well as its day,
    // bound ONCE as the first placeholder through a one-row CTE, so the
    // callers' positional bind order `[nowIso, ...where]` is unchanged.
    "WITH outbox_clock(now_iso) AS (SELECT ?) " +
    `INSERT INTO analytics_outbox (${OUTBOX_COLUMNS}) ` +
    `SELECT 'evt_' || lower(hex(randomblob(16))), ${spec.passportExpr ?? "passport_id"}, '${kind}', substr((SELECT now_iso FROM outbox_clock), 1, 10), NULL, NULL, ` +
    `${k.scopeKind === null ? "NULL" : `'${k.scopeKind}'`}, ${scopeIdExpr ?? "NULL"}, ${spec.dimsExpr ?? "NULL"}, ${amountExpr ?? "NULL"}, (SELECT now_iso FROM outbox_clock) ` +
    `FROM ${spec.table} WHERE ${spec.extraWhere === undefined ? spec.where : `(${spec.where}) AND (${spec.extraWhere})`}`
  );
}

/**
 * I20-FXD (I20-RG-TOOLS-01): THE predicate for "this SQL text writes `table`" — every spelling SQLite accepts for a
 * row write: `INSERT`, `INSERT OR <conflict>`, `REPLACE`, in any case; an optional schema prefix (`main.`); the table
 * bare or quoted (`"t"`, `` `t` ``, `[t]`). A longer table sharing the prefix (`analytics_outbox_dead`) is NOT the
 * table. Shared by `s002-events.test.ts` "one emitter" and the INV-051 census (`doorsOf`), so the two scans cannot
 * disagree about what a write is. `table` must be a plain identifier.
 */
export function sqlWritesTable(sql: string, table: string): boolean {
  if (!IDENT_RE.test(table)) throw new EventError("sqlWritesTable: table must be a plain identifier");
  const q = (x: string) => `(?:"${x}"|\`${x}\`|\\[${x}\\]|${x}(?![\\w$]))`;
  return new RegExp(`(?<![\\w$])(?:INSERT(?:\\s+OR\\s+\\w+)?|REPLACE)\\s+INTO\\s+(?:${q("\\w+")}\\s*\\.\\s*)?${q(table)}`, "i").test(sql);
}

/** A SHELVED erasure row: an erasure decrement (`kinds`, bound as ONE json_each parameter) in `analytics_outbox_dead`
 * for any reason but `delivered_above_gap` (that one was applied — the ledger holds its key — and needs no re-send).
 * The one definition: `outboxRedriveSql` below and every `analytics-outbox.ts` re-drive/alarm statement use it. */
export const shelvedErasureWhere = (alias = ""): string => `${alias}event_type IN (SELECT value FROM json_each(?)) AND ${alias}reason <> 'delivered_above_gap'`;

/**
 * #946(o) (I19-JN-ORCH:4): the OPERATOR RE-DRIVE's live-outbox INSERT, built HERE so this file is the one emitter with no
 * exemption (`s002-events.test.ts` "one emitter" scans every other `src/` file). It re-offers a row this emitter already
 * built, UNCHANGED — its own `seq` and `event_key`, so the consumer's ledger absorbs a row that did land — copied from
 * `analytics_outbox_dead` as owed (`attempts` 0, `published_at` NULL, `created_at` = the first bound value), and never
 * over a live row holding its seq or event_key. Caller: `analytics-outbox.ts` `redriveShelvedErasures`, which binds
 * `[nowIso, JSON.stringify(seqs), erasureKindsJson]` — the seqs as ONE json_each parameter, so the statement's bind count
 * does not grow with the list (D1's 100-parameter cap). Not a producer door: no event is minted, and the next analytics
 * tick publishes the owed row.
 */
export function outboxRedriveSql(columns: readonly string[], ...extra: never[]): string {
  // I20-FX2 (FXD-01b): the dead-row condition is fixed HERE, never passed in — a caller cannot widen the re-drive.
  if (extra.length > 0) throw new EventError("outboxRedriveSql: the re-drive's condition is fixed (shelved erasures); it takes no predicate");
  if (columns.length === 0 || columns.some((c) => !IDENT_RE.test(c))) throw new EventError("outboxRedriveSql: columns must be plain identifiers");
  if (!columns.includes("seq") || !columns.includes("event_key")) throw new EventError("outboxRedriveSql: a re-drive keeps the row's seq and event_key");
  const cols = columns.join(", ");
  return (
    `INSERT INTO analytics_outbox (${cols}, attempts, published_at, created_at) ` +
    `SELECT ${cols}, 0, NULL, ? FROM analytics_outbox_dead d WHERE d.seq IN (SELECT value FROM json_each(?)) AND (${shelvedErasureWhere("d.")}) ` +
    "AND NOT EXISTS (SELECT 1 FROM analytics_outbox o WHERE o.seq = d.seq OR o.event_key = d.event_key)"
  );
}
