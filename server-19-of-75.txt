#!/usr/bin/env node
/**
 * THE REAL-D1 FIDELITY SUITE, WITH A FLOOR UNDER IT.
 *
 * ============================================================================
 * WHY THIS EXISTS
 * ============================================================================
 *
 * `pnpm test:fidelity` has been a CI step and a documented HARD GATE
 * (docs/GO-LIVE-CHECKLIST.md) for a long time. It is the ONLY thing in this
 * repository that runs real workerd + real D1, and it exists to catch the class
 * of assumption the better-sqlite3 double cannot check: bound-parameter
 * ceilings, `meta.changes`, batch atomicity, `ON CONFLICT`, and — since slice 20
 * — the exact wording of a driver error that four separate refusal matchers key
 * on.
 *
 * It was nevertheless BROKEN AND SILENT for many slices. Migration 0024 added
 * eight `CREATE TRIGGER … BEGIN … ; … END;` blocks; the suite's statement
 * splitter cut them into fragments; D1 answered `incomplete input`; the shared
 * `beforeAll` died and every test reported as SKIPPED. That particular failure
 * exits nonzero, so CI would have caught it — but nothing in the LOCAL build
 * loop runs this suite, and the local loop is where every slice is actually
 * built and delivered from.
 *
 * MEASURED, and this is the part worth the script: a fidelity run whose tests
 * are all SKIPPED but whose setup does not throw **exits 0**.
 *
 *     Test Files  1 skipped (1)
 *           Tests  2 skipped (2)
 *     ALL-SKIPPED EXIT: 0
 *
 * So the suite is one `describe.skip`, one bad `include` glob, or one renamed
 * file away from passing CI while proving nothing at all — which is precisely
 * the shape of defect it was built to prevent, turned on itself.
 *
 * ============================================================================
 * WHAT IT ENFORCES
 * ============================================================================
 *
 * The same three things `scripts/check-invariants.mjs` enforces for the
 * invariant suite, and for the same reasons:
 *
 *   1. the runner actually ran THIS time (the JSON reporter's own stdout
 *      marker — a stub that copies a stale passing report does not print it);
 *   2. no test was skipped, todo, or left without an assertion;
 *   3. at least `fidelityMinTests` tests EXECUTED, a floor pinned in
 *      fixtures/invariant-manifest.json — the same file, so lowering it is a
 *      visible diff in a reviewed fixture rather than a quiet edit here.
 *
 * It deliberately does NOT re-pin file hashes: the invariant meta-guard already
 * content-pins `test-fidelity/ledger.d1.test.ts` and `test-fidelity/schema.ts`,
 * so a tampered fidelity test is already a two-file diff. This script's job is
 * only to make sure the suite RAN.
 */

// `globalThis.it`, `{ it }`, `(it)` and `[it]` bound a test function all the same. So, beside them, EVERY occurrence of

const VALUE_KEYWORDS = new Set(["return", "yield", "await", "typeof", "void", "new", "throw", "case", "in", "of", "default", "extends", "const", "let", "var", "function", "class", "as", "delete", "instanceof"]);

  if (GLOBAL_READ.test(raw)) out.push("a test function read off a global object");

    const before = code.slice(0, m.index).replace(/\s+$/, "");

    if (after.startsWith(":") && !after.startsWith("::") && (prev === "{" || prev === ",")) continue; // an object key: the literal shapes above judge `{ it: X } =`

    if (after.startsWith(".") && statementStart) continue; // it.skip(...) / it.skipIf(c)(...) as a statement
    const line = before.split("\n").length;
    out.push(`a value use of \`${m[0]}\` at line ${line}`);
  }

}

#!/usr/bin/env node
/**
 * Meta-guard for the protected invariant suites. Every check below fails
 * closed: if the script cannot determine something, it exits nonzero.
 *
 * The job is narrower than it looks. It is not "run the guards" — vitest does
 * that. It is "prove that the code that RAN is the code that was REVIEWED".
 * These things have to hold for that claim to mean anything:
 *
 *  0. HARDENED PARENT — the process that decides trust must not execute any
 *     attacker-reachable code. Two levers used to reach it: `.npmrc`
 *     `node-options=--import …` (honoured by `pnpm check:invariants`, NOT by the
 *     `node scripts/check-invariants.mjs` CI form) and an inherited NODE_OPTIONS.
 *     Both `--import` a module BEFORE this script runs, which could patch a
 *     hashing primitive's prototype and forge every content pin. So the very
 *     first thing we do is RE-EXEC ourselves once in a child with a scrubbed
 *     environment (no NODE_OPTIONS, launched via `node`, which never reads
 *     `.npmrc`). The first process does no work and hashes nothing.
 *  1. COVERAGE  — every file under a protected directory is pinned in the
 *     manifest. Without this a new guard file ships unpinned (free to be
 *     neutered later with no manifest diff) and a DELETED guard file leaves no
 *     trace, which is the whole point of invariant 5.
 *  2. CONTENT   — every pinned file (guards, the load-path machinery, AND the
 *     vitest runner entry points under node_modules) hashes to its pinned
 *     sha256, so neutering a guard, the config, a setup file, or the test runner
 *     in place requires a same-change manifest edit reviewers see. CONTENT runs
 *     BEFORE anything reads or trusts those files.
 * 2b. RUNNER GRAPH — pinning the two vitest ENTRY SHIMS was not the same as
 *     pinning the runner. `vitest.mjs` is two lines and `dist/cli.js` is a list
 *     of imports; test execution, result tallying, the JSON reporter, the stdout
 *     marker and the process exit code all live in files those two REACH. A
 *     review edited two unpinned chunks — the JSON reporter (force
 *     numFailedTests to 0 and flip fail->pass while preserving each file's
 *     assertionResults.length so the per-file counts still matched) and cli-api
 *     (neuter the `process.exitCode = 1` sites) — and this script printed
 *     `OK: … 196/196 invariant tests passed` over a genuinely red guard, marker
 *     present and startTime fresh, because the real reporter still wrote the
 *     report. So the whole runner MODULE GRAPH is pinned now, per file, in
 *     fixtures/runner-graph.json (whose own sha256 is pinned above, in 2).
 *  3. LOAD PATH — the config and the setup files are pinned, and the config's
 *     structure is asserted by STATIC INSPECTION of its now-content-verified
 *     text. The config is NEVER imported by this process: importing runs
 *     attacker-controlled top-level code in the trust-deciding process, which is
 *     exactly the hole a reordering alone does not close. `.pnpmfile.cjs` (a
 *     `pnpm install` code-execution hook) is banned and `.npmrc` is asserted to
 *     carry no `node-options`.
 *  4. EXECUTION — every guard file that exists actually reported results, the
 *     report is FRESH (its startTime is after we launched, so a copied stale
 *     report is caught), vitest actually wrote it (its stdout marker is
 *     present), no test failed or was skipped, each file ran EXACTLY its
 *     manifest-pinned test count, and the total is at or above the floor.
 *
 * The suites are WALKED from PROTECTED_DIRS (plus the NAMED PROTECTED_FILES,
 * #625 -- a guard living in an unprotected product test dir), so adding a
 * guard file needs no edit here: drop it in the directory, pin it, record its
 * test count, done. On any mismatch this script prints the exact manifest line
 * to paste — the hash, the test-count line, and the minTests floor — so raising
 * protection stays one command plus a copy-paste, while lowering it stays a
 * conscious, visible edit.
 *
 * PINNED PATHS ARE A TWO-FILE DIFF BY CONSTRUCTION. The residual — editing a
 * guard and its pin together, or editing `.github/workflows/ci.yml` to skip
 * this step — is closed by repo settings, not by code: see "Guard integrity" in
 * README.md (branch protection + CODEOWNERS on the pinned paths). Those
 * residuals are genuinely outside this code path.
 */

// I27-PINS (QUEUE #971, D-1018): a STALE PIN is collected, not fatal on the spot, so one run prints every
// stale pin and count (each in its own unchanged `FAIL:` block, the form the re-pin loop parses) instead of
// one per run. fail() prints every collected pin first; failIfDeferred() fails the run when any were
// collected, before anything is written or OK is printed. A BLOCKING pin (the runner or a load-path file the
// spawn trusts) stops the run before vitest is spawned.
const deferredPins = [];
function deferPin(msg, blocksExecution) {
  deferredPins.push({ msg, blocksExecution: blocksExecution === true });
}
function fail(msg) {
  for (const d of deferredPins.splice(0)) console.error(`FAIL: ${d.msg}`);
  console.error(`FAIL: ${msg}`);
  process.exit(1);
}
function failIfDeferred(context) {
  if (deferredPins.length > 0) {
    fail(`${deferredPins.length} stale pin/count problem(s) reported above in one pass${context} — apply every one, then re-run`);
  }
}
// QUEUE #499 (B23-C4): a TREE PRECONDITION that stops the run before any guard executes is a
// REFUSAL, not a guard failure. Same `FAIL:` line and same exit 1 (nothing downstream re-keys);

const REFUSED_MARKER = "check-invariants: REFUSED — a precondition on the tree failed before any guard executed";

/**
 * Remove the environment levers that can `--import`/`--require` code or
 * otherwise steer a Node process. Absent is the safe value for every one of
 * these, so deleting is always correct.
 * QUEUE #253: every `GIT_*` variable too. `GIT_INDEX_FILE`, `GIT_DIR`,
 * `GIT_WORK_TREE`, `GIT_CONFIG_PARAMETERS` … redirect what `git ls-files`
 * returns, and that listing IS every guard's population (0c below;
 * tests/invariants/repo-files.ts runs git inside the vitest child).
 * `GIT_INDEX_FILE=/nonexistent` used to hand every guard an empty population.
 */
function scrubEnv(base) {
  const env = { ...base };
  delete env["NODE_OPTIONS"];
  delete env["NODE_REPL_EXTERNAL_MODULE"];
  delete env["NODE_V8_COVERAGE"];
  for (const key of Object.keys(env)) {
    if (key.startsWith("VITEST_") || key.startsWith("VITE_") || key.startsWith("GIT_")) delete env[key];
  }
  return env;
}

// ---------------------------------------------------- CLI: --shard / --merge
// QUEUE #266. Section 4 (EXECUTION) is the only section that spawns vitest,
// and it is the ~64s+132s cost that no longer fits one ~175s call over the
// full protected population from this persistent build tree. `--shard i/N`
// restricts section 4's vitest spawn to the i-th slice of a stable,
// path-sorted partition of guardTests and defers the checks that need the
// WHOLE population (missing-file / per-file-count / minTests) to a later
// `--merge N`, which spawns no vitest and instead reads every shard's
// persisted result. Neither flag changes anything above section 4. Without
// either flag, behavior is exactly what it was before this flag existed.
const SHARD_STATE_DIR = abs(".scratch/invariants");
function parseShardArgs(argv) {
  let shardSpec = null;
  let mergeSpec = null;
  for (let i = 0; i < argv.length; i++) {
    const a = argv[i];
    if (a === "--shard") shardSpec = argv[++i];
    else if (a.startsWith("--shard=")) shardSpec = a.slice("--shard=".length);
    else if (a === "--merge") mergeSpec = argv[++i];
    else if (a.startsWith("--merge=")) mergeSpec = a.slice("--merge=".length);
  }
  if (shardSpec !== null && mergeSpec !== null) fail("--shard and --merge are mutually exclusive");
  if (shardSpec !== null) {
    const m = /^(\d+)\/(\d+)$/.exec(shardSpec);
    if (!m) fail(`--shard requires "i/N" (got ${JSON.stringify(shardSpec)})`);
    const i = Number(m[1]);
    const n = Number(m[2]);
    if (n < 1 || i < 1 || i > n) fail(`--shard i/N requires 1 <= i <= N (got ${i}/${n})`);
    return { mode: "shard", i, n };
  }
  if (mergeSpec !== null) {
    const n = Number(mergeSpec);
    if (!Number.isInteger(n) || n < 1) fail(`--merge requires a positive integer N (got ${JSON.stringify(mergeSpec)})`);
    return { mode: "merge", n };
  }
  return { mode: "full" };
}
const cli = parseShardArgs(process.argv.slice(2));

// ------------------------------------------------- CLI: --retire / --docket
// QUEUE #229 / D-0424(2). A floor recorded in the manifest (minTests, a
// mutationFloors[file] entry, or a per-file "tests" count) may go DOWN from
// its value at HEAD only when the caller names the EXCEPTION-DOCKET.md row
// that ruled the retirement -- see the "0d. FLOOR RATCHET" section below,
// which is the only reader of this value. Naming --retire/--docket here
// never itself permits anything; it only earns a look at that row.
// TOOLING-EXTRACT-BEGIN parseRetireArgs
function parseRetireArgs(argv) {
  let retireInv = null;
  let docket = null;
  for (let i = 0; i < argv.length; i++) {
    const a = argv[i];
    if (a === "--retire") retireInv = argv[++i] ?? null;
    else if (a.startsWith("--retire=")) retireInv = a.slice("--retire=".length);
    else if (a === "--docket") docket = argv[++i] ?? null;
    else if (a.startsWith("--docket=")) docket = a.slice("--docket=".length);
  }
  if (retireInv !== null && docket === null) fail("--retire requires --docket <n>");
  if (docket !== null && retireInv === null) fail("--docket requires --retire <INV-id>");
  // JN-2 (S-081 wave A judge, round 2): --retire used to accept any string at
  // all, including "" or a substring of a real id -- the ONLY thing standing
  // between an arbitrary floor decrease and a "permitted" result was
  // docketRowRuled's row match (see below), so a loose id here made that
  // match easier to satisfy by accident. Require the manifest's own id shape.
  if (retireInv !== null && !/^INV-\d{3}$/.test(retireInv)) {
    fail(`--retire requires an invariant id of the form INV-NNN (got ${JSON.stringify(retireInv)})`);
  }
  if (docket !== null && !/^\d+$/.test(docket)) fail(`--docket requires a row number (got ${JSON.stringify(docket)})`);
  return retireInv !== null ? { inv: retireInv, docket: Number(docket) } : null;
}

function sha256File(fileAbs) {
  return createHash("sha256").update(readFileSync(fileAbs)).digest("hex");
}

/**
 * Strip `//` line and block comments from JS source, preserving string and
 * template-literal contents, so the config's own explanatory prose does not
 * trip the structural checks below. Not a full parser — enough for a config
 * whose only strings are simple path/name literals.
 */
function stripComments(src) {
  let out = "";
  let i = 0;
  const n = src.length;
  let quote = null; // current string delimiter, or null
  while (i < n) {
    const c = src[i];
    const d = src[i + 1];
    if (quote) {
      out += c;
      if (c === "\\") {
        out += d ?? "";
        i += 2;
        continue;
      }
      if (c === quote) quote = null;
      i += 1;
      continue;
    }
    if (c === '"' || c === "'" || c === "`") {
      quote = c;
      out += c;
      i += 1;
      continue;
    }
    if (c === "/" && d === "/") {
      while (i < n && src[i] !== "\n") i += 1;
      continue;
    }
    if (c === "/" && d === "*") {
      i += 2;
      while (i < n && !(src[i] === "*" && src[i + 1] === "/")) i += 1;
      i += 2;
      continue;
    }
    out += c;
    i += 1;
  }
  return out;
}

/** Every file under `dirAbs`, relative + POSIX + sorted, NOT following symlinks
 *  and NOT descending into a nested `node_modules` (under pnpm those are links
 *  into the store, and following them would walk the whole dependency tree). */
// POPULATION: CHECKOUT — installed runner packages, not repo files; the runner graph pins node_modules as installed
function walkPackage(dirAbs) {
  const out = [];
  const recurse = (abs, prefix) => {
    for (const entry of readdirSync(abs).sort()) {
      if (entry === "node_modules") continue;
      const full = join(abs, entry);
      const st = lstatSync(full);
      if (st.isSymbolicLink()) continue;
      if (st.isDirectory()) recurse(full, `${prefix}${entry}/`);
      else out.push(`${prefix}${entry}`);
    }
  };
  recurse(dirAbs, "");
  return out.sort();
}

/** Resolve an installed package's real directory, walking up from `fromAbs`. */
function packageDir(fromAbs, name) {
  let dir = fromAbs;
  for (;;) {
    const candidate = join(dir, "node_modules", ...name.split("/"));
    if (existsSync(join(candidate, "package.json"))) return realpathSync(candidate);
    const parent = dirname(dir);
    if (parent === dir) return null;
    dir = parent;
  }
}

/**
 * The runner graph as it exists on disk right now: vitest's version, the
 * package names in scope, and `<package>/<path>` -> sha256 for every file in
 * them. Paths are keyed by PACKAGE NAME, never by the on-disk location, because
 * under pnpm the store directory carries a peer-dependency hash
 * (`vitest@3.2.7_@types+node@22.20.1_tsx@4.23.11`) that changes for reasons
 * that have nothing to do with the runner's bytes.
 */
function computeRunnerGraph() {
  const vitestDir = realpathSync(join(root, "node_modules", "vitest"));
  let pkg;
  try {
    pkg = JSON.parse(readFileSync(join(vitestDir, "package.json"), "utf8"));
  } catch (e) {
    fail(`cannot read the vitest package manifest: ${e.message}`);
  }
  const names = ["vitest", ...Object.keys(pkg.dependencies ?? {}).filter((n) => !n.startsWith("@types/"))].sort();
  const files = {};
  for (const name of names) {
    const dir = name === "vitest" ? vitestDir : packageDir(vitestDir, name);
    // FAIL CLOSED: a runner dependency we cannot find is a runner we cannot pin.
    if (dir === null) fail(`runner package not installed: ${name} — cannot pin a graph with a hole in it`);
    for (const rel of walkPackage(dir)) files[`${name}/${rel}`] = sha256File(join(dir, rel));
  }
  return { vitest: String(pkg.version ?? ""), packages: names, files };
}

// -------------------------------------------------------- 0d. FLOOR RATCHET
// QUEUE #229 / D-0424(2). A protective floor recorded in the manifest
// (minTests, a mutationFloors[file] entry, or a per-file "tests" count) may
// only ever go UP in a reviewed change -- a lower value is the same lever as
// weakening a guard, spelled as a one-line manifest edit instead. "previous"
// is the value committed at HEAD (`git show HEAD:fixtures/invariant-manifest.json`);
// this compares the manifest already parsed above (the one ON DISK now, which
// may carry an uncommitted edit) against it. A decrease is permitted ONLY
// when the caller passes `--retire <INV-id> --docket <n>` naming an
// org/01-current/EXCEPTION-DOCKET.md row that IS row <n> and that names both
// the INV id and the literal text "RULED" -- an approved, on-the-record
// retirement, never a builder's own judgment call.
// TOOLING-EXTRACT-BEGIN docketRowRuled
function docketRowRuled(docketText, n, invId) {
  // Every docket row is one line, "| <n> | ...", per org/01-current/EXCEPTION-DOCKET.md's
  // own table format -- anchoring at the start of a line on "| <n> |" cannot
  // match inside another row's prose (a stray "| 5 |" mid-sentence would not
  // be at column 0), and `n` is unique per row by construction (the docket is
  // a hand-maintained numbered table, never renumbered in place).
  const re = new RegExp(`^\\|\\s*${n}\\s*\\|.*$`, "m");
  const m = re.exec(docketText);
  if (!m) return { ok: false, reason: `no row "| ${n} |" found` };
  const row = m[0];
  // JN-2: a bare `row.includes(invId)` substring test also matched invId as
  // a SUBSTRING of a different, unrelated id in the row -- e.g. --retire
  // INV-9 would match a row naming only INV-99, and an id embedded in a
  // longer token (INV-042X) would match too. invId is validated to
  // /^INV-\d{3}$/ by parseRetireArgs before this ever runs, so no regex
  // metacharacter escaping is needed for that caller, but this function is
  // exercised directly by tests with arbitrary strings too, so escape it
  // defensively and match only on a word/hyphen boundary either side.
  const escapedInv = invId.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
  const invRe = new RegExp(`(?<![\\w-])${escapedInv}(?![\\w-])`);
  if (!invRe.test(row)) return { ok: false, reason: `row ${n} does not name ${invId}` };
  // REG-2: a bare `row.includes("RULED")` substring test also matched a
  // NEGATED ruling -- "NOT RULED", "NOT YET RULED" and "UNRULED" all contain
  // the substring "RULED" -- the opposite of an approved retirement. Require
  // the bold **RULED token (the convention every affirmative ruling in
  // EXCEPTION-DOCKET.md uses), and additionally refuse a match whose
  // immediately preceding text ends in "NOT" even when the bold span wraps
  // only the verdict word ("NOT **RULED**") rather than the whole phrase.
  const RULED_RE = /\*\*RULED\b/g;
  let affirmed = false;
  for (const m2 of row.matchAll(RULED_RE)) {
    const before = row.slice(Math.max(0, m2.index - 8), m2.index);
    if (!/NOT\s*$/.test(before)) {
      affirmed = true;
      break;
    }
  }
  if (!affirmed) return { ok: false, reason: `row ${n} does not carry an affirmative "**RULED" token` };
  return { ok: true };
}
// TOOLING-EXTRACT-END docketRowRuled

// QUEUE #301 (S-081 wave B). Maps each `INV-NNN` row in docs/INVARIANTS.md to
// the guard files its own "how enforced" column names, so a floor decrease
// can be scoped to the specific files an INV id's row actually cites,
// instead of a valid --retire/--docket pair blanket-permitting every
// decrease found in the whole manifest. A row's guard column is prose, not
// structured data: every backtick-quoted token ending in `.test.ts`/`.tsx`
// is a candidate; an exact `tests` map path wins outright, otherwise its
// basename is matched (glob-aware, so `run0001-frozen-*.test.ts` resolves
// to every `tests` map path whose basename fits) against every `tests` map
// key's own basename. An INV row with no such token (a "no guard" row, or
// one naming only prose like `enforced (partial)`) maps to no files at all
// -- retiring it permits nothing.
// TOOLING-EXTRACT-BEGIN invariantGuardMap
function parseInvariantGuardMap(docsText, testsMapKeys) {
  const byBasename = new Map();
  for (const k of testsMapKeys) {
    const base = k.split("/").pop();
    if (!byBasename.has(base)) byBasename.set(base, []);
    byBasename.get(base).push(k);
  }
  function globToRe(glob) {
    const esc = glob.replace(/[.+^${}()|[\]\\]/g, "\\$&").replace(/\*/g, ".*");
    return new RegExp(`^${esc}$`);
  }
  const invToFiles = new Map();
  const rowRe = /^\| (INV-\d{3}) \|.*\|\s*$/gm;
  for (const m of docsText.matchAll(rowRe)) {
    const invId = m[1];
    const row = m[0];
    const files = invToFiles.get(invId) ?? new Set();
    const codeRe = /`([^`]+)`/g;
    for (const c of row.matchAll(codeRe)) {
      const token = c[1].trim();
      const base = token.split("/").pop();
      if (!/\.test\.tsx?$/.test(base)) continue; // not a guard-file-shaped token
      if (testsMapKeys.includes(token)) {
        files.add(token);
        continue;
      }
      if (base.includes("*")) {
        const re = globToRe(base);
        for (const [b, paths] of byBasename) {
          if (re.test(b)) for (const p of paths) files.add(p);
        }
        continue;
      }
      const exact = byBasename.get(base);
      if (exact) for (const p of exact) files.add(p);
    }
    if (files.size > 0) invToFiles.set(invId, files);
  }
  return invToFiles;
}
// TOOLING-EXTRACT-END invariantGuardMap

// QUEUE #301: `--merge`'s (or a full run's) own success is the only thing
// that writes this -- a SEPARATE signal from HEAD's committed manifest, so a
// floor lowered inside the very commit that re-pins the manifest (the wave +
// repin commit JN-3 flagged as invisible to a HEAD-only comparison) is still
// caught: the ratchet below compares against the HIGHER of HEAD's value and
// this file's, and this file is written from what actually, recently,
// verifiably ran green -- never from the manifest text alone.
// TOOLING-EXTRACT-BEGIN lastGreen
function lastGreenPath(stateDir) {
  return join(stateDir, "last-green.json");
}
function readLastGreen(stateDir) {
  let data;
  try {
    data = JSON.parse(readFileSync(lastGreenPath(stateDir), "utf8"));
  } catch {
    return null;
  }
  if (!data || typeof data !== "object") return null;
  return data;
}
function higherOf(a, b) {
  if (typeof a !== "number" || !Number.isFinite(a)) return typeof b === "number" && Number.isFinite(b) ? b : undefined;
  if (typeof b !== "number" || !Number.isFinite(b)) return a;
  return Math.max(a, b);
}
// TOOLING-EXTRACT-END lastGreen

// QUEUE #301: the ratchet's decision logic is three small pure functions
// (baseline, which decreases exist, whether a named retirement's own
// mapped files cover them) so each is independently testable against
// synthetic manifests/baselines, with no fail()/fs access of its own. The
// block below is the thin runtime wrapper: it supplies real data and turns
// a "not ok" result into fail().
// TOOLING-EXTRACT-BEGIN floorRatchetDecreases
function computeFloorBaseline(prevManifest, lastGreen) {
  const baseMinTests = higherOf(prevManifest?.minTests, lastGreen?.minTests);
  const baseFidelityMinTests = higherOf(prevManifest?.fidelityMinTests, lastGreen?.fidelityMinTests);
  const baseMutationFloors = {};
  for (const f of new Set([...Object.keys(prevManifest?.mutationFloors ?? {}), ...Object.keys(lastGreen?.mutationFloors ?? {})])) {
    const v = higherOf(prevManifest?.mutationFloors?.[f], lastGreen?.mutationFloors?.[f]);
    if (v !== undefined) baseMutationFloors[f] = v;
  }
  const baseTests = {};
  for (const f of new Set([...Object.keys(prevManifest?.tests ?? {}), ...Object.keys(lastGreen?.tests ?? {})])) {
    const v = higherOf(prevManifest?.tests?.[f], lastGreen?.tests?.[f]);
    if (v !== undefined) baseTests[f] = v;
  }
  return { baseMinTests, baseFidelityMinTests, baseMutationFloors, baseTests };
}

function computeFloorDecreases(baseline, manifest) {
  const { baseMinTests, baseMutationFloors, baseTests } = baseline;
  const minTestsDecrease = Number.isInteger(baseMinTests) && Number.isInteger(manifest.minTests) && manifest.minTests < baseMinTests;
  const mutationFloorDecreases = [];
  for (const [f, baseFloor] of Object.entries(baseMutationFloors)) {
    const curFloor = manifest.mutationFloors && manifest.mutationFloors[f];
    if (typeof curFloor === "number" && curFloor < baseFloor) mutationFloorDecreases.push([f, baseFloor, curFloor]);
  }
  const testsDecreases = [];
  for (const [f, baseCount] of Object.entries(baseTests)) {
    const curCount = manifest.tests && manifest.tests[f];
    if (typeof curCount === "number" && curCount < baseCount) testsDecreases.push([f, baseCount, curCount]);
  }
  return { minTestsDecrease, mutationFloorDecreases, testsDecreases };
}

// QUEUE #301: mutationFloors is intentionally OUT of scope here -- it is
// never bound to an INV id (docs/INVARIANTS.md's guard column names TEST
// FILES, never a mutation subject), so it has no place in a --retire's
// scope check; a mutationFloors decrease is rejected before this is even
// called (see the runtime wrapper).
function checkRetireScope(decreases, baseline, manifest, invFiles) {
  const { minTestsDecrease, testsDecreases } = decreases;
  const { baseMinTests, baseTests } = baseline;
  const unmappedTestsDecreases = testsDecreases.filter(([f]) => !invFiles.has(f));
  if (unmappedTestsDecreases.length > 0) {
    return { ok: false, reason: "unmapped-tests-decrease", unmappedTestsDecreases };
  }
  if (minTestsDecrease) {
    const budget = [...invFiles].reduce((sum, f) => sum + (baseTests[f] ?? 0), 0);
    const actualDrop = baseMinTests - manifest.minTests;
    if (actualDrop > budget) {
      return { ok: false, reason: "minTests-budget-exceeded", budget, actualDrop };
    }
  }
  return { ok: true };
}

function indexedUnder(dir, what) {
  const ignoredStrays = git([...GIT_IGNORED_UNTRACKED, "--", dir]).split("\0").filter((p) => p !== "");
  if (ignoredStrays.length > 0) {

      `ignored, untracked file(s) under the ${what} ${dir} — an ignore rule does not exempt it:\n` +
        ignoredStrays.map((f) => `    ${f}`).join("\n") +
        "\nstage it (`git add`), or delete it.",
    );
  }
  return tracked.filter((f) => f.startsWith(`${dir}/`)).sort();
}

// ----------------------------------------------- 1a. TRANSITIVE HELPER IMPORTS
// QUEUE #191. A pinned guard importing an UNPINNED test helper is a one-file,
// zero-manifest-diff lever over what the guard actually exercises (the same
// shape S-061 found for `services/image/test/doubles.ts` and
// `services/trust-core/test/fan-session.ts`, now pinned by hand — which is
// exactly the problem: the load-path enumeration was maintained by hand).
// Walk every protected file's static relative imports TRANSITIVELY, through
// helper files, stopping at any path under a `src/` directory (production
// code — never required to be pinned, the manifest's model only pins the TEST
// load path) and at node_modules (npm packages, covered by the lockfile /
// runner graph instead). Any non-src helper reached this way that is not a
// `manifest.files` key fails closed. Regex-scanned, not parsed: enough for
// this repo's uniform `import { x } from "../../a/b.js"` shape.
// TOOLING-EXTRACT-BEGIN importSpecRegexes
const IMPORT_SPEC_RE = /\bfrom\s+["']([^"']+)["']/g;
// QUEUE #276: the original regex only ever matched a `from "..."` clause
// (a normal import AND `export ... from "..."`, both already covered by the
// word "from"). Three load-path edges it could not see: a bare side-effect
// import (`import "./x"`, no `from`), a dynamic `import("./x")` with a
// literal specifier, and `vi.importActual("./x")` / `vi.mock("./x")` --
// calls, not import statements, but each one puts an unpinned file on a
// guard's load path exactly like a real import does.
const SIDE_EFFECT_IMPORT_RE = /\bimport\s+["']([^"']+)["']/g;
const DYNAMIC_IMPORT_RE = /\bimport\(\s*["']([^"']+)["']\s*\)/g;
const VI_MOCK_RE = /\bvi\.(?:importActual|mock)\(\s*["']([^"']+)["']/g;
const ALL_IMPORT_SPEC_RES = [IMPORT_SPEC_RE, SIDE_EFFECT_IMPORT_RE, DYNAMIC_IMPORT_RE, VI_MOCK_RE];
// TOOLING-EXTRACT-END importSpecRegexes
function staticRelativeImportSpecs(fileAbs) {
  let text;
  try {
    text = readFileSync(fileAbs, "utf8");
  } catch {
    return [];
  }
  const specs = [];
  for (const re of ALL_IMPORT_SPEC_RES) {
    for (const m of text.matchAll(re)) {
      const spec = m[1];
      if (spec.startsWith("./") || spec.startsWith("../")) specs.push(spec);
    }
  }
  return specs;
}
function resolveRelativeImport(fromRel, spec) {
  const joined = join(dirname(fromRel), spec).split(sep).join("/");
  const candidates = /\.(m|c)?[jt]sx?$/.test(joined)
    ? [joined.replace(/\.(m|c)?jsx?$/, ".ts"), joined.replace(/\.(m|c)?jsx?$/, ".tsx"), joined]
    : [`${joined}.ts`, `${joined}.tsx`, `${joined}/index.ts`, joined];
  for (const c of candidates) {
    if (existsSync(abs(c)) && statSync(abs(c)).isFile()) return c;
  }
  return null;
}
// Q974-B2 + D-1059(17) (I30-RG-REG NEW-2): the file the RUNNER loads for a relative import, or null when nothing loads.
// `resolveRelativeImport` above tries `.ts` first, but vite (vitest 3 / vite 7 tryCleanFsResolve) loads the EXACT file
// first, so `./h.js` loads h.js and `./g` loads g.mjs or g.js even when a pinned h.ts / g.ts sits beside it. Callers
// verify THIS file: unnamed by the resolver -> unverifiable, named differently -> shadowed; both refused. `node` (a
// script node runs, e.g. family-index.mjs) loads the exact path only. An import that loads nothing cannot run (most
// such specs are fixture text inside a guard's own string literals, which the regex scan also sees).
// TOOLING-EXTRACT-BEGIN runnerLoads
const VITE_EXTENSIONS = [".mjs", ".js", ".mts", ".ts", ".jsx", ".tsx", ".json"];
function runnerLoads(fromRel, spec, runner) {
  const joined = join(dirname(fromRel), spec).split(sep).join("/");
  const isFile = (c) => existsSync(abs(c)) && statSync(abs(c)).isFile();
  if (isFile(joined)) return joined;
  if (runner === "node") return null;
  const tsSource = /\.(?:m|c)?jsx?$/.test(joined) ? [joined.replace(/js(x?)$/, "ts$1"), ...(joined.endsWith(".js") ? [`${joined.slice(0, -3)}.tsx`] : [])] : [];
  for (const c of [...tsSource, ...VITE_EXTENSIONS.map((e) => `${joined}${e}`)]) if (isFile(c)) return c;
  // a directory: vite loads its package.json entry when one exists (an entry this check cannot name), else index.<ext>
  if (isFile(`${joined}/package.json`)) return `${joined}/package.json`;
  for (const e of VITE_EXTENSIONS) if (isFile(`${joined}/index${e}`)) return `${joined}/index${e}`;
  return null;
}
// TOOLING-EXTRACT-END runnerLoads

// QUEUE #280. `shardSliceOf` used to be round-robin by POSITION only, blind
// to how long each guard file actually takes to run -- a shard could land
// every slow file and another every fast one. `--shard` now bin-packs the
// CURRENT population into `n` shards greedy-longest-first (largest file
// first, always into the currently-lightest bin) using the per-file wall
// times (`endTime - startTime` from vitest's own JSON report) a prior
// successful `--merge` persisted to `shard-timings.json` -- see JN-1 below
// for why a shard never reads timings its own pass produced. A file with no
// recorded time yet (new, or never shown in a merged run) is estimated at
// the mean of the times that ARE known, so it still lands somewhere balanced
// rather than always in bin 0. First run ever (nothing persisted yet) has
// nothing to read and falls back to round-robin, exactly as before this
// queue item.
// TOOLING-EXTRACT-BEGIN shardPlan
// JN-1 (S-081 wave A judge, round 2): the REG-1 fix above still recomputed
// the plan from whatever shard result files happened to be on disk AT THE
// MOMENT OF EACH `--shard` CALL -- including files the SAME pass's earlier
// shards had just written. Shard 2 of a pass then saw shard 1's fresh timing
// data and replanned the whole population differently than shard 1 assumed,
// so a later `--merge` found a partition that did not match what any single
// shard actually ran. A pass now uses ONE frozen plan: `--shard` writes
// `shard-plan.json` (tree digest, n, the slices, a planId) the first time it
// finds no plan matching the CURRENT tree digest + n; every later `--shard`
// call of the same pass reads that file back verbatim instead of
// recomputing. The timings a plan is built from live in a separate file,
// `shard-timings.json`, written ONLY by a successful `--merge` (never by
// `--shard`) from the shard files it just read -- so no shard of a pass ever
// observes a timing measurement taken during that same pass. A successful
// merge also deletes `shard-plan.json`, so the NEXT pass (even against the
// same tree digest, if nothing changed) replans from the freshly persisted
// timings rather than reusing the old plan forever. Both paths are single
// named-file reads/writes -- no directory is ever listed (REG-1).
function shardPlanPath(stateDir) {
  return join(stateDir, "shard-plan.json");
}
function shardTimingsPath(stateDir) {
  return join(stateDir, "shard-timings.json");
}
function readPersistedTimings(stateDir) {
  const timings = new Map(); // file -> ms, from the last successful --merge only
  let data;
  try {
    data = JSON.parse(readFileSync(shardTimingsPath(stateDir), "utf8"));
  } catch {
    return timings; // no merge has ever persisted timings yet -- round-robin fallback
  }
  if (!data || typeof data.fileTimingsMs !== "object") return timings;
  for (const [f, ms] of Object.entries(data.fileTimingsMs)) {
    if (Number.isFinite(ms) && ms >= 0) timings.set(f, ms);
  }
  return timings;
}

// QUEUE #304 (S-081 wave B). Before this fix, a pass with NO persisted
// per-file timings (the very first pass ever, or any pass after every prior
// merge failed) fell back to round-robin AND printed every bin's predicted
// time as the identical literal "unknown (no prior timing data)" -- every
// shard really did show the same "total" on the console, because there was
// nothing at all to distinguish them with. Round-robin is replaced by a
// SECOND greedy bin-pack, weighted by each file's byte size on disk (`wc -c`
// of the test file itself -- a cheap, always-available proxy correlated
// with how much there is to parse/run, computed by the caller and passed in
// as `sizeBytes` so this function stays a pure `Map`-in, plan-out helper
// with no fs access of its own). This is explicitly NOT a time estimate --
// `totalMs` stays `null` (so `--merge`'s and the CLI's ms-based comparisons
// are untouched) and `totalBytes` carries the size-based total instead, so
// the console line reads "~N KB (…) — sized estimate, not a timing" rather
// than implying second-accuracy it does not have. It still gets shards
// visibly UNEQUAL in composition and printed value from the very first
// pass, instead of four identical placeholder lines.
function computeShardBins(population, n, timings, sizeBytes) {
  if (timings.size === 0) {
    const sizes = population.map((f) => [f, sizeBytes instanceof Map && sizeBytes.has(f) ? sizeBytes.get(f) : 0]);
    sizes.sort((a, b) => b[1] - a[1] || a[0].localeCompare(b[0]));
    const bins = Array.from({ length: n }, () => ({ files: [], totalMs: null, totalBytes: 0 }));
    for (const [f, sz] of sizes) {
      let best = 0;
      for (let k = 1; k < n; k++) if (bins[k].totalBytes < bins[best].totalBytes) best = k;
      bins[best].totalBytes += sz;
      bins[best].files.push(f);
    }
    for (const b of bins) b.files.sort();
    return { bins, source: "greedy longest-first by file size (no prior timing data yet)" };
  }
  const knownValues = [...timings.values()];
  const mean = knownValues.reduce((a, b) => a + b, 0) / knownValues.length;
  const withTimes = population.map((f) => [f, timings.has(f) ? timings.get(f) : mean]);
  // Largest first; ties broken by path so the plan is deterministic for a
  // fixed (population, timings) pair, same guarantee the old round-robin gave.
  withTimes.sort((a, b) => b[1] - a[1] || a[0].localeCompare(b[0]));
  const bins = Array.from({ length: n }, () => ({ files: [], totalMs: 0 }));
  for (const [f, t] of withTimes) {
    let best = 0;
    for (let k = 1; k < n; k++) if (bins[k].totalMs < bins[best].totalMs) best = k;
    bins[best].totalMs += t;
    bins[best].files.push(f);
  }
  for (const b of bins) b.files.sort();
  return { bins, source: "greedy longest-first over persisted per-file timings" };
}

function loadOrCreatePlan(stateDir, population, n, digest, sizeBytes) {
  const p = shardPlanPath(stateDir);
  let existing = null;
  try {
    existing = JSON.parse(readFileSync(p, "utf8"));
  } catch {
    existing = null;
  }
  if (
    existing &&
    existing.treeDigest === digest &&
    existing.n === n &&
    Array.isArray(existing.bins) &&
    existing.bins.length === n &&
    typeof existing.planId === "string"
  ) {
    return existing; // frozen plan for this pass -- never recompute mid-pass
  }
  const timings = readPersistedTimings(stateDir);
  const { bins, source } = computeShardBins(population, n, timings, sizeBytes);
  const planId = createHash("sha256")
    .update(`${digest}:${n}:${Date.now()}:${Math.random()}`)
    .digest("hex")
    .slice(0, 16);
  const plan = { treeDigest: digest, n, planId, source, bins };
  mkdirSync(stateDir, { recursive: true });
  writeFileSync(p, JSON.stringify(plan, null, 2));
  return plan;
}
// TOOLING-EXTRACT-END shardPlan


// QUEUE #266: a digest of the manifest's OWN file pins (path:sha256 pairs,
// sorted) — not a `git` call. Section 2 (CONTENT, above) already proved every
// one of those hashes matches the file on disk in THIS process, so this
// digest stands in for "the reviewed, pinned tree" the same way an individual
// pin does, and a shard run needs no extra git access beyond what section 0c
// already required. `--merge` fails if two shards' digests disagree — a
// pinned file changed between shard runs.
function pinHash() {
  const h = createHash("sha256");
  for (const k of Object.keys(manifest.files).sort()) {
    h.update(k);
    h.update("\0");
    h.update(manifest.files[k]);
    h.update("\n");
  }
  return h.digest("hex");
}

// REG-SHARD-STALE: pinHash() covers only the manifest's OWN pins; production
// src/ is never pinned, so a shard's green result must also carry a digest of
// the WHOLE tracked tree (index state + any uncommitted diff against HEAD).
// `--merge` recomputes this now and requires every shard to match it, so an
// unpinned source edit between one shard's run and the merge is caught even
// though no pinned file changed.
function treeDigest() {
  const indexHash = createHash("sha256").update(git(["ls-files", "-s", "-z"])).digest("hex");
  const diffHash = createHash("sha256").update(git(["diff", "--binary", "HEAD"])).digest("hex");
  return `${indexHash}:${diffHash}`;
}

function runVitestOn(testFiles) {
  const REPORT_DIR = mkdtempSync(join(tmpdir(), "dhwaja-invariants-"));
  const ENV_DIR = mkdtempSync(join(tmpdir(), "dhwaja-invariant-env-"));
  const OUT = join(REPORT_DIR, "report.json");
  const MARKER = `JSON report written to ${OUT}`;
  const launchedAt = Date.now();
  function cleanup() {
    for (const d of [REPORT_DIR, ENV_DIR]) {
      try {
        rmSync(d, { recursive: true, force: true });
      } catch {
        /* best effort — a leaked temp dir is not worth failing the run over */
      }
    }
  }
  const env = scrubEnv(process.env);
  env["INVARIANT_INCLUDE"] = testFiles.join("\n");
  env["INVARIANT_ENV_DIR"] = ENV_DIR; // the config reads its envDir from here

  let stdout = "";
  try {
    stdout = execFileSync(
      process.execPath,
      [VITEST_BIN, "run", "--config", join(root, CONFIG_REL), "--reporter=json", `--outputFile=${OUT}`],
      { cwd: root, stdio: ["ignore", "pipe", "inherit"], env, encoding: "utf8" },
    );
  } catch (e) {
    if (typeof e?.stdout === "string") process.stdout.write(e.stdout);
    // FIX2 item 4: vitest exits nonzero on ANY red test, which used to
    // short-circuit straight to a bare "invariant suite exited nonzero" —
    // true but useless, since the JSON report was usually still written. If
    // the marker printed, best-effort read + parse it here (before cleanup)
    // and print each failing test's name and first failure line, so the
    // caller sees WHAT failed without re-running by hand.
    if (typeof e?.stdout === "string" && e.stdout.includes(MARKER)) {
      try {
        const report = JSON.parse(readFileSync(OUT, "utf8"));
        const lines = [];
        for (const r of report.testResults ?? []) {
          for (const a of r.assertionResults ?? []) {
            if (a.status !== "passed") {
              const firstFailureLine = (a.failureMessages?.[0] ?? "").split("\n")[0];
              lines.push(`  ${relative(root, r.name).split(sep).join("/")} :: ${a.fullName ?? a.title} — ${firstFailureLine}`);
            }
          }
        }
        if (lines.length > 0) console.error(`FAILING TESTS:\n${lines.join("\n")}`);
      } catch {
        /* best effort — an unreadable report here falls through to the generic fail() below */
      }
    }
    cleanup();
    fail("invariant suite exited nonzero");
  }
  process.stdout.write(stdout);

  // Vitest's json reporter prints this line to stdout when it writes the
  // file. A stub that copies a passing report and exits 0 does not print it
  // — this is the tell that "the report exists" is not the same as "vitest
  // wrote it this run".
  if (!stdout.includes(MARKER)) {
    cleanup();
    fail(
      "vitest did not report writing the JSON output this run (missing 'JSON report written to …'). " +
        "The runner may have been replaced by a stub that copies a stale report.",
    );
  }

  let report;
  try {
    report = JSON.parse(readFileSync(OUT, "utf8"));
  } catch (e) {
    cleanup();
    fail(`cannot read vitest JSON output: ${e.message}`);
  }

  // FRESHNESS: the report must have been produced by THIS run. A copied stale
  // report carries an older startTime. Allow a small skew backward for clock
  // coarseness but reject anything predating our launch by more than a
  // second.
  const startTime = Number(report.startTime);
  if (!Number.isFinite(startTime) || startTime < launchedAt - 1000) {
    cleanup();
    fail(
      `vitest report is not fresh (startTime ${report.startTime}, this run launched at ${launchedAt}). ` +
        "A stale report was substituted for a real run.",
    );
  }

  // QUEUE #230: a count the report does not carry cannot be read as 0.
  for (const k of ["numTotalTests", "numPassedTests", "numFailedTests", "numPendingTests", "numTodoTests"]) {
    if (!Number.isInteger(report[k])) {
      cleanup();
      fail(`vitest JSON output cannot be read: "${k}" is ${JSON.stringify(report[k])}, not an integer`);
    }
  }

  const perFile = new Map();
  // QUEUE #280: vitest's JSON reporter carries `startTime`/`endTime` (ms
  // epoch) per file alongside the assertion results -- the wall time the
  // shard-by-runtime planner (computeShardBins, above) needs. Missing/non-finite
  // on either field (a stub reporter, a future vitest shape change) degrades
  // to "no timing for this file" rather than a fail: timing is an
  // OPTIMIZATION computeShardBins already falls back from, never a
  // correctness input.
  const perFileTime = new Map();
  for (const r of report.testResults ?? []) {
    const rel = relative(root, r.name).split(sep).join("/");
    perFile.set(rel, (r.assertionResults ?? []).length);
    const start = Number(r.startTime);
    const end = Number(r.endTime);
    if (Number.isFinite(start) && Number.isFinite(end) && end >= start) perFileTime.set(rel, end - start);
  }
  const ran = new Set(perFile.keys());
  const missing = testFiles.filter((f) => !ran.has(f));
  if (missing.length > 0) {
    cleanup();
    fail(`guard file(s) present on disk but produced no test results:\n${missing.map((f) => `  ${f}`).join("\n")}`);
  }

  cleanup();
  return {
    total: report.numTotalTests,
    passed: report.numPassedTests,
    failed: report.numFailedTests,
    pending: report.numPendingTests,
    todo: report.numTodoTests,
    perFile,
    perFileTime,
  };
}

// Real-red-test checks that need only the run just made (no other shard's
// data) — kept immediate per shard so a genuinely failing guard does not wait
// for `--merge` to surface. The checks that DO need the whole population
// (missing-file across every protected file, per-file counts, the minTests
// floor) are exactly the ones `--shard` defers to `--merge` (QUEUE #266 D1).
function checkNoRedTests(failed, pending, todo, label) {
  if (failed > 0) fail(`${failed} invariant test(s) failing${label}`);
  if (pending > 0) fail(`${pending} invariant test(s) skipped — skipping a guard is a failure${label}`);
  if (todo > 0) fail(`${todo} invariant test(s) marked todo — a todo guard is not a guard${label}`);
}

// QUEUE #552 (B23-C4): a "tests" key outside the population is TWO different facts. Absent from
// disk -> a stale entry, and removal is the right advice. Present on disk -> a SCOPE fact (the
// file is not in this run's guard population), and "remove ... no such guard test file" was a
// false statement that told the reader to delete a live pin. Both still fail the run; only the
// removal advice is gated on the file's real absence.
// TOOLING-EXTRACT-BEGIN staleTestsEntryLines
function staleTestsEntryLines(testKeys, population, exists) {
  const lines = [];
  for (const f of testKeys) {
    if (population.includes(f)) continue;
    lines.push(
      exists(f)
        ? `    ${JSON.stringify(f)} — pinned in "tests" but not in this run's guard population (the file EXISTS: do NOT remove it; it is outside the protected population, or ran elsewhere)`
        : `    remove ${JSON.stringify(f)} — no such guard test file`,
    );
  }
  return lines;
}
// TOOLING-EXTRACT-END staleTestsEntryLines

// The population / per-file-count / floor checks, run against a POPULATION
// and a PERFILE map that between them are the union across however many
// spawns (one, in `full` mode; N, merged, in `--merge` mode) produced the
// results.
function checkPopulationAndFloor(population, perFile, total) {
  const ran = new Set(perFile.keys());
  const missing = population.filter((f) => !ran.has(f));
  if (missing.length > 0) {
    fail(`guard file(s) in the protected population produced no test results in any shard:\n${missing.map((f) => `  ${f}`).join("\n")}`);
  }
  const countProblems = [];
  for (const f of population) {
    const expected = manifest.tests[f];
    const actual = perFile.get(f) ?? 0;
    if (!Number.isInteger(expected)) {
      countProblems.push(`    ${JSON.stringify(f)}: ${actual},   // MISSING from manifest "tests"`);
    } else if (expected !== actual) {
      countProblems.push(`    ${JSON.stringify(f)}: ${actual},   // was pinned at ${expected}`);
    }
  }
  // A stale "tests" entry for a file that no longer exists is also a problem
  // — it would let a deleted file's count survive in the manifest.
  countProblems.push(...staleTestsEntryLines(Object.keys(manifest.tests), population, (f) => existsSync(abs(f))));
  if (countProblems.length > 0) {
    deferPin(
      `per-file test count mismatch (manifest "tests" map):\n${countProblems.join("\n")}\n` +
        `Update the "tests" map in fixtures/invariant-manifest.json to match, in the same reviewed change. ` +
        `Per-file counts make a redistribution of tests between files visible; a global floor alone does not.`,
    );
  }
  if (total < manifest.minTests) {
    deferPin(
      `invariant suite ran ${total} tests but manifest requires >= ${manifest.minTests}. ` +
        `A guard was deleted or filtered. If removal was intentional, lower minTests in a reviewed change ` +
        `(the current true total is ${total}).`,
    );
  }
  // The floor should track the true total so a future deletion is visible
  // against it, not just against per-file counts. Advisory only — never fail
  // for being ABOVE the floor, but tell the developer the value to set.
  if (total > manifest.minTests) {
    console.error(
      `note: invariant suite ran ${total} tests; manifest minTests is ${manifest.minTests}. ` +
        `Consider raising "minTests" to ${total} so a later deletion trips the floor too (never lower it).`,
    );
  }
}

function judgeRuledEveryRow(bdir, d, judge) {

// judge report: `JUDGED-BY: <repo-relative path to a REPORT.md> sha256=<64-hex>`. The file counts only if
// EVERY line (blank lines aside) names a REPORT.md inside --root whose sha256 matches, whose sibling
// PROMPT.md CARRIES room type `judge` (promptType — no dir-name fallback: the PROMPT must say it), and
// whose SCOPE PACKET names `<batch>` as a whole token. The batch DIRECTORY name, not the bare id: every
// RJ-550 judge's scope quotes D-0694, which names all six bare ids. Any failing line refuses the file.
// sha256 by the system hasher, never node:crypto — the close-run.mjs hashFile shape ([single-hasher]).
function hashFile(file) {
  try {
    return execFileSync("sha256sum", [file], { encoding: "utf8" }).trim().split(/\s+/)[0];
  } catch (e) {
    if (e.code !== "ENOENT") throw e;
    return execFileSync("shasum", ["-a", "256", file], { encoding: "utf8" }).trim().split(/\s+/)[0];
  }
}

  }
  return null;
}

#!/usr/bin/env node
/**
 * Structural guard for org/01-current/QUEUE.md, the ordered work queue.
 *
 * WHY THIS EXISTS, and what it cost to find out. Until 2026-09-14 the queue had
 * seven columns and no status column. `mode` was meant to carry the dispatch
 * (ADR-0005 §2: serial / org / design / code) and it had silently absorbed the
 * state instead, in free prose, one improvisation at a time:
 * `**RETIRED 2026-09-11** — fixed S-006 as CL:A-01 (swept RUN-0014)`,
 * `org — **C-058 DONE, C-057 stopped as a docket question (S-026 REBUILD)**`,
 * `**DONE 2026-09-08 (S-005a; items 1–5, 8)** — org (scripts)`,
 * `proposed — needs Jason`, `3 of 4 done`, `done except the 5 copy rulings`.
 * Every one of those is meaningful to a person reading that row and useless to
 * anything counting. Asked how many of the 208 rows were complete, the honest
 * answer was "somewhere between 20 and 124" — the low number if you trust only
 * rows whose cell begins with DONE, the high one if you count every cell with
 * the substring "done" anywhere in it. That is not an answer, and the failure is
 * invisible until somebody asks a counting question, which is precisely why it
 * survived 208 rows and roughly forty batches.
 *
 * So the fix was a real `status` column with a closed vocabulary, and this
 * script is the half that stops it rotting back. A vocabulary nothing enforces
 * is a naming convention, and this org already knows what those become: the
 * mode column WAS one.
 *
 * The checks, each fails closed and each names the offending row number and
 * prints the row, because a validator that says "invalid" without saying where
 * is a second problem rather than a fix:
 *
 *  1. ONE TABLE, CONTIGUOUS. Every row line must sit in one unbroken block
 *     under one `| # | status | ... |` header. This is not pedantry: before the
 *     rewrite, two closure paragraphs ("Docket rows touched…", "Last re-sort…")
 *     sat between rows 16 and 17 with no second header after them, so 192 of the
 *     208 rows were not in a markdown table at all — they rendered as plain
 *     paragraphs full of pipe characters, and had done for weeks. Nobody noticed
 *     because everybody reads this file as source, not as rendered output. A
 *     narrative paragraph may live above or below the table; it may not live
 *     inside it.
 *  2. COLUMN COUNT. Exactly 8 cells per row, splitting on UNESCAPED pipes.
 *     Six rows were already malformed under the old seven-column layout: #21,
 *     #24 and #92 carried an EXTRA cell (a status narrative wedged in beside
 *     `mode` — the defect this column replaces, caught in the act), #34 was
 *     MISSING `mode` entirely, and #38 and #41 had a literal `|` inside a code
 *     span (`<scope|finding text>`, `evidence: executed|read|inferred`) which
 *     markdown reads as a column break. The last two are why the split here is
 *     on `(?<!\\)\|` and why those pipes are now written `\|`.
 *  3. VOCABULARY. The status cell starts with exactly one of the seven tokens.
 *     Nothing outside them, ever — the point of a controlled vocabulary is that
 *     the eighth token is a bug and not a judgement call.
 *  4. FORMAT PER TOKEN, enforced and not suggested. `done` carries a date and
 *     the batch or run that did it, because a `done` nobody can trace to a
 *     closure is a claim, not a record — this rewrite found two DONE rows whose
 *     stated effect could not be corroborated anywhere in the tree. `retired`
 *     carries a date and a reason. `partial` names what REMAINS, after
 *     `remains:`, because "3 of 4 done" without the fourth named is how a limb
 *     goes missing for a month. `blocked` names its blocker AND has a non-empty
 *     `blocked on` cell. `ruling` says what Jason has to decide: these are the
 *     rows that sit forever if nobody surfaces them, so they are the ones worth
 *     making legible.
 *  5. IDENTITY. `#` values unique and ascending. They are how Jason picks a row
 *     ("Jason picks by number"), so a duplicate is an ambiguous instruction. The
 *     ascending part is a weaker property than it looks — before the rewrite the
 *     ids ran 1..53, then 26, 29, 27, 28, 22, then 54..208, five rows displaced
 *     by an old re-sort. Priority lives in the order of the queue as a document;
 *     identity lives in the number. Letting the two fight makes the number
 *     useless as an address.
 *  6. TALLY. Printed on every run, green or red. `node scripts/check-queue.mjs`
 *     is the answer to "how many are done" from now on; the previous answer was
 *     a person reading 208 cells and guessing.
 *
 * Usage:  node scripts/check-queue.mjs [--file <path>] [--open] [--row N]
 *           [--status <token>] [--count] [--min-rows N]
 * Exit 0 when every check holds, 1 on any violation. New flags never change the
 * structural checks above -- they validate first, same as no flags, and only then
 * print their view: --open lists every non-done/retired row; --row N prints one row's
 * 8 cells labelled, full text; --status <token> filters to that vocabulary token;
 * --count prints the tally only (drops the "check-queue: OK" line).
 */
import { readFileSync, existsSync } from "node:fs";

const argv = process.argv.slice(2);
const fileIdx = argv.indexOf("--file");
const FILE = fileIdx >= 0 ? argv[fileIdx + 1] : "org/01-current/QUEUE.md";
const openFlag = argv.includes("--open");
const countFlag = argv.includes("--count");
const rowArgIdx = argv.indexOf("--row");
const ROW_ARG = rowArgIdx >= 0 ? argv[rowArgIdx + 1] : null;
const statusArgIdx = argv.indexOf("--status");
const STATUS_ARG = statusArgIdx >= 0 ? argv[statusArgIdx + 1] : null;
const ownerlessFlag = argv.includes("--ownerless");
// QUEUE #448. A structural guard whose input can be EMPTY passes trivially: zero rows means
// zero duplicates, zero bad statuses and zero bad columns, and the script prints OK. The
// zero-row case is already refused below, but ONE row is just as vacuous for IDENTITY (a
// single id cannot collide with itself) and for ASCENT, and a caller that has silently been
// handed a truncated file — a half-written candidate, the wrong tree, a `--file` typo — gets
// a green. `--min-rows N` makes the population itself an assertion, so a harness can pin the
// size it believes it is checking and a shrunken file is a failure rather than a pass.
const minRowsIdx = argv.indexOf("--min-rows");
const MIN_ROWS = minRowsIdx >= 0 ? Number(argv[minRowsIdx + 1]) : 1;
if (!Number.isInteger(MIN_ROWS) || MIN_ROWS < 1) {
  console.error(`check-queue: --min-rows must be a positive integer; got ${JSON.stringify(argv[minRowsIdx + 1])}.`);
  process.exit(1);
}

const HEADER = "| # | status | mode | item | why now | shape | blocked on | claimed_by |";
const COLUMNS = 8;
const TOKENS = ["open", "wip", "partial", "done", "retired", "ruling", "blocked"];
// The mode taxonomy is DERIVED from what the rows actually use, not invented:
// `org`/`org (gate)`/`org (guard)`/…, `code`/`code (T1)`/`code (T2)`/`code (deps)`,
// `design`/`design → serial`/`design → code`, `serial`, and `—` for the rows that
// record no dispatch. Only the head word is checked, so a new parenthetical
// qualifier needs no edit here; a new DISPATCH does, deliberately.
// QUEUE #467 (B23-D3): a PACKAGE or AREA is never a head. #417-#420 once read
// `trust-core (privacy)` / `docs (legal)` and kept this guard red; they now read
// `code (trust-core: privacy)` / `code (docs: legal)`. Widening this list to take package
// names would make `mode` carry two different things again — the defect this column was
// rebuilt to end. Defended by tests/tooling/check-queue-mode.test.ts.
const MODE_HEADS = ["org", "code", "design", "serial", "—"];

const failures = [];
const fail = (row, msg) => failures.push({ row, msg });

if (!existsSync(FILE)) {
  console.error(`check-queue: no such file: ${FILE}`);
  process.exit(1);
}
const lines = readFileSync(FILE, "utf8").split("\n");

// ------------------------------------------------------ 1. ONE TABLE, CONTIGUOUS
const headerIdxs = lines.map((l, i) => (l.trim() === HEADER ? i : -1)).filter((i) => i >= 0);
if (headerIdxs.length !== 1) {
  console.error(
    headerIdxs.length === 0
      ? `check-queue: no queue table header found. Expected exactly this line:\n  ${HEADER}`
      : `check-queue: ${headerIdxs.length} queue table headers found (lines ${headerIdxs.map((i) => i + 1).join(", ")}); expected exactly 1.`,
  );
  process.exit(1);
}
const headerLine = headerIdxs[0];
if (!/^\|[-\s|:]+\|$/.test(lines[headerLine + 1] ?? "")) {
  console.error(`check-queue: line ${headerLine + 2} is not a markdown separator under the header:\n  ${lines[headerLine + 1]}`);
  process.exit(1);
}

const isRowLine = (l) => /^\|\s*\d+\s*\|/.test(l);
const rowIdxs = lines.map((l, i) => (isRowLine(l) ? i : -1)).filter((i) => i >= 0);
if (rowIdxs.length === 0) {
  console.error("check-queue: the table has no rows.");
  process.exit(1);
}
const first = rowIdxs[0];
const last = rowIdxs[rowIdxs.length - 1];
// QUEUE #448. These two were `process.exit(1)` and that is how #448 happened. The contiguity
// check is the FIRST thing that runs and it used to exit the process the moment it found a
// stray line — so on a file with one interrupting paragraph, checks 2 through 7 never ran AT
// ALL. Identity (5) is one of them. The result: this script carried a working duplicate-id
// refusal for weeks while three ids (#410, #411, #412) sat duplicated in the file it guards,
// because a narrative paragraph eleven lines from the end exited before identity was reached.
// A guard that stops at the first problem is a guard that reports the CHEAPEST problem and
// hides every expensive one behind it. So these accumulate now: the run still FAILS, and it
// fails saying everything it found.
if (first !== headerLine + 2) {
  fail("table", `the first row is at line ${first + 1} but the separator is at line ${headerLine + 2} — rows must start immediately under the header.`);
}
for (let i = first; i <= last; i++) {
  if (!isRowLine(lines[i])) {
    fail("table", `line ${i + 1} interrupts the table between rows — a non-row line inside the block means every row after it renders as a paragraph, not a table row:\n     ${lines[i].slice(0, 160)}`);
  }
}
// A row line OUTSIDE the contiguous block is the shape #448 actually took: three rows appended
// below the closing paragraphs, invisible to a reader and to markdown, and carrying ids that
// were already in use. `rowIdxs` spans the WHOLE file precisely so identity sees them.
let blockEnd = headerLine + 1;
while (isRowLine(lines[blockEnd + 1] ?? "")) blockEnd++;
for (const i of rowIdxs) {
  if (i > blockEnd) fail("table", `line ${i + 1} is a table row sitting OUTSIDE the table block (which ends at line ${blockEnd + 1}). It renders as a paragraph, no counter of this file sees it, and it still occupies an id.`);
}

// -------------------------------------------------------------- 2..5. per row
// Markdown treats `\|` as a literal pipe inside a cell, so the split has to as well.
const splitCells = (line) => line.split(/(?<!\\)\|/).slice(1, -1).map((s) => s.trim());
const EMPTY = new Set(["", "—", "-", "none", "n/a"]);
const DATE = /\d{4}-\d{2}-\d{2}/;

const rows = [];
for (const i of rowIdxs) {
  const line = lines[i];
  const cells = splitCells(line);
  const lineNo = i + 1;
  const num = Number((cells[0] ?? "").trim());

  // 2. COLUMN COUNT
  if (cells.length !== COLUMNS) {
    fail(num, `line ${lineNo}: ${cells.length} columns, expected ${COLUMNS} (| # | status | mode | item | why now | shape | blocked on | claimed_by |). A literal pipe inside a cell must be written \\|.`);
    rows.push({ num, lineNo, line, cells, bad: true });
    continue;
  }
  const [, status, mode, item, , , blockedOn] = cells;
  rows.push({ num, lineNo, line, cells, status, mode });

  // 3. VOCABULARY
  const token = status.split(/\s+/)[0];
  if (!TOKENS.includes(token)) {
    fail(num, `line ${lineNo}: status starts with "${token || "(empty)"}", which is not in the vocabulary (${TOKENS.join(", ")}).`);
    continue;
  }

  // 4. FORMAT PER TOKEN
  if (token === "done") {
    const m = /^done (\d{4}-\d{2}-\d{2})(?: (.+))?$/.exec(status);
    if (!m) {
      fail(num, `line ${lineNo}: "done" must carry a date — "done YYYY-MM-DD <batch/run>". Got: ${JSON.stringify(status)}`);
    } else {
      const ref = (m[2] ?? "").split(" — ")[0].trim();
      if (ref === "" || ref.startsWith("—")) {
        fail(num, `line ${lineNo}: "done" must name the batch or run that did it after the date — "done ${m[1]} S-026 DOMGATE". Got: ${JSON.stringify(status)}`);
      }
    }
  } else if (token === "retired") {
    if (!/^retired \d{4}-\d{2}-\d{2} — \S.*$/.test(status)) {
      fail(num, `line ${lineNo}: "retired" must carry a date and a reason — "retired YYYY-MM-DD — <why>". Got: ${JSON.stringify(status)}`);
    }
  } else if (token === "partial") {
    if (!/^partial — \S.*$/.test(status)) {
      fail(num, `line ${lineNo}: "partial" must carry a note — "partial — <what landed>; remains: <what does not>". Got: ${JSON.stringify(status)}`);
    } else if (!/remains:/.test(status)) {
      fail(num, `line ${lineNo}: "partial" must name what REMAINS, after "remains:". A partial with no remainder named is how a limb goes missing. Got: ${JSON.stringify(status)}`);
    }
  } else if (token === "blocked") {
    if (!/^blocked — \S.*$/.test(status)) {
      fail(num, `line ${lineNo}: "blocked" must name its blocker — "blocked — <what is in the way>". Got: ${JSON.stringify(status)}`);
    }
    if (EMPTY.has(blockedOn.toLowerCase())) {
      fail(num, `line ${lineNo}: "blocked" requires a non-empty "blocked on" cell; got ${JSON.stringify(blockedOn)}.`);
    }
  } else if (token === "ruling") {
    if (!/^ruling — \S.*$/.test(status)) {
      fail(num, `line ${lineNo}: "ruling" must say what Jason has to decide — "ruling — <the decision>". These are the rows that sit forever unless somebody surfaces them. Got: ${JSON.stringify(status)}`);
    }
  } else if (!new RegExp(`^${token}(?: — \\S.*)?$`).test(status)) {
    fail(num, `line ${lineNo}: anything after "${token}" must be an em-dash note — "${token} — <note>". Got: ${JSON.stringify(status)}`);
  }

  // The old prose shape resurfacing: a SHOUTED closure verb next to a date, inside
  // a row that claims to be unstarted. That exact pair — `**DONE 2026-09-12 (S-026
  // DOMGATE)**`, `**RETIRED 2026-09-11**`, `FIXED 2026-09-12 (S-020 HARNESS)` — is
  // how the state got into the mode column in the first place. Deliberately keyed on
  // the UPPERCASE spelling: ordinary lower-case narrative ("the blocker was
  // discharged when RUN-0012 closed 2026-09-07") is a legitimate note on an open row
  // and must not trip this.
  if ((token === "open" || token === "wip") && DATE.test(status) && /\b(DONE|FIXED|CLOSED|RETIRED|BUILT|SHIPPED)\b/.test(status)) {
    fail(num, `line ${lineNo}: an "${token}" row carrying a shouted closure verb and a date is a closure recorded as prose — use done/retired/partial. Got: ${JSON.stringify(status)}`);
  }

  // mode reverted to the dispatch it was always for.
  const head = mode.split(/[\s(]/)[0];
  if (mode === "" || !MODE_HEADS.includes(head)) {
    fail(num, `line ${lineNo}: mode is ${JSON.stringify(mode)}; expected one of ${MODE_HEADS.join(" / ")} (optionally qualified, e.g. "org (gate)", "code (T2)", "design → serial"). A package or area goes INSIDE the qualifier — "code (trust-core: privacy)", never "trust-core (privacy)" (QUEUE #467).`);
  }
  if (item === "") fail(num, `line ${lineNo}: item cell is empty.`);
}

// ------------------------------------------------------------------ 5. IDENTITY
const seen = new Map();
for (const r of rows) {
  if (!Number.isInteger(r.num) || r.num < 1) {
    fail(r.num, `line ${r.lineNo}: "#" is not a positive integer.`);
    continue;
  }
  if (seen.has(r.num)) fail(r.num, `line ${r.lineNo}: duplicate # ${r.num} (first seen at line ${seen.get(r.num)}). Jason picks rows by number; a duplicate is an ambiguous instruction.`);
  else seen.set(r.num, r.lineNo);
}
for (let i = 1; i < rows.length; i++) {
  if (rows[i].num <= rows[i - 1].num) {
    fail(rows[i].num, `line ${rows[i].lineNo}: # ${rows[i].num} is not greater than the preceding # ${rows[i - 1].num} — ids must ascend down the file.`);
  }
}

// All the "new modes" below and the LIVENESS/ownerless checks share these -- defined once,
// ahead of LIVENESS, so LIVENESS's `tokenByNum` and the --ownerless report (with the TALLY,
// before the failure exit) can both use `rowToken` instead of re-deriving it.
const stripMd = (s) =>
  s
    .replace(/\*\*(.*?)\*\*/g, "$1")
    .replace(/`([^`]*)`/g, "$1")
    .replace(/\[([^\]]*)\]\([^)]*\)/g, "$1")
    .trim();
const rowToken = (r) => (r.status ?? "").split(/\s+/)[0];
const LABELS = ["#", "status", "mode", "item", "why now", "shape", "blocked on", "claimed_by"];
const listLine = (r) => `${r.num} | ${rowToken(r)} | ${r.mode} | ${stripMd(r.cells[3]).slice(0, 140)} | ${r.cells[6]}`;

// ------------------------------------------------------------- 6. LIVENESS (QUEUE #212/#133)
// A `blocked on` cell can go stale two ways this org has actually hit: it names ITSELF (#117:
// "QUEUE #117") or it names another row that has since gone `done` -- the blocker closed and
// nobody removed the reference, so the row now sits behind a ghost blocker forever. `docket #N`
// is a DIFFERENT id space (EXCEPTION-DOCKET.md, not this table) and is never a row reference --
// stripped before the bare `#N` scan below, including a docket chain ("docket #3/#4").
// #294 (Wave B): a `done`/`retired`/`ruling` row can carry a leftover "blocked on" reference
// (nobody clears it once the row itself closes) -- that reference is stale paperwork, not a
// live problem, and flagging it over-reports: LIVENESS only means something for a row still
// waiting on its blocker, so it applies only to rows whose OWN status is open/partial/blocked.
const LIVE_STATUS = new Set(["open", "partial", "blocked"]);
const tokenByNum = new Map(rows.filter((r) => !r.bad).map((r) => [r.num, rowToken(r)]));
const REF_RE = /docket\s*#\d+(?:\s*[/,]\s*#\d+)*|#(\d+)/gi;
for (const r of rows) {
  if (r.bad) continue;
  if (!LIVE_STATUS.has(rowToken(r))) continue;
  const blockedOn = r.cells[6] ?? "";
  if (EMPTY.has(blockedOn.toLowerCase())) continue;
  for (const m of blockedOn.matchAll(REF_RE)) {
    if (!m[1]) continue; // consumed as part of a docket reference, not a row number
    const ref = Number(m[1]);
    if (ref === r.num) {
      fail(r.num, `line ${r.lineNo}: "blocked on" names itself (#${ref}) — a row cannot be blocked on its own number.`);
    } else if (tokenByNum.get(ref) === "done") {
      fail(r.num, `line ${r.lineNo}: "blocked on" names #${ref}, which is done — the blocker is discharged and the reference is stale.`);
    }
  }
}

// ---------------------------------------------------------------------- 7. TALLY
const tally = Object.fromEntries(TOKENS.map((t) => [t, 0]));
let unknown = 0;
for (const r of rows) {
  const t = (r.status ?? "").split(/\s+/)[0];
  if (TOKENS.includes(t)) tally[t]++;
  else unknown++;
}
console.log(`queue: ${FILE} — ${rows.length} rows`);
console.log(`rows-examined: ${rows.length}`);
if (rows.length < MIN_ROWS) {
  fail("table", `${rows.length} rows examined, which is below the --min-rows floor of ${MIN_ROWS}. A guard run over an empty or truncated queue proves nothing: with no rows there are no duplicate ids to find, no statuses to reject and no columns to count, and the run would otherwise print OK.`);
}
for (const t of TOKENS) console.log(`  ${t.padEnd(8)} ${String(tally[t]).padStart(3)}`);
if (unknown > 0) console.log(`  ${"(none)".padEnd(8)} ${String(unknown).padStart(3)}  rows whose status is outside the vocabulary`);

// --ownerless (QUEUE #212/#133): print-only, never a violation -- a row that has been ROUTED
// (its status moved past `open`, or it is out of the `ruling` state that belongs to Jason, not
// a room) yet carries no `claimed_by` is the org's own concurrency marker (ADR-0005 §3.4: "the
// account that started it; the other account skips it") going unset on work that has, by its
// own status, already started. Printed here, with the tally, because it must surface even on a
// file the structural checks above are about to fail on -- that is exactly when an operator
// most wants the diagnostic.
if (ownerlessFlag) {
  const ownerless = rows.filter((r) => !r.bad).filter((r) => {
    const t = rowToken(r);
    if (t === "open" || t === "ruling") return false;
    return EMPTY.has((r.cells[7] ?? "").toLowerCase());
  });
  console.log(`\n-- ownerless (routed, no claimed_by): ${ownerless.length}`);
  for (const r of ownerless) console.log(listLine(r));
}

if (failures.length > 0) {
  console.error(`\n${failures.length} violation(s):`);
  const byRow = new Map();
  for (const f of failures) {
    if (!byRow.has(f.row)) byRow.set(f.row, []);
    byRow.get(f.row).push(f.msg);
  }
  for (const [row, msgs] of byRow) {
    const r = rows.find((x) => x.num === row);
    console.error(`\n-- row #${row}`);
    for (const m of msgs) console.error(`   ${m}`);
    if (r) console.error(`   ${r.line.length > 300 ? `${r.line.slice(0, 300)} …[+${r.line.length - 300} chars]` : r.line}`);
  }
  console.error(`\ncheck-queue: FAILED (${failures.length} violation(s) over ${rows.length} rows)`);
  process.exit(1);
}
if (!countFlag) {
  console.log(`check-queue: OK — ${rows.length} rows, ${COLUMNS} columns each, every status in the vocabulary, ids unique and ascending.`);
}

// -------------------------------------------------------------- new modes
// All reuse the already-validated `rows`/`cells` from above (D3: same escaped-pipe split
// everywhere, no re-parsing). --row/--status/--open only run once the file is structurally
// sound -- a violation above already exited 1 for every one of those. --ownerless is printed
// earlier, with the TALLY (see above) -- it is diagnostic, not a violation, and unlike the
// other three it is useful precisely when the file is NOT clean, so it must not wait behind
// the failure exit.

if (ROW_ARG !== null) {
  const n = Number(ROW_ARG);
  const r = rows.find((x) => x.num === n);
  if (!r) {
    console.error(`check-queue: --row ${ROW_ARG}: no such row number.`);
    process.exit(1);
  }
  console.log(`\n-- row #${r.num}`);
  for (let i = 0; i < COLUMNS; i++) console.log(`${LABELS[i]}: ${r.cells[i]}`);
}

if (STATUS_ARG !== null) {
  if (!TOKENS.includes(STATUS_ARG)) {
    console.error(`check-queue: --status ${STATUS_ARG}: not in the vocabulary (${TOKENS.join(", ")}).`);
    process.exit(1);
  }
  console.log(`\n-- status: ${STATUS_ARG}`);
  for (const r of rows) if (rowToken(r) === STATUS_ARG) console.log(listLine(r));
}

if (openFlag) {
  console.log(`\n-- open (not done/retired)`);
  for (const r of rows) {
    const t = rowToken(r);
    if (t !== "done" && t !== "retired") console.log(listLine(r));
  }
}

process.exit(0);

#!/usr/bin/env node
/**
 * QUEUE #431 — a `path:NNNN` citation that points at the file it is written in.
 *
 * WHY THIS IS A CLASS AND NOT A TYPO. An absolute line number is a coordinate in a file, and a
 * comment is text IN that file. Writing the comment moves the coordinate. So a comment that
 * cites a line of its own file is falsified by the act of writing it, before anyone has had a
 * chance to be wrong about anything — it is not stale later, it is wrong on arrival. The filed
 * case is the org's own claims-hygiene correction: a comment in
 * `services/trust-core/src/domain-claims.ts` was rewritten three times (NA-W309-17), and its
 * third attempt cited that file at a bare line number, which was TRUE when the author read it
 * and was pushed three lines down
 * by the three lines that carried the citation.
 *
 * ANTI-024 clause 3 already says a comment is a CLAIM and never evidence, and that a false
 * comment is filed as a CLAIMS HYGIENE finding. This is the sub-species it does not name: the
 * claim whose truth value is destroyed by its own insertion. It needs no reader to go stale and
 * no edit to falsify, so "check it against the code" — clause 2's remedy — cannot save it. The
 * only correction is to stop writing the coordinate: name the STATEMENT, the symbol, or the
 * clause, which survives an edit above it.
 *
 * WHAT THIS SCANS. In a SOURCE file, only text inside a comment (`//` to end of line, and
 * `/* … *\/` blocks) — code that computes a string like `${f}:${n}` is not a claim. In a
 * MARKDOWN file, the whole file: markdown IS commentary, there is no executable half to
 * exclude, and the record commits this defect as readily as the source does
 * (`RUN-ORDER-2026-09-21.md`'s item 0.3 cited a stage number the edit adding stages had
 * already moved).
 *
 * TWO CLASSES, reported separately, because they carry different certainty:
 *   SELF   — `<path>:<line>` where <path> resolves to the file the citation lives in. This is
 *            unambiguous: the author named the file and it is this one.
 *   BARE   — a naked `:<line>` inside a comment with no path in front of it. This is the filed
 *            case's exact spelling — a colon and a number, no path — and it is self-referential by default: a reader
 *            has nothing else to resolve it against — but it is a judgement call when another
 *            file is named in the same sentence, so it is counted apart and never merged in.
 *
 * NOT FLAGGED, deliberately: a RANGE (`:106-110`) is still rot, but it is usually written when
 * quoting another file and the false-positive rate is not worth it here; a citation of ANOTHER
 * file is out of scope (it rots too, but not by construction — somebody else has to edit that
 * file first, which is ordinary staleness and not this class).
 *
 * Usage: node scripts/check-self-citations.mjs [--root <dir>] [--json] [--quiet]
 *                                              [--min-files N] [--include <glob-ish substring>]
 * Exit 0 when the population is empty, 1 when any citation is found, 2 on a refusal.
 *
 * `--min-files N` exists for the same reason `check-queue.mjs --min-rows` does: a scanner that
 * walked zero files finds zero defects and prints a clean bill of health. `--root` defaulting
 * to cwd is the org's own repeat defect (QUEUE #432/#434), so the resolved root and the file
 * count are printed on EVERY run, clean or not.
 */

// ---------------------------------------------------------------- comment extraction
// Returns, per line, the substring of that line that is inside a comment. A source line with
// no comment yields "". Strings are tracked only well enough to keep a `//` inside a quoted
// string or a template literal from opening a comment; that is the only false-open that
// matters here, and over-caution costs a missed finding rather than a false one.
function commentText(src) {
  const lines = src.split("\n");
  const out = new Array(lines.length).fill("");
  let inBlock = false;
  for (let i = 0; i < lines.length; i++) {
    const l = lines[i];
    let acc = "", j = 0, q = null;
    while (j < l.length) {
      if (inBlock) {
        const end = l.indexOf("*/", j);
        if (end < 0) { acc += l.slice(j); j = l.length; }
        else { acc += l.slice(j, end); j = end + 2; inBlock = false; }
        continue;
      }
      const c = l[j];
      if (q) { if (c === "\\") { j += 2; continue; } if (c === q) q = null; j++; continue; }
      if (c === '"' || c === "'" || c === "`") { q = c; j++; continue; }
      if (c === "/" && l[j + 1] === "/") { acc += l.slice(j + 2); break; }
      if (c === "/" && l[j + 1] === "*") { inBlock = true; j += 2; continue; }
      j++;
    }
    out[i] = acc;
  }
  return out;
}

// -------------------------------------------------------------------- the two patterns
// A path, then `:`, then a line number that is NOT the start of a range and NOT followed by
// another digit. The path must carry a recognised extension so that `http://host:8080`,
// `12:30` and `INV-061:3` cannot match.
const SELF_RE = new RegExp(String.raw`(?<![\w./-])((?:[\w.@-]+[/\\])*[\w.@-]+\.(?:${CITED_EXT}))\s*:\s*(\d{1,5})(?![\d\-–])`, "g");
// A naked `:NNNN` — preceded by whitespace, `(`, `` ` ``, `[` or the start of the text, so that
// `foo.ts:12` (handled above) and `2026-09-21:1` style noise do not double-count.
const BARE_RE = /(?<=^|[\s(`[])[:：]\s*(\d{2,5})(?![\d\-–:])/g;

// Does a cited path name the file the citation is written in?
function isSelf(cited, relPath) {
  const norm = (s) => s.split(/[\\/]/).filter(Boolean).join("/");
  const c = norm(cited), f = norm(relPath);
  if (c === f) return true;
  if (f.endsWith(`/${c}`)) return true;                       // a suffix: `domain-claims.ts` in `services/.../domain-claims.ts`
  const rel = norm(relative(ROOT, resolve(dirname(join(ROOT, relPath)), cited)));
  return rel === f;                                           // resolved against the citing file's own directory
}

  if (f.exempt) exempt.files++;

  const bareOut = f.exempt ? { push: () => exempt.bare++ } : bareHits;

    // file you are reading — there is nothing else it could mean, which is exactly why the filed
    // case wrote a bare colon-and-number and not a path. In MARKDOWN it means the opposite: prose names a path
    // once and then continues with further bare colon-numbers, and reading those as self-references produced
    // 70 hits in org/01-current alone, every one of them pointing at some other file. A detector
    // that has to be argued with is not one, so the markdown half is dropped rather than tuned.

#!/usr/bin/env node
// clock-forward-scan.mjs — which tests turn red when the calendar moves on? (QUEUE #890, FAMILY-005)
//
// EXECUTES, never guesses: runs the chosen vitest files with the process clock moved forward N days
// (scripts/clock-forward-shim.mjs, preloaded through NODE_OPTIONS=--import) and reports every file that
// is GREEN at +0 and RED at +N — a test that will go red on that date with no code change. The static
// horizon lookout is scripts/s4-calendar-scan.mjs; this is its executed counterpart.
//
//   node scripts/clock-forward-scan.mjs [--days 30,90,365] [--state <file.json>] [--budget-ms 35000]
//        [--chunk 12] [--chunk-timeout-ms <ms>] [--only-red-at <N>] [--summary] [--list] <file-or-dir> ...
//
// --days        offsets to run; +0 (the baseline) is always added — a file red at +0 is BASELINE RED,
//               never reported as clock-bound.
// --state       JSON file accumulating per-offset results across invocations, so a long population can
//               be run in several short calls; files already recorded for an offset are skipped.
// --budget-ms   stop starting new chunks after this much wall time (default: no limit).
// --chunk-timeout-ms  kill one chunk after this long (default: a third of --budget-ms when a budget is set, else
//               none). A chunk killed by it is split; a single file killed by it is recorded `timeout`
//               (reported NOT-REACHED) and the pass moves on (QUEUE #913): without it one chunk that cannot
//               finish took the whole budget, recorded nothing, and every later pass restarted the same chunk.
//               A third, so a killed chunk and then its hung file alone both fit in one pass.
//               A `timeout` is retried ONCE, on the next pass over the same --state (a file that timed out only because
//               the machine was loaded is not NOT-REACHED for ever); if the retry times out too it stays `timeout`
//               for good (state.retried marks it) and only a hand edit of the state file clears it (I14-FX3, RGT-05).
//               A CRASH — vitest exits without writing a report (an unloadable config or --import, a spawn failure) —
//               is its own state, `crash`, recorded when a single file crashes (a multi-file chunk is split first, to
//               find the file); it is reported `NOT-REACHED <file> @+N (crash)`, counts as unreached (exit 3), and is
//               NOT retried: a crash is not a slow run (I14-FX3, I14-CLOCK-913-3).
//               Every flag that takes a number takes a positive WHOLE number: `--chunk-timeout-ms 1500.5` or `soon` is
//               refused with a usage message and exit 2, never a crash with exit 1 (1 means "a file is clock-bound").
// --only-red-at restrict the population to files the state records RED at offset N (run the widest
//               offset first, then classify its reds at the narrower ones).
// --summary     print the verdict from the state without running anything.
// --list        print the expanded file population and exit.
// Exit: 0 every file reached at every offset and none clock-bound; 1 at least one clock-bound file;
//       3 the run is incomplete (a file not yet reached, timed out, or crashed);
//       2 a usage error or the scan itself failed (never 1: that would read as a clock-bound file), including a
//         run that scanned ZERO files — an empty population, an --only-red-at that matched nothing, or every
//         file EXCLUDED by the vitest config — which says why on its own line (I28-GATECHECKS G3); and a complete run
//         that classified ZERO files against the clock because every scanned file is BASELINE-RED (I28-FX-GATE X2).
// Paths resolve against the repository root derived from this file (import.meta.url), not the cwd.

import { spawnSync } from "node:child_process";
import { existsSync, mkdtempSync, readFileSync, readdirSync, realpathSync, rmSync, statSync, writeFileSync } from "node:fs";
import { tmpdir } from "node:os";
import { dirname, join, relative, resolve } from "node:path";
import { fileURLToPath } from "node:url";

const ROOT = resolve(dirname(fileURLToPath(import.meta.url)), "..");
const SHIM = join(ROOT, "scripts", "clock-forward-shim.mjs");
const VITEST = join(ROOT, "node_modules", "vitest", "vitest.mjs");
const DAY_MS = 86_400_000;
const TEST_RE = /\.test\.(ts|tsx|mts|mjs|js)$/;
const SKIP_DIRS = new Set(["node_modules", ".git", "dist", ".wrangler", ".scratch", ".stryker-tmp"]);

export class UsageError extends Error {}
const wholeMs = (flag, v) => {
  const n = Number(v);
  if (String(v).trim() === "" || !Number.isInteger(n) || n < 1) throw new UsageError(`clock-forward-scan: ${flag} takes a positive whole number of milliseconds, got ${JSON.stringify(v)}`);
  return n;

export function parseArgs(argv) {
  const o = { days: [30, 90, 365], state: null, budgetMs: Infinity, chunk: 12, chunkTimeoutMs: null, onlyRedAt: null, summary: false, list: false, paths: [], root: ROOT };
  for (let i = 0; i < argv.length; i++) {
    const a = argv[i];
    const next = () => {
      const v = argv[++i];
      if (v === undefined) throw new UsageError(`clock-forward-scan: ${a} needs a value`);
      return v;
    };
    if (a === "--days") o.days = next().split(",").map((d) => Number(d));
    else if (a === "--state") o.state = resolve(next());
    else if (a === "--budget-ms") o.budgetMs = wholeMs(a, next());
    else if (a === "--chunk") o.chunk = Number(next());
    else if (a === "--chunk-timeout-ms") o.chunkTimeoutMs = wholeMs(a, next());
    else if (a === "--only-red-at") o.onlyRedAt = Number(next());
    else if (a === "--root") o.root = resolve(next());
    else if (a === "--summary") o.summary = true;
    else if (a === "--list") o.list = true;
    else if (a.startsWith("--")) throw new UsageError(`clock-forward-scan: unknown flag ${a}`);
    else o.paths.push(a);
  }
  for (const d of o.days) if (!Number.isInteger(d) || d < 1) throw new UsageError(`clock-forward-scan: --days takes positive whole days, got ${d}`);
  if (!Number.isInteger(o.chunk) || o.chunk < 1) throw new UsageError("clock-forward-scan: --chunk must be a positive integer");
  if (o.onlyRedAt !== null && !Number.isInteger(o.onlyRedAt)) throw new UsageError("clock-forward-scan: --only-red-at takes whole days");
  if (o.chunkTimeoutMs === null) o.chunkTimeoutMs = Number.isFinite(o.budgetMs) ? Math.max(1, Math.floor(o.budgetMs / 3)) : Infinity;
  if (!(o.chunkTimeoutMs > 0)) throw new UsageError("clock-forward-scan: --chunk-timeout-ms must be a positive number of milliseconds");
  return o;
}

export function expand(root, paths) {
  const out = new Set();
  const walk = (abs) => {
    const st = statSync(abs);
    if (st.isDirectory()) {
      for (const e of readdirSync(abs).sort()) if (!SKIP_DIRS.has(e)) walk(join(abs, e));
    } else if (TEST_RE.test(abs)) out.add(relative(root, abs));
  };
  for (const p of paths) {
    const abs = resolve(root, p);
    if (!existsSync(abs)) throw new UsageError(`clock-forward-scan: no such path: ${p}`);
    walk(abs);
  }
  return [...out].sort();
}

const emptyState = () => ({ version: 1, results: {} });
export function loadState(file) {
  if (!file || !existsSync(file)) return emptyState();
  const s = JSON.parse(readFileSync(file, "utf8"));
  if (s.version !== 1 || typeof s.results !== "object") throw new UsageError(`clock-forward-scan: ${file} is not a clock-forward state file`);
  return s;
}

/** The word runChunk returns when vitest exited WITHOUT a report (not killed, not "no test files"): an unloadable config or `--import`, a spawn failure, a truncated report. */
export const CRASH = "crash";

/** Runs one chunk at one offset; returns {file: "pass"|"fail"|"excluded"}, null when the run was killed (timeout), or CRASH when vitest exited without a usable report. */
export function runChunk(root, files, days, timeoutMs) {
  const dir = mkdtempSync(join(process.env.TMPDIR || tmpdir(), "cfs-"));
  const out = join(dir, "report.json");
  const env = { ...process.env, CLOCK_FORWARD_MS: String(days * DAY_MS), NODE_OPTIONS: `${process.env.NODE_OPTIONS ?? ""} --import ${SHIM}`.trim() };
  let rep;
  try {
    const r = spawnSync(process.execPath, [VITEST, "run", "--reporter=json", `--outputFile=${out}`, ...files], { cwd: root, env, timeout: Number.isFinite(timeoutMs) ? timeoutMs : undefined, encoding: "utf8", maxBuffer: 64 * 1024 * 1024 });
    // killed by OUR timeout (spawnSync marks it ETIMEDOUT even when vitest traps the signal and exits with a status): a kill, never a crash
    if (r.signal || r.error?.code === "ETIMEDOUT") return null;
    // every file of the chunk filtered out by the vitest config: nothing ran, nothing failed
    if (!existsSync(out) && /No test files found/.test(`${r.stdout}${r.stderr}`)) return Object.fromEntries(files.map((f) => [f, "excluded"]));
    if (!existsSync(out)) return CRASH;
    try {
      rep = JSON.parse(readFileSync(out, "utf8"));
    } catch {
      return CRASH;
    }
  } finally {
    rmSync(dir, { recursive: true, force: true }); // only the directory this call created
  }
  const res = {};
  // a file vitest did not report was filtered out by its config (e.g. test-fidelity/**): EXCLUDED, never a
  // pass and never a fail — a file that fails to load is still reported, with status "failed"
  for (const f of files) res[f] = "excluded";
  // vitest reports each file by its REAL path; a root reached through a symlink (macOS TMPDIR
  // /var -> /private/var) would key every name as "../.." and read every file "excluded" (I27).
  const realRoot = realpathSync(root);
  for (const t of rep.testResults ?? []) {
    const rel = relative(realRoot, t.name);
    if (rel in res) res[rel] = t.status === "passed" && (t.assertionResults ?? []).every((a) => a.status !== "failed") ? "pass" : "fail";
  }
  return res;
}

export function verdict(state, files, days) {
  const r = state.results;
  const at = (f, d) => r[f]?.[String(d)];
  const offsets = [0, ...days];
  const unreached = [];
  const excluded = files.filter((f) => offsets.some((d) => at(f, d) === "excluded"));
  for (const f of files) {
    if (excluded.includes(f)) continue;
    for (const d of offsets) if (at(f, d) === undefined || at(f, d) === "timeout" || at(f, d) === CRASH) unreached.push(`${f} @+${d}${at(f, d) === "timeout" || at(f, d) === CRASH ? ` (${at(f, d)})` : ""}`);
  }
  const baselineRed = files.filter((f) => at(f, 0) === "fail");
  const clockBound = [];
  for (const f of files) {
    if (at(f, 0) !== "pass") continue;
    const red = days.filter((d) => at(f, d) === "fail");
    if (red.length) clockBound.push({ file: f, redAt: red });
  }
  return { clockBound, baselineRed, unreached, excluded };
}

          }
          save();
          continue;
        }
        msPerFile = (Date.now() - started) / chunk.length;
        shrunk -= chunk.length;
        if (shrunk <= 0) size = o.chunk;
        for (const [f, s] of Object.entries(res)) record(f, d, s);
        todo = todo.slice(chunk.length);
        save();
        console.error(`clock-forward-scan: +${d}d ${chunk.length} file(s), ${todo.length} left at this offset`);
      }
    }
  }
  const v = verdict(state, files, o.days);
  console.log(`clock-forward-scan: ${files.length} file(s) at +0 and +${o.days.join("/+")} days`);
  for (const c of v.clockBound) console.log(`CLOCK-BOUND ${c.file} — green at +0, red at +${c.redAt.join(", +")}d`);
  for (const f of v.baselineRed) console.log(`BASELINE-RED ${f} — red at +0 (not attributable to the clock)`);
  for (const f of v.excluded) console.log(`EXCLUDED ${f} — filtered out by the vitest config, never run`);
  for (const u of v.unreached) console.log(`NOT-REACHED ${u}`);
  console.log(`summary: clock_bound=${v.clockBound.length} baseline_red=${v.baselineRed.length} not_reached=${v.unreached.length} excluded=${v.excluded.length}`);
  // I28-GATECHECKS G3 (I27 JA-02): a run that scanned nothing checked nothing, and used to exit 0.
  if (files.length - v.excluded.length === 0) {
    const why =
      files.length === 0
        ? o.onlyRedAt !== null
          ? `--only-red-at ${o.onlyRedAt} matched no file in the state`
          : `no test file under ${o.paths.length ? o.paths.join(", ") : "(no path given)"}`
        : `every one of the ${files.length} file(s) was EXCLUDED by the vitest config`;
    console.log(`clock-forward-scan: 0 file(s) scanned — ${why}; nothing was checked, so this run is not a pass`);
    return 2;
  }
  // I28-FX-GATE X2 (RG-GC-02): a file is classified against the clock only when it is green at +0; a COMPLETE run in
  // which none is (every scanned file BASELINE-RED) checked nothing against the calendar, and used to exit 0.
  const classified = files.filter((f) => !v.excluded.includes(f) && state.results[f]?.["0"] === "pass").length;
  if (v.unreached.length === 0 && classified === 0) {
    console.log(`clock-forward-scan: 0 file(s) classified against the clock — every one of the ${files.length - v.excluded.length} scanned file(s) is BASELINE-RED (red at +0), so nothing was checked against the calendar; this run is not a pass`);
    return 2;
  }
  if (v.clockBound.length) return 1;
  return v.unreached.length ? 3 : 0;
}

if (process.argv[1] && resolve(process.argv[1]) === fileURLToPath(import.meta.url)) {
  try {
    process.exitCode = main();
  } catch (e) {
    console.error(e instanceof UsageError ? e.message : `clock-forward-scan: the scan itself failed: ${e instanceof Error ? (e.stack ?? e.message) : String(e)}`);
    process.exitCode = 2;
  }
}

// clock-forward-shim.mjs — loaded with `node --import` by scripts/clock-forward-scan.mjs (QUEUE #890).
// Moves this process's wall clock forward by CLOCK_FORWARD_MS milliseconds: `Date.now()`, `new Date()`
// and `Date()` read the shifted instant, and so do the no-argument `Intl.DateTimeFormat#format()` / `#formatToParts()`;
// `new Date(x)`, `Date.parse`, `Date.UTC` are untouched.
// It replaces globalThis.Date BEFORE vitest loads, so vitest's fake timers capture the shifted Date as
// their "real" one: `vi.useRealTimers()` restores the shifted clock, and a test that pins its own clock
// with `vi.setSystemTime` is unaffected — which is exactly the property the scan measures.
const off = Number(process.env.CLOCK_FORWARD_MS ?? "0");
if (!Number.isFinite(off)) throw new Error(`clock-forward-shim: CLOCK_FORWARD_MS is not a number: ${process.env.CLOCK_FORWARD_MS}`);
if (off !== 0) {
  const RealDate = globalThis.Date;
  const realNow = RealDate.now.bind(RealDate);
  const shiftedNow = () => realNow() + off;
  globalThis.Date = new Proxy(RealDate, {
    apply: () => new RealDate(shiftedNow()).toString(),
    construct: (target, args, newTarget) => Reflect.construct(target, args.length === 0 ? [shiftedNow()] : args, newTarget),
    get: (target, prop, receiver) => (prop === "now" ? shiftedNow : Reflect.get(target, prop, receiver)),
  });
  // `Intl.DateTimeFormat#format()` and `#formatToParts()` with NO argument read the engine's intrinsic clock, which
  // replacing globalThis.Date does not move (I13-RG-DATA RGD-890-scan-intrinsic): hand them the shifted instant.
  // Any argument is passed through untouched; only `undefined` (the no-argument case) becomes the shifted instant.
  const proto = Intl.DateTimeFormat.prototype;
  const fmt = Object.getOwnPropertyDescriptor(proto, "format");
  const parts = proto.formatToParts;
  Object.defineProperty(proto, "format", {
    ...fmt,
    get() {
      const bound = fmt.get.call(this);
      return (date) => bound(date === undefined ? shiftedNow() : date);
    },
  });
  proto.formatToParts = function formatToParts(date) {
    return parts.call(this, date === undefined ? shiftedNow() : date);
  };
}
