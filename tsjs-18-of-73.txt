/**
 * Creator session tokens — HMAC-signed, stateless. Issued after a proven
 * claim; identify the creator by the passport they proved control of.
 *
 * HMAC (crypto.subtle.sign, NOT a hashing primitive) with a server secret from
 * env. No secret configured → sessions are DISABLED (issue returns null): a
 * claim is still provable and recorded, we just don't hand out a session. We
 * never fall back to an unsigned or guessable token.
 */

const encoder = new TextEncoder();

/**
 * THE ONE PLACE A SESSION SECRET ENTERS THE SYSTEM.
 *
 * This used to be a LENGTH CHECK — `secret.length` against 16 — written twice
 * (issue + verify) and nowhere else, and a length check is not a secret check.
 * Measured before the fix: `issueSession("TODO-set-a-real-session-secret")`
 * MINTED A TOKEN, signed with a string published verbatim in DEPLOY.md, while
 * `/ops/config` reported `sessions: false` because `configHealth` had its own,
 * different rule (`!/TODO/i`). Sixteen spaces minted a token AND reported
 * healthy. A forged session is takeover of any fan, creator or agency account,
 * so the two rules disagreeing IS the finding: the operator surface said the
 * feature was off while the feature was on and signing with a published key.
 *
 * There is now ONE predicate. `issueSession`, `verifySession`, every
 * `sessions_unavailable` 503 in trust-core, and `configHealth`'s `sessions`
 * boolean all call it, so the health surface and the behaviour agree BY
 * CONSTRUCTION rather than by two lists somebody has to keep in step.
 *
 * The rule is the one `registry-build.ts` already applies to
 * `REGISTRY_DECOY_SEED` (`isUsableDecoySeed`): long enough, and not an obvious
 * stand-in. It is reimplemented rather than imported because a package may not
 * depend on a service; if a third reader ever appears, hoist this one — do not
 * write a third rule.
 *
 * Absent or placeholder DISABLES sessions (issue returns null, verify returns
 * null, the sign-in routes 503). It never weakens them and it never falls back
 * to an unsigned or guessable token — invariant 6, fail closed.
 */
export const SESSION_SECRET_MIN_LEN = 32;
const SECRET_PLACEHOLDER_RE = /TODO|CHANGE[-_ ]?ME|PLACEHOLDER|EXAMPLE|^0+$|^(.)\1*$/i;

/** True iff `v` is a session secret this system is willing to sign with. */
export function usableSessionSecret(v: unknown): v is string {
  return typeof v === "string" && v.length >= SESSION_SECRET_MIN_LEN && v.trim() !== "" && !SECRET_PLACEHOLDER_RE.test(v);
}

function b64url(bytes: Uint8Array): string {
  let bin = "";
  for (const b of bytes) bin += String.fromCharCode(b);
  return btoa(bin).replace(/\+/g, "-").replace(/\//g, "_").replace(/=+$/, "");
}
function b64urlToBytes(s: string): Uint8Array {
  const pad = s.length % 4 === 0 ? "" : "=".repeat(4 - (s.length % 4));
  const bin = atob(s.replace(/-/g, "+").replace(/_/g, "/") + pad);
  const out = new Uint8Array(bin.length);
  for (let i = 0; i < bin.length; i++) out[i] = bin.charCodeAt(i);
  return out;
}

function hmacKey(secret: string) {
  return crypto.subtle.importKey("raw", encoder.encode(secret), { name: "HMAC", hash: "SHA-256" }, false, ["sign", "verify"]);
}

export type SubjectKind = "creator" | "fan" | "agency";

/**
 * THERE IS NO `tier` CLAIM, AND ADDING ONE BACK IS THE BUG.
 *
 * A `tier?: "free" | "premium"` claim used to sit on `SessionSubject` and `t?`
 * on the wire, documented as the mechanism by which a SEPARATE worker enforces
 * per-tier quota off the signed token without a cross-service call. It never
 * worked: no production call site of `issueSession` ever passed a tier — all
 * thirteen passed `{kind, id, ep[, member]}` — so `t` was never stamped,
 * `sub.tier` was always undefined, and its one reader (the image service's
 * `/find` quota) always resolved `free`. A paying Premium fan was quietly
 * metered as free, and only the image service's own tests ever minted a tiered
 * token, which is the only reason the type survived at all.
 *
 * STAMPING IT WOULD HAVE BEEN WORSE THAN THE BUG. `SESSION_TTL_SECONDS` is 30
 * days and the entitlement lives in trust-core's billing tables, so a stamped
 * tier is a 30-day UNREVOCABLE grant read by a worker that cannot see the
 * cancellation: a fan who cancels or charges back keeps premium quota in the
 * image service for up to a month, with no way to revoke it short of rotating
 * the session secret and logging every user in the product out.
 *
 * The claim is therefore DELETED rather than left optional-and-ignored, and the
 * image service reads the tier live over the internal session-check seam below
 * (`internalServiceCredential`), where a cancellation propagates in seconds.
 * If a future service needs a tier, it asks that seam; it does not stamp a
 * month-long entitlement into a bearer token.
 */

export interface SessionSubject {
  kind: SubjectKind;
  id: string; // passport_id for creators, account_id for fans
  ep: number; // session epoch it was minted at (for logout-everywhere; 0 when unused)
  /** Agency sessions only (S97): the member_id of the human who signed in. The
   * member's ROLE is deliberately NOT stamped — authority is re-read from
   * agency_members on every request, so a demotion/removal changes what a LIVE
   * session can do immediately. REQUIRED on an agency session since 2026-08-15:
   * a memberless agency token used to resolve to full admin, grandfathering a
   * population of zero. Absent `m` on an agency token is now rejected outright. */
  member?: string;
  /** Agency sessions minted by a SEAT key only (I3-1740, D-0860(3)): the seat
   * and the seat epoch it was minted at. `agencyPrincipal` refuses the token
   * once that seat is revoked or its epoch moves, so revoking one seat ends
   * that seat's sessions and nobody else's. Absent on an agency-KEY session. */
  seat?: string;
  seatEp?: number;
  /** Agency sessions only (I3-0686, D-0920): the instant (unix seconds) of the
   * SIGN-IN this session descends from — the `oiat` claim, or the token's own
   * `iat` for a token minted before the claim existed. `/agency/refresh` refuses
   * once it is more than AGENCY_ABSOLUTE_SECONDS old, so a stolen token cannot
   * be renewed forever by its holder. */
  oiat?: number;
}

/** I3-0686 (D-0920): how long an agency session may be RENEWED after the sign-in
 * it descends from. Past it, `/agency/refresh` answers 401 and the member signs
 * in again (key + identity). A token minted by the last permitted refresh still
 * lives out its own SESSION_TTL_SECONDS. */
export const AGENCY_ABSOLUTE_SECONDS = 90 * 24 * 60 * 60;

interface SessionClaims {
  k: SubjectKind;
  id: string;
  iat: number;
  exp: number;
  ep?: number; // session epoch (absent → 0, i.e. pre-revocation tokens)
  m?: string; // agency member id — REQUIRED on an agency token (absent → rejected)
  st?: string; // agency seat id — only with `sep`, only on an agency token
  sep?: number; // seat epoch at mint
  oiat?: number; // agency only: the originating sign-in's iat (I3-0686)
}

 * placeholder secret returns null, and both ends of the seam answer 503 on null
 * rather than skipping the check.

function hex(bytes: Uint8Array): string {
  let out = "";
  for (const b of bytes) out += b.toString(16).padStart(2, "0");
  return out;
}

/** The static credential a sibling worker presents (as `X-Internal-Auth`) to
 * reach one internal trust-core seam, or null when sessions are disabled. Both
 * the caller and the verifier derive it the same way from the same secret. */
export async function internalServiceCredential(secret: string | undefined, audience: string): Promise<string | null> {
  if (!usableSessionSecret(secret)) return null;
  if (typeof audience !== "string" || audience.length === 0) return null;
  // I3-1350 (D-0920): the image key is never handed out through this door. I20-FXC (I20-RG-SEC-03): nor is any
  // other key DERIVED as HMAC(session secret, label) — every such label is in RESERVED_KEY_LABELS.
  if ((RESERVED_KEY_LABELS as readonly string[]).includes(audience)) return null;
  // The eight image seams are keyed under the DERIVED image key, so trust-core
  // (which passes SESSION_SECRET here) and the image worker (which holds only
  // the derived key and calls `imageSeamCredential`) compute the same value.
  if ((IMAGE_SEAM_AUDIENCES as readonly string[]).includes(audience)) return imageSeamCredential(await deriveImageScopeKey(secret), audience);
  const key = await hmacKey(secret);
  return hex(new Uint8Array(await crypto.subtle.sign("HMAC", key, encoder.encode(audience))));
}

/**

 * The image worker used to hold SESSION_SECRET itself, so a compromise of that
 * worker was a compromise of every login in the product. It now holds only
 * `IMAGE_SCOPE_KEY` = hex HMAC-SHA256(SESSION_SECRET, IMAGE_SCOPE_KEY_LABEL):
 * derived, one-way, no new secret value. With it the image worker can
 *   - verify an IMAGE token (`verifyImageToken`), which trust-core mints with
 *     `issueImageToken` for a live session, valid IMAGE_TOKEN_TTL_SECONDS;
 *   - present / check the eight image-seam credentials (`imageSeamCredential`).
 * It cannot mint or verify a session: an image token is `img1.<payload>.<sig>`

 * session token is two parts, which `verifyImageToken` refuses.
 */
export const IMAGE_SCOPE_KEY_LABEL = "dhwaja/image-scope-key.v1";
/** D-0960(13): the label of the dispute-fingerprint key, HMAC(SESSION_SECRET, this) (trust-core account_data.ts
 * disputeIdentityFingerprint). Declared HERE so the credential door refuses the same constant the derivation uses. */
export const DISPUTE_FINGERPRINT_KEY_LABEL = "dhwaja/erased-dispute-fingerprint/key.v1";

 * internalServiceCredential would return for that audience — so the door refuses each of them. */
export const RESERVED_KEY_LABELS = [IMAGE_SCOPE_KEY_LABEL, DISPUTE_FINGERPRINT_KEY_LABEL] as const;
/** I65-IMAGE (Jason D-1569): image -> trust-core, "may this passport's face data be erased now?" (trust-core

export const FACE_HOLD_ASK_AUDIENCE = "dhwaja/internal/face-hold-ask.v1"; // image -> trust-core

export const FACE_ERASE_AUDIENCE = "dhwaja/internal/face-erase.v1"; // trust-core -> image
export const IMAGE_SEAM_AUDIENCES = [SESSION_CHECK_AUDIENCE, SUPPRESSION_CHECK_AUDIENCE, LIKENESS_REVOKE_AUDIENCE, LIKENESS_REVOKED_AUDIENCE, EXACT_HASH_REVIEWS_AUDIENCE, CREATOR_IMAGE_DATA_AUDIENCE, FACE_HOLD_ASK_AUDIENCE, FACE_ERASE_AUDIENCE] as const;
export const IMAGE_TOKEN_TTL_SECONDS = 15 * 60;
const IMAGE_TOKEN_PREFIX = "img1";

/** The derived image key, or null when sessions are disabled. */
export async function deriveImageScopeKey(sessionSecret: string | undefined): Promise<string | null> {
  if (!usableSessionSecret(sessionSecret)) return null;
  const key = await hmacKey(sessionSecret);
  return hex(new Uint8Array(await crypto.subtle.sign("HMAC", key, encoder.encode(IMAGE_SCOPE_KEY_LABEL))));
}

/** True iff `v` has the derived key's exact shape (64 lowercase hex). */
export function usableImageScopeKey(v: unknown): v is string {
  return typeof v === "string" && /^[0-9a-f]{64}$/.test(v) && !/^(.)\1*$/.test(v);
}

/** An image-seam credential from the derived key; null for any other audience. */
export async function imageSeamCredential(imageKey: string | null | undefined, audience: string): Promise<string | null> {
  if (!usableImageScopeKey(imageKey)) return null;
  if (!(IMAGE_SEAM_AUDIENCES as readonly string[]).includes(audience)) return null;
  const key = await hmacKey(imageKey);
  return hex(new Uint8Array(await crypto.subtle.sign("HMAC", key, encoder.encode(audience))));
}

interface ImageClaims {
  k: "creator" | "fan";
  id: string;
  iat: number;
  exp: number;
  ep: number;
}

/** trust-core side: an image token for an ALREADY-ADMITTED subject, or null. */
export async function issueImageToken(sessionSecret: string | undefined, subject: { kind: SubjectKind; id: string; ep?: number }, nowSec: number): Promise<string | null> {
  const imageKey = await deriveImageScopeKey(sessionSecret);
  if (!imageKey) return null;
  if (subject.kind !== "creator" && subject.kind !== "fan") return null;
  if (typeof subject.id !== "string" || subject.id.length === 0 || subject.id.length > 128) return null;
  const claims: ImageClaims = { k: subject.kind, id: subject.id, iat: nowSec, exp: nowSec + IMAGE_TOKEN_TTL_SECONDS, ep: subject.ep ?? 0 };
  const payload = b64url(encoder.encode(JSON.stringify(claims)));
  const key = await hmacKey(imageKey);
  const sig = new Uint8Array(await crypto.subtle.sign("HMAC", key, encoder.encode(`${IMAGE_TOKEN_PREFIX}.${payload}`)));
  return `${IMAGE_TOKEN_PREFIX}.${payload}.${b64url(sig)}`;
}

/** image side: verify an image token with the derived key only. */
export async function verifyImageToken(imageKey: string | null | undefined, token: string, nowSec: number): Promise<SessionSubject | null> {
  if (!usableImageScopeKey(imageKey)) return null;
  if (typeof token !== "string") return null;
  const parts = token.split(".");
  if (parts.length !== 3 || parts[0] !== IMAGE_TOKEN_PREFIX) return null;
  const payload = parts[1] as string;
  const sig = parts[2] as string;
  if (!/^[A-Za-z0-9_-]+$/.test(payload) || !/^[A-Za-z0-9_-]+$/.test(sig)) return null;
  const key = await hmacKey(imageKey);
  let ok: boolean;
  try {
    ok = await crypto.subtle.verify("HMAC", key, b64urlToBytes(sig) as unknown as ArrayBuffer, encoder.encode(`${IMAGE_TOKEN_PREFIX}.${payload}`) as unknown as ArrayBuffer);
  } catch {
    return null;
  }
  if (!ok) return null;
  let c: ImageClaims;
  try {
    c = JSON.parse(new TextDecoder().decode(b64urlToBytes(payload))) as ImageClaims;
  } catch {
    return null;
  }
  if (!c || (c.k !== "creator" && c.k !== "fan") || typeof c.id !== "string" || c.id.length === 0) return null;
  if (!Number.isSafeInteger(c.iat) || !Number.isSafeInteger(c.exp) || !Number.isSafeInteger(c.ep) || c.ep < 0) return null;
  if (c.exp < nowSec || c.exp - c.iat > IMAGE_TOKEN_TTL_SECONDS || c.iat > nowSec + 60) return null;
  return { kind: c.k, id: c.id, ep: c.ep };
}

/**
 * Length-independent-ish constant-time string compare (per-char XOR over the
 * longer length, with the length difference folded in). Used on both sides of
 * the internal seam so a secret comparison never leaks through timing. Kept
 * here rather than in either service so there is ONE implementation of it on
 * the seam's two ends.
 */
export function constantTimeEqual(a: string, b: string): boolean {
  const len = Math.max(a.length, b.length);
  let diff = a.length === b.length ? 0 : 1;
  for (let i = 0; i < len; i++) diff |= (a.charCodeAt(i) || 0) ^ (b.charCodeAt(i) || 0);
  return diff === 0;
}

/** Accepted limit (I3-1654, docs/AUTH-SOCIAL-SIGNIN.md): a 30-day bearer with no
 * token id (no `jti` in SessionClaims) and no idle expiry. Revocation is all at
 * once only — logout-all bumps the epoch (`ep`); one token cannot be revoked alone. */
export const SESSION_TTL_SECONDS = 30 * 24 * 60 * 60;

/** Issue a signed session, or null if sessions are disabled (no secret). */
export async function issueSession(
  secret: string | undefined,
  subject: { kind: SubjectKind; id: string; ep?: number; member?: string; seat?: string; seatEp?: number; oiat?: number; expCap?: number },
  nowSec: number,
): Promise<string | null> {
  if (!usableSessionSecret(secret)) return null;
  if (subject.kind !== "creator" && subject.kind !== "fan" && subject.kind !== "agency") return null;
  // I3-2055 (D-1010(2)): a renewal passes the presented token's own exp as `expCap`, so a renewed session never
  // outlives the sign-in it descends from (that sign-in's exp = its iat + SESSION_TTL_SECONDS).
  const exp = typeof subject.expCap === "number" && Number.isSafeInteger(subject.expCap) ? Math.min(nowSec + SESSION_TTL_SECONDS, subject.expCap) : nowSec + SESSION_TTL_SECONDS;
  if (exp <= nowSec) return null;
  const claims: SessionClaims = { k: subject.kind, id: subject.id, iat: nowSec, exp, ep: subject.ep ?? 0, ...(subject.member ? { m: subject.member } : {}), ...(subject.kind === "agency" && subject.seat ? { st: subject.seat, sep: subject.seatEp ?? 0 } : {}), ...(subject.kind === "agency" ? { oiat: subject.oiat ?? nowSec } : {}) };
  const payload = b64url(encoder.encode(JSON.stringify(claims)));
  const key = await hmacKey(secret);
  const sig = new Uint8Array(await crypto.subtle.sign("HMAC", key, encoder.encode(payload)));
  return `${payload}.${b64url(sig)}`;
}

/** Verify a session token. Constant-time MAC via crypto.subtle.verify. Returns
 * the subject, or null. Callers check `kind` to keep fan/creator tokens from
 * crossing surfaces. */
export async function verifySession(
  secret: string | undefined,
  token: string,
  nowSec: number,
): Promise<SessionSubject | null> {
  if (!usableSessionSecret(secret)) return null;
  const parts = token.split(".");
  if (parts.length !== 2) return null;
  const [payload, sig] = parts as [string, string];
  if (!/^[A-Za-z0-9_-]+$/.test(payload) || !/^[A-Za-z0-9_-]+$/.test(sig)) return null;
  const key = await hmacKey(secret);
  let ok: boolean;
  try {
    ok = await crypto.subtle.verify("HMAC", key, b64urlToBytes(sig) as unknown as ArrayBuffer, encoder.encode(payload) as unknown as ArrayBuffer);
  } catch {
    return null;
  }
  if (!ok) return null;
  let claims: SessionClaims;
  try {
    claims = JSON.parse(new TextDecoder().decode(b64urlToBytes(payload))) as SessionClaims;
  } catch {
    return null;
  }
  if ((claims.k !== "creator" && claims.k !== "fan" && claims.k !== "agency") || typeof claims.id !== "string" || !Number.isFinite(claims.exp) || claims.exp < nowSec) {
    return null;
  }
  // A PRESENT-but-malformed member claim kills the token outright rather than
  // silently degrading to a memberless subject — dropping it would turn a bad
  // claim into an ESCALATION.
  if (claims.m !== undefined && (typeof claims.m !== "string" || claims.m.length === 0 || claims.m.length > 64)) {
    return null;
  }
  // And an AGENCY token with no member claim at all is rejected here, not just
  // at the caller. It used to resolve to {memberId: null, role: "admin"},
  // bypassing the per-seat system to grandfather pre-S97 tokens that never
  // existed. Enforcing it in the verifier means no other service can resolve
  // one either, whatever its own caller does.
  if (claims.k === "agency" && typeof claims.m !== "string") {
    return null;
  }
  // The seat claim (I3-1740) comes whole or not at all, and only on an agency
  // token: a present-but-malformed one kills the token like `m`, because
  // dropping it would turn a seat session into an agency-key session that no
  // seat revocation reaches.
  if (claims.st !== undefined || claims.sep !== undefined) {
    if (claims.k !== "agency" || typeof claims.st !== "string" || claims.st.length === 0 || claims.st.length > 64) return null;
    if (typeof claims.sep !== "number" || !Number.isSafeInteger(claims.sep) || claims.sep < 0) return null;
  }
  // I3-0686: the origin claim is agency-only and comes well-formed or kills the
  // token — never silently dropped, because a dropped `oiat` would fall back to
  // `iat` and turn a renewal ceiling into a fresh one.
  if (claims.oiat !== undefined) {
    if (claims.k !== "agency" || typeof claims.oiat !== "number" || !Number.isSafeInteger(claims.oiat) || claims.oiat < 0) return null;
    if (typeof claims.iat === "number" && claims.oiat > claims.iat) return null;
  }
  const oiat = claims.k !== "agency" ? undefined : claims.oiat ?? (typeof claims.iat === "number" && Number.isFinite(claims.iat) ? claims.iat : undefined);
  return {
    kind: claims.k,
    id: claims.id,
    ep: typeof claims.ep === "number" && Number.isFinite(claims.ep) ? claims.ep : 0,
    ...(typeof claims.m === "string" ? { member: claims.m } : {}),
    ...(typeof claims.st === "string" ? { seat: claims.st, seatEp: claims.sep as number } : {}),
    ...(oiat === undefined ? {} : { oiat }),
  };
}

/**
 * #242 candidate doctrines as pluggable reroll predicates — SIM ONLY. The
 * engine's `Doctrine` union stays `"guardian"` until Jason rules (D-0434,
 * D-0794). Names and thresholds are candidates; Jason sets the real ones.
 */
import { rerollExpectedDamageNumerator } from "../src/dials.js";
import { STAT_COUNT } from "../src/index.js";
import type { RerollPolicy } from "./resolver.js";

/** The engine's rule, reproduced: the hit would KO, or the Champion takes >= championGuardThreshold and (championGuardEv 1,
 * I3-1626 D-1082(8)) more than the re-thrown round is expected to do to it, each outcome capped at its remaining HP (D-1083(3)). */
export const guardian: RerollPolicy = (c) =>
  c.damage >= c.fieldedHp ||
  (c.isChampion &&
    c.damage >= c.dials.championGuardThreshold &&
    (c.dials.championGuardEv === 0 ||
      rerollExpectedDamageNumerator(c.dials.rollSides, c.dials.damageMultiplier, c.ownMod, c.otherMod, c.fieldedHp) < c.damage * c.dials.rollSides * c.dials.rollSides));

/** Never rerolls — the floor every doctrine is measured against. */
export const none: RerollPolicy = () => false;

/** AGGRESSOR: reroll a THIN loss (margin <= t), turning near-misses into a fresh throw. */
export const aggressor = (t: number): RerollPolicy => (c) => c.margin <= t;

/** GAMBLER: reroll any loss in the first `n` rounds, then never — front-load or bust. */
export const gambler = (n: number): RerollPolicy => (c) => c.round <= n;

/**
 * SPECIALIST: reroll when the battleground stat is one of the `k` stats the
 * loser's roster is weakest in (lowest roster sum; ties to the lower index).
 */
export const specialist = (k: number): RerollPolicy => {
  const cache = new WeakMap<object, Set<number>>();
  return (c) => {
    let weak = cache.get(c.side.roster);
    if (!weak) {
      const sums = Array.from({ length: STAT_COUNT }, (_, s) => c.side.roster.reduce((a, card) => a + card.stats[s]!, 0));
      const order = sums.map((v, s) => [v, s] as const).sort((x, y) => x[0] - y[0] || x[1] - y[1]);
      weak = new Set(order.slice(0, k).map((e) => e[1]));
      cache.set(c.side.roster, weak);
    }
    return weak.has(c.statIndex);
  };
};

  "aggressor(15)": aggressor(15),
  "gambler(3)": gambler(3),
  "gambler(10)": gambler(10),
  "specialist(2)": specialist(2),
});

/** A random side: stats uniform 1..7, bonds uniform 0..DEFAULT_DIALS.bondMax, Champion = highest stat sum (ties to lower index). */
export function randomSide(label: string, n: number, rnd: () => number, clientSeed: string): WarSide {
  const roster: WarCard[] = Array.from({ length: n }, (_, i) => ({
    passportId: `psp_${label}${i}`,
    stats: Array.from({ length: STAT_COUNT }, () => 1 + Math.floor(rnd() * 7)),
  }));
  const bonds = roster.map(() => Math.floor(rnd() * (DEFAULT_DIALS.bondMax + 1)));
  let champ = 0;
  const sum = (c: WarCard) => c.stats.reduce((a, b) => a + b, 0);
  for (let i = 1; i < n; i++) if (sum(roster[i]!) > sum(roster[champ]!)) champ = i;
  return { player: label, roster, championIndex: champ, bonds, doctrine: "guardian", clientSeed };
}

/**
 * kind: fan = m on every card of side A; creator = m on A's Champion only; stacked = both flags at m each (2m on the
 * Champion, m elsewhere). I31 (I30 JASON-QUESTIONS 43, D-0388 as built — war-match.ts cardFlags puts the creator flag on
 * the flagged creator's card in ANY roster slot): creator_any = m on the flagged card wherever it sits; stacked_any = m on
 * every card plus m more on the flagged card. The flagged card's slot is drawn per trial from its own stream
 * (`flaggedSlot`), uniform over A's roster, so it is the Champion one trial in `roster` and the older kinds' draws are
 * unchanged.
 */
export type FlagKind = "fan" | "creator" | "stacked" | "creator_any" | "stacked_any";
export interface FlagCell {
  kind: FlagKind;
  m: number;
  roster: number;
}

export const FLAG_MODS = [-21, -14, -10, -7, -3, 0, 3, 7, 10, 14, 21];
export const FLAG_CELLS: FlagCell[] = [
  ...FLAG_MODS.map((m) => ({ kind: "fan" as const, m, roster: 10 })),
  ...FLAG_MODS.map((m) => ({ kind: "creator" as const, m, roster: 10 })),
  ...[-10, -7, -3, 3, 7].map((m) => ({ kind: "stacked" as const, m, roster: 10 })),
  ...[-14, -7, 7, 14].map((m) => ({ kind: "fan" as const, m, roster: 30 })),
  // I31: appended, so every older cell keeps its chunk-independent seeds and reproduces byte for byte.
  ...FLAG_MODS.map((m) => ({ kind: "creator_any" as const, m, roster: 10 })),
  ...[-10, -7, -3, 3, 7].map((m) => ({ kind: "stacked_any" as const, m, roster: 10 })),
];

/** I31: the flagged card's roster slot for trial `t` of a `*_any` cell — its own stream, uniform over the roster. */
export function flaggedSlot(masterSeed: number, roster: number, t: number): number {
  return Math.floor(prng(masterSeed, EXP_FLAGS, roster, t, 3)() * roster);
}

/** `slot` is the flagged card's index for the `*_any` kinds (the Champion when omitted). */
export function flagVector(kind: FlagKind, m: number, side: WarSide, slot: number = side.championIndex): number[] {
  return side.roster.map((_, i) => {
    if (kind === "fan") return m;
    if (kind === "creator") return i === side.championIndex ? m : 0;
    if (kind === "creator_any") return i === slot ? m : 0;
    if (kind === "stacked_any") return m + (i === slot ? m : 0);
    return m + (i === side.championIndex ? m : 0);
  });
}

/** Side A carries the flag; trials t0..t1-1. Common random numbers: trial t draws the same two sides in every cell of a roster size. */
export async function runFlagCell(masterSeed: number, cell: FlagCell, t0: number, t1: number): Promise<Tally> {
  const tally: Tally = { wins: 0, n: 0, rounds: 0, rerolls: 0 };
  for (let t = t0; t < t1; t++) {
    const rnd = prng(masterSeed, EXP_FLAGS, cell.roster, t);
    const a = randomSide("a", cell.roster, rnd, seedHex(masterSeed, EXP_FLAGS, cell.roster, t, 1));
    const b = randomSide("b", cell.roster, rnd, seedHex(masterSeed, EXP_FLAGS, cell.roster, t, 2));
    const zero = b.roster.map(() => 0);
    const log = await simulateMatch({
      serverSeed: seedHex(masterSeed, EXP_FLAGS, cell.roster, t, 0),
      sides: [a, b],
      policies: [guardian, guardian],
      flags: [flagVector(cell.kind, cell.m, a, cell.kind.endsWith("_any") ? flaggedSlot(masterSeed, cell.roster, t) : a.championIndex), zero],
    });
    tally.n++;
    if (log.result.winner === 0) tally.wins++;
    tally.rounds += log.result.rounds;
    tally.rerolls += log.result.rerollsUsed[0];
  }
  return tally;
}

function emptyFlag(): FlagStats {
  return {
    heldWeeks: 0, handovers: 0, distinctHolders: 0, maxTenureWeeks: 0, sumTenureWeeks: 0, tenures: 0, sumValueAtPass: 0,
    minValue: Number.POSITIVE_INFINITY, holderDays: 0, negativeHolderDays: 0,
    bucket: { pos: { n: 0, wins: 0 }, mid: { n: 0, wins: 0 }, neg: { n: 0, wins: 0 } }, topHolderWeeks: 0,
  };
}

/**
 * Weekly hand-over, #238/#239 as ruled (D-0385..D-0390): the top scorer takes
 * the flag; a sole-top incumbent keeps it WITHOUT refresh; an incumbent never
 * wins a tie; among challengers the fewest losses wins; a remaining tie goes to
 * the lowest index (SIM ASSUMPTION — no ruling on a full tie). Zero wins = no change.
 */
export function rotate(incumbent: number | null, wins: readonly number[], losses: readonly number[]): number | null {
  let best = 0;
  for (const w of wins) if (w > best) best = w;
  if (best === 0) return incumbent;
  const top = wins.map((w, i) => (w === best ? i : -1)).filter((i) => i >= 0);
  if (incumbent !== null && top.length === 1 && top[0] === incumbent) return incumbent;
  const challengers = top.filter((i) => i !== incumbent);
  let pick = challengers[0]!;
  for (const i of challengers) if (losses[i]! < losses[pick]!) pick = i;
  return pick;
}

interface Holder {
  id: number | null;
  captureDay: number;
  since: number;
}

export async function runSeason(masterSeed: number, cfg: SeasonConfig, season: number, cfgIndex: number): Promise<SeasonResult> {
  const rnd = prng(masterSeed, EXP_SEASON, cfgIndex, season);
  // Creators and their market popularity (strong creators sit in more rosters).
  const creators: WarCard[] = Array.from({ length: cfg.creators }, (_, c) => ({
    passportId: `psp_c${c}`,
    stats: Array.from({ length: STAT_COUNT }, () => 1 + Math.floor(rnd() * 7)),
  }));
  const weight = creators.map((c) => c.stats.reduce((a, b) => a + b, 0) ** 2);
  const rosters: number[][] = [];
  const bonds: number[][] = [];
  for (let f = 0; f < cfg.fans; f++) {
    const pool = creators.map((_, c) => c);
    const w = [...weight];
    const pick: number[] = [];
    for (let k = 0; k < cfg.roster; k++) {
      const tot = w.reduce((a, b) => a + b, 0);
      let x = rnd() * tot;
      let j = 0;
      while (j < pool.length - 1 && x >= w[j]!) x -= w[j++]!;
      pick.push(pool[j]!);
      pool.splice(j, 1);
      w.splice(j, 1);
    }
    rosters.push(pick);
    // REG3-1b (I30-TOOLS; I21-Q30(2), D-1055(1)): uniform 0..DEFAULT_DIALS.bondMax, as randomSide (:26) draws.
    bonds.push(pick.map(() => Math.floor(rnd() * (DEFAULT_DIALS.bondMax + 1))));
  }
  const statSum = creators.map((c) => c.stats.reduce((a, b) => a + b, 0));
  const crValue = (h: Holder, day: number) => cfg.bonus - cfg.decay * (day - h.captureDay);
  const anySlot = cfg.creatorFlagSlot === "any";
  const fanValue = (h: Holder, day: number) => (cfg.fanBonus ?? cfg.bonus) - (cfg.fanDecay ?? cfg.decay) * (day - h.captureDay);

  const fanH: Holder = { id: null, captureDay: 0, since: 0 };
  const crH: Holder = { id: null, captureDay: 0, since: 0 };
  const fs = emptyFlag();
  const cs = emptyFlag();
  const fanHeld = new Map<number, number>();
  const crHeld = new Map<number, number>();
  const stacked = { n: 0, wins: 0, minValue: Number.POSITIVE_INFINITY, maxValue: Number.NEGATIVE_INFINITY };
  let matches = 0;
  let flaggedChamp = 0;

  const bucketOf = (v: number) => (v > 0 ? "pos" : v >= -7 ? "mid" : "neg");

  for (let week = 0; week < cfg.weeks; week++) {
    const fanWins = new Array<number>(cfg.fans).fill(0);
    const fanLosses = new Array<number>(cfg.fans).fill(0);
    const crWins = new Array<number>(cfg.creators).fill(0);
    const crLosses = new Array<number>(cfg.creators).fill(0);
    if (fanH.id !== null) fanHeld.set(fanH.id, (fanHeld.get(fanH.id) ?? 0) + 1);
    if (crH.id !== null) crHeld.set(crH.id, (crHeld.get(crH.id) ?? 0) + 1);
    if (fanH.id !== null) fs.heldWeeks++;
    if (crH.id !== null) cs.heldWeeks++;

    for (let d = 0; d < 7; d++) {
      const day = week * 7 + d;
      const fv = fanH.id !== null ? fanValue(fanH, day) : 0;
      const cv = crH.id !== null ? crValue(crH, day) : 0;
      if (fanH.id !== null) {
        fs.holderDays++;
        if (fv < 0) fs.negativeHolderDays++;
        fs.minValue = Math.min(fs.minValue, fv);
      }
      if (crH.id !== null) {
        cs.holderDays++;
        if (cv < 0) cs.negativeHolderDays++;
        cs.minValue = Math.min(cs.minValue, cv);
      }
      // Champion choice (SIM ASSUMPTION): each fan fields argmax(statSum + bond + creator flag if that card is flagged).
      const champ = rosters.map((r, f) =>
        r.reduce((best, c, i) => {
          const s = (k: number) => statSum[r[k]!]! + bonds[f]![k]! + (r[k] === crH.id ? cv : 0);
          return s(i) > s(best) ? i : best;
        }, 0),
      );
      for (let m = 0; m < cfg.matchesPerDay; m++) {
        const order = Array.from({ length: cfg.fans }, (_, i) => i);
        for (let i = order.length - 1; i > 0; i--) {
          const j = Math.floor(rnd() * (i + 1));
          [order[i], order[j]] = [order[j]!, order[i]!];
        }
        for (let p = 0; p + 1 < order.length; p += 2) {
          const fa = order[p]!;
          const fb = order[p + 1]!;
          const mk = (f: number, lab: string): WarSide => ({
            player: `fan${f}`,
            roster: rosters[f]!.map((c) => creators[c]!),
            championIndex: champ[f]!,
            bonds: bonds[f]!,
            doctrine: "guardian",
            clientSeed: seedHex(masterSeed, EXP_SEASON, cfgIndex, season, day, m, p, lab === "a" ? 1 : 2),
          });
          const flagFor = (f: number) =>
            rosters[f]!.map((c, i) => (f === fanH.id ? fv : 0) + (c === crH.id && (anySlot || i === champ[f]) ? cv : 0));
          const log = await simulateMatch({
            serverSeed: seedHex(masterSeed, EXP_SEASON, cfgIndex, season, day, m, p, 0),
            sides: [mk(fa, "a"), mk(fb, "b")],
            policies: [guardian, guardian],
            flags: [flagFor(fa), flagFor(fb)],
          });
          matches++;
          const wf = log.result.winner === 0 ? fa : fb;
          const lf = log.result.winner === 0 ? fb : fa;
          fanWins[wf]!++;
          fanLosses[lf]!++;
          crWins[rosters[wf]![champ[wf]!]!]!++;
          crLosses[rosters[lf]![champ[lf]!]!]!++;
          for (const f of [fa, fb]) {
            const champCreator = rosters[f]![champ[f]!]!;
            if (champCreator === crH.id) flaggedChamp++;
            // Whether this fan's side carries the creator flag this match: as Champion only, or (I31 any slot) anywhere.
            const carries = crH.id !== null && (anySlot ? rosters[f]!.includes(crH.id) : champCreator === crH.id);
            const won = f === wf ? 1 : 0;
            if (f === fanH.id) {
              const b = fs.bucket[bucketOf(fv)];
              b.n++;
              b.wins += won;
              if (carries) {
                stacked.n++;
                stacked.wins += won;
                stacked.minValue = Math.min(stacked.minValue, fv + cv);
                stacked.maxValue = Math.max(stacked.maxValue, fv + cv);
              }
            }
            if (carries) {
              const b = cs.bucket[bucketOf(cv)];
              b.n++;
              b.wins += won;
            }
          }
        }
      }
    }
    // End of week (Sunday 23:59 UTC): hand over. Next capture day = first day of next week.
    const nextDay = (week + 1) * 7;
    for (const [h, st, wins, losses, value] of [
      [fanH, fs, fanWins, fanLosses, fanValue],
      [crH, cs, crWins, crLosses, crValue],
    ] as const) {
      const next = rotate(h.id, wins, losses);
      if (next !== h.id) {
        if (h.id !== null) {
          st.handovers++;
          st.sumValueAtPass += value(h, nextDay - 1);
          const ten = week + 1 - h.since;
          st.sumTenureWeeks += ten;
          st.tenures++;
          st.maxTenureWeeks = Math.max(st.maxTenureWeeks, ten);
        }
        h.id = next;
        h.captureDay = nextDay;
        h.since = week + 1;
      }
    }
  }
  // Close the open tenures at season end (right-censored; counted in max only).
  if (fanH.id !== null) fs.maxTenureWeeks = Math.max(fs.maxTenureWeeks, cfg.weeks - fanH.since);
  if (crH.id !== null) cs.maxTenureWeeks = Math.max(cs.maxTenureWeeks, cfg.weeks - crH.since);
  fs.distinctHolders = fanHeld.size;
  cs.distinctHolders = crHeld.size;
  fs.topHolderWeeks = Math.max(0, ...fanHeld.values());
  cs.topHolderWeeks = Math.max(0, ...crHeld.values());
  return { config: cfg.name, season, matches, fan: fs, creator: cs, stacked, flaggedChampionShare: flaggedChamp / (2 * matches) };
}

}

/**
 * Does a max-Bond card still beat a 2-point stat edge on its own? Side A: random stats 1..5, every card at Bond =
 * `bondMax`. Side B: the SAME cards with every stat +2 (so B holds a 2-point edge in every duel), Bond 0. Both
 * guardian. Common random numbers: trial t draws the same rosters and seeds for every maximum.
 */
export async function runBondEdgeCell(masterSeed: number, bondMax: number, t0: number, t1: number): Promise<BondEdgeTally> {
  const tally: BondEdgeTally = { wins: 0, n: 0, rounds: 0, rerolls: 0, duels: 0, duelsA: 0 };
  for (let t = t0; t < t1; t++) {
    const rnd = prng(masterSeed, EXP_BOND_EDGE, t);
    const roster: WarCard[] = Array.from({ length: 10 }, (_, i) => ({ passportId: `psp_e${i}`, stats: Array.from({ length: STAT_COUNT }, () => 1 + Math.floor(rnd() * 5)) }));
    const a: WarSide = { player: "a", roster, championIndex: 0, bonds: roster.map(() => bondMax), doctrine: "guardian", clientSeed: seedHex(masterSeed, EXP_BOND_EDGE, t, 1) };
    const b: WarSide = { player: "b", roster: roster.map((c) => ({ passportId: c.passportId, stats: c.stats.map((x) => x + 2) })), championIndex: 0, bonds: roster.map(() => 0), doctrine: "guardian", clientSeed: seedHex(masterSeed, EXP_BOND_EDGE, t, 2) };
    const log = await simulateMatch({ serverSeed: seedHex(masterSeed, EXP_BOND_EDGE, t, 0), sides: [a, b], policies: [guardian, guardian], dials: { ...DEFAULT_DIALS, bondMax } });
    tally.n++;
    if (log.result.winner === 0) tally.wins++;
    tally.rounds += log.result.rounds;
    tally.rerolls += log.result.rerollsUsed[0];
    for (const r of log.rounds) {
      if (r.loser === null) continue;
      tally.duels++;
      if (r.loser === 1) tally.duelsA++;
    }
  }
  return tally;
}

/**
 * B27-SIM (QUEUE #353): a SHADOW of `resolveMatch` (src/engine.ts) with two
 * hooks the engine does not have, for pricing only — never for play:
 *   - `policy`: the reroll predicate. The engine hardcodes guardian
 *     (engine.ts, the `wouldFire` expression) and rejects any other doctrine
 *     in `validateSide`. #242's candidates live HERE, not in the engine.
 *   - `flags`: a per-side, per-card addend that may be negative (the #238
 *     creator flag and #239 fan flag), added to the modifier exactly where
 *     Bond is added (engine.ts, `modA`/`modB`).
 * `sim-determinism.test.ts` proves that with `guardian` and zero flags this
 * loop returns a MatchLog deep-equal to `resolveMatch`, so every other number
 * it produces differs from the engine ONLY by the hook under test.
 * Inputs are assumed valid: the sim builds them itself.
 */
import { DEFAULT_DIALS, RandomStream, STAT_COUNT, type Dials, type MatchLog, type MatchResult, type RollAttempt, type RoundLog, type WarSide } from "../src/index.js";

export interface RerollContext {
  round: number;
  /** The side that lost this attempt and is deciding whether to reroll. */
  loser: 0 | 1;
  damage: number;
  /** |totalA - totalB| before the damage multiplier. */
  margin: number;
  fieldedHp: number;
  isChampion: boolean;
  /** This round's modifier of the deciding side and of its opponent (engine.ts `modA`/`modB`), for the Guardian's re-throw expectation (I3-1626). */
  ownMod: number;
  otherMod: number;
  statIndex: number;
  /** The loser's roster. */
  side: WarSide;
  dials: Dials;
}

/** Called only while the loser still holds a reroll (the engine checks the pool first). */
export type RerollPolicy = (ctx: RerollContext) => boolean;

export interface SimMatchInput {
  serverSeed: string;
  sides: readonly [WarSide, WarSide];
  policies: readonly [RerollPolicy, RerollPolicy];
  /** Per-card addend per side (length = roster size); may be negative. Omitted = zeros. */
  flags?: readonly [readonly number[], readonly number[]];
  dials?: Dials;
}

export async function simulateMatch(input: SimMatchInput): Promise<MatchLog> {
  const dials = input.dials ?? DEFAULT_DIALS;
  const [A, B] = input.sides;
  const n = A.roster.length;
  const zero = A.roster.map(() => 0);
  const flags = input.flags ?? [zero, zero];
  const stream = new RandomStream(input.serverSeed, A.clientSeed, B.clientSeed);
  const hp: [number[], number[]] = [A.roster.map(() => dials.cardHp), B.roster.map(() => dials.cardHp)];
  const living: [number[], number[]] = [A.roster.map((_, i) => i), B.roster.map((_, i) => i)];
  const rerollsLeft: [number, number] = [dials.rerolls, dials.rerolls];
  const sides = [A, B] as const;
  const rounds: RoundLog[] = [];
  let decidedBy: MatchResult["by"] = "wipeout";
  let winner: 0 | 1 | null = null;

  for (let round = 1; living[0].length > 0 && living[1].length > 0; round++) {
    if (round > dials.maxRounds) {
      const hpSum0 = living[0].reduce((s, i) => s + hp[0][i]!, 0);
      const hpSum1 = living[1].reduce((s, i) => s + hp[1][i]!, 0);
      if (hpSum0 !== hpSum1) {
        winner = hpSum0 > hpSum1 ? 0 : 1;
        decidedBy = "hp";
      } else if (living[0].length !== living[1].length) {
        winner = living[0].length > living[1].length ? 0 : 1;
        decidedBy = "cards";
      } else {
        winner = (await stream.uniform(2)) as 0 | 1;
        decidedBy = "coin";
      }
      break;
    }
    let cardA: number;
    let cardB: number;
    if (round === 1) {
      cardA = A.championIndex;
      cardB = B.championIndex;
    } else {
      cardA = living[0][await stream.uniform(living[0].length)]!;
      cardB = living[1][await stream.uniform(living[1].length)]!;
    }
    const statIndex = await stream.uniform(STAT_COUNT);
    const statA = A.roster[cardA]!.stats[statIndex]!;
    const statB = B.roster[cardB]!.stats[statIndex]!;
    const edge = Math.abs(statA - statB);
    const modA = (statA > statB ? edge : 0) + A.bonds[cardA]! + flags[0][cardA]!;
    const modB = (statB > statA ? edge : 0) + B.bonds[cardB]! + flags[1][cardB]!;

    const attempts: RollAttempt[] = [];
    let loser: 0 | 1 | null = null;
    let damage = 0;
    for (;;) {
      const rollA = await stream.roll(dials.rollSides);
      const rollB = await stream.roll(dials.rollSides);
      const totalA = rollA + modA;
      const totalB = rollB + modB;
      attempts.push({ rollA, rollB, totalA, totalB });
      if (totalA === totalB) {
        loser = null;
        damage = 0;
        break;
      }
      loser = totalA > totalB ? 1 : 0;
      const margin = Math.abs(totalA - totalB);
      damage = margin * dials.damageMultiplier;
      const fielded = loser === 0 ? cardA : cardB;
      const wouldFire =
        rerollsLeft[loser] > 0 &&
        input.policies[loser]({
          round,
          loser,
          damage,
          margin,
          fieldedHp: hp[loser][fielded]!,
          isChampion: fielded === sides[loser].championIndex,
          ownMod: loser === 0 ? modA : modB,
          otherMod: loser === 0 ? modB : modA,
          statIndex,
          side: sides[loser],
          dials,
        });
      if (!wouldFire) break;
      rerollsLeft[loser]--;
      attempts[attempts.length - 1]!.rerolledBy = loser;
    }

    let hpAfter: number | undefined;
    let ko = false;
    if (loser !== null && damage > 0) {
      const fielded = loser === 0 ? cardA : cardB;
      const next = Math.max(0, hp[loser][fielded]! - damage);
      hp[loser][fielded] = next;
      hpAfter = next;
      if (next === 0) {
        ko = true;
        const list = living[loser];
        list.splice(list.indexOf(fielded), 1);
      }
    }
    const rec: RoundLog = { round, cardA, cardB, statIndex, statA, statB, edge, bondA: A.bonds[cardA]!, bondB: B.bonds[cardB]!, attempts, loser, damage, ko };
    // Q642: the engine records the fielded flags whenever a side carries any; given flags, so does the shadow.
    if (input.flags !== undefined) {
      rec.flagA = flags[0][cardA]!;
      rec.flagB = flags[1][cardB]!;
    }
    if (hpAfter !== undefined) rec.hpAfter = hpAfter;
    rounds.push(rec);
  }

  if (winner === null) winner = living[0].length > 0 ? 0 : 1;
  const result: MatchResult = {
    winner,
    by: decidedBy,
    rounds: rounds.length,
    rerollsUsed: [dials.rerolls - rerollsLeft[0], dials.rerolls - rerollsLeft[1]],
    championSurvived: [hp[0][A.championIndex]! > 0, hp[1][B.championIndex]! > 0],
    hpRemaining: [living[0].reduce((s, i) => s + hp[0][i]!, 0), living[1].reduce((s, i) => s + hp[1][i]!, 0)],
    cardsRemaining: [living[0].length, living[1].length],
  };
  return { engine: "war-engine/v1", dials, players: [A.player, B.player], rosterSize: n, champions: [A.championIndex, B.championIndex], rounds, result, bytesConsumed: stream.bytesConsumed };
}

/**
 * Deterministic seed derivation and interval arithmetic for the sim. No
 * digest here (single-hasher invariant): seeds are an integer mix (murmur3
 * fmix32) rendered as lowercase hex for `RandomStream`, which does the
 * hashing through @dhwaja/canonical.
 */
function fmix32(h: number): number {
  h ^= h >>> 16;
  h = Math.imul(h, 0x85ebca6b);
  h ^= h >>> 13;
  h = Math.imul(h, 0xc2b2ae35);
  h ^= h >>> 16;
  return h >>> 0;
}

function mix(parts: readonly number[], salt: number): number {
  let h = fmix32(salt ^ 0x9e3779b9);
  for (const p of parts) h = fmix32((h ^ fmix32((p >>> 0) + 0x632be5ab)) + Math.imul(h, 5));
  return h;
}

/** 16 lowercase hex chars, a pure function of (parts). */
export function seedHex(...parts: number[]): string {
  return mix(parts, 1).toString(16).padStart(8, "0") + mix(parts, 2).toString(16).padStart(8, "0");
}

/** A small deterministic PRNG for season set-up (rosters, pairings) — mulberry32. */
export function prng(...parts: number[]): () => number {
  let a = mix(parts, 3);
  return () => {
    a = (a + 0x6d2b79f5) >>> 0;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

/** Wilson score interval, 95%. */
export function wilson(wins: number, n: number): [number, number] {
  if (n === 0) return [0, 1];
  const z = 1.959963984540054;
  const p = wins / n;
  const d = 1 + (z * z) / n;
  const c = (p + (z * z) / (2 * n)) / d;
  const h = (z * Math.sqrt((p * (1 - p)) / n + (z * z) / (4 * n * n))) / d;
  return [Math.max(0, c - h), Math.min(1, c + h)];
}

export function pct(x: number): string {
  return (x * 100).toFixed(1);
}

/**
 * Creator War — balance dials. Every tunable constant in one place, so the
 * simulator (`packages/war-engine/sim/`) and the resolver read the SAME numbers and
 * a balance change is a one-line diff with a visible blast radius.
 *
 * These are game-feel constants, not architecture: changing them changes match
 * texture (and breaks golden replay fixtures on purpose — a dial change must
 * re-pin goldens, the same discipline as fold goldens).
 */

export interface Dials {
  /** Roll is uniform in [1, rollSides]. d100 keeps luck the loudest input. */
  readonly rollSides: number;
  /** Every card enters with this HP. */
  readonly cardHp: number;
  /** Rerolls each side carries into a match, spent by doctrine. */
  readonly rerolls: number;
  /**
   * Guardian doctrine: a reroll fires when the incoming damage would KO the
   * fielded card, or when the fielded card is the Champion and the damage is
   * at least this threshold (and, under `championGuardEv` 1, above the re-throw's
   * expected damage). DEFAULT_DIALS sets it to 0, no floor: under
   * `championGuardEv` 1 the expectation alone decides (I3-1626, D-1082(8)).
   */
  readonly championGuardThreshold: number;
  /**
   * I3-1626 (D-1082(8), correcting D-1078(24)): 1 = the Guardian rerolls a hit on its Champion (one that would not KO it)
   * only when the re-thrown round's expected damage to that Champion (`rerollExpectedDamageNumerator`, this round's
   * modifiers) is lower than the hit it replaces; 0 = the threshold alone decides, the rule every log stamped before this
   * dial ran on (INITIAL_DIALS).
   */
  readonly championGuardEv: number;
  /** Damage = margin × this multiplier (kept ×1; scale for match length). */
  readonly damageMultiplier: number;
  /**
   * Termination backstop. Ties do no damage, so an (astronomically unlikely)
   * unbounded tie streak must still terminate deterministically: at this many
   * rounds the match is decided by remaining HP, then living cards, then a
   * final draw from the committed stream. Sim shows real matches end in well
   * under 400 rounds even at 50 cards.
   */
  readonly maxRounds: number;
  /** Roster size bounds (matchmaking brackets are multiples of 10 inside). */
  readonly minRoster: number;
  readonly maxRoster: number;
  /** Stat bars are integers in [1, statMax] — the 1–7 retro bar. */
  readonly statMax: number;
  /** Bond is an integer in [0, bondMax]. */
  readonly bondMax: number;
}

/**
 * I3-1626 (D-1078(24)): the expected damage of a round's hit, E|a - b| x multiplier for two independent uniform dice on
 * [1, rollSides]: exactly (n^2 - 1) / (3n) x multiplier (33.33 for d100 at x1). A reroll re-throws BOTH dice, so this is
 * what the replacement hit is worth on average, before the round's stat, Bond and flag modifiers.
 */
export function expectedHitDamage(rollSides: number, damageMultiplier: number): number {
  return ((rollSides * rollSides - 1) / (3 * rollSides)) * damageMultiplier;
}
/** The smallest whole-damage Champion threshold at or above `expectedHitDamage` (34 for d100 x1). D-1078(24)'s default; since
 * D-1082(8) no dial default reads it (the Guardian compares `rerollExpectedDamageNumerator` instead). */
export function guardThresholdFor(rollSides: number, damageMultiplier: number): number {
  return Math.ceil(expectedHitDamage(rollSides, damageMultiplier));
}
/**
 * I3-1626 (D-1082(8), D-1083(3)): rollSides^2 x the expected damage a fresh throw of BOTH dice does to the side deciding
 * whether to reroll, at this round's modifiers, each outcome capped at that side's remaining HP (damage past a KO does
 * not count): the sum over k = otherRoll - ownRoll of (rollSides - |k|) x min(max(0, k + otherMod - ownMod) x
 * damageMultiplier, remainingHp) (a tie or a win does it no damage). An exact integer for an integer `remainingHp`, so
 * the comparison with a hit (`numerator < damage x rollSides^2`) has no rounding. Pass `Infinity` for no cap.
 */
export function rerollExpectedDamageNumerator(rollSides: number, damageMultiplier: number, ownMod: number, otherMod: number, remainingHp: number): number {
  const shift = otherMod - ownMod;
  let sum = 0;
  for (let k = Math.max(1 - rollSides, 1 - shift); k <= rollSides - 1; k++) sum += (rollSides - Math.abs(k)) * Math.min((k + shift) * damageMultiplier, remainingHp);
  return sum;
}

/**
 * Dials come from server config, not users — but a misconfigured (or
 * attacker-influenced) dial must fail loudly at the door, not as a
 * million-round CPU grind or fractional-HP weirdness. Integers, GAME-SIZED
 * ceilings (the ceilings are a work bound: worst case ≈ maxRounds × a few
 * hashes — keep that inside a Workers CPU budget), unknown keys rejected
 * (the repo-wide unknown-fields-are-rejected invariant applies to dials too).
 */
const DIAL_KEYS = new Set<string>(Object.keys(DEFAULT_DIALS));

function mergeDials(override: Partial<Dials> | undefined): Dials {
  if (override === undefined) return DEFAULT_DIALS;
  if (typeof override !== "object" || override === null || Array.isArray(override)) {
    throw new WarInputError("dials: must be a plain object");
  }
  for (const k of Object.keys(override)) {
    if (!DIAL_KEYS.has(k)) throw new WarInputError(`dials: unknown key "${k}"`);
  }
  // Copy known keys ONLY — nothing an attacker smuggles into the override
  // object (unknown fields, __proto__ as an own key, numeric keys) can reach
  // the frozen dials or the logged artifact.
  const merged = { ...DEFAULT_DIALS } as Record<string, number>;
  for (const k of DIAL_KEYS) {
    const v = (override as Record<string, unknown>)[k];
    if (v !== undefined) {
      if (typeof v !== "number") throw new WarInputError(`dials: ${k} must be a number`);
      merged[k] = v;
    }
  }
  return Object.freeze(merged as unknown as Dials);
}

function validateDials(d: Dials): void {
  const intIn = (v: number, lo: number, hi: number) => Number.isInteger(v) && v >= lo && v <= hi;
  if (!intIn(d.rollSides, 2, 10_000)) throw new WarInputError("dials: rollSides out of range");
  if (!intIn(d.cardHp, 1, 10_000)) throw new WarInputError("dials: cardHp out of range");
  if (!intIn(d.rerolls, 0, 10)) throw new WarInputError("dials: rerolls out of range");
  if (!intIn(d.championGuardThreshold, 0, 10_000)) throw new WarInputError("dials: championGuardThreshold out of range");
  if (!intIn(d.championGuardEv, 0, 1)) throw new WarInputError("dials: championGuardEv out of range");
  if (!intIn(d.damageMultiplier, 1, 100)) throw new WarInputError("dials: damageMultiplier out of range");
  if (!intIn(d.maxRounds, 0, 20_000)) throw new WarInputError("dials: maxRounds out of range");
  if (!intIn(d.minRoster, 1, 100) || !intIn(d.maxRoster, d.minRoster, 100)) throw new WarInputError("dials: roster bounds invalid");
  if (!intIn(d.statMax, 1, 100)) throw new WarInputError("dials: statMax out of range");
  if (!intIn(d.bondMax, 0, 100)) throw new WarInputError("dials: bondMax out of range");
}

function validateSide(side: WarSide, label: string, dials: Dials): void {
  if (typeof side.player !== "string" || side.player.length === 0 || side.player.length > 200) {
    throw new WarInputError(`${label}: player label must be a non-empty string (≤200 chars)`);
  }
  const n = side.roster.length;
  if (n < dials.minRoster || n > dials.maxRoster) {
    throw new WarInputError(`${label}: roster size ${n} outside [${dials.minRoster}, ${dials.maxRoster}]`);
  }
  if (!Number.isInteger(side.championIndex) || side.championIndex < 0 || side.championIndex >= n) {
    throw new WarInputError(`${label}: champion index out of range`);
  }
  if (side.bonds.length !== n) throw new WarInputError(`${label}: bonds length must equal roster length`);
  if (side.doctrine !== "guardian") throw new WarInputError(`${label}: unknown doctrine`);
  const seen = new Set<string>();
  for (let i = 0; i < n; i++) {
    const card = side.roster[i]!;
    if (typeof card.passportId !== "string" || card.passportId.length === 0 || card.passportId.length > 64) {
      throw new WarInputError(`${label}: card ${i} passportId invalid`);
    }
    if (seen.has(card.passportId)) throw new WarInputError(`${label}: duplicate creator in roster (one card per creator)`);
    seen.add(card.passportId);
    if (card.stats.length !== STAT_COUNT) throw new WarInputError(`${label}: card ${i} must have ${STAT_COUNT} stats`);
    for (const s of card.stats) {
      if (!Number.isInteger(s) || s < 1 || s > dials.statMax) {
        throw new WarInputError(`${label}: card ${i} stat outside [1, ${dials.statMax}]`);
      }
    }
    const bond = side.bonds[i]!;
    if (!Number.isInteger(bond) || bond < 0 || bond > dials.bondMax) {
      throw new WarInputError(`${label}: card ${i} bond outside [0, ${dials.bondMax}]`);
    }
  }
  if (side.flags !== undefined) {
    if (!Array.isArray(side.flags) || side.flags.length !== n) throw new WarInputError(`${label}: flags length must equal roster length`);
    for (const f of side.flags) if (!Number.isInteger(f) || Math.abs(f) > MAX_FLAG_MAGNITUDE) throw new WarInputError(`${label}: flag outside [-${MAX_FLAG_MAGNITUDE}, ${MAX_FLAG_MAGNITUDE}]`);
  }
}

/** Q642: a flag is unbounded below by ruling; this is only a sanity bound on a malformed input. */
const MAX_FLAG_MAGNITUDE = 100_000;

/**
 * Resolve a full match. Deterministic; async only because sha256 is.
 */
export async function resolveMatch(input: WarMatchInput): Promise<MatchLog> {
  const dials: Dials = mergeDials(input.dials);
  return resolveOn(input, dials, dials);
}

/**
 * Re-resolve a match from the dial set that was PINNED with it (I26-INT, REGISTRY hand-off, D-1014(6)): the same
 * run/stamp split `verifyMatch` uses, for a caller that holds the pinned dials but not the log (trust-core
 * war-match.ts getReplay regenerating a lost replay object). Runs every dial the pinned set lacks at its INITIAL value
 * (never today's default), ignores a key the engine no longer reads, and stamps the pinned set EXACTLY into the log,
 * so the regenerated replay is byte-identical to the one the row's hash pins.
 */
export async function resolveMatchAsLogged(input: Omit<WarMatchInput, "dials">, logged: Readonly<Record<string, unknown>>): Promise<MatchLog> {
  if (typeof logged !== "object" || logged === null || Array.isArray(logged)) throw new WarInputError("dials: must be a plain object");
  return resolveOn(input, runFromLogged(logged), Object.freeze({ ...logged }) as unknown as Dials); // resolveOn never reads input.dials
}

/** INITIAL_DIALS overlaid by every dial key the logged set carries (QUEUE #193, D-0814(3)). */
function runFromLogged(logged: Readonly<Record<string, unknown>>): Dials {
  const run = { ...INITIAL_DIALS } as Record<string, unknown>;
  for (const k of DIAL_KEYS) if (Object.prototype.hasOwnProperty.call(logged, k)) run[k] = logged[k];
  return Object.freeze(run) as unknown as Dials;
}

/** The resolver proper: runs on `dials`, stamps `stamp` into the log (the two
 * differ only for `verifyMatch` and `resolveMatchAsLogged`, which stamp the logged set they replay). */
async function resolveOn(input: WarMatchInput, dials: Dials, stamp: Dials): Promise<MatchLog> {
  validateDials(dials);
  if (!Array.isArray(input.sides) || input.sides.length !== 2) {
    throw new WarInputError("exactly two sides required");
  }
  const [A, B] = input.sides;
  validateSide(A, "side A", dials);
  validateSide(B, "side B", dials);
  if (A.roster.length !== B.roster.length) {
    throw new WarInputError("rosters must be the same size (matchmaking brackets guarantee this)");
  }

  const stream = new RandomStream(input.serverSeed, A.clientSeed, B.clientSeed);
  const n = A.roster.length;
  const flagged = A.flags !== undefined || B.flags !== undefined;
  const flagOf = (s: WarSide, i: number): number => s.flags?.[i] ?? 0;

  // Mutable battle state: HP per card, living index lists (in roster order).
  const hp: [number[], number[]] = [A.roster.map(() => dials.cardHp), B.roster.map(() => dials.cardHp)];
  const living: [number[], number[]] = [A.roster.map((_, i) => i), B.roster.map((_, i) => i)];
  const rerollsLeft: [number, number] = [dials.rerolls, dials.rerolls];
  const sides = [A, B] as const;
  // I3-1626: the re-throw's expected damage depends only on the modifier gap and (D-1083(3)) the Champion's remaining HP,
  // which caps each outcome, so it is computed once per (gap, HP) per match.
  const evByGapHp = new Map<string, number>();
  const rerollEv = (ownMod: number, otherMod: number, remainingHp: number): number => {
    const key = `${otherMod - ownMod}:${remainingHp}`;
    let v = evByGapHp.get(key);
    if (v === undefined) {
      v = rerollExpectedDamageNumerator(dials.rollSides, dials.damageMultiplier, ownMod, otherMod, remainingHp);
      evByGapHp.set(key, v);
    }
    return v;
  };

  const rounds: RoundLog[] = [];
  let decidedBy: MatchResult["by"] = "wipeout";
  let winner: 0 | 1 | null = null;

  for (let round = 1; living[0].length > 0 && living[1].length > 0; round++) {
    if (round > dials.maxRounds) {
      // Deterministic backstop: HP, then cards, then a final committed draw.
      const hpSum0 = living[0].reduce((s, i) => s + hp[0][i]!, 0);
      const hpSum1 = living[1].reduce((s, i) => s + hp[1][i]!, 0);
      if (hpSum0 !== hpSum1) {
        winner = hpSum0 > hpSum1 ? 0 : 1;
        decidedBy = "hp";
      } else if (living[0].length !== living[1].length) {
        winner = living[0].length > living[1].length ? 0 : 1;
        decidedBy = "cards";
      } else {
        winner = (await stream.uniform(2)) as 0 | 1;
        decidedBy = "coin";
      }
      break;
    }

    // 1. Fielded cards: round 1 is ALWAYS the Champion duel; after that, a
    //    uniform draw over each side's living cards.
    let cardA: number;
    let cardB: number;
    if (round === 1) {
      cardA = A.championIndex;
      cardB = B.championIndex;
    } else {
      cardA = living[0][await stream.uniform(living[0].length)]!;
      cardB = living[1][await stream.uniform(living[1].length)]!;
    }

    // 2. Battleground stat.
    const statIndex = await stream.uniform(STAT_COUNT);
    const statA = A.roster[cardA]!.stats[statIndex]!;
    const statB = B.roster[cardB]!.stats[statIndex]!;
    const edge = Math.abs(statA - statB);
    const modA = (statA > statB ? edge : 0) + A.bonds[cardA]! + flagOf(A, cardA);
    const modB = (statB > statA ? edge : 0) + B.bonds[cardB]! + flagOf(B, cardB);

    // 3. Roll, then let the LOSER's doctrine spend rerolls. A reroll re-throws
    //    BOTH dice (the round is re-rolled), and the new loser evaluates next —
    //    bounded by the two sides' remaining reroll pools.
    const attempts: RollAttempt[] = [];
    let loser: 0 | 1 | null = null;
    let damage = 0;
    for (;;) {
      const rollA = await stream.roll(dials.rollSides);
      const rollB = await stream.roll(dials.rollSides);
      const totalA = rollA + modA;
      const totalB = rollB + modB;
      attempts.push({ rollA, rollB, totalA, totalB });
      if (totalA === totalB) {
        loser = null;
        damage = 0;
        break; // whiff — nothing to save anyone from
      }
      loser = totalA > totalB ? 1 : 0;
      damage = Math.abs(totalA - totalB) * dials.damageMultiplier;
      // Guardian: fire if the hit would KO my fielded card, or my Champion
      // takes a hit of at least the threshold that (championGuardEv 1, I3-1626 D-1082(8)) is larger than what the
      // re-thrown round is expected to do to it at this round's modifiers, each outcome capped at its remaining HP
      // (D-1083(3)) — and I still hold a reroll.
      const fielded = loser === 0 ? cardA : cardB;
      const fieldedHp = hp[loser][fielded]!;
      const isChampion = fielded === sides[loser].championIndex;
      const wouldFire =
        rerollsLeft[loser] > 0 &&
        (damage >= fieldedHp ||
          (isChampion &&
            damage >= dials.championGuardThreshold &&
            (dials.championGuardEv === 0 ||
              rerollEv(loser === 0 ? modA : modB, loser === 0 ? modB : modA, fieldedHp) < damage * dials.rollSides * dials.rollSides)));
      if (!wouldFire) break;
      rerollsLeft[loser]--;
      attempts[attempts.length - 1]!.rerolledBy = loser;
      // loop: fresh dice replace this attempt's outcome
    }

    // 4. Apply.
    let hpAfter: number | undefined;
    let ko = false;
    if (loser !== null && damage > 0) {
      const fielded = loser === 0 ? cardA : cardB;
      const next = Math.max(0, hp[loser][fielded]! - damage);
      hp[loser][fielded] = next;
      hpAfter = next;
      if (next === 0) {
        ko = true;
        const list = living[loser];
        list.splice(list.indexOf(fielded), 1);
      }
    }

    const rec: RoundLog = {
      round,
      cardA,
      cardB,
      statIndex,
      statA,
      statB,
      edge,
      bondA: A.bonds[cardA]!,
      bondB: B.bonds[cardB]!,
      attempts,
      loser,
      damage,
      ko,
    };
    if (flagged) {
      rec.flagA = flagOf(A, cardA);
      rec.flagB = flagOf(B, cardB);
    }
    if (hpAfter !== undefined) rec.hpAfter = hpAfter;
    rounds.push(rec);
  }

  if (winner === null) winner = living[0].length > 0 ? 0 : 1;

  const result: MatchResult = {
    winner,
    by: decidedBy,
    rounds: rounds.length,
    rerollsUsed: [dials.rerolls - rerollsLeft[0], dials.rerolls - rerollsLeft[1]],
    championSurvived: [hp[0][A.championIndex]! > 0, hp[1][B.championIndex]! > 0],
    hpRemaining: [
      living[0].reduce((s, i) => s + hp[0][i]!, 0),
      living[1].reduce((s, i) => s + hp[1][i]!, 0),
    ],
    cardsRemaining: [living[0].length, living[1].length],
  };

  return {
    engine: "war-engine/v1",
    dials: stamp,
    players: [A.player, B.player],
    rosterSize: n,
    champions: [A.championIndex, B.championIndex],
    rounds,
    result,
    bytesConsumed: stream.bytesConsumed,
  };
}

/**
 * Verify a replay: re-resolve from the (now-revealed) inputs and compare
 * byte-for-byte. True ⇒ not a single draw, roll, or reroll was cooked.
 *
 * QUEUE #193, D-0810 (12): the replay runs on `log.dials`, the snapshot
 * `resolveMatch` stamped into this log, never on today's DEFAULT_DIALS, and
 * uses that logged set EXACTLY, key set included — so a later dial edit, a
 * dial ADDED to the defaults or one RETIRED from them, never flips the verdict
 * of a match resolved before it. Every dial the engine reads comes from the
 * log when the log carries it; a dial the log predates runs at the value
 * it had when introduced (`INITIAL_DIALS`, D-0814(3)), never today's default,
 * so re-defaulting a dial later does not flip an older log (a new dial must be
 * introduced at its pre-dial behaviour, or the rounds differ and the verdict is
 * false, as it should be); a key the engine no
 * longer reads is carried through, not refused. The replayed log is stamped
 * with the logged set itself, so the rounds and result decide. An input that
 * names dials must agree with that snapshot on every key it names.
 */
export async function verifyMatch(input: WarMatchInput, log: MatchLog): Promise<boolean> {
  try {
    const snap: unknown = log.dials;
    if (typeof snap !== "object" || snap === null || Array.isArray(snap)) return false;
    if (input.dials !== undefined) {
      if (typeof input.dials !== "object" || input.dials === null || Array.isArray(input.dials)) return false;
      for (const [k, v] of Object.entries(input.dials)) {
        if (!Object.prototype.hasOwnProperty.call(snap, k) || (snap as Record<string, unknown>)[k] !== v) return false;
      }
    }
    const logged = snap as Record<string, unknown>;
    const fresh = await resolveOn(input, runFromLogged(logged), logged as unknown as Dials);
    return deepEqual(fresh, log);
  } catch {
    return false;
  }
}

/**
 * Structural equality — key-ORDER independent (PS-010: `JSON.stringify`
 * equality is a false negative the moment two equivalent objects are built
 * with fields in a different order; nothing here relies on insertion order).
 */
function deepEqual(a: unknown, b: unknown): boolean {
  if (a === b) return true;
  if (typeof a !== "object" || typeof b !== "object" || a === null || b === null) return false;
  if (Array.isArray(a) !== Array.isArray(b)) return false;
  if (Array.isArray(a) && Array.isArray(b)) {
    if (a.length !== b.length) return false;
    return a.every((v, i) => deepEqual(v, b[i]));
  }
  const ao = a as Record<string, unknown>;
  const bo = b as Record<string, unknown>;
  const aKeys = Object.keys(ao);
  const bKeys = Object.keys(bo);
  if (aKeys.length !== bKeys.length) return false;
  return aKeys.every((k) => Object.prototype.hasOwnProperty.call(bo, k) && deepEqual(ao[k], bo[k]));
}

import { sha256Bytes } from "@dhwaja/canonical";

/**
 * The provably-fair random stream. Every draw in a match — card picks, stat
 * picks, rolls, rerolls, the last-resort tiebreak — consumes this ONE stream,
 * in order, so a match is a pure function of (inputs, seeds) and anyone can
 * replay it from the revealed server seed.
 *
 * Construction: block i = sha256( seedLine ":" i ) where seedLine =
 * `serverSeed:clientSeedA:clientSeedB`. Seeds are VALIDATED lowercase hex, so
 * ":" can never appear inside a component — no way to shift bytes between
 * fields and produce a colliding seed line (a client seed is attacker-chosen;
 * without the alphabet restriction, "ab:1" / "ab" + ":1" would collide).
 *
 * sha256 comes from @dhwaja/canonical (single-hasher invariant) — the engine
 * must never reach for its own digest primitive.
 *
 * Bounded ints use REJECTION sampling (never modulo): draw enough bytes to
 * cover the range, reject values ≥ the largest multiple of n, redraw. This
 * keeps every face of the d100 exactly equally likely — a biased die in a
 * for-keeps game is a house edge nobody agreed to.
 */

const HEX_RE = /^[0-9a-f]{8,128}$/;

/** A seed usable in the stream: 8–128 chars of lowercase hex. */
export function isValidSeed(s: string): boolean {
  return typeof s === "string" && HEX_RE.test(s);
}

export class RandomStream {
  private readonly seedLine: string;
  private block: Uint8Array = new Uint8Array(0);
  private blockIndex = 0;
  private offset = 0;
  /** Total bytes consumed — recorded in the log so replays can sanity-check. */
  bytesConsumed = 0;

  constructor(serverSeed: string, clientSeedA: string, clientSeedB: string) {
    if (!isValidSeed(serverSeed)) throw new Error("invalid server seed (want 8-128 lowercase hex chars)");
    if (!isValidSeed(clientSeedA)) throw new Error("invalid client seed A (want 8-128 lowercase hex chars)");
    if (!isValidSeed(clientSeedB)) throw new Error("invalid client seed B (want 8-128 lowercase hex chars)");
    this.seedLine = `${serverSeed}:${clientSeedA}:${clientSeedB}`;
  }

  private async nextByte(): Promise<number> {
    if (this.offset >= this.block.length) {
      const material = new TextEncoder().encode(`${this.seedLine}:${this.blockIndex}`);
      this.block = await sha256Bytes(material);
      this.blockIndex++;
      this.offset = 0;
    }
    const b = this.block[this.offset]!;
    this.offset++;
    this.bytesConsumed++;
    return b;
  }

  /**
   * Uniform integer in [0, n). Rejection sampling over the minimal number of
   * bytes covering n. n must be an integer in [1, 2^32].
   */
  async uniform(n: number): Promise<number> {
    if (!Number.isInteger(n) || n < 1 || n > 0x1_0000_0000) throw new Error("uniform: n out of range");
    if (n === 1) return 0;
    let k = 1;
    let space = 256;
    while (space < n) {
      k++;
      space *= 256;
    }
    const limit = space - (space % n); // largest multiple of n ≤ space
    for (;;) {
      let v = 0;
      for (let i = 0; i < k; i++) v = v * 256 + (await this.nextByte());
      if (v < limit) return v % n;
      // rejected — redraw; probability of rejection < 1/2 per attempt
    }
  }

  /** Uniform die roll in [1, sides]. */
  async roll(sides: number): Promise<number> {
    return (await this.uniform(sides)) + 1;
  }
}

#!/usr/bin/env node
// Enumerate every audit_log ACTION literal in services/trust-core/src and say which are covered by
// the erasure sweep's action list — INV-061's population, computed rather than retyped.
//
// WHY THIS EXISTS. This enumeration has been got wrong three times, each time by a hand-written
// grep pasted into a scope packet:
//   W3-01 widened the sweep list 3 -> 6 while adding a SEVENTH action it did not list.
//   W3-02 widened it 6 -> 8; the judge then found two more limbs still open.
//   W3-03's scope packet (mine) told a room to enumerate with a SINGLE-QUOTED character class.
//     The literals in this codebase are overwhelmingly DOUBLE-quoted: that command returns 23 of
//     131 actions — 18% — and it omitted both subjects of its own brief. A room that ran it
//     verbatim would have reported a confident, complete-looking enumeration of a sixth of the
//     population. That is the frame-falsification failure this org has a standing rule about, and
//     I wrote it into the packet myself.
// A grep in prose gets retyped and re-broken. A script gets run.
//
// WHAT IT DOES NOT DO, deliberately: it does not decide whether a given action's `item_id` holds an
// ACCOUNT id (as opposed to a release, credential or passport id), because that is a read of the
// call site and a judgement about what the column means, and a script that guessed would be exactly
// the claim-shaped artifact D-0440 forbids. It gives you the true population and the true covered
// set; classifying the difference is the room's work, and it is a much smaller job than finding it.
import { readFileSync, readdirSync } from "node:fs";
import { join, resolve } from "node:path";

const ROOT = resolve(process.argv[2] ?? ".");
const SRC = join(ROOT, "services/trust-core/src");
const SWEEP_FILE = join(SRC, "account_data.ts");

// Both quotings, and backticks: an action literal is `<namespace>:<name>`, and which quote a given
// call site used is an accident of whether it sits in a SQL string or a TypeScript argument.
const LITERAL = /["'`]([a-z][a-z0-9_]*:[a-z0-9_]+)["'`]/g;

function actionsIn(text) {
  return new Set([...text.matchAll(LITERAL)].map((m) => m[1]));
}

// Read the sweep FIRST, because its own UPDATE text names every action it covers — and if that
// text counts as a usage, a swept entry can never be reported as stale no matter how long ago the
// literal left the codebase. The guard for this script caught exactly that: a fixture naming
// `account:renamed_away` in the sweep and nowhere else still read as present. So the sweep's own
// statement is cut out of the file before its literals are collected.
const sweepSrcRaw = readFileSync(SWEEP_FILE, "utf8");
const sweepStmt = sweepSrcRaw.match(/UPDATE audit_log SET item_id[\s\S]{0,400}?action IN \(([^)]*)\)/);
if (!sweepStmt) {
  console.error(`audit-action-inventory: no "UPDATE audit_log SET item_id ... action IN (...)" found in ${SWEEP_FILE}`);
  console.error("The sweep moved or changed shape. That is a finding, not a reason to fall back to a hardcoded list.");
  process.exit(2);
}
const swept = new Set([...sweepStmt[1].matchAll(/'([^']+)'/g)].map((m) => m[1]));

const files = readdirSync(SRC).filter((f) => f.endsWith(".ts"));
const all = new Map(); // action -> Set(file)
for (const f of files) {
  let text = readFileSync(join(SRC, f), "utf8");
  if (join(SRC, f) === SWEEP_FILE) text = text.replace(sweepStmt[0], "");
  for (const a of actionsIn(text)) {
    if (!all.has(a)) all.set(a, new Set());
    all.get(a).add(f);
  }
}

const uncovered = [...all.keys()].filter((a) => !swept.has(a)).sort();
const ghost = [...swept].filter((a) => !all.has(a)).sort();

console.log(`audit-action-inventory — ${ROOT}`);
console.log(`ACTION LITERALS IN src/: ${all.size}   SWEPT BY ERASURE: ${swept.size}   NOT SWEPT: ${uncovered.length}`);
console.log("");
console.log("SWEPT (the erasure sweep pseudonymises audit_log.item_id for these):");
for (const a of [...swept].sort()) console.log(`  ${a}${all.has(a) ? "" : "   << NOT FOUND in src/ — a stale entry, or the literal moved"}`);
if (ghost.length) console.log(`\n${ghost.length} swept action(s) have no literal in src/ — stale list entries are their own finding.`);
console.log("");
console.log("NOT SWEPT — for each, read the call site and decide whether item_id holds an ACCOUNT id:");
for (const a of uncovered) console.log(`  ${a}   (${[...all.get(a)].sort().join(", ")})`);
console.log("");
console.log(`Classifying the ${uncovered.length} above is the room's work; finding them is not. An action whose`);
console.log("item_id is an account id and which can fire AFTER erasure needs write-time pseudonymisation too —");
console.log("the sweep runs once and can never reach a row written later.");
process.exit(0);

// process to be echoed back in a condition's evidence.
//
// SEGMENTS, NOT JUST THE BASENAME. The first form tested `basename(p)` alone, so a secret that
// lives in a DIRECTORY name walked straight through the "refuses by name" check — `.aws/credentials`
// and `secrets/api.txt` both have innocent basenames. The anchors also made the shapes exact, so a
// suffix defeated them: `.dev.vars.production` (a documented Cloudflare per-env secrets file, see
// DEPLOY.md) and `secrets.env.bak` are the two that matter. Every segment of the path is tested now.
//
// MEASURED BEFORE IT WAS FROZEN, which is the whole discipline here (W3-02-R7's first list matched
// 14 of 6935 tracked paths and it correctly rejected itself: a gate that refuses on `credentials.ts`
// every run is a gate somebody disables). THIS list matches 0 of 6935 `git ls-files` paths, so a
// match is a signal and not noise, and it refuses every shape the basename form refused.
// `SOURCE_TAIL` is why: a segment carrying a source extension is code ABOUT secrets, never a secret
// — `src/credentials.ts`, `docs/secrets.md`, `lib/service-account.ts` — and it is exempt. That
// exemption subtracts nothing from the basename form; no shape below can end in a source extension.
// Pinned by tests/tooling/auto-commit-secret-shape.test.ts, which fails if the list is emptied.
const SECRET_SHAPES = [
  /^\.env(\..+)?$/i, /^\.envrc$/i, /^\.npmrc$/i, /^\.netrc$/i, /^\.pgpass$/i, /^\.htpasswd$/i,
  /^\.git-credentials$/i, /^\.?dev\.vars(\..+)?$/i, /^\.?secrets?$/i, /^\.?credentials?$/i,
  /^\.?(secrets?|credentials?)\.(json|ya?ml|env|txt|ini|toml|cfg|conf|properties)(\..+)?$/i,
  /^id_(rsa|dsa|ecdsa|ed25519)$/i,
  /(^|[-_.])service[-_]?accounts?([-_.][^/]*)?\.json$/i,
  /\.(pem|p12|pfx|jks|keystore|ppk|kdbx|asc|gpg|key)$/i,
];
const SOURCE_TAIL = /\.(ts|tsx|js|jsx|mjs|cjs|md|mdx|html|css|scss|sql|py|rs|go|java|rb|sh|snap)$/i;

  p.split("/").find((seg) => seg && !SOURCE_TAIL.test(seg) && SECRET_SHAPES.some((re) => re.test(seg)));

// A token out of SERIAL-NOTE.md that is not a plain repo-relative path is not a pathspec entry:
// a leading `-` is read by git as an option, and `..`/absolute leaves the repo entirely.
const unsafeToken = (rel) => rel.startsWith("/") || rel.startsWith("-") || rel.includes("\0") || /(^|\/)\.\.(\/|$)/.test(rel);

// PATHSPEC (G-10). "Only ever writes org/auto" is a BRANCH name; the content scope is this:
// everything under org/, plus the product files the stage's own SERIAL-NOTE entry names as
// touched. The entry is the `=== <HEADING>` block whose heading token appears in the stage
// dir's basename; its backticked paths are kept when they exist on disk. No entry matched =>
// `org` alone, stated on stdout. The pathspec never widens silently: a product file that is
// dirty but unlisted stays out of the commit.
function stagePathspec() {
  const spec = ["org"];
  const notes = [];
  const rejected = [];
  const notePath = join(ROOT, "SERIAL-NOTE.md");
  const base = basename(stageDir);
  if (!existsSync(notePath)) return { spec, rejected, evidence: "SERIAL-NOTE.md absent at repo root — pathspec is org alone" };
  let text = "";
  try { text = readFileSync(notePath, "utf8"); }
  catch (e) { return { spec, rejected, evidence: `SERIAL-NOTE.md unreadable (${e.message}) — pathspec is org alone` }; }
  const blocks = text.split(/^===\s+/m).slice(1);
  for (const b of blocks) {
    const heading = b.split(/\s|\n/)[0];
    if (!heading || !base.toUpperCase().includes(heading.toUpperCase())) continue;
    notes.push(heading);
    for (const m of b.matchAll(/`([^`\s]+\.[a-z]{1,4})`/g)) {
      const rel = m[1].replace(/^\.\//, "");
      if (rel.startsWith("org/") || spec.includes(rel)) continue;
      // SERIAL-NOTE.md is written by any room, wave builder or contributor: a token out of it is
      // UNTRUSTED INPUT to `git add`, not a fact. Two shapes are never added, and each one that is
      // refused is carried out of here so `pathspec-scope` can FAIL on it by name (I3-0004).
      if (unsafeToken(rel)) { rejected.push(`${rel} — not a repo-relative path (leading -, absolute, or ..)`); continue; }
      const seg = secretSegment(rel);
      if (seg !== undefined) {
        // The basename wording is kept verbatim for the basename case: W3-02-R7's harness pins it.
        rejected.push(seg === basename(rel) ? `${rel} — secret-shaped basename` : `${rel} — secret-shaped path segment "${seg}"`);
        continue;
      }
      if (existsSync(join(ROOT, rel))) spec.push(rel);
    }
  }
  const evidence = notes.length
    ? `org + ${spec.length - 1} product file(s) from SERIAL-NOTE entr${notes.length > 1 ? "ies" : "y"} ${notes.join(", ")}`
    : `no SERIAL-NOTE entry matches ${base} — pathspec is org alone`;
  return { spec, rejected, evidence };

// QUEUE #861 (D-0871(10), I10-FX1): a room id is a FILE NAME under --out-dir (`<id>.md`) and an owns entry

const ROOM_ID_RE = /^[A-Za-z0-9][A-Za-z0-9._-]*$/;
export function ownsPathProblem(o) {
  const s = String(o);
  if (/[\0\\]/.test(s)) return "carries a NUL or a backslash";
  // #868c (I16-W01F): an invisible/format character hides a leading `/` or `~` from the test below, and a
  // `${...}` HOME spelling, a drive letter or a URL scheme is absolute without starting `/` or `$HOME`.
  if (/[\u200b-\u200f\u202a-\u202e\u2060-\u2064\ufeff]/.test(s)) return "carries an invisible (zero-width or bidi) character";

/** Every problem with a spec, as one line each; [] when it can be rendered. */

  if (!Array.isArray(spec.rooms) || !spec.rooms.length) { out.push("spec.rooms is missing or empty"); return out; }
  const seen = new Set();
  spec.rooms.forEach((r, i) => {
    const at = `rooms[${i}]${r && r.id ? ` (${r.id})` : ""}`;

/** Exactly-once capture → { names: string[] | null, values: number[] }. */
function captureOnce(text, source, what, flags = "") {
  const matches = [...text.matchAll(new RegExp(source, `g${flags}`))];
  if (matches.length === 0) fail(`${what} found nothing`);
  if (matches.length > 1) fail(`${what} is ambiguous (${matches.length} matches)`);
  const m = matches[0];
  if (m.groups && Object.keys(m.groups).length) {
    const names = Object.keys(m.groups).sort();
    return { names, values: names.map((n) => num(m.groups[n], `${what} group ${n}`)) };
  }
  if (m.length < 2) fail(`${what} has no capture group`);
  return { names: null, values: m.slice(1).map((v, i) => num(v, `${what} group ${i + 1}`)) };
}

function derive(d) {
  if (d.cmd !== undefined) {
    let out;
    try { out = execFileSync("sh", ["-c", d.cmd], { cwd: ROOT, encoding: "utf8", stdio: ["ignore", "pipe", "pipe"] }); }
    catch (e) { fail(`derive cmd exited ${e.status ?? "?"}: ${String(e.stderr ?? e.message).trim().split("\n")[0]}`); }
    return captureOnce(out, d.capture, "derive capture", d.flags);
  }
  switch (d.builtin) {
    case "file": return captureOnce(readRel(d.file), d.capture, `derive capture in ${d.file}`, d.flags);
    case "line-count": { const t = readRel(d.file); return { names: null, values: [t.split("\n").length - (t.endsWith("\n") ? 1 : 0)] }; }
    case "json": {
      let v;
      try { v = JSON.parse(readRel(d.file)); } catch (e) { if (e instanceof ClaimError) throw e; fail(`${d.file}: ${e.message}`); }
      for (const k of String(d.path).split(".")) { if (v === null || typeof v !== "object" || !(k in v)) fail(`${d.file}: no path ${d.path}`); v = v[k]; }
      const coll = Array.isArray(v) ? v : v && typeof v === "object" ? Object.values(v) : null;
      if (d.op === "value") return { names: null, values: [num(v, `${d.file} ${d.path}`)] };
      if (!coll) fail(`${d.file} ${d.path} is not a collection`);
      if (d.op === "count") return { names: null, values: [coll.length] };
      if (d.op === "sum") return { names: null, values: [coll.reduce((a, x) => a + num(x, `${d.file} ${d.path} member`), 0)] };
      return fail(`json op must be value|count|sum (got ${d.op})`);
    }
    case "git-ls-count": {
      let out;
      try { out = execFileSync("git", ["-C", ROOT, "ls-files", "--", d.pathspec], { encoding: "utf8", stdio: ["ignore", "pipe", "pipe"] }); }
      catch (e) { fail(`git ls-files failed (fail closed): ${String(e.stderr ?? e.message).trim().split("\n")[0]}`); }
      const files = out.split("\n").filter(Boolean);
      if (files.length === 0) fail(`git index holds nothing under ${d.pathspec}`);
      const re = d.match ? new RegExp(d.match) : null;
      return { names: null, values: [re ? files.filter((f) => re.test(f)).length : files.length] };
    }
    case "vitest-totals": {
      const blocks = readRel(d.file).split(/^(?===\s)/m).filter((b) => /^==\s+vitest shard \d+\/\d+\s*$/m.test(b.split("\n")[0]));
      if (blocks.length === 0) fail(`${d.file} has no "== vitest shard" block`);
      const t = { failed: 0, files: 0, passed: 0, skipped: 0 };
      for (const b of blocks) {
        const fm = /Test Files\s+(?:(\d+) failed \| )?(\d+) passed(?: \| (\d+) skipped)? \((\d+)\)/.exec(b);
        const tm = /^\s*Tests\s+(?:(\d+) failed \| )?(\d+) passed(?: \| (\d+) skipped)? \((\d+)\)/m.exec(b);
        if (!fm || !tm) fail(`${d.file}: shard block "${b.split("\n")[0]}" has no Test Files/Tests line`);
        t.files += Number(fm[4]);
        t.passed += Number(tm[2]);
        t.skipped += Number(tm[3] ?? 0);
        t.failed += Number(tm[1] ?? 0);
      }
      const names = Object.keys(t).sort();
      return { names, values: names.map((n) => t[n]) };
    }
    default: return fail(`unknown builtin ${d.builtin}`);
  }
}

#!/usr/bin/env node
/**
 * CHECK-CONTRACT-MIRRORS — the room contracts, over the POPULATION, over BOTH TREES.
 *
 * WHY THIS EXISTS, and it is not a style preference. A mirror check already existed
 * (`scripts/preflight.mjs`, check 5). It missed BOTH defects in QUEUE #445 and it missed them
 * for the same reason:
 *
 *   for (const f of readdirSync("org/rooms")) { ...compare with .claude/agents/f... }
 *
 * That loop iterates the PAIRS IT ALREADY KNOWS ABOUT, in ONE TREE, in ONE DIRECTION.
 *   - `org/rooms/teachback.md` had no `.claude/agents/` counterpart in EITHER tree. A loop over
 *     `org/rooms` should have caught that one... except it ran only in the build tree, where the
 *     file was also absent from `org/rooms`, so the pair was never enumerated at all. Fourteen room
 *     contracts, thirteen agent files, and the check that exists to find exactly this saw nothing.
 *   - `.claude/agents/_PROMPT-CORE.md` was SEVEN LINES SHORT of `org/rooms/_PROMPT-CORE.md` — the
 *     whole `<high>` WEAKEST-MUTATION rule (D-0555). Every mutation proof taken by a room reading
 *     the agents copy was weaker than the record claims. One tree had the drift; the other did not.
 *
 * So this check does four things the old one did not:
 *   1. It enumerates the UNION of `org/rooms/*.md` and `.claude/agents/*.md`, so a file that exists
 *      on ONLY ONE side is a name in the population rather than an iteration that never happens.
 *   2. It runs over EVERY TREE it is given, and compares the population ACROSS trees too, so a
 *      contract that exists in the repo and not in the build tree is a failure, not a silence.
 *   3. It NAMES THE DIRECTION of every drift — which side has the file, which side has the bytes,
 *      and which side is authoritative (`org/rooms/` always; `.claude/agents/` is the copy).
 *   4. It FAILS ON AN EMPTY SET. A population of zero is how a path typo reports success, and
 *      INV-059 applies to this file as much as to anything it checks: a check that verified nothing
 *      says NOT VERIFIED and exits non-zero.
 *
 * IDENTITY HERE IS BYTE EQUALITY, NOT A DIGEST (W7-A, 2026-09-21). This file shipped comparing the
 * two sides by sha256, and `tests/invariants/single-hasher.test.ts` was right to redden on it: a
 * second hashing primitive outside `packages/canonical` is the canonical-divergence failure mode
 * that invariant exists to prevent, and a guard that breaks the tree's own invariants has no
 * standing to enforce anything. The remedy is NOT to route the digest through the sanctioned
 * module -- it is that this script never needed a digest at all. The question asked is "are these
 * two files byte-identical", and `Buffer.compare` answers exactly that. A digest answers a strictly
 * weaker question (equal digests imply equal bytes only up to a collision), so bytes are both
 * simpler and stronger. Drift now reports the OFFSET of the first differing byte, which is more
 * actionable than a digest prefix nobody can do anything with.
 *
 * USAGE
 *   node scripts/check-contract-mirrors.mjs [--tree <path>]... [--require-trees <n>] [--json]
 *
 *   --tree <path>        A tree root to check. Repeatable. An explicitly named tree that does not
 *                        exist, or that is missing either contract directory, is a FAILURE.
 *   --require-trees <n>  Minimum number of trees that must actually have been checked (default 1).
 *                        Pass 2 from the gate when both the repo and the build tree are mounted.
 *   --json               Machine-readable result on stdout.
 *
 *   With no --tree, the script auto-discovers: the current directory, then $DHWAJA_REPO, then
 *   $HOME/mnt/dhwaja, then $HOME/mnt/dhwaja-linux-build — keeping each that exists. Auto-discovered
 *   trees that were NOT found are printed by name, so a one-tree run can never be read as a
 *   two-tree run.
 *
 * EXIT: 0 only when at least --require-trees trees were checked, the population was non-empty, every
 * name had a counterpart on both sides in every tree, and every counterpart was byte-identical.
 */

/**
 * check-defenders.mjs — W3-03 head item 4 (Jason's ruling 2026-09-18), the
 * MECHANICAL form of rule 33.
 *
 * Rule 33 asks, of a landed change: if this change were undone, would anything
 * in the tree notice? Not "is the line executed" (68 in-tree payout assertions
 * executed services/trust-core/src/creator-payout.ts:1191 while its binding
 * check was neutered and money reached a disconnected Connect account), and
 * not "did a room NAME a test" (a declaration is a claim, D-0440/ANTI-024 v2:
 * every one of those 68 files exists and is green, so a registry pointing at
 * any of them would have returned PASS on that exact defect). The only
 * mechanical form of the question is: change the line, and see if the tree
 * goes red.
 *
 * So: mutation, scoped to the DIFF, with the mutant operator set taken from
 * Stryker's own instrumenter (@stryker-mutator/instrumenter allMutators) —
 * derived from the AST, never an operator list this script's author guessed.
 * Stryker's RUNNER is not used: its dry run plus sandbox copy does not fit
 * this box's ~170 s bash call (scripts/mutate-file.mjs documents that at
 * length). This script generates the same mutants offline and runs them
 * itself, one vitest invocation per mutant, resumable and shardable.
 *
 * PREDICATE (the load-bearing choice; tests/tooling/check-defenders.test.ts
 * is its guard):
 *   a changed REGION is DEFENDED only if EVERY mutant in it is Killed.
 *   ANY mutant that Survived or had NoCoverage is an UNDEFENDED finding.
 * The weaker "at least one mutant died" predicate is what fails the known
 * positive: at creator-payout.ts:1191 the `if (true)` mutant holds every
 * payout and dies loudly, while `if (false)` — skip the binding check, pay a
 * disconnected account — is the one nothing notices. The reported unit is
 * therefore the surviving MUTANT, not a per-region score.
 *
 * FAIL CLOSED (INV-059, and the D-0459 shape auto-commit.mjs now carries):
 * this leg prints exactly one of
 *     defenders: VERIFIED <r> region(s) / <m> mutant(s) in <f> file(s); <u> UNDEFENDED
 *     defenders: NOT VERIFIED — <reason>
 * NOT VERIFIED always exits non-zero. A missing/unparseable result, a zero
 * candidate test set, a region that generated no mutants, a restore that is
 * not byte-identical: each is NOT VERIFIED. Absence of an answer is never
 * the answer, and "ok" is never printed for a run that verified nothing.
 *
 * Usage:
 *   node scripts/check-defenders.mjs --base <ref> [--tree <dir>] [--shard i/N] [--json]

// Resolved THROUGH the instrumenter package so babel and the mutators are the
// exact versions Stryker itself would use; a divergence here would make this
// leg measure a different operator set than `pnpm mutate` does.
const INSTRUMENTER = "@stryker-mutator/instrumenter";
export async function loadMutators(root) {

  let pkgMain;
  try {
    pkgMain = req.resolve(INSTRUMENTER);
  } catch {

    pkgMain = join(hits, "dist/src/index.js");
  }
  const base = pkgMain.replace(/\/dist\/src\/index\.js$/, "");
  const req2 = createRequire(join(base, "package.json"));

// ------------------------------------------------------------ diff -> regions
const DEFAULT_GLOBS = [/^services\/[^/]+\/src\/.*\.ts$/, /^packages\/[^/]+\/src\/.*\.ts$/];

/**
 * Parse `git diff -U0` output into changed regions. Added/modified lines only:
 * a deletion-only hunk has no line left to mutate and is reported
 * NOT VERIFIABLE, never as defended.
 */
export function parseDiffRegions(diffText, globs = DEFAULT_GLOBS) {
  const regions = [];
  const deletions = [];
  let file = null;
  for (const line of String(diffText).split("\n")) {
    const f = /^\+\+\+ b\/(.+)$/.exec(line);
    if (f) {
      file = globs.some((g) => g.test(f[1])) ? f[1] : null;
      continue;
    }
    if (line.startsWith("--- ") || line.startsWith("diff --git")) continue;
    const h = /^@@ -\d+(?:,\d+)? \+(\d+)(?:,(\d+))? @@/.exec(line);
    if (!h || !file) continue;
    const start = Number(h[1]);
    const count = h[2] === undefined ? 1 : Number(h[2]);
    if (count === 0) {
      deletions.push({ file, atLine: start, deletionOnly: true });
      continue;
    }
    regions.push({ file, start, end: start + count - 1 });
  }
  return { regions, deletions };
}

// ------------------------------------------------------ region -> mutant list
/** Derive every Stryker mutant whose node STARTS inside one of `ranges`. */
export async function deriveMutants(root, file, ranges, loaded) {
  const { allMutators, parser, traverse, generate } = loaded;
  const abs = join(root, file);
  const source = readFileSync(abs, "utf8");
  const ast = parser.parse(source, {
    sourceType: "module",
    plugins: ["typescript", "decorators", "importAttributes", "explicitResourceManagement"],
    errorRecovery: true,
    ranges: true,
  });
  const inRange = (loc) => loc && ranges.some((r) => loc.start.line >= r.start && loc.start.line <= r.end);
  const out = [];
  const seen = new Set();
  traverse(ast, {
    enter(path) {
      if (!inRange(path.node.loc)) return;
      for (const mutator of allMutators) {
        let replacements;
        try {
          replacements = [...mutator.mutate(path, { isExpressionContext: false })];
        } catch {
          continue; // a mutator that cannot read this node contributes nothing
        }
        for (const node of replacements) {
          let code;
          try {
            code = generate(node).code;
          } catch {
            continue;
          }
          const start = path.node.start;
          const end = path.node.end;
          if (start == null || end == null) continue;
          const original = source.slice(start, end);
          if (code === original) continue;
          const key = `${start}:${end}:${code}`;
          if (seen.has(key)) continue;
          seen.add(key);
          out.push({
            file,
            line: path.node.loc.start.line,
            column: path.node.loc.start.column + 1,
            mutator: mutator.name,
            start,
            end,
            original: original.length > 120 ? `${original.slice(0, 117)}...` : original,
            replacement: code.length > 120 ? `${code.slice(0, 117)}...` : code,
            replacementFull: code,
          });
        }
      }
    },
  });
  return out;
}

// ---------------------------------------------------------------- apply/restore
// NOT a crypto hash, on purpose. `tests/invariants/single-hasher.test.ts` forbids hashing
// primitives outside packages/canonical, and it has an EXEMPT_FILES list this script would fit —
// the uses below are local file identity and a cache key, none of which reach a subject hash, a
// bucket prefix or a verdict. I did not take the exemption. Rooms in this wave were told not to
// reach for a guard's escape hatch to go green, and the orchestrator does not get a different
// rule; an exemption list that grows whenever it is inconvenient stops being a guard.
//
// The honest trade, stated rather than buried: FNV-1a 64-bit detects ACCIDENTAL change, which is
// the whole job here — "is this file still the text I saved" and "has the test population moved".
// It would not detect a deliberate collision, and nothing in this leg defends against an adversary
// who can already write to the scratch tree. If a use ever needs collision resistance, that use
// belongs in packages/canonical, not in an exemption.
const sha = (s) => {
  let h = 0xcbf29ce484222325n;
  const b = Buffer.from(String(s), "utf8");
  for (let i = 0; i < b.length; i++) { h ^= BigInt(b[i]); h = (h * 0x100000001b3n) & 0xffffffffffffffffn; }
  return h.toString(16).padStart(16, "0");
};

/**
 * Write the mutated source WITHOUT writing through the inode: the scratch tree
 * this leg measures in is hardlinked to the shared build tree, and an in-place
 * write would mutate a file four other rooms are editing. temp + rename only.
 */
export function writeDetached(abs, text) {
  const tmp = `${abs}.defenders.tmp`;
  writeFileSync(tmp, text, "utf8");
  renameSync(tmp, abs);
}

export function applyMutant(root, mutant, originalSource) {
  const abs = join(root, mutant.file);
  const mutated = originalSource.slice(0, mutant.start) + mutant.replacementFull + originalSource.slice(mutant.end);
  writeDetached(abs, mutated);
  return mutated;
}

export function restore(root, file, originalSource, originalSha) {
  const abs = join(root, file);
  writeDetached(abs, originalSource);
  const now = readFileSync(abs, "utf8");
  return { ok: sha(now) === originalSha, sha: sha(now) };
}

// ------------------------------------------------------------------ predicate
export const SURVIVING = Object.freeze(["Survived", "NoCoverage"]);

/**
 * THE PREDICATE. A region is defended only if every mutant in it was Killed.
 * Any Survived / NoCoverage mutant is an UNDEFENDED finding, reported per
 * MUTANT (its replacement text is the actionable thing), never aggregated
 * into a per-region score.
 */
export function classify(mutants) {
  const undefended = mutants.filter((m) => SURVIVING.includes(m.status));
  const killed = mutants.filter((m) => m.status === "Killed");
  const unknown = mutants.filter((m) => !SURVIVING.includes(m.status) && m.status !== "Killed");
  return { undefended, killed, unknown, defended: mutants.length > 0 && undefended.length === 0 && unknown.length === 0 };
}

export function findingLine(m) {
  return (
    `UNDEFENDED ${m.file}:${m.line}\n` +
    `  mutant [${m.mutator}] \`${m.original}\` -> \`${m.replacement}\` ${m.status}\n` +
    `  ran ${m.ranTestFiles ?? "?"} of ${m.baselineTestFiles ?? "?"} baseline-GREEN test file(s) reachable from this file; none failed.\n` +
    `  Nothing in the tree notices if this changed region is undone.`
  );
}

/**
 * The one line this leg is allowed to print as its answer. `ok` is not in the
 * vocabulary: a run that verified nothing says NOT VERIFIED and exits non-zero.
 */
export function summaryLine(s) {
  if (s.notVerified) return `defenders: NOT VERIFIED — ${s.notVerified}`;
  return `defenders: VERIFIED ${s.regions} region(s) / ${s.mutants} mutant(s) in ${s.files} file(s); ${s.undefended} UNDEFENDED`;
}

// ----------------------------------------------------------------------- CLI
function runVitest(root, file, extraArgs) {
  // `vitest related <src>` is the CANDIDATE SET: every test file whose module
  // graph reaches the changed file. A test cannot execute a line in a module it
  // does not import, so this is a strict SUPERSET of execution coverage —
  // scoping a mutant run to it can never manufacture a false UNDEFENDED. A
  // hand-listed testFiles (what stryker.config.mjs carries for its nine pinned
  // subjects) would silently under-run any other subject and do exactly that.
  const r = spawnSync(
    "bash",
    ["-o", "pipefail", "-c",
      `cd ${JSON.stringify(root)} && npx vitest related ${JSON.stringify(file)} --run --reporter=dot ${extraArgs} 2>&1 | grep -E "^ (FAIL|✓|❯)|Test Files|Tests  |Error:" | tail -60`],
    { encoding: "utf8", maxBuffer: 64 * 1024 * 1024 },
  );
  return { code: r.status, out: `${r.stdout ?? ""}${r.stderr ?? ""}` };
}

/** File-level `FAIL  <path>` lines: the test FILES that were red in this run. */
export function failingFiles(out) {
  const set = new Set();
  for (const line of String(out).split("\n")) {
    const m = /^\s*FAIL\s+(\S+\.test\.[cm]?[tj]sx?)/.exec(line);
    if (m) set.add(m[1]);
  }
  return [...set];
}

function parseVitestTail(out) {
  const files = /Test Files\s+(?:(\d+) failed \|)?\s*(\d+) passed(?:\s*\|\s*(\d+) skipped)?\s*\((\d+)\)/.exec(out);
  const failed = /Test Files\s+(\d+) failed/.test(out);
  return { failed, ranTestFiles: files ? Number(files[4]) : null };
}

/**
 * BASELINE. A mutant run that goes red is only a KILL if the same candidate
 * set was GREEN unmutated. The first build of this leg skipped that and read
 * a scratch tree whose workspace symlinks did not resolve — 166/166 test files
 * failed to LOAD — as "Killed", i.e. as "this change is defended". That is the
 * exact failure shape this leg exists to catch, in the leg itself: a red that
 * proves nothing, reported as a pass. So the baseline is mandatory and a
 * non-green baseline is NOT VERIFIED, never a kill.
 */
function baselineFor(root, file, fileSha, cachePath) {
  let cache = {};
  try { cache = JSON.parse(readFileSync(cachePath, "utf8")); } catch { cache = {}; }
  // The baseline is a property of BOTH the source file and the test tree: the
  // known-positive experiment differs from its control only by one test file
  // being absent, and a cache keyed on the source alone would hand the second
  // run the first run's answer. The key therefore carries the test tree's git
  // index state too; an unreadable index makes the entry uncacheable, never
  // silently reusable.
  let treeKey = null;
  try {
    treeKey = sha(execFileSync("git", ["--no-optional-locks", "ls-files", "-s", "--", "*.test.ts", "*.test.tsx"], { cwd: root, encoding: "utf8", maxBuffer: 64 * 1024 * 1024 })).slice(0, 16);
  } catch { treeKey = null; }
  const key = `${file}@${fileSha}@${treeKey ?? "uncacheable"}`;
  if (treeKey === null) cache = {};
  if (cache[key]) return cache[key];
  // No --bail here: the baseline must enumerate EVERY already-red test file,
  // not stop at the first. Those files are then excluded from the mutant runs,
  // so a mutant is Killed only when a test file that was GREEN unmutated turns
  // red. Demanding a wholly green tree instead would make the leg unusable on
  // any real repo and, worse, tempt a later room to drop the baseline entirely.
  const { out } = runVitest(root, file, "");
  const parsed = parseVitestTail(out);
  const failing = failingFiles(out);
  const total = parsed.ranTestFiles;
  const greenFiles = total == null ? null : total - failing.length;
  const rec = { failing, ranTestFiles: total, greenFiles, tail: out.split("\n").slice(-4).join(" | ") };
  cache[key] = rec;
  try { mkdirSync(dirname(cachePath), { recursive: true }); writeFileSync(cachePath, JSON.stringify(cache, null, 1)); } catch { /* cache is an optimisation, never a result */ }
  return rec;
}

  // unrecognised flag is refused rather than dropped (QUEUE #432) — this leg

  let deletions = [];

  try {

  } catch (e) {
    console.log(summaryLine({ notVerified: e.message }));
    process.exit(1);
  }

  const byFile = new Map();
  for (const r of regions) {
    if (!byFile.has(r.file)) byFile.set(r.file, []);
    byFile.get(r.file).push(r);
  }

  const noMutants = [];
  for (const [file, ranges] of byFile) {
    if (!existsSync(join(root, file))) { noMutants.push(`${file} — not present in ${root}`); continue; }

    if (!ms.length) noMutants.push(`${file}:${ranges.map((r) => `${r.start}-${r.end}`).join(",")} — no mutants generated (type-only, import or comment change)`);

  const shardSpec = opt("--shard", null);

    shard = { i, n };

    process.exit(1);

  for (const file of byFile.keys()) {
    const text = readFileSync(join(root, file), "utf8");
    originals.set(file, { text, sha: sha(text) });
  }

  // first build of this leg was killed mid-mutant and left `if (false) {` in
  // the source; the NEXT invocation then read the mutated file as its own
  // baseline and derived mutants from it. That is the hazard
  // scripts/mutate-file.mjs's leak scan exists for (an interrupted inPlace run
  // left `// @ts-nocheck` in 773 files), reproduced here. So: every original is
  // written to a sentinel BEFORE the first mutation, restored on any exit path
  // including SIGTERM/SIGINT, and a run that finds a stale sentinel whose
  // recorded sha does not match the file on disk REFUSES rather than measuring
  // a tree it cannot vouch for.
  const sentinelPath = join(root, ".scratch", "defenders-inflight.json");

    if (stale) {
      for (const [f, rec] of Object.entries(stale.files ?? {})) {
        const now = existsSync(join(root, f)) ? sha(readFileSync(join(root, f), "utf8")) : null;

          originals.set(f, { text: rec.text, sha: rec.sha });
        }

#!/usr/bin/env node
// HR-6 — every row Jason reads opens in plain language.
//
// The check: a row's "what" cell must OPEN with a sentence a reader outside this
// codebase could understand — before any backticked symbol, path, or line number.
//
// SCOPING BY DATE (QUEUE #576, D-0767): HR-6(e) makes this binding on rows filed after
// 2026-09-22 and advisory on the backlog. A row's filed date is read from the sidecar
// keyed by its id (`org/01-current/QUEUE-FILED.jsonl`, `org/08-control/DOCKET-FILED.jsonl`,
// written by `scripts/filed-dates.mjs`), never from its id: the id cutoff this replaced
// (#575) was a proxy. A row the sidecar does not date is scored BINDING and named, so a
// gap in the sidecar fails toward the rule; a missing sidecar file exits 2.
const BINDING_AFTER = '2026-09-22';
const SIDECARS = {
  'org/01-current/QUEUE.md': 'org/01-current/QUEUE-FILED.jsonl',
  'org/01-current/EXCEPTION-DOCKET.md': 'org/08-control/DOCKET-FILED.jsonl',
};

import fs from 'node:fs';
import path from 'node:path';

const args = process.argv.slice(2);
const rootArg = args.findIndex((a) => a === '--root');
// --root is REQUIRED and has no default. check-orchestrator-compliance.mjs shipped
// twice with a cwd default and twice reported a confident ZERO from the wrong tree.
if (rootArg === -1 || !args[rootArg + 1]) {
  console.error('check-eli5: --root <repo> is required (no default; see D-0669)');
  process.exit(2);
}
const ROOT = path.resolve(args[rootArg + 1]);
const strict = args.includes('--strict'); // fail on backlog too

/** A first sentence fails HR-6(a) if it leans on the tree to be understood. */
function eli5Failure(cell) {
  const text = cell.trim();
  if (!text) return 'empty';
  // Strip leading bold/emphasis markers and severity tags — they are not the sentence.
  const body = text.replace(/^\*+/, '').replace(/^(HIGH|CRITICAL|MEDIUM|LOW)\s*[—-]\s*/i, '');
  // The opening sentence: up to the first sentence-ending period, or 240 chars.
  const m = body.match(/^[^.!?]{0,240}[.!?]/);
  const first = (m ? m[0] : body.slice(0, 240));
  if (/`[^`]+`/.test(first)) return 'opens with a code symbol';
  if (/\b[\w./-]+\.(ts|tsx|js|mjs|md|sql|json|swift|kt|yml|yaml)\b/.test(first)) return 'opens with a file path';
  if (/:\d{2,}/.test(first)) return 'opens with a line number';
  if (/\b(I3-\d+|FAMILY-\d+|INV-\d+|PS-\d+|D-\d{4})\b/.test(first)) return 'opens with an internal identifier';
  if (first.trim().length < 25) return 'no opening sentence';
  return null;
}

/** id -> filed date (YYYY-MM-DD) from a table's sidecar; exits 2 when the sidecar is absent. */
function filedDates(file) {
  const side = path.join(ROOT, SIDECARS[file]);
  if (!fs.existsSync(side)) {
    console.error(`check-eli5: ${SIDECARS[file]} not found under ${ROOT} (node scripts/filed-dates.mjs --derive)`);
    process.exit(2);
  }
  const out = new Map();
  for (const line of fs.readFileSync(side, 'utf8').split('\n')) {
    if (!line.trim()) continue;
    const r = JSON.parse(line);
    if (Number.isInteger(r.id) && typeof r.filed === 'string') out.set(r.id, r.filed);
  }
  return out;
}

function rowsOf(file, idCol, whatCol) {
  const full = path.join(ROOT, file);
  if (!fs.existsSync(full)) {
    console.error(`check-eli5: ${file} not found under ${ROOT}`);
    process.exit(2);
  }
  const out = [];
  for (const line of fs.readFileSync(full, 'utf8').split('\n')) {
    if (!line.startsWith('|')) continue;
    const cells = line.split('|');
    if (cells.length < 5) continue;
    const id = cells[idCol]?.trim();
    if (!/^\d+$/.test(id)) continue;
    out.push({ id: Number(id), what: cells[whatCol] ?? '' });
  }
  return out;
}

let binding = 0, backlog = 0, ok = 0;
const bindingFails = [], backlogFails = [];

for (const [file, idCol, whatCol] of [
  ['org/01-current/QUEUE.md', 1, 4],
  ['org/01-current/EXCEPTION-DOCKET.md', 1, 2],
]) {
  const filed = filedDates(file);
  for (const row of rowsOf(file, idCol, whatCol)) {
    const fail = eli5Failure(row.what);
    if (!fail) { ok++; continue; }
    const day = filed.get(row.id);
    const rec = `${file.split('/').pop()} #${row.id} — ${fail}${day ? '' : ' (no filed date in the sidecar)'}`;
    if (day === undefined || day > BINDING_AFTER) { binding++; bindingFails.push(rec); }
    else { backlog++; backlogFails.push(rec); }
  }
}

console.log(`check-eli5 (HR-6): ${ok} rows open in plain language.`);
console.log(`  binding failures (filed after HR-6): ${binding}`);
console.log(`  backlog failures (filed before HR-6, advisory): ${backlog}`);
for (const f of bindingFails) console.log(`  FAIL  ${f}`);
if (backlog && (strict || process.env.ELI5_SHOW_BACKLOG)) {
  for (const f of backlogFails) console.log(`  note  ${f}`);
}
if (binding > 0 || (strict && backlog > 0)) {
  console.log('check-eli5: FAILED');
  process.exit(1);
}
console.log('check-eli5: OK');
