#!/usr/bin/env node
/**
 * Derive `services/trust-core/src/compound-words.ts` — the common-word vocabulary
 * `domain-gate.ts` uses to tell a LAUNDERING occurrence from a SEAM COLLISION.
 *
 * WHY THIS EXISTS. The gate refuses a domain whose label CONTAINS a listed
 * brand. That rule is right for `netflixsupport` and wrong for `youtubereviews`,
 * where `uber` exists only because `youtube` ends in `ube` and `reviews` begins
 * with `r`. Slice 33 measured the cost of not telling those apart and found the
 * threshold it shipped was a FLOOR on the false-positive count, not a ceiling —
 * `visa` was measured at 74 refused dictionary words and refuses thousands of
 * two-word compounds. A creator was refused `youtubereviews.com` because Uber is
 * listed.
 *
 * The fix specified at the time was to ANCHOR containment for short brands.
 * Measuring it against a generated corpus killed it: on 15,525 generated
 * `prefix+brand+suffix` laundering labels, anchoring blocks **0%**. It is the
 * one shape anchoring cannot see.
 *
 * What works is asking about the OCCURRENCE rather than the brand: can the label
 * be segmented into common words such that a word boundary falls STRICTLY INSIDE
 * the brand, with at least one of the two words reaching outside it? If so, the
 * brand is an artefact of the seam and not a name anybody wrote.
 *
 *   youtube│reviews   → the boundary splits `uber`, `youtube` starts before it
 *                       → a collision, cleared
 *   visa│support      → the boundary is at the brand's own edge, not inside it
 *                       → `visa` is a word somebody chose, blocked
 *   master│card       → the boundary splits the brand but NEITHER word reaches
 *                       outside it → the brand is its own compound, blocked
 *
 * Measured on 107,292 generated cases (91,767 two-common-word compounds whose
 * seam creates a brand; 15,525 prefix+brand+suffix laundering labels), against
 * only the brands that reach the containment arm:
 *
 *     rule                             seam FPs cleared   laundering blocked
 *     no test at all (before this)                 0.0%             100.00%
 *     anchored containment                       100.0%               0.00%
 *     interior minWord 3                          97.9%              97.89%
 *     interior minWord 4                          89.5%              99.81%
 *     minWord 4 + edge 3, straddle as `||`       100.0%              99.54%
 *     minWord 4 + edge 3, straddle as `&&`  →     96.3%              99.98%  ← shipped
 *
 * The last two rows are the correction adversarial review forced: the `||` form
 * shipped for an hour and was broken with a plural (`master│cards│hop`). See the
 * straddle clause in the generated file for the full argument.
 *
 * HOW TO RE-RUN IT:
 *
 *   npm install most-common-words-by-language     # in a scratch directory
 *   node scripts/derive-compound-words.mjs <path-to-english.txt> [minWord]
 *
 * The list is FREQUENCY-RANKED, and that is the whole point: it is the
 * vocabulary people actually compound into domain names. An alphabetical
 * dictionary is the wrong corpus here — it admits `visaac` and `tvisa` as
 * "compounds" and would clear laundering that no reader would call accidental.
 */
import { createHash } from "node:crypto";
import { readFileSync, writeFileSync } from "node:fs";

const listPath = process.argv[2];
const minWord = Number(process.argv[3] ?? 4);
// Words this short are admitted ONLY as the first or last segment of a label.
// See the measurement in the header: `you│tube│reviews` and `creative│bay` both
// need one, and both put it at an edge; a three-letter word in the INTERIOR is
// what manufactures spurious readings (`thee│bay│support`).
const edgeWord = 3;
if (!listPath) {
  console.error("usage: node scripts/derive-compound-words.mjs <english.txt> [minWord]");
  process.exit(2);
}
if (!Number.isInteger(minWord) || minWord < edgeWord) {
  // Not `>= 2`: a minWord below the edge floor emits a header claiming an
  // interior floor that `admits()` does not enforce, which is a generated file
  // that lies about itself.
  console.error(`minWord must be an integer >= edgeWord (${edgeWord}), got ${process.argv[3]}`);
  process.exit(2);
}

const rawText = readFileSync(listPath, "utf8");
// THE CORPUS IS PART OF THE RULE, so it is identified rather than assumed. Two
// people running the documented command against different releases of the same
// npm package would otherwise ship two different gates, and nothing but
// `COMPOUND_WORD_COUNT` would notice (adversarial review, MEDIUM).
const corpusSha = createHash("sha256").update(rawText).digest("hex");
const raw = rawText.split("\n").map((w) => w.trim().toLowerCase().replace(/[^a-z0-9]/g, ""));
// Words shorter than `edgeWord` can never be used by the segmentation, so
// shipping them is bytes for nothing. This is why the blob is smaller than the
// list it comes from.
const words = [...new Set(raw.filter((w) => w.length >= edgeWord))].sort();
const blob = words.join(" ");
const longest = words.reduce((m, w) => Math.max(m, w.length), 0);

const out = `// GENERATED by scripts/derive-compound-words.mjs — DO NOT EDIT BY HAND.
//
// corpus:   most-common-words-by-language / english.txt
// corpus sha256: ${corpusSha}
// corpus lines:  ${raw.length}
// minWord:  ${minWord} in a label's interior, ${edgeWord} as its first or last segment
// words:    ${words.length}
//
// Re-derive with:
//   node scripts/derive-compound-words.mjs <english.txt> ${minWord}
//
// WHAT THIS IS. The vocabulary \`domain-gate.ts\` segments a domain label against,
// to tell a laundering occurrence of a brand from a SEAM COLLISION between two
// ordinary words. See the script's header for the rule and the measurement that
// chose it.
//
// THE CORPUS IS FREQUENCY-RANKED AND TRUNCATED; THIS FILE IS ITS SORTED,
// DEDUPLICATED PROJECTION. An earlier version of this comment claimed the file
// itself was frequency-ranked, which it plainly is not — it is alphabetical, and
// the generator sorts the rank away because a Set does not care. What the rank
// buys is WHICH words are here at all: the corpus is truncated to the ten
// thousand most common, and that truncation is the point.
//
// A bigger, alphabetical dictionary was measured rather than assumed. Swapping
// in the 76,240-stem hunspell en_US list does NOT clear any of the laundering
// roster — the failure is on the FALSE-POSITIVE half, where obscure stems make
// spurious readings and stop explaining the collisions this exists for. The
// corpus sha256 above and the guard's roster are what say whether a swap changed
// an answer that matters.
const BLOB =
  ${JSON.stringify(blob)};

/** The vocabulary, built ON FIRST USE.
 *
 * Lazy because \`index.ts\` reaches this module through the route table, so an
 * eager Set would be built at isolate startup for every cold start — including
 * \`/health\` and \`/v1/bucket/:prefix\`, which never segment anything. The domain
 * -add verify path is the only caller, and it is the only one that should pay. */
let VOCAB: Set<string> | null = null;
function vocab(): Set<string> {
  if (VOCAB === null) VOCAB = new Set(BLOB.split(" "));
  return VOCAB;
}

/** How many words the vocabulary holds — pinned by the guard, so a hand-edit or
 * a silent re-derivation against a different corpus is visible. */
export const COMPOUND_WORD_COUNT = ${words.length};

/** The shortest word the segmentation will use. */
export const COMPOUND_MIN_WORD = ${minWord};

/** The shortest word admitted as a label's FIRST or LAST segment.
 *
 * Measured, on 107,292 generated cases: allowing three-letter words everywhere
 * takes laundering blocked from 99.81% down to 97.89%, because a short word in
 * the INTERIOR manufactures readings that explain away real laundering
 * (\`thee│bay│support\`). Allowing them only at the edges costs 0.34 points of
 * laundering and buys 8.4 points of false positives — \`you│tube│reviews\` and
 * \`creative│bay\`, the two cases the previous slice shipped as its worst
 * examples, are both of exactly this shape. */
export const COMPOUND_EDGE_WORD = ${edgeWord};

/** The longest word considered at any position — DERIVED from the vocabulary,
 * not asserted about it. A hard-coded bound below the longest entry ships a word
 * the segmentation can never use, silently (adversarial review, LOW). */
const MAX_WORD = ${longest};

/** Is \`label[q, p)\` a usable segment? Full length in the interior, \`edgeWord\` at
 * either end of the label. */
function admits(W: ReadonlySet<string>, label: string, q: number, p: number, n: number): boolean {
  const len = p - q;
  if (len < COMPOUND_EDGE_WORD) return false;
  if (len < COMPOUND_MIN_WORD && q !== 0 && p !== n) return false;
  return W.has(label.slice(q, p));
}

/**
 * Analyse one label ONCE, and return a predicate over occurrences within it.
 *
 * The two passes are the expensive part and they depend only on the label, so a
 * caller that tests many brands against one domain pays for them once. That is
 * the shape \`domainAddGate\` needs: one target, thousands of listed brands.
 *
 *   back[p] = the start index of a word ENDING at p, on some segmentation of
 *             label[0..p) into vocabulary words; -1 when there is none.
 *   fwd[p]  = the end index of a word STARTING at p, on some segmentation of
 *             label[p..n); -1 when there is none.
 *
 * THE SCAN DIRECTION IS LOAD-BEARING AND THE HEDGE THAT USED TO BE HERE
 * UNDERSOLD IT. \`back[p]\` scans \`q\` UPWARD and breaks, so it stores the MINIMUM
 * valid \`q\` — the longest word ending at p. \`fwd[p]\` scans downward and stores
 * the MAXIMUM. Because the straddle test is an existential over the words
 * meeting at p, those two extremes are exactly its witnesses, which makes the
 * predicate EXACT rather than approximate: verified against a complete
 * enumeration of every segmentation over 25,260 random labels, zero
 * disagreements.
 *
 * Reverse either loop to shortest-word-first — a natural "cheap words first"
 * optimisation — and reachability is unchanged, so every test still passes while
 * the analyzer silently becomes incomplete (\`discordless\` vs \`discord\` is the
 * divergence, and it is pinned in the guard). Do not reorder these loops.
 */
export function makeSeamTest(): (target: string) => (lo: number, hi: number) => boolean {
  // THE MEMO LIVES HERE, not in the caller. \`seamAnalyzer\` costs ~17µs per label
  // and the gate tests thousands of brands against a handful of target strings,
  // so a caller that forgets to memoize turns a 7ms scan into a 300ms one — one
  // arm's entire CPU budget. The contract used to be a sentence in a docblock
  // and the only example in the repo (a test) violated it, which is how the next
  // caller would have learned the wrong idiom (adversarial review, LOW).
  const memo = new Map<string, (lo: number, hi: number) => boolean>();
  return (target) => {
    let a = memo.get(target);
    if (!a) {
      a = seamAnalyzer(target);
      memo.set(target, a);
    }
    return a;
  };
}

export function seamAnalyzer(label: string): (lo: number, hi: number) => boolean {
  const W = vocab();
  const n = label.length;
  const back = new Int16Array(n + 1).fill(-1);
  back[0] = 0;
  for (let p = 1; p <= n; p++) {
    for (let q = Math.max(0, p - MAX_WORD); q < p; q++) {
      if (back[q] !== -1 && admits(W, label, q, p, n) ) {
        back[p] = q;
        break;
      }
    }
  }
  const fwd = new Int16Array(n + 1).fill(-1);
  fwd[n] = n;
  for (let p = n - 1; p >= 0; p--) {
    for (let q = Math.min(n, p + MAX_WORD); q > p; q--) {
      if (fwd[q] !== -1 && admits(W, label, p, q, n)) {
        fwd[p] = q;
        break;
      }
    }
  }
  return (lo: number, hi: number): boolean => {
    for (let p = lo + 1; p < hi; p++) {
      // STRICTLY INTERIOR. A boundary at the occurrence's own edge means the
      // brand IS one of the words — \`visa│support\` — which is a name somebody
      // chose, not a collision.
      if (back[p] === -1 || fwd[p] === -1) continue;
      // STRADDLE, AND IT IS A CONJUNCTION. Both words meeting at the seam must
      // reach OUT of the brand, because that is what a collision IS: two words
      // that each exist for their own reasons and happen to spell the brand
      // where they meet.
      //
      // It shipped for an hour as a DISJUNCTION — either word reaching out was
      // enough — and adversarial review broke it in one minute with a plural.
      // \`master│card\` is blocked, but \`master│cards│hop\` was CLEARED, because
      // the trailing \`s\` put the forward word one character past the brand and
      // the clause asked only "did anything cross at all". Eighteen realistic
      // laundering domains were self-addable through the live gate that way:
      // \`playstationshop\`, \`coinbaseshop\`, \`targetshop\`, \`getslacksupport\`,
      // \`gotwitchsupport\`, \`acerestore\`, \`microsoftware\`. One character of
      // overhang is not evidence of anything.
      //
      // THE ONE RELAXATION, and it is asymmetric on purpose. When the brand's
      // tail is the label's FINAL segment there is no word after it to reach
      // out — \`creative│bay\` is a real collision with nothing on its right — so
      // \`fwd[p] === n\` counts. The mirror (\`back[p] === 0\`) is NOT admitted: a
      // label that STARTS with the brand is the laundering shape, not the
      // collision shape, and admitting it re-opens \`mastercardshop\`.
      //
      // Measured over 97,133 seam labels and 54,720 laundering labels:
      //
      //     clause                          seam cleared   laundering blocked
      //     a || b            (shipped 1h)        100.0%              99.54%
      //     a && b                                 93.6%              99.98%
      //     a && (b || fwd===n)  ← now             96.3%              99.98%
      //
      // 3.7 points of false positives for a 23x reduction in escaped laundering.
      if (back[p]! < lo && (fwd[p]! > hi || fwd[p]! === n)) return true;
    }
    return false;
  };
}
`;

writeFileSync(new URL("../services/trust-core/src/compound-words.ts", import.meta.url), out);
console.log(`wrote services/trust-core/src/compound-words.ts — ${words.length} words at minWord ${minWord}, ${(blob.length / 1024).toFixed(1)}KB blob`);

#!/usr/bin/env node
/**
 * Derive `services/trust-core/src/generic-skeletons.ts` — the set of domain-label
 * skeletons too common in ordinary English to be treated as a brand.
 *
 * WHY THIS EXISTS. `domain-gate.ts` refuses a domain that CONTAINS a listed
 * brand's name, which is the rule that catches `netflixsupport.com`. Applied to a
 * short brand it is unusable: `meta` sits inside 457 English words, so protecting
 * Meta that way would refuse `metadata.io` and `metalworks.com` with a sentence
 * that explains nothing. The first version of the gate handled that with a LENGTH
 * floor — containment only for brands of five characters or more — and length
 * turned out to be a bad proxy for the thing that actually matters. Measured over
 * 274,937 English words: `visa` sits in 74 words, `nike` in 10, `ebay` in 8,
 * `xbox` and `espn` in 0, while five-character brands the gate already protected
 * sit in as many as 61 (`dream`) and 49 (`label`). The floor was refusing
 * protection to brands that would have cost less than the ones it granted.
 *
 * So the gate asks the real question instead — how common is this string in
 * ordinary English — and this script is where that question is answered, once,
 * offline, against a corpus.
 *
 * HOW TO RE-RUN IT:
 *
 *   npm install an-array-of-english-words        # in a scratch directory
 *   node scripts/derive-generic-skeletons.mjs <path-to-word-list.json> [threshold]
 *
 * The word list is any JSON array of words. It is deliberately NOT vendored into
 * this repository: it is 3.5MB of data that ships nowhere, changes nothing at
 * runtime, and would have to be reviewed on every bump. What IS vendored is the
 * derived set and, in the guard, the behavioural consequences — the roster of
 * real brands that must and must not be in it. If the corpus is ever swapped,
 * that guard is what says whether the swap changed any answer that matters.
 *
 * THE THRESHOLD IS A PRODUCT DECISION, NOT A TUNING PARAMETER. It sets the
 * ceiling on how many ordinary domains may be refused to protect one brand.
 * 250 was chosen deliberately: it protects `uber` (151) and `acer` (161), and of
 * 73 real brands measured it stop-lists exactly one, `meta` (457). See
 * `MANIFEST.md` for the alternatives that were on the table and what each cost.
 */
import { readFileSync, writeFileSync } from "node:fs";

const listPath = process.argv[2];
const threshold = Number(process.argv[3] ?? 250);
if (!listPath) {
  console.error("usage: node scripts/derive-generic-skeletons.mjs <word-list.json> [threshold]");
  process.exit(2);
}
if (!Number.isInteger(threshold) || threshold < 1) {
  console.error(`threshold must be a positive integer, got ${process.argv[3]}`);
  process.exit(2);
}

// The skeleton alphabet `labelSkeleton` produces: lowercase a-z0-9, everything
// else dropped. Words are folded the same way so the count means what the gate
// will actually compare.
const words = [
  ...new Set(
    JSON.parse(readFileSync(listPath, "utf8"))
      .map((w) => String(w).toLowerCase().replace(/[^a-z0-9]/g, ""))
      .filter((w) => w.length >= 3 && w.length <= 40),
  ),
];

// MIN is `GATE_MIN_BRAND`. MAX is DERIVED, not assumed: the set is complete only
// if no string longer than MAX would have qualified, and that depends on the
// threshold. At 250 nothing of length 10 qualifies (the highest is `ablenesses`,
// 237) — but a reviewer running the documented command at a lower threshold got
// a SILENTLY INCOMPLETE set, because 225 admits `ablenesses` and 150 admits five
// more. So the loop widens until a length yields nothing, and refuses to guess.
const MIN = 4;
const MAX_CEILING = 24;

// One pass over the corpus. `seen` is per-word so a word containing the same
// substring twice counts once: the question is "how many WORDS would this refuse",
// not "how many times does it occur".
const counts = new Map();
for (const w of words) {
  const seen = new Set();
  for (let len = MIN; len <= MAX_CEILING; len++) {
    // STRICT containment only. `gateMatch` treats an exact equality as `same`,
    // which is a different arm with a different justification, so a word that IS
    // the brand is not a cost of the containment rule.
    if (w.length <= len) continue;
    for (let i = 0; i + len <= w.length; i++) seen.add(w.slice(i, i + len));
  }
  for (const s of seen) counts.set(s, (counts.get(s) ?? 0) + 1);
}

const qualifying = [...counts].filter(([, n]) => n > threshold).map(([s]) => s);
// The longest length that actually qualifies at THIS threshold. Reported in the
// header so the set's completeness is a stated fact rather than an assumption,
// and asserted below so a ceiling that was too low can never ship quietly.
const MAX = qualifying.reduce((m, s) => Math.max(m, s.length), MIN);
if (MAX >= MAX_CEILING) {
  console.error(`threshold ${threshold} admits strings of length ${MAX_CEILING}; raise MAX_CEILING and re-run, or the set is incomplete.`);
  process.exit(1);
}

// The DENSITY around the switch, so whoever moves the threshold can see how much
// of the set is within a wobble of it. 74% of the shipped set sits in (200, 320]
// at threshold 250 — a corpus bump of a few percent moves a lot of strings.
const density = [0.5, 0.8, 1.0, 1.25, 2].map((f) => {
  const t = Math.round(threshold * f);
  return `${t}:${[...counts].filter(([, n]) => n > t).length}`;
});

const entries = qualifying.sort();
const blob = entries.join(" ");

const out = `// GENERATED by scripts/derive-generic-skeletons.mjs — DO NOT EDIT BY HAND.
//
// corpus:    an-array-of-english-words (${words.length} usable words)
// threshold: >${threshold} words strictly containing the string
// entries:   ${entries.length} (lengths ${MIN}-${MAX}, MAX derived — nothing longer qualifies)
// density:   entries by threshold — ${density.join("  ")}
//
// Re-derive with:
//   node scripts/derive-generic-skeletons.mjs <word-list.json> ${threshold}
//
// WHAT THIS IS. Domain-label skeletons too common in ordinary English to be used
// as a brand by \`domain-gate.ts\`'s containment arm. A brand in this set keeps
// identity and edit-distance matching and loses containment, because refusing
// every ordinary domain that happens to contain it costs more than the
// impersonation it prevents. Of 73 real brands measured, exactly one is in here.
//
// THE THRESHOLD IS THE PRODUCT DECISION and it is the ceiling on how many
// ordinary domains may be refused to protect one brand. Raising it protects more
// brands and refuses more honest creators; lowering it does the reverse. It is
// NOT a knob to turn when a particular brand is inconvenient — change it, re-run
// the script, and read what the guard says about the roster.
//
// Stored as one space-separated string rather than a Set literal: the same data
// costs ~${(blob.length / 1024).toFixed(0)}KB here against ~${((entries.length * 9) / 1024).toFixed(0)}KB as quoted array elements, and this file ships to a
// Worker. It is split once at module load.
const BLOB =
  ${JSON.stringify(blob)};

const GENERIC: ReadonlySet<string> = new Set(BLOB.split(" "));

/** How many entries the derived set holds — pinned by the guard, so a hand-edit
 * or a silent re-derivation against a different corpus is visible. */
export const GENERIC_SKELETON_COUNT = GENERIC.size;

/** The threshold this set was derived at, carried so the guard can state it. */
export const GENERIC_THRESHOLD = ${threshold};

/**
 * Is this skeleton too common in ordinary English to be treated as a brand?
 *
 * Consulted ONLY by the containment arm. Identity and edit distance stay on,
 * because neither has the runaway false-positive behaviour containment has: a
 * brand matches its own name exactly however common it is, and a bounded edit
 * ball around a four-character string is small.
 */
export function isGenericSkeleton(skeleton: string): boolean {
  return GENERIC.has(skeleton);
}
`;

writeFileSync(new URL("../services/trust-core/src/generic-skeletons.ts", import.meta.url), out);
console.log(`wrote services/trust-core/src/generic-skeletons.ts — ${entries.length} entries at threshold ${threshold}, ${(blob.length / 1024).toFixed(1)}KB blob`);

//   scripts/family-index.mjs:129 counts occurrences ONLY between `## Occurrences` and the next

//   node scripts/family-append.mjs --root <dir> --family FAMILY-NNN [--section <name>]
//        --row "a|b|c|d|e|f" [--row ...] [--dry-run] [--no-index]

import fs from 'node:fs';
import path from 'node:path';

export const SECTIONS = ['Occurrences', 'Sibling searches', 'Outcomes'];

const isTableLine = (l) => l.trim().startsWith('|');
const isSeparator = (l) => /^\s*\|[\s|:-]+\|\s*$/.test(l);

/** Split a markdown table line into trimmed cells. `\|` is a literal pipe inside a cell. */
export function rowCells(line) {
  const body = line.trim().replace(/^\|/, '').replace(/\|$/, '');
  const out = [];
  let cur = '';
  for (let i = 0; i < body.length; i++) {
    if (body[i] === '\\' && body[i + 1] === '|') { cur += '|'; i++; continue; }
    if (body[i] === '|') { out.push(cur.trim()); cur = ''; continue; }
    cur += body[i];
  }
  out.push(cur.trim());
  return out;
}

export const headerCells = rowCells;

/** Render cells back to a table line, escaping literal pipes so the round-trip is stable. */
export function buildRow(cells) {
  return `| ${cells.map((c) => String(c).replace(/\|/g, '\\|').trim()).join(' | ')} |`;
}

/**
 * Every `## ` section of a family log, with its table block located by LINE INDEX.
 * `rowStart..rowEnd` is the half-open range of data rows; `insertAt` is where a new row goes —
 * the end of THIS section's table, never the end of the file.
 */
export function parseSections(text) {
  const lines = text.split('\n');
  const heads = [];
  lines.forEach((l, i) => { if (/^##\s+\S/.test(l)) heads.push({ heading: l.replace(/^##\s+/, '').trim(), at: i }); });
  return heads.map((h, n) => {
    const end = n + 1 < heads.length ? heads[n + 1].at : lines.length;
    const tbl = [];
    for (let i = h.at + 1; i < end; i++) if (isTableLine(lines[i])) tbl.push(i);
    const headerLine = tbl.length ? lines[tbl[0]] : null;
    const hasSep = tbl.length > 1 && isSeparator(lines[tbl[1]]);
    const rowStart = tbl.length ? tbl[0] + (hasSep ? 2 : 1) : h.at + 1;
    const rowEnd = tbl.length ? tbl[tbl.length - 1] + 1 : h.at + 1;
    return {
      heading: h.heading, at: h.at, end,
      headerLine, hasSep,
      columns: headerLine && hasSep ? headerCells(headerLine).length : null,
      rowStart, rowEnd,
      insertAt: Math.max(rowStart, rowEnd),
      rows: tbl.filter((i) => i >= rowStart).map((i) => lines[i]),
    };
  });
}

export function sectionOf(text, heading) {
  return parseSections(text).find((s) => s.heading.toLowerCase() === String(heading).toLowerCase()) ?? null;
}

/** family-index.mjs:129's occurrence regex, VERBATIM — the count this script must move. */
export function occurrenceCountLikeIndex(logText) {
  const m = logText.match(/## Occurrences\n\|[^\n]*\n\|[-\s|]*\n([\s\S]*?)(?=\n## |\n?$)/);
  if (!m) return 0;
  return m[1].split('\n').filter((l) => l.trim().startsWith('|')).length;
}

export class AppendError extends Error {
  constructor(code, message) { super(message); this.code = code; }
}

/**
 * Pure. Splice `rows` into `heading`'s table. Throws AppendError rather than guessing.
 * Guarantees, all asserted here and not merely intended:
 *   G1 the rows land at the END OF THAT SECTION, before the next `## ` heading
 *   G2 each row's cell count equals THIS FILE's own header cell count for THIS section
 *   G3 every pre-existing line survives byte-identically and in order; delta == rows added
 */
export function appendRows(text, heading, rows) {
  if (!Array.isArray(rows) || rows.length === 0) throw new AppendError('NO_ROWS', 'no rows given');
  const sec = sectionOf(text, heading);
  if (!sec) throw new AppendError('NO_SECTION', `no "## ${heading}" section in this file`);
  if (!sec.headerLine || !sec.hasSep) {
    throw new AppendError('NO_HEADER',
      `"## ${heading}" has no header row + separator, so there is nothing to assert a row against — ` +
      `write the header first (this is QUEUE #421's root: 8 of 19 logs are in this state)`);
  }
  const want = sec.columns;
  // QUEUE #421a (I16-W01F): a count-correct row can still be SHIFTED — the `run/item` split across two
  // cells pushes the `vN` lesson version into `delivered to` (FAMILY-002.log.md:18's shape). That column
  // names who a lesson was delivered to, never a version, so a `vN` there is refused. The column is
  // located by THIS file's header, never by a hand index.
  const deliveredAt = headerCells(sec.headerLine).findIndex((c) => /^delivered to$/i.test(c.trim()));
  const lines = [];
  rows.forEach((r, i) => {
    const cells = Array.isArray(r) ? r : rowCells(r);
    if (cells.length !== want) {
      throw new AppendError('COLUMNS',
        `row ${i + 1} has ${cells.length} cells, "## ${heading}" header has ${want} ` +
        `(${headerCells(sec.headerLine).join(' | ')}) — refused, nothing written`);
    }
    if (deliveredAt >= 0 && /^v\d+\b/i.test(String(cells[deliveredAt] ?? '').trim())) {
      throw new AppendError('SHIFTED',
        `row ${i + 1} carries the lesson version ${JSON.stringify(String(cells[deliveredAt]).trim())} in "delivered to" ` +
        `(column ${deliveredAt + 1}) — the row is shifted left; refused, nothing written`);
    }
    lines.push(buildRow(cells));
  });

  const orig = text.split('\n');
  const before = orig.slice(0, sec.insertAt);
  const after = orig.slice(sec.insertAt);
  const next = [...before, ...lines, ...after];

  // G3, checked rather than assumed: delete exactly the inserted indices and demand the
  // original text back, byte for byte. A splice that dropped or reordered a line fails here.
  const restored = next.filter((_, i) => i < sec.insertAt || i >= sec.insertAt + lines.length);
  if (restored.join('\n') !== text) throw new AppendError('NOT_APPEND_ONLY', 'a pre-existing line did not survive the splice');
  if (next.length - orig.length !== lines.length) throw new AppendError('DELTA', `line delta ${next.length - orig.length} != rows added ${lines.length}`);

  return { text: next.join('\n'), added: lines.length, before: orig.length, after: next.length, rows: lines, insertAt: sec.insertAt };
}

// delivered to. It is the strongest evidence for `applies_to`, and in 7 of 19 logs it holds a
// `vN` literal because the row's author split `run/item` across two cells and the whole row
// shifted left by one. It is REGENERATED from source here, never hand-repaired.

// THE SOURCE IS THE COMPILED PROMPT ITSELF: org/07-findings/<batch>/<room>/PROMPT.md carries a
// `## ROOM CONTRACT (org/rooms/<persona>.md)` heading — the persona the compiler actually gave

// taken from the ROOM CONTRACT heading and NEVER from the directory name (FAMILY-022 P1).
const PROMPT_GLOB_DEPTH = 2;

export function readDeliveryRecord(root) {
  const base = path.join(root, 'org/07-findings');
  const rows = [];
  const unresolved = [];
  const batches = [];
  if (!fs.existsSync(base)) return { rows, unresolved, prompts: 0, batches };
  let prompts = 0;
  for (const batch of fs.readdirSync(base)) {
    const bdir = path.join(base, batch);
    if (!fs.statSync(bdir).isDirectory()) continue;
    batches.push(batch);
    for (const room of fs.readdirSync(bdir)) {
      const p = path.join(bdir, room, 'PROMPT.md');
      if (!fs.existsSync(p)) continue;
      prompts++;
      const text = fs.readFileSync(p, 'utf8');
      const rc = text.match(/^## ROOM CONTRACT \(org\/rooms\/([A-Za-z0-9_-]+)\.md\)/m);
      const rt = text.match(/^- ROOM TYPE:\s*([A-Za-z0-9_-]+)/m);

      // I34-TOOLS-C (B4 limb 4): a NOT-delivered shadow block is a record, not a delivery -- one predicate, shared.
      const fams = deliveredFamilyIds(text).sort();

/** Persona(s) actually delivered to `rooms` in `batchToken`, from the record. Never guessed. */
export function personaFor(record, batchToken, rooms) {
  // Derive the batch set from the DIRECTORY population, never from the rows that happen to
  // carry a family — otherwise a batch whose prompts were never persisted reports as "no batch
  // dir", which is a different and false claim than "that room has no compiled prompt".
  const batches = record.batches ?? [...new Set(record.rows.map((r) => r.batch))];
  const batch = batches.find((b) => b === batchToken) ?? batches.find((b) => b.startsWith(`${batchToken}-`)) ?? null;
  const found = [];
  const missing = [];
  for (const room of rooms) {
    const hits = record.rows.filter((r) => r.room === room && (batch ? r.batch === batch : true));
    if (hits.length === 0) { missing.push(room); continue; }
    for (const h of hits) if (!found.includes(h.persona)) found.push(h.persona);
  }
  return { batch, personas: found, missing };
}

/** Room tokens out of a shifted row's second cell: `Z01 / TAKEDOWN / REDIRECT (note)` -> 3 tokens. */

    .split('/')
    .map((s) => s.replace(/\(.*?\)/g, '').trim())
    .filter((s) => /^[A-Za-z0-9][A-Za-z0-9._-]*$/.test(s));
}

/**

export function regenerateOutcomes(text, record, familyId) {
  const lines = text.split('\n');
  const sec = sectionOf(text, 'Outcomes');
  if (!sec) return { text, changed: 0, notes: ['no ## Outcomes section'] };
  const notes = [];
  const body = [];
  for (let i = sec.at + 1; i < sec.end; i++) if (isTableLine(lines[i])) body.push(i);
  if (body.length === 0) return { text, changed: 0, notes: ['## Outcomes is empty'] };

  const first = rowCells(lines[body[0]]);
  const hadHeader = first.includes('delivered to');
  const hasSep = hadHeader && body.length > 1 && isSeparator(lines[body[1]]);
  const dataIdx = body.slice(hadHeader ? (hasSep ? 2 : 1) : 0);

  let changed = 0;
  const out = [];
  // Walk EVERY line of the section's table region, not only the table rows: a blank line that
  // separates two row groups is part of the record and dropping it is a loss like any other.
  const dataSet = new Set(dataIdx);
  const regionStart = hadHeader ? body[0] + (hasSep ? 2 : 1) : body[0];
  for (let i = regionStart; i <= body[body.length - 1]; i++) {
    if (!dataSet.has(i)) { out.push(lines[i]); continue; }
    const c = rowCells(lines[i]);
    if (!(c.length === 7 && /^v\d+(\.\d+)*$/.test(c[2]))) { out.push(lines[i]); continue; }
    const rooms = roomTokens(c[1]);
    const { batch, personas, missing } = personaFor(record, c[0], rooms);
    let delivered;
    if (personas.length > 0) {
      delivered = personas.join(' / ');
      if (missing.length) { delivered += ` (${missing.join(', ')}: no compiled prompt on disk)`; notes.push(`${familyId} row "${c[0]} ${c[1]}": ${missing.length} of ${rooms.length} room(s) have no PROMPT.md`); }
    } else if (rooms.length === 0) {
      delivered = `— (second cell "${c[1]}" names no room)`;
      notes.push(`${familyId} row "${c[0]} ${c[1]}": second cell is not a room token — delivered-to left UNDERIVABLE, not guessed`);
    } else {
      delivered = `— (UNDERIVABLE: no PROMPT.md for ${rooms.join(', ')}${batch ? ` under ${batch}` : ` and no batch dir matching ${c[0]}`})`;
      notes.push(`${familyId} row "${c[0]} ${c[1]}": UNDERIVABLE from the delivery record`);
    }
    out.push(buildRow([`${c[0]} ${c[1]}`, c[2], delivered, c[3], c[4], c[5], c[6]]));
    changed++;
  }

  // PRESERVE an existing header verbatim — several logs carry a RICHER `result (...)` column
  // label and regenerating over it would destroy a real distinction. Only a section with no
  // header at all gets the canonical one, and that is the only place this function invents text.
  const head = hadHeader
    ? (hasSep ? [lines[body[0]], lines[body[1]]] : [lines[body[0]], OUTCOMES_SEP])
    : [OUTCOMES_HEADER, OUTCOMES_SEP];

  else if (!hasSep) notes.push(`${familyId}: "## Outcomes" had a header with no separator — separator restored`);

  return { text: next.join('\n'), changed, notes, headerRestored: !hadHeader };
}

// A family's "opening date": the ISO date embedded in its `opened_by` frontmatter line if one is
// there (today only FAMILY-005 carries one: "opened_by: S-003 (orchestrator L0, 2026-09-07 ...)"),
// else the earliest commit date in ROOT's own git history that added that FAMILY-NNN.md (git log
// --follow --diff-filter=A --reverse — first line is the add commit). A file git cannot date at all
// (no git history reaching it — a bare test fixture with no commits) is UNRESOLVED and excluded from
// the count, named on stderr rather than silently dropped.
function familyOpenedAt(id, text) {
  const m = text.match(/^opened_by:\s*(.*)$/m);
  // REG-FAM-DAY: an opened_by line that names its own batch ("S-076 REG (...)") carries an
  // S-NUMBER, which ranks correctly even when two batches close/open on the same calendar day;
  // a bare day-string compare cannot.
  const sMatch = m && m[1].match(/\bS-(\d+)\b/);
  const embedded = m && m[1].match(/\d{4}-\d{2}-\d{2}/);
  if (embedded) return { date: embedded[0], sNum: sMatch ? Number(sMatch[1]) : null, instant: null, source: 'opened_by frontmatter (embedded date)' };
  try {
    const rel = path.relative(ROOT, path.join(DIR, `${id}.md`));
    // NOTE: --follow + --diff-filter=A + --reverse together return EMPTY on git 2.34.1 (a known
    // --follow/--reverse interaction, reproduced against the real repo — grepped for a git version
    // bump in the tree, none found) even though --follow + --diff-filter=A alone finds the add
    // commit. Newest-first output, oldest (add) commit is the LAST line — no --reverse needed.
    const out = execFileSync('git', ['-C', ROOT, 'log', '--follow', '--diff-filter=A', '--format=%aI', '--', rel], { encoding: 'utf8' }).trim();
    const lines = out.split('\n').filter(Boolean);
    const earliest = lines[lines.length - 1];
    // REG-FAM-DAY: keep the FULL instant (not just its first 10 chars) — the day-only slice is
    // still printed, but the cycle-count compare below needs the full timestamp to rank two
    // events on the same day without a same-day miss.
    if (earliest) return { date: earliest.slice(0, 10), sNum: null, instant: earliest, source: `git log --follow --diff-filter=A (${rel})` };
  } catch { /* fall through to unresolved */ }
  return { date: null, sNum: null, instant: null, source: 'unresolved — no embedded date, no git history' };
}

// Boundary = the previous batch's closure (D2, "say which"): the LARGEST `closed YYYY-MM-DD` date
// among org/01-current/CURRENT.md's `## S-NNN ... closed YYYY-MM-DD` headers — the most recently
// CLOSED batch on record, not the highest S-number (a higher-numbered batch can still be mid-flight
// with no closed header yet — true of S-075 as of this fix). --boundary overrides for testing.
function cycleBoundary(explicit) {
  if (explicit) return { date: explicit, sNum: null, instant: null, source: '--boundary' };
  const curPath = path.join(ROOT, 'org/01-current/CURRENT.md');
  if (!fs.existsSync(curPath)) throw new Error(`family-index --cycle-count: ${curPath} cannot be read and no --boundary was given`);
  const text = fs.readFileSync(curPath, 'utf8');
  const hits = [...text.matchAll(/^## S-(\d+)[^\n]*\bclosed\s+(\d{4}-\d{2}-\d{2})/gm)];
  if (hits.length === 0) throw new Error(`family-index --cycle-count: no "## S-NNN ... closed YYYY-MM-DD" header found in ${curPath} and no --boundary was given`);
  // REG-FAM-DAY: rank CLOSED batches by S-NUMBER, not the calendar date string — S-073..S-076 can
  // all close on one day, and picking the lexicographically-newest DATE among ties is arbitrary.
  const newest = hits.reduce((a, b) => (Number(b[1]) > Number(a[1]) ? b : a));
  let instant = null;
  try {
    // The full closure INSTANT (CURRENT.md's own latest git commit time here), for the git-log
    // fallback path below when a family carries no opened_by S-number of its own.
    instant = execFileSync('git', ['-C', ROOT, 'log', '-1', '--format=%aI', '--', path.relative(ROOT, curPath)], { encoding: 'utf8' }).trim() || null;
  } catch { /* no git history for CURRENT.md here (e.g. a bare test fixture) — date-only fallback below */ }
  return { date: newest[2], sNum: Number(newest[1]), instant, source: `org/01-current/CURRENT.md "## S-${newest[1]} ... closed ${newest[2]}"` };
}

function parseFrontmatter(text) {
  const m = text.match(/^---\n([\s\S]*?)\n---\n/);
  if (!m) return {};
  const fm = {};
  for (const line of m[1].split('\n')) {
    const mm = line.match(/^([a-zA-Z_]+):\s*(.*)$/);
    if (mm) fm[mm[1]] = mm[2];
  }
  return fm;
}

function parseTriggers(raw) {
  if (!raw) return [];
  const m = raw.match(/\[(.*)\]/);
  if (!m) return [];
  return m[1].split(',').map(s => s.trim().replace(/^"|"$/g, '')).filter(Boolean);
}

function lessonVersion(text) {
  // the delivered block's own `version:` line (frontmatter `version: 1.0.0` is the FILE version, not the lesson's) — W2 fix 2026-09-11
  const m = text.match(/## Lesson[^\n]*\n(?:[^\n]*\n){0,2}?version:\s*(\d+)/);
  return m ? `v${m[1]}` : 'v0';
}

function occurrenceCount(logText) {
  const m = logText.match(/## Occurrences\n\|[^\n]*\n\|[-\s|]*\n([\s\S]*?)(?=\n## |\n?$)/);
  if (!m) return 0;
  const rows = m[1].split('\n').filter(l => l.trim().startsWith('|'));
  return rows.length;
}

function mechanismOneLine(text) {
  const m = text.match(/^# (FAMILY-\d{3}) — (.+)$/m);
  return m ? m[2] : '';
}

// REGB-1 (S-087 REGB, MEDIUM): compile-room-prompt.mjs's own tag matcher used a plain substring
// test (qLower.includes(trigger)), unlike the word-boundary regex below (W2 fix 2026-09-11) --
// "callback" wrongly hit FAMILY-003's "ack" trigger there. Exported as a pure, parameterized
// function -- no closure over `queryText`/module state -- so compile-room-prompt.mjs (and anyone
// else) can import the SAME matcher instead of re-deriving it and drifting again. Declared above
// the CLI body below (which still runs its own argv parsing, `help()`'s `process.exit`, and file
// I/O unconditionally at module scope when this file is the process ENTRY point -- see `isMain`
// below) so importing this one symbol never depends on, or triggers, any of that.
export function triggerHits(triggers, text) {
  const lower = text.toLowerCase();
  return triggers.filter(t => new RegExp('(^|[^a-z0-9])' + t.toLowerCase().replace(/[.*+?^${}()|[\]\\]/g, '\\$&') + '([^a-z0-9]|$)', 'i').test(lower));
}

// Selection used to be ONE channel (prose scored against prose), so which lessons

/** `triggers: ["a", "b"]` / `applies_to: [x, y]` -> ["a","b"]. One parser for all three fields. */
export function parseListField(raw) {
  if (!raw) return [];
  const m = raw.match(/\[(.*)\]/);
  if (!m) return [];

/**
 * CHANNEL 2 -- PERSONA. A family about GUARDS is relevant to a ROLE, not to a
 * scope text: every builder ships one, every attacker mutates one, the judge

export function personaHits(appliesTo, persona) {

  return (appliesTo || []).filter(a => String(a).trim().toLowerCase() === p);

/**
 * CHANNEL 3 -- PATH. A family that has previously occurred in `warnings.ts` is
 * delivered to a room whose OWNS list names `warnings.ts` -- because the defect
 * literally happened there, not because the words matched.
 * A declared path matches an OWNS entry when it is EQUAL to it, is a
 * path-SEGMENT suffix of it (`warnings.ts` declared, `services/trust-core/src/warnings.ts`
 * owned), or glob-matches it (`*` inside a segment, `**` across segments).
 * Segment-anchored on purpose: a bare substring test would make `ings.ts` a hit.
 */
export function pathHits(declared, ownsPaths) {
  const owns = (ownsPaths || []).map(o => String(o).replace(/^\.\//, '').trim()).filter(Boolean);
  if (!owns.length) return [];
  const hits = [];
  for (const d0 of declared || []) {
    const d = String(d0).replace(/^\.\//, '').trim();
    if (!d) continue;
    const isGlob = d.includes('*');
    const re = isGlob
      ? new RegExp('^(?:.*/)?' + d.split('/').map(seg =>
          seg === '**' ? '.*' : seg.replace(/[.+?^${}()|[\]\\]/g, '\\$&').replace(/\*/g, '[^/]*'),
        ).join('/') + '$')
      : null;
    for (const o of owns) {
      // A declared path ENDING IN `/` is a DIRECTORY prefix: it matches any OWNS entry
      // beneath it. The trailing slash is the whole line -- a bare `src` keeps the
      // exact/segment-suffix rule below and so never matches `a/src/b.ts` (S-094 REG-02,
      // `pathHits` docblock contract). Segment anchoring survives: `'/' + d` is
      // slash-terminated on both sides, so `ings.ts` is still not a hit.
      const isDir = !isGlob && d.endsWith('/');
      const hit = isGlob ? re.test(o)
        : isDir ? (o.startsWith(d) || o.includes('/' + d))
        : (o === d || o.endsWith('/' + d));
      if (hit) { hits.push(`${d} -> ${o}`); break; }
    }
  }
  return hits;
}

/**
 * The packet's OWNS list is the INPUT to channel 3: the backticked tokens under
 * an OWNS declaration, in any of the THREE spellings the corpus actually writes
 * (S-094-FX1 census over org/**.md -- counts are in that room's REPORT.md):
 *   `## OWNS ...`                            heading, 34 -- and ALL 34 are in a
 *                                            compiled PROMPT.md, i.e. in this
 *                                            channel's own OUTPUT, never in a
 *                                            SCOPE.md. 0 of 804 SCOPE.md.
 *   `**Owns (the ONLY files you may edit):**` label + bullet run, 106 -- the
 *                                            dominant PRODUCTION shape.
 *   `Owns: \`a\`, \`b\``                       inline label, 296 lines.
 * This docblock used to claim the heading was "the shape every scope packet
 * uses". It was the shape NONE of them used, and the parser matched only it, so
 * the channel had zero doors for a whole batch while its guard stayed green.
 * A packet with NO parseable OWNS section returns [] and channel 3 then
 * contributes nothing -- it never falls back to scanning the whole packet,
 * because "a path mentioned in prose" is the prose-scoring defect again.
 */
export function extractOwns(packetText) {
  const text = String(packetText || '');
  // I34-TOOLS-B (I33 B17 limb 2, D-1103(3)): the sections are the grant parser's, scripts/lib/room-grants.mjs ownsSections --
  // (A) the HEADING form, `#` to `######` OWNS to the next heading of any level (this read only `##+`), and (B)/(C) the
  // LABEL form: `**Owns (the ONLY files you may edit):**` with an EMPTY tail (body = the bullet/blank run that follows), or
  // `Owns: `a`, `b`` with a NON-EMPTY tail (body = the tail); between `Owns` and the colon ONLY a parenthetical is allowed
  // (S-094-FX1's census of the accepted spellings and the refused prose). The FILES (OWNS) line form is not read here.
  const secs = ownsSections(text);
  const bodies = ['heading', 'label'].flatMap((form) => secs.filter((s) => s.form === form).map((s) => s.body));
  const out = [];
  for (const body of bodies) {
    for (const t of body.matchAll(/`([^`\n]+)`/g)) {
      const tok = t[1].trim();
      if (/^[\w./@-]+$/.test(tok) && (tok.includes('/') || /\.[a-z0-9]+$/i.test(tok))) out.push(tok.replace(/^\.\//, ''));
    }
  }
  return [...new Set(out)];
}

      id: f.id,
      tag: tagHits(f),
      bm25: Number(bm25Norm.toFixed(3)),
      bm25Raw: Number(bm25Raw.toFixed(3)),
      match,
    };
  }).sort((a, b) => (b.tag.length - a.tag.length) || (b.bm25 - a.bm25));

  if (jsonOut) {
    console.log(JSON.stringify(rows, null, 2));
  } else {
    console.log('| family | tag hit | bm25 | bm25Raw | match |');
    console.log('|---|---|---|---|---|');
    for (const r of rows) {
      console.log(`| ${r.id} | ${r.tag.length ? r.tag.join(', ') : '—'} | ${r.bm25} | ${r.bm25Raw} | ${r.match ? 'yes' : '—'} |`);
    }
  }
}

//   deliveries   — rooms (a directory holding PROMPT.md under org/07-findings/<batch>/) whose PROMPT.md, or a LESSONS.md

/** Every PROMPT.md / REPORT.md under one batch directory, at any depth, through symlinked directories (stat follows
 *  the link); a directory reached twice by realpath is walked once. */
function walkBatch(dir, out, seen = new Set()) {
  let real;
  try { real = realpathSync(dir); } catch { return; }
  if (seen.has(real)) return;
  seen.add(real);
  let names;
  try { names = readdirSync(dir); } catch { return; }
  for (const name of names) {
    const p = join(dir, name);
    let st;
    try { st = statSync(p); } catch { continue; }
    if (st.isDirectory()) { if (!SKIP_DIRS.has(name)) walkBatch(p, out, seen); }
    else if (name === "PROMPT.md") out.prompts.push(p);
    else if (name === "LESSONS.md") out.lessons.push(p);
    else if (name === "REPORT.md") out.reports.push(p);
  }
}

function tableCells(line) {
  const t = line.trim();
  if (!t.startsWith("|")) return null;
  return t.replace(/^\|/, "").replace(/\|$/, "").split("|").map((c) => c.trim());
}
const isSeparator = (cells) => cells.length > 0 && cells.every((c) => /^:?-{3,}:?$/.test(c));
const isBlankRow = (cells) => cells.every((c) => c === "" || /^[—–-]+$/.test(c));

/** The `## Occurrences` rows of one log: [{ finding, runCell, cells }]. The run column is taken from the most recent
 *  header row (a row followed by a separator) inside the section; rows appended after a blank line keep that header,
 *  and a row before any header has no run cell. */
export function occurrenceRows(text) {
  const lines = text.split("\n");
  const rows = [];
  let inSection = false;
  let runIdx = -1;
  for (let i = 0; i < lines.length; i++) {
    const line = lines[i];
    if (/^## /.test(line)) { inSection = /^## Occurrences\b/.test(line); continue; }
    if (!inSection) continue;
    const cells = tableCells(line);
    if (!cells || isSeparator(cells)) continue;
    const next = tableCells(lines[i + 1] ?? "");
    if (next && isSeparator(next)) { runIdx = cells.findIndex((c) => /^run\b/i.test(c)); continue; }
    if (isBlankRow(cells)) continue;
    rows.push({ finding: cells[0], runCell: runIdx >= 0 ? cells[runIdx] : undefined, cells });
  }
  return rows;
}

/** A cell's batch directory: a token that IS a batch name, or a token that is the unique prefix of one
 *  (`RUN-0010` -> `RUN-0010-sweep-s002`). */
export function resolveBatch(text, batchNames) {
  if (typeof text !== "string") return null;
  for (const tok of text.replace(/[`*]/g, " ").match(/[A-Za-z0-9][A-Za-z0-9_.-]*/g) ?? []) {
    const t = tok.replace(/[.-]+$/, "");
    if (batchNames.has(t)) return t;
    const pre = [...batchNames].filter((b) => b.startsWith(`${t}-`));
    if (pre.length === 1) return pre[0];
  }
  return null;
}

export function familyOutcomes(root) {
  const FIND = join(resolve(root), "org/07-findings");
  if (!existsSync(FIND)) throw new Error(`${FIND} not found — pass --root <the tree whose org/07-findings to read>`);
  const batchNames = new Set();

  const lessonsByDir = new Map(files.lessons.filter((p) => promptDirs.has(dirname(p))).map((p) => [dirname(p), p]));

    }

#!/usr/bin/env node
// QUEUE #576 / D-0767 — the day a QUEUE or docket row was FILED, kept in a sidecar.
//
// Neither QUEUE.md nor EXCEPTION-DOCKET.md has a filed-date column, and D-0767 rules
// that none is added (every parser keys on column position). The date lives in two
// JSONL sidecars keyed by row id, one object per line: {"id": N, "filed": "YYYY-MM-DD",
// "source": "git" | "filed", "commit"?: "<sha>"}.
//
//   --derive  (once, from the REPO's history): a row's date is the committer date, in
//             the committer's own offset, of the FIRST commit whose diff ADDS a line
//             `| <id> |` to the file (git log --follow --reverse -p -U0, read-only,
//             --no-optional-locks). That is the day the row reached the repo, an upper
//             bound on the day it was written. A row in the file with no such commit
//             (filed since the last commit) is NOT given a date: it is listed on stderr
//             and left for --append, never guessed.
//   --append  (at filing, by the orchestrator): `--append queue|docket <id> [--date D]`
//             adds one line, source "filed", date D or today. Refuses an id already
//             present, and an id the table does not carry.
//   --check   every row id in each table has exactly one sidecar line with a
//             well-formed date; a missing, duplicated or malformed entry exits 1.
//
// --root is REQUIRED (no cwd default; D-0669). With --derive, --root is the repo whose
// history is read and --out the tree the sidecars are written into (refuses to
// overwrite an existing sidecar without --force). An unrecognised flag exits 2.
import { execFileSync } from "node:child_process";
import fs from "node:fs";
import path from "node:path";
import { fileURLToPath } from "node:url";
import { parseArgs } from "node:util";

export const TABLES = {
  queue: { table: "org/01-current/QUEUE.md", sidecar: "org/01-current/QUEUE-FILED.jsonl" },
  docket: { table: "org/01-current/EXCEPTION-DOCKET.md", sidecar: "org/08-control/DOCKET-FILED.jsonl" },
};
const DATE_RE = /^\d{4}-\d{2}-\d{2}$/;

/** The row id a table line carries, by the rule check-eli5 uses: a `|` line of at least five cells whose first cell is digits. */
export function rowIdOf(line) {
  if (!line.startsWith("|")) return null;
  const cells = line.split("|");
  if (cells.length < 5) return null;
  const id = cells[1]?.trim();
  return /^\d+$/.test(id ?? "") ? Number(id) : null;
}

export function tableIds(text) {
  const ids = [];
  for (const line of text.split("\n")) {
    const id = rowIdOf(line);
    if (id !== null) ids.push(id);
  }
  return ids;
}

/** First-adding commit per row id, from `git log --follow --reverse -p -U0` of one file. */
export function deriveFromHistory(repo, file) {
  const out = execFileSync(
    "git",
    ["--no-optional-locks", "-C", repo, "log", "--follow", "--reverse", "-p", "-U0", "--no-color", "--format=@@COMMIT %H %cI", "--", file],
    { encoding: "utf8", maxBuffer: 1 << 30 },
  );
  const first = new Map();
  let commit = null;
  for (const line of out.split("\n")) {
    if (line.startsWith("@@COMMIT ")) {
      const [, sha, iso] = line.split(" ");
      commit = { sha, day: iso.slice(0, 10) };
      continue;
    }
    if (!commit || !line.startsWith("+|")) continue;
    const id = rowIdOf(line.slice(1));
    if (id !== null && !first.has(id)) first.set(id, { id, filed: commit.day, source: "git", commit: commit.sha });
  }
  return first;
}

export function readSidecar(file) {
  if (!fs.existsSync(file)) return null;
  return fs
    .readFileSync(file, "utf8")
    .split("\n")
    .filter((l) => l.trim())
    .map((l, i) => {
      try {
        return JSON.parse(l);
      } catch {
        throw new Error(`${file}:${i + 1}: not JSON`);
      }
    });
}

    console.log(`filed-dates: ${name}: ${ids.length} row(s), ${rows.length} sidecar line(s)`);
  }
  console.log(bad ? "filed-dates: FAILED" : "filed-dates: OK");
  return bad ? 1 : 0;
}

// observation around it (JUDGE-0008). A command is recognised by the shapes this org actually
// runs: a node/npx/pnpm/vitest invocation, a `-t "name"` selector, or an explicit `<cmd> -> <obs>`.
//
// W3-02's judge found the sharp edge: this cell truncated an invocation MID-COMMAND, before the

// pointer to the guard that turned out to be the wave's systemic finding. A command cut short of
// its path is not shortened, it is destroyed: nobody can re-run it at any length. So EV_MAX is a
// budget for the OBSERVATION, never for the command. `keepWhole` holds a command intact even past
// the budget, and `cutsAPath` is the backstop for the general case: a truncation that would drop a
// path-shaped token the text still needs is refused and the cell keeps the whole token.
const EV_MAX = 160;
const PATHISH = /[\w.@-]*\/[\w./@-]*\.[A-Za-z][\w]{0,5}\b|\b[\w-]+\.(?:ts|tsx|js|mjs|cjs|json|kt|sql|md|sh|py)\b/g;
// Truncate `t` to at most `n` chars WITHOUT ending inside a path-shaped token: if the cut lands
// mid-path, back up to the token's start; if the token started at 0 (the whole budget is one
// path), keep the token whole and overrun the budget rather than emit an unusable fragment.
function cutKeepingPaths(t, n) {
  if (t.length <= n) return t;
  for (const m of t.matchAll(PATHISH)) {
    const start = m.index, end = start + m[0].length;
    if (start < n && end > n) return (start === 0 ? t.slice(0, end) : t.slice(0, start).trimEnd()) + "…";
  }
  return t.slice(0, n) + "…";
}
// The contract names the field `evidence` and every room in W3-02 wrote it as a STRING — but the
// contract's own JUDGE-0008 wording says "when a finding's `evidence.kind` is `executed`", which
// reads as the object shape rooms use for CLAIMS in REPORT.md, and the judge's packets say the
// same. A room writing `{kind, ref}` here is following the text it was given; a bare String() on
// it renders "[object Object]" and destroys the whole cell — worse than any truncation. Accept
// both, and prefer `ref` (the invocation) over any other member.
function evidenceText(raw) {
  if (raw == null) return "";
  if (typeof raw === "string") return raw;
  if (typeof raw !== "object") return String(raw);
  const ref = raw.ref ?? raw.invocation ?? raw.cmd ?? raw.command;
  if (ref) return String(ref);
  const vals = Object.values(raw).filter((v) => typeof v === "string" && v.trim());
  return vals.length ? vals.join(" · ") : "";
}
function evidenceCell(raw) {
  const ev = evidenceText(raw).replace(/\s+/g, " ").trim();
  if (!ev) return "";
  const esc = (t) => t.replace(/\|/g, "\\|");
  if (ev.length <= EV_MAX) return esc(ev);
  // An explicit `command -> observation` keeps the command WHOLE — at any length — and spends what
  // is left of the budget on the observation. A command over budget on its own still ships whole.
  const arrow = ev.match(/^(.*?)(?:\s*(?:->|→|=>)\s*)([\s\S]*)$/);
  if (arrow) {
    const cmd = arrow[1].trim(), obs = arrow[2].trim();
    const room = EV_MAX - cmd.length - 6;
    return esc(room > 8 ? `${cmd} → ${cutKeepingPaths(obs, room)}` : `${cmd} → …`);
  }
  // Otherwise find the invocation anywhere in the line and lead with it — again, whole.
  const inv = ev.match(/((?:npx |pnpm |node |timeout \d+ )?(?:npx )?(?:vitest|node|pnpm)[^;|]*?(?:-t\s+(?:"[^"]*"|'[^']*'))?)(?:[;|]|$)/);
  if (inv && inv[1] && inv[1].trim().length >= 12) {
    const cmd = inv[1].trim();
    const rest = ev.replace(cmd, "").replace(/^[\s;|]+/, "");
    const room = EV_MAX - cmd.length - 3;
    return esc(room > 8 ? `${cmd} · ${cutKeepingPaths(rest, room)}` : cmd);
  }
  return esc(cutKeepingPaths(ev, EV_MAX));
}

function tail(text, n = TAIL_LINES) {

// W6-A (ORG-01-batch, 2026-09-21) — ROOT CAUSE of the "0 failed beside 8 RED PARTS" line.
// A leg's body is written to the log as its TAIL only (n lines). For a GREEN vitest shard that is
// harmless: vitest's summary is the last thing in the captured body. For a RED shard it is fatal.
// The body is `${stdout}\n${stderr}\n${countBody}` and a red shard's stderr failure dump runs to
// hundreds of lines, so the 40-line window lands entirely INSIDE the dump and vitest's own
// `Tests  3 failed | 567 passed (570)` line — the only line in the whole block that carries the
// failed count — never reaches the log. --summarize then had nothing to read for that shard and
// added nothing, so seven red shards contributed 0 failed. Absence summed as zero.
// The repair is to KEEP the line that carries the count, not to infer the count from elsewhere:
// lines matching KEEP_LINES survive the truncation wherever they sit in the body, in their
// original order, never duplicated (the window and the keeps are merged as an index set).
// Capped, so a body that prints thousands of summary-shaped lines cannot blow the block up; the
// rescued set is capped at KEEP_CAP, keeping BOTH ends (see the split below).
const KEEP_CAP = 20;
const KEEP_LINES = [
  /^\s*Test Files\s+\S/,                     // vitest's per-shard file summary
  /^\s*Tests\s+\S/,                          // vitest's per-shard test summary (carries `N failed`)
  /^\s*==\s+totals:/,                         // --summarize's own totals line, mirrored into the log
  /^\s*TOTALS (UNRECONCILED|CONTRADICTION|PARTIAL)/, // and its reconciliation verdict
  /^\s*EXIT [01]:/,                           // and the gate verdict itself
  // ORG-02-R3 (#499): a NON-SHARD leg's own evidence-of-execution line. --summarize now reads
  // these to tell a refused leg from a clean one, so truncation destroying one would manufacture
  // a red on a LONG GREEN leg (the body is `${stdout}\n${stderr}`, and a chatty stderr can push
  // `OK:` out of the window) instead of catching a refusal. Rescued for the same reason the
  // `Tests` line is: the line that carries the verdict must survive the window that hides it.
  /^\s*OK:/,                                  // check-fidelity / check-invariants success line
  /^\s*FAIL:/,                                // check-invariants.mjs's fail() refusal line
  /^check-claims: /,                          // the claims leg's own verdict line
  /^\s*LEG UNRECONCILED/,                     // and --summarize's own per-leg reconciliation
];
function tailKeeping(text, n = TAIL_LINES, keep = KEEP_LINES) {
  const lines = text.split("\n");
  const start = Math.max(0, lines.length - n);
  const idx = new Set();
  for (let i = start; i < lines.length; i++) idx.add(i);
  const rescued = [];
  for (let i = 0; i < start; i++) if (keep.some((re) => re.test(lines[i]))) rescued.push(i);
  // Both ENDS, not just the last: the earliest rescued lines are a block's own header-ish
  // summary lines (`== totals:`), the latest are the ones nearest vitest's real trailer. Keeping
  // only one end would let a long body evict whichever end it pleased.
  const back = Math.floor(KEEP_CAP / 2); // NOT slice(-back) unguarded: slice(-0) returns EVERYTHING
  const rescuedKept = rescued.length <= KEEP_CAP
    ? rescued
    : [...rescued.slice(0, KEEP_CAP - back), ...(back > 0 ? rescued.slice(-back) : [])];
  for (const i of rescuedKept) idx.add(i);
  return [...idx].sort((a, b) => a - b).map((i) => lines[i]).join("\n");
}

function runCapture(cmd, args) {
  let stdout = "";
  let stderr = "";
  let code = 0;
  try {
    // S-092 gate0: Node's default 1 MB stdout buffer KILLED two shards whose combined output
    // reached ~1.6 MB (verbose registry logging), so the child died before writing its JSON report
    // and the leg read as RED while a raw run of the same command passed 576/576 and 444/444.
    // A gate leg must fail on the tests, never on how much its child printed.
    stdout = execFileSync(cmd, args, { cwd: process.cwd(), encoding: "utf8", stdio: ["ignore", "pipe", "pipe"], maxBuffer: 256 * 1024 * 1024 });
  } catch (e) {
    stdout = e.stdout ?? "";
    stderr = e.stderr ?? "";
    code = typeof e.status === "number" ? e.status : 1;
  }
  return { stdout, stderr, code };
}

// ORG-02-R3 (#499), found while probing this mode: the three labels above are interpolated into

// as a pattern, `^summarize (gate verdict)$` is a capture group and matches the literal
// "summarize gate verdict" — never the label itself. MEASURED: re-running --summarize over a log
// that already carried its own mirrored block printed
// `UNKNOWN BLOCKS: summarize (gate verdict)`, i.e. the gate accused ITSELF of being the "foreign
// writer (a tee, a hand edit, a merged log)" that KNOWN_HEADERS' comment describes. Escaped at
// the one place a label becomes a pattern, so a label may carry any character.
const reLit = (s) => s.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
function appendBlock(label, body, code, n = TAIL_LINES) {
  // P2-G1 (S-081 REG phase 2): raw captured leg output can itself contain a line starting with
  // "==" (a test named that way printing to console, or a hostile one) which --summarize's own
  // block-header regex (`/^==\s+/m`) cannot tell apart from a REAL header. Escaped at the one
  // place captured output enters the log, so it can never masquerade as a header again.
  const safeBody = body.replace(/^==/gm, "  ==");
  const block = `== ${label}\n${tailKeeping(safeBody, n).replace(/\n?$/, "\n")}exit ${code}\n`;
  const existing = existsSync(outFile) ? readFileSync(outFile, "utf8") : "";
  writeFileSync(outFile, existing.replace(/\n?$/, existing ? "\n" : "") + block);
}

// I10-GATE (D-0870(8)): the log parse and the shard reconciliation, lifted VERBATIM out of --summarize so
// --summarize, --prove-full and (through --prove-full) batch-close.mjs read a log through ONE implementation
// (FAMILY-004 P1). Bodies unchanged; only the enclosing function and its return are new.
function parseGateLog(text) {
  const blockRe = /^==\s+(.+)$/gm;

  // A block's own "exit N" line is the LAST one in its span (the exact trailer gate.mjs's own
  // appendBlock writes, always the final line of a real block, since nothing is written after it
  // until the next appendBlock call). P2-G1 (S-081 REG phase 2): the OLD non-global `.exec` took
  // the FIRST "exit N"-shaped line instead, so a leg whose captured output happened to contain an
  // earlier line merely ENDING in "exit N" (forged or coincidental) was read as that block's real
  // exit code. `^exit N$` (whole line, anchored) plus taking the LAST match makes gate.mjs's own
  // trailer the only thing this can ever read, since a forger can only write BEFORE it, never
  // after — appendBlock appends the real trailer itself, after capture, from the real exit code.
  const starts = [...text.matchAll(blockRe)];
  const parts = starts.map((m, i) => {
    const begin = m.index;
    const end = i + 1 < starts.length ? starts[i + 1].index : text.length;
    const body = text.slice(begin, end);
    const exitMatches = [...body.matchAll(/^exit (-?\d+)\s*$/gm)];
    const exitMatch = exitMatches.length ? exitMatches[exitMatches.length - 1] : null;
    return { header: m[1].trim(), exitCode: exitMatch ? Number(exitMatch[1]) : null, body };
  });

  // QUEUE #290: a re-run leg (a red typecheck fixed by a later re-run of the same mode, a shard
  // re-run after a flake) appends a SECOND block under the same header rather than replacing the
  // first — every downstream check below keeps only the LAST block per exact header (a shard's
  // header already embeds i/N, so this dedups "per shard label" for free); totals and red/missing
  // checks are computed from `deduped` only, never the raw `parts`.
  const lastIndexByHeader = new Map();
  const collapsedByHeader = new Map();
  parts.forEach((p, i) => {
    if (lastIndexByHeader.has(p.header)) collapsedByHeader.set(p.header, (collapsedByHeader.get(p.header) ?? 0) + 1);
    lastIndexByHeader.set(p.header, i);
  });
  const keptIdx = new Set(lastIndexByHeader.values());
  const deduped = parts.filter((_, i) => keptIdx.has(i));
  return { parts, collapsedByHeader, keptIdx, deduped };
}
function reconcileShards(deduped) {
  let filesTotal = 0;
  let passedTotal = 0;
  let skippedTotal = 0;
  let failedTotal = 0;
  // KM:KM-065 (W3-01-R3): vitest prints FAILURES FIRST — `Tests  3 failed | 567 passed (570)`. The
  // old regexes anchored on `passed` before any optional `failed` group, so they matched NOTHING on
  // exactly the shards that mattered: a red shard contributed 0 to every total and the line always
  // printed `0 failed` beside a correct RED PARTS line. Each count is read by its own word now,
  // from the summary line itself, in whatever order vitest chose.
  const countOn = (line, word) => { const m = new RegExp(`(\\d+)\\s+${word}\\b`).exec(line); return m ? Number(m[1]) : 0; };
  const totalOn = (line) => { const m = /\((\d+)\)\s*$/.exec(line.trim()); return m ? Number(m[1]) : 0; };
  // W6-A (ORG-01-batch, 2026-09-21): KM-065 fixed the REGEX. It did not fix the ARITHMETIC. On the
  // 2026-09-21 16-shard run this line still printed `495 files, 5765 passed, 3 skipped, 0 failed`
  // in the same output that listed 8 RED PARTS — 495 = 9 x 55 = the NINE GREEN shards, because the
  // seven red shards had no `Tests` line in the log AT ALL (appendBlock's tail had cut it: see
  // tailKeeping, :160) and a shard with no line to read contributed NOTHING rather than being
  // counted as UNKNOWN. A totals line that sums `passed` over the parts it can see, while silently
  // skipping the parts it cannot, reports the green subset as the whole run.
  //
  // It is a RECONCILIATION now, with the property a reconciliation has: PARTS COUNTED must equal
  // PARTS SUMMED. The two are derived independently — `partsCounted` from the block HEADERS in the
  // log, `partsSummed` from the BODIES that actually yielded both count lines — so they can be
  // checked against each other without the check being a tautology. Any gap is stated in the
  // output and reds the gate; it can never be dropped in silence. When the gap is non-zero the
  // line refuses to print a total at all: "UNKNOWN" is an honest output, `0 failed` over seven
  // unread shards is not.
  const shardParts = deduped.filter((p) => /^vitest shard \d+\/\d+$/.test(p.header));
  const partsCounted = shardParts.length;
  let partsSummed = 0;
  const unaccounted = [];
  for (const p of shardParts) {
    // LAST match, for the same reason the `exit N` trailer is read last (the `exitMatches`
    // read in this mode's block scan): a chatty or
    // hostile test can print a summary-shaped line BEFORE vitest's own trailer, never after it.
    const fileMatches = [...p.body.matchAll(/^\s*Test Files\s+\S.*$/gm)];
    const testMatches = [...p.body.matchAll(/^\s*Tests\s+\S.*$/gm)];
    if (fileMatches.length === 0 || testMatches.length === 0) {
      const why = testMatches.length === 0 ? 'no "Tests" line' : 'no "Test Files" line';
      unaccounted.push(`${p.header} (${why}, exit ${p.exitCode})`);
      continue;
    }
    const fileLine = fileMatches[fileMatches.length - 1][0];
    const testLine = testMatches[testMatches.length - 1][0];
    filesTotal += totalOn(fileLine);
    passedTotal += countOn(testLine, "passed");
    skippedTotal += countOn(testLine, "skipped");
    failedTotal += countOn(testLine, "failed");
    partsSummed++;
  }
  // Derived from the blocks' own exit trailers, NOT from any count parsed above — that
  // independence is the whole reason the cross-check below means anything.
  const redShardCount = shardParts.filter((p) => p.exitCode !== 0).length;
  const totalsProblems = [];
  // THE EMPTY CASE IS NOT A PASS. A reconciliation over zero parts is trivially satisfied, and an
  // empty or shard-less log is the likeliest input a fixture ever hands this code. `0 files,
  // 0 passed, 0 skipped, 0 failed` reads exactly like a clean run, so it is refused outright.
  if (partsCounted === 0) {
    totalsProblems.push('TOTALS UNRECONCILED: no "vitest shard i/N" part in this log — there is nothing to total, and 0 is not the answer');
  } else if (partsSummed !== partsCounted) {
    // One line PER unaccounted part, each carrying the loud prefix. A joined list is one line a
    // reader's eye slides over; a line per part is greppable and countable, and the count in the
    // header can be checked against the number of lines under it.
    totalsProblems.push(`TOTALS UNRECONCILED: parts counted ${partsCounted} != parts summed ${partsSummed} — ${unaccounted.length} shard part(s) carry no readable count:`);
    for (const u of unaccounted) totalsProblems.push(`TOTALS UNRECONCILED:   ${u}`);
  } else {
    // Both sides readable: the exit trailers and vitest's own text must now agree about whether
    // anything failed. They are two different measurements of one fact; disagreement is a defect
    // in the instrument and is stated, never averaged away.
    if (redShardCount > 0 && failedTotal === 0) {
      totalsProblems.push(`TOTALS CONTRADICTION: ${redShardCount} shard part(s) exited non-zero, but the summed "Tests" lines report 0 failed`);
    }
    if (redShardCount === 0 && failedTotal > 0) {
      totalsProblems.push(`TOTALS CONTRADICTION: every shard part exited 0, but the summed "Tests" lines report ${failedTotal} failed`);
    }
  }
  return { filesTotal, passedTotal, skippedTotal, failedTotal, partsCounted, partsSummed, totalsProblems };
}

// I10-GATE (QUEUE #456, D-0870(8)): `mode: full` is PROVED FROM THE LOG, never declared. It holds only when
// the log's vitest shard blocks share ONE N (the declared N when the caller gives one), every k in 1..N is
// present (last block per header, #290), no k is out of range, and reconcileShards() finds no problem (no
// UNKNOWN totals). Read by --summarize (which then emits the mode line itself) and by --prove-full (which
// batch-close.mjs runs) — one predicate, two doors.
function proveFull(text, declaredN = null) {
  const { deduped } = parseGateLog(text);
  const problems = [];
  const shards = deduped
    .map((p) => /^vitest shard (\d+)\/(\d+)$/.exec(p.header))
    .filter(Boolean)
    .map((m) => ({ k: Number(m[1]), n: Number(m[2]) }));
  const ns = [...new Set(shards.map((x) => x.n))].sort((a, b) => a - b);
  if (shards.length === 0) problems.push('no "vitest shard k/N" block in the log');
  if (ns.length > 1) problems.push(`the log's shard blocks carry ${ns.length} different N (${ns.map((n) => `k/${n}`).join(", ")}); one full run has one N`);
  if (declaredN !== null && ns.length > 0 && !(ns.length === 1 && ns[0] === declaredN)) problems.push(`the mode line declares ${declaredN} shard(s), but the log's shard blocks are ${ns.map((n) => `k/${n}`).join(", ")}`);
  const n = declaredN ?? (ns.length === 1 ? ns[0] : null);
  if (n !== null) {
    const seen = new Set(shards.filter((x) => x.n === n).map((x) => x.k));
    const missing = Array.from({ length: n }, (_, i) => i + 1).filter((k) => !seen.has(k));
    if (missing.length > 0) problems.push(`vitest shard(s) ${missing.join(", ")}/${n} missing from the log`);
    const outOfRange = [...seen].filter((k) => k < 1 || k > n).sort((a, b) => a - b);
    if (outOfRange.length > 0) problems.push(`vitest shard(s) ${outOfRange.join(", ")}/${n} are out of range 1..${n}`);
  }
  const totals = reconcileShards(deduped);
  for (const t of totals.totalsProblems) problems.push(t);
  // I10-FX3 (QUEUE #456, D-0871(10)): the POPULATION anchor. Every --shard run writes, after its captured
  // output, `-- suite population: M test file(s) (vitest list --filesOnly); this block: shard k/N`. The LAST
  // such line in a kept block is the gate's (a forger writes only before the trailer). Each block's own k/N
  // must equal its header (a relabelled 1/4 -> 1/2 block is refused), all blocks must record ONE M, and the
  // files the shards ran must sum to M (half a suite, or a one-file 1/1 log, is not "full").
  let population = null;
  if (n !== null) {
    const ms = new Set();
    for (const p of deduped) {
      const h = /^vitest shard (\d+)\/(\d+)$/.exec(p.header);
      if (!h || Number(h[2]) !== n) continue;
      const lines = [...p.body.matchAll(/^-- suite population: (\d+) test file\(s\) \(vitest list --filesOnly\); this block: shard (\d+)\/(\d+)\s*$/gm)];
      const pl = lines.length ? lines[lines.length - 1] : null;
      if (!pl) { problems.push(`${p.header} carries no "-- suite population" line (gate.mjs --shard writes one on every run), so its run cannot be tied to the suite`); continue; }
      if (Number(pl[2]) !== Number(h[1]) || Number(pl[3]) !== Number(h[2])) problems.push(`${p.header} was run as shard ${pl[2]}/${pl[3]} (its gate-written population line), not as its header says — a relabelled block`);
      ms.add(Number(pl[1]));
    }
    if (ms.size > 1) problems.push(`the shard blocks record ${ms.size} different suite populations (${[...ms].sort((a, b) => a - b).join(", ")}); one full run has one`);
    if (ms.size === 1) {
      population = [...ms][0];
      if (totals.totalsProblems.length === 0 && totals.filesTotal !== population) problems.push(`the log's shards ran ${totals.filesTotal} test file(s), but the suite collected ${population} (vitest list --filesOnly): not the whole suite`);
    }
  }
  return { proven: problems.length === 0, n, problems, totals, population };
}

// I10-GATE (D-0870(12)): per-file test timings. --shard APPENDS one row per test file vitest's JSON
// reporter named — {shard, file, duration_ms, tests, at} — to GATE-TIMINGS.jsonl BESIDE --out. It never
// rewrites the file (shards run in parallel; a read-modify-write would drop a sibling's rows): `at` stamps
// the run, and --summarize keeps only each shard's latest `at` (a re-run supersedes itself, as #290 does).
// duration_ms is the JSON report's endTime - startTime for the file (its tests' wall time, not collect).
// A report that cannot be read or a file that cannot be written leaves a line in the shard block, so a
// missing row is never silent; it never changes the shard's exit code (timings are advisory).
const timingsFile = () => join(dirname(outFile), "GATE-TIMINGS.jsonl");
// I27-PORT-GIT G4: process.cwd() is already a realpath (macOS: /private/var/...) while a report's
// file name may be spelled through a symlink (/var/...); relative() between the two spellings is a
// `../../..` path, not the repo-relative file. Both sides are resolved through their nearest
// existing ancestor (the file itself need not exist) before relative() is taken (scripts/lib/real-or-tail.mjs).

function recordTimings(spec, jsonPath) {
  let results;
  try {
    results = JSON.parse(readFileSync(jsonPath, "utf8"));
  } catch (e) {
    return `-- timings NOT RECORDED: could not read/parse ${jsonPath}: ${e?.message || e}\n`;
  }
  const at = new Date().toISOString();
  const rows = (results?.testResults ?? []).map((tf) => {
    const start = Number(tf.startTime);
    const end = Number(tf.endTime);
    return {
      shard: spec,
      file: relative(realOrTail(process.cwd()), realOrTail(String(tf.name))).split(sep).join("/"),
      duration_ms: Number.isFinite(start) && Number.isFinite(end) ? Math.max(0, Math.round(end - start)) : null,
      tests: (tf.assertionResults ?? []).length,
      at,
    };
  });
  try {
    if (rows.length > 0) appendFileSync(timingsFile(), rows.map((r) => `${JSON.stringify(r)}\n`).join(""));
  } catch (e) {
    return `-- timings NOT RECORDED: could not write ${timingsFile()}: ${e?.message || e}\n`;
  }
  return rows.length === 0 ? `-- timings: the vitest JSON report named 0 test files for shard ${spec}\n` : "";
}

const ORCH_COMPLIANCE_LABEL = "orchestrator compliance (HR-4)";
// #551b (D-0920 option 1): the population cutoff --summarize applies to the compliance block. The
// batch families below are every family under org/07-findings at i16-base that predates the cutoff
// (`ls org/07-findings | grep -- -batch$`, 2026-10-06); B<n> and I<n> are cut by number. Anything
// else binds. Driven end to end through --summarize by tests/tooling/i16-w02e-gate.test.ts.
const ORCH_CUTOFF_B = 28;
const ORCH_CUTOFF_I = 16;
const ORCH_HISTORY = [/^S-/, /^W3-/, /^RUN-/, /^ORG-0[1-4]-batch$/, /^RJ-550b?-batch$/, /^GATE-0923-batch$/];
function orchBatchBinds(name) {
  const b = /^B(\d+)-/.exec(name);
  if (b) return Number(b[1]) > ORCH_CUTOFF_B;
  const i = /^I(\d+)-/.exec(name);
  if (i) return Number(i[1]) > ORCH_CUTOFF_I;
  return !ORCH_HISTORY.some((re) => re.test(name));
}
// Reads the LAST compliance block's own FAIL listings (check-orchestrator-compliance.mjs prints each
// failing batch on a 7-space-indented line under its `FAIL HR-…` header) and splits them by the cutoff.
function orchSplit(part, logPath) {
  const out = { red: false, exitCode: part ? part.exitCode : null, binding: [], advisory: [], hr6: false, unattributed: false, inFlight: null, orphanRows: 0 };
  if (!part || part.exitCode === 0) return out;
  const m = logPath ? /(?:^|[\\/])org[\\/]07-findings[\\/]([^\\/]+)[\\/]/.exec(resolvePath(logPath)) : null;
  out.inFlight = m ? m[1] : null;
  let section = null;
  const batches = new Set();
  for (const line of part.body.split("\n")) {
    if (/^FAIL HR-6\b/.test(line)) { section = "hr6"; out.hr6 = true; continue; }
    if (/^FAIL HR-/.test(line)) { section = "batch"; continue; }
    if (/^(ok|FAIL)\s/.test(line) || /^\S/.test(line)) { section = null; continue; }
    const row = /^ {7}(\S+)\s/.exec(line);
    if (row && section === "batch") batches.add(row[1]);
    // A listing row under NO `FAIL HR-` header: the header was cut (a log tailed to 40 lines before the
    // fix below) or the script's shape moved. It cannot be attributed to history, so it binds.
    else if (row && section === null) out.orphanRows++;
  }
  for (const b of batches) (orchBatchBinds(b) && b !== out.inFlight ? out.binding : out.advisory).push(b);
  out.unattributed = part.exitCode !== 1 || (batches.size === 0 && !out.hr6) || out.orphanRows > 0;
  out.red = out.binding.length > 0 || out.hr6 || out.unattributed;
  return out;
}

      const filesMismatch = testFilesTotal !== g.files.length;
      const failed = [...out.matchAll(/^\s*(?:FAIL|❯)\s+(\S+\.test\.ts)/gm)].map((m) => basename(m[1]).replace(PREFIX, ""));
      const red = [...new Set(failed)];
      // QUEUE #271: a retired file carrying `live_tests` gets one targeted re-run (whole file,
      // json reporter — cheap, these files are a handful of its) so a failure among exactly
      // those titles still counts as LIVE RED; a failure elsewhere in the same file stays
      // retired. Only fires for a file already in `red` (something in it failed at all).
      const liveViaLiveTests = [];
      for (const fileName of red) {
        const liveTitles = liveTestsByFile.get(fileName);
        if (!liveTitles || !retiredSet.has(fileName)) continue;
        const planEntry = plan.find((q) => basename(q.src) === fileName && !!q.fidelity === (g.label === "fidelity"));
        if (!planEntry) continue; // fileName came from THIS group's own red set, so this always resolves
        const liveJson = `.scratch/gate-harness-live-${fileName}.json`;
        runCapture("npx", ["vitest", "run", "--config", g.config, "--reporter=json", `--outputFile.json=${liveJson}`, planEntry.run]);
        let anyLiveFail = true; // fail closed: an unreadable/unparsable targeted re-run counts as live red
        try {
          anyLiveFail = failedLiveTitles(JSON.parse(readFileSync(liveJson, "utf8")), liveTitles).length > 0;
        } catch { /* keep the fail-closed default above */ }
        try { rmSync(liveJson); } catch { /* best-effort */ }
        if (anyLiveFail) liveViaLiveTests.push(fileName);
      }
      const liveRed = red.filter((n) => !retiredSet.has(n) || liveViaLiveTests.includes(n));
      const retiredRed = red.filter((n) => retiredSet.has(n) && !liveViaLiveTests.includes(n));
      const runOf = (n) => (/^(run\d{4})/.exec(n)?.[1] ?? "other");
      const placedBy = {};
      for (const q of plan.filter((x) => !!x.fidelity === (g.label === "fidelity"))) { const k = runOf(basename(q.src)); placedBy[k] = (placedBy[k] || 0) + 1; }
      const redBy = {};
      for (const n of red) { const k = runOf(n); redBy[k] = (redBy[k] || 0) + 1; }
      body += `-- ${g.label}: ${g.files.length} harness files — ${sum}\n   red files ${red.length} (${liveRed.length} live, ${retiredRed.length} retired): ${Object.keys(placedBy).sort().map((k) => `${k} ${redBy[k] ?? 0}/${placedBy[k]}`).join(", ")}\n${liveRed.length ? `   LIVE: ${liveRed.join(", ")}\n` : ""}${retiredRed.length ? `   RETIRED (not blocking): ${retiredRed.join(", ")}\n` : ""}${liveViaLiveTests.length ? `   LIVE-VIA-LIVE_TESTS (retired file, specific its still live): ${liveViaLiveTests.join(", ")}\n` : ""}${filesMismatch ? `   FILES MISMATCH: vitest reported Test Files ${Number.isNaN(testFilesTotal) ? "(none)" : testFilesTotal} of ${g.files.length} handed to it — some files were silently not collected\n` : ""}${r.code !== 0 && red.length === 0 ? `   CRASH: vitest exit ${r.code} with no red file named — ${(/^.*(?:Error|error)[^\n]*$/m.exec(out)?.[0] ?? "no error line").trim().slice(0, 200)}\n` : ""}`;
      // QUEUE #67: --harnesses exits non-zero only on a LIVE RED file (a harness a retired: entry
      // does not name), a Test Files count mismatch (N1: vitest silently dropped a handed file),
      // or a non-test crash (r.code with no red files at all, e.g. a config error).
      if (liveRed.length || filesMismatch || (r.code !== 0 && red.length === 0)) code = r.code || 1;
    }
  } catch (e) {
    // RUN-0015 F-9: an I/O failure in the copy phase used to escape this block, so --out was
    // never written and --summarize printed `== harnesses: not run` — a CRASH spelled exactly
    // like a leg nobody ran. The block is appended either way, carrying what happened.
    body += `-- CRASHED after ${copies.length} copies: ${e?.message || e}\n`;
    code = code || 1;
  } finally {
    for (const c of copies) { try { rmSync(c); } catch {} }
    // Success-path cleanup (unchanged) removed every copy this run made; the registry naming
    // them is now stale and is cleared too, so the NEXT run's "remove what a previous run's
    // registry lists" step (above) finds nothing to do.
    clearRegistry(REGISTRY);
    // QUEUE #275: released on every path out of the try (success or the catch above) — a run
    // killed by the ~170s call budget before reaching here is the only case that leaves the lock
    // for the next run's staleness check to reclaim. `lockOwner` (P2-G2) makes this a no-op if
    // this run never actually acquired the lock (unreachable here in practice, since every
    // pre-acquire path returns before this point — kept for symmetry with `refuse`).
    releaseHarnessLock(LOCK_FILE, lockOwner);
  }
  // QUEUE #197 (S-075): three disjoint classes that sum to the pinned population, the FULL list, and an
  // unplaced harness that no retired: entry names is a LIVE RED (an unplaced harness is unmeasured, not green).
  const unplacedRetired = unplaced.filter((u) => retiredSet.has(basename(u.file)));
  const unplacedUnmeasured = unplaced.filter((u) => !retiredSet.has(basename(u.file)));
  const classSum = plan.length + unplacedRetired.length + unplacedUnmeasured.length;
  body += `-- COUNTS: placed ${plan.length} + UNPLACED-RETIRED ${unplacedRetired.length} + UNPLACED-UNMEASURED ${unplacedUnmeasured.length} = ${classSum} of ${pinned.length} pinned\n`;
  if (classSum !== pinned.length) { body += `-- COUNT MISMATCH: the three classes sum to ${classSum}, not ${pinned.length}\n`; code = code || 1; }
  if (unplacedRetired.length) body += `-- UNPLACED-RETIRED (${unplacedRetired.length}, not blocking — ${RETIRED_FILE}):\n${unplacedRetired.map((u) => `   ${u.file} (${u.reason})`).join("\n")}\n`;
  if (unplacedUnmeasured.length) { body += `-- UNPLACED-UNMEASURED (${unplacedUnmeasured.length}) — LIVE RED:\n${unplacedUnmeasured.map((u) => `   ${u.file} (${u.reason})`).join("\n")}\n`; code = code || 1; }
  const header = `${pinned.length} pinned org harnesses; ${plan.length} placed (${plan.filter((p) => p.dest === dirname(p.src)).length} in place), ${unplaced.length} unplaced (${unplacedRetired.length} retired, ${unplacedUnmeasured.length} unmeasured); ${copies.length} copies removed`;
  appendBlock(HARNESS_LABEL, `${header}\n${body}`, code, Number.POSITIVE_INFINITY);
  console.log(header);
  console.log(tail(body, 20));
  console.log(`harnesses exit ${code} — appended to ${outFile}`);
  process.exit(code);
}

// ------------------------------------------------- QUEUE #446: the green ledger
// The sampling policy lives HERE, as two constants, because the gate is what makes the ledger
// safe to trust -- not the ledger. green-ledger.mjs will happily record a "pass" nobody observed
// (`record --result pass` writes an assertion; it runs nothing), so the only thing standing
// between a recorded green and a fabricated one is that a sample of the skips is EXECUTED.
const LEDGER_AUDIT_RATE = 0.25; // 25% of the skip set, per --ledger-verify run
const LEDGER_AUDIT_CAP = 12;    // one audit leg still has to fit a ~170s bash call
const LEDGER_PLAN_LABEL = "green-ledger plan";
const LEDGER_VERIFY_LABEL = "green-ledger verify";
// `ceil(K * RATE)`, floored at 1 whenever anything at all was skipped. The floor is the whole
// point: `for (const s of []) …` finishes without executing anything, and an audit that executed
// nothing used to exit 0. A skip set of 1 still owes one executed suite.
function ledgerAuditFor(skipped) {
  if (skipped <= 0) return 0;
  return Math.min(LEDGER_AUDIT_CAP, Math.max(1, Math.ceil(skipped * LEDGER_AUDIT_RATE)));
}
// The plan leg's own recorded skip count, read back out of the shared log. This is the coupling:
// --ledger-verify does not take the caller's word for how much was skipped, and --summarize
// re-reads the same two numbers from the same two blocks.
function lastLedgerSkip(text) {
  const rows = [...String(text).matchAll(/^-- ledger-plan: scope (\d+), run (\d+), skip (\d+)$/gm)];
  if (!rows.length) return null;
  const r = rows[rows.length - 1];
  return { scope: Number(r[1]), run: Number(r[2]), skip: Number(r[3]) };
}
function lastLedgerAudited(text) {
  const rows = [...String(text).matchAll(/^green-ledger: audited (\d+) of (\d+) hit\(s\)/gm)];
  if (!rows.length) return null;
  const r = rows[rows.length - 1];
  return { audited: Number(r[1]), hits: Number(r[2]) };
}

if (mode === "--ledger-plan") {
  const i = argv.indexOf("--files");
  const files = [];
  if (i >= 0) for (let k = i + 1; k < argv.length && !argv[k].startsWith("--"); k++) files.push(argv[k]);
  // QUEUE #230's absent-input rule. An empty scope is not "nothing changed, skip everything": it
  // is a caller that named nothing, and answering it with a plan would be answering a question
  // that was never asked.
  if (!files.length) {
    appendBlock(LEDGER_PLAN_LABEL, "REFUSED: --ledger-plan needs --files <a> <b> ... (the changed scope); no file named", 1);
    console.error("--ledger-plan REFUSED — no --files named");
    process.exit(1);
  }
  const { stdout, stderr, code } = runCapture("node", ["scripts/green-ledger.mjs", "plan", "--json", "--files", ...files]);
  let planned = null;
  try { planned = JSON.parse(stdout); } catch { planned = null; }
  if (code !== 0 || !planned || !Array.isArray(planned.skip) || !Array.isArray(planned.run)) {
    appendBlock(LEDGER_PLAN_LABEL, `REFUSED: green-ledger plan did not return a usable plan (exit ${code})\n${stdout}\n${stderr}`, code || 1);
    console.error(`--ledger-plan REFUSED — green-ledger plan exit ${code}`);
    process.exit(code || 1);
  }
  const skip = planned.skip.length;
  const owed = ledgerAuditFor(skip);
  // Every skipped suite is NAMED in the log. A skip list you cannot read afterwards is a skip
  // list nobody can ever check, which is the same defect one level up.
  const body = [
    `-- ledger-plan: scope ${planned.scope}, run ${planned.run.length}, skip ${skip}`,
    `-- ledger-plan: this log now owes ${owed} EXECUTED audit(s) (rate ${LEDGER_AUDIT_RATE}, floor 1, cap ${LEDGER_AUDIT_CAP}); run: node scripts/gate.mjs --ledger-verify --out ${outFile}`,
    planned.run.length ? `RUN: ${planned.run.join(" ")}` : "RUN: (nothing)",
    ...planned.skip.map((s) => `SKIP: ${s}`),
  ].join("\n");

  // skip set, so there is nothing whose honesty this could establish -- and a green here would

  if (!plan) {

  }

  // state HAS an evidence line, so the loop above cannot see it — an `OK:` line reporting 0 is

    if (!executed || !tests) {

    } else if (Number(executed[1]) === 0 || Number(tests[2]) === 0) {

    }

  if (totalsPartialLine) console.log(totalsPartialLine);

  const lastInv = invariantsParts.length ? invariantsParts[invariantsParts.length - 1] : null;
  const invRefused = lastInv ? refusedLine(lastInv) : null;
  legLines.push(`== invariants: ${invRefused !== null ? `REFUSED (exit ${lastInv.exitCode}) — did not start, no guard executed: ${invRefused.slice(0, 200)}` : legShow(invariantsParts, invariantsLine, '"OK: ... invariant ..."')}`);
  const shardsRefused = invariantsShardParts.filter((p) => refusedLine(p) !== null).length;
  legLines.push(`== invariants (sharded): ${invariantsShardParts.length} shard block(s)${shardsRefused ? `, ${shardsRefused} REFUSED (did not start)` : ""}; merge ${invariantsMergeParts.length ? (invariantsMergeParts[invariantsMergeParts.length - 1].exitCode === 0 ? "OK" : `FAILED (exit ${invariantsMergeParts[invariantsMergeParts.length - 1].exitCode})`) : "not run"}`);
  legLines.push(`== lint:arch: ${lintArchParts.length ? (lintArchParts[0].exitCode === 0 ? "OK" : `FAILED (exit ${lintArchParts[0].exitCode})`) : "MISSING"}`);
  legLines.push(`== claims: ${legShow(claimsParts, claimsLine, '"check-claims: ..."')}`);

  legLines.push(`== harnesses: ${harnessLine ?? "not run"}${harnessCrash ? ` — ${harnessCrash}` : ""}${harnessRed}`);

  // from the summary alone rather than by re-reading the log.
  {
    const planned = ledgerPlanParts.length ? lastLedgerSkip(ledgerPlanParts[ledgerPlanParts.length - 1].body) : null;
    const got = ledgerVerifyParts.length ? lastLedgerAudited(ledgerVerifyParts[ledgerVerifyParts.length - 1].body) : null;
    legLines.push(`== green-ledger: ${planned ? `${planned.skip} skipped of ${planned.scope} (owed ${ledgerAuditFor(planned.skip)} audit(s))` : "plan not run"}; ${got ? `${got.audited} audited of ${got.hits} hit(s)` : "verify not run"}`);
  }

  {
    const p = orchParts.length ? orchParts[orchParts.length - 1] : null;
    const v = p ? /^check-orchestrator-compliance: .*$/m.exec(p.body)?.[0] ?? `no verdict line (exit ${p.exitCode})` : null;
    const state = !p ? "not run (gate.mjs --orch-compliance --repo <dir>)" : p.exitCode === 0 ? "OK" : p.exitCode === 2 ? `REFUSED (exit 2)` : `FAILED (exit ${p.exitCode})`;

      console.log(`== timings: the ${top.length} slowest file(s) take ${pct(topSum)} of ${total} ms`);
    }
  }
  console.log(verdict);
  // The block mirrors the verdict; it is not a gate LEG, so it is excluded from redParts above and

  // ORG-02-R3 (#499) — THE CLOSURE HALF, and it is the half that reaches an auditor. Before this

}

/**
 * Regenerate the browser extension's content-script `matches` from the platform
 * table — the ONE place platform hosts live.
 *
 *   npx tsx scripts/gen-extension-matches.mjs          # write
 *   npx tsx scripts/gen-extension-matches.mjs --check   # verify, non-zero on drift
 *
 * Run this after adding or removing a platform (or a host). The extension's
 * manifest test (`apps/extension/test/manifest.test.ts`) asserts the shipped
 * manifest equals `injectionMatches()` exactly, so a forgotten run fails CI with
 * the exact fix. Only the `content_scripts[0].matches` array is touched; the rest
 * of the manifest is preserved verbatim (2-space JSON, trailing newline).
 */
import { readFileSync, writeFileSync } from "node:fs";
import { fileURLToPath } from "node:url";
import { injectionMatches } from "../packages/canonical/src/platforms.ts";

const MANIFEST = fileURLToPath(new URL("../apps/extension/manifest.json", import.meta.url));
const check = process.argv.includes("--check");

const raw = readFileSync(MANIFEST, "utf8");
const manifest = JSON.parse(raw);
const want = injectionMatches();
const cs = manifest.content_scripts?.[0];
if (!cs || !Array.isArray(cs.matches)) {
  console.error("manifest has no content_scripts[0].matches to update");
  process.exit(2);
}

const same = JSON.stringify([...cs.matches].sort()) === JSON.stringify(want);
if (check) {
  if (same) {
    console.log(`OK: extension matches are in sync (${want.length} hosts).`);
    process.exit(0);
  }
  console.error("DRIFT: extension matches differ from the platform table. Run: npx tsx scripts/gen-extension-matches.mjs");
  process.exit(1);
}

cs.matches = want;
const out = JSON.stringify(manifest, null, 2) + "\n";
if (out === raw) {
  console.log(`No change: extension matches already current (${want.length} hosts).`);
} else {
  writeFileSync(MANIFEST, out);
  console.log(`Wrote ${want.length} content-script matches to apps/extension/manifest.json.`);
}

#!/usr/bin/env node
// scripts/green-ledger.mjs — `(suite, sha256 of the suite and its transitive inputs) -> passed`,
// so a suite whose inputs have not moved since a recorded green can be skipped.
//
//   node scripts/green-ledger.mjs key    --suite <s> [--root <dir>]
//   node scripts/green-ledger.mjs check  --suite <s>            exit 0 = HIT (skippable), 1 = MISS
//   node scripts/green-ledger.mjs record --suite <s> [--result pass|fail]
//   node scripts/green-ledger.mjs plan   --files <a> <b> …      the scope, split into HIT / MISS
//   node scripts/green-ledger.mjs verify [--run N] [--require M]  re-derive every key; RUN a sample
//                                                               of the HITS (least-recently-audited
//                                                               first) and fail if one is red, or if
//                                                               fewer than M were actually executed
//   common: [--ledger <file>] [--json]
//
// THE FAILURE MODE THIS IS DESIGNED AGAINST IS A STALE KEY — a hit returned for a suite whose
// inputs HAVE moved. That turns the ledger into a guard that cannot fail, which is the family
// this org has spent weeks fighting. Five rules, each of which can be executed against:
//
//   1. CONTENT, NEVER MTIME. The key is sha256 over the bytes. `touch` on an input changes
//      nothing; one byte changes everything. Nothing in this file reads a timestamp, and
//      tests/tooling/green-ledger.test.ts proves it by touching a file and re-checking.
//   2. AN UNCOMPUTABLE KEY IS A MISS. A suite whose input set cannot be bounded (test-scope's
//      OPAQUE class — a whole-tree reader, a `git ls-files` shell-out) has no honest key, so it
//      never hits. It is not "assume unchanged"; it is "cannot answer", and the two must not
//      produce the same behaviour. A missing input file is likewise a MISS, not a skipped term.
//   3. THE KEY COVERS THE DERIVER. sha256 of test-scope.mjs and of this file are IN the key. The
//      classic stale key is not a changed input — it is a changed notion of "input": widen the
//      graph and every recorded green silently describes a narrower world. Changing either
//      instrument invalidates the whole ledger, by construction.
//   4. THE INPUT SET IS RE-DERIVED AT CHECK TIME, never read back from the record. Keying on the
//      hashes of the files you knew about is how a NEW input slips in unnoticed; because the key
//      is a hash of the (path, hash) SET, a file entering the set changes the key.
//   5. A STALE HIT IS DETECTABLE. `record` stores the manifest beside the key, so any later
//      reader can see exactly what was hashed; `verify --run N` re-derives every key and then
//      EXECUTES a sample of the suites the ledger would have skipped, and exits non-zero if one
//      of them is red. A ledger nobody audits is a ledger that cannot fail.
import { execFileSync } from "node:child_process";
import { createHash } from "node:crypto";
import { existsSync, mkdirSync, readFileSync, realpathSync, writeFileSync } from "node:fs";
import { dirname, join, resolve } from "node:path";
import { fileURLToPath } from "node:url";
import { createGraph, expandTargets, scopeFor } from "./test-scope.mjs";

export const SCHEMA = "green-ledger/1";
const HERE = dirname(fileURLToPath(import.meta.url));
const sha = (b) => createHash("sha256").update(b).digest("hex");

/** The two instruments' own bytes. Rule 3: a changed deriver invalidates every key. */
export function deriverDigest() {
  return sha(Buffer.concat([readFileSync(join(HERE, "test-scope.mjs")), readFileSync(join(HERE, "green-ledger.mjs"))]));
}

/** The manifest a key is computed over: every input path with the sha256 of its CONTENT, sorted.
 *  Returns `{ ok:false, reason }` where no honest key exists (rule 2). */
export function manifestFor(graph, suite) {
  if (!graph.suites.has(suite)) return { ok: false, reason: `not in the suite population (${graph.popSource})` };
  const { files, dirs, opaque } = graph.inputsOf(suite);
  if (opaque) return { ok: false, reason: `UNBOUNDED INPUTS — ${opaque}` };
  const paths = new Set(files);
  for (const d of dirs) for (const f of graph.treeFiles) if (f === d || f.startsWith(d + "/")) paths.add(f);
  const rows = [];
  for (const p of [...paths].sort()) {
    if (!graph.treeSet.has(p)) return { ok: false, reason: `input not in the tree: ${p}` };
    let buf;
    try { buf = readFileSync(join(graph.root, p)); } catch (e) { return { ok: false, reason: `input unreadable: ${p} — ${String(e.message).split("\n")[0]}` }; }
    rows.push(`${sha(buf)}  ${p}`);
  }
  return { ok: true, rows };
}

export function keyFor(graph, suite, deriver = deriverDigest()) {
  const m = manifestFor(graph, suite);
  if (!m.ok) return m;
  const body = [`${SCHEMA}`, `deriver ${deriver}`, `suite ${suite}`, ...m.rows].join("\n");
  return { ok: true, key: sha(body), rows: m.rows, count: m.rows.length };
}

const defaultLedger = (root) => join(root, "node_modules", ".cache", "green-ledger", "ledger.json");
export function loadLedger(file) {
  if (!existsSync(file)) return { schema: SCHEMA, entries: {} };
  try {
    const j = JSON.parse(readFileSync(file, "utf8"));
    // A ledger written by another schema is not partially usable. Discard it loudly.
    if (j.schema !== SCHEMA) { process.stderr.write(`green-ledger: ledger schema ${j.schema} != ${SCHEMA}; ignoring every entry\n`); return { schema: SCHEMA, entries: {} }; }
    return { schema: SCHEMA, entries: j.entries ?? {} };
  } catch (e) {
    process.stderr.write(`green-ledger: ledger unreadable (${String(e.message).split("\n")[0]}); treating as empty\n`);
    return { schema: SCHEMA, entries: {} };
  }
}
export function saveLedger(file, ledger) {
  mkdirSync(dirname(file), { recursive: true });
  writeFileSync(file, JSON.stringify(ledger, null, 1));
}

/** HIT iff a recorded PASS exists under the freshly-derived key. Anything else is a MISS, with
 *  the reason attached — "cannot compute" and "changed" are different answers and say so. */
export function check(graph, ledger, suite, deriver = deriverDigest()) {
  const k = keyFor(graph, suite, deriver);
  if (!k.ok) return { hit: false, reason: `MISS (no key): ${k.reason}` };
  const rec = ledger.entries[suite];
  if (!rec) return { hit: false, key: k.key, reason: "MISS: never recorded" };
  if (rec.key !== k.key) return { hit: false, key: k.key, reason: `MISS: inputs moved (recorded ${rec.key.slice(0, 12)}, now ${k.key.slice(0, 12)}, ${k.count} inputs)` };
  // Belt and braces on rule 3. The deriver digest is already INSIDE the key, so a mismatch here
  // is unreachable through `record`; it is reachable through a ledger written by hand or carried
  // between trees, and an entry whose recorded deriver is not this one is not evidence about
  // this one. Stating it as its own MISS also gives the guard a surface it can drive.
  if (rec.deriver && rec.deriver !== deriver) return { hit: false, key: k.key, reason: `MISS: recorded under deriver ${String(rec.deriver).slice(0, 12)}, running ${deriver.slice(0, 12)}` };
  if (rec.result !== "pass") return { hit: false, key: k.key, reason: `MISS: recorded result is ${rec.result}` };
  return { hit: true, key: k.key, reason: `HIT: ${k.count} inputs unchanged since ${rec.at}` };
}

function runSuite(root, suite) {
  const bin = join(root, "node_modules", ".bin", "vitest");
  try {
    execFileSync(bin, ["run", suite, "--reporter=dot"], { cwd: root, encoding: "utf8", stdio: ["ignore", "pipe", "pipe"], timeout: 170_000 });
    return { pass: true };
  } catch (e) {
    return { pass: false, out: String(e.stdout ?? "").split("\n").slice(-6).join("\n") };
  }
}

    if (flag("json")) console.log(JSON.stringify({ schema: SCHEMA, targets, scope: scope.length, skip: hits, run: misses }, null, 1));
    else {

    }

  if (cmd === "verify") {
    // R03 / QUEUE #446. TWO changes to this mode, both aimed at the same defect class: an audit
    // that cannot fail.
    //
    // (1) THE SAMPLE IS ROUND-ROBIN, NOT A PREFIX. This used to be `hits.slice(0, sample)` over
    //     `Object.keys(ledger.entries)` — INSERTION order. A suite recorded after the first N
    //     was then never audited, ever, by any number of runs: a permanent blind spot, fixed in
    //     place, that an actor who knows the order can aim at. Selecting LEAST-RECENTLY-AUDITED
    //     first instead turns the rate into a BOUND rather than a probability: at `--run N` over
    //     H hits, every hit is EXECUTED at least once every ceil(H/N) verifies. `lastAudit` is
    //     written back so the rotation survives across runs; it is ledger bookkeeping and is not
    //     an input to any key, so recording it cannot move a key.
    // (2) `--require M` — AUDITING NOTHING IS NOT PASSING. `for (const s of []) …` completes
    //     without executing anything and the old code then exited 0: an empty hit set, an empty
    //     ledger, or `--run 0` all printed a green. That is the vacuous-pass shape (QUEUE #230's
    //     absent-input rule, ANTI-024). With `--require M` this mode exits non-zero unless M
    //     suites were ACTUALLY EXECUTED. gate.mjs --ledger-verify always passes a --require.
    const sample = Number(opt("run", "0"));
    const required = Number(opt("require", "0"));
    let hit = 0, miss = 0, gone = 0;
    const hits = [];
    for (const suite of Object.keys(ledger.entries)) {
      if (!graph.suites.has(suite)) { gone++; out.push(`GONE  ${suite} — recorded but no longer in the population`); continue; }
      const r = check(graph, ledger, suite, deriver);
      if (r.hit) { hit++; hits.push(suite); } else { miss++; }
    }
    console.log(`entries: ${Object.keys(ledger.entries).length}  hits: ${hit}  misses: ${miss}  gone: ${gone}  deriver: ${deriver.slice(0, 12)}`);
    for (const l of out) console.log(l);
    // RULE 5: execute a sample of the suites this ledger would have let a room skip. A hit that
    // is red is a stale key, and this is the only thing that can tell you so.
    hits.sort((a, b) => {
      const la = ledger.entries[a].lastAudit ?? "";
      const lb = ledger.entries[b].lastAudit ?? "";
      return la === lb ? (a < b ? -1 : a > b ? 1 : 0) : la < lb ? -1 : 1;
    });
    const want = Number.isFinite(sample) ? Math.max(0, Math.trunc(sample)) : 0;
    const now = new Date().toISOString();
    let red = 0, audited = 0;
    for (const s of hits.slice(0, want)) {
      const r = runSuite(root, s);
      audited++;
      console.log(`${r.pass ? "AUDIT-OK  " : "AUDIT-RED "}${s}${r.pass ? "" : `\n${r.out}`}`);
      ledger.entries[s].lastAudit = now;
      ledger.entries[s].lastAuditResult = r.pass ? "pass" : "fail";
      if (!r.pass) red++;
    }
    if (audited) { try { saveLedger(ledgerFile, ledger); } catch (e) { process.stderr.write(`green-ledger: could not persist audit rotation: ${String(e.message).split("\n")[0]}\n`); } }
    // The one line gate.mjs --summarize parses. `audited` is the number of suites this run
    // actually EXECUTED, never the number it intended to.
    console.log(`green-ledger: audited ${audited} of ${hit} hit(s) (least-recently-audited first); ${red} RED`);
    if (red) { process.stderr.write(`green-ledger: ${red} audited HIT(s) are RED — the ledger returned a stale key\n`); process.exit(1); }
    if (audited < required) {
      process.stderr.write(`green-ledger: NOT VERIFIED — ${audited} suite(s) executed, --require ${required}; ${hit} hit(s) available. An audit that ran nothing is not a pass.\n`);
      process.exit(1);
    }
    process.exit(0);
  }

  process.stderr.write("green-ledger: usage — key|check|record|plan|verify (see the docblock)\n");
  process.exit(2);
}

    // tried LAST — reading it first let a stale appended "EXIT 0" mask that dir's own
    // SUMMARY.txt EXIT 1.
    const EVIDENCE_NAMES = ["SUMMARY.txt", "GATE-FINAL.md", "GATE-LOGS.md"];
    const BATCH_SUMMARY = "GATE-SUMMARY.md";
    // I34-FX3-TOOLS (I34-RG-W3T TC:8): the letter in either case, and the per-run DIRECTORY shape I26/I27 wrote,
    // `<x>-gate-<n>[letter]/SUMMARY.txt`, read like the flat file.
    const MAC_SUMMARY_RE = /^[A-Za-z0-9_.-]*-gate-(\d+)([a-zA-Z]?)-SUMMARY\.txt$/;
    const MAC_DIR_RE = /^[A-Za-z0-9_.-]*-gate-(\d+)([a-zA-Z]?)$/;
    const MAC_FAILED_LEG_RE = /^FAIL[ \t]+--\S/m;
    const CANONICAL_EXIT_RE = /^EXIT (\d): gate /gm;
    // Ordering (D-0728, I3-1525): by the time of the last git commit that touched the gate dir
    // (`git log -1 --format=%ct -- <dir>`), NEVER by parsing the run dir's name. The name rank this
    // replaced put every unrecognised shape (all W3-* and B*-batch dirs) at [-1, 0], below every S-*
    // dir, so the tripwire was pinned to S-092 and saw none of the later waves. A dir no commit has
    // touched is the live batch and sorts newest. The only tie-break is inside ONE run dir, by the gate
    // subdir's own number (gate/gate0 < gate1 < gate2); a tie ACROSS run dirs at the newest-two
    // boundary cannot be ordered by the ruled key and is never broken by name: every admissible ordering
    // of the tie is read, and only a verdict that DEPENDS on the ordering is ORDER-AMBIGUOUS (D-0733).
    let entries = [];
    try {
      for (const runDir of readdirSync(FINDINGS_DIR)) {
        const runPath = join(FINDINGS_DIR, runDir);
        let subs = [];
        try {
          if (statSync(runPath).isDirectory()) subs = readdirSync(runPath);
        } catch {
          continue;
        }
        // QUEUE #536 RG4 (D-0849(10)): since B23 a batch writes its gate verdict at the BATCH ROOT
        // (GATE-SUMMARY.md), not in a gate*/ subdir, so a discovery that only walked gate*/ dirs went
        // blind to every later batch and read the stale W3-09a dir as newest. A batch-root
        // GATE-SUMMARY.md is its own entry, keyed by that FILE's last commit, and it is read ONLY for
        // the canonical verdict line gate.mjs prints ("EXIT N: gate …") — a per-command "EXIT 0" row
        // in a summary table is never a verdict.
        if (subs.includes(BATCH_SUMMARY)) {
          const file = join(runPath, BATCH_SUMMARY);
          try {
            if (statSync(file).isFile()) entries.push({ gateDir: file, runDir, present: [file], subNum: 0, time: null, canonical: true });
          } catch {}
        }
        // I34-TOOLS-C (I32 B18 / L4-21): since I26 a batch's gate runs on the Mac and its verdict is
        // `<run>/_plan/mac-gate/<x>-gate-<n>[letter]-SUMMARY.txt`. Each is its own entry, keyed by that FILE's last
        // commit like GATE-SUMMARY.md and read only for the canonical verdict line; inside one run dir the gate number
        // (3b after 3) is the tie-break, as gate0 < gate1 is for gate*/ dirs.
        let macs = [];
        const macDir = join(runPath, "_plan", "mac-gate");
        try {
          if (statSync(macDir).isDirectory()) macs = readdirSync(macDir);
        } catch {}
        for (const name of macs) {
          const flat = MAC_SUMMARY_RE.exec(name);
          const mm = flat ?? MAC_DIR_RE.exec(name);
          if (!mm) continue;
          const file = flat ? join(macDir, name) : join(macDir, name, "SUMMARY.txt");
          const subNum = Number(mm[1]) + (mm[2] ? (mm[2].toLowerCase().charCodeAt(0) - 96) / 100 : 0);
          try {
            if (statSync(file).isFile()) entries.push({ gateDir: file, runDir, present: [file], subNum, time: null, canonical: true, mac: true });
          } catch {}
        }
        for (const sub of subs) {
          if (!GATE_DIR_RE.test(sub)) continue;
          const gateDir = join(runPath, sub);
          if (!statSync(gateDir).isDirectory()) continue;
          const present = EVIDENCE_NAMES.map((name) => join(gateDir, name)).filter((p) => existsSync(p));
          const subNum = Number(/^gate(\d*)$/.exec(sub)[1] || 0);
          entries.push({ gateDir, runDir, present, subNum, time: null });
        }
      }
    } catch (e) {
      push({ name: "gate_red_twice", ok: false, evidence: `cannot list gate dirs: ${e.message}` });
      return;
    }
    for (const e of entries) {
      try {
        e.time = commitTime(e.gateDir);
      } catch (err) {
        push({ name: "gate_red_twice", ok: false, evidence: `cannot read the last-commit time of ${e.gateDir} (git log): ${String(err.message).split("\n")[0]} — evidence cannot be read` });
        return;
      }
    }
    const override = val("--gate-log");
    if (override) {
      entries = entries.filter((e) => !e.present.includes(override));
      if (existsSync(override)) entries.push({ gateDir: dirname(override), runDir: dirname(dirname(override)), present: [override], subNum: 0, time: Infinity });
    }
    entries.sort((a, b) => (a.time === b.time ? (a.runDir === b.runDir ? b.subNum - a.subNum : 0) : b.time > a.time ? 1 : -1));
    const newest2 = entries.slice(0, 2);
    if (newest2.length < 2) {
      push({ name: "gate_red_twice", ok: false, evidence: `fewer than two gate dirs found (${newest2.length}) — evidence cannot be read` });
      return;
    }
    // D1 "absent evidence = fail, not ok": a ranked-newest gate dir with no evidence file, or none
    // of its present files carrying an EXIT line, fails this tripwire — it is never skipped in
    // favour of an older dir that happens to have readable evidence.
    const exitCache = new Map();
    const readExit = (e) => {
      if (!exitCache.has(e)) {
        let found = { gateDir: e.gateDir, p: e.gateDir, exit: null };
        for (const p of e.present) {
          const text = readFileSync(p, "utf8");
          const m = [...text.matchAll(e.canonical ? CANONICAL_EXIT_RE : /EXIT (\d)/g)];
          if (m.length) {
            found = { gateDir: e.gateDir, p, exit: Number(m[m.length - 1][1]) };
            // D-1120 (I34-TOOLS-C): a Mac-gate summary is RED only when a leg RAN and FAILED (a `FAIL  --<leg>` line); an
            // EXIT 1 with no failed leg is INCOMPLETE (a leg not run on the Mac, e.g. the HR-4 compliance leg), never red.
            if (e.mac && found.exit !== 0 && !MAC_FAILED_LEG_RE.test(text)) {
              const missing = /^MISSING PARTS: (.*)$/m.exec(text);
              found.incomplete = `incomplete: no leg ran and failed${missing ? `; MISSING PARTS: ${missing[1]}` : ""}`;
            }
            break;
          }
        }
        exitCache.set(e, found);
      }
      return exitCache.get(e);
    };
    const judgePair = (pair) => {
      const exits = pair.map(readExit);
      if (exits.some((e) => e.exit == null)) {
        return { ok: false, evidence: `no EXIT line in ${exits.filter((e) => e.exit == null).map((e) => e.gateDir).join(", ")} — evidence cannot be read` };
      }
      return { ok: !exits.every((e) => e.exit !== 0 && !e.incomplete), evidence: exits.map((e) => `${e.p}: EXIT ${e.exit}${e.incomplete ? ` (${e.incomplete})` : ""}`).join(" | ") };
    };
    const t2 = newest2[1].time;
    const tied = entries.filter((e) => e.time === t2);
    const known = entries.filter((e) => e.time > t2);
    const slots = 2 - known.length;
    if (tied.length > slots && new Set(tied.map((e) => e.runDir)).size > 1) {
      // D-0733: read EVERY admissible filling of the open slots from the tie (inside one run dir the gate
      // number still orders, so a lower gate is admissible only beside its higher siblings). Every
      // filling red twice -> TRIPPED; none -> ok; a MIXED set is the only ORDER-AMBIGUOUS.
      const admissible = (sel) => sel.every((e) => tied.every((o) => o.runDir !== e.runDir || o.subNum <= e.subNum || sel.includes(o)));
      const fillings = (slots === 1 ? tied.map((e) => [e]) : tied.flatMap((a, i) => tied.slice(i + 1).map((b) => [a, b]))).filter(admissible);
      const verdicts = fillings.map((sel) => judgePair([...known, ...sel]));
      const red = verdicts.filter((v) => v.ok === false).length;
      const where = `${tied.map((e) => e.gateDir).join(", ")} share last-commit time ${t2 === UNCOMMITTED ? "(uncommitted)" : t2} across run dirs`;
      if (red === 0 || red === verdicts.length) {
        push({
          name: "gate_red_twice",
          ok: red === 0,
          evidence: `${where}; all ${verdicts.length} admissible ordering(s) of the tie agree (D-0733) — e.g. ${verdicts[0].evidence}`,
        });
        return;
      }
      push({
        name: "gate_red_twice",
        ok: "ORDER-AMBIGUOUS",
        evidence: `newest two gate dirs cannot be ordered: ${where}; ${red} of ${verdicts.length} admissible ordering(s) read red twice and the rest do not (D-0733)`,
      });
      return;
    }
    const v = judgePair(newest2);
    push({ name: "gate_red_twice", ok: v.ok, evidence: v.evidence });
  })();

  // ---- 2. budget mode FREEZE ----
  (function budgetTripwire() {
    try {
      // S-076 #269/discovered live: this used to shell out with no --dry-run of its own, so even
      // `halt.mjs --check --dry-run` silently wrote a fresh org/08-control/BUDGET.json for real
      // (budget-pace.mjs's own save() was unconditional before this room's --dry-run fix there) —
      // reproduced against the real mounted repo, reverted by hand. --root/--dry-run now forward.
      const budgetArgs = [join(ROOT, "scripts/budget-pace.mjs"), "--root", ROOT, ...(dryRun ? ["--dry-run"] : [])];
      const out = execFileSync("node", budgetArgs, { cwd: ROOT, encoding: "utf8" });
      const m = /\bMODE:\s*(\S+)/.exec(out) || /"mode":\s*"(\w+)"/.exec(out);
      if (!m) {
        push({ name: "budget_freeze", ok: false, evidence: "budget-pace.mjs output has no MODE — evidence cannot be read" });
        return;
      }
      if (!BUDGET_MODES.has(m[1])) {
        push({ name: "budget_freeze", ok: false, evidence: `budget-pace.mjs MODE: ${m[1]} — not one of ${[...BUDGET_MODES].join("/")} (budget-pace.mjs:81-86); the unrecognised is TRIPPED` });
        return;
      }
      push({ name: "budget_freeze", ok: m[1] !== "FREEZE", evidence: `budget-pace.mjs MODE: ${m[1]}` });
    } catch (e) {
      push({ name: "budget_freeze", ok: false, evidence: `budget-pace.mjs failed: ${e.message}` });
    }
  })();

  // ---- 3. seeded recall < 2/4 ----
  // D5 means SEEDED recall and nothing else (docket #43 (d), ruled (i) by Jason 2026-09-12):
  // `metrics.recall` (retrospective, a different denominator — RUN-0015's 5/13) is never read.
  // QUEUE #211/#230 (S-075): an ABSENT figure is not "not applicable". The sweep instance
  // declares `seeds` (RUN.schema 0.7.0); only an explicit `declared: false` with a reason
  // passes without a measured figure. Undeclared, planted-but-unscored, or never measured
  // at all -> TRIPPED.
  // SEEDS2 R2: a ratio over fewer than MIN_SEEDS seeds is not a measurement (1/1 clears 0.50).
  (function recallTripwire() {
    const MIN_SEEDS = 3;
    let src = val("--recall");
    let evidence;
    if (src) {
      evidence = `--recall ${src}`;
    } else {
      try {
        const files = readdirSync(INST_DIR).filter((f) => f.endsWith(".json")).sort();
        if (files.length === 0) throw new Error("no instances");
        const insts = files.map((f) => ({ f, j: JSON.parse(readFileSync(join(INST_DIR, f), "utf8")), time: null }));
        // D-0729/D-0733: "latest" is by last-commit time, never by filename (RG4-F2-T3-NAMEORDER). The
        // filename sort above only fixes an arbitrary order INSIDE a tie; the tie itself is enumerated below.
        for (const i of insts) {
          try {
            i.time = commitTime(join(INST_DIR, i.f));
          } catch (err) {
            push({ name: "seeded_recall", ok: false, evidence: `cannot read the last-commit time of ${i.f} (git log): ${String(err.message).split("\n")[0]} — evidence cannot be read` });
            return;
          }
        }
        const sweeps = insts.filter((i) => isSweepInstance(i.j));
        if (sweeps.length === 0) throw new Error("no SWEEP instance (composition.modes)");
        // One reading for ONE chronological order. P2-C2: the pick is the SAME shared selection
        // control-signals.mjs uses (scripts/lib/seed-recall.mjs), which takes the caller's order as given.
        const evalOrder = (ordered) => {
          const pick = pickLatestScoredSweepSeeds(ordered.map((i) => i.j));
          const S = ordered[pick.sweepIndex];
          if (S.j.seeds === undefined) {
            return {
              ok: false,
              evidence: ordered.some((i) => i.j?.seeds !== undefined)
                ? `latest sweep ${S.f} declares no seeds — evidence cannot be read`
                : `seeded recall has never been measured (QUEUE #32/#90): no instance declares seeds; latest sweep ${S.f}`,
            };
          }
          const d = S.j.seeds?.declared;
          if (typeof d !== "boolean" || (d === false && !(typeof S.j.seeds.reason_if_false === "string" && S.j.seeds.reason_if_false.trim()))) {
            return { ok: false, evidence: `${S.f} seeds.declared=${JSON.stringify(d)} without a reason_if_false — evidence cannot be read` };
          }
          if (!pick.ok) {
            return { ok: true, evidence: `${S.f} seeds.declared=false — not applicable: ${S.j.seeds.reason_if_false.trim()} (no seeded sweep has been scored yet)` };
          }
          const T = ordered[pick.scoredIndex];
          const { planted, scored, seeded_recall: r } = T.j.seeds;
          if (typeof r !== "number") return { ok: false, evidence: `${T.f} seeds planted, never scored (seeded_recall ${JSON.stringify(r)})` };
          if (Number.isInteger(planted) && planted < MIN_SEEDS) {
            return { ok: false, evidence: `${T.f} seeds planted=${planted} — too few seeds to measure (a sweep's seeds count only when planted >= ${MIN_SEEDS}; ADR-0010 "seeded recall < 2/4")` };
          }
          const hits = r * planted;
          if (!Number.isInteger(planted) || planted < 1 || scored !== planted || !(r >= 0 && r <= 1) || Math.abs(hits - Math.round(hits)) > 1e-9) {
            return { ok: false, evidence: `${T.f} seeds planted=${planted} scored=${scored} seeded_recall=${r} — evidence cannot be read` };
          }
          return { ok: r >= 0.5, evidence: `${T.f} seeds.seeded_recall: ${r} = ${Math.round(hits)}/${planted} (ratio ${r.toFixed(2)}, floor 2/4=0.50)` };
        };
        // The reading depends only on WHICH sweep is latest (S) and which SWEEP declaring seeds is latest
        // at or before it (T). Enumerate every (S, T) a commit-time tie admits: S is any sweep of the newest
        // sweep time; T is S when S declares, else any declaring sweep S's tie can place before it, else any
        // of the newest declaring sweep time below. `pins` puts those last inside their own tie.
        const declares = (i) => i.j?.seeds?.declared === true;
        const topT = Math.max(...sweeps.map((i) => i.time));
        const pinsList = [];
        for (const S of sweeps.filter((i) => i.time === topT)) {
          if (declares(S)) { pinsList.push([S]); continue; }
          const same = sweeps.filter((i) => i.time === topT && i !== S && declares(i));
          const older = sweeps.filter((i) => i.time < topT && declares(i));
          const olderT = older.length ? Math.max(...older.map((i) => i.time)) : null;
          const Ts = same.length ? same : older.filter((i) => i.time === olderT);
          if (Ts.length === 0) pinsList.push([S]);
          else for (const T of Ts) pinsList.push([T, S]);
        }
        const readings = pinsList.map((pins) => evalOrder([...insts].sort((a, b) => (a.time !== b.time ? (a.time < b.time ? -1 : 1) : pins.indexOf(a) - pins.indexOf(b)))));
        const red = readings.filter((x) => x.ok === false).length;
        if (readings.length === 1) {
          push({ name: "seeded_recall", ...readings[0] });
        } else if (red === 0 || red === readings.length) {
          push({ name: "seeded_recall", ok: red === 0, evidence: `instances tie on last-commit time; all ${readings.length} admissible latest-sweep picks agree (D-0733) — e.g. ${readings[0].evidence}` });
        } else {
          push({
            name: "seeded_recall",
            ok: "ORDER-AMBIGUOUS",
            evidence: `latest sweep cannot be ordered: ${sweeps.filter((i) => i.time === topT).map((i) => i.f).join(", ")} share last-commit time ${topT === UNCOMMITTED ? "(uncommitted)" : topT}; ${red} of ${readings.length} admissible pick(s) read TRIPPED and the rest do not (D-0733)`,
          });
        }
        return;
      } catch (e) {
        push({ name: "seeded_recall", ok: false, evidence: `${e.message} — evidence cannot be read` });
        return;
      }
    }
    const m = /^(\d+)\/(\d+)$/.exec(src);
    if (!m) {
      push({ name: "seeded_recall", ok: false, evidence: `unparseable recall '${src}' — evidence cannot be read` });
      return;
    }
    if (Number(m[2]) < MIN_SEEDS) {
      push({ name: "seeded_recall", ok: false, evidence: `${evidence} — too few seeds to measure (denominator < ${MIN_SEEDS})` });
      return;
    }
    const ratio = Number(m[1]) / Number(m[2]);
    push({ name: "seeded_recall", ok: ratio >= 0.5, evidence: `${evidence} (ratio ${ratio.toFixed(2)}, floor 2/4=0.50)` });
  })();

  // ---- 4. any CRITICAL confirmed at HEAD ----
  (function criticalTripwire() {
    if (!existsSync(REGISTRY_PATH)) {
      push({ name: "critical_at_head", ok: false, evidence: "REGISTRY.md missing — evidence cannot be read" });
      return;
    }
    const lines = readFileSync(REGISTRY_PATH, "utf8").split("\n");
    const open = [];
    let header = [];
    for (const line of lines) {
      if (!line.startsWith("|")) { header = []; continue; }
      const cells = line.split("|").map((c) => c.trim());
      // header row of the current table: it carries a severity-named cell, or its first column
      // is id-like (`id`, `finding`, `Item ID`) — not the single literal `id`. A table titled
      // otherwise left `header` stale from the previous table and pointed `sevIdx` into the
      // wrong columns (REGISTRY.md:391 is such a table).
      if (cells.some((c) => /^sev[a-z]*$/i.test(c)) || /^[a-z0-9 ]*\b(id|finding|item)\b[a-z0-9 ]*$/i.test(cells[1] ?? "")) { header = cells; continue; }
      if (/^-+$/.test((cells[1] ?? "").replace(/-/g, "-"))) continue; // separator row
      // THE TABLE SHAPE (G-7, second half). The column is located by HEADER on /sev/i, not
      // /severity/i — a REGISTRY table headed `sev` used to have no severity column at all, so
      // every CRITICAL row in it read ok. When there is no header row to read (`sevIdx === -1`),
      // the row's OWN cell decides: a cell that spells a severity word IS the severity column.
      // A shape this brake does not recognise must not read as "no unresolved CRITICAL row".
      let sevIdx = header.findIndex((c) => /sev/i.test(c));
      if (sevIdx === -1) sevIdx = cells.findIndex((c) => /^\*{0,2}(CRITICAL|HIGH|MEDIUM|LOW)\b/i.test(c));
      // contains-CRITICAL, not equals: `**CRITICAL**`, `CRITICAL (confirmed)`, `critical — at HEAD`.
      if (sevIdx === -1 || !/\bCRITICAL\b/i.test(cells[sevIdx] ?? "")) continue;
      // every status-shaped column for this row (any column whose header names status/class/found)
      const statusIdxs = header.map((h, i) => (/status|class/i.test(h) ? i : -1)).filter((i) => i >= 0);
      const statusText = statusIdxs.map((i) => cells[i] ?? "").join(" | ");
      const closedRe = /fixed|closed|wontdo|rejected/i;
      if (!closedRe.test(statusText)) open.push(`${cells[1]} (${statusText || "no status column found"})`);
    }
    push({
      name: "critical_at_head",
      ok: open.length === 0,
      evidence: open.length ? `unresolved CRITICAL rows: ${open.join("; ")}` : "no unresolved CRITICAL row in REGISTRY.md",
    });
  })();

  // ---- 5. reversal rate rising two cycles running ----
  // QUEUE #288: control-signals.mjs now appends the numeric orchestrator_reversal_rate to
  // SIGNALS-HISTORY.jsonl every run (it used to append only the fired boolean) — this tripwire
  // can finally read a real series instead of tripping "unknown" forever.
  (function reversalTripwire() {
    if (!existsSync(SIGNALS_HISTORY_PATH)) {
      push({ name: "reversal_rate_rising", ok: "INSUFFICIENT", evidence: "org/08-control/SIGNALS-HISTORY.jsonl does not exist yet — a series of 0, need 3 (D-0728)" });
      return;
    }
    const lines = readFileSync(SIGNALS_HISTORY_PATH, "utf8").split("\n").filter(Boolean);
    const rows = lines.map((l) => { try { return JSON.parse(l); } catch { return null; } }).filter((r) => r && typeof r.orchestrator_reversal_rate === "number");
    // P2-C1 (S-081 REG phase 2, fix round 1): control-signals.mjs now skips the append when
    // nothing closed since the last line (one line per NEW closure), but OLDER history can still
    // carry several lines for the same last_closed_run — collapse to the LAST value per
    // last_closed_run before taking the last 3, so a repeated value between real closures can
    // never mask (or fake) a rise. A missing/absent last_closed_run never collapses with another
    // (each keyed uniquely), since we cannot tell those apart as the same closure.
    const collapsed = [];
    for (const r of rows) {
      const key = r.last_closed_run ?? Symbol("no-last-closed-run");
      if (collapsed.length && collapsed[collapsed.length - 1].key === key) collapsed[collapsed.length - 1].rate = r.orchestrator_reversal_rate;
      else collapsed.push({ key, rate: r.orchestrator_reversal_rate });
    }
    const last3 = collapsed.slice(-3).map((c) => c.rate);
    if (last3.length < 3) {
      push({ name: "reversal_rate_rising", ok: "INSUFFICIENT", evidence: `only ${last3.length} recorded orchestrator_reversal_rate value(s) in history (one per closure) — need 3 (D-0728)` });
      return;
    }
    const rising = last3[0] < last3[1] && last3[1] < last3[2];
    push({ name: "reversal_rate_rising", ok: !rising, evidence: `last 3 orchestrator_reversal_rate values: ${last3.join(" -> ")}${rising ? " (rising two cycles running)" : ""}` });
  })();

  // ---- 6. (removed) lesson_conflicting — QUEUE #556, D-0684 (Jason, 2026-09-22) ----
  // `conflicting` is one of the five LEGAL lesson answers (_OUTPUT-CONTRACT.md), so an OCCURRENCE of
  // it is not a halt condition. The mtime-keyed window it used (reports newer than the tag
  // verified-2026-09-07-s003, aged by statSync().mtimeMs — no ordering on a copied tree, per
  // tripwire 1's own comment) went with it: nothing else in this file reads an mtime. D-0684 allows
  // a RATE of conflicting answers as a tripwire; none is built, because no rate, window or threshold
  // has been decided. Test: tests/tooling/b23h-halt-tripwires.test.ts.

  // ---- 7. antipattern_recurrence > 0 ----
  (function antipatternTripwire() {
    try {
    if (!existsSync(SIGNALS_PATH)) {
      push({ name: "antipattern_recurrence", ok: false, evidence: "SIGNALS.json missing — evidence cannot be read" });
      return;
    }
    const s = JSON.parse(readFileSync(SIGNALS_PATH, "utf8"));
    const sig = s.signals?.antipattern_recurrence;
    if (!sig || typeof sig.fired !== "boolean") {
      push({ name: "antipattern_recurrence", ok: false, evidence: "SIGNALS.json signals.antipattern_recurrence.fired unreadable — evidence cannot be read" });
      return;
    }
    push({ name: "antipattern_recurrence", ok: !sig.fired, evidence: `SIGNALS.json signals.antipattern_recurrence.fired: ${sig.fired} (count=${sig.count ?? "n/a"})` });
    } catch (e) {
      // guarded like its five siblings: an unparseable SIGNALS.json is TRIPPED, and the five
      // verdicts printed before it stay printed.
      push({ name: "antipattern_recurrence", ok: false, evidence: `SIGNALS.json unreadable: ${e.message} — evidence cannot be read` });
    }
  })();

  // D-0728/D-0729: every verdict is COUNTED by state, INSUFFICIENT and ORDER-AMBIGUOUS apart from TRIPPED and from unknown.
  // The tally is keyed by TRIPWIRE_STATES itself, so a state added to the export is counted without an edit here.
  console.log(`tripwires evaluated: ${results.length} — ${TRIPWIRE_STATES.map((st) => `${st} ${results.filter((r) => r.label === st).length}`).join(", ")}`);
  const tripped = results.filter((r) => r.ok === false);
  // QUEUE #520(b), D-0920 option (2): `--authoritative a,b` names the PROMOTED tripwires — only a TRIPPED one of those
  // authors HALT.md; every other trip is printed, counted and exits 2 exactly as before, but authors nothing. A name in
  // the list that this file did not evaluate is unreadable, and the unreadable is TRIPPED (ADR-0010 D5): it authors.
  // Without the flag every trip authors, as it always has, except an OBSERVE_ONLY_TRIPWIRES name, which never authors (D-0696, D-1597).
  const authArg = val("--authoritative");
  const authList = has("--authoritative") ? (authArg ?? "").split(",").map((x) => x.trim()).filter(Boolean) : null;
  if (authList !== null) {
    for (const name of authList) {
      if (results.some((r) => r.name === name)) continue;
      const r = { name, ok: false, evidence: `named by --authoritative but evaluated by no tripwire in this halt.mjs — the unreadable is TRIPPED`, label: "TRIPPED" };
      console.log(`tripwire ${name}: TRIPPED (${r.evidence})`);
      tripped.push(r);
    }
  }
  const authoring = (authList === null ? tripped : tripped.filter((t) => authList.includes(t.name))).filter((t) => !OBSERVE_ONLY_TRIPWIRES.includes(t.name));
  const observed = tripped.filter((t) => !authoring.includes(t));
  if (authList === null && observed.length) console.log(`observe-only (D-0696, D-1597; never authors HALT.md): ${observed.map((t) => t.name).join(", ")}`);
  if (authList !== null) console.log(`authoritative: [${authList.join(", ")}] — authoring ${authoring.map((t) => t.name).join(", ") || "none"}; observe-only trips: ${observed.map((t) => t.name).join(", ") || "none"}`);
  if (tripped.length > 0 && authoring.length === 0) {
    console.log(`HALT.md not written: every TRIPPED tripwire is observe-only (${observed.map((t) => t.name).join(", ")})`);
    process.exit(2);
  }
  if (tripped.length > 0) {
    // I3-0821 (W3-15-R8): this write used to be unconditional, so an automatic --check destroyed
    // the reason text a human had written with --write — the file whose own body says "Only Jason
    // deletes this file", and whose reason exists nowhere else. A halt already in force is never
    // overwritten by the brake itself. Strictly fail-closed: an existing HALT.md already means
    // halted, the tripwires are still printed, and process.exit(2) below is unchanged on both paths.
    const standing = !dryRun ? readHaltFile() : null;
    if (standing) {
      console.log(`HALT.md already present since ${standing.at ?? "(unknown time)"} — left unmodified; tripwires now: ${tripped.map((t) => t.name).join(", ")}`);
    } else if (!dryRun) {
      mkdirSync(dirname(HALT_PATH), { recursive: true });
      const at = new Date().toISOString();
      const body = [
        "---",
        "written_by: halt.mjs",
        `at: ${at}`,
        `tripwires: [${authoring.map((t) => t.name).join(", ")}]`,
        "---",
        "",
        ...authoring.map((t) => `- ${t.name}: ${t.evidence}`),
        ...(observed.length ? ["", "Also TRIPPED, observe-only (not promoted; these did not author this file):", ...observed.map((t) => `- ${t.name}: ${t.evidence}`)] : []),
        "",
        "Only Jason deletes this file.",
        "",
      ].join("\n");
      writeFileSync(HALT_PATH, body);
      console.log(`HALT.md written: ${authoring.map((t) => t.name).join(", ")}`);
    }
    process.exit(2);
  }
  process.exit(0);
}

if (IS_MAIN) {
  console.error("usage: halt.mjs --status | --check [--gate-log <file>] [--recall a/b] [--dry-run] [--authoritative <name,name>] | --write \"<reason>\" [--replace]");
  process.exit(2);
}
