/** Expand a syntactically-plausible IPv6 literal to its 8 hextet groups,
 * resolving `::` elision. Returns null when the input is not a well-formed
 * IPv6 literal — the caller must fail closed on null, never fall back to
 * echoing the raw string (PS-131). */
function expandIpv6(ip: string): string[] | null {
  const addr = ip.split("%")[0] ?? ""; // strip a zone id (fe80::1%eth0)
  const halves = addr.split("::");
  if (halves.length > 2) return null; // more than one "::" is never valid
  const headRaw: string | undefined = halves[0];
  const tailRaw: string | undefined = halves.length === 2 ? halves[1] : undefined;
  const splitGroups = (s: string | undefined) => (s ? s.split(":") : []);
  const head = splitGroups(headRaw);
  const tail = splitGroups(tailRaw);
  const isHextet = (g: string) => /^[0-9a-fA-F]{1,4}$/.test(g);
  if (!head.every(isHextet) || !tail.every(isHextet)) return null;
  const elided = halves.length === 2;
  const missing = 8 - head.length - tail.length;
  if (!elided && missing !== 0) return null; // no "::" must give exactly 8 groups
  if (elided && missing < 1) return null; // "::" must stand in for at least one group
  const full = elided ? [...head, ...new Array(missing).fill("0"), ...tail] : head;
  return full.map((g) => g.toLowerCase().replace(/^0+(?=.)/, "") || "0");
}

/** Mask a client IP to a coarse prefix (v4 /24, v6 /48) so the throttle table
 * holds no precise fan identifier.
 *
 * Fail closed (PS-131): an input that is not a well-formed IPv4 or IPv6
 * literal is NEVER echoed back raw — every well-formed address is masked,
 * so a malformed one must never be the sole exception that is stored in
 * full. It collapses to a shared, non-identifying sentinel, the same
 * fail-closed shape `NO_IP_BUCKET` already uses for an absent header.
 *
 * IPv4-mapped/compatible IPv6 literals (`::ffff:1.2.3.4`, `::1.2.3.4`) are
 * masked by their embedded v4 address (AF-13/V-06): every such client
 * shares hextets 0-2 (all-zero) by construction, so masking them as v6
 * would collapse ALL of them into one /48 bucket regardless of
 * canonicalisation — the "too few buckets" failure the finder describes.
 * Routing them through the v4 rule instead restores per-/24 granularity. */
export function maskIp(ip: string): string {
  const INVALID = "invalid::/0";
  const lastColon = ip.lastIndexOf(":");
  const lastDot = ip.lastIndexOf(".");
  if (ip.includes(":") && lastDot > lastColon) {
    return maskIp(ip.slice(lastColon + 1));
  }
  if (ip.includes(":")) {
    const groups = expandIpv6(ip);
    return groups ? `${groups[0]}:${groups[1]}:${groups[2]}::/48` : INVALID;
  }
  const parts = ip.split(".");
  return parts.length === 4 ? `${parts[0]}.${parts[1]}.${parts[2]}.0/24` : INVALID;
}

/**
 * Attach a push token to an EXISTING (authenticated) account, and return the
 * device id. This is the folded-in replacement for anonymous device
 * registration: the caller must already hold a fan session (the route resolves
 * it via fanId), so a device can only ever be attached to the account that
 * proved it.
 *
 * A push token is globally unique (one physical install), so a re-attach MOVES
 * the token to `accountId` — if the same device was signed in as a different
 * fan before, its old owner stops receiving this device's pushes (correct: you
 * must never deliver one fan's private alerts to whoever holds the device now).
 * Idempotent for the same (account, token).
 */
export async function attachDevice(
  db: DbLike,
  accountId: string,
  platform: string,
  pushToken: string,
  now: () => string = () => new Date().toISOString(),
): Promise<{ device_id: string }> {
  if (platform !== "apns" && platform !== "fcm") throw new AccountError("platform must be apns|fcm");
  if (typeof pushToken !== "string" || pushToken.length === 0 || pushToken.length > TOKEN_MAX) {
    throw new AccountError("invalid push token");
  }
  await requireAccount(db, accountId); // fail closed if the session's account vanished
  const deviceId = mintId("dev");
  const ts = now();
  // Upsert on the unique push_token: a known token is re-pointed at this
  // account (and its platform/timestamp refreshed); a new one inserts.
  // I3-1223: a MOVE to another account is a new binding and gets a NEW
  // device_id. The fan-out's dead-token cleanup deletes by the device_id it
  // sent to (alerts.ts `DELETE FROM devices WHERE device_id IN`), so reusing
  // the old id let a failure observed for the previous owner's binding delete
  // the one just made. Same-account refresh keeps its id (idempotent re-attach).
  await db
    .prepare(
      `INSERT INTO devices (device_id, account_id, platform, push_token, created_at)
       VALUES (?, ?, ?, ?, ?)
       ON CONFLICT(push_token) DO UPDATE SET
         device_id = CASE WHEN devices.account_id = excluded.account_id THEN devices.device_id ELSE excluded.device_id END,
         account_id = excluded.account_id,
         platform = excluded.platform,
         created_at = excluded.created_at`,
    )
    .bind(deviceId, accountId, platform, pushToken, ts)
    .run();
  // On conflict the pre-existing row keeps its device_id — return the real one.
  const row = await db.prepare("SELECT device_id FROM devices WHERE push_token = ?").bind(pushToken).first<{ device_id: string }>();
  return { device_id: row?.device_id ?? deviceId };
}

/**
 * Register a device → new account + device, in one transaction. This is the
 * account-bootstrap primitive: it is NOT exposed by any route (the public
 * `/account/device` endpoint is authenticated and calls `attachDevice`
 * instead). Kept for internal callers and test fixtures. Caller mints the
 * session.
 */
export async function registerDevice(
  db: DbLike,
  platform: string,
  pushToken: string,
  now: () => string = () => new Date().toISOString(),
  opts: { throttleKey?: string; nowMs?: number } = {},
): Promise<RegisterResult> {
  if (platform !== "apns" && platform !== "fcm") throw new AccountError("platform must be apns|fcm");
  if (typeof pushToken !== "string" || pushToken.length === 0 || pushToken.length > TOKEN_MAX) {
    throw new AccountError("invalid push token");
  }
  if (opts.throttleKey) {
    await checkThrottle(db, `reg:${opts.throttleKey}`, opts.nowMs ?? Date.now(), MAX_REGISTRATIONS_PER_IP_DAY);
  }
  const accountId = mintId("acc");
  const deviceId = mintId("dev");
  const ts = now();
  await db.batch([
    db.prepare(ACCOUNT_INSERT_SQL).bind(accountId, ts),
    db.prepare("INSERT INTO devices (device_id, account_id, platform, push_token, created_at) VALUES (?, ?, ?, ?, ?)").bind(deviceId, accountId, platform, pushToken, ts),
  ]);
  return { account_id: accountId, device_id: deviceId };
}

/** How long a device-attach challenge stays valid. Short — it only has to
 * survive one push round-trip. */
export const DEVICE_CHALLENGE_TTL_MS = 5 * 60 * 1000;

/**
 * Create a device-attach proof-of-possession challenge. Stores the secret
 * `code` (delivered to the device out-of-band via push) bound to the account +
 * token, with a short expiry. Validates platform/token like attachDevice.
 */
export async function createDeviceChallenge(
  db: DbLike,
  accountId: string,
  platform: string,
  pushToken: string,
  challengeId: string,
  code: string,
  nowMs: number,
  now: () => string = () => new Date().toISOString(),

  if (platform !== "apns" && platform !== "fcm") throw new AccountError("platform must be apns|fcm");
  if (typeof pushToken !== "string" || pushToken.length === 0 || pushToken.length > TOKEN_MAX) throw new AccountError("invalid push token");
  await requireAccount(db, accountId);

    .bind(challengeId, accountId, platform, pushToken, code, now(), expiresAt)
    .run();
}

/**
 * Redeem a challenge single-use: the row must exist, belong to THIS account,
 * match the code, and be unexpired — all enforced (and the row consumed)
 * atomically by a `DELETE … RETURNING`, so a replay or a concurrent second
 * attempt finds nothing. Returns the (platform, push_token) to attach, or null.
 */
export async function redeemDeviceChallenge(db: DbLike, accountId: string, challengeId: string, code: string, nowMs: number): Promise<{ platform: string; push_token: string } | null> {
  if (typeof challengeId !== "string" || typeof code !== "string" || code.length === 0) return null;
  const nowIso = new Date(nowMs).toISOString();
  const row = await db
    .prepare("DELETE FROM device_challenges WHERE challenge_id = ? AND account_id = ? AND code = ? AND expires_at > ? RETURNING platform, push_token")
    .bind(challengeId, accountId, code, nowIso)
    .first<{ platform: string; push_token: string }>();
  return row ?? null;
}

/** Cron prune: drop expired challenges. */
export async function pruneDeviceChallenges(db: DbLike, beforeIso: string): Promise<number> {
  const r = await db.prepare("DELETE FROM device_challenges WHERE expires_at < ?").bind(beforeIso).run();
  return r.changes;
}

/** Save a creator (idempotent). Fan-scoped.
 *
 * The target is resolved through `livePassport` — the ONE gated point-read —
 * so retracted, suppressed and never-listed are the same 404. This route used
 * to answer 200 for a SUPPRESSED subject and 404 for one that was never listed:
 * a fan session plus a passport id was enough to ask "did this person ask to be
 * delisted?" and get an answer (review HIGH 3). `requestCreator` had been given
 * the same fix by hand and this one was missed, which is why the gate now lives
 * in the lookup rather than in each caller. */
export async function saveCreator(
  db: DbLike,
  accountId: string,
  passportId: string,
  now: () => string = () => new Date().toISOString(),
  /** `ANALYTICS_FAN_SECRET`. Undefined is refused by `fanKey` (ADR-0006 §2):
   * a follow that cannot be counted is not silently uncounted. */
  fanSecret?: string,
  pubEnv?: AnalyticsTickEnv,
  pubCtx?: { waitUntil(p: Promise<unknown>): void },
): Promise<void> {
  await requireAccount(db, accountId);
  if (!/^psp_[0-9a-f]{16}$/.test(passportId)) throw new AccountError("bad passport id");
  if (!(await livePassport(db, passportId))) throw new AccountError("no such creator", 404);
  // Per-account save cap (bounds table growth / fan-out abuse). Only consulted
  // when the target isn't already saved — an idempotent re-save is always ok,
  // cap or no cap.
  const already = await db.prepare("SELECT 1 AS x FROM saves WHERE account_id = ? AND passport_id = ?").bind(accountId, passportId).first();
  if (already) return;
  // THE CAP IS IN THE INSERT, not a COUNT read before it. Counted-then-inserted,
  // it is overshot by the width of a concurrent burst — measured at 5003 saves
  // against a cap of 5000 with four savers held on that read, the same defect
  // that measured 13 passkeys against a cap of 10 (fan-auth.ts) and 12 OTP codes
  // against a cap of 5. An unwritten row is the cap; the PK still makes a repeat
  // save a silent no-op rather than a limit error.
  //
  // THE ANALYTICS ROW RIDES THIS BATCH (ADR-0006 §2). One `db.batch` and not a
  // second write afterwards, so a Worker that dies between them cannot leave a
  // creator short a supporter with nothing to detect it from.
  //
  // AND THE EVENT IS GATED ON THE FOLLOW NOT ALREADY EXISTING, evaluated INSIDE
  // this transaction and AHEAD of the INSERT (`DbLike.batch` is ordered and
  // all-or-nothing — db.ts:19-20). The `already` pre-read above sits OUTSIDE the
  // transaction and closes the SEQUENTIAL repeat only: two concurrent saves both
  // read no row, both arrive here, `(account_id, passport_id)` is the PRIMARY KEY
  // (0004_fan_accounts.sql:155-160) so the second INSERT is silently no-op'd, and
  // the cap predicate — which used to be this event's ONLY guard — is true for
  // both of them. That was +1 `new_supporters` per concurrent request for one
  // real follow, and `supporters` is a stock the creator never gets back.
  // Ordered first, the two batches serialise and exactly one sees no row.
  const ts = now();
  const cap = await fanSavedCap(db, accountId, Date.parse(ts));
  const key = await fanKey(fanSecret, accountId, passportId);
  // The INSERT's own admission condition, spelled the same way and bound the same
  // way so the two cannot drift: not already following, and under the cap.
  //
  // QUALIFIED AT THE DOOR (QUEUE #526, D-0819(5), Option A+): the follow is
  // COUNTED — the `fan_followed` row written and `saves.counted_at` set — only
  // when the saver is in `qualified_fan_accounts` inside this batch. Both the
  // gate and the INSERT's CASE read that view in the same transaction, so they
  // see one answer. An unqualified save is stored uncounted and
  // `sweepQualifiedFollows` counts it once, on the day the account qualifies.
  const admits = {
    sql: `NOT EXISTS (SELECT 1 FROM saves WHERE account_id = ? AND passport_id = ?) AND (SELECT COUNT(*) FROM saves WHERE account_id = ?) < ? AND ${QUALIFIED_SQL}`,
    params: [accountId, passportId, accountId, cap, accountId],
  };
  const span = outboxSpan(db);
  const spanRes = await db.batch([
    span.open,
    // `source` is omitted: this door has no codex source in hand, and a dim
    // invented at the call site is not a closed enum (INV-051).
    ...outboxStatements(db, { kind: "fan_followed", passportId, occurredDay: occurredDay(ts), fanKey: key }, { accountId, extraWhere: admits }),
    db
      .prepare(
        // #680 (I16-W04C): COUNTED iff the fan_followed row above was written — qualified AND logging not paused
        // (analytics-events.ts LOGGING_NOT_PAUSED gates that row). A paused fan's save stays uncounted and
        // `sweepQualifiedFollows` counts it once, on a tick after logging resumes.
        `INSERT INTO saves (account_id, passport_id, created_at, counted_at)
       SELECT ?, ?, ?, CASE WHEN ${QUALIFIED_SQL} AND ${LOGGING_NOT_PAUSED} THEN ? END WHERE (SELECT COUNT(*) FROM saves WHERE account_id = ?) < ?
       ON CONFLICT(account_id, passport_id) DO NOTHING`,
      )
      .bind(accountId, passportId, ts, accountId, accountId, ts, accountId, cap),
    // #839a (I16-FXC, D-0924(7)): the save's UTC day is its COMMIT statement's (SQLite `now`), recorded beside the row this
    // batch inserted (matched on the request instant it carries); `ratings.ts populationValues` reads it before created_at.
    db
      .prepare(
        `INSERT OR IGNORE INTO community_settled_days (account_id, passport_id, kind, settled_day)
         SELECT account_id, passport_id, 'save', ${TOMBSTONE_COMMIT_DAY} FROM saves WHERE account_id = ? AND passport_id = ? AND created_at = ?`,
      )
      .bind(accountId, passportId, ts),
    span.close,
  ]);
  const seqs = outboxSpanSeqs(spanRes[0], spanRes[spanRes.length - 1]);
  if (pubEnv && pubCtx) publishAfterCommit(pubEnv, pubCtx, seqs);
  // `DbLike.batch` returns `unknown[]` — no `changes` (db.ts:20), so the cap is
  // read off the TABLE rather than off the statement. The re-read is the one
  // this branch already did; it now runs on every save instead of only on a
  // refusal, which is the price of the atomic emit.
  const landed = await db.prepare("SELECT 1 AS x FROM saves WHERE account_id = ? AND passport_id = ?").bind(accountId, passportId).first();
  if (!landed) throw new AccountError(SAVE_LIMIT_REFUSAL, 400);
}

/**
 * QUEUE #793 (I8RG-PRIV-01/-02): the `community_tombstones` row (migration 0157)
 * an un-save / un-collect leaves for the rest of its UTC day, so the community
 * legs of `ratings.ts populationValues` (live arm day-pure: row made before T,
 * fan qualified before T; tombstone arm `created_day < T AND erased_day >= T`)
 * do not lose a counted vote mid-day. The SAME expression as the erasure path
 * (`account_data.ts deleteAccount`): first counted day = the later of the row
 * day and the qualification day, weight `fanDayWeightExpr` at the erasure day.
 * Only a row the live arm counts on `day` gets one (`first counted day < day`;
 * the inner qualified join drops a never-qualified fan). It reads the row it
 * tombstones, so it must sit in the DELETE's batch, immediately before it: a
 * second concurrent DELETE then finds no row and writes nothing.
 */
// #839b: no day parameter — the day is the commit statement's own (TOMBSTONE_COMMIT_DAY), so no caller can choose it.
export function communityTombstoneStmt(db: DbLike, kind: "save" | "collect", accountId: string, passportId: string): DbStatement {
  const table = kind === "save" ? "saves" : "card_collections";
  // #839a (I16-FXC): the row's day is its settled (commit) day when one was recorded (0202), else its created_at day.
  const firstCounted = "MAX(COALESCE(sd.settled_day, substr(t.created_at, 1, 10)), substr(qf.qualified_at, 1, 10))";
  // #839b (D-0924(7), I16-L05): an erasure belongs to the UTC day of ITS COMMIT STATEMENT'S server clock — SQLite's
  // own `now` evaluated by this INSERT — never a request clock read before the batch: a request read at 23:59:59
  // whose batch commits at 00:00:01 used to stamp yesterday, and `erased_day >= T` then dropped the fan mid-day.
  return db
    .prepare(
      `INSERT INTO community_tombstones (passport_id, kind, created_day, erased_day, fan_weight)
       SELECT t.passport_id, '${kind}', ${firstCounted}, ${TOMBSTONE_COMMIT_DAY}, ${COMMIT_DAY_FAN_WEIGHT}
         FROM ${table} t LEFT JOIN community_settled_days sd ON sd.account_id = t.account_id AND sd.passport_id = t.passport_id AND sd.kind = '${kind}'
         ${qualifiedFanJoin("t.account_id")} WHERE t.account_id = ? AND t.passport_id = ? AND ${firstCounted} < ${TOMBSTONE_COMMIT_DAY}`,
    )

/** Unsave (idempotent). The DELETE always runs; the `fan_unfollowed` event is
 * emitted only when there was a follow to end.
 *
 * THE GUARD IS IN THE BATCH, NOT IN THE PRE-READ. An unconditional emit would
 * let any client decrement a creator's `supporters` stock once per request by
 * unsaving somebody they never saved, so the emit carries `EXISTS (… FROM saves
 * …)` as its `extraWhere`, evaluated inside the transaction and ahead of the
 * DELETE. The pre-read below is a skip-read that avoids the work and decides
 * nothing authoritative (ADR-0003 §11). It USED to be the event's only guard,
 * and that is a check-then-act window a same-instant double-unsave walked
 * straight through: both callers read a row, both emitted, and the stock went
 * one below what the `saves` detail can account for — the phantom this docblock
 * used to claim was impossible. The DELETE is left unconditional. */
export async function unsaveCreator(
  db: DbLike,
  accountId: string,
  passportId: string,
  now: () => string = () => new Date().toISOString(),
  fanSecret?: string,
  pubEnv?: AnalyticsTickEnv,
  pubCtx?: { waitUntil(p: Promise<unknown>): void },
): Promise<void> {
  await requireAccount(db, accountId);
  const del = db.prepare("DELETE FROM saves WHERE account_id = ? AND passport_id = ?").bind(accountId, passportId);
  const following = await db.prepare("SELECT 1 AS x FROM saves WHERE account_id = ? AND passport_id = ?").bind(accountId, passportId).first();
  const ts = now();
  // QUEUE #793: the tombstone rides with EVERY DELETE of this door, the skip-read miss included.
  const tomb = communityTombstoneStmt(db, "save", accountId, passportId);
  // #839a (I16-FXC): the settled day goes with the save it settles, after the tombstone has read it.
  const unsettle = db.prepare("DELETE FROM community_settled_days WHERE account_id = ? AND passport_id = ? AND kind = 'save'").bind(accountId, passportId);
  if (!following) {
    await db.batch([tomb, del, unsettle]);
    return;
  }
  const key = await fanKey(fanSecret, accountId, passportId);
  // THE GUARD IS RESTATED INSIDE THE BATCH, AHEAD OF THE DELETE. The `following`
  // pre-read above is outside the transaction, so two concurrent unsaves both
  // read a row and, before this, both emitted — a PHANTOM decrement of a stock
  // the `saves` detail can never account for. Serialised, exactly one sees it.
  // COUNTED saves only (QUEUE #526): an uncounted save never added a supporter,
  // so ending it must not subtract one.
  const wasFollowing = { sql: "EXISTS (SELECT 1 FROM saves WHERE account_id = ? AND passport_id = ? AND counted_at IS NOT NULL)", params: [accountId, passportId] };
  const span = outboxSpan(db);
  // #680 CONVERSE (D-0924(7), I16-L05): `fan_unfollowed` is fan-keyed, so the outbox gate drops it while this fan's
  // logging is paused (analytics-events.ts LOGGING_NOT_PAUSED) — and a COUNTED save un-saved then left the creator a
  // supporter forever. The complement emits the creator-scoped `supporter_erased` (no fan key, no account id: the
  // decrement deleteAccount uses) for exactly that case: counted AND paused. Both read the row before the DELETE in
  // this batch, and their pause terms are one constant and its negation, so exactly one of the two fires for a
  // counted save and neither for an uncounted one.
  const pausedDecrement = db
    .prepare(outboxSelectSql("supporter_erased", { table: "saves", where: "account_id = ? AND passport_id = ? AND counted_at IS NOT NULL", extraWhere: LOGGING_PAUSED }))
    .bind(ts, accountId, passportId, accountId);
  const spanRes = await db.batch([span.open, ...outboxStatements(db, { kind: "fan_unfollowed", passportId, occurredDay: occurredDay(ts), fanKey: key }, { accountId, extraWhere: wasFollowing }), pausedDecrement, tomb, del, unsettle, span.close]);
  const seqs = outboxSpanSeqs(spanRes[0], spanRes[spanRes.length - 1]);
  if (pubEnv && pubCtx) publishAfterCommit(pubEnv, pubCtx, seqs);
}

/** THE CATCH-UP (QUEUE #526, D-0819(5), Option A+). Counts, once, every save
 * made while its account was unqualified and whose account is qualified now:
 * the `fan_followed` row the save door withheld is written dated TODAY (the
 * tick's day — a sealed past day is never re-opened), and `counted_at` is set.
 *
 * Per save ONE batch, gate FIRST and claim SECOND, both bound to the same
 * predicate — the save still exists, is still uncounted, and its account is
 * qualified (INV-058): an un-save or an erasure landing between the read and
 * this batch makes both statements no-ops, and a second tick (or a concurrent
 * one) finds `counted_at` set and writes nothing. The message is the save
 * door's, field for field (no dims), so the analytics store learns nothing it
 * did not learn before. Returns the number of saves it attempted. With
 * `pubEnv`/`pubCtx` it publishes once after the loop, as the save door does. */
export async function sweepQualifiedFollows(
  db: DbLike,
  fanSecret: string | undefined,
  now: () => string = () => new Date().toISOString(),
  limit: number = QUALIFIED_FOLLOW_SWEEP_LIMIT,
  pubEnv?: AnalyticsTickEnv,
  pubCtx?: { waitUntil(p: Promise<unknown>): void },
): Promise<number> {
  const { results } = await db
    .prepare(
      `SELECT s.account_id, s.passport_id FROM saves s
        WHERE s.counted_at IS NULL AND EXISTS (SELECT 1 FROM qualified_fan_accounts q JOIN ${FAN_COSTED_QUALIFICATION_TABLE} fcq ON fcq.account_id = q.account_id WHERE q.account_id = s.account_id)
          AND NOT EXISTS (SELECT 1 FROM codex_settings c WHERE c.account_id = s.account_id AND c.logging = 0)
        ORDER BY s.created_at, s.account_id, s.passport_id LIMIT ?`,
    )
    .bind(limit)
    .all<{ account_id: string; passport_id: string }>();
  // #862: the seqs THESE batches wrote, so the one publish below sends them and no other request's row.
  const seqs: number[] = [];
  for (const r of results) {
    const ts = now();
    const key = await fanKey(fanSecret, r.account_id, r.passport_id);
    const uncounted = {
      sql: `EXISTS (SELECT 1 FROM saves WHERE account_id = ? AND passport_id = ? AND counted_at IS NULL) AND ${QUALIFIED_SQL}`,
      params: [r.account_id, r.passport_id, r.account_id],
    };
    const span = outboxSpan(db);
    const spanRes = await db.batch([
      span.open,
      ...outboxStatements(db, { kind: "fan_followed", passportId: r.passport_id, occurredDay: occurredDay(ts), fanKey: key }, { accountId: r.account_id, extraWhere: uncounted }),
      db
        // #680: and only while logging is not paused, the same predicate the fan_followed row above carries.
        .prepare(`UPDATE saves SET counted_at = ? WHERE account_id = ? AND passport_id = ? AND counted_at IS NULL AND ${QUALIFIED_SQL} AND ${LOGGING_NOT_PAUSED}`)
        .bind(ts, r.account_id, r.passport_id, r.account_id, r.account_id),
      span.close,
    ]);
    seqs.push(...outboxSpanSeqs(spanRes[0], spanRes[spanRes.length - 1]));
  }
  // ONE publish for the tick, after every batch has committed: with no seqs it sends
  // everything owed, so the rows above leave now instead of waiting for a drain (E-1).
  if (pubEnv && pubCtx && results.length > 0) publishAfterCommit(pubEnv, pubCtx, seqs);
  return results.length;
}

/**
 * `action`/`item_id` filter shapes for `/ops/audit` (QUEUE #84 part 3). Closed,
 * bounded alphabets — a value outside them cannot be a real `audit_log` row
 * and is REFUSED (400, at the route) rather than run as a query: an
 * unrecognised shape is never widened into a `LIKE` scan.
 *
 * `action` mirrors every literal this service writes to `audit_log.action`:
 * lowercase, colon-separated, snake_case segments (`account_added:refused`,
 * `enqueue:claim`, `warning:second_look`, …). There is no DB `CHECK` or
 * central enum to walk (the column is free TEXT, written from a dozen call
 * sites) — the shape below is the closed population this filter can match,
 * not a hand-list of today's action strings, so a new call site that follows
 * the same convention is filterable without touching this file.
 *
 * `item_id` mirrors the two id shapes an audit row's `item_id` actually
 * carries: an opaque mint (`ids.ts`, e.g. a queue id) and a subject hash
 * (`sha256:<hex>`, e.g. `warnSubject` in `events.ts`'s `account_added:refused`
 * rows). Both are closed alphabets (hex/underscore/hyphen/colon), bounded to
 * 128 chars (`ids.ts`'s and a sha256 hex digest's own bound).
 */
const AUDIT_ACTION_SHAPE = /^[a-z][a-z0-9_]*(:[a-z0-9_]+)+$/;
const AUDIT_ITEM_SHAPE = /^[A-Za-z0-9][A-Za-z0-9_:-]{0,127}$/;

export function isValidAuditAction(s: string): boolean {
  return s.length <= 64 && AUDIT_ACTION_SHAPE.test(s);
}

export function isValidAuditItem(s: string): boolean {
  return AUDIT_ITEM_SHAPE.test(s);
}

/**
 * QUEUE #167 S-10 — the `actor` filter's shapes. The stored actor is an
 * `op_<hex>` id or a `system:*`/`AUTO_*` literal (a closed alphabet, the same
 * bound as an item id), or an operator's EMAIL, which `auditList` resolves to
 * the op_ id(s) of the PERSON behind it through `operator_identities` — an
 * email is never itself stored in `audit_log`, so it is never matched there.
 */
const AUDIT_ACTOR_EMAIL = /^[^\s@<>"]{1,64}@[^\s@<>"]{1,190}$/;
export function isValidAuditActor(s: string): boolean {
  return AUDIT_ITEM_SHAPE.test(s) || (s.length <= 254 && AUDIT_ACTOR_EMAIL.test(s));
}

/**

 * optionally narrowed to one `action` and/or one `item_id` — both exact
 * equality on indexed-shape columns, never a scan. Callers (the route) are
 * expected to have already refused a malformed `action`/`item_id` with
 * `isValidAuditAction`/`isValidAuditItem`; this function does not re-validate,
 * the same division the route already draws for `type` on `/ops/review`.
 *
 * The stored actor is an opaque `op_<hex>` (operators.ts) so no staff email is
 * ever written to an audit row. Accountability is not lost by that: the id is
 * resolved back to a human HERE, on the read, through the one mutable table

export async function auditList(db: DbLike, opts: { limit?: number; before?: number; action?: string; item_id?: string; actor?: string } = {}): Promise<AuditEntry[]> {
  const limit = Math.max(1, Math.min(200, opts.limit ?? 50));
  const conditions: string[] = [];
  const params: unknown[] = [];
  if (opts.actor !== undefined) {
    // QUEUE #668 / D-0817(7): a LEGACY row's actor is `op:<email>` (the shape
    // before `op_` ids, operators.ts:11). An actor given in that shape is read
    // as its email; each arm below also matches the person's legacy rows.

    conditions.push(await actorCondition(db, actor, params));

  const q = db
    .prepare(`SELECT audit_id, actor_id, actor_kind, action, detail, item_id, created_at FROM audit_log ${where} ORDER BY audit_id DESC LIMIT ?`)
    .bind(...params, limit);
  const { results } = await q.all<AuditRow>();

  const legacy = [...new Set(results.map((r) => r.actor_id).filter((a) => LEGACY_OP_ACTOR.test(a)).map((a) => a.slice(3).toLowerCase()))];
  const opOfEmail = new Map<string, string>();

    const { results: ids } = await db
      .prepare(`SELECT op_id, email FROM operator_identities WHERE email IN (${chunk.map(() => "?").join(",")})`)

/** QUEUE #773 (D-0858(9)): the most identities `auditList` binds per list; above it, the subquery form. */
export const AUDIT_ACTOR_BIND_CAP = 40;

/**
 * QUEUE #773 limb A (D-0858(9)): the actor filter as ONE bracketed condition
 * (D-0823(6)), its identities resolved by a read of `operator_identities` FIRST
 * and BOUND, so `(actor_id IN (?..) OR LEGACY_ADDR IN (?..))` seeks
 * idx_audit_actor_id and 0150's idx_audit_legacy_addr_id — a subquery on either
 * arm plans SCAN audit_log (i6-773-audit-actor-index.test.ts). Semantics as the
 * subquery form: an email names every op_ id of the same person (the
 * COALESCE(person_id, op_id) the dual-control gate counts by) and every legacy
 * address of theirs plus the email itself (D-0825(1): LEGACY_ADDR is lowercased,
 * stored emails are lowercased); an op_/system actor adds its own address. An
 * unknown email binds only itself, so it matches nothing but its own legacy
 * rows — never everything. Above AUDIT_ACTOR_BIND_CAP identities the subquery
 * form (correct, slower) is used. The two reads are not atomic: an identity
 * linked between them is missed by that one page load.
 */
async function actorCondition(db: DbLike, actor: string, params: unknown[]): Promise<string> {
  let ids: string[];
  let addrs: string[];
  if (actor.includes("@")) {
    const addr = actor.toLowerCase();
    const addrForms = legacyAddrForms(addr);
    const { results } = await db
      .prepare(
        "SELECT o.op_id AS op_id, o.email AS email FROM operator_identities o WHERE COALESCE(o.person_id, o.op_id) IN " +
          "(SELECT COALESCE(person_id, op_id) FROM operator_identities WHERE email = ?) LIMIT ?",
      )
      .bind(addr, AUDIT_ACTOR_BIND_CAP + 1)
      .all<{ op_id: string; email: string }>();
    addrs = [...new Set([...results.map((r) => r.email), addr])];
    const expanded = [...new Set(addrs.flatMap(legacyAddrForms))];
    if (results.length > AUDIT_ACTOR_BIND_CAP || expanded.length > AUDIT_ACTOR_BIND_CAP) {
      params.push(addr, addr, ...addrForms);
      return (
        "(actor_id IN (SELECT o.op_id FROM operator_identities o WHERE COALESCE(o.person_id, o.op_id) IN " +
        "(SELECT COALESCE(person_id, op_id) FROM operator_identities WHERE email = ?)) " +
        `OR ${LEGACY_ADDR} IN (SELECT o.email FROM operator_identities o WHERE COALESCE(o.person_id, o.op_id) IN ` +
        `(SELECT COALESCE(person_id, op_id) FROM operator_identities WHERE email = ?)) OR ${LEGACY_ADDR} IN (${addrForms.map(() => "?").join(", ")}))`
      );
    }
    ids = results.map((r) => r.op_id);
    addrs = expanded;
  } else {
    const own = await db.prepare("SELECT email FROM operator_identities WHERE op_id = ?").bind(actor).first<{ email: string }>();
    ids = [actor];
    addrs = own ? legacyAddrForms(own.email) : [];
  }
  const arms: string[] = [];
  if (ids.length) {
    arms.push(`actor_id IN (${ids.map(() => "?").join(", ")})`);
    params.push(...ids);
  }
  if (addrs.length) {
    arms.push(`${LEGACY_ADDR} IN (${addrs.map(() => "?").join(", ")})`);
    params.push(...addrs);
  }
  return `(${arms.join(" OR ")})`;
}

/**
 * Every non-ASCII code point whose JS `toLowerCase()` is another string, keyed by that result. I7-FX1: a code
 * point that lowercases to SEVERAL code points (U+0130 İ -> "i" + U+0307) is keyed by the whole sequence, so
 * `legacyAddrForms` can find it; they were skipped, which left 'op:İbrahim@x.com' unreachable by its own lowercase.
 */
function loweredFrom(): Map<string, string[]> {
  if (lowerSources) return lowerSources;
  const m = new Map<string, string[]>();
  for (let cp = 0x80; cp <= 0x1ffff; cp++) {
    if (cp >= 0xd800 && cp <= 0xdfff) continue;
    const ch = String.fromCodePoint(cp);
    const low = ch.toLowerCase();
    if (low === ch) continue;
    const l = m.get(low);
    if (l) l.push(ch);
    else m.set(low, [ch]);
  }
  lowerSources = m;
  return m;
}
/**
 * I7 #689: what SQLite `lower()` (ASCII only) can make of a stored legacy address whose JS lowercase is `addr`.
 * A stored 'ÉLODIE' reads 'Élodie' in LEGACY_ADDR while the filter is 'élodie'; each non-ASCII letter may have been
 * stored as itself or as any character that lowercases to it. `addr` itself is always first; the product is capped
 * at LEGACY_ADDR_FORMS_MAX (an address with more case-variable non-ASCII letters than that is matched by its
 * as-given and first-source forms only).
 */
export function legacyAddrForms(addr: string): string[] {
  const src = loweredFrom();
  // Units: a two-code-point lowercase sequence that some character lowercases to (i + U+0307) is ONE unit.
  const cps = [...addr];
  const chars: string[] = [];
  for (let i = 0; i < cps.length; i++) {
    const pair = i + 1 < cps.length ? cps[i]! + cps[i + 1]! : "";
    if (pair && [...pair].length === 2 && src.has(pair)) {
      chars.push(pair);
      i++;
    } else chars.push(cps[i]!);
  }
  const opts = chars.map((ch) => [ch, ...(src.get(ch) ?? [])]);
  const out = new Set<string>([addr, chars.map((ch, i) => opts[i]![opts[i]!.length > 1 ? 1 : 0]!).join("")]);
  let forms = [""];
  for (const o of opts) {
    const next: string[] = [];
    for (const f of forms) for (const c of o) if (next.length < LEGACY_ADDR_FORMS_MAX) next.push(f + c);
    forms = next;
  }
  for (const f of forms) if (out.size < LEGACY_ADDR_FORMS_MAX) out.add(f);
  return [...out];
}

async function closureAging(db: DbLike, where: string, col: "appeal_opened_at" | "appeal_deadline", cutoff: string, nowMs: number): Promise<ClosureAging> {
  const c = await db.prepare(`SELECT COUNT(*) AS n FROM account_closures WHERE ${where} AND ${col} < ?`).bind(cutoff).first<{ n: number }>();

/** I3-0818: two 6-hourly anchor periods. */
export const CHECKPOINT_STALE_MS = 12 * 60 * 60 * 1000;
/** History objects `checkpointSigningStale` reads back looking for the last verified one (16 days at 6-hourly). */
export const CHECKPOINT_STALE_READS = 64;

async function readCheckpoint(store: ObjectStore, key: string): Promise<CheckpointRecord | null> {
  try {
    const raw = await store.get(key);
    if (raw === null || raw.length > CHECKPOINT_READ_MAX_BYTES) return null;
    const r = JSON.parse(raw) as CheckpointRecord;
    return r !== null && typeof r === "object" ? r : null;
  } catch {
    return null;
  }
}

/**
 * I3-0818 (D-0829): IntegrityState keeps 'unverified' as not-broken; this is the
 * operator signal beside it. With published keys, `latest.json` verifying ends the
 * check (null). Otherwise the history is read newest first until a checkpoint
 * VERIFIES; `since` is the oldest `taken_at` in the unverified run above it (an
 * unsigned head re-anchors every run, so `latest.json`'s own time is never old).
 * Stale when `!(now - since <= CHECKPOINT_STALE_MS)` — an unreadable time is stale.
 * No checkpoint at all is not a signing signal (null).
 */
export async function checkpointSigningStale(store: ObjectStore, keys: CheckpointPublicKey[], nowMs: number): Promise<OverviewStats["checkpoint_signing_stale"]> {
  const latest = await readCheckpoint(store, CHECKPOINT_LATEST_KEY);
  if (latest && (await verifyCheckpointSignature(latest, keys)).state === "verified") return null;
  const history = (await store.list("checkpoints/")).filter((k) => parseCheckpointKey(k) !== null).sort().reverse().slice(0, CHECKPOINT_STALE_READS);
  let since: string | null = latest && typeof latest.taken_at === "string" ? latest.taken_at : null;
  let headSeq: number | null = latest && typeof latest.seq === "number" ? latest.seq : null;
  for (const key of history) {
    const rec = await readCheckpoint(store, key);
    if (rec && (await verifyCheckpointSignature(rec, keys)).state === "verified") break;
    const meta = parseCheckpointKey(key)!;
    if (headSeq === null) headSeq = meta.seq;
    since = meta.taken_at;
  }
  if (since === null) return null;
  return !(nowMs - Date.parse(since) <= CHECKPOINT_STALE_MS) ? { head_seq: headSeq, since } : null;
}

async function analyticsBuildFailure(db: DbLike): Promise<OverviewStats["analytics_build_failed"]> {
  const f = await db.prepare("SELECT cursor_id, updated_at FROM sweep_cursor WHERE name = ?").bind(BUILD_SWEEP_FAILURE_ROW).first<{ cursor_id: string; updated_at: string }>();
  if (!f?.updated_at) return null;
  const ok = await db.prepare("SELECT updated_at FROM sweep_cursor WHERE name = ?").bind(BUILD_SWEEP_REPORT_ROW).first<{ updated_at: string }>();
  if (ok?.updated_at && ok.updated_at > f.updated_at) return null;
  let err = "Error";
  try {
    const p: unknown = JSON.parse(f.cursor_id);
    if (p !== null && typeof p === "object" && typeof (p as { err?: unknown }).err === "string") err = (p as { err: string }).err;
  } catch { /* an unreadable record is still a failure */ }
  return { at: f.updated_at, err };
}

/** QUEUE #161 (D-0823(6)): the pre-build sweep runs on the 6-hourly trigger
 * (`wrangler.jsonc`, minute 17 of every sixth hour, which falls through to the artifact sweep that
 * calls `buildSweep`, index.ts). A recorded report older than TWO periods is
 * `stale`: a sweep that stopped without reaching its catch (isolate killed,
 * never scheduled) records nothing, so age is the only signal left. */
export const BUILD_SWEEP_PERIOD_MS = 6 * 60 * 60 * 1000;

/** The share of `BUILD_SWEEP_LIMIT` at which the backlog is flagged BEFORE it overflows (D-0709 "approaches"). */
export const BUILD_SWEEP_WARN_PERCENT = 80;

async function analyticsBuildBacklog(db: DbLike): Promise<OverviewStats["analytics_build"]> {
  const row = await db.prepare("SELECT cursor_id, updated_at FROM sweep_cursor WHERE name = ?").bind(BUILD_SWEEP_REPORT_ROW).first<{ cursor_id: string; updated_at: string }>();
  let r: Record<string, unknown> = {};
  try {
    const parsed: unknown = row ? JSON.parse(row.cursor_id) : null;
    if (parsed !== null && typeof parsed === "object") r = parsed as Record<string, unknown>;
  } catch { /* an unreadable report reads as no report */ }
  const n = (v: unknown): number | null => (typeof v === "number" && Number.isFinite(v) ? v : null);
  const granted = n(r.granted);
  const unreached = n(r.unreached);
  const limit = n(r.limit);
  const near_limit = (unreached !== null && unreached > 0) || (granted !== null && limit !== null && granted * 100 >= limit * BUILD_SWEEP_WARN_PERCENT);
  return { ran_at: row && granted !== null ? row.updated_at : null, day: typeof r.day === "string" ? r.day : null, granted, examined: n(r.examined), unreached, limit, near_limit };
}

/** QUEUE #161 (D-0823(6)): a recorded report older than two sweep periods is stale.
 * An unreadable time is stale, never healthy: `!(age <= bound)` is true for NaN.
 * No report at all is not claimed stale (`ran_at` null; the tile says none recorded). */
export function analyticsBuildStale(b: OverviewStats["analytics_build"], nowMs: number): boolean {
  return b.ran_at !== null && !(nowMs - Date.parse(b.ran_at) <= 2 * BUILD_SWEEP_PERIOD_MS);
}

  const slaCutoff = new Date(nowMs - APPEAL_DECISION_SLA_DAYS * DAY_MS).toISOString();
  const stats: OverviewStats = {

   * answer to "did someone already look at this / who decided this", and until
   * this field existed that answer lived only on the separate, 100-row-capped
   * audit screen. Sourced from `review_decisions` keyed on `(item_id,
   * from_state = 'open')` — every `approveItem`/`rejectItem` decision is
   * written from `open` (review.ts `reviewDecisionStmt`),
   * so this is the same row the write side already guarantees is unique. Null
   * for an item still `open` (nobody has decided it yet). */
  decided_by: string | null;
  decided_at: string | null;
  /** QUEUE #163 PS-014 — what OUR OWN DATA says about the account a manual
   * claim names, for the operator deciding it. Present only on an open-or-closed
   * `manual_claim` claim item whose account row still exists; null otherwise.
   * `first_observed_at` is when THIS LEDGER first recorded the account
   * (`passport_accounts.observed_at`), NOT the platform account's age — that
   * needs a platform API and is not sourced. `persistent_id` is whether the
   * claimed account is itself a platform pid (not recyclable) and
   * `pid_on_passport` how many active pid-kind accounts on the same platform
   * the same passport already holds — a handle beside a pid is anchored; a lone
   * handle can be a recycled name. */
  claim_signals: { id_kind: string; first_observed_at: string; persistent_id: boolean; pid_on_passport: number; status: string } | null;
  /** I38-OAUTHREV (INV-015): on an `oauth_claim` item only — the provider id the claimant's OAuth ceremony proved
   * (null on an item enqueued before it was recorded), and whether a proven OAuth proof of this account, or of this
   * passport on the same platform, recorded a different one (read now, not at enqueue). Null otherwise. */
  oauth_identity: { provider_pid: string | null; earlier_provider_pid_differs: boolean | null } | null;
  queued_at: string | null;
  created_at: string;
  updated_at: string;
}

/** Full review-queue row (incl. workflow columns), for the item detail view. */
export async function reviewItemDetail(db: DbLike, itemId: string): Promise<ReviewDetail | null> {
  const r = await db
    .prepare(
      `SELECT q.item_id, q.item_type, q.state, q.subject, q.payload, q.wording_id, q.second_look, q.second_look_by,
              q.disposition, q.disposition_note, q.queued_at, q.created_at, q.updated_at,
              d.decided_by AS decided_by, d.decided_at AS decided_at
         FROM review_queue q
         LEFT JOIN review_decisions d ON d.item_id = q.item_id AND d.from_state = 'open'
        WHERE q.item_id = ?`,
    )
    .bind(itemId)
    .first<Omit<ReviewDetail, "payload"> & { payload: string }>();
  if (!r) return null;
  let payload: unknown = r.payload;
  try {
    payload = JSON.parse(r.payload);
  } catch {
    /* leave raw */
  }
  const labels = await operatorLabels(db, [r.second_look_by, r.decided_by].filter((v): v is string => !!v));
  let claim_signals: ReviewDetail["claim_signals"] = null;
  const mp = payload !== null && typeof payload === "object" ? (payload as Record<string, unknown>) : null;
  if (r.item_type === "claim" && (mp?.source === "manual_claim" || mp?.source === "oauth_claim") && typeof mp.account_key === "string") {
    const acct = await db
      .prepare(
        "SELECT a.id_kind AS id_kind, a.observed_at AS observed_at, a.status AS status, " +
          "(SELECT COUNT(*) FROM passport_accounts b WHERE b.passport_id = a.passport_id AND b.platform = a.platform AND b.id_kind = 'pid' AND b.status = 'active') AS pids " +
          "FROM passport_accounts a WHERE a.account_key = ?",
      )
      .bind(mp.account_key)
      .first<{ id_kind: string; observed_at: string; status: string; pids: number }>();
    if (acct) claim_signals = { id_kind: acct.id_kind, first_observed_at: acct.observed_at, persistent_id: acct.id_kind === "pid", pid_on_passport: num(acct.pids), status: acct.status };
  }
  let oauth_identity: ReviewDetail["oauth_identity"] = null;
  if (r.item_type === "claim" && mp?.source === "oauth_claim" && typeof mp.account_key === "string") {
    const pid = typeof mp.provider_pid === "string" && mp.provider_pid.length > 0 ? mp.provider_pid : null;
    oauth_identity = {
      provider_pid: pid,
      earlier_provider_pid_differs: await earlierProviderPidDiffers(db, r.subject, mp.account_key, String(mp.platform ?? ""), pid),
    };
  }
  return {
    ...r,
    payload,
    claim_signals,
    oauth_identity,
    second_look_by: r.second_look_by ? labelFor(labels, r.second_look_by) : null,
    decided_by: r.decided_by ? labelFor(labels, r.decided_by) : null,
  };
}

/**
 * OU-05 — BOTH OPERATOR SEARCHES ADVERTISE `@handle` AND BOTH RENDER HANDLES
 * WITH THE MARKER ON, BUT THE HANDLE IS STORED BARE.
 *
 * `canonicalAccount` strips the leading `@` when it builds a handle subject
 * (packages/canonical), and `user_profiles.handle` holds the bare form. So the
 * exact string the console PRINTS — `@starfan` — went into `LIKE '%@starfan%'`
 * and matched nothing, and the screen answered with the healthy empty result:
 * "No account matches". On an operator surface a false empty is worse than an
 * error, because there is nothing to distinguish it from the subject genuinely
 * not existing.
 *
 * ONE leading marker, and only for the HANDLE limb. The id limb stays exact and
 * the email limb stays untouched — an email's `@` is internal, never leading,
 * so nothing about full-email lookup changes, and `@@x` is still nobody.
 */
function handleTerm(term: string): string {
  return term.startsWith("@") ? term.slice(1) : term;
}

/**
 * I3-0297/I3-0970 (KM-037/AA-03): D1's real LIKE-pattern ceiling is ~50 bytes
 * (measured, `aa_glob.mjs`; recovery.ts SIGNOFF_PREFIX_ID_BOUND is the same
 * finding fixed once already). The old `slice(0, 100)` bounded the TERM, not
 * the pattern D1 actually receives (`%term%`, +2 chars) — a 49+ char term
 * produced a 51+ byte pattern past the ceiling, and a 100-char term up to 102.
 * 44 keeps the wrapped pattern (44 + 2 = 46 bytes worst case for single-byte
 * input) under the ceiling with the same margin recovery.ts's bound uses.
 */
export const SEARCH_TERM_BOUND = 44;

/**
 * I3-0961 (A-11): a wildcard or a one/two-character term wraps to `%a%` and
 * matches almost every row — an operator "search" that is really a full
 * table browse. Below this length the LIKE limb is asked to match a needle
 * that cannot occur (a NUL byte: no stored display_name/email/handle should
 * ever contain one), leaving the EXACT-match limb (id/email/handle, unaffected
 * by this bound) as the only way a short input finds anything — which is
 * correct, since a short input can never usefully disambiguate by substring.
 */
export const MIN_LIKE_TERM = 3;

/**
 * I3-1022/I3-1752: each LIKE limb of an operator search stops after this many
 * hits, and only the union of the limbs' hits (<= 1 + 2 x cap ids) is joined and
 * sorted. A leading-wildcard LIKE cannot use an index, so a broad term used to
 * be evaluated over every joined row before `ORDER BY ... LIMIT` cut it down.
 * Past the cap the page is the first `limit` of a cap-sized sample, which a
 * narrower term resolves; below it the result is unchanged.
 */
export const SEARCH_SCAN_CAP = 500;
/**
 * I11-FX2 (I11-RG-MONEY D4): the ceiling is BYTES, and `SEARCH_TERM_BOUND` counts
 * characters (17 CJK characters, or 30 'é', is a 53-byte or 62-byte pattern that
 * real D1 refuses). Every operator LIKE (creator, account, subscription) is built
 * here, so the byte cap lives here: the wildcard-stripped term is cut to this many
 * UTF-8 bytes at a code-point boundary (truncated, as the character cap does), which
 * keeps `%term%` at <= 46 bytes whatever the script.
 */
export const SEARCH_TERM_BYTE_BOUND = 44;
const UTF8 = new TextEncoder();
function clampUtf8Bytes(s: string, maxBytes: number): string {
  let out = "";
  let used = 0;
  for (const ch of s) {
    const n = UTF8.encode(ch).length;
    if (used + n > maxBytes) break;
    out += ch;
    used += n;
  }
  return out;
}
export function likePattern(term: string): string {
  const stripped = clampUtf8Bytes(term.replace(/[%_]/g, ""), SEARCH_TERM_BYTE_BOUND); // strip LIKE wildcards from operator input
  if (stripped.length < MIN_LIKE_TERM) return "\u0000\u0000"; // never matches a real row
  return `%${stripped}%`;
}

/** Search creators by passport id (exact), display name, or account handle. */
export async function searchCreators(db: DbLike, q: string, opts: { limit?: number } = {}): Promise<CreatorSummary[]> {
  const term = (q ?? "").trim().slice(0, SEARCH_TERM_BOUND);
  if (!term) return [];
  const limit = Math.max(1, Math.min(100, opts.limit ?? 30));
  const like = likePattern(term);
  const handleLike = likePattern(handleTerm(term)); // OU-05: the advertised @handle form
  const { results } = await db
    .prepare(
      `WITH hit(passport_id) AS (
         SELECT passport_id FROM passports WHERE passport_id = ?
         UNION SELECT passport_id FROM (SELECT passport_id FROM passports WHERE display_name LIKE ? LIMIT ?)
         UNION SELECT passport_id FROM (SELECT passport_id FROM passport_accounts WHERE id LIKE ? LIMIT ?)
       )
       SELECT p.passport_id AS passport_id, p.display_name AS display_name, p.claim_state AS claim_state, p.retracted AS retracted, f.rank AS founding_rank
         FROM hit h
         JOIN passports p ON p.passport_id = h.passport_id
         LEFT JOIN founding f ON f.passport_id = p.passport_id
        ORDER BY p.display_name LIMIT ?`,
    )
    .bind(term, like, SEARCH_SCAN_CAP, handleLike, SEARCH_SCAN_CAP, limit)
    .all<CreatorSummary>();
  return results;
}

/** Mask a payout value (an address/handle) — enough to verify, not to copy. */
function maskTag(v: string): string {
  if (v.length <= 10) return v[0] + "…" + v.slice(-1);
  return v.slice(0, 5) + "…" + v.slice(-4);
}

/** Full operator view of one creator/passport. Null if unknown. */
export async function creatorDetail(db: DbLike, passportId: string): Promise<CreatorDetail | null> {
  const p = await db.prepare("SELECT display_name, claim_state, retracted FROM passports WHERE passport_id = ?").bind(passportId).first<{ display_name: string; claim_state: string; retracted: number }>();
  if (!p) return null;
  const founding = await db.prepare("SELECT rank FROM founding WHERE passport_id = ?").bind(passportId).first<{ rank: number }>();
  const accounts = (await db.prepare("SELECT account_key, platform, id_kind, id, url, status, relationship_basis FROM passport_accounts WHERE passport_id = ? ORDER BY platform, id").bind(passportId).all<CreatorDetail["accounts"][number]>()).results;
  const releases = (await db.prepare("SELECT release_id, title, version, status, item_count, created_at FROM releases WHERE passport_id = ? ORDER BY created_at DESC LIMIT 100").bind(passportId).all<CreatorDetail["releases"][number]>()).results;
  const tagRows = (await db.prepare("SELECT tag_type, tag_value, status, effective_at FROM payout_tags WHERE passport_id = ? AND tag_value <> '' ORDER BY created_at DESC LIMIT 50").bind(passportId).all<{ tag_type: string; tag_value: string; status: string; effective_at: string }>()).results;
  const reviewRows = (
    await db
      .prepare(
        `SELECT q.item_id, q.item_type, q.state, q.disposition, q.created_at, d.decided_by AS decided_by
           FROM review_queue q
           LEFT JOIN review_decisions d ON d.item_id = q.item_id AND d.from_state = 'open'
          WHERE q.subject = ? ORDER BY q.created_at DESC LIMIT 50`,
      )
      .bind(passportId)
      .all<Omit<CreatorDetail["reviews"][number], "decided_by"> & { decided_by: string | null }>()
  ).results;
  const reviewLabels = await operatorLabels(db, reviewRows.map((r) => r.decided_by).filter((v): v is string => !!v));
  const reviews = reviewRows.map((r) => ({ ...r, decided_by: r.decided_by ? labelFor(reviewLabels, r.decided_by) : null }));
  const nowMs = Date.now();
  const vs = await verificationStatus(db, passportId, nowMs);
  const standings = await creatorAccountStandings(db, passportId, nowMs);
  const grants = await liveGrantsForPassport(db, passportId);
  // Slice 15: WHAT this subject is, WHO answers for it, and whether that binding
  // is live right now. `characterStanding` evaluates the fold's own gate string
  // rather than a copy of it, so this screen and the claim gate cannot disagree.
  const character = await characterStanding(db, passportId, nowMs);
  const meta = await storedProfileMeta(db, passportId);
  // Read straight from the table that decides it. `isRequestSuppressed` answers
  // the same question with a bare boolean; this screen needs the stamp too, and
  // an operator-only surface is the one place the stamp belongs.
  const delistRow = await db
    .prepare("SELECT operator, created_at FROM request_suppressions WHERE passport_id = ?")
    .bind(passportId)
    .first<{ operator: string; created_at: string }>();
  // Through the SAME resolver `auditList` uses. It is a no-op on the emails the
  // three current writers store (`operatorLabels` shape-filters to `op_` ids and
  // `labelFor` falls back to the raw value), which is exactly the behaviour
  // wanted: today it changes nothing, and the day a writer starts storing the
  // `op_` id this screen resolves it like every other attribution instead of
  // printing a hex string at an operator.
  const delistLabels = delistRow ? await operatorLabels(db, [delistRow.operator]) : null;
  const delisting = delistRow && delistLabels ? { operator: labelFor(delistLabels, delistRow.operator), created_at: delistRow.created_at } : null;
  return {
    passport_id: passportId,
    display_name: p.display_name,
    claim_state: p.claim_state,
    retracted: p.retracted,
    founding_rank: founding?.rank ?? null,
    accounts,
    releases,
    payout_tags: tagRows.map((t) => ({ tag_type: t.tag_type, masked_value: maskTag(t.tag_value), status: t.status, effective_at: t.effective_at })),
    reviews,
    verification: { expires_at: vs?.expires_at ?? null, expired_at: vs?.expired_at ?? null, renewal_pending: vs?.renewal_pending ?? false },
    account_standing: Object.fromEntries(standings.map((a) => [`${a.platform}|${a.id}`, a.standing])),
    agency_grants: grants,
    // `character` is null only when the passport row is gone between the read
    // above and this one — `creatorDetail` already returned null for an unknown
    // id. It cannot happen today; the defaults are stated rather than left to a
    // chain of `??` so that the fail-open direction (a character shown as a
    // person, on the one screen carrying the specific reason) is a deliberate
    // line somebody has to edit rather than an accident of punctuation.
    subject_kind: character?.subject_kind ?? "person",
    // OPERATOR-ONLY, and this is the ONE place it is served. `passport-view.ts`
    // keeps `operated_by` off every public view because publishing it hands
    // anyone the org chart behind every brand mascot in the registry; an
    // authenticated console seat is not anyone.
    operated_by: character?.operated_by ?? null,
    operator_live: character?.accountable ?? true,
    // THE SPECIFIC REASON, which exists here and nowhere else. The public claim
    // refusal stays uniform on purpose (claims.ts `NOT_CLAIMABLE`) — naming the
    // failed predicate to an anonymous caller leaks whether a passport is
    // delisted or its agency suspended. Telling an operator is the whole job.
    operator_reason: character ? STANDING_SENTENCE[character.reason] : STANDING_SENTENCE.person,
    profile_meta: meta,
    delisting,
  };
}

export interface WarningRow {
  item_id: string;
  subject: string;
  state: string; // queued_warning | published | retracted
  warning_class: string | null;
  linkage_basis: string | null;
  wording_id: string | null;
  wording_status: string | null; // status of the chosen wording (null if none / unknown)
  wording_class_match: boolean; // wording's class === warning_class
  second_look: number;
  /** WHO confirmed the second look (slice 9) — null until a DIFFERENT operator
   * does. An ops console already shows operator ids throughout `/ops/audit`, and
   * "who signed this accusation" is the accountability question this screen
   * exists to answer. */
  second_look_by: string | null;
  /** True iff the VIEWER drafted this warning, so the console can grey out a
   * confirm button that is not theirs to press instead of letting them discover
   * it as a 409. A boolean rather than the drafter's id, because that is all the
   * button needs and the server is the enforcement either way. */
  you_drafted: boolean;
  queued_at: string | null;
  cooling_off_elapsed: boolean;
  cooling_off_remaining_ms: number; // 0 once elapsed
  publishable: boolean; // all three gates green (only meaningful in queued_warning)
  disposition: string | null;
  created_at: string;
  updated_at: string;
}

/**
 * The warnings pipeline for the moderation view: every report item that has
 * entered the accusatory funnel — drafted (queued_warning), published, or
 * retracted — with the live gate state each needs to publish. `publishable`
 * folds all three server-side gates (cooling-off elapsed, second-look
 * confirmed, an approved wording of the matching class) so the dashboard can
 * show exactly what's still blocking a publish without re-deriving the rules.
 */
export async function warningsList(db: DbLike, nowMs: number, viewer: string | null = null): Promise<WarningRow[]> {
  const viewerPerson = viewer === null ? null : await personOf(db, viewer);
  const { results } = await db
    .prepare(
      `SELECT q.item_id, q.subject, q.state, q.payload, q.wording_id, q.second_look, q.second_look_by, q.queued_at,
              q.disposition, q.created_at, q.updated_at,
              w.status AS wording_status, w.class AS wording_class, d.decided_by AS drafted_by,
              -- THE PERSON behind each id, resolved the same way publishWarning
              -- and confirmSecondLook resolve it. Comparing the raw op_ ids here
              -- made this mirror disagree with the server the moment two ids were
              -- linked: the console offered an enabled Publish button on an item
              -- the server then refused with a 409.
              COALESCE((SELECT person_id FROM operator_identities WHERE op_id = d.decided_by), d.decided_by) AS drafted_person,
              COALESCE((SELECT person_id FROM operator_identities WHERE op_id = q.second_look_by), q.second_look_by) AS second_look_person,
              -- D-0814(5) (J7): once published, the basis shown is the LEDGER's —
              -- publishWarning drops an inherited basis the operator never set, so
              -- review_queue.payload can carry one the warning never published.
              EXISTS (SELECT 1 FROM warning_publications wp WHERE wp.item_id = q.item_id) AS was_published,
              (SELECT CASE WHEN json_valid(e.payload) = 1 THEN json_extract(e.payload, '$.linkage_basis') END
                 FROM warning_publications wp JOIN trust_events e ON e.event_id = wp.event_id
                WHERE wp.item_id = q.item_id) AS published_basis
         FROM review_queue q
         LEFT JOIN wordings w ON w.wording_id = q.wording_id
         -- At most one row by the primary key (item_id, from_state), so this
         -- cannot fan the result out, and it is the same source publishWarning
         -- re-derives the drafter from
         LEFT JOIN review_decisions d
                ON d.item_id = q.item_id AND d.from_state = 'open' AND d.disposition = 'drafted'
        WHERE q.item_type = 'report' AND q.state IN ('queued_warning','published','retracted','withdrawn')
        -- Actionable rows FIRST. Withdrawal is meant to be routine now, and it
        -- shares this cap with published and retracted history — without the
        -- leading key a busy week silently pushes drafts awaiting a second look
        -- off the only screen that shows them
        ORDER BY (q.state = 'queued_warning') DESC, q.updated_at DESC LIMIT 200`,
    )
    .bind()
    .all<{
      item_id: string; subject: string; state: string; payload: string; wording_id: string | null;
      second_look: number; second_look_by: string | null; queued_at: string | null; disposition: string | null;
      created_at: string; updated_at: string;
      wording_status: string | null; wording_class: string | null; drafted_by: string | null;
      drafted_person: string | null; second_look_person: string | null;
      was_published: number; published_basis: unknown;
    }>();
  return results.map((r) => {
    let warning_class: string | null = null;
    let linkage_basis: string | null = null;
    try {
      const p = JSON.parse(r.payload) as { warning_class?: unknown; linkage_basis?: unknown };
      if (typeof p.warning_class === "string") warning_class = p.warning_class;
      if (typeof p.linkage_basis === "string") linkage_basis = p.linkage_basis;
    } catch { /* leave null */ }
    // Published (or since retracted): only what the ledger event carries; an id
    // still '__pending__' joins no event and shows none.
    if (r.was_published === 1) linkage_basis = typeof r.published_basis === "string" ? r.published_basis : null;
    const queuedMs = r.queued_at ? Date.parse(r.queued_at) : NaN;
    const elapsed = Number.isFinite(queuedMs) ? nowMs - queuedMs >= COOLING_OFF_MS : false;
    const remaining = Number.isFinite(queuedMs) ? Math.max(0, COOLING_OFF_MS - (nowMs - queuedMs)) : COOLING_OFF_MS;
    const wording_class_match = r.wording_class !== null && r.wording_class === warning_class;
    // Mirrors publishWarning's gates EXACTLY, two-person rule included — and
    // "exactly" now means PEOPLE, not `op_` id strings, because that is what the
    // server compares. A `publishable: true` the server would then refuse is
    // worse than useless on a console whose whole job is showing an operator
    // what is still blocking.
    const twoPerson = r.second_look_person !== null && r.drafted_person !== null && r.second_look_person !== r.drafted_person;
    const publishable =
      r.state === "queued_warning" &&
      elapsed &&
      r.second_look === 1 &&
      twoPerson &&
      r.wording_status === "approved" &&
      wording_class_match;
    return {
      item_id: r.item_id,
      subject: r.subject,
      state: r.state,
      warning_class,
      linkage_basis,
      wording_id: r.wording_id,
      wording_status: r.wording_status,
      wording_class_match,
      second_look: r.second_look,
      second_look_by: r.second_look_by,
      // Also by person: a viewer whose OTHER address drafted this must not be
      // offered an enabled Confirm button the server will refuse.
      you_drafted: viewer !== null && r.drafted_person !== null && r.drafted_person === (viewerPerson ?? viewer),
      queued_at: r.queued_at,
      cooling_off_elapsed: elapsed,
      cooling_off_remaining_ms: remaining,
      publishable,
      disposition: r.disposition,
      created_at: r.created_at,
      updated_at: r.updated_at,
    };
  });
}

export interface ImportBatchRow {
  batch_id: string;
  kind: string;
  status: string; // applied | retracted
  accepted: number;
  rejected: number;
  skipped: number;
  created_at: string;
  /** I3-1421 (I4-05-R7 P2): 'applying' with no heartbeat for IMPORT_STALE_MS — crashed, not slow. */
  stuck?: boolean;
  /** I3-1207 residual (JN-NEW-1): a chunked retraction began and has not completed
   * (importer.ts `incompleteRetractions`) — the console's "continue" is drawn from this. */
  retraction_incomplete: boolean;
  /** QUEUE #775 (D-0858(11)): an apply stopped at IMPORT_APPLY_PER_CALL and is not finished
   * (importer.ts `incompleteApplies`) — the console's "continue (re-select the file)". */
  apply_incomplete: boolean;
}

/** Applied/retracted seed batches, newest first. Dry-runs write nothing, so
 * they never appear here. */
export async function importBatchList(db: DbLike, opts: { limit?: number } = {}): Promise<ImportBatchRow[]> {
  const limit = Math.max(1, Math.min(200, opts.limit ?? 50));
  const { results } = await db
    .prepare("SELECT batch_id, kind, status, accepted, rejected, skipped, created_at FROM import_batches ORDER BY created_at DESC, batch_id DESC LIMIT ?")
    .bind(limit)
    .all<Omit<ImportBatchRow, "retraction_incomplete" | "apply_incomplete">>();
  const stale = new Set((await staleImportBatches(db)).map((b) => b.batch_id));
  const halfRun = await incompleteRetractions(db, results.map((r) => r.batch_id));
  const halfApplied = await incompleteApplies(db, results.map((r) => r.batch_id));
  return results.map((r) => ({
    ...(r.status === "applying" ? { ...r, stuck: stale.has(r.batch_id) } : r),
    retraction_incomplete: halfRun.has(r.batch_id),
    apply_incomplete: halfApplied.has(r.batch_id),
  }));
}

export interface ImportBatchDetail {
  batch: ImportBatchRow;
  profiles: { passport_id: string; display_name: string | null; claim_state: string | null; retracted: number | null }[];
  determinations: { item_id: string; subject: string | null; state: string | null; disposition: string | null }[];
  /** I3-1708: how many of each the batch really produced, and whether the lists
   * above stop short of it (each is capped at IMPORT_DETAIL_CAP rows). */
  profiles_total: number;
  determinations_total: number;
  truncated: { profiles: boolean; determinations: boolean };
}

/** Rows per kind `importBatchDetail` lists; past it the detail says `truncated`. */
export const IMPORT_DETAIL_CAP = 500;

/**
 * One batch with the LIVE state of everything it produced — resolved against the
 * ledger-derived tables, not the import bookkeeping, so it reflects reality now:
 * a profile a creator has since claimed shows as claimed, a determination an
 * operator published shows as published. This is what makes "should I retract
 * this batch?" an informed decision. Null if the batch id is unknown.
 */
export async function importBatchDetail(db: DbLike, batchId: string): Promise<ImportBatchDetail | null> {
  const row = await db
    .prepare("SELECT batch_id, kind, status, accepted, rejected, skipped, created_at FROM import_batches WHERE batch_id = ?")
    .bind(batchId)
    .first<Omit<ImportBatchRow, "retraction_incomplete" | "apply_incomplete">>();
  if (!row) return null;
  // I3-1421 (I34-FX2-CLIENTS): the detail carries `stuck` exactly as importBatchList does, so the console's batch pane
  // and its list read one rule (importer.ts staleImportBatches).
  const stuck = row.status === "applying" ? { stuck: (await staleImportBatches(db)).some((s) => s.batch_id === batchId) } : {};
  const batch: ImportBatchRow = {
    ...row,
    ...stuck,
    retraction_incomplete: (await incompleteRetractions(db, [batchId])).has(batchId),
    apply_incomplete: (await incompleteApplies(db, [batchId])).has(batchId),
  };
  const profiles = (
    await db
      .prepare(
        `SELECT r.passport_id AS passport_id, p.display_name AS display_name, p.claim_state AS claim_state, p.retracted AS retracted
           FROM import_records r LEFT JOIN passports p ON p.passport_id = r.passport_id
          WHERE r.batch_id = ? AND r.kind = 'profile' AND r.passport_id IS NOT NULL
          ORDER BY r.id LIMIT ?`,
      )
      .bind(batchId, IMPORT_DETAIL_CAP)
      .all<ImportBatchDetail["profiles"][number]>()
  ).results;
  const determinations = (
    await db
      .prepare(
        `SELECT r.item_id AS item_id, q.subject AS subject, q.state AS state, q.disposition AS disposition
           FROM import_records r LEFT JOIN review_queue q ON q.item_id = r.item_id
          WHERE r.batch_id = ? AND r.kind = 'determination' AND r.item_id IS NOT NULL
          ORDER BY r.id LIMIT ?`,
      )
      .bind(batchId, IMPORT_DETAIL_CAP)
      .all<ImportBatchDetail["determinations"][number]>()
  ).results;
  // I3-1708: "should I retract this batch?" must not be decided on a silent
  // sample — the same predicates as the lists above, counted.
  const totals = await db
    .prepare(
      `SELECT COALESCE(SUM(kind = 'profile' AND passport_id IS NOT NULL), 0) AS p,
              COALESCE(SUM(kind = 'determination' AND item_id IS NOT NULL), 0) AS d
         FROM import_records WHERE batch_id = ?`,
    )
    .bind(batchId)
    .first<{ p: number; d: number }>();
  const profiles_total = Number(totals?.p ?? 0);
  const determinations_total = Number(totals?.d ?? 0);
  return {
    batch,
    profiles,
    determinations,
    profiles_total,
    determinations_total,
    truncated: { profiles: profiles_total > profiles.length, determinations: determinations_total > determinations.length },
  };
}

// --- Ledger integrity (the "prove the chain is intact" view) ---

export interface AnchorRow { seq: number; subject: string; created_at: string }

export interface LedgerIntegrity {
  ok: boolean; // chain hashes + linkage recompute cleanly
  events: number;
  brokenAtSeq?: number;
  reason?: string;
  anchored: boolean; // an external checkpoint anchor was compared
  anchorSeq?: number;
  anchorReason?: string;
  head: { seq: number; event_hash: string } | null;
  recent_anchors: AnchorRow[]; // edition-set ledger anchors (tamper-evidence artifacts)
  /**
   * Item 12 — THE SIGNATURE AXIS, carried through to the operator.
   *
   * `ok` keeps its old meaning for existing consumers (the chain, and its
   * anchor, are sound). `state` is the field a UI switches on, because the
   * strongest claim now needs a THIRD thing beyond a clean chain and a matching
   * anchor: a signature made by a key this service never holds.
   *
   *  verified   chain + anchor + a signature that verifies under a published key
   *  unverified true and narrower — nothing could be checked on the signature
   *             axis (signing off, signer was down, key retired, no anchor yet)
   *  broken     the alarm, including a checkpoint signed under a published key
   *             whose signature does not verify
   *
   * These are ALWAYS present. A response that omits them would be read by a
   * fail-closed console as "unverified", which is safe but needlessly narrow.
   */
  state: IntegrityState;
  /**
   * WHICH AXIS is broken, present exactly when `ok` is false — the field that
   * tells an operator whether the scheduled sweep is still running. `state` is
   * `broken` for all three; only `chain` and `ledger` stop the world. See
   * checkpoint.ts BreachAxis and DEPLOY.md §3e for the recovery runbook.
   */
  breach?: BreachAxis;
  /** #131E-11b (I16-W07A H-6): the object key the verdict condemns the log against (checkpoint.ts
   * `breachObject`), so the operator can name the object the DEPLOY.md runbook asks for first. */
  breachObject?: string;
  signature: SignatureState;
  signatureReason?: string;
  signatureKeyId?: string;
  /**
   * THE OTHER MUTABLE-PROJECTION AXIS. The chain proves nothing was rewritten in
   * `trust_events`; a published warning's SENTENCE lives in the mutable
   * `wordings` table, and the ledger commits to its bytes (`wording_hash`). So a
   * green chain and a rewritten accusation are compatible states, and this is
   * where the second one becomes visible. `rows` empty is the healthy case; the
   * scheduled sweep raises the same finding as a `WORDING DRIFT` alarm so nobody
   * has to come here to learn it. See warnings.ts `wordingDrift`.
   */
  wording_drift: WordingDriftReport;
  /**
   * I3-0275 — THE OPERATOR AUDIT LOG IS A SEPARATE AXIS. Everything above covers
   * `trust_events` only; `audit_log` is an ordinary table, sealed after the fact
   * by `sweepAuditLogSeal` into `audit_log_seal`. This says how far the seal
   * reaches and whether it still recomputes — it never feeds `ok`/`state`.
   */
  audit_log: AuditLogSealCoverage;
}

/**
 * Live ledger-integrity check for the operator. With a store binding it runs the
 * STRONGEST check (verifyAgainstCheckpoint: recompute the whole chain, confirm
 * the highest external anchor still matches — catching a full
 * rewrite-from-genesis that internal verification alone cannot — and, when a
 * published key set is passed, check that the anchor was SIGNED by a key this
 * service never holds). Without a store it falls back to the internal chain
 * recompute, which is a narrower claim and reports itself as one. Also returns
 * the current head and the recent edition-set anchors so the operator can
 * eyeball tamper-evidence coverage.
 *
 * `keys` comes from `checkpointKeysFromEnv` at the route. Absent/empty is the
 * feature being OFF: signature "disabled", state "unverified" — never a pass.
 */
export async function ledgerIntegrity(
  db: DbLike,
  store?: ObjectStore,
  keys?: CheckpointPublicKey[] | null,
  /** I3-1101 (I23-W1-TRUST hand-off 5): this deployment's id (`checkpointDeploymentIdFromEnv`), so the console's read
   * sets a foreign deployment's v2 anchor aside exactly as the cron does. The console caller (hub) passes it. */
  opts: { deploymentId?: string | null } = {},
): Promise<LedgerIntegrity> {
  let verdict: AnchoredVerification;
  if (store) {
    // QUEUE #124: verify at the CRON'S persisted tail cursor, read-only. The
    // cron (index.ts `checkpoint_anchors`) throws on a chain/ledger breach
    // BEFORE it persists `nextTailCursor`, so a halted cron leaves the cursor on
    // the window that held the contradiction; reading that same window here
    // reports the same breach. With the default cursor the console re-read the
    // newest tail window forever and said "verified" while the cron had halted.
    // Never written from here: a GET that moved the cursor would race the cron.
    const tailCursor = num((await db.prepare("SELECT cursor_seq FROM sweep_cursor WHERE name = 'checkpoint_anchors'").bind().first<{ cursor_seq: number }>())?.cursor_seq);
    verdict = await verifyAgainstCheckpoint(db, store, keys, { tailCursor, deploymentId: opts.deploymentId ?? null });
  } else {
    // NO STORE BINDING: nothing external was read, so nothing on the signature
    // axis could be checked either — and a check that did not run is never a
    // pass. This is the narrow-but-true state, not `verified`.
    //
    // "disabled" rather than "absent" because with no object store there is no
    // anchor here to have been signed in the first place; the anchor half of
    // the operator's banner already says no external checkpoint was compared.
    const chain = await verifyChain(db);
    verdict = {
      ...chain,
      anchored: false,
      state: chain.ok ? "unverified" : "broken",
      ...(chain.ok ? {} : { breach: "chain" as const }),
      signature: "disabled",
      signatureReason: "no object store binding on this deployment — no external anchor was read, so no signature could be checked",
    };
  }
  const head = await chainHead(db);
  // I3-2168 a: a read — the operator's look never advances the scheduled sweep's cursor (warnings.ts wordingDrift).
  const drift = await wordingDrift(db, Date.now(), { advance: false });
  const auditSeal = await auditLogSealCoverage(db);
  const { results: anchors } = await db
    .prepare("SELECT seq, subject, created_at FROM trust_events WHERE event_type = 'edition_set_anchored' ORDER BY seq DESC LIMIT 10")
    .bind()
    .all<AnchorRow>();
  return {
    ok: verdict.ok,
    events: verdict.events,
    ...(verdict.brokenAtSeq !== undefined ? { brokenAtSeq: verdict.brokenAtSeq } : {}),
    ...(verdict.reason !== undefined ? { reason: verdict.reason } : {}),
    anchored: verdict.anchored,
    ...(verdict.anchorSeq !== undefined ? { anchorSeq: verdict.anchorSeq } : {}),
    ...(verdict.anchorReason !== undefined ? { anchorReason: verdict.anchorReason } : {}),
    head: head ? { seq: head.seq, event_hash: head.event_hash } : null,
    recent_anchors: anchors,
    state: verdict.state,
    ...(verdict.breach !== undefined ? { breach: verdict.breach } : {}),
    ...(verdict.breachObject !== undefined ? { breachObject: verdict.breachObject } : {}),
    signature: verdict.signature,
    ...(verdict.signatureReason !== undefined ? { signatureReason: verdict.signatureReason } : {}),
    ...(verdict.signatureKeyId !== undefined ? { signatureKeyId: verdict.signatureKeyId } : {}),
    wording_drift: drift,
    audit_log: auditSeal,
  };
}

// --- I3-0275: the operator audit log seal (D-0920 option A) ---

/** A row is sealed once it is this old, so a batch still being written is never cut. */
export const AUDIT_SEAL_MIN_AGE_MS = 10 * 60 * 1000;
/** Rows per seal, and seals per sweep tick — the tick's statement budget. */
export const AUDIT_SEAL_MAX_ROWS = 500;
export const AUDIT_SEAL_MAX_SEALS_PER_TICK = 4;
/** Seals whose rows `auditLogSealCoverage` recomputes (newest first); older seals are linkage-checked only. */
export const AUDIT_SEAL_VERIFY_SEALS = 50;

export const AUDIT_LOG_NOT_CHAINED = "covers the trust ledger only; the operator audit log is not chained";

export interface AuditLogSealCoverage {
  /** At least one seal exists. */
  chained: boolean;
  /** Every checked seal recomputes and links; true when there is nothing to check. */
  ok: boolean;
  seals: number;
  /** Highest audit_id under a seal, or null. */
  sealed_through: number | null;
  /** audit_log rows above `sealed_through` — not yet tamper-evident. */
  unsealed_rows: number;
  /** Lowest audit_id whose row bytes were recomputed this read (null: none). */
  verified_from: number | null;
  broken_seal?: number;
  reason?: string;
  /** The sentence an operator reads. */
  note: string;
}

interface AuditRowForSeal { audit_id: number; actor_id: string; action: string; created_at: string; actor_kind: string }
interface SealRow { seal_id: number; from_audit_id: number; to_audit_id: number; row_count: number; prev_hash: string; seal_hash: string }

/** The sealed bytes: id, actor, action, time, actor kind — never `detail` or `item_id`, which erasure rewrites. */
async function sealHash(prev: string, from: number, to: number, rows: readonly AuditRowForSeal[]): Promise<string> {
  return sha256Hex([prev, String(from), String(to), ...rows.map((r) => JSON.stringify([r.audit_id, r.actor_id, r.action, r.created_at, r.actor_kind]))].join("\n"));
}

/**
 * I3-0275 — chain closed audit_log ranges into `audit_log_seal`. A range is the
 * rows after the last seal, in id order, up to the first row younger than
 * `AUDIT_SEAL_MIN_AGE_MS` (so a range never skips a row still being written).
 * The INSERT's primary key (`seal_id = last + 1`) is the claim: a racing tick
 * collides and writes nothing. Returns the number of seals written.
 */
export async function sweepAuditLogSeal(db: DbLike, nowMs: number): Promise<number> {
  const cutoff = new Date(nowMs - AUDIT_SEAL_MIN_AGE_MS).toISOString();
  let written = 0;
  for (let i = 0; i < AUDIT_SEAL_MAX_SEALS_PER_TICK; i++) {
    const last = await db.prepare("SELECT seal_id, to_audit_id, seal_hash FROM audit_log_seal ORDER BY seal_id DESC LIMIT 1").bind().first<{ seal_id: number; to_audit_id: number; seal_hash: string }>();
    const { results } = await db
      .prepare("SELECT audit_id, actor_id, action, created_at, actor_kind FROM audit_log WHERE audit_id > ? ORDER BY audit_id LIMIT ?")
      .bind(last?.to_audit_id ?? 0, AUDIT_SEAL_MAX_ROWS)
      .all<AuditRowForSeal>();
    const young = results.findIndex((r) => !(r.created_at <= cutoff));
    const rows = young === -1 ? results : results.slice(0, young);
    if (rows.length === 0) break;
    const from = rows[0]!.audit_id;
    const to = rows[rows.length - 1]!.audit_id;
    const prev = last?.seal_hash ?? "";
    const hash = await sealHash(prev, from, to, rows);
    try {
      await db
        .prepare("INSERT INTO audit_log_seal (seal_id, from_audit_id, to_audit_id, row_count, prev_hash, seal_hash, sealed_at) VALUES (?, ?, ?, ?, ?, ?, ?)")
        .bind((last?.seal_id ?? 0) + 1, from, to, rows.length, prev, hash, new Date(nowMs).toISOString())
        .run();
    } catch (e) {
      if (/UNIQUE constraint failed|PRIMARY KEY/i.test(e instanceof Error ? e.message : String(e))) break; // lost the claim to a racing tick
      throw e;
    }
    written++;
    if (young !== -1) break;
  }
  return written;
}

/**
 * I3-0275 — how far the audit-log seal reaches, and whether it still holds. Every
 * seal is linkage-checked (prev_hash = the previous seal_hash, ranges ascending
 * and disjoint); the newest `AUDIT_SEAL_VERIFY_SEALS` are recomputed from the
 * live rows, so a changed actor/action/time/kind, or a row removed or inserted
 * inside a sealed range, reads as broken.
 */
export async function auditLogSealCoverage(db: DbLike): Promise<AuditLogSealCoverage> {
  let seals: SealRow[];
  try {
    seals = (await db.prepare("SELECT seal_id, from_audit_id, to_audit_id, row_count, prev_hash, seal_hash FROM audit_log_seal ORDER BY seal_id").bind().all<SealRow>()).results;
  } catch (e) {
    // Migration 0167 not applied: nothing is sealed, which is exactly what the note says. Any other fault throws.
    if (!/no such table/i.test(e instanceof Error ? e.message : String(e))) throw e;
    seals = [];
  }
  const sealedThrough = seals.length ? seals[seals.length - 1]!.to_audit_id : null;
  const unsealed = (await db.prepare("SELECT COUNT(*) AS n FROM audit_log WHERE audit_id > ?").bind(sealedThrough ?? 0).first<{ n: number }>())?.n ?? 0;
  const base = { chained: seals.length > 0, seals: seals.length, sealed_through: sealedThrough, unsealed_rows: unsealed };
  const broken = (seal: number, reason: string): AuditLogSealCoverage => ({
    ...base, ok: false, verified_from: null, broken_seal: seal, reason,
    note: `BROKEN at seal ${seal}: ${reason}. The operator audit log has been changed since it was sealed.`,
  });
  let prevHash = "";
  let prevTo = 0;
  for (let i = 0; i < seals.length; i++) {
    const s = seals[i]!;
    if (s.seal_id !== i + 1) return broken(s.seal_id, "seal ids are not contiguous");
    if (s.prev_hash !== prevHash) return broken(s.seal_id, "prev_hash does not match the previous seal");
    if (!(s.from_audit_id > prevTo && s.to_audit_id >= s.from_audit_id)) return broken(s.seal_id, "sealed ranges overlap or run backwards");
    prevHash = s.seal_hash;
    prevTo = s.to_audit_id;
  }
  const pruned = await prunedSealRanges(db);
  const checkFrom = Math.max(0, seals.length - AUDIT_SEAL_VERIFY_SEALS);
  for (const s of seals.slice(checkFrom)) {
    const { results: rows } = await db
      .prepare("SELECT audit_id, actor_id, action, created_at, actor_kind FROM audit_log WHERE audit_id BETWEEN ? AND ? ORDER BY audit_id")
      .bind(s.from_audit_id, s.to_audit_id)
      .all<AuditRowForSeal>();
    // I4-Q9: a range pruneSealedAuditRanges recorded (0239) must be EMPTY, and its record must match the seal.
    const p = pruned.get(s.seal_id);
    if (p) {
      if (p.from_audit_id !== s.from_audit_id || p.to_audit_id !== s.to_audit_id || p.row_count !== s.row_count) return broken(s.seal_id, "the pruned-range record does not match its seal");
      if (rows.length !== 0) return broken(s.seal_id, `the range was pruned and ${rows.length} rows are there now`);
      continue;
    }
    if (rows.length !== s.row_count) return broken(s.seal_id, `${s.row_count} rows were sealed and ${rows.length} are there now`);
    if ((await sealHash(s.prev_hash, s.from_audit_id, s.to_audit_id, rows)) !== s.seal_hash) return broken(s.seal_id, "a sealed row's actor, action, time or kind changed");
  }
  const verifiedFrom = seals.length ? seals[checkFrom]!.from_audit_id : null;
  const prunedNote = pruned.size ? ` ${pruned.size} sealed range${pruned.size === 1 ? " was" : "s were"} pruned after the money-record period; ${pruned.size === 1 ? "its seal still links" : "their seals still link"}.` : "";
  const note = seals.length === 0
    ? `This check ${AUDIT_LOG_NOT_CHAINED}: no audit_log row has been sealed yet.`
    : `The trust ledger is checked above. The operator audit log is sealed through #${sealedThrough} (${seals.length} seal${seals.length === 1 ? "" : "s"}, rows from #${verifiedFrom} recomputed now); ${unsealed} newer row${unsealed === 1 ? " is" : "s are"} not yet sealed. Only actor, action, time and kind are sealed — the detail text is not.${prunedNote}`;
  return { ...base, ok: true, verified_from: verifiedFrom, note };
}

interface PrunedRangeRow { seal_id: number; from_audit_id: number; to_audit_id: number; row_count: number }
/** I4-Q9: the ranges pruneSealedAuditRanges recorded (0239), by seal id. Migration not applied: none. */
async function prunedSealRanges(db: DbLike): Promise<Map<number, PrunedRangeRow>> {
  try {
    const { results } = await db.prepare("SELECT seal_id, from_audit_id, to_audit_id, row_count FROM audit_log_pruned_ranges").bind().all<PrunedRangeRow>();
    return new Map(results.map((r) => [r.seal_id, r]));
  } catch (e) {
    if (!/no such table/i.test(e instanceof Error ? e.message : String(e))) throw e;
    return new Map();
  }
}

/** I4-Q9: sealed ranges pruned per call (one batch each); `more: true` = call again. */
export const AUDIT_PRUNE_MAX_SEALS_PER_TICK = 4;

/**
 * D-1037(8) (I29-RG-ERASE RECORDS-07): a sealed range holds a held subject's trail when any of its audit rows names, by
 * `item_id`, a subject an open legal hold names — every LEGAL_HOLD_SUBJECT_KINDS kind (a report item, a passport, an
 * account) — or names a report item, or the review item a report item was escalated into (`review_item_id`), that
 * itemPurgeHeldSql holds (the item, an uploader account of its evidence, the passport it names; D-1046(2)) — reached
 * through the review item's own `payload.fan_report_item` too, as TRAIL_HELD_SQL does, because reject and withdraw NULL
 * `review_item_id` (I34-FX3-HOLDS) — or names a
 * review item whose subject is a held passport or account (verification.ts:235, claims, recovery.ts:564 creator cases;
 * recovery.ts:1820 fan cases; D-1046(1)).
 * Over the SQL expressions `fromExpr`/`toExpr` (the range's bounds); the range is skipped while it holds.
 */
const auditRangeHeldSql = (fromExpr: string, toExpr: string): string =>
  `EXISTS (SELECT 1 FROM audit_log a WHERE a.audit_id BETWEEN ${fromExpr} AND ${toExpr} AND (${LEGAL_HOLD_SUBJECT_KINDS.map((k) => legalHoldOpenSql(k, "a.item_id")).join(" OR ")} OR ${itemPurgeHeldSql("a.item_id")} OR EXISTS (SELECT 1 FROM fan_report_items f WHERE f.review_item_id = a.item_id AND ${itemPurgeHeldSql("f.item_id")}) OR ${itemPurgeHeldSql("(SELECT CASE WHEN json_valid(q.payload) = 1 THEN json_extract(q.payload, '$.fan_report_item') END FROM review_queue q WHERE q.item_id = a.item_id)")} OR EXISTS (SELECT 1 FROM review_queue q WHERE q.item_id = a.item_id AND (${legalHoldOpenSql("passport", "q.subject")} OR ${legalHoldOpenSql("account", "q.subject")}))))`;

/**
 * I4-Q9 (I3-0796, I3-1403; D-1030 "Keep audit rows as long as the money records they explain, with erased people's ids
 * pseudonymised" — the pseudonymising is account_data.ts's erasure; census pick A). Deletes audit_log rows only inside
 * ranges audit_log_seal already covers, oldest seal first, a whole range at a time, and only when EVERY row of the range
 * is past the money-record period (money-record-period.ts moneyRecordRetentionYears, default 7), counted as the money
 * records count it — from the END of the tax year of the range's newest row (R2-I5-Q5, moneyRecordCutoffFromYears;
 * D-1112(7)) — AND the range
 * still recomputes to its seal — a tampered range is evidence and is never pruned. Each range is one batch: the
 * `audit_log_pruned_ranges` record first (0239), then the DELETE of exactly the sealed range. Stops at
 * the first range that is too young, broken or unsealed, so an unsealed row is never reached. scheduled(): I29-INT.
 */
export async function pruneSealedAuditRanges(db: DbLike, nowMs: number = Date.now()): Promise<{ pruned: number; rows: number; more: boolean }> {
  const cutoff = moneyRecordCutoffFromYears(nowMs, await moneyRecordRetentionYears(db));
  const { results: seals } = await db
    .prepare(`SELECT seal_id, from_audit_id, to_audit_id, row_count, prev_hash, seal_hash FROM audit_log_seal WHERE seal_id NOT IN (SELECT seal_id FROM audit_log_pruned_ranges) AND NOT ${auditRangeHeldSql("audit_log_seal.from_audit_id", "audit_log_seal.to_audit_id")} AND NOT ${auditRangeOwedSql("audit_log_seal.from_audit_id", "audit_log_seal.to_audit_id")} ORDER BY seal_id LIMIT ?`) // I65-HOLD F2: nor one holding an OWED row its arm would still take
    .bind(AUDIT_PRUNE_MAX_SEALS_PER_TICK)
    .all<SealRow>();
  let pruned = 0;
  let rowsGone = 0;
  for (const s of seals) {
    const { results: rows } = await db
      .prepare("SELECT audit_id, actor_id, action, created_at, actor_kind FROM audit_log WHERE audit_id BETWEEN ? AND ? ORDER BY audit_id")
      .bind(s.from_audit_id, s.to_audit_id)
      .all<AuditRowForSeal>();
    if ((await sealHash(s.prev_hash, s.from_audit_id, s.to_audit_id, rows)) !== s.seal_hash) break; // the hash covers every row, so a missing one fails it too
    const newest = rows.reduce((m, r) => (r.created_at > m ? r.created_at : m), "");
    if (!(newest < cutoff)) break;
    // D-1037(8): the record is written only while no hold names a row of the range (a hold opened after the read
    // above wins), and the DELETE is bound to THAT record, so a refused range is neither recorded nor deleted.
    let recorded = 0;
    try {
      const res = (await db.batch([
      db
        .prepare(`INSERT INTO audit_log_pruned_ranges (seal_id, from_audit_id, to_audit_id, row_count, newest_created_at, pruned_at) SELECT ?, ?, ?, ?, ?, ? WHERE NOT ${auditRangeHeldSql("?", "?")} AND NOT ${auditRangeOwedSql("?", "?")} RETURNING seal_id`)
        .bind(s.seal_id, s.from_audit_id, s.to_audit_id, s.row_count, newest, new Date(nowMs).toISOString(), s.from_audit_id, s.to_audit_id, s.from_audit_id, s.to_audit_id),
      db
        .prepare("DELETE FROM audit_log WHERE audit_id BETWEEN ? AND ? AND EXISTS (SELECT 1 FROM audit_log_pruned_ranges WHERE seal_id = ? AND pruned_at = ?)")
        .bind(s.from_audit_id, s.to_audit_id, s.seal_id, new Date(nowMs).toISOString()),
      ])) as { results?: unknown[] }[];
      recorded = res[0]?.results?.length ?? 0;
    } catch (e) {
      if (/UNIQUE constraint failed|PRIMARY KEY/i.test(e instanceof Error ? e.message : String(e))) break; // a racing tick recorded it first
      throw e;
    }
    if (recorded !== 1) break; // a hold landed on the range after the read: nothing recorded, nothing deleted
    pruned += 1;
    rowsGone += rows.length;
  }
  return { pruned, rows: rowsGone, more: seals.length === AUDIT_PRUNE_MAX_SEALS_PER_TICK && pruned === seals.length };
}

// --- I21-Q103: the accusation decision trail becomes hash-only ---

/** I21-Q103 (I3-2071; D-1030 "(a) while the warning stands plus 3 years, then hash-only; subject export on request"). */
export const DECISION_TRAIL_RETENTION_YEARS = 3;
/** Items hashed per sweep call (one batch each); `more: true` = call again. */
export const DECISION_TRAIL_SWEEP_BATCH = 50;
/** A `report` item whose warning no longer stands: never published (rejected), withdrawn, or retracted. */
const TRAIL_STOPPED_SQL = "item_type = 'report' AND state IN ('rejected','withdrawn','retracted') AND trail_hashed_at IS NULL";
/** The fan report item this review item was escalated from, by the review item's own record of it (reports.ts escalateReport
 * writes `payload.fan_report_item` once); NULL for a malformed payload or an item no fan report was escalated into. */
const TRAIL_REPORT_OF_ITEM_SQL = "(CASE WHEN json_valid(review_queue.payload) = 1 THEN json_extract(review_queue.payload, '$.fan_report_item') END)";
/** D-1037(8) (I29-RG-ERASE RECORDS-07; D-1044(2) I29-LINTFIX): a report item escalated into this review item is held — an open
 * legal hold on the item, its evidence uploader's account or the passport it names, asked through the purge doors' own
 * predicate (reports.ts itemPurgeHeldSql -> evidence.ts evidencePurgeHeldSql); its trail is not hashed while it holds.
 * I34-FX3-HOLDS (RG3R-TRAIL-HOLD-BACKLINK): the report is reached through `fan_report_items.review_item_id` OR through
 * TRAIL_REPORT_OF_ITEM_SQL, because the S65 back-link (review.ts rejectItem, warnings.ts withdrawWarning) sets
 * review_item_id to NULL when an item is rejected or withdrawn, two of the three states TRAIL_STOPPED_SQL selects. */
const TRAIL_HELD_SQL = `(EXISTS (SELECT 1 FROM fan_report_items f WHERE f.review_item_id = review_queue.item_id AND ${itemPurgeHeldSql("f.item_id")}) OR ${itemPurgeHeldSql(TRAIL_REPORT_OF_ITEM_SQL)})`;
/** When it stopped standing: its ending decision (review.ts reviewDecisionStmt), or `updated_at` for a legacy row. */
const TRAIL_ENDED_AT_SQL =
  "COALESCE((SELECT MAX(d.decided_at) FROM review_decisions d WHERE d.item_id = review_queue.item_id AND d.disposition IN ('rejected','withdrawn','retracted')), review_queue.updated_at)";

/**
 * I21-Q103 — DECISION_TRAIL_RETENTION_YEARS after a warning stops standing, replace that item's audit_log `detail` and
 * `item_id` and its review_decisions `decided_by` with their sha256Hex, and stamp `review_queue.trail_hashed_at` (0238).
 * Lives here beside the audit seal it must not break (the one other non-subject sha256Hex site). One batch per item; each write binds the value it READ (a row changed since is left alone) and re-asks that the item
 * is still stopped and unhashed; the stamp is last. audit_log_seal covers neither column (admin.ts sealHash), so every
 * seal still recomputes. Disposition, states and dates stay readable. scheduled() step: I29-INT.
 */
export async function sweepDecisionTrails(db: DbLike, nowMs: number = Date.now()): Promise<{ items: number; audit_rows: number; decisions: number; more: boolean }> {
  const cut = new Date(nowMs);
  cut.setUTCFullYear(cut.getUTCFullYear() - DECISION_TRAIL_RETENTION_YEARS);
  const cutoff = cut.toISOString();
  const ts = new Date(nowMs).toISOString();
  const { results } = await db
    .prepare(`SELECT item_id FROM review_queue WHERE ${TRAIL_STOPPED_SQL} AND NOT ${TRAIL_HELD_SQL} AND ${TRAIL_ENDED_AT_SQL} <= ? ORDER BY item_id LIMIT ?`)
    .bind(cutoff, DECISION_TRAIL_SWEEP_BATCH)
    .all<{ item_id: string }>();
  let items = 0;
  let auditRows = 0;
  let decisions = 0;
  for (const { item_id } of results) {
    const still = `EXISTS (SELECT 1 FROM review_queue WHERE item_id = ? AND ${TRAIL_STOPPED_SQL} AND NOT ${TRAIL_HELD_SQL})`;
    const audits = (await db.prepare("SELECT audit_id, detail FROM audit_log WHERE item_id = ?").bind(item_id).all<{ audit_id: number; detail: string }>()).results;
    const decs = (await db.prepare("SELECT from_state, decided_by FROM review_decisions WHERE item_id = ?").bind(item_id).all<{ from_state: string; decided_by: string }>()).results;
    const hashedItem = await sha256Hex(item_id);
    const stmts: DbStatement[] = [];
    for (const a of audits)
      stmts.push(db.prepare(`UPDATE audit_log SET detail = ?, item_id = ? WHERE audit_id = ? AND item_id = ? AND detail = ? AND ${still}`).bind(await sha256Hex(a.detail), hashedItem, a.audit_id, item_id, a.detail, item_id));
    for (const d of decs)
      stmts.push(db.prepare(`UPDATE review_decisions SET decided_by = ? WHERE item_id = ? AND from_state = ? AND decided_by = ? AND ${still}`).bind(await sha256Hex(d.decided_by), item_id, d.from_state, d.decided_by, item_id));
    stmts.push(db.prepare(`UPDATE review_queue SET trail_hashed_at = ? WHERE item_id = ? AND ${TRAIL_STOPPED_SQL} AND NOT ${TRAIL_HELD_SQL} RETURNING item_id`).bind(ts, item_id));
    const res = (await db.batch(stmts)) as { results?: unknown[] }[];
    if ((res[res.length - 1]?.results?.length ?? 0) === 1) {
      items += 1;
      auditRows += audits.length;
      decisions += decs.length;
    }
  }
  return { items, audit_rows: auditRows, decisions, more: results.length === DECISION_TRAIL_SWEEP_BATCH };
}

export interface DecisionTrail {
  subject: string;
  items: {
    item_id: string;
    state: string;
    disposition: string | null;
    created_at: string;
    trail_hashed_at: string | null;
    decisions: { from_state: string; disposition: string; decided_by: string; decided_at: string }[];
    audit: { audit_id: number; actor_id: string; action: string; detail: string; created_at: string }[];
  }[];
}

/**
 * I21-Q103 — one subject's decision trail, for the subject export on request (route: I29-OPS, full seat, audited).
 * `subject` is the review item's `subject` exactly as stored (a `report` item's is the reported account's subject
 * hash, reports.ts escalateReport). Only `report` items of that subject; audit rows are matched by the live item id
 * and, after sweepDecisionTrails, by its sha256Hex — so a hashed trail still exports, as hashes.
 */
export async function exportDecisionTrail(db: DbLike, subject: string): Promise<DecisionTrail> {
  const { results: rows } = await db
    .prepare("SELECT item_id, state, disposition, created_at, trail_hashed_at FROM review_queue WHERE item_type = 'report' AND subject = ? ORDER BY created_at, item_id")
    .bind(subject)
    .all<{ item_id: string; state: string; disposition: string | null; created_at: string; trail_hashed_at: string | null }>();
  const items: DecisionTrail["items"] = [];
  for (const r of rows) {
    const decisions = (
      await db
        .prepare("SELECT from_state, disposition, decided_by, decided_at FROM review_decisions WHERE item_id = ? ORDER BY decided_at, from_state")
        .bind(r.item_id)
        .all<DecisionTrail["items"][number]["decisions"][number]>()
    ).results;
    const audit = (
      await db
        .prepare("SELECT audit_id, actor_id, action, detail, created_at FROM audit_log WHERE item_id IN (?, ?) ORDER BY audit_id")
        .bind(r.item_id, await sha256Hex(r.item_id))
        .all<DecisionTrail["items"][number]["audit"][number]>()
    ).results;
    items.push({ ...r, decisions, audit });
  }
  return { subject, items };
}

// --- Abuse signals (dead-credential abuse, referral farming, throttle pressure) ---

export interface AbuseSignals {
  redemption_attempts: { redeemed: number; dead: number; unknown: number };
  leaked_credentials: { credential: string; dead_attempts: number; distinct_accounts: number }[];
  noisy_accounts: { account: string; dead_attempts: number }[];
  /** I3-1676 (D-0829): `unqualified_referees` = live referees with no `fan_qualification` row (erased `del_` referees are neither). */
  referral_clusters: { referrer: string; referees: number; qualified_referees: number; unqualified_referees: number; rewards_granted: number }[];
  /** I3-1676 (I22): the tail BELOW `referral_clusters`' >= 3 floor, as one platform-wide total — a farm that
   * splits its signups two per referrer account never reaches the floor and was invisible. Counts only. */
  referral_fragments: { referrers: number; referees: number; unqualified_referees: number; rewards_granted: number };
  /** `refused` (QUEUE #483): the bucket's `~` refusal marker, stripped from `window_start`. */
  throttle_pressure: { key: string; window_start: string; count: number; refused: boolean }[];
}

/**
 * Operator-wide abuse read. Dead/unknown redemption attempts are the primary
 * credential-sharing/leak signal (a live credential redeems once; repeated dead
 * hits on the same credential across accounts = a leaked code being tried).
 * Referral clusters surface many-referees-per-referrer (farming shape; the
 * verified-gate + self-referral-impossible constraints already prevent the
 * payout, this just makes the pattern visible). Throttle pressure lists the
 * hottest current rate-limit windows. Credential hashes are masked; nothing here
 * exposes a usable secret.
 */
export async function abuseSignals(db: DbLike): Promise<AbuseSignals> {
  const outcomes = (await db.prepare("SELECT outcome, COUNT(*) AS n FROM redemption_attempts GROUP BY outcome").bind().all<{ outcome: string; n: number }>()).results;
  const oc = { redeemed: 0, dead: 0, unknown: 0 };
  for (const r of outcomes) if (r.outcome === "redeemed" || r.outcome === "dead" || r.outcome === "unknown") oc[r.outcome] = num(r.n);

  const leaked = (await db
    .prepare("SELECT credential_hash, COUNT(*) AS n, COUNT(DISTINCT account_id) AS accts FROM redemption_attempts WHERE outcome IN ('dead','unknown') GROUP BY credential_hash HAVING n > 1 ORDER BY n DESC, accts DESC LIMIT 20")
    .bind()
    .all<{ credential_hash: string; n: number; accts: number }>()).results;

  const noisy = (await db
    .prepare("SELECT account_id, COUNT(*) AS n FROM redemption_attempts WHERE outcome IN ('dead','unknown') GROUP BY account_id ORDER BY n DESC LIMIT 20")
    .bind()
    .all<{ account_id: string; n: number }>()).results;

  const clusters = (await db
    .prepare(
      `SELECT r.referrer_account AS referrer_account, COUNT(*) AS referees,
              SUM(CASE WHEN q.account_id IS NOT NULL THEN 1 ELSE 0 END) AS qualified,
              SUM(CASE WHEN q.account_id IS NULL AND substr(r.referee_account, 1, 4) <> 'del_' THEN 1 ELSE 0 END) AS unqualified,
              COALESCE(SUM(r.fan_reward_granted),0) AS granted
         FROM referrals r LEFT JOIN fan_qualification q ON q.account_id = r.referee_account
        GROUP BY r.referrer_account HAVING referees >= 3 ORDER BY unqualified DESC, referees DESC LIMIT 20`,
    )
    .bind()
    .all<{ referrer_account: string; referees: number; qualified: number; unqualified: number; granted: number }>()).results;

  // I3-1676 (I22): the same per-referrer grouping and the same `unqualified` rule as `clusters` above,
  // summed over the referrers the floor leaves out (referees < 3), so fragmentation shows as a growing total.
  const fragments = await db
    .prepare(
      `SELECT COUNT(*) AS referrers, COALESCE(SUM(referees),0) AS referees, COALESCE(SUM(unqualified),0) AS unqualified, COALESCE(SUM(granted),0) AS granted FROM (
         SELECT COUNT(*) AS referees,
                SUM(CASE WHEN q.account_id IS NULL AND substr(r.referee_account, 1, 4) <> 'del_' THEN 1 ELSE 0 END) AS unqualified,
                COALESCE(SUM(r.fan_reward_granted),0) AS granted
           FROM referrals r LEFT JOIN fan_qualification q ON q.account_id = r.referee_account
          GROUP BY r.referrer_account HAVING referees < 3)`,
    )
    .bind()
    .first<{ referrers: number; referees: number; unqualified: number; granted: number }>();

  const throttle = (await db
    .prepare("SELECT rl_key, window_start, count FROM rate_limits WHERE count > 1 ORDER BY count DESC, window_start DESC LIMIT 30")
    .bind()
    .all<{ rl_key: string; window_start: string; count: number }>()).results;

  return {
    redemption_attempts: oc,
    leaked_credentials: leaked.map((r) => ({ credential: maskTag(r.credential_hash), dead_attempts: num(r.n), distinct_accounts: num(r.accts) })),
    noisy_accounts: noisy.map((r) => ({ account: r.account_id, dead_attempts: num(r.n) })),
    referral_clusters: clusters.map((r) => ({ referrer: r.referrer_account, referees: num(r.referees), qualified_referees: num(r.qualified), unqualified_referees: num(r.unqualified), rewards_granted: num(r.granted) })),
    referral_fragments: { referrers: num(fragments?.referrers), referees: num(fragments?.referees), unqualified_referees: num(fragments?.unqualified), rewards_granted: num(fragments?.granted) },
    // QUEUE #483 (B27-RATE hand-off): a bucket whose refusal was recorded this
    // window carries a trailing `~` on `window_start` (ratelimit.ts). Strip it
    // and say so as `refused`, so the operator reads "this ceiling fired".
    throttle_pressure: throttle.map((r) => ({ key: r.rl_key, window_start: r.window_start.replace(/~$/, ""), count: num(r.count), refused: r.window_start.endsWith("~") })),
  };
}

// --- Billing & subscriptions ---

/** Provider derived from the customer_id prefix — the only place source lives
 * (there's no dedicated column). apple:/google: prefixes are set by the IAP
 * layer; a bare id is a web (hosted-processor) customer. */
function subSource(customerId: string | null): "web" | "apple" | "google" | "none" {
  if (!customerId) return "none";
  if (customerId.startsWith("apple:")) return "apple";
  if (customerId.startsWith("google:")) return "google";
  return "web";
}

export interface SubscriptionRow {
  account_id: string;
  email: string | null;
  status: string; // active | canceled | expired | none
  source: "web" | "apple" | "google" | "none";
  current_period_end: string | null;
  in_grace: boolean; // active, past period end, still inside the 3-day grace
  premium: boolean; // effective entitlement now (active + within period+grace)
  has_customer: boolean;
  updated_at: string;
}

/**
 * Subscriptions joined to accounts, newest activity first. `source` is derived
 * from the customer_id prefix; `premium`/`in_grace` are computed with the same
 * rule the entitlement gate uses (premium iff active AND now < period_end +
 * grace), so the operator sees the effective state, not just the stored status.
 * Optional status filter and an account_id/email search term.
 */
export async function subscriptionList(db: DbLike, nowMs: number, opts: { status?: string; q?: string; limit?: number } = {}): Promise<SubscriptionRow[]> {
  const limit = Math.max(1, Math.min(200, opts.limit ?? 50));
  const where: string[] = [];
  const bind: unknown[] = [];
  if (opts.status) { where.push("s.status = ?"); bind.push(opts.status); }
  const term = (opts.q ?? "").trim().slice(0, SEARCH_TERM_BOUND);
  // JN-04 lead (I23-FX-TC): the email LIKE limb stops at SEARCH_SCAN_CAP hits, as the accounts/creators searches do.
  if (term) { where.push("s.account_id IN (SELECT ? UNION SELECT account_id FROM (SELECT account_id FROM accounts WHERE email LIKE ? LIMIT ?))"); bind.push(term, likePattern(term), SEARCH_SCAN_CAP); }
  const sql = `SELECT s.account_id AS account_id, a.email AS email, s.status AS status, s.customer_id AS customer_id, s.current_period_end AS current_period_end, s.updated_at AS updated_at
                 FROM subscriptions s JOIN accounts a ON a.account_id = s.account_id
                 ${where.length ? "WHERE " + where.join(" AND ") : ""}
                ORDER BY s.updated_at DESC LIMIT ?`;
  const { results } = await db.prepare(sql).bind(...bind, limit).all<{ account_id: string; email: string | null; status: string; customer_id: string | null; current_period_end: string | null; updated_at: string }>();
  return results.map((r) => {
    const endMs = r.current_period_end ? Date.parse(r.current_period_end) : NaN;
    const withinGrace = Number.isFinite(endMs) ? nowMs < endMs + SUBSCRIPTION_GRACE_MS : false;
    const premium = r.status === "active" && withinGrace;
    const inGrace = r.status === "active" && Number.isFinite(endMs) && nowMs >= endMs && nowMs < endMs + SUBSCRIPTION_GRACE_MS;
    return {
      account_id: r.account_id,
      email: r.email,
      status: r.status,
      source: subSource(r.customer_id),
      current_period_end: r.current_period_end,
      in_grace: inGrace,
      premium,
      has_customer: r.customer_id !== null,
      updated_at: r.updated_at,
    };
  });
}

export interface AccountSummary {
  account_id: string;
  email: string | null;
  tier: string;
  handle: string | null; // public profile handle, if any
  created_at: string;
}

/** Search FAN accounts (not creators) by account_id (exact), email, or public
 * profile handle. Distinct from searchCreators, which is passport-scoped. */
export async function accountSearch(db: DbLike, q: string, opts: { limit?: number } = {}): Promise<AccountSummary[]> {
  const term = (q ?? "").trim().slice(0, SEARCH_TERM_BOUND);
  if (!term) return [];
  const limit = Math.max(1, Math.min(100, opts.limit ?? 30));
  const like = likePattern(term);
  const handleLike = likePattern(handleTerm(term)); // OU-05: see handleTerm
  const { results } = await db
    .prepare(
      `WITH hit(account_id) AS (
         SELECT account_id FROM accounts WHERE account_id = ?
         UNION SELECT account_id FROM (SELECT account_id FROM accounts WHERE email LIKE ? LIMIT ?)
         UNION SELECT account_id FROM (SELECT account_id FROM user_profiles WHERE handle LIKE ? LIMIT ?)
       )
       SELECT a.account_id AS account_id, a.email AS email, a.tier AS tier, u.handle AS handle, a.created_at AS created_at
         FROM hit h
         JOIN accounts a ON a.account_id = h.account_id
         LEFT JOIN user_profiles u ON u.account_id = a.account_id
        ORDER BY a.created_at DESC LIMIT ?`,
    )
    .bind(term, like, SEARCH_SCAN_CAP, handleLike, SEARCH_SCAN_CAP, limit)
    .all<AccountSummary>();
  return results;
}

export interface AccountDetail {
  account_id: string;
  email: string | null;
  /**
   * THE `accounts.tier` COLUMN, WHICH IS A STORE-RAIL CACHE — not the fan's
   * effective tier (audit item #30, named rather than fixed). Only
   * `applySubscriptionEvent` writes it, so it reads `free` for a fan paying on
   * Stripe, and `billing.premium` below is likewise the STORE rails' fold. The
   * authoritative answer is `effectiveFanTier`, which the GDPR export now uses
   * — but reading it HERE means naming an entitlement symbol inside
   * `fn accountDetail`, which is a symbol-scoped exemption in
   * tests/invariants/paywall.test.ts's `REGION_ALLOW`, and widening that list is
   * a decision about what the operator console may see, not a refactor. Left for
   * whoever makes it; written down so the next reader does not mistake this
   * field for the tier.
   */
  tier: string;
  created_at: string;
  email_alerts: number;
  session_epoch: number;
  identities: { provider: string; email: string | null }[];
  profile: { handle: string; display_name: string | null; visibility: string } | null;
  billing: { status: string; source: string; current_period_end: string | null; premium: boolean; in_grace: boolean; has_customer: boolean };
  counts: { devices: number; saves: number; collection: number; acquisitions: number; following: number; followers: number; notifications: number; requests: number };
  credits: number;
}

/** Compact operator view of one FAN account: identity + effective billing +
 * activity counts. No secrets (push tokens, iap_token, raw credential hashes are
 * never selected). Null if the account id is unknown. */
export async function accountDetail(db: DbLike, accountId: string, nowMs: number): Promise<AccountDetail | null> {
  const a = await db.prepare("SELECT account_id, email, tier, created_at, email_alerts, session_epoch FROM accounts WHERE account_id = ?").bind(accountId).first<{ account_id: string; email: string | null; tier: string; created_at: string; email_alerts: number; session_epoch: number }>();
  if (!a) return null;
  const identities = (await db.prepare("SELECT provider, email FROM oauth_identities WHERE account_id = ? ORDER BY provider").bind(accountId).all<{ provider: string; email: string | null }>()).results;
  const profile = await db.prepare("SELECT handle, display_name, visibility FROM user_profiles WHERE account_id = ?").bind(accountId).first<{ handle: string; display_name: string | null; visibility: string }>();
  // ONE ROW PER (ACCOUNT, RAIL) since 0022, so this point-read has to SAY which
  // row it wants instead of taking whichever the engine hands back. The one that
  // matters to an operator looking at a fan is the grant reaching furthest
  // forward — the same rail `billingStatus` reports below, so the two halves of
  // this view cannot disagree. (`bs` is the authoritative fold over all rails;
  // this row supplies only the display `source` and the grace flag.)
  // BF-05(b) — ONE RESOLVER, NOT TWO SELECTIONS.
  //
  // This used to pick its own row with `ORDER BY current_period_end DESC LIMIT
  // 1`, REGARDLESS OF LIVENESS, and take `source` + the grace flag from it —
  // while `status`, the period and `premium` came from the rail fold. Those two
  // choices disagree on a supported state: a revoked Apple annual reaching
  // further forward than a live Google month (`iap.ts` marks a revoked Apple
  // transaction `canceled` and keeps its period end). The dossier then named
  // the wrong store as the source of a paid entitlement, and — because the
  // canceled row is not "active" — reported `in_grace: false` for a live
  // subscription that was days from lapsing. It also folded NO Stripe rail at
  // all, so a Stripe-paying fan read `premium: false` on a screen the admin UI
  // describes as effective entitlement.
  //
  // `combinedBillingStatus` answers all of it from one place: which grant is
  // holding this fan up, its period, its grace, and which rail it is on.
  const cb = await combinedBillingStatus(db, accountId, nowMs);
  // The store row is now looked up BY THE RAIL the resolver chose, so
  // `subSource` describes that grant rather than whichever row sorted first.
  const sub = cb.rail
    ? await db
        .prepare("SELECT status, customer_id, current_period_end FROM subscriptions WHERE account_id = ? AND rail = ?")
        .bind(accountId, cb.rail)
        .first<{ status: string; customer_id: string | null; current_period_end: string | null }>()
    : null;
  const bs = cb;
  // `web` is the admin vocabulary for hosted web checkout, which is what the
  // Stripe rail IS — the same word `subSource` already returns for a
  // non-store customer id.
  const source = cb.provider === "stripe" ? ("web" as const) : subSource(sub?.customer_id ?? null);
  const inGrace = cb.in_grace;
  const one = async (sql: string): Promise<number> => num((await db.prepare(sql).bind(accountId).first<{ n: number }>())?.n);
  const counts = {
    devices: await one("SELECT COUNT(*) AS n FROM devices WHERE account_id = ?"),
    saves: await one("SELECT COUNT(*) AS n FROM saves WHERE account_id = ?"),
    collection: await one("SELECT COUNT(*) AS n FROM card_collections WHERE account_id = ?"),
    acquisitions: await one("SELECT COUNT(*) AS n FROM acquisitions WHERE account_id = ?"),
    following: await one("SELECT COUNT(*) AS n FROM follows WHERE follower_id = ?"),
    followers: await one("SELECT COUNT(*) AS n FROM follows WHERE followee_id = ?"),
    notifications: await one("SELECT COUNT(*) AS n FROM notifications WHERE account_id = ?"),
    requests: await one("SELECT COUNT(*) AS n FROM creator_requests WHERE account_id = ?"),
  };
  const credits = num((await db.prepare("SELECT COALESCE(SUM(credits),0) AS n FROM credit_ledger WHERE account_id = ?").bind(accountId).first<{ n: number }>())?.n);
  return {
    account_id: a.account_id,
    email: a.email,
    tier: a.tier,
    created_at: a.created_at,
    email_alerts: a.email_alerts,
    session_epoch: a.session_epoch,
    identities,
    profile: profile ? { handle: profile.handle, display_name: profile.display_name, visibility: profile.visibility } : null,
    billing: { status: bs.status, source, current_period_end: bs.current_period_end, premium: bs.tier === "premium", in_grace: !!inGrace, has_customer: bs.has_customer },
    counts,
    credits,
  };
}

export interface AgencyRow { agency_id: string; display_name: string; status: string; created_at: string; active_grants: number }

/** All provisioned agencies with their live grant count. */
export async function listAgencies(db: DbLike): Promise<AgencyRow[]> {
  const { results } = await db.prepare("SELECT agency_id, display_name, status, created_at FROM agencies ORDER BY created_at DESC").bind().all<{ agency_id: string; display_name: string; status: string; created_at: string }>();

/** One agency with the creators it currently holds a grant on (+ scope). Null if
 * the agency id is unknown. Revocation is a creator-passkey ledger act, so this
 * always reflects the live ledger state.
 *
 * RETRACT2 (S-032, QUEUE #105) — DELIBERATELY NOT GATED ON `retracted`. This is
 * an operator-facing read (the M7 admin surface), not a fan surface: an agency
 * grant on a retracted passport is exactly the row an operator investigating a
 * takedown, an agency dispute, or a stale grant needs to see. Suppressing it
 * here would hide the operator's own audit trail behind the same control that
 * protects fans from a delisted creator's name — and there is no name to
 * protect: `display_name` is already '' post-retraction (M30), so this reads
 * as a blank rather than a leak. See
 * `services/trust-core/test/s032-retract2-readers.test.ts` for the pinned
 * control (the grant row still serves after retraction). */
export async function agencyDetail(db: DbLike, agencyId: string): Promise<AgencyDetail | null> {
  const a = await db.prepare("SELECT agency_id, display_name, status, created_at FROM agencies WHERE agency_id = ?").bind(agencyId).first<{ agency_id: string; display_name: string; status: string; created_at: string }>();
  if (!a) return null;
  const grants = (await activeGrants(db)).filter((g) => g.agency_id === agencyId);

  }));
  const members = await listAgencyMembers(db, agencyId);
  return { ...a, active_grants: detailed.length, grants: detailed, members };
}

// --- Payout monitor (M7) ---

export interface PayoutChange { passport_id: string; display_name: string | null; tag_type: string; masked_value: string; status: string; effective_at: string; created_at: string }

/** Recent payout-tag changes across ALL creators — the monitor for the one
 * money-adjacent field. Values are masked (enough to recognize, not to copy).
 * Payout tags are NOT ledger events, so this reads the fold table directly. */
/**
 * UI-AUDIT — THE REPEAT-CHARGEBACK SIGNAL, and the shape of the decision behind
 * it.
 *
 * A chargeback ban lives on `accounts.banned_at`. Erasing the account clears it,
 * and re-registering yields a fresh account id — so the ban does not survive an
 * erase-and-return loop. The alternative considered was persisting Stripe's
 * `payment_method.card.fingerprint` and refusing purchases from a card with a
 * prior dispute, which would close the loop and would also mean Dhwaja starts
 * holding a durable financial identifier that outlives account erasure: a new
 * category of data in the privacy policy and in every DSAR, on a product whose
 * whole posture is that the trust layer stores as little as it can (money: D-0381 — Dhwaja handles money in exactly one direction: fans buy Credits, creators get paid out).
 *
 * DECIDED: keep the ban per-account, store nothing new, and rely on the
 * processor. Stripe Radar already blocks repeat fraudsters at the payment layer,
 * before a credit is ever granted — that is a control we get without holding the
 * identifier ourselves.
 *
 * WHAT THAT LEAVES, stated rather than implied: a determined fan can erase,
 * re-register and dispute again, and nothing in Dhwaja alone will link the two.
 * This read is the residual mitigation — it makes the pattern VISIBLE to a human
 * without storing anything to detect it automatically. It correlates on the
 * verified sign-in identity we already hold in `oauth_identities`, which is
 * enough to notice "these three banned accounts are one person" and is
 * deliberately NOT wired to any automatic refusal: an identity is trivially
 * changed, so acting on it belongs to an operator with the rest of the picture,
 * not to a gate that a new email address defeats.
 *
 * READ-ONLY, and it writes nothing.
 */
export interface DisputeRow {
  account_id: string;
  banned_at: string | null;
  disputes: number;
  refunds: number;
  credits_clawed: number;
  reserve_loss_cents: number;
  last_at: string;
  /** Verified sign-in identities on this account, provider-qualified. Present so
   * an operator can see the correlation below; never used as a gate. */
  identities: string[];
  /** OTHER accounts sharing one of those identities that also carry a dispute.
   * This is the repeat-fraud signal: an empty list is the normal case.
   * D-0960(13): also ERASED disputers (their money tombstone, `del_`…) whose kept fingerprint
   * (account_data.ts erased_dispute_fingerprints) matches one of this account's verified identities. */
  linked_accounts: string[];
}

/** D-1003(3): `previousSecrets` (sessions.ts previousSessionSecrets) — a live row's fingerprints are ALSO computed under
 * every retired SESSION_SECRET, so a tombstone that kept a fingerprint keyed before a rotation still matches. */
export async function disputeMonitor(db: DbLike, opts: { limit?: number; fingerprintSecret?: string; previousSecrets?: readonly string[] } = {}): Promise<DisputeRow[]> {
  const limit = Math.max(1, Math.min(200, opts.limit ?? 50));
  const { results } = await db
    .prepare(
      `SELECT r.account_id AS account_id,
              a.banned_at AS banned_at,
              SUM(CASE WHEN r.kind = 'dispute' THEN 1 ELSE 0 END) AS disputes,
              SUM(CASE WHEN r.kind = 'refund' THEN 1 ELSE 0 END) AS refunds,
              SUM(r.credits_clawed) AS credits_clawed,
              SUM(r.reserve_loss_cents) AS reserve_loss_cents,
              MAX(r.at) AS last_at
         FROM credit_reversals r LEFT JOIN accounts a ON a.account_id = r.account_id
        GROUP BY r.account_id
        ORDER BY MAX(r.at) DESC
        LIMIT ?`,
    )
    .bind(limit)
    .all<{ account_id: string; banned_at: string | null; disputes: number; refunds: number; credits_clawed: number; reserve_loss_cents: number; last_at: string }>();

  const out: DisputeRow[] = [];
  for (const r of results) {
    // VERIFIED EMAILS ONLY, on both sides of the join. `oauth_identities` states
    // the rule for cross-provider linking in its own schema comment — "an
    // unverified email can never link, so there is no account-takeover surface"
    // — and it applies here for a sharper reason: a provider-asserted email that
    // nobody verified is attacker-choosable, so correlating on one would let
    // somebody hang a repeat-fraud flag on an account by claiming its address.
    // An operator screen that can be poisoned into accusing the wrong person is
    // worse than no screen.
    const ids = (await db.prepare("SELECT provider, email FROM oauth_identities WHERE account_id = ? AND email IS NOT NULL AND email_verified = 1 ORDER BY provider").bind(r.account_id).all<{ provider: string; email: string | null }>()).results;
    const identities = ids.filter((i) => i.email).map((i) => `${i.provider}:${i.email}`);
    // Correlate on an identity we ALREADY hold. Bounded to accounts that also
    // have a reversal, so this is a repeat-dispute signal and not a general
    // "who else shares this email" lookup — an operator screen should answer the
    // question it is for and no wider one.
    const linked = new Set<string>();
    for (const i of ids) {
      if (!i.email) continue;
      const { results: peers } = await db
        .prepare(
          `SELECT DISTINCT o.account_id AS account_id FROM oauth_identities o
             WHERE o.provider = ? AND o.email = ? AND o.email_verified = 1 AND o.account_id <> ?
               AND EXISTS (SELECT 1 FROM credit_reversals x WHERE x.account_id = o.account_id AND x.kind = 'dispute')`,
        )
        .bind(i.provider, i.email, r.account_id)
        .all<{ account_id: string }>();
      for (const p of peers) linked.add(p.account_id);
    }
    // D-0960(13): erased disputers. A live row's fingerprints are computed from its verified identities (needs the
    // secret: without it this arm links nothing); a tombstone row's are the ones its erasure kept. Either way the
    // match is another tombstone that kept the same fingerprint and still carries a dispute reversal.
    const fps = r.account_id.startsWith("del_")
      ? (await db.prepare("SELECT fingerprint FROM erased_dispute_fingerprints WHERE tombstone_id = ?").bind(r.account_id).all<{ fingerprint: string }>()).results.map((x) => x.fingerprint)
      : opts.fingerprintSecret
        ? [...new Set((await Promise.all([opts.fingerprintSecret, ...(opts.previousSecrets ?? [])].map((k) => verifiedIdentityFingerprints(db, r.account_id, k)))).flat().map((f) => f.fingerprint))]
        : [];
    for (const chunk of inChunks(fps)) {
      const { results: erased } = await db
        .prepare(
          `SELECT DISTINCT e.tombstone_id AS account_id FROM erased_dispute_fingerprints e
             WHERE e.fingerprint IN (${chunk.map(() => "?").join(",")}) AND e.tombstone_id <> ?
               AND EXISTS (SELECT 1 FROM credit_reversals x WHERE x.account_id = e.tombstone_id AND x.kind = 'dispute')`,
        )
        .bind(...chunk, r.account_id)
        .all<{ account_id: string }>();
      for (const p of erased) linked.add(p.account_id);
    }
    out.push({ ...r, identities, linked_accounts: [...linked].sort() });
  }
  return out;
}

/** W3-09-batch J-01 stage 2 (D-0598/D-0596) — ONE reversal's creator linkage,
 * for a human to check BEFORE any automatic claw exists.
 *
 * WHY THIS IS PER-REVERSAL AND NOT A LISTING. D-0596 lifts INV-004's k-floor for
 * this screen on a scope that is the reason it is safe: authenticated `/ops`,
 * per-creator, serving only figures the operator is already entitled to see
 * about that ONE creator — "never an aggregate across creators, never a
 * population count, never a ranking", and a cross-creator view is a stop. A
 * listing over many reversals fans out over arbitrary creators, so this reader
 * takes a single `reversal_id` (the operator already holds it from
 * `/ops/disputes`) and orders by `passport_id`, a stable key and not a ranking.
 *
 * WHAT IS ATTRIBUTED AND WHAT IS NOT — the whole point of the row.
 * `recovered` is derived from the marks the reversal itself made:
 * `reversed_by = ? AND reversed = 1`, the SAME predicate `refunds.ts:324` and
 * `:326` already use for the reversal record's two totals, so this screen and
 * those totals are the same set of sales by construction.
 * THE OBJECT SPEAKS ONE DENOMINATION AND SAYS WHICH: GROSS BACKING CENTS, in
 * `reconcile_in`. It used to speak two at once — the remainder GROSS, every
 * per-creator figure NET — with this docblock inviting the operator to reconcile
 * them, so one 200c sale read as remainder 200 against candidate 160 and the 40c
 * gap was an artefact of the units rather than anything about the money. Gross is
 * the side that cannot be converted: the remainder is the charged-back money NO
 * sale row accounts for, so there is no `platform_cut_cents` to subtract. The
 * per-creator side CAN be converted and is, from `sales.gross_backing_cents`, so
 * every row carries both and the reconciliation happens inside one unit.
 *
 * `unattributed_remainder_gross_cents` is `credit_reversals.reserve_loss_cents` and
 * carries NO creator field, because which creator received a past-hold spend is
 * NOT a fact this database holds: `refunds.ts:265` restricts marking to
 * `clears_at > ?`, the pool is capped at `MAX_REVERSAL_SALES` (`refunds.ts:51`,
 * bound `:281`), credits are fungible and there is no purchase -> sale edge
 * anywhere in the schema. `past_hold_candidates` is therefore a CANDIDATE set
 * and every row says `attributed: false`. A column that read "creator who owes
 * this" over the remainder is the error this screen exists to avoid.
 *
 * NO WRITE. This function issues four reads and nothing else; the claw is a
 * separate room and the attribution rule is not decided.
 *
 * ERASURE (FAMILY-002). A FAN erasure: `account_data.ts` `deleteAccount` (its `UPDATE credit_reversals SET account_id`
 * and `UPDATE sales SET account_id` statements) re-keys
 * `credit_reversals.account_id` and `sales.account_id` onto one shared money
 * tombstone, and touches neither `reversal_id`, `reversed_by`,
 * `passport_id` nor `creator_net_cents`. So every figure this reader SERVES is
 * byte-identical before and after a fan erasure; the only field that moves is
 * `account_id`, surfaced as the tombstone exactly as `disputeMonitor` already
 * surfaces it (`admin.ts` disputes query, `account_id` straight off the row).
 * `account_is_tombstoned` states which it is, so an operator reading a `del_`
 * id is not left guessing whether it is a bug.
 * A CREATOR erasure is different: `creator-erasure.ts` `eraseCreator` (`UPDATE sales SET passport_id = ?`) re-keys
 * that creator's `sales.passport_id` onto its `del_` tombstone, and `recovered` and `past_hold_candidates` group by
 * and serve `sales.passport_id`, so after it that creator's rows are served under the `del_` id (the sums are unchanged).
 */
export const REVERSAL_CANDIDATE_LIMIT = 200;

export interface ReversalCreatorRow {
  passport_id: string;
  /** Creator NET. Named, because `recovered_gross_cents` sits beside it. */
  recovered_net_cents: number;
  /** GROSS backing cents — the denomination `reconcile_in` names, and the only
   * one in which this row is comparable with `unattributed_remainder_gross_cents`. */
  recovered_gross_cents: number;
  sales_reversed: number;
  attributed: true;
}

export interface ReversalCandidateRow {
  passport_id: string;
  /** Creator NET — what a claw would actually take from this creator. */
  creator_net_cents: number;
  /** GROSS backing cents for the same sales, so the candidate side can be read
   * in the denomination the remainder is in (`reconcile_in`). */
  gross_backing_cents: number;
  sales_count: number;
  /** FALSE, always, and it is a literal so no caller can read it as a claim. */
  attributed: false;
}

/** I20-JN-B01 W1A-07 (D-0960(1)): a lost dispute's past-hold remainder IS attributed — the copy says how. */
export const LOST_DISPUTE_ATTRIBUTION = "creator_dispute_debits — a lost dispute's past-hold remainder is attributed newest-first to the fan's cleared sales since the purchase and netted against each creator's next payout (D-0960(1))" as const;

export interface ReversalAttribution {
  reversal_id: string;
  account_id: string;
  account_is_tombstoned: boolean;
  kind: string;
  at: string;
  credits_clawed: number;
  /** The reversal record's own total, for the operator to reconcile against the
   * sum of `recovered_net_cents` below — the two are the same predicate at HEAD,
   * so a disagreement is a real signal and not a rounding artefact. NET: it is
   * `SUM(creator_net_cents)` at `refunds.ts:324`, and the name now says so. */
  sales_reversed_net_cents: number;
  recovered: ReversalCreatorRow[];
  /** `none` vs `present`: an EMPTY recovery and an absent one are never
   * byte-identical (FAMILY-012). `none` is the normal case for a reversal that
   * found no within-hold sale at all. */
  recovered_state: "none" | "present";
  /** GROSS BACKING CENTS, and the name is the fix. It is
   * `MAX(0, clawBackingCents - SUM(gross_backing_cents))` (`refunds.ts:326`):
   * the charged-back money NO sale row accounts for. There is no sale behind it,
   * so there is no `platform_cut_cents` to subtract and gross -> net is not a
   * computable conversion — which is why THIS is the denomination the object
   * speaks and the per-creator side carries the converted twin instead. */
  unattributed_remainder_gross_cents: number;
  /** The ONE axis on which the two halves of this object are comparable. Fixed
   * string, so a renderer states it rather than inferring it from field names. */
  reconcile_in: "gross_backing_cents";
  /** Fixed strings. A REFUND's past-hold remainder has no attribution rule and this screen does not mint one; a
   * LOST DISPUTE's is attributed by D-0960(1) (I20-JN-B01 W1A-07): refunds.ts writes creator_dispute_debits rows
   * (newest-first over the fan's cleared sales since the purchase) netted against each creator's next payout. */
  remainder_attribution: "none — not derivable from the schema" | typeof LOST_DISPUTE_ATTRIBUTION;
  past_hold_candidates: ReversalCandidateRow[];
  candidates_state: "none" | "present";
  /** The lower time bound the candidate list was taken under —
   * `credit_purchases.at` for this reversal's own purchase. NULL means the
   * purchase row could not be read, and then the list is EMPTY by construction
   * rather than unbounded (see `reconciliation`). */
  candidates_from: string | null;
  /** Sales that were in the candidate population and have since been taken by a
   * DIFFERENT reversal. The remainder is frozen at `at`; this list is live. A
   * non-zero count is the reconciliation the docblock invites silently ceasing
   * to be possible, so it is served as a number rather than left to be noticed. */
  candidates_moved_since_reversal: number;
  /** `possible` means the remainder and the candidate list still describe the
   * same moment. `broken` means they do not, and `reconciliation_note` says why
   * and who acts next. */
  reconciliation: "possible" | "broken";
  reconciliation_note: string;
  /** TRUE means the candidate list was cut at `REVERSAL_CANDIDATE_LIMIT`
   * creators. FAMILY-003: the caller may conclude NOTHING about the creators
   * beyond the cut except that at least one exists — not their number, not
   * their share, and above all not that the remainder is attributable to the
   * ones shown. */
  candidates_truncated: boolean;
}

export async function reversalAttribution(db: DbLike, reversalId: string): Promise<ReversalAttribution | null> {
  const rev = await db
    .prepare("SELECT reversal_id, account_id, purchase_id, kind, credits_clawed, sales_reversed_cents, reserve_loss_cents, at FROM credit_reversals WHERE reversal_id = ?")
    .bind(reversalId)
    .first<{ reversal_id: string; account_id: string; purchase_id: string; kind: string; credits_clawed: number; sales_reversed_cents: number; reserve_loss_cents: number; at: string }>();
  if (!rev) return null;

  // THE ATTRIBUTED HALF. `reversed = 1` is not decoration: `reversed_by` is set
  // in the same UPDATE as the flip, so a row carrying this id that the UPDATE
  // did not reverse is not a recovery (the landed-flip rule refunds.ts:324/:326
  // and the analytics emit at refunds.ts:284 all apply).
  const recoveredRows = (
    await db
      .prepare(
        `SELECT passport_id, COALESCE(SUM(creator_net_cents),0) AS recovered_net_cents, COALESCE(SUM(gross_backing_cents),0) AS recovered_gross_cents, COUNT(*) AS sales_reversed
           FROM sales WHERE reversed_by = ? AND reversed = 1
          GROUP BY passport_id ORDER BY passport_id`,
      )
      .bind(reversalId)
      .all<{ passport_id: string; recovered_net_cents: number; recovered_gross_cents: number; sales_reversed: number }>()
  ).results;

  // THE UNATTRIBUTED HALF, as a candidate set and nothing more. Past-hold at the
  // instant the reversal ran (`clears_at <= rev.at`) is exactly the population
  // `refunds.ts:265`'s `clears_at > ?` excluded from marking, read back on the
  // account the reversal names — which after an erasure is the money tombstone,
  // the same identifier the reversal itself was re-keyed onto, so the two sides
  // still meet.
  // THE LOWER TIME BOUND (W3-09 row 3). Without one this list named sales that
  // CLEARED before the disputed purchase existed — money that provably cannot
  // have been funded by the credits being clawed. The bound is not a judgment
  // call: `credit_reversals.purchase_id` is a UNIQUE FK onto the purchase, whose
  // `at` is the instant the credits came into existence, and credits cannot be
  // spent before they are granted. So the bound goes on `at`, the SPEND instant,
  // not on `clears_at`: `clears_at = at + PAYOUT_HOLD`
  // (`migrations/0058_credit_journal.sql:99`), so a `clears_at` bound still
  // admits a sale spent before the purchase landed.
  const purchase = await db
    .prepare("SELECT at FROM credit_purchases WHERE purchase_id = ?")
    .bind(rev.purchase_id)
    .first<{ at: string }>();

  // THE ZERO-REMAINDER CASE (W3-09 row 1). A candidate list is the answer to
  // "who might the UNATTRIBUTED money have gone to". With nothing unattributed
  // the question has no subject, and this is not cosmetic: the object's only
  // consumer is an operator holding a clawback question, and a list of real
  // creators with real cents figures rendered beside that question is read as a
  // shortlist however it is labelled. The claw takes money from a real person's
  // balance and is not reversible (Jason's constraint), so an empty question
  // gets an empty answer, stated as `candidates_state: "none"` and never a
  // silent omission (FAMILY-012).
  //
  // AND A BOUND WE COULD NOT READ IS NOT A BOUND WE MAY SKIP (FAMILY-003): if
  // the purchase row is unreadable the population is unbounded below, so the
  // list is empty by construction and `reconciliation` says so, rather than the
  // read quietly widening to every past-hold sale on the account.
  const remainderGrossCents = rev.reserve_loss_cents;
  const candidatesApply = remainderGrossCents > 0 && purchase != null;
  const candidateRows = !candidatesApply
    ? []
    : (
        await db
          .prepare(
            `SELECT passport_id, COALESCE(SUM(creator_net_cents),0) AS creator_net_cents, COALESCE(SUM(gross_backing_cents),0) AS gross_backing_cents, COUNT(*) AS sales_count
               FROM sales WHERE account_id = ? AND at >= ? AND clears_at <= ? AND reversed = 0
              GROUP BY passport_id ORDER BY passport_id LIMIT ?`,
          )
          .bind(rev.account_id, purchase!.at, rev.at, REVERSAL_CANDIDATE_LIMIT + 1)
          .all<{ passport_id: string; creator_net_cents: number; gross_backing_cents: number; sales_count: number }>()
      ).results;
  const truncated = candidateRows.length > REVERSAL_CANDIDATE_LIMIT;

  // A FROZEN REMAINDER BESIDE A LIVE LIST (W3-09 row 4). `reserve_loss_cents` is
  // a stored column, fixed at `rev.at`; the candidate query above runs now. Any
  // LATER reversal that takes one of these sales moves the list and not the
  // remainder, so the reconciliation this object invites stops being possible —
  // and used to stop being possible with no visible change at all. This counts
  // the same population under `reversed = 1 AND reversed_by <> this reversal`,
  // which is exactly the set that has left the list since. `reversed_by IS NULL`
  // is included: a reversed row that carries no id was not taken by THIS
  // reversal either.
  const movedRow = !candidatesApply
    ? null
    : await db
        .prepare(
          `SELECT COUNT(*) AS n FROM sales
            WHERE account_id = ? AND at >= ? AND clears_at <= ? AND reversed = 1
              AND (reversed_by IS NULL OR reversed_by <> ?)`,
        )
        .bind(rev.account_id, purchase!.at, rev.at, rev.reversal_id)
        .first<{ n: number }>();
  const moved = movedRow?.n ?? 0;
  const reconciliation: "possible" | "broken" = purchase == null || moved > 0 ? "broken" : "possible";
  const reconciliationNote =
    purchase == null
      ? `broken — the purchase row \`${rev.purchase_id}\` this reversal names could not be read, so the candidate population has no lower time bound and no candidate list is served. An operator cannot repair this from this screen: it is a data question for whoever owns \`credit_purchases\`.`
      : moved > 0
        ? `broken — ${moved} sale(s) in this reversal's past-hold population have since been reversed by something else. The remainder is frozen at ${rev.at}; the candidate list is live, so the two no longer describe the same moment and subtracting one from the other is meaningless. The operator's next act is to read the LATER reversal, not this one.`
        : "possible — the remainder and the candidate list still describe the same moment.";

  return {
    reversal_id: rev.reversal_id,
    account_id: rev.account_id,
    account_is_tombstoned: rev.account_id.startsWith("del_"),
    kind: rev.kind,
    at: rev.at,
    credits_clawed: rev.credits_clawed,
    sales_reversed_net_cents: rev.sales_reversed_cents,
    recovered: recoveredRows.map((r) => ({ ...r, attributed: true as const })),
    recovered_state: recoveredRows.length === 0 ? "none" : "present",
    unattributed_remainder_gross_cents: remainderGrossCents,
    reconcile_in: "gross_backing_cents",
    remainder_attribution: rev.kind === "dispute" ? LOST_DISPUTE_ATTRIBUTION : "none — not derivable from the schema",
    past_hold_candidates: candidateRows.slice(0, REVERSAL_CANDIDATE_LIMIT).map((r) => ({ ...r, attributed: false as const })),
    candidates_state: candidateRows.length === 0 ? "none" : "present",
    candidates_from: purchase?.at ?? null,
    candidates_moved_since_reversal: moved,
    reconciliation,
    reconciliation_note: reconciliationNote,
    candidates_truncated: truncated,
  };
}

/** I3-1226 (D-0829): the window `payoutSummary` aggregates, by `created_at`. */
export const PAYOUT_SUMMARY_DAYS = 30;
export const PAYOUT_STATUSES = ["pending", "paid", "failed", "voided"] as const;
export interface PayoutSummary { since: string; by_status: Record<(typeof PAYOUT_STATUSES)[number], { count: number }> }

/** I3-1226: count of `creator_payouts` by status, created in the last PAYOUT_SUMMARY_DAYS. Aggregate only — no
 * creator id, no row. Every status the 0099 CHECK admits is present, zero-filled. I5-Q4 (D-1030 "(a) counts only"):
 * no amount, for any role — the SELECT below reads no `amount_cents`. */
export async function payoutSummary(db: DbLike, nowMs: number): Promise<PayoutSummary> {
  const since = new Date(nowMs - PAYOUT_SUMMARY_DAYS * DAY_MS).toISOString();
  const { results } = await db
    .prepare("SELECT status, COUNT(*) AS n FROM creator_payouts WHERE created_at >= ? GROUP BY status")
    .bind(since)
    .all<{ status: string; n: number }>();
  const by_status = Object.fromEntries(PAYOUT_STATUSES.map((s) => [s, { count: 0 }])) as PayoutSummary["by_status"];
  for (const r of results) if ((PAYOUT_STATUSES as readonly string[]).includes(r.status)) by_status[r.status as (typeof PAYOUT_STATUSES)[number]] = { count: num(r.n) };
  return { since, by_status };
}

export interface VerificationRow { item_id: string; subject: string; state: string; disposition: string | null; kyc_ref: string | null; expires_at: string | null; created_at: string; updated_at: string }
export interface VerificationActivity { open_count: number; recent: VerificationRow[] }

/** Verification queue depth + recent decisions. kyc_ref / expires_at live in the
 * REVIEW ITEM's payload, so they're parsed from there. Only `expires_at` rides
 * into the `verification_granted` event; `kyc_ref` was removed from the ledger
 * (window item 2 — a shape-free, human-typed identity-document reference had no
 * business in an append-only log) and is written by review.ts to the mutable
 * `verification_refs` table instead, in the same atomic batch as the approval.
 * This operator read still shows it because it is the operator's own
 * bookkeeping.
 *
 * EXPIRY IS ENFORCED (audit item 6, docs/VERIFIED-MINT.md §1). `expires_at` is
 * no longer a decorative field: `verification-sweep.ts` appends
 * `verification_expired` once the deadline passes and the fold drops the
 * passport verified → claimed, so the date in this column is the date the
 * creator loses the badge, the card, the payout-tag gate and every mint path.
 * `approveItem` refuses to grant without a parseable one. The exception an
 * operator still has to know about: grants approved BEFORE this landed may
 * carry an empty/unparseable `expires_at`, and those are deliberately left
 * verified (the sweep counts them rather than mass-downgrading them) — they
 * show here with a blank Expires cell and need re-verifying by hand. */
export async function verificationActivity(db: DbLike, opts: { limit?: number } = {}): Promise<VerificationActivity> {
  const limit = Math.max(1, Math.min(100, opts.limit ?? 40));
  const openRow = await db.prepare("SELECT COUNT(*) AS n FROM review_queue WHERE item_type = 'verification' AND state = 'open'").bind().first<{ n: number }>();
  const { results } = await db
    .prepare("SELECT item_id, subject, state, disposition, payload, created_at, updated_at FROM review_queue WHERE item_type = 'verification' ORDER BY updated_at DESC LIMIT ?")
    .bind(limit)
    .all<{ item_id: string; subject: string; state: string; disposition: string | null; payload: string; created_at: string; updated_at: string }>();
  const recent = results.map((r) => {
    let kyc_ref: string | null = null;
    let expires_at: string | null = null;
    try {
      const p = JSON.parse(r.payload) as { kyc_ref?: unknown; expires_at?: unknown };
      if (typeof p.kyc_ref === "string") kyc_ref = p.kyc_ref;
      if (typeof p.expires_at === "string") expires_at = p.expires_at;
    } catch { /* leave null */ }
    return { item_id: r.item_id, subject: r.subject, state: r.state, disposition: r.disposition, kyc_ref, expires_at, created_at: r.created_at, updated_at: r.updated_at };
  });
  return { open_count: num(openRow?.n), recent };
}

export interface LikenessReasonCount { status: string; reason: string | null; jobs: number; passports: number; last_7d: number; latest: string }

/**
 * QUEUE #185 F-35: the likeness view counted per REASON, so a PATTERN of
 * refusals (the minor-related pre-filter above all) is one line with a
 * number beside it rather than one row per attempt. `passports` is how many
 * distinct creators it spans; `last_7d` is the recent share. Rejected and
 * failed jobs only - `done`/`queued`/`generating` carry no reason.
 */
export async function likenessReasonCounts(db: DbLike, nowMs: number = Date.now()): Promise<LikenessReasonCount[]> {
  const weekAgo = new Date(nowMs - 7 * 24 * 60 * 60 * 1000).toISOString();
  const { results } = await db
    .prepare(
      `SELECT status, reason, COUNT(*) AS jobs, COUNT(DISTINCT passport_id) AS passports,
              SUM(CASE WHEN created_at >= ? THEN 1 ELSE 0 END) AS last_7d, MAX(created_at) AS latest
         FROM likeness_jobs WHERE status IN ('rejected', 'failed')
        GROUP BY status, reason ORDER BY jobs DESC, status ASC, reason ASC LIMIT 50`,
    )
    .bind(weekAgo)
    .all<LikenessReasonCount>();
  return results.map((r) => ({ status: r.status, reason: r.reason, jobs: num(r.jobs), passports: num(r.passports), last_7d: num(r.last_7d), latest: r.latest }));
}

// --- Registry build health (M7) ---

/** The cron runs every 6h (see `triggers.crons` in wrangler.jsonc), so a served
 * manifest older than TWO of them is a rebuild that did not happen rather than a
 * slow one. */
export const REGISTRY_STALE_AFTER_HOURS = 12;

export interface RegistryHealth {
  store_available: boolean;
  manifest: { v: number; width: number; built_at: string; buckets: number; pad?: number; churn?: number; seq?: number; prefix?: string } | null;
  current_real_accounts: number; // live count now, to compare against the last build
  age_hours: number | null; // age of the served manifest; null = no manifest, or a `built_at` that won't parse
  stale: boolean; // FAIL CLOSED: an age we cannot compute reads STALE, never fresh
  /**
   * THE STAGED BUILD, ON A SURFACE (ADR-0012 §7, REG-9). A generation is now
   * assembled across cron ticks, so "is a build running, and where has it got
   * to?" is a question an operator can only answer from here — the log lines
   * below say what HAPPENED, this says what is happening now. `null` when no
   * build is open, which under the publication promise should be the state for
   * most of every six-hour cycle.
   */
  build: { seq: number; state: string; phase: string; cursor: number; failures: number; started_at: string; updated_at: string } | null;
  /** The last generation that reached the wire, as its row records it. */
  last_committed: { seq: number; ended_at: string | null } | null;
  /** OR-01: safety entries the reserved lane could not admit, and the oldest
   * generation one has been waiting since. A non-zero `pending` with an old
   * `since_seq` is the rung step not happening. */
  safety_admission: { pending: number; admitted: number; oldest_pending_seq: number | null };
  /**
   * THE LOG LINES TO ALERT ON, named here so an operator surface does not have
   * to know them by heart and so the names cannot drift away from the emitter
   * unnoticed (`s092-ops-surface.test.ts` pins each one to its call site in
   * registry-build.ts).
   */
  alarms: readonly string[];
}

/** Every alarm and metric line the registry build emits, in the order an
 * operator would triage them: the publication promise first, then the envelope,
 * then the three capacity signals, then the fence.
 *
 * `registry_listing_floor_short` is the feasibility floor's own premise
 * (`hwFloor(w) >= maxND(w)`, registry-build.ts:2745): it fires never by
 * construction, and it is advertised for that reason — the build keeps
 * publishing when it fires, so the log line is the only way an operator learns
 * the floor every padding property rests on went short. */
export const REGISTRY_ALARMS = [
  "registry_publication_stale",
  "registry_capacity_exhausted",
  "registry_bucket_ceiling_breached",
  "registry_bucket_at_capacity",
  "registry_listing_floor_short",
  "registry_safety_backlog",
  // I3-1650 (I4-02-R1 hand-off): index.ts emits both at error level after a
  // publication that left removals over the per-bucket quota.
  "registry_delisting_backlog",
  "registry_retraction_backlog",
  "registry_pointer_cas_lost",
  // I3-1110 (D-0858(22), I6-REGBUILD hand-off): registry-build.ts stepRung logs it at error
  // level whenever the planned width is >= 2 (the watch cover is complete only at width 1).
  "registry_width_cover_weakened",
  // I7 judge audit (JA-05): registry-build.ts logs it when the ladder-state object and the manifest
  // disagree on sequence — a copy from another build, i.e. a possible rollback — so it is advertised,
  // not informational. The build falls back and continues.
  "registry_ladder_state_stale_fallback",
] as const;

/** Last registry build (from the served manifest.json) + the current real-entry
 * count, so the operator can see build freshness and drift since the last build.
 * totalEntries/realCount aren't persisted in the manifest, so the live count is
 * recomputed here.
 *
 * STALENESS IS THE LOAD-BEARING FIELD. A skipped rebuild is otherwise INVISIBLE:
 * `REGISTRY_DECOY_SEED` absent or placeholder makes the cron skip the build with
 * one console.error (index.ts), every other surface stays healthy, and the
 * buckets keep serving yesterday's answer — so a removal request never reaches
 * `/v1/bucket/:prefix` and the only signal is a `built_at` that stops moving and
 * that nobody is watching. `stale` turns that into a boolean an operator surface
 * can draw in red. No manifest, or one whose `built_at` will not parse, is
 * `age_hours: null` and `stale: true` — if we cannot tell how old it is, it is
 * stale. A manifest dated in the FUTURE reports its (negative) age and is stale
 * past an hour of clock skew: a build clock we cannot trust is a freshness claim
 * we cannot make. `nowMs` is threaded so the boundary is testable. */
export async function registryBuildHealth(db: DbLike, store?: ObjectStore, nowMs: number = Date.now()): Promise<RegistryHealth> {
  let manifest: RegistryHealth["manifest"] = null;
  if (store) {
    try {
      const raw = await store.get("manifest.json");
      if (raw) {
        const m = JSON.parse(raw) as { v?: number; width?: number; built_at?: string; buckets?: number; pad?: number; churn?: number; seq?: number; prefix?: string };
        if (typeof m.built_at === "string") {
          manifest = {
            v: num(m.v),
            width: num(m.width),
            built_at: m.built_at,
            buckets: num(m.buckets),
            // ADR-0012's own numbers, so the panel can show the rung, the
            // departure quota and WHICH generation is being served.
            pad: num(m.pad),
            churn: num(m.churn),
            seq: num(m.seq),
            ...(typeof m.prefix === "string" ? { prefix: m.prefix } : {}),
          };
        }
      }
    } catch { /* no/invalid manifest → null */ }
  }
  const real = num((await db.prepare("SELECT COUNT(*) AS n FROM passport_accounts a JOIN passports p ON p.passport_id = a.passport_id WHERE a.status = 'active' AND p.retracted = 0").bind().first<{ n: number }>())?.n);
  // The staged build's own state, straight off the fence's table.
  const build = await db
    .prepare("SELECT seq, state, phase, cursor, failures, started_at, updated_at FROM registry_builds WHERE state = 'open' ORDER BY seq DESC LIMIT 1")
    .bind()
    .first<{ seq: number; state: string; phase: string; cursor: number; failures: number; started_at: string; updated_at: string }>();
  const lastCommitted = await db
    .prepare("SELECT seq, ended_at FROM registry_builds WHERE state = 'committed' ORDER BY seq DESC LIMIT 1")
    .bind()
    .first<{ seq: number; ended_at: string | null }>();
  const safety = await db
    .prepare(
      `SELECT SUM(CASE WHEN state = 'pending' THEN 1 ELSE 0 END) AS pending,
              SUM(CASE WHEN state = 'admitted' THEN 1 ELSE 0 END) AS admitted,
              MIN(CASE WHEN state = 'pending' THEN since_seq END) AS oldest
         FROM registry_safety_admission`,
    )
    .bind()
    .first<{ pending: number | null; admitted: number | null; oldest: number | null }>();
  const builtMs = manifest ? Date.parse(manifest.built_at) : Number.NaN;
  const ageHours = Number.isFinite(builtMs) ? Math.round(((nowMs - builtMs) / 3_600_000) * 10) / 10 : null;
  const stale = ageHours === null || ageHours > REGISTRY_STALE_AFTER_HOURS || ageHours < -1;
  return {
    store_available: !!store,
    manifest,
    current_real_accounts: real,
    age_hours: ageHours,
    stale,
    build: build ? { seq: num(build.seq), state: build.state, phase: build.phase, cursor: num(build.cursor), failures: num(build.failures), started_at: build.started_at, updated_at: build.updated_at } : null,
    last_committed: lastCommitted ? { seq: num(lastCommitted.seq), ended_at: lastCommitted.ended_at } : null,
    safety_admission: { pending: num(safety?.pending), admitted: num(safety?.admitted), oldest_pending_seq: safety?.oldest === null || safety?.oldest === undefined ? null : num(safety.oldest) },
    alarms: REGISTRY_ALARMS,
  };
}

/**
 * LIVE facts `configHealth` cannot read from `env` alone. Absent = unknown = the row
 * that needs it reports false (fails closed).
 */
export interface ConfigProbes {
  /** The image service's own answer: `/health?ready=1` `checks.embedder_wired === true`. */
  embedder_wired?: boolean;
}

/**
 * I11-FX2 (I11-FACEGUARD, I3-0027): ask the image service whether a real face
 * embedder is wired. Same call and same fail-closed reading as `faceTriageArmed`
 * (index.ts): true ONLY for a 2xx whose body has `checks.embedder_wired === true`;
 * a throw, a timeout, a non-2xx, an unparseable body or a missing/non-boolean field
 * is false. Never called when the binding or secret is absent (nothing to ask).
 */
export async function probeEmbedderWired(env: Record<string, unknown>, timeoutMs = 5_000): Promise<boolean> {
  const svc = env.IMAGE_SERVICE as { fetch?: (url: string, init?: RequestInit) => Promise<Response> } | undefined;
  if (!svc || typeof svc.fetch !== "function") return false;
  const ctrl = new AbortController();
  const timer = setTimeout(() => ctrl.abort(), timeoutMs);
  const ask = (async (): Promise<boolean> => {
    const res = await svc.fetch!("https://image.internal/health?ready=1", { method: "GET", signal: ctrl.signal });
    if (!res.ok) return false;
    const body = (await res.json()) as { checks?: { embedder_wired?: unknown } } | null;
    return body?.checks?.embedder_wired === true;
  })();
  const timedOut = new Promise<boolean>((resolve) => ctrl.signal.addEventListener("abort", () => resolve(false)));
  try {
    return await Promise.race([ask, timedOut]);
  } catch {
    return false;
  } finally {
    clearTimeout(timer);
    ask.catch(() => {});
  }
}

/**
 * `configHealth` plus the live facts it cannot read from `env`: what `/ops/config`
 * (console.ts) serves. The image service is asked only when its binding is present and
 * `faceTriageSecretFromEnv` answers non-null (nothing to ask otherwise).
 */
export async function configHealthLive(env: Record<string, unknown>, probeTimeoutMs = 5_000): Promise<Record<string, boolean>> {
  // I59-FX-CONFIG (D-1474): the secret predicate faceTriageArmed (index.ts) reads, so the row and the sweep agree.
  const configured = !!env.IMAGE_SERVICE && faceTriageSecretFromEnv(env) !== null;
  return configHealth(env, { embedder_wired: configured ? await probeEmbedderWired(env, probeTimeoutMs) : false });
}

/**
 * I3-0261 (D-1082(4)): the operator role-gating VALUES `accessConfigFromEnv` reads — `OPS_FULL_GROUP` (null when
 * unset: every operator is support) and `OPS_GROUPS_CLAIM` — plus how many addresses `OPERATOR_EMAILS` names. Group
 * and claim names are configuration, not secrets; served beside `configHealth`'s booleans on `GET /ops/config` so a
 * changed group shows as a changed value, never only as "configured". Null when the access config is incomplete.
 */
export function opsAccessValues(env: Record<string, unknown>): { full_group: string | null; groups_claim: string; operator_emails: number } | null {
  const cfg = accessConfigFromEnv(env);
  return cfg === null ? null : { full_group: cfg.fullGroup ?? null, groups_claim: cfg.groupsClaim ?? "groups", operator_emails: cfg.operatorEmails.length };
}

/**
 * Which subsystems are configured/live vs fail-closed, as BOOLEANS only — never
 * secret values. Lets the operator see at a glance what will work in production.
 */
export function configHealth(env: Record<string, unknown>, probes: ConfigProbes = {}): Record<string, boolean> {
  const has = (k: string): boolean => {
    const v = env[k];
    return typeof v === "string" && v.trim().length > 0 && !/TODO/i.test(v);
  };
  // `has()` is only ever correct for a var with NO shape (a shared secret, an
  // opaque group name). Anything with a shape — an origin, an endpoint, a token
  // pair, a binding — is read through the predicate the FEATURE uses, so this
  // surface cannot report a subsystem healthy that will refuse the next request.
  const crypto = cryptoConfigFromEnv(env);
  const faceTriageConfigured = !!env.IMAGE_SERVICE && faceTriageSecretFromEnv(env) !== null;
  return {

/** A RUN of the characters the shared display rule refuses (packages/schema UNSAFE_DISPLAY_TEXT_RE, built from its source), so one run is one space. */
const PEER_AUDIT_UNSAFE_RUN_RE = new RegExp(`(?:${UNSAFE_DISPLAY_TEXT_RE.source})+`, "gu");
/** I43-FX-DIGEST (D-1254(5)): the at-sign is every code point whose NFKC form is `@` (U+0040, U+FE6B SMALL, U+FF20 FULLWIDTH),
 * so an address written with one is redacted like an ASCII one; built from code points, so no look-alike byte is in this file. */
const PEER_AUDIT_AT = `[@${String.fromCodePoint(0xfe6b, 0xff20)}]`;
const PEER_AUDIT_AT_RE = new RegExp(PEER_AUDIT_AT, "u");
/** I43-FX-DIGEST (D-1254(5)): code points outside `[\p{L}\p{N}.-]` whose NFKC form is wholly inside it (U+FF0E FULLWIDTH FULL
 * STOP, U+FE52 SMALL FULL STOP, U+2024 ONE DOT LEADER, U+FF0D / U+FE63 hyphen-minus forms, squared and circled letters), as ranges,
 * so a domain continues past a refused run or an invisible joiner as its ASCII form would (i43-fxdg census over every code point). */
const PEER_AUDIT_FOLD_ALPHA_RANGES: readonly (readonly [number, number])[] = Object.freeze([
  [0x2024, 0x2026], [0x20a8, 0x20a8], [0x2116, 0x2116], [0x2120, 0x2122], [0x213b, 0x213b], [0x24b6, 0x24e9], [0x2e9f, 0x2e9f], [0x2ef3, 0x2ef3], [0x2f00, 0x2fd5], [0x3196, 0x319f], [0x3244, 0x3247], [0x3250, 0x3250], [0x3260, 0x327e], [0x328a, 0x32b0], [0x32c0, 0x33a6], [0x33a9, 0x33ad], [0x33b0, 0x33c5], [0x33c7, 0x33dd], [0x33e0, 0x33ff], [0xfdfc, 0xfdfc], [0xfe19, 0xfe19], [0xfe30, 0xfe30], [0xfe52, 0xfe52], [0xfe63, 0xfe63], [0xff0d, 0xff0e], [0x1ccd6, 0x1ccef], [0x1f12b, 0x1f12e], [0x1f130, 0x1f14f], [0x1f16a, 0x1f16c], [0x1f190, 0x1f190], [0x1f200, 0x1f202], [0x1f210, 0x1f23b], [0x1f250, 0x1f251],
]);
/** One character of domain alphabet: a letter, digit, `.` or `-`, or a code point that NFKC-folds to them. */
const PEER_AUDIT_DOMAIN_ALPHA = `[\\p{L}\\p{N}.${PEER_AUDIT_FOLD_ALPHA_RANGES.map(([a, b]) => `${String.fromCodePoint(a)}-${String.fromCodePoint(b)}`).join("")}-]`;
/** One character of an address's local part: neither whitespace nor `@`, OR any character the shared display rule refuses
 * (built from its source, as the run above is), so a refused character JS `\s` also matches (U+2028, U+2029, TAB, LF, VT,
 * FF, CR) cannot split an address either (I41-JN-02 DIG-03); a Zs space, which the rule does not refuse, still ends one. */
const PEER_AUDIT_ADDRESS_PART = `(?:[^\\s@]|${UNSAFE_DISPLAY_TEXT_RE.source})`;
/** One character of its domain: the same, except that a refused run is part of the domain only when a domain character
 * (letter, digit, `.`, `-`) follows it, so `x@evil.example<LF>#999 ...` still ends the address at the line break. */
const PEER_AUDIT_DOMAIN_PART = `(?:[^\\s@]|\\p{Cf}|${UNSAFE_DISPLAY_TEXT_RE.source}(?=(?:${UNSAFE_DISPLAY_TEXT_RE.source})*${PEER_AUDIT_DOMAIN_ALPHA}))`;
/** An address, matched BEFORE unsafe runs become spaces, its parts admitting an invisible format character (\p{Cf}:
 * ZWSP, LRM, WORD JOINER, BOM), so `bob<ZWSP>@example.com` is redacted, not mailed as `bob @example.com` (I39-JUDGE-AUDIT JA-02). */
const PEER_AUDIT_ADDRESS_RE = new RegExp(`${PEER_AUDIT_ADDRESS_PART}+${PEER_AUDIT_AT}${PEER_AUDIT_DOMAIN_PART}+`, "gu");
/** I43-DIGEST (D-1245 (7) c): the marker a digest line carries where an address run took visible text beside the address. */
export const PEER_AUDIT_TEXT_REMOVED = "[text removed]";
/** A refused run made only of invisible format characters (\p{Cf}); any other refused run is a break. */
const PEER_AUDIT_INVISIBLE_RUN_RE = /^\p{Cf}+$/u;
/** A piece that is wholly domain alphabet (letter, digit, `.`, `-`). */
const PEER_AUDIT_DOMAIN_PIECE_RE = new RegExp(`^${PEER_AUDIT_DOMAIN_ALPHA}+$`, "u");
/** The replacement for one PEER_AUDIT_ADDRESS_RE run (the whole run is still removed). Its pieces between refused runs: the address
 * is the piece holding `@`, grown left while its local side is empty or the joiner is invisible-only, and right while its domain
 * side is empty or the joiner is invisible-only and the piece is domain alphabet; any other non-empty piece is removed text, so
 * `[text removed]` is written on the side it came from (D-1249(2): `bob<ZWSP>@example.com` reads `[address]`). */
function peerAuditAddressMark(run: string): string {
  const p = run.split(new RegExp(`(${PEER_AUDIT_UNSAFE_RUN_RE.source})`, "u"));
  let lo = p.findIndex((x) => PEER_AUDIT_AT_RE.test(x)); // a joiner is refused characters only, never an at-sign
  let hi = lo;
  let local = p[lo]!.slice(0, p[lo]!.search(PEER_AUDIT_AT_RE));
  let domain = p[hi]!.slice(p[hi]!.search(PEER_AUDIT_AT_RE) + 1);
  while (lo >= 2 && (local === "" || PEER_AUDIT_INVISIBLE_RUN_RE.test(p[lo - 1]!))) {
    lo -= 2;
    local = p[lo]! + local;
  }
  while (hi + 2 < p.length && (domain === "" || (PEER_AUDIT_INVISIBLE_RUN_RE.test(p[hi + 1]!) && PEER_AUDIT_DOMAIN_PIECE_RE.test(p[hi + 2]!)))) {
    hi += 2;
    domain += p[hi]!;
  }
  const before = p.slice(0, lo).some((x, i) => i % 2 === 0 && x !== "");
  const after = p.slice(hi + 1).some((x, i) => i % 2 === 1 && x !== "");
  return `${before ? `${PEER_AUDIT_TEXT_REMOVED} ` : ""}[address]${after ? ` ${PEER_AUDIT_TEXT_REMOVED}` : ""}`;
}
/** The markers peerAuditText writes; the bound drops any one it would cut, whole (D-1254(5)). */
const PEER_AUDIT_MARKS: readonly string[] = Object.freeze([PEER_AUDIT_TEXT_REMOVED, "[address]", "[account]"]);
/** The row's free text made safe for a mail line: one line (a reason cannot forge a second `#` line), no address
 * (the body carries ids, never an email), no account id (dual_control.ts NAMES_AN_ACCOUNT's shape), bounded; a marker the bound
 * would cut — `[text removed]`, `[address]` or `[account]` — is dropped whole (I43-DIGEST, W0 amendment 4; D-1254(5)). */
function peerAuditText(s: string, max: number): string {
  const t = s
    .replace(PEER_AUDIT_ADDRESS_RE, peerAuditAddressMark)
    .replace(PEER_AUDIT_UNSAFE_RUN_RE, " ")
    .replace(/acc[_#][0-9a-f]{16,}/gi, "[account]")
    .trim();
  let cut = max;
  for (const k of PEER_AUDIT_MARKS) {
    const m = t.indexOf(k, max - k.length + 1); // the first one that starts before the bound and ends past it
    if (m >= 0 && m < cut) cut = m;
  }
  return cut < max ? t.slice(0, cut).trimEnd() : t.slice(0, max);
}

/**
 * I38-DIGEST — ONE DIGEST LINE: what was done, to whom, of what class, and what the operator recorded as the decision.
 * `#<id> <at> <action> on <item> by <op>`, then ` | subject <s>` (the review item's subject; a fan report's
 * `<platform>/<handle>`, found by item id or, for `caution:*`, by its subject hash; a `psp_` item is its own
 * subject), ` | class <warning_class>`, ` | passport <id>` (the protected or suspected passport) and
 * ` | decision <detail>` (the row's `detail`: `warning:retract` writes `<disposition>: <reason>`, warnings.ts).
 */
export async function peerAuditLine(db: DbLike, r: PeerAuditRow): Promise<string> {
  const parts = [`#${r.audit_id} ${r.created_at} ${r.action}${r.item_id ? ` on ${peerAuditText(r.item_id, 120)}` : ""} by ${r.actor_id.startsWith("op:") ? "a legacy operator" : r.actor_id}`];
  let subject: string | null = null;
  let klass: string | null = null;
  let passport: string | null = null;
  if (r.item_id) {
    const rq = await db.prepare("SELECT subject, payload FROM review_queue WHERE item_id = ?").bind(r.item_id).first<{ subject: string; payload: string }>();
    if (rq) {
      subject = rq.subject;
      let payload: Record<string, unknown> = {};
      try {
        payload = JSON.parse(rq.payload) as Record<string, unknown>;
      } catch {
        payload = {};
      }
      if (typeof payload.warning_class === "string") klass = payload.warning_class;
      if (typeof payload.protected_passport_id === "string") passport = payload.protected_passport_id;
    } else {
      const fr = await db
        .prepare("SELECT platform, handle, suspected_passport_id FROM fan_report_items WHERE item_id = ? OR subject_hash = ? LIMIT 1")
        .bind(r.item_id, r.item_id)
        .first<{ platform: string; handle: string; suspected_passport_id: string | null }>();
      if (fr) {
        subject = `${fr.platform}/${fr.handle}`;
        passport = fr.suspected_passport_id;
      } else if (r.item_id.startsWith("psp_")) subject = r.item_id;
    }
  }
  if (subject) parts.push(`subject ${peerAuditText(subject, 120)}`);
  if (klass) parts.push(`class ${peerAuditText(klass, 40)}`);
  if (passport) parts.push(`passport ${peerAuditText(passport, 40)}`);
  const detail = r.detail ? peerAuditText(r.detail, PEER_AUDIT_DETAIL_MAX) : "";
  if (detail) parts.push(`decision ${detail}`);
  return parts.join(" | ");
}

export interface PeerAuditOutcome {
  rows: number;
  sent: number;
  failed: number;
  /** the audit_id the cursor stands at after this run */
  cursor: number;
  advanced: boolean;
  /** more listed rows wait past this run's batch */
  more: boolean;
}

export async function peerAuditDigest(
  db: DbLike,
  env: Record<string, unknown>,
  mailer: { send(template: string, to: string, vars: Record<string, string>): Promise<boolean> },
  now: () => string = () => new Date().toISOString(),
): Promise<PeerAuditOutcome> {
  const cur = await db.prepare("SELECT cursor_seq FROM sweep_cursor WHERE name = ?").bind(PEER_AUDIT_CURSOR).first<{ cursor_seq: number }>();
  const from = Number(cur?.cursor_seq ?? 0);
  // ONE bound parameter for the action set (json_each, as agencies.ts / alerts.ts do), never a placeholder per member.
  const { results: rows } = await db
    .prepare("SELECT audit_id, actor_id, action, item_id, detail, created_at FROM audit_log WHERE audit_id > ? AND action IN (SELECT value FROM json_each(?)) ORDER BY audit_id ASC LIMIT ?")
    .bind(from, JSON.stringify(PEER_AUDIT_ACTIONS), PEER_AUDIT_BATCH + 1)
    .all<PeerAuditRow>();
  const more = rows.length > PEER_AUDIT_BATCH;
  const batch = rows.slice(0, PEER_AUDIT_BATCH);
  const out: PeerAuditOutcome = { rows: batch.length, sent: 0, failed: 0, cursor: from, advanced: false, more };
  if (batch.length === 0) return out;
  const raw = typeof env.OPERATOR_EMAILS === "string" ? env.OPERATOR_EMAILS : "";
  const recipients = [...new Set(raw.split(",").map((e) => e.trim().toLowerCase()).filter((e) => e.length > 0 && !/todo/i.test(e)))];
  if (recipients.length === 0) return out; // nobody to tell: hold the cursor, never mark rows as told
  const personOfEmail = async (email: string): Promise<string> => {
    const canon = canonicalizeEmail(email);
    const row = await db
      .prepare("SELECT op_id FROM operator_identities WHERE email = ? OR canonical_email = ? ORDER BY op_id LIMIT 1")
      .bind(email, canon)
      .first<{ op_id: string }>();
    return row ? personOf(db, row.op_id) : `email:${canon}`;
  };
  const actorPerson = new Map<string, string>();
  for (const r of batch) {
    if (actorPerson.has(r.actor_id)) continue;
    actorPerson.set(r.actor_id, r.actor_id.startsWith("op:") ? await personOfEmail(r.actor_id.slice(3).toLowerCase()) : await personOf(db, r.actor_id));
  }
  const lineOf = new Map<number, string>();
  for (const r of batch) lineOf.set(r.audit_id, await peerAuditLine(db, r));
  for (const to of recipients) {
    const me = await personOfEmail(to);
    const theirs = batch.filter((r) => actorPerson.get(r.actor_id) !== me);
    if (theirs.length === 0) continue;
    const lines = theirs.map((r) => lineOf.get(r.audit_id)!);
    let ok = false;
    try {
      ok = await mailer.send(PEER_AUDIT_TEMPLATE, to, { count: String(theirs.length), lines: lines.join("\n") });
    } catch {
      ok = false;
    }
    if (ok) out.sent += 1;
    else out.failed += 1;
  }
  if (out.failed > 0) return out;
  const last = batch[batch.length - 1]!.audit_id;
  await db
    .prepare(
      "INSERT INTO sweep_cursor (name, cursor_seq, cursor_id, updated_at) VALUES (?, ?, '', ?) " +
        "ON CONFLICT(name) DO UPDATE SET cursor_seq = excluded.cursor_seq, updated_at = excluded.updated_at WHERE excluded.cursor_seq > sweep_cursor.cursor_seq",
    )
    .bind(PEER_AUDIT_CURSOR, last, now())
    .run();
  return { ...out, cursor: last, advanced: true };
}

/**
 * I65-HOLD F2 (I64 teachback U1; W0 am.7 as D-1571 reads PLAN ruling 5): a sealed range is kept while one of its rows is
 * an OWED marker that its own sweep arm (requests.ts sweepSubjectEraseOwed and its arms) would still take — the arm's own
 * select, minus the hold and the LIMIT: the OWED action with no DONE of its pair at or after it, plus, for the subject and
 * the granted-removal arms, the standing suppression it was written at or after (and, for the subject arm, a name left to
 * erase). `creator:face_erasure_owed` and `system:refund_owed` have no DONE action and never pin a range; every face-data
 * OWED is answered by a DONE the arm writes even when the image side has nothing left (no new retention extension, D-1570).
 * `image:face_erase_owed` -> `image:face_erase_done` is listed for the I65-IMAGE contract (D-1569 (e)).
 */
function auditRangeOwedSql(fromExpr: string, toExpr: string): string {
  const pending = (owed: string, done: string): string =>
    `(a.action = '${owed}' AND NOT EXISTS (SELECT 1 FROM audit_log d WHERE d.item_id = a.item_id AND d.action = '${done}' AND d.created_at >= a.created_at))`;
  const suppressed = "EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = a.item_id AND a.created_at >= s.created_at)";
  const namesLeft = "(EXISTS (SELECT 1 FROM passport_names n WHERE n.passport_id = a.item_id) OR EXISTS (SELECT 1 FROM passports p WHERE p.passport_id = a.item_id AND p.display_name <> ''))";
  const arms = [
    `(${pending("request:subject_erase_owed", "request:subject_erase_done")} AND ${suppressed} AND ${namesLeft})`,
    `(${pending("retraction:names_erase_owed", "retraction:names_erase_done")} AND EXISTS (SELECT 1 FROM passports p WHERE p.passport_id = a.item_id AND p.retracted = 1))`,
    pending("rename:names_erase_owed", "rename:names_erase_done"),
    `(${pending("removal:names_erase_owed", "removal:names_erase_done")} AND ${suppressed})`,
    pending("face:revoke_owed", "face:revoke_done"),
    pending("image:face_erase_owed", "image:face_erase_done"),
  ];
  return `EXISTS (SELECT 1 FROM audit_log a WHERE a.audit_id BETWEEN ${fromExpr} AND ${toExpr} AND (${arms.join(" OR ")}))`;
}
