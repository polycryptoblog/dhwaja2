import { Hono, type Context } from "hono";
import { bodyLimit } from "hono/body-limit";
import { d1 } from "./db.js";
import { isVerifiedPassport } from "./intelligence.js";
import { deleteMediaAsset, edgePurgerFromEnv, MediaError, mediaKey, mediaServeState, mediaThumbKey, MEDIA_THUMB_TYPE, storeSignedMedia, thumbMakerFromEnv } from "./media.js";
import { mediaScreenerFromEnv } from "./media-screen.js";
import { creatorId, mediaStore, NOT_FOUND_CACHE_60 } from "./route-helpers.js";

/**
 * I3-0764 (D-0833(4)) — the fallback for a throw no typed arm above it claimed:
 * a storage fault, a bug, an outage is ours — logged, answered 500, never a 400
 * that tells the caller their request was the problem.
 */
function serverFault(c: Context<{ Bindings: Env }>, route: string, e: unknown): Response {
  logError({ cid: crypto.randomUUID(), hop: "http", route, msg: "unexpected route fault", err: e instanceof Error ? e.name : typeof e });
  return c.json({ error: "internal" }, 500);
}


/**
 * S91 MEDIA STORAGE — Dhwaja stores creator-SIGNED collectible art, and only
 * that.
 *
 * BOUNDARY. The one creator upload (`POST /account/media`, capped at the runtime
 * layer), the immutable content-addressed read (`GET /media/:hash`) and the
 * creator's own delete. Bytes are accepted ONLY when their sha256 matches a
 * fingerprint the creator's passkey signed into a PUBLISHED release — there is
 * no open upload — and an unprovisioned bucket fails the whole surface closed.
 *
 * The OPERATOR takedown path is not here: it is an /ops route in console.ts,
 * behind Access and dual control.
 */

// Media upload is the one route that takes raw image bytes. Cap the read at the
// runtime layer (not just after buffering) so a chunked/omitted-Content-Length body
// can't stream past the 8 MiB asset cap and burn isolate memory.
/** Q143-D24: a pending answer must not be cached — it turns into a 200 the moment the screen passes. */
const PENDING_NO_STORE = { "Cache-Control": "no-store" };

const mediaBodyLimit = bodyLimit({
  maxSize: 8 * 1024 * 1024,
  onError: (c) => c.json({ error: "too_large" }, 413),
});

export function makeMediaRoutes(): Hono<{ Bindings: Env }> {
  const app = new Hono<{ Bindings: Env }>();


  // --- S91 media storage: Dhwaja stores creator-SIGNED collectible art ------------
  // Bytes are stored ONLY when their sha256 matches a fingerprint the creator's
  // passkey signed into their own PUBLISHED release (no open uploads). Content-
  // addressed; served immutably; png/jpeg only (magic-byte validated). Fail-closed
  // on an unprovisioned MEDIA_BUCKET.

  app.post("/account/media", mediaBodyLimit, async (c) => {
    const id = await creatorId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const store = mediaStore(c.env.MEDIA_BUCKET);
    if (!store) return c.json({ error: "media_unavailable" }, 503);
    const db = d1(c.env.TRUST_DB);
    if (!(await isVerifiedPassport(db, id))) return c.json({ error: "verified creators only" }, 403);
    const cl = Number(c.req.header("content-length"));
    if (Number.isFinite(cl) && cl > 8 * 1024 * 1024) return c.json({ error: "too_large" }, 413);
    const bytes = new Uint8Array(await c.req.arrayBuffer());
    if (bytes.length > 8 * 1024 * 1024) return c.json({ error: "too_large" }, 413);
    try {
      // Q143-D24 / RC-I3-0802: the screen, the purger a screen match's takedown needs, and the variant maker —
      // each null when unconfigured (no screen = the row stays pending and is not served).
      const env = c.env as unknown as Record<string, unknown>;
      const r = await storeSignedMedia(db, store, id, bytes, undefined, { screener: mediaScreenerFromEnv(env), purge: edgePurgerFromEnv(env), thumbs: thumbMakerFromEnv(env) });
      return c.json({ ok: true, ...r });
    } catch (e) {
      if (e instanceof MediaError) return c.json({ error: e.message }, e.status);
      return serverFault(c, "/account/media", e);
    }
  });

  // Public serve — content-addressed, immutable, the validated content-type only.
  app.get("/media/:hash", async (c) => {
    // I3-0211 — ONE CACHE KEY PER ASSET. Cloudflare's default cache key carries
    // the full query string, so `/media/<h>?utm=x` was its own year-long entry
    // holding these bytes, and the takedown purge (`media.ts` cloudflareEdgePurger,
    // `{"files":[<bare url>]}`) cannot enumerate those. Any `?` form is sent to
    // the bare URL instead and never carries the bytes; the bare URL is the one
    // the purge clears.
    const q = c.req.url.indexOf("?");
    if (q !== -1) return c.redirect(c.req.url.slice(0, q), 301);
    const store = mediaStore(c.env.MEDIA_BUCKET);
    if (!store) return c.json({ error: "media_unavailable" }, 503);
    // Q143-D24: served only once screened clean (media.ts isServableVerdict); pending is a 404 nobody caches.
    const st = await mediaServeState(d1(c.env.TRUST_DB), c.req.param("hash"));
    if (st.state === "pending") return c.json({ error: "pending" }, 404, PENDING_NO_STORE);
    if (st.state !== "ok") return c.json({ error: "not_found" }, 404, NOT_FOUND_CACHE_60);
    const asset = st.asset;
    const obj = await store.getBytes(mediaKey(asset.content_hash));
    if (!obj) return c.json({ error: "not_found" }, 404, NOT_FOUND_CACHE_60);
    return serveMediaBytes(obj.bytes, asset.content_type, asset.content_hash);
  });

  // RC-I3-0802: the small variant for grids — the same gate, the same headers, the same purge (cloudflareEdgePurger
  // purges this URL with the original's). No variant (no IMAGES binding, or one that failed) => the original bytes.
  app.get("/media/:hash/thumb", async (c) => {
    const q = c.req.url.indexOf("?");
    if (q !== -1) return c.redirect(c.req.url.slice(0, q), 301);
    const store = mediaStore(c.env.MEDIA_BUCKET);
    if (!store) return c.json({ error: "media_unavailable" }, 503);
    const st = await mediaServeState(d1(c.env.TRUST_DB), c.req.param("hash"));
    if (st.state === "pending") return c.json({ error: "pending" }, 404, PENDING_NO_STORE);
    if (st.state !== "ok") return c.json({ error: "not_found" }, 404, NOT_FOUND_CACHE_60);
    const variant = st.thumb ? await store.getBytes(mediaThumbKey(st.asset.content_hash)) : null;
    if (variant) return serveMediaBytes(variant.bytes, MEDIA_THUMB_TYPE, st.asset.content_hash);
    const obj = await store.getBytes(mediaKey(st.asset.content_hash));
    if (!obj) return c.json({ error: "not_found" }, 404, NOT_FOUND_CACHE_60);
    return serveMediaBytes(obj.bytes, st.asset.content_type, st.asset.content_hash);
  });

  /** The one image response both media GETs return. */
  function serveMediaBytes(bytes: Uint8Array, contentType: string, contentHash: string): Response {
    // The stored, magic-byte-validated type — NEVER a client-supplied header. nosniff +
    // a locked-down CSP + sandbox so a served image can't be coerced into script.
    //
    // NO `frame-ancestors` HERE, AND `sandbox` IS NOT THE REASON. `sandbox`
    // constrains what the sandboxed document may DO once loaded (opaque origin, no
    // scripts, no forms, no top-level navigation); it says nothing about WHO may
    // embed it. Only `frame-ancestors` (or X-Frame-Options) decides that, and this
    // response sends neither, so this URL IS framable. That is fine, on two
    // grounds that do hold:
    //   - There is nothing to clickjack. The body is an image, not an interactive
    //     document: no controls to overlay, and no authenticated state-changing
    //     action a click on it can reach.
    //   - Framing grants nothing new. `frame-ancestors` never restricted <img>
    //     either, and any page can already `<img src>` this content-addressed URL
    //     and get the identical load/error oracle ("does this hash exist").
    // The server-rendered HTML pages are the opposite case and DO send it (SSR_CSP
    // above) — those have UI to overlay and a verdict to dress up.
    //
    // CSAM SCANNABILITY (load-bearing — see docs/CSAM-SCANNING.md): this response
    // MUST stay PUBLICLY CACHEABLE. Cloudflare's CSAM Scanning Tool scans media at
    // CACHE-FILL time; a `private`/`no-store`/cache-bypass response, or serving via
    // an r2.dev subdomain, silently EXCLUDES the object from scanning. Keep it
    // `public … immutable` and served through the proxied custom domain. The 8 MiB
    // upload cap keeps every object well under the cache-eligibility ceiling.
    return new Response(bytes, {
      status: 200,
      headers: {
        "Content-Type": contentType,
        "X-Content-Type-Options": "nosniff",
        "Content-Security-Policy": "default-src 'none'; sandbox",
        "Content-Disposition": "inline",
        "Cache-Control": "public, max-age=31536000, immutable",
        // THE PURGE HANDLE FOR THE YEAR-LONG CACHE ABOVE. The cache is what gets
        // the object CSAM-scanned; it is also what keeps serving a DMCA/NCII
        // takedown for a year unless something purges it. `deleteMediaAsset`
        // purges by URL — the mechanism that works on EVERY Cloudflare plan —
        // and this header is the second lever, not the first:
        //
        //   PURGE BY `Cache-Tag` IS ENTERPRISE-ONLY. On any lower plan a
        //   tag purge is refused by the API. Nothing in the takedown path may
        //   ever depend on it; treat it as a bulk instrument for an Enterprise
        //   deployment ("purge every stored asset", "purge this one hash from a
        //   dashboard") and as a fallback if a by-URL purge is refused.
        //
        // Two tags: `media` for the whole surface, `media-<hash>` for one asset.
        // Both are derived from the 64-hex content hash, which is validated by
        // `getMediaAsset` before it reaches here, so nothing caller-controlled
        // can shape this header.
        "Cache-Tag": `media,media-${contentHash}`,
      },
    });
  }

  // Creator self-deletion (GDPR / mistake). Ownership-gated inside deleteMediaAsset.
  //
  // THE EDGE PURGER IS PASSED HERE TOO, and a creator delete is not the lesser
  // case. Their bytes were served with the same one-year immutable cache as
  // everything else, so a self-delete that stops at R2 leaves the image being
  // served worldwide while the creator has been told it is gone — which is a
  // GDPR erasure that did not erase. Absent config ⇒ `edgePurgerFromEnv` is null
  // ⇒ `deleteMediaAsset` refuses with 503 without destroying anything, so the
  // creator can retry once the deployment can actually purge.
  //
  // `c.env` is cast because these two keys are read through
  // `Record<string, unknown>` predicates in env-config.ts (the same shape
  // `configHealth` and every other *FromEnv reader takes) rather than being
  // added to the `Env` type.
  app.delete("/account/media/:hash", async (c) => {
    const id = await creatorId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const store = mediaStore(c.env.MEDIA_BUCKET);
    if (!store) return c.json({ error: "media_unavailable" }, 503);
    try {
      const purge = edgePurgerFromEnv(c.env as unknown as Record<string, unknown>);
      await deleteMediaAsset(d1(c.env.TRUST_DB), store, c.req.param("hash"), id, undefined, purge);
      return c.json({ ok: true });
    } catch (e) {
      if (e instanceof MediaError) return c.json({ error: e.message }, e.status);
      return serverFault(c, "/account/media/:hash", e);
    }
  });

  return app;
}

import { evidenceScanConfigFromEnv, httpEvidenceScanner } from "./evidence-scan.js";

/**
 * I30-SAFETY Q143-D24 — THE SCREEN A STORED IMAGE MUST PASS BEFORE `GET /media/:hash` SERVES IT
 * (D-1030: "Require screening before media is served; don't accept the first-serve gap").
 *
 * `MEDIA_SCREEN_PROVIDER` names the provider. Unset, empty or unknown -> `mediaScreenerFromEnv` returns null and
 * every newly stored image stays pending (media.ts `storeSignedMedia`), so nothing new is served: fail closed.
 * Jason names the provider; the one wired today is `http-shim`, the hash-matching/classifier HTTP contract the
 * report-evidence scan already speaks (evidence-scan.ts `httpEvidenceScanner`: POST the bytes, read one boolean
 * `match`), configured by MEDIA_SCREEN_URL / MEDIA_SCREEN_TOKEN / MEDIA_SCREEN_TIMEOUT_MS.
 */

/** clean: serve it. flagged: the provider matched it. error: no answer (it stays pending and is retried). */
export type MediaScreenOutcome = { verdict: "clean" } | { verdict: "flagged"; category: string | null } | { verdict: "error"; reason: string };

export interface MediaScreener {
  readonly provider: string;
  screen(bytes: Uint8Array, contentType: string): Promise<MediaScreenOutcome>;
}

/** The provider values `MEDIA_SCREEN_PROVIDER` accepts. Anything else is unconfigured. */
export const MEDIA_SCREEN_PROVIDERS = ["http-shim"] as const;

export function mediaScreenerFromEnv(env: Record<string, unknown>, fetchImpl: typeof fetch = fetch): MediaScreener | null {
  const provider = typeof env.MEDIA_SCREEN_PROVIDER === "string" ? env.MEDIA_SCREEN_PROVIDER.trim() : "";
  if (provider !== "http-shim") return null;
  const cfg = evidenceScanConfigFromEnv({
    EVIDENCE_SCAN_URL: env.MEDIA_SCREEN_URL,
    EVIDENCE_SCAN_TOKEN: env.MEDIA_SCREEN_TOKEN,
    EVIDENCE_SCAN_TIMEOUT_MS: env.MEDIA_SCREEN_TIMEOUT_MS,
  });
  if (!cfg) return null;
  const http = httpEvidenceScanner(cfg, fetchImpl);
  return {
    provider,
    async screen(bytes, contentType) {
      const r = await http.scan(bytes, contentType);
      if (r.verdict === "clear") return { verdict: "clean" };
      if (r.verdict === "match") return { verdict: "flagged", category: r.category };
      return { verdict: "error", reason: r.reason };
    },
  };
}

/**
 * S91 — media storage. Dhwaja stores the creator-SIGNED collectible art it is given
 * (and, later, the AI art it produces).
 *
 * ATTRIBUTION GUARD, NOT A CONTENT CONTROL (I3-1057): bytes are stored ONLY when
 * their sha256 matches a `media_fingerprints.exact_hash` the creator's passkey
 * already signed into their own PUBLISHED release (step 3 of `storeSignedMedia`)
 * — never an open upload. That proves who put the bytes here; it says nothing
 * about what the image shows. The content-safety control is SCREEN BEFORE SERVE
 * (Q143-D24): a new row is stored pending and `getMediaAsset` serves nothing the
 * screen named by MEDIA_SCREEN_PROVIDER has not passed (media-screen.ts; unset =
 * nothing new served); Cloudflare's CSAM Scanning Tool on the proxied
 * `GET /media/:hash` path is the second layer (docs/CSAM-SCANNING.md), and the
 * operator takedown below (`deleteMediaAsset`) is the removal path for both. Content-addressed
 * (R2 key `media/<hash>`); the stored content-type is read from the bytes' magic
 * bytes, never from a client-supplied header.
 *
 * The heavy decode never happens here: `assertWithinCaps` reads only the image
 * HEADER (fail-closed to png/jpeg) to enforce the byte + declared-pixel caps before
 * anything decodes, defeating a decode bomb.
 */

export class MediaError extends Error {
  override name = "MediaError";
  constructor(
    message: string,
    public readonly status: 400 | 401 | 403 | 404 | 409 | 413 | 415 | 502 | 503 = 400,
  ) {
    super(message);
  }
}

/**
 * THE EDGE-CACHE HALF OF A TAKEDOWN.
 *
 * `GET /media/:hash` answers `Cache-Control: public, max-age=31536000,
 * immutable`, and that is load-bearing rather than lazy: Cloudflare's CSAM
 * Scanning Tool scans at CACHE-FILL, so a private/no-store response would
 * silently exclude every stored object from scanning (docs/CSAM-SCANNING.md §2).
 *
 * The bill for that decision falls due here. Deleting the R2 object and the row
 * removes the ORIGIN copy; it removes nothing that a fan is actually served. Any
 * PoP that already filled keeps the identical bytes for up to a YEAR, and
 * `immutable` tells every browser that has them never to revalidate. Until this
 * interface existed, `deleteMediaAsset` stopped after the R2 delete and
 * `DELETE /ops/media/:hash` answered `{ok:true, removed:true}` over material
 * still being served worldwide — under a commitment that reads "We remove or
 * disable the material" (docs/legal/DMCA-AND-TAKEDOWNS.md).
 *
 * This is exactly the case Cloudflare's own block does NOT cover. A DMCA or NCII
 * report with no hash-list match gets no edge block at all; the console already
 * says so in its own words — "Cloudflare's edge block is not the takedown".
 */
export interface EdgePurger {
  /** The absolute URL the purge targets. Named in the failure so an operator
   * holding a partial takedown knows precisely what to purge by hand. */
  mediaUrl(hash: string): string;
  /** Purge that URL from the edge. NEVER THROWS — every failure comes back as
   * `ok:false` with a detail, so the takedown decides what a failure means
   * rather than an exception deciding it by escaping. */
  purgeMedia(hash: string): Promise<PurgeOutcome>;
}

export type PurgeOutcome = { ok: true; url: string } | { ok: false; url: string; detail: string };

/**
 * A takedown whose ORIGIN half completed and whose EDGE half did not. A subclass
 * of `MediaError` so every existing call site keeps mapping it to a status
 * without knowing it exists — and 502 rather than 503 because the distinction is
 * the operator's next action: 503 is "this deployment cannot purge, nothing was
 * touched, fix the config", 502 is "the removal happened, the bytes are still on
 * the edge, purge them by hand NOW".
 */
export class MediaEdgeNotPurged extends MediaError {
  override name = "MediaEdgeNotPurged";
  constructor(
    public readonly url: string,
    public readonly detail: string,
  ) {
    super(`removed from storage but the EDGE CACHE WAS NOT PURGED (${detail}) — cached copies are still being served from ${url} and ${url}/thumb; purge those URLs now`, 502);
  }
}

/** Deadline for one purge attempt. An unbounded outbound call inside a takedown
 * is its own defect: it holds the invocation open and the operator gets a
 * hanging console instead of an answer. Same shape as `push.ts`'s
 * `fetchWithDeadline` — the AbortController is what makes the deadline real,
 * because without it the request keeps running after we stop waiting. */
export const MEDIA_PURGE_TIMEOUT_MS = 8_000;

/**
 * Two attempts, not one and not five. `mailer.ts` and `push.ts` deliberately do
 * not retry, because a retried notification is amplification and the caller
 * retries anyway. NEITHER IS TRUE HERE: a takedown has exactly one shot — its
 * row is already gone, so a retry of the WHOLE takedown answers 404 and reads to
 * an operator as a mistyped hash. One retry buys a transient TCP reset without
 * turning a takedown into a hammer.
 *
 * A 4xx IS NEVER RETRIED (see `purgeOnce`): an expired token or a zone the token
 * has no `Cache Purge` permission on will refuse identically forever, and
 * retrying it only delays the operator being told to escalate.
 */
export const MEDIA_PURGE_ATTEMPTS = 2;

const CF_API_BASE = "https://api.cloudflare.com/client/v4/zones";

/**
 * The Cloudflare zone purge-by-URL adapter.
 *
 * PURGE BY URL IS THE PRIMARY AND MUST STAY SO: `{"files":[…]}` works on EVERY
 * Cloudflare plan. Purge by `Cache-Tag` — which `media-routes.ts` emits — is
 * ENTERPRISE-ONLY, so it is a bulk lever for deployments that have it and never
 * something a takedown may depend on.
 *
 * `fetchImpl` is injected (the `push.ts` / `mailer.ts` convention) so the call
 * itself is testable: the URL, the auth header shape, the body, a non-2xx, a
 * `success:false` 200, a transport error and the deadline.
 */
export function cloudflareEdgePurger(cfg: EdgePurgeConfig, fetchImpl: typeof fetch = fetch): EdgePurger {
  const mediaUrl = (hash: string): string => `${cfg.mediaOrigin}/media/${hash}`;
  /** One bounded attempt. Returns `null` for "failed, and a retry might help",
   * or a detail string for "failed permanently — do not retry". */
  const purgeOnce = async (url: string): Promise<{ done: true } | { done: false; detail: string; retry: boolean }> => {
    const ctl = new AbortController();
    const timer = setTimeout(() => ctl.abort(), MEDIA_PURGE_TIMEOUT_MS);
    try {
      const res = await fetchImpl(`${CF_API_BASE}/${cfg.zoneId}/purge_cache`, {
        method: "POST",
        headers: { authorization: `Bearer ${cfg.apiToken}`, "content-type": "application/json", accept: "application/json" },
        // RC-I3-0802 (I30-SAFETY): the original AND its /thumb URL (media-routes.ts), so every purge door clears both.
        body: JSON.stringify({ files: [url, `${url}/thumb`] }),
        redirect: "manual", // never chase a redirect while holding a zone-purge token
        signal: ctl.signal,
      });
      // manual mode surfaces a 3xx as a response instead of throwing; it
      // must be checked BEFORE !res.ok and treated as the same failure the
      // transport-error catch below already reaches.
      if (res.status >= 300 && res.status < 400) {
        return { done: false, detail: "unreachable", retry: true };
      }
      if (!res.ok) {
        // 4xx is a configuration answer and will not change on a retry; 5xx and
        // 429 are the transient ones. The body is NOT read into the detail: it
        // can echo request data and it reaches an operator screen.
        return { done: false, detail: `http ${res.status}`, retry: res.status >= 500 || res.status === 429 };
      }
      // A 200 IS NOT AN ANSWER. The Cloudflare API returns 200 with
      // `{"success":false,"errors":[…]}` for a rejected purge, so trusting the
      // status line here would report a takedown as complete over a purge that
      // never happened — the same class of lie this whole change removes.
      let ok = false;
      try {
        ok = ((await res.json()) as { success?: unknown }).success === true;
      } catch {
        ok = false;
      }
      return ok ? { done: true } : { done: false, detail: "api reported success:false", retry: false };
    } catch (e) {
      // Timeout and transport failure are the SAME outcome: not purged. The
      // error text is not surfaced — it can carry the request URL and this
      // string is rendered to an operator.
      return { done: false, detail: (e as Error)?.name === "AbortError" ? "timed out" : "unreachable", retry: true };
    } finally {
      clearTimeout(timer);
    }
  };
  return {
    mediaUrl,
    async purgeMedia(hash: string): Promise<PurgeOutcome> {
      const url = mediaUrl(hash);
      let last = "no attempt made";
      for (let attempt = 0; attempt < MEDIA_PURGE_ATTEMPTS; attempt++) {
        const r = await purgeOnce(url);
        if (r.done) return { ok: true, url };
        last = r.detail;
        if (!r.retry) break;
      }
      return { ok: false, url, detail: last };
    },
  };
}

/**
 * Build the purger from env, or null when the deployment cannot purge. Null is
 * the fail-closed signal `deleteMediaAsset` refuses on — it is never a "skip the
 * purge" flag, and the takedown treats an ABSENT argument identically, so a call
 * site that has not been wired for purge cannot quietly inherit the old
 * behaviour. `pushSenderFromEnv` is the convention this follows.
 */
export function edgePurgerFromEnv(env: Record<string, unknown>, fetchImpl: typeof fetch = fetch): EdgePurger | null {
  const cfg = edgePurgeConfigFromEnv(env);
  return cfg ? cloudflareEdgePurger(cfg, fetchImpl) : null;
}

/** A bytes-capable object store — the sibling of the string-only `ObjectStore`.
 * `list` (I3-1701) is what lets `sweepMediaOrphans` find bytes no row references;
 * it is optional in the type only so a store without it still stores and serves —
 * `sweepMediaOrphans` refuses (503) on a store that lacks it, never skips silently. */
export interface MediaStore {
  putBytes(key: string, bytes: Uint8Array, contentType: string): Promise<void>;
  getBytes(key: string): Promise<{ bytes: Uint8Array; contentType: string } | null>;
  delete(key: string): Promise<void>;
  list?(prefix: string, cursor: string | null): Promise<MediaListPage>;
}

const SHA256_HEX = /^[0-9a-f]{64}$/;
/** I3-1442 (D-0920 option A): the ONE text of the cross-creator duplicate 409, at both doors (the pre-read and the
 * race loser). Naming "another creator" told the uploader that someone else holds these exact bytes. */
export const MEDIA_DUPLICATE_REFUSAL = "this file can't be stored";
const MAX_MEDIA_BYTES = 8 * 1024 * 1024; // 8 MiB per asset
const MAX_MEDIA_PIXELS = 40_000_000; // 40 MP declared-pixel cap (decode-bomb guard)
/** Per-creator volume cap. Exported so the concurrency guard can hold the cap
 * to a race rather than restating the number. */
export const MAX_ASSETS_PER_CREATOR = 500; // volume cap — Dhwaja is not a free CDN

function hex(b: Uint8Array): string {
  let s = "";
  for (const x of b) s += x.toString(16).padStart(2, "0");
  return s;
}

/** R2 object key for a content-addressed media asset. */
export function mediaKey(hash: string): string {
  return `media/${hash}`;
}

/** RC-I3-0802 (I30-SAFETY): R2 key of the one small variant of a content-addressed asset, derived from its hash. */
export function mediaThumbKey(hash: string): string {
  return `media/${hash}/thumb`;
}

/** RC-I3-0802: the variant's width (px), its format, and the largest variant kept (a bigger one is not stored and
 * `GET /media/:hash/thumb` serves the original instead). */
export const MEDIA_THUMB_WIDTH = 320;
export const MEDIA_THUMB_TYPE = "image/webp";
export const MEDIA_THUMB_MAX_BYTES = 100 * 1024;

/** RC-I3-0802: makes the variant. Absent (no IMAGES binding) means no variant, which is today's behaviour. */
export interface ThumbMaker {
  make(bytes: Uint8Array, contentType: string): Promise<Uint8Array>;
}

/** The subset of the Cloudflare Images binding (`env.IMAGES`) the variant uses — no npm dependency. */
export interface ImagesBindingLike {
  input(stream: ReadableStream<Uint8Array>): {
    transform(t: { width: number }): { output(o: { format: string }): Promise<{ response(): Response }> };
  };
}

export function imagesThumbMaker(images: ImagesBindingLike): ThumbMaker {
  return {
    async make(bytes) {
      // D-1061(6): typed for the DOM lib (apps/creator-console compiles this file; the line is I30 6e9cbe8's).
      const out = await images.input(new Blob([bytes as Uint8Array<ArrayBuffer>]).stream()).transform({ width: MEDIA_THUMB_WIDTH }).output({ format: MEDIA_THUMB_TYPE });
      return new Uint8Array(await out.response().arrayBuffer());
    },
  };
}

/** The `IMAGES` binding as a ThumbMaker, or null when the deployment has none. */
export function thumbMakerFromEnv(env: Record<string, unknown>): ThumbMaker | null {
  const images = env.IMAGES as ImagesBindingLike | undefined;
  return images && typeof images.input === "function" ? imagesThumbMaker(images) : null;
}

/** Q143-D24: the `media_takedowns.reason` a screen match is taken down under (the existing takedown path). */
export const MEDIA_SCREEN_TAKEDOWN_REASON = "screen match";

/** What `storeSignedMedia` needs beyond the bytes. Every field absent = unconfigured: no screen (the row stays
 * pending, never served), no purger (a flagged row's takedown refuses and the row stays unservable until
 * `screenPendingMedia` finishes it), no variant. */
export interface StoreMediaOptions {
  screener?: MediaScreener | null;
  purge?: EdgePurger | null;
  thumbs?: ThumbMaker | null;
}

/** Q143-D24: the ONE serve predicate. NULL (pending) and 'flagged' never serve. */
export function isServableVerdict(v: string | null | undefined): boolean {
  return v === "clean" || v === "legacy";
}

/** The screen state a stored row reports to its uploader; a flagged row is refused as removed. */
function uploaderScreenState(v: string | null): "clean" | "pending" {
  if (v === "flagged") throw new MediaError("this image was removed and cannot be re-stored", 403);
  return isServableVerdict(v) ? "clean" : "pending";
}

export interface StoredMedia {
  content_hash: string;
  content_type: string;
  width: number | null;
  height: number | null;
  /** Q143-D24: "pending" until the configured screen passes it; only "clean" is served. */
  screen: "clean" | "pending";
}

/**
 * Store bytes for an ALREADY-SIGNED release fingerprint owned by this creator.
 * CALLER-GATING: the HTTP layer passes the authenticated creator's own verified
 * passport. Guards (all fail-closed): image caps + png/jpeg-only (magic bytes);
 * sha256(bytes) must equal a `media_fingerprints.exact_hash` on a PUBLISHED release
 * of this creator; per-creator volume cap. Idempotent (content-addressed).
 */
export async function storeSignedMedia(db: DbLike, store: MediaStore, passportId: string, bytes: Uint8Array, now: () => string = () => new Date().toISOString(), opts: StoreMediaOptions = {}): Promise<StoredMedia> {
  // 1. Raw byte cap (413), then fail-closed format detection + declared-pixel cap.
  //    The returned header's format IS the trusted content-type (never the client's);
  //    an unrecognized/over-pixel image is refused as unsupported (415).
  if (bytes.length > MAX_MEDIA_BYTES) throw new MediaError("image too large", 413);
  let header: ReturnType<typeof assertWithinCaps>;
  try {
    header = assertWithinCaps(bytes, { maxBytes: MAX_MEDIA_BYTES, maxPixels: MAX_MEDIA_PIXELS });
  } catch (e) {
    if (e instanceof OversizeImage) throw new MediaError("unsupported or oversized image (png/jpeg only)", 415);
    throw new MediaError("unreadable image", 400);
  }
  const contentType = header.format === "png" ? "image/png" : "image/jpeg";

  // 2. Hash the bytes with THE canonical hasher.
  const contentHash = hex(await sha256Bytes(bytes));

  // 2b. Operator takedown is durable: a purged hash is banned forever, regardless of
  //     what release still signs it (else a takedown is cosmetic — the creator just
  //     re-POSTs the identical bytes and the positive guard re-admits them).
  //     THIS READ IS AN EARLY-OUT, NOT THE GUARD. It is five statements ahead of the
  //     write, and the creator controls when they POST: a takedown that landed in
  //     between used to be walked straight through, leaving the hash tombstoned AND
  //     the object back in R2 and servable. The guard that actually holds is on the
  //     INSERT at step 6, which tests the tombstone in the same statement.
  const banned = await db.prepare("SELECT 1 AS x FROM media_takedowns WHERE content_hash = ? LIMIT 1").bind(contentHash).first();
  if (banned) throw new MediaError("this image was removed and cannot be re-stored", 403);

  // 3. The ATTRIBUTION guard: the hash must match a signed fingerprint on a
  //    PUBLISHED release owned by this creator. No signed match → refuse — Dhwaja
  //    never stores an arbitrary upload. Content safety is NOT decided here: it is
  //    the screen at step 7 (see the file header and docs/CSAM-SCANNING.md).
  const signed = await db
    .prepare("SELECT 1 AS x FROM media_fingerprints f JOIN releases r ON r.release_id = f.release_id WHERE f.exact_hash = ? AND r.passport_id = ? AND r.status = 'published' LIMIT 1")
    .bind(contentHash, passportId)
    .first();
  if (!signed) throw new MediaError("no published release of yours signed this image", 403);

  // 4. Idempotent: already stored → return it (content-addressed; identical bytes).
  //
  // ME-05 (S-068). IDEMPOTENT FOR THE OWNER, A NAMED CONFLICT FOR ANYONE ELSE.
  //
  // `media_assets` is keyed on the content hash and carries exactly ONE
  // `passport_id` (0007_releases_credentials.sql). Two creators can each
  // legitimately sign the SAME licensed image into their own published release
  // — the positive guard at step 3 admits both — and this early-out then told
  // the second one "stored" while creating no association for them at all.
  // Every consumer is owner-scoped: `releases.ts:681` and `feed.ts:185` both
  // read `WHERE passport_id = ?`, so their archive/feed art is simply absent,
  // and `deleteMediaAsset` checks that same single owner, so they have no
  // management authority over bytes they were told were stored. The first
  // owner's self-delete can also remove the backing content out from under
  // them, because there is no reference model to stop it.
  //
  // The finder's two options were "separate shared content from per-creator
  // associations" (a schema change, a reference count and a defined removal
  // rule — real design) and "reject the conflict explicitly until sharing is
  // supported". This takes the second, which is the one that can be taken
  // safely today: 409, naming the situation, rather than a silent confirmation
  // of an association that was never created. It is strictly a new refusal on
  // a path that previously reported a success it had not delivered, and it is
  // reversible the day the reference model lands.
  //
  // ME-06 is the other half of the same read. A row whose object is gone —
  // an older self-delete finishing after a later accepted re-upload published
  // — used to answer "stored" from the metadata alone, forever, while
  // `GET /media/:hash` 404'd and the only remedy the product offers (upload it
  // again) was the very call that kept reporting success. These are the same
  // bytes, already positively guarded above and already tombstone-checked, so
  // re-putting them repairs the row rather than creating anything new.
  const existing = await db.prepare("SELECT passport_id, content_type, width, height FROM media_assets WHERE content_hash = ?").bind(contentHash).first<{ passport_id: string; content_type: string; width: number | null; height: number | null }>();
  if (existing) {
    if (existing.passport_id !== passportId) {
      throw new MediaError(MEDIA_DUPLICATE_REFUSAL, 409);
    }
    // Q143-D24: a flagged row is refused BEFORE the ME-06 heal below could re-put its bytes.
    const verdictRow = await db.prepare("SELECT screen_verdict FROM media_assets WHERE content_hash = ?").bind(contentHash).first<{ screen_verdict: string | null }>();
    const screen = uploaderScreenState(verdictRow?.screen_verdict ?? null);
    // ME-06 heal. Bounded (one HEAD-equivalent read on an already-authorized
    // retry), and it CANNOT defeat a takedown: step 2b refused a tombstoned
    // hash before we got here, and `deleteMediaAsset` writes the tombstone
    // before it drops the object, so a takedown racing this re-put is swept by
    // the takedown's own final delete exactly as step 6 already documents.
    if ((await store.getBytes(mediaKey(contentHash))) === null) {
      await store.putBytes(mediaKey(contentHash), bytes, existing.content_type);
      // I3-0908: the argument above holds only if the takedown's final R2 delete
      // runs AFTER this put, and nothing orders the two. A takedown that finished
      // between the row read and this put (tombstone, row, object all done) left
      // the banned bytes back in the bucket with no row — outside every operator
      // lever. Re-test the tombstone AFTER the put, the same shape step 6 uses:
      // `deleteMediaAsset` writes the tombstone before its delete, so either its
      // delete follows this put or this read sees the tombstone.
      const takenDown = await db.prepare("SELECT 1 AS x FROM media_takedowns WHERE content_hash = ? LIMIT 1").bind(contentHash).first();
      if (takenDown) {
        await store.delete(mediaKey(contentHash));
        throw new MediaError("this image was removed and cannot be re-stored", 403);
      }
    }
    return { content_hash: contentHash, content_type: existing.content_type, width: existing.width, height: existing.height, screen };
  }

  // 5. Per-creator volume cap. AN EARLY-OUT, NOT THE GUARD — same as the takedown
  //    read at 2b, and for the same reason: counted-then-inserted, a cap is
  //    overshot by the width of a concurrent burst (the shape already fixed in
  //    `accounts.ts saveCreator` and `collections.ts collectCard`). The clause
  //    that actually holds is in the INSERT at step 6. This read stays because it
  //    refuses BEFORE the R2 put, so an over-cap creator does not write bytes we
  //    then have to delete.
  const count = await db.prepare("SELECT COUNT(*) AS n FROM media_assets WHERE passport_id = ?").bind(passportId).first<{ n: number }>();
  if ((count?.n ?? 0) >= MAX_ASSETS_PER_CREATOR) throw new MediaError("media storage limit reached", 409);

  // 6. Bytes first, then the row. A row without bytes would 404 on serve (bad); bytes
  //    without a row are an orphan that `sweepMediaOrphans` collects after 24 h —
  //    so write bytes first.
  //
  //    THE TOMBSTONE IS TESTED IN THE INSERT ITSELF, not by the read at 2b. R2 has
  //    no multi-object transaction, so the ordering and this one D1 statement are
  //    what carry an operator takedown: `deleteMediaAsset` writes the tombstone
  //    FIRST and drops the R2 object LAST, so any re-store that gets its row in has
  //    already put its bytes before the tombstone existed, and the takedown's final
  //    delete sweeps them. A re-store that loses sees the tombstone here, inserts
  //    nothing, and removes the bytes it just wrote. There is no interleaving that
  //    ends tombstoned-and-servable: serving requires this row (GET /media/:hash).
  await store.putBytes(mediaKey(contentHash), bytes, contentType);
  // ME-07 (S-068). THE THROWN INSERT HAD NO CLEANUP BRANCH AT ALL.
  //
  // Every undo below lives inside `if (ins.changes !== 1)` — the CONDITIONAL
  // REFUSAL path. A D1 exception or a worker interruption on the way to that
  // `await` never reaches any of them, so the object stays in R2 with no row:
  // outside `media_assets` enumeration, outside the per-creator volume cap,
  // and unreachable by the deletion helper, which starts from metadata.
  //
  // A try/catch is NOT general reconciliation and is not claimed to be — a
  // worker termination still strands the object. The durable half is
  // `sweepMediaOrphans` below (I3-1701), which lists `media/` and removes an
  // object older than 24 h that no row references. What this closes is the ordinary
  // case the finder names, a transient metadata failure after a successful
  // object write, which today retains creator bytes indefinitely and silently.
  // The put is re-derivable (content-addressed, identical bytes), so undoing
  // it loses nothing a retry cannot redo.
  // D-1079(2) (I32-VERIFY H3): ONE instant — the row's created_at and the standing re-check's grant-expiry instant.
  const at = now();
  let ins: { changes: number };
  try {
    ins = await db
    .prepare(
      `INSERT INTO media_assets (content_hash, passport_id, content_type, byte_size, width, height, kind, created_at, screen_verdict)
       SELECT ?, ?, ?, ?, ?, ?, 'signed_edition', ?, NULL
        WHERE NOT EXISTS (SELECT 1 FROM media_takedowns WHERE content_hash = ?)
          AND NOT EXISTS (SELECT 1 FROM media_assets WHERE content_hash = ?)
          AND (SELECT COUNT(*) FROM media_assets WHERE passport_id = ?) < ?
          AND ${VERIFIED_AT_SUBQUERY} IS NOT NULL`,
    )
    .bind(contentHash, passportId, contentType, bytes.length, header.width, header.height, at, contentHash, contentHash, passportId, MAX_ASSETS_PER_CREATOR, passportId, at)
    .run();
  } catch (e) {
    // Best-effort, and deliberately non-throwing: the metadata failure is the
    // fault the caller must hear about, and a storage fault on the way out
    // must not replace it with a different, less true one. If this delete is
    // the thing that fails, the object is exactly as orphaned as it was
    // before — no worse, and the caller still gets the real error.
    try {
      await store.delete(mediaKey(contentHash));
    } catch {
      /* the orphan survives; the metadata error below is still the truth */
    }
    throw e;
  }
  if (ins.changes !== 1) {
    // Either a takedown landed while we were working, or a concurrent store of the
    // identical bytes won (content-addressed, so its row is ours too). Distinguish,
    // because only one of them may keep the object.
    const tombstoned = await db.prepare("SELECT 1 AS x FROM media_takedowns WHERE content_hash = ? LIMIT 1").bind(contentHash).first();
    if (tombstoned) {
      // Undo our own put. Worst case (a crash before this line) leaves an orphan
      // object with no row — unservable, and the takedown's own delete already ran
      // or will run against the same key.
      await store.delete(mediaKey(contentHash));
      throw new MediaError("this image was removed and cannot be re-stored", 403);
    }
    const existingNow = await db.prepare("SELECT content_type, width, height, passport_id, screen_verdict FROM media_assets WHERE content_hash = ?").bind(contentHash).first<{ passport_id: string; content_type: string; width: number | null; height: number | null; screen_verdict: string | null }>();
    if (existingNow) {
      // I3-1271: the concurrent winner may be ANOTHER creator. Step 4's ME-05
      // refusal applies here exactly as it does to a row that was already there —
      // the loser was told "stored" for an association that was never created.
      // The object is the winner's (identical bytes, same key), so it stays.
      if (existingNow.passport_id !== passportId) {
        throw new MediaError(MEDIA_DUPLICATE_REFUSAL, 409);
      }
      return { content_hash: contentHash, content_type: existingNow.content_type, width: existingNow.width, height: existingNow.height, screen: uploaderScreenState(existingNow.screen_verdict) };
    }
    // …or the volume cap filled between the read at 5 and this write. Undo our
    // own put — the bytes are not referenced by any row, so leaving them would be
    // an orphan this creator's cap already refused to pay for.
    const nowCount = await db.prepare("SELECT COUNT(*) AS n FROM media_assets WHERE passport_id = ?").bind(passportId).first<{ n: number }>();
    if ((nowCount?.n ?? 0) >= MAX_ASSETS_PER_CREATOR) {
      await store.delete(mediaKey(contentHash));
      throw new MediaError("media storage limit reached", 409);
    }
    // …or STANDING went away between the route's read and this write (audit v11,
    // D5). This module has no standing read of its own — `media-routes.ts` asks
    // `isVerifiedPassport` five statements upstream — so before the predicate
    // above existed, a creator downgraded or suppressed inside that window stored
    // bytes and a row under standing they no longer had. Asked LAST, after the
    // three causes that are cheaper and more likely, and it is a read on the
    // refusal path only. The bytes come back out: an unreferenced object is an
    // orphan this creator was not admitted to write.
    //
    // It re-asks THE SAME PREDICATE the INSERT just used, not a second
    // formulation of it. `isVerifiedPassport` (intelligence.ts) is the reader
    // standing.ts names as defect #3 — it omits the suppression clause — and
    // standing.ts says in terms that new code must not add another call site of
    // it. Binding `VERIFIED_AT_SUBQUERY` at the INSERT's own instant means this refusal message can
    // never disagree with the refusal that produced it.
    const stillVerified = await db.prepare(`SELECT 1 AS x WHERE ${VERIFIED_AT_SUBQUERY} IS NOT NULL`).bind(passportId, at).first();
    if (!stillVerified) {
      await store.delete(mediaKey(contentHash));
      throw new MediaError("verified creators only", 403);
    }
    throw new MediaError("could not store this image", 409);
  }
  // 7. Q143-D24: the row is in and PENDING (screen_verdict NULL above) — `getMediaAsset` serves nothing pending.
  //    Ask the configured screen now; no screen, or no answer, leaves it pending for `screenPendingMedia`.
  const verdict = await screenStoredMedia(db, store, contentHash, bytes, contentType, opts, now);
  if (verdict === "flagged") throw new MediaError("this image was removed and cannot be re-stored", 403);
  // 8. RC-I3-0802: the small variant, only for bytes the screen did not match.
  await storeThumb(db, store, contentHash, bytes, contentType, opts.thumbs ?? null);
  return { content_hash: contentHash, content_type: contentType, width: header.width, height: header.height, screen: verdict };
}

/**
 * Q143-D24: screen one stored PENDING row and record the answer. clean -> served from now on; flagged -> one
 * SCAN_MATCH alert line (I21-Q65), recorded, then handed to the existing operator takedown (`deleteMediaAsset(…, null, …)`, unchanged), whose refusal or fault
 * leaves the row flagged (never served) for `screenPendingMedia` to finish; no screener or no answer -> pending.
 */
async function screenStoredMedia(db: DbLike, store: MediaStore, hash: string, bytes: Uint8Array, contentType: string, opts: StoreMediaOptions, now: () => string): Promise<"clean" | "pending" | "flagged"> {
  if (!opts.screener) return "pending";
  let out: Awaited<ReturnType<MediaScreener["screen"]>>;
  try {
    out = await opts.screener.screen(bytes, contentType);
  } catch {
    out = { verdict: "error", reason: "screen_threw" };
  }
  if (out.verdict === "clean") {
    await db.prepare("UPDATE media_assets SET screen_verdict = 'clean', screened_at = ? WHERE content_hash = ? AND screen_verdict IS NULL").bind(now(), hash).run();
    return "clean";
  }
  if (out.verdict === "flagged") {
    // I21-Q65 (D-1073 SHOULD; I31-TCFIX TCFIX-05, I31-JN-04): the same SCAN_MATCH line a report-evidence match writes
    // (evidence.ts recordScanMatch) — digest and category only — before the record and the takedown, so a failed
    // write or takedown still alerts. Both doors reach here: storeSignedMedia and screenPendingMedia.
    logError({ cid: "media-screen", hop: "http", msg: SCAN_MATCH_MARKER, marker: SCAN_MATCH_MARKER, detail: `sha256=${hash} category=${out.category ?? "-"}` });
    await db.prepare("UPDATE media_assets SET screen_verdict = 'flagged', screened_at = ? WHERE content_hash = ?").bind(now(), hash).run();
    try {
      await deleteMediaAsset(db, store, hash, null, now, opts.purge ?? null, MEDIA_SCREEN_TAKEDOWN_REASON);
    } catch (e) {
      logError({ cid: "media-screen", hop: "http", msg: "media_screen_takedown_failed", err: e instanceof Error ? e.name : typeof e });
    }
    return "flagged";
  }
  logError({ cid: "media-screen", hop: "http", msg: "media_screen_no_answer", detail: out.reason });
  return "pending";
}

/** RC-I3-0802: make, cap and record the variant. Any failure leaves no variant (the original is served at /thumb). */
async function storeThumb(db: DbLike, store: MediaStore, hash: string, bytes: Uint8Array, contentType: string, thumbs: ThumbMaker | null): Promise<void> {
  if (!thumbs) return;
  let out: Uint8Array;
  try {
    out = await thumbs.make(bytes, contentType);
  } catch (e) {
    logError({ cid: "media-thumb", hop: "http", msg: "media_thumb_failed", err: e instanceof Error ? e.name : typeof e });
    return;
  }
  if (out.length === 0 || out.length > MEDIA_THUMB_MAX_BYTES) return;
  await store.putBytes(mediaThumbKey(hash), out, MEDIA_THUMB_TYPE);
  const rec = await db.prepare("UPDATE media_assets SET thumb_bytes = ? WHERE content_hash = ?").bind(out.length, hash).run();
  // A takedown that removed the row while the variant was being made: the variant goes too (same shape as I3-0908).
  if ((rec.changes ?? 0) === 0) await store.delete(mediaThumbKey(hash));
}

export interface MediaAsset {
  content_hash: string;
  content_type: string;
}

/**
 * Public read: the asset row for a content hash (existence + validated type)
 * — AND the owning creator's standing RIGHT NOW.
 *
 * D-23: this used to ask only "does a row exist for this hash", so a creator
 * who was retracted or asked to be delisted (`request_suppressions`) after
 * storing art kept serving it forever — the one gate `storeSignedMedia`
 * enforces at STORE time (`VERIFIED_AT_SUBQUERY`, step 6) had no counterpart
 * at SERVE time. Mirrors the exact predicate `publicArchive` already uses for
 * the same two facts (`releases.ts` — retracted/suppressed → the release list
 * for that passport reads as empty, not as a 403 or a different shape), so a
 * suppressed/retracted creator's media 404s exactly like a hash that was never
 * stored — no new existence oracle, same shape the archive already commits to.
 * Deliberately narrower than the store gate's `VERIFIED_AT_SUBQUERY`: it does NOT also require
 * `claim_state = 'verified'`, because a lapsed-but-not-retracted verification
 * (`reverification_required`) is not the defect this row names, and
 * `publicArchive` does not gate releases on it either — widening past
 * retracted/suppressed would be a new, undiscussed product decision.
 */
export async function getMediaAsset(db: DbLike, hash: string): Promise<MediaAsset | null> {
  const r = await readMediaRow(db, hash);
  // Q143-D24: a pending or flagged row is NOT served (isServableVerdict) — every door that reads this primitive
  // (GET /media/:hash, /thumb, the creator avatar in creator-profile.ts) inherits it.
  return r && isServableVerdict(r.screen_verdict) ? { content_hash: r.content_hash, content_type: r.content_type } : null;
}

/** Q143-D24 + RC-I3-0802: what the serve routes answer. `pending` = stored under a live owner, not yet screened
 * clean (404 "pending"); `absent` = everything else that is not served, a flagged row included (no flag oracle). */
export async function mediaServeState(db: DbLike, hash: string): Promise<{ state: "ok"; asset: MediaAsset; thumb: boolean } | { state: "pending" } | { state: "absent" }> {
  const r = await readMediaRow(db, hash);
  if (!r) return { state: "absent" };
  if (isServableVerdict(r.screen_verdict)) return { state: "ok", asset: { content_hash: r.content_hash, content_type: r.content_type }, thumb: (r.thumb_bytes ?? 0) > 0 };
  return r.screen_verdict === null ? { state: "pending" } : { state: "absent" };
}

/** The row under the owner's standing RIGHT NOW (D-23 below), with its screen verdict and variant size. */
async function readMediaRow(db: DbLike, hash: string): Promise<{ content_hash: string; content_type: string; screen_verdict: string | null; thumb_bytes: number | null } | null> {
  if (typeof hash !== "string" || !SHA256_HEX.test(hash)) return null;
  const r = await db
    .prepare(
      `SELECT content_hash, content_type, screen_verdict, thumb_bytes FROM media_assets m
        WHERE m.content_hash = ?
          AND NOT EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = m.passport_id)
          AND NOT EXISTS (SELECT 1 FROM passports p WHERE p.passport_id = m.passport_id AND p.retracted = 1)`,
    )
    .bind(hash)
    .first<{ content_hash: string; content_type: string; screen_verdict: string | null; thumb_bytes: number | null }>();
  return r ?? null;
}

/** Delete a media asset — the creator's own (self-service) OR any (operator). When
 * `passportId` is given, ownership is enforced; when null, it's an operator takedown.
 * Removes the R2 object + the row + THE EDGE-CACHED COPY. An OPERATOR takedown also
 * tombstones the hash so it can never be re-stored (a creator self-delete does NOT —
 * an honest mistake stays re-uploadable). Returns whether a row was removed.
 *
 * `purge` IS NOT OPTIONAL IN MEANING, only in arity. It is `EdgePurger | null`
 * where null means "this deployment cannot purge" and an OMITTED argument means
 * exactly the same thing — a call site that has not been wired must not inherit
 * the old, silent behaviour by saying nothing. See the refusal below. */
export async function deleteMediaAsset(db: DbLike, store: MediaStore, hash: string, passportId: string | null, now: () => string = () => new Date().toISOString(), purge?: EdgePurger | null, reason: string = "operator takedown"): Promise<boolean> {
  if (typeof hash !== "string" || !SHA256_HEX.test(hash)) throw new MediaError("no such asset", 404);
  const row = await db.prepare("SELECT passport_id FROM media_assets WHERE content_hash = ?").bind(hash).first<{ passport_id: string }>();
  if (!row) {
    // I3-1270 — A TAKEDOWN WHOSE PURGE FAILED IS FINISHED BY RETRYING IT. The
    // tombstone is written first and the row dropped before the purge, so a
    // failed purge leaves exactly this state: tombstoned, no row. Answering 404
    // here made the manual URL the only completion path. An OPERATOR retry now
    // skips steps 1-3 (already done) and re-issues step 4. A creator self-delete
    // writes no tombstone, so it still 404s — and so does a hash never taken down.
    if (passportId === null && (await db.prepare("SELECT 1 AS x FROM media_takedowns WHERE content_hash = ?").bind(hash).first())) {
      if (!purge) {
        throw new MediaError("edge cache purge is not configured — refusing a takedown that would leave cached copies served (set CF_PURGE_ZONE_ID, CF_PURGE_API_TOKEN and PUBLIC_BASE_URL)", 503);
      }
      await purgeAndStamp(db, purge, hash, now);
      return true;
    }
    throw new MediaError("no such asset", 404);
  }
  if (passportId !== null && row.passport_id !== passportId) throw new MediaError("not your asset", 403);

  // ─── FAIL CLOSED ON A DEPLOYMENT THAT CANNOT PURGE — BEFORE ANY DESTRUCTION ───
  //
  // WHAT WAS CHOSEN, AND WHY IT IS THE REFUSAL AND NOT THE HALF-TAKEDOWN. The
  // honest options were (a) refuse the takedown, or (b) perform it and report
  // loudly that the edge was not purged. Option (b) removes strictly more bytes
  // in the moment, and it was still rejected, for one reason that decides it: a
  // half-performed takedown CANNOT BE FINISHED BY RETRYING IT. The row and the
  // object are gone, so the retry an operator reaches for after fixing the
  // config answers 404 — which this product's own console reads as "nearly
  // always a mistyped or truncated hash". The lever would have destroyed its own
  // completion path and then lied about why.
  //
  // Refusing keeps every state repairable: nothing is tombstoned, nothing is
  // deleted, the console says "nothing was taken down · check config health"
  // (which is TRUE, and is copy that already exists for the absent-bucket case),
  // and the retry after two config values are set completes the takedown for
  // real — origin AND edge. Minutes of delay against a partial removal that can
  // never be made whole is not a close call when the duty is "remove or disable
  // the material".
  //
  // It is also just invariant 6: absent config DISABLES a feature (503), never
  // silently narrows it. The feature here is the takedown, and a takedown that
  // leaves the material served worldwide for a year is not a narrower takedown,
  // it is a false one.
  //
  // AN OMITTED ARGUMENT IS TREATED AS "CANNOT PURGE" and not as "the old
  // behaviour". That is the whole point: before this, every caller purged
  // nothing and every caller reported success.
  if (!purge) {
    throw new MediaError("edge cache purge is not configured — refusing a takedown that would leave cached copies served (set CF_PURGE_ZONE_ID, CF_PURGE_API_TOKEN and PUBLIC_BASE_URL)", 503);
  }

  // TOMBSTONE FIRST, R2 LAST. This is the CSAM takedown lever, so the ordering is
  // the guarantee, not a nicety: the tombstone used to be written AFTER the object
  // delete and was only CHECKED by a read five statements ahead of the re-store's
  // write, so a creator who re-POSTed the identical bytes in that window got the
  // object back — tombstoned and servable, with the console reporting success, and
  // they chose the timing. Now:
  //   1. tombstone   — after this, `storeSignedMedia`'s guarded INSERT can never
  //                    admit the hash again (it tests this table in the statement).
  //   2. drop the row — nothing serves without it.
  //   3. delete R2   — LAST, so it runs after any concurrent put that could still
  //                    have got a row in, and sweeps those bytes too.
  // A creator self-delete (passportId given) writes no tombstone by design — an
  // honest mistake stays re-uploadable — and takes its own order, below.
  if (passportId === null) {
    // F-19/F-37: this used to hard-code the literal 'operator takedown' for
    // EVERY ground (DMCA, CSAM, ordinary abuse), so `media_takedowns.reason`
    // could never distinguish them — no repeat-infringer count, no way to
    // tell a CSAM removal from a copyright one in the one table that records
    // why a hash is banned forever. `reason` is now the caller's own word for
    // the ground, defaulting to the prior literal so every existing call site
    // (media-routes.ts has none with passportId===null; console.ts's
    // `/ops/media/:hash` is the only one) keeps its current behaviour until
    // that route is wired to collect and forward the real ground — WRONG
    // OWNER for this room (console.ts), see VERDICTS.md.
    await db.prepare("INSERT OR IGNORE INTO media_takedowns (content_hash, reason, created_at) VALUES (?, ?, ?)").bind(hash, reason, now()).run();
  }
  // I3-1270 (creator door, I22): A SELF-DELETE KEEPS ITS ROW UNTIL THE EDGE IS
  // PURGED. Order: R2 object -> edge purge -> row. The row was dropped first,
  // so a failed purge left nothing a retry could find (404) and nothing that
  // recorded the hash still cached for a year. Now the row IS that record, and
  // the creator's retry passes the ownership precheck and re-issues the delete
  // and the purge. Serving needs the object as well as the row
  // (media-routes.ts GET /media/:hash), so nothing refills the edge meanwhile.
  // I3-1770 kept: ownership is re-asserted by statement immediately before the
  // R2 delete, and the row DELETE itself is ownership-bound.
  if (passportId !== null) {
    const mine = await db.prepare("SELECT 1 AS x FROM media_assets WHERE content_hash = ? AND passport_id = ?").bind(hash, passportId).first();
    if (!mine) throw new MediaError("no such asset", 404);
    await store.delete(mediaKey(hash));
    await deleteThumbIfAny(store, hash);
    const purgedSelf = await purge.purgeMedia(hash);
    if (!purgedSelf.ok) throw new MediaEdgeNotPurged(purgedSelf.url, purgedSelf.detail);
    const gone = await db.prepare("DELETE FROM media_assets WHERE content_hash = ? AND passport_id = ?").bind(hash, passportId).run();
    if ((gone.changes ?? 0) === 0) throw new MediaError("no such asset", 404);
    return true;
  }
  await db.prepare("DELETE FROM media_assets WHERE content_hash = ?").bind(hash).run();
  await store.delete(mediaKey(hash));
  await deleteThumbIfAny(store, hash);

  //   4. PURGE THE EDGE — LAST, and after the object is gone.
  // Ordering again, and again it is the guarantee rather than tidiness: a purge
  // issued while the object is still in R2 (or the row still readable) is undone
  // by the very next request, which refills the same PoP from origin. Purging
  // after step 3 leaves nothing to refill from — `GET /media/:hash` 404s.
  //
  // A FAILED PURGE IS A FAILED TAKEDOWN. It throws rather than returning,
  // because every caller of this function reports its RETURN VALUE as success
  // (`{ok:true, removed}`) and there is no return value that can mean "the bytes
  // are still being served". The origin removal is NOT rolled back: the tombstone
  // is the durable ban and R2 has no transaction, so undoing it would trade a
  // known partial state for an unknown one. The operator is handed the exact URL
  // to purge by hand.
  await purgeAndStamp(db, purge, hash, now);
  // An operator takedown is reported as done only when it IS done: the hash is
  // banned, the object is gone AND the edge has been purged, whether or not this
  // call was the one that removed the row (a concurrent delete removing it does
  // not un-take the takedown).
  return true;
}

/** RC-I3-0802: the variant goes wherever the original goes — in the same door, right after it. Probed in R2 (not a
 * second D1 read) so the takedown's statement order is unchanged; a variant written after the probe has no row to
 * serve it and is collected by `sweepMediaOrphans`. */
async function deleteThumbIfAny(store: MediaStore, hash: string): Promise<void> {
  if ((await store.getBytes(mediaThumbKey(hash))) !== null) await store.delete(mediaThumbKey(hash));
}

/** Step 4 of an operator takedown: purge the edge, then record that it happened
 * (`media_takedowns.purged_at`, I3-1270). The stamp is written only after a
 * successful purge; a failure throws `MediaEdgeNotPurged` and leaves it NULL,
 * which is exactly what `sweepUnpurgedTakedowns` and the ops count read. */
async function purgeAndStamp(db: DbLike, purge: EdgePurger, hash: string, now: () => string): Promise<void> {
  const purged = await purge.purgeMedia(hash);
  if (!purged.ok) throw new MediaEdgeNotPurged(purged.url, purged.detail);
  await db.prepare("UPDATE media_takedowns SET purged_at = ? WHERE content_hash = ? AND purged_at IS NULL").bind(now(), hash).run();
}

/** Tombstones older than this with no recorded purge are retried by the cron. */
export const TAKEDOWN_PURGE_RETRY_AGE_MS = 5 * 60 * 1000;
/** At most this many takedown purges are retried per cron run. */
export const TAKEDOWN_PURGE_RETRY_BATCH = 20;

/** Ops overview (I3-1270 (4)): operator takedowns whose edge purge has not been
 * recorded as done — each is material the edge may still be serving. */
export async function countUnpurgedTakedowns(db: DbLike): Promise<number> {
  const r = await db.prepare("SELECT COUNT(*) AS n FROM media_takedowns WHERE purged_at IS NULL").first<{ n: number }>();
  return Number(r?.n ?? 0);
}

/**
 * I3-1270 (3) — THE CRON HALF OF A FAILED-PURGE TAKEDOWN. Re-runs
 * `deleteMediaAsset(…, null, …)` — the one takedown path, not a second one — for
 * at most TAKEDOWN_PURGE_RETRY_BATCH tombstones with `purged_at IS NULL` that are
 * older than TAKEDOWN_PURGE_RETRY_AGE_MS (so a takedown still in flight is left
 * to its own request). A tombstone with no row gets the purge re-issued; one
 * whose row survived a fault gets the whole takedown finished, same ordering.
 * Every failure is logged and left NULL for the next run. A deployment that
 * cannot purge is a 503 (thrown), never a silent skip.
 */
export async function sweepUnpurgedTakedowns(db: DbLike, store: MediaStore, purge: EdgePurger | null | undefined, now: () => string = () => new Date().toISOString(), cid = "media-purge-retry"): Promise<{ attempted: number; purged: number; failed: number }> {
  if (!purge) {
    throw new MediaError("edge cache purge is not configured — cannot retry unpurged takedowns (set CF_PURGE_ZONE_ID, CF_PURGE_API_TOKEN and PUBLIC_BASE_URL)", 503);
  }
  const nowIso = now();
  const cutoff = new Date(Date.parse(nowIso) - TAKEDOWN_PURGE_RETRY_AGE_MS).toISOString();
  const { results } = await db
    .prepare("SELECT content_hash FROM media_takedowns WHERE purged_at IS NULL AND created_at <= ? ORDER BY created_at, content_hash LIMIT ?")
    .bind(cutoff, TAKEDOWN_PURGE_RETRY_BATCH)
    .all<{ content_hash: string }>();
  let purged = 0;
  let failed = 0;
  for (const r of results) {
    try {
      await deleteMediaAsset(db, store, r.content_hash, null, now, purge);
      purged++;
    } catch (e) {
      failed++;
      logError({ cid, hop: "cron", msg: "media_takedown_purge_retry_failed", err: e instanceof Error ? e.name : typeof e });
    }
  }
  return { attempted: results.length, purged, failed };
}

/** Q143-D24: rows `screenPendingMedia` handles per run. */
export const MEDIA_SCREEN_BATCH = 20;
/** D-1059(7): cron screens of one PENDING row that may go unanswered (no verdict, or no stored object) before the row
 * is marked screen-failed (`screen_failed_at`, migration 0269): still pending, never served, counted for the operator. */
export const MEDIA_SCREEN_MAX_ATTEMPTS = 6;
/** D-1059(7): the wait before the next cron screen of an unanswered row is this times 2^(attempts - 1). */
export const MEDIA_SCREEN_BACKOFF_BASE_MS = 5 * 60 * 1000;

/** D-1059(7): the instant an unanswered row is next due, after its `attempts`-th unanswered cron screen. */
export function mediaScreenNextAt(nowIso: string, attempts: number): string {
  return new Date(Date.parse(nowIso) + MEDIA_SCREEN_BACKOFF_BASE_MS * 2 ** Math.max(0, attempts - 1)).toISOString();
}

/** D-1059(7) ops overview: pending uploads the screen gave up on (never served; an operator decides them). */
export async function countScreenFailedMedia(db: DbLike): Promise<number> {
  const r = await db.prepare("SELECT COUNT(*) AS n FROM media_assets WHERE screen_verdict IS NULL AND screen_failed_at IS NOT NULL").first<{ n: number }>();
  return Number(r?.n ?? 0);
}

/**
 * Q143-D24 — THE RETRY HALF OF SCREEN-BEFORE-SERVE (a scheduled() step, wired by the integration room). Screens
 * at most MEDIA_SCREEN_BATCH PENDING rows (an upload whose screen was unconfigured, unanswered or faulted), oldest
 * first (pending before flagged, so a stuck takedown cannot starve the screen), and re-runs the existing takedown for FLAGGED rows a refused or faulted takedown left behind. A pending
 * row whose object is missing is skipped (`probeMediaRows` reports it). No screener is a 503 (thrown), never a
 * silent skip: with none configured nothing new is ever served.
 */
export async function screenPendingMedia(db: DbLike, store: MediaStore, screener: MediaScreener | null | undefined, purge: EdgePurger | null | undefined, now: () => string = () => new Date().toISOString()): Promise<{ attempted: number; clean: number; flagged: number; pending: number; takedowns: number; failed: number }> {
  if (!screener) throw new MediaError("media screening is not configured — newly stored media stays pending and is not served (set MEDIA_SCREEN_PROVIDER)", 503);
  const out = { attempted: 0, clean: 0, flagged: 0, pending: 0, takedowns: 0, failed: 0 };
  // D-1059(7) (I30-RG-OPS NEW-media-screen-starve): a pending row is taken only while it is not screen-failed and is
  // due, so rows the provider cannot answer back off instead of holding the batch's head forever.
  const tickIso = now();
  const { results } = await db
    .prepare(
      "SELECT content_hash, content_type, screen_verdict, screen_attempts FROM media_assets" +
        " WHERE (screen_verdict IS NULL AND screen_failed_at IS NULL AND (screen_next_at IS NULL OR screen_next_at <= ?)) OR screen_verdict = 'flagged'" +
        " ORDER BY (screen_verdict IS NULL) DESC, created_at, content_hash LIMIT ?",
    )
    .bind(tickIso, MEDIA_SCREEN_BATCH)
    .all<{ content_hash: string; content_type: string; screen_verdict: string | null; screen_attempts: number }>();
  // An unanswered screen (no verdict, or no object to screen) is recorded on the row: the next due instant and, at
  // the bound, the screen-failed mark. The row stays pending (verdict NULL), so nothing about it is ever served.
  const unanswered = async (r: { content_hash: string; screen_attempts: number }): Promise<void> => {
    const attempts = Number(r.screen_attempts ?? 0) + 1;
    const failed = attempts >= MEDIA_SCREEN_MAX_ATTEMPTS;
    await db
      .prepare("UPDATE media_assets SET screen_attempts = ?, screen_next_at = ?, screen_failed_at = ? WHERE content_hash = ? AND screen_verdict IS NULL")
      .bind(attempts, mediaScreenNextAt(tickIso, attempts), failed ? tickIso : null, r.content_hash)
      .run();
    if (failed) {
      out.failed++;
      logError({ cid: "media-screen", hop: "cron", msg: "media_screen_failed" });
    }
  };
  for (const r of results) {
    out.attempted++;
    if (r.screen_verdict === "flagged") {
      try {
        await deleteMediaAsset(db, store, r.content_hash, null, now, purge ?? null, MEDIA_SCREEN_TAKEDOWN_REASON);
        out.takedowns++;
      } catch (e) {
        logError({ cid: "media-screen", hop: "cron", msg: "media_screen_takedown_failed", err: e instanceof Error ? e.name : typeof e });
      }
      continue;
    }
    const obj = await store.getBytes(mediaKey(r.content_hash));
    if (!obj) {
      out.pending++;
      await unanswered(r);
      continue;
    }
    const v = await screenStoredMedia(db, store, r.content_hash, obj.bytes, r.content_type, { screener, purge: purge ?? null }, now);
    out[v]++;
    if (v === "pending") await unanswered(r);
  }
  return out;
}

/** I3-0799: subrequests (statements + purge calls) one hidden-media sweep may spend. */
export const HIDDEN_MEDIA_PURGE_BUDGET = 30;
/** I3-0799: the two `sweep_cursor` rows the hidden-media sweep resumes from. */
export const HIDDEN_MEDIA_CURSORS = { retracted: "media_purge_retracted", suppressed: "media_purge_suppressed" } as const;

interface HiddenCursor {
  /** The hide this sweep has reached: [seq] (retraction) or [rowid, passport_id, created_at] (suppression — RGP-L5). */
  k: string[];
  /** The passport being purged, or "" between passports. */
  p: string;
  /** The last content_hash purged for `p` ("" before its first). */
  h: string;
}

/**
 * I3-0799 — THE EDGE HALF OF A HIDE THAT NO DOOR PURGED. Media is served
 * `public, max-age=31536000, immutable` (media-routes.ts), so a passport that
 * leaves public view keeps its images at the edge until each URL is purged. The
 * removal / importer / console doors call `purgePassportMedia` themselves, but a
 * retraction recorded through the ledger (`retract_passport`, events.ts) never
 * did, and a door whose purge failed had no retry. This walks BOTH sources of a
 * hide in order — `correction` events whose action is `retract_passport`, by seq, `request_suppressions` rows
 * by rowid — and purges every `media_assets` hash of each
 * hidden passport, ONCE per hide (a door that already purged costs one repeat).
 *
 * Progress lives in `sweep_cursor` (HIDDEN_MEDIA_CURSORS), down to the last hash
 * purged, so a passport with more assets than one run's budget resumes where it
 * stopped. A purge that fails HOLDS the cursor at the last success and ends that
 * source's walk: the same hash is retried next run, never skipped. Every
 * statement and purge call is charged to `budget` before it is sent. A
 * deployment with no purger is a 503 (thrown), like `sweepUnpurgedTakedowns`.
 * Re-showing a passport needs nothing: a later suppression is a new key.
 */
export async function sweepHiddenPassportMedia(
  db: DbLike,
  purge: EdgePurger | null | undefined,
  now: () => string = () => new Date().toISOString(),
  cid = "media-hidden-purge",
  budget: number = HIDDEN_MEDIA_PURGE_BUDGET,
): Promise<{ purged: number; failed: number; passports: number; spent: number }> {
  if (!purge) {
    throw new MediaError("edge cache purge is not configured — cannot purge hidden passports' media (set CF_PURGE_ZONE_ID, CF_PURGE_API_TOKEN and PUBLIC_BASE_URL)", 503);
  }
  const out = { purged: 0, failed: 0, passports: 0, spent: 0 };
  const next = async (src: keyof typeof HIDDEN_MEDIA_CURSORS, k: string[]): Promise<{ k: string[]; p: string } | null> => {
    if (src === "retracted") {
      const r = await db
        .prepare("SELECT seq, subject FROM trust_events WHERE event_type = 'correction' AND seq > ? AND json_extract(payload, '$.action') = 'retract_passport' ORDER BY seq LIMIT 1")
        .bind(Number(k[0] ?? 0))
        .first<{ seq: number; subject: string }>();
      return r ? { k: [String(r.seq)], p: r.subject } : null;
    }
    // RGP-L5 (I15-FX4): COMMIT ORDER, not `created_at`. The stamp is taken before
    // the row commits, so a suppression stamped earlier but committed after the
    // walk passed that instant was never visited. SQLite assigns rowids under its
    // single writer, so rowid order is commit order — while the row the cursor
    // stands on still exists. Un-suppressing DELETEs rows (requests.ts), and
    // SQLite reuses freed TOP rowids, so once that anchor row is gone (or is a
    // different row) a new suppression can sit at or below the cursor: the walk
    // then restarts from the first row — a repeated purge is harmless, a skipped
    // one is not. An older [created_at, passport_id] cursor restarts the same way.
    const [rid, pid, at] = k.length === 3 ? [Number(k[0]), k[1] ?? "", k[2] ?? ""] : [0, "", ""];
    const r = await db
      .prepare(
        "SELECT rowid AS rid, passport_id, created_at FROM request_suppressions WHERE rowid > (CASE WHEN EXISTS (SELECT 1 FROM request_suppressions WHERE rowid = ? AND passport_id = ? AND created_at = ?) THEN ? ELSE 0 END) ORDER BY rowid LIMIT 1",
      )
      .bind(Number.isFinite(rid) ? rid : 0, pid, at, Number.isFinite(rid) ? rid : 0)
      .first<{ rid: number; passport_id: string; created_at: string }>();
    return r ? { k: [String(r.rid), r.passport_id, r.created_at], p: r.passport_id } : null;
  };
  for (const src of ["retracted", "suppressed"] as const) {
    if (out.spent + 1 > budget) break;
    out.spent += 1;
    const row = await db.prepare("SELECT cursor_id FROM sweep_cursor WHERE name = ?").bind(HIDDEN_MEDIA_CURSORS[src]).first<{ cursor_id: string }>();
    let cur: HiddenCursor = { k: [], p: "", h: "" };
    try {
      if (row?.cursor_id) cur = JSON.parse(row.cursor_id) as HiddenCursor;
    } catch {
      // An unreadable cursor restarts the walk: a repeated purge is harmless, a skipped one is not.
    }
    const start = JSON.stringify(cur);
    walk: while (out.spent + 2 <= budget) {
      if (!cur.p) {
        out.spent += 1;
        const nx = await next(src, cur.k);
        if (!nx) break;
        cur = { k: nx.k, p: nx.p, h: "" };
        out.passports += 1;
      }
      // One page read, then as many purges as the budget still holds (the cursor write is reserved).
      const room = budget - out.spent - 2;
      if (room < 1) break;
      out.spent += 1;
      const { results } = await db
        .prepare("SELECT content_hash FROM media_assets WHERE passport_id = ? AND content_hash > ? ORDER BY content_hash LIMIT ?")
        .bind(cur.p, cur.h, room)
        .all<{ content_hash: string }>();
      for (const r of results) {
        out.spent += 1;
        let ok = false;
        try {
          ok = (await purge.purgeMedia(r.content_hash)).ok;
        } catch {
          ok = false;
        }
        if (!ok) {
          out.failed += 1;
          logError({ cid, hop: "cron", msg: "media_hidden_purge_failed", detail: src });
          break walk;
        }
        out.purged += 1;
        cur.h = r.content_hash;
      }
      if (results.length < room) cur = { k: cur.k, p: "", h: "" };
    }
    if (JSON.stringify(cur) !== start) {
      out.spent += 1;
      await db
        .prepare("INSERT INTO sweep_cursor (name, cursor_seq, cursor_id, updated_at) VALUES (?, 0, ?, ?) ON CONFLICT(name) DO UPDATE SET cursor_id = excluded.cursor_id, updated_at = excluded.updated_at")
        .bind(HIDDEN_MEDIA_CURSORS[src], JSON.stringify(cur), now())
        .run();
    }
  }
  return out;
}

/**
 * #205b (D-0920 option 1) — THE OTHER DIRECTION: a `media_assets` row whose object is gone (a delete that raced a
 * re-upload, a lost write) serves 404 until the creator happens to re-upload. Probes at most
 * MEDIA_MISSING_PROBE_BUDGET rows older than `minAgeIso`, in content_hash order from the `walk` cursor; each missing
 * object is a `media_object_missing` error line at once, and a completed pass records its count in the `found` row.
 * A probe that throws ends the run with the cursor held before that row. Never deletes anything.
 */
async function probeMediaRows(db: DbLike, store: MediaStore, nowIso: string, minAgeIso: string): Promise<{ probed: number; missing: number }> {
  const cur = await db.prepare("SELECT cursor_seq, cursor_id FROM sweep_cursor WHERE name = ?").bind(MEDIA_MISSING_CURSORS.walk).first<{ cursor_seq: number; cursor_id: string }>();
  let after = cur?.cursor_id ?? "";
  let passMissing = Number(cur?.cursor_seq ?? 0);
  const { results } = await db
    .prepare("SELECT content_hash FROM media_assets WHERE content_hash > ? AND created_at <= ? ORDER BY content_hash LIMIT ?")
    .bind(after, minAgeIso, MEDIA_MISSING_PROBE_BUDGET)
    .all<{ content_hash: string }>();
  let probed = 0;
  let missing = 0;
  let broke = false;
  for (const r of results) {
    let obj: Awaited<ReturnType<MediaStore["getBytes"]>>;
    try {
      obj = await store.getBytes(mediaKey(r.content_hash));
    } catch {
      broke = true;
      break;
    }
    probed++;
    if (obj === null) {
      missing++;
      passMissing++;
      logError({ cid: "media-orphans", hop: "cron", msg: "media_object_missing", detail: r.content_hash });
    }
    after = r.content_hash;
  }
  const upsert = "INSERT INTO sweep_cursor (name, cursor_seq, cursor_id, updated_at) VALUES (?, ?, ?, ?) ON CONFLICT(name) DO UPDATE SET cursor_seq = excluded.cursor_seq, cursor_id = excluded.cursor_id, updated_at = excluded.updated_at";
  if (!broke && results.length < MEDIA_MISSING_PROBE_BUDGET) {
    await db.prepare(upsert).bind(MEDIA_MISSING_CURSORS.found, passMissing, "", nowIso).run();
    after = "";
    passMissing = 0;
  }
  await db.prepare(upsert).bind(MEDIA_MISSING_CURSORS.walk, passMissing, after, nowIso).run();
  return { probed, missing };
}

/**
 * I3-1701 — THE MEDIA ORPHAN COLLECTOR. Bytes with no `media_assets` row (a put
 * whose INSERT never ran, a worker killed mid-store) are creator art no erasure
 * or takedown can reach, because both start from the row. This walks the
 * `media/` prefix through `MediaStore.list` and deletes an object ONLY when its
 * key is `media/<sha256>`, it was uploaded at least MEDIA_ORPHAN_MIN_AGE_MS ago,
 * AND no `media_assets` row has that hash — at most MEDIA_ORPHAN_MAX_DELETES per
 * run. Progress lives in `sweep_cursor` row 'media_orphans': `cursor_id` is the
 * store's listing cursor ('' = start), `cursor_seq` is 1 mid-walk and 0 idle, and
 * an idle walk restarts only MEDIA_ORPHAN_WALK_INTERVAL_MS after it finished —
 * so it can ride the six-hourly cron and still run weekly. When the delete cap
 * stops a run mid-page, the cursor stays at that page's start and the next run
 * re-lists it (deleted keys are simply gone). A store without `list` is a 503.
 */
export async function sweepMediaOrphans(
  db: DbLike,
  store: MediaStore,
  now: () => string = () => new Date().toISOString(),
): Promise<{ skipped: boolean; scanned: number; deleted: number; complete: boolean; probed: number; missing: number }> {
  if (typeof store.list !== "function") throw new MediaError("media store cannot list — orphan collection unavailable", 503);
  const nowIso = now();
  const nowMs = Date.parse(nowIso);
  const minAgeIso = new Date(nowMs - MEDIA_ORPHAN_MIN_AGE_MS).toISOString();
  // #205b: the rows->bucket probe runs on EVERY call, including one whose bucket walk is skipped below.
  const { probed, missing } = await probeMediaRows(db, store, nowIso, minAgeIso);
  const cur = await db.prepare("SELECT cursor_seq, cursor_id, updated_at FROM sweep_cursor WHERE name = 'media_orphans'").first<{ cursor_seq: number; cursor_id: string; updated_at: string }>();
  const walking = Number(cur?.cursor_seq ?? 0) === 1;
  if (cur && !walking && Date.parse(cur.updated_at) > nowMs - MEDIA_ORPHAN_WALK_INTERVAL_MS) {
    return { skipped: true, scanned: 0, deleted: 0, complete: false, probed, missing };
  }
  let cursor: string | null = walking ? cur?.cursor_id || null : null;
  let scanned = 0;
  let deleted = 0;
  let complete = false;
  for (let page = 0; page < MEDIA_ORPHAN_MAX_PAGES; page++) {
    const listed = await store.list("media/", cursor);
    let capped = false;
    for (const o of listed.objects) {
      scanned++;
      const m = MEDIA_KEY_RE.exec(o.key);
      if (!m) continue;
      const up = Date.parse(o.uploaded);
      if (!Number.isFinite(up) || up > Date.parse(minAgeIso)) continue;
      const ref = await db.prepare("SELECT 1 AS x FROM media_assets WHERE content_hash = ?").bind(m[1]).first();
      if (ref) continue;
      if (deleted >= MEDIA_ORPHAN_MAX_DELETES) {
        capped = true;
        break;
      }
      await store.delete(o.key);
      deleted++;
    }
    if (capped) break; // resume at this page's start next run
    cursor = listed.cursor;
    if (cursor === null) {
      complete = true;
      break;
    }
  }
  await db
    .prepare("INSERT INTO sweep_cursor (name, cursor_seq, cursor_id, updated_at) VALUES ('media_orphans', ?, ?, ?) ON CONFLICT(name) DO UPDATE SET cursor_seq = excluded.cursor_seq, cursor_id = excluded.cursor_id, updated_at = excluded.updated_at")
    .bind(complete ? 0 : 1, complete ? "" : (cursor ?? ""), nowIso)
    .run();
  return { skipped: false, scanned, deleted, complete, probed, missing };
}

/**
 * I3-1269 — A VISIBILITY TRANSITION MUST PURGE THE EDGE TOO. `getMediaAsset`
 * stops serving a retracted or suppressed creator's media at the ORIGIN, but a
 * PoP that already filled keeps it for a year (`public, max-age=31536000,
 * immutable`, media-routes.ts), so without a purge the transition is cosmetic
 * at the edge — the same gap `deleteMediaAsset` closes for a takedown. This
 * purges every asset URL the passport owns. Bounded by MAX_ASSETS_PER_CREATOR
 * (the store's own cap), so the loop is at most that many purge calls.
 *
 * Fail-closed like the takedown: a null/absent purger is a 503, never a silent
 * skip. It does not throw on a per-URL failure; it returns the failures so the
 * caller (the door that flipped visibility) can name them to the operator, and
 * `purged + failed.length` equals the number of rows it read.
 *
 * WIRING IS NOT IN THIS FILE: the doors that flip visibility (`requests.ts
 * setRequestSuppression`, `removal.ts` suppression inserts, the retraction path
 * through `events.ts`) must call it with `edgePurgerFromEnv(env)` — see the
 * I4-04-R2 report's WRONG OWNER rows.
 */
export async function purgePassportMedia(db: DbLike, passportId: string, purge?: EdgePurger | null): Promise<{ purged: number; failed: { url: string; detail: string }[] }> {
  if (typeof passportId !== "string" || !/^psp_[0-9a-f]{16}$/.test(passportId)) throw new MediaError("bad passport id", 400);
  if (!purge) {
    throw new MediaError("edge cache purge is not configured — refusing a visibility change that would leave cached copies served (set CF_PURGE_ZONE_ID, CF_PURGE_API_TOKEN and PUBLIC_BASE_URL)", 503);
  }
  const { results } = await db.prepare("SELECT content_hash FROM media_assets WHERE passport_id = ? ORDER BY content_hash").bind(passportId).all<{ content_hash: string }>();
  let purged = 0;
  const failed: { url: string; detail: string }[] = [];
  for (const r of results) {
    const out = await purge.purgeMedia(r.content_hash);
    if (out.ok) purged++;
    else failed.push({ url: out.url, detail: out.detail });
  }
  return { purged, failed };
}

/**
 * services/trust-core/src/money-reconcile.ts
 *
 * I53-RECONCILE (D-1382 item 3, D-1383(3)): the money books' conservation identities, read once per money tick
 * (index.ts, the money group's one leg, after reconcileCashOutflows). READ-ONLY: every statement is a SELECT.
 * Each identity is ONE aggregate statement (one subrequest) counting violating rows; the LIMIT inside the COUNT
 * bounds the rows the subquery returns (so the count is capped at MONEY_RECONCILE_ROW_CAP), NOT the rows it scans.
 * A non-zero total logs ONE line, msg `money_reconcile_drift`, carrying `count` (violating rows, all identities)
 * and `scanned` (identities checked) only: no id, no account, no amount (log.ts ALLOWED_KEYS, INV-061).
 * The identities are I53-MONEY's as amended by I53-W0; "live" is per key (an erased fan's rows sit on a tombstone
 * that has no accounts row, an erased creator's on a `del_` passport).
 */
import { CASH_ORIGINS_SQL, CREATOR_SHARE_DEN, CREATOR_SHARE_NUM, CREDIT_BACKING_CENTS } from "./credits.js";
import type { DbLike } from "./db.js";
import { logError } from "./log.js";
import { RESERVED_PAYOUT_STATUS_SQL } from "./payout-status-sql.js";

/** The cap on each identity's count (the LIMIT inside its COUNT subquery). */
export const MONEY_RECONCILE_ROW_CAP = 1000;

export type MoneyIdentity = "purchase_fold" | "reserved_over_cleared" | "lot_sum" | "non_cash_spend" | "sale_split";

const capped = (inner: string): string => `SELECT COUNT(*) AS n FROM (${inner} LIMIT ?)`;

/**
 * One statement per identity, each binding (cap) last; `reserved_over_cleared` binds (nowIso, cap).
 * - purchase_fold: a live purchase whose ledger rows (dispute freeze and unfreeze excluded) do not net to
 *   credits_granted minus its reversal's credits_clawed.
 * - reserved_over_cleared: a non-`del_` passport whose reserved payouts exceed its cleared unreversed net. A
 *   negative available after a later dispute debit is the designed owed state and is NOT counted.
 * - lot_sum: a live sale with at least one lot edge whose edges do not sum to credits_spent (an edge-less sale is
 *   not decidable: nothing records the 0221 cut-over).
 * - non_cash_spend: a live cash-bearing spend (store_spend, tip) after which the account's cash-origin balance, in
 *   insertion order through that row, is below zero (CASH_GUARD_SQL asks it at or above the spend inside the debit).
 * - sale_split: a sale whose split is not the product's: direct_buy and pack floor the 80/100 share, tip is 100%.
 */
export const MONEY_RECONCILE_SQL: Readonly<Record<MoneyIdentity, string>> = Object.freeze({
  purchase_fold: capped(
    "SELECT 1 FROM credit_purchases p WHERE EXISTS (SELECT 1 FROM accounts a WHERE a.account_id = p.account_id) " +
      "AND (SELECT COALESCE(SUM(l.credits),0) FROM credit_ledger l WHERE l.account_id = p.account_id AND l.ref = p.purchase_id AND l.reason NOT IN ('dispute_freeze','dispute_unfreeze')) " +
      "<> p.credits_granted - COALESCE((SELECT r.credits_clawed FROM credit_reversals r WHERE r.purchase_id = p.purchase_id),0)",
  ),
  reserved_over_cleared: capped(
    `SELECT 1 FROM creator_payouts cp WHERE cp.status IN ${RESERVED_PAYOUT_STATUS_SQL} AND cp.passport_id NOT GLOB 'del_*' ` +
      "GROUP BY cp.passport_id HAVING SUM(cp.amount_cents) > " +
      "(SELECT COALESCE(SUM(s.creator_net_cents),0) FROM sales s WHERE s.passport_id = cp.passport_id AND s.reversed = 0 AND s.clears_at <= ?)",
  ),
  lot_sum: capped(
    "SELECT 1 FROM sales s JOIN credit_spend_lots e ON e.sale_or_tip_id = s.sale_id " +
      "WHERE EXISTS (SELECT 1 FROM accounts a WHERE a.account_id = s.account_id) GROUP BY s.sale_id HAVING SUM(e.credits) <> s.credits_spent",
  ),
  non_cash_spend: capped(
    "SELECT 1 FROM (SELECT w.account_id AS account_id, w.origin AS origin, " +
      `SUM(CASE WHEN w.origin IN ${CASH_ORIGINS_SQL} THEN w.credits ELSE 0 END) OVER (PARTITION BY w.account_id ORDER BY w.rowid ROWS UNBOUNDED PRECEDING) AS cash ` +
      "FROM credit_ledger w) x WHERE x.origin IN ('store_spend','tip') AND x.cash < 0 " +
      "AND EXISTS (SELECT 1 FROM accounts a WHERE a.account_id = x.account_id)",
  ),
  sale_split: capped(
    `SELECT 1 FROM sales s WHERE NOT (s.gross_backing_cents = s.credits_spent * ${CREDIT_BACKING_CENTS} AND (` +
      `(s.product IN ('direct_buy','pack') AND s.creator_net_cents = (s.credits_spent * ${CREDIT_BACKING_CENTS} * ${CREATOR_SHARE_NUM}) / ${CREATOR_SHARE_DEN} AND s.platform_cut_cents = s.gross_backing_cents - s.creator_net_cents) ` +
      "OR (s.product = 'tip' AND s.platform_cut_cents = 0 AND s.creator_net_cents = s.gross_backing_cents)))",
  ),
});

export interface MoneyReconcileResult {
  counts: Record<MoneyIdentity, number>;
  /** Violating rows over every identity (each capped at MONEY_RECONCILE_ROW_CAP). */
  total: number;
  /** Identities checked. */
  checked: number;
}

/** Run every identity. Writes nothing. A failed statement throws (the caller's try logs it). */
export async function reconcileMoneyBooks(db: DbLike, nowMs: number): Promise<MoneyReconcileResult> {
  const nowIso = new Date(nowMs).toISOString();
  const counts = {} as Record<MoneyIdentity, number>;
  let total = 0;
  let checked = 0;
  for (const id of Object.keys(MONEY_RECONCILE_SQL) as MoneyIdentity[]) {
    const binds: unknown[] = id === "reserved_over_cleared" ? [nowIso, MONEY_RECONCILE_ROW_CAP] : [MONEY_RECONCILE_ROW_CAP];
    const row = await db.prepare(MONEY_RECONCILE_SQL[id]).bind(...binds).first<{ n: number }>();
    const n = Number(row?.n ?? 0);
    counts[id] = n;
    total += n;
    checked++;
  }
  return { counts, total, checked };
}

/** The money tail's reconcile leg: on drift, ONE operator-alarm line with counts only. */
export async function runMoneyReconcile(db: DbLike, nowMs: number, cid: string): Promise<MoneyReconcileResult> {
  const r = await reconcileMoneyBooks(db, nowMs);
  if (r.total > 0) logError({ cid, hop: "cron", msg: "money_reconcile_drift", count: r.total, scanned: r.checked });
  return r;
}

/** Groups (one `del_` id each) deleted per run. FAMILY-025: one run a day reaches 365 x this per year; the population
 * is one group per erased buyer plus one per erased creator, which this covers while erasures stay under ~18,000 a

export const MONEY_SWEEP_GROUPS_PER_RUN = 50;

/** A fingerprint carries a UTC DAY; read it as the END of that day so the age is never under-stated. */
const FP_TS = "created_day || 'T23:59:59.999Z'";

/** The cutoff, bound once per statement as `c(v)`. */

/**

  AND NOT EXISTS (SELECT 1 FROM credit_checkout_pending x WHERE x.account_id = ${key} AND MAX(x.created_at, COALESCE(x.resolved_at, '')) >= ${C})
  AND NOT EXISTS (SELECT 1 FROM credit_settlement_refusals x WHERE x.account_id = ${key} AND (MAX(x.at, COALESCE(x.resolved_at, '')) >= ${C} OR x.status = 'refund_owed'))
  AND NOT EXISTS (SELECT 1 FROM sales x WHERE x.account_id = ${key} AND x.at >= ${C})
  AND NOT EXISTS (SELECT 1 FROM sales x WHERE x.passport_id = ${key} AND x.at >= ${C})
  AND NOT EXISTS (SELECT 1 FROM erased_dispute_fingerprints x WHERE x.tombstone_id = ${key} AND x.${FP_TS} >= ${C})
  AND NOT EXISTS (SELECT 1 FROM pending_dispute_fingerprints x WHERE x.tombstone_id = ${key} AND x.${FP_TS} >= ${C})
  AND NOT EXISTS (SELECT 1 FROM creator_payouts x WHERE x.passport_id = ${key} AND (MAX(x.created_at, COALESCE(x.paid_at, ''), COALESCE(x.voided_at, ''), COALESCE(x.returned_at, '')) >= ${C} OR x.status IN ('pending','failed')))
  AND NOT EXISTS (SELECT 1 FROM creator_closures x WHERE x.passport_id = ${key} AND x.closed_at >= ${C})
  AND NOT EXISTS (SELECT 1 FROM creator_dispute_debits x WHERE x.passport_id = ${key} AND x.at >= ${C})
  AND NOT EXISTS (SELECT 1 FROM creator_reserve_losses x WHERE x.passport_id = ${key} AND x.recorded_at >= ${C})

  AND NOT EXISTS (SELECT 1 FROM credit_ledger f WHERE ${OPEN_FREEZE_ON_ACCOUNT_FOR(key)}))`;

  let unsubscribe: string | null = null;
  try {
    const minted = (await mintUnsubscribeTokens(db, "fan_newsletter", [accountId], nowIso)).get(accountId);
    if (minted) {
      await db.prepare("UPDATE newsletter_subscriptions SET unsubscribe_token = ? WHERE subscription_id = ?").bind(minted, subscriptionId).run();
      unsubscribe = unsubscribeUrl(args.publicBase, minted);
    }
  } catch (e) {
    logError({ cid: crypto.randomUUID(), hop: "http", route: "newsletter-subscribe", msg: "unsubscribe mint failed", err: e instanceof Error ? e.name : typeof e });
  }
  const vars: Record<string, string> = { creator: creator.display_name.slice(0, 64), confirm_url: newsletterConfirmUrl(args.appOrigin, passportId, token) };
  if (unsubscribe) vars.unsubscribe_url = unsubscribe;
  let sent = false;
  try {
    sent = await mailer.send("newsletter-confirm", email.address, vars, undefined, unsubscribe ? { listUnsubscribe: unsubscribe } : undefined);
  } catch {
    sent = false;
  }
  if (!sent) return drop();
  return { state: "pending", mailed: true };
}

/** The link the confirmation mail carries: the creator's public page in the app (apps/web newsletter.ts reads the

export function newsletterConfirmUrl(appOrigin: string, passportId: string, token: string): string {
  return `${appOrigin.replace(/\/+$/, "")}/#/p/${passportId}/newsletter/${token}`;

/** The guarded confirm, one statement, so two confirms of one token (back to back or in one batch) confirm once. */
export function newsletterConfirmStatement(db: DbLike, accountId: string, token: string, nowIso: string, cutoffIso: string): DbStatement {
  return db
    .prepare(
      `UPDATE newsletter_subscriptions SET confirmed_at = ? WHERE confirm_token = ? AND account_id = ? AND confirmed_at IS NULL AND requested_at >= ? AND ${newsletterCreatorOpenSql("newsletter_subscriptions.passport_id")} RETURNING passport_id`,
    )
    .bind(nowIso, token, accountId, cutoffIso);

/**
 * The signed-in fan follows the link. 400 bad_token (shape), 404 unknown_token (no row, or another account's), 409
 * used_token (already confirmed), 410 expired_token (the pending row is removed), 404 creator_unavailable (removed).
 */
export async function confirmNewsletterSubscription(db: DbLike, accountId: string, token: string, nowMs: number): Promise<{ state: "confirmed"; passport_id: string }> {
  if (!CONFIRM_TOKEN_RE.test(token)) throw new NewsletterError("bad_token", 400);
  const ttl = await newsletterConfirmTtlMs(db);
  const cutoffIso = new Date(nowMs - ttl).toISOString();
  const done = (await newsletterConfirmStatement(db, accountId, token, new Date(nowMs).toISOString(), cutoffIso).all<{ passport_id: string }>()).results;
  if (done.length === 1) return { state: "confirmed", passport_id: done[0]!.passport_id };
  const row = await db
    .prepare("SELECT subscription_id, account_id, confirmed_at, requested_at FROM newsletter_subscriptions WHERE confirm_token = ?")
    .bind(token)
    .first<{ subscription_id: string; account_id: string; confirmed_at: string | null; requested_at: string }>();
  if (!row || row.account_id !== accountId) throw new NewsletterError("unknown_token", 404);
  if (row.confirmed_at !== null) throw new NewsletterError("used_token", 409);
  await db.prepare("DELETE FROM newsletter_subscriptions WHERE subscription_id = ? AND confirmed_at IS NULL").bind(row.subscription_id).run();
  if (row.requested_at < cutoffIso) throw new NewsletterError("expired_token", 410);
  throw new NewsletterError("creator_unavailable", 404);
}

/** Unconfirmed rows whose token has expired. The only removal of an unconfirmed row besides its use (W0-05(2)). */
export async function pruneNewsletterPending(db: DbLike, nowMs: number): Promise<void> {
  const ttl = await newsletterConfirmTtlMs(db);
  await db.prepare("DELETE FROM newsletter_subscriptions WHERE confirmed_at IS NULL AND requested_at < ?").bind(new Date(nowMs - ttl).toISOString()).run();
}

  if (clean.length === 0 || clean.length !== ids.length || clean.length > NEWSLETTER_CLEAR_MAX) throw new NewsletterError("bad_selection", 400);
  // Chunked under D1's bound-parameter cap (db.ts inChunks; one extra bind, the passport id), one atomic batch.

/** D-1246 (Jason: "12 months"), D-1248(1): the one-click clear removes subscriptions confirmed more than this many
 * calendar months (UTC) before now. A constant, not a setting. */
export const NEWSLETTER_CLEAR_AGE_MONTHS = 12;
/** D-1254(1): the same UTC time, NEWSLETTER_CLEAR_AGE_MONTHS calendar months back, the day clamped to the target
 * month's last day (2028-02-29T10:00Z -> 2027-02-28T10:00Z), never overflowing into the next month. */
export function newsletterClearAgeCutoffIso(nowMs: number): string {
  const d = new Date(nowMs);
  const year = d.getUTCFullYear();
  const month = d.getUTCMonth() - NEWSLETTER_CLEAR_AGE_MONTHS;
  const lastDay = new Date(Date.UTC(year, month + 1, 0)).getUTCDate();
  d.setUTCMonth(month, Math.min(d.getUTCDate(), lastDay));
  return d.toISOString();
}
/** One statement: every CONFIRMED row of this passport confirmed before the cutoff (a row confirmed exactly 12 months
 * ago is kept). Unconfirmed requests are never touched. A confirmed fan whose address is unverified is not on the list
 * (NEWSLETTER_SUBSCRIBERS_SQL) and is cleared too. No audit row, as the selected clear writes none. */
export async function clearNewsletterSubscribersOlderThan(db: DbLike, passportId: string, nowMs: number): Promise<{ cleared: number }> {
  const gone = (
    await db
      .prepare("DELETE FROM newsletter_subscriptions WHERE passport_id = ? AND confirmed_at IS NOT NULL AND confirmed_at < ? RETURNING subscription_id")
      .bind(passportId, newsletterClearAgeCutoffIso(nowMs))
      .all()
  ).results;
  return { cleared: gone.length };
}

import { canonicalAccount, canonicalize } from "@dhwaja/canonical";
import type { DbLike } from "./db.js";
import { consumeRateLimit } from "./ratelimit.js";
import { auditStmt } from "./review.js";
import { mintId } from "./ids.js";
import { legalHoldOpenSql } from "./legal-hold-sql.js";
import { MAX_REQUESTS_PER_ACCOUNT } from "./requests.js";
// THE "has this subject asked to be removed?" read — one statement, two legs
// (the `suppressed_account_keys` snapshot ∪ the live join). Shared with the
// importer and the report shield so the three can never disagree.
import { suppressedKeyStmt } from "./seeding.js";

/**
 * flag 4 — fan nominations: the suggestion box, never the printing press.
 *
 * A fan nomination creates an OPERATOR QUEUE ITEM and NEVER a passport row.
 * That is the whole design. Operators still source-check and mint every
 * listing, which preserves the product guarantee ("every listing passed a
 * human check") and structurally blocks the killer attack: a scammer
 * nominating their own fake as "the known original" and laundering it into
 * the registry. Like fan reports, a nomination is a plain queue row — NOT a
 * trust event, no new event type, nothing in the fold.
 *
 * Three cases collapse to ONE response body, `{ok:true}`:
 *   1. new/repeat nomination  → queued (counter bumped for a distinct fan)
 *   2. already listed         → nothing written (no listedness oracle: a
 *                               probe must not learn whether we already
 *                               carry a creator by nominating them)
 *   3. subject SUPPRESSED     → accept-and-drop, silently. The "never
 *                               re-seeded" promise must not be probeable —
 *                               a removal has to be indistinguishable from
 *                               an ordinary queued nomination, or the
 *                               nomination endpoint becomes a "did this
 *                               person ask to be delisted?" oracle.
 *
 * The nominator count is the operator's prioritization signal and is never
 * public, so there is no status to farm. It is DERIVED at read time (never a
 * stored counter) and shown next to a distinct-masked-prefix count, so a
 * sybil-inflated spike is visible rather than merely bounded.
 */

export class NominationError extends Error {
  override name = "NominationError";
  constructor(
    message: string,
    public readonly status: 400 | 401 | 409 | 429 = 400,
  ) {
    super(message);
  }
}

/** Per-fan daily cap. Generous for a real fan (nominating ten creators a day
 * is already unusual), tight enough that one account cannot seed a queue. */
export const NOMINATE_CAP_PER_DAY = 10;
/** Per-IP hourly cap, applied at the route (mirrors the fan-auth surfaces):
 * caps a single address driving many fresh accounts. */
export const NOMINATE_CAP_PER_IP_HOUR = 30;

export const NOMINATION_STATES = ["open", "listed", "declined"] as const;
export type NominationState = (typeof NOMINATION_STATES)[number];

/**
 * File one nomination. Returns nothing: every non-error path is the SAME
 * constant response at the route, so this function deliberately reports no
 * outcome the caller could leak.
 *
 * Throws only for a client-fixable condition (unsupported link) or the
 * throttle — neither of which reveals anything about the subject.
 */
export async function fileNomination(
  db: DbLike,
  accountId: string,
  input: { url?: string; platform?: string; id?: string; ipPrefix?: string | null },
  nowMs: number,
  now: () => string = () => new Date().toISOString(),
): Promise<void> {
  // CHIP2 (QUEUE #101 docket #55, ruling (a)): two input shapes, ONE resolver.
  // A platform+id pair (the bare-handle lookup shape apps/web/src/registry.ts
  // checkAccount already uses) and a profile url converge to the SAME
  // canonical account key through @dhwaja/canonical ONLY — canonicalAccount /
  // canonicalize — never a second parser (tests/invariants/single-hasher.test.ts).
  // A pair takes precedence when both are present; a malformed request (neither
  // shape complete) is refused naming what IS accepted, not a generic 400.
  const hasPair = typeof input.platform === "string" && typeof input.id === "string";
  let subject: ReturnType<typeof canonicalize>;
  if (hasPair) {
    try {
      subject = canonicalAccount(input.platform as string, "handle", input.id as string);
    } catch {
      throw new NominationError("that platform and handle are not a supported profile", 400);
    }
  } else if (typeof input.url === "string") {
    try {
      subject = canonicalize(input.url);
    } catch {
      throw new NominationError("that link is not a supported profile", 400);
    }
  } else {
    throw new NominationError("a profile link, or a platform and handle, is required", 400);
  }
  // A bare domain / un-routable URL is not a nominatable creator profile:
  // "list this person" needs a platform account we can source-check. I15-Q16
  // (D-1030, Jason: "Allow, to match reports and contests"): a generic `link`
  // (M3 Slice 4, `v1|link|url|<norm>`) IS nominated, with the link as the
  // subject — it rides the same constant-work path as a platform account.
  if (subject.platform === "url" || subject.idKind === "norm") {
    throw new NominationError("that link is not a supported profile", 400);
  }

  // Throttle BEFORE any read, so the endpoint cannot be used as a cheap
  // enumeration probe even by an authenticated fan.
  if (!(await consumeRateLimit(db, `nom:${accountId}`, nowMs, NOMINATE_CAP_PER_DAY))) {
    throw new NominationError("nomination limit reached for today", 429);
  }

  const key = subject.canonical;

  // ---- CONSTANT WORK FROM HERE ON ------------------------------------------
  // Equal response bodies are not enough. Listedness is already public via the
  // k-anon lookup, so if the drop paths did LESS work than the queued path, ONE
  // authenticated request against a named, known-listed person would separate
  // "suppressed" from "genuinely new" by latency alone — precisely the "did
  // they ask to be delisted?" oracle this module exists to prevent (caps bound
  // mass enumeration, never targeted probing).
  //
  // So: both lookups ALWAYS run, the queue read ALWAYS runs, and the drop paths
  // spend the same three writes as the queued path against an absorbing
  // rate-limit key. Every outcome is 3 reads + 4 writes = 7 statements.
  // (Asserted in nominations.test.ts by counting prepared statements.)

  // I3-0045 (I22-FX-MONEY): the CURRENT binding only. `historical` is what a `revoke_account` correction leaves
  // (events.ts revoke_account: the handle was wrongly attached), so admitting it filed the fan's ask as demand for the
  // passport the handle was corrected OFF.
  const listed = await db
    .prepare(
      `SELECT p.passport_id, p.claim_state FROM passport_accounts a JOIN passports p ON p.passport_id = a.passport_id
       WHERE a.account_key = ? AND a.status = 'active' AND p.retracted = 0 LIMIT 1`,
    )
    .bind(key)
    .first<{ passport_id: string; claim_state: string }>();

  // SUPPRESSED: unfiltered by account status and by retraction on purpose —
  // any historical link from this key to a suppressed passport means that
  // person asked to leave, and the seeding side must never re-list them.
  //
  // ONE statement, TWO legs (`seeding.ts` `suppressedKeyStmt`): the
  // `suppressed_account_keys` snapshot frozen at suppression time, UNION the
  // live join this used to run alone. The live join hangs off
  // `passport_accounts.passport_id`, which `account_transferred` MOVES — one
  // transfer off the delisted passport used to make this shield stop firing, so
  // a person who had asked to leave became nominatable again (audit HIGH,
  // 2026-08-26). Still one statement, so the 3 reads + 4 writes every outcome
  // spends is unchanged.
  const suppressed = await suppressedKeyStmt(db, key).first<{ x: number }>();

  // RETRACTED (U-05 fix): `listed` above requires `p.retracted = 0`, so a
  // retracted passport reads as "not listed" rather than "must not be
  // re-seeded" — and `correction/retract_passport` writes neither
  // `suppressed_account_keys` nor `request_suppressions`, so `suppressed`
  // above misses it too. A FOURTH read, unconditional on every outcome, so the
  // constant-work property (`nominations.test.ts` "CONSTANT WORK") holds at 4
  // reads + 4 writes instead of 3 + 4.
  const retracted = await db
    .prepare(
      `SELECT 1 AS x FROM passport_accounts a JOIN passports p ON p.passport_id = a.passport_id
       WHERE a.account_key = ? AND p.retracted = 1 LIMIT 1`,
    )
    .bind(key)
    .first<{ x: number }>();

  // ALREADY LISTED → the nomination has no work to do (and recording it would
  // leak listedness into the operator queue). SUPPRESSED or RETRACTED → never
  // re-seeded. All drop, all indistinguishably.
  const drop = listed !== null || suppressed !== null || retracted !== null;

  const ts = now();
  const existing = await db.prepare("SELECT nomination_id FROM nominations WHERE account_key = ?").bind(key).first<{ nomination_id: string }>();

  // WRITE 1 — mint the queue row, or absorb.
  let nominationId: string | null = drop ? null : (existing?.nomination_id ?? null);
  if (!drop && existing === null) {
    const id = mintId("nom");
    try {
      // RG2FR-01 (I3-1298): bound to the nominator's account existing, like
      // WRITE 2 — an erasure landing first mints no queue row (changes 0).
      const w = await db
        .prepare("INSERT INTO nominations (nomination_id, account_key, platform, handle, first_nominated_at, last_nominated_at) SELECT ?, ?, ?, ?, ?, ? WHERE EXISTS (SELECT 1 FROM accounts WHERE account_id = ?)")
        .bind(id, key, subject.platform, subject.id, ts, ts, accountId)
        .run();
      nominationId = w.changes > 0 ? id : null;
    } catch {
      // Concurrent first-nomination: the UNIQUE(account_key) index collapses
      // the race and the loser re-reads the winner's row. (The one path whose
      // statement count differs — a self-inflicted race, not a subject signal.)
      const raced = await db.prepare("SELECT nomination_id FROM nominations WHERE account_key = ?").bind(key).first<{ nomination_id: string }>();
      if (!raced) throw new NominationError("could not record the nomination", 409);
      nominationId = raced.nomination_id;
    }
  } else {
    await absorb(db, accountId, nowMs);
  }

  // WRITE 2 — ONE FAN COUNTS ONCE: the voters PK makes a repeat nomination
  // idempotent, so the derived count never inflates from a fan tapping twice.
  // I3-1298 (I4-08-R9): bound to the account's lifetime (`EXISTS accounts`), so
  // an erasure landing before this write leaves no vote behind.
  if (nominationId !== null) {
    await db
      .prepare("INSERT OR IGNORE INTO nomination_voters (nomination_id, account_id, ip_prefix, created_at) SELECT ?, ?, ?, ? WHERE EXISTS (SELECT 1 FROM accounts WHERE account_id = ?)")
      .bind(nominationId, accountId, input.ipPrefix ?? null, ts, accountId)
      .run();
  } else {
    await absorb(db, accountId, nowMs);
  }

  // WRITE 3 — freshness only. State is NEVER touched by fan input: a declined
  // item stays declined however many nominations arrive, so volume cannot
  // overturn an operator's decision.
  if (nominationId !== null) {
    await db
      .prepare("UPDATE nominations SET last_nominated_at = ? WHERE nomination_id = ? AND EXISTS (SELECT 1 FROM accounts WHERE account_id = ?)")
      .bind(ts, nominationId, accountId)
      .run();
  } else {
    await absorb(db, accountId, nowMs);
  }

  // WRITE 4 — I3-0045 (D-0981(1)): an ask for a LISTED, UNCLAIMED creator is recorded as a creator request
  // (requests.ts requestCreator's gates: live, unsuppressed, unclaimed, under MAX_REQUESTS_PER_ACCOUNT), never
  // discarded. ONE statement on EVERY outcome, gated by a bound flag, so the constant-work property above holds
  // and the body is unchanged: a fresh, suppressed or claimed subject runs the same INSERT and it inserts nothing.
  const requestable = listed !== null && listed.claim_state === "unclaimed" && suppressed === null && retracted === null;
  await db
    .prepare(
      `INSERT INTO creator_requests (account_id, passport_id, created_at)
       SELECT ?, p.passport_id, ? FROM passports p
        WHERE ? = 1 AND p.passport_id = ? AND p.claim_state = 'unclaimed' AND p.retracted = 0
          AND NOT EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = p.passport_id)
          AND EXISTS (SELECT 1 FROM accounts WHERE account_id = ?)
          AND (SELECT COUNT(*) FROM creator_requests WHERE account_id = ?) < ?
       ON CONFLICT(account_id, passport_id) DO NOTHING`,
    )
    .bind(accountId, ts, requestable ? 1 : 0, requestable ? listed.passport_id : "", accountId, accountId, MAX_REQUESTS_PER_ACCOUNT)
    .run();
}

 * returns nothing rather than silently listing everything.

export async function nominationQueue(
  db: DbLike,

  const offset = Math.max(0, Math.min(100_000, Math.floor(opts.offset ?? 0) || 0));

    .prepare(
      `SELECT n.nomination_id, n.platform, n.handle, n.account_key, n.state, n.first_nominated_at, n.last_nominated_at,

    )

    .all<NominationRow>();

/**
 * OAuth claim flow. start → authorize URL (state + PKCE stored); the platform
 * redirects the user back to callback → we verify+consume the state, exchange
 * the code, fetch the identity, and claim the passport ONLY if the identity is
 * the target account (identityMatchesTarget). Same claim_proven write and
 * single-claim lock as the bio-code path.
 */

export const OAUTH_TTL_MS = 15 * 60 * 1000; // 15 min to complete the dance
export const HANDOFF_TTL_MS = 2 * 60 * 1000; // 2 min: the console exchanges the code immediately on return

function randToken(bytes: number): string {
  return [...crypto.getRandomValues(new Uint8Array(bytes))].map((b) => b.toString(16).padStart(2, "0")).join("");
}

// --- one-time session handoff (OAuth callback → console) ----------------------
// The callback claims the account server-side, then hands the CONSOLE a session
// without ever putting a session token in a redirect URL. It stashes a single-
// use, short-TTL, 256-bit code bound to the passport id here and redirects the
// console with just that code; the console POSTs it to /claim/oauth/exchange,
// which mints a fresh session and deletes the code. An expired/used/unknown code
// yields nothing (fail closed).

const HANDOFF_CODE_RE = /^[0-9a-f]{64}$/;

/** Mint a one-time handoff code for a just-claimed passport, BOUND to the OAuth
 * `state` the initiating browser holds. Only an exchange presenting the matching
 * state can redeem it, so a leaked code can't sign anyone else in (anti login-CSRF). */
export async function createOAuthHandoff(db: DbLike, passportId: string, state: string, now: () => string = () => new Date().toISOString(), nowMs: number = Date.now()): Promise<string> {
  const code = randToken(32); // 64 hex = 256 bits
  const expires = new Date(nowMs + HANDOFF_TTL_MS).toISOString();
  await db.prepare("INSERT INTO oauth_handoffs (code, passport_id, flow_state, created_at, expires_at) VALUES (?, ?, ?, ?, ?)").bind(code, passportId, state, now(), expires).run();
  return code;
}

/** Redeem a handoff code: returns the passport id only when the code is valid,
 * unexpired, AND the presented `state` matches the one bound at mint (so only the
 * browser that started the flow can redeem — a leaked code alone is useless).
 * The code is consumed on ANY found+unexpired attempt (one shot per code), so a
 * wrong-state attempt burns the code rather than leaving it open to repeated
 * probing; two racing exchanges can't both win (the DELETE's `changes===1`). */
export async function redeemOAuthHandoff(db: DbLike, code: unknown, state: unknown, nowMs: number = Date.now()): Promise<string | null> {
  if (typeof code !== "string" || !HANDOFF_CODE_RE.test(code) || typeof state !== "string" || state.length === 0) return null;
  const nowIso = new Date(nowMs).toISOString();
  const row = await db.prepare("SELECT passport_id, flow_state FROM oauth_handoffs WHERE code = ? AND expires_at > ?").bind(code, nowIso).first<{ passport_id: string; flow_state: string }>();
  if (!row) {
    await db.prepare("DELETE FROM oauth_handoffs WHERE code = ?").bind(code).run(); // reap an expired/unknown row
    return null;
  }
  const del = await db.prepare("DELETE FROM oauth_handoffs WHERE code = ?").bind(code).run(); // consume: one attempt per code
  if (del.changes !== 1) return null; // lost the single-use race
  return row.flow_state === state ? row.passport_id : null; // wrong state → consumed, no session
}

/** Best-effort reaping of expired handoff codes (called opportunistically). */
export async function pruneOAuthHandoffs(db: DbLike, nowMs: number = Date.now()): Promise<void> {
  await db.prepare("DELETE FROM oauth_handoffs WHERE expires_at <= ?").bind(new Date(nowMs).toISOString()).run();
}

/**
 * How long a settled `oauth_challenges` row is kept before it is dropped.
 * Matches `claims.ts` CHALLENGE_RETENTION_MS deliberately — the two tables are
 * the two halves of one funnel and an operator reading either should not have
 * to hold two numbers.
 */
export const OAUTH_CHALLENGE_RETENTION_MS = 30 * 24 * 60 * 60 * 1000;

/**
 * Cron hygiene for `oauth_challenges` — the twin of `pruneClaimChallenges`, and
 * for the same reason: this table held a SINGLE-USE SECRET indefinitely past
 * the moment it could ever be used again.
 *
 * `code_verifier` is the PKCE verifier. It is read in EXACTLY one place — the
 * `status = 'issued'` row `completeOAuthClaim` loads, immediately before it
 * flips that row to 'spent' — so from the instant a row leaves 'issued', its
 * verifier is dead weight that nothing in the product will ever read again. It
 * was nonetheless kept forever, including on the `proven` rows that are KEPT
 * forever on purpose, so the steady state of a working registry was a growing
 * pile of retired OAuth secrets sitting next to the account they authorized.
 *
 * Three statements, in this order:
 *
 *  1. live rows past their TTL become 'expired' — `completeOAuthClaim` already
 *     refuses them (`outcome: "expired"`), so this only makes the table agree
 *     with the decision the route has been making all along;
 *  2. the verifier is BLANKED on every row that is no longer 'issued'. This is
 *     the point of the sweep: it reaches `proven` rows, which are never deleted;
 *  3. settled non-`proven` rows past the retention window are dropped, so an
 *     anonymous start leg cannot grow the table without bound.
 *
 * `proven` is deliberately absent from (3) at every age: that row IS the control
 * proof `isAccountProven`, `registry-build.ts`, `removal.ts` and `reports.ts`
 * read, and all four read `account_key` + `status`, never the verifier.
 *
 * Dropping a 'spent' row does not reopen a replay: the single-use guard is the
 * `claim_consumed` row keyed `oauth:<state>`, which outlives this table, and a
 * state whose challenge row is gone fails the `!chg` check first anyway.
 */
export async function pruneOAuthChallenges(db: DbLike, nowMs: number = Date.now()): Promise<void> {
  const nowIso = new Date(nowMs).toISOString();
  await db.prepare("UPDATE oauth_challenges SET status = 'expired' WHERE status = 'issued' AND expires_at <= ?").bind(nowIso).run();
  await db.prepare("UPDATE oauth_challenges SET code_verifier = '' WHERE status <> 'issued' AND code_verifier <> ''").run();
  await db
    .prepare("DELETE FROM oauth_challenges WHERE status IN ('spent','expired','failed') AND created_at < ?")
    .bind(new Date(nowMs - OAUTH_CHALLENGE_RETENTION_MS).toISOString())
    .run();
}

 * operator's detail view use it to say a recycled handle may be claiming a creator's passport. NULL when the item has no
 * recorded provider id (`pid` null or empty: an item queued before the id was recorded) — unknown, not "does not differ"

  db: DbLike,
  env: Record<string, unknown>,
  platform: string,

   * is anonymous, so the daily cap below is keyed per (account, caller). */
  caller: string | null,
  now: () => string = () => new Date().toISOString(),
  nowMs: number = Date.now(),
  /** D-1060(3): the minimum age the creator confirmed at the door (claim-routes.ts, `ageAckMatches`); absent reads the
   * `creator_min_age` setting that door checked. Stored on the challenge INSERT (migration 0267). */
  minAge?: number,

  const provider = PROVIDERS[platform];
  const cfg = oauthConfigFromEnv(env, platform);
  if (!provider || !cfg) throw new ClaimError("OAuth is not available for this platform", 409);

  // M39: fold the hand-typed account key to canonical before lookup, the daily
  // cap, and the stored challenge — the same free-text field as the bio-code path.
  // M-03: that field also accepts a profile URL now (resolveClaimAccountKey,
  // claims.ts) — throws AccountKeyFormatError on an unresolvable one, which the
  // route maps to a 400 naming the accepted shapes.
  const accountKey = resolveClaimAccountKey(rawAccountKey);

    .prepare(

       FROM passport_accounts a JOIN passports p ON p.passport_id = a.passport_id
       WHERE a.account_key = ? AND a.passport_id = ? AND a.platform = ?`,
    )

    .first<{ status: string; claim_state: string; retracted: number }>();

  if (!acct || (await accountIsShielded(db, passportId, accountKey))) throw new ClaimError("no such account on that passport", 404);

  // per account alone (adversarial review, LAUNCH-BLOCKING — the class, not just
  // `/claim/start`). This leg is anonymous and both of its inputs are published
  // by `GET /p/:id/json`, so an account-only cap is a budget a stranger spends on
  // the victim's behalf: five anonymous requests and the creator's OAuth claim
  // answered 409 for the rest of the day. That mattered more here than anywhere
  // else, because `x` is the only platform with an OAuth path — a creator whose
  // bio-code-verifiable accounts were also wedged had no self-serve route left.

  // The counter is `rate_limits`, not a COUNT over `oauth_challenges`: it needs a

  const state = randToken(24);
  const codeVerifier = randToken(48); // 96 hex chars — within PKCE 43..128

      )

}

/** I3-1721: the account-proof completion. Runs only after the state was spent, the code exchanged and
 * `identityMatchesTarget` (with the recorded provider subject) passed. ONE statement: the spent->proven flip,
 * dated and stamped like the claim path's, whose VALUE re-asks the evidence (the account still active on THIS
 * passport, the passport still claimed and not retracted) — a withdrawn support yields NULL into the NOT NULL
 * column and the statement fails, the same in-statement abort the claim path uses (BF-03). No ledger event,
 * like the bio-code and DNS account proofs (the proven row is the proof), and no handoff. */
async function completeOAuthAccountProof(db: DbLike, state: string, pid: string, nowMs: number): Promise<OAuthCompleteResult> {
  let changed = 0;
  try {
    const r = await db
      .prepare(
        "UPDATE oauth_challenges SET status = (SELECT CASE WHEN EXISTS (" +
          "SELECT 1 FROM passport_accounts a JOIN passports p ON p.passport_id = a.passport_id" +
          " WHERE a.account_key = oauth_challenges.account_key AND a.passport_id = oauth_challenges.passport_id" +
          " AND a.status = 'active' AND p.retracted = 0 AND p.claim_state <> 'unclaimed') THEN 'proven' END)" +
          ", proved_at = ?, method = 'oauth', provider_pid = ? WHERE state = ? AND status = 'spent' AND purpose = 'account_proof'",
      )
      .bind(new Date(nowMs).toISOString(), pid, state)
      .run();
    changed = Number(r.changes ?? 0);
  } catch (e) {
    if (supportWithdrawn(e)) return { ok: false, outcome: "support_withdrawn" };
    throw e;
  }
  return changed === 1 ? { ok: true, outcome: "account_proven" } : { ok: false, outcome: "expired" };
}

/** True iff the claim was refused by the liveness predicate on the proving
 * statement below (:362) — the supporting account was detached, re-pointed or
 * deactivated during the provider round trip. A NARROW regex on the one column,
 * the twin of `isUnaccountableCharacter` (events.ts:1454): `UNIQUENESS_LOST_RE`
 * (events.ts:365) matches UNIQUE only, so this error reaches the caller's catch
 * unwrapped rather than arriving as a conflict refusal. */
function supportWithdrawn(e: unknown): boolean {
  return e instanceof Error && /NOT NULL constraint failed:\s*oauth_challenges\.status/i.test(e.message);
}

export interface OAuthCompleteResult {
  ok: boolean;
  /** `account_proven` (I3-1721): an account-proof ceremony proved its account; it carries NO passport_id, so
   * no caller can mint a sign-in handoff from it. */
  outcome: "verified" | "identity_mismatch" | "exchange_failed" | "expired" | "bad_state" | "support_withdrawn" | "pending_review" | "account_proven";
  passport_id?: string;
  event_id?: string;
  /** `pending_review` only: the bearer ref for the review (manual-claims.ts). */
  claim_ref?: string;
  /** I3-1078: the one-time console handoff code, minted IN the claim's batch
   * (only when `mintHandoff` was asked and this call committed the claim). */
  handoff?: string;
}

interface ChallengeRow {
  state: string;
  passport_id: string;
  account_key: string;
  platform_key: string;
  code_verifier: string;
  status: string;
  expires_at: string;
  /** 0219 (I3-1721): 'claim' | 'account_proof'. */
  purpose?: string;
}

export async function completeOAuthClaim(
  db: DbLike,
  env: Record<string, unknown>,
  platform: string,
  code: string,
  state: string,
  now: () => string = () => new Date().toISOString(),
  nowMs: number = Date.now(),
  fetchImpl: typeof fetch = fetch,
  /** QUEUE #19: `publishHook(...)`'s thunk, run once after the batch commits. */
  afterCommit?: () => void,
  /** I3-1078: mint the console handoff code in the SAME batch as the claim, so a
   * claim can never commit without the handoff that lets its creator sign in. */
  mintHandoff = false,
): Promise<OAuthCompleteResult> {
  const provider = PROVIDERS[platform];
  const cfg = oauthConfigFromEnv(env, platform);
  if (!provider || !cfg) throw new ClaimError("OAuth is not available for this platform", 409);
  if (!/^[0-9a-f]{1,128}$/.test(state)) return { ok: false, outcome: "bad_state" };

  // `account_key <> 'fanconn'` — THE CREATOR PATH NEVER TOUCHES A FAN LINK'S
  // ROW. F8 (`fan-connections.ts`) reuses this table for fan platform links, so
  // it inherits the cron hygiene below rather than needing a second sweep; its
  // rows are marked by the `fanconn` sentinel in `account_key`. This function
  // SPENDS a row — `claim_consumed` plus `issued → spent` — before it checks
  // any identity, so without this clause a creator callback carrying a fan's
  // state would kill that fan's in-flight link. There is no escalation in it
  // (the identity check fails against the sentinel either way), but a feature
  // that silently stops working is worse than one that fails loudly. The two
  // provider maps are also asserted disjoint at module load in
  // `fan-connections.ts`; this is the second lock, because the day somebody
  // adds Instagram to `PROVIDERS` above is the day the first one matters.
  const chg = await db
    .prepare("SELECT * FROM oauth_challenges WHERE state = ? AND platform_key = ? AND account_key <> 'fanconn'")
    .bind(state, platform)
    .first<ChallengeRow>();
  if (!chg || chg.status !== "issued") return { ok: false, outcome: "bad_state" };
  // I3-2053 a (I22-W1-AUTH): an expired state is settled `expired` and answered BEFORE the spend batch, so it
  // writes no `claim_consumed` row and burns nothing; the later check below stays for a state that expires
  // between this line and the spend.
  if (Date.parse(chg.expires_at) < nowMs) {
    await db.prepare("UPDATE oauth_challenges SET status = 'expired' WHERE state = ? AND status = 'issued'").bind(state).run();
    return { ok: false, outcome: "expired" };
  }

  // ATOMIC single-use SPEND, before the exchange. Inserting the state into
  // claim_consumed and flipping issued->spent ride ONE batch; a concurrent or
  // replayed callback with the same state violates the primary key, its batch
  // rolls back, and it never reaches the token exchange. This serializes at the
  // DB (no rows-affected check needed) and prevents double-EXCHANGE, not just
  // double-claim — which is why it cannot move into the ledger batch below.
  //
  // 'spent' IS NOT 'proven'. It only records that the token was burnt. Nothing
  // reads it as a control proof; see the status split in 0003 and `isAccountProven`.
  const consumeTs = now();
  let spentRows = 0;
  try {
    const res = await db.batch([
      db.prepare("INSERT INTO claim_consumed (challenge_id, consumed_at) VALUES (?, ?)").bind(`oauth:${state}`, consumeTs),
      db.prepare("UPDATE oauth_challenges SET status = 'spent' WHERE state = ? AND status = 'issued'").bind(state),
    ]);
    // `DbLike.batch` is typed `unknown[]` deliberately: D1 reports rows-affected
    // per element at `meta.changes` and the better-sqlite3 double reports it
    // top-level, so the shape is read here exactly the way `db.ts`'s own D1Stmt
    // adapter reads it — accept either, without runtime type-sniffing. Safe on
    // this table specifically: `meta.changes` is a `total_changes()` DELTA that
    // AFTER triggers inflate, and the four trigger-bearing tables are
    // `fan_email`, `fan_passkeys`, `oauth_identities` and `acquisitions` —
    // `oauth_challenges` carries none, so one row written reports exactly 1.
    const spend = res?.[1] as { meta?: { changes?: number }; changes?: number } | undefined;
    spentRows = Number(spend?.meta?.changes ?? spend?.changes ?? 0);
  } catch (e) {
    if (e instanceof Error && /UNIQUE|constraint/i.test(e.message)) return { ok: false, outcome: "bad_state" };
    throw e;
  }
  // THE ACQUISITION IS ONLY ACQUIRED IF IT CHANGED A ROW (S-070 SJ-15).
  //
  // A conditional UPDATE matching zero rows does NOT throw — it succeeds having
  // changed nothing — so the batch above committed its permanent
  // `claim_consumed` tombstone even when the `issued -> spent` half hit nothing.
  // `pruneOAuthChallenges` flips every past-`expires_at` row `issued -> expired`
  // on the ordinary cron tick, and a callback admitted just before that instant
  // carries an entry-time clock still inside the TTL: the expiry check below
  // passes, the provider exchange succeeds, and the `claim_proven` event commits
  // with its own companion `... WHERE status = 'spent'` ALSO matching zero rows
  // and ALSO not guarding anything. The result was a legitimate, successful
  // claim whose control proof did not exist — `status` still `expired`, excluded
  // by every current-proof reader (`account-standing.ts`), and deleted outright
  // by retention. The account was added and unserved and needed another ceremony
  // nobody knew to run.
  //
  // ASKED OF THE WRITE, NOT OF A RE-READ (S-070 W2). The first fix confirmed the
  // spend with `SELECT 1 ... WHERE state = ? AND status = 'spent'`, which is a
  // correct acquisition check and an INDISTINGUISHABLE ONE: `ledger-contention`
  // scans this whole service for any read of `oauth_challenges` keyed on a
  // status other than 'proven', because a reader added tomorrow that treats
  // 'spent' as control is the conflation that guard exists to stop. That scan
  // cannot see the difference between "did my own UPDATE land" and "has this
  // account proved control", and it should not have to: the two are told apart
  // by asking the WRITE how many rows it changed, which is the fact actually
  // wanted and costs no second read. D1 answers `meta.changes` per batch
  // element, and the double was made to answer the same shape (AE-05).
  //
  // The tombstone stays committed, which is correct — the state is burnt and
  // must never be replayable — and the ceremony is refused as what it is.
  //
  // RESIDUAL, STATED: the spent->proven flip below is still an unguarded
  // conditional update. It is safe only because nothing else moves a row OUT of
  // `spent` — the sweep expires `issued` rows only, and retention deletes by an
  // age this row cannot have reached inside one ceremony. That is an argument
  // about the current writer set, not a guard; a future writer that settles a
  // `spent` row re-opens this exact shape one line down.
  if (spentRows !== 1) return { ok: false, outcome: "expired" };

  if (Date.parse(chg.expires_at) < nowMs) {
    return { ok: false, outcome: "expired" };
  }

  const identity = await exchangeAndIdentify(provider, cfg, code, chg.code_verifier, fetchImpl);
  if (!identity) return { ok: false, outcome: "exchange_failed" };

  // I3-1953 (D-0920): PROVIDER-IDENTITY CONTINUITY. A handle-kind account key
  // is satisfied by whoever holds that handle TODAY (identityMatchesTarget), and
  // handles are recycled (INV-015). Once a provider identity has proven this
  // account, a later OAuth proof of the same key must be the SAME provider
  // subject (`provider_pid`, recorded on every proven challenge since migration
  // 0123): a different subject holding the recycled handle is not the account
  // that was proven. Answered as `identity_mismatch` — the account authorized is
  // not the account on record — before any write, so the challenge stays burnt
  // like every other refusal on this door. I3-0113 (D-0920 option d): the
  // recorded pid is an ARGUMENT of THE guard below, so the pid rule lives in
  // identityMatchesTarget itself rather than beside its one caller.
  const priorPid = await db
    .prepare(
      "SELECT provider_pid FROM oauth_challenges WHERE account_key = ? AND status = 'proven' AND provider_pid IS NOT NULL ORDER BY proved_at DESC LIMIT 1",
    )
    .bind(chg.account_key)
    .first<{ provider_pid: string }>();

  // THE guard: the authenticated identity must be the target account — and,
  // once the account has been OAuth-proven, the same provider subject.
  if (!identityMatchesTarget(platform, identity, chg.account_key, priorPid?.provider_pid ?? null)) {
    return { ok: false, outcome: "identity_mismatch" };
  }

  // I3-1721: an account-proof state never reaches the claim path below (no claim_proven, no handoff).
  if (chg.purpose === "account_proof") return completeOAuthAccountProof(db, state, identity.pid, nowMs);

  // THE GATE, RE-ASKED — the same call, on the same const, at the same point in
  // the ceremony as the bio-code door (`claims.ts:1398`) and the DNS door
  // (`domain-claims.ts:869`). `startOAuthClaim` decided this before the browser
  // left for the provider (:150-162) and the challenge row is a record of that
  // decision, not of the world.
  //
  // WHAT IT ADDS, MEASURED CLAUSE BY CLAUSE rather than asserted. Four of
  // `stillClaimable`'s five clauses are already refused on this door by
  // something else: transfer and deactivation by the liveness predicate on the
  // proving statement below (BF-03, :400-410), retraction by
  // `requireLivePassport` (events.ts:1544), a concurrent claim by "claim_proven
  // requires unclaimed" (events.ts:909), and an unaccountable character by
  // `accountablePassport` (events.ts:1446). The clause NONE of them carries is
  // SUPPRESSION — `seeding.ts suppressedKeyStmt` — and `claims.ts:1239-1246`
  // says why it cannot ride the batch: it is deliberately a BOUND statement, so
  // a half-bound copy of it as a subquery would fail OPEN, and the fold does
  // not ask it at all. A hand-rolled second copy is the exact drift that helper
  // exists to prevent. So this door, and only this door, claimed a passport
  // whose subject had asked to be delisted, or whose account key had been
  // frozen by a removal, while its two siblings refused the same ceremony.
  //
  // ASKED HERE, AFTER THE IDENTITY CHECK, for the reason the other two doors
  // give: reaching this line means control of the account has just been
  // demonstrated, and an earlier or a more specific refusal would be a
  // listedness oracle on an anonymous route.
  //
  // THE STATE STAYS BURNT. The `claim_consumed` tombstone written above is
  // permanent and must be — a refused ceremony is not a replayable one.
  //
  // IT REFUSES THROUGH `support_withdrawn`, NOT A THROW AND NOT A NEW MEMBER,
  // and that is three decisions, each measured:
  //
  //  1. NOT a throw. `ClaimError(NOT_CLAIMABLE, 409)` is what the two sibling
  //     doors raise, and raising it here turned W3-08's shipped `FIXED:` guards
  //     red — `w308a-r3-oauth-support-liveness.test.ts:97,:105` assert the
  //     callback RETURNS `ok: false`, so a refusal that escapes as an exception
  //     breaks the channel they defend while leaving the property intact.
  //  2. NOT a new `outcome` member. The member would need a mapping in
  //     `claim-routes.ts:400-444`, which is another room's file this round.
  //  3. `support_withdrawn` SPECIFICALLY, because the refusal must be UNIFORM.
  //     `claims.ts:1160-1170` is explicit that naming WHICH clause failed is
  //     what puts the delisting bit back on the wire, and that is the one bit
  //     this gate is here to protect. Every clause of `stillClaimable` now
  //     leaves this door by the one sentence, which is what `NOT_CLAIMABLE`
  //     buys on the doors that can throw. The cost is that `support_withdrawn`
  //     now has a SECOND producer, and it is no longer only the NOT NULL
  //     constraint `supportWithdrawn` (:193-199) matches — hence this comment
  //     and the `||` that follows, rather than a quiet second `return`.
  //
  // IT DOES NOT REPLACE BF-03, and must not be read as doing so. This is a READ
  // before the write, so it is raced by construction for the clauses that have
  // an in-batch guard; those guards stay the deciders of record and are still
  // exercised (`w308a-r3-oauth-support-liveness.test.ts:113` drives a move that
  // lands AFTER this read and is still refused). What this read adds that no
  // in-batch guard can is the suppression clause above.
  if (!(await stillClaimable(db, chg.passport_id, chg.account_key, nowMs))) {
    return { ok: false, outcome: "support_withdrawn", passport_id: chg.passport_id };
  }

  // I3-1659 / #527 (D-0810(4)): CONTINUITY, the bio-code door's rule
  // (claims.ts claimNeedsReview, verifyClaim) asked at the same point — after
  // the proof, before the write. An old account on an operator-seeded passport
  // goes to an operator instead of `claim_proven`; the burnt state stays burnt.
  if (await claimNeedsReview(db, chg.passport_id, chg.account_key, nowMs)) {
    const ts = now();
    const claimRef = crypto.randomUUID().replace(/-/g, "") + crypto.randomUUID().replace(/-/g, "");
    const itemId = mintId("itm");
    // I38-OAUTHREV (INV-015): the provider's stable subject id rides the review item, so the approval can pin it
    // (review.ts approveItem) and the operator sees whether an earlier proof recorded a different one.
    const payloadJson = JSON.stringify({
      source: "oauth_claim",
      method: "oauth_review",
      platform,
      account_key: chg.account_key,
      claim_ref_hash: await claimRefHashOf(claimRef),
      provider_pid: identity.pid,
      earlier_provider_pid_differs: await earlierProviderPidDiffers(db, chg.passport_id, chg.account_key, platform, identity.pid),
    });
    await db.batch([
      db
        .prepare("INSERT INTO review_queue (item_id, item_type, state, subject, payload, created_at, updated_at, import_batch_id) VALUES (?, 'claim', 'open', ?, ?, ?, ?, NULL)")
        .bind(itemId, chg.passport_id, payloadJson, ts, ts),
      db
        .prepare("INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) VALUES ('system', 'enqueue:claim', ?, ?, ?, 'system')")
        .bind(itemId, ts, itemId), // I34-W3-SEAM-P (D-1119(8), D-1120(6)): the review-queue audit copy keeps only the item id
    ]);
    return { ok: false, outcome: "pending_review", passport_id: chg.passport_id, claim_ref: claimRef };
  }

  // `extraStatements`, exactly as `appendCreatorSigned` rides a passkey mark —
  // so a control proof cannot exist in mutable state without its ledger row
  // (invariant 2). It was previously flipped in the batch above, and a racing
  // append left `status='used'` + a claim_consumed row + NO claim_proven, with
  // `isAccountProven` reporting true. The `WHERE status = 'spent'` also makes
  // this the only door: a row that never spent can never be flipped to proven.
  //
  // tryAppendEvents, NOT a message regex: `prev_hash` UNIQUE matched
  // `/UNIQUE|constraint/i`, so a contention loss was reported to the creator as
  // `identity_mismatch` — "the account you authorized is not the one you are
  // claiming" — when the identity had in fact matched.
  //
  // AND WRAPPED, because a second atomic gate refuses through a THROW rather
  // than a refusal. `accountablePassport` (events.ts) fails a character with
  // nobody live answering for it by feeding NULL to `passports.claim_state`;
  // that is a NOT NULL constraint error, which `tryAppendEvents` deliberately
  // rethrows unchanged, so it sailed past the `!r.ok` block below and out of
  // this function as a 500 on the OAuth callback. It means exactly what the
  // "already claimed" branch means — this passport can no longer be claimed —
  // and must answer the same way, or the callback page distinguishes them.
  // I3-1078: the handoff row rides the same batch (see `mintHandoff`). Same
  // columns, code size and TTL as `createOAuthHandoff`.
  const handoff = mintHandoff ? randToken(32) : undefined;
  const r = await tryAppendEvents(
    db,
    // I3-1953 (D-0920): the provider's stable subject id rides the event, the
    // ledger record of WHICH provider identity proved this account.
    [{ event_type: "claim_proven", subject: chg.passport_id, actor_kind: "system", actor_id: `oauth:${platform}`, payload: { method: "oauth", pid: identity.pid } }],
    now,
    [],
    // The spent→proven flip also DATES the proof (migration 0014). `proved_at`
    // is the verify instant, not `created_at` — an OAuth challenge is issued
    // when the creator is bounced to the provider and proven only when they come
    // back, and the gap is however long the consent screen took. `valid_until`
    // stays NULL so the default TTL in `account-standing.ts` governs: the
    // PROVIDER's token lifetime is not this proof's lifetime (we keep no token),
    // and there is nothing here that carries a deadline of its own. `method =
    // 'oauth'` mirrors the `claim_proven` payload one line above.
    [
      db
        .prepare(
          // BF-03. The supporting account is the EVIDENCE, and it is re-examined
          // HERE, in the statement that writes the proof — not in a read before
          // it. `startOAuthClaim` checked attachment and `active` before the
          // browser left for the provider (:150-162) and nothing has looked at
          // the row since; `account_transferred` (events.ts:892),
          // `account_revoked` (:763) and the historical sweep (:1753) all move
          // it, and the ledger's own `claim_proven` handler (events.ts:907-921)
          // has no account predicate. A separate SELECT before this batch would
          // be raced by any of the three.
          //
          // NOT a `WHERE` clause: a conditional UPDATE matching zero rows does
          // not throw, so the event would commit with the flip silently missing
          // — the S-070 shape this function already carries a comment about
          // (:290-315). The predicate is on the VALUE, so a withdrawn support
          // yields NULL into a NOT NULL column (0003_creator_identity.sql:63),
          // the batch aborts and the `claim_proven` event rolls back with it.
          // Same in-batch abort `accountablePassport` uses on
          // `passports.claim_state` (events.ts:1454-1456).
          "UPDATE oauth_challenges SET status = (SELECT CASE WHEN EXISTS (" +
            "SELECT 1 FROM passport_accounts a WHERE a.account_key = oauth_challenges.account_key" +
            " AND a.passport_id = oauth_challenges.passport_id AND a.status = 'active') THEN 'proven' END)" +
            // I4-07-R3 WO-E: `result_event` is bound IN THIS BATCH — the sub-select
            // reads the `claim_proven` row this same batch appended before its
            // extraStatements, so no '__pending__' is committed and no second write
            // backfills it. I3-0188 (I4-01-R4, column from migration 0123): the
            // provider subject id that proved the challenge is recorded with it.
            ", result_event = (SELECT event_id FROM trust_events WHERE subject = ? AND event_type = 'claim_proven' ORDER BY seq DESC LIMIT 1)" +
            ", proved_at = ?, method = 'oauth', provider_pid = ? WHERE state = ? AND status = 'spent'",
        )
        .bind(chg.passport_id, new Date(nowMs).toISOString(), identity.pid, state),
      ...(handoff === undefined
        ? []
        : [
            db
              .prepare("INSERT INTO oauth_handoffs (code, passport_id, flow_state, created_at, expires_at) VALUES (?, ?, ?, ?, ?)")
              .bind(handoff, chg.passport_id, state, now(), new Date(nowMs + HANDOFF_TTL_MS).toISOString()),
          ]),
    ],
    ["system"], // S27: system actor now requires explicit authorization
    afterCommit,
  ).catch((e: unknown) => {
    if (isUnaccountableCharacter(e)) return { ok: false as const, refusal: { kind: "state" as const, message: "requires unclaimed", error: e as Error } };
    if (supportWithdrawn(e)) return { ok: false as const, refusal: { kind: "state" as const, message: "support withdrawn", error: e as Error } };
    throw e;
  });
  if (!r.ok) {
    // Same narrow translation as the bio-code and DNS paths: only "somebody
    // already claimed this passport" is reported to the caller; any other ledger
    // fault is rethrown rather than shown as an identity mismatch.
    if (r.refusal.message === "support withdrawn") return { ok: false, outcome: "support_withdrawn", passport_id: chg.passport_id };
    const alreadyClaimed = r.refusal.kind === "conflict" || /requires unclaimed/.test(r.refusal.message);
    if (!alreadyClaimed) throw r.refusal.error;
    // I4-04-R8 (sibling of I3-2061): the identity DID match, so this is never
    // `identity_mismatch`. If the claim that won the window proved THIS account
    // on THIS passport (the creator's own second tab), the claim succeeded and
    // the answer is `verified` with the winner's event. Anything else is the
    // door's uniform not-claimable refusal (see `support_withdrawn` above).
    const won = await db
      .prepare(
        "SELECT result_event FROM oauth_challenges WHERE passport_id = ? AND account_key = ? AND status = 'proven' AND state <> ? ORDER BY proved_at DESC LIMIT 1",
      )
      .bind(chg.passport_id, chg.account_key, state)
      .first<{ result_event: string | null }>();
    if (won) return { ok: true, outcome: "verified", passport_id: chg.passport_id, ...(won.result_event ? { event_id: won.result_event } : {}) };
    return { ok: false, outcome: "support_withdrawn", passport_id: chg.passport_id };
  }
  const event = r.events[0];
  return { ok: true, outcome: "verified", passport_id: chg.passport_id, event_id: event!.event_id, ...(handoff === undefined ? {} : { handoff }) };
}

import { canonicalAccount } from "@dhwaja/canonical";
import { logWarn } from "./log.js";

/**
 * OAuth providers for social-account claim verification.
 *
 * The security-critical function here is identityMatchesTarget: OAuth proves a
 * user controls SOME account; we claim a passport ONLY if the authenticated
 * identity is THE target account. Authorizing your own unrelated account can
 * never claim a victim's passport, because the canonical key derived from the
 * OAuth identity won't equal the target's key. Once a target has been
 * OAuth-proven, its recorded provider pid must ALSO match (I3-0113).
 *
 * Endpoints are constants (no SSRF): the only outbound calls go to these fixed
 * platform hosts. Tokens are used once to fetch identity and then discarded —
 * never stored or logged.
 */

export interface OAuthProvider {
  platform: string; // canonical platform key
  authorizeUrl: string;
  tokenUrl: string;
  identityUrl: string;
  scope: string;
  /** Parse the platform's identity response into {pid, handle}. */
  parseIdentity: (body: unknown) => { pid: string; handle: string } | null;
}

function str(v: unknown): string | null {
  return typeof v === "string" && v.trim().length > 0 && v.length <= 128 ? v : null;
}

export const PROVIDERS: Record<string, OAuthProvider> = {
  x: {
    platform: "x",
    authorizeUrl: "https://twitter.com/i/oauth2/authorize",
    tokenUrl: "https://api.twitter.com/2/oauth2/token",
    identityUrl: "https://api.twitter.com/2/users/me",
    scope: "users.read tweet.read",
    parseIdentity: (body) => {
      const d = (body as { data?: { id?: unknown; username?: unknown } } | null)?.data;
      const pid = str(d?.id);
      const handle = str(d?.username);
      return pid && handle ? { pid, handle } : null;
    },
  },
  // Instagram / TikTok follow the same shape; add when the operator registers
  // apps and confirms the identity response fields.
};

export interface OAuthAppConfig {
  clientId: string;
  clientSecret: string;
  redirectUri: string;
}

/** Read per-platform OAuth app config from env, or null if not configured. */
export function oauthConfigFromEnv(env: Record<string, unknown>, platform: string): OAuthAppConfig | null {
  const up = platform.toUpperCase();
  const clientId = str(env[`OAUTH_${up}_CLIENT_ID`]);
  const clientSecret = str(env[`OAUTH_${up}_CLIENT_SECRET`]);
  const redirectUri = str(env[`OAUTH_${up}_REDIRECT_URI`]);
  if (!clientId || !clientSecret || !redirectUri) return null;
  // A-36: clientSecret gets the same placeholder test as every other secret in
  // env-config.ts (CF_API_TOKEN, SESSION_SECRET); redirectUri gets the same
  // https-origin shape test appOriginFromEnv applies to APP_ORIGIN. Without
  // these, "OAUTH_X_CLIENT_SECRET=TODO-set-real-secret" or a bare host
  // redirectUri passes oauthConfigFromEnv and the platform looks configured
  // until the first callback fails. configHealth (admin.ts) carries an
  // `oauth_<platform>` row computed from this same function, so a platform
  // refused here reads as unconfigured to the operator too.
  if (/TODO/i.test(clientId) || /TODO/i.test(clientSecret) || /TODO/i.test(redirectUri)) return null;
  if (!/^https:\/\//i.test(redirectUri)) return null;
  return { clientId, clientSecret, redirectUri };
}

export function buildAuthorizeUrl(
  provider: OAuthProvider,
  cfg: OAuthAppConfig,
  state: string,
  codeChallenge: string,
): string {
  const u = new URL(provider.authorizeUrl);
  u.searchParams.set("response_type", "code");
  u.searchParams.set("client_id", cfg.clientId);
  u.searchParams.set("redirect_uri", cfg.redirectUri);
  // An EMPTY scope is omitted rather than sent as `scope=`. A provider that
  // needs no scope to answer "who is this" (Twitch's helix/users) must not be
  // asked for one, and a literal empty parameter is rejected by some and
  // silently defaulted by others — neither of which is "ask for nothing".
  if (provider.scope.length > 0) u.searchParams.set("scope", provider.scope);
  u.searchParams.set("state", state);
  u.searchParams.set("code_challenge", codeChallenge);
  u.searchParams.set("code_challenge_method", "S256");
  return u.toString();
}

/**
 * Exchange the authorization code for a token and fetch the authenticated
 * identity. `fetchImpl` is injectable for tests. Never returns/stores the
 * token. Fails closed (null) on any error.
 */
/** I3-2014: every outbound call on this rail carries its own deadline, the
 * house rule stated repeatedly elsewhere (email.ts, media.ts, push.ts,
 * registry-sign.ts, mailer.ts, evidence-scan.ts, domain-claims.ts,
 * checkpoint.ts) and, until now, absent from both fetches here. Without an
 * AbortController, a platform that accepts the connection and never answers
 * keeps this Worker's invocation alive until the runtime's own hard limit,
 * not this rail's — the fail-closed `fail()` below never fires because the
 * call never rejects. */
const OAUTH_FETCH_TIMEOUT_MS = 8_000;

async function fetchWithDeadline(fetchImpl: typeof fetch, url: string, init: RequestInit): Promise<Response> {
  const ctl = new AbortController();
  const timer = setTimeout(() => ctl.abort(), OAUTH_FETCH_TIMEOUT_MS);
  try {
    return await fetchImpl(url, { ...init, redirect: "manual", signal: ctl.signal });
  } finally {
    clearTimeout(timer);
  }
}

/** I3-2014 (reopened): `fetchWithDeadline` only bounded getting HEADERS —
 * `clearTimeout` fired the moment `fetchImpl` resolved, which is BEFORE the
 * body is read. A server that answers headers immediately and then stalls the
 * body was never aborted: `tokenRes.json()` / `idRes.json()` carried no
 * signal and no timer of their own, so the exact hang the original fix was
 * for moved four lines down, to the `.json()` call. This wraps a
 * Response-body read (`.json()`, injected as `read`) in ITS OWN
 * AbortController tied to the SAME deadline budget, and — because `fetch`'s
 * signal does not reach a `Response` already returned to the caller — calls
 * `res.body?.cancel()` on abort so the underlying connection is actually
 * freed, not just abandoned with the read promise still pending. A caller
 * cannot tell a body-stall timeout from a `fail()` refusal: both surface as
 * bare `null` through the same `fail(stage)` path below. */
async function readJsonWithDeadline<T>(res: Response, read: () => Promise<T>): Promise<T> {
  const ctl = new AbortController();
  const timer = setTimeout(() => {
    ctl.abort();
    void res.body?.cancel().catch(() => {});
  }, OAUTH_FETCH_TIMEOUT_MS);
  const aborted = new Promise<never>((_, reject) => {
    ctl.signal.addEventListener("abort", () => reject(new DOMException("body deadline exceeded", "AbortError")), { once: true });
  });
  try {
    return await Promise.race([read(), aborted]);
  } finally {
    clearTimeout(timer);
  }
}

export async function exchangeAndIdentify(
  provider: OAuthProvider,
  cfg: OAuthAppConfig,
  code: string,
  codeVerifier: string,
  fetchImpl: typeof fetch = fetch,
): Promise<OAuthIdentity | null> {
  // E-47: every failure below used to return bare `null`, collapsing "wrong
  // code", "network blip" and "the platform changed its response shape" into
  // one indistinguishable outcome with no per-platform signal anywhere (no
  // per-exchange counter; configHealth's `oauth_<platform>` row in admin.ts
  // says only whether the platform is configured). `fail()` tags what an
  // operator can grep for: platform + the stage that gave up. It does not change the return type (still null) so
  // every caller's `outcome: "exchange_failed"` contract is unchanged.
  const fail = (stage: string): null => {
    logWarn({ cid: "oauth", hop: "http", msg: "oauth: exchangeAndIdentify failed", platform: provider.platform, stage });
    return null;
  };
  const basic = btoa(`${cfg.clientId}:${cfg.clientSecret}`);
  let tokenRes: Response;
  try {
    tokenRes = await fetchWithDeadline(fetchImpl, provider.tokenUrl, {
      method: "POST",
      headers: { "Content-Type": "application/x-www-form-urlencoded", Authorization: `Basic ${basic}` },
      body: new URLSearchParams({
        grant_type: "authorization_code",
        code,
        redirect_uri: cfg.redirectUri,
        code_verifier: codeVerifier,
        client_id: cfg.clientId,
      }).toString(),
    });
  } catch {
    return fail("token_request");
  }
  if (!tokenRes.ok) return fail("token_response");
  const tokenBody = (await readJsonWithDeadline(tokenRes, () => tokenRes.json()).catch(() => null)) as { access_token?: unknown } | null;
  // Access tokens are opaque and can be long (hundreds of chars) — do NOT use
  // the 128-char `str()` cap here (that would reject real tokens and silently
  // break the feature). Bound generously against absurd payloads only.
  const raw = tokenBody?.access_token;
  const token = typeof raw === "string" && raw.length > 0 && raw.length <= 8192 ? raw : null;
  if (!token) return fail("token_shape");

  let idRes: Response;
  try {
    // #177 PV-04: `Client-Id` on EVERY provider's identity GET. Twitch Helix
    // refuses `/helix/users` without it; the value is the public app id that
    // every authorize URL already carries, so sending it elsewhere leaks nothing.
    idRes = await fetchWithDeadline(fetchImpl, provider.identityUrl, { method: "GET", headers: { Authorization: `Bearer ${token}`, "Client-Id": cfg.clientId } });
  } catch {
    return fail("identity_request");
  }
  if (!idRes.ok) return fail("identity_response");
  const idBody = await readJsonWithDeadline(idRes, () => idRes.json()).catch(() => null);
  const identity = provider.parseIdentity(idBody);
  if (!identity) return fail("identity_parse");
  return identity;
}

/**
 * THE anti-hijack guard. The target is a specific account_key. The OAuth
 * identity yields two candidate canonical keys (by immutable pid, and by
 * current handle); the target must be one of them. So the ONLY way to satisfy
 * a claim is to authenticate as the actual target account.
 *
 * I3-0113 (D-0920 option d): handles are recycled, so the current-handle
 * candidate is trusted only for a target never OAuth-proven. `recordedPid` is
 * the provider pid recorded when this account key was last OAuth-proven
 * (`oauth_challenges.provider_pid`), or null if it never was; when it is set the
 * identity's pid must equal it. Required, so no caller can leave it out.
 */
export function identityMatchesTarget(platform: string, identity: OAuthIdentity, targetAccountKey: string, recordedPid: string | null): boolean {
  if (recordedPid !== null && identity.pid !== recordedPid) return false;
  const candidates = new Set<string>();
  try {
    candidates.add(canonicalAccount(platform, "pid", identity.pid).canonical);
  } catch {
    /* pid may be non-conforming; ignore */
  }
  try {
    candidates.add(canonicalAccount(platform, "handle", identity.handle).canonical);
  } catch {
    /* handle may be non-conforming; ignore */
  }
  return candidates.has(targetAccountKey);
}
