/**
 * I3-0465b (D-0954(6), I19-L01): the ledger, witness and likeness-revoke verdicts below are logged as NAMED fields
 * on log.ts's allowlist, never as a stringified-object `detail` dump — the second raw-JSON convention, which
 * `redact()` cut at MAX_STRING_LEN (200) mid-object, so a long mismatch list lost the very seqs an operator needed.
 * Booleans with no allowlisted name ride `reason` as a comma list of the flags that hold (the
 * `ledger_full_walk_overdue` convention, checkpoint.ts). `detail` keeps only its sanctioned ledger use: one object
 * key (`breachObject`, the anchor a breach condemns the log against — DEPLOY.md "name the object first").
 */
type VerdictFields = Record<string, string | number | boolean | undefined>;
const flagList = (flags: Record<string, boolean>): string | undefined => Object.keys(flags).filter((k) => flags[k]).join(",") || undefined;
function chainLogFields(v: Awaited<ReturnType<typeof verifyChainTick>>): VerdictFields {
  return { count: v.events, seq: v.brokenAtSeq, reason: v.reason };
}
function anchorLogFields(v: Awaited<ReturnType<typeof verifyAgainstCheckpoint>>): VerdictFields {
  return { count: v.events, seq: v.anchorSeq ?? v.brokenAtSeq, kind: v.breach, arm: v.state, reason: v.anchorReason ?? v.signatureReason ?? v.reason, detail: v.breachObject };
}
function witnessLogFields(w: WitnessVerdict): VerdictFields {
  return { scanned: w.checked, count: w.mismatches.length, seq: w.mismatches[0]?.seq, failed: w.unreadable, reason: flagList({ truncated: w.truncated, nothing_compared: w.nothing_compared, ever_sealed: w.ever_sealed }) };
}

 * under. One const so the recorder and the operator query name the same string

export const UNERASED_SUBJECT_ACTION = "purge_frozen_analytics:subject_skipped";

/**
 * ONE `audit_log` ROW PER RETRACTED CREATOR WHOSE PURGE THREW.
 *
 * `sweepFrozenSnapshots` records a COUNT (`analytics-build.ts:376`) because the
 * count is all that may go to a platform log: erasure reaches the DB and the
 * object bucket, and it does not reach a log line, so a passport id written
 * there survives the erasure that was supposed to remove it. The count alone
 * leaves an operator unable to re-run erasure for a NAMED creator, which is the
 * whole of the FX2 sink. The subject therefore lands HERE — `item_id` on a DB
 * row, inside the erasure perimeter, queryable by action — exactly the shape
 * `eraseAccountWithSubscriptions`'s post-erasure `subscriptionAudit` (account_data.ts) uses for `account_delete:post_erasure_failed`.
 *
 * Best-effort by construction, for the same reason that one does: the record of
 * a residual must never be able to fail the sweep it is about.
 */
async function recordUnerasedSubject(db: DbLike, actorId: string, passportId: string, ts: string): Promise<void> {
  try {
    // I14-CRON (QUEUE #923, D-0904(2)): ONE row per subject per UTC day of `ts`, not one per lap of the purge
    // cursor. Still ONE statement (the purge reserves one per failure), served by idx_audit_item_action; no
    // audit row is ever deleted (retention is not this sweep's).
    await db
      .prepare(
        "INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, ?, ?, ?, ?, 'system' WHERE NOT EXISTS (SELECT 1 FROM audit_log WHERE item_id = ? AND action = ? AND substr(created_at, 1, 10) = ?)",
      )
      .bind(actorId, UNERASED_SUBJECT_ACTION, "frozen day objects remain in the bucket; erasure must be re-run for this subject", ts, passportId, passportId, UNERASED_SUBJECT_ACTION, ts.slice(0, 10))
      .run();
  } catch {
    /* an audit row that fails must not fail the erasure sweep it is about */
  }
}

function jwksFor(domain: string, nowMs: number, wantKid?: string): Promise<Jwk[]> {
  let cache = accessJwksCaches.get(domain);
  if (!cache) {
    cache = new JwksCache(`${domain}/cdn-cgi/access/certs`);
    accessJwksCaches.set(domain, cache);
  }
  return cache.get(nowMs, fetch, wantKid);
}

 * `{error:"internal"}`, never a 400 telling the caller the request was wrong.
 */
function routeFault(c: Context<{ Bindings: Env; Variables: { cid: string } }>, route: string, e: unknown): Response {
  logError({ cid: c.get("cid"), hop: "http", route, msg: "unexpected route fault", err: e instanceof Error ? e.name : typeof e });

}

// (I3-1756: a caller's `x-request-id` is never reused as it; a bounded one is only
// logged beside ours, below). It is not echoed as a header (see the note after
// `next()`); it reaches the caller only inside the generic 500/503 JSON body. This is
// the ONLY new id source PS-089 needs (org/07-findings/S-034-batch/OBS/PLAN.md).
app.use("*", async (c, next) => {

  // every route; a cookie is honoured ONLY under `Authorization: Dhwaja-Csrf <handle>` and is then handed to the

  if ((c.res.headers.get("content-type") ?? "").startsWith("text/html") && !deployIsProduction(c.env)) {
    c.res.headers.set("X-Robots-Tag", "noindex, nofollow");
  }

    }

  const liveSecret = c.env?.SESSION_SECRET;

});

/**
 * ── POST /internal/session-check — THE REVOCATION READ FOR SIBLING WORKERS ────
 *
 * THE DEFECT THIS CLOSES. `services/image` verifies the same HMAC-signed tokens
 * this worker issues, and it used to stop there: it read `sub.kind` and `sub.id`
 * and NEVER read `sub.ep`, because it has no copy of `accounts.session_epoch` or
 * `creator_sessions` and no channel to ask. Every revocation lever in this
 * product — `recovery.ts`'s force_logout + reset_passkeys, `/account/logout-all`,
 * `/account/creator-logout-all`, `revokeFanPasskey`/`revokePasskey` — works by
 * BUMPING AN EPOCH, so all of them stopped at trust-core's edge. A hijacked
 * creator whose passport had just been recovered was told they were safe while
 * the attacker's token kept working against `POST /enroll` and `POST /consent`
 * for the remainder of the 30-day session TTL — enrolling BIOMETRIC FACE VECTORS
 * against the victim's passport. This endpoint is the channel that was missing.
 *
 * IT IS NOT AN ENUMERATION ORACLE, AND THAT IS STRUCTURAL. The request does not
 * name a subject; it carries a TOKEN, and the token is verified here before any
 * table is touched. A caller therefore cannot ask about an id it does not
 * already hold a validly-signed session for — which is a session that already
 * opens every gated route in this worker, so the answer discloses strictly
 * nothing the caller could not learn by calling one. And every refusal is the
 * SAME refusal: an unknown account, a deleted account, a bumped epoch, an
 * expired token, a forged token, a wrong-kind token and an agency token all
 * answer the identical `{"live":false,"tier":"free"}` with status 200. Nothing
 * in the response separates "no such subject" from "revoked".
 *
 * IT IS NOT A PUBLIC ROUTE. `X-Internal-Auth` must carry the machine credential
 * derived from SESSION_SECRET (see `internalServiceCredential` in
 * @dhwaja/session for why that secret and not a second one), compared in
 * constant time. Absent or placeholder SESSION_SECRET → 503, never open:
 * invariant 6, and the same fail-closed shape `services/image`'s own
 * `INTERNAL_TRIAGE_SECRET` seam already uses.
 *
 * THE TIER RIDES ALONG on purpose. It is the second half of the same defect:
 * the image service's `/find` quota used to read a `tier` claim off the token
 * that nothing ever stamped, so every paying fan was metered as free — and
 * stamping it would have made a 30-day unrevocable entitlement (see
 * @dhwaja/session). Reading it HERE, from `effectiveFanTier` — the same
 * two-rail read every fan gate in this worker uses — means a cancellation or a
 * chargeback reaches the image service in seconds, because it is answered live
 * on a request the image service was going to make anyway.
 *
 * THE EPOCH RULES ARE THE ONES THIS WORKER APPLIES TO ITSELF, deliberately:
 * `fanCaller` treats an absent account as dead, and `creatorId` treats an absent
 * `creator_sessions` row as epoch 0 (pre-S69 tokens stay valid until the first
 * logout-all). A sibling worker must not be stricter or looser than the origin
 * of the session, or the two disagree about who is signed in.
 *
 * AGENCY TOKENS ANSWER NOT-LIVE. No sibling worker accepts an agency principal
 * today, and an agency's liveness is more than an epoch (`agencyPrincipal` also
 * re-reads status and the member's CURRENT seat). Answering "not live" is the
 * fail-closed answer; the day a sibling needs agency principals, this branch
 * grows the full `agencyPrincipal` check rather than a partial one.
 */
/**
 * POST /internal/likeness-revoke — I3-0603 (D-0838(1)): the image service's
 * mirror of a biometric-consent withdrawal. Gated like /internal/session-check:
 * an X-Internal-Auth credential derived from SESSION_SECRET for
 * LIKENESS_REVOKE_AUDIENCE (503 without a usable secret, 401 otherwise), body
 * read only after the gate. Records `likeness_consent_revoked` as system/svc:image
 * via `revokeLikenessConsentAsSystem` (idempotent; never queues the reverse call).
 */
app.post("/internal/likeness-revoke", anonBodyLimit, async (c) => {
  const expected = await internalServiceCredential(c.env.SESSION_SECRET, LIKENESS_REVOKE_AUDIENCE);
  if (!expected) return c.json({ error: "not_configured" }, 503);
  if (!constantTimeEqual(c.req.header("X-Internal-Auth") ?? "", expected)) return c.json({ error: "unauthorized" }, 401);
  const raw = await c.req.text();
  if (raw.length > 1024) return c.json({ error: "passport_id required" }, 400);
  let b: { passport_id?: unknown } | null = null;
  try {
    b = JSON.parse(raw) as { passport_id?: unknown } | null;
  } catch {
    b = null;
  }
  if (!b || typeof b !== "object" || typeof b.passport_id !== "string" || Object.keys(b).length !== 1) return c.json({ error: "passport_id required" }, 400);
  try {
    return c.json({ ok: true, ...(await revokeLikenessConsentAsSystem(d1(c.env.TRUST_DB), b.passport_id)) });
  } catch (e) {
    if (e instanceof LikenessError) return c.json({ error: e.message }, 400);
    throw e;
  }
});

app.post("/internal/session-check", anonBodyLimit, async (c) => {
  const secret = c.env.SESSION_SECRET;
  const expected = await internalServiceCredential(secret, SESSION_CHECK_AUDIENCE);
  if (!expected) return c.json({ error: "not_configured" }, 503); // no usable session secret → fail closed
  if (!constantTimeEqual(c.req.header("X-Internal-Auth") ?? "", expected)) return c.json({ error: "unauthorized" }, 401);
  // Body read only AFTER the gate, and capped: an unauthenticated caller never
  // buffers anything here. A session token is ~200 bytes; 4 KiB is slack.
  const raw = await c.req.text();
  if (raw.length > 4096) return c.json({ error: "bad_request" }, 400);
  let token = "";
  try {
    const b = JSON.parse(raw) as unknown;
    if (b && typeof b === "object" && !Array.isArray(b) && typeof (b as { token?: unknown }).token === "string") token = (b as { token: string }).token;
  } catch {
    /* malformed JSON is a CALLER bug, not a fact about a subject — 400 below */
  }
  if (!token) return c.json({ error: "bad_request" }, 400);
  // THE ONE REFUSAL. Every not-live case returns this exact body, so the
  // response cannot be read as an existence oracle.
  const dead = { live: false, tier: "free" as const };
  const nowMs = Date.now();
  // I3-1350 (D-0920, I16-W05F H1): the image worker presents an IMAGE token; a session token is still answered.
  const nowSec = Math.floor(nowMs / 1000);
  // I7-Q4 (I30-SESSION): a session verifies under SESSION_SECRET or any SESSION_SECRET_PREVIOUS entry. The image token key
  // follows SESSION_SECRET only; image tokens live IMAGE_TOKEN_TTL_SECONDS (15 minutes), so a rotation costs at most one re-issue.
  const s = (await verifySessionRotating(c.env, token, nowSec))?.subject ?? (await verifyImageToken(await deriveImageScopeKey(secret), token, nowSec));
  if (!s) return c.json(dead); // expired, forged, malformed, or a memberless agency token
  const db = d1(c.env.TRUST_DB);
  if (s.kind === "fan") {
    const epoch = await currentSessionEpoch(db, s.id);
    // QUEUE #171: `!==`, not `<` — a token stamped ABOVE the stored epoch is refused, the
    // same predicate `fanCallerWith` (route-helpers.ts) applies at every `fanId` door.
    if (epoch === null || s.ep !== epoch) return c.json(dead); // account deleted, signed out everywhere, or epoch regressed
    return c.json({ live: true, tier: await effectiveFanTier(db, s.id, nowMs) });
  }
  if (s.kind === "creator") {
    // D-0843(5) (I4-RG-OPS rgops-05): live exactly when `creatorId` — the ONE creator
    // predicate (epoch, and I3-1422's suspended/terminated founder) — resolves this token.
    const adm = await creatorAdmissionForSubject(c.env, s); // the ONE creator predicate, on the verified subject (session or image token)
    if (adm?.id !== s.id) return c.json(dead); // logged out everywhere / revoked / recovered / founder closed
    return c.json({ live: true, tier: "free" as const }); // tier is a FAN concept; creator quota is not metered by it
  }
  return c.json(dead);
});

/**
 * I25-W1-COOKIE: the browser bootstrap. A page holds its CSRF handle in memory only, so after a reload it asks here:
 * the session cookie for `kind` is read directly (this route is a GET and returns no data but the handle, which is
 * useless without the cookie; credentialed CORS is exact-origin, cors.ts), resolved through the SAME live predicate
 * its surface uses (fanId / creatorIdForDataRights / agencyPrincipal), and answered with the handle — or 401 and the cookie
 * cleared. A creator answer also names the passport the console is signed in as.
 */
/** The session `token` resolves to, when it is LIVE by the SAME predicate its own surface uses (fanId /
 * creatorIdForDataRights / agencyPrincipal — epoch, closure, rights scope, seat), else null. */
async function liveSessionSubject(c: Context<{ Bindings: Env; Variables: { cid: string } }>, token: string | null): Promise<SessionSubject | null> {
  // I7-Q4 (I30-SESSION): verified under SESSION_SECRET or any SESSION_SECRET_PREVIOUS entry.
  const s = token ? ((await verifySessionRotating(c.env, token, Math.floor(Date.now() / 1000)))?.subject ?? null) : null;
  if (!s) return null;
  const asBearer = { env: c.env, req: { header: (k: string) => (k.toLowerCase() === "authorization" ? `Bearer ${token}` : c.req.header(k)) } };
  if (s.kind === "fan") return (await fanId(asBearer)) === s.id ? s : null;
  // RG-06 (D-1010(1)): the creator console's session is the one its export, appeal and payout pages are admitted with —
  // creatorIdForDataRights (epoch; refuses an open account_compromise), which also admits a suspended or self-closed
  // creator. Only the bootstrap and sign-out read this; every service door keeps its own predicate.
  if (s.kind === "creator") return (await creatorIdForDataRights(asBearer)) === s.id ? s : null;
  return (await agencyPrincipal(asBearer))?.agencyId === s.id ? s : null;
}
/** RG-01/RG-06: `liveSessionSubject`, or a fan session `fanIdForDataRights` still admits (the closed-account data-rights
 * session a closed fan's browser sign-in put in the cookie). A cookie this answers non-null for is never expired by a
 * 401 from another door, nor by the bootstrap; sign-out still expires it. */
async function cookieSessionLive(c: Context<{ Bindings: Env; Variables: { cid: string } }>, token: string | null): Promise<SessionSubject | null> {
  const live = await liveSessionSubject(c, token);
  if (live || !token) return live;
  const s = (await verifySessionRotating(c.env, token, Math.floor(Date.now() / 1000)))?.subject ?? null;
  if (s?.kind !== "fan") return null;
  const asBearer = { env: c.env, req: { header: (k: string) => (k.toLowerCase() === "authorization" ? `Bearer ${token}` : c.req.header(k)) } };
  return (await fanIdForDataRights(asBearer)) === s.id ? s : null;
}
app.get("/session/csrf", async (c) => {
  const kind = SESSION_KINDS.find((k) => k === c.req.query("kind"));
  if (!kind) return c.json({ error: "unauthorized" }, 401); // no session of an unnamed kind
  const secret = c.env.SESSION_SECRET;
  if (!usableSessionSecret(secret)) return c.json({ error: "sessions_unavailable" }, 503);
  const cookie = readSessionCookie(c.req.raw, kind, c.env);
  const s = await liveSessionSubject(c, cookie);
  if (s?.kind !== kind) {
    // RG-01: a data-rights cookie is no signed-in session (401), but it is kept for the export/erase pages.
    const keep = (await cookieSessionLive(c, cookie))?.kind === kind;
    return c.json({ error: "unauthorized" }, 401, { ...(keep ? {} : { "Set-Cookie": clearSessionCookie(kind, c.env) }), "Cache-Control": "no-store" });
  }
  return c.json({ csrf: await csrfHandle(secret, s), ...(kind === "creator" ? { passport_id: s.id } : {}) }, 200, { "Cache-Control": "no-store" });
});

/** I25-W1-COOKIE: sign-out on this browser. The browser calls it under `Dhwaja-Csrf <handle>` (the wildcard turns that
 * into the cookie's Bearer); a LIVE session's cookie expires now (Max-Age=0). Anything else is 401 `unauthorized` —
 * which, under the scheme, makes the wildcard expire the cookie anyway (csrf.ts finishSessionResponse), so a dead
 * session's cookie is cleared too while a data-rights or forged token gets the same answer (QUEUE #417's equality).
 * Server-side revocation stays where it was: /account/logout-all and its creator/agency siblings. */
app.post("/session/signout", async (c) => {
  const secret = c.env.SESSION_SECRET;
  if (!usableSessionSecret(secret)) return c.json({ error: "sessions_unavailable" }, 503);
  const token = /^Bearer\s+(.+)$/.exec(c.req.header("Authorization") ?? "")?.[1] ?? null;
  const s = await liveSessionSubject(c, token);
  if (!s) {
    // RG-01: a data-rights cookie gets the forged token's answer (401, cookie expired — the wildcard expires the forged
    // one), so QUEUE #417's equality holds; it is expired here because the wildcard keeps a cookie that is still live.
    const rights = await cookieSessionLive(c, token);
    return c.json({ error: "unauthorized" }, 401, rights ? { "Set-Cookie": clearSessionCookie(rights.kind, c.env) } : {});
  }
  return c.json({ ok: true }, 200, { "Set-Cookie": clearSessionCookie(s.kind, c.env), "Cache-Control": "no-store" });
});

/**
 * I3-0144 / I3-1646 (I23-W1-CONSOLES H4, landed by I23-L01): the creator's face-data withdraw and erase doors, proxied
 * to the dhwaja-image worker so the creator console needs no second origin. Same admission as /session/image-token
 * (a live creator session), then a fresh image-scope token for THIS creator, forwarded as the Bearer. The POST body is
 * forwarded byte-for-byte: an ABSENT body means "withdraw every consent" (image http.ts /consent/revoke parses strictly),
 * so it is never synthesised. The worker's status and JSON come back unchanged (a 503 stays 503).
 * JN-08 sibling (I23-FX-HUB): admitted through creatorIdForDataRights, not creatorAdmission — a privacy exit is never
 * blocked by a closure, so a staff-suspended, self-closed or founder-closed creator still withdraws and erases; an
 * account_compromise suspension still refuses (the bearer may be the hijacker).
 */
async function creatorFaceDoor(c: Context<{ Bindings: Env; Variables: { cid: string } }>, method: "POST" | "DELETE", path: "/consent/revoke" | "/enrollments"): Promise<Response> {
  const secret = c.env.SESSION_SECRET;
  if (!usableSessionSecret(secret)) return c.json({ error: "sessions_unavailable" }, 503);
  const nowSec = Math.floor(Date.now() / 1000);
  const m = /^Bearer\s+(.+)$/.exec(c.req.header("Authorization") ?? "");
  const s = m ? ((await verifySessionRotating(c.env, m[1]!, nowSec))?.subject ?? null) : null; // I7-Q4 (I30-SESSION)
  if (s?.kind !== "creator" || (await creatorIdForDataRights(c)) !== s.id) return c.json({ error: "unauthorized" }, 401);
  const svc = c.env.IMAGE_SERVICE;
  if (!svc) return c.json({ error: "image_service_unavailable" }, 503);
  const token = await issueImageToken(secret, { kind: "creator", id: s.id, ep: s.ep }, nowSec);
  if (!token) return c.json({ error: "sessions_unavailable" }, 503);
  const body = method === "POST" ? await c.req.text() : "";
  const res = await svc.fetch(`https://image.internal${path}`, {
    method,
    headers: { Authorization: `Bearer ${token}`, ...(body !== "" ? { "Content-Type": "application/json" } : {}) },
    ...(body !== "" ? { body } : {}),
    redirect: "manual",
  });
  return new Response(await res.text(), { status: res.status, headers: { "Content-Type": res.headers.get("content-type") ?? "application/json", "Cache-Control": "no-store" } });
}

/**
 * ── POST /internal/suppression-check — THE SUPPRESSION ORACLE FOR SIBLING WORKERS ──
 *
 * THE DEFECT THIS CLOSES (D-0527, W3-07-R6 I3-0358/I3-0915). `services/image`'s
 * find-the-creator cascade checks only its own `IMAGE_DB.consents.revoked` —
 * it has no channel to THIS worker's suppression sources
 * (`request_suppressions`, `passports.retracted`), which live in a separate D1
 * with no join path. A subject who opted out of discovery, or whose passport
 * was retracted, stayed fully matchable through image search — the exact
 * thing `dropSuppressed` (rankings.ts) exists to prevent on every discovery
 * board, just never enforced here because there was no seam to ask. This
 * endpoint is that seam, shaped like `/internal/session-check` above: same
 * transport convention (`TRUST_CORE` service binding or `TRUST_CORE_URL`),
 * same machine credential derived from `SESSION_SECRET`, same fail-closed
 * contract on the client side (see `SuppressionOracle` in
 * `services/image/src/cascade.ts` — null means REFUSE, never "assume clear").
 *
 * A CLOSED-WORLD ORACLE, NOT AN ENUMERATION ENDPOINT. The request carries a
 * bounded batch of passport ids the CALLER ALREADY HOLDS as live cascade
 * candidates — there is no cursor, no range, no "list the next suppressed
 * id". The response is a SUBSET of exactly the ids submitted: an id that was
 * not in the request is never named in the answer (checked below), so this
 * cannot be walked to build a suppression list. It answers only about
 * identities the calling service's OWN biometric match already produced —
 * the same bound `/internal/session-check` uses (a caller can only ask about
 * a token it already holds a valid signature for); here the caller can only
 * ask about a passport_id its own cascade already surfaced as a live
 * enrollment candidate. `SUPPRESSION_CHECK_MAX_BATCH` bounds the request
 * itself so a compromised or buggy caller cannot turn one authenticated call
 * into a corpus-wide sweep.
 *
 * NOT DISTINGUISHING WHY. `suppressed` folds `request_suppressions` (opt-out)
 * and `passports.retracted` (erasure) into ONE flag — the same union
 * `dropSuppressed`'s `sup2` computes — so a sibling worker learns "hide
 * this", never "hide this because X"; which control produced the hide is not
 * observable off this seam, matching rankings.ts's own invariant that a
 * suppressed subject and a never-listed one read identically.
 *
 * NOT A PUBLIC ROUTE — the same `X-Internal-Auth` machine credential as
 * session-check, but a DOMAIN-SEPARATED audience
 * (`SUPPRESSION_CHECK_AUDIENCE`, not `SESSION_CHECK_AUDIENCE`) — both imported
 * from @dhwaja/session, so the two workers cannot drift apart silently — so a
 * credential minted for one internal seam can never be replayed against the
 * other. Absent/placeholder `SESSION_SECRET` → 503, the same fail-closed
 * shape session-check and `INTERNAL_TRIAGE_SECRET` already use.
 *
 * FAIL DIRECTION (the client's, but stated here since it is the reason this
 * endpoint's answer shape is what it is): `services/image/src/cascade.ts`'s
 * `SuppressionOracle.check` returns `null` on ANYTHING short of a well-formed
 * 200 — unreachable, timed out, wrong credential, malformed body, or an id in
 * the response the caller never asked about — and its caller (`findIdentity`)
 * treats null as "withhold every candidate in this request", never "assume
 * not suppressed". A suppression that stops working when this endpoint is
 * unreachable would not be a suppression.
 */
const SUPPRESSION_CHECK_MAX_BATCH = 16; // cascade only ever asks about ONE candidate per call today; bounded well below any plausible sweep

app.post("/internal/suppression-check", anonBodyLimit, async (c) => {
  const secret = c.env.SESSION_SECRET;
  const expected = await internalServiceCredential(secret, SUPPRESSION_CHECK_AUDIENCE);
  if (!expected) return c.json({ error: "not_configured" }, 503); // no usable session secret → fail closed
  if (!constantTimeEqual(c.req.header("X-Internal-Auth") ?? "", expected)) return c.json({ error: "unauthorized" }, 401);
  // Body read only AFTER the gate: an unauthenticated caller never buffers
  // anything here. `SUPPRESSION_CHECK_MAX_BATCH` ids at typical passport_id
  // length comfortably fit; 4 KiB is the same slack session-check gives itself.
  const raw = await c.req.text();
  if (raw.length > 4096) return c.json({ error: "bad_request" }, 400);
  let ids: string[] = [];
  try {
    const b = JSON.parse(raw) as unknown;
    if (b && typeof b === "object" && !Array.isArray(b) && Array.isArray((b as { passport_ids?: unknown }).passport_ids)) {
      const arr = (b as { passport_ids: unknown[] }).passport_ids;
      if (arr.length > 0 && arr.every((x) => typeof x === "string" && x.length > 0)) ids = arr as string[];
    }
  } catch {
    /* malformed JSON is a CALLER bug, not a fact about a subject — 400 below */
  }
  const distinct = [...new Set(ids)];
  // Empty or over-batch is a CALLER bug, not "nothing suppressed" — refuse
  // rather than silently answering about a truncated subset of the request.
  if (distinct.length === 0 || distinct.length > SUPPRESSION_CHECK_MAX_BATCH) return c.json({ error: "bad_request" }, 400);
  const db = d1(c.env.TRUST_DB);
  const suppressed = new Set<string>();
  for (const chunk of inChunks(distinct)) {
    const placeholders = chunk.map(() => "?").join(",");
    const { results: sup } = await db.prepare(`SELECT passport_id FROM request_suppressions WHERE passport_id IN (${placeholders})`).bind(...chunk).all<{ passport_id: string }>();
    for (const r of sup) suppressed.add(r.passport_id);
    const { results: ret } = await db.prepare(`SELECT passport_id FROM passports WHERE passport_id IN (${placeholders}) AND retracted = 1`).bind(...chunk).all<{ passport_id: string }>();
    for (const r of ret) suppressed.add(r.passport_id);
    // I3-0833 a (I23-W1-MONEY H7, landed by I23-L01): a creator under a staff suspension (creator_suspensions) or
    // whose founder account is suspended/terminated (account_closures) is suppressed for sibling workers too.
    // Two reads, each binding the chunk once, so the parameter count stays the chunk's size if the batch cap ever rises.
    const { results: own } = await db.prepare(`SELECT passport_id FROM creator_suspensions WHERE state = 'suspended' AND passport_id IN (${placeholders})`).bind(...chunk).all<{ passport_id: string }>();
    for (const r of own) suppressed.add(r.passport_id);
    const { results: founderClosed } = await db
      .prepare(`SELECT f.passport_id AS passport_id FROM founder f JOIN account_closures ac ON ac.account_id = f.account_id WHERE ac.state IN ${FROZEN_CLOSURE_STATES_SQL} AND f.passport_id IN (${placeholders})`) // I3-0389: the ONE spelling (credits.ts)

      .bind(...chunk)
      .all<{ passport_id: string }>();
    for (const r of founderClosed) suppressed.add(r.passport_id);
    // JN-08 sibling (I23-FX-HUB): a creator who closed their own creator account (creator_closures, creator-close.ts
    // closeCreatorAccount) is suppressed for sibling workers too.
    const { results: selfClosed } = await db.prepare(`SELECT passport_id FROM creator_closures WHERE passport_id IN (${placeholders})`).bind(...chunk).all<{ passport_id: string }>();
    for (const r of selfClosed) suppressed.add(r.passport_id);
  }
  // Closed-world contract: the answer names only ids drawn from `distinct`
  // (the request) -- structurally, since both queries above are WHERE ... IN
  // (the request's own ids), never a source that could add an unsolicited id.
  return c.json({ suppressed: [...suppressed] });
});

/**
 * W2 — the reader's language plus the cache directive that goes with it.
 *
 * `negotiate` is total and bounded (it truncates and caps an attacker-supplied
 * header before parsing it, and returns `en` for anything it cannot place), so
 * the value reaching a renderer is always a key that has a catalogue.
 *
 * VARY ONLY WHEN IT IS TRUE. These responses vary by `Accept-Language` only once
 * a second catalogue exists; advertising it with one locale fragments every
 * shared cache for no benefit, on pages a search engine crawls. Derived from the
 * catalogues rather than hardcoded, so the day a second language lands the
 * header appears with it and nobody has to remember.
 */
function localeFor(c: { req: { header: (n: string) => string | undefined } }): { locale: string; vary: Record<string, string> } {
  return { locale: negotiate(c.req.header("accept-language")), vary: localeVary() };
}

/** The Vary half of `localeFor`, which reads no request: a response that renders no
 * locale (a JSON or error 404 cached like its 200) takes only this, so `localeFor`
 * stays one call per SSR render (psa-i18n.test.ts counts them). */
function localeVary(): Record<string, string> {
  return supportedLocales().length > 1 ? { Vary: "Accept-Language" } : {};
}

/**
 * #183 — THE DISCOVERY RAILS CARRY A LIVE ANSWER, SO THEY GET NO REUSE WINDOW.
 *
 * `/discover` and `/browse/:category` render one artifact plus ONE live query.
 * The artifact is sweep-precomputed and hours old on purpose, and the product
 * says as much ("the checks fans run catch up at the next rebuild, within a few
 * hours" — i18n `how.wrong_off`). The live query is `dropSuppressed`, which
 * exists precisely "so an opt-out takes effect immediately, not at the next
 * sweep" (rankings.ts), and which removal.ts states as the stronger commitment
 * still: a suppressed subject is INDISTINGUISHABLE from a never-listed one.
 *
 * These two routes used to answer `public, max-age=300`. That number was
 * inherited from the ARTIFACT-read routes (`card.svg`, `card/ratings`), whose
 * bodies really are five minutes' worth of stale-safe; it was never reasoned
 * about for a body that computes a delisting per request. Its effect was to
 * license every browser and shared cache to replay the PRE-suppression page —
 * a delisted creator's passport id and display name — for five more minutes,
 * in stores this service can neither see nor purge, while the subject who asked
 * to be gone has no way to know. The live query was computed and then handed
 * back at the HTTP layer. Half a promise is the wrong half to keep.
 *
 * WHY max-age=0 AND NOT A SMALLER NUMBER. The guarantee is binary. At 60s a
 * suppressed subject is still distinguishable from a never-listed one for 60s,
 * which is the same defect with a smaller constant; there is no N>0 that is
 * consistent with "immediately". So the freshness lifetime of a response is the
 * MINIMUM of the freshness of its parts, and the live part's is zero.
 * `must-revalidate` is the half that matters operationally: it forbids the one
 * behaviour that does the harm — a cache serving the stale copy — including
 * under the stale-on-error and stale-on-disconnect allowances.
 *
 * WHY NOT A CONSENT-GATED-CATEGORY SPLIT (`private`/`no-store` on adult only).
 * Wrong axis. After #184 a consent-gated category has no board and no names in
 * its body at all, so that split would harden the one page with nothing left to
 * protect and leave the other sixteen — where the delisting risk is identical —
 * exactly as they were. Delisting is not category-shaped.
 *
 * WHAT IT COSTS, PLAINLY. These routes lose browser and proxy reuse: every view
 * is an origin render. Two things bound that cost. First, `public` is KEPT and
 * `no-store` is NOT used, so the response stays shareable and storable — this
 * is a revalidation requirement, not a withdrawal of caching from the crawler
 * front door, and it leaves the door open to serving a real 304 once these
 * routes carry an ETag (the way to buy the origin relief back without selling
 * the immediacy). Second, the relief was largely notional already: as three
 * separate comments in this tree record, a bare `Cache-Control` does not
 * edge-cache a Worker response absent a Cache Rule — so the header was not
 * saving the origin work today, it was only standing by to authorize a future
 * Cache Rule to do the harmful thing. What actually bounds origin load on these
 * two routes is the PS-093 per-IP limit below, which is a control this service
 * enforces rather than a hint to infra that may or may not be listening.
 */
const DISCOVERY_RAIL_CACHE = "public, max-age=0, must-revalidate";

// SUPPORTER QUEUE PRIORITY — the ONE place money is allowed near the report
// path, and the exact reason tests/invariants/paywall.test.ts carries a
// region exemption for the two functions below. (Product decision 2026-08-15;
// the `fan_report_reporters.supporter_boost` column.)
//
// WHAT IT BUYS: queue ORDER, and only among reports whose EARNED priority is
// already comparable (reports.ts `compareQueueItems` /
// `PRIORITY_COMPARABILITY_STEP`). An operator looks at a supporter's report
// sooner. That is the entire product.
//
// WHAT IT MUST NEVER BUY, and does not: corroboration (caution.ts still gates
// a machine-published Caution on `MAX(earned_band) >= 10`, the EARNED band), the
// per-reporter daily cap (buying volume is buying corroboration one step
// removed), the priority SCORE itself, or any decision to publish a warning.
// Weight is earned; only attention is for sale.
//
// The two functions are deliberately split so each exemption names the
// narrowest possible symbol set: one asks "can anyone pay at all here?", the
// other asks "did this fan pay?". Neither is a route, and neither can be
// widened without editing the guard's allowlist in a visible, pinned diff.
// ===========================================================================

/**
 * Is a billing processor configured at all? Absent config ⇒ nobody on this
 * deployment can have paid ⇒ the whole feature is inert and every report sorts
 * by its earned signal alone. Config PRESENCE only — never a subject's tier.
 */
function supporterQueueEnabled(env: Record<string, unknown>): boolean {
  return stripeConfigFromEnv(env) !== null || appleIapConfigFromEnv(env) !== null || googlePlayConfigFromEnv(env) !== null;
}

/**
 * The purchased ORDERING signal for one fan reporter, resolved at report time.
 *
 * PAYWALL-GUARD EXEMPTION, DELIBERATE AND NARROW: this function reads a fan's
 * paid tier. It is exempted BY FUNCTION, not by route or module, and the value
 * it returns reaches exactly one consumer — the operator queue's sort order. It
 * must never widen to corroboration, to the per-reporter cap, or to anything
 * that decides publication. `tests/invariants/paywall.test.ts` (see the
 * "supporter queue priority is ORDERING ONLY" test) pins that separation, and
 * `test/supporter-queue-priority.test.ts` proves behaviourally that a queue of
 * paying-but-untrusted reporters cannot cross the Caution threshold.
 *
 * FAIL NEUTRAL, NEVER OPEN: an unreadable entitlement, a missing table, or any
 * thrown error yields 0. There is no branch that returns 1 on an unknown.
 */
async function supporterQueueBoost(db: DbLike, accountId: string, env: Record<string, unknown>, nowMs: number): Promise<SupporterQueueBoost> {
  try {
    if (!supporterQueueEnabled(env)) return 0;
    return (await effectiveFanTier(db, accountId, nowMs)) === "free" ? 0 : 1;
  } catch {
    return 0; // an unreadable entitlement lowers no report and raises none
  }
}

// --- S65: fan reports (docs/FAN-REPORT-TRIAGE-PLAN.md) ---
// Session REQUIRED (locked decision): a fan session reports at its EARNED band,
// a creator session at the verified band only if the passport is actually
// verified right now (claimed → trusted band). Anonymous → 401. The real-link
// guard, throttles, dedup, and evidence live in reports.ts.
//
// MONEY IS NOT AN INPUT TO THE BAND. This route used to resolve the fan band

  }

// --- S99: self-service recovery REQUEST intake ---
// A signed-in fan/creator/agency reports a passport they can't fix themselves
// (lost passkey / hijack / dispute). This is a REQUEST, not an action: it enqueues
// an operator-adjudicated `dispute` case and executes nothing. Session required
// (the requester need not control the disputed passport), rate-limited + dedup'd
// in recovery.ts. The requester never chooses the remediation — the operator does.
// ============================================================================
// THE CREATOR'S WAY BACK IN — the two legs of an inbox proof (0043)
// ============================================================================
//
// `POST /recovery/request` below resolves fan → creator → agency and 401s with
// no principal. So a creator who has lost their passkey and whose 30-day
// session has expired cannot file the very request this product points them at:
// the one door is behind the credential they lost. Fans have a complete channel
// here — verified backup email, unauthenticated OTP sign-in, a last-way-in
// guard — and creators, whose passports carry payout destinations, have none.
//
// These two legs close that, and ONLY that. A proven inbox files into the
// operator queue with an identity attached. It mints no session, enrols no
// passkey and changes nothing on the passport — see creator-recovery.ts for why
// a session would have been one hop from the payout address.
/** Redemption attempts per masked IP per hour. Named rather than inline so it
 * cannot drift silently — every sibling cap in this file comes from a constant
 * or from `otp-policy.ts`. Generous against a creator's typos (the per-CODE
 * budget is 5 and is the real ceiling), tight enough that blind guessing across
 * a refreshed code set is not free. */
const CREATOR_RECOVERY_REDEEM_PER_IP_HOUR = 60;

/** M27 status reads per masked IP per hour. Matched to `REMOVAL_STATUS_PER_IP_HOUR`
 * because it is the same act on the same shape of capability: somebody who holds
 * a ticket checking back — half of it, because a recovery has one ticket where a
 * removal filer may hold several. Generous for a person refreshing a page, and — since
 * the bucket is consumed BEFORE the lookup — it bounds blind ticket guessing
 * without the limiter itself becoming the existence signal. */
const RECOVERY_STATUS_PER_IP_HOUR = 60;
/**
 * RFC 8058 one-click POSTs per masked IP per hour.
 *
 * THIS BUCKET IS AIMED AT THE ONE CALLER THE ROUTE EXISTS FOR, which is the
 * uncomfortable fact to state before picking the number. `fanIpAllowed` keys on
 * a /24 (`maskIp`), and the legitimate caller is a mailbox provider's egress
 * range — Gmail's and Yahoo's one-click POSTs for one sender concentrate onto a
 * handful of /24s. A cap tuned like the fan-facing routes would refuse the
 * provider, and a refusal here is not a neutral 429: it is a fan pressing
 * "unsubscribe", their client reporting success, and the mail continuing to
 * arrive.
 *
 * So it is deliberately set three orders of magnitude above any plausible honest
 * rate — one POST per second, sustained, from a single /24 — because what it is
 * actually defending is the database against an unauthenticated flood, and
 * nothing else. Guessing the token space is infeasible at 128 bits, so this is
 * not a brute-force control and must not be tuned as if it were.
 *
 * IT IS CONSUMED BEFORE THE LOOKUP, so the limiter cannot become a "does this
 * token exist" oracle — pinned by test.
 *
 * KNOWN GAP, written down rather than discovered: the bucket is SHARED per /24,
 * so an attacker who spends 36000 requests in an hour can deny one provider's
 * range for the remainder of it. That is a real cost of keying on the caller's
 * address for a route whose caller is infrastructure; the next step is a
 * provider-agnostic global bound once this route's real cost profile is
 * measured, not a lower per-IP cap.
 */
const UNSUBSCRIBE_PER_IP_HOUR = 36000; // I3-0615 (D-0920): ten per second per /24, sustained — a bulk send's one-click POSTs from one provider range

// --- Caution soft class: public CONTEST intake ---
// Anyone (no session) can dispute a caution by the profile's URL. This only
// ENQUEUES an operator review item — it never changes what is served. The response
// is uniform (`{ok:true}`) whether or not a caution/dispute exists, so it's not an
// oracle; a per-IP cap + a per-subject rate limit + open-dispute dedup bound the queue.
const CAUTION_CONTESTS_PER_IP_HOUR = 20;
// A-15: the per-subject budget below is the ONLY thing standing between a
// contest and the operator queue, and it is shared across every caller — an
// address well inside its own 20/hour cap can burn all 5 of a SUBJECT's daily
// slots by itself, and the uniform `{ok:true}` response means the subject's
// own later, genuine contest is silently dropped with no sign anything went
// wrong. Capping ONE address to ONE of the five daily slots per subject (this
// is 1/day, not the hourly IP cap above) does not stop a distributed attacker,
// but it closes the trivial single-actor case and forces the cost the
// per-subject cap was written to impose in the first place.
const CAUTION_CONTESTS_PER_IP_PER_SUBJECT_DAY = 1;
app.post("/caution/contest", anonBodyLimit, async (c) => {
  // M78: a PER-IP cap, BEFORE any canonicalize/hash work or the per-subject cap.
  // The per-subject cap alone bounded one target but not one ADDRESS: a script
  // could iterate distinct subjects, each with its own fresh 5-budget, writing a
  // rate_limits row per fabricated hash without bound. Gating the address first
  // bounds both the operator-queue pressure (now that population-1 subjects
  // enqueue too, per M34) and the rate_limits row growth to the IP budget.
  if (!(await fanIpAllowed(c, "cautioncontest", CAUTION_CONTESTS_PER_IP_HOUR))) return c.json(refusal("rate limit exceeded", RATE_LIMIT_MESSAGE), 429);
  const body = (await c.req.json().catch(() => null)) as { url?: unknown; reason?: unknown } | null;
  if (!body || typeof body.url !== "string") return c.json({ error: "url required" }, 400);
  let hash: string;
  try {
    let subject = canonicalize(body.url);
    // A url|norm fallback folds to its DOMAIN. A generic `link` (M3 Slice 4) does
    // NOT (I3-1165): a report on a link is filed under the link's own subject
    // (reports.ts fileReport folds url|norm only), so folding it here contested
    // a domain nobody reported and silently did nothing.
    if (subject.platform === "url" || subject.idKind === "norm") {
      let host = "";
      try {
        host = new URL(body.url).hostname.toLowerCase();
      } catch {
        host = "";
      }
      const dom = host ? canonicalAccount("domain", "domain", host) : null;
      if (!dom) return c.json({ error: "that link is not a supported profile" }, 400);
      subject = dom;
    }
    hash = await subjectHash(subject.canonical);
  } catch {
    return c.json({ error: "that link is not a supported profile" }, 400);
  }
  const db = d1(c.env.TRUST_DB);
  // A-15: one address gets at most ONE of the subject's five daily slots.
  // Checked and consumed BEFORE the shared per-subject budget, so an address
  // that has already used its slot against this subject today never touches
  // the shared counter again — same uniform response either way.
  if (!(await consumeRateLimit(db, fanIpBucketKey(c, `cautioncontest-subj:${hash}`), Date.now(), CAUTION_CONTESTS_PER_IP_PER_SUBJECT_DAY))) {
    return c.json({ ok: true }, 200);
  }
  // Per-subject cap: one target can't flood the operator queue. Over the cap we
  // still return the same benign response (never reveal the throttle or the class).
  if (await consumeRateLimit(db, `caution-contest:${hash}`, Date.now(), 5)) {
    try {
      await contestCaution(db, hash, typeof body.reason === "string" ? body.reason : null);
    } catch {
      /* contest is best-effort; never surface internal state */
    }
  }
  return c.json({ ok: true }, 200);
});

// #157 RV-01 (D-0921(2), I16-W04C hand-off, landed by I16-L05): SPEND A RE-ENTRY GRANT. A creator whose hijack
// resolution reset every passkey is mailed a single-use, 72-hour grant at the one contact address the operators
// spared (recovery.ts issueReentryGrant, dual control). Redeeming it mints a creator session at the passport's
// CURRENT epoch, from which the console registers a new passkey. Shaped like POST /recovery/status: the bearer
// travels in the BODY (never a logged URL), the rate bucket is spent BEFORE the lookup, and every refusal —
// unknown, malformed, expired, used, or an address changed since issue — is the same 404 (redeemReentryGrant).
// The session secret is checked BEFORE the grant is consumed, so a misconfigured deploy cannot burn a grant.
const REENTRY_REDEEMS_PER_IP_HOUR = 20;
app.post("/recovery/reentry/redeem", anonBodyLimit, async (c) => {
  if (!(await fanIpAllowed(c, "rgrredeem", REENTRY_REDEEMS_PER_IP_HOUR))) return c.json(refusal("rate limit exceeded", RATE_LIMIT_MESSAGE), 429);
  const body = (await c.req.json().catch(() => null)) as { grant?: unknown } | null;
  if (!body || typeof body.grant !== "string") return c.json({ error: "grant required" }, 400);
  if (!usableSessionSecret(c.env.SESSION_SECRET)) return c.json({ error: "sessions_unavailable" }, 503);
  const db = d1(c.env.TRUST_DB);
  let passportId: string;
  const redeemedAt = Date.now();
  try {
    passportId = (await redeemReentryGrant(db, body.grant, redeemedAt)).passport_id;
  } catch (e) {
    if (e instanceof RecoveryError && e.status === 404) return c.json({ error: "not_found" }, 404);
    throw e;
  }
  // I22-L01 (I3-0338#2): a session-epoch bump AFTER the grant was issued (the creator's own logout-all, or any
  // later recovery) kills the grant, like the OAuth hand-off's bump-after-grant refusal. One read binds the grant this
  // call just spent (used_at = redeemedAt) to the epoch row, and the session is minted AT the epoch it compared
  // (INV-058), never at a re-read one. Same 404 as every other refusal.
  const bound = await db
    .prepare(
      `SELECT g.issued_at AS issued_at, g.issued_epoch AS issued_epoch, s.session_epoch AS ep, s.updated_at AS bumped_at FROM creator_reentry_grants g
        LEFT JOIN creator_sessions s ON s.passport_id = g.passport_id WHERE g.passport_id = ? AND g.used_at = ? LIMIT 1`,
    )
    .bind(passportId, new Date(redeemedAt).toISOString())
    .first<{ issued_at: string; issued_epoch: number | null; ep: number | null; bumped_at: string | null }>();
  if (!bound) return c.json({ error: "not_found" }, 404);
  const ep = bound.ep ?? 0;
  // migration 0212 stamps issued_epoch at INSERT; a grant issued before it is judged by the bump instant instead.
  if (bound.issued_epoch !== null ? bound.issued_epoch !== ep : bound.bumped_at !== null && bound.bumped_at > bound.issued_at) return c.json({ error: "not_found" }, 404);
  const session = await issueSession(c.env.SESSION_SECRET, { kind: "creator", id: passportId, ep }, Math.floor(Date.now() / 1000));
  if (!session) return c.json({ error: "sessions_unavailable" }, 503);
  return c.json({ ok: true, passport_id: passportId, session });
});

// C-37: `grantCredits` REFUSES a settlement into a frozen (suspended/terminated)
// account by design (credits.ts — "a settlement to a frozen account is a refund
// question"), and that refusal is right: crediting an account nobody can spend
// from is worse. But every one of its three callers below answered the refusal
// with a bare `{ok:true, granted:false}` and NOTHING ELSE — no console line, no
// audit_log row — while every SIBLING refusal branch in this same file (unknown
// bucket, paid-below-backing, partial refund) logs for manual review. Money a
// rail actually SETTLED (Stripe/crypto/IAP already took it) then left no trace
// on our side at all: not in `credit_purchases` (grantCredits never inserts on
// this branch) and not in any operator-visible log — the "refund question" the
// design docblock says this is was never handed to anyone who could answer it.
// This single helper is the fix for all three call sites (one mechanism, one
// patch) — it does not change what is granted, only whether the refusal is seen.
// I3-0687: returns false when the audit row could not be written, and every caller
// answers 503 (record-then-ack) so the rail redelivers; the WHERE NOT EXISTS below
// makes that redelivery write exactly one row.
export async function logFrozenSettlement(db: DbLike, cid: string, rail: "stripe" | "crypto" | "apple" | "google", accountId: string, grossCents: number, credits: number, externalTxnId: string): Promise<boolean> {
  // accountId/externalTxnId are deliberately NOT passed to logError — see
  // REPORT.md's flagged live leak: this call site used to log a raw
  // account id and a raw payment-rail transaction id. audit_log below still
  // records them (a compliance ledger, not a log surface INV-004 governs).
  logError({ cid, hop: "http", msg: "credit settlement into frozen account", rail, grossCents, credits });
  try {
    // PSEUDONYMIZE AT WRITE TIME, because this row is written AFTER erasure.
    // credits.ts:81's second limb routes an ERASED account down this very
    // branch, so the settlement that follows an erasure would mint a FRESH
    // row naming the erased id — a row deleteAccount's erasure batch (account_data.ts) already
    // ran and can never reach (INV-061). `CASE … EXISTS (accounts)` writes
    // the raw id only while the account still exists (a suspension, which
    // the sweep does cover) and otherwise the SAME `del_<hex>` shape erasure
    // writes — indistinguishable from a tombstoned row, so it is not a
    // durable "this id was erased" marker. The operator handle for a stray
    // settlement is the rail's `externalTxnId` in `detail`, which is intact
    // either way. One statement: nothing is read and then acted on.
    await db
      .prepare(
        "INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) " +
          "SELECT ?, 'credit:frozen_settlement', ?, ?, " +
          "CASE WHEN EXISTS (SELECT 1 FROM accounts WHERE account_id = ?) THEN ? ELSE 'del_' || lower(hex(randomblob(16))) END, 'system' " +
          // I3-1606: a rail REDELIVERY of the same settlement is the same refund
          // question — one row per (rail, txn), keyed on the detail it writes.
          "WHERE NOT EXISTS (SELECT 1 FROM audit_log WHERE action = 'credit:frozen_settlement' AND detail = ?)",
      )
      .bind(`rail:${rail}`, JSON.stringify({ rail, grossCents, credits, externalTxnId }), new Date().toISOString(), accountId, accountId, JSON.stringify({ rail, grossCents, credits, externalTxnId }))
      .run();
    return true;
  } catch (e) {
    logError({ cid, hop: "http", msg: "credit_frozen_settlement_audit_write_failed", err: e instanceof Error ? e.name : typeof e });
    return false;
  }
}

/**
 * I3-0726 d (D-0993(1), census pick (a)): Play's voided-purchases list can lag a real void by minutes.
 * A void Play does not list yet is RETRIED (503, Pub/Sub redelivers) for this long after its order id was
 * first seen, then parked for an operator as before. The first-seen instant is an insert-once
 * `credit:store_void_pending` audit row keyed by the order id (append-only; no new table).
 */
export const PLAY_VOID_LAG_WINDOW_MS = 24 * 60 * 60 * 1000;
export const PLAY_VOID_PENDING_ACTION = "credit:store_void_pending";
export const playVoidPendingDetail = (externalTxnId: string): string => JSON.stringify({ rail: "google", externalTxnId });
/** The order id's first-seen instant (ms), recording it on first sight. Throws when the audit store is down. */
async function playVoidFirstSeenMs(db: DbLike, externalTxnId: string, nowMs: number): Promise<number> {
  const detail = playVoidPendingDetail(externalTxnId);
  await db
    .prepare(
      "INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) " +
        `SELECT 'rail:google', '${PLAY_VOID_PENDING_ACTION}', ?, ?, NULL, 'system' ` +
        `WHERE NOT EXISTS (SELECT 1 FROM audit_log WHERE action = '${PLAY_VOID_PENDING_ACTION}' AND detail = ?)`,
    )
    .bind(detail, new Date(nowMs).toISOString(), detail)
    .run();
  const row = await db.prepare(`SELECT MIN(created_at) AS at FROM audit_log WHERE action = '${PLAY_VOID_PENDING_ACTION}' AND detail = ?`).bind(detail).first<{ at: string | null }>();
  const at = row?.at ? Date.parse(row.at) : Number.NaN;
  return Number.isFinite(at) ? at : nowMs;
}

// --- M7: Stripe Billing (the real rail) -------------------------------------
// One webhook + one checkout + one portal for BOTH products (Fan Premium,
// Creator Pro), backed by the non-ledger `billing_entitlements` table.
// Money is not trust: nothing here writes a trust_events row, and nothing that
// SERVES A VERDICT may read what it writes (tests/invariants/paywall.test.ts).

// Signature-gated, NOT a session. Absent STRIPE_WEBHOOK_SECRET → 503 (fail
// closed: an unconfigured webhook grants nothing, ever). The signature is
// verified over the RAW body before a single field is trusted, the timestamp
// must be inside Stripe's ±5-minute window, and the event id makes an
// at-least-once redelivery a no-op.
app.post("/billing/webhook/stripe", anonBodyLimit, async (c) => {
  const secret = stripeWebhookSecret(c.env as unknown as Record<string, unknown>);
  if (!secret) return c.json({ error: "billing_not_configured" }, 503);
  const raw = await c.req.text();
  if (raw.length > 64 * 1024) return c.json({ error: "too_large" }, 413);
  const nowMs = Date.now();
  const ok = await verifyStripeSignature(raw, c.req.header("Stripe-Signature") ?? "", secret, nowMs);
  if (!ok) return c.json({ error: "bad_signature" }, 400);
  const parsed = parseStripeEvent(raw);
  if (!parsed) return c.json({ error: "bad_event" }, 400);
  // #747, D-0855(1): a checkout since #747 carries a SEALED subject; open it
  // under SUBJECT_SEAL_KEY (+ PREVIOUS), never the webhook secret Stripe holds.
  // Legacy raw metadata passes through untouched; a seal that does not open —
  // or any seal while SUBJECT_SEAL_KEY is unset — attributes nothing.
  const sealKeys = subjectSealKeys(c.env as unknown as Record<string, unknown>);
  if (!sealKeys) logError({ cid: c.get("cid"), hop: "http", msg: "subject_seal_key_missing", rail: "stripe" });
  const { event, unopened } = await openStripeSubjects(parsed, sealKeys, { db: d1(c.env.TRUST_DB) });
  if (unopened > 0) logError({ cid: c.get("cid"), hop: "http", msg: "stripe_subject_seal_unopened", count: unopened });
  // S-CW6a: a completed, paid CREDIT checkout grants credits — idempotent on
  // the session id (credit_purchases UNIQUE), a wholly separate path from the
  // subscription state machine. Everything else falls through below.
  // S-CW6d: a refund or dispute on a credit purchase reverses the grant. Maps to
  // the purchase via the PaymentIntent (never the session id). D-0749: a
  // chargeback (at open), an escalation or a loss reverses in proportion to
  // `disputedCents` (D-0745) and bans; an INQUIRY only freezes the purchase's
  // credits and a won/warning_closed releases them (D-0744); a full refund
  // claws credits without a ban. A PI that isn't
  // a credit purchase (e.g. a subscription charge) finds no row and is a safe
  // no-op. Idempotent on the purchase (credit_reversals UNIQUE). Runs BEFORE the
  // grant branch — a refund/dispute and a completed checkout never coincide.
  const refundIntent = refundIntentFromEvent(event);
  if (refundIntent) {
    // Review F4: a partial refund is surfaced for manual review, never
    // auto-clawed (a fixed credit bucket is not something we partial-refund).
    //
    // SCOPED TO CREDIT PURCHASES, for exactly the reason the FULL-refund branch
    // below now falls through. This branch used to answer 200 for ANY partial
    // refund, and a subscription charge is indistinguishable from a credit
    // purchase here until somebody looks the PaymentIntent up — so a partial
    // refund on a subscription never reached `applyStripeOutcome` and never
    // wrote its dedup marker, one branch over from the swallow that had made
    // the whole subscription-reversal path unreachable. The subscription
    // reducer happens to answer `ignore` for a positively-identified partial
    // refund today (money partly returned is not the purchase reversed), so the
    // OUTCOME was the same — but an early return in front of a reducer, taken
    // on a question this branch has not asked, is the defect, and it bites the
    // moment either side's partial handling changes.
    //
    // I21-Q70 / I21-Q86 (D-1030 (a) "in proportion"; I29-REVERSALS H1, landed by I29-INT): a partial refund now takes
    // back its share through `reverseGrant` below, keyed by Stripe's CUMULATIVE `amount_refunded` (minor units, read
    // once here from the verified event) as both the take-back figure and the recorded cash figure. Only a partial whose
    // figure is unreadable (not a positive safe integer) stays here: it is parked for manual review, never clawed whole.
    const refundedObj: Record<string, unknown> = event.object ?? {};
    const amountRefunded = Number.isSafeInteger(refundedObj.amount_refunded) && (refundedObj.amount_refunded as number) > 0 ? (refundedObj.amount_refunded as number) : null;
    // I3-2210 b / REFUND-CURRENCY (I30-CONSOLE H3): Stripe's `currency` for the refunded figure; null when absent or malformed.
    const refundCurrency = typeof refundedObj.currency === "string" && /^[a-z]{3}$/.test(refundedObj.currency) ? refundedObj.currency : null;
    if (refundIntent.partial && amountRefunded === null) {
      const creditRow = await d1(c.env.TRUST_DB)
        .prepare("SELECT purchase_id FROM credit_purchases WHERE payment_ref = ?")
        .bind(refundIntent.paymentRef)
        .first<{ purchase_id: string }>();
      if (creditRow) {
        logError({ cid: c.get("cid"), hop: "http", msg: "stripe_partial_credit_refund_manual_review" });
        // S-065-batch FX-02: a log line alone is not a queryable operator
        // trail -- the sibling "no matching purchase" case already writes a
        // durable `audit_log` row (refunds.ts:128-144, refund:unreconciled_
        // dispute) for exactly this reason. Best-effort, same as that sibling:
        // never let logging block the (correct) manual-review ack.
        // I3-1225: the row IS the manual-review queue (nothing else records a
        // partial refund), so it is no longer best-effort: a failed write
        // answers 500 and Stripe redelivers; a redelivery of a written event is
        // a no-op (NOT EXISTS on the same detail, which carries the event id).
        // I3-2210 b (I22-L01): the review row says HOW MUCH was refunded and in WHAT currency (Stripe's charge
        // `amount_refunded`, minor units, and `currency`), read from the verified event; null when absent or malformed.
        const detail = JSON.stringify({ paymentRef: refundIntent.paymentRef, purchaseId: creditRow.purchase_id, eventId: refundIntent.eventId, amountRefunded, currency: refundCurrency });
        try {
          await d1(c.env.TRUST_DB)
            .prepare(
              "INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, 'refund:partial_credit_manual_review', ?, ?, ?, 'system' " +
                "WHERE NOT EXISTS (SELECT 1 FROM audit_log WHERE action = 'refund:partial_credit_manual_review' AND detail = ?)",
            )
            .bind(`rail:stripe`, detail, new Date().toISOString(), creditRow.purchase_id, detail)
            .run();
        } catch (e) {
          logError({ cid: c.get("cid"), hop: "http", msg: "stripe_partial_credit_refund_trail_failed", err: e instanceof Error ? e.name : typeof e });
          return c.json({ error: "error" }, 500);
        }
        return c.json({ ok: true, partial: true }, 200);
      }
      // Not a credit purchase → this is somebody else's event. Fall through
      // to the subscription reducer below, exactly as the full-refund branch
      // does when `reverseGrant` answers `unknown`.
    } else {
      try {
        const r = await reverseGrant(
          d1(c.env.TRUST_DB),
          {
            rail: "stripe",
            kind: refundIntent.kind,
            processorEventId: refundIntent.eventId,
            paymentRef: refundIntent.paymentRef,
            disputedCents: refundIntent.disputedCents,
            disputeId: refundIntent.disputeId,
            // I21-Q70: a partial's cumulative refunded cents set the take-back; a full refund takes the whole grant.
            ...(refundIntent.partial ? { refundedCents: amountRefunded ?? undefined } : {}),
            // I21-Q86: the cash the processor reported refunded, recorded on the reversal row (refunds only).
            ...(refundIntent.kind === "refund" ? { cashRefundedCents: amountRefunded } : {}),
            // REFUND-CURRENCY (I30-CONSOLE H3, 0268): the currency of that cash figure (refunds.ts validates again).
            ...(refundIntent.kind === "refund" ? { currency: refundCurrency } : {}),
          },
          () => "crv_" + crypto.randomUUID().replace(/-/g, "").slice(0, 20),
          () => "cle_" + crypto.randomUUID().replace(/-/g, "").slice(0, 20),
          () => new Date().toISOString(),
          ...analyticsPublishArgs(c),
        );
        if (r.reversed) logError({ cid: c.get("cid"), hop: "http", msg: "credit_reversal", kind: refundIntent.kind });
        // FALL THROUGH WHEN THIS PAYMENT WAS NOT A CREDIT PURCHASE. `unknown` is
        // reverseGrant's answer for "no credit_purchases row behind this
        // PaymentIntent", which is what EVERY SUBSCRIPTION CHARGE looks like from
        // here — and this branch used to answer 200 for it and return, so
        // applyStripeOutcome never saw a `charge.refunded` or a
        // `charge.dispute.created` at all and the whole subscription reversal
        // path (stripe.ts chargeReversal → REVOKE_BY_SUBSCRIPTION /
        // REVOKE_BY_CUSTOMER) was unreachable in production: correct, tested, and
        // inert. The two rails are mutually exclusive on one charge — a payment
        // is a credit bucket or a subscription invoice, never both — so handing
        // an unrecognised one on costs the credit rail nothing.
        // `reversed` (a credit purchase was clawed) and `duplicate` (it already
        // had been) both mean the credit rail owns this event: answer for it.
        if (r.reversed || r.reason !== "unknown") return c.json({ ok: true, reversed: r.reversed });
      } catch (e) {
        logError({ cid: c.get("cid"), hop: "http", msg: 'stripe refund webhook', err: e instanceof Error ? e.name : typeof e });
        return c.json({ error: "error" }, 500); // a lost reversal write → Stripe redelivers
      }
    }
  }
  const creditIntent = creditIntentFromEvent(event);
  if (creditIntent) {
    // Re-validate the bucket even though we set the metadata (defense in
    // depth): an amount we did not offer is never granted.
    if (!(CREDIT_BUCKETS as readonly number[]).includes(creditIntent.credits)) {
      logError({ cid: c.get("cid"), hop: "http", msg: "stripe_credit_webhook_unknown_bucket", credits: creditIntent.credits });
      return c.json({ ok: true, ignored: true }, 200); // signed but not a bucket we sell — do not retry
    }
    // Review F1: bind the credits granted to the money actually PAID. Tax is inside
    // the bucket price by default and on top of it when the operator sets it exclusive
    // (I21-Q55), so amount_total must be at least the bucket's backing value either way; anything less means a discount/coupon/partial-capture path
    // we never offer — fail closed (grant nothing) rather than hand out a full
    // bucket for less money. 200 so Stripe does not retry a decision we made.
    if (creditIntent.grossCents < creditIntent.credits * CREDIT_BACKING_CENTS) {
      logError({ cid: c.get("cid"), hop: "http", msg: "stripe_credit_webhook_paid_below_bucket_backing", grossCents: creditIntent.grossCents, credits: creditIntent.credits });
      return c.json({ ok: true, ignored: true }, 200);
    }
    try {
      // IA-02 (W3-08 invariant-auditor). THE TWO RAILS NOW REFUSE THE SAME SET.
      //
      // The crypto rail has refused a chargeback-BANNED fan since review (LOW)
      // — "the ban would otherwise keep accruing spendable credits" — and this
      // rail had no such gate: it went straight to `grantCredits`, which refuses
      // only a FROZEN account. `accountBanned` (accounts.banned_at) and
      // `accountFrozenForCredit` are DISJOINT predicates, so a fan who was
      // chargeback-banned but never frozen was refused on crypto and credited on
      // Stripe, for the same money. The auditor measured that adding this gate
      // reddened nothing and removing crypto's reddened nothing either: the
      // behaviour was not decided, it was unspecified.
      //
      // WHY THIS DIRECTION, and the two are not equivalent. A Stripe charge is
      // REVERSIBLE and a crypto settlement is not. Refusing here is therefore the
      // recoverable error — if the ban was wrong the payment can be refunded and
      // the credits granted — while crediting a banned account is the error that
      // cannot be taken back, on the one rail where it is cheapest to repeat.
      // The ban exists to stop exactly this accrual; it now stops it everywhere.
      if (await accountBanned(d1(c.env.TRUST_DB), creditIntent.subjectId)) {
        if (!(await logFrozenSettlement(d1(c.env.TRUST_DB), c.get("cid"), "stripe", creditIntent.subjectId, creditIntent.grossCents, creditIntent.credits, creditIntent.externalTxnId))) return c.json({ error: "audit_unavailable" }, 503);
        // I21-Q80 (D-1030 (a) "record the refund owed"; I29-REVERSALS H1c, landed by I29-INT): the refused settlement's
        // refund_owed row (settlement-refusals.ts refundOwedOnBanStmt, INSERT OR IGNORE on the settlement id). A lost
        // write answers 503 like the audit write above, so Stripe redelivers.
        try {
          await refundOwedOnBanStmt(d1(c.env.TRUST_DB), { accountId: creditIntent.subjectId, rail: "stripe", externalTxnId: creditIntent.externalTxnId, paymentRef: creditIntent.paymentRef, grossCents: creditIntent.grossCents, credits: creditIntent.credits }, new Date().toISOString()).run();
        } catch (e) {
          logError({ cid: c.get("cid"), hop: "http", msg: "refund_owed_on_ban_unwritten", rail: "stripe", err: e instanceof Error ? e.name : typeof e });
          return c.json({ error: "audit_unavailable" }, 503);
        }
        return c.json({ ok: true, ignored: "suspended" }, 200);
      }
      const r = await grantCredits(
        d1(c.env.TRUST_DB),
        {
          accountId: creditIntent.subjectId,
          rail: "stripe",
          externalTxnId: creditIntent.externalTxnId,
          grossCents: creditIntent.grossCents,
          netCents: creditIntent.credits * CREDIT_BACKING_CENTS,
          credits: creditIntent.credits,
          paymentRef: creditIntent.paymentRef,
          // I3-1213 (D-0920): what Stripe reported — the session's charged amount; the fee is not known here.
          observed: { grossCents: creditIntent.grossCents, feeCents: null, currency: creditIntent.currency ?? null },
        },
        () => "cpu_" + crypto.randomUUID().replace(/-/g, "").slice(0, 20),
        () => "cle_" + crypto.randomUUID().replace(/-/g, "").slice(0, 20),
      );
      if (r.reason === "frozen" && !(await logFrozenSettlement(d1(c.env.TRUST_DB), c.get("cid"), "stripe", creditIntent.subjectId, creditIntent.grossCents, creditIntent.credits, creditIntent.externalTxnId))) return c.json({ error: "audit_unavailable" }, 503);
      if (r.granted) {
        // I3-1691 (0119): resolve the attempt this grant paid for. A session
        // carrying `dhwaja_nonce` resolves exactly its own row; a nonce-less one
        // (created before the nonce was stamped) claims the oldest OPEN attempt for
        // the same fan and bucket — each in ONE statement, so two concurrent grants
        // cannot both claim a row. Best-effort: the credits are already granted.
        try {
          const at = new Date(nowMs).toISOString();
          await (creditIntent.nonce
            ? d1(c.env.TRUST_DB)
                 // D-0843(1): a late webhook grant also clears a sweep's operator-review mark ('reconciled').
                .prepare("UPDATE credit_checkout_pending SET state = 'granted', resolved_at = ? WHERE nonce = ? AND account_id = ? AND credits = ? AND rail = 'stripe' AND state IN ('open','reconciled')")
                .bind(at, creditIntent.nonce, creditIntent.subjectId, creditIntent.credits)
            : d1(c.env.TRUST_DB)
                // TCHIGH H6 (I34-W2-MONEY H3): a sweep-reconciled row is still the attempt this grant paid for, as in the
                // nonce arm above; left 'reconciled' it would be counted by the spend cap beside the granted credits.
                .prepare(
                  "UPDATE credit_checkout_pending SET state = 'granted', resolved_at = ? WHERE state IN ('open','reconciled') AND nonce = (SELECT nonce FROM credit_checkout_pending WHERE account_id = ? AND credits = ? AND rail = 'stripe' AND state IN ('open','reconciled') ORDER BY created_at, nonce LIMIT 1)",
                )
                .bind(at, creditIntent.subjectId, creditIntent.credits)
          ).run();
        } catch (e) {
          logError({ cid: c.get("cid"), hop: "http", msg: "credit_checkout_pending_unresolved", err: e instanceof Error ? e.name : typeof e });
        }
      }
      return c.json({ ok: true, granted: r.granted });
    } catch (e) {
      // A lost write (not a duplicate) → 500 so Stripe redelivers.
      logError({ cid: c.get("cid"), hop: "http", msg: 'stripe credit webhook', err: e instanceof Error ? e.name : typeof e });
      return c.json({ error: "error" }, 500);
    }
  }
  try {
    // The configured price set is passed in so the GRANTED TIER is bound to
    // the price actually purchased, not inferred from the subject's id shape.
    // No prices configured → nothing can be granted (a lapse still applies).
    // I3-1353 (I23-W1-HUB H-3): the price ids ALONE (stripePricesFromEnv), not the whole checkout config — a signed
    // webhook grants on the price purchased even while STRIPE_SECRET_KEY / APP_ORIGIN are unset.
    const prices = stripePricesFromEnv(c.env as unknown as Record<string, unknown>);

    const r = await applyStripeOutcome(d1(c.env.TRUST_DB), event.id, event.type, event.created * 1000, stripeOutcome(event, nowMs, prices));
    // A write that was supposed to land and did not is NOT a success: answer
    // 5xx so Stripe redelivers (the dedup marker was withheld for exactly this).
    if (r.reason === "unresolved") {
      logError({ cid: c.get("cid"), hop: "http", msg: "stripe_webhook_unresolved", event_type: event.type });
      return c.json({ error: "unresolved" }, 503);
    }
    // 200 on duplicate/ignored too — a non-2xx would make Stripe retry an
    // event we have already (correctly) decided about.
    return c.json({ ok: true, applied: r.applied });
  } catch (e) {
    // A constraint failure here means the write was LOST. 500, so Stripe
    // retries — never a quiet 200 that buries a dropped entitlement.
    logError({ cid: c.get("cid"), hop: "http", msg: 'stripe webhook', err: e instanceof Error ? e.name : typeof e });
    return c.json({ error: "error" }, 500);
  }
});

  const creator = fan ? null : await creatorId(c);

  const subjectId = (fan ?? creator) as string;

/**
 * QUEUE #419 (D-0577, "ADD THE ABILITY TO CANCEL") — the ONE principal a
 * SUSPENDED fan resolves to, read at `GET /billing/portal` and at no other door.
 *
 * `fanId` folds a closure into "no session" (`oidc.ts currentSessionEpoch`), so
 * suspension used to lock the fan out of the only cancel lever while Stripe
 * kept charging. This resolves a fan token when ALL of: it verifies and is of
 * kind `fan`; the account's closure is `suspended` (terminated is refused);
 * its epoch equals the account's un-folded epoch (`accountSessionEpoch`, so
 * logout-everywhere still kills it); and it was ISSUED BEFORE `suspended_at`.
 *
 * THE LAST CLAUSE is what keeps B27-OIDC's closed-account DATA-RIGHTS session
 * off this door, whatever shape that session takes: every ordinary sign-in
 * path refuses a closed account, so a fan token minted after the suspension
 * can only be that scoped one. `iat` is read from the payload only after
 * `verifySession` has authenticated it. Not `fanIdForDataRights`, which would
 * admit exactly that session.
 *
 * D-0773 (#419 follow-up): that data-rights session IS admitted here, for a
 * `suspended` or `terminated` closure, and only here — recognised by its own
 * encoding (`ep === dataRightsEpoch(raw epoch)`, route-helpers.ts), never by
 * `fanIdForDataRights`. The door it reaches is the `frozen` branch below, which
 * serves `createStripeCancelFlowSession` and nothing else.
 */
async function suspendedFanForCancel(c: { env: Env; req: { header: (k: string) => string | undefined } }): Promise<string | null> {
  const token = /^Bearer\s+(.+)$/.exec((c.req.header("Authorization") ?? "").trim())?.[1];
  if (!token) return null;
  const s = (await verifySessionRotating(c.env, token, Math.floor(Date.now() / 1000)))?.subject ?? null; // I7-Q4 (I30-SESSION)
  if (!s || s.kind !== "fan") return null;
  const db = d1(c.env.TRUST_DB);
  const closure = await closureFor(db, s.id);
  if (!closure || (closure.state !== "suspended" && closure.state !== "terminated")) return null;
  const epoch = await accountSessionEpoch(db, s.id);
  if (epoch === null) return null;
  if (s.ep === dataRightsEpoch(epoch)) return s.id; // D-0773: the closed-account rights session, suspended or terminated
  if (closure.state !== "suspended" || s.ep !== epoch) return null;
  let iat: unknown;
  try {
    iat = (JSON.parse(atob(token.split(".")[0]!.replace(/-/g, "+").replace(/_/g, "/"))) as { iat?: unknown }).iat;
  } catch {
    return null;
  }
  const suspendedMs = Date.parse(closure.suspended_at);
  if (typeof iat !== "number" || !Number.isFinite(suspendedMs) || iat * 1000 >= suspendedMs) return null;
  return s.id;
}

 * answer GET /account/billing gives (Stripe folded in), not the store-only fold `applySubscriptionEvent`

export async function applyIapGrant(c: { env: Env }, accountId: string, grant: { ok: boolean; event_id: string; status: "active" | "canceled" | "expired" | "none"; current_period_end: string | null; event_time: string; event_tiebreak?: string; customer_id: string | null }): Promise<Response> {

}

// S-CW6e: IAP CREDIT purchases (consumables). The SAME fixed-backing credits as
// the web rail — the in-app SKU price covers the store cut (the fan pays the
// uplift), the granted credits are fixed by SKU. Server-verified against the
// store API, bound to the account's IAP token, idempotent on the store txn id
// (credit_purchases UNIQUE). A banned fan cannot buy. Fail closed without config.
// INT-14 (D-1051(3), I21-Q80 IAP limb): VERIFY FIRST, THEN REFUSE. A verified store purchase for a banned account is not
// granted and is recorded as refund owed (settlement-refusals.ts refundOwedOnBanStmt, keyed by the store transaction id),
// as the Stripe and crypto ban branches do; a lost write answers 503 so the app retries.
/** D-0843(2): THIS store transaction is owed back (a refund_owed row keyed by its external id), never "the account has
 * some refusal". The ONE read both grantIapCredits answers give as `refund_pending` (D-1059(15)). */
async function iapRefundPending(db: ReturnType<typeof d1>, accountId: string, rail: "apple" | "google", externalTxnId: string): Promise<boolean> {
  return (await db.prepare("SELECT 1 AS x FROM credit_settlement_refusals WHERE external_txn_id = ? AND rail = ? AND account_id = ? AND status = 'refund_owed'").bind(externalTxnId, rail, accountId).first()) !== null;
}
async function grantIapCredits(cid: string, db: ReturnType<typeof d1>, accountId: string, rail: "apple" | "google", v: CreditPurchaseVerified): Promise<Response> {
  try {
    const net = v.credits * CREDIT_BACKING_CENTS; // fixed backing; the store cut is not our revenue (reconciled to the store report)
    if (await accountBanned(db, accountId)) {
      if (!(await logFrozenSettlement(db, cid, rail, accountId, net, v.credits, v.externalTxnId))) return Response.json({ error: "audit_unavailable" }, { status: 503 });
      try {
        await refundOwedOnBanStmt(db, { accountId, rail, externalTxnId: v.externalTxnId, grossCents: net, credits: v.credits }, new Date().toISOString()).run();
      } catch (e) {
        logError({ cid, hop: "http", msg: "refund_owed_on_ban_unwritten", rail, err: e instanceof Error ? e.name : typeof e });
        return Response.json({ error: "audit_unavailable" }, { status: 503 });
      }
      // D-1059(15) (I30-RG-OPS NEW-iap-ban-refund-pending-claim): the guarded INSERT writes nothing for a purchase already
      // credited (or disputed/refunded) before the ban, so the answer reads THIS transaction's row back, as the arm below does.
      return Response.json({ error: "account_suspended", refund_pending: await iapRefundPending(db, accountId, rail, v.externalTxnId) }, { status: 403 });
    }
    const r = await grantCredits(
      db,
      { accountId, rail, externalTxnId: v.externalTxnId, grossCents: net, netCents: net, credits: v.credits, monthlyCapGate: (await import("./fan-spend-limits.js")).spendLimitCapOnlyGate(accountId, net, Date.now()) }, // I54 M1 a (INV-072, D-1399, J1): the fan's own monthly cap (not the pause), asked IN the purchase INSERT at nominal credits; over it, no grant and one refund_owed row keyed by the store transaction id. Imported in place: this room's grant of index.ts is the handler, not the import lines.
      () => "cpu_" + crypto.randomUUID().replace(/-/g, "").slice(0, 20),
      () => "cle_" + crypto.randomUUID().replace(/-/g, "").slice(0, 20),
    );
    if (r.reason === "frozen" && !(await logFrozenSettlement(db, cid, rail, accountId, net, v.credits, v.externalTxnId))) return Response.json({ error: "audit_unavailable" }, { status: 503 });
    // I4-18-R9 hand-off (D-0841(4)): the IAP grant answer carries a
    // `refund_pending` flag, so the app that just paid learns the settlement is
    // owed back, not merely ungranted. It is NOT the predicate GET
    // /account/war/credits serves (that one is account-scoped: any refund_owed row).
    // D-0843(2) (I4-RG-MONEY-05): this one is per TRANSACTION — THIS settlement is owed back —
    // never "the account has some refusal", which a genuine duplicate would inherit.
    return Response.json({ ok: true, granted: r.granted, credits: v.credits, balance: await creditBalance(db, accountId), refund_pending: await iapRefundPending(db, accountId, rail, v.externalTxnId) });
  } catch (e) {
    logError({ cid, hop: "http", msg: "iap_credit_grant", err: e instanceof Error ? e.name : typeof e });
    return Response.json({ error: "error" }, { status: 500 });
  }
}

app.post("/account/payout-tags", anonBodyLimit, async (c) => {

    const r = await setPayoutTags(db, creator, tags, Date.now(), () => new Date().toISOString(), (pid, changeId, change) =>
      buildPayoutChangeStatements(db, pid, changeId, () => new Date().toISOString(), change), { assertion, cfg },

    if (e instanceof VerifiedCreatorError) return c.json({ error: e.message }, e.status);

// The OAuth callback (public — Stripe redirects the creator's browser here with
// ?code&state). The `state` binds it to the passport; the exchanged account is
// stored PENDING, awaiting the passkey confirm below. No session is trusted
// here — the state is the binding, and nothing becomes a payout destination
// until the passkey confirm activates it.
// S6-18 (I59): per-/24 callbacks per hour that reach completeConnect; the value is the claim OAuth callback's.
const CONNECT_CALLBACKS_PER_IP_HOUR = 60; // DRAFT (Jason): JASON-QUESTIONS I59 1 (claim-routes.ts OAUTH_CALLBACKS_PER_IP_HOUR until answered)
app.get("/connect/oauth/callback", async (c) => {
  // I3-2004: Hono answers HEAD by running this GET handler and dropping the
  // body, which would spend the one-time `state` on a link prefetch.
  if (c.req.method === "HEAD") return c.body(null, 405, { Allow: "GET" });
  // I3-1728: every early failure (Stripe's own `?error=access_denied` carries
  // no code) lands in the console like the catch below (I3-1915) does.
  // S6-18 (I59): `message` is the per-network sentence an IP-keyed 429 carries (D-1004(1)); refusal(reason) alone is
  // byte-identical to the `{ error: reason }` every other early failure answered.
  const failTo = (reason: string, status: ContentfulStatusCode, message?: string) => {
    const back = consoleOrigin(c.env) ?? appOrigin(c.env);
    return back ? c.redirect(`${back}/#/earnings?connect_error=${encodeURIComponent(reason)}`, 302) : c.json(refusal(reason, message), status);
  };
  const cfg = connectConfigFromEnv(c.env as unknown as Record<string, unknown>);
  if (!cfg) return failTo("connect_not_configured", 503);
  const state = c.req.query("state") ?? "";
  const code = c.req.query("code") ?? "";
  if (!state || !code) return failTo(c.req.query("error") === "access_denied" ? "access_denied" : "state and code required", 400);
  try {
    // S6-18 (I59): decided before completeConnect reads the state, so a refused callback leaves the state unspent and
    // lands in the console like every early failure; a limiter fault lands there like a completeConnect one (I3-1915).
    if (!(await fanIpAllowed(c, "connectcb", CONNECT_CALLBACKS_PER_IP_HOUR))) return failTo("rate_limited", 429, RATE_LIMIT_MESSAGE);
    const r = await completeConnect(d1(c.env.TRUST_DB), cfg, state, code, Date.now());
    // Redirect the creator back into the console to run the passkey confirm.
    const back = consoleOrigin(c.env) ?? appOrigin(c.env);
    if (back) return c.redirect(`${back}/#/earnings?connected=1`, 302);
    return c.json({ ok: true, pending: true, passport_id: r.passport_id });
  } catch (e) {
    // I3-1915: a failed callback lands the creator back in the console like the
    // success path does, carrying the reason, instead of a raw JSON page.
    const back = consoleOrigin(c.env) ?? appOrigin(c.env);
    const reason = e instanceof ConnectError ? e.message : "error";
    // I3-0764: an unexpected fault is logged even when the caller is redirected.
    if (back && !(e instanceof ConnectError)) logError({ cid: c.get("cid"), hop: "http", route: "GET /connect/oauth/callback", msg: "unexpected route fault", err: e instanceof Error ? e.name : typeof e });
    if (back) return c.redirect(`${back}/#/earnings?connect_error=${encodeURIComponent(reason)}`, 302);
    if (e instanceof ConnectError) return c.json({ error: e.message }, e.status);
    return routeFault(c, "GET /connect/oauth/callback", e);
  }
});

// Passkey CONFIRM (verify leg): a valid assertion activates the binding.
app.post("/account/creator-connect/confirm", anonBodyLimit, async (c) => {
  const creator = await creatorIdForPayouts(c); // I20-RG-MONEY-03 (D-0966): a closed creator is still paid
  if (!creator) return c.json({ error: "unauthorized" }, 401);
  const cfg = passkeyConfig(c.env);
  if (!cfg) return c.json({ error: "passkeys_unavailable" }, 503);
  const raw = await c.req.text();
  if (raw.length > 16 * 1024) return c.json({ error: "too_large" }, 413);
  const body = ((): { assertion?: unknown } | null => {
    try {
      return JSON.parse(raw) as { assertion?: unknown };
    } catch {
      return null;
    }
  })();
  const assertion = body ? parseAssertion(body.assertion) : null;
  if (!assertion) return c.json({ error: "assertion required" }, 400);
  const db = d1(c.env.TRUST_DB);
  const pending = await pendingConnect(db, creator);
  if (!pending) {
    // I3-2108: a retry whose first response was lost finds the binding already
    // ACTIVE. Nothing is written here; the retry is told what is true, which is
    // what the console's `connected` branch renders.
    if ((await boundConnect(db, creator))?.status === "active") return c.json({ ok: true, connected: true, already_active: true });
    return c.json({ error: "no pending payout account to confirm" }, 409);
  }
  try {
    // Standing again, on the leg that actually makes the destination LIVE.
    // Re-asked rather than inherited from `/start`: an OAuth round trip can take
    // as long as the creator likes, and a suppression or retraction that lands
    // in between must be decisive. Same rule as `requireVerifiedCreator`'s own
    // "verified AT THE MOMENT OF THE ACTION, never inherited".
    await requireCurrentlyVerifiedCreator(db, creator);
    await verifyConnectBindAssertion(db, creator, "bind", pending.stripe_account_id, pending.bind_epoch, assertion, cfg, Date.now()); // throws on failure
    // BF-14: and the MUTATION carries the reviewed generation, so a rebind that
    // landed between the read and the write is a conflict, not an activation.

// A verified creator uploads a batch of their platform's discount/unlock codes;
// Dhwaja hands one to each QUALIFYING fan (the claim is an active-follower

app.post("/account/reverse-codes/pool", anonBodyLimit, async (c) => {
  const creator = await creatorId(c);
  if (!creator) return c.json({ error: "unauthorized" }, 401);
  const db = d1(c.env.TRUST_DB);
  try {
    // PRIVILEGE FIRST, before the body is even read. Two reasons: an
    // unprivileged caller learns nothing about the shape, and the check cannot
    // be routed around by a malformed request. `requireVerifiedCreator` is the
    // ONLY producer of the value `createPool` demands (reverse-codes.ts), which
    // is why the two sibling handlers below can no longer omit it the way they
    // omitted the `isVerifiedPassport` line that used to live only here.
    const verified = await requireVerifiedCreator(db, creator);

    // creator-scoped inside createPool). The body is capped before it is parsed
    // — a rule is bounded to a few hundred bytes by its own limits.
    const raw = await c.req.text();

    const body = ((): { title?: unknown; eligibility_kind?: unknown; eligibility_ref?: unknown; eligibility_rule?: unknown; min_supporter_tier?: unknown } | null => {
      try {
        return JSON.parse(raw) as { title?: unknown; eligibility_kind?: unknown; eligibility_ref?: unknown; eligibility_rule?: unknown; min_supporter_tier?: unknown };
      } catch {
        return null;
      }
    })();
    if (!body || typeof body.title !== "string" || typeof body.eligibility_kind !== "string") return c.json({ error: "title and eligibility_kind required" }, 400);
    const r = await createPool(
      db,
      verified,
      {
        title: body.title,
        eligibility_kind: body.eligibility_kind,
        ...(typeof body.eligibility_ref === "string" ? { eligibility_ref: body.eligibility_ref } : {}),
        ...(body.eligibility_rule !== undefined ? { eligibility_rule: body.eligibility_rule } : {}),
        ...(typeof body.min_supporter_tier === "string" ? { min_supporter_tier: body.min_supporter_tier } : {}),
      },
      Date.now(),
    );
    return c.json({ ok: true, ...r });
  } catch (e) {
    if (e instanceof VerifiedCreatorError || e instanceof ReverseCodeError) return c.json(errorBody(e), e.status);
    return routeFault(c, "POST /account/reverse-codes/pool", e);
