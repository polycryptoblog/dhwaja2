/**
 * Pure view renderers: data → HTML-string fragment. Every dynamic value flows
 * through esc(); the only navigation is data-nav hash links the controller
 * wires up — a rendered value can never carry an untrusted URL into an href.
 * Fetching, ceremonies, and DOM wiring live in main.ts; these never touch the
 * DOM or the network (same contract as the fan SPA's views).
 */

export function esc(s: unknown): string {
  return String(s ?? "").replace(/[&<>"']/g, (c) => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" })[c]!);
}

function clamp(s: unknown, n: number): string {
  return typeof s === "string" ? s.slice(0, n) : "";
}

/** PS-285: a literal parenthesized-s suffix is not a plural, it's a placeholder
 * nobody resolved — it reads exactly the same for 1 as for 100. One helper so
 * every count in the console picks "report" vs "reports" the way English
 * actually does. */
function plural(n: number, word: string): string {
  return `${word}${n === 1 ? "" : "s"}`;
}

/** I3-0201: verb/pronoun agreement ("is"/"are", "it"/"them", "an account"/"accounts")
 * was hand-written per call site as `n === 1 ? a : b` ternaries — correct for
 * English but with no shared seam an ICU swap could replace, unlike `plural()`
 * above. One helper via `Intl.PluralRules`, so every agreement decision in
 * this file resolves through the same point. Still English-only by itself:
 * real locale pluralization additionally needs the "few"/"many"/zero
 * categories `Intl.PluralRules` exposes and this file's callers don't branch
 * on yet — wiring those in is a catalogue integration (I3-0133), out of
 * this file's reach; this closes the in-file half (I3-0201). */
function pick(n: number, singular: string, pluralForm: string): string {
  return new Intl.PluralRules("en-US").select(n) === "one" ? singular : pluralForm;
}

/** #132PS-273 / I3-0328: the link for the page on screen carries `aria-current="page"` (and `.active`), so a
 * screen reader and a sighted reader both get "you are here". `current` is the route hash main.ts painted. */
function nav(hash: string, label: string, current?: string | null): string {
  const here = current === hash ? ` class="active" aria-current="page" style="font-weight:700;text-decoration:none"` : "";
  return `<a href="${esc(hash)}" data-nav="${esc(hash)}"${here}>${esc(label)}</a>`;
}

/**

 *
 * `regionsSection` already carries this law in its own words (S-070 CA-02) and

}

/**

function asOfLine(generatedAt?: string | null): string {
  if (typeof generatedAt !== "string" || generatedAt.length === 0) return "";

/**
 * Why a frozen-report read failed, IN THE SERVER'S OWN TERMS (CA-01).
 *
 * `intelligence-routes.ts` answers 503 for two different situations and is
 * careful to distinguish them: a report whose day has not been built yet comes
 * back as `{ error, retry: "daily" }`, while an unconfigured or unreachable
 * snapshot store comes back as `analytics_unavailable`. The console collapsed
 * both — and every other 503 — into one sentence, "Plain-language explanations
 * are not switched on for this deployment."
 *
 * That sentence is a confident, specific, WRONG diagnosis in the ordinary case:
 * a creator who opens Intelligence before today's build has finished is told a
 * feature is switched off, when it is switched on and simply not built yet. The
 * recovery is opposite too — nothing they do switches a feature on, and waiting
 * until tomorrow is exactly what would have worked.
 *
 * The server already sent the distinction. This reads it instead of guessing
 * from the status code, which is the whole of the fix: `retry: "daily"` is a
 * machine contract, and a client that ignores it invents its own recovery
 * advice out of a number that cannot carry one.
 */
export function frozenReadFailure(status: number, body?: { error?: string; retry?: string } | null): string {
  if (status !== 503) return "Could not read your numbers just now.";
  // Order matters: the not-ready case is the common one and the only one with a
  // true recovery, so it is tested before the deployment-shaped explanations.
  if (body?.retry === "daily") {
    return "Today's numbers haven't been built yet. These reports are rebuilt once a day — check back tomorrow. Nothing is wrong with your account.";
  }
  if (body?.error === "analytics_unavailable") {
    return "Analytics aren't available on this deployment yet.";
  }
  return "Plain-language explanations are not switched on for this deployment.";
}

/**
 * THE LEGAL FOOTER — Terms and Privacy on the fan app, opened in a new tab.
 *
 * The pages live on the PUBLIC origin (the one a fan uses), so the links are
 * built from the same validated `PUBLIC_BASE` the share link is, with the same
 * origin-only check `shareLink` applies: anything that is not a bare https
 * origin renders NO link and a one-line note instead of a dead or hostile href.
 * A build without a public origin fails closed to the note.
 *
 * CONSOLES H3b (I3-1646 b, I23-L05): when the API origin is configured (`helpBase`, the same origin as the Help
 * link), Terms and Privacy open trust-core's server-rendered `/terms` and `/privacy` (path form, CONSOLES H3); the
 * fan app's `#/terms` / `#/privacy` remains the fallback when only the public origin is set. Both render the one
 * `@dhwaja/legal` text.
 */
export function legalFooter(publicBase: string | null | undefined, helpBase?: string | null): string {
  const base = typeof publicBase === "string" && /^https:\/\/[a-z0-9.-]+(:\d+)?$/i.test(publicBase) ? publicBase : null;
  // I3-1594: the Help link is the trust-core `/how-it-works` page on the API origin (the web app's `ssrBase`), validated
  // with the same bare-https-origin check; an absent or unusable base renders NO help anchor rather than a dead one.
  const hb = typeof helpBase === "string" && /^https:\/\/[a-z0-9.-]+(:\d+)?$/i.test(helpBase) ? helpBase : null;
  // I21-Q146 (D-1072(3)): About is trust-core's server-rendered GET /about (about-page.ts) on the same API origin as Help.
  const help = hb ? `<a href="${esc(`${hb}/how-it-works`)}" target="_blank" rel="noopener noreferrer">Help</a> · <a href="${esc(`${hb}/about`)}" target="_blank" rel="noopener noreferrer">About</a>` : "";
  if (!base) return `<footer class="foot"><nav><span class="hint">Legal pages not configured.</span>${help ? ` · ${help}` : ""}</nav></footer>`;
  const link = (path: "/terms" | "/privacy", label: string): string => `<a href="${esc(hb ? `${hb}${path}` : `${base}/#${path}`)}" target="_blank" rel="noopener noreferrer">${esc(label)}</a>`;
  return `<footer class="foot"><nav>${link("/terms", "Terms")} · ${link("/privacy", "Privacy")}${help ? ` · ${help}` : ""}</nav></footer>`;
}

/**

 *
 * The console had ZERO `aria-live` regions while the fan SPA has them
 * throughout, and almost every consequential thing this app says — a proof
 * verified, a payout refused, a session expired — is said by dropping one of

export function banner(kind: "ok" | "err" | "warn", message: string): string {

  const role = kind === "err" ? "alert" : "status";
  return `<div class="ds-banner banner ${cls}" role="${role}">${esc(clamp(message, 300))}</div>`;
}

/**
 * THE 14-DAY CONTROL-PROOF WARNING, ON EVERY PAGE.
 *
 * The server half of this is `noticeProofsExpiring` (an `alert` digest on the
 * email/push/in-app channel); this is the belt-and-braces half, derived on every
 * paint from the creator's own accounts read, so a creator who never set up a
 * contact point or turned digests off still cannot miss it once they sign in.
 * Exactly the shape `verificationBanner` uses for the other deadline, because
 * the two feel identical to the person receiving them.
 *
 * It names what LAPSING COSTS, not just that a date is coming: an expired proof
 * stops the handle being served to fans as confirmed AND stops it shielding that
 * handle from impersonation reports. A creator who does not know the second half
 * has no way to price two minutes with a bio code.
 */
export function proofExpiryBanner(rows: AccountStandingDto[] | null): string {
  if (!Array.isArray(rows)) return "";
  const soon = rows.filter((r) => r.status === "active" && r.expiring_soon === true && typeof r.expires_at === "string");
  if (soon.length === 0) return "";
  // The nearest deadline leads: it is the one with a clock on it.
  const first = soon.reduce((a, b) => (Date.parse(a.expires_at ?? "") <= Date.parse(b.expires_at ?? "") ? a : b));
  const d = typeof first.days_remaining === "number" ? ` (${whenDays(first.days_remaining)})` : "";
  const which = soon.length === 1
    ? `The control proof on <code>@${esc(clamp(first.id, 80))}</code> expires on ${day(first.expires_at)}${esc(d)}.`
    : `${soon.length} of your accounts have a control proof expiring, the first on ${day(first.expires_at)}${esc(d)}.`;
  return `<div class="ds-banner banner banner-warn" role="alert" style="margin:0 0 12px">
  <strong>${which}</strong> When a proof lapses that handle stops being shown to fans as confirmed and stops being shielded from impersonation reports — you keep the account, the listing and everything on your passport. Renewing takes one short code in that profile's bio. ${nav("#/accounts", "Renew it →")}
</div>`;
}

 * button that died with the tab. This panel is that memory, restored from the
 * per-passport store, and it is on the SIGN-IN screen because that is the page a
 * returning, session-less creator actually lands on.
 */
export function pendingClaimsPanel(pending: import("./session.js").PendingManualClaim[]): string {
  if (!Array.isArray(pending) || pending.length === 0) return "";
  const rows = pending
    .slice(0, 5)
    .map(
      (p) => `<li><code>${esc(clamp(p.passport_id, 24))}</code>${p.platform ? ` · ${esc(clamp(p.platform, 40))}` : ""}${p.handle ? ` <code>@${esc(clamp(p.handle, 80))}</code>` : ""}${p.started_at ? ` <span class="roi-sub">started ${esc(clamp(p.started_at, 10))}</span>` : ""} <button class="ds-btn" data-act="manual-resume" data-psp="${esc(clamp(p.passport_id, 24))}">Check status</button></li>`,
    )
    .join("");
  return `<div class="ds-banner banner banner-ok" role="status"><strong>You have a claim under review.</strong> We kept the one-time reference from when you started it, on this browser. Once an operator has confirmed your posted code, checking here signs you in.</div>
  <ul class="pending-claims">${rows}</ul>
  <div id="manual-resume-result" aria-live="polite"></div>`;
}

const OAUTH_REASON_TEXT: Record<string, string> = {
  access_denied: "Sign-in was cancelled — you did not finish granting access.",
  identity_mismatch: "That account does not match the passport you are signing in to.",
  exchange_failed: "Dhwaja could not confirm that sign-in with the provider.",
  expired: "That sign-in link had expired. ",
  bad_state: "That sign-in link was no longer valid.",
  support_withdrawn: "That account no longer supports this platform's sign-in.",
};
export function oauthReasonText(reason?: string): string {
  const known = reason ? OAUTH_REASON_TEXT[reason] : undefined;
  return `OAuth sign-in didn't complete: ${known ?? "something went wrong on the provider's side."}`;
}

/**
 * I3-0529 d: what a recovering creator is told about the recovery they filed (`POST /recovery/status`, one word).
 * Mirrors the fan app's `recoveryStatusOutcome`. A 404 is NOT evidence the reference is wrong (unknown, malformed and
 * pruned references are answered identically on purpose), so the wording never says "no such reference".
 */
export function recoveryStatusView(r: { ok: boolean; status: number; data: { status?: string; error?: string } | null }): string {
  if (r.status === 429) return banner("warn", serverMessage(r) ?? "Too many checks just now. Give it an hour — nothing about your case changed."); // #963
  if (!r.ok || r.status !== 200 || typeof r.data?.status !== "string") {
    return banner("warn", "We can't show you anything for that reference. Check for a typo — a reference is trk_ followed by thirty-two characters. If it is right and this keeps happening, reply to the email we sent you.");
  }
  switch (r.data.status) {
    case "resolved":
      return banner("ok", "An operator has resolved it. If you still can't get in, file again and say so — that is new information, not a repeat.");
    case "closed":
      return banner("warn", "An operator closed it without action. If circumstances changed, file again and describe what's different.");
    case "mixed":
      return banner("warn", "Different outcomes: this reference covered more than one passport and they didn't all end the same way. Reply to the email we sent you and a person will tell you which is which.");
    default:
      return banner("warn", "Still with an operator. Recovery is a human review with no queue position to give you; checking more often will not move it.");
  }
}

/**

 *
 * Two of the server's 409s are dead ends as written, and both are reachable by
 * doing exactly what the console asked:
 *

 *    (account, caller), one-hour TTL. A creator who closed the page has a code
 *    live on their own profile that they can no longer read, and the refusal

 *    the console now hands the code straight back; this is what is left.)

 *    whose manual claim was approved while the tab was shut, it reads as somebody
 *    else having taken their passport, and it is the last thing the product says
 *    to them before they give up.
 *
 * The server's sentence is kept verbatim and a second one is added. Anything
 * unrecognized passes through untouched — this must never paper over a refusal
 * it does not understand.

/** I21-Q20 (I30-APIS): the age the box names until the server names its own (`min_age` on an age_ack_required refusal):

export function claimStartRefusal(error: string | undefined): string {
  const why = error ?? "could not start the claim";
  if (/already in progress/i.test(why)) {
    return `${why} A code stays live for up to an hour and there can only be one — the code you were given is still the one to place. If you no longer have it, wait for it to run out and start again; nothing is lost by waiting.`;
  }
  // Kept under `banner`'s 300-character clamp on purpose: a sentence that names
  // the remedy and is then cut off mid-word is worse than a shorter one.
  if (/already claimed/i.test(why)) {
    return `${why} If that was YOU — a manual claim of yours was approved — use "Lost the reference from a claim you started?" below with the code you posted, or ask for account recovery. If it was not you, that is exactly what recovery exists for.`;
  }
  // I3-1918 (I4-02-R8 hand-off): these five reasons fell through to the bare
  // server string with no recourse text, same as the two arms above.
  if (/too many .* attempts/i.test(why)) {
    return `${why} There's a daily cap on attempts for this account — wait for it to reset and try again; nothing about your account changed.`;
  }
  if (/account not active/i.test(why)) {
    return `${why} That account isn't active on its own passport right now, so a claim can't start from it.`;
  }
  if (/manual verification/i.test(why)) {
    return `${why} This platform doesn't expose a page Dhwaja can read a code from, so it needs the manual route below (post a code, then use "Instagram / TikTok / OnlyFans") instead of a bio-code claim.`;
  }
  if (/prove control via DNS/i.test(why)) {
    return `${why} Use "Or prove a brand domain" below instead of a bio-code claim.`;
  }
  return why;
}

/**
 * The Back Catalog Wizard: what fans asked you to dig out (archive demand),
 * upcoming anniversaries (throwback candidates), and your existing releases.
 * Dhwaja hosts no media — each item becomes a SIGNED collectible/edition (mint it
 * via the drop, stamping its true original date), which links out to the real
 * thing on your own platform. Every value escaped; navigation is data-nav only.
 */
/**
 * S88 — the ARCHIVE-DEMAND section, rendered from its own read
 * (`GET /account/archive-demand`, NOT verified-gated) so a claimed-but-not-yet-
 * verified creator still sees what fans asked them to dig out. `undefined` means
 * the caller didn't fetch it (back-compat); `null` means the read failed, which
 * is a display fact, never a claim that nobody asked.
 */
/**
 * THE SERVER'S DEMAND SHAPE, as this file needs to read it.
 *
 * `ArchiveDemandDto` in api.ts now matches the wire exactly (`request_count:
 * number | null`, `withheld`, `floor`, `through`, and each note's `day` — the
 * `created_at` alias is gone). This type stays DELIBERATELY LOOSER than that
 * DTO: it is what a RENDERER may assume about a body that arrived over the
 * network, and the optional fields are the ones whose absence must degrade to
 * silence rather than to a claim about fans. `request_count` is the one field
 * that is not optional, because "withheld" and "not fetched" must never collapse
 * into each other here.
 */
type DemandLike = {
  request_count: number | null;
  withheld?: boolean;
  floor?: number;
  through?: string;
  notes?: Array<{ note: string; day?: string }>;
  /** RUN-0006: the count is served and the NOTES are withheld (the live
   * population is below the floor while the counted one is not). Optional for
   * the same reason as the fields above - a body that arrived over the network
   * may not carry it - and its absence degrades to the existing behaviour. */
  notes_withheld?: boolean;
};

/**
 * WHAT A WITHHELD DEMAND COUNT MAY SAY.
 *
 * `request_count` is null whenever the population is below the reporting floor —
 * and that includes ZERO, deliberately: if "0" and "withheld" rendered
 * differently, the difference between the two screens would itself prove that a
 * fan had asked, which is the inference the floor exists to prevent. So every
 * population from nobody up to floor-1 must produce the SAME sentence.
 *
 * IT MUST ALSO NOT BE A QUANTITY (Jason, 2026-09-05). The sentence used to read
 * "Fewer than 5 fans have asked…", which is byte-identical across the withheld
 * range and still wrong: an upper bound IS a claim about how many fans asked,
 * and a creator who watches it become a number learns the crossing. The rule is
 * now three-fold and all three parts are load-bearing:
 *
 *   - no number, and `demand.floor` is NEVER rendered (the server sends it for
 *     inspection, not for copy);
 *   - no claim that anybody asked, and no claim that nobody did — "No archive
 *     requests yet" was already forbidden as a false statement to a creator who
 *     has four, and "some fans have asked" is forbidden for the mirror reason;
 *   - identical bytes for every population from nobody up to floor-1.
 *
 * What is left to say truthfully is the CONDITION and the CLOCK, neither of
 * which carries fan data: demand appears once the floor is met, and today's
 * requests are never in today's answer. `through` is derived from the clock
 * alone, so it is the same string for every population and may be shown.
 */
function demandCountOf(d: DemandLike): number | null {
  const n = d.request_count;
  return typeof n === "number" && Number.isFinite(n) ? n : null;
}

/** "…through 2026-09-04." — today is never counted, and saying so is the honest
 * alternative to a number that silently lags. */
function demandThrough(d: DemandLike): string {
  return typeof d.through === "string" && /^\d{4}-\d{2}-\d{2}$/.test(d.through) ? ` Counted through ${esc(d.through)}; today's requests aren't in yet.` : "";
}

function demandWithheldLine(d: DemandLike): string {
  return `<p class="roi-note">Back-catalog demand appears here once enough fans have asked, so that no single fan can be picked out of the number.${demandThrough(d)}</p>`;
}

/**
 * NOTES WITHHELD UNDER A SERVED COUNT (RUN-0006, orchestrator ruling).
 * A dedicated sentence for the notes-only gate — never `demandWithheldLine`,
 * which is about the count and carries `through`. No digit, no "none"/"no
 * notes"/"zero", and no presence claim either way.
 */
function notesWithheldLine(): string {
  return `<p class="muted">Fan notes appear here once enough fans have asked, so no single note can be picked out.</p>`;
}

/**
 * THE SAME SENTENCE, FOR THE TILES THAT CARRY THE COUNT AND NOT THE SECTION.
 *
 * The ROI dashboard and the audience-intelligence screen each print the archive
 * count as one figure in a list, and `stat()` renders a withheld one as "—".
 * That dash is correct as a CELL VALUE and insufficient on its own: under a
 * heading a creator is reading for a number, an unexplained dash reads as ZERO,
 * which is a claim about how many fans asked and false for every population from
 * one upward. (The agency ROI screen was held to exactly this standard in
 * `run0002-builder-l1-agency-render`: the dash stays, the explanation goes
 * beside it.)
 *
 * So the dash stays and this adds the sentence — and it adds THE SENTENCE,
 * routed through `demandWithheldLine` rather than retyped, so the four surfaces
 * that speak about a withheld archive population (this, the archive-demand
 * section, the back-catalog body, and the agency screen) cannot drift into four
 * slightly different claims that could be differenced against each other.
 *
 * The ROI and intelligence wires carry no `through`, so no clock line is
 * rendered here; every population from nobody up to floor-1 produces the same
 * bytes either way, which is the property that matters.
 */
function demandTileWithheldNote(n: number | null | undefined): string {
  return typeof n === "number" && Number.isFinite(n) ? "" : demandWithheldLine({ request_count: null });
}

/**
 * THE SAME STANDARD, FOR THE WITHHELD DEAD-ATTEMPT BLOCK.
 *
 * `intelligence.ts` floors the dead-attempt counts on their own account
 * population with ZERO INCLUDED, so `stat()` prints "—" for every population
 * from nobody up to floor-1. The dash is right as a cell value and insufficient
 * under a heading a creator reads for a number, exactly as the archive-demand
 * tile above: an unexplained dash reads as ZERO, which is a claim about how many
 * accounts attempted dead codes and false from one upward.
 *
 * It is NOT `demandWithheldLine` — that sentence is about back-catalog demand
 * and would be a false description here. It carries no digit and no
 * presence/absence claim, so every population below the floor renders the same
 * bytes.
 */
function integrityWithheldNote(withheld: boolean): string {
  return withheld
    ? `<p class="roi-note">Dead-redemption figures appear here once enough accounts are in the population, so that no single account can be picked out of the number.</p>`
    : "";
}

export function archiveDemandSection(demand?: DemandLike | null): string {
  if (demand === undefined) return "";
  if (demand === null) {
    return `<h3>Fans have asked you to catalog</h3><p class="empty ${UNREAD_CLASS}">We couldn't load your archive demand just now. ${unreadReassurance()} It says nothing about whether fans have asked. Try again in a moment.</p>`;
  }
  const n = demandCountOf(demand);
  // Below the floor the server sends no notes at all (one note under a withheld
  // count would prove the population is at least one), so there is nothing to
  // render and nothing to decide here.
  if (n === null) return `<h3>Fans have asked you to catalog</h3>${demandWithheldLine(demand)}`;
  const notes = Array.isArray(demand.notes) ? demand.notes : [];
  // NOTES WITHHELD UNDER A SERVED COUNT (RUN-0006). The server sends no notes
  // when the LIVE population is below the floor even though the counted one is
  // not, and the empty list must not render as silence here: silence beside a
  // count reads as "nobody wrote anything", which is false for a creator whose
  // fans did. It is THE sentence, routed through `demandWithheldLine` like the
  // three other surfaces, with no `through` because the count line below
  // already carries the clock.
  const list = demand.notes_withheld === true
    ? notesWithheldLine()
    : notes.length > 0
      ? `<ul class="cat-demand">${notes.slice(0, 20).map((d) => `<li>“${esc(clamp(d.note, 200))}” <span class="roi-sub">${esc(clamp(d.day ?? "", 10))}</span></li>`).join("")}</ul>`
      : "";
  const count = `<p><strong>${n}</strong> fan${n === 1 ? " has" : "s have"} asked you to catalog your back catalog.${demandThrough(demand)}</p>`;
  return `<h3>Fans have asked you to catalog</h3>${count}${list}`;
}

  // that is true for every population below the floor, and never "No specific

  const demand = demandCount === null

      // count take the shared sentence, never "No written notes with those

      ? `<p><strong>${demandCount}</strong> fan${demandCount === 1 ? " has" : "s have"} asked for your back catalog.${demandThrough(bcDemand)}</p>${notesWithheldLine()}`

      ? `<p><strong>${demandCount}</strong> fan${demandCount === 1 ? " has" : "s have"} asked for your back catalog.${demandThrough(bcDemand)}</p><ul class="cat-demand">${demandNotes.slice(0, 20).map((d) => `<li>“${esc(clamp(d.note, 200))}”</li>`).join("")}</ul>`

/**
 * Ready-to-run campaign recipes. A "campaign" is never a new mechanism — it's the
 * signed drop (which can attach reward codes) run toward a goal, with inputs
 * pre-computed from your own state. Nothing is minted or distributed here; each
 * available card links into the drop flow, where your passkey signs the action.
 * Unavailable cards say what's needed. Every value escaped; nav is data-nav only.
 */
function campaignCard(t: CampaignTemplateDto): string {
  const runnable = t.available && t.action !== null;
  const pre = t.prefill ?? {};
  const hints: string[] = [];
  if (typeof pre.suggested_codes === "number" && Number.isFinite(pre.suggested_codes)) hints.push(`suggested batch: ${Math.trunc(pre.suggested_codes)} codes`);
  if (typeof pre.released_at === "string" && pre.released_at) hints.push(`original date: ${esc(clamp(pre.released_at, 10))}`);
  const hintLine = hints.length > 0 ? `<p class="roi-sub">${hints.join(" · ")}</p>` : "";
  const cta = runnable
    // M40(a): a "pool" template opens the reverse-codes authoring page (#/codes),
    // NOT #/drop — the else branch used to send every non-drop template to the
    // drop screen, so a code-pool campaign dead-ended on the wrong flow.
    ? nav(t.action === "drop" ? dropHref({ ...(pre.title ? { title: pre.title } : {}), ...(pre.released_at ? { released_at: pre.released_at } : {}) }) : t.action === "pool" ? "#/codes" : "#/drop", "Set it up →")
    : t.unlocks
      ? `<p class="roi-sub">Unlocks: ${esc(clamp(t.unlocks, 80))}</p>`
      : `<p class="roi-sub">Not available right now.</p>`;
  return `<li class="campaign${runnable ? "" : " campaign-off"}">
    <h3>${esc(clamp(t.title, 80))}</h3>
    <p>${esc(clamp(t.blurb, 200))}</p>
    <p class="roi-note">${esc(clamp(t.readiness, 200))}</p>
    ${hintLine}${cta}</li>`;
}

/**
 * The Recognition Queue + granted list. Recognition is the scarce, signed ladder:
 * follow a fan from your own audience first, then like / thank / shout out /
 * spotlight them (each a passkey-signed act, tier-capped). Every dynamic value —
 * account ids, kinds, messages — is escaped; the only navigation is data-act
 * buttons the controller wires (account ids ride data-attrs, never an href).
 */
const RECOGNIZE_NAMED = ["thanks", "shout_out", "spotlight"];
function recBtn(acct: string, kind: string): string {
  return `<button class="ds-btn" data-act="recognize" data-acct="${acct}" data-kind="${esc(kind)}">${esc(kind.replace("_", " "))}</button>`;
}
function recognizeFanRow(f: QueueFanDto): string {
  const acct = esc(clamp(f.account_id, 40));
  const given = new Set(Array.isArray(f.given) ? f.given : []);
  const btns: string[] = [];
  // `like` needs no follow (audience-gated). follow, then the named recognitions,
  // need a follow first.
  if (!f.following) btns.push(recBtn(acct, "follow"));
  if (!given.has("like")) btns.push(recBtn(acct, "like"));
  if (f.following) for (const k of RECOGNIZE_NAMED) if (!given.has(k)) btns.push(recBtn(acct, k));
  const givenLabels = [...given].map((k) => esc(k.replace("_", " "))).join(", ");
  const status = givenLabels || esc(clamp(f.source, 8));
  return `<li><code>${esc(clamp(f.account_id, 40))}</code> <span class="roi-sub">${status}</span> ${btns.join(" ")}</li>`;
}
function recapLine(recap: RecognitionRecapDto | null): string {
  if (!recap) return "";
  const k = recap.granted_by_kind ?? {};
  const parts = ["like", "thanks", "shout_out", "spotlight"].filter((x) => (k[x] ?? 0) > 0).map((x) => `${esc(String(k[x]))} ${esc(x.replace("_", " "))}`);
  const detail = parts.length > 0 ? ` (${parts.join(", ")})` : "";
  return `<p class="roi-note">This week: <strong>${esc(String(recap.granted_total))}</strong> recognitions given${detail} · <strong>${esc(String(recap.new_follows))}</strong> new follows · <strong>${esc(String(recap.active_spotlights))}</strong> fans currently spotlighted.</p>`;
}

/** The cursors of the page on screen, read from the hash by main.ts: `answers` is the published-answers
 * cursor (`qa`/`qat`), `questions` the open-questions one (`qq`/`qqat`). */
export interface QuestionsCursors {
  answers?: { question_id: string; answered_at: string } | null;
  questions?: { question_id: string; created_at: string } | null;
}
/** I3-0242: the `#/questions?…` hash for a page. Both lists page independently, so every link
 * carries the OTHER list's current cursor - paging one never resets the other. */
export function questionsHash(answers: QuestionsCursors["answers"], questions: QuestionsCursors["questions"]): string {
  const q: string[] = [];
  if (answers) q.push(`qa=${encodeURIComponent(answers.question_id)}`, `qat=${encodeURIComponent(answers.answered_at)}`);
  if (questions) q.push(`qq=${encodeURIComponent(questions.question_id)}`, `qqat=${encodeURIComponent(questions.created_at)}`);
  return q.length > 0 ? `#/questions?${q.join("&amp;")}` : "#/questions";
}

/** A date a person reads, from an ISO instant. Never throws on junk. */
function day(iso: string | null | undefined): string {
  if (typeof iso !== "string" || iso.length < 10) return "—";
  const ms = Date.parse(iso);
  if (!Number.isFinite(ms)) return esc(clamp(iso, 24));
  return new Date(ms).toISOString().slice(0, 10);
}

/** "in 12 days" / "today" / "9 days ago". */
function whenDays(n: number): string {
  if (n === 0) return "today";
  return n > 0 ? `in ${n} day${n === 1 ? "" : "s"}` : `${-n} day${n === -1 ? "" : "s"} ago`;
}

    const d = typeof v.days_remaining === "number" ? ` (${whenDays(v.days_remaining)})` : "";
    return `<div class="ds-banner banner banner-warn" role="alert" style="margin:0 0 12px">
  <strong>Your verification expires on ${day(v.expires_at)}${esc(d)}.</strong> ${esc(LAPSE_COSTS)} ${
    v.renewal_pending
      ? "Your renewal request is already with an operator — nothing about your badge changes while they decide."
      : `${nav("#/verification", "Start the renewal →")}`
  }
</div>`;
  }
  return "";
}

/**
 * THE VERIFICATION PAGE. Before this existed, expiry was a one-way door: the
 * sweep could take verification away and nothing in the product could ask for
 * it back, because nothing enqueued a `verification` review item and
 * `verification_renewed` had no writer at all.
 *
 * The button files an ordinary review item. It is offered to a verified creator
 * as well as a lapsed one, and the copy is explicit that a request changes
 * nothing on its own — that is what makes renewing BEFORE the deadline safe,
 * and a renewal granted while still verified never opens a downgrade window
 * (the ledger takes `verification_renewed` and the badge does not blink).
 */
export function verificationView(v: VerificationDto | null, read?: { status: number }): string {
  // `read` carries no 403 arm HERE — this is the page a creator lands on
  // precisely BECAUSE their standing is in question — but it does carry the 401
  // one, which is about the console rather than about their verification.
  if (!v) return consoleUnread("verification", "Verification", "your verification status", read);
  const head = `<section id="verification"><h2>Verification</h2>`;
  const request = v.renewal_pending
    ? `<p class="ds-banner banner banner-ok">Your request is with an operator. Renewal is a human review and takes a few days; nothing about your badge changes while they decide, and you will not lose anything by having asked.</p>`
    : `<p><button class="ds-btn" data-act="request-verification">${v.claim_state === "verified" ? "Request renewal" : "Request verification"}</button></p>
  <p class="hint">This files a request for an operator to review. It changes nothing by itself — your badge, your pages and your standing stay exactly as they are until a decision is made.</p>`;
  if (v.claim_state === "unclaimed") {
    return `${head}<p>This passport has not been claimed yet. Prove control of one of its listed accounts, or a brand domain if you're not listed at all — verification is granted on top of a proven claim.</p>
  <p>${nav("#/onboarding", "Get set up →")}</p></section>`;
  }
  if (v.claim_state !== "verified") {
    const lapsed = v.expired_at
      // I3-1595 - a lapse is routine and recoverable (Start the renewal, below),
      // not a failure; `banner-err` is this app's reserved destructive-red and is
      // spent elsewhere on things like a refused request or an expired session.
      ? `<p class="ds-banner banner banner-warn"><strong>Your verification lapsed on ${day(v.expired_at)}.</strong> ${esc(LAPSE_COSTS)}</p>`
      : `<p>You are <strong>Claimed</strong>. Verification is the operator-reviewed step above it.</p>`;
    return `${head}${lapsed}${request}<div id="verification-result" aria-live="polite"></div></section>`;
  }
  const expiry =
    v.expires_at === null
      ? `<p class="ds-banner banner banner-warn"><strong>Your grant states no readable deadline.</strong> That is a legacy record: the expiry sweep deliberately leaves these alone rather than downgrading you, so nothing is at risk today. Ask for a renewal anyway — it replaces the unreadable date with a real one.</p>`
      // I3-1595 - the verified verdict was bare text, the one state on this
      // page with no color or icon at all, next to two lapse states that both
      // get a banner. `banner-ok` gives it the same visual verdict treatment.
      : `<p class="ds-banner banner banner-ok"><strong>Verified.</strong> Your current grant runs to <strong>${day(v.expires_at)}</strong>${typeof v.days_remaining === "number" ? ` — ${esc(whenDays(v.days_remaining))}` : ""}.</p>
  <p>${esc(LAPSE_COSTS)}</p>`;
  return `${head}${expiry}${request}<div id="verification-result" aria-live="polite"></div>
  <p class="hint">Renewing early is safe and is the point: a renewal granted while you are still verified moves the deadline without your badge ever going away.</p>
  </section>`;
}

/**
 * THE COPY ON A VERIFIED-GATED PAGE, DRIVEN BY THE REASON.

 * Seven screens rendered "Unlocks at verified status." / "this unlocks once
 * you're verified" — present and future tense, describing a creator who has not
 * arrived yet. After the sweep downgrades somebody overnight those seven

export function verifiedGateNote(v?: VerificationDto | null): string {

  }
  return `Unlocks at verified status. <a href="#/verification" data-nav="#/verification">Verification →</a>`;
}

/**
 * WHAT A FAN CHECKING THIS HANDLE ACTUALLY SEES, IN ONE SENTENCE.
 *
 * The screen used to tell a creator that an unproven account is "not shown to
 * fans at all". That is true at today's default settings and FALSE the moment an
 * operator turns the per-account standing model on, at which point exactly the
 * same row is served as an amber **caution** — "listed, not confirmed" — to
 * every fan who checks that handle. A creator deciding what to do about their
 * own listing was reading a statement about a setting nobody had told them
 * existed, and the sentence flipped meaning without a deploy.
 *
 * The server now returns the class the registry build would publish
 * (`AccountStanding.serving`), so the sentence is derived from it rather than
 * guessed. `undefined` means an older server that does not send it, and that
 * arm deliberately says NOTHING about what fans see — silence is the one answer
 * that cannot be wrong.
 */
function servingNote(serving: AccountStandingDto["serving"]): string {
  switch (serving) {
    case "verified":
      return ` Fans checking this handle see it at your <strong>Verified</strong> standing.`;
    case "claimed":
      return ` Fans checking this handle see it as <strong>Claimed</strong>.`;
    case "unclaimed":
      return ` Fans checking this handle see it as <strong>Unclaimed</strong> — listed, with nobody having demonstrated control.`;
    case "caution":
      return ` <strong>Fans checking this handle right now see an amber "listed, not confirmed" caution</strong> — not nothing, and not your standing.`;
    case "not_served":
      return ` Fans checking this handle get "not listed" — Dhwaja publishes nothing for it.`;
    default:
      return ``;
  }
}

/**
 * WHAT FANS SEE, IN SIX WORDS — the welcome screen's version.
 *
 * `servingNote` + `standingLine` write a full paragraph per row, which is right
 * on `#/accounts`: that screen is where a creator goes to WORK on one account,
 * and the remedy has to be spelled out next to it. It is wrong here. The welcome
 * table is six rows long on a first visit, and repeating forty words of
 * explanation in every one buries the two things this screen is actually for —
 * the handle, and the button beside it. Measured by running the built console:
 * the roster ran past two screenfuls before the third handle appeared.
 *
 * So a row here is a chip plus one short clause, and the reasoning that used to
 * be repeated in every row is stated ONCE at the top of the page, where it
 * belongs — and again, in full, in the outcome banner after a filing.
 */
function servingShort(serving: AccountStandingDto["serving"]): string {
  switch (serving) {
    case "verified":
      return `Fans see your <strong>Verified</strong> standing.`;
    case "claimed":
      return `Fans see it as <strong>Claimed</strong>.`;
    case "unclaimed":
      return `Fans see it as <strong>Unclaimed</strong> — nobody has proved it.`;
    case "caution":
      return `<strong>Fans see an amber "listed, not confirmed" caution.</strong>`;
    case "not_served":
      return `Fans checking it get "not listed".`;
    default:
      return ``;
  }
}

/** How a standing reads to the person who owns the account. Each one names what
 * fans see and what to do about it — the standing vocabulary itself
 * (`proven_stale` etc.) is never shown. */
function standingLine(a: AccountStandingDto): { chip: string; note: string; action: "prove" | "reprove" | null } {
  const fans = servingNote(a.serving);
  switch (a.standing) {
    case "proven_current": {
      // INSIDE THE 14-DAY WINDOW IT IS A DIFFERENT SENTENCE. "The proof stays
      // good until <date>" is technically true right up to the deadline and
      // reads as reassurance on the day before it lapses.
      if (a.expiring_soon && a.expires_at) {
        const d = typeof a.days_remaining === "number" ? ` — ${whenDays(a.days_remaining)}` : "";
        return {
          chip: `<span class="mark unconfirmed">! proof expiring</span>`,
          note: `Confirmed, but this proof runs out on ${day(a.expires_at)}${esc(d)}. When it lapses the handle stops being shown to fans as confirmed and stops being shielded from impersonation reports — nothing else changes, and renewing now costs you nothing.`,
          action: "reprove",
        };
      }
      return {
        chip: `<span class="mark creator">✓ confirmed</span>`,
        note: a.expires_at ? `Shown to fans as confirmed. The proof stays good until ${day(a.expires_at)}.` : `Shown to fans as confirmed.`,
        action: null,
      };
    }
    case "proven_stale":
      return {
        chip: `<span class="mark unconfirmed">! proof expired</span>`,
        note: `You proved this account${a.proved_at ? ` on ${day(a.proved_at)}` : ""} and the proof has lapsed${a.expires_at ? ` (it ran out on ${day(a.expires_at)})` : ""}. It is still on your passport, and it has stopped shielding this handle from impersonation reports.${fans} Only you can renew it, and it takes one short code.`,
        action: "reprove",
      };
    case "asserted_unproven":
      return {
        chip: `<span class="mark unconfirmed">! never proven</span>`,
        note: `You added this account but never proved control of it.${fans} Place a short code in that profile's bio and it goes live. Prove you control this account — proving it removes the caution.`,
        action: "prove",
      };
    case "transferred":
      return {
        chip: `<span class="mark unconfirmed">! proof belongs elsewhere</span>`,
        note: `The proof on file for this handle belongs to a different passport — the handle changed hands.${fans} Prove it again from here to claim it for yours. Prove you control this account — proving it removes the caution.`,
        action: "prove",
      };
    default:
      return {
        chip: `<span class="mark none">○ listed by Dhwaja</span>`,
        note: `Listed from public information; nobody has demonstrated control of it.${fans} Proving it is what turns it into an answer fans can rely on. Prove you control this account — proving it removes the caution.`,
        action: "prove",
      };
  }
}

/**
 * EVERY ACCOUNT, ITS STANDING, AND A BUTTON.
 *
 * Proofs expire at 180 days and re-proof was made possible in the same change —
 * and the only surface in the product that rendered `data-act="prove-start"`
 * was injected into the account-ADD success handler, holding its key in a
 * closure over the just-submitted form. Any repaint destroyed it. The console
 * listed a creator's accounts nowhere, so at day 180 the remedy existed
 * server-side with no door on it, and the row quietly stopped being served.
 *
 * This is the door. It is also where the lapse becomes VISIBLE BEFORE IT BITES:
 * the weekly digest counts stale proofs, this screen names them.
 *
 * AND IT IS WHERE THE BUTTON STOPPED LYING. Every unproven row used to carry
 * "Prove it", wired to `/account/prove/start` — a route that answers 409 "this
 * platform requires manual verification" for every platform without a fetchable
 * public profile, which is all but four of them. The creator pressed it, got a
 * refusal, and the fallback the refusal names (a manual claim) answered
 * "passport already claimed", because manual ACCOUNT proof on an already-claimed
 * passport does not exist. Two dead ends, one after the other, on the account
 * they had just added. A button that cannot work is worse than no button: it
 * spends the creator's trust and teaches them the product is broken.

 * `autoProvable` decides, from the SAME platform table the server reads, and a

 */

/**
 * X1 — the assertion budget, said out loud.
 *
 * An account you add and never prove holds its key against everybody else, so
 * how many of those one passport may hold at once is bounded. The ceiling is
 * enforced inside the write and its refusal arrives mid-ceremony, which is the
 * worst possible moment to learn a number — so the number is on the screen the
 * creator is already looking at, and it goes quiet until it is worth reading.
 *
 * Absent (an older server) renders nothing rather than a guess.
 */
function assertionBudgetLine(b?: { outstanding: number; max: number }, stuck = 0, reprovable = 0): string {
  if (!b || !Number.isFinite(b.outstanding) || !Number.isFinite(b.max) || b.outstanding < b.max - 2) return "";
  const left = Math.max(0, b.max - b.outstanding);
  // I3-1199 — EVERY ROW THIS LINE COUNTS IS NAMED WITH ITS WAY OUT. `assertions.ts`
  // counts what the passport is HOLDING with no `status` filter (a revoked row still
  // reserves the key). Since I23 the table draws a Prove button on a held revoked row
  // too (`heldRevoked` in accountsView), so `reprovable` counts the revoked rows that
  // HAVE a button and `stuck` only the ones on a platform with no self-serve proof.
  const freeable = reprovable > 0
    ? ` ${reprovable} of them ${pick(reprovable, "is", "are")} on ${pick(reprovable, "an account", "accounts")} you have already revoked: a revoked account still holds its handle, so it still holds its slot, and proving it (the Prove button on its row) frees the slot without putting it back on your page.`
    : "";
  // I52 W48 (D-1365): DRAFT copy (I52-CONSOLES REPORT.md ## JASON): a revoked row draws no Continue with X (accountsView
  // oauthBtn needs status active), so "no bio-code proof" names what is missing without calling X unreadable.
  const barred = stuck > 0
    ? ` <strong>${stuck} of them ${pick(stuck, "is", "are")} on a revoked account on a platform with no bio-code proof</strong>, so there is no self-serve proof for ${pick(stuck, "it", "them")} and nothing on this screen frees ${pick(stuck, "that slot", "those slots")}. If that is all of them, ask us — it is not something you can clear yourself.`
    : "";
  return left === 0
    ? `<p class="ds-banner banner banner-warn" role="status"><strong>You're holding ${b.max} unproven accounts.</strong> That's the limit — an account nobody has proved still reserves that handle against everyone else, so prove one of the ones already here to add another.${freeable}${barred}</p>`
    : `<p class="roi-note"><strong>${left} unproven ${left === 1 ? "account" : "accounts"} left</strong> before you'd need to prove one. You're holding ${b.outstanding} of ${b.max}; proving any of them frees a slot.${freeable}${barred}</p>`;
}

export function accountsView(rows: AccountStandingDto[] | null, read?: { status: number }, budget?: { outstanding: number; max: number }): string {
  if (!rows) return consoleUnread("accounts", "Your accounts", "your linked accounts", read);
  const active = rows.filter((r) => r.status === "active");
  const unproven = active.filter((r) => r.standing === "proven_stale" || r.standing === "asserted_unproven" || r.standing === "transferred");
  const needing = unproven.filter((r) => autoProvable(r.platform, r.id_kind ?? "handle", r.id)).length;
  // I3-1199 (I23, D-0993(1) pick A): a revoked or historical row the passport asserted still holds
  // its assertion slot (assertions.ts counts with no status filter), and proving it is the one thing
  // that frees the slot -- the server accepts that re-proof (claims.ts / domain-claims.ts F5 legs).
  // So the Prove button below is drawn on those rows too, and the rows counted here are only the
  // ones still left with NO self-serve proof. Derived from the same `rows` the table renders.
  const heldRevoked = (r: AccountStandingDto): boolean =>
    r.status !== "active" && (r.added_by === "creator" || r.added_by === "agency") && (r.standing === "proven_stale" || r.standing === "asserted_unproven" || r.standing === "transferred");
  const stuckSlots = rows.filter((r) => heldRevoked(r) && !autoProvable(r.platform, r.id_kind ?? "handle", r.id)).length;
  const reprovable = rows.filter((r) => heldRevoked(r) && autoProvable(r.platform, r.id_kind ?? "handle", r.id)).length;
  const waiting = unproven.length - needing;
  const nag =
    needing > 0
      ? `<p class="ds-banner banner banner-warn" role="status"><strong>${needing} of your accounts ${pick(needing, "needs", "need")} a control proof.</strong> An account without a current proof is not served to fans as confirmed and does not shield that handle from impersonation reports. Each one takes a short code in the profile's bio.</p>`
      : "";
  // The other population gets its OWN sentence, because the answer for it is
  // "there is nothing you can do here yet" and mixing the two would make the nag
  // above a demand a creator cannot satisfy.
  // I52 W48 (D-1365): DRAFT copy (I52-CONSOLES REPORT.md ## JASON): an active unproven X row is counted here and draws
  // Continue with X (oauthBtn below), so the banner names that route instead of calling X uncheckable.
  const held =
    waiting > 0
      ? `<p class="ds-banner banner banner-warn" role="status"><strong>${waiting} of your accounts ${pick(waiting, "is", "are")} on a platform with no bio-code proof.</strong> A bio-code proof needs a public profile Dhwaja can fetch, and only ${provablePlatforms().slice(0, 6).map((p) => esc(p)).join(", ")} (plus brand domains, by DNS) offer one today. An X account can be proven by X's own sign-in instead: the Continue with X button on its row. For any other there is no button yet — not because you missed a step, and not against you. Proving any account Dhwaja <em>can</em> check strengthens the whole passport, and it is what an operator looks at for verification.</p>`
      : "";
  const body =
    rows.length === 0
      ? `<p class="empty">No accounts on this passport yet. ${nav("#/onboarding", "Add one under Get set up")}.</p>`
      : `<table class="ds-table accounts"><thead><tr><th scope="col">Platform</th><th scope="col">Handle or ID</th><th scope="col">Standing</th><th scope="col">On your public page</th><th scope="col"></th></tr></thead><tbody>${rows
          .map((a) => {
            const s = standingLine(a);
            const state = a.status !== "active" ? ` <span class="m">(${esc(a.status)})</span>` : "";
            // A FILED CORRECTION MUST BE VISIBLE WHEREVER THE ACCOUNT IS. This
            // screen is where a creator comes back to check on their accounts,
            // and a row they have already reported as not theirs would otherwise
            // read here exactly like one they never touched — so they would file
            // it again (a no-op they cannot see) or conclude nothing happened.
            const corr = a.correction_pending === true
              ? `<p class="hint"><strong>You told us this isn't yours.</strong> It's queued for a person at Dhwaja and stays listed until they act. ${nav("#/welcome", "See what we found →")}</p>`
              : "";
            // A domain account proves via DNS (H18), not a bio code: its button
            // opens the TXT-record flow instead of /account/prove/*.
            const isDomain = a.platform === "domain" || a.id_kind === "domain";
            const provable = autoProvable(a.platform, a.id_kind ?? "handle", a.id);
            const revokedHold = heldRevoked(a);
            const btn =
              (a.status === "active" || revokedHold) && s.action
                ? !provable
                  ? `<span class="roi-sub">no self-serve proof yet</span>`
                  : isDomain
                    ? `<button class="ds-btn" data-act="acct-prove-domain" data-domain="${esc(clamp(a.id, 253))}">${s.action === "reprove" ? "Re-prove" : "Prove it"}</button>`
                    : `<button class="ds-btn" data-act="acct-prove" data-key="${esc(clamp(a.account_key, 600))}">${s.action === "reprove" ? "Re-prove" : "Prove it"}</button>`
                : "";
            // A revoked or historical row is not on any public roster to begin
            // with, so offering to shield it would be a control with no effect.
            const vis = a.status === "active" ? visibilityCell(a) : "";
            const freeNote = revokedHold && provable ? `<p class="hint">${REVOKED_PROOF_NOTE}</p>` : "";
            // #147M-13 (I16-W01D): the self-remove control removal.ts:1712 sends a
            // creator to ("remove it from your accounts screen instead"). Drawn on
            // exactly the rows that refusal covers — active, added on the
            // passport's own authority — and signed with the passkey
            // (`account_revoked` is CREATOR_SIGNABLE, passkeys.ts:675). An unknown
            // `added_by` draws nothing (fail-closed, as welcomeView does).
            const own = a.status === "active" && (a.added_by === "creator" || a.added_by === "agency");
            // I3-1721 (I25-INT, I25-W1-IDENTITY H8): an active X row that needs a proof can also be proven by X's own
            // sign-in (POST /account/prove/oauth/x/start); the label is the console's existing "Continue with X".
            const oauthBtn = a.status === "active" && s.action && a.platform === "x" ? ` <button class="ds-btn" data-act="acct-prove-oauth" data-platform="x" data-key="${esc(clamp(a.account_key, 600))}">Continue with X</button>` : "";
            const rm = own ? ` <button class="ds-btn" data-act="acct-remove" data-key="${esc(clamp(a.account_key, 2100))}" data-label="${esc(clamp(a.id, 80))}">Remove from my passport</button>` : "";
            return `<tr><td>${esc(clamp(a.platform, 40))}</td><td><code>@${esc(clamp(a.id, 80))}</code>${state}</td><td>${s.chip}<p class="hint">${s.note}</p>${corr}</td><td>${vis}</td><td>${oauthBtn && !provable ? "" : btn}${oauthBtn}${rm}${freeNote}</td></tr>`;
          })
          .join("")}</tbody></table>`;
  return `<section id="accounts"><h2>Your accounts</h2>
  ${nag}
  ${held}
  <p>Every account on your passport, and whether anybody has demonstrated control of it. A proof lasts about six months — that is deliberate, because handles change hands — so renewing one occasionally is normal and is not a mark against you.</p>
  ${assertionBudgetLine(budget, stuckSlots, reprovable)}
  <details class="shield-help"><summary>What "on your public page" means</summary><p>${esc(SHIELD_EXPLAINER)}</p><p>${esc(UNSHIELD_WARNING)}</p></details>
  ${body}
  <div id="ob-prove" aria-live="polite"></div>
  <div id="accounts-result" aria-live="polite"></div>
  ${brandContestPanel()}</section>`;
}

/**
 * WHAT A CREATOR IS ACTUALLY OWED IN THEIR FIRST TEN SECONDS.
 *
 * Every one of the four claim flows (bio-code, manual review, domain/DNS, OAuth)
 * dropped a newly-claimed creator on a screen that asked them for something —
 * `#/onboarding` on two of them, `#/dashboard` (an empty supporter-stats page)
 * on the other two. Nothing anywhere showed them THE THING THAT WAS ALREADY
 * TRUE: Dhwaja had been publishing a list of accounts under their name, built by
 * scraping public information, before they ever arrived. That list is the
 * product's first and largest factual claim about a person, it was made without
 * asking, and the person it is about was the only party who could not see it.
 *
 * So this screen leads with the list and with the correction, in that order,
 * and asks for nothing above the fold. The checklist still exists and is linked
 * at the bottom; it is not what a stranger's first screen should be.
 *
 * THE SPLIT IS BY PROVENANCE, NOT BY STANDING, and that is the whole design.
 * "We found these" can only be said about rows Dhwaja itself listed
 * (`added_by` operator/system). A row the creator or their agency added is
 * their own claim and belongs in the second group, where the answer to a wrong
 * entry is to manage it, not to correct us. The server enforces the same split
 * in `fileMisattribution`, so the button is drawn exactly where it would be
 * honoured — an unknown `added_by` (older server) draws nothing, which is the
 * fail-closed direction.
 *
 * WHAT THE BUTTON HONESTLY PROMISES. It files a correction for an operator; it
 * does not take the row down on the spot, and the copy says the account stays
 * listed until someone looks. Overstating that would be the easiest sentence to
 * write here and the one that would cost the most: a creator who is told "done"
 * and then finds the impostor still on their public page learns that this
 * product's statements about itself are decoration.
 */
/**
 * THE SECOND BEAT — "and here's what we're already watching for you".
 *
 * The screen leads with the creator's accounts, which is what they have an
 * opinion about. This is the half that says the product is doing something FOR
 * them, and until now it could not be said at all: the only creator-facing
 * number was published warnings, which is zero on day one — for exactly the
 * person this screen exists for.
 *
 * NO COUNT AND NO TARGETS for anything unreviewed. A report names a suspected
 * impostor nobody has checked; a tally is a number that goes up, and that is
 * pressure to go hunting on platforms Dhwaja does not control, against accounts
 * that may not be impostors at all.
 *
 * A FAILED READ RENDERS NOTHING, not a reassuring zero. This block's whole
 * content is a claim about what Dhwaja is doing; an outage saying "nothing is
 * being reviewed" would be the most comforting possible lie, on the screen whose
 * job is to be straight with a creator on their first day.
 */
/**
 * WHY DHWAJA THINKS IT IS ABOUT YOU — one sentence, chosen by the server's
 * `review_basis`, sitting between the alarm and the ask-not-to-go-hunting.
 *
 * The screen already hedged that "an account that looks like an impostor
 * sometimes isn't". What it could not say is whether the hedge applied to THEIR
 * case — a reviewer's own identification and a bare substring collision
 * produced identical copy.
 *
 * THE HEADLINE ABOVE USED TO SAY "A PERSON AT DHWAJA IS LOOKING AT" and it does
 * not any more, because it was false on a path this field makes visible.
 * `face-triage.ts` sets `state = 'escalated'` itself, from the actor
 * `system:face-triage`, after an auto-draft — so an item can enter the reviewed
 * population with no human having opened it, and the `likeness` sentence below
 * is exactly the one such an item renders. "in our review queue" is true of both
 * doors. Provisioning-gated today, and the rule this repo works to is that the
 * sentence has to be right before the gate flips, not after.
 *
 * A basis this client does not recognise, or a server that sends none, renders
 * NOTHING. The unqualified paragraph is the honest fallback; inventing a
 * confidence to fill the gap is the failure this fixes.
 */
function basisLine(basis: NamingBasisDto | null | undefined): string {
  switch (basis) {
    case "operator":
      return `<p>Someone here matched at least one of them to you by hand \u2014 that part is not a machine's guess.</p>`;
    case "identity":
      return `<p>At least one is a close copy of a handle or domain of yours: the same once padding and punctuation come off (an <em>official</em> or <em>backup</em> on the end, a dot, a 1 standing in for an l), or a letter or two away from it, or your name on a different ending, or your name written in another alphabet. That is a machine reading the spelling \u2014 it says nothing about who is behind the account.</p>`;
    case "likeness":
      return `<p>At least one was matched to you by our image check: the picture on it is one of yours \u2014 the same file, or a near-copy of it. A stolen photo is not proof of who is using it, and the comparison was a machine's.</p>`;
    case "resemblance":
      return `<p><strong>These sit on our weakest signal.</strong> Something about them resembles you and the machine cannot tell what the resemblance means \u2014 it cannot separate a fan account, a tribute, or a shop from an impostor. It is a reason for a person here to look, not a conclusion.</p>`;
    default:
      return "";
  }
}

/** I3-0094: the PSA (public "someone is impersonating you" page) URL a
 * creator can share was never actually rendered — only a generic
 * "There is a page you can share" link to `#/report`, the console's OWN
 * report screen, not the shareable PSA. Building the real link needs the
 * deployment's public origin (`publicBase`, the same value `legalFooter`
 * and `shareLink` already validate), which this function didn't receive;
 * it's threaded through as an optional param so the caller (`welcomeView`,
 * and its own caller in main.ts) can supply it. */
function psaLink(base: string | null | undefined, psaPath: string | null): string | null {
  if (typeof base !== "string" || !/^https:\/\/[a-z0-9.-]+(:\d+)?$/i.test(base)) return null;
  if (!psaPath) return null;
  return `${base}${psaPath.startsWith("/") ? "" : "/"}${psaPath}`;
}

/**
 * WHAT ELSE A RECOVERY TOOK, AND HOW TO GET IT BACK.
 *
 * A hijack resolution does not stop at the passkeys. It also retires every
 * payout destination (live AND staged — an attacker's staged swap would
 * otherwise go live a day later) and revokes every agency grant on the
 * passport. Both are correct and both are INVISIBLE from the creator's side:
 * the money screen goes quiet and the agency stops being able to act, with no
 * explanation anywhere.
 *
 * Fixing a security hole by manufacturing a support ticket is not fixing it, so
 * the same panel that explains the device reset explains these two and names
 * the way back. The route back is deliberately the ordinary one — reauthorize
 * the agency, re-add the destinations through the normal cooling-off — because
 * recovery is operator-adjudicated and never self-service (docs/DIRECTION.md),
 * and because BOTH require a passkey.
 *
 * IT NO LONGER HANGS OFF THE PASSKEY COUNT. It used to render only when
 * `needs_reenroll` was true, which meant it never appeared at all for a hijack
 * resolved without `reset_passkeys` (not a forced action — an operator handling
 * a stolen SESSION reasonably ticks `force_logout` alone, and the payout
 * retraction and grant revocation happen anyway), and it deleted itself the
 * instant the creator registered a device — the moment they start needing it.
 * The server now returns `recovery_aftermath`, keyed to the recovery and good
 * for 30 days, and this shows on Passkeys, Payout and Agencies: the three
 * screens a recovered creator actually visits looking for what is missing.
 */
export function recoveryAftermathPanel(cleared?: RecoveryClearedDto): string {

  const agency = cleared?.agency_grants === true;

/** Digit grouping without Intl (the console ships no locale data): 131072 →
 * "131,072". Numbers a creator has to act on must be readable. */
function num(n: unknown): string {
  return typeof n === "number" && Number.isFinite(n) ? Math.trunc(n).toString().replace(/\B(?=(\d{3})+(?!\d))/g, ",") : "0";
}

/**
 * WHAT VERIFICATION IS, AND HOW A CREATOR GETS IT.
 *
 * `POST /account/reverse-codes/pool` answers 403 to an unverified creator. That
 * is not an error and it must not read like one: it is the product telling
 * someone their standing does not reach this feature yet, and the only useful
 * reply explains what the standing is and what moves it. "Forbidden" would be a
 * dead end on the one page they came to make money on.
 *
 * It also must not invent a path. Verification is an operator review — there is
 * no self-serve request route in trust-core and no "apply" button in this
 * console — so this copy says so out loud and points at the two things the
 * creator actually controls, both of which live under Get set up.
 */
export type VerificationSubject = "pool" | "upload" | "generate";

/** What was refused, in the creator's terms. One sentence per producer, so the
 * explainer never tells someone filling a pool that "no pool was created". */
const VERIFICATION_LEDE: Record<VerificationSubject, string> = {
  pool: "so no pool was created",
  upload: "so nothing was added to that pool",
  generate: "so nothing was minted and no vouchers exist",
};

/**
 * AS-069: same class of bug `verifiedGateNote` exists to fix (see its
 * docblock) — "isn't verified yet" is false for a creator who WAS verified and
 * lapsed, and this refusal never got the same lapse-vs-never split. `expiredAt`
 * is optional and defaults to the old wording so this stays a no-op until a
 * caller threads the creator's `VerificationDto.expired_at` through.
 */
export function rcVerificationNeededView(subject: VerificationSubject = "pool", expiredAt?: string | null): string {
  const standing = expiredAt ? `your verification lapsed on ${esc(day(expiredAt))}` : "your passport isn't verified yet";
  return `<div class="ds-banner banner banner-warn">Code pools are for <strong>verified</strong> creators — ${standing}, ${VERIFICATION_LEDE[subject]}.</div>
  <div class="explainer">
  <p><strong>What verified means here.</strong> <em>Claimed</em> means you proved you control an account. <em>Verified</em> means a person at Dhwaja reviewed the accounts on your passport and vouched that they are really yours. It is granted by an operator and it can be taken away again — that revocability is the only reason it means anything to a fan.</p>
  <p><strong>Why this feature waits for it.</strong> A pool is a monetization surface: fans qualify, claim a code, and go spend money on your platform. Dhwaja only lets a passport it has actually vetted stand behind that, because an impostor with a pool is exactly the harm this product exists to stop.</p>
  <p><strong>How you get there.</strong> The Verification page carries the button that asks, and pressing it files a request and changes nothing by itself — it is a queue, not a switch, because verification is an operator review, not a purchase and not a form. Two things move that review, and both are yours: <strong>prove more of your accounts</strong> (each proof is a separate check an operator can see) and <strong>set a verified contact email</strong> so we can reach you during review. Both live under Get set up.</p>
  <p>Nothing about your account changed. Your passport, your proofs and every warning protecting you are exactly as they were.</p>
  </div>
  <p>${nav("#/onboarding", "Get set up →")}</p>`;
}

/**
 * CREATE → OUTCOME ADAPTER. Same three-way discipline as `rcVerifyOutcome`, with
 * one more axis: this is a WRITE, so a request that never completed leaves the
 * outcome genuinely UNKNOWN — the pool may exist. "Failed, try again" would be a
 * guess, and the guess that costs the creator a duplicate pool.
 */
export function rcCreateOutcome(res: import("./api.js").ApiResult<{ ok?: boolean; pool_id?: string }>, expiredAt?: string | null): string {
  if (res.ok && res.data?.pool_id) {
    const id = esc(clamp(res.data.pool_id, 32));
    return `<div class="ds-banner banner banner-ok">Pool created — <code>${id}</code>.</div>
  <p class="roi-note">It is live and <strong>empty</strong>: a fan who qualifies right now is told it's sold out. Add codes to it below.</p>`;
  }
  // A 200 THAT DID NOT PARSE IS NOT A REFUSAL. `{ok:true,status:200,data:null}`
  // is what the client produces for a body it could not read (a proxy's error
  // page, a truncated response), and the pool may well have been created — this
  // header's own words, "the guess that costs the creator a duplicate pool",
  // described exactly what the old fall-through then guessed.
  if (res.ok) {
    return `<div class="ds-banner banner banner-warn ${UNREAD_CLASS}">Dhwaja answered, but not with anything we could read — so we can't tell you whether that pool was created. It may well exist. Reload this page and look at the list before you create it again, or you'll have two.</div>`;
  }
  if (isVerificationRefusal(res)) return rcVerificationNeededView("pool", expiredAt);
  if (res.status === 401) return sessionExpiredWrite("No pool was created", "create the pool again");

  if (res.status === 0 || res.status >= 500) {
    return `<div class="ds-banner banner banner-warn ${UNREAD_CLASS}">We couldn't tell whether that pool was created — Dhwaja didn't answer. This says <strong>nothing</strong> about your verification status, and nothing about the pool: it may well exist. Reload this page and look at the list before you create it again, or you'll have two.</div>`;
  }
  const why = res.data?.error;
  return `<div class="ds-banner banner banner-err">Dhwaja refused it${why ? `: ${esc(clamp(why, 200))}` : ""}. No pool was created.</div>`;
}

/**
 * THE BATCH THAT IS TOO BIG, CAUGHT BEFORE IT IS SENT.
 *
 * The upload route caps the RAW BODY at 128 KiB and answers 413, and a creator
 * with a real list of a few thousand codes hits that on their first try. A 413
 * surfacing as "error" tells them nothing; worse, they retry it. So the batch is
 * counted and sized first and this view says the number to split at.
 *
 * It does NOT split for them. Chunking an oversize upload into several automatic
 * requests would defeat the cap the server put there on purpose.
 */
export function rcUploadPreflightView(plan: CodeBatchPlan): string {
  if (plan.found === 0) return `<div class="ds-banner banner banner-err">Nothing to upload — paste your codes, or choose a file, first.</div>`;
  if (plan.malformedTotal > 0) {
    const shown = plan.malformed
      .map((m) => `<li>line ${num(m.line)}: <code>${esc(clamp(m.value, 80))}</code></li>`)
      .join("");
    const more = plan.malformedTotal > plan.malformed.length ? `<li class="roi-sub">…and ${num(plan.malformedTotal - plan.malformed.length)} more</li>` : "";
    return `<div class="ds-banner banner banner-err">${num(plan.malformedTotal)} of your ${num(plan.found)} entries aren't valid code values — nothing has been uploaded.</div>
  <p class="roi-note">Dhwaja accepts or refuses an upload as a whole batch, not line by line, so one bad entry stops all ${num(plan.found)}. A code is 1–${MAX_CODE_LEN} characters of A–Z, a–z, 0–9 and <code>. _ % + -</code> — no spaces and no other punctuation. Fix or delete these, then send again:</p>
  <ul class="badcodes">${shown}${more}</ul>`;
  }
  if (plan.overCount || plan.overSize) {
    return `<div class="ds-banner banner banner-err">Too big to send in one go — nothing has been uploaded.</div>
  <p class="roi-note">This batch is <strong>${num(plan.codes.length)} codes</strong> (${num(plan.bytes)} characters). One upload is capped at ${num(MAX_CODES_PER_BATCH)} codes and ${num(MAX_UPLOAD_BODY)} characters, and we don't quietly split it into several uploads for you — that cap is deliberate. <strong>Split it yourself: send the first ${num(plan.fitsPerRequest)}, then paste the rest and send again.</strong> As many batches into the same pool as you like; a code already in the pool is skipped, never duplicated.</p>`;
  }
  return "";
}

/**
 * UPLOAD → OUTCOME ADAPTER. A transport failure is not a rejection: the batch may
 * have landed. The one thing that makes this safe to say is that the server
 * de-dups by value within the pool, so re-sending the same list adds nothing
 * twice — which is exactly what the creator is told to do.
 */
export function rcUploadOutcome(res: import("./api.js").ApiResult<{ ok?: boolean; added?: number; skipped?: number }>, plan: CodeBatchPlan, expiredAt?: string | null): string {
  if (res.ok && res.data) {
    const added = typeof res.data.added === "number" ? res.data.added : 0;
    const skipped = typeof res.data.skipped === "number" ? res.data.skipped : 0;
    const dupNote = plan.duplicates > 0 ? `<p class="roi-note">${num(plan.duplicates)} repeat${plan.duplicates === 1 ? "" : "s"} within your own list ${plan.duplicates === 1 ? "was" : "were"} collapsed before sending.</p>` : "";
    const skipNote = skipped > 0 ? `<p class="roi-note">${num(skipped)} ${skipped === 1 ? "was" : "were"} already in this pool and ${skipped === 1 ? "was" : "were"} skipped — a value is stored once per pool, so re-sending a list is always safe.</p>` : "";
    // I3-1969 - the success banner said the codes were added and stopped,
    // with no pointer to where they actually reach a fan.
    const nextStep = `<p class="roi-note">Now get them to fans: ${nav("#/broadcast", "send a broadcast")} or fold this pool into ${nav("#/drop", "your next drop")}.</p>`;
    return `<div class="ds-banner banner banner-ok">Added ${num(added)} code${added === 1 ? "" : "s"} to the pool.</div>${nextStep}${skipNote}${dupNote}`;
  }
  // Same rule as create: an unreadable 200 is not a refusal. The batch may have
  // landed, and re-sending is safe because the pool de-dups by value.
  if (res.ok) {
    return `<div class="ds-banner banner banner-warn ${UNREAD_CLASS}">Dhwaja answered, but not with anything we could read — so we can't tell you how much of that batch landed. Reload this page and check the pool's code count; either way, <strong>sending the same list again is safe</strong> — codes already in the pool are skipped, never duplicated.</div>`;
  }
  // 403 IS TWO ANSWERS on this route: `verified creators only` (a standing
  // problem, which has an explainer) and `not your pool` (an ownership problem,
  // which does not). Only the exact token routes to the explainer — telling an
  // owner-mismatch that they aren't verified would be the same collapse the
  // outage branch below exists to prevent.
  if (isVerificationRefusal(res)) return rcVerificationNeededView("upload", expiredAt);
  if (res.status === 413) {
    return `<div class="ds-banner banner banner-err">Dhwaja refused the batch as too large — nothing was added.</div>
  <p class="roi-note">Send at most ${num(plan.fitsPerRequest)} of these at a time (the cap is ${num(MAX_CODES_PER_BATCH)} codes and ${num(MAX_UPLOAD_BODY)} characters per upload), then send the rest.</p>`;
  }
  if (res.status === 401) return sessionExpiredWrite("No codes were added", "re-send the batch");

  if (res.status === 0 || res.status >= 500) {
    return `<div class="ds-banner banner banner-warn ${UNREAD_CLASS}">We couldn't tell whether that batch landed — Dhwaja didn't answer. It may have. Reload this page and check the pool's code count; either way, <strong>sending the same list again is safe</strong> — codes already in the pool are skipped, never duplicated.</div>`;
  }
  const why = res.data?.error;
  return `<div class="ds-banner banner banner-err">Dhwaja refused the batch${why ? `: ${esc(clamp(why, 200))}` : ""}. Nothing was added.</div>`;
}

/**
 * GENERATE → OUTCOME ADAPTER. The response body IS the product here, so the
 * failure modes are unusually sharp:
 *
 *  - A request that never completed may still have MINTED the vouchers. They
 *    would then exist, be claimable by fans, and be unreadable by the creator
 *    until they press Show codes (I3-0231: `GET /pool/:id/generated` lists a pool's
 *    generated values), so this outcome points there. Saying "it
 *    failed, try again" would be false AND would quietly double the pool.
 *  - A refusal is the server's sentence, shown verbatim — EXCEPT the one that
 *    is about standing rather than about this request. `/pool/:id/generate` is
 *    gated on a CURRENTLY-verified passport, so a creator who made a pool while
 *    verified and was then de-verified gets `403 verified creators only` here;
 *    that is the explainer's case, not a dead end. The other 403 on this route
 *    (`not your pool`) is an ownership answer and stays verbatim.
 *
 * IT NEVER RENDERS THE VALUES. It returns them, and `showOneTimeCodes` in
 * main.ts is the single place that puts voucher values on a screen — rendering
 * and arming the leave-the-page guard as one act, so a producer cannot do the
 * first without the second. That is why this returns a union rather than HTML.
 */
export type RcGenerateOutcome = { kind: "minted"; codes: string[] } | { kind: "message"; html: string };

export function rcGenerateOutcome(res: import("./api.js").ApiResult<{ ok?: boolean; added?: number; codes?: string[] }>, count: number, expiredAt?: string | null): RcGenerateOutcome {
  const msg = (html: string): RcGenerateOutcome => ({ kind: "message", html });
  const minted = res.ok && res.data && Array.isArray(res.data.codes) ? res.data.codes.filter((c): c is string => typeof c === "string") : null;
  if (minted && minted.length > 0) return { kind: "minted", codes: minted };
  if (res.status === 401) return msg(sessionExpiredWrite("Nothing was minted", "generate again"));

  if (!res.ok && (res.status === 0 || res.status >= 500)) {
    return msg(`<div class="ds-banner banner banner-err ${UNREAD_CLASS}">We asked Dhwaja for ${num(count)} voucher${count === 1 ? "" : "s"} and got no answer — so we have no codes to show you, and we cannot tell you whether they were minted.</div>
  <p class="roi-note">If they were, they are in the pool and fans can still claim them — you just don't hold the list yet. Reload this page, then press <strong>Show codes</strong> beside that pool to read back every code Dhwaja generated for it. Verifying also works when a supporter presents one. <strong>Compare the pool's code count before you generate again</strong>: generating mints a fresh batch on top, it never recovers this one.</p>`);
  }
  if (res.ok) return msg(`<div class="ds-banner banner banner-err">Dhwaja answered without any codes, so there is nothing to show you. Reload this page and check the pool's code count before generating again.</div>`);
  if (isVerificationRefusal(res)) return msg(rcVerificationNeededView("generate", expiredAt));
  const why = res.data?.error;
  return msg(`<div class="ds-banner banner banner-err">Dhwaja refused it${why ? `: ${esc(clamp(why, 200))}` : ""}. Nothing was minted.</div>`);
}

/**

 * `generateCodes` returns the values once; nothing in trust-core will ever hand
 * them back. So this view is built around that constraint rather than mentioning

 *    both the clipboard and the download can still select and save them by hand

  const shown = codes.slice(0, MAX_CODES_PER_BATCH).map((c) => clamp(c, 64));

/**
 * I3-0231 - THE RECOVERY VIEW: what "Show codes" paints. Returns HTML for every outcome (it is
 * not the one-time panel: it holds no leave-the-page guard, because the values can be read
 * again at will). `ok === false` is "we could not ask" and is kept apart from a pool that holds
 * no generated codes: telling a creator "nothing was generated" over a dropped connection is the
 * same collapse `rcVerifyOutcome` exists to prevent. The unused values go in a read-only box
 * (selectable by hand); the redeemed ones are only counted, because they are already spent.
 */
export function rcReviewOutcome(res: import("./api.js").ApiResult<{ codes?: import("./api.js").GeneratedCodeDto[] }>, poolId: string): string {
  if (res.status === 401) return sessionExpiredWrite("Nothing was changed", "show the codes again");
  if (!res.ok) {
    if (res.status === 0 || res.status >= 500) return `<div class="ds-banner banner banner-err ${UNREAD_CLASS}">We asked Dhwaja for this pool's codes and got no answer, so we can't tell you what it holds. Nothing was changed. Try again in a moment.</div>`;
    return `<div class="ds-banner banner banner-err">${esc(clamp(errorCopy(res.data?.error, "Dhwaja couldn't show that pool's codes."), 200))}</div>`;
  }
  const rows = Array.isArray(res.data?.codes) ? res.data!.codes!.filter((c) => c !== null && typeof c === "object" && typeof c.code_value === "string") : null;
  if (!rows) return `<div class="ds-banner banner banner-err ${UNREAD_CLASS}">Dhwaja answered without a list, so there is nothing to show. Nothing was changed. Try again in a moment.</div>`;
  if (rows.length === 0) return `<div class="ds-banner banner banner-warn" role="status">This pool holds no codes that Dhwaja generated. Codes you uploaded yourself are never listed here - you already have those.</div>`;
  const unused = rows.filter((c) => c.redeemed_at === null || c.redeemed_at === undefined);
  const used = rows.length - unused.length;
  const shown = unused.slice(0, 5000).map((c) => clamp(c.code_value, 64));
  const pool = POOL_ID_RE.test(poolId) ? `<code>${esc(poolId)}</code>` : "this pool";
  const textRows = Math.min(20, Math.max(4, shown.length));
  return `<div class="reviewed" id="rc-reviewed">
  <p><strong>${num(rows.length)} code${rows.length === 1 ? "" : "s"} generated for ${pool}</strong> &mdash; ${num(unused.length)} unused, ${num(used)} already used.</p>
  ${shown.length > 0 ? `<label for="rc-reviewed-text">Unused codes</label>
  <textarea id="rc-reviewed-text" rows="${textRows}" readonly spellcheck="false">${esc(shown.join("\n"))}</textarea>` : `<p class="roi-note">Every code in this pool has been used.</p>`}
  ${shownOf(shown.length, unused.length, "unused codes")}
  <p class="roi-note">These are the live values. Anyone who sees them can present one, so treat this list like the batch you first generated.</p>
</div>`;
}

/**
 * DID THE SAVE ACTUALLY HAPPEN? Copy and download are two independent browser
 * capabilities and either can be refused (clipboard permission, a blocked
 * download, no `navigator.clipboard` outside a secure context). Reporting them
 * as one "saved!" would be the reassurance-shaped lie the copy law forbids —
 * and here it costs the creator the batch.
 *
 * THE TWO HALVES ARE NOT EQUALLY KNOWABLE, and this copy no longer pretends
 * they are. `navigator.clipboard.writeText` RESOLVING is an observation: the
 * text is on the clipboard. A download is NOT observable from a page — the Blob,
 * the object URL and the synthetic click all succeed while the browser silently
 * drops the file ("Downloads blocked", an enterprise policy, a sandboxed frame).
 * So the file is reported as OFFERED, with the checking left to the only party
 * who can do it, and no wording here releases the leave-the-page guard; the
 * acknowledgement button does that, and it is a statement by the creator.
 */
export function rcCodesSaveStatus(r: { copied: boolean; offered: boolean; filename: string }): string {
  const file = `<code>${esc(clamp(r.filename, 80))}</code>`;
  const check = `<p class="roi-note">We can't see your downloads folder — no page can — so we can't tell you the file arrived. <strong>Open ${file} and check it holds every code</strong>, then press "I've saved them". If you lose them, Show codes beside the pool on the Reverse codes page reads them back.</p>`;
  if (r.copied && r.offered) {
    return `<div class="ds-banner banner banner-ok">Copied to your clipboard, and ${file} was handed to your browser to download.</div>${check}`;
  }
  if (r.copied && !r.offered) {
    return `<div class="ds-banner banner banner-err">Copied to your clipboard — but your browser <strong>refused the download outright</strong>, so there is no file. Your clipboard is one copy away from losing them: paste them somewhere permanent now, or select the box above and save them by hand. If you lose them, Show codes beside the pool on the Reverse codes page reads them back.</div>`;
  }
  if (!r.copied && r.offered) {
    return `<div class="ds-banner banner banner-err">The <strong>copy to your clipboard failed</strong>, so nothing is on your clipboard. ${file} was handed to your browser — but we can't see whether it saved. Open it and check; if it isn't there, select the box above and copy them by hand. If you lose them, Show codes beside the pool on the Reverse codes page reads them back.</div>`;
  }
  return `<div class="ds-banner banner banner-err"><strong>Nothing was saved.</strong> Your browser refused both the copy and the download. The codes are still in the box above — select them and save them by hand before you leave this page. If you lose them, Show codes beside the pool on the Reverse codes page reads them back.</div>`;
}

/**
 * THE RESPONSE → OUTCOME ADAPTER, exported so the three-way split is under test
 * rather than living inline in a DOM handler. `ok === false` means the request
 * did not complete — a dropped connection, a 500, an expired console session —
 * and it must never reach the same branch as a server that answered "no such
 * code". That collapse is what made the console tell a creator to refuse a
 * paying customer's valid code because the network blipped.
 */
export function rcVerifyOutcome(
  code: string,
  res: import("./api.js").ApiResult<import("./api.js").VerifyResultDto>,
): string {
  return rcVerifyResultView(code, res.ok ? (res.data ?? null) : null, { unreachable: !res.ok });
}

/** The verify outcome + (when valid & unused) a mark-used button. `code` is the
 * creator's own presented value — echoed back escaped so the redeem button can
 * carry it; a foreign/unknown value shows the same neutral "not found" (no
 * cross-creator oracle, matching the server).
 *
 * THREE OUTCOMES, NOT TWO. `unreachable` is the one this view was missing: the
 * caller collapsed a transport failure — a dropped connection, a 500, an expired
 * console session — into the same `null` as a genuine miss, so the console told
 * the creator "Not one of your codes — Don't honor it" about a code it had never
 * managed to ask about. That is a paying customer standing at a checkout being
 * refused because the network blipped, and the creator has no way to tell.
 *
 * A failed lookup is not a verdict about a code. It says only that we could not
 * look, and the instruction that follows from it is "don't act", not "refuse".
 */
export function rcVerifyResultView(
  code: string,
  r: import("./api.js").VerifyResultDto | null,
  opts: { unreachable?: boolean } = {},
): string {
  if (opts.unreachable) {
    return `<div class="ds-banner banner banner-warn ${UNREAD_CLASS}">Couldn't check this code — Dhwaja didn't answer. This says <strong>nothing</strong> about the code: it may be perfectly valid. Don't mark it used and don't refuse it on this basis — try again in a moment.</div>`;
  }
  if (!r || !r.found) return `<div class="ds-banner banner banner-err">Not one of your codes — or it doesn't exist. Don't honor it.</div>`;
  const where = r.title ? ` in <strong>${esc(clamp(r.title, 60))}</strong>` : "";
  if (r.redeemed) return `<div class="ds-banner banner banner-err">Already used${where}${r.redeemed_at ? ` on ${esc(clamp(r.redeemed_at, 10))}` : ""} — do not honor it again.</div>`;
  const claimedNote = r.claimed ? ` <span class="roi-sub">(claimed by a fan on Dhwaja)</span>` : "";
  return `<div class="ds-banner banner banner-ok">Valid &amp; unused${where}.${claimedNote}</div>
  <button class="ds-btn" data-act="rc-redeem" data-code="${esc(clamp(code, 64))}"${r.pool_id ? ` data-pool="${esc(clamp(r.pool_id, 32))}"` : ""}>Mark this code used</button>
  <div id="rc-redeem-result" aria-live="polite"></div>`;
}

/** #182a / I3-1050 (I16-W01D): the "Attach art" file control for one collectible

function attachArtControl(collectibleId: string, artHash: string): string {
  if (!/^[0-9a-f]{64}$/.test(artHash)) return "";
  return `<div data-attach class="inline-form"><label>Attach art <input type="file" accept="image/png,image/jpeg" data-act="attach-collectible-art" data-id="${esc(clamp(collectibleId, 80))}" data-hash="${artHash}"></label> <span class="hint">Pick the exact image you minted with — any other file is refused.</span> <span data-attach-result class="rmsg" aria-live="polite"></span></div>`;

/** #182a / I3-1050: every collectible on this passport still waiting for its art,

export function signResultView(r: { release_id?: string; stored?: boolean; storeError?: string; retryable?: boolean }): string {

/** Only https share links built from a shape-validated passport id are ever
 * rendered as hrefs. */
function shareLink(base: string, passportId: string): string | null {
  if (!/^https:\/\/[a-z0-9.-]+(:\d+)?$/i.test(base) || !/^psp_[0-9a-f]{16}$/.test(passportId)) return null;
  return `${base}/p/${passportId}`;
}

export function onboardingView(steps: StepState[], done: number, activeId: string | null, panel: string): string {
  const items = steps
    .map((s) => {
      const mark = s.done ? "✓" : "○";
      const cls = s.done ? "step done" : s.id === activeId ? "step active" : "step";
      return `<li class="${cls}"${!s.done && s.id === activeId ? ` aria-current="step"` : ""}><button class="ds-btn" data-act="ob-step" data-step="${esc(s.id)}">${esc(mark)} ${esc(s.title)}</button><p>${esc(s.detail)}</p></li>`;
    })
    .join("");
  // THE CROWN ONLY FIRES WHEN THE APPOINTMENT IS ALSO CLEAR (F9). It used to
  // fire on step count alone, so "All set — you're provably you 🎉" was the last
  // thing a creator heard on a passport where not one account was being served
  // to fans yet. A checklist that ends in a false all-clear is worse than one
  // that ends in a date.
  const pending = steps.find((s) => s.id === "appointment" && !s.done);
  const crown = pending
    ? `<p class="ds-banner banner banner-warn" role="status"><strong>Nearly.</strong> ${esc(pending.detail)}</p>`
    : done === steps.length
      ? `<p class="ds-banner banner banner-ok">All set — you're provably you. 🎉</p>`
      : `<p>${done}/${steps.length} complete</p>`;
  return `<section id="onboarding"><h2>Get set up</h2>${crown}<ol class="steps">${items}</ol><div id="ob-panel">${panel}</div></section>`;
}

/** #425: the ONE neutral sentence every refusal at the verify door gets. The server answers every eligibility
 * refusal with one uniform 409 (brand-enrolment.ts `uniform` -> NOT_CONTESTABLE), so this copy must not turn it
 * into a reason: it never says the record failed to propagate and never says there is nothing to contest. */
export const BRAND_CONTEST_REFUSED = "We can't file a dispute from this proof. If you own this domain and believe this is wrong, contact support — a person can look.";

/** #425: verify answer -> copy. Each outcome gets its own sentence; a 409 with no `outcome` is the uniform refusal. */
export function brandContestOutcome(res: { ok: boolean; status: number; data?: { outcome?: string; contest_id?: string; attempts_remaining?: number } | null }): string {
  const d = res.data ?? null;
  if (res.ok && d?.outcome === "filed") {
    return banner("ok", "Proved and filed. A person at Dhwaja has your dispute with the DNS proof attached; the current holder has been told a dispute was filed. Nothing moves until a person decides.") +
      (d.contest_id ? banner("ok", `Your dispute reference is ${d.contest_id}. Keep it for any question to support.`) : "");
  }
  if (res.status === 401) return banner("err", "Sign in to file a dispute — filing is done from your own passport. Your DNS record stays valid while you sign in.");
  if (res.status === 429) return banner("err", serverMessage({ ok: res.ok, status: res.status, data: res.data ?? null }) ?? "Too many attempts for now. Wait a bit, then try again."); // #963
  if (res.status === 503) return banner("err", "Domain checks aren't available right now. Nothing was filed; try again later.");
  // RG-CLIENT:1 (I16-FXE): no answer (status 0) or a server fault is NOT a refusal — the copy law says unknown is
  // never told as a decision. The server may or may not have filed; say so, and send them to check, not to support.
  if (res.status === 0 || res.status >= 500) return banner("err", "We couldn't get an answer from Dhwaja just now, so we can't tell whether your dispute was filed. Your DNS record stays valid for 24 hours — wait a minute and verify again; a dispute that was already filed is not filed twice.");
  if (d?.outcome === "expired") return banner("err", "That record expired — it's good for 24 hours. Start again and you'll get a fresh one.");
  if (d?.outcome === "record_absent") {
    const left = typeof d.attempts_remaining === "number" ? ` ${String(d.attempts_remaining)} ${plural(d.attempts_remaining, "attempt")} left.` : "";
    return banner("err", `We didn't see that TXT record on the domain. Check the name and value match exactly, then verify again.${left}`);
  }
  return banner("err", BRAND_CONTEST_REFUSED);
}

/**

 * not the API one.
 *
 * This rendered `${API_BASE}/p/<psp>`. In every normal deployment those are
 * different origins (`api.dhwaja.…` vs the public site), which meant the console

 * IS the proof" story depends on. `PUBLIC_BASE` is now its own build-time
 * constant, validated exactly like `API_BASE`, and it falls back to the API base
 * only when it is unset (a single-origin deployment), so nothing regresses where

/**
 * THE MONEY SCREEN. A payout tag is the highest-stakes value in the whole
 * console — a wrong one sends a fan's money to a thief — so this page renders
 * nothing it cannot stand behind:
 *  - A failed read of the current set is the WHOLE page failing closed. Editing
 *    is full-set replacement, and a creator who edits over a set they cannot
 *    see removes destinations without knowing it.
 *  - The copy states the server's semantics out loud (payout.ts): a change is
 *    passkey-signed onto the ledger, STAGES for 24 hours before it renders
 *    (the previous set keeps showing until then), and every change fires a
 *    "payout changed — verify before you send" alert to followers, so a
 *    briefly-compromised account can't silently swap an address and drain a
 *    payment window.

  // staged set if one is pending, else the live set (`currentIntentKey` in
  // trust-core payout.ts) — never just `tags` (what fans currently see). The
  // box has to prefill from that same intent: prefilling from `tips` (live)

  // silently discarded the pending change and re-staged the old live set with

  const editing = view.pending && Array.isArray(view.pending.tags) ? view.pending.tags : tips;
  const rows = tips.length > 0
    ? `<ul class="stats">${tips.slice(0, MAX_PAYOUT_TAGS).map((t) => `<li><strong>${esc(clamp(PAYOUT_TYPE_LABEL[t.type] ?? t.type, 30))}</strong> <code>${esc(clamp(t.value, 128))}</code></li>`).join("")}</ul>`

  return `<section id="payout"><h2>Payout destinations</h2>

That is exactly why this is the most protected thing you can edit here.</p>
  <h3>What fans currently see</h3>
  ${payoutDisplayNotice(view)}
  ${rows}
  ${payoutPendingPanel(view)}
  <h3>Change the set</h3>
  <p class="roi-note">One destination per line, as <code>type value</code> (types: ${types}) — e.g. <code>eth 0x1234…</code> or <code>cashapp $yourname</code>. At most ${MAX_PAYOUT_TAGS}. <strong>Submitting replaces the whole set</strong>; an empty box removes every destination.${view.pending ? /** A plain address only: no query/fragment/header characters can reach a mailto. */
const SUPPORT_MAILTO_RE = /^[^\s@?&#%/\\<>"'`=;:,]+@[^\s@?&#%/\\<>"'`=;:,]+\.[^\s@?&#%/\\<>"'`=;:,]+$/;

/** "contact support" with its destination (I3-1166 / I4-15-R6 R6-W2): a mailto
 * when it is an address, the plain text otherwise, the bare words when the
 * server named nothing. Exported (I4-18-R9) so earnings.ts can re-export this
 * copy instead of holding its own byte-identical one (earnings.ts imports
 * views.ts, so the import direction is this way to avoid a depcruise cycle). */
export function supportContact(to: string | null): string {
  if (to === null) return "contact support";
  if (SUPPORT_MAILTO_RE.test(to)) return `contact support at <a href="mailto:${esc(to)}" data-support-to>${esc(to)}</a>`;
  return `contact support (<span data-support-to>${esc(to)}</span>)`;
}

/**
 * PAYOUT CHANGE → OUTCOME ADAPTER. Same three-way discipline as the write
 * adapters on #/codes, sharpened for money:
 *  - a failure on the start/sign leg wrote NOTHING (the start is read-only and
 *    an unsigned set cannot stage) — safe to say "nothing changed";
 *  - a transport failure on the FINISH leg is an UNKNOWN OUTCOME: the change
 *    may be staged, with its follower alert out. Saying "failed, try again"
 *    could stage it twice and alert followers twice;
 *  - 503 `passkeys_unavailable` is the fail-closed door: no passkey config
 *    means NO payout changes, and the honest sentence is that this is the safe
 *    direction, not an error to work around.
 */
/** AS-069: `expiredAt` is optional and defaults to the old "isn't verified yet"
 * wording, same reasoning as `rcVerificationNeededView` above — this refusal
 * has the identical never-vs-lapsed gap and no caller threads the creator's
 * lapse date through yet. */
export function payoutOutcome(r: { ok: boolean; unchanged?: boolean; effective_at?: string; leg?: "start" | "sign" | "finish"; status?: number; error?: string }, expiredAt?: string | null): string {
  if (r.ok && r.unchanged) return `<div class="ds-banner banner banner-ok">No change — that set is already exactly what's staged or live. Nothing was re-staged and your followers were not re-alerted.</div>`;
  if (r.ok) {
    const when = r.effective_at ? ` at <strong>${esc(clamp(r.effective_at, 24))}</strong>` : ` after the ${PAYOUT_COOLING_OFF_HOURS}-hour cooling-off`;
    return `<div class="ds-banner banner banner-ok">Signed &amp; staged.</div>
  <p class="roi-note">Your new destinations go live${when}. Until then fans keep seeing the previous set, and your followers are being alerted to verify before they send. The change is on your ledger, signed by your passkey.</p>`;
  }
  if (r.status === 503 && r.error === PASSKEYS_UNAVAILABLE_ERROR) {
    return `<div class="ds-banner banner banner-err">Payout changes are signed with a passkey, and passkeys aren't configured on this deployment — so <strong>no payout destination can be changed at all right now</strong>, by you or by anyone who took your session. That's the safe direction; nothing about your current destinations changed.</div>`;
  }
  if (r.status === 401) return sessionExpiredWrite("Nothing was staged, so nothing was changed", "re-enter the destinations");

  if (r.status === 403 && r.error === PAYOUT_VERIFIED_ONLY_ERROR) {
    const standing = expiredAt ? `your verification lapsed on ${esc(day(expiredAt))}` : "your passport isn't verified yet";
    return `<div class="ds-banner banner banner-warn">Payout destinations are for <strong>verified</strong> creators — ${standing}, so nothing was staged.</div>
  <p class="roi-note">This is where a fan's money goes, so only a passport a person at Dhwaja has actually vetted may stand behind one — an impostor with a payout address is exactly the theft this product exists to stop. Verification is an operator review; proving more of your accounts and setting a verified contact email (both under ${nav("#/onboarding", "Get set up")}) are what move it.</p>`;
  }
  if (r.leg === "finish" && (r.status === 0 || (r.status !== undefined && r.status >= 500))) {
    return `<div class="ds-banner banner banner-warn ${UNREAD_CLASS}">We couldn't tell whether that change was staged — Dhwaja didn't answer after your passkey signed it. It <strong>may</strong> be staged, with the follower alert already out. Reload this page and check "Change pending" — it shows a staged change immediately, before the cooling-off, so you don't have to wait to find out. If it's not there, it's safe to sign the same set again: resubmitting an unchanged set is a no-op and never re-alerts your followers twice.</div>`;
  }
  return `<div class="ds-banner banner banner-err">Dhwaja refused it${r.error && errorCopy(r.error, "") ? `: ${esc(clamp(errorCopy(r.error, ""), 200))}` : ""}. Nothing was staged and your destinations are unchanged.</div>`;
}

/**
 * CANCEL → OUTCOME ADAPTER (M14). Same three-way discipline as payoutOutcome:
 *  - start/sign failure wrote NOTHING;
 *  - a transport failure on FINISH is an UNKNOWN OUTCOME (the cancel may have
 *    committed) — never "failed, try again";
 *  - 503 is the fail-closed door (a cancel is a payout mutation, so no passkey
 *    config means it can't be cancelled here either).
 * A cancel never re-alerts followers, so the unknown-outcome copy is calmer than
 * the set adapter's — there is no double-alert to warn about, only a double-sign.
 */
export function cancelStagedOutcome(r: { ok: boolean; unchanged?: boolean; leg?: "start" | "sign" | "finish"; status?: number; error?: string }): string {
  if (r.ok && r.unchanged) return `<div class="ds-banner banner banner-ok">There was no pending change to cancel — nothing was staged. Your current destinations are unchanged.</div>`;
  if (r.ok) return `<div class="ds-banner banner banner-ok">The pending change was cancelled. Your current destinations stay exactly as they are, and your followers were not alerted — the public set never changed.</div>`;
  if (r.status === 503 && r.error === PASSKEYS_UNAVAILABLE_ERROR) {
    return `<div class="ds-banner banner banner-err">Cancelling a payout change is signed with a passkey, and passkeys aren't configured on this deployment — so it can't be cancelled here right now. Nothing changed.</div>`;
  }
  if (r.status === 401) return sessionExpiredWrite("The pending change was not cancelled", "reload the payout page and try again");

  if (r.leg === "finish" && (r.status === 0 || (r.status !== undefined && r.status >= 500))) {
    return `<div class="ds-banner banner banner-warn ${UNREAD_CLASS}">We couldn't tell whether the cancellation went through — Dhwaja didn't answer after your passkey signed it. Reload this page and check whether "Change pending" still shows before signing again.</div>`;
  }
  return `<div class="ds-banner banner banner-err">Dhwaja refused the cancellation${r.error && errorCopy(r.error, "") ? `: ${esc(clamp(errorCopy(r.error, ""), 200))}` : ""}. Your pending change is unchanged.</div>`;
}

export function digestsView(contact: import("./api.js").ContactStateDto | null, digests: import("./api.js").DigestDto[] | null, read?: { status: number }, devices?: { list: import("./api.js").CreatorPushDeviceDto[] | null; status: number; pending?: readonly import("./api.js").CreatorPushRemovalHoldDto[] }): string {

  // said "Turn digest email off" and wrote `digests_enabled`, which stops the
  // digest PUSH as well. The email leg now has its own column — the one RFC 8058
  // one-click writes — so the button's words are true, and the master switch
  // gets its own control rather than being operated under a false label.

/** The compact form of `basisLine`, for a one-line row in the digest list. Same
 * fallback rule: an absent or unknown basis adds nothing. */
function basisTag(basis: NamingBasisDto | null | undefined): string {
  switch (basis) {
    case "operator":
      return " (matched to you by a person here)";
    case "identity":
      return " (matched on a close copy of a handle or domain of yours, by machine)";
    case "likeness":
      return " (matched on one of your own pictures, by machine)";
    case "resemblance":
      return " (our weakest signal — a resemblance, with nothing said about what it means)";
    default:
      return "";
  }
}
