#!/usr/bin/env node
// scripts/intake/room-write-audit.mjs — QUEUE #113 (D2, S-078 ISO)
//
// Detects a room's write footprint MECHANICALLY (ANTI-015 ruling: "detect mechanically, not by
// prose" — prose warnings measurably do not stop rooms writing to the mounted repo, ROWS.md
// #113). `snapshot` records what `git status` already sees as dirty, plus a content hash of each
// such path; a later `check` re-reads the same view and names every path that is newly dirty, or
// whose hash moved, since the snapshot — restricted to paths outside the room's `--allow` dirs.
//
//   node scripts/intake/room-write-audit.mjs snapshot <label> [--repo <path>]
//   node scripts/intake/room-write-audit.mjs check <label> --allow <dir> [<dir> ...] [--repo <path>]
//
// Snapshots land at `<repo>/.scratch/audit/<label>.json` (`--repo` defaults to cwd — so run from
// the tree being audited, same convention as `git status` itself).
//
// Exit 0 = clean (or snapshot written); exit 1 = violation(s) found; exit 2 = usage/config error.

// W02A-868a (I16-W04L): REPO and AUDIT_DIR are compared below, so both are REALPATHS — a symlink spelling of an
// in-tree dir resolved() only would read as out-of-tree and the tool would audit its own bookkeeping. A path not
// yet created (an --audit-dir the snapshot makes) keeps its tail on the realpath of its nearest existing ancestor
// (scripts/lib/real-or-tail.mjs).
const REPO = realOrTail(FLAGS.repo || process.cwd());
if (!existsSync(join(REPO, ".git"))) usage(`config error: ${REPO} is not a git repo (no .git)`);
const AUDIT_DIR = FLAGS["audit-dir"] ? realOrTail(FLAGS["audit-dir"]) : join(REPO, ".scratch/audit");
// The audit dir's tree-relative path when it lies inside REPO, else "". An audit dir that IS the tree or
// holds it is refused: the bookkeeping exclusion in isAuditPath would then exclude every path.
const isUnder = (child, parent) => { const r = relative(parent, child); return r !== "" && r !== ".." && !r.startsWith(`..${sep}`) && resolve(parent, r) === child; };
if (AUDIT_DIR === REPO || isUnder(REPO, AUDIT_DIR)) usage(`config error: --audit-dir ${AUDIT_DIR} is the audited tree or contains it`);
// JA2-03 (D-0873(2)): an in-tree audit dir is EXCLUDED from the audit (isAuditPath), so --audit-dir <repo>/src
// would hide every write under src. Inside the tree it must lie BELOW a directory named exactly `.scratch`
// (scratch, never audited content): the tree's own .scratch/audit, or the `.scratch/audit` an in-tree --out
// compile puts beside its PROMPT.md (audit-store.mjs compileAuditRoot, D-0870(1)). Outside the tree: accepted.
const inTreeScratch = (d) => relative(REPO, d).split(sep).slice(0, -1).includes(".scratch");
if (isUnder(AUDIT_DIR, REPO) && !inTreeScratch(AUDIT_DIR)) usage(`config error: --audit-dir ${AUDIT_DIR} is inside the audited tree but not under a .scratch/ dir (it would be excluded from the audit)`);
const AUDIT_DIR_REL = isUnder(AUDIT_DIR, REPO) ? relative(REPO, AUDIT_DIR).split(sep).join("/") : "";

// git status --porcelain=v1 -z: NUL-separated records. Each record is `XY path`; a rename/copy
// record (X or Y is R/C) is followed by one MORE NUL-separated token, the original path — which
// this audit does not need, but must still consume or the next record parses as garbage.
// This tool's own bookkeeping lives at `.scratch/audit/` inside the repo it audits. Writing a
// snapshot there is itself a filesystem write, which `git status` will report — as the leaf file
// once `.scratch/audit/` is tracked-dirty, or collapsed to `.scratch/` (or `.scratch/audit/`) when
// the directory is still wholly untracked. Left in, the tool would flag its own first snapshot as
// a violation on the very next `check`. Excluded unconditionally, not by relying on a gitignore
// entry existing in whatever tree or temp repo this runs against.
const AUDIT_REL = ".scratch/audit";
// QUEUE #836 (I10-FX12): an --audit-dir inside the tree is this tool's bookkeeping too: that dir and what is
// under it are excluded, never its ancestors (an ancestor such as a room's report dir is audited as before).
function isAuditPath(p) {
  const norm = p.replace(/\/+$/, "");
  if (AUDIT_DIR_REL && (norm === AUDIT_DIR_REL || norm.startsWith(AUDIT_DIR_REL + "/"))) return true;
  return norm === AUDIT_REL || norm.startsWith(AUDIT_REL + "/") || (AUDIT_REL + "/").startsWith(norm + "/");
}

// QUEUE #543 (B23-D9): also returns `removed`, the paths GIT ITSELF classes as a removal of a tracked
// file (X or Y is `D`, or a rename's orig). Such a path cannot be hashed, so without this set it was
// absent from `baseline.hashes` and read as newly dirty at every later check, forever.
function gitStatusPaths(repo) {
  const raw = execFileSync("git", ["status", "--porcelain=v1", "-z", "--untracked-files=all"], { cwd: repo, encoding: "utf8", maxBuffer: GIT_MAX_BUFFER });
  const tokens = raw.split("\0").filter((t) => t.length > 0);
  const paths = [];
  const removed = new Set();
  let i = 0;
  while (i < tokens.length) {
    const rec = tokens[i];
    const xy = rec.slice(0, 2);
    const path = rec.slice(3);
    if (!isAuditPath(path)) { paths.push(path); if (xy.includes("D")) removed.add(path); }
    i++;
    if (xy[0] === "R") {
      // a rename's orig path is a REMOVAL at that location — flag it too (unless audit bookkeeping).
      const orig = tokens[i];
      if (orig !== undefined && !isAuditPath(orig)) { paths.push(orig); removed.add(orig); }
      i++;
    } else if (xy[0] === "C") {
      i++; // a copy leaves orig unchanged in place — skip-only, do not flag it
    }
  }
  return { paths, removed };
}

// #302a (S-081 judge JN-6): `git status` never reports an UNTRACKED path that matches
// .gitignore -- a room dropping a secret or scratch output under a path it assumes is invisible
// defeats gitStatusPaths above for exactly that reason (a TRACKED file's edits still show up in
// ordinary `git status` regardless of .gitignore, so that case is already covered). Watch a
// fixed, small set of roots known to matter and fold any ignored-but-present file under them
// into the SAME path list the rest of this tool audits -- hashing, snapshot/check diffing and
// --allow filtering then treat it identically to an ordinary dirty path, with no separate branch
// to keep in sync. `git ls-files --others --ignored --exclude-standard` reports individual FILES
// (not a collapsed directory entry the way `git status --ignored=matching` does for a wholly
// ignored directory), which is what a content-hash diff needs.
const WATCHED_IGNORED_ROOTS = ["Claude outputs", ".env*"];
function watchedIgnoredPaths(repo) {
  const raw = execFileSync("git", ["ls-files", "-z", "--others", "--ignored", "--exclude-standard", "--", ...WATCHED_IGNORED_ROOTS], { cwd: repo, encoding: "utf8", maxBuffer: GIT_MAX_BUFFER });
  return raw.split("\0").filter((p) => p.length > 0 && !isAuditPath(p));
}
// The set this tool actually audits: everything `git status` sees as dirty, PLUS any watched
// ignored path present on disk. Both snapshot and check must walk the identical union or the
// comparison between them is meaningless.
function allAuditedPaths(repo) {
  const st = gitStatusPaths(repo);
  return { paths: [...new Set([...st.paths, ...watchedIgnoredPaths(repo)])], removed: st.removed };
}
// A removal git reports AND that is still absent on disk. Recorded in the snapshot as `deleted`, a
// field of its own rather than a sentinel inside `hashes` (validate-room-report.mjs reads `hashes`).
function absentRemovals(repo, removed) {
  return [...removed].filter((p) => !existsSync(join(repo, p))).sort();
}

// [single-hasher] (docs/INVARIANTS.md, tests/invariants/single-hasher.test.ts): SHA-256 may
// only be computed inside packages/canonical; this audit tool hashes FILES for integrity (never
// a subject/URL) and shells out to a system sha256 hasher instead of node:crypto, the same shape
// as scripts/close-run.mjs's hashFile() / scripts/intake/w2-sync.mjs's own exemption reasoning.
//
// Batches every file into ONE process invocation per snapshot/check — `xargs -0 sha256sum -z`
// (NUL-delimited in and out, safe for exotic filenames) — instead of spawning a hasher per file;
// xargs itself re-chunks the argument list across multiple sha256sum invocations if it would
// exceed ARG_MAX, so this scales to a long dirty-file list without this script doing that work.
// Falls back to `xargs -0 shasum -a 256` (no -z; BSD/macOS shasum has no NUL-output mode, so its
// output is parsed on newlines instead) when sha256sum is not on PATH, and fails naming both
// hashers if neither is (fail-closed, matches close-run.mjs's hashFile()).
function isCommandNotFound(e) {
  return e?.code === "ENOENT" || e?.status === 127;
}

function parseHashRecords(out, sep) {
  const hashes = {};
  for (const rec of out.split(sep)) {
    if (!rec) continue;
    const m = rec.match(/^([0-9a-fA-F]{64})\s{2}(.*)$/s);
    if (m) hashes[m[2]] = m[1].toLowerCase();
  }
  return hashes;
}

function hashFilesBatch(repo, fullPaths) {
  if (fullPaths.length === 0) return {};
  const input = fullPaths.map((p) => p + "\0").join("");
  try {
    const out = execFileSync("xargs", ["-0", "sha256sum", "-z"], { cwd: repo, input, encoding: "utf8", maxBuffer: 64 * 1024 * 1024 });
    return parseHashRecords(out, "\0");
  } catch (e) {
    if (!isCommandNotFound(e)) throw e;
    try {
      const out = execFileSync("xargs", ["-0", "shasum", "-a", "256"], { cwd: repo, input, encoding: "utf8", maxBuffer: 64 * 1024 * 1024 });
      return parseHashRecords(out, "\n");
    } catch (e2) {
      if (!isCommandNotFound(e2)) throw e2;
      console.error(`refused: no sha256 hasher on PATH (tried "sha256sum" and "shasum -a 256")`);
      process.exit(1);
    }
  }
}

function hashListedPaths(repo, paths) {
  const existing = paths.filter((p) => {
    const full = join(repo, p);
    return existsSync(full) && statSync(full).isFile();
  });
  const fullPaths = existing.map((p) => join(repo, p));
  const byFull = hashFilesBatch(repo, fullPaths);
  const hashes = {};
  for (const p of existing) {
    const full = join(repo, p);
    if (full in byFull) hashes[p] = byFull[full];
  }
  return hashes;
}

function isAllowed(path, allowDirs) {
  return allowDirs.some((d) => {
    const norm = d.replace(/\/+$/, "");
    return path === norm || path.startsWith(norm + "/");
  });
}

// FX-WRITE-AUDIT. A GRANT is one of two shapes, and the shape is read off the token, not declared:
//   a PATH  (contains "/")  -- exact match or directory prefix, the same rule `isAllowed` uses.
//   a NAME  (no "/")        -- a bare basename. Rule 33 orders a room to copy its harness INTO
//                              the tree, and the tree path is chosen by the suite layout, not by
//                              the room, so that grant cannot be written as a path. A name grant
//                              admits exactly the one file the room itself wrote into its own
//                              report dir's harness/, wherever rule 33 put it -- narrower than any
//                              directory grant, and it is why W3-12-R3's own harness stopped being
//                              charged to W3-12-R3.
// SIDECAR: `<granted path>.<seg>` also matches, up to THREE strips. D-0631 REQUIRES the mutate/
// restore idiom `cp <f> <f>.bak`, so the contract mandates creating a file the audit would
// otherwise charge -- FAMILY-022's trap verbatim (a check that rejects the form another rule
// demands). Only resolves when the STRIPPED path is itself a real grant: `secrets.txt.stolen`
// strips to `secrets.txt`, which nobody owns, and stays a violation.
function matchGrant(path, grants) {
  for (const g0 of grants) {
    const g = String(g0).replace(/\/+$/, "");
    if (!g) continue;
    if (!g.includes("/")) { if (basename(path) === g) return g; continue; }
    if (path === g || path.startsWith(g + "/")) return g;
    let p = path;
    for (let i = 0; i < 3; i++) {
      const cut = p.lastIndexOf(".");
      if (cut <= 0) break;
      p = p.slice(0, cut);
      if (p === g) return g;
    }
  }
  return null;
}

// P2-V2 (REG Phase 2): a room can hide an out-of-scope write by COMMITTING it (git status shows
// nothing dirty once committed) or by `git update-index --skip-worktree`/`--assume-unchanged`
// (git status treats such a path as clean even while its content differs from the index/HEAD).
// Both defeat a dirty-tree-only audit. Record HEAD and the skip-worktree/assume-unchanged path
// set at snapshot time; compare both at check time.
function gitHead(repo) {
  return execFileSync("git", ["rev-parse", "HEAD"], { cwd: repo, encoding: "utf8" }).trim();
}

// `git ls-files -v`: each line is `<tag><SP><path>`. An uppercase tag (of the letters that have
// an upper/lower pair — notably `S`) marks skip-worktree; a lowercase tag marks assume-unchanged.
// Ordinary tracked files are tagged `H` (or `M`/`R`/`C`/`K` for other normal states) and are not
// flagged. `-v` alone (no `-z`) is used; embedded-NUL filenames are already out of scope for the
// rest of this tool (hashFilesBatch/gitStatusPaths use -z, but ls-files' tag-prefixed format has
// no -z NUL-per-record mode that keeps the tag machine-parseable), so this matches that tool's
// existing filename-encoding assumptions rather than introducing a stricter one just here.
function gitLsFilesFlags(repo) {
  const raw = execFileSync("git", ["ls-files", "-v"], { cwd: repo, encoding: "utf8", maxBuffer: GIT_MAX_BUFFER });
  const flagged = [];
  for (const line of raw.split("\n")) {
    if (!line) continue;
    const tag = line[0];
    const path = line.slice(2);
    // uppercase "S" = skip-worktree; any lowercase tag (h/m/r/c/k/?) = assume-unchanged variant
    // of that base state. Both defeat `git status`'s dirty-tree view and must be flagged.
    if (!isAuditPath(path) && (tag === "S" || /[a-z]/.test(tag))) flagged.push(path);
  }
  return flagged.sort();
}

// #302b (S-081 judge JN-6): a room can defeat the whole check by deleting its own snapshot file
// after the fact -- `check` then reports "config error: no snapshot" (exit 2), which reads as an
// environment problem, not the violation it actually is. Record every snapshot ever taken for a
// label, append-only, in `<repo>/.scratch/audit/INDEX.jsonl` (one JSON object per line: label,
// the snapshot file's own sha256, and when it was taken) -- appendFileSync only ever adds a line,
// never rewrites one, so deleting the snapshot cannot also erase its entry from the index. `check`
// can then tell "never snapshotted" (no index entry either -- a real config error) apart from
// "snapshotted, and the index proves it, but the file is gone now" (a violation).
const INDEX_FILE = join(AUDIT_DIR, "INDEX.jsonl");
function sha256OfFile(repo, fullPath) {
  return hashFilesBatch(repo, [fullPath])[fullPath];
}
function appendIndexEntry(entry) {
  mkdirSync(AUDIT_DIR, { recursive: true });
  appendFileSync(INDEX_FILE, JSON.stringify(entry) + "\n");
}
// Malformed/truncated lines (e.g. a crash mid-append) are skipped rather than failing the whole
// read -- an index is a best-effort trail, not itself something to crash the audit over.
function readIndexEntriesForLabel(label) {
  if (!existsSync(INDEX_FILE)) return [];
  const out = [];
  for (const line of readFileSync(INDEX_FILE, "utf8").split("\n")) {
    if (!line.trim()) continue;
    try {
      const entry = JSON.parse(line);
      if (entry && entry.label === label) out.push(entry);
    } catch { /* skip a malformed line */ }
  }
  return out;
}

// B23-D9, QUEUE #608 follow-on: the orchestrator's OWN deliberate writes (the eight stray backups it
// deleted on Jason's instruction) were charged to every room whose window they fell in. A RECORD, not
// a skip: one JSONL object per path, `{path, state, actor, ref}`, with `state` either "deleted" or
// "sha256:<64 hex>". It exempts that EXACT path only while the path's CURRENT state equals the
// recorded one — so a different file created at an exempted deleted path is still a violation. Read
// from `<repo>/org/08-control/audit-exemptions.jsonl` when present, plus every `--exemptions <file>`.
// A malformed record is a usage error (exit 2), never skipped: a skipped record reads as a violation,
// and a lenient parse is how a typo becomes a blanket grant.
const DEFAULT_EXEMPTIONS = "org/08-control/audit-exemptions.jsonl";
let EXEMPT_LIST = [];
// I46-FXR-TOOLS (D-1307 F10; I46-RG-TOOLS TO-C48, TO-C61, NEW-07): the `append` helper's records live in this AUDITED file,
// never in the audit dir (which the audit cannot see), and check no longer reads `<audit-dir>/EXEMPTIONS.jsonl`. An
// --exemptions file inside the audit dir is refused for the same reason.
const APPEND_EXEMPTIONS = "org/08-control/audit-append-exemptions.jsonl";
function loadExemptions() {
  const files = [...(existsSync(join(REPO, DEFAULT_EXEMPTIONS)) ? [join(REPO, DEFAULT_EXEMPTIONS)] : []), ...(FLAGS.exemptions ?? []).map((f) => resolve(f))];
  const out = [];
  for (const f of files) {
    if (!existsSync(f)) usage(`check: --exemptions ${f} does not exist`);
    const rf = realOrTail(f);
    if (rf === AUDIT_DIR || isUnder(rf, AUDIT_DIR)) usage(`check: --exemptions ${f} lies inside the audit dir ${AUDIT_DIR}, which the audit does not see, so a record there is not evidence of anything`);
    readFileSync(f, "utf8").split("\n").forEach((line, i) => {
      const t = line.trim();
      if (!t || t.startsWith("#")) return;
      let r;
      try { r = JSON.parse(t); } catch (e) { usage(`check: ${f}:${i + 1} is not JSON (${e.message})`); }
      const ok = r && typeof r.path === "string" && r.path && !r.path.startsWith("/") && !/[*?[]/.test(r.path) && !/(^|\/)\.\.(\/|$)/.test(r.path)
        && (r.state === "deleted" || /^sha256:[0-9a-f]{64}$/.test(String(r.state))) && typeof r.actor === "string" && r.actor && typeof r.ref === "string" && r.ref;
      if (!ok) usage(`check: ${f}:${i + 1} is not an exemption record {path (exact, tree-relative, no glob), state ("deleted" | "sha256:<hex>"), actor, ref}`);
      out.push({ ...r, source: `${f}:${i + 1}` });
    });
  }
  EXEMPT_LIST = out;
  return out;
}
// I46-FXR-TOOLS (D-1307 F10): an append record is {path, before, appended, appended_bytes, actor, ref}, exactly those keys:
// the sha256 of the file before the append, of the appended bytes, and their count.
const APPEND_KEYS = ["actor", "appended", "appended_bytes", "before", "path", "ref"];
function isAppendRecord(r) {
  return !!r && Object.keys(r).sort().join(",") === APPEND_KEYS.join(",")
    && typeof r.path === "string" && r.path !== "" && r.path !== APPEND_EXEMPTIONS && !r.path.startsWith("/") && !/[*?[]/.test(r.path) && !/(^|\/)\.\.(\/|$)/.test(r.path)
    && /^sha256:[0-9a-f]{64}$/.test(String(r.before)) && /^sha256:[0-9a-f]{64}$/.test(String(r.appended))
    && Number.isInteger(r.appended_bytes) && r.appended_bytes > 0 && typeof r.actor === "string" && r.actor !== "" && typeof r.ref === "string" && r.ref !== "";
}
// sha256 of a byte buffer, through the same hashers hashFilesBatch uses (no in-process digest: single-hasher invariant).
function sha256OfBytes(buf) {
  for (const [bin, args] of [["sha256sum", []], ["shasum", ["-a", "256"]]]) {
    try { return execFileSync(bin, args, { input: buf, encoding: "utf8" }).slice(0, 64); } catch (e) { if (!isCommandNotFound(e)) throw e; }
  }
  console.error(`refused: no sha256 hasher on PATH (tried "sha256sum" and "shasum -a 256")`);
  process.exit(1);
}
// The sha256 (hex) a path held at snapshot time, or null when it held nothing: the snapshot's own hash for a path dirty
// then, nothing for a path deleted then, else the blob at the snapshot's HEAD (absent there: nothing).
let BASELINE = null;
function baselineSha(p) {
  if (p in BASELINE.hashes) return BASELINE.hashes[p];
  if (BASELINE.deletedSet.has(p)) return null;
  try { return sha256OfBytes(execFileSync("git", ["show", `${BASELINE.head}:${p}`], { cwd: REPO, maxBuffer: GIT_MAX_BUFFER, stdio: ["ignore", "pipe", "ignore"] })); } catch { return null; }
}
// The append records written in THIS window: the whole lines added to APPEND_EXEMPTIONS since the snapshot, each one a
// well-formed append record (last line first). Any other change to that file (an edited, removed or malformed line) trusts none.
let WINDOW_APPENDS = null;
function windowAppendRecords() {
  if (WINDOW_APPENDS) return WINDOW_APPENDS;
  WINDOW_APPENDS = [];
  const full = join(REPO, APPEND_EXEMPTIONS);
  if (!existsSync(full) || !statSync(full).isFile()) return WINDOW_APPENDS;
  const base = baselineSha(APPEND_EXEMPTIONS);
  const buf = readFileSync(full);
  const found = [];
  let end = buf.length;
  for (;;) {
    if (base === null ? end === 0 : sha256OfBytes(buf.subarray(0, end)) === base) break;
    if (buf[end - 1] !== 0x0a) return WINDOW_APPENDS;
    const start = buf.subarray(0, end - 1).lastIndexOf(0x0a) + 1;
    let r = null;
    try { r = JSON.parse(buf.subarray(start, end - 1).toString("utf8")); } catch { /* not JSON, so not a record: refused below */ }
    if (!isAppendRecord(r)) return WINDOW_APPENDS;
    found.push({ ...r, source: `${APPEND_EXEMPTIONS}:${buf.subarray(0, start).toString("utf8").split("\n").length}` });
    end = start;
  }
  WINDOW_APPENDS = found;
  return WINDOW_APPENDS;
}
// The window's records that rebuild `p` exactly: its current bytes are its snapshot bytes plus each record's appended bytes,
// in order, each record's `before` the sha256 of what preceded it. Null when no such chain exists (a room's own edit of the
// file before, between or after the appends breaks it, and the path is charged as before).
function appendChain(p) {
  const full = join(REPO, p);
  if (!existsSync(full) || !statSync(full).isFile()) return null;
  const base = baselineSha(p);
  const mine = windowAppendRecords().filter((r) => r.path === p);
  const used = [];
  let cur = readFileSync(full);
  while (sha256OfBytes(cur) !== base) {
    const r = mine.find((x) => `sha256:${sha256OfBytes(cur.subarray(cur.length - x.appended_bytes))}` === x.appended
      && `sha256:${sha256OfBytes(cur.subarray(0, cur.length - x.appended_bytes))}` === x.before);
    if (!r) return null;
    used.push(r);
    cur = cur.subarray(0, cur.length - r.appended_bytes);
  }
  return used.length ? used : null;
}
// True (and recorded) when a would-be violation at `p` matches a record by path AND current state, or is exactly the
// window's recorded appends to its snapshot bytes; APPEND_EXEMPTIONS itself only when every record it gained is so used.
function exemptBy(p) {
  const full = join(REPO, p);
  const present = existsSync(full);
  for (const r of EXEMPT_LIST) {
    if (r.path !== p) continue;
    const match = r.state === "deleted" ? !present : present && statSync(full).isFile() && `sha256:${sha256OfFile(REPO, full)}` === r.state;
    if (match) { exemptedSink.push(`${p} — ${r.state === "deleted" ? "deleted" : "content"} as recorded by ${r.actor} (${r.ref}; ${r.source})`); return true; }
  }
  if (p === APPEND_EXEMPTIONS) {
    const w = windowAppendRecords();
    if (w.length === 0) return false;
    const used = new Set(w.flatMap((r) => appendChain(r.path)));
    if (!w.every((r) => used.has(r))) return false;
    exemptedSink.push(`${p} — ${w.length} append record(s) added in the window, each matched by the append it records`);
    return true;
  }
  const chain = appendChain(p);
  if (!chain) return false;
  for (const r of chain.slice().reverse()) exemptedSink.push(`${p} — ${r.appended_bytes} appended byte(s) as recorded by ${r.actor} (${r.ref}; ${r.source})`);
  return true;
}

  // be EMPTY passes trivially). A room that wrote NOTHING produces an empty dirty set, walks every

  // is an audit with nothing in it, and the two must not print the same word.
  if (OWNS.length) {

    console.log(`OWN-WRITES: ${k}/${OWNS.length} — ${k === 0

      : `of this room's own grant(s) {${OWNS.join(", ")}} were written since snapshot '${label}' — the check ran against a non-empty own-write set`}`);
  } else {
    console.log(`OWN-WRITES: none — NO GRANT RESOLVED for '${label}'. Every dirty path is therefore judged against the org allow-list alone, which cannot express a builder's ownership; a PASS here is weaker than a PASS with a grant.`);
  }
  if (violations.length) {
    console.error(`VIOLATIONS (${violations.length}):`);
    for (const v of violations) console.error(" - " + v);
    process.exit(1);
  }
  console.log(`OK: check '${label}' against grant {${OWNS.join(", ") || "none"}} + {${allowDirs.join(", ")}} — no writes outside this room's ownership`);
  process.exit(0);

/**
 * Runtime hasher tripwire — second line of defence for INVARIANT [single-hasher].
 *
 * `tests/invariants/single-hasher.test.ts` reads source and matches patterns.
 * Pattern matching is a losing game against escaping: `"\x6eode:crypto"`,
 * `globalThis["cry" + "pto"].subtle["dig" + "est"]`, a specifier assembled at
 * run time. The scanner normalises the obvious encodings, but "normalise harder"
 * is an arms race with a fixed loser.
 *
 * This layer does not read code — it watches the primitive. During the protected
 * run every SHA-256 entry point is wrapped and throws unless the call comes from
 * inside `packages/canonical`. However the specifier was written, when the call
 * EXECUTES it arrives here.
 *
 * WHAT EACH LAYER CATCHES, plainly:
 *   scanner   — any second hasher present in source, executed or not; the only
 *               layer that can see dead code, unreachable branches, and files
 *               the protected suites never import. Blind to escaping it does
 *               not normalise.
 *   tripwire  — any second hasher that actually runs during the protected
 *               suites, however it was spelled. Blind to code the protected
 *               suites never execute.
 * Neither subsumes the other; the pair is the point.
 *
 * COVERAGE LIMITS, stated rather than implied:
 *   - `import { createHash } from "node:crypto"` (a NAMED ESM import of a Node
 *     builtin) binds to a snapshot of the module's exports taken when Node built
 *     the builtin's ESM facade. Reassigning `createHash` on the CJS exports
 *     object afterwards cannot rebind it. The default-import, `require()`, and
 *     `crypto.createHash(...)` property-access forms ARE caught. The named-import
 *     form is caught by the source scanner instead — which is exactly why both
 *     layers exist.
 *   - There is no blanket exemption for `node_modules`. A prior version of this
 *     file admitted any caller frame under that directory segment; the
 *     population of such a blanket is the installed dependency tree, which
 *     this repo does not author and which changes on every install, so it
 *     cannot be enumerated by file (docket #37 (e), W0 AMENDMENTS). Measured
 *     over the protected suite the blanket admitted zero frames — it was not
 *     load-bearing — so it was DELETED rather than enumerated. A caller frame
 *     from inside `node_modules` is judged the same as any other caller: by
 *     ALLOWED_ORIGINS alone.
 *   - Code the protected suites never execute is invisible here, by definition.
 *
 * ABSENT PRIMITIVE: if `SubtleCrypto` or `node:crypto` is unavailable in this
 * runtime, the corresponding wrapper is skipped and the source scanner still
 * runs. This file never silently disables the guard it belongs to — it only
 * ever adds a way to fail.
 */

// `vitest run --no-isolate`, or any runner that evaluates setup files more than
// once per realm — re-wrapping would nest wrappers one layer deeper per test
// file, and the nested frames crowd out the caller. Wrapping is therefore
// IDEMPOTENT: install once per realm, and never wrap a wrapper.
const WRAPPED = Symbol.for("dhwaja.invariant.hashWrapped");

// wrapper frames plus a realistic async boundary, cheap enough to pay on every
// hash. Only raised for the duration of one stack capture.

/**

 * still counts toward `minTests` and toward its per-file expected count. An

 * pad the numbers while enforcing nothing. After every test in the PROTECTED

/** Origin predicate for an HMAC-keyed `subtle.sign`: a KEYED_MAC_ORIGINS file or a test tree, under ROOT (or the real checkout root). */
function keyedMacOriginAllowed(line) {

    const at = line.indexOf(root);
    if (at === -1) return false;
    const rel = line.slice(at + root.length);
    return KEYED_MAC_ORIGINS.some((o) => rel === o || rel.startsWith(`${o}:`) || rel.startsWith(`${o}?`)) || keyedMacTestTree(rel);
  });
}

const hashOriginAllowed = (line) => ALLOWED_ORIGINS.some((origin) => line.includes(origin));

/** Decide from ONE captured stack. Returns null when no caller frame was
 * visible — the stack was all our own wrapper frames, or it was truncated. */
function decideFrom(stack, allowed) {
  if (typeof stack !== "string") return null;
  for (const line of stack.split("\n").slice(1)) {
    // Skip our own wrapper frames; the first frame below them is the caller.
    if (line.includes(SELF)) continue;
    if (!line.includes("/") && !line.includes("\\")) continue;
    return allowed(line);
  }
  return null;
}

// I20-FX2 (JN-B03 W1F-invariant-setup-stack-cost): the stack is formatted by
// V8's own CallSite#toString — the text V8 prints for each frame — under a
// prepareStackTrace installed for this one capture, so a host formatter
// (vitest's source-map rewrite, measured at 8.8 s of 11.1 s on
// scheduled-integrity) is never invoked. Same frames, same paths, same order.
const v8Format = (err, frames) => `${err}\n${frames.map((f) => `    at ${f}`).join("\n")}`;
function rawStack() {
  const prior = Error.prepareStackTrace;
  Error.prepareStackTrace = v8Format;
  try {
    return new Error().stack;
  } finally {
    Error.prepareStackTrace = prior;
  }
}

function callerIsAllowed(allowed = hashOriginAllowed) {
  // FAST PATH: capture at whatever depth the runtime already uses (Node's
  // default is 10). This runs on every hash in the suite, and a deep capture
  // here costs real wall-clock — v8 materialises every frame.
  const decided = decideFrom(rawStack(), allowed);
  if (decided !== null) return decided;
  // UNDECIDED: every visible frame was our own, or the stack was truncated.
  // Before this module became idempotent, a shared global (`vitest run
  // --no-isolate`) wrapped the wrapper once per test file, and ten nested
  // wrapper frames crowded the caller off the end of the stack — so every
  // LEGITIMATE canonical hash failed closed. Retry once, deeper, rather than
  // refuse on a truncation artefact. Still fails closed if nothing is visible.
  const priorLimit = Error.stackTraceLimit;
  Error.stackTraceLimit = STACK_FRAMES_NEEDED;
  const retried = decideFrom(rawStack(), allowed);
  Error.stackTraceLimit = priorLimit;
  return retried === null ? false : retried;
}

function refuse(what) {
  if (/^crypto\.subtle\.\w+\((HMAC|HKDF|PBKDF2)\)$/.test(what)) {
    return new Error(
      `INVARIANT [single-hasher] violated at run time: ${what} was called from outside ` +
        `KEYED_MAC_ORIGINS and the test trees. A keyed MAC is legal only at a named site that ` +
        `meets D-0695 (keyed with a secret not derivable from the hashed content, and scoped per ` +
        `relying party so outputs cannot be joined across contexts) — add it to KEYED_MAC_ORIGINS ` +
        `and KEYED_MAC_EXEMPTIONS together, or hash through @dhwaja/canonical.`,
    );
  }
  return new Error(
    `INVARIANT [single-hasher] violated at run time: ${what} was called from outside ` +
      `packages/canonical. SHA-256 must come from @dhwaja/canonical (sha256Hex / ` +
      `subjectHash / sha256Bytes). A second hasher produces a second subject hash ` +
      `for the same URL, which breaks every verdict lookup and the ledger preimage.`,
  );
}

// --- webcrypto: the primitive @dhwaja/canonical actually uses, and the only
// --- one available on Workers. This is the high-value wrapper.
const SubtleCryptoCtor = globalThis.SubtleCrypto;
if (typeof SubtleCryptoCtor === "function" && SubtleCryptoCtor.prototype) {
  const original = SubtleCryptoCtor.prototype.digest;
  if (typeof original === "function" && !original[WRAPPED]) {
    Object.defineProperty(SubtleCryptoCtor.prototype, "digest", {
      configurable: true,
      writable: true,
      value: Object.defineProperty(
        function digest(...args) {
          if (!callerIsAllowed()) throw refuse("crypto.subtle.digest");
          return original.apply(this, args);
        },
        WRAPPED,
        { value: true },
      ),
    });
  }
  // KEYED MAC (QUEUE #189, D-0695): `subtle.sign` with HMAC is keyed SHA-256.
  // Refused unless the caller is a KEYED_MAC_ORIGINS file or a test tree. HMAC
  // is recognised from the algorithm, snapshotted so the name checked is the
  // name WebCrypto dispatches on (see snapshotAlgorithm, isHmacSign). ECDSA /
  // RSASSA / Ed25519 signatures are not hashers and pass through untouched.
  const originalSign = SubtleCryptoCtor.prototype.sign;
  if (typeof originalSign === "function" && !originalSign[WRAPPED]) {
    Object.defineProperty(SubtleCryptoCtor.prototype, "sign", {
      configurable: true,
      writable: true,
      value: Object.defineProperty(
        function sign(algorithm, ...rest) {
          const alg = snapshotAlgorithm(algorithm);
          if (isHmacSign(alg) && !callerIsAllowed(keyedMacOriginAllowed)) {
            throw refuse("crypto.subtle.sign(HMAC)");
          }
          return originalSign.call(this, alg, ...rest);
        },
        WRAPPED,
        { value: true },
      ),
    });
  }
  // KEYED DERIVATION (QUEUE #189, I5-RG-PRIVACY I5RG-PRIV-04): HKDF-SHA256 and
  // PBKDF2-HMAC-SHA256 hand back keyed-SHA-256 bytes without ever calling
  // `sign` — deriveBits directly, deriveKey through an extractable key. Same
  // origin rule and the same snapshot as `sign`; ECDH / X25519 derivation is
  // not a hasher and passes untouched.
  for (const op of ["deriveBits", "deriveKey"]) {
    const originalDerive = SubtleCryptoCtor.prototype[op];
    if (typeof originalDerive !== "function" || originalDerive[WRAPPED]) continue;
    Object.defineProperty(SubtleCryptoCtor.prototype, op, {
      configurable: true,
      writable: true,
      value: Object.defineProperty(
        {
          [op](algorithm, ...rest) {
            const alg = snapshotAlgorithm(algorithm);
            const kdf = keyedKdfName(alg);
            if (kdf !== null && !callerIsAllowed(keyedMacOriginAllowed)) {
              throw refuse(`crypto.subtle.${op}(${kdf})`);
            }
            return originalDerive.call(this, alg, ...rest);
          },
        }[op],
        WRAPPED,
        { value: true },
      ),
    });
  }
}

/**
 * The algorithm argument with its `name` read ONCE and fixed. WebCrypto
 * dispatches on the name IT reads, and a getter can answer "ECDSA" to this
 * wrapper and "HMAC" to WebCrypto; passing the snapshot on means the operation
 * performed is the one that was checked. Other members stay live through the
 * prototype link. A non-object (a string, or garbage WebCrypto will reject) is
 * returned as is.
 */
function snapshotAlgorithm(algorithm) {
  if (algorithm === null || typeof algorithm !== "object") return algorithm;
  const raw = algorithm.name;
  const name = typeof raw === "string" || raw === undefined || raw === null ? raw : String(raw);
  return Object.create(algorithm, { name: { value: name, enumerable: true } });
}

/**
 * True when a (snapshot) algorithm is HMAC, in any case. The snapshot is what
 * makes this sufficient: WebCrypto dispatches on the name this read fixed, and
 * refuses a key whose own algorithm name differs, so a call this calls
 * non-HMAC cannot produce an HMAC tag. Never throws.
 */
function isHmacSign(algorithm) {
  const name = algorithm !== null && typeof algorithm === "object" ? algorithm.name : algorithm;
  return typeof name === "string" && name.toUpperCase() === "HMAC";
}

/** "HKDF" / "PBKDF2" when a (snapshot) algorithm names a keyed KDF, in any case; else null. Never throws. */
function keyedKdfName(algorithm) {
  const name = algorithm !== null && typeof algorithm === "object" ? algorithm.name : algorithm;
  if (typeof name !== "string") return null;
  const upper = name.toUpperCase();
  return upper === "HKDF" || upper === "PBKDF2" ? upper : null;
}

/** The finding a test engineer may see: FINDING_FIELDS only, fix-artifact tokens withheld. Pure. */

/**
 * The `## Items` entries of a scope: each `- ` or `N. ` line at column 0 plus its indented continuations, one string
 * each. A scope with no `## Items` (the I10 shape, B9) is read from its `## Rows assigned to this room` section, whose
 * entries are `#<n>: ` lines at column 0.
 */
export function parseItems(text) {
  const lines = String(text).split(/\r?\n/);
  const rowsForm = !lines.some((l) => /^## Items\s*$/.test(l));
  const items = [];
  let inItems = false;
  let cur = null;
  for (const l of lines) {
    if (l.startsWith("## ")) { inItems = /^## Items\s*$/.test(l) || (rowsForm && /^## Rows assigned to this room\s*$/.test(l)); cur = null; continue; }
    if (!inItems) continue;
    const m = (rowsForm ? /^(#\d+[a-z]?: .*)$/ : /^(?:- |\d+\. )(.*)$/).exec(l);
    if (m) { cur = [m[1].trim()]; items.push(cur); }
    else if (cur && /^\s+\S/.test(l)) cur.push(l.trim());
    else cur = null;
  }
  return items.map((p) => p.join(" "));
}

  const ord = rowsForm ? null : /(\d+)$/.exec(String(b.id ?? ""));

/** `--rule <judge room>:<line>=<text>` values -> Map("room:line" -> text); each must name a --rows row, once, on one line. */
export function parseRules(rules, refs) {
  const out = new Map();
  for (const r of rules ?? []) {
    const m = /^([^:=]+):(\d+)=([\s\S]*)$/.exec(r);
    if (!m) fail(`--rule ${JSON.stringify(r)} is not <judge room>:<line>=<text>`);
    const key = `${m[1]}:${Number(m[2])}`;
    if (!refs.some((x) => `${x.room}:${x.line}` === key)) fail(`--rule ${key} names a row --rows does not`);
    if (out.has(key)) fail(`--rule ${key} is given twice`);
    if (!m[3].trim()) fail(`--rule ${key} has no text`);
    if (/[\r\n]/.test(m[3])) fail(`--rule ${key} text spans lines (it would open lines in the scope)`);
    out.set(key, m[3].trim());
  }
  return out;
}

/** --ref resolves to a commit of `tree` (git rev-parse) that is not the tree's HEAD (the candidate). Returns the sha. */
export function checkRef(tree, ref) {
  const rev = (r) => { try { return execFileSync("git", ["--no-optional-locks", "-C", tree, "rev-parse", "--verify", "--quiet", `${r}^{commit}`], { encoding: "utf8", stdio: ["ignore", "pipe", "ignore"] }).trim(); } catch { return null; } };
  const sha = rev(ref);
  if (!sha) fail(`--ref ${JSON.stringify(ref)} does not resolve to a commit in ${tree} (git rev-parse --verify ${ref}^{commit})`);
  if (sha === rev("HEAD")) fail(`--ref ${ref} is ${sha.slice(0, 12)}, the candidate HEAD of ${tree}: a DRAFT runs at the pre-fix ref`);
  return sha;
}

/** INV ids named by a finding: its `invariant` field and any INV-NNN in its carried text. */
export function invIdsOf(f) {
  const ids = new Set();
  for (const v of Object.values(f)) for (const m of String(v).matchAll(/\bINV-\d{3}\b/g)) ids.add(m[0]);
  return [...ids];
}

/** Every id has a `| INV-NNN |` row in <root>/docs/INVARIANTS.md; throws naming the ones that do not. */
export function checkInvIds(root, ids) {
  const bad = ids.filter((id) => !INV_RE.test(id));
  if (bad.length) fail(`not an invariant id (INV-NNN): ${bad.join(", ")}`);
  if (!ids.length) return;
  const reg = join(root, "docs/INVARIANTS.md");
  let lines;
  try { lines = readFileSync(reg, "utf8").split("\n"); } catch (e) { fail(`docs/INVARIANTS.md unreadable at ${reg} (${e.message}) — no id can be checked`); }
  const unknown = ids.filter((id) => !lines.some((l) => l.startsWith(`| ${id} |`)));
  if (unknown.length) fail(`unknown invariant id(s), no row in ${reg}: ${unknown.join(", ")}`);
}

function testPaths(names, id, given) {

/** Every CONFIRMED row with status fixed in <root>/org/07-findings/<batch>/*\/FINDINGS.jsonl. */
const isFile = (p) => { try { return statSync(p).isFile(); } catch { return false; } };
/** Per id: the backticked tokens of its `| INV-NNN |` row in <root>/docs/INVARIANTS.md (`:line` suffix cut) that are files under root. */
export function invFiles(root, ids) {
  const lines = readFileSync(join(root, "docs/INVARIANTS.md"), "utf8").split("\n");
  return ids.map((id) => {
    const row = lines.find((l) => l.startsWith(`| ${id} |`)) ?? "";
    const named = [...new Set([...row.matchAll(/`([^`\s]+)`/g)].map((m) => m[1].replace(/:\d[\d,:-]*$/, "")))];
    const files = named.filter((p) => !p.startsWith("/") && !p.split("/").includes("..") && isFile(join(root, p)));
    return { id, files, unresolved: named.length - files.length };
  });
}

export function confirmedFixed(root, batch) {
  const dir = join(root, "org/07-findings", batch);
  if (!existsSync(dir)) fail(`${dir} does not exist`);
  const out = [];
  for (const room of readdirSync(dir).sort()) {
    const p = join(dir, room, "FINDINGS.jsonl");
    if (!existsSync(p)) continue;
    readFileSync(p, "utf8").split("\n").forEach((l, i) => {
      let r;
      try { r = JSON.parse(l); } catch { return; }
      if (r && r.verdict === "CONFIRMED" && r.status === "fixed") out.push({ ref: { room, line: i + 1 }, finding: findingOf(r).finding });
    });
  }
  return out;
}

function readIndexLabels(indexFile) {
  if (!existsSync(indexFile)) return new Set();
  const labels = new Set();
  for (const line of readFileSync(indexFile, "utf8").split("\n")) {
    if (!line.trim()) continue;
    try {
      const entry = JSON.parse(line);
      if (entry && entry.label) labels.add(entry.label);
    } catch { /* skip a malformed line */ }
  }
  return labels;
}

// I10-FX11 (QUEUE #836, RG-TOOLS1:9): runSnapshot's leftovers are the CLI's OWN output, byte-identical to the
// copy just moved to the audit root, so on the undeletable mount they are left AS WRITTEN. Emptying them made a
// 0-byte <label>.json in the code tree that the validator reports as an unreadable dispatch snapshot; this
// writes nothing. (runCheck stages nothing since I10-FX12, so its old removeOrEmpty -- a 0-byte file per check -- is gone.)
function removeOrKeep(file) {
  try {
    unlinkSync(file);
  } catch (e) {
    if (!(e && (e.code === "EPERM" || e.code === "EACCES" || e.code === "ENOENT"))) throw e;
  }
}

// QUEUE #310 JA-1 (S-088 JUDGE round 2): on the undeletable repo mount, the old removeOrEmpty left a

function existsNonEmpty(file) {
  try {
    return statSync(file).size > 0;
  } catch {
    return false;
  }

function spawn(auditScript, args) {
  try {
    const out = execFileSync(process.execPath, [auditScript, ...args], { encoding: "utf8", stdio: ["ignore", "pipe", "pipe"] });
    return { status: 0, out };
  } catch (e) {
    return { status: e.status ?? 1, out: `${e.stdout ?? ""}${e.stderr ?? ""}` };
  }
}

// QUEUE #389 test hook, ONLY: a synchronous delay inserted between staging the shared repo-local
// file and spawning the CLI that reads it, so a two-process race can be reproduced DETERMINISTICALLY
// (widen the window instead of hoping real OS scheduling loses the race) rather than depending on

// (validate-room-report.mjs, compile-room-prompt.mjs) are plain scripts, not async pipelines.
function maybeTestDelay() {
  const ms = Number(process.env.AUDIT_STORE_TEST_RACE_DELAY_MS || 0);
  if (ms > 0) Atomics.wait(new Int32Array(new SharedArrayBuffer(4)), 0, 0, ms);
}

const LOCK_POLL_MS = 20;
const LOCK_STALE_MS = 30_000; // a holder older than this, or whose pid is dead, is presumed crashed
const LOCK_TIMEOUT_MS = 20_000;

// QUEUE #836 (D-0871(4)): the lock lives in the audit root it serializes -- `<repoRoot>/.scratch/audit`,
// the staging dir room-write-audit.mjs hardcodes -- never in os.tmpdir(). Keyed in TMPDIR, two compiles
// with different TMPDIRs held two different locks over ONE staging dir and erased each other's INDEX
// lines (I10-RG-TOOLS1 race-run.sh: 15/15 lost). The mount refuses unlink (EPERM) but permits rename
// over an existing file, so a lock that cannot be deleted is renamed onto `<lock>.released`: the lock
// NAME is free again either way, which is what O_EXCL acquisition needs (the 2026-09-19 wedge was a
// truncated-but-present lock file).
function lockPath(repoRoot) {
  return join(repoRoot, ".scratch/audit", ".audit-store.lock");
}

function dropLock(lockFile) {
  try {
    unlinkSync(lockFile);
  } catch (e) {
    if (e && (e.code === "EPERM" || e.code === "EACCES")) renameSync(lockFile, `${lockFile}.released`);
    else if (!(e && e.code === "ENOENT")) throw e;
  }
}

function pidAlive(pid) {
  try {
    process.kill(pid, 0);
    return true;
  } catch {
    return false;
  }
}

function tryAcquireLock(lockFile) {
  try {
    const fd = openSync(lockFile, "wx"); // atomic exclusive create; EEXIST if someone else holds it
    writeFileSync(fd, JSON.stringify({ pid: process.pid, at: Date.now() }));
    closeSync(fd);
    return true;
  } catch (e) {
    if (e && e.code === "EEXIST") return false;
    // a borrowed runSnapshot removes the staging dir it created once it has released the lock; the
    // next poll re-creates it (rmdir of a dir holding a live lock fails, so a held lock is never lost)
    if (e && e.code === "ENOENT") { mkdirSync(dirname(lockFile), { recursive: true }); return false; }
    throw e;
  }
}

// Crash safety: a lock file is stolen (removed so the next tryAcquireLock can succeed) once its
// recorded holder is either dead (pid no longer exists) or older than LOCK_STALE_MS -- a process
// that dies mid-critical-section cannot wedge every future runSnapshot/runCheck against this repo.
function stealLockIfStale(lockFile) {
  let holder;
  try {
    holder = JSON.parse(readFileSync(lockFile, "utf8"));
  } catch {
    // MEASURED WEDGE (orchestrator, 2026-09-19): this branch used to `return true` WITHOUT
    // removing the file. "Safe to steal" is a verdict, not an act -- the caller's next
    // tryAcquireLock still hit EEXIST on the same file, every poll, until the 20 s timeout, and
    // then every later run did the same. A ZERO-BYTE lock (a process killed between create and
    // write -- which is what a timed-out test run leaves behind) wedged every compile against
    // this repo, i.e. exactly the failure this function exists to prevent, in the one branch
    // that was never executed on the way in. Steal means UNLINK.
    try { dropLock(lockFile); } catch { /* already gone, or the real owner released it */ }
    return true;
  }
  if (!holder || typeof holder.pid !== "number") return true;
  const age = Date.now() - (holder.at || 0);
  if (age > LOCK_STALE_MS || !pidAlive(holder.pid)) {
    try {
      dropLock(lockFile);
    } catch { /* raced with the real owner's own release -- fine */ }
    return true;
  }
  return false;
}

// QUEUE #389: serializes every runSnapshot/runCheck against the SAME repoRoot's shared
// `.scratch/audit` staging area (see the file-header note). room-write-audit.mjs cannot be told
// to look anywhere else, so the fix is not "a different path per process" for the file the CLI
// itself reads -- it is exclusive access to the stage -> spawn -> cleanup sequence that touches
// that fixed path, with a per-process lock file (pid + timestamp) as the mutex and staleness as
// the crash-recovery rule.
export function withAuditLock(repoRoot, fn) {
  mkdirSync(join(repoRoot, ".scratch/audit"), { recursive: true });
  const lockFile = lockPath(repoRoot);
  const deadline = Date.now() + LOCK_TIMEOUT_MS;
  for (;;) {
    if (tryAcquireLock(lockFile)) break;
    stealLockIfStale(lockFile);
    if (Date.now() > deadline) throw new Error(`audit-store: timed out waiting for lock at ${lockFile}`);
    Atomics.wait(new Int32Array(new SharedArrayBuffer(4)), 0, 0, LOCK_POLL_MS);
  }
  try {
    return fn();
  } finally {
    dropLock(lockFile);
  }
}

/** QUEUE #836: split an INDEX.jsonl text into the lines NEW since `prior` (multiset) whose `label` is
 *  `label` (`own`), and everything else in original order (`rest`). Both end in "\n" when non-empty. */
export function splitIndexForLabel(now, prior, label) {
  const lines = String(now).split("\n").filter((l) => l !== "");
  const before = new Map();
  for (const l of String(prior).split("\n")) if (l !== "") before.set(l, (before.get(l) ?? 0) + 1);
  const own = [];
  const rest = [];
  for (const l of lines) {
    const n = before.get(l) ?? 0;
    if (n > 0) { before.set(l, n - 1); rest.push(l); continue; }
    let lab = null;
    try { lab = JSON.parse(l)?.label ?? null; } catch { /* a malformed line is never ours */ }
    (lab === label ? own : rest).push(l);
  }
  const join_ = (a) => (a.length ? `${a.join("\n")}\n` : "");
  return { own: join_(own), rest: join_(rest) };
}

// Runs `room-write-audit.mjs snapshot <label> --repo <repoRoot>`, then relocates the JSON it
// wrote from <repoRoot>/.scratch/audit/<label>.json to <auditRoot>/<label>.json (a no-op when
// auditRoot already IS <repoRoot>/.scratch/audit). Never throws; a subprocess failure is
// returned in `status`/`out` for the caller to log.
export function runSnapshot({ auditScript, repoRoot, label, auditRoot }) {
  // QUEUE #836: when the destination is NOT the CLI's own staging dir, the staging dir is only
  // borrowed. What was there before (another label's snapshot, a persistent INDEX.jsonl when
  // repoRoot is the build tree) is put back byte-identical, only the CLI's NEW index lines move,
  // and a staging dir this call created is removed again.
  const stageDir = join(repoRoot, ".scratch/audit");
  const borrowed = resolve(stageDir) !== resolve(auditRoot);
  const hadScratch = existsSync(join(repoRoot, ".scratch"));
  const hadStage = existsSync(stageDir);
  const result = withAuditLock(repoRoot, () => {
    const src = repoAuditPath(repoRoot, label);
    const indexSrc = repoIndexPath(repoRoot);
    const priorLabel = borrowed && existsSync(src) ? readFileSync(src) : null;
    const priorIndex = borrowed && existsSync(indexSrc) ? readFileSync(indexSrc, "utf8") : null;
    // I10-FX12 (QUEUE #836): the CLI is told the audit root, so it writes the snapshot and its INDEX line
    // there directly and the staging dir is never written. The relocation below is a fallback for a CLI that
    // ignores --audit-dir (a stand-in in a test) and moves ONLY bytes that CLI wrote: a prior same-label file
    // left in the staging dir is never copied over the snapshot just written to the audit root.
    const result = spawn(auditScript, ["snapshot", label, "--repo", repoRoot, "--audit-dir", auditRoot]);
    if (result.status === 0 && borrowed) {
      const dst = join(auditRoot, `${label}.json`);
      if (existsSync(src) && (priorLabel === null || !readFileSync(src).equals(priorLabel))) {
        mkdirSync(dirname(dst), { recursive: true }); // #591b: a nested label's parent, not only auditRoot
        writeFileSync(dst, readFileSync(src));
        if (priorLabel !== null) writeFileSync(src, priorLabel);
        else removeOrKeep(src); // leave the mounted repo as clean as room-write-audit.mjs found it
      }
      // #302b: APPEND (never overwrite) the line(s) this snapshot added onto the shared, persistent
      // index. Only the NEW tail moves: the CLI appends, so the prior bytes are a prefix.
      const indexDst = join(auditRoot, "INDEX.jsonl");
      if (existsSync(indexSrc) && readFileSync(indexSrc, "utf8") !== (priorIndex ?? "")) {
        // QUEUE #836 (D-0871(4)): an --out compile touches only its OWN index line(s); every other
        // line -- including one a concurrent writer appended -- stays in the tree index.
        const { own, rest } = splitIndexForLabel(readFileSync(indexSrc, "utf8"), priorIndex ?? "", label);
        mkdirSync(auditRoot, { recursive: true });
        if (own) appendFileSync(indexDst, own);
        if (rest || priorIndex !== null) writeFileSync(indexSrc, rest);
        else removeOrKeep(indexSrc);
      }
    }
    return result;
  });
  if (borrowed && !hadStage) {
    try { rmdirSync(stageDir); } catch { /* not empty, or the undeletable mount */ }
    if (!hadScratch) try { rmdirSync(join(repoRoot, ".scratch")); } catch { /* same */ }
  }
  return result;
}

// Stages <auditRoot>/<label>.json at <repoRoot>/.scratch/audit/<label>.json (room-write-audit.mjs's
// hardcoded lookup location) before running `check`, then removes the staged copy afterward.
// `snapshotMissing` is true when no snapshot exists at either the audit root or the repo's own
// bookkeeping location — the caller (validate-room-report.mjs) treats that as a #113 violation.
export function runCheck({ auditScript, repoRoot, label, auditRoot, allowDirs, ownsGrants = [], attributedGrants = [] }) {
  return withAuditLock(repoRoot, () => {
    const src = join(auditRoot, `${label}.json`);
    const dst = repoAuditPath(repoRoot, label);
    const indexSrc = join(auditRoot, "INDEX.jsonl");
    const indexDst = repoIndexPath(repoRoot);
    // #302b: "never snapshotted" (a real config gap) is now judged against the PERSISTENT index too,
    // not just file presence -- a label the index has never heard of, with no file either, is the
    // only case that is still a config error; a label the index DOES know but whose file is gone is
    // a violation (room-write-audit.mjs's own check enforces that once it reads that index).
    const indexedForLabel = readIndexLabels(indexSrc).has(label);
    const snapshotMissing = !existsNonEmpty(src) && !existsNonEmpty(dst) && !indexedForLabel;
    // I10-FX12 (QUEUE #836): nothing is staged. The CLI is pointed (--audit-dir) at the dir that holds this
    // label -- the audit root when it has the file or its index knows the label, else the tree's own
    // .scratch/audit when that does (a snapshot taken by the CLI directly), else the audit root (the CLI then
    // reports the missing snapshot). Staging copied the snapshot into the tree and, on the mount that refuses
    // unlink, left a 0-byte <label>.json and INDEX.jsonl there after every check of an --out room.
    const repoKnows = existsNonEmpty(dst) || readIndexLabels(indexDst).has(label);
    const readDir = existsNonEmpty(src) || indexedForLabel || !repoKnows ? auditRoot : dirname(dst);
    // QUEUE #389 test hook: see maybeTestDelay's own comment. No-op in every real invocation.
    maybeTestDelay();
    // room-write-audit.mjs's --allow is ONE flag taking MULTIPLE values -- "--allow", ...allowDirs, then the
    // next "--" flag stops it. `--owns` and `--attributed` are variadic the same way, and each is omitted
    // entirely when its list is empty: a bare `--owns` with no values would collect the next flag's value.
    const ownsArgs = ownsGrants.length ? ["--owns", ...ownsGrants] : [];
    const attrArgs = attributedGrants.length ? ["--attributed", ...attributedGrants] : [];
    const result = spawn(auditScript, ["check", label, "--allow", ...allowDirs, ...ownsArgs, ...attrArgs, "--repo", repoRoot, "--audit-dir", readDir]);
    return { ...result, snapshotMissing };
  });
}

//   CENSUS: <label> | <regex> | <path> [<path> ...]
// makes scripts/compile-room-prompt.mjs enumerate every match into a `## CENSUS — <label>` block, so the
// room's population is a list it answers line by line instead of a lesson it is told to apply.
// Pure apart from reading the named files: (text) -> specs, (spec, root) -> matches, (result) -> block text.
// Every refusal is a CensusError with a plain message; the compiler prints it and writes nothing.
import { readdirSync, readFileSync, realpathSync, statSync } from "node:fs";

export class CensusError extends Error {

/** A directory census reads files with these extensions; a path named as a FILE is read whatever it is. */
export const SOURCE_EXTENSIONS = Object.freeze(["ts", "tsx", "mts", "cts", "js", "jsx", "mjs", "cjs", "sql", "kt", "kts", "swift", "java", "py", "sh", "html", "css"]);

export const CENSUS_LINE_CHARS = 160;
export const CENSUS_ANSWER_SENTENCE = "Answer every census line above in your FINDINGS evidence: changed, already correct (with its path:line), or out of scope (with the reason).";

// I33-FX2-COMPILE (I33-RG-ORG:4), I33-FX3-COMPILE: every common markdown shape of a census line produces the block — any
// number of `>` quotes and `-`/`*`/`+`/`1.`/`1)` list markers in any order, an optional `#` heading (then an optional list
// marker), `**`/`__`/`*`/`_` emphasis around the word or around `CENSUS:` nested up to two deep (`***CENSUS:***`,
// `**_CENSUS:_**`), in any case. A line in one of these shapes that does not parse refuses (parseCensusSpec); any other line
// with `census` and a colon refuses too (censusLike, below).
const CENSUS_LINE_RE = /^[ \t]*(?:>[ \t]*|(?:[-*+]|\d{1,3}[.)])[ \t]+)*(?:#{1,6}[ \t]+)?(?:(?:[-*+]|\d{1,3}[.)])[ \t]+)?(\*\*|__|\*|_)?(\*\*|__|\*|_)?CENSUS\2?\1?[ \t]*:(?:[ \t]*\2)?(?:[ \t]*\1)?[ \t]*(.*)$/i;

/** Parse one CENSUS line's body (the text after `CENSUS:`). The label is before the FIRST `|`, the paths after the LAST,
 *  and the regex is everything between, so a regex alternation (`a|b`) survives. Throws CensusError. */
export function parseCensusSpec(body, where = "CENSUS line") {
  const first = body.indexOf("|");
  const last = body.lastIndexOf("|");
  if (first < 0 || last === first) throw new CensusError(`${where}: expected \`CENSUS: <label> | <regex> | <path> [<path> ...]\`, got \`CENSUS: ${body}\``);
  const label = body.slice(0, first).trim();
  const regex = body.slice(first + 1, last).trim();
  const paths = body.slice(last + 1).trim().split(/\s+/).filter(Boolean);
  if (!label) throw new CensusError(`${where}: the label is empty`);
  if (!regex) throw new CensusError(`${where} (${label}): the regex is empty`);
  if (!paths.length) throw new CensusError(`${where} (${label}): no path is named`);
  try { new RegExp(regex); } catch (e) { throw new CensusError(`${where} (${label}): the regex \`${regex}\` does not compile: ${e.message}`); }
  return { label, regex, paths };
}

// I33-FX3-COMPILE (I33-RJ-01 residual): a line whose second or later `|` cell, code spans removed, is census-shaped is a census
// line inside a table cell; the table's `|` splits the census's own fields, so it is refused, never read or dropped.

// I33-FX4-CLASSES (D-1097; I33-RJ-03's record on I33-FX3-COMPILE:5, D-1096(1)'s open row): the shapes above are what
// PARSES; what REFUSES is a class. A line where the word `census` (any case; a letter or digit before it, or after it
// where a marker stands, makes it another word) is followed by a colon with nothing but markers between -- whitespace,
// punctuation, HTML tags -- and that the pattern above did not read, refuses, whatever markers surround it
// (`- [ ] CENSUS:`, `<b>CENSUS:</b>`, `#CENSUS:`, and `I21 census: ...` in prose too). It is tested on the line with its
// code spans removed (a quoted `CENSUS: ...` in a sentence is a mention), and on the whole line with backticks read as
// markers when the census word opens the line's text (a census line written inside a code span is not a mention).
// I34-TOOLS-B (I33 B8, D-1100(1)): a digit run that does not follow a letter is a marker between the word and the colon
// (`census 2:`; `census2` is another word), and the colon may be full-width or small (U+FF1A, U+FE55, U+FE13). A
// parenthesised label (`census (x):`) and the no-colon field shape (`CENSUS x | re | paths`) are NOT read here: the first
// would newly refuse 18 recorded packets, two of them current I34 scopes (I34-batch/_scopes/I34-W3-CLIENTS.md:18,
// I34-W3-TC.md:14), and existing cases assert both shapes silent (tests/tooling/i33-fx2-compile-fixes.test.ts:140,
// tests/tooling/i33-fx3-compile-branches.test.ts:161), so they wait on the orchestrator.
const CENSUS_WORD_COLON = String.raw`(?<![A-Za-z0-9])census(?:\s|<\/?[A-Za-z][^<>\n]*>|(?<![A-Za-z])\d+|[^\w\s]|_)*[:\uFF1A\uFE55\uFE13]`;
const CENSUS_LIKE_RE = new RegExp(CENSUS_WORD_COLON, "i");
const CENSUS_OPENS_RE = new RegExp(String.raw`^(?:[^A-Za-z0-9\n]|<\/?[A-Za-z][^<>\n]*>|\[[A-Za-z]\]|[A-Za-z0-9]{1,3}[.)])*` + CENSUS_WORD_COLON, "i");
const censusLike = (line) => CENSUS_LIKE_RE.test(line.replace(/`[^`\n]*`/g, "")) || CENSUS_OPENS_RE.test(line);

/** Every CENSUS line of a scope packet, in order; lines inside a fenced block are quoted examples and are skipped. A census
 *  line inside a table cell throws CensusError. */
export function parseCensusLines(text) {
  const specs = [];
  let fenced = false;
  String(text).split("\n").forEach((line, i) => {
    if (FENCE_RE.test(line)) { fenced = !fenced; return; }
    if (fenced) return;
    const m = line.match(CENSUS_LINE_RE);
    if (m) specs.push(parseCensusSpec(m[3], `scope packet line ${i + 1}`));
    else if (censusInTableCell(line)) throw new CensusError(`scope packet line ${i + 1}: a CENSUS line inside a table cell is not read (the table's \`|\` splits its fields); write it on a line of its own as \`CENSUS: <label> | <regex> | <path> [<path> ...]\`, got \`${line.trim()}\``);
    else if (censusLike(line)) throw new CensusError(`scope packet line ${i + 1}: \`census\` followed by a colon is read as a census line, and this one does not parse; write it on a line of its own as \`CENSUS: <label> | <regex> | <path> [<path> ...]\` (an optional list marker, quote, heading or emphasis may surround \`CENSUS:\`), or reword it so \`census\` is not followed by a colon; got \`${line.trim()}\``);
  });
  return specs;
}

/** What a symlink points at, for the skipped list: a directory, a file, or nothing (a broken link). */
function symlinkKind(p) {
  try { return statSync(p).isDirectory() ? "symlink to a directory" : "symlink to a file"; } catch { return "broken symlink"; }
}
// I33-FX2-COMPILE (I33-RG-ORG:5): a symlink met in a directory walk (file or directory) is NEVER followed (no loop, no read
// outside the tree) and never dropped silently: it is pushed to `skipped` with its kind, and the block lists it.
function walk(abs, out, skipped) {
  for (const e of readdirSync(abs, { withFileTypes: true }).sort((x, y) => (x.name < y.name ? -1 : x.name > y.name ? 1 : 0))) {
    const p = join(abs, e.name);
    if (e.isSymbolicLink()) skipped.push({ abs: p, kind: symlinkKind(p) });
    else if (e.isDirectory()) { if (!skipDir(e.name)) walk(p, out, skipped); }
    else if (e.isFile() && isSource(e.name)) out.push(p);
  }
}

/** Enumerate every line under `spec.paths` (relative to `root`) that `spec.regex` matches. Throws CensusError for a
 *  path that is absolute, escapes the root or does not exist. Returns { ...spec, count, matches: [{ path, line, text }] }. */
export function runCensus(spec, root) {
  const re = new RegExp(spec.regex);
  const files = [];
  const skippedAbs = [];
  for (const p of spec.paths) {
    if (isAbsolute(p)) throw new CensusError(`CENSUS (${spec.label}): \`${p}\` is absolute; census paths are relative to the build-tree root ${root}`);
    const abs = resolve(root, p);
    if (!isInside(root, abs, { allowSelf: true })) throw new CensusError(`CENSUS (${spec.label}): \`${p}\` lies outside the build-tree root ${root}`);
    let st;
    try { st = statSync(abs); } catch { throw new CensusError(`CENSUS (${spec.label}): the path \`${p}\` does not exist under ${root}`); }
    // I34-TOOLS-B (I33-JUDGE-AUDIT JA-03): isInside above is lexical and statSync follows a symlink, so the containment is
    // tested again on the REAL paths: a named path that is, or runs through, a symlink leaving the tree is refused.
    const realAbs = realpathSync(abs);
    if (!isInside(realpathSync(root), realAbs, { allowSelf: true })) throw new CensusError(`CENSUS (${spec.label}): \`${p}\` resolves through a symlink to ${realAbs}, outside the build-tree root ${root}; a census path never follows a symlink out of the tree`);
    if (st.isDirectory()) walk(abs, files, skippedAbs); else files.push(abs);
  }
  const relOf = (f) => relative(root, f).split(sep).join("/");
  const matches = [];
  for (const f of [...new Set(files)]) {
    const rel = relOf(f);
    const lines = readFileSync(f, "utf8").split("\n");
    // I33-FX2-COMPILE (I33-RG-ORG:6): the empty element after a file's final newline is not a line (`wc -l` does not count it).
    if (lines[lines.length - 1] === "") lines.pop();
    lines.forEach((raw, i) => {
      const line = raw.replace(/\r$/, "");
      if (re.test(line)) matches.push({ path: rel, line: i + 1, text: line.trim().slice(0, CENSUS_LINE_CHARS) });
    });
  }
  const skipped = [...new Map(skippedAbs.map((s) => [s.abs, { path: relOf(s.abs), kind: s.kind }])).values()];
  return { ...spec, count: matches.length, matches, skipped };
}

/** The `## CENSUS — <label>` block: the total, one line per match (capped, remainder counted), the answer sentence. */
export function renderCensusBlock(result, { maxLines = CENSUS_MAX_LINES } = {}) {
  const shown = result.matches.slice(0, maxLines);
  const rest = result.count - shown.length;
  return [
    `## CENSUS — ${result.label}`,
    `regex \`${result.regex}\` over ${result.paths.join(" ")} (relative to the build-tree root; directories recurse over ${SOURCE_EXTENSIONS.join("/")} files, skipping ${SKIPPED_DIRS.join(", ")}; a symlink inside a directory is never followed and is listed as skipped)`,
    `total: ${result.count} match${result.count === 1 ? "" : "es"}${result.count === 0 ? " — the census found NOTHING; say so, do not read it as coverage" : ""}`,
    ...shown.map((m) => `- ${m.path}:${m.line} — ${m.text}`),
    ...(rest > 0 ? [`- ... and ${rest} more match${rest === 1 ? "" : "es"} not listed (cap ${maxLines} lines); rerun the regex over the same paths to read them`] : []),
    ...((result.skipped ?? []).length ? [`skipped, NOT read (symlinks are never followed): ${result.skipped.map((s) => `${s.path} (${s.kind})`).join(", ")}`] : []),
    CENSUS_ANSWER_SENTENCE,
    "",
  ].join("\n");
}

// QUEUE #855 (I10-CLOSER, D-0870(5)): the ONE finding-id matcher for scripts/close-run.mjs and scripts/audit-precheck.mjs
// (it was a byte-identical copy in each). Declarations: finding-id.d.mts (the root tsconfig typechecks tests/**).
// An id is letters/digits in >= 2 dash-joined segments (`--` allowed), then any `.`-joined tail (`I3-0304.b`,
// `...-demand-optout.harness`); it starts with a letter of either case (`i8-jn-priv`, `q180-oidc`).

/** The id shape, unanchored (for building scanners). */
export const FINDING_ID_SRC = String.raw`[A-Za-z][A-Za-z0-9]*(?:--?[A-Za-z0-9]+)+(?:\.[A-Za-z0-9]+)*`;
const WHOLE_RE = new RegExp(`^${FINDING_ID_SRC}$`);
const SEV_WORD_RE = /^(?:critical|high|medium|low|info)$/i;

/** True when `s` is exactly one finding id. */
export function isFindingId(s) {
  return WHOLE_RE.test(String(s ?? ""));
}

/**
 * The judge ADJUDICATION-table row filter (I9-FX1/FX4, QUEUE #586): the id token of a table row, or null for a header,
 * aggregate or prose row. A DIGIT-FREE id (`TB-STATUS-VOCAB`) shares its shape with `Sub-total`/`Finding-ID`, so it is a
 * finding only when its severity cell (cells[2]) is EXACTLY a severity word (emphasis/backtick wrappers stripped).
 */
export function judgeTableFindingId(cells) {
  if (!Array.isArray(cells) || cells.length < 3) return null;
  const idTok = String(cells[0] ?? "").split(/\s+/)[0];
  if (/^id$/i.test(idTok) || !WHOLE_RE.test(idTok)) return null;
  if (!/\d/.test(idTok) && !SEV_WORD_RE.test(String(cells[2] ?? "").replace(/^[\s*_`]+|[\s*_`]+$/g, ""))) return null;
  return idTok;
}

/** Every id-shaped token in `text`, whole, with its index (a sentence-final `.` is not part of an id). */
export function findingIdTokens(text) {
  const re = new RegExp(`(?<![A-Za-z0-9-])${FINDING_ID_SRC}(?![A-Za-z0-9-])`, "g");
  return [...String(text ?? "").matchAll(re)].map((m) => ({ id: m[0], index: m.index }));
}

/**
 * The table id a DISPATCH token names, or null. (1) exact; (2) case-insensitive, only when EXACTLY ONE table id fits
 * (`i8-jn-priv` -> `I8-JN-PRIV`); (3) the legacy upper-case short form `X-N` (DISPATCH dropping the room segment),
 * only onto exactly one row `X-...-N`. Never a prefix or suffix match: `I3-0304` does not name `I3-0304.b`.
 */
export function matchTableId(token, ids) {
  const t = String(token ?? "");
  const list = [...(ids ?? [])].map(String);
  if (list.includes(t)) return t;
  const ci = list.filter((id) => id.toLowerCase() === t.toLowerCase());
  if (ci.length === 1) return ci[0];
  if (ci.length > 1) return null;
  if (/^[A-Z]-\d+$/.test(t)) {
    const [letter, num] = t.split("-");
    const fits = list.filter((id) => id.startsWith(`${letter}-`) && id.endsWith(`-${num}`));
    if (fits.length === 1) return fits[0];
  }
  return null;
}

/**
 * A RegExp finding `id` only on an id boundary: no [A-Za-z0-9-] either side and no `.`-joined continuation after it
 * (`I3-0304` is not found inside `I3-0304.b`; a sentence-final `I3-0304.` is). `ignoreCase` for whole ids only.
 */
export function idBoundaryRe(id, { ignoreCase = false } = {}) {
  const esc = String(id).replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
  return new RegExp(`(?<![A-Za-z0-9-])${esc}(?![A-Za-z0-9-]|\\.[A-Za-z0-9])`, ignoreCase ? "i" : "");
}
