/**
 * Evidence scan-on-upload — the adapter seam (F1, 2026-09-03).
 *
 * `POST /report/:item_id/evidence` is the product's ONLY stranger-supplied-image
 * path, and Cloudflare's CSAM Scanning Tool structurally cannot see it: that tool
 * scans at cache-fill, and evidence is Access-gated + `no-store` on purpose
 * (`docs/CSAM-SCANNING.md` § "Report evidence"). `docs/GO-LIVE-CHECKLIST.md`
 * therefore names a pre-GA HARD GATE — a scan-on-upload hook at the intake
 * seam, before the bytes commit, failing closed. This module is that seam.
 *
 * WHAT IT IS AND IS NOT. It is a bounded, typed HTTP call to a vendor the
 * OPERATOR wires (PhotoDNA, Safer, IWF — each has its own API, so the contract
 * here is a tiny shim any of them can sit behind; DEPLOY.md documents it). It
 * is not a classifier and not a face pipeline. `media/` is screened by its own
 * module, not by this route's hook: since I30 (Q143-D24, migration 0261)
 * `media.ts storeSignedMedia` stores a new row PENDING (`media_assets.screen_verdict`
 * NULL) and asks the screen `media-screen.ts mediaScreenerFromEnv` names
 * (MEDIA_SCREEN_PROVIDER; its one provider, `http-shim`, reuses this module's
 * `httpEvidenceScanner` contract with its own MEDIA_SCREEN_* keys; unset = no
 * screen, the row stays pending); `screenPendingMedia` retries pending rows;
 * `screened_at` records when the screen answered; and the serve routes answer
 * only a `clean` row or a `legacy` one (`isServableVerdict`). A `legacy` row was
 * stored before 0261 under the old posture, where the only thing that touched
 * those bytes was Cloudflare's CSAM Scanning Tool at CACHE-FILL — REACTIVE (the
 * first fetch that fills the cache is served before any block can land) and
 * conditional on the response staying publicly cacheable (`docs/CSAM-SCANNING.md`
 * § "Residual risks") — and it keeps serving.
 *
 * FAIL CLOSED, THREE TIMES. (1) Unconfigured — `evidenceScannerFromEnv` returns
 * null and the route answers 503 BEFORE reading the body; `storeEvidence`
 * refuses too, so no future caller can skip the scan. (2) Unreachable /
 * malformed / oversize / non-2xx — `error`, and the upload is refused with
 * nothing stored; never `clear`. (3) The vendor's answer is a hostile input:
 * measured before it is read, capped at 4 KiB, parsed to exactly one boolean.
 *
 * THE ONE HASHER. The request carries `x-dhwaja-sha256` computed by
 * `@dhwaja/canonical`'s `sha256Bytes` — the same digest a match event records —
 * so an operator can reconcile a vendor's log against `evidence_scan_events`
 * without Dhwaja retaining a byte of the material.
 */

import { sha256Bytes } from "@dhwaja/canonical";

export type ScanResult =
  | { verdict: "clear" }
  | { verdict: "match"; category: string | null }
  | { verdict: "error"; reason: string };

export interface EvidenceScanner {
  /** Never throws. `error` is the answer for everything that is not a clean
   * `clear` or `match` from the vendor. */
  scan(bytes: Uint8Array, contentType: string): Promise<ScanResult>;
}

export interface EvidenceScanConfig {
  url: string;
  token: string;
  timeoutMs: number;
}

/** A bare `https://` URL with an optional path — no query, no fragment, no
 * userinfo, no placeholder host. The same shape rule the checkpoint signer uses. */
const SCAN_URL_RE = /^https:\/\/[a-z0-9.-]+(?::\d{2,5})?(?:\/[A-Za-z0-9._~\-\/]*)?$/;
const TOKEN_RE = /^[A-Za-z0-9._\-]{16,512}$/;
const DEFAULT_TIMEOUT_MS = 8000;
/** A real answer is `{"match":false}`. Anything past this is not an answer. */
export const MAX_SCAN_RESPONSE_BYTES = 4096;
/** Vendor category labels are opaque strings; bounded before they are stored. */
const MAX_CATEGORY_LEN = 64;

/**
 * `EVIDENCE_SCAN_URL` (launch_var — declared empty in every vars block; empty
 * fails the upload route closed) + `EVIDENCE_SCAN_TOKEN` (secret). Either
 * absent, misshapen, or carrying `TODO` → null → 503. Optional
 * `EVIDENCE_SCAN_TIMEOUT_MS`, clamped to [250, 12000] (I3-1765: under the 15 s client abort).
 */
export function evidenceScanConfigFromEnv(env: Record<string, unknown>): EvidenceScanConfig | null {
  const url = typeof env.EVIDENCE_SCAN_URL === "string" ? env.EVIDENCE_SCAN_URL.trim() : "";
  const token = typeof env.EVIDENCE_SCAN_TOKEN === "string" ? env.EVIDENCE_SCAN_TOKEN.trim() : "";
  if (!SCAN_URL_RE.test(url) || /dhwaja\.example/i.test(url) || /TODO/i.test(url)) return null;
  if (!TOKEN_RE.test(token) || /TODO/i.test(token)) return null;
  const t = Number(env.EVIDENCE_SCAN_TIMEOUT_MS);
  // I3-1765: the ceiling stays under every client's request abort (apps/web api.ts API_TIMEOUT_MS = 15 s), with headroom for the upload itself.
  const timeoutMs = Number.isFinite(t) && t >= 250 && t <= 12_000 ? Math.trunc(t) : DEFAULT_TIMEOUT_MS;
  return { url, token, timeoutMs };
}

export function evidenceScannerFromEnv(env: Record<string, unknown>, fetchImpl: typeof fetch = fetch): EvidenceScanner | null {
  const cfg = evidenceScanConfigFromEnv(env);
  if (!cfg) return null;
  return httpEvidenceScanner(cfg, fetchImpl);
}

/**
 * The vendor contract, in one paragraph (also in DEPLOY.md): `POST <url>` with
 * the image bytes as the body, `content-type` = the magic-byte type Dhwaja
 * detected, `authorization: Bearer <token>`, `x-dhwaja-sha256: <hex>`. Reply
 * `2xx` with JSON `{"match": <boolean>, "category"?: <string>}`. Everything
 * else is `error`.
 */
export function httpEvidenceScanner(cfg: EvidenceScanConfig, fetchImpl: typeof fetch = fetch): EvidenceScanner {
  return {
    async scan(bytes, contentType) {
      const ctl = new AbortController();
      const timer = setTimeout(() => ctl.abort(), cfg.timeoutMs);
      try {
        const digest = hex(await sha256Bytes(bytes));
        const res = await fetchImpl(cfg.url, {
          method: "POST",
          headers: {
            authorization: `Bearer ${cfg.token}`,
            "content-type": contentType,
            "x-dhwaja-sha256": digest,
            accept: "application/json",
          },
          body: bytes as Uint8Array<ArrayBuffer>, // D-1061(6): typed for the DOM lib (apps/creator-console compiles this file since I30 6e9cbe8)
          signal: ctl.signal,
          // NEVER CHASE A REDIRECT WITH THE TOKEN ATTACHED — the house rule
          // every other token-bearing outbound call follows (media.ts,
          // mailer.ts, verify-fetch.ts). A 3xx is a vendor that is not where it
          // was configured: it lands in the catch as `vendor_unreachable`.
          redirect: "manual",
        });
        // manual mode surfaces a 3xx as a response instead of throwing; it
        // must be checked BEFORE !res.ok and routed into the same
        // vendor_unreachable value the catch below already reaches.
        if (res.status >= 300 && res.status < 400) return { verdict: "error", reason: "vendor_unreachable" };
        if (!res.ok) return { verdict: "error", reason: `vendor_status_${res.status}` };
        // MEASURE BEFORE READING. `checkpoint.ts` reads then measures, which a
        // hostile vendor can turn into an unbounded body; this seam declines
        // on the declared length first, then reads through a capped reader.
        const declared = Number(res.headers.get("content-length"));
        if (Number.isFinite(declared) && declared > MAX_SCAN_RESPONSE_BYTES) return { verdict: "error", reason: "vendor_response_too_large" };
        const text = await readCapped(res, MAX_SCAN_RESPONSE_BYTES);
        if (text === null) return { verdict: "error", reason: "vendor_response_too_large" };
        let parsed: unknown;
        try {
          parsed = JSON.parse(text);
        } catch {
          return { verdict: "error", reason: "vendor_response_not_json" };
        }
        if (typeof parsed !== "object" || parsed === null || Array.isArray(parsed)) return { verdict: "error", reason: "vendor_response_shape" };
        const match = (parsed as { match?: unknown }).match;
        if (typeof match !== "boolean") return { verdict: "error", reason: "vendor_response_shape" };
        if (!match) return { verdict: "clear" };
        const rawCat = (parsed as { category?: unknown }).category;
        const category = typeof rawCat === "string" && rawCat.length > 0 ? rawCat.replace(/[^A-Za-z0-9_.:\- ]/g, "").slice(0, MAX_CATEGORY_LEN) : null;
        return { verdict: "match", category: category && category.length > 0 ? category : null };
      } catch (e) {
        return { verdict: "error", reason: ctl.signal.aborted ? "vendor_timeout" : "vendor_unreachable" };
      } finally {
        clearTimeout(timer);
      }
    },
  };
}

/** Read at most `cap` bytes; null when the body is longer. Works with and
 * without a streaming body (the test double hands back plain Responses). */
async function readCapped(res: Response, cap: number): Promise<string | null> {
  const body = res.body;
  if (!body) {
    const t = await res.text();
    return t.length > cap ? null : t;
  }
  const reader = body.getReader();
  const chunks: Uint8Array[] = [];
  let total = 0;
  for (;;) {
    const { done, value } = await reader.read();
    if (done) break;
    if (value) {
      total += value.byteLength;
      if (total > cap) {
        await reader.cancel().catch(() => undefined);
        return null;
      }
      chunks.push(value);
    }
  }
  const out = new Uint8Array(total);
  let off = 0;
  for (const c of chunks) {
    out.set(c, off);
    off += c.byteLength;
  }
  return new TextDecoder().decode(out);
}

/** The actor stamped on a scan-match escalation and its audit line. A system
 * actor, like `system:face-triage` — the console headline "in our review
 * queue" is already written to be true of both doors. */
export const EVIDENCE_SCAN_ACTOR = "system:evidence-scan";

/** Lowercase hex of a digest — the one hasher's bytes, spelled out. */
function hex(b: Uint8Array): string {
  let out = "";
  for (const x of b) out += x.toString(16).padStart(2, "0");
  return out;
}

/**
 * flag 13 — report evidence uploads. THE ONLY FAN UPLOAD IN THE PRODUCT.
 *
 * Mock 20 lets a fan attach up to four screenshots to an impersonation report.
 * That is a categorically different risk from everything else Dhwaja stores, so
 * it is built as a bomb-disposal box: a narrow intake, a separate store, and no
 * way out except an operator's screen.
 *
 * WHY THIS IS NOT THE S91 SIGNED-MEDIA STORE (load-bearing — do not merge them):
 * `media.ts` is safe because of ONE positive guard — bytes are stored only when
 * their sha256 matches a fingerprint a creator's PASSKEY already signed into a
 * published release. Evidence has no such signature and never can: it is a
 * screenshot of somebody else's page, supplied by a stranger. Putting it in the
 * S91 store would mean loosening that guard to admit unsigned bytes, which
 * would retire the guard for signed art too. So evidence gets its own table and
 * its own R2 namespace — `evidence/<id>`, deliberately DISJOINT from `media/
 * <hash>` — and its safety argument is different in kind:
 *
 *   - NOT CONTENT-ADDRESSED. A content-addressed URL is a capability anyone who
 *     has the bytes can compute. Evidence ids are CSPRNG and unguessable, and
 *     the id alone is still not enough (see below).
 *   - NO PUBLIC ROUTE. The only reader is `GET /ops/fan-reports/:item_id/
 *     evidence/:evidence_id`, behind Cloudflare Access. There is no public
 *     serve path, no signed URL, no redirect to R2. The route-surface invariant
 *     test asserts this, in both directions.
 *   - NO CACHING. Unlike S91's deliberately public-cacheable art, this response
 *     is `no-store` + `nosniff` + `default-src 'none'; sandbox`.
 *   - INTAKE IS AUTHORIZED, NOT OPEN. Only a fan who is ALREADY A REPORTER on
 *     that exact item may attach to it, so the upload surface is bounded by the
 *     report throttles rather than by the internet.
 *
 * RETENTION RULE (stated here and in the MANIFEST): evidence dies with the
 * report. It is deleted when an operator DISMISSES or SUPPRESSES the item (the
 * terminal transitions in reports.ts), and when the uploading fan erases their
 * account (account_data.ts). Nothing here is a trust event and nothing here is
 * retained for its own sake — it exists to let one operator make one decision.
 *
 * FAIL CLOSED: absent bucket binding → the upload route 503s. It never accepts
 * bytes it cannot store, and never records a row it cannot serve.
 *
 * KNOWN GAP — CSAM SCANNING. Written up in full at
 * `docs/CSAM-SCANNING.md` § "Report evidence — NOT covered by this tool", with
 * two matching hard gates in `docs/GO-LIVE-CHECKLIST.md`.
 *
 * Cloudflare's CSAM Scanning Tool scans at CACHE-FILL time, so it only ever
 * sees PUBLICLY CACHEABLE objects. Evidence is deliberately the opposite —
 * Access-gated and `no-store` — so these objects are NOT covered by it, and
 * this is the product's ONLY stranger-supplied-image path. The trade is
 * deliberate: making evidence scannable would mean making a stranger's
 * screenshot of someone else's page public, which is worse than the gap.
 *
 * That makes the compensating controls load-bearing, and every one of them is
 * in this file or its callers: intake restricted to existing reporters on a
 * LIVE item, ≤4 images per (item, account) with byte/format/pixel caps, volume
 * bounded by the report-band throttles, retention tied to the report's terminal
 * state and to account erasure, operator review as the only way to see an
 * object, and an unconditional `deleteEvidence` takedown lever.
 *
 * Before GA the operator owes two things the code cannot supply: a
 * SCAN-ON-UPLOAD hook at this intake seam (right here in `storeEvidence`,
 * before the bytes are committed, failing closed), and the NCMEC CyberTipline
 * reporting duty extended to evidence — the media runbook is triggered by
 * Cloudflare's match email, which evidence never produces, so that duty
 * attaches to operator review instead.
 */

export interface StoredEvidence {
  evidence_id: string;
  content_type: string;
  byte_size: number;
}

/** The live (item, fan reporter) relationship: a FAN REPORTER row for this
 * account on this item. Bound as `(itemId, accountId)`. */
const LIVE_REPORTER_SQL = "EXISTS (SELECT 1 FROM fan_report_reporters WHERE item_id = ? AND reporter_kind = 'fan' AND reporter_id = ?)";

/** Step 1's authorization, re-used by every later answer that must read
 * exactly like it: a fan reporter on this item AND the item still live. */
async function isLiveReporter(db: DbLike, itemId: string, accountId: string): Promise<boolean> {
  const mine = await db
    .prepare(
      `SELECT 1 AS x FROM fan_report_reporters r
         JOIN fan_report_items i ON i.item_id = r.item_id AND i.state IN ('open','escalated')
        WHERE r.item_id = ? AND r.reporter_kind = 'fan' AND r.reporter_id = ? LIMIT 1`,
    )
    .bind(itemId, accountId)
    .first<{ x: number }>();
  return !!mine;
}

/** I3-1211: a stored object of this (item, account) whose bytes are identical
 * to `bytes`. Only same-size rows are fetched (at most MAX_EVIDENCE_PER_REPORT
 * of them), and a row whose object is missing is not a match. */
async function storedDuplicate(db: DbLike, store: EvidenceStore, itemId: string, accountId: string, bytes: Uint8Array): Promise<StoredEvidence | null> {
  const { results } = await db
    .prepare("SELECT evidence_id, r2_key, content_type, byte_size FROM report_evidence WHERE item_id = ? AND account_id = ? AND byte_size = ? ORDER BY created_at, evidence_id LIMIT ?")
    .bind(itemId, accountId, bytes.length, MAX_EVIDENCE_PER_REPORT)
    .all<{ evidence_id: string; r2_key: string; content_type: string; byte_size: number }>();
  for (const r of results) {
    const obj = await store.getBytes(r.r2_key);
    if (obj && sameBytes(obj.bytes, bytes)) return { evidence_id: r.evidence_id, content_type: r.content_type, byte_size: r.byte_size };
  }
  return null;
}

function sameBytes(a: Uint8Array, b: Uint8Array): boolean {
  if (a.length !== b.length) return false;
  for (let i = 0; i < a.length; i++) if (a[i] !== b[i]) return false;
  return true;
}

/**
 * Attach ONE image to a fan report.
 *
 * AUTHORIZATION: the caller must already be a FAN REPORTER on this exact item.
 * A non-reporter — and an unknown item id — get the SAME 404, so the endpoint
 * cannot be used to probe which subjects have open reports (report volume and
 * queue state are operator-only signals, per reports.ts).
 *
 * Guards run cheapest-and-most-authoritative first: authorization, then the
 * per-report count cap, then the byte cap, then the header-only format/pixel
 * caps. The content type is taken from the MAGIC BYTES and never from a client
 * header — the same rule as S91.
 */
export async function storeEvidence(
  db: DbLike,
  store: EvidenceStore,
  scanner: EvidenceScanner | null,
  itemId: string,
  accountId: string,
  bytes: Uint8Array,
  now: () => string = () => new Date().toISOString(),
): Promise<StoredEvidence> {
  // THE SCAN GATE IS INSIDE THE FUNCTION, not only at the route. A null
  // scanner refuses here too, so no future caller — a console convenience, an
  // importer, a test helper promoted to production — can store a stranger's
  // image without it having been scanned. 503, never 200 (invariant 6).
  if (scanner === null) throw new EvidenceError("evidence_scan_unconfigured", 503);
  if (typeof itemId !== "string" || !ITEM_ID_RE.test(itemId)) throw new EvidenceError("no such report", 404);

  // 1. AUTHORIZATION — a reporter on this item AND the item still LIVE.
  //
  //    The state gate is load-bearing, not cosmetic (adversarial review).
  //    Retention purges on the TERMINAL transitions, and those transitions only
  //    accept certain prior states: `dismissReport` requires `open`, and
  //    `suppressReport` requires open/dismissed/retracted. So evidence attached
  //    AFTER an item was closed could never be cleared by either lever — a
  //    reporter could re-upload onto a suppressed item and the bytes would be
  //    stuck forever, breaking "evidence dies with the report" on the product's
  //    only stranger-supplied-image path. Uploads are therefore accepted only
  //    while the item is open or escalated (the states an operator is still
  //    working), and a CLOSED item returns the SAME 404 as an unknown one, so
  //    this adds no queue-state oracle.
  if (!(await isLiveReporter(db, itemId, accountId))) throw new EvidenceError("no such report", 404);

  // 1b. IDEMPOTENT RETRY (I3-1211). An upload whose response was lost after
  //     the commit used to be answered, on the fan's retry, by a SECOND row for
  //     the same bytes: one more of the four slots gone, invisibly. Byte-
  //     identical bytes from the same (item, account) are the same evidence, so
  //     the stored row is the answer — before the slot count, so a retry at the
  //     cap still succeeds, and before the scan, because these bytes were
  //     scanned clear when they were stored. Compared by content, never by id.
  if (bytes.length > 0 && bytes.length <= MAX_EVIDENCE_BYTES) {
    const prior = await storedDuplicate(db, store, itemId, accountId, bytes);
    if (prior) return prior;
  }

  // 2. Per (item, account) count cap — mock 20's four screenshots. A scan
  //    MATCH consumes a slot too (F1 review, persona 89): a refused upload that
  //    cost nothing bought an unbounded number of hash-membership probes
  //    against the vendor, and every probe was a metered call. Four attempts
  //    per (item, account), stored or refused, and the route's per-account
  //    limiter sits in front of this.
  const count = await db
    .prepare(
      "SELECT (SELECT COUNT(*) FROM report_evidence WHERE item_id = ? AND account_id = ?) + (SELECT COUNT(*) FROM evidence_scan_events WHERE item_id = ? AND account_id = ?) AS n",
    )
    .bind(itemId, accountId, itemId, accountId)
    .first<{ n: number }>();
  if ((count?.n ?? 0) >= MAX_EVIDENCE_PER_REPORT) {
    throw new EvidenceError(`at most ${MAX_EVIDENCE_PER_REPORT} screenshots per report`, 409);
  }

  // 3. Byte cap, then fail-closed format detection + declared-pixel cap. The
  //    header's format IS the trusted content type; anything unrecognized or
  //    over-pixel is refused as unsupported rather than stored "just in case".
  if (bytes.length === 0) throw new EvidenceError("empty upload", 400);
  if (bytes.length > MAX_EVIDENCE_BYTES) throw new EvidenceError("image too large", 413);
  let header: ReturnType<typeof assertWithinCaps>;
  try {
    header = assertWithinCaps(bytes, { maxBytes: MAX_EVIDENCE_BYTES, maxPixels: MAX_EVIDENCE_PIXELS });
  } catch (e) {
    if (e instanceof OversizeImage) throw new EvidenceError("unsupported or oversized image (png/jpeg only)", 415);
    throw new EvidenceError("unreadable image", 400);
  }
  const contentType = header.format === "png" ? "image/png" : "image/jpeg";

  // 4. SCAN BEFORE COMMIT (F1, 2026-09-03 — the pre-GA hard gate in
  //    docs/GO-LIVE-CHECKLIST.md). The bytes have passed every cheap cap and
  //    are about to become an object; this is the last moment nothing of them
  //    persists, and the only moment that matters.
  //      error → 503 `evidence_scan_unavailable` (distinct from the
  //              missing-bucket `evidence_unavailable`, so the web copy can say
  //              "try again in a moment" rather than "switched off"), nothing
  //              stored — a vendor that cannot answer is not a vendor that
  //              answered "clear". The vendor-shaped reason is logged; it never
  //              carries fan data or the token.
  //      match → nothing stored, no R2 object, no bytes anywhere. A HASH-ONLY
  //              event row is written (the vendor got the same digest in
  //              `x-dhwaja-sha256`, so the two logs reconcile) and an audit
  //              line from `system:evidence-scan`. THE ITEM'S STATE IS NOT
  //              TOUCHED. The first cut escalated it, and four review personas
  //              showed that `escalated` is read by the uploader's own
  //              `/account/reports` ladder, by the suspected creator's
  //              `under_review`, by the digest, and by `rejectItem`'s caution
  //              clearer — a fact about the UPLOADER was being broadcast as a
  //              fact about the SUBJECT. The operator sees the match as a
  //              `scan_matches` count on the open queue instead. The uploader
  //              receives the SAME 415 a malformed PNG gets. HONEST LIMIT: an
  //              uploader who verified their own image locally can still read
  //              a 415 as "matched" — the status code hides nothing from them;
  //              what bounds the probe is the slot cap above and the route's
  //              limiter. The NCMEC duty stays with operator review
  //              (docs/CSAM-SCANNING.md); whether matched material must be
  //              PRESERVED rather than refused is a legal question recorded as
  //              a known gap, not decided in code.
  //   I3-0924 a (I23, D-0993(1) pick option 1): A FINGERPRINT ALREADY MATCHED IS REFUSED WITHOUT ASKING AGAIN.
  //   `evidence_scan_events.sha256` is the digest of bytes the vendor once matched; the same bytes are refused with
  //   the same 415 and recorded again (a fresh row + audit line under this item), whatever the vendor says today —
  //   a vendor that drifts to "clear", or is down, cannot let a known match through. Read before the scan so the
  //   vendor is not asked about bytes we already know. Indexed by migration 0217.
  const digest = hex(await sha256Bytes(bytes));
  const known = await db
    .prepare("SELECT category FROM evidence_scan_events WHERE sha256 = ? ORDER BY created_at DESC LIMIT 1")
    .bind(digest)
    .first<{ category: string | null }>();
  if (known) {
    await recordScanMatch(db, itemId, accountId, digest, known.category, now);
    throw new EvidenceError("unsupported or oversized image (png/jpeg only)", 415);
  }
  const scan = await scanner.scan(bytes, contentType);
  if (scan.verdict === "error") {
    logError({ cid: "unknown", hop: "http", msg: "evidence_scan_unavailable", kind: scan.reason }); // one of the adapter's fixed strings
    throw new EvidenceError("evidence_scan_unavailable", 503);
  }
  if (scan.verdict === "match") {
    await recordScanMatch(db, itemId, accountId, digest, scan.category, now);
    throw new EvidenceError("unsupported or oversized image (png/jpeg only)", 415);
  }

  // 5. THE POINTER FIRST, THEN THE BYTES (I3-0942, I3-1367). This used to be
  //    bytes-then-row, excused by "swept by the same GC as other orphans" —
  //    no such collector exists (media.ts states none can enumerate keys), and
  //    evidence keys are CSPRNG, so no retry can ever adopt an orphan. An
  //    INSERT that THREW after the put stranded a stranger's screenshot with
  //    nothing pointing at it. Now: a row without bytes is the harmless
  //    direction (the ops serve answers 404 and every purge deletes through
  //    it), so the row is written first and every failure after it removes
  //    what this call wrote.
  //
  //    The INSERT re-asserts, atomically at commit: the COMBINED slot cap
  //    (PS-065/PV-10 — scan matches consume slots too, so N concurrent uploads
  //    that all read "3 so far" cannot all land); ITEM LIVENESS (PS-067 — the
  //    terminal transitions purge what exists THEN, and an insert landing after
  //    would never be swept); and, I3-1334/I3-1335, THE LIVE REPORTER
  //    RELATIONSHIP — the account erasure batch pseudonymises this fan's
  //    `fan_report_reporters` rows (account_data.ts), so an upload authorised
  //    at step 1 cannot land after that batch and keep the erased id.
  //
  //    #792 (D-0859): THE PRE-PUT INTENT. The upload's work order (`upload:<id>`)
  //    commits in the SAME batch as the pointer, predicated on the pointer having
  //    landed, so it exists BEFORE `putBytes`. A Worker that dies during or after
  //    the put leaves it; an erasure that deletes the pointer meanwhile skips the
  //    key (ON CONFLICT DO NOTHING at `carryAccountEvidenceStmt`) and
  //    `drainErasedEvidence` deletes the bytes once the intent is older than
  //    `UPLOAD_INTENT_GRACE_MS` and no pointer names the key. It is deleted after
  //    the `kept` re-read below; a pointer that survives makes the drain drop the
  //    intent and keep the bytes.
  const evidenceId = newEvidenceId();
  const key = evidenceKey(evidenceId);
  const ts = now();
  const [res] = (await db.batch([
    db
      .prepare(
        `INSERT INTO report_evidence (evidence_id, item_id, account_id, r2_key, content_type, byte_size, created_at)
       SELECT ?, ?, ?, ?, ?, ?, ?
        WHERE (SELECT COUNT(*) FROM report_evidence WHERE item_id = ? AND account_id = ?)
            + (SELECT COUNT(*) FROM evidence_scan_events WHERE item_id = ? AND account_id = ?) < ?
          AND EXISTS (SELECT 1 FROM fan_report_items WHERE item_id = ? AND state IN ('open','escalated'))
          AND ${LIVE_REPORTER_SQL}`,
      )
      .bind(evidenceId, itemId, accountId, key, contentType, bytes.length, ts, itemId, accountId, itemId, accountId, MAX_EVIDENCE_PER_REPORT, itemId, itemId, accountId),
    db.prepare(UPLOAD_INTENT_SQL).bind(key, `upload:${evidenceId}`, ts, evidenceId),
  ])) as Array<{ meta?: { changes?: number }; changes?: number }>;
  if ((res?.meta?.changes ?? res?.changes ?? 0) !== 1) {
    // Nothing was written to storage. Which guard tripped changes the answer:
    // a slot cap is the 409, but an item that closed (or a reporter
    // relationship that ended) mid-upload reads exactly like step 1's unknown
    // item (404) — never a distinct code (a queue-state oracle).
    if (!(await isLiveReporter(db, itemId, accountId))) throw new EvidenceError("no such report", 404);
    throw new EvidenceError(`at most ${MAX_EVIDENCE_PER_REPORT} screenshots per report`, 409);
  }
  try {
    await store.putBytes(key, bytes, contentType);
  } catch (e) {
    // D-0861(4), RGP-L1-02: a put that throws may still have stored the bytes
    // (a timeout after the write). Same order as the `!kept` cleanup below:
    // work order first, then the delete (an absent object deletes cleanly). The
    // pointer row goes — freeing the slot — only once the object is gone or a
    // work order carries it; with neither, it stays as the pointer every purge
    // deletes through. The caller always gets the put's own error.
    const carried = await db
      .prepare(UPLOAD_WORK_ORDER_SQL)
      .bind(key, `upload:${evidenceId}`, now())
      .run()
      .then(
        () => true,
        () => false,
      );
    let gone = false;
    try {
      await store.delete(key);
      gone = true;
    } catch {
      logError({ cid: "evidence-upload", hop: "http", msg: "upload_evidence_delete_deferred" });
    }
    if (gone && carried) await db.prepare("DELETE FROM erasure_evidence_purges WHERE r2_key = ?").bind(key).run().catch(() => undefined);
    if (gone || carried) await db.prepare("DELETE FROM report_evidence WHERE evidence_id = ?").bind(evidenceId).run().catch(() => undefined);
    throw e;
  }
  // The row was committed before the put; a terminal transition (or the
  // erasure batch) that ran while the bytes were in flight has already swept
  // the rows it could see and will never come back for these bytes. Re-read
  // the pointer against a LIVE item AFTER the bytes exist: gone or closed →
  // this call removes what it wrote and answers like an unknown item.
  const kept = await db
    .prepare(`SELECT 1 AS x FROM report_evidence e JOIN fan_report_items i ON i.item_id = e.item_id AND i.state IN ('open','escalated') WHERE e.evidence_id = ?`)
    .bind(evidenceId)
    .first<{ x: number }>();
  if (!kept) {
    // QUEUE #779 (D-0858(15), within D-0848(3)): whatever swept this item or
    // account has gone, so this delete is the last pointer to the bytes. A work
    // order goes into `erasure_evidence_purges` FIRST; a delete that throws
    // leaves it for `drainErasedEvidence`, a delete that lands takes it with it.
    const carried = await db
      .prepare(UPLOAD_WORK_ORDER_SQL)
      .bind(key, `upload:${evidenceId}`, now())
      .run()
      .then(
        () => true,
        () => false,
      );
    try {
      await store.delete(key);
    } catch (e) {
      if (!carried) throw e; // no work order: nothing else can reach the bytes, so fail loud
      logError({ cid: "evidence-upload", hop: "http", msg: "upload_evidence_delete_deferred" });
      await db.prepare("DELETE FROM report_evidence WHERE evidence_id = ?").bind(evidenceId).run().catch(() => undefined);
      throw new EvidenceError("no such report", 404);
    }
    if (carried) await db.prepare("DELETE FROM erasure_evidence_purges WHERE r2_key = ?").bind(key).run();
    await db.prepare("DELETE FROM report_evidence WHERE evidence_id = ?").bind(evidenceId).run();
    throw new EvidenceError("no such report", 404);
  }
  // #792: the upload is whole and pointed at; its intent is spent. A failed delete
  // is not an error to the fan: the drain finds the live pointer and drops the row.
  await db
    .prepare("DELETE FROM erasure_evidence_purges WHERE r2_key = ? AND erasure_id = ?")
    .bind(key, `upload:${evidenceId}`)
    .run()
    .catch(() => logError({ cid: "evidence-upload", hop: "http", msg: "upload_intent_delete_deferred" }));
  return { evidence_id: evidenceId, content_type: contentType, byte_size: bytes.length };
}

/**
 * D-0647 (Jason, 2026-09-21) — PUBLISHED EVIDENCE IS RETAINED, AND BOUNDED.
 *
 * The three terminal WITHDRAWAL transitions purge (`reports.ts:1686` dismiss,
 * `:1708` suppress, `:1965` retract). PUBLICATION does not, and that is
 * deliberate: a published warning is an ongoing public claim about a named
 * person, and destroying the proof of it while the claim stands leaves this
 * platform unable to substantiate its own statement to the accused, to a
 * reviewer or to a court.
 *
 * THE RULING IS RETAIN **AND BOUND IT**, NOT RETAIN. Before D-0647 that
 * decision existed ONLY as the comment at `reports.ts:1963-1964`
 * ("Publication is NOT a closure and keeps its evidence"), with no period, no
 * end state, and nothing in `docs/legal/PRIVACY.md` — whose §1.4 bullet and
 * retention table said the opposite and marked it "Implemented". A comment is
 * not a policy (D-0440). The bound below is the policy, and
 * `docs/legal/PRIVACY.md` §1.4 / the retention table now carry it in the place
 * a person affected by it would actually look.
 *
 * THE CLOCK IS `report_evidence.created_at`, NOT `fan_report_items.updated_at`.
 * `updated_at` is bumped by an ordinary new report on the same subject
 * (`reports.ts:755`) and by the face pass (`face-triage.ts:174`, `:275`), so a
 * bound keyed to it can be pushed forward indefinitely by third-party activity
 * the accused does not control — a bound that is unbounded in the adversarial
 * case, which is precisely what D-0647 forbids. Upload always precedes
 * publication, so measuring from upload is never the looser reading.
 * That holds for the READ gate below. The DELETION sweep waits longer (D-0740):
 * see `purgeExpiredPublishedEvidence`.
 */
export const PUBLISHED_EVIDENCE_RETENTION_DAYS = 365;

/**
 * Rows uploaded at or before this instant are past the bound. Exported because
 * the bound must be assertable from a test: see
 * `test/w312-b3-evidence-retention.test.ts`, whose S3 mutation is this
 * arithmetic.
 */
export function publishedEvidenceCutoff(nowIso: string): string {
  return new Date(Date.parse(nowIso) - PUBLISHED_EVIDENCE_RETENTION_DAYS * 86_400_000).toISOString();
}

/**
 * THE BOUND, ENFORCED ON READ. `listEvidence` and `getEvidence` are the ONLY
 * routes in this tree that serve evidence to anyone (`console.ts:566`, `:593`),
 * so gating both of them here is what makes the retention period a fact rather
 * than a sentence in a document. It applies ONLY to a PUBLISHED item: an open
 * or escalated case is still live work, and the retention table carries that
 * as its own open row.
 */
function pastCutoff(createdAt: string, nowIso: string): boolean {
  return Date.parse(createdAt) <= Date.parse(publishedEvidenceCutoff(nowIso));
}

async function itemIsPublished(db: DbLike, itemId: string): Promise<boolean> {
  const item = await db.prepare("SELECT state FROM fan_report_items WHERE item_id = ?").bind(itemId).first<{ state: string }>();
  return item?.state === "published";
}

export interface EvidenceRow {
  evidence_id: string;
  account_id: string;
  content_type: string;
  byte_size: number;
  created_at: string;
}

/** OPERATOR read: the evidence attached to one report item (metadata only —
 * bytes come from the separate serve route, one object at a time). */
export async function listEvidence(db: DbLike, itemId: string, nowIso: string = new Date().toISOString()): Promise<EvidenceRow[]> {
  if (typeof itemId !== "string" || !ITEM_ID_RE.test(itemId)) return [];
  const { results } = await db
    .prepare("SELECT evidence_id, account_id, content_type, byte_size, created_at FROM report_evidence WHERE item_id = ? ORDER BY created_at ASC, evidence_id ASC")
    .bind(itemId)
    .all<EvidenceRow>();
  // D-0647 THE BOUND. One state read, and only when a row is actually past the
  // cutoff — an item with nothing expired pays nothing.
  if (!results.some((r) => pastCutoff(r.created_at, nowIso))) return results;
  if (!(await itemIsPublished(db, itemId))) return results;
  return results.filter((r) => !pastCutoff(r.created_at, nowIso));
}

/** OPERATOR read: one object, scoped to its item. The (item, evidence) pairing
 * is enforced in the query so a mismatched pair is a 404 rather than a way to
 * read any evidence id from any item's URL. */
export async function getEvidence(db: DbLike, store: EvidenceStore, itemId: string, evidenceId: string, nowIso: string = new Date().toISOString()): Promise<{ bytes: Uint8Array; content_type: string } | null> {
  if (typeof itemId !== "string" || !ITEM_ID_RE.test(itemId)) return null;
  if (typeof evidenceId !== "string" || !EVIDENCE_ID_RE.test(evidenceId)) return null;
  const row = await db
    .prepare("SELECT r2_key, content_type, created_at FROM report_evidence WHERE evidence_id = ? AND item_id = ?")
    .bind(evidenceId, itemId)
    .first<{ r2_key: string; content_type: string; created_at: string }>();
  if (!row) return null;
  // D-0647 THE BOUND, on the one route that serves the BYTES. Past the period a
  // published item's evidence is served to nobody — operator included — whether
  // or not the sweep below has reached it yet.
  if (pastCutoff(row.created_at, nowIso) && (await itemIsPublished(db, itemId))) return null;
  const obj = await store.getBytes(row.r2_key);
  if (!obj) return null;
  // The STORED, magic-byte-validated type — never the object's own metadata and
  // never a client header.
  return { bytes: obj.bytes, content_type: row.content_type };
}

/**
 * OPERATOR TAKEDOWN of a single evidence object — the counterpart of
 * `DELETE /ops/media/:hash`, and the lever that must exist because retention is
 * otherwise driven entirely by report state (adversarial review).
 *
 * Without it, an operator who opens an attachment on a PUBLISHED or already-
 * suppressed item and finds illegal content has nothing to press: neither
 * terminal transition accepts those states, so the bytes would stay servable.
 * Any evidence id, any item state — except under an open legal hold (P34 a, D-1399): while a hold names the item, the
 * uploader's account or the passport the item names (`PURGE_LEGAL_HOLD_SQL`, the purge doors' predicate) this refuses
 * 409 `legal_hold` before the bytes go, and the row DELETE asks it again (a hold landing between keeps the row; the bytes
 * may already be gone, the residual `purgeItemEvidence` carries too).
 *
 * Bytes first, then the row: the failure mode that matters is "we said we
 * removed it and it is still fetchable". No tombstone (unlike media takedown):
 * evidence is not content-addressed and there is no stable hash to ban — the
 * re-upload path is closed by the item-state gate on intake instead.
 */
export async function deleteEvidence(db: DbLike, store: EvidenceStore | null, itemId: string, evidenceId: string): Promise<boolean> {
  if (typeof itemId !== "string" || !ITEM_ID_RE.test(itemId)) throw new EvidenceError("no such evidence", 404);
  if (typeof evidenceId !== "string" || !EVIDENCE_ID_RE.test(evidenceId)) throw new EvidenceError("no such evidence", 404);
  const row = await db
    .prepare(`SELECT e.r2_key AS r2_key, ${PURGE_LEGAL_HOLD_SQL} AS held FROM report_evidence e LEFT JOIN fan_report_items i ON i.item_id = e.item_id WHERE e.evidence_id = ? AND e.item_id = ?`)
    .bind(evidenceId, itemId)
    .first<{ r2_key: string; held: number }>();
  if (!row) throw new EvidenceError("no such evidence", 404);
  if (row.held === 1) throw new EvidenceError("legal_hold", 409);
  // FAIL CLOSED (review HIGH 2). With no store binding this used to delete the
  // ROW and return success: the console toasted "the bytes and the row are
  // gone" while nothing had touched storage — and deleting the row destroys the
  // only pointer to the r2 key, so the object becomes unreachable-but-present
  // FOREVER. The operator reaching for this lever is by definition looking at
  // content they must be able to remove, so an unprovisioned store is a refusal,
  // never a false success. The row is left intact on purpose: it is the pointer
  // a later, provisioned purge needs.
  if (!store) throw new EvidenceError("evidence storage unavailable — refusing to purge, the bytes would survive", 503);
  await store.delete(row.r2_key);
  const res = await db
    .prepare(`DELETE FROM report_evidence WHERE evidence_id IN (SELECT e.evidence_id FROM report_evidence e LEFT JOIN fan_report_items i ON i.item_id = e.item_id WHERE e.evidence_id = ? AND NOT ${PURGE_LEGAL_HOLD_SQL})`)
    .bind(evidenceId)
    .run();
  return res.changes > 0;
}

/**
 * RETENTION: drop every object and row for one report item. Called from the
 * terminal transitions in reports.ts (dismiss / suppress).
 *
 * Bytes first, rows second: the failure mode that matters is "we said we
 * deleted it and the bytes are still fetchable", so the fetchable thing goes
 * first, and a row goes only with its own object: after a mid-flight failure
 * the rows left are exactly the objects left (I3-0880).
 */
export async function purgeItemEvidence(db: DbLike, store: EvidenceStore | null, itemId: string, createdThrough?: string): Promise<number> {
  // D-1046(1): a hold landing after the closing door's UPDATE (which asked the same predicate) keeps every row.
  // I39-FX-OPS: `createdThrough` (the retraction countersign only) keeps every row created after that instant.
  const through = createdThrough === undefined ? "" : " AND created_at <= ?";
  const { results } = await db
    .prepare(`SELECT evidence_id, r2_key FROM report_evidence WHERE item_id = ? AND NOT ${itemPurgeHeldSql("?")}${through}`)
    .bind(itemId, itemId, ...(createdThrough === undefined ? [] : [createdThrough]))
    .all<{ evidence_id: string; r2_key: string }>();
  if (results.length === 0) return 0; // nothing to purge — an unprovisioned store is fine here
  // Same fail-closed rule as `deleteEvidence`: rows are the ONLY pointer to the
  // r2 keys, so dropping them with no store to delete through would strand the
  // objects permanently. The callers wrap this in `purgeItemEvidenceQuietly`,
  // which logs the refusal and lets the (already committed) state change stand —
  // the rows then survive for the operator takedown lever to finish the job.
  if (!store) throw new EvidenceError("evidence storage unavailable — refusing to drop the rows that point at the bytes", 503);
  // I3-0880: PER OBJECT. A mid-loop R2 fault used to abort with objects
  // 1..k-1 gone and EVERY row standing, and the item-wide DELETE could also
  // take a row an in-flight upload committed after the SELECT above. Only the
  // rows whose objects this call actually deleted are dropped, by id; a failed
  // object keeps its row (the pointer a retry or the takedown lever needs), and
  // the call then refuses so the caller records the failure.
  const done: string[] = [];
  for (const r of results) {
    try {
      await store.delete(r.r2_key);
      done.push(r.evidence_id);
    } catch {
      /* counted below; the row keeps the only pointer */
    }
  }
  let purged = 0;
  for (const chunk of inChunks(done)) {
    const res = await db
      .prepare(`DELETE FROM report_evidence WHERE evidence_id IN (${chunk.map(() => "?").join(", ")})`)
      .bind(...chunk)
      .run();
    purged += res.changes;
  }
  const failed = results.length - done.length;
  if (failed > 0) throw new EvidenceError(`evidence purge incomplete — ${failed} object(s) could not be deleted; their rows are kept as the pointer`, 503);
  return purged;
}

/** The system actor on the retention audit rows the terminal transitions
 * leave behind (`evidence:purge`, `evidence:purge_failed`). */
export const EVIDENCE_RETENTION_ACTOR = "system:evidence-retention";

/** One retention audit row. Never throws: it runs after a committed state
 * change, and a failed audit write must not unwind or 500 the transition. */
async function auditRetention(db: DbLike, itemId: string, action: "evidence:purge" | "evidence:purge_failed" | "evidence:purge_held", detail: string): Promise<void> {
  try {
    await db
      .prepare("INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) VALUES (?, ?, ?, ?, ?, 'system')")
      .bind(EVIDENCE_RETENTION_ACTOR, action, detail.slice(0, 500), new Date().toISOString(), itemId)
      .run();
  } catch (e) {
    logError({ cid: "evidence-purge", hop: "http", msg: "evidence_retention_audit_failed", err: e instanceof Error ? e.name : typeof e });
  }
}

/**
 * The form the terminal report transitions call: purge, but NEVER let a storage
 * fault propagate (adversarial review). The state change has already COMMITTED
 * by the time this runs, so a throwing R2 delete would surface as a 500 to the
 * operator and — worse — skip the audit write that happens after it, leaving a
 * committed dismissal with no audit line while the objects stay servable.
 *
 * NOT SILENT (I3-0880, I3-1349, I3-1838). A bulk destruction writes an
 * `evidence:purge` audit row naming the count — the single-object takedown
 * (console.ts `evidence:delete`) always did, and this one destroyed more with
 * no record. A failure writes an `evidence:purge_failed` row naming how many
 * rows still point at bytes that were not deleted, so the refusal is a durable
 * operator-visible fact on the item's audit trail rather than a console line;
 * those rows remain for the takedown lever. There is no GC sweep for them.
 */
export async function purgeItemEvidenceQuietly(db: DbLike, store: EvidenceStore | null, itemId: string, createdThrough?: string): Promise<number> {
  try {
    // D-1046(1): held -> nothing purged, and the `evidence:purge_held` row the console's held-evidence list reads
    // (console.ts, D-1042(2)) is where an operator finds it to take down after release (as the retraction leg, D-1038(2)).
    if ((await db.prepare(`SELECT ${itemPurgeHeldSql("?")} AS held`).bind(itemId).first<{ held: number }>())?.held === 1) {
      await auditRetention(db, itemId, "evidence:purge_held", "item purge refused: an open legal hold names this report item, an uploader of its evidence or its passport; its evidence is kept — take it down once the hold is released");
      return 0;
    }
    const n = await purgeItemEvidence(db, store, itemId, createdThrough);
    if (n > 0) await auditRetention(db, itemId, "evidence:purge", `${n} evidence object(s) and their rows destroyed by the item's terminal transition`);
    return n;
  } catch (e) {
    logError({ cid: "evidence-purge", hop: "http", msg: "item_evidence_purge_failed", err: e instanceof Error ? e.name : typeof e });
    let left = -1;
    try {
      left = (await db.prepare("SELECT COUNT(*) AS n FROM report_evidence WHERE item_id = ?").bind(itemId).first<{ n: number }>())?.n ?? -1;
    } catch {
      /* the count is best-effort; the audit row still records the failure */
    }
    await auditRetention(db, itemId, "evidence:purge_failed", `purge refused or incomplete: ${left < 0 ? "unknown number of" : left} evidence row(s) still point at bytes that were not deleted — use the evidence takedown`);
    return 0;
  }
}

/** Rows (and R2 objects) one `purgeExpiredPublishedEvidence` call may take:
 * one R2 delete each plus two D1 statements, inside the 6-hourly `scheduled()`
 * run's subrequest budget. Clamped to 100, D1's bound-parameter cap. */
export const PUBLISHED_EVIDENCE_SWEEP_LIMIT = 50;

/**
 * I21-Q29 (D-1030): an open legal hold keeps a published item's evidence past the bound. The sweep skips a row while
 * a hold names its report item, its uploader's account, or the passport the item names (`suspected_passport_id`);
 * once the hold is released the next sweep takes it. Over aliases `e` (report_evidence) and `i` (fan_report_items).
 * D-1041(4): exported; itemPurgeHeldSql below asks it item-wide for the item purge doors.
 */
export const evidencePurgeHeldSql = (e: string, i: string): string =>
  `(${legalHoldOpenSql("report_item", `${i}.item_id`)} OR ${legalHoldOpenSql("account", `${e}.account_id`)} OR ${legalHoldOpenSql("passport", `${i}.suspected_passport_id`)})`;
const PURGE_LEGAL_HOLD_SQL = evidencePurgeHeldSql("e", "i");

/**
 * D-1041(4) (I29-RJ-03): the item purge doors honour every hold the published-evidence sweep honours — the report item,
 * the uploader account of any of its evidence rows, or the passport it names — by asking evidencePurgeHeldSql item-wide.
 * LEFT JOIN: an item with no evidence still answers its item and passport holds. `itemExpr` is `?` or a column.
 * Lives here (I29-FX9) because the purge itself asks it; reports.ts re-exports it for its doors, admin.ts and console.ts.
 */
export const itemPurgeHeldSql = (itemExpr: string): string =>
  `EXISTS (SELECT 1 FROM fan_report_items i LEFT JOIN report_evidence e ON e.item_id = i.item_id WHERE i.item_id = ${itemExpr} AND ${evidencePurgeHeldSql("e", "i")})`;

/**
 * D-0647 THE BOUND, byte half — the retention sweep for PUBLISHED items.
 * Called from `scheduled()` as the step "purgeExpiredPublishedEvidence" (D-0685).
 *
 * `purgeItemEvidence` above is the WITHDRAWAL purge and is item-scoped: a case
 * closed, everything for it goes. This one is ROW-scoped, because a published
 * item's attachments age individually and only the ones past
 * `PUBLISHED_EVIDENCE_RETENTION_DAYS` may be taken — taking the rest would
 * destroy the proof of a public claim that is still standing, which is the
 * thing D-0647 ruled against.
 *
 * Bytes first, rows second, and the same FAIL-CLOSED `!store` rule as
 * `purgeItemEvidence`: the rows are the only pointer to the r2 keys, so
 * dropping them with no store to delete through strands the objects forever.
 * Wrap the call site in the caller's own non-throwing guard, as
 * `purgeItemEvidenceQuietly` does for the withdrawal legs.
 */
export async function purgeExpiredPublishedEvidence(
  db: DbLike,
  store: EvidenceStore | null,
  nowIso: string = new Date().toISOString(),
  limit: number = PUBLISHED_EVIDENCE_SWEEP_LIMIT,
): Promise<number> {
  const cutoff = publishedEvidenceCutoff(nowIso);
  const take = Math.max(1, Math.min(100, Math.floor(limit)));
  // BOUNDED (QUEUE #562): oldest first, at most `take` rows per call. A caller
  // may conclude nothing about rows beyond the limit; the next call resumes,
  // because every row this call deletes leaves the selected set.
  // D-0740: the SWEEP deletes only when 365 days have passed since the LATER
  // of upload and publication. Publication is `warning_publications.published_at`
  // for the item's review case: written once in the publish batch, never moved.
  // NOT `fan_report_items.updated_at`, which any new report or face pass on the
  // subject bumps (see the constant's docblock), so third parties could extend
  // it forever (D-0647). A published item with no publication row (no writer
  // produces one) falls back to the upload clock. The read gate is unchanged.
  const { results } = await db
    .prepare(
      // #636a: rows whose delete has failed before go to the BACK of the window
      // (migration 0168), so `take` permanently failing objects cannot starve the rest.
      `SELECT e.evidence_id AS evidence_id, e.r2_key AS r2_key FROM report_evidence e JOIN fan_report_items i ON i.item_id = e.item_id LEFT JOIN warning_publications wp ON wp.item_id = i.review_item_id LEFT JOIN evidence_sweep_failures f ON f.evidence_id = e.evidence_id WHERE i.state = 'published' AND e.created_at <= ? AND COALESCE(wp.published_at, e.created_at) <= ? AND NOT ${PURGE_LEGAL_HOLD_SQL} ORDER BY COALESCE(f.failures, 0), e.created_at, e.evidence_id LIMIT ?`,
    )
    .bind(cutoff, cutoff, take)
    .all<{ evidence_id: string; r2_key: string }>();
  if (results.length === 0) return 0; // nothing past the bound — an unprovisioned store is fine here
  if (!store) throw new EvidenceError("evidence storage unavailable — refusing to drop the rows that point at the bytes", 503);
  // RGB-562-poison-wedge: one object whose delete keeps failing must not stop
  // the rest. It is skipped THIS call (its row stays, so the next call retries
  // it) and only the rows whose objects were actually deleted are dropped.
  // Logged as a count, never a key or id (INV-061). #636a: each failure is also
  // counted per row (0168), which moves that row behind every row that has not
  // failed, so failing objects no longer fill the window ahead of deletable ones.
  const done: string[] = [];
  for (const r of results) {
    try {
      await store.delete(r.r2_key);
      done.push(r.evidence_id);
    } catch {
      // the row keeps the only pointer for the retry; the count reorders it
      await db
        .prepare("INSERT INTO evidence_sweep_failures (evidence_id, failures, last_failed_at) VALUES (?, 1, ?) ON CONFLICT(evidence_id) DO UPDATE SET failures = failures + 1, last_failed_at = excluded.last_failed_at")
        .bind(r.evidence_id, nowIso)
        .run()
        .catch(() => undefined); // the order is best-effort; the kept row and the logged count below are the record
    }
  }
  const failed = results.length - done.length;
  if (failed > 0) logError({ cid: "evidence-sweep", hop: "cron", msg: "published_evidence_delete_failed", failed, count: done.length });
  let purged = 0;
  // Only rows whose objects were deleted; chunked under D1's bind cap (one
  // chunk at `take` <= 100 minus the helper's margin).
  for (const chunk of inChunks(done)) {
    const marks = chunk.map(() => "?").join(", ");
    const res = await db.prepare(`DELETE FROM report_evidence WHERE evidence_id IN (${marks})`).bind(...chunk).run();
    purged += res.changes;
    // #636a: the deleted rows' failure counts go too. Best-effort and after the
    // pointer delete: a count left behind names a random id that no longer exists
    // and orders nothing (the sweep joins it to live rows only).
    await db
      .prepare(`DELETE FROM evidence_sweep_failures WHERE evidence_id IN (${marks})`)
      .bind(...chunk)
      .run()
      .catch(() => undefined);
  }
  return purged;
}

/**
 * The profile PAGE URL for the face fetch, built ONLY from the platform
 * allowlist table + the canonical (ASCII-validated) handle — never a
 * reporter-supplied string. Distinct from verify-fetch's bio-code target
 * (which points at API-ish pages like reddit's about.json): the face stage
 * wants the HTML page whose og:image is the profile picture. Platforms whose
 * URL shape isn't a simple handle path (e.g. youtube's mixed forms) return
 * null → the stage records no_target and the human path continues.
 */
export function faceProfileUrl(platform: string, idKind: string, handle: string): string | null {
  if (idKind !== "handle") return null;
  const spec = PLATFORMS[platform];
  if (!spec || spec.hosts.length === 0) return null;
  if (!/^[a-z0-9._-]{1,64}$/i.test(handle)) return null; // canonical guarantees this; re-checked anyway
  const host = spec.hosts[0]!;
  const ex = spec.extract as { kind: string; requiresAt?: boolean; prefixes?: { seg: string }[] };
  if (ex.kind === "first-segment") return `https://${host}/${ex.requiresAt ? "@" : ""}${encodeURIComponent(handle)}`;
  if (ex.kind === "prefix" && ex.prefixes && ex.prefixes.length > 0) return `https://${host}/${ex.prefixes[0]!.seg}/${encodeURIComponent(handle)}`;
  return null;
}

/** The five named entities HTML defines unconditionally (no DTD lookup needed)
 * plus numeric character references — the only decoding a bare `content="..."`
 * attribute value ever needs. PV-09: a real page commonly writes a query-string
 * `&` as `&amp;` inside the attribute (signed CDN/S3 URLs almost always carry
 * one); left undecoded, the fetch below requests a literal `&amp;` and gets the
 * wrong bytes or a 404 — a false NEGATIVE (the sweep silently stops matching),
 * not a false positive, but still real signal loss on a safety mechanism. */
function decodeHtmlAttributeEntities(s: string): string {
  return s
    .replace(/&#x([0-9a-f]{1,6});/gi, (_, hex: string) => String.fromCodePoint(parseInt(hex, 16)))
    .replace(/&#(\d{1,7});/g, (_, dec: string) => String.fromCodePoint(parseInt(dec, 10)))
    .replace(/&amp;/gi, "&")
    .replace(/&lt;/gi, "<")
    .replace(/&gt;/gi, ">")
    .replace(/&quot;/gi, '"')
    .replace(/&(?:apos|#39|#x27);/gi, "'");
}

/** Extract the profile image URL from page HTML: og:image / twitter:image
 * meta (either attribute order), https only, bounded length. Regex over a
 * capped page is fine here — we need one attribute, not a DOM. */
export function extractProfileImageUrl(html: string): string | null {
  const head = html.slice(0, 256 * 1024);
  const metas = head.match(/<meta[^>]{1,600}>/gi) ?? [];
  for (const tag of metas) {
    if (!/(?:property|name)\s*=\s*["'](?:og:image(?::secure_url)?|twitter:image(?::src)?)["']/i.test(tag)) continue;
    const m = /content\s*=\s*["']([^"']{1,2048})["']/i.exec(tag);
    if (!m) continue;
    const url = decodeHtmlAttributeEntities(m[1]!.trim());
    if (/^https:\/\//i.test(url)) return url;
  }
  return null;
}

    scanned: out.scanned,
    detail: `high=${out.matched_high} medium=${out.matched_medium} drafted=${out.drafted} errored=${out.errored} unavailable=${out.unavailable}`,
  });
  return out;
}

/**
 * I14-CAPS (QUEUE #912, D-0904(1)): THE SWEEP'S SUBREQUEST CAP. It runs on the digests tail
 * (alerts.ts THE DIGESTS TAIL: 836 of the ~1000 ceiling budgeted before it, 164 left), and
 * index.ts spends 3 of those on this leg before calling it (the two settings reads and the
 * armed probe), so the sweep itself may spend this many. One subrequest = one fetch hop, one
 * matcher call, or one D1 statement (a batch counts per statement).
 */
export const FACE_TRIAGE_SUBREQUEST_BUDGET = 160;
/**
 * The most one item can spend, worst path: the page and the image at
 * 1 + PAGE_LIMITS/IMAGE_LIMITS.maxRedirects hops each (6), the matcher (1), the passport and
 * own-account reads (2), the evidence UPDATE (1), the review enqueue batch (2), draftWarning,
 * the review_item_id UPDATE, and a throw's face_state UPDATE. Measured on every path by
 * test/i14-face-triage-cap.test.ts, which fails if any path spends more.
 */
export const FACE_TRIAGE_ITEM_MAX_SUBREQUESTS = 24;

/** The most items one sweep tick may take, whatever `deps.batch` asks for. The image
 * service's receiver-side daily ceiling on /internal/triage-find is derived from it
 * (services/image/src/http.ts TRIAGE_FIND_DAILY_CEILING = ticks/day x this; I3-2178). */
export const FACE_TRIAGE_MAX_BATCH = 50;

export async function faceTriageSweep(db: DbLike, deps: FaceSweepDeps): Promise<SweepOutcome> {
  const now = deps.now ?? (() => new Date().toISOString());
  const fetchImpl = deps.fetchImpl ?? fetch;
  // I14-CAPS: the item SELECT (1) plus whole items only, so the sweep stops at an item boundary;
  // the items it leaves stay `pending`, oldest first, and are the next tick's first picks.
  const fit = Math.floor(((deps.budget ?? FACE_TRIAGE_SUBREQUEST_BUDGET) - 1) / FACE_TRIAGE_ITEM_MAX_SUBREQUESTS);
  const batch = Math.max(0, Math.min(FACE_TRIAGE_MAX_BATCH, deps.batch ?? 10, fit));
  if (batch === 0) return logSweep({ scanned: 0, matched_high: 0, matched_medium: 0, drafted: 0, errored: 0, unavailable: 0 }, deps.cid);
  const out: SweepOutcome = { scanned: 0, matched_high: 0, matched_medium: 0, drafted: 0, errored: 0, unavailable: 0 };
  const retryCutoff = new Date(Date.parse(now()) - RETRY_AFTER_MS).toISOString();

  // I14-FX1 (RG-CAPS-04, D-0905(1)): reports never yet triaged go AHEAD of retries, so items that
  // keep failing cannot starve newer ones; and a retry BACKS OFF — it waits at least as long as the
  // item has already been failing (now - face_checked_at >= face_checked_at - created_at), so each
  // wait doubles, with RETRY_AFTER_MS the floor. Still one statement; no column added.
  // I3-0834 (D-0920 option B): PENDING items are taken by per-reporter rank, then created_at — each
  // reporter's oldest pending item before any reporter's second, so one reporter's crowd of items cannot
  // push another reporter's report to the back. An item's reporter is the one who FILED it (its earliest
  // fan_report_reporters row); an item with none left ranks as its own reporter. Retries stay FIFO.
  const { results: items } = await db
    .prepare(
      "SELECT item_id, subject_hash, canonical, platform, handle, evidence, suspected_passport_id FROM (" +
        "SELECT item_id, subject_hash, canonical, platform, handle, evidence, suspected_passport_id, created_at, " +
        "CASE WHEN face_state = 'pending' THEN 0 ELSE 1 END AS lane, " +
        "(SELECT r.reporter_kind || ':' || r.reporter_id FROM fan_report_reporters r WHERE r.item_id = fan_report_items.item_id " +
        "ORDER BY r.created_at, r.reporter_kind, r.reporter_id LIMIT 1) AS filer FROM fan_report_items " +
        "WHERE state = 'open' AND (face_state = 'pending' OR (face_state IN ('unavailable', 'error') AND face_checked_at <= ? " +
        "AND julianday(?) - julianday(face_checked_at) >= julianday(face_checked_at) - julianday(created_at)))) " +
        "ORDER BY lane, CASE WHEN lane = 0 THEN ROW_NUMBER() OVER (PARTITION BY lane, COALESCE(filer, item_id) ORDER BY created_at, item_id) ELSE 0 END, " +
        "created_at ASC LIMIT ?",
    )
    .bind(retryCutoff, now(), batch)
    .all<{ item_id: string; subject_hash: string; canonical: string; platform: string; handle: string; evidence: string; suspected_passport_id: string | null }>();

  for (const item of items) {
    out.scanned++;
    try {
      const result = await triageOne(db, item, deps.matcher, fetchImpl, deps.wordingId, now);
      if (result === "match_high") out.matched_high++;
      if (result === "match_high_drafted") {
        out.matched_high++;
        out.drafted++;
      }
      if (result === "match_medium") out.matched_medium++;
      if (result === "unavailable") out.unavailable++;
    } catch {
      // Isolation: one hostile/broken profile never stops the sweep.
      out.errored++;
      await setFaceState(db, item.item_id, "error", now).catch(() => {});
    }
  }
  return logSweep(out, deps.cid);
}

async function setFaceState(db: DbLike, itemId: string, state: string, now: () => string): Promise<void> {
  await db.prepare("UPDATE fan_report_items SET face_state = ?, face_checked_at = ?, updated_at = ? WHERE item_id = ?").bind(state, now(), now(), itemId).run();
}

type TriageResult = "no_target" | "no_image" | "unavailable" | "no_match" | "self" | "match_medium" | "match_high" | "match_high_drafted";

async function triageOne(
  db: DbLike,
  item: { item_id: string; subject_hash: string; canonical: string; platform: string; handle: string; evidence: string; suspected_passport_id: string | null },
  matcher: FaceMatcher,
  fetchImpl: typeof fetch,
  wordingId: string | null,
  now: () => string,
): Promise<TriageResult> {
  // Rebuild the profile URL from the CANONICAL subject (v1|platform|idKind|id)
  // via the platform allowlist — never a reporter-supplied string.
  const parts = item.canonical.split("|");
  const targetUrl = parts.length === 4 ? faceProfileUrl(item.platform, parts[2]!, parts[3]!) : null;
  if (!targetUrl) {
    await setFaceState(db, item.item_id, "no_target", now);
    return "no_target";
  }

  // 1. Profile page (SSRF-guarded, size/time-capped, re-guarded per redirect).
  let pageBytes: Uint8Array;
  try {
    const page = await safeFetch(targetUrl, PAGE_LIMITS, fetchImpl);
    pageBytes = page.bytes;
  } catch {
    // Blocked bot / login wall / guard refusal — the stage is best-effort.
    await setFaceState(db, item.item_id, "no_image", now);
    return "no_image";
  }
  const imageUrl = extractProfileImageUrl(new TextDecoder().decode(pageBytes));
  if (!imageUrl) {
    await setFaceState(db, item.item_id, "no_image", now);
    return "no_image";
  }

  // 2. The image itself, under the same guard (og:image is page-controlled —
  //    i.e. attacker-controlled — so it gets the FULL hostile-URL treatment).
  let imageBytes: Uint8Array;
  let contentType: string | null;
  try {
    const img = await safeFetch(imageUrl, IMAGE_LIMITS, fetchImpl);
    imageBytes = img.bytes;
    contentType = img.contentType;
  } catch {
    await setFaceState(db, item.item_id, "no_image", now);
    return "no_image";
  }

  // 3. Match against the consented corpus (ephemeral on the service side).
  const match = await matcher(imageBytes, contentType);
  if (match === null) {
    await setFaceState(db, item.item_id, "unavailable", now);
    return "unavailable";
  }
  if (!match.matched || typeof match.passport_id !== "string" || !/^psp_[0-9a-f]{16}$/.test(match.passport_id)) {
    await setFaceState(db, item.item_id, "no_match", now);
    return "no_match";
  }

  // 4. Guards before anything actionable.
  //    (a) The matched creator must be VERIFIED and live right now.
  const p = await db.prepare("SELECT claim_state, retracted FROM passports WHERE passport_id = ?").bind(match.passport_id).first<{ claim_state: string; retracted: number }>();
  if (!p || p.retracted !== 0 || p.claim_state !== "verified") {
    await setFaceState(db, item.item_id, "no_match", now);
    return "no_match";
  }
  //    (b) The reported account must NOT be the creator's own (any status —
  //        if it is, this is the real person, e.g. their unclaimed alt).
  const own = await db
    .prepare("SELECT 1 AS x FROM passport_accounts WHERE passport_id = ? AND account_key = ? LIMIT 1")
    .bind(match.passport_id, item.canonical)
    .first<{ x: number }>();
  if (own) {
    await setFaceState(db, item.item_id, "self", now);
    return "self";
  }

  // Evidence merge (outcome only — no embedding, no bytes).
  let evidence: Record<string, unknown> = {};
  try {
    evidence = JSON.parse(item.evidence) as Record<string, unknown>;
  } catch {
    evidence = {};
  }
  const high = match.stage === "exact" || match.near_duplicate === true;
  evidence.face = {
    stage: String(match.stage ?? "embedding"),
    near_duplicate: match.near_duplicate === true,
    exact_conflict: match.exact_conflict === true,
    matched_passport_id: match.passport_id,
    // imageUrl is attacker-controlled (og:image). It's already https-only from
    // extractProfileImageUrl, but strip markup chars and re-assert the scheme
    // so a future HTML/href sink in the ops console can't be tricked.
    image_url: /^https:\/\//i.test(imageUrl) ? imageUrl.replace(/[<>"'`\s]/g, "").slice(0, 512) : "",
    confidence: high ? "high" : "medium",
    checked_at: now(),
  };
  const ts = now();
  await db
    .prepare("UPDATE fan_report_items SET evidence = ?, suspected_passport_id = COALESCE(suspected_passport_id, ?), face_state = ?, face_checked_at = ?, updated_at = ? WHERE item_id = ?")
    .bind(JSON.stringify(evidence), match.passport_id, high ? "match_high" : "match_medium", ts, ts, item.item_id)
    .run();

  if (!high) return "match_medium";

  // 5. AUTO-FLAG (high only): create the warning DRAFT so the cooling-off
  //    clock starts now and the operator's remaining work is the human gates.
  //    Needs a configured approved wording; absent → stay on the human path.
  //    HIGH means the fake is using the creator's actual enrolled photo
  //    (exact bytes or perceptual near-duplicate) — same-face-different-photo
  //    (lookalike/twin risk) never auto-drafts.
  if (!wordingId) return "match_high";

  // AUTO-DRAFT GUARD (review finding #1): a photo/face match ALONE cannot
  // distinguish "impersonator stole the creator's photo" from "the creator's
  // own UNREGISTERED alt reusing their own photo" — both look identical to the
  // face stage, and the registered-account self-guard above can't see an
  // unregistered alt. What disambiguates is the HANDLE: a genuine impersonator
  // camps a look-alike of the creator's handle; a creator's own alt almost
  // never does. So auto-DRAFT (the machine writing an accusation) requires the
  // Stage-3 handle evidence to point at the SAME matched creator. Without it,
  // the item stays match_high — pinned to the TOP of the human queue with the
  // face evidence attached — but a human, not the machine, opens the draft.
  const sim = evidence.handle_similarity as { suspected_passport_id?: string } | undefined;
  if (!sim || sim.suspected_passport_id !== match.passport_id) return "match_high";
  let reviewId: string | null = null;
  try {
    reviewId = await enqueue(
      db,
      "system:face-triage",
      {
        item_type: "report",
        subject: `sha256:${item.subject_hash}`,
        payload: {
          source: "face_triage",
          fan_report_item: item.item_id,
          matched_passport_id: match.passport_id,
          face_stage: String(match.stage ?? "embedding"),
          near_duplicate: match.near_duplicate === true,
          exact_conflict: match.exact_conflict === true,
        },
      },
      () => ts,
    );
    await draftWarning(
      db,
      "system:face-triage",
      reviewId,
      { warning_class: "impersonation", wording_id: wordingId, protected_passport_id: match.passport_id },
      () => ts,
    );
    // I3-0266: deliberately NOT flipping fan_report_items.state to 'escalated'
    // here. `reports.ts fanReportStatus()` maps 'escalated' to the fan-visible
    // "reviewing" bucket (vs "received" for 'open'), and `protection.ts`'s own
    // docblock already names this exact defect: "escalated IS NOT ONLY A
    // REVIEWER'S ACT ... face-triage.ts sets it ... so an item can enter this
    // population with no human having opened it." This leg runs with no
    // operator in the loop and no meter beyond the ordinary per-day report
    // cap, so writing the visible bit here turns a fan's own report-status
    // poll into an unmetered, machine-answered biometric-confirmation oracle:
    // file a report against a candidate account, poll your own report,
    // "reviewing" vs "received" answers "does this face match a verified
    // creator's enrolled photo?". The drafted warning above already queues
    // (review_queue, keyed by reviewId, independent of this row's `state`) for
    // an operator to act on fast; the fan-visible state moves only when a
    // HUMAN escalates (reports.ts escalateReport) or acts on the draft
    // (warnings.ts publish/withdraw).
    // RESTORED by the orchestrator (D-0497): the removed statement wrote TWO things, and only
    // one of them was the leak. `state = 'escalated'` is fan-visible through
    // fanReportStatus() and stays gone. `review_item_id` is an INTERNAL pointer from the fan
    // report to the review_queue row this leg just drafted, and dropping it left the draft
    // with nothing linking it back to the report it came from -- operator traceability lost as
    // collateral from a privacy fix. Caught by test/face-triage.test.ts:117, which looks the
    // review row up BY that pointer.
    await db
      .prepare("UPDATE fan_report_items SET review_item_id = ?, updated_at = ? WHERE item_id = ? AND state = 'open'")
      .bind(reviewId, ts, item.item_id)
      .run();
    return "match_high_drafted";

  } catch {
    // Draft failure (e.g. wording retired mid-sweep) degrades to the human
    // path. Reject the review row we just enqueued so it doesn't linger as an
    // orphan open 'report' item (review finding #5); the fan item stays open at
    // match_high, pinned to the queue top for a human to open manually.
    if (reviewId) {
      // THE CLEANUP TAKES THE SHARED DECISION LOCK TOO (migration 0031), even
      // though this is a machine tidying up after itself. `open` is a state five
      // doors leave, and this one is the door migration 0030 named when it left
      // the cross-module race open: a console operator who opened this item
      // between the enqueue above and this line has already approved or rejected
      // it, and an unguarded flip here would have overwritten a human's decision
      // with `rejected` — the machine, silently, having the last word on an item
      // a person had already ruled on. In the same batch as the flip, so a loser
      // writes nothing at all.
      //
      // Still best-effort (`.catch`), and deliberately: this is a degradation
      // path, the fan item stays `open` at match_high pinned to the top of the
      // human queue either way, and a lost race means somebody has already
      // decided the item — a better outcome than the cleanup, not a worse one.
      const ts = now();
      await db
        .batch([
          reviewDecisionStmt(db, reviewId, "open", "rejected", "system:face-triage", ts, null),
          db
            .prepare("UPDATE review_queue SET state = 'rejected', disposition = 'rejected', updated_at = ? WHERE item_id = ? AND state = 'open'")
            .bind(ts, reviewId),
        ])
        .catch(() => {});
    }
    return "match_high";
  }
}

/** Random lowercase hex, `bytes` bytes → 2*bytes chars (crypto RNG). */
function randHex(bytes: number): string {
  const a = new Uint8Array(bytes);
  crypto.getRandomValues(a);
  let s = "";
  for (const b of a) s += b.toString(16).padStart(2, "0");
  return s;
}

function backupEmailError(c: Context<{ Bindings: Env }>, e: FanAuthError): Response {
  if (e.status === 429 && e.detail) {
    const retry = e.detail.retryAfterSeconds;
    if (typeof retry === "number") {
      c.header("Retry-After", String(retry));
      return c.json({ ...errorBody(e), reason: e.detail.reason, retry_after_seconds: retry }, 429);
    }
    return c.json({ ...errorBody(e), reason: e.detail.reason }, 429);
  }
  return c.json(errorBody(e), e.status);
}

export function makeFanAccountRoutes(): Hono<{ Bindings: Env }> {
  const app = new Hono<{ Bindings: Env }>();


  // --- Fan account surfaces (fan-session gated; NOT Access, NOT anonymous) ---












  // Device registration is now a PROOF-OF-POSSESSION two-step, so a fan who merely
  // knows another fan's push-token string can't seize it:
  //   1) POST /account/device/challenge {platform, push_token} → the server pushes
  //      a secret code to that token (only the real device receives it) and returns
  //      a challenge_id (never the code).
  //   2) POST /account/device {challenge_id, code} → the app echoes the code it got
  //      via push; the server verifies + consumes it single-use, THEN attaches.
  app.post("/account/device/challenge", anonBodyLimit, async (c) => {
    const id = await fanId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const body = (await c.req.json().catch(() => null)) as { platform?: unknown; push_token?: unknown } | null;
    if (!body || (body.platform !== "apns" && body.platform !== "fcm") || typeof body.push_token !== "string") {
      return c.json({ error: "platform (apns|fcm) and push_token required" }, 400);
    }
    const db = d1(c.env.TRUST_DB);
    // ─────────────────────────────────────────────────────────────────────────
    // THE ONLY ROUTE THAT SENDS A REAL PUSH TO A CALLER-SUPPLIED ADDRESS.
    // ─────────────────────────────────────────────────────────────────────────
    // It had no IP gate, no per-account limit and no cap on `device_challenges`
    // rows, which made one cheap fan session three things at once: an UNMETERED
    // PUSH AMPLIFIER (post somebody else's token, they get a silent push, repeat
    // — and repeated unsolicited pushes are how a bundle id loses its APNs
    // standing), an unbounded writer to a table only a redeem ever reads, and a
    // TOKEN-VALIDITY ORACLE, because the response distinguished "the provider
    // accepted this token" (200 + challenge_id) from "it did not" (503).
    //
    // Two buckets, for the two different attackers. The per-ACCOUNT cap bounds
    // one session; the per-IP cap bounds somebody minting sessions, which is
    // cheap here by design. Both are consumed BEFORE any provider call, so a
    // throttled caller costs nothing outbound.
    if (!(await consumeRateLimit(db, `devchal:${id}`, Date.now(), DEVICE_CHALLENGES_PER_ACCOUNT_HOUR, HOUR_MS))) {
      return c.json(refusal("rate limit exceeded"), 429);
    }
    if (!(await fanIpAllowed(c, "devchal", DEVICE_CHALLENGES_PER_IP_HOUR))) return c.json(refusal("rate limit exceeded", RATE_LIMIT_MESSAGE), 429);
    const code = randHex(8); // the secret that must arrive via the push channel
    const challengeId = `dch_${randHex(20)}`;
    try {
      await createDeviceChallenge(db, id, body.platform, body.push_token, challengeId, code, Date.now());
    } catch (e) {
      if (e instanceof AccountError) return c.json(errorBody(e), e.status);
      return serverFault(c, "/account/device/challenge", e);
    }
    // Deliver the code out-of-band via a SILENT push. Only whoever actually
    // holds the token receives it, which is the whole proof-of-possession.
    //
    // THE RESPONSE IS THE SAME EITHER WAY, and that is the oracle fix. This used
    // to answer 503 `push_unavailable` when the provider rejected the token and
    // 200 with a challenge_id when it accepted, so anyone with a fan session
    // could ask "is this push token live?" about any token they had scraped —
    // and get an answer that costs us an outbound call. The challenge row is
    // written before the send either way, so a caller learns nothing from the
    // response and everything from whether the code actually arrives on the
    // device. A genuinely unconfigured provider is now indistinguishable from a
    // dead token here; that is deliberate, and `POST /account/device` still
    // fails closed (the code never arrives, so the challenge cannot be redeemed).
    // The failure is logged so an operator is not left guessing.
    const delivered = await sendChallengePush(c.env as unknown as Record<string, unknown>, { device_id: "", platform: body.platform, push_token: body.push_token }, code);
    if (!delivered) logError({ cid: crypto.randomUUID(), hop: "http", msg: "device challenge push not accepted" });
    return c.json({ challenge_id: challengeId });
  });

  app.post("/account/device", anonBodyLimit, async (c) => {
    const id = await fanId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const body = (await c.req.json().catch(() => null)) as { challenge_id?: unknown; code?: unknown } | null;
    if (!body || typeof body.challenge_id !== "string" || typeof body.code !== "string") {
      return c.json({ error: "challenge_id and code required" }, 400);
    }
    const db = d1(c.env.TRUST_DB);
    if (!(await consumeRateLimit(db, `devredeem:${id}`, Date.now(), DEVICE_REDEEMS_PER_ACCOUNT_HOUR, HOUR_MS))) {
      return c.json(refusal("rate limit exceeded"), 429);
    }
    // Verify + consume the challenge (proves the caller's device received the
    // pushed code); only then bind the token.
    const redeemed = await redeemDeviceChallenge(db, id, body.challenge_id, body.code, Date.now());
    if (!redeemed) return c.json({ error: "invalid_challenge" }, 401);
    try {
      const r = await attachDevice(db, id, redeemed.platform, redeemed.push_token);
      return c.json({ device_id: r.device_id, ok: true });
    } catch (e) {
      if (e instanceof AccountError) return c.json(errorBody(e), e.status);
      return serverFault(c, "/account/device", e);
    }
  });

  // I3-0679 — the fan's own push-device removal (sign-out on a phone, "stop
  // notifying this device"). Scoped by BOTH keys in the statement, so a
  // device_id belonging to another account deletes nothing and answers exactly
  // like one that never existed: `removed` reports only the caller's own row.
  app.delete("/account/device/:device_id", async (c) => {
    const id = await fanId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const deviceId = c.req.param("device_id");
    try {
      const r = await d1(c.env.TRUST_DB).prepare("DELETE FROM devices WHERE device_id = ? AND account_id = ?").bind(deviceId, id).run();
      return c.json({ ok: true, removed: r.changes > 0 });
    } catch (e) {
      return serverFault(c, "/account/device/:device_id", e);
    }
  });

  // Social sign-in (Google / Apple). The client obtains a provider ID token (web
  // redirect+PKCE, or the native Apple/Google button) and posts it here. We VERIFY
  // it (signature + iss/aud/exp/nonce), find-or-create the account by (provider,
  // sub), and hand back a fan session. Sign-up and login are the same call.
  // I3-1778 (D-1080(5)): the nonce the client gives the provider is ISSUED here first. Metered per masked IP in
  // its own bucket at the mint door's number, so fetching a nonce never spends a sign-in and a flood of issues is
  // capped. The value is stored for the fan door only, expiring `OIDC_NONCE_TTL_MS` after issue.
  app.post("/account/oauth/nonce", anonBodyLimit, async (c) => {
    if (!(await fanIpAllowed(c, "oidcnonce", FAN_MINTS_PER_IP_HOUR))) return c.json(refusal("rate limit exceeded", RATE_LIMIT_MESSAGE), 429);
    const { nonce } = await issueNonce(d1(c.env.TRUST_DB), "fan", Date.now());
    return c.json({ nonce, expires_in: OIDC_NONCE_TTL_MS / 1000 });
  });
  app.post("/account/oauth/:provider/signin", anonBodyLimit, async (c) => {
    const provider = c.req.param("provider");
    if (provider !== "google" && provider !== "apple") return c.json({ error: "unknown_provider" }, 404);
    const cfg = providerConfig(provider as OidcProvider, c.env as Record<string, unknown>);
    if (!cfg) return c.json({ error: "provider_not_configured" }, 503); // fail closed
    // If sessions are disabled we can't complete sign-in anyway — bail BEFORE
    // spending a nonce or minting an account (avoids orphaned accounts on a
    // misconfigured deploy).
    if (!usableSessionSecret(c.env.SESSION_SECRET)) return c.json({ error: "sessions_unavailable" }, 503);
    // Metered BEFORE the JWKS fetch and the signature verify below, so a
    // throttled caller costs no outbound call and no crypto — the rule
    // `/account/device/challenge` states for its own provider call.
    if (!(await fanIpAllowed(c, "fanmint", FAN_MINTS_PER_IP_HOUR))) return c.json(refusal("rate limit exceeded", RATE_LIMIT_MESSAGE), 429);
    const body = (await c.req.json().catch(() => null)) as { id_token?: unknown; nonce?: unknown; referral_code?: unknown } | null;
    if (!body || typeof body.id_token !== "string") return c.json({ error: "id_token required" }, 400);
    // Nonce is now REQUIRED (replay defense): the token must carry a matching
    // nonce, and that nonce is spent single-use server-side (see below).
    if (typeof body.nonce !== "string" || body.nonce.length === 0) return c.json({ error: "nonce required" }, 400);
    const nonce = body.nonce;
    try {
      const nowMs = Date.now();
      // I3-0483: the presented token's kid, so a just-rotated provider key refreshes the cache early (bounded in JwksCache).
      const jwks = await oidcJwks(cfg.jwksUrl, nowMs, unverifiedKid(body.id_token));
      const claims = await verifyIdToken(body.id_token, cfg, jwks, Math.floor(nowMs / 1000), nonce);
      const db = d1(c.env.TRUST_DB);
      // I21-Q152: a provider identity this deployment has never seen is a sign-up (or a first link), so it takes the
      // bot check; a returning identity is never asked. Before the nonce is spent, so a refused caller can retry.
      // The read is oidc.ts signInWithOidc's own first statement (provider, sub): oidcIdentityAccount, one spelling.
      const knownIdentity = await oidcIdentityAccount(db, provider, claims.sub);
      if (knownIdentity === null) {
        const refused = await botRefusal(c, body);
        if (refused) return refused;
      }
      // Spend the nonce AFTER the signature/claims verify. I3-1778: only a nonce `/account/oauth/nonce` issued for
      // this door, unexpired and unspent, is accepted, and it is spent once — a replay (same token, or any token
      // carrying this nonce) and a client-chosen value both change no row and are refused. token.nonce == body.nonce
      // alone can't do this, because a stolen token supplies its own matching nonce.
      if (!(await consumeNonce(db, nonce, "fan", nowMs))) return c.json({ error: NONCE_REFUSED }, 401);
      // D-0706 (QUEUE #180): the link notice needs a real mailer, off the response path.
      const r = await signInWithOidc(db, provider as OidcProvider, claims, undefined, mailerFromEnv(c.env as unknown as Record<string, unknown>), deferrer(c), c.env as unknown as Record<string, unknown>);
      // Optional referral code — linked at account birth only (this is now where
      // accounts are born). Best-effort: a bad code must never fail sign-in, and
      // only a brand-new account can attribute a referral (no back-dating).
      if (r.created && typeof body.referral_code === "string") {
        try {
          await linkReferral(db, r.account_id, body.referral_code);
        } catch (e) {
          logError({ cid: crypto.randomUUID(), hop: "http", msg: "referral link failed at sign-in", err: e instanceof Error ? e.name : typeof e });
        }
      }
      // AUDIT ITEM #24: the founder auto-follow used to be applied here, at
      // account birth. A new account now starts with an empty social graph and
      // stays that way until its owner does something.
      const session = await issueSession(c.env.SESSION_SECRET, { kind: "fan", id: r.account_id, ep: r.session_epoch }, Math.floor(nowMs / 1000));
      if (!session) return c.json({ error: "sessions_unavailable" }, 503);
      return c.json({ session, account_id: r.account_id, created: r.created, linked: r.linked });
    } catch (e) {
      // F18: the closed-account notice on the OAuth door, ahead of the uniform
      // OidcError refusal — the provider's id token verified, so control is
      // proved and the ground and appeal route are owed.
      if (e instanceof OidcAccountClosedError) return c.json(await closedSignInBody(c.env, e), e.status);
      if (e instanceof OidcError) return c.json(errorBody(e), e.status);
      return serverFault(c, "/account/oauth/:provider/signin", e);
    }
  });

  // Logout everywhere — bump the account's session epoch so every outstanding
  // session (which carries the older epoch) stops verifying. The caller then
  // signs in again to get a fresh session.
  //
  // SESSIONS ONLY, DELIBERATELY, AND THE COPY NOW SAYS SO. Every client used to
  // sit this button under "“Everywhere” revokes every signed-in device — use it
  // if a device is lost", and the lost-device case is the one thing it cannot
  // do: the CREDENTIAL on the lost phone is untouched, so a usernameless
  // assertion signs it straight back in at the new epoch. The fix is not to make
  // this route revoke credentials —
  //
  //   • it would strand every passkey-only account at zero sign-in methods, the
  //     exact state removeFanEmail refuses to create; and
  //   • the two acts are not the same act. A session bump is CHEAP AND
  //     REVERSIBLE (sign in again and you are back), which is what makes it the
  //     right answer for "tidy up my sessions" and for a suspected token leak. A
  //     revoke is IRREVERSIBLE for that credential and needs its own naming, its
  //     own confirmation and its own refusal to strand the account.
  //
  // So the lost-device case gets its own explicit control (POST
  // /account/fan-passkey/revoke below), and that one carries the epoch bump
  // INSIDE it — because revoking a credential without severing the sessions it
  // already minted is not a severing either.
  app.post("/account/logout-all", async (c) => {
    const who = await fanCaller(c);
    if (who.kind !== "live") return c.json({ error: "unauthorized" }, 401);
    // I41-FANDRAFT (D-1193(9) RGA-03): it revokes recently added passkeys, so it needs the fresh sign-in of
    // POST /account/fan-passkey/revoke, before any write; a stale session changes nothing.
    // I43-AUTHMSG (D-1245(2) a): its own code and Jason's sentence; the fan is still signed in here.
    if (!freshFanSignIn(who.exp, Math.floor(Date.now() / 1000))) return c.json(refusal(LOGOUT_ALL_REAUTH_ERROR, LOGOUT_ALL_REAUTH_MESSAGE), 403);
    const id = who.id;
    // I3-0215 (D-0920 B): the epoch bump AND the revoke of passkeys added in the last 24 h, one batch.
    const r = await logoutAllFan(d1(c.env.TRUST_DB), id);
    return c.json({ ok: true, session_epoch: r.session_epoch, passkeys_revoked: r.passkeys_revoked });
  });

  // Account settings (fan-scoped): the email-alerts opt-out, the promotional
  // broadcast mute, and the AFFILIATION lever (audit item #12).
  //
  // The affiliation flag lives on `user_profiles`, so it arrives on a LEFT JOIN —
  // one statement, not a second read, and NULL when the fan has no profile (a fan
  // with no profile cannot be listed on anybody's page by construction, so `null`
  // is the honest answer and not a defaulted "public" the fan never chose).
  //
  // `email` is the address alert mail is actually SENT to (QUEUE #420, D-0693): the
  // fan's VERIFIED fan_email, else accounts.email — the same JOIN and COALESCE as
  // alerts.ts ALERT_ADDRESS_JOIN / ALERT_ADDRESS_SQL (not exported, so re-typed
  // here; b23-f2-settings-alert-address.test.ts pins the two to one answer).
  app.get("/account/settings", async (c) => {
    const id = await fanId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const row = await d1(c.env.TRUST_DB)
      .prepare(
        // #127 (D-0819(4)): the accounts.email fallback carries alerts.ts's
        // verified bar (ALERT_ADDRESS_JOIN / ALERT_ADDRESS_SQL, D-0817(2)): an
        // address only an UNVERIFIED provider claim vouches for is not shown.
        `SELECT COALESCE(fe.address, CASE WHEN ou.email_verified = 0 THEN NULL ELSE a.email END) AS email, a.email_alerts AS email_alerts, a.broadcasts_muted AS broadcasts_muted,
                a.email_report_outcomes AS email_report_outcomes, u.affiliation_visibility AS affiliation_visibility
           FROM accounts a LEFT JOIN user_profiles u ON u.account_id = a.account_id
           LEFT JOIN fan_email fe ON fe.account_id = a.account_id AND fe.verified_at IS NOT NULL
           LEFT JOIN oauth_identities ou ON ou.rowid = (SELECT o.rowid FROM oauth_identities o WHERE o.account_id = a.account_id AND o.email = a.email ORDER BY o.email_verified DESC LIMIT 1)
          WHERE a.account_id = ?`,
      )
      .bind(id)
      .first<{ email: string | null; email_alerts: number; broadcasts_muted: number; email_report_outcomes: number; affiliation_visibility: string | null }>();
    return c.json({
      email: row?.email ?? null,
      email_alerts: (row?.email_alerts ?? 1) === 1,
      // I3-2072 (D-0995(5)): the report-outcome email's own switch (0217, DEFAULT 1).
      email_report_outcomes: (row?.email_report_outcomes ?? 1) === 1,
      broadcasts_muted: (row?.broadcasts_muted ?? 0) === 1,
      affiliation_visibility: row?.affiliation_visibility ?? null,
    });
  });

  // "List me on creator pages" — the separable half of `visibility` (item #12).
  // Fan-session gated, the same shape and the same class of consent valve as
  // POST /account/settings/broadcasts one screen over.
  app.post("/account/settings/affiliation", anonBodyLimit, async (c) => {
    const id = await fanId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const body = (await c.req.json().catch(() => null)) as { affiliation_visibility?: unknown } | null;
    if (!body || !isAffiliationVisibility(body.affiliation_visibility)) return c.json({ error: "affiliation_visibility must be public or private" }, 400);
    try {
      const v = await setAffiliationVisibility(d1(c.env.TRUST_DB), id, body.affiliation_visibility);
      return c.json({ ok: true, affiliation_visibility: v });
    } catch (e) {
      if (e instanceof AffiliationError) return c.json(errorBody(e), e.status);
      return serverFault(c, "/account/settings/affiliation", e);
    }
  });

  // I3-0909 (I4-08-R8 W5): the fan's own switch for one creator's recognitions
  // (follow/like/thanks/shout-out/spotlight). Fan-session gated and self-scoped:
  // the account is the session's, never the body's. Idempotent both ways; the
  // creator is told nothing (their grant is refused as ordinary ineligibility).
  app.post("/account/recognition-blocks/:passport_id", anonBodyLimit, async (c) => {
    const id = await fanId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const body = (await c.req.json().catch(() => null)) as { blocked?: unknown } | null;
    if (!body || typeof body.blocked !== "boolean") return c.json({ error: "blocked (bool) required" }, 400);
    try {
      const r = await setRecognitionBlock(d1(c.env.TRUST_DB), id, c.req.param("passport_id"), body.blocked);
      return c.json({ ok: true, blocked: r.blocked });
    } catch (e) {
      if (e instanceof RecognitionError) return c.json(errorBody(e), e.status);
      return serverFault(c, "/account/recognition-blocks/:passport_id", e);
    }
  });

  // Opt in/out of email alerts (push is unaffected; the in-app pull always works).
  // I3-2072 (D-0995(5)): the same door carries the report-outcome email's own switch, `email_report_outcomes`
  // (accounts.email_report_outcomes, 0217; read by reports.ts emailReportersOnWarning). Either field alone, or both;
  // a present field must be a boolean. Each written field is answered back.
  app.post("/account/settings/email-alerts", anonBodyLimit, async (c) => {
    const id = await fanId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const body = (await c.req.json().catch(() => null)) as { enabled?: unknown; email_report_outcomes?: unknown } | null;
    const hasEnabled = body !== null && body.enabled !== undefined;
    const hasOutcomes = body !== null && body.email_report_outcomes !== undefined;
    if (!body || (!hasEnabled && !hasOutcomes) || (hasEnabled && typeof body.enabled !== "boolean")) return c.json({ error: "enabled (bool) required" }, 400);
    if (hasOutcomes && typeof body.email_report_outcomes !== "boolean") return c.json({ error: "email_report_outcomes (bool) required" }, 400);
    const db = d1(c.env.TRUST_DB);
    const writes = [
      ...(hasEnabled ? [db.prepare("UPDATE accounts SET email_alerts = ? WHERE account_id = ?").bind(body.enabled ? 1 : 0, id)] : []),
      ...(hasOutcomes ? [db.prepare("UPDATE accounts SET email_report_outcomes = ? WHERE account_id = ?").bind(body.email_report_outcomes ? 1 : 0, id)] : []),
    ];
    await db.batch(writes);
    return c.json({ ok: true, ...(hasEnabled ? { email_alerts: body.enabled } : {}), ...(hasOutcomes ? { email_report_outcomes: body.email_report_outcomes } : {}) });
  });

  // Data rights (GDPR/CCPA). Export = the fan's own data (access/portability);
  // delete = erasure. Both are strictly self-scoped to the authenticated account.
  app.get("/account/export", async (c) => {
    // `fanIdForDataRights`, not `fanId`: a suspended or terminated account keeps
    // access/portability. See the docblock at `route-helpers.ts` for why this
    // door and `POST /account/delete` are the only two.
    const id = await fanIdForDataRights(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const db = d1(c.env.TRUST_DB);
    if (!(await consumeRateLimit(db, `acctexport:${id}`, Date.now(), ACCOUNT_EXPORT_PER_ACCOUNT_HOUR, HOUR_MS))) {
      return c.json(refusal("rate limit exceeded"), 429);
    }
    return c.json(await exportAccount(db, id));
  });

  // Irreversible account deletion. Requires an explicit {confirm:true} so a stray
  // POST can't wipe an account; the session dies with the account (fanId rejects
  // once the row is gone), so every other device is logged out too.
  // What deleting takes with it, BEFORE the confirm. The three payment rails
  // have three different fates and only one of them is ours to end, so every
  // client renders all three from this one read. Rail WORD only — no store id,
  // no customer id, no amount, no period end (INV-025's fan-facing clause).
  app.get("/account/delete/preview", async (c) => {
    const id = await fanId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    return c.json(await deleteSubscriptionPreview(d1(c.env.TRUST_DB), id));
  });

  app.post("/account/delete", anonBodyLimit, async (c) => {
    // `fanIdForDataRights`, not `fanId`: a suspended or terminated account keeps
    // erasure. The second and last door with this reader.
    const id = await fanIdForDataRights(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const body = (await c.req.json().catch(() => null)) as { confirm?: unknown } | null;
    if (!body || body.confirm !== true) return c.json({ error: "confirmation required" }, 400);
    // CAPTURED BEFORE THE ERASURE, DELIBERATELY. Hono's `executionCtx` is a
    // GETTER that THROWS when the request carries none; it used to be read
    // after `deleteAccount` had committed, inside this route's try, so a fan
    // whose account was gone was answered `400 {"error":"error"}` and every
    // client told them "everything is still here". Nothing after the commit may
    // decide this response.
    const waitUntil = captureWaitUntil(c);
    try {
      const db = d1(c.env.TRUST_DB);
      const env = c.env as unknown as Record<string, unknown>;
      const [pubEnv, pubCtx] = analyticsPublishArgs(c);
      const r = await eraseAccountWithSubscriptions(db, id, {
        actorId: id,
        actorKind: "fan",
        env,
        waitUntil,
        deferredCancelAnswer: "pending",
        evidence: evidenceStore(c.env),
        nowMs: Date.now(),
        ...(pubEnv && pubCtx ? { pubEnv, pubCtx } : {}),
      });
      return c.json({ ok: true, deleted: r.deleted, store_subscriptions: r.store_subscriptions, stripe_subscription: r.stripe_subscription, stripe_cancel: r.stripe_cancel, subscriptions_unknown: r.subscriptions_unknown });
    } catch (e) {
      // Only the erasure itself can reach here now (401 unknown account, 409
      // recovery hold). On a 409 nothing was erased, so nothing was cancelled,
      // mailed or audited.
      if (e instanceof AccountError) return c.json(errorBody(e), e.status);
      return serverFault(c, "/account/delete", e);
    }
  });

  // --- M2: fan passkey-first auth (flag 11) ---
  // A brand-new fan joins with ONE passkey tap: the anonymous start issues an
  // unbound challenge and the finish creates the account + credential atomically
  // and mints the session. The SAME routes, hit with a valid fan session, add a
  // second passkey to that account instead (the challenge is bound to the
  // account at start and the binding is re-checked at finish) — presence of the
  // session is the join/add switch, no separate route. Fans have their OWN
  // tables (fan_passkeys / fan_webauthn_challenges); the creator passkey paths
  // are untouched. Optional email (verify + OTP fallback sign-in) below.

/**
 * M2 — fan passkey-first auth (flag 11). A fan's primary credential is a
 * passkey: the anonymous registration ceremony IS account creation (one tap,
 * no form), a signed-in fan can add a second passkey, and sign-in is a
 * usernameless assertion. An OPTIONAL email is the alerts + fallback-sign-in
 * channel, proven by a 6-digit OTP (limits pinned in otp-policy.ts).
 *
 * Email reconciliation (deliberate, do not merge): `accounts.email` is the
 * OAuth-provider-attested address captured at social sign-in — provider
 * METADATA, possibly an Apple private relay, never a sign-in factor.
 * `fan_email` is the address the fan EXPLICITLY added and proved control of
 * (verified_at) — only it can receive fallback sign-in OTPs. The two coexist;
 * neither overwrites the other.
 *
 * One OTP primitive, TWO PURPOSES (0015). Verify-email and fallback-sign-in are
 * the same "prove you hold this inbox" proof, so there is ONE fan_otp table —
 * but the row now records which act the code was mailed for and `redeemOtp`
 * demands the expected one. Purpose-blindness was harmless only by accident
 * while a replacement was destructive: the sign-in leg's stamp carried
 * `AND verified_at IS NULL`, so a cross-purpose redemption was a no-op. With a
 * PENDING address it is not. A `fan-otp-signin` code is mailed to the OLD,
 * already-verified address A; posted to `/account/email/verify` it would have
 * promoted the pending address B, so whoever can read A (or has stolen a
 * session and can make the victim's own inbox answer) could complete a change
 * to an address they do not control. The unauthenticated sign-in path still
 * only ever maps VERIFIED addresses to accounts (and only sends to them), so an
 * unverified address can neither receive nor redeem a sign-in code.
 *
 * REPLACING A VERIFIED ADDRESS IS A PENDING SWAP, NOT AN OVERWRITE (0015). See
 * `addFanEmail` and `redeemOtp` below: while a change is in flight the account
 * holds BOTH — the verified address goes on working (alerts, report outcomes,
 * fallback sign-in, the last-way-in guards) and the new one is proved in the
 * `pending_email` column. Nothing about the account's ways in changes until the
 * new address proves itself, and the promotion is one atomic statement.
 *
 * No account-existence oracle: BOTH unauthenticated legs (send and verify)
 * return identical surface results AND issue the identical sequence of D1
 * statements whether or not the address maps to an account. The mechanism is
 * the sentinel ids below — a miss is re-bound to an id that can never exist
 * and then runs the same code path, rather than returning early. Identical
 * responses over unequal database work is not "no oracle": a write is the most
 * expensive thing in these requests and was directly measurable. Nothing here
 * ever logs an address or a code.
 *
 * This module REUSES the webauthn.ts verifier primitives (no new crypto) and
 * writes only the fan tables — the creator passkey tables/paths are never
 * touched.
 */

/**
 * Null epoch → WHY. Called only where control has just been proved; everywhere
 * else the null is passed through as the uniform 401 it already was.
 *
 * Order matters: the closure lookup happens ONLY on the null branch, so a live
 * account pays nothing and a deleted account is still one statement — the extra
 * read is on the path that is already refusing.
 */
export async function explainNullEpoch(db: DbLike, accountId: string, fallback: string, env?: Record<string, unknown> | null): Promise<never> {
  const notice = await closureNotice(db, accountId, env);
  if (notice) {
    // D-0810(1): no data-rights subject under `account_compromise` — the
    // credential that just proved control is the one the closure says is not
    // the owner's. The owner exports after recovery resolves.
    const epoch = notice.category === "account_compromise" ? null : await accountSessionEpoch(db, accountId);
    throw new AccountClosedError(notice as unknown as Record<string, string>, epoch === null ? undefined : { accountId, epoch });
  }
  throw new FanAuthError(fallback, 401);
}

const defaultRandom = (): Uint8Array => crypto.getRandomValues(new Uint8Array(CHALLENGE_BYTES));

/** Constant-time compare of two hex digests (per-char XOR over the longer
 * length), the same primitive services/image/src/http.ts uses for its internal
 * secret. An OTP digest is not steerable by the attacker, so this is belt-and-
 * braces — but a secret comparison in this repo is never `===`. */
function timingSafeHexEqual(a: string, b: string): boolean {
  const len = Math.max(a.length, b.length);
  let diff = a.length === b.length ? 0 : 1;
  for (let i = 0; i < len; i++) {
    diff |= (a.charCodeAt(i) || 0) ^ (b.charCodeAt(i) || 0);
  }
  return diff === 0;
}

/** Uniform 6-digit code via rejection sampling (no modulo bias). */
function newOtpCode(): string {
  const buf = new Uint32Array(1);
  for (;;) {
    crypto.getRandomValues(buf);
    const v = buf[0]!;
    if (v < 4_294_000_000) return String(v % 1_000_000).padStart(6, "0");
  }
}

/** Validate a fan challenge against (purpose, expiry), then consume it
 * atomically (single-use: only the caller that stamps used_at proceeds).
 * Returns the challenge's account binding (null = anonymous ceremony). */
async function consumeFanChallenge(
  db: DbLike,
  challenge: string,
  purpose: "register" | "auth",
  nowMs: number,
  now: () => string,
): Promise<{ ok: boolean; accountId: string | null }> {
  if (!/^[A-Za-z0-9_-]{1,256}$/.test(challenge)) return { ok: false, accountId: null };
  const row = await db
    .prepare("SELECT challenge_id, account_id, purpose, created_at, used_at FROM fan_webauthn_challenges WHERE challenge = ?")
    .bind(challenge)
    .first<{ challenge_id: string; account_id: string | null; purpose: string; created_at: string; used_at: string | null }>();
  if (!row) return { ok: false, accountId: null };
  if (row.used_at !== null) return { ok: false, accountId: null };
  if (row.purpose !== purpose) return { ok: false, accountId: null };
  if (Date.parse(row.created_at) + FAN_CHALLENGE_TTL_MS <= nowMs) return { ok: false, accountId: null };
  const res = await db
    .prepare("UPDATE fan_webauthn_challenges SET used_at = ? WHERE challenge_id = ? AND used_at IS NULL")
    .bind(now(), row.challenge_id)
    .run();
  return { ok: res.changes === 1, accountId: row.account_id };
}

/** Cron prune: drop used or expired fan challenges (bounds the table). */
export async function pruneFanAuth(db: DbLike, nowMs: number): Promise<void> {
  const challengeCutoff = new Date(nowMs - FAN_CHALLENGE_TTL_MS).toISOString();
  await db.prepare("DELETE FROM fan_webauthn_challenges WHERE used_at IS NOT NULL OR created_at <= ?").bind(challengeCutoff).run();
  await pruneFanOtp(db, nowMs);
  // Stale unverified emails: the address squat expires (their pending OTPs
  // first — they reference the dying claim).
  const staleEmailCutoff = new Date(nowMs - FAN_EMAIL_UNVERIFIED_TTL_MS).toISOString();
  await db
    .prepare("DELETE FROM fan_otp WHERE account_id IN (SELECT account_id FROM fan_email WHERE verified_at IS NULL AND created_at <= ?)")
    .bind(staleEmailCutoff)
    .run();
  await db.prepare("DELETE FROM fan_email WHERE verified_at IS NULL AND created_at <= ?").bind(staleEmailCutoff).run();
  // THE SWEEP IS SPLIT, AND THE SPLIT IS THE POINT (0015). The DELETE above is
  // scoped to `verified_at IS NULL` and always was, so a row that has PROVED an
  // address is never swept — but an abandoned REPLACEMENT still has to expire,
  // or a pending claim would sit on the row forever. It expires on its own
  // clock (`pending_created_at`) and only the pending half goes: the verified
  // address, the row, and therefore every way in the account has, survive
  // untouched. A fan who starts a change and walks away is exactly where they
  // started, which is the whole property this migration exists for.
  await db
    .prepare("UPDATE fan_email SET pending_email = NULL, pending_created_at = NULL WHERE verified_at IS NOT NULL AND pending_created_at IS NOT NULL AND pending_created_at <= ?")
    .bind(staleEmailCutoff)
    .run();
}

  attestation: "none";
}

/** Begin a fan passkey registration. `accountId` null → the JOIN path (the

    attestation: "none",
  };
}

export interface FanRegisterResult {
  account_id: string;
  credential_id: string;
  created: boolean;
  session_epoch: number;
}

/**
 * Finish a fan passkey registration: verify the attestation with the REUSED
 * webauthn.ts primitives, consume the challenge single-use, then either create
 * the account + credential atomically (join path) or attach the credential to
 * the challenge's account (add path — the presented session MUST match the
 * account the challenge was bound to at start; an anonymous challenge must be
 * finished anonymously). Caller mints the session from the returned epoch.
 */
export async function finishFanRegister(
  db: DbLike,
  resp: AttestationResponse,
  /** The presented session: the account AND the epoch admission compared
   * against (`fanCaller`'s `ep`). The ADD insert binds that epoch (I12-RG-AUTH-06). */
  session: { accountId: string; ep: number } | null,
  cfg: WebAuthnConfig,
  nowMs: number,
  now: () => string = () => new Date().toISOString(),
  opts: { throttleOk?: (nowMs: number) => Promise<boolean> } = {},
): Promise<FanRegisterResult> {
  const sessionAccountId = session === null ? null : session.accountId;
  let clientData: ClientData, authData: AuthData, jwk: Es256Jwk;
  try {
    clientData = parseClientData(b64urlToBytes(resp.clientDataJSON));
    if (clientData.type !== "webauthn.create") throw new WebAuthnError("wrong clientData type");
    assertOriginAllowed(cfg.origins, clientData.origin);
    const att = decodeCbor(b64urlToBytes(resp.attestationObject));
    if (!(att instanceof Map)) throw new WebAuthnError("attestationObject not a map");
    const authDataBytes = att.get("authData");
    if (!(authDataBytes instanceof Uint8Array)) throw new WebAuthnError("authData missing");
    authData = parseAuthData(authDataBytes);
    if (!authData.up) throw new WebAuthnError("user-presence flag not set");
    if (!authData.uv) throw new WebAuthnError("user-verification required");
    if (!authData.at || !authData.cosePublicKey || !authData.credentialId) throw new WebAuthnError("no attested credential");
    if (!(await rpIdHashMatches(authData, cfg.rpId))) throw new WebAuthnError("rpIdHash mismatch");
    jwk = coseToEs256Jwk(authData.cosePublicKey).jwk;
  } catch (e) {
    if (e instanceof WebAuthnError) throw new FanAuthError(`registration rejected: ${e.message}`);
    throw new FanAuthError("registration rejected");
  }
  // Consume the challenge only after the response is structurally valid.
  const consumed = await consumeFanChallenge(db, clientData.challenge, "register", nowMs, now);
  // I3-2206: a stale/spent register challenge is RETRYABLE — 400 like every other
  // "registration rejected" answer on this door, never the 401 of a dead session.
  if (!consumed.ok) throw new FanAuthError("challenge invalid or expired", 400);
  // The account binding fixed at start must match the finish's auth context —
  // a challenge issued to one account can never enroll a key on another, and
  // an anonymous (join) challenge can't be replayed under someone's session.
  if (consumed.accountId !== sessionAccountId) throw new FanAuthError("challenge was not issued for this session", 401);

  const credId = bytesToB64url(authData.credentialId!);
  const ts = now();
  if (consumed.accountId === null) {
    // JOIN: throttle account creation (coarse per-IP backstop, edge is primary).
    if (opts.throttleOk && !(await opts.throttleOk(nowMs))) throw Object.assign(new FanAuthError("rate limit exceeded", 429), { ipKeyed: true }); // masked-IP key: route-helpers.ts errorBody
    const accountId = newId("acc");
    try {
      await db.batch([
        db.prepare(ACCOUNT_INSERT_SQL).bind(accountId, ts),
        db.prepare("INSERT INTO fan_passkeys (credential_id, account_id, public_key, sign_count, created_at, status) VALUES (?, ?, ?, ?, ?, 'active')").bind(credId, accountId, JSON.stringify(jwk), authData.signCount, ts),
      ]);
    } catch (e) {
      // credential_id is the PK — re-registering an existing credential is a
      // conflict (never a rebind), and the batch rolls the account back with it.
      // I3-0765: ONLY that collision; any other D1 failure propagates (a 500).
      if (isDuplicateCredential(e)) throw new FanAuthError("credential already registered", 409);
      throw e;
    }
    return { account_id: accountId, credential_id: credId, created: true, session_epoch: 0 };
  }
  // ADD: attach to the bound account under the active-passkey cap. The cap is
  // IN the insert, not a count read before it — one ceremony needs one
  // single-use challenge, but a fan can hold several at once, so a counted-then-
  // inserted cap is overshot by the width of the burst. `changes === 0` is the
  // cap; a zero-row insert returns NO rows and a duplicate credential still
  // fails the primary key and is reported as the rebind refusal it has always
  // been.
  // Non-null here (the binding check above), and -1 matches no epoch: fail closed.
  const admittedEp = session === null ? -1 : session.ep;
  let added: { results: { credential_id: string }[] };
  try {
    added = await db
      .prepare(
        "INSERT INTO fan_passkeys (credential_id, account_id, public_key, sign_count, created_at, status) " +
          "SELECT ?, ?, ?, ?, ?, 'active' WHERE (SELECT COUNT(*) FROM fan_passkeys WHERE account_id = ? AND status = 'active') < ? " +
          // I12-RG-AUTH-06: the session's ADMITTED epoch, re-asked in the insert —
          // a logout-everywhere after admission enrols nothing.
          "AND (SELECT session_epoch FROM accounts WHERE account_id = ?) = ? " +
          "RETURNING credential_id",
      )
      .bind(credId, consumed.accountId, JSON.stringify(jwk), authData.signCount, ts, consumed.accountId, MAX_FAN_PASSKEYS, consumed.accountId, admittedEp)
      .all<{ credential_id: string }>();
  } catch (e) {
    if (isDuplicateCredential(e)) throw new FanAuthError("credential already registered", 409);
    throw e;
  }
  // THE CAP COUNTS ACTIVE CREDENTIALS, and now that `revokeFanPasskey` exists
  // that is a cap a fan can actually get under. It was not: with no way to free
  // a slot, every lost device occupied one permanently, and at
  // MAX_FAN_PASSKEYS recovery's final and load-bearing step — "register a new
  // passkey now" — could never succeed again, leaving the account email-only
  // forever while ONLY_SIGNIN_METHOD correctly refused to remove the email. The
  // message says what to do about it rather than naming the wall; the clients
  // read the STATUS, so this sentence is the fan's only instruction here.
  // S-006 X1 (Z-01): THE CAP'S VERDICT IS THE RETURNED ROW, NOT A COUNT.
  // `fan_passkeys` bears fan_qualification_passkey_insert (migrations/0024:158-197),
  // and D1 reports `changes` as a total_changes() delta — trigger writes included —
  // so a row that landed reported 2 and this guard refused a device it had just
  // stored. `RETURNING` enumerates THIS statement's own rows: the cap's zero-row
  // case is `[]` (length 0 → the message below, unchanged), success is length 1.
  if (added.results.length !== 1 && (await currentSessionEpoch(db, consumed.accountId)) !== admittedEp) throw new FanAuthError("session ended — sign in again", 401);
  if (added.results.length !== 1) throw new FanAuthError(`that's ${MAX_FAN_PASSKEYS} passkeys — the most an account can hold. Remove one you no longer use from Sign-in & recovery, then add this device.`, 400);
  const epoch = await currentSessionEpoch(db, consumed.accountId);
  return { account_id: consumed.accountId, credential_id: credId, created: false, session_epoch: epoch ?? 0 };
}

export interface FanAuthResult {
  account_id: string;
  credential_id: string;
  session_epoch: number;
}

/** Finish a fan sign-in: verify the assertion against the credential's stored
 * key (single-use challenge consumed BEFORE trusting the signature, so a
 * captured assertion can never replay), enforce sign-count monotonicity the
 * same way the creator flow does, and return the account for session minting. */
export async function finishFanAuth(
  db: DbLike,
  resp: AssertionResponse,
  cfg: WebAuthnConfig,
  nowMs: number,
  now: () => string = () => new Date().toISOString(),
  env?: Record<string, unknown> | null,
): Promise<FanAuthResult> {
  let clientData: ClientData, authData: AuthData, cdjBytes: Uint8Array, adBytes: Uint8Array, sigBytes: Uint8Array;
  try {
    cdjBytes = b64urlToBytes(resp.clientDataJSON);
    adBytes = b64urlToBytes(resp.authenticatorData);
    sigBytes = b64urlToBytes(resp.signature);
    clientData = parseClientData(cdjBytes);
    if (clientData.type !== "webauthn.get") throw new WebAuthnError("wrong clientData type");
    assertOriginAllowed(cfg.origins, clientData.origin);
    authData = parseAuthData(adBytes);
    if (!authData.up) throw new WebAuthnError("user-presence flag not set");
    if (!authData.uv) throw new WebAuthnError("user-verification required");
    if (!(await rpIdHashMatches(authData, cfg.rpId))) throw new WebAuthnError("rpIdHash mismatch");
  } catch (e) {
    if (e instanceof WebAuthnError) throw new FanAuthError(`assertion rejected: ${e.message}`, 401);
    throw new FanAuthError("assertion rejected", 401);
  }
  if (typeof resp.credentialId !== "string" || !/^[A-Za-z0-9_-]{1,512}$/.test(resp.credentialId)) {
    throw new FanAuthError("assertion rejected", 401);
  }
  // I3-1711: THE CHALLENGE IS SPENT BEFORE THE CREDENTIAL IS LOOKED UP. The
  // lookup used to answer first, so a forged or stale challenge told an outsider
  // "unknown credential" vs "challenge invalid" for any credential id, for free.
  // Now a bad challenge is one answer for every id, and every existence answer
  // costs a real, single-use challenge.
  const consumed = await consumeFanChallenge(db, clientData.challenge, "auth", nowMs, now);
  if (!consumed.ok) throw new FanAuthError("challenge invalid or expired", 401);
  const row = await db
    .prepare("SELECT account_id, public_key, sign_count FROM fan_passkeys WHERE credential_id = ? AND status = 'active'")
    .bind(resp.credentialId)
    .first<{ account_id: string; public_key: string; sign_count: number }>();
  if (!row) throw new FanAuthError("unknown credential", 401);

  let ok = false;
  try {
    const key = await importEs256(JSON.parse(row.public_key) as Es256Jwk);
    const base = await assertionSignatureBase(adBytes, cdjBytes);
    ok = await verifyEs256(key, base, sigBytes);
  } catch {
    ok = false;
  }
  if (!ok) throw new FanAuthError("signature invalid", 401);

  // Signed-counter clone detection (same rule as the creator flow): if either
  // side is non-zero the counter must strictly increase; always-zero
  // authenticators are exempt.
  const newCount = authData.signCount;
  if (!(newCount === 0 && row.sign_count === 0) && newCount <= row.sign_count) {
    throw new FanAuthError("authenticator counter did not increase (possible clone)", 401);
  }
  // M31: the advance is a GUARDED UPDATE, not a read-then-write. The check above
  // is check-then-act — two concurrent assertions replayed from the same cloned
  // authenticator both read the same `row.sign_count`, both pass, and both write,
  // so the clone the counter exists to catch slips through. Binding the WHERE to
  // the exact counter we read serializes them at the row: the first advance wins
  // and the second matches zero rows (its stale counter no longer exists), which
  // is precisely the clone signal. Zero-counter authenticators (exempt) keep
  // sign_count = 0 on both, so their concurrent writes both match and both pass —
  // correct, since there is no counter to compare.
  const upd = await db
    .prepare("UPDATE fan_passkeys SET sign_count = ?, last_used_at = ? WHERE credential_id = ? AND sign_count = ? RETURNING credential_id")
    .bind(newCount, now(), resp.credentialId, row.sign_count)
    .all<{ credential_id: string }>();
  // S-006 X1 (Z-01): the CAS verdict is "did MY stale counter still match a row",
  // which is the enumerated row, not `changes` — fan_qualification_passkey_update
  // adds its own write to D1's count, which turned the one correctly-behaving
  // authenticator into a clone refusal for every fan at sign-in. The WHERE is
  // unchanged: a real replay still matches zero rows and is still refused here.
  if (upd.results.length !== 1) {
    throw new FanAuthError("authenticator counter did not increase (possible clone)", 401);
  }

  const epoch = await currentSessionEpoch(db, row.account_id);
  // The assertion verified against a credential bound to this account, so the
  // caller has proved control — and a closure is told, not hidden (F18).
  if (epoch === null) await explainNullEpoch(db, row.account_id, "unknown credential", env); // never returns
  return { account_id: row.account_id, credential_id: resp.credentialId, session_epoch: epoch as number };
}

// --- email (alerts + fallback) -----------------------------------------------

/** Issue an OTP for `accountId` under the pinned send limits and mail it.
 * Returns whether a send was attempted+accepted ("blocked" = a policy limit
 * suppressed it; callers on the no-oracle path must NOT surface which).
 * `mailer` null = issue-shaped work only, nothing is ever sent (the no-oracle
 * miss branch). */
/** A fan_otp row that is NOT a rotation (the key is inlined as a literal, so the
 * bind order of the statements that embed it is unchanged). */
const notRotation = (): string => `NOT EXISTS (SELECT 1 FROM rate_limits rk WHERE rk.rl_key = '${FAN_OTP_ROTATION_KEY}' || fan_otp.otp_id)`;
/** I3-0215: the rows that spend the send budget (cap + cooldown). A CONSUMED swap code is not one of them: it went
 * to the account's own verified inbox and its owner has already entered it, so it must not cooldown-block the
 * new-address code the same request then sends. An unconsumed swap code still counts. */
const budgetRow = (): string => `${notRotation()} AND NOT (fan_otp.purpose = '${PURPOSE_SWAP}' AND fan_otp.consumed_at IS NOT NULL)`;

async function issueOtp(
  db: DbLike,
  mailer: Mailer | null,
  accountId: string,
  address: string,
  /** The mailer template AND the row's `purpose` — one value, so the code a fan
   * is sent and the act it can complete can never drift apart. */
  template: FanOtpPurpose,
  ipBucketCount: number,
  nowMs: number,
  now: () => string,
  defer?: (p: Promise<unknown>) => void,
  /** I3-0964: the HMAC key for the stored code hash — the route passes c.env.SESSION_SECRET (otpCodeHash). */
  otpSecret?: unknown,
): Promise<
  | { status: "sent" }
  | { status: "send_failed" }
  | { status: "blocked"; reason: "cooldown" | "email_hourly_cap" | "ip_hourly_cap"; retryAfterSeconds?: number | undefined }
> {
  // Code generated + hashed BEFORE the limit decision, so a blocked issue
  // burns the same hash cost as an allowed one (constant-shape work for the
  // no-oracle caller; harmless extra work on the authed path).
  const code = newOtpCode();
  const hash = await otpCodeHash(otpSecret, "fan", code);
  const windowStart = new Date(nowMs - 60 * 60 * 1000).toISOString();
  const { results } = await db
    // #669: rotation codes (rotateFanOtpAtCeiling) never spend the request budget,
    // so a stranger's guessing cannot use up the owner's fresh request.
    // I3-1852: BOUNDED. The sentinel account collects one row per unknown-address
    // start from anyone, so an unbounded window read cost a miss more than a hit.
    // canSendOtp decides on `count >= cap` and the newest instant; the `cap`
    // newest rows decide both exactly as the whole window would.
    .prepare(`SELECT issued_at FROM fan_otp WHERE account_id = ? AND issued_at > ? AND ${budgetRow()} ORDER BY issued_at DESC LIMIT ?`)
    .bind(accountId, windowStart, SEND_HOURLY_CAP_PER_EMAIL)
    .all<{ issued_at: string }>();
  const decision = canSendOtp(results.map((r) => Date.parse(r.issued_at)), ipBucketCount, nowMs);
  // CONSTANT-SHAPE WRITE. The INSERT is UNCONDITIONAL — only its account
  // binding varies. A blocked issue (and the no-oracle miss branch, which
  // arrives here already bound to NULL_ACCOUNT_ID) writes an unreachable row
  // instead of skipping the write.
  //
  // Why: a D1 write is by far the most expensive operation in this request, so
  // "hit ⇒ 2 reads + 1 write, miss ⇒ 2 reads" was an account-existence oracle
  // that needed no latency calibration at all — POST the same address twice
  // ~5s apart and compare the PAIR (a real address gives slow-then-blocked; an
  // unknown address gives fast-then-fast). Now every branch does exactly
  // SELECT → INSERT, so the pair is uninformative.
  //
  // THE PER-EMAIL CEILING IS DECIDED IN THIS STATEMENT, not before it. The
  // binding used to be `decision.ok ? accountId : NULL_ACCOUNT_ID` — a value
  // computed from the SELECT above — which is check-then-act, and this is the
  // ONLY per-victim ceiling in the mail-cannon guard (the per-IP bucket bounds
  // the SENDER, not the recipient; an attacker with addresses to spare or a
  // pool of IPs is bounded by nothing else). Measured with a barrier holding
  // every racer on that SELECT (test/db-recorder.ts): 12 concurrent
  // `/fan/otp/start` posts for one verified address mailed 12 codes against a
  // cap of 5 — rate limiting on paper, an open relay pointed at a chosen inbox
  // in fact. The hourly count and the cooldown are therefore re-asserted
  // INSIDE the INSERT, against rows already committed, so N racers converge on
  // the cap instead of all reading zero.
  //
  // The numbers are otp-policy.ts's and are not restated here; `canSendOtp`
  // still runs, because it also carries the per-IP cap (a different table) and
  // supplies the pure module's reason. It is the FAST path — the SQL is the
  // authority, and which one blocked is read back below rather than assumed.
  // D-0823(2) (the creator door's D-0819(2) fix, creator-recovery.ts): the
  // hourly COUNT cap alone no longer refuses — the INSERT admits a request past
  // it while the account holds no live request code, so a stranger's five
  // requests cannot starve the owner. The per-IP cap and the cooldown still refuse.
  const usable = (decision.ok || decision.reason === "email_hourly_cap") && mailer !== null;
  const cooldownStart = new Date(nowMs - SEND_COOLDOWN_MS).toISOString();
  const bound = await db
    .prepare(
      "INSERT INTO fan_otp (otp_id, account_id, code_hash, issued_at, attempts, consumed_at, purpose, address) VALUES (?, " +
        "CASE WHEN ? = 1 " +
        `AND ((SELECT COUNT(*) FROM (SELECT 1 FROM fan_otp WHERE account_id = ? AND issued_at > ? AND ${budgetRow()} LIMIT ?)) < ? ` +
        // Past the cap only while no request code is live (unconsumed, under its
        // attempt cap, inside its TTL): one per TTL, so an inbox takes at most
        // 5 + 6 request codes an hour and the owner always holds a live one.
        `OR NOT EXISTS (SELECT 1 FROM fan_otp WHERE account_id = ? AND consumed_at IS NULL AND attempts < ? AND issued_at > ? AND ${notRotation()})) ` +
        `AND NOT EXISTS (SELECT 1 FROM fan_otp WHERE account_id = ? AND issued_at > ? AND ${budgetRow()}) ` +
        "THEN ? ELSE ? END, ?, ?, 0, NULL, ?, ?) RETURNING account_id",
    )
    .bind(
      newId("otp"),
      usable ? 1 : 0,
      accountId,
      windowStart,
      SEND_HOURLY_CAP_PER_EMAIL,
      SEND_HOURLY_CAP_PER_EMAIL,
      accountId,
      OTP_MAX_ATTEMPTS,
      // Against the clock that STAMPS issued_at, as creator-recovery.ts does.
      new Date(Date.parse(now()) - OTP_TTL_MS).toISOString(),
      accountId,
      cooldownStart,
      accountId,
      NULL_ACCOUNT_ID,
      hash,
      now(),
      // A CONSTANT, BOUND — never branched on. The purpose is fixed by the
      // caller's act, not by anything about the address or the account, so the
      // write stays exactly the one shape the no-oracle argument rests on. The
      // per-account window and cooldown above stay purpose-BLIND on purpose:
      // the cap is a ceiling on mail to a recipient, and splitting it per
      // purpose would double it.
      template,
      // THE ADDRESS THIS CODE IS BEING MAILED TO (0025). Bound, like the
      // purpose, as one more constant of the caller's act — the statement text
      // is unchanged for every input, so the constant-shape / no-oracle
      // argument above is untouched. Recording it is what lets `redeemOtp`
      // refuse to stamp any OTHER address: the OTP was bound to
      // `(account_id, purpose)` and to nothing else, so a code mailed to B
      // stamped whatever `pending_email` happened to hold, and a refused
      // second change had already moved that column to C. See
      // migrations/0025_otp_address_binding.sql.
      address,
    )
    .first<{ account_id: string }>();
  // The row's OWN binding is the verdict: it is the sentinel whenever any
  // limit held, including one a racer consumed after this request read the
  // window. Sentinel rows are unreachable (see NULL_ACCOUNT_ID) and swept on
  // the same hourly cutoff, so a blocked issue still costs exactly one write —
  // the constant shape the no-oracle argument rests on.
  if (bound?.account_id !== accountId || mailer === null) {
    // Prefer the pure module's own reason; a race lost against concurrent
    // racers converging on the SQL-enforced cap (see comment above) can land
    // here with `decision.ok === true`, since the row — not the earlier
    // in-memory read — is the actual verdict. That race is always the
    // per-account ceiling (the only per-victim limit re-asserted inside the
    // INSERT), so it is reported as the hourly cap.
    return decision.ok
      ? { status: "blocked", reason: "email_hourly_cap" }
      : { status: "blocked", reason: decision.reason, retryAfterSeconds: decision.retryAfterSeconds };
  }
  if (defer) {
    // The send rides OFF the response path (waitUntil) — but a failed send of
    // an auth-critical mail is never silently dropped: the deferred chain
    // writes the operator audit line itself (account id only, never the
    // address or the code).
    defer(
      mailer
        .send(template, address, { code })
        .then(async (ok) => {
          // PSEUDONYMIZE AT WRITE TIME, because this row can be written AFTER
          // erasure: the send rides on `defer`, so a `deleteAccount` landing
          // between the send and its settlement leaves the sweep
          // (deleteAccount's `UPDATE audit_log SET item_id = 'del_' …`) already run and unable to reach this row.
          // Same `CASE … EXISTS (accounts)` and the same `del_<hex>` expression
          // index.ts logFrozenSettlement writes, so a post-erasure row is indistinguishable
          // from a tombstoned one. `actor_kind` is 'system' (I3-2161): no operator acted.
          if (!ok)
            await db
              .prepare(
                "INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) " +
                  "SELECT 'system:fan-auth', 'fan_auth:otp_send_failed', ?, ?, " +
                  "CASE WHEN EXISTS (SELECT 1 FROM accounts WHERE account_id = ?) THEN ? ELSE 'del_' || lower(hex(randomblob(16))) END, 'system'",
              )
              .bind(`${template} could not be sent (mailer disabled or provider rejected)`, now(), accountId, accountId)
              .run();
        })
        .catch(() => {}),
    );
    return { status: "sent" };
  }
  return (await mailer.send(template, address, { code })) ? { status: "sent" } : { status: "send_failed" };
}

/** Who is guessing on the sign-in leg (D-0648): the masked edge prefix the
 * route reads from CF-Connecting-IP (route-helpers `callerPrefix`), never the
 * body; plus the mailer/defer a rotation needs. */
export interface FanOtpCaller {
  prefix?: string | null | undefined;
  /** I3-1246: the address the sign-in caller typed; a sign-in code redeems only
   * if it was mailed to this address AND the account still holds it. */
  address?: string | undefined;
  mailer?: Mailer | null | undefined;
  defer?: ((p: Promise<unknown>) => void) | undefined;
}
const FAN_OTP_HOUR_MS = 60 * 60 * 1000;
/** One caller's hourly wrong-guess budget on the fan sign-in door: the creator
 * door's number (creator-recovery.ts CREATOR_RECOVERY_MISSES_PER_CALLER_HOUR),
 * one person's honest worst hour — every code an address can be sent, each
 * mistyped to its per-code limit. */
export const FAN_OTP_MISSES_PER_CALLER_HOUR = OTP_MAX_ATTEMPTS * SEND_HOURLY_CAP_PER_EMAIL;
export const FAN_OTP_MISS_BUCKET = "fotpmiss";
/** #669 per-code global ceiling: twice one caller's budget, so no lone caller
 * can rotate the owner's code; per-code success <= 50 / 10^6. */
export const FAN_OTP_GUESSES_PER_CODE = 2 * FAN_OTP_MISSES_PER_CALLER_HOUR;
/** Rotations per account per hour = the request door's hourly cap, so an inbox
 * takes at most 5 request + 5 rotation codes an hour. Past it a code at the
 * ceiling STAYS LIVE (D-0819(2) accepts: the caller meter bounds guessing). */
export const FAN_OTP_ROTATIONS_PER_ACCOUNT_HOUR = SEND_HOURLY_CAP_PER_EMAIL;
/** rate_limits keys (no migration): per-code compared-guess count, and the
 * marker that says a fan_otp row was minted by a rotation (so the request
 * door's window never counts it). */
export const FAN_OTP_GUESS_KEY = "fotpg:";
export const FAN_OTP_ROTATION_KEY = "fotprot:";
/** #678 EQUAL COST: the sign-in compare window (request codes + rotation codes an
 * account can hold live in an hour). Every redemption compares EXACTLY this many
 * slots and every sign-in miss writes EXACTLY this many charge rows, so a wrong
 * guess costs the same for an address with ten live codes, one, or no account. */
export const FAN_OTP_COMPARE_WINDOW = SEND_HOURLY_CAP_PER_EMAIL + FAN_OTP_ROTATIONS_PER_ACCOUNT_HOUR;
/** One FIXED rate_limits key per empty slot of the window (`fotppad:<slot>`,
 * never per account or per address, so the table grows by at most the window
 * and a pad row carries nothing about anyone). Outside the `fotpg:` prefix, so
 * the per-code charge still counts code rows only (D-0824(1)). */
export const FAN_OTP_PAD_KEY = "fotppad:";
/** The digest an empty slot is compared against; `matchInWindow` never lets it match. */
const FAN_OTP_DECOY_HASH = "0".repeat(64);
/** Compare `submittedHash` against every slot of the window — no early exit, a
 * decoy digest for an empty slot — and return the first real row that matched. */
function matchInWindow<R extends { code_hash: string }>(rows: readonly R[], submittedHash: string): R | undefined {
  let found: R | undefined;
  for (let i = 0; i < FAN_OTP_COMPARE_WINDOW; i++) {
    const r = rows[i];
    const hit = timingSafeHexEqual(r?.code_hash ?? FAN_OTP_DECOY_HASH, submittedHash);
    if (hit && r !== undefined && found === undefined) found = r;
  }
  return found;
}
/** Rotation rows of one account in a window; binds (rotation key, account, since). */
const FAN_OTP_ROTATIONS_SQL =
  "(SELECT COUNT(*) FROM fan_otp rf WHERE EXISTS (SELECT 1 FROM rate_limits rr WHERE rr.rl_key = ? || rf.otp_id) AND rf.account_id = ? AND rf.issued_at > ?)";
/** Shared METERED buckets for an absent/unresolved caller — never an exemption
 * (the creator door's NO_CALLER_PREFIX_BUCKET / UNATTRIBUTED_CALLER_BUCKET). */
function fanOtpMissBucket(prefix: string | null | undefined): string {
  const who = prefix === null || prefix === undefined ? "unattributed" : prefix === "" ? "noip" : prefix;
  return `${FAN_OTP_MISS_BUCKET}:${who}`;
}

/**
 * ROTATE AT THE CEILING (#669, D-0819(2)), the creator door's rotateAtCeiling
 * on fan_otp. Runs on every sign-in miss, known account or not (one batch
 * either way). One transaction: mint a sign-in code for the account's VERIFIED
 * address — only when a live code has reached FAN_OTP_GUESSES_PER_CODE, the
 * rotation budget is left and a mailer exists — mark it as a rotation, and
 * retire the ceiling rows bound to the minted id (no mint, no retirement). The
 * new code is mailed to that verified address and nowhere else.
 */
async function rotateFanOtpAtCeiling(db: DbLike, mailer: Mailer | null, accountId: string, nowMs: number, now: () => string, defer?: (p: Promise<unknown>) => void, otpSecret?: unknown): Promise<void> {
  const code = newOtpCode();
  const hash = await otpCodeHash(otpSecret, "fan", code);
  const otpId = newId("otp");
  const ts = now();
  const out = (await db.batch([
    db
      .prepare(
        "INSERT INTO fan_otp (otp_id, account_id, code_hash, issued_at, attempts, consumed_at, purpose, address) " +
          "SELECT ?, fe.account_id, ?, ?, 0, NULL, ?, fe.address FROM fan_email fe WHERE fe.account_id = ? AND fe.verified_at IS NOT NULL AND ? = 1 " +
          "AND EXISTS (SELECT 1 FROM fan_otp f JOIN rate_limits g ON g.rl_key = ? || f.otp_id WHERE f.account_id = ? AND f.purpose = ? AND f.consumed_at IS NULL AND f.attempts < ? AND f.issued_at > ? AND g.count >= ?) " +
          `AND ${FAN_OTP_ROTATIONS_SQL} < ? RETURNING address`,
      )
      .bind(
        otpId,
        hash,
        ts,
        PURPOSE_SIGNIN,
        accountId,
        mailer !== null ? 1 : 0,
        FAN_OTP_GUESS_KEY,
        accountId,
        PURPOSE_SIGNIN,
        OTP_MAX_ATTEMPTS,
        new Date(nowMs - OTP_TTL_MS).toISOString(),
        FAN_OTP_GUESSES_PER_CODE,
        FAN_OTP_ROTATION_KEY,
        accountId,
        new Date(nowMs - FAN_OTP_HOUR_MS).toISOString(),
        FAN_OTP_ROTATIONS_PER_ACCOUNT_HOUR,
      ),
    db.prepare("INSERT INTO rate_limits (rl_key, window_start, count) SELECT ? || otp_id, ?, 1 FROM fan_otp WHERE otp_id = ?").bind(FAN_OTP_ROTATION_KEY, ts, otpId),
    db
      .prepare(
        "UPDATE fan_otp SET consumed_at = ? WHERE account_id = ? AND purpose = ? AND consumed_at IS NULL AND otp_id <> ? " +
          "AND EXISTS (SELECT 1 FROM rate_limits g WHERE g.rl_key = ? || fan_otp.otp_id AND g.count >= ?) AND EXISTS (SELECT 1 FROM fan_otp m WHERE m.otp_id = ?)",
      )
      .bind(ts, accountId, PURPOSE_SIGNIN, otpId, FAN_OTP_GUESS_KEY, FAN_OTP_GUESSES_PER_CODE, otpId),
  ])) as Array<{ results?: { address: string }[] } | undefined>;
  const address = out[0]?.results?.[0]?.address;
  if (!address || mailer === null) return;
  const sent = mailer.send(PURPOSE_SIGNIN, address, { code }).then(async (ok) => {
    // Same audit line (and the same write-time pseudonymisation) issueOtp writes.
    if (!ok)
      await db
        .prepare(
          "INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) " +
            "SELECT 'system:fan-auth', 'fan_auth:otp_send_failed', ?, ?, " +
            "CASE WHEN EXISTS (SELECT 1 FROM accounts WHERE account_id = ?) THEN ? ELSE 'del_' || lower(hex(randomblob(16))) END, 'system'",
        )
        .bind(`${PURPOSE_SIGNIN} rotation could not be sent (mailer disabled or provider rejected)`, now(), accountId, accountId)
        .run();
  });
  if (defer) defer(sent.catch(() => {}));
  else await sent.catch(() => {});
}

/**
 * Redeem a code against the account's newest live OTP. Uniform false on every
 * failure mode.
 *
 * DEBIT-FIRST (the 5-attempt cap, for real). The previous shape read
 * `attempts = A`, decided, and then burned the attempt with
 * `UPDATE ... WHERE otp_id = ? AND attempts = A`. That is a TOCTOU: k
 * concurrent guesses all read the same A, exactly ONE update matches, and the
 * other k-1 return `changes = 0` and burn NOTHING. Measured: 20 concurrent
 * verifies with 20 distinct wrong codes left `attempts = 1`, where the same 20
 * sequentially left `attempts = 21`. The cap therefore bounded WAVES, not
 * guesses — 5 waves of unlimited width per code, 5 codes/hour, against a 10^6
 * space, for a full fan session (the email leg is a complete sign-in factor).
 *
 * So the increment now runs FIRST, is NOT guarded on `attempts`, and happens
 * BEFORE the code is compared: every request costs exactly one attempt whether
 * or not it wins any race. The policy then judges the state as of THIS
 * attempt, read back authoritatively. `otp-policy.ts` pins "dead after 5 wrong
 * attempts"; this is what makes that true of the system and not just the pure
 * function.
 *
 * Single-use consumption is unchanged and still atomic: the consume is guarded
 * on `consumed_at IS NULL` (the single-use fence) AND on the exact `attempts`
 * value we judged, so the row we consume is the row we approved.
 *
 * CONSTANT SHAPE. The debit and the re-read run even when no live code exists,
 * against a sentinel otp id no row can ever hold. Together with the sentinel
 * ACCOUNT id the caller passes on a lookup miss, that makes "known address,
 * wrong code" and "unknown address" issue the identical statement sequence —
 * previously 3 ops vs 1, an account-existence oracle sitting behind an
 * identical `401 {"error":"invalid_code"}`.
 */
/** Result of a redemption: `false` on every failure; on success `replaced` is the
 * verified address a SWAP just displaced (I3-1113), else null. */
type RedeemResult = false | { replaced: string | null };
async function redeemOtp(db: DbLike, accountId: string, code: string, purpose: FanOtpPurpose, nowMs: number, now: () => string, caller?: FanOtpCaller, otpSecret?: unknown): Promise<RedeemResult> {
  if (!isWellFormedOtp(code)) return false;
  const submittedHash = await otpCodeHash(otpSecret, "fan", code);
  // D-0648 / D-0819(2): ON THE UNAUTHENTICATED SIGN-IN LEG A WRONG GUESS IS
  // CHARGED TO THE CALLER, never to the fan's codes. The verify leg is
  // session-gated: its caller IS the owner, so it keeps the per-code debit below.
  // D-0821(2): the caller's unit is DEBITED HERE, before anything is compared,
  // by one atomic upsert whose verdict is the budget check — a refused debit
  // compares nothing, so N concurrent guesses from one caller are compared at
  // most FAN_OTP_MISSES_PER_CALLER_HOUR times. A matched code refunds its unit
  // below, so only wrong guesses cost.
  const signin = purpose === PURPOSE_SIGNIN;
  const missBucket = fanOtpMissBucket(caller?.prefix);
  if (signin && !(await consumeRateLimit(db, missBucket, nowMs, FAN_OTP_MISSES_PER_CALLER_HOUR, FAN_OTP_HOUR_MS))) return false;
  // AUTH-01 (dead-row shadowing — the DoS `creator-recovery.ts redeemCreatorOtp`
  // fixed first). Selecting only the newest unconsumed row lets a DEAD-BY-
  // ATTEMPTS row shadow a live one, and it is outsider-createable: the victim
  // requests a code, ≥60s later an attacker who knows the verified address
  // (sendFanOtp needs nothing else) requests another — mailed to the VICTIM, so
  // the attacker never reads it — and spends 5 wrong guesses on it. That row is
  // now exhausted but still `consumed_at IS NULL`, so it stays the newest and
  // every later redemption compares against ITS hash: the victim's real,
  // still-live code stops working with no error that explains why.
  //
  // So: the CANDIDATE SET, not "the newest row" — every row that is not yet
  // dead (unconsumed, under the attempt cap, inside its TTL) — and the
  // submitted code is compared against the whole set. A code always finds its
  // own row. `address` rides the same read (0025): the stamps below are gated
  // on it, so the code can only ever confirm the address it was actually
  // mailed to. The set is bounded by the per-account hourly ceiling, so this is
  // at most SEND_HOURLY_CAP_PER_EMAIL rows and one statement — the same cost
  // the single read had.
  const ttlCutoff = new Date(nowMs - OTP_TTL_MS).toISOString();
  const { results: candidates } = await db
    .prepare(
      "SELECT otp_id, code_hash, issued_at, attempts, consumed_at, address FROM fan_otp " +
        "WHERE account_id = ? AND purpose = ? AND consumed_at IS NULL AND attempts < ? AND issued_at > ? " +
        // I3-1246: ON THE SIGN-IN LEG A CODE IS BOUND TO ITS RECIPIENT. It must
        // have been mailed to the address being signed in with, and the account
        // must not hold a DIFFERENT address now — a code mailed to an address the
        // fan has since replaced dies with the replacement. Same statement for
        // both purposes (the verify leg's address binding is in its stamps), and
        // the sentinel has no fan_email row, so its shape is unchanged.
        "AND (? <> 'fan-otp-signin' OR (address = ? AND NOT EXISTS (SELECT 1 FROM fan_email fe WHERE fe.account_id = fan_otp.account_id AND fe.address <> fan_otp.address))) " +
        // I3-0215: a swap code proves the inbox the account holds VERIFIED now, and no other.
        "AND (? <> 'fan-email-swap' OR EXISTS (SELECT 1 FROM fan_email fs WHERE fs.account_id = fan_otp.account_id AND fs.verified_at IS NOT NULL AND fs.address = fan_otp.address)) " +
        "ORDER BY issued_at DESC, otp_id DESC LIMIT ?",
    )
    .bind(accountId, purpose, OTP_MAX_ATTEMPTS, ttlCutoff, purpose, caller?.address ?? "", purpose, FAN_OTP_COMPARE_WINDOW)
    .all<{ otp_id: string; code_hash: string; issued_at: string; attempts: number; consumed_at: string | null; address: string }>();
  // #669 PER-CODE GLOBAL CEILING (sign-in leg): every live code is charged one
  // compared guess in ONE statement with its own RETURNING verdict, and only the
  // charged rows are compared (INV-058), so N concurrent guesses cost N and a
  // code is compared at most FAN_OTP_GUESSES_PER_CODE times. A code already at
  // the ceiling is left out ONLY while a rotation can replace it; otherwise it
  // stays a candidate — a code is never retired unreplaced (no lockout).
  let live = candidates;
  if (signin) {
    const { results: charged } = await db
      .prepare(
        "INSERT INTO rate_limits (rl_key, window_start, count) " +
          "SELECT ? || otp_id, ?, 1 FROM fan_otp WHERE otp_id IN (SELECT value FROM json_each(?)) AND account_id = ? AND purpose = ? AND consumed_at IS NULL AND attempts < ? AND issued_at > ? " +
          // #678: one fixed pad key per empty slot, so the rows this statement
          // writes are FAN_OTP_COMPARE_WINDOW whatever the address holds.
          "UNION ALL SELECT ? || value, ?, 1 FROM json_each(?) WHERE true " +
          "ON CONFLICT(rl_key) DO UPDATE SET count = rate_limits.count + 1 " +
          `WHERE substr(rate_limits.rl_key, 1, ?) = ? OR NOT (rate_limits.count >= ? AND ? = 1 AND ${FAN_OTP_ROTATIONS_SQL} < ?) RETURNING rl_key`,
      )
      .bind(
        FAN_OTP_GUESS_KEY,
        now(),
        // D-0824(1): EXACTLY the rows this guess is compared against — the
        // candidate window read above (LIMIT 10) — never every live row of the
        // account: the NULL_ACCOUNT_ID sentinel holds one live row per
        // unknown-address start from any caller, and was swept whole per miss.
        JSON.stringify(candidates.map((r) => r.otp_id)),
        accountId,
        purpose,
        OTP_MAX_ATTEMPTS,
        ttlCutoff,
        FAN_OTP_PAD_KEY,
        now(),
        JSON.stringify(Array.from({ length: FAN_OTP_COMPARE_WINDOW - candidates.length }, (_, i) => candidates.length + i)),
        FAN_OTP_PAD_KEY.length,
        FAN_OTP_PAD_KEY,
        FAN_OTP_GUESSES_PER_CODE,
        caller?.mailer ? 1 : 0,
        FAN_OTP_ROTATION_KEY,
        accountId,
        new Date(nowMs - FAN_OTP_HOUR_MS).toISOString(),
        FAN_OTP_ROTATIONS_PER_ACCOUNT_HOUR,
      )
      .all<{ rl_key: string }>();
    const chargedIds = new Set(charged.filter((r) => r.rl_key.startsWith(FAN_OTP_GUESS_KEY)).map((r) => r.rl_key.slice(FAN_OTP_GUESS_KEY.length)));
    live = candidates.filter((r) => chargedIds.has(r.otp_id));
  }
  // The row whose code this actually is, if any. On the SIGN-IN leg a miss
  // returns just below, before any fan_otp row is touched: its cost is the
  // caller's unit debited above (D-0648, D-0821(2)). Only the session-gated
  // VERIFY leg reaches `matched ?? live[0]`, where a miss debits the NEWEST live
  // row so a wrong guess still costs an attempt (same fallback as the creator leg).
  // #678: every slot of the window is compared (matchInWindow), not "until found".
  const matched = matchInWindow(live, submittedHash);
  // D-0821(2): a correct code gives back the unit it was debited above.
  if (signin && matched) await refundRateLimit(db, missBucket, nowMs, FAN_OTP_HOUR_MS);
  if (signin && !matched) {
    // THE MISS COSTS THE CALLER (D-0648): its unit was debited before the
    // compare; no fan_otp row is debited, so a stranger's guesses never kill the
    // code the owner holds.
    await rotateFanOtpAtCeiling(db, caller?.mailer ?? null, accountId, nowMs, now, caller?.defer, otpSecret);
    return false;
  }
  const row = matched ?? live[0];
  const otpId = row?.otp_id ?? NULL_OTP_ID;
  const debit = await db.prepare("UPDATE fan_otp SET attempts = attempts + 1 WHERE otp_id = ? AND consumed_at IS NULL").bind(otpId).run();
  const after = await db
    .prepare("SELECT code_hash, issued_at, attempts, consumed_at FROM fan_otp WHERE otp_id = ?")
    .bind(otpId)
    .first<{ code_hash: string; issued_at: string; attempts: number; consumed_at: string | null }>();
  if (!row || debit.changes !== 1 || !after || after.consumed_at !== null) return false;
  // `after.attempts` already counts THIS attempt; the policy is handed the
  // state as it stood immediately before it.
  const record: OtpRecord = { codeHash: after.code_hash, issuedAtMs: Date.parse(after.issued_at), attempts: after.attempts - 1, consumedAtMs: null };
  const outcome = verifyOtp(record, timingSafeHexEqual(after.code_hash, submittedHash), nowMs);
  if (outcome.outcome !== "consumed") return false;
  const res = await db
    .prepare("UPDATE fan_otp SET consumed_at = ? WHERE otp_id = ? AND consumed_at IS NULL AND attempts = ?")
    .bind(now(), otpId, after.attempts)
    .run();
  if (res.changes !== 1) return false; // lost the race — treat as dead (single-use)
  // THE ADDRESS CONTEST IS DECIDED HERE, not at claim time. An address is
  // exclusive only once VERIFIED (partial unique index, 0004_fan_accounts.sql),
  // so any number of accounts may hold an unverified claim on it and this stamp
  // is what one of them wins. A losing stamp violates the index and is reported
  // as the SAME uniform false a wrong code gets — which is what keeps
  // `POST /account/email` from answering differently for a registered address.
  // The OTP is already consumed above (single use), so the failure cannot be
  // replayed to probe, and the caller's unverified row simply stays unverified
  // until the sweep takes it.
  //
  // TWO STAMPS, BOTH PURPOSE-GUARDED AND ADDRESS-GUARDED IN SQL, AND THEY ARE
  // MUTUALLY EXCLUSIVE:
  //
  //   1. THE SWAP. A row that holds a PENDING replacement is promoted in ONE
  //      statement — the new address becomes the address, the pending columns
  //      clear, and `verified_at` is re-stamped together, so there is no
  //      instant at which the account has no verified address. If the pending
  //      address was verified by SOMEBODY ELSE in the meantime the partial
  //      unique index refuses the whole statement, nothing moves, and the
  //      failure is the same uniform `false` a wrong code gets — no oracle, and
  //      the fan's existing address is exactly as it was.
  //   2. FIRST ENROLMENT, unchanged: an unverified row becomes verified.
  //
  // A row can satisfy at most one (the swap sets `verified_at`, which is what
  // the second requires to be NULL), so `stamped` is 0 or 1 and never 2. Both
  // statements ALWAYS run and carry the purpose as a BOUND value compared
  // against a literal, so the sign-in leg issues the identical pair of
  // statements and both are no-ops for it — the constant shape the no-oracle
  // argument needs, and a defence in depth against a caller that asked for the
  // wrong purpose.
  //
  // AND EACH IS BOUND TO THE ADDRESS THE CODE WAS MAILED TO (0025, audit v11
  // D3). `otpAddress` comes off the row this call just CONSUMED, so a code can
  // only ever confirm its own recipient:
  //
  //   * the swap fires only while `pending_email` is still that address;
  //   * first enrolment fires only while `address` is still that address.
  //
  // 0015 closed this class on the `purpose` axis and left the address axis
  // open: the OTP recorded no recipient, so a code mailed to B stamped whatever
  // `pending_email` held at redemption time. `addFanEmail` moved that column
  // BEFORE asking whether a code could be sent and did not roll it back when
  // the answer was no, so a mistyped correction refused by the 60-second
  // cooldown left B's code pointing at C — and C became a full sign-in factor
  // nobody had proved control of.
  //
  // THE COMPARISON IS IN THE UPDATE, not in JS between the read and the write.
  // A JS check would be check-then-act: a concurrent `POST /account/email`
  // landing in that window moves the column after the check and before the
  // stamp. Same argument the upsert in `addFanEmail` makes for itself.
  const otpAddress = row.address;
  const stampTs = now();
  // D-0823(1): the swap row names WHICH address the swap replaced, by the same
  // per-account one-way tag the removal row carries (recovery.ts
  // fanEmailRemovalTag), so fanEmailClearEligibility can read a chain of swaps
  // and removals back to the address held before it. Read unconditionally and
  // hashed unconditionally (constant shape for the sign-in leg). Why the read
  // cannot go stale before the swap UPDATE fires: see the INSERT's note.
  const prior = await db.prepare("SELECT address FROM fan_email WHERE account_id = ?").bind(accountId).first<{ address: string }>();
  const priorAddress = prior?.address ?? "";
  const swapTag = await fanEmailRemovalTag(accountId, priorAddress);
  let stamped = 0;
  let swapped = false;
  try {
    // F-12 — A SWAP RETIRES THE ACCOUNT'S ONE-CLICK TOKENS, IN THE SWAP'S OWN
    // BATCH. Same reasoning as `digest.ts`'s creator-side confirm (0051, S-010
    // W0): `unsubscribe_tokens` for `fan_alerts` is keyed on `subject_id =
    // account_id`, not on an address (`unsubscribe.ts:172` — there is no
    // per-address key), so every unspent token is a capability over WHATEVER
    // address the account holds, and the mailbox it was mailed to is about to
    // stop being that address. Left alone, whoever still reads the OLD inbox —
    // no longer the fan's own way in — could later spend that token and turn
    // off `email_alerts` (including the was-this-you security notice) for the
    // account's NEW address, with no session and no proof of anything. Batched
    // with the swap itself so there is no instant with the new address live and
    // an old token still spendable; guarded on the same pending-address fact so
    // it only ever fires for a swap that actually happened. ONLY THE SWAP ARM
    // retires — first enrolment moves no mailbox, so it retires nothing (the
    // creator side's own rule, carried here).
    const armed = (await db.batch([
      db
        .prepare(
          "DELETE FROM unsubscribe_tokens WHERE stream IN ('fan_alerts', 'fan_report_outcomes', 'fan_newsletter') AND subject_id = ? " +
            "AND EXISTS (SELECT 1 FROM fan_email WHERE account_id = ? AND pending_email IS NOT NULL AND pending_email = ?)",
        )
        .bind(accountId, accountId, otpAddress),
      db
        .prepare(
          "UPDATE fan_email SET address = pending_email, pending_email = NULL, pending_created_at = NULL, verified_at = ?, created_at = ? " +
            "WHERE account_id = ? AND pending_email IS NOT NULL AND pending_email = ? AND ? = 'fan-email-verify' " +
            "RETURNING account_id",
        )
        .bind(stampTs, stampTs, accountId, otpAddress, purpose),
      // D-0817(1): THE SWAP IS RECORDED, in its own batch and bound to its
      // EFFECT: the row exists only if the UPDATE above just wrote this call's
      // stamp to both `verified_at` and `created_at` (a first enrolment never
      // re-stamps `created_at`; a refused swap rolls the batch back). recovery.ts
      // `fanEmailClearEligibility` reads it to tell a REPLACED email from a
      // first enrolment. D-0823(1): its detail carries the one-way tag of the
      // address the swap replaced (`swapTag`, read above) — the UPDATE fires only
      // while `pending_email` holds this code's address, and a row carrying a
      // pending change is a verified row whose `address` no other door rewrites
      // without clearing `pending_email`. Account id and tag only, never an address.
      db
        .prepare(
          "INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) " +
            "SELECT 'system:fan-auth', 'fan_email:swap', ?, ?, ?, 'system' " +
            "WHERE ? = 'fan-email-verify' AND EXISTS (SELECT 1 FROM fan_email WHERE account_id = ? AND created_at = ? AND verified_at = ?)",
        )
        .bind(`the account email was replaced by a confirmed change (${swapTag})`, stampTs, accountId, purpose, accountId, stampTs, stampTs),
    ])) as Array<{ results?: { account_id: string }[] } | undefined>;
    const swap = { results: armed[1]?.results ?? [] };
    const first = await db
      .prepare(
        "UPDATE fan_email SET verified_at = ? WHERE account_id = ? AND verified_at IS NULL AND address = ? AND ? = 'fan-email-verify' " +
          "RETURNING account_id",
      )
      .bind(stampTs, accountId, otpAddress, purpose)
      .all<{ account_id: string }>();
    // S-006 X1 (Z-01): THE SUM OF THE ENUMERATED ROWS, not of two `changes`.
    // Both statements set `verified_at`, so on D1 each fires
    // fan_qualification_email_{insert,update} and reports one more than it wrote
    // — a swap that committed summed to 2 and was reported to the fan as a wrong
    // or expired code while the address had already changed. The mutual
    // exclusivity argued above is unaffected: at most one statement matches, so
    // this sum is 0 or 1 and never 2.
    stamped = swap.results.length + first.results.length;
    swapped = swap.results.length === 1;
  } catch {
    return false;
  }
  // A VERIFY THAT CONFIRMED NOTHING IS NOT A VERIFY. The sign-in leg never
  // stamps (its address is already verified and it holds no pending claim), so
  // it does not ask. The verify leg does: if the fan cancelled the change, or
  // the row was swept, or someone else took the address, there is nothing this
  // code confirmed and saying "confirmed" would be a claim about a state that
  // does not exist.
  if (purpose === PURPOSE_VERIFY && stamped !== 1) return false;
  return { replaced: swapped && priorAddress !== "" ? priorAddress : null };
}

/** D-1004(1): an OTP send refused by the route's per-IP `fanmail` bucket (otp-policy.ts `ip_hourly_cap`, fed by
 * fan-account-routes.ts `fanIpBucketKey(c, "fanmail")`) is network-keyed: marked so route-helpers.ts errorBody answers the
 * per-network sentence. The per-address cap and the cooldown stay unmarked. */
function otpRefusal(e: FanAuthError): FanAuthError {
  return e.detail?.reason === "ip_hourly_cap" ? Object.assign(e, { ipKeyed: true }) : e;
}

/**
 * Add — or CHANGE — the account's alerts+recovery email: normalize + validate,
 * record the claim, and mail a verify OTP. AUTHENTICATED path — limit errors
 * may be surfaced here (the caller proved who they are; there is no oracle to
 * protect).
 *
 * THE TWO STATES, AND WHY THE SECOND EXISTS.
 *
 *   • NO VERIFIED ADDRESS YET (none, or one still unproven) → today's
 *     behaviour, unchanged: the row holds the claim UNVERIFIED and an unproven
 *     address signs nobody in.
 *   • A VERIFIED ADDRESS ALREADY → the new address goes to `pending_email` and
 *     `address`/`verified_at` are NOT TOUCHED. The fan keeps their way in for
 *     the whole of the change.
 *
 * The second state is the fix for audit item 27. This function used to run one
 * `ON CONFLICT(account_id) DO UPDATE SET address = excluded.address,
 * verified_at = NULL` — a de-facto REMOVAL of the verified address, performed
 * before a code was sent and before the new address had proved anything, with
 * NONE of the protection `removeFanEmail` applies to the very same act (it
 * refuses to strip an address unless ≥1 active passkey remains). The
 * consequences ran from "fallback sign-in is dead for both addresses" to "an
 * account with no passkey has no way in at all", and the fan was told nothing:
 * the destructive write happened BEFORE `issueOtp`, so a 429 (5/hour or the 60s
 * cooldown) or a 503 destroyed the address and mailed nothing, and the fan read
 * "That's enough codes for now". Writing the claim to a column that is not a
 * way in fixes that ordering bug for free — a blocked or undeliverable code now
 * costs a pending column and nothing else.
 *
 * NO REGISTRATION ORACLE. Every well-formed address gets the identical answer
 * and the identical work, whether or not somebody else already registered it —
 * see the insert below and `redeemOtp`, where the exclusivity actually lives.
 *
 * MAIL-CANNON NOTE. Fan accounts are free, anonymous and one passkey tap to
 * mint, and `ON CONFLICT(account_id) DO UPDATE` below lets ONE account rotate
 * through unlimited addresses — so the per-account 5/hour + 60s cooldown is
 * NOT a ceiling on attacker-chosen recipients. `ipBucketCount` therefore has
 * to be real here (it used to be hardcoded 0, which made
 * `SEND_HOURLY_CAP_PER_IP` inert on this route) and the route additionally
 * consumes an atomic per-IP bucket before we are reached. Uniformity did not
 * widen that surface: an already-registered address was ALREADY reachable by
 * this route (it just got a 409 instead of a code), and every free address
 * always was.
 */
export async function addFanEmail(
  db: DbLike,
  mailer: Mailer,
  accountId: string,
  rawAddress: string,
  nowMs: number,
  now: () => string = () => new Date().toISOString(),
  ipBucketCount = 0,
  defer?: (p: Promise<unknown>) => void,
  /** I3-0964: the HMAC key for the stored code hash — the route passes c.env.SESSION_SECRET (otpCodeHash). */
  otpSecret?: unknown,
  /** I3-0215 (D-0920 A): the code mailed to the CURRENT verified address; required to change it. */
  currentCode?: string,
): Promise<"confirm_current" | "sent"> {
  if (typeof rawAddress !== "string" || rawAddress.length > 254) throw new FanAuthError("that doesn't look like an email address");
  const address = rawAddress.trim().toLowerCase();
  if (!EMAIL_RE.test(address)) throw new FanAuthError("that doesn't look like an email address");
  const ts = now();
  // I3-0215 (D-0920 A) — A CHANGE OF A VERIFIED ADDRESS STARTS ONLY WITH A CODE FROM THAT ADDRESS. A stolen live
  // session used to be enough to put the thief's own inbox in `pending_email` and confirm it, taking the account's
  // recovery address. Without `currentCode` the code goes to the address the account holds verified and NOTHING
  // is written to `fan_email`; with it, the code is redeemed (bound to that address, redeemOtp) before the claim.
  // `gated` is carried into the claim's own statement below, so a verified address that appears between this read
  // and that write cannot take a pending change that never passed the gate.
  const current = await db
    .prepare("SELECT address FROM fan_email WHERE account_id = ? AND verified_at IS NOT NULL")
    .bind(accountId)
    .first<{ address: string }>();
  let gated = false;
  if (current && current.address !== address) {
    if (currentCode === undefined) {
      const swap = await issueOtp(db, mailer, accountId, current.address, PURPOSE_SWAP, ipBucketCount, nowMs, now, defer, otpSecret);
      if (swap.status === "blocked") {
        const message = swap.reason === "cooldown" ? "too many codes requested — wait a bit and ask again" : "too many codes requested this hour — try again later";
        throw otpRefusal(new FanAuthError(message, 429, { reason: swap.reason, retryAfterSeconds: swap.retryAfterSeconds }));
      }
      if (swap.status === "send_failed") throw new FanAuthError("verification email could not be sent", 503);
      return "confirm_current";
    }
    const proved = typeof currentCode === "string" ? await redeemOtp(db, accountId, currentCode.trim(), PURPOSE_SWAP, nowMs, now, undefined, otpSecret) : false;
    if (!proved) throw new FanAuthError("the code for your current address is wrong or expired", 401);
    gated = true;
  }
  // NO CONFLICT BRANCH, BY CONSTRUCTION. This insert never writes a VERIFIED
  // address — it writes an unverified `address` or a `pending_email`, and
  // `fan_email(address)` is unique only WHERE `verified_at IS NOT NULL`
  // (0004_fan_accounts.sql), with NO index at all on `pending_email` (0015).
  // So a claim on an address someone else already
  // registered lands exactly like a claim on a free one — same statement, same
  // outcome, same response. It used to fail the UNIQUE index and surface as a
  // 409 against the 200 a free address got, which made this endpoint an
  // email-registration oracle for anyone holding a free fan account, with the
  // address's owner never notified (review MEDIUM 7). The owner IS notified now:
  // the verification code goes to the address on every claim, so an unrequested
  // one is itself the "someone is trying to use your address" notice.
  //
  // The exclusivity did not disappear, it MOVED to the moment of proof —
  // `redeemOtp` is where a second claimant's stamp fails, uniformly, exactly
  // like a wrong code. That also retires the anti-squatting displacement this
  // block used to do: with unverified claims non-exclusive there is no lockout
  // to defend against, so no other account's pending claim (or its OTPs) is
  // destroyed by someone else typing the same address.
  //
  // THE CODE IS ISSUED FIRST, AND THE CLAIM IS ONLY RECORDED IF IT WENT OUT.
  //
  // This used to be the other way round, and the ordering was the bug (audit
  // v11, D3). The upsert moved `pending_email` with a bare `.run()` outside any
  // batch, and only THEN did `issueOtp` decide whether a code could be sent. A
  // refusal — the 60-second `SEND_COOLDOWN_MS`, the 5/hour cap, a provider 503 —
  // threw without putting the column back, so a mistyped correction two seconds
  // after a real change left the account pending on an address nothing had ever
  // been mailed to, holding a live code for the PREVIOUS one. The fan then typed
  // the code they actually received and it stamped the typo verified. That
  // address is a full sign-in factor (`POST /fan/otp/start` mails a session code
  // to it) which nobody proved control of, and the real recovery address is
  // gone.
  //
  // 0025 makes the stamp itself address-bound, so the code CANNOT confirm the
  // wrong address any more — that is the property that closes the class. This
  // ordering is the other half of the same fix, and it is what the reviewer
  // asked for in so many words: the pending write and the send outcome must not
  // be able to disagree. Now they cannot. `issueOtp` throws before anything
  // touches `fan_email`, so a blocked or undeliverable code costs exactly one
  // unreachable `fan_otp` row and NOTHING on the account.
  //
  // The failure direction is the safe one: if the process dies between the send
  // and the write, the fan holds a code for an address the row never learned
  // about, and the address-bound stamp refuses it. Fail closed, retry works.
  //
  // NO ORACLE IS INTRODUCED. This is the AUTHENTICATED path and the branch is on
  // the RATE LIMIT, which is a function of the caller's own recent sends and
  // their IP bucket — never of the address, never of whether anyone else has
  // registered it. A claim on a taken address still does the identical work and
  // gets the identical answer as a claim on a free one, which is the property
  // `redeemOtp`'s partial-index contest exists to hold.
  const sent = await issueOtp(db, mailer, accountId, address, PURPOSE_VERIFY, ipBucketCount, nowMs, now, defer, otpSecret);
  if (sent.status === "blocked") {
    // AS-127: distinguish the 60s cooldown from the 5/hour cap instead of
    // collapsing both into one message a fan can't act on — waiting a
    // minute only ever clears the cooldown.
    const message =
      sent.reason === "cooldown"
        ? "too many codes requested — wait a bit and ask again"
        : "too many codes requested this hour — try again later";
    throw otpRefusal(new FanAuthError(message, 429, { reason: sent.reason, retryAfterSeconds: sent.retryAfterSeconds }));
  }
  // `send_failed` is only reachable when the caller did NOT defer (direct
  // callers / tests). Deferred, an undeliverable code is an operator audit
  // line written by the deferred chain, not a 503 held open on a hung
  // provider — the route already 503s when mail is unconfigured at all.
  if (sent.status === "send_failed") throw new FanAuthError("verification email could not be sent", 503);

  // ONE STATEMENT DECIDES WHICH OF THE TWO STATES THIS IS, and it decides it
  // against the row as committed rather than against a value read a moment
  // earlier. A `SELECT verified_at` followed by a branch is check-then-act:
  // two concurrent posts (or a verify racing a change) could each read
  // "unverified" and both take the destructive branch. `verified_at` is not in
  // the SET list AT ALL, so no path through this statement can clear it.
  const claimed = await db
    .prepare(
      `INSERT INTO fan_email (account_id, address, verified_at, created_at, pending_email, pending_created_at) VALUES (?, ?, NULL, ?, NULL, NULL)
       ON CONFLICT(account_id) DO UPDATE SET
         address            = CASE WHEN fan_email.verified_at IS NULL THEN excluded.address     ELSE fan_email.address     END,
         created_at         = CASE WHEN fan_email.verified_at IS NULL THEN excluded.created_at  ELSE fan_email.created_at  END,
         pending_email      = CASE WHEN fan_email.verified_at IS NULL THEN NULL                 WHEN ? = 1 OR fan_email.address = excluded.address THEN excluded.address    ELSE fan_email.pending_email      END,
         pending_created_at = CASE WHEN fan_email.verified_at IS NULL THEN NULL                 WHEN ? = 1 OR fan_email.address = excluded.address THEN excluded.created_at ELSE fan_email.pending_created_at END
       RETURNING verified_at, address, pending_email`,
    )
    .bind(accountId, address, ts, gated ? 1 : 0, gated ? 1 : 0)
    .first<{ verified_at: string | null; address: string; pending_email: string | null }>();
  // The statement's own verdict: a verified row that did not take this address as its pending change refused it
  // (a verified address appeared after the gate read). Nothing was claimed; the fan starts the change again.
  if (claimed && claimed.verified_at !== null && claimed.address !== address && claimed.pending_email !== address) {
    throw new FanAuthError("your confirmed address changed while this was in flight — start the change again", 409);
  }
  return "sent";
}

/** AUTHENTICATED verify: redeem the mailed code against the caller's own
 * account; success stamps verified_at, or promotes a pending replacement over
 * the old address in one statement (both inside redeemOtp).
 *
 * ONLY A VERIFY-PURPOSE CODE. A `fan-otp-signin` code is mailed to the address
 * the account ALREADY holds, and before 0015 it would have been accepted here —
 * which, with a pending replacement on the row, means a code sent to the old
 * address could complete a change to a new one. */
export async function verifyFanEmail(
  db: DbLike,
  accountId: string,
  code: string,
  nowMs: number,
  now: () => string = () => new Date().toISOString(),
  /** I3-1113: sends the displaced-address notice; null = none configured (audited). */
  mailer: Mailer | null = null,
  defer?: (p: Promise<unknown>) => void,
  /** I3-0964: the HMAC key for the stored code hash — the route passes c.env.SESSION_SECRET (otpCodeHash). */
  otpSecret?: unknown,
): Promise<void> {
  const redeemed = typeof code === "string" ? await redeemOtp(db, accountId, code.trim(), PURPOSE_VERIFY, nowMs, now, undefined, otpSecret) : false;
  if (!redeemed) throw new FanAuthError("code is wrong or expired", 401);
  // I3-1113: A CONFIRMED SWAP TELLS THE ADDRESS IT DISPLACED. Whoever still reads
  // the old inbox is the one person who can say "that wasn't me", and the swap
  // itself was proved only from the NEW inbox. Same shape as the was-this-you
  // notice: off the response path, and never silently dropped — no mailer, a
  // refused send or a throw writes an audit row (account id only, never an
  // address, pseudonymised at write time like its siblings).
  if (redeemed.replaced === null) return;
  const displaced = redeemed.replaced;
  const whenUtc = new Date(nowMs).toISOString().replace("T", " ").replace(/\.\d+Z$/, " UTC");
  const notice = (async () => {
    const delivered = mailer ? await mailer.send("fan-email-replaced", displaced, { when_utc: whenUtc }).catch(() => false) : false;
    if (!delivered)
      await db
        .prepare(
          "INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) " +
            "SELECT 'system:fan-auth', 'fan_auth:email_replaced_notice_undeliverable', ?, ?, " +
            "CASE WHEN EXISTS (SELECT 1 FROM accounts WHERE account_id = ?) THEN ? ELSE 'del_' || lower(hex(randomblob(16))) END, 'system'",
        )
        .bind("the account email was replaced but the notice to the displaced address could not be sent", now(), accountId, accountId)
        .run();
  })().catch(() => {});
  if (defer) defer(notice);
  else await notice;
}

/**
 * UNAUTHENTICATED fallback sign-in, leg 1: maybe-send an OTP. Exists ONLY for
 * accounts with a VERIFIED fan_email. NO ACCOUNT-EXISTENCE ORACLE: identical
 * return (void) and the identical statement sequence — two reads and one write
 * to fan_otp — whether or not the address maps to a verified account.
 * Per-IP limiting happens at the route (atomic consumeRateLimit; uniform 429
 * independent of the address); `ipBucketCount` feeds the pinned policy check.
 */
export async function sendFanOtp(
  db: DbLike,
  mailer: Mailer,
  emailAddress: string,
  ipBucketCount: number,
  nowMs: number,
  now: () => string = () => new Date().toISOString(),
  defer?: (p: Promise<unknown>) => void,
  /** I3-0964: the HMAC key for the stored code hash — the route passes c.env.SESSION_SECRET (otpCodeHash). */
  otpSecret?: unknown,
): Promise<void> {
  const address = typeof emailAddress === "string" ? emailAddress.trim().toLowerCase() : "";
  if (!EMAIL_RE.test(address)) return; // shape-invalid: same silence as unknown
  const match = await db
    .prepare("SELECT account_id FROM fan_email WHERE address = ? AND verified_at IS NOT NULL")
    .bind(address)
    .first<{ account_id: string }>();
  // ONE code path for hit and miss. The miss runs the SAME issue sequence
  // bound to the sentinel account and with the mailer withheld, so both
  // branches perform exactly `SELECT fan_email` → `SELECT fan_otp` window →
  // `INSERT fan_otp` — the same statement kinds in the same order, one write
  // each. (Before: hit = 2 reads + 1 write, miss = 2 reads, and a
  // cooldown-blocked hit = 2 reads, so two posts ~5s apart told an attacker
  // whether the address had an account without any timing calibration.)
  //
  // The send itself is DEFERRED off the response path (waitUntil at the route)
  // so provider latency can never become an oracle either, and a limit-blocked
  // hit is swallowed as silently as a miss — surfacing it would reveal the
  // address has an account.
  const bound = match?.account_id ?? NULL_ACCOUNT_ID;
  await issueOtp(db, match ? mailer : null, bound, address, PURPOSE_SIGNIN, ipBucketCount, nowMs, now, defer, otpSecret);
}

export interface OtpSignInResult {
  account_id: string;
  session_epoch: number;
}

/**
 * UNAUTHENTICATED fallback sign-in, leg 2: redeem address+code → account.
 * Uniform null on every failure (unknown/unverified address, wrong, expired,
 * burned, replayed). On success fires the was-this-you security notice; when
 * the mailer cannot send it, an operator AUDIT line is written — a security
 * notice is never silently dropped. The audit line carries the account id,
 * never the address.
 */
export async function fanOtpSignIn(
  db: DbLike,
  mailer: Mailer,
  emailAddress: string,
  code: string,
  nowMs: number,
  now: () => string = () => new Date().toISOString(),
  defer?: (p: Promise<unknown>) => void,
  env?: Record<string, unknown> | null,
  /** The masked edge prefix of whoever is guessing (D-0648); undefined = a
   * direct module caller, metered in a shared bucket. */
  callerPrefix?: string | null,
  /** I3-0964: the HMAC key for the stored code hash — the route passes c.env.SESSION_SECRET
   * (otpCodeHash), the SAME value it passes `sendFanOtp`, so issue and redeem cannot disagree. */
  otpSecret?: unknown,
): Promise<OtpSignInResult | null> {
  const address = typeof emailAddress === "string" ? emailAddress.trim().toLowerCase() : "";
  if (!EMAIL_RE.test(address) || typeof code !== "string") return null;
  const match = await db
    .prepare("SELECT account_id FROM fan_email WHERE address = ? AND verified_at IS NOT NULL")
    .bind(address)
    .first<{ account_id: string }>();
  // The M2 no-oracle fix was never applied to THIS leg: the miss branch used
  // to hash `code.slice(0,16)` and return, equalizing the hash but not the DB
  // work — hit = SELECT fan_email + SELECT fan_otp + UPDATE attempts (3 ops,
  // one of them a write), miss = SELECT fan_email (1 op), both behind the same
  // `401 {"error":"invalid_code"}`. Now the miss runs the SAME redeem against
  // the sentinel account: identical statement kinds, identical order,
  // identical hash work, uniform null. (redeemOtp is itself constant-shape —
  // it debits and re-reads even when it found no row — so "known address with
  // no live code" is not a third shape either.)
  const bound = match?.account_id ?? NULL_ACCOUNT_ID;
  // SIGN-IN PURPOSE ONLY. Symmetrically to `verifyFanEmail`: a verify-purpose
  // code is one the fan asked for in order to CHANGE an address, and it must
  // not double as a session.
  const redeemed = await redeemOtp(db, bound, code.trim(), PURPOSE_SIGNIN, nowMs, now, { prefix: callerPrefix, mailer, defer, address }, otpSecret);
  if (!match || !redeemed) return null;
  // The was-this-you security notice rides OFF the response path (finding: a
  // hung provider otherwise stalls fan sign-in until the platform subrequest
  // limit fires). It is never silently dropped — the deferred chain writes the
  // operator audit line itself, carrying the account id and never the address.
  // M85: the template renders "signed in … at {{when_utc}}" and had been sent
  // with EMPTY vars, so the one fact the notice exists to convey — WHEN — came
  // out blank. Supply a readable UTC timestamp of this sign-in. `approx_location`
  // stays absent on purpose: it is an OPTIONAL Mustache section (the "from
  // around …" clause simply drops), and this codebase deliberately holds no
  // geo-IP to fill it.
  const whenUtc = new Date(nowMs).toISOString().replace("T", " ").replace(/\.\d+Z$/, " UTC");
  const notice = mailer
    .send("fan-was-this-you", address, { when_utc: whenUtc })
    .then(async (delivered) => {
      if (!delivered) {
        // PSEUDONYMIZE AT WRITE TIME — same reason as `issueOtp`'s `fan_auth:otp_send_failed` row: the
        // notice is deferred, so an erasure can land between the send and this
        // write, after deleteAccount's `UPDATE audit_log SET item_id = 'del_' …` sweep has already run.
        await db
          .prepare(
            "INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) " +
              "SELECT 'system:fan-auth', 'fan_auth:was_this_you_undeliverable', ?, ?, " +
              "CASE WHEN EXISTS (SELECT 1 FROM accounts WHERE account_id = ?) THEN ? ELSE 'del_' || lower(hex(randomblob(16))) END, 'system'",
          )
          .bind("email fallback sign-in succeeded but the security notice could not be sent (mailer disabled)", now(), match.account_id, match.account_id)
          .run();
      }
    })
    .catch(() => {});
  if (defer) defer(notice);
  else await notice; // direct callers (unit tests / cron) still settle it
  const epoch = await currentSessionEpoch(db, match.account_id);
  // A CORRECT CODE FOR A CLOSED ACCOUNT IS PROOF OF CONTROL TOO (F18). The
  // `?? 0` below would have minted a token whose epoch can never match, so the
  // session was already inert — fail-closed by construction — but the fan would
  // have been handed a "signed in" that 401s on every subsequent request with
  // nothing said. They are told instead.
  if (epoch === null) await explainNullEpoch(db, match.account_id, "unknown code", env);
  return { account_id: match.account_id, session_epoch: epoch ?? 0 };
}

/**
 * THE DESTRUCTIVE ACT ON THIS ROW, and the ONE place a way in may be taken
 * away. Two outcomes, decided by the row and not by the caller:
 *
 *   1. A CHANGE IS IN FLIGHT (`pending_email IS NOT NULL`) → the CHANGE is
 *      cancelled and nothing else happens. The verified address, and therefore
 *      every way in the account has, is untouched.
 *   2. OTHERWISE → remove the email → back to passkey-only. REQUIRES ≥1 active
 *      fan passkey (an account must never be stranded with zero sign-in
 *      methods) — the guard is folded into the DELETEs so a concurrent path
 *      can't strip the last method — and kills every pending OTP with it.
 *
 * WHY THE CANCEL IS THE FIRST OF THE TWO, and why it is the same act rather
 * than a second route. A fan looking at "confirmed A, waiting on B" has exactly
 * one destructive control on the screen, and the two things it could mean are
 * not equally safe: cancelling a change costs nothing and removing an address
 * can be the last way in. Ordering the harmless reading first means the
 * dangerous one is never reachable by accident from that state, and it is
 * ATOMIC — the cancel is a guarded UPDATE against the row as committed, so the
 * fall-through to the removal happens only when the database itself says there
 * was no change to cancel. The clients label the button for the state they
 * read, and read the account back afterwards rather than asserting what
 * happened (see `cancelPendingOutcome`).
 *
 * Removing the row takes any pending claim with it, so a cancelled-by-removal
 * change cannot outlive the address it was replacing.
 */
/** I3-1247: a linked Google/Apple identity is a way in (oidc.ts signs in by
 * (provider, sub) alone). Binds one `account_id`. Shared by both last-way-in guards. */
const OIDC_WAY_IN = "EXISTS (SELECT 1 FROM oauth_identities WHERE account_id = ? AND provider IN ('google', 'apple'))";

export async function removeFanEmail(
  db: DbLike,
  accountId: string,
  now: () => string = () => new Date().toISOString(),
  /** #934: tells the removed VERIFIED address it was removed; null = none configured (audited). */
  mailer: Mailer | null = null,
  defer?: (p: Promise<unknown>) => void,
  /** I16-RG-AUTH:4: the fan-email-swap code mailed to the VERIFIED address; required to remove it. */
  currentCode?: string,
  /** I3-0964: the HMAC key for the stored code hash — the route passes c.env.SESSION_SECRET (otpCodeHash). */
  otpSecret?: unknown,
  ipBucketCount = 0,
): Promise<"removed" | "pending_cancelled" | "confirm_current"> {
  const row = await db.prepare("SELECT 1 AS x FROM fan_email WHERE account_id = ?").bind(accountId).first();
  if (!row) throw new FanAuthError("no email on this account", 404);
  // Guarded on `pending_email IS NOT NULL`, so this is the row's own verdict
  // rather than a read acted on afterwards. No passkey guard: cancelling a
  // change cannot reduce the account's ways in — it restores the state the
  // account was in before the change was started.
  const cancelled = await db
    .prepare(
      "UPDATE fan_email SET pending_email = NULL, pending_created_at = NULL WHERE account_id = ? AND pending_email IS NOT NULL RETURNING account_id",
    )
    .bind(accountId)
    .all<{ account_id: string }>();
  // S-006 X1 (Z-01): THE CANCEL IS THE FIRST OF TWO ACTS ON ONE CONTROL AND THE
  // FALL-THROUGH IS THE DESTRUCTIVE ONE, so its verdict must be this statement's
  // own rows. On a row that was already VERIFIED the cancel fires
  // fan_qualification_email_update (its WHEN clause reads `verified_at`, which
  // this UPDATE does not touch), D1 reported 2, `=== 1` read that as "no change
  // to cancel", and the fan's verified address was DELETED by a cancel.
  if (cancelled.results.length === 1) return "pending_cancelled";
  // I16-RG-AUTH:4 (D-0920 A, D-0929(9)) — REMOVING A VERIFIED ADDRESS TAKES A CODE FROM THAT ADDRESS, exactly as
  // changing it does (addFanEmail): otherwise a stolen live session removes the owner's address and then adds and
  // verifies its own with no swap gate, because the gate keys on a CURRENT verified row. Without `currentCode` the
  // code goes to the verified address and nothing is removed; an unverified-only row was never proven and needs none.
  // I3-1247: a linked Google/Apple identity signs in on its own (oidc.ts), so it is a way in.
  const guard = `((SELECT COUNT(*) FROM fan_passkeys WHERE account_id = ? AND status = 'active') >= 1 OR ${OIDC_WAY_IN})`;
  const verified = await db.prepare("SELECT address FROM fan_email WHERE account_id = ? AND verified_at IS NOT NULL").bind(accountId).first<{ address: string }>();
  if (verified) {
    // A removal the guard would refuse anyway is refused before any code is mailed (the batch below re-checks it).
    const leaves = await db.prepare(`SELECT ${guard} AS ok`).bind(accountId, accountId).first<{ ok: number }>();
    if (!leaves?.ok) throw new FanAuthError("add a passkey before removing your email — an account must keep a sign-in method", 409);
    const nowMs = Date.parse(now());
    if (currentCode === undefined) {
      const swap = await issueOtp(db, mailer, accountId, verified.address, PURPOSE_SWAP, ipBucketCount, nowMs, now, defer, otpSecret);
      if (swap.status === "blocked") {
        const message = swap.reason === "cooldown" ? "too many codes requested — wait a bit and ask again" : "too many codes requested this hour — try again later";
        throw otpRefusal(new FanAuthError(message, 429, { reason: swap.reason, retryAfterSeconds: swap.retryAfterSeconds }));
      }
      if (swap.status === "send_failed") throw new FanAuthError("verification email could not be sent", 503);
      return "confirm_current";
    }
    const proved = typeof currentCode === "string" ? await redeemOtp(db, accountId, currentCode.trim(), PURPOSE_SWAP, nowMs, now, undefined, otpSecret) : false;
    if (!proved) throw new FanAuthError("the code for your current address is wrong or expired", 401);
  }
  // D-0820(5): the removal row names WHICH address it removed by a per-account
  // one-way tag (recovery.ts fanEmailRemovalTag), never the address; the INSERT
  // binds the address read here, so a swap landing in between writes no row.
  const current = await db.prepare("SELECT address FROM fan_email WHERE account_id = ? AND verified_at IS NOT NULL").bind(accountId).first<{ address: string }>();
  const removedTag = current ? await fanEmailRemovalTag(accountId, current.address) : "";
  await db.batch([
    // D-0819(1): THE REMOVAL IS RECORDED, in its own batch and under the SAME
    // guard, iff a VERIFIED email is removed (an unproven claim was never the
    // account's email). recovery.ts fanEmailClearEligibility reads it: a new
    // email enrolled within 30 days after it counts as a replacement. Account
    // id only, never an address. Runs first: the EXISTS reads the row the
    // DELETE below takes.
    db
      .prepare(
        "INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) " +
          "SELECT 'system:fan-auth', 'fan_email:removed', ?, ?, ?, 'system' " +
          `WHERE EXISTS (SELECT 1 FROM fan_email WHERE account_id = ? AND verified_at IS NOT NULL AND address = ?) AND ${guard}`,
      )
      .bind(`the account email was removed (${removedTag})`, now(), accountId, accountId, current?.address ?? "", accountId, accountId),
    db.prepare(`DELETE FROM fan_otp WHERE account_id = ? AND ${guard}`).bind(accountId, accountId, accountId),
    db.prepare(`DELETE FROM fan_email WHERE account_id = ? AND ${guard}`).bind(accountId, accountId, accountId),
    // F-12 — REMOVAL RETIRES THE SAME TOKENS, for the identical reason the swap
    // arm above does: an unspent `fan_alerts` one-click token is keyed on
    // `account_id`, not an address, so it survives the address it was mailed
    // under going away entirely and would otherwise let whoever still reads
    // that now-detached mailbox switch the account's email alerts off forever
    // with no session. Carries the SAME guard as the two deletes above, so it
    // fires exactly when — and only when — the removal itself does.
    db.prepare(`DELETE FROM unsubscribe_tokens WHERE stream IN ('fan_alerts', 'fan_report_outcomes', 'fan_newsletter') AND subject_id = ? AND ${guard}`).bind(accountId, accountId, accountId),
  ]);
  const still = await db.prepare("SELECT 1 AS x FROM fan_email WHERE account_id = ?").bind(accountId).first();
  if (still) throw new FanAuthError("add a passkey before removing your email — an account must keep a sign-in method", 409);
  // #934: the removed VERIFIED address is the one inbox that can say "that wasn't me" (a stolen session can remove
  // it) — the I3-1113 notice shape, on "removed" only, never for an unverified address. Off the response path;
  // an undeliverable notice is an operator-visible audit row, never silence, and never names the address.
  if (current) {
    const removedAddress = current.address;
    const whenUtc = new Date(Date.parse(now())).toISOString().replace("T", " ").replace(/\.\d+Z$/, " UTC");
    const notice = (async () => {
      const delivered = mailer ? await mailer.send("fan-email-removed", removedAddress, { when_utc: whenUtc }).catch(() => false) : false;
      if (!delivered)
        await db
          .prepare(
            "INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) " +
              "SELECT 'system:fan-auth', 'fan_auth:email_removed_notice_undeliverable', ?, ?, " +
              "CASE WHEN EXISTS (SELECT 1 FROM accounts WHERE account_id = ?) THEN ? ELSE 'del_' || lower(hex(randomblob(16))) END, 'system'",
          )
          .bind("the account email was removed but the notice to the removed address could not be sent", now(), accountId, accountId)
          .run();
    })().catch(() => {});
    if (defer) defer(notice);
    else await notice;
  }
  return "removed";
}

// --- device self-management (list + revoke a lost device) --------------------

/**
 * THE ONE ACT THAT SEVERS A DEVICE, and the reason this section exists at all.
 *
 * `fan_passkeys.status` has carried a `'revoked'` value since 0004 and NOTHING
 * in the product ever wrote it. A fan whose phone was lost had exactly one
 * control — `POST /account/logout-all` — and it bumps the session epoch and
 * nothing else, so the credential on the lost device was untouched: whoever
 * held the phone finished a usernameless assertion and came straight back at
 * the NEW epoch, while the fan was told every signed-in device had been
 * revoked. The only severing act a fan actually had was deleting the account.
 *
 * Two facts about a fan (as against a creator) decide the shape below.
 *
 * NO ASSERTION CEREMONY. `passkeys.ts revokePasskey` makes a creator PROVE
 * possession of a current passkey, because a creator's passkey is signing
 * authority over the ledger and S96 operator recovery exists behind it. Neither
 * holds here. A fan has no break-glass — `operatorResetPasskeys` is
 * creator-only — so a self-lockout is permanent; and the proof is
 * unsatisfiable in exactly the case this feature is FOR, because the only
 * credential the account has is the one on the lost phone (and at
 * MAX_FAN_PASSKEYS it is unsatisfiable forever). The authorization is therefore
 * the fan session, which is not a bare bearer: it is minted only by a passkey
 * assertion, a proven mailbox, or an OAuth sign-in. Decisively, that same
 * session may already `POST /account/delete`, which severs the thief by
 * destroying the account and everything in it — demanding MORE proof for the
 * proportionate act than for the catastrophic one is theatre, not defence.
 *
 * REVOKING SEVERS THE SESSIONS TOO. Killing the credential alone leaves the
 * thief's LIVE session working for the rest of its 30-day TTL, which is not a
 * severing at all — so the epoch bump and the status flip are one atomic batch
 * carrying the identical predicate: both apply or neither does. The caller's
 * own session dies with everyone else's, and the route mints them a fresh one
 * at the new epoch, so the fan doing the revoking stays in and every other
 * device — including the one they are afraid of — is out.
 *
 * AND IT CANNOT STRAND THE ACCOUNT. `removeFanEmail` above refuses to leave
 * zero sign-in methods; this is the same rule from the other side, and the two
 * have to be read together rather than each be right alone. A METHOD is an
 * active passkey OR a VERIFIED email (an unverified one signs nobody in), so:
 * the email may go while ≥1 passkey remains, and a passkey may go while another
 * passkey OR a verified email remains. Both guards live INSIDE their writes,
 * against rows already committed, so two concurrent removals cannot each see
 * the other's method and strip both.
 */

/** One enrolled device, as its owner may honestly be shown it. Dhwaja stores no
 * nickname, no platform and no AAGUID for a fan credential, so what the fan gets
 * is what the fan_passkeys row actually holds: when it was added and when it
 * last signed in. The screens say that out loud rather than implying a device
 * name Dhwaja does not have. */
export interface FanPasskeyInfo {
  credential_id: string;
  created_at: string;
  last_used_at: string | null;
}

/** The account's ACTIVE devices, oldest first (the same projection and order
 * `listPasskeys` gives a creator). Revoked rows are not listed: the list is
 * "what can sign in", and a row that cannot is noise on a screen a frightened
 * person is reading. */
export async function listFanPasskeys(db: DbLike, accountId: string): Promise<FanPasskeyInfo[]> {
  const { results } = await db
    .prepare("SELECT credential_id, created_at, last_used_at FROM fan_passkeys WHERE account_id = ? AND status = 'active' ORDER BY created_at, credential_id")
    .bind(accountId)
    .all<FanPasskeyInfo>();
  return results;
}

export interface FanRevokeResult {
  revoked: string;
  /** Active passkeys left on the account after this revoke. */
  remaining: number;
  /** The epoch every other session was just invalidated against. */
  session_epoch: number;
}

/**
 * Revoke ONE of the caller's own passkeys and sign every device out.
 *
 * NO CROSS-ACCOUNT EXISTENCE ORACLE. `credential_id` is a GLOBAL primary key,
 * so an unscoped read of it would answer "does this credential exist anywhere
 * in Dhwaja?" for any fan holding a free account. Every statement here is
 * scoped by `account_id`, and a credential that belongs to somebody else, one
 * that was already revoked, one of the wrong shape and one that never existed
 * are the SAME 404 — there is nothing in the answer to tell them apart.
 *
 * The refusal that IS distinguishable, deliberately, is the last-way-in 409:
 * it is a fact about the CALLER'S OWN account, told to a caller who has already
 * proved they hold it, and it has to be sayable or the screen degrades into
 * "something went wrong" for the one refusal a fan can act on.
 */
export async function revokeFanPasskey(
  db: DbLike,
  accountId: string,
  credentialId: string,
  now: () => string = () => new Date().toISOString(),
  /** I3-0983 c: the epoch the caller's session was ADMITTED at (route: `who.ep`). Absent, the epoch read just
   * before the revoke batch stands in for it. */
  admittedEpoch?: number,
): Promise<FanRevokeResult> {
  // Shape first, and it answers 404 rather than 400: the regex is public and
  // identical for everyone, so collapsing it into the miss costs nothing and
  // keeps ONE answer for every credential this account cannot revoke.
  if (typeof credentialId !== "string" || !/^[A-Za-z0-9_-]{1,512}$/.test(credentialId)) throw new FanAuthError("no such passkey on this account", 404);
  const target = await db
    .prepare("SELECT status FROM fan_passkeys WHERE credential_id = ? AND account_id = ?")
    .bind(credentialId, accountId)
    .first<{ status: string }>();
  if (!target || target.status !== "active") throw new FanAuthError("no such passkey on this account", 404);
  // THE JOINT GUARD, folded into BOTH writes rather than checked before them.
  // A count read then acted on is check-then-act, and this is the only ceiling
  // stopping an account being left with nothing to sign in with — two
  // concurrent revokes would both read "2 active" and strip the pair.
  // I3-0983 c (I22-W1-AUTH): the baseline the replacement session is judged against, fixed BEFORE the batch.
  const baseline = admittedEpoch ?? (await currentSessionEpoch(db, accountId)) ?? 0;
  const keepsAWayIn =
    "((SELECT COUNT(*) FROM fan_passkeys WHERE account_id = ? AND status = 'active') > 1 " +
    "OR EXISTS (SELECT 1 FROM fan_email WHERE account_id = ? AND verified_at IS NOT NULL) " +
    `OR ${OIDC_WAY_IN})`;
  // J-54 — THE SEVERING ACT LEAVES A ROW. Revoking a fan passkey signs every
  // device on the account out and can be the act that locks a thief out (or,
  // done by the thief, the act that strands the fan), and it recorded nothing:
  // no operator could answer "when did this account's devices change" after the
  // fact. The INSERT carries the IDENTICAL predicate as the two writes below,
  // in the same batch, so the row exists exactly when the revoke does — an
  // audit line for an act that did not happen is worse than none. It runs
  // BEFORE the status flip because its predicate reads `status = 'active'`.
  // `actor_kind` is `system` (auditStmt's sixth parameter, review.ts:131): the
  // actor is the fan's own session, not an operator, and filing it as an
  // operator decision is the exact drift F-31 closed.
  await db.batch([
    db
      .prepare(
        "INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) " +
          "SELECT 'system:fan-auth', 'fan_auth:passkey_revoked', ?, ?, ?, 'system' " +
          `WHERE EXISTS (SELECT 1 FROM fan_passkeys WHERE credential_id = ? AND account_id = ? AND status = 'active') AND ${keepsAWayIn}`,
      )
      .bind(`fan revoked credential ${credentialId}; every session on this account was signed out`, now(), accountId, credentialId, accountId, accountId, accountId, accountId),
    // The epoch bump runs SECOND and carries the IDENTICAL predicate, including
    // `status = 'active'` — inside one transaction it therefore either fires
    // alongside the revoke or not at all. Ordering it first also means the only
    // survivable partial state is "everyone signed out, credential alive",
    // which fails closed; the reverse would leave the thief's live session.
    db
      .prepare(
        `UPDATE accounts SET session_epoch = session_epoch + 1 WHERE account_id = ? ` +
          `AND EXISTS (SELECT 1 FROM fan_passkeys WHERE credential_id = ? AND account_id = ? AND status = 'active') AND ${keepsAWayIn}`,
      )
      .bind(accountId, credentialId, accountId, accountId, accountId, accountId),
    db
      .prepare(`UPDATE fan_passkeys SET status = 'revoked' WHERE credential_id = ? AND account_id = ? AND status = 'active' AND ${keepsAWayIn}`)
      .bind(credentialId, accountId, accountId, accountId, accountId),
  ]);
  // The ROW is the verdict, not a pre-check: read back what the transaction
  // actually did. A racer that revoked the same credential first also satisfies
  // the fan's request (and bumped the epoch doing it), so it is a success here.
  const after = await db
    .prepare("SELECT status FROM fan_passkeys WHERE credential_id = ? AND account_id = ?")
    .bind(credentialId, accountId)
    .first<{ status: string }>();
  if (after?.status !== "revoked") {
    throw new FanAuthError("that's the only way into this account — add another passkey, or confirm a backup email, before removing this one", 409);
  }
  const left = await db.prepare("SELECT COUNT(*) AS n FROM fan_passkeys WHERE account_id = ? AND status = 'active'").bind(accountId).first<{ n: number }>();
  // I3-0983 c: the replacement session is minted at the epoch THIS revoke produced (baseline + 1) only when that
  // is still the account's epoch; if anything else moved it (a sign-out-everywhere in the same moment), the
  // caller gets the dead baseline epoch — signed out with everyone else — never the newer epoch adopted.
  const epoch = (await currentSessionEpoch(db, accountId)) ?? 0;
  return { revoked: credentialId, remaining: left?.n ?? 0, session_epoch: epoch === baseline + 1 ? epoch : baseline };
}

/** I3-0215 (D-0920 B): how far back "sign out everywhere" reaches into the passkeys themselves. */
export const LOGOUT_ALL_PASSKEY_WINDOW_MS = 24 * 60 * 60 * 1000;

/**
 * I3-0215 (D-0920 B) — `POST /account/logout-all`. A stolen live session could ENROL its own passkey
 * (`/fan/passkey/register` takes the live session alone), and the epoch bump alone left that credential
 * working: the thief signed straight back in at the new epoch. So sign-out-everywhere also revokes every
 * fan passkey ADDED in the last LOGOUT_ALL_PASSKEY_WINDOW_MS, in the SAME batch as the epoch bump.
 *
 * It never strands the account (the rule `revokeFanPasskey` and `removeFanEmail` hold): the revoke carries
 * the predicate that a way in SURVIVES it — an active passkey older than the window, a verified email, or an
 * OIDC identity. Without one, the recent passkeys stay and `passkeys_revoked` is 0 (the sessions still end).
 * The audit row carries the identical predicate, so it exists exactly when the revoke does.
 */
export async function logoutAllFan(
  db: DbLike,
  accountId: string,
  now: () => string = () => new Date().toISOString(),
): Promise<{ session_epoch: number; passkeys_revoked: number }> {
  const ts = now();
  const cutoff = new Date(Date.parse(ts) - LOGOUT_ALL_PASSKEY_WINDOW_MS).toISOString();
  const recent = "account_id = ? AND status = 'active' AND created_at > ?";
  // I16-RG-AUTH:5: an email VERIFIED inside the window is as suspect as a passkey added inside it (a stolen session
  // can add and verify one), so it never counts as the surviving way in.
  const survivingWayIn =
    "(EXISTS (SELECT 1 FROM fan_passkeys o WHERE o.account_id = ? AND o.status = 'active' AND o.created_at <= ?) " +
    "OR EXISTS (SELECT 1 FROM fan_email WHERE account_id = ? AND verified_at IS NOT NULL AND verified_at <= ?) " +
    `OR ${OIDC_WAY_IN})`;
  // ...and it is removed with the recent passkeys whenever an active passkey or an OIDC identity remains AFTER the
  // passkey revoke (the statements below run in batch order), so a recent email that is the only way in stays.
  const recentEmail = "account_id = ? AND verified_at IS NOT NULL AND verified_at > ?";
  const wayInWithoutEmail = `(EXISTS (SELECT 1 FROM fan_passkeys p WHERE p.account_id = ? AND p.status = 'active') OR ${OIDC_WAY_IN})`;
  const out = (await db.batch([
    db
      .prepare(
        "INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) " +
          "SELECT 'system:fan-auth', 'fan_auth:logout_all_passkeys_revoked', " +
          `'fan signed out everywhere; ' || (SELECT COUNT(*) FROM fan_passkeys WHERE ${recent}) || ' passkey(s) added in the last 24 hours were revoked', ?, ?, 'system' ` +
          `WHERE EXISTS (SELECT 1 FROM fan_passkeys WHERE ${recent}) AND ${survivingWayIn}`,
      )
      .bind(accountId, cutoff, ts, accountId, accountId, cutoff, accountId, cutoff, accountId, cutoff, accountId),
    db
      .prepare(`UPDATE fan_passkeys SET status = 'revoked' WHERE ${recent} AND ${survivingWayIn} RETURNING credential_id`)
      .bind(accountId, cutoff, accountId, cutoff, accountId, cutoff, accountId),
    db
      .prepare(
        "INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) " +
          "SELECT 'system:fan-auth', 'fan_auth:logout_all_email_removed', " +
          "'fan signed out everywhere; an email verified in the last 24 hours was removed', ?, ?, 'system' " +
          `WHERE EXISTS (SELECT 1 FROM fan_email WHERE ${recentEmail}) AND ${wayInWithoutEmail}`,
      )
      .bind(ts, accountId, accountId, cutoff, accountId, accountId),
    db.prepare(`DELETE FROM fan_otp WHERE account_id = ? AND EXISTS (SELECT 1 FROM fan_email WHERE ${recentEmail}) AND ${wayInWithoutEmail}`).bind(accountId, accountId, cutoff, accountId, accountId),
    db
      .prepare(`DELETE FROM unsubscribe_tokens WHERE stream IN ('fan_alerts', 'fan_report_outcomes', 'fan_newsletter') AND subject_id = ? AND EXISTS (SELECT 1 FROM fan_email WHERE ${recentEmail}) AND ${wayInWithoutEmail}`)
      .bind(accountId, accountId, cutoff, accountId, accountId),
    db.prepare(`DELETE FROM fan_email WHERE ${recentEmail} AND ${wayInWithoutEmail}`).bind(accountId, cutoff, accountId, accountId),
    db.prepare("UPDATE accounts SET session_epoch = session_epoch + 1 WHERE account_id = ?").bind(accountId),
  ])) as Array<{ results?: { credential_id: string }[] } | undefined>;
  const epoch = await currentSessionEpoch(db, accountId);
  return { session_epoch: epoch ?? 0, passkeys_revoked: out[1]?.results?.length ?? 0 };
}

export interface FanAuthState {
  passkey_count: number;
  email: {
    address_masked: string;
    verified: boolean;
    /** The address a change is WAITING ON, masked the same way, or null when no
     * change is in flight. Present so the screen can say "confirmed A, waiting
     * on B — A still works until B is confirmed" instead of leaving the fan to
     * guess whether their way in survived. Masked for the same reason
     * `address_masked` is: a borrowed session must not be able to read either
     * address off the settings page. */
    pending_masked: string | null;
  } | null;
  has_acquisition: boolean;
}

/** Mask an address for display: first char + "…" + domain (never the full
 * local part — the settings page must not become an address oracle for a
 * borrowed session). */
function maskAddress(address: string): string {
  const at = address.indexOf("@");
  if (at <= 0) return "…";
  return `${address[0]}…@${address.slice(at + 1)}`;
}

export async function fanAuthState(db: DbLike, accountId: string): Promise<FanAuthState> {
  const count = await db.prepare("SELECT COUNT(*) AS n FROM fan_passkeys WHERE account_id = ? AND status = 'active'").bind(accountId).first<{ n: number }>();
  const email = await db
    .prepare("SELECT address, verified_at, pending_email FROM fan_email WHERE account_id = ?")
    .bind(accountId)
    .first<{ address: string; verified_at: string | null; pending_email: string | null }>();
  const acq = await db.prepare("SELECT 1 AS x FROM acquisitions WHERE account_id = ? LIMIT 1").bind(accountId).first();
  return {
    passkey_count: count?.n ?? 0,
    email: email
      ? {
          address_masked: maskAddress(email.address),
          verified: email.verified_at !== null,
          pending_masked: email.pending_email === null ? null : maskAddress(email.pending_email),
        }
      : null,
    has_acquisition: acq !== null,
  };
}

/**
 * CONNECTED FAN PASSPORT ROUTES (F8, master list #45).
 *
 * Every one of these is FAN-SESSION GATED and self-scoped: a fan links, lists,
 * unlinks and names their OWN connections and nobody else's. There is no
 * operator route (an operator has no business linking somebody's Twitch), no
 * creator route (a creator reads a COUNT through the eligibility predicate,
 * never a list), and — most importantly — NO PUBLIC ROUTE THAT TAKES A
 * PROVIDER ACCOUNT. Nothing here can be asked "which Dhwaja fan is this Twitch
 * id", in either direction, at any privilege level below database access.
 *
 * The public half of this feature is ONE NUMBER on `GET /u/:handle`
 * (`constellation`), served by `social.ts` under the profile's own visibility
 * rules.
 */
export function makeFanConnectionRoutes(): Hono<{ Bindings: Env }> {
  const app = new Hono<{ Bindings: Env }>();

  // What this deployment can actually complete, plus what the fan already has.
  // The two ride one response on purpose: a client that rendered the provider
  // list without the fan's existing links would offer a button that 409s.
  //
  // `available` names which provider apps the operator registered — a fact
  // about the DEPLOYMENT, not about any person, which is why it is served
  // plainly to any signed-in fan. `startFanConnection`'s single 503 for
  // "unsupported" and "unregistered" is not trying to hide it; it is there so
  // that ONE refusal covers both, rather than two refusals a prober could tell
  // apart on a route that writes.
  app.get("/account/connections", async (c) => {
    const id = await fanId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const db = d1(c.env.TRUST_DB);
    return c.json({
      connections: await listOwnConnections(db, id),
      available: availableProviders(c.env as unknown as Record<string, unknown>),
      all_providers: FAN_PROVIDER_KEYS,
      relink_cooldown_days: RELINK_COOLDOWN_DAYS,
    });
  });

  // Begin a link. Returns the provider's authorize URL; the daily ceiling is
  // enforced inside the INSERT (F6's lesson) and is keyed on the fan's own
  // session, so no stranger can spend it.
  app.post("/account/connections/:provider/start", async (c) => {
    const id = await fanId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    if (!(await fanIpAllowed(c, "fconn", RANKING_WRITES_PER_IP_HOUR))) return c.json(refusal("rate limit exceeded", RATE_LIMIT_MESSAGE), 429);
    try {
      const r = await startFanConnection(d1(c.env.TRUST_DB), c.env as unknown as Record<string, unknown>, id, c.req.param("provider"));
      return c.json(r);
    } catch (e) {
      if (e instanceof ConnectionError) return c.json(errorBody(e), e.status);
      return serverFault(c, "/account/connections/:provider/start", e);
    }
  });

  // Finish a link. The fan's session is required on THIS leg too — the callback
  // carries a `state` the challenge row binds to one account, and the row is
  // read with that account in the WHERE, so somebody else's state reads exactly
  // like one that never existed.
  //
  // The provider's raw redirect lands on the SPA, which posts the code HERE, so
  // no Dhwaja API route ever sees it in a URL. The redirect itself is still a
  // GET to the SPA origin — that is how OAuth works — so the code does reach
  // that origin's access log and could ride a `Referer` on a same-origin
  // subresource until the client strips it. `main.ts finishPendingConnection`
  // strips it with `history.replaceState` BEFORE the first render for exactly
  // that reason. The code is single-use and already spent by the time this
  // returns, so the window is narrow — but it is a window, and saying "there is
  // no way to leak it" would have been wrong.
  app.post("/account/connections/finish", anonBodyLimit, async (c) => {
    const id = await fanId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const raw = await c.req.text();
    if (raw.length > 4 * 1024) return c.json({ error: "too_large" }, 413);
    const body = ((): { state?: unknown; code?: unknown } | null => {
      try {
        return JSON.parse(raw) as { state?: unknown; code?: unknown };
      } catch {
        return null;
      }
    })();
    if (!body) return c.json({ error: "state and code required" }, 400);
    const r = await completeFanConnection(d1(c.env.TRUST_DB), c.env as unknown as Record<string, unknown>, id, body.state, body.code);
    // The outcome is named rather than collapsed: "already connected to another
    // account" and "held" are facts the person in front of the screen needs, and
    // both are about an account they just proved they control.
    return c.json(r, r.ok ? 200 : 409);
  });

  // Unlink. Always allowed — a connection a fan cannot withdraw is one they did
  // not really give — and it reserves the external account to them for the
  // cooldown, so this is not a way to walk one Twitch account up a fleet.
  app.post("/account/connections/:provider/unlink", async (c) => {
    const id = await fanId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const removed = await unlinkFanConnection(d1(c.env.TRUST_DB), id, c.req.param("provider"));
    return c.json({ ok: true, removed, relink_cooldown_days: RELINK_COOLDOWN_DAYS });
  });

  // The per-link opt-in: name this platform on my public profile, or don't.
  // Default off, and the fan can move it either way at any time.
  app.post("/account/connections/:provider/visibility", anonBodyLimit, async (c) => {
    const id = await fanId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const body = (await c.req.json().catch(() => null)) as { public?: unknown } | null;
    if (!body || typeof body.public !== "boolean") return c.json({ error: "public (boolean) required" }, 400);
    const changed = await setProviderPublic(d1(c.env.TRUST_DB), id, c.req.param("provider"), body.public);
    if (!changed) return c.json({ error: "you have not connected that platform" }, 404);
    return c.json({ ok: true, provider_public: body.public });
  });

  // The fan's own constellation, exactly as a stranger would see it. Owner-
  // scoped, and it exists so a fan can check what they are publishing about
  // themselves without having to open their own public profile in a private
  // window — the same reason `GET /account/demand-visibility` exists.
  app.get("/account/constellation", async (c) => {
    const id = await fanId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    return c.json(await constellation(d1(c.env.TRUST_DB), id));
  });

  return app;
}

/**
 * THE READ HALF of the Connected Fan Passport (F8, master list #45) — the
 * COUNT, and nothing that could ever reach an OAuth provider.
 *
 * ---------------------------------------------------------------------------
 * WHY THIS IS A SEPARATE FILE, AND IT IS NOT TIDINESS
 * ---------------------------------------------------------------------------
 * `social.ts` (the public `/u/:handle` page) and `eligibility.ts` (the F4
 * predicate) both need "how many platforms vouch for this fan". Both are inside
 * the VERDICT CLOSURE that `tests/invariants/paywall.test.ts` walks — a walk
 * that follows every static import edge to the end and takes NO exemptions of
 * any kind.
 *
 * The link machinery lives in `fan-connections.ts`, which imports `oauth.ts`,
 * which reads `env[\`OAUTH_${up}_CLIENT_ID\`]` — a COMPUTED env read. The
 * guard refuses those inside the closure, and it is right to: a computed key
 * is one a reviewer cannot rule out as an entitlement binding, and the whole
 * point of that layer is that a verdict path cannot see billing config even by
 * accident. Importing the write half from a public page would have dragged the
 * entire OAuth surface onto the verdict closure to serve one integer.
 *
 * So it is SPLIT, the `recognition-public.ts` / `fan-quality.ts` precedent: this
 * file imports the `DbLike` TYPE and nothing else, which makes it safe for the
 * closure to contain by construction rather than by review.
 *
 * DO NOT ADD AN IMPORT HERE. Not `oauth.ts`, not `fan-connections.ts`, not a
 * config reader. If a public surface needs something this file cannot compute
 * from the database alone, that is the signal to ask whether a public surface
 * should have it at all.
 */

/**
 * What a stranger reads on a fan's profile: how many INDEPENDENT external
 * platforms have vouched, by OAuth, for this being a person.
 *
 * `named` carries only the providers the fan opted to publish, PER LINK, and is
 * empty by default. Naming a platform somebody is on is a fact about them they
 * should have to volunteer; the count says the thing that matters to a stranger
 * deciding whether to trust a profile, and says it without naming anything.
 */
export interface Constellation {
  size: number;
  named: string[];
}

export async function constellation(db: DbLike, accountId: string): Promise<Constellation> {
  const { results } = await db
    .prepare("SELECT provider, provider_public FROM fan_connections WHERE account_id = ? ORDER BY provider")
    .bind(accountId)
    .all<{ provider: string; provider_public: number }>();
  return { size: results.length, named: results.filter((r) => r.provider_public === 1).map((r) => r.provider) };
}

/** How many platforms vouch for this fan. The `eligibility.ts` `connected` leaf
 * reads this, and reads NOTHING ELSE — the leaf cannot express "connected on
 * Twitch", deliberately (see `eligibility.ts`). */
export async function connectionCount(db: DbLike, accountId: string): Promise<number> {
  const r = await db.prepare("SELECT COUNT(*) AS n FROM fan_connections WHERE account_id = ?").bind(accountId).first<{ n: number }>();
  return r?.n ?? 0;
}
