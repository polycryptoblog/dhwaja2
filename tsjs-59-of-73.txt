/**
 * Passkey "touched-by" provenance — the ceremony layer + the ledger binding.
 *
 * The guarantee: an event carrying actor_kind='creator_passkey' (the "by
 * creator" mark) is appended ONLY through `appendCreatorSigned`, which requires
 * a WebAuthn assertion the server verified against a registered credential over
 * a fresh, single-use challenge BOUND to the exact drafts being appended. The
 * ledger's general path refuses creator_passkey outright (see appendEvents'
 * privileged-actor gate), so there is no other way to mint the creator mark.
 * The private key never leaves the creator's device → an agency can never
 * forge it.
 */

export const CHALLENGE_TTL_MS = 5 * 60 * 1000;
const CHALLENGE_BYTES = 32;

/** I3-0034 (Jason I11-Q7, D-1110(8)): the code mailed before a creator's FIRST passkey lives this many minutes —
 * the setting PASSKEY_FIRST_CODE_MINUTES, read by `firstPasskeyCodeTtlMs`; its register challenge expires with it. */
export const FIRST_PASSKEY_CODE_MINUTES_DEFAULT = 10;
/** D-1122(2): the setting's bounds, in minutes. */
export const FIRST_PASSKEY_CODE_MINUTES_MIN = 1;
export const FIRST_PASSKEY_CODE_MINUTES_MAX = 60;
const firstCodeSettingRefused = new Set<string>();
/** The setting's parse: a number of minutes in FIRST_PASSKEY_CODE_MINUTES_MIN..MAX (a number or a numeric string).
 * Absent -> the default. Anything else set -> the default, and one config warning per distinct refused value
 * (D-1122(2)), so a bad setting can never make the start answer 500 or live past an hour. */
export function firstPasskeyCodeTtlMs(raw: unknown): number {
  const m = typeof raw === "number" ? raw : typeof raw === "string" && raw.trim() !== "" ? Number(raw) : Number.NaN;
  if (Number.isFinite(m) && m >= FIRST_PASSKEY_CODE_MINUTES_MIN && m <= FIRST_PASSKEY_CODE_MINUTES_MAX) return m * 60_000;
  const shown = String(raw).slice(0, 40);
  if (raw !== undefined && raw !== null && !firstCodeSettingRefused.has(shown)) {
    firstCodeSettingRefused.add(shown);
    logWarn({ cid: "config", hop: "http", msg: "passkeys: PASSKEY_FIRST_CODE_MINUTES is outside 1..60 or not a number; using the default", env_key: "PASSKEY_FIRST_CODE_MINUTES", configured: shown, using: FIRST_PASSKEY_CODE_MINUTES_DEFAULT });
  }
  return FIRST_PASSKEY_CODE_MINUTES_DEFAULT * 60_000;
}
/** The lifetime the first-passkey mail states (`{{lifetime}}`, ops/email/templates/creator-first-passkey.*): the

/** OTP_DIGITS decimal digits, uniform (rejection sampling: 4294 * 10^6 is the largest multiple of 10^6 under 2^32). */
function newFirstPasskeyCode(): string {
  const space = 10 ** OTP_DIGITS;
  const limit = Math.floor(0x1_0000_0000 / space) * space;
  for (;;) {
    const v = crypto.getRandomValues(new Uint32Array(1))[0]!;
    if (v < limit) return String(v % space).padStart(OTP_DIGITS, "0");
  }
}

async function createChallenge(
  db: DbLike,
  passportId: string,
  purpose: "register" | "assert",
  actionHash: string | null,
  nowMs: number,
  now: () => string,
  randomBytes: () => Uint8Array,
  /** The bucket this challenge is charged to: the passport's, except on the
   * anonymous sign-in start (D-0785), which passes its caller's own bucket. */
  meter: { key: string; cap: number; ipKeyed?: true } = { key: `pkchal:${passportId}`, cap: PASSKEY_CHALLENGES_PER_HOUR },
  /** I3-0034: the first-enrolment challenge lives the code's lifetime and its action_hash is derived from it. */
  first?: { ttlMs: number; hashFor: (challenge: string) => Promise<string> },
): Promise<string> {
  // #482: metered BEFORE the INSERT, so a refusal writes no row.
  if (!(await consumeRateLimit(db, meter.key, nowMs, meter.cap, PASSKEY_CHALLENGE_WINDOW_MS))) {
    // D-1004(1): a meter keyed on the caller's network alone (the sign-in start) answers the per-network sentence (route-helpers.ts errorBody).
    throw meter.ipKeyed ? Object.assign(new PasskeyChallengeRateLimitError(), { ipKeyed: true }) : new PasskeyChallengeRateLimitError();
  }
  const challenge = randomChallenge(randomBytes);
  const expires = new Date(nowMs + (first ? first.ttlMs : CHALLENGE_TTL_MS)).toISOString();
  await db
    .prepare("INSERT INTO passkey_challenges (challenge, passport_id, purpose, action_hash, created_at, expires_at, consumed) VALUES (?, ?, ?, ?, ?, ?, 0)")
    .bind(challenge, passportId, purpose, first ? await first.hashFor(challenge) : actionHash, now(), expires)
    .run();
  return challenge;
}

/** Validate a challenge against (passport, purpose, action, expiry) then consume
 * it atomically (single-use). Returns true only if all pass AND this call is the
 * one that consumed it. */
async function consumeChallenge(
  db: DbLike,
  challenge: string,
  passportId: string,
  purpose: "register" | "assert",
  actionHash: string | null,
  nowMs: number,
): Promise<boolean> {
  if (!/^[A-Za-z0-9_-]{1,256}$/.test(challenge)) return false;
  const row = await db
    .prepare("SELECT passport_id, purpose, action_hash, expires_at, consumed FROM passkey_challenges WHERE challenge = ?")
    .bind(challenge)
    .first<{ passport_id: string; purpose: string; action_hash: string | null; expires_at: string; consumed: number }>();
  if (!row) return false;
  if (row.consumed === 1) return false;
  if (row.passport_id !== passportId) return false;
  if (row.purpose !== purpose) return false;
  if ((row.action_hash ?? null) !== (actionHash ?? null)) return false;
  if (Date.parse(row.expires_at) <= nowMs) return false;
  // Atomic single-use: only the caller that flips 0→1 proceeds.
  const res = await db.prepare("UPDATE passkey_challenges SET consumed = 1 WHERE challenge = ? AND consumed = 0").bind(challenge).run();
  return res.changes === 1;
}

/** Cron prune: drop consumed or expired challenges so the table can't grow
 * unbounded (a challenge is only useful until consumed or until it expires). */
export async function prunePasskeyChallenges(db: DbLike, nowMs: number): Promise<number> {
  const nowIso = new Date(nowMs).toISOString();
  const res = await db.prepare("DELETE FROM passkey_challenges WHERE consumed = 1 OR expires_at <= ?").bind(nowIso).run();
  return res.changes;
}

export interface RegistrationOptions {
  challenge: string;
  rp: { id: string };
  pubKeyCredParams: { type: "public-key"; alg: number }[];
  authenticatorSelection: { residentKey: string; userVerification: string };
  attestation: "none";
  // AH-01 fix: every currently-active credential for this passport, so the
  // browser refuses to let the SAME authenticator silently overwrite one of
  // them as a "new" discoverable credential (WebAuthn replaces a resident key
  // sharing rpId+user handle on that authenticator). Without this, ordinary
  // re-enrollment on an already-enrolled device can locally replace a key the
  // server still lists active, and finishPasskeyRegistration never retires the
  // old row (services/trust-core/test/passkeys.test.ts: "excludeCredentials").
  excludeCredentials: { type: "public-key"; id: string }[];
  /** I3-0034: present on a FIRST enrolment — a code was mailed to the verified contact address; finish needs it. */
  email_code?: { sent: true; expires_at: string };
}

/**
 * I3-0034 — A SECOND PASSKEY NEEDS THE FIRST. A creator session alone used to
 * be enough to enrol a passkey, so a stolen session could add its own and sign
 * as the creator for good. Now:
 *  - 0 active passkeys (first enrolment, or re-enrolment after the operator
 *    break-glass `operatorResetPasskeys`, which revokes every one): the session
 *    AND a one-time code mailed to the VERIFIED contact address (Jason I11-Q7,
 *    D-1110(8)); no verified address -> `FirstPasskeyContactRequired`. The
 *    register challenge's action_hash is `FIRST_PASSKEY_HASH_PREFIX` + the code's
 *    otp hash over this challenge, and it expires with the code;
 *  - >=1 active and no `stepUp`: an `assert` challenge bound to the add-a-passkey
 *    op is minted and refused back as `PasskeyStepUpRequired` (403);
 *  - >=1 active with `stepUp`: `verifyAssertion` (single-use, CHALLENGE_TTL_MS,
 *    this passport's active credential) and then ONE register challenge whose
 *    action_hash names the approving credential (`ADD_PASSKEY_HASH_PREFIX`).
 * The door that cannot be skipped is the INSERT in `finishPasskeyRegistration`,
 * which re-asserts the same rule in its own statement.
 */
export class PasskeyStepUpRequired extends PasskeyError {
  override name = "PasskeyStepUpRequired";
  constructor(public readonly stepUp: AssertionOptions) {
    super("existing_passkey_required", 403);
  }
}

/** The register challenge's action_hash after a step-up: names the credential
 * that approved it, so finish can require it is STILL active. */
const ADD_PASSKEY_HASH_PREFIX = "passkey-add:v1:";
const CREDENTIAL_ID_RE = /^[A-Za-z0-9_-]{1,512}$/;

  return manageActionHash({ op: "add_passkey", target: passportId, passport: passportId });

export async function startPasskeyRegistration(
  db: DbLike,
  passportId: string,
  cfg: WebAuthnConfig,
  nowMs: number,
  now: () => string = () => new Date().toISOString(),
  randomBytes: () => Uint8Array = defaultRandom,
  stepUp?: AssertionResponse,
  /** I3-0034: required at ZERO active passkeys (absent there -> 503 email_unavailable, fail closed). */
  firstCode?: FirstPasskeyCodeDeps,
): Promise<RegistrationOptions> {
  if (!/^psp_[0-9a-f]{16}$/.test(passportId)) throw new PasskeyError("bad passport id");
  const existing = await activeCredentials(db, passportId);
  const options = (challenge: string): RegistrationOptions => ({
    challenge,
    rp: { id: cfg.rpId },
    pubKeyCredParams: [{ type: "public-key", alg: -7 }], // ES256 only
    authenticatorSelection: { residentKey: "preferred", userVerification: "required" },
    attestation: "none",
    excludeCredentials: existing.map((id) => ({ type: "public-key", id })),
  });
  if (existing.length === 0) {
    if (!firstCode) throw new PasskeyError("email_unavailable", 503);
    const to = (await db.prepare("SELECT email FROM creator_contacts WHERE passport_id = ? AND email_verified = 1 AND email IS NOT NULL").bind(passportId).first<{ email: string }>())?.email ?? null;
    if (to === null) throw new FirstPasskeyContactRequired();
    // The other OTP doors' send limits (otp-policy.ts), per passport: one code a minute, SEND_HOURLY_CAP_PER_EMAIL an hour.
    if (!(await consumeRateLimit(db, `pkfirst-min:${passportId}`, nowMs, 1, SEND_COOLDOWN_MS)) || !(await consumeRateLimit(db, `pkfirst-hour:${passportId}`, nowMs, SEND_HOURLY_CAP_PER_EMAIL, HOUR_MS_))) {
      throw new PasskeyChallengeRateLimitError();
    }
    const code = newFirstPasskeyCode();
    const challenge = await createChallenge(db, passportId, "register", null, nowMs, now, randomBytes, undefined, {
      ttlMs: firstCode.ttlMs,
      hashFor: async (ch) => `${FIRST_PASSKEY_HASH_PREFIX}${await firstCode.hash(firstCodeHashInput(ch, code))}`,
    });
    // The code goes to the mail and nowhere else: never logged, never in an audit row or an answer.
    if (!(await firstCode.mailer.send(FIRST_PASSKEY_CODE_TEMPLATE, to, { code }).catch(() => false))) {
      await db.prepare("UPDATE passkey_challenges SET consumed = 1 WHERE challenge = ?").bind(challenge).run();
      throw new PasskeyError("email_send_failed", 503);
    }
    return { ...options(challenge), email_code: { sent: true, expires_at: new Date(nowMs + firstCode.ttlMs).toISOString() } };
  }
  let registerHash: string | null = null;
  {
    const addHash = await addPasskeyActionHash(passportId);
    if (!stepUp) {
      const ch = await createChallenge(db, passportId, "assert", addHash, nowMs, now, randomBytes);
      throw new PasskeyStepUpRequired({ challenge: ch, rpId: cfg.rpId, allowCredentials: existing.map((id) => ({ type: "public-key", id })), userVerification: "required" });
    }
    const approved = await verifyAssertion(db, passportId, addHash, stepUp, cfg, nowMs, now);
    registerHash = `${ADD_PASSKEY_HASH_PREFIX}${approved.credential_id}`;
  }
  return options(await createChallenge(db, passportId, "register", registerHash, nowMs, now, randomBytes));
}

export async function finishPasskeyRegistration(
  db: DbLike,
  passportId: string,
  resp: AttestationResponse,
  cfg: WebAuthnConfig,
  nowMs: number,
  now: () => string = () => new Date().toISOString(),
  /** I3-0338 / I12-RG-AUTH-04: the creator-session epoch the route ADMITTED the
   * caller at (`creatorAdmission`). REQUIRED: an entry-time read here would be a
   * generation the request was never admitted in. */
  admittedEpoch: number,
  /** I3-0034: the code the creator typed (route body `email_code`) and the same otp hash start used. A first-enrolment
   * challenge is refused without them. */
  firstCode?: { code: unknown; hash: (raw: string) => Promise<string> },
): Promise<{ credential_id: string }> {
  if (!/^psp_[0-9a-f]{16}$/.test(passportId)) throw new PasskeyError("bad passport id");
  if (!Number.isInteger(admittedEpoch)) throw new PasskeyError("session ended — sign in again", 401);
  const admitted = admittedEpoch;
  let clientData: ClientData, authData: AuthData, jwk: Es256Jwk;
  try {
    clientData = parseClientData(b64urlToBytes(resp.clientDataJSON));
    if (clientData.type !== "webauthn.create") throw new WebAuthnError("wrong clientData type");
    assertOriginAllowed(cfg.origins, clientData.origin);
    const att = decodeCbor(b64urlToBytes(resp.attestationObject));
    if (!(att instanceof Map)) throw new WebAuthnError("attestationObject not a map");
    const authDataBytes = att.get("authData");
    if (!(authDataBytes instanceof Uint8Array)) throw new WebAuthnError("authData missing");
    authData = parseAuthData(authDataBytes);
    if (!authData.up) throw new WebAuthnError("user-presence flag not set");
    if (!authData.uv) throw new WebAuthnError("user-verification required"); // biometric/PIN, not a bare tap
    if (!authData.at || !authData.cosePublicKey || !authData.credentialId) throw new WebAuthnError("no attested credential");
    if (!(await rpIdHashMatches(authData, cfg.rpId))) throw new WebAuthnError("rpIdHash mismatch");
    jwk = coseToEs256Jwk(authData.cosePublicKey).jwk;
  } catch (e) {
    if (e instanceof WebAuthnError) throw new PasskeyError(`registration rejected: ${e.message}`);
    throw new PasskeyError("registration rejected");
  }
  // Consume the challenge only after the response is structurally valid. Its
  // action_hash says which door minted it (I3-0034): `passkey-first:v1:<code hash>`
  // = first enrolment, needing the mailed code; `passkey-add:v1:<cred>` = approved
  // by that existing passkey. Anything else (a NULL one included: the session-only
  // first enrolment Jason ruled out, I11-Q7) is not a register challenge this
  // function honours.
  const minted = /^[A-Za-z0-9_-]{1,256}$/.test(clientData.challenge)
    ? await db.prepare("SELECT action_hash FROM passkey_challenges WHERE challenge = ? AND passport_id = ? AND purpose = 'register'").bind(clientData.challenge, passportId).first<{ action_hash: string | null }>()
    : null;
  const firstArm = minted?.action_hash?.startsWith(FIRST_PASSKEY_HASH_PREFIX) === true;
  const approvedBy = minted?.action_hash?.startsWith(ADD_PASSKEY_HASH_PREFIX) ? minted.action_hash.slice(ADD_PASSKEY_HASH_PREFIX.length) : null;
  if (!minted || (!firstArm && (approvedBy === null || !CREDENTIAL_ID_RE.test(approvedBy)))) throw new PasskeyError("challenge invalid or expired", 401);
  // THE FIRST-PASSKEY CODE (D-1110(8)): recomputed over THIS challenge and compared in constant time. A missing or
  // wrong code BURNS the challenge, so each mailed code gets one guess (sends are metered at start).
  let firstHash: string | null = null;
  if (firstArm) {
    const code = firstCode?.code;
    const submitted = firstCode && typeof code === "string" && isWellFormedOtp(code) ? `${FIRST_PASSKEY_HASH_PREFIX}${await firstCode.hash(firstCodeHashInput(clientData.challenge, code))}` : null;
    if (submitted === null || !constantTimeEqual(submitted, minted.action_hash!)) {
      await db.prepare("UPDATE passkey_challenges SET consumed = 1 WHERE challenge = ? AND passport_id = ? AND purpose = 'register'").bind(clientData.challenge, passportId).run();
      throw new PasskeyError(submitted === null ? "email_code_required" : "email_code_invalid", 401);
    }
    firstHash = submitted;
  }
  const ok = await consumeChallenge(db, clientData.challenge, passportId, "register", minted.action_hash, nowMs);
  if (!ok) throw new PasskeyError("challenge invalid or expired", 401);

  const credId = bytesToB64url(authData.credentialId!);
  const aaguidHex = authData.aaguid ? hex(authData.aaguid) : null;
  const ts = now();
  // THE GATE IS IN THE STATEMENT THAT WRITES (INV-058). A first-enrolment
  // challenge inserts only while the passport STILL has no active passkey, so
  // one minted at zero cannot be spent after a passkey exists, AND only while its
  // challenge row is consumed by THIS passport with the hash of the code just
  // checked, AND while the passport still holds a verified contact address (the
  // start's verdict read it; D-1110(8)); an approved one
  // inserts only while the approving credential is STILL this passport's and
  // active, so a revoke or break-glass in the window refuses it.
  // I3-0338: AND the session generation the caller was admitted in is STILL the
  // passport's — a recovery/reset committing after the route's `creatorAdmission`
  // (a reset puts the passport back at ZERO active, where the first arm is true)
  // must not let that ended session's enrolment land.
  const generation = "COALESCE((SELECT session_epoch FROM creator_sessions WHERE passport_id = ?), 0) = ?";
  const gate = approvedBy === null
    ? {
        sql: `NOT EXISTS (SELECT 1 FROM passkeys WHERE passport_id = ? AND status = 'active') AND ${generation} AND EXISTS (SELECT 1 FROM passkey_challenges WHERE challenge = ? AND passport_id = ? AND purpose = 'register' AND consumed = 1 AND action_hash = ?) AND EXISTS (SELECT 1 FROM creator_contacts WHERE passport_id = ? AND email_verified = 1 AND email IS NOT NULL)`,
        args: [passportId, passportId, admitted, clientData.challenge, passportId, firstHash, passportId],
      }
    : { sql: `EXISTS (SELECT 1 FROM passkeys WHERE credential_id = ? AND passport_id = ? AND status = 'active') AND ${generation}`, args: [approvedBy, passportId, passportId, admitted] };
  // The audit row and the creator notice ride the same batch and exist iff the
  // new credential row does.
  const landed = "EXISTS (SELECT 1 FROM passkeys WHERE credential_id = ? AND passport_id = ? AND status = 'active' AND created_at = ?)";
  const via = approvedBy === null ? "first_enrolment" : "existing_passkey";
  const hint = credId.slice(0, 8);
  const periodKey = `passkey-added:${credId}`;
  const content = JSON.stringify({ kind: "alert", period_key: periodKey, passkey_added: { credential_hint: hint, added_at: ts, via } });
  try {
    await db.batch([
      db
        .prepare(`INSERT INTO passkeys (credential_id, passport_id, public_key_jwk, alg, sign_count, aaguid, status, created_at) SELECT ?, ?, ?, ?, ?, ?, 'active', ? WHERE ${gate.sql}`)
        .bind(credId, passportId, JSON.stringify(jwk), -7, authData.signCount, aaguidHex, ts, ...gate.args),
      db
        .prepare(`INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, 'creator:passkey_added', ?, ?, ?, 'creator' WHERE ${landed}`)
        .bind(passportId, `${via}${approvedBy ? ` approved_by=${approvedBy.slice(0, 8)}` : ""} credential=${hint}`, ts, passportId, credId, passportId, ts),
      db
        // D-1037(12) (I29-FX-ERASE H2): a wiped creator is written no digest of any kind (digest.ts creatorNotWipedSql).
        .prepare(`INSERT OR IGNORE INTO creator_digests (digest_id, passport_id, kind, period_key, content, created_at, emailed, pushed) SELECT ?, ?, 'alert', ?, ?, ?, 0, 0 WHERE ${landed} AND ${creatorNotWipedSql("?")}`)
        .bind(`dig_${crypto.randomUUID()}`, passportId, periodKey, content, ts, credId, passportId, ts, passportId),
    ]);
  } catch (e) {
    // credential_id is the PK — a re-registration of the same credential is a
    // conflict, not a silent overwrite (never rebind an existing key). Any other
    // fault (the batch now carries the audit + notice) is ours, not a 409.
    if (e instanceof Error && /UNIQUE|PRIMARY KEY/i.test(e.message) && /passkeys/.test(e.message)) throw new PasskeyError("credential already registered", 409);
    throw e;
  }
  // THE ROW IS THE VERDICT (as in `revokePasskey`): read back what the batch did.
  const row = await db.prepare(`SELECT 1 AS x FROM passkeys WHERE ${landed}`).bind(credId, passportId, ts).first();
  if (!row) {
    if ((await currentCreatorEpoch(db, passportId)) !== admitted) throw new PasskeyError("session ended — sign in again", 401);
    if (approvedBy === null && !(await db.prepare("SELECT 1 AS x FROM creator_contacts WHERE passport_id = ? AND email_verified = 1 AND email IS NOT NULL").bind(passportId).first())) throw new FirstPasskeyContactRequired();
    throw approvedBy === null
      ? new PasskeyError("existing_passkey_required", 403)
      : new PasskeyError("the passkey that approved this is no longer active", 401);
  }
  return { credential_id: credId };
}

// --- assertion (the touched-by proof) ---------------------------------------

export interface AssertionOptions {
  challenge: string;
  rpId: string;
  allowCredentials: { type: "public-key"; id: string }[];
  userVerification: string;
}

/** Stable digest binding a challenge to the EXACT action being authorized. The
 * signed challenge and the appended drafts must produce the same hash. */
export async function actionHashForDrafts(drafts: DraftEvent[], binding?: string): Promise<string> {
  const canonical = drafts.map((d) => ({
    event_type: d.event_type,
    subject: d.subject,
    actor_kind: d.actor_kind,
    payload: d.payload,
    supersedes: d.supersedes ?? [],
  }));
  // BF-04 — A PRIVATE BINDING: part of WHAT WAS AUTHORIZED, but never part of
  // what is APPENDED.
  //
  // Some ceremonies decide something the ledger deliberately does not record.
  // Recognition is the case that drove this: the signed drafts carry
  // `recognition_id` + kind + message and NEVER the fan's `acc_` id (window
  // item 1 — a deleted fan must take the link with them), so the RECIPIENT was
  // outside the signed action hash entirely. The start leg validated a selected
  // fan; the finalize leg accepted a recipient again, from the request body, and
  // put THAT account into the grant. A genuine, unconsumed assertion collected
  // for fan A completed for fan B — consuming the creator's quota and
  // delivering the recognition to the wrong account.
  //
  // `binding` closes that without moving a fan identifier into the permanent
  // log. It is folded into the digest the CHALLENGE ROW is keyed on
  // (`passkey_challenges.action_hash`), which lives in our own database, is
  // single-use, and is pruned — while `drafts`, the thing that is actually
  // appended, is byte-for-byte unchanged. A changed recipient produces a
  // different hash, so `consumeChallenge` finds nothing and the ceremony is
  // refused with no grant and no quota spent.
  //
  // Absent binding hashes exactly as before, so every existing ceremony's
  // action hash is unchanged.
  const body = binding === undefined ? stableStringify(canonical) : `${stableStringify(canonical)}\u0000bind:${binding}`;
  return hex(await sha256Bytes(new TextEncoder().encode(body)));
}

export async function startAssertionForAction(
  db: DbLike,
  passportId: string,
  drafts: DraftEvent[],
  cfg: WebAuthnConfig,
  nowMs: number,
  now: () => string = () => new Date().toISOString(),
  randomBytes: () => Uint8Array = defaultRandom,
  /** BF-04: extra, PRIVATE facts this signature authorizes that the appended
   * drafts deliberately do not carry. See `actionHashForDrafts`. */
  binding?: string,
): Promise<AssertionOptions> {
  if (!/^psp_[0-9a-f]{16}$/.test(passportId)) throw new PasskeyError("bad passport id");
  const creds = await activeCredentials(db, passportId);
  if (creds.length === 0) throw new PasskeyError("no registered passkey", 403);
  refuseMisKindedAccounts(drafts);
  const actionHash = await actionHashForDrafts(drafts, binding);
  const challenge = await createChallenge(db, passportId, "assert", actionHash, nowMs, now, randomBytes);
  return {
    challenge,
    rpId: cfg.rpId,
    allowCredentials: creds.map((id) => ({ type: "public-key", id })),
    userVerification: "required",
  };
}

/**
 * H2 fix — passkey SIGN-IN (signed-out creator re-entry).
 *
 * A creator who has enrolled a passkey must be able to return to the console
 * after any session ends (logout-all, epoch bump, the 30-day TTL) WITHOUT
 * re-running the one-shot claim. This is a pure authentication ceremony: prove
 * possession of a registered credential for a KNOWN passport id, mint a fresh
 * creator session, append nothing to the ledger.
 *
 * WHY NO ORACLE: the caller supplies the `passport_id` themselves — the same id
 * printed in their console and already public at `GET /p/:id/json` — so nothing
 * new about existence leaks; there is deliberately NO handle→passport lookup
 * (the reason F14 was parked). WHY NO CROSS-USE: the sign-in challenge is bound
 * to the fixed sentinel `SIGNIN_ACTION_HASH`, which can never equal a real
 * draft's action hash (those are sha256 of a draft array), so a sign-in
 * challenge can never authorize a ledger append and an append challenge can
 * never sign anyone in — while reusing the exact same single-use,
 * counter-checked, origin-pinned `verifyAssertion` machinery. */
export const SIGNIN_ACTION_HASH = "signin:v1";

/**
 * ACCEPTED RISK (I8 / QUEUE #169, E-50 limb B; DECIDED, low severity): the
 * anonymous sign-in start returns `allowCredentials`, one entry per ACTIVE
 * credential, so a caller who knows a passport id learns how many passkeys it
 * has and their ids. Kept because registration asks for residentKey
 * "preferred", not "required": enrolled non-resident devices can only sign in
 * from this list, and "required" would strand them and force re-enrolment (a
 * product change). The passport id is already public, and the list carries no
 * key material. `i8-169-signin-allowcredentials-accepted.test.ts` pins this
 * shape so a change to it is noticed, not silent.
 */
export async function startSigninAssertion(
  db: DbLike,
  passportId: string,
  cfg: WebAuthnConfig,
  nowMs: number,
  now: () => string = () => new Date().toISOString(),
  randomBytes: () => Uint8Array = defaultRandom,
  /** The route's MASKED caller prefix (`callerPrefix`), or null for a headerless
   * caller, which shares one bucket — metered collectively, never uncapped. */
  caller: string | null = null,
): Promise<AssertionOptions> {
  if (!/^psp_[0-9a-f]{16}$/.test(passportId)) throw new PasskeyError("bad passport id");
  const creds = await activeCredentials(db, passportId);
  if (creds.length === 0) throw new PasskeyError("no registered passkey", 403);
  const challenge = await createChallenge(db, passportId, "assert", SIGNIN_ACTION_HASH, nowMs, now, randomBytes, {
    key: `pkchal-signin:${caller ?? "noip"}`,
    cap: SIGNIN_CHALLENGES_PER_IP_HOUR,
    ipKeyed: true,
  });
  return { challenge, rpId: cfg.rpId, allowCredentials: creds.map((id) => ({ type: "public-key", id })), userVerification: "required" };
}

/** Verify a sign-in assertion (possession of a registered passkey for the
 * passport). Returns the credential id on success; throws PasskeyError on any
 * failure. Appends nothing. */
export async function verifySigninAssertion(
  db: DbLike,
  passportId: string,
  resp: AssertionResponse,
  cfg: WebAuthnConfig,
  nowMs: number,
  now: () => string = () => new Date().toISOString(),
): Promise<{ credential_id: string }> {
  return verifyAssertion(db, passportId, SIGNIN_ACTION_HASH, resp, cfg, nowMs, now);
}

/** D-0840(1): the sign-in refusal for a passport that is not LIVE. Runs the same
 * parse checks, lookup and challenge consume `verifySigninAssertion` runs for a
 * never-listed id, then refuses with the same challenge refusal — so a delisted
 * creator's real credential and a never-listed id answer identically. */
export async function refuseSigninAssertion(db: DbLike, passportId: string, resp: AssertionResponse, cfg: WebAuthnConfig, nowMs: number): Promise<never> {
  await verifyAssertion(db, passportId, SIGNIN_ACTION_HASH, { ...resp, credentialId: "" }, cfg, nowMs, undefined, true);
  throw new PasskeyError(CHALLENGE_REFUSAL, 401);
}

export interface AssertionResponse {
  credentialId: string; // base64url
  clientDataJSON: string; // base64url
  authenticatorData: string; // base64url
  signature: string; // base64url (DER ECDSA)
}

/**
 * S-CW6c: passkey-gate BINDING a Connect payout account — a non-ledger action,
 * exactly the pattern signin/revoke use. The action hash is bound to the exact
 * account id, so an assertion collected for account A can never activate
 * account B, and the `connect-bind:` prefix can never equal a real draft's
 * action hash (those are 64-hex sha256), so this challenge can never authorize
 * a ledger append and vice versa. Appends nothing; the caller flips the
 * creator_connect row to 'active' on success. This is why a stolen session
 * cannot redirect a creator's earnings: starting the OAuth flow needs only the
 * session, but MAKING an account the payout destination needs the device key.
 */
export type ConnectBindOp = "bind" | "unbind";

export function connectBindActionHash(op: ConnectBindOp, stripeAccountId: string, bindEpoch: number): string {
  // BF-14 — v2 CARRIES THE OPERATION AND THE BINDING GENERATION.
  //
  // v1 was `connect-bind:v1:<acct>`: the destination and nothing else. Both the
  // activation and the removal ceremonies used these helpers, and
  // `boundConnect` returns a PENDING binding as well as an active one — so for
  // a pending row BOTH doors were open at once and produced the SAME action
  // value. A creator who connected the wrong Stripe account and approved
  // DISCARDING it thereby produced, byte for byte, an approval to ACTIVATE it.
  // The assertion is single-use, but it was single-use for the wrong act.
  //
  // The epoch is the second half. `startConnect` bumps `creator_connect
  // .bind_epoch` on every restart, so an approval the creator reviewed for one
  // destination generation could otherwise be spent on whatever generation
  // happened to be current when it was submitted.
  //
  // The `connect-bind:` prefix still cannot equal a real draft's action hash
  // (those are 64-hex sha256), so this challenge can still never authorize a
  // ledger append, and vice versa.
  return `connect-bind:v2:${op}:${stripeAccountId}:${bindEpoch}`;
}

export async function startConnectBindAssertion(
  db: DbLike,
  passportId: string,
  op: ConnectBindOp,
  stripeAccountId: string,
  bindEpoch: number,
  cfg: WebAuthnConfig,
  nowMs: number,
  now: () => string = () => new Date().toISOString(),
  randomBytes: () => Uint8Array = defaultRandom,
): Promise<AssertionOptions> {
  if (!/^psp_[0-9a-f]{16}$/.test(passportId)) throw new PasskeyError("bad passport id");
  if (!/^acct_[A-Za-z0-9]+$/.test(stripeAccountId)) throw new PasskeyError("bad account id");
  if (!Number.isInteger(bindEpoch) || bindEpoch < 0) throw new PasskeyError("bad binding generation");
  const creds = await activeCredentials(db, passportId);
  if (creds.length === 0) throw new PasskeyError("no registered passkey", 403);
  const challenge = await createChallenge(db, passportId, "assert", connectBindActionHash(op, stripeAccountId, bindEpoch), nowMs, now, randomBytes);
  return { challenge, rpId: cfg.rpId, allowCredentials: creds.map((id) => ({ type: "public-key", id })), userVerification: "required" };
}

/** Verify a Connect-bind assertion — possession of a registered passkey, bound
 * to the exact OPERATION, account id and binding generation (BF-14). Returns
 * the credential id; throws on failure. */
export async function verifyConnectBindAssertion(
  db: DbLike,
  passportId: string,
  op: ConnectBindOp,
  stripeAccountId: string,
  bindEpoch: number,
  resp: AssertionResponse,
  cfg: WebAuthnConfig,
  nowMs: number,
  now: () => string = () => new Date().toISOString(),
): Promise<{ credential_id: string }> {
  return verifyAssertion(db, passportId, connectBindActionHash(op, stripeAccountId, bindEpoch), resp, cfg, nowMs, now, false, true);
}

async function activeCredentials(db: DbLike, passportId: string): Promise<string[]> {
  const { results } = await db.prepare("SELECT credential_id FROM passkeys WHERE passport_id = ? AND status = 'active'").bind(passportId).all<{ credential_id: string }>();
  return results.map((r) => r.credential_id);
}

const CHALLENGE_REFUSAL = "challenge invalid, expired, or not for this action";

/** Verify an assertion is a valid signature by one of the passport's registered
 * credentials over the challenge bound to `actionHash`. Fail closed.
 * D-0840(1): SELECT -> consume -> refuse. The challenge decides first, so a caller
 * holding no live challenge learns nothing about which credentials exist, and an
 * assertion naming an unregistered credential spends the challenge it rode on.
 * `nonLive` (refuseSigninAssertion only) refuses after the consume. */
export async function verifyAssertion(
  db: DbLike,
  passportId: string,
  actionHash: string,
  resp: AssertionResponse,
  cfg: WebAuthnConfig,
  nowMs: number,
  now: () => string = () => new Date().toISOString(),
  nonLive = false,
  /** I11-Q7: a money action — the credential must be older than the cooling-off (`coolingOffMs`). */
  money = false,
): Promise<{ credential_id: string }> {
  let clientData: ClientData, authData: AuthData, cdjBytes: Uint8Array, adBytes: Uint8Array, sigBytes: Uint8Array;
  try {
    cdjBytes = b64urlToBytes(resp.clientDataJSON);
    adBytes = b64urlToBytes(resp.authenticatorData);
    sigBytes = b64urlToBytes(resp.signature);
    clientData = parseClientData(cdjBytes);
    if (clientData.type !== "webauthn.get") throw new WebAuthnError("wrong clientData type");
    assertOriginAllowed(cfg.origins, clientData.origin);
    authData = parseAuthData(adBytes);
    if (!authData.up) throw new WebAuthnError("user-presence flag not set");
    if (!authData.uv) throw new WebAuthnError("user-verification required");
    if (!(await rpIdHashMatches(authData, cfg.rpId))) throw new WebAuthnError("rpIdHash mismatch");
  } catch (e) {
    if (e instanceof WebAuthnError) throw new PasskeyError(`assertion rejected: ${e.message}`, 401);
    throw new PasskeyError("assertion rejected", 401);
  }

  // The credential must belong to THIS passport and be active.
  const row = await db
    .prepare("SELECT public_key_jwk, sign_count, created_at FROM passkeys WHERE credential_id = ? AND passport_id = ? AND status = 'active'")
    .bind(resp.credentialId, passportId)
    .first<{ public_key_jwk: string; sign_count: number; created_at: string }>();

  // Consume the bound challenge (single-use) BEFORE trusting the signature, so a
  // valid assertion can never be replayed even if captured — and before the
  // credential verdict, so the challenge refusal is the first answer (D-0840(1)).
  const consumed = await consumeChallenge(db, clientData.challenge, passportId, "assert", actionHash, nowMs);
  if (!consumed || nonLive) throw new PasskeyError(CHALLENGE_REFUSAL, 401);
  if (!row) throw new PasskeyError("unknown credential", 401);

  let ok = false;
  try {
    const key = await importEs256(JSON.parse(row.public_key_jwk) as Es256Jwk);
    const base = await assertionSignatureBase(adBytes, cdjBytes);
    ok = await verifyEs256(key, base, sigBytes);
  } catch {
    ok = false;
  }
  if (!ok) throw new PasskeyError("signature invalid", 401);
  // I11-Q7: asked only of a proven holder (after the signature), so the credential's age is never an oracle. An
  // unparseable created_at is refused (the comparison is false for NaN).
  if (money && !(Date.parse(row.created_at) <= nowMs - coolingOffMs(cfg))) throw new PasskeyError(PASSKEY_COOLING_OFF_REFUSAL, 403);

  // Signed-counter clone detection: if either side is non-zero, the counter must
  // strictly increase. (Authenticators that always report 0 are exempt.)
  const newCount = authData.signCount;
  if (!(newCount === 0 && row.sign_count === 0) && newCount <= row.sign_count) {
    throw new PasskeyError("authenticator counter did not increase (possible clone)", 401);
  }
  // INV-058 — THE FINAL WRITE CARRIES ITS OWN VERDICT. The `status = 'active'`
  // fact was established by the SELECT at :444 and is acted on here, in a later
  // statement; a revoke committing in the window between them (`revokePasskey`,
  // `operatorResetPasskeys` — break-glass on a hijacked passport is exactly when
  // this races) left that SELECT the decider, and this function RESOLVED, so the
  // caller minted a session and stamped a revoked credential. Both writes now
  // re-assert the credential is STILL this passport's and STILL active, and
  // `changes === 0` is a refusal — which is what makes `verifyAssertion` a
  // generation fence for `verifySigninAssertion` (:327) and `appendCreatorSigned`
  // (:611) rather than a stale read.
  const fenced = "credential_id = ? AND passport_id = ? AND status = 'active'";
  if (newCount === 0 && row.sign_count === 0) {
    // Always-zero authenticator: nothing to advance, just stamp the use.
    const w = await db.prepare(`UPDATE passkeys SET last_used_at = ? WHERE ${fenced}`).bind(now(), resp.credentialId, passportId).run();
    if (w.changes === 0) throw new PasskeyError("credential is no longer active", 401);
  } else {
    // GUARDED write, not a blind one. The JS check above is check-then-act: two
    // concurrent assertions (real device + clone, each holding its own fresh
    // challenge) could both read count N, both present N+1, and both pass it —
    // losing exactly the signal the counter exists for — while the later commit
    // regressed the stored value and widened the window for the next pair.
    // Re-asserting `sign_count < ?` IN the UPDATE makes the compare and the
    // write one atomic statement: the loser matches no row (`changes === 0`)
    // and is treated as the clone alarm, and the counter can never regress.
    const w = await db.prepare(`UPDATE passkeys SET sign_count = ?, last_used_at = ? WHERE ${fenced} AND sign_count < ?`).bind(newCount, now(), resp.credentialId, passportId, newCount).run();
    if (w.changes === 0) {
      // Two reasons the row didn't match; say which, so the clone alarm keeps
      // meaning a clone. Failure path only.
      const still = await db.prepare("SELECT 1 AS x FROM passkeys WHERE credential_id = ? AND passport_id = ? AND status = 'active'").bind(resp.credentialId, passportId).first();
      throw new PasskeyError(still ? "authenticator counter did not increase (possible clone)" : "credential is no longer active", 401);
    }
  }

  return { credential_id: resp.credentialId };
}

/**
 * #119 (D-0761) — THE PLATFORM'S ID KIND, ASKED BY THE SERVER. A platform is
 * pid-only when every shape its `PLATFORMS` extract rule can produce is a pid:
 * `music-id`, or a `prefix` rule whose every prefix is `pid` (nine platforms).
 * The same predicate the console signs by (`apps/creator-console/src/
 * platforms.ts pidOnlyPlatform`), read from the same table, so the two cannot
 * disagree. Steam mixes `/id/` and `/profiles/` and stays handle; a platform
 * whose rule CAN yield a pid (youtube, facebook, steam, x via OAuth) is not
 * refused either kind, because the fan canonicalizer can emit both there.
 */
function pidOnlyPlatformKey(platformKey: string): boolean {
  const extract = PLATFORMS[platformKey]?.extract;
  if (!extract) return false;
  if (extract.kind === "music-id") return true;
  return extract.kind === "prefix" && extract.prefixes.length > 0 && extract.prefixes.every((p) => p.idKind === "pid");
}

/** The refusal sentence for an `(platform, id_kind)` pair whose stored key no
 * fan canonicalizer could ever derive, or null. An unknown platform, a `domain`
 * and a `link` are not this function's question (the canonicalizer and the
 * domain/link add gate answer them) and return null. */
export function accountIdKindProblem(platform: string, idKind: string): string | null {
  if (idKind === "domain" || idKind === "link") return null;
  let key: string;
  try {
    key = normalizePlatform(platform);
  } catch {
    return null;
  }
  if (pidOnlyPlatformKey(key) && idKind !== "pid") {
    return `${key} identifies accounts by profile id: id_kind must be "pid", not "${idKind}"`;
  }
  return null;
}

/** #119 (D-0761): a creator-signed `account_added` whose id_kind is not the
 * platform's is refused on BOTH legs, before a challenge is written or an
 * assertion verified — stored, it would be a key no fan canonicalizer derives. */
function refuseMisKindedAccounts(drafts: DraftEvent[]): void {
  for (const d of drafts) {
    if (d.event_type !== "account_added") continue;
    const p = (d.payload ?? {}) as Record<string, unknown>;
    const problem = accountIdKindProblem(String(p.platform ?? ""), String(p.id_kind ?? ""));
    if (problem) throw new PasskeyError(problem, 400);
    // I3-1136 (I6-LEDGER hand-off 3): the write door's url predicate
    // (events.ts appendEventsOnce step 1), run here so the refusal lands before
    // createChallenge (start leg) and before verifyAssertion spends the
    // challenge (finish leg). An unmappable account is left to the door.
    if (typeof p.url === "string") {
      let s: { platform: string; id: string } | null = null;
      try {
        s = subjectForAccount(String(p.platform), String(p.id_kind), String(p.id));
      } catch {
        s = null;
      }
      if (s && !urlNamesAccount(s.platform, s.id, p.url)) {
        throw new PasskeyError(`that link doesn't point at this account — use the account's own profile URL (it must name ${s.id}), or leave it empty`, 400);
      }
    }
  }
}

/** Build the drafts for a set of creator actions — identically on the start
 * (challenge) and finish (append) legs so their action hashes match. The server
 * fixes subject/actor_kind/actor_id; the client supplies only type + payload. */
export function buildCreatorDrafts(passportId: string, actions: CreatorAction[]): DraftEvent[] {
  return actions.map((a) => ({
    event_type: a.event_type,
    subject: passportId,
    actor_kind: "creator_passkey" as ActorKind,
    actor_id: "passkey",
    payload: a.payload,
    supersedes: a.supersedes ?? [],
  }));
}

export async function appendCreatorSigned(
  db: DbLike,
  passportId: string,
  drafts: DraftEvent[],
  assertion: AssertionResponse,
  cfg: WebAuthnConfig,
  nowMs: number,
  now: () => string = () => new Date().toISOString(),
  extraStatements: DbStatement[] = [],
  /** BF-04: must equal the value the `/start` leg bound, or the challenge is
   * not found and the ceremony is refused. */
  binding?: string,
  /** QUEUE #19: `publishHook(...)`'s thunk, forwarded to `appendEvents`, which runs it
   * once after the batch commits (so outbox rows in `extraStatements` publish now). */
  afterCommit?: () => void,
): Promise<TrustEventEnvelope[]> {
  if (drafts.length === 0) throw new PasskeyError("no drafts");
  if (drafts.length > 20) throw new PasskeyError("too many drafts in one signed action");
  for (const d of drafts) {
    if (d.subject !== passportId) throw new PasskeyError("a creator may only sign actions on their own passport", 403);
    if (d.actor_kind !== "creator_passkey") throw new PasskeyError("creator-signed drafts must be actor_kind creator_passkey");
    if (!CREATOR_SIGNABLE.has(d.event_type)) throw new PasskeyError(`a creator may not sign ${d.event_type}`, 403);
  }
  refuseMisKindedAccounts(drafts);
  const actionHash = await actionHashForDrafts(drafts, binding);
  await verifyAssertion(db, passportId, actionHash, assertion, cfg, nowMs, now, false, drafts.some((d) => MONEY_SIGNABLE.has(d.event_type))); // throws on failure
  const privileged: ActorKind[] = ["creator_passkey"];
  // The agency/likeness/release signable types aren't on the general
  // APPEND_ALLOWED list — this passkey-verified path is their one governed door.
  // extraStatements ride the SAME atomic batch (e.g. release/fingerprint rows
  // committed with the signed release_registered event, all-or-nothing).
  return appendEvents(db, drafts, now, [...CREATOR_SIGNABLE], extraStatements, privileged, afterCommit);
}

/**
 * True iff this passport was BREAK-GLASS reset and hasn't re-enrolled yet: it has
 * ZERO active passkeys but at least one REVOKED one. This is a precise, derived
 * signal — no ledger event or marker needed — because the ONLY way to reach zero
 * active passkeys is `operatorResetPasskeys` (S96 recovery): self-revoke is
 * guarded against removing your last passkey (revokePasskey's last-one check), so
 * a creator can never self-reduce to zero. A brand-new creator who simply never
 * enrolled has NO revoked rows → false (the onboarding golden path covers them,
 * not this break-glass banner). Clears to false the moment a fresh device enrolls.
 */
export async function needsPasskeyReenroll(db: DbLike, passportId: string): Promise<boolean> {
  if (!/^psp_[0-9a-f]{16}$/.test(passportId)) return false;
  const active = await db.prepare("SELECT COUNT(*) AS n FROM passkeys WHERE passport_id = ? AND status = 'active'").bind(passportId).first<{ n: number }>();
  if ((active?.n ?? 0) > 0) return false; // still has a signing device → nothing to prompt
  const revoked = await db.prepare("SELECT 1 AS x FROM passkeys WHERE passport_id = ? AND status = 'revoked' LIMIT 1").bind(passportId).first();
  return revoked !== null; // zero active + at least one revoked ⟹ an operator break-glass reset
}

/** Action hash for a NON-ledger passkey-management op (revoke). A distinct
 * preimage namespace from `actionHashForDrafts`, so a ledger-action assertion can
 * never be replayed as a management op or vice versa. */
async function manageActionHash(op: { op: string; target: string; passport: string }): Promise<string> {
  return hex(await sha256Bytes(new TextEncoder().encode(`passkey-manage-v1\n${stableStringify(op)}`)));
}

/** Begin a passkey-management assertion (e.g. revoke a device): the creator must
 * prove possession of a CURRENT passkey. Same single-use challenge machinery as a
 * signed action, bound to the management op hash — not to a ledger draft. */
export async function startRevokeAssertion(
  db: DbLike,
  passportId: string,
  targetCredentialId: string,
  cfg: WebAuthnConfig,
  nowMs: number,
  now: () => string = () => new Date().toISOString(),
  randomBytes: () => Uint8Array = defaultRandom,
): Promise<AssertionOptions> {
  if (!/^psp_[0-9a-f]{16}$/.test(passportId)) throw new PasskeyError("bad passport id");
  if (typeof targetCredentialId !== "string" || !/^[A-Za-z0-9_-]{1,512}$/.test(targetCredentialId)) throw new PasskeyError("bad credential id");
  const creds = await activeCredentials(db, passportId);
  if (creds.length === 0) throw new PasskeyError("no registered passkey", 403);
  const actionHash = await manageActionHash({ op: "revoke_passkey", target: targetCredentialId, passport: passportId });
  const challenge = await createChallenge(db, passportId, "assert", actionHash, nowMs, now, randomBytes);
  return { challenge, rpId: cfg.rpId, allowCredentials: creds.map((id) => ({ type: "public-key", id })), userVerification: "required" };
}

/**
 * Revoke one of a passport's passkeys, PROVEN by an assertion from a current
 * passkey (so a stolen session alone can't strip a creator's devices). Refuses
 * revoking the LAST active passkey — self-lockout goes through operator recovery,
 * not here. Fills the previously-dead `status='revoked'` column.
 *
 * REVOKING SEVERS THE SESSIONS TOO — the creator-side twin of the fan fix.
 *
 * This flipped `status` and stopped there, so a revoked creator device kept a
 * LIVE session for the rest of its TTL: the credential could no longer sign in
 * and could no longer sign the ledger (`verifyAssertion` requires
 * `status = 'active'`), but the token it already held went on reading and
 * writing everything a creator session reaches. That is the same half-severing
 * `revokeFanPasskey` was fixed for, and it is worse here, because the thing a
 * creator is revoking is usually a device they have lost.
 *
 * SO THE EPOCH BUMP AND THE STATUS FLIP ARE ONE ATOMIC BATCH carrying the
 * IDENTICAL predicate: both apply or neither does. The bump runs FIRST, so the
 * only survivable partial state is "every session dead, credential alive",
 * which fails closed; the reverse would leave exactly the live session this
 * exists to kill. The caller's own session dies with everyone else's and the
 * route mints them a fresh one at the new epoch (`POST /account/passkey/revoke`),
 * so the creator doing the revoking stays in and the device they are afraid of
 * is out.
 *
 * WHY THE LEDGER NEEDS NOTHING FURTHER, stated because a creator passkey is
 * SIGNING AUTHORITY and "the session is dead" is not obviously enough.
 * `appendCreatorSigned` cannot be reached without a fresh WebAuthn assertion,
 * `verifyAssertion` reads `WHERE … AND status = 'active'`, and a challenge is
 * single-use and bound to the exact drafts. So the revoked credential cannot
 * sign a new event whether or not any session survives; what survived was the
 * SESSION, and that is what this closes. `trust_events` is untouched — events
 * already signed stay signed, which is the whole point of an append-only log.
 *
 * The epoch lives in the `creator_sessions` SIDE TABLE and not on `passports`,
 * because `passports` is a fold of the ledger that a pure reducer must
 * reproduce byte for byte (see creator-session.ts).
 */
export async function revokePasskey(
  db: DbLike,
  passportId: string,
  targetCredentialId: string,
  assertion: AssertionResponse,
  cfg: WebAuthnConfig,
  nowMs: number,
  now: () => string = () => new Date().toISOString(),
  /** I12-RG-AUTH-01: the epoch the ROUTE admitted the caller at
   * (`creatorAdmission`), never one read here — a bump between admission and
   * this function would otherwise become the replacement session's epoch. */
  admittedEpoch: number,
): Promise<{ revoked: string; session_epoch: number }> {
  if (!Number.isInteger(admittedEpoch)) throw new PasskeyError("session ended — sign in again", 401);
  if (typeof targetCredentialId !== "string" || !/^[A-Za-z0-9_-]{1,512}$/.test(targetCredentialId)) throw new PasskeyError("bad credential id");
  const target = await db.prepare("SELECT status FROM passkeys WHERE credential_id = ? AND passport_id = ?").bind(targetCredentialId, passportId).first<{ status: string }>();
  if (!target || target.status !== "active") throw new PasskeyError("no such active passkey", 404);
  const active = await activeCredentials(db, passportId);
  if (active.length <= 1) throw new PasskeyError("cannot remove your only passkey — register another first, or use account recovery", 409);
  const actionHash = await manageActionHash({ op: "revoke_passkey", target: targetCredentialId, passport: passportId });
  // Proves possession of a current passkey AND consumes the single-use challenge.
  await verifyAssertion(db, passportId, actionHash, assertion, cfg, nowMs, now);
  // The last-passkey guard is folded INTO both writes (count>1 as a WHERE
  // clause), so two concurrent revokes of DIFFERENT devices can't both pass the
  // pre-check and strip the final one — the second serialized write sees
  // count==1 and changes 0 rows. The pre-check above is just for the friendly
  // message.
  const stillRevocable =
    "EXISTS (SELECT 1 FROM passkeys WHERE credential_id = ? AND passport_id = ? AND status = 'active') " +
    "AND (SELECT COUNT(*) FROM passkeys WHERE passport_id = ? AND status = 'active') > 1";
  const ts = now();
  const removedHint = targetCredentialId.slice(0, 8);
  const removedDetail = `credential=${removedHint}`;
  const removedKey = `passkey-removed:${targetCredentialId}`;
  const removedContent = JSON.stringify({ kind: "alert", period_key: removedKey, passkey_removed: { credential_hint: removedHint, removed_at: ts } });
  const res = (await db.batch([
    // FIRST, and with the identical predicate — inside one transaction it fires
    // alongside the revoke or not at all. `creator_sessions` may have no row
    // yet (absent = epoch 0), so this is the same upsert `bumpCreatorEpoch`
    // uses, with the predicate carried on the SELECT.
    db
      .prepare(
        `INSERT INTO creator_sessions (passport_id, session_epoch, updated_at) SELECT ?, 1, ? WHERE ${stillRevocable} ` +
          `ON CONFLICT(passport_id) DO UPDATE SET session_epoch = session_epoch + 1, updated_at = excluded.updated_at RETURNING session_epoch`,
      )
      .bind(passportId, ts, targetCredentialId, passportId, passportId),
    db
      .prepare(`UPDATE passkeys SET status = 'revoked' WHERE credential_id = ? AND passport_id = ? AND status = 'active' AND (SELECT COUNT(*) FROM passkeys WHERE passport_id = ? AND status = 'active') > 1`)
      .bind(targetCredentialId, passportId, passportId),
    // I11-JN-AUTH N1: the audit row and the creator notice exist iff THIS batch
    // revoked the credential. `changes()` is the row count of the statement
    // just above (same connection, same transaction), so a racer's revoke, a
    // break-glass in the window or the last-passkey floor all write nothing.
    // MUST stay directly after the UPDATE. The notice keys on the audit row.
    db
      .prepare("INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, 'creator:passkey_removed', ?, ?, ?, 'creator' WHERE changes() = 1")
      .bind(passportId, removedDetail, ts, passportId),
    db
      .prepare(
        "INSERT OR IGNORE INTO creator_digests (digest_id, passport_id, kind, period_key, content, created_at, emailed, pushed) SELECT ?, ?, 'alert', ?, ?, ?, 0, 0 " +
          `WHERE EXISTS (SELECT 1 FROM audit_log WHERE actor_id = ? AND action = 'creator:passkey_removed' AND item_id = ? AND created_at = ? AND detail = ?) AND ${creatorNotWipedSql("?")}`,
      )
      // D-1037(12) (I29-FX-ERASE H2): no digest for a wiped creator (digest.ts creatorNotWipedSql).
      .bind(`dig_${crypto.randomUUID()}`, passportId, removedKey, removedContent, ts, passportId, passportId, ts, removedDetail, passportId),
  ])) as { results?: { session_epoch: number }[] }[];
  // THE ROW IS THE VERDICT, not a pre-check: read back what the transaction
  // actually did, exactly as `revokeFanPasskey` does. A racer that revoked the
  // same credential first also satisfied the creator's request (and bumped the
  // epoch doing it), so it is a success here.
  const after = await db.prepare("SELECT status FROM passkeys WHERE credential_id = ? AND passport_id = ?").bind(targetCredentialId, passportId).first<{ status: string }>();
  if (after?.status !== "revoked") throw new PasskeyError("could not revoke — the passkey is already gone or is now your only one; try again", 409);
  // I3-0338 — THE EPOCH HANDED BACK IS THE ONE THIS BATCH WROTE, and only when
  // nothing else moved it since admission (`admittedEpoch + 1`). The route mints
  // the caller's replacement session from it; a read AFTER the batch returned
  // whatever a recovery/reset in between had just written, so the session
  // outlived the reset. Otherwise (a racer or a reset revoked first, or another
  // bump interleaved) the admitted epoch comes back: the request is satisfied
  // and the session it mints is already dead, never live past someone's bump.
  const wrote = res[0]?.results?.[0]?.session_epoch;
  return { revoked: targetCredentialId, session_epoch: wrote === admittedEpoch + 1 ? wrote : admittedEpoch };
}

/**
 * OPERATOR break-glass: revoke ALL of a passport's active passkeys at once. This
 * deliberately bypasses the last-passkey self-lockout guard — it is exactly what
 * you want when a passport is compromised (a hijacker holds a device) or the
 * creator lost every device: kill every credential so no one can sign, then the
 * rightful owner re-enrolls fresh. NOT a self-service path — it is only reachable
 * from operator-governed recovery (S96), which is full-operator + dual-control +
 * audited. No WebAuthn assertion is required (the point is the creator can't
 * assert); the authorization is the operator gate at the route.
 */
export async function operatorResetPasskeys(
  db: DbLike,
  passportId: string,
  now: () => string = () => new Date().toISOString(),
  /** QUEUE #942(a) (D-0946(1)): the caller's audit row rides THIS batch, FIRST, so a failed audit insert
   * rolls the reset back and an INSERT ... SELECT there reads the pre-reset state (the keys about to be
   * revoked, the epoch about to be bumped). */
  extraStatements: DbStatement[] = [],
): Promise<{ revoked: number; session_epoch: number }> {
  if (!/^psp_[0-9a-f]{16}$/.test(passportId)) throw new PasskeyError("bad passport id");
  // KILLING THE CREDENTIALS IS NOT KILLING THE AUTHORITY. With every passkey
  // revoked the passport is back at ZERO active, where enrolling needs a live
  // creator session and the code mailed to the verified contact address
  // (`startPasskeyRegistration`'s first-enrolment arm, D-1110(8); I3-0034's
  // step-up applies only while a passkey is active),
  // so a revoke-all that leaves sessions standing hands the passport to whoever
  // holds a token at reset time: they re-enroll and become its SOLE signer,
  // which is precisely the hijacker this break-glass exists to evict. The
  // recovery case only forces `force_logout` alongside for the `hijack` reason
  // (recovery.ts:109); every other reason could run this alone.
  //
  // I3-0338 — THE BUMP AND THE REVOKE ARE ONE BATCH. As two statements, a
  // creator sign-in that read the epoch AFTER the bump and stamped its still-
  // active credential BEFORE the revoke minted a session at the post-bump epoch
  // that nothing later killed (claim-routes.ts reads and re-checks that same,
  // already-bumped value). One transaction leaves no such window, and still has
  // no partial state at all. The upsert is `bumpCreatorEpoch`'s, INVOLUNTARY —
  // this is the operator/recovery door the claim-ref watermark is about
  // (creator-session.ts:23-:37); `RETURNING` is the epoch THIS batch wrote.
  const ts = now();
  const res = (await db.batch([
    ...extraStatements,
    db
      .prepare(
        "INSERT INTO creator_sessions (passport_id, session_epoch, updated_at, last_involuntary_bump_at) VALUES (?, 1, ?, ?) " +
          "ON CONFLICT(passport_id) DO UPDATE SET session_epoch = session_epoch + 1, updated_at = excluded.updated_at, " +
          "last_involuntary_bump_at = MAX(COALESCE(creator_sessions.last_involuntary_bump_at, ''), excluded.last_involuntary_bump_at) RETURNING session_epoch",
      )
      .bind(passportId, ts, ts),
    db.prepare("UPDATE passkeys SET status = 'revoked' WHERE passport_id = ? AND status = 'active'").bind(passportId),
  ])) as { results?: { session_epoch: number }[]; meta?: { changes?: number } }[];
  const at = extraStatements.length;
  const session_epoch = res[at]?.results?.[0]?.session_epoch;
  if (typeof session_epoch !== "number") throw new Error("operatorResetPasskeys: the epoch bump returned no row");
  return { revoked: res[at + 1]?.meta?.changes ?? 0, session_epoch };
}

/**
 * F15 / DIRECTION §3 — THE HREF FOR AN ACCOUNT'S ANSWER PAGE, or null.
 *
 * THE PAGE HAD NO INBOUND LINK. `GET /is/:platform/:handle` is this product's
 * search-engine front door — a fan googles a handle and lands on the verdict —
 * and a grep of every `.ts` under `services/trust-core/src` and `apps/*\/src`
 * for an `/is/` path found the route, its own canonical-URL builder, and eleven
 * prose comments. Zero `href`s. It is not in `sitemap.xml` either, and
 * `renderSitemapXml`'s docblock justified that omission with a premise —
 * "discovered the same way /p/:id is: by crawling the links that point at
 * them" — that was simply false: nothing pointed at them. The surface was
 * reachable only by typing the URL.
 *
 * THIS FUNCTION IS THE ROUTE'S PRECONDITIONS, RUN FORWARDS. `resolveIsPage`
 * below decides what it will serve; a link emitted for anything else is a link
 * to a 404, and a 404 that only SOME rows produce is itself a difference worth
 * measuring. So the two share ONE vocabulary gate
 * (`isSeoPlatform`), ONE syntax gate (`SEO_HANDLE_RE`) and THE canonicaliser — and
 * the round-trip assertion is the same one, for the same reason: the URL we
 * link to must be the URL the route resolves.
 *
 * `idKind` IS REQUIRED AND IS NOT COSMETIC. An account stored as a platform
 * `pid` folds to `v1|<platform>|pid|<id>`, and this page canonicalises what it
 * finds in the path as a HANDLE — a different account key, so the route would
 * 404 on a link built from the same two strings. The caller must pass the row's
 * real kind rather than assuming.
 *
 * WHAT IT DOES NOT DO is decide whether the subject may be linked at all. That
 * is the passport's business and the caller's: `getPassportView` has already
 * applied retraction, suppression and the shield, and the consent-by-claim rule
 * (an unclaimed subject's page is `noindex,nofollow`) belongs to the renderer,
 * which marks the link `nofollow` to match.
 */
export function isPagePath(platform: string, idKind: string, handle: string, status: string): string | null {
  if (idKind !== "handle" || status !== "active") return null;
  if (!isSeoPlatform(platform) || !SEO_HANDLE_RE.test(handle)) return null;
  try {
    const subject = canonicalAccount(platform, "handle", handle);
    if (subject.platform !== platform || subject.id !== handle.toLowerCase()) return null;
    return `/is/${subject.platform}/${subject.id}`;
  } catch {
    return null;
  }
}

   * a platform pid, its platform has an answer-page shape, and it is active.
   * It is here rather than in the renderer because the ID KIND that decides it
   * lives on the database row and nowhere else, and because a native client
   * showing a verdict should be able to link to the same page the web page
   * links to — one source of truth for "does this account have an answer page",
   * not two that can disagree.
   */
  is_path?: string;
  /** Who added this account, from the log's actor_kind. Only "creator" (a
   * passkey-signed action) is a trust-bearing mark and it is unforgeable. */
  touched_by: TouchedBy;
  /**
   * Has THIS account's own control proof been demonstrated, by THIS passport,
   * and is it still current? (`accountProofStanding(...) === "proven_current"`.)
   *
   * A SECOND, ORTHOGONAL AXIS TO `touched_by`, and the page needs both. The
   * provenance mark answers "who asserted this row"; this answers "did anybody
   * ever check". They come apart in exactly the case that matters: an
   * operator-listed row (`touched_by: "operator"`) that nobody has ever proved
   * looks, under a Verified banner, indistinguishable from one the creator
   * proved last week — and the passport-level model made it indistinguishable
   * in the SERVED data too.
   *
   * Published for every listed account and honest whatever the ops settings
   * say. The setting decides what is SERVED at which class; it never decides
   * whether the creator is told the truth about their own page.
   */
  confirmed: boolean;
}

export interface PassportView {
  passport_id: string;
  display_name: string;
  claim_state: "unclaimed" | "claimed" | "verified";
  /**
   * WHAT this subject is — a person, an AI character, or a brand character.
   *
   * An AXIS, not a verdict: it says nothing about how trustworthy the listing is
   * and never colours one. A character reaches `verified` on exactly the proofs a
   * human does; what differs is that the identity confirmed belongs to the entity
   * OPERATING it. Public, because a fan deciding whether to send money is owed
   * the fact that they are talking to a persona.
   *
   * `operated_by` is deliberately NOT here. Which agency or person runs a
   * character is operator information; publishing it would hand anyone the org
   * chart behind every brand mascot in the registry.
   */
  subject_kind: "person" | "ai_character" | "brand";
  /**
   * THE HONEST LINE — AND IT IS FROZEN ENGLISH, ON PURPOSE (slice 26).
   *
   * This field is served verbatim over `GET /p/:id/json`, which is a
   * machine-readable public read that third-party integrations consume. It does
   * NOT vary by `Accept-Language` and it never will: a JSON field whose value
   * changes with a request header is a field no integration can compare, cache
   * or store, and the caller who suffers is the one who never sent the header
   * and got a language they cannot parse.
   *
   * That is why localizing the verdict vocabulary was blocked rather than
   * merely unstarted, and slice 26 settles it in the direction the wire
   * requires: the HTML surfaces localize freely through `@dhwaja/i18n`; this
   * one string stays English. `claim_state` and `all_accounts_confirmed` below
   * are the machine axes it is composed from, so a client that wants its own
   * wording — in any language — can build it without parsing this sentence.
   *
   * `slice26-verdict-contract.test.ts` pins the invariance against a header
   * sweep, because "we did not localize it" is a fact about today's code and
   * "it does not vary" is the promise being made.
   */
  label: string;
  /**
   * THE SECOND AXIS `label` IS BUILT FROM, published so the sentence is
   * reconstructible rather than merely readable.
   *
   * `honestLabel` picks between two tables per claim state — the strong form
   * ("this creator controls these accounts") and the passport-level form used
   * when any ACTIVE account row is unconfirmed. Before slice 26 that choice was
   * observable only by reading the English, so a machine consumer could recover
   * `claim_state` from the JSON but not WHICH sentence it was going to get, and
   * therefore could not render its own equivalent.
   *
   * Named for what it is rather than for the sentence it selects: it is a fact
   * about the accounts, and the roster in `accounts[].confirmed` is the same
   * fact row by row. `true` when there is at least one active row and every
   * active row is confirmed — which is the condition the strong form is true in.
   *
   * COMPUTED OVER `kept`, i.e. AFTER the shield drop and the S101
   * asserted-unproven drop, so it is a pure function of the roster published
   * beside it. A shielded account cannot move it, which is what keeps it from
   * being a second channel for the fact the shield removes.
   *
   * NO VACUOUS CASE (I3-1952): with no active row on the published roster
   * (every account shielded, moved off by a transfer, or revoked) this is
   * `false`, so `label` is the passport-level form rather than "controls these
   * accounts" over nothing. Still a pure function of the published roster.
   */
  all_accounts_confirmed: boolean;
  accounts: PublicAccount[];
  /** flag 1 (2026-08-14): how many ACTIVE PUBLISHED impersonation warnings
   * point at this passport as the protected creator. A bare integer only —
   * never hashes or subjects — and OMITTED entirely when zero (hidden-at-zero
   * is the API contract, so a client can never render a "0 impostors" chip).
   * Derived from PUBLISHED warnings only, never raw reports. */
  impostor_count?: number;
  /** I23-Q8c: present (true) only when the creator closed their own account (`creatorSelfClosed`); absent otherwise. */
  closed?: true;
}

/**
 * THE HONEST LINE, AND WHY IT HAS TWO FORMS PER CLAIM STATE.
 *
 * `claim_state` is a fact about a PASSPORT. Both of the strong labels below were
 * written as facts about the ACCOUNTS — "this creator controls these accounts",
 * "account control and identity confirmed by Dhwaja" — and the page renders them
 * directly above a roster in which any number of rows can read `! not
 * confirmed`. Confirmed live: the header said "Claimed — this creator controls
 * these accounts" while both rows underneath said nobody had confirmed either
 * one. A page that contradicts itself teaches a fan that the marks in the table
 * do not mean anything, which is the opposite of what they are for.
 *
 * So a passport with any unconfirmed ACTIVE row gets the passport-level form:
 * what Dhwaja actually knows (this creator claimed / was verified) plus a
 * pointer at the rows that say the rest. The strong form is kept for the case it
 * is true in — every listed account confirmed.
 *
 * ACTIVE rows only. A revoked or historical row is already marked as such in the
 * table and is not a claim about anything the creator controls today; letting an
 * old row hold the header hostage would mean a creator could never get the
 * strong sentence back after retiring an account.
 */
const LABELS: Record<string, string> = {
  unclaimed: "Unclaimed — listed from public information, not confirmed by this creator",
  claimed: "Claimed — this creator controls these accounts. Identity not independently verified.",
  verified: "Verified — account control and identity confirmed by Dhwaja",
};

/** The passport-level form, used when some listed account is unconfirmed. */
const LABELS_UNCONFIRMED: Record<string, string> = {
  unclaimed: LABELS.unclaimed!,
  claimed: "Claimed — this creator has claimed a Dhwaja passport. Accounts marked “not confirmed” below are listed under it and Dhwaja has not confirmed them. Identity not independently verified.",
  verified: "Verified — Dhwaja confirmed this creator's identity. Accounts marked “not confirmed” below are listed under their passport and Dhwaja has not confirmed them.",
};

/** The honest line for a claim state, told which of its rows are confirmed. */
export function honestLabel(claimState: string, allActiveConfirmed: boolean): string {
  const table = allActiveConfirmed ? LABELS : LABELS_UNCONFIRMED;
  return table[claimState] ?? table.unclaimed!;
}

/**
 * flag 1 — count of ACTIVE PUBLISHED impersonation warnings whose
 * `protected_passport_id` pointer targets this passport: published minus
 * retracted, derived from the ledger on every read (the same fold as
 * registry-build.ts / psa.ts: seq-ordered, latest event per subject wins,
 * with the defensive pointer re-shape-check — a malformed pointer degrades
 * to "not counted", never to a crash or a junk match). Counts distinct
 * warned SUBJECTS, i.e. distinct impostor accounts on record.
 *
 * I3-1174: the statement reads only the subjects that ever had a publication
 * whose payload pointer is THIS passport (the `e.subject IN (...)` clause), with
 * every event of those subjects — so the fold below is unchanged and a passport
 * nobody warned about reads zero rows instead of the platform's whole history.
 * A subject outside that set cannot count: its latest publish cannot point here.
 *
 * I3-0193: the sub-select's pointer expression is byte-identical to the partial
 * expression index `idx_trust_events_warning_pointer` (migration 0163), which is
 * what makes it an index lookup; change one and the other in the same edit
 * (test/i15-warning-pointer-index.test.ts reads the plan).
 */
export async function impostorCount(db: DbLike, passportId: string): Promise<number> {
  if (!PASSPORT_ID_RE.test(passportId)) return 0;
  const { results } = await db
    .prepare(
      `SELECT e.subject AS subject, e.event_type AS event_type, e.payload AS payload, ${operatorSetPointerSql("e")} AS ptr_operator
         FROM trust_events e
        WHERE e.event_type IN ('warning_published','warning_retracted')
          AND e.subject IN (SELECT w.subject FROM trust_events w
                             WHERE w.event_type = 'warning_published'
                               AND (CASE WHEN json_valid(w.payload) = 1 THEN json_extract(w.payload, '$.protected_passport_id') END) = ?)
        ORDER BY e.seq ASC`,
    )
    .bind(passportId)
    .all<{ subject: string; event_type: string; payload: string; ptr_operator: number | null }>();
  const active = new Map<string, boolean>();
  const targets = new Map<string, boolean>(); // subject → latest publish points at THIS passport
  for (const r of results) {
    if (r.event_type === "warning_published") {
      active.set(r.subject, true);
      try {
        const p = JSON.parse(r.payload) as { warning_class?: unknown; protected_passport_id?: unknown };
        // Defensive re-shape-check (mirrors registry-build.ts): only a
        // well-formed pointer on an impersonation-class warning counts.
        targets.set(
          r.subject,
          String(p.warning_class) === "impersonation" &&
            r.ptr_operator === 1 &&
            typeof p.protected_passport_id === "string" &&
            PASSPORT_ID_RE.test(p.protected_passport_id) &&
            p.protected_passport_id === passportId,
        );
      } catch {
        targets.set(r.subject, false);
      }
    } else {
      active.set(r.subject, false); // a later retraction wins (seq ASC)
    }
  }
  let n = 0;
  for (const [subject, isActive] of active) if (isActive && targets.get(subject) === true) n++;
  return n;
}

  display_name: string;

/**
 * THE passport-id shape — one constant, for every reader on every surface.
 *
 * It was four shapes. The server mints `psp_<16 hex>` and every point-read here
 * enforced exactly that, while the find-the-real-one POINTER — the redirect a
 * frightened fan is offered on a Reported verdict — was validated against
 * `psp_[0-9a-z]{8,64}` in the registry builder and in all four clients, and the
 * SPA's own router then required 16 hex again to route it. Nothing was broken
 * only because every id ever minted happens to be 16 hex: the day one is not,
 * the builder serves the pointer, the extension and both native apps render "The
 * real X is here →", and the link lands the fan on the home page with no
 * explanation. A widened id must fail at the VALIDATOR, where the answer is "no
 * pointer" (the warning still stands), not at the router, where the answer is a
 * broken promise.
 *
 * The clients cannot import this constant (they are separate bundles, and two of
 * them are not even TypeScript), so each carries a copy and
 * `tests/invariants/safety-handoff-parity.test.ts` holds all of them to this
 * one, by value.
 */
export const PASSPORT_ID_RE = /^psp_[0-9a-f]{16}$/;

/** I23-Q8c (D-1030): the creator closed their own account (creator_closures, 0205). NOT credits.ts CREATOR_CLOSED_SQL,
 * whose suspension arm is an operator's act, not the creator's. One spelling, read by getPassportView and
 * creator-profile.ts publicCreatorProfile. */
export const SELF_CLOSED_SQL = "SELECT 1 FROM creator_closures WHERE passport_id = ?";
export async function creatorSelfClosed(db: DbLike, passportId: string): Promise<boolean> {
  return (await db.prepare(`SELECT EXISTS (${SELF_CLOSED_SQL}) AS c`).bind(passportId).first<{ c: number }>())?.c === 1;
}

// D-0814(1): moved to the leaf warning-pointer.ts; re-exported so no caller changes.
export { operatorSetPointerSql } from "./warning-pointer.js";

/**
 * THE gated point-read of `passports` for every PUBLIC or FAN-FACING surface:
 * "is there a listing here that this caller is allowed to be told about?"
 * Live (`retracted = 0`) AND not opted out (`request_suppressions`), or null.
 *
 * WHY A SHARED FUNCTION AND NOT A PER-CALLER CHECK. `getPassportView` learned
 * the suppression rule; `saveCreator` did not, and `POST /account/save` answered
 * 200 for a suppressed subject and 404 for one that was never listed — the exact
 * "did this person ask to be delisted?" question the removal feature exists to
 * make unanswerable, reachable by any fan with a session and a passport id
 * (review HIGH 3). `requestCreator` had the rule but reached it late, after a
 * 409 that answered the same question for a suppressed CLAIMED subject. A rule
 * that each caller has to remember is a rule that the next caller forgets, so
 * the lookup itself now carries it and a new surface inherits the gate by using
 * the only reader there is.
 *
 * ONE STATEMENT, DELIBERATELY. The gate rides the SAME statement as the read
 * (`NOT EXISTS`), so a suppressed subject costs exactly the round trips a
 * never-listed one costs. Split across two statements it was a timing oracle
 * behind byte-identical 404s — 2 D1 round trips for suppressed, 1 for
 * never-listed, no calibration needed (review MEDIUM 5). Same idiom
 * `taxonomy.ts` getProfileMeta and `releases.ts` publicArchive already use.
 *
 * NOT for operator/creator-side reads: an operator console must still see a
 * suppressed subject (that is how a suppression gets lifted), and the creator's
 * own authenticated surfaces are covered by their own rules.
 */
export async function livePassport(db: DbLike, passportId: string): Promise<LivePassport | null> {
  if (!PASSPORT_ID_RE.test(passportId)) return null;
  const row = await db
    .prepare(
      `SELECT passport_id, display_name, claim_state, subject_kind
         FROM passports p
        WHERE p.passport_id = ?
          AND p.retracted = 0
          AND NOT EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = p.passport_id)`,
    )
    .bind(passportId)
    .first<LivePassport>();
  return row ?? null;
}

/**
 * `livePassport` PLUS the narrow demand opt-out (`demand_optouts`, migration
 * 0017) — the gated point-read for the REQUEST-COUNTER family only:
 * /p/:id/requests, /p/:id/demand, /p/:id/demand.svg and the passport page's
 * request bar.
 *
 * WHY A SECOND READER RATHER THAN A CLAUSE IN `livePassport`. The two flags mean
 * genuinely different things. `request_suppressions` is "this subject is gone" —
 * it must gate every public surface, which is exactly why the base reader carries
 * it. A demand opt-out is "this subject stays listed and checkable; the public
 * counter about them comes off" (LISTING-POLICY.md §4.4). Folding it into
 * `livePassport` would silently break saving, requesting and archive-requesting
 * for a subject who only asked for the counter to stop, i.e. would quietly turn a
 * narrow opt-out back into the full delisting it exists to avoid.
 *
 * ONE STATEMENT, for the same reason the base reader is one statement: an opted-
 * out subject must cost exactly the round trips an opted-in one costs, or the
 * 404 is byte-identical and the TIMING is not.
 */
export async function livePassportWithDemand(db: DbLike, passportId: string, nowMs: number = Date.now()): Promise<LivePassport | null> {
  if (!PASSPORT_ID_RE.test(passportId)) return null;
  const row = await db
    .prepare(
      `SELECT passport_id, display_name, ${CLAIM_STATE_AT_SQL} AS claim_state, subject_kind
         FROM passports p
        WHERE p.passport_id = ?
          AND p.retracted = 0
          AND NOT EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = p.passport_id)
          AND NOT EXISTS (SELECT 1 FROM demand_optouts d WHERE d.passport_id = p.passport_id)`,
    )
    .bind(new Date(nowMs).toISOString(), passportId)
    .first<LivePassport>();
  return row ?? null;
}

export async function getPassportView(db: DbLike, passportId: string, nowMs: number = Date.now()): Promise<PassportView | null> {
  // Live + unsuppressed in ONE statement — see `livePassport`. This function is
  // the single source for /p/:id, /p/:id/json, /p/:id/psa, /p/:id/presentation,
  // the SEO page and the proof type-gate; `request_suppressions` used to be
  // consulted here only for the impostor COUNT, so /p/:id served the whole page
  // while /is/:platform/:handle 404'd the same subject (review MEDIUM 4).
  // D-1083(1): the claim state is the one a fan is shown at `nowMs` (`CLAIM_STATE_AT_SQL`): a lapsed grant reads claimed.
  const p = await livePassportAt(db, passportId, nowMs);
  if (!p) return null;

  // I3-0197: ONE statement for the roster AND every row's proof standing — the
  // same four aggregates `accountProofDetail` reads per account, grouped per
  // row, and `deriveStanding` (the one rule) applied below. Each proof arm is
  // restricted to this passport's own account keys, so the read is bounded by
  // the roster, not by the proof tables.
  const { results } = await db
    .prepare(
      `SELECT a.account_key AS account_key, a.platform AS platform, a.id AS id, a.id_kind AS id_kind,
              a.url AS url, a.status AS status, a.added_by AS added_by,
              COUNT(CASE WHEN pr.passport_id =  a.passport_id THEN 1 END) AS mine_count,
              COUNT(CASE WHEN pr.passport_id != a.passport_id THEN 1 END) AS other_count,
              MAX(CASE WHEN pr.passport_id =  a.passport_id THEN pr.valid_until END) AS mine_valid_until,
              MAX(CASE WHEN pr.passport_id =  a.passport_id AND pr.valid_until IS NULL THEN pr.proved_at END) AS mine_proved_at
         FROM passport_accounts a
         LEFT JOIN (SELECT c.account_key AS account_key, c.passport_id AS passport_id,
                           COALESCE(c.proved_at, c.created_at) AS proved_at, c.valid_until AS valid_until
                      FROM claim_challenges c
                     WHERE c.status = 'used' AND c.purpose != 'contest'
                       AND c.account_key IN (SELECT k.account_key FROM passport_accounts k WHERE k.passport_id = ?)
                    UNION ALL
                    SELECT o.account_key AS account_key, o.passport_id AS passport_id,
                           COALESCE(o.proved_at, o.created_at) AS proved_at, o.valid_until AS valid_until
                      FROM oauth_challenges o
                     WHERE o.status = 'proven'
                       AND o.account_key IN (SELECT k.account_key FROM passport_accounts k WHERE k.passport_id = ?)) pr
           ON pr.account_key = a.account_key
        WHERE a.passport_id = ?
        GROUP BY a.account_key, a.platform, a.id, a.id_kind, a.url, a.status, a.added_by
        ORDER BY a.status, a.platform, a.id`,
    )
    .bind(passportId, passportId, passportId)
    .all<{
      account_key: string;
      platform: string;
      id: string;
      id_kind: string;
      url: string | null;
      status: PublicAccount["status"];
      added_by: string | null;
      mine_count: number;
      other_count: number;
      mine_valid_until: string | null;
      mine_proved_at: string | null;
    }>();

  // Touched-by provenance is a READ over the log (actor_kind per account_added),
  // keyed by the canonical account_key so it matches the stored account rows.
  const prov = await accountProvenance(db, passportId);

  // S101: an ASSERTED account (creator/agency self-add, unforgeable actor_kind)
  // that has NOT passed its own control-proof is not "the link is the proof" —
  // showing it on the public share page under a Verified banner would launder an
  // unproven famous handle. Drop such accounts from the public listing (mirror
  // of the registry serving gate + report shield).
  //
  // DROPPED vs MARKED, and why the two populations are treated differently.
  // An ASSERTED-and-unproven row is a claim the creator made about themselves
  // that nothing backs, so it is removed outright — leaving it on the page in
  // any form is the laundering this gate exists to stop. An OPERATOR/SYSTEM-
  // listed row that nobody has proved is not a claim anybody made: it is
  // Dhwaja's own guess from public information, and dropping every one of them
  // would empty the page for the entire OSINT coverage tier. Those are kept and
  // MARKED (`confirmed: false`) — which is also the nag, because the fastest way
  // to get an account confirmed is for its owner to see, on their own page,
  // that it is not.
  //
  // "Proven" is passport-bound and TTL-bound. An account whose proof belongs to
  // a different passport (a handle that changed hands) or whose proof has aged
  // out reads exactly like one that was never proved — which is the point:
  // inheriting a stranger's proof, or a decade-old one, is the defect.
  const asserted = await assertedKeysForPassport(db, passportId);
  // P1: THE ROSTER IS THE LINK. Everything else this page does is about whether
  // an account is TRUE; this is about whether the creator agreed to have it
  // listed NEXT TO THE OTHERS. A shielded account keeps its registry answer (a
  // fan checking that URL still gets a verdict, and that verdict carries no
  // passport ref) and loses its place on every public roster — which is the only
  // surface that says "these accounts are the same person".
  //
  // Applied HERE, with the other public-listing gates, because `getPassportView`
  // is the single source for /p/:id, /json, /psa, /presentation and the proof
  // type-gate — and, through `seo-page.ts`'s re-check of `view.accounts`, for
  // /is/:platform/:handle, which 404s for a shield-only handle. That 404 is the
  // decision taken, not an accident: /is/ is a public render carrying the
  // display name and the rest of the roster. Its stated cost: an observer
  // comparing the registry answer to the 404 learns the handle belongs to SOME
  // creator who shielded it — never to which one.
  //
  // A DROP, NOT A MARK, and unlike the S101 gate below there is no honest way to
  // mark it: "this creator has an account they asked us not to name" is the
  // linkage the feature exists to remove.
  const shielded = await shieldedAccountKeys(db, passportId);
  const kept: { row: (typeof results)[number]; confirmed: boolean }[] = [];
  for (const r of results) {
    if (shielded.has(r.account_key)) continue;
    const standing = deriveStanding(r, nowMs).standing;
    const confirmed = standing === "proven_current";
    // A PROOF THAT AGED OUT IS NOT A CLAIM THAT WAS NEVER BACKED, and the drop
    // test used to treat them as one thing.
    //
    // `if (asserted && !confirmed) continue` removed BOTH `asserted_unproven`
    // (never proved — correctly dropped, that is the laundering case) and
    // `proven_stale` (proved by THIS passport, and the proof has passed
    // PROOF_TTL_MS). The second one is a creator who did everything asked of
    // them: on day 180 a platform silently vanished from the page they put in
    // their own bio, with no notice, no mark, and — until the accounts screen
    // shipped in the creator console — no way to put it back. The codebase names
    // this failure itself: *"An expiry a person cannot see coming is a trap"*
    // (digest.ts).
    //
    // So a lapsed proof is KEPT AND MARKED instead. It costs the anti-laundering
    // control nothing: `proven_stale` is reachable only when a proof for this
    // exact (passport, account) pair exists, so the row was demonstrated once by
    // the passport now showing it — which is precisely what an unproven
    // assertion never was. `transferred` (somebody ELSE's proof) stays dropped
    // with `asserted_unproven`: an inherited handle is the case the gate is for.
    //
    // The MARK is the existing `! not confirmed`, not a new one, and that is
    // deliberate. Its wording already covers both worlds ("…or the last proof
    // has expired") and distinguishing them on a PUBLIC page would publish how
    // long ago a creator last touched their own console — a timing oracle the
    // page has no reason to serve. The distinction is drawn where it is useful
    // and safe: on the creator's own authenticated `#/accounts` screen, which
    // names the standing, the date and the remedy.
    if (asserted.has(r.account_key) && !confirmed && standing !== "proven_stale") continue;
    kept.push({ row: r, confirmed });
  }
  // NOT gated on `servingPerAccountStanding`, deliberately. That setting decides
  // what the REGISTRY serves and at which class; whether this page tells the
  // truth about which of its own rows nobody has confirmed is not an operator
  // preference. Saying less here would also cost the flag its cheapest path to
  // being turned on, which is creators noticing and proving their accounts.

  // flag 1: the public impostor-count chip. Served for any listed passport
  // regardless of claim state (same stance as the flag-5 pointer: the warning
  // itself carries the operator sign-offs). Zero → the field is OMITTED. A
  // suppressed passport has no count because it has no page at all — the
  // suppression check above already returned null (it used to be applied only
  // here, which is what left the page itself served).
  const impostors = await impostorCount(db, passportId);
  const closed = await creatorSelfClosed(db, passportId);

  // The header must not contradict the roster underneath it — see `honestLabel`.
  // Computed from the rows this page is about to render, so the two can never
  // drift apart.
  // I3-1952: and never VACUOUSLY. With no active row (every account moved,
  // revoked or shielded) there is nothing for the strong form to be true of.
  const allActiveConfirmed =
    kept.some(({ row: r }) => r.status === "active") && !kept.some(({ row: r, confirmed }) => r.status === "active" && !confirmed);

  return {
    passport_id: p.passport_id,
    display_name: p.display_name,
    claim_state: p.claim_state,
    // NO `?? "person"` FALLBACK. The column is NOT NULL with a default, so a
    // reader that somehow lacks it is a producer bug — and defaulting a
    // DISCLOSURE to "this is a human" is fail-OPEN in the one direction that
    // matters, silently un-disclosing every character. TypeScript makes the
    // field required on `LivePassport`, so a producer that forgets it fails to
    // compile rather than shipping a lie.
    subject_kind: p.subject_kind,
    label: honestLabel(p.claim_state, allActiveConfirmed),
    all_accounts_confirmed: allActiveConfirmed,
    accounts: kept.map(({ row: r, confirmed }) => {
      // The answer-page path, decided by `seo-page.ts` so the link and the route
      // cannot drift; ABSENT rather than null when there is none, so the JSON
      // gains nothing for the rows that have no page.
      const isPath = isPagePath(r.platform, r.id_kind, r.id, r.status);
      return {
        platform: r.platform,
        handle: r.id,
        url: urlNamesAccount(r.platform, r.id, r.url) ? r.url : null,
        status: r.status,
        touched_by: prov[r.account_key] ?? "operator",
        confirmed,
        ...(isPath ? { is_path: isPath } : {}),
      };
    }),
    ...(impostors > 0 ? { impostor_count: impostors } : {}),
    ...(closed ? { closed: true as const } : {}),
  };
}

/**
 * I3-1716 (D-0858(22)): the ONE comparison key for "same name" — name-fold.ts
 * `confusableSkeleton` (`foldName`, a case fold and the UTS #39 look-alike letters;
 * QUEUE #789, #794a — shared with the display-name screen in events.ts) plus a
 * collapse of space runs. Namesakes are legitimate and are never
 * refused; this key only decides whether the public page DISCLOSES one.
 */
export function namesakeKey(name: string): string {
  return confusableSkeleton(name)
    .split(/[\s\p{White_Space}\p{Z}]+/u)
    .filter(Boolean)
    .join(" ");
}

/**
 * RGP-L6-03 (D-0861(5)): one page of the verified-live set past a watermark. It
 * SEEKS migration 0140's partial index `idx_passports_verified_live(updated_seq,
 * passport_id)` (the leading `>=` is the range, the OR the tie-break — the shape
 * verification-sweep.ts resumes on), so a view reads at most `pageRows` rows and
 * never the whole verified population. Binds: seq, seq, seq, passport_id, limit.
 */
export const NAMESAKE_PAGE_SQL = `SELECT passport_id, display_name, updated_seq FROM passports
  WHERE claim_state = 'verified' AND retracted = 0 AND updated_seq >= ?
    AND (updated_seq > ? OR (updated_seq = ? AND passport_id > ?))
  ORDER BY updated_seq, passport_id LIMIT ?`;
const NAMESAKE_PAGE_ROWS = 500;
/** Candidates revalidated per view: a name shared by more verified passports links among the lowest ids indexed. */
const NAMESAKE_CANDIDATES = 20;
type NamesakeIndex = { seq: number; id: string; byKey: Map<string, Set<string>>; keyOf: Map<string, string> };
const namesakeIndexes = new WeakMap<object, NamesakeIndex>();

/**
 * I3-1716: the live, unsuppressed VERIFIED passport whose `namesakeKey` equals
 * this UNVERIFIED view's, or null. Never computed for a verified view.
 *
 * NFKD is not expressible in SQL, so the keys are folded in JS into a per-isolate
 * index (keyed by `opts.cacheKey`, the D1 binding in production; the db by
 * default) that each call ADVANCES by one bounded page (`NAMESAKE_PAGE_SQL`).
 * The index is only a candidate list: every candidate is re-read by primary key
 * under `livePassport`'s predicate (verified, not retracted, no
 * `request_suppressions` row) and a grant not lapsed at `opts.nowMs` (standing.ts `grantLiveAtSql`, D-1083(1)) and re-keyed from its CURRENT name, so a
 * suppression, retraction, un-verify or name erasure (events.ts erases without
 * bumping `updated_seq`) applies on the next view. On a cold isolate with more
 * verified passports than one page the line can be absent until the watermark
 * catches up — a missed disclosure, never a wrong one. With several matches the
 * lowest passport id is linked and the count is not disclosed.
 */
export async function verifiedNamesake(
  db: DbLike,
  v: Pick<PassportView, "passport_id" | "display_name" | "claim_state">,
  opts: { cacheKey?: object; pageRows?: number; nowMs?: number } = {},
): Promise<{ passport_id: string } | null> {
  if (v.claim_state === "verified") return null;
  const key = namesakeKey(v.display_name);
  if (key === "") return null;
  const cacheKey = opts.cacheKey ?? db;
  let ix = namesakeIndexes.get(cacheKey);
  if (!ix) {
    ix = { seq: -1, id: "", byKey: new Map(), keyOf: new Map() };
    namesakeIndexes.set(cacheKey, ix);
  }
  const { results: page } = await db
    .prepare(NAMESAKE_PAGE_SQL)
    .bind(ix.seq, ix.seq, ix.seq, ix.id, Math.max(1, Math.min(opts.pageRows ?? NAMESAKE_PAGE_ROWS, NAMESAKE_PAGE_ROWS)))
    .all<{ passport_id: string; display_name: string; updated_seq: number }>();
  for (const r of page) {
    const was = ix.keyOf.get(r.passport_id);
    if (was !== undefined) ix.byKey.get(was)?.delete(r.passport_id);
    const k = namesakeKey(r.display_name);
    ix.keyOf.set(r.passport_id, k);
    if (k !== "") {
      const ids = ix.byKey.get(k) ?? new Set<string>();
      ids.add(r.passport_id);
      ix.byKey.set(k, ids);
    }
    if (r.updated_seq > ix.seq || (r.updated_seq === ix.seq && r.passport_id > ix.id)) {
      ix.seq = r.updated_seq;
      ix.id = r.passport_id;
    }
  }
  const candidates = [...(ix.byKey.get(key) ?? [])].filter((id) => id !== v.passport_id).sort().slice(0, NAMESAKE_CANDIDATES);
  if (candidates.length === 0) return null;
  const live: { passport_id: string; display_name: string }[] = [];
  for (const chunk of inChunks(candidates)) {
    const { results } = await db
      .prepare(
        `SELECT passport_id, display_name FROM passports p
          WHERE passport_id IN (${chunk.map(() => "?").join(", ")})
            AND claim_state = 'verified' AND retracted = 0 AND length(display_name) > 0
            AND NOT EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = p.passport_id)
            AND ${grantLiveAtSql("p.passport_id")}
          ORDER BY passport_id`,
      )
      .bind(...chunk, new Date(opts.nowMs ?? Date.now()).toISOString())
      .all<{ passport_id: string; display_name: string }>();
    live.push(...results);
  }
  const hit = live.find((r) => namesakeKey(r.display_name) === key);
  return hit ? { passport_id: hit.passport_id } : null;
}

/** A locale-bound lookup. The catalogue returns TEXT and this file escapes it,

type Translate = (key: MessageKey, vars?: Readonly<Record<string, string>>) => string;

/**
 * The chip and the plain sentence for a subject's kind — ONE derivation, used by
 * `/p/:id`, `/is/:platform/:handle` and the PSA.
 *
 * Three surfaces disclosing the same fact in three separately-written phrasings
 * is how one of them ends up saying something slightly different about what an
 * AI character is, on the page a fan happens to land on.
 */
export function kindDisclosure(kind: string, m: Translate): { chip: string; plain: string } {
  const keys = DISCLOSURE_KEYS[kind as keyof typeof DISCLOSURE_KEYS];
  if (!keys) return { chip: "", plain: "" };
  return { chip: kindChip(kind, m(keys.chip)), plain: m(keys.plain) };
}

/**
 * THE TABLE, AND WHY IT IS A TABLE.

/**
 * Only an absolute https URL built from CONFIGURED env may reach a canonical
 * tag, an og:url or an app hand-off link. `esc()` makes a value inert as
 * markup; it does NOT stop `javascript:` or `data:` in an href, so escaping is
 * not the control here — this is (review LOW 7). Shared verbatim by
 * `seo-page.ts` and `psa.ts` so the three most-shared pages cannot drift apart;
 * the class includes `_`, `.` and `-` because handles do, and `extra` adds `#`
 * for the SPA's hash routes.
 */
export function absHttps(u: string | null | undefined, extra = ""): string | null {
  return u && new RegExp(`^https://[a-z0-9._\\-/${extra}]+$`, "i").test(u) ? u : null;
}

/** Verdict presentation per claim state. Icon + text label are MANDATORY and
 * the reserved hue is only an accent (DESIGN-SYSTEM §4) — these pages are
 * screenshotted and greyscaled, so a verdict must never be colour alone.
 * `unclaimed` is deliberately neutral: slate, an outline dot, and copy that
 * says absence of a claim is evidence of nothing. */
const STATE_VERDICT: Record<PassportView["claim_state"], VerdictKey> = { verified: "verified", claimed: "claimed", unclaimed: "unclaimed" };
const STATE_PILL: Record<PassportView["claim_state"], string> = { verified: "VERIFIED CREATOR", claimed: "CLAIMED", unclaimed: "UNCLAIMED" };

/** One account row, with TWO marks in the last cell and they answer different
 * questions.
 *
 * THE PROVENANCE MARK says WHO ASSERTED THIS, never that it is true: "✓ by
 * creator" is the passkey-signed, agency-unforgeable mark; "by authorized
 * agency" is the distinct, deliberately weaker delegated one; anything else is
 * explicitly marked as neither, so an operator-listed row can never borrow a
 * creator's signature.
 *
 * THE CONFIRMATION MARK says whether anybody has ever DEMONSTRATED control of
 * this exact account for this exact passport, and whether that demonstration is
 * still current. It appears only in the negative — a confirmed row says nothing
 * extra, an unconfirmed one says so — because the honest default on a page whose
 * whole job is "which accounts are really theirs" is that absence of proof is
 * visible rather than inferred from the absence of a badge.
 *
 * The two come apart precisely where the old model was wrong. A row Dhwaja
 * listed from public information and nobody has ever checked rendered
 * identically to one the creator proved last week, under the same Verified
 * banner, with only "○ listed by Dhwaja" (a statement about WHO, not about
 * WHETHER) between them.
 */
/**
 * ONE RULE FOR PRINTING AN ACCOUNT, and every surface that prints one uses it.
 *
 * A generic Tier-2 `link` (M3) is a URL, not a platform. `@livejasmin.com/en/chat/RileyMoon`
 * is a sentence Dhwaja has no business writing: the `@` form is how this product
 * spells "an account on a platform we model", and a link is by definition an
 * account on a platform we do NOT model and have vouched for nothing about. The
 * rule was written down and fixed HERE, in this one file, while the SPA's result
 * card, the crawlable `/is/` page and both native result screens went on
 * rendering the fake handle — three surfaces, one of them the search-engine
 * front door.
 *
 * So the rule is a function, exported, and the sites that cannot import it
 * (Swift, Kotlin) mirror it under `tests/invariants/safety-handoff-parity.test.ts`.
 *
 * Returns PLAIN TEXT, unescaped: each call site escapes for its own sink (HTML
 * here and in `seo-page.ts`, `Text(verbatim:)` on iOS, a Compose `Text` on
 * Android, `textClamp` in the SPA). Escaping here and again there is how a
 * creator called "D'Angelo" ends up as "D&#39;Angelo" on a page.
 */
export const LINK_PLATFORM = "link";

export interface AccountDisplay {
  /** What goes in the Platform column / platform slot. */
  platform: string;
  /** What goes in the identity slot: `@handle`, or a bare URL for a link. */
  ident: string;
}

export function accountDisplay(platform: unknown, handle: unknown): AccountDisplay {
  const p = typeof platform === "string" ? platform : "";
  const h = typeof handle === "string" ? handle : "";
  return p === LINK_PLATFORM ? { platform: "web link", ident: h } : { platform: p, ident: `@${h}` };
}

/** The Platform-column label — the shared rule, escaped for HTML. */
function platformLabel(a: PublicAccount): string {
  return esc(accountDisplay(a.platform, a.handle).platform);
}

/** The identity cell: the @handle (or, for a `link`, the bare URL — an @<url>
 * would look wrong and imply a platform account we vouch for), its status, and
 * the outbound anchor. Defense in depth: only an http(s) URL is ever clickable,
 * so a javascript:/data: value that somehow reached the DB is inert. */
/** I3-1136: does this account's url name its own handle (or is it a `link`)?
 * The SAME predicate the ledger's write door applies (@dhwaja/schema urlNamesAccount). */
function urlNamesHandle(a: PublicAccount): boolean {
  return urlNamesAccount(a.platform, a.handle, a.url);
}

function accountIdent(a: PublicAccount, m?: Translate, indexable = true): string {
  // I3-1136: the url column is supplied beside the proof and never bound to
  // it (verificationTarget uses platform/id_kind/id only). An Open link beside
  // a proved account must point AT that account: a `link` account (the url IS
  // the identity) or a url whose host/path names the handle. Otherwise none.
  const link = a.url && /^https?:\/\//i.test(a.url) && urlNamesHandle(a) ? ` <a href="${esc(a.url)}" rel="nofollow noopener">open</a>` : "";
  const state = a.status !== "active" ? ` <span class="m">(${esc(a.status)})</span>` : "";
  return `<code>${esc(accountDisplay(a.platform, a.handle).ident)}</code>${state}${link}${m ? answerLink(a, m, indexable) : ""}`;
}

/**
 * F15 / DIRECTION §3 — THE LINK THE `/is/` PAGE NEVER HAD.
 *
 * `renderSitemapXml`'s docblock kept those pages out of the sitemap on the
 * grounds that they are "discovered the same way /p/:id is: by crawling the
 * links that point at them". Nothing pointed at them, in the whole tree. This
 * is that link, emitted from the page that already publishes the same two
 * strings — which is also the page a human is on when they are asking the
 * question, so the affordance is not only for crawlers.
 *
 * THE CRAWL PATH IT COMPLETES: `sitemap.xml` → `/discover` and
 * `/browse/:category` → `/p/:id` (every board row links there) → here. A crawler
 * reaches an answer page only by first reaching a passport page, which 404s for
 * a suppressed or retracted subject at the same instant the boards drop it.
 *
 * That is why this is a link and not the `sitemap-is.xml` the TODO proposed —
 * but the distinction is narrower than "no artifact anybody can archive", and
 * saying that would be false: getting these pages crawled IS getting them
 * archived, by the index, by Common Crawl, by every SEO tool's URL history. The
 * difference that matters is AUTHORITY. A first-party `sitemap-is.xml` states
 * the set; absence on Tuesday is Dhwaja saying the URL is gone. A crawl-derived
 * index is partial, lags by days or weeks, and its absences are ambiguous
 * between delisted, not-recrawled and deranked. A delisting oracle needs a
 * signal the observer can trust, and that is precisely the property a sitemap
 * hands over and a crawl does not.
 *
 * `nofollow` FOR AN UNCLAIMED SUBJECT, because the page it points at is
 * `noindex,nofollow` for exactly that subject — the consent-by-claim rule this
 * renderer already applies to its own `canonical` and `og:url`. Inviting a
 * crawler toward a page that will tell it not to index is not merely wasteful;
 * it is this page recommending the indexing of somebody who has not consented.
 */
/** The exact shape `isPagePath` produces, re-asserted at the point of use.
 * Every other href on this page is scheme-checked (`/^https?:/` for an account
 * link, `absHttps` for the canonical and the app hand-off) and `ssr-shell`'s own
 * guard exists because "esc() does nothing about a `javascript:` scheme inside
 * an href" — a comment that then said "the account link is the ONLY anchor a row
 * can carry", which stopped being true the moment this function was added. The
 * producer is trusted today; the layer is what the siblings all have. */
const IS_PATH_RE = /^\/is\/[a-z0-9_]{1,24}\/[a-z0-9][a-z0-9._-]{0,63}$/;

function answerLink(a: PublicAccount, m: Translate, indexable: boolean): string {
  if (!a.is_path || !IS_PATH_RE.test(a.is_path)) return "";
  const rel = indexable ? "" : ' rel="nofollow"';
  return ` <a class="isk" href="${esc(a.is_path)}"${rel}>${esc(m("pv.is_link"))}</a>`;
}

/** The provenance + confirmation marks — WHO asserted this row, and whether
 * control was ever demonstrated. Wording is about the CHECK, never an
 * insinuation about the person; a stale and a never-taken proof read the same on
 * purpose (the ask, "prove it", is identical, and distinguishing them here would
 * publish how long ago a creator last touched their console). */
function accountMarks(a: PublicAccount, m: Translate): string {
  // M-24: each mark's fuller explanation lived ONLY in `title=` on a
  // non-focusable `<span>` — unreachable with a keyboard and on touch, with no
  // script on this page to add a click-triggered alternative. `tabindex="0"`
  // is the smallest fix that does not need one: the short visible label was
  // already text (never hidden), and the span is now a stop a keyboard user
  // can reach, which is what native `title` tooltips key off.
  const mark =
    a.touched_by === "creator"
      ? `<span class="mark creator" tabindex="0" title="${esc(m("pv.mark_creator_title"))}">${esc(m("mark.by_creator"))}</span>`
      : a.touched_by === "agency"
        ? `<span class="mark agency" tabindex="0" title="${esc(m("pv.mark_agency_title"))}">${esc(m("mark.by_agency"))}</span>`
        : `<span class="mark none" tabindex="0" title="${esc(m("pv.mark_none_title"))}">${esc(m("mark.listed_by_dhwaja"))}</span>`;
  const confirm = a.confirmed
    ? ""
    : ` <span class="mark unconfirmed" tabindex="0" title="${esc(m("pv.unconfirmed_title"))}">${esc(m("pv.unconfirmed"))}</span>`;
  return `${mark}${confirm}`;
}

/** ONE account on one platform → a plain three-column row. */
function accountRow(a: PublicAccount, m: Translate, indexable = true): string {
  return `<tr${a.status !== "active" ? ' class="inert"' : ""}><td>${platformLabel(a)}</td><td>${accountIdent(a, m, indexable)}</td><td>${accountMarks(a, m)}</td></tr>`;
}

/**
 * SEVERAL legitimate profiles on ONE platform (a creator with two official
 * Facebook pages, three regional TikToks, …) → a CHOOSER instead of N look-alike
 * rows. The fan clicks the platform and the official alternates disclose, so they
 * pick from a vetted set rather than guessing which "Facebook" is real — the
 * anti-impersonation point of the page, applied to the ambiguous case. Native
 * M-09: this used to be a `<details>` — closed by default, ALWAYS, whether or
 * not the alternates agreed on confirmation. This file's own no-JS contract
 * ("the pages still work with JavaScript entirely off" / "nothing meaningful
 * is hidden from a text reader", `ssr-shell.test.ts`) makes a per-alternate
 * "! not confirmed" mark exactly the kind of content that must not wait on a
 * click nothing on the page prompts a fan to make — two profiles disagreeing
 * about confirmation is the case this exists to surface, not to gate. Each
 * alternate keeps its OWN identity + marks, so an unconfirmed one in the group
 * still reads "! not confirmed" and is never laundered by a confirmed
 * sibling. Spans the Account+asserted columns; the platform stays in its own
 * column.
 */
function accountGroupRow(accounts: PublicAccount[], m: Translate, indexable = true): string {
  const anyInert = accounts.some((a) => a.status !== "active");
  const items = accounts
    .map(
      (a) =>
        `<li${a.status !== "active" ? ' class="inert"' : ""}>${accountIdent(a, m, indexable)}<span class="altmarks">${accountMarks(a, m)}</span></li>`,
    )
    .join("");
  // The summary is deliberately NEUTRAL ("N profiles"), not "N official
  // profiles": a group can hold an operator-listed row that is not control-proven
  // (it still renders "! not confirmed" inside), and calling the set "official"
  // would launder that. The per-alternate marks carry the confirmed/unconfirmed
  // truth; the chooser only says there is more than one to pick from.
  return `<tr${anyInert ? ' class="inert"' : ""}><td>${platformLabel(accounts[0]!)}</td><td colspan="2"><span class="altbadge">${accounts.length} profiles on record</span><ul class="rows alt">${items}</ul></td></tr>`;
}

/**
 * The host a Tier-2 link points at, for grouping only (never rendered). Falls
 * back to the raw id when the URL will not parse — two unparseable ids only
 * group together if they are identical, which is the safe direction.
 */
function linkHost(a: PublicAccount): string {
  const raw = (a.url && /^https?:\/\//i.test(a.url) ? a.url : `https://${a.handle}`).trim();
  try {
    return new URL(raw).hostname.toLowerCase();
  } catch {
    return a.handle.toLowerCase();
  }
}

/**
 * What makes two rows "the same platform" for the chooser below.
 *
 * FOR A TIER-2 LINK IT IS THE HOST, NOT THE `link` KEY. `link` is a synthetic
 * bucket meaning "somewhere we do not model yet", so grouping on it put a
 * creator's Patreon, their own website and a cam profile into one row reading
 * "web link · 3 profiles on record · choose one" — a chooser whose whole purpose
 * is to help a fan pick between profiles that are genuinely alternatives of each
 * other. Three different places are not alternatives, and asking a fan to choose
 * between them is the confusion this page exists to remove. Grouped by host, two
 * links only share a chooser when they are two pages on the same site, which is
 * exactly the case the chooser is for.
 */
function groupKey(a: PublicAccount): string {
  return a.platform === LINK_PLATFORM ? `${LINK_PLATFORM}:${linkHost(a)}` : a.platform;
}

/** Group a passport's accounts by platform (by HOST for a Tier-2 link),
 * preserving the incoming order (the query already sorts by platform then id, so
 * same-platform rows are contiguous). A group with one account renders as a
 * plain row; several become a chooser. */
function groupAccountsByPlatform(accounts: readonly PublicAccount[]): PublicAccount[][] {
  const order: string[] = [];
  const byPlatform = new Map<string, PublicAccount[]>();
  for (const a of accounts) {
    const key = groupKey(a);
    const bucket = byPlatform.get(key);
    if (bucket) bucket.push(a);
    else {
      byPlatform.set(key, [a]);
      order.push(key);
    }
  }
  return order.map((p) => byPlatform.get(p)!);
}

import type { DbLike, DbStatement } from "./db.js";

/**
 * Verified-creator payout tags. A creator displays payout destinations (crypto
 * addresses / payment-platform handles) that fans pay DIRECTLY — Dhwaja never
 * takes custody. A payout tag is the highest-stakes piece of creator trust: a
 * wrong tag sends real money to a thief. So:
 *  - Only VERIFIED creators may set tags.
 *  - Each tag is validated by type (crypto address charset, or an allowlisted
 *    platform handle) — a malformed value is refused, not displayed.
 *  - A CHANGE stages with a cooling-off before it renders, AND fires a Save-
 *    Creator alert so followers notice — an attacker who briefly compromised an
 *    account can't silently swap the address and drain a payment window.
 */

export const PAYOUT_COOLING_OFF_MS = 24 * 60 * 60 * 1000; // 24h before a change goes live
export const MAX_TAGS = 10;

export type TagType = (typeof PAYOUT_TAG_TYPES)[number];

export class PayoutError extends Error {
  override name = "PayoutError";
  constructor(message: string, public readonly status: 400 | 403 | 404 = 400) {
    super(message);
  }
}

export interface PayoutTagInput {
  type: TagType;
  value: string;
}

/** One vocabulary: the ledger's grammar and this validator read the same list. */
const TAG_TYPES = new Set<string>(PAYOUT_TAG_TYPES);

/** Per-type validation. Charset/format only — no chain calls. Fail closed. */
function validateTag(t: PayoutTagInput): void {
  // Shape first: untyped input (missing/non-string fields) is refused as a
  // PayoutError, not a raw TypeError.
  if (typeof t !== "object" || t === null || typeof (t as PayoutTagInput).type !== "string" || typeof (t as PayoutTagInput).value !== "string") {
    throw new PayoutError("each tag needs a string type and value");
  }
  if (!TAG_TYPES.has(t.type)) throw new PayoutError("unknown tag type");
  const v = t.value.trim();
  if (!v || v.length > 128) throw new PayoutError("tag value out of range");
  switch (t.type) {
    case "btc":
      if (!/^(bc1[0-9a-z]{20,80}|[13][1-9A-HJ-NP-Za-km-z]{25,39})$/.test(v)) throw new PayoutError("invalid btc address");
      break;
    case "eth":
      if (!/^0x[0-9a-fA-F]{40}$/.test(v)) throw new PayoutError("invalid eth address");
      break;
    case "sol":
      if (!/^[1-9A-HJ-NP-Za-km-z]{32,44}$/.test(v)) throw new PayoutError("invalid sol address");
      break;
    case "cashapp":
      if (!/^\$?[A-Za-z0-9_]{1,20}$/.test(v)) throw new PayoutError("invalid cashapp tag");
      break;
    case "venmo":
      if (!/^@?[A-Za-z0-9_-]{1,30}$/.test(v)) throw new PayoutError("invalid venmo handle");
      break;
    case "throne":
      if (!/^[A-Za-z0-9_-]{1,40}$/.test(v)) throw new PayoutError("invalid throne handle");
      break;
    case "other_url":
      if (!/^https:\/\/[a-z0-9.-]+(\/[^\s"'`<>]*)?$/i.test(v)) throw new PayoutError("other_url must be a plain https url");
      break;
    default:
      throw new PayoutError("unknown tag type");
  }
}

/** Builds the change-alert statements — for the creator's followers AND for the

/** Normalized, order-independent key for a tag set — used to detect a no-op
 * resubmission (identical to what's already staged-or-live). */
function tagKey(tags: { tag_type: string; tag_value: string }[]): string {
  return tags.map((t) => `${t.tag_type}\u0000${t.tag_value}`).sort().join("");
}

/**
 * THE EMPTY STAGED SET (B24-599). A staged set IS its rows, so "remove every
 * destination" used to stage nothing — and with nothing staged, nothing ever
 * replaced the live set: the old destination stayed live forever. An empty plan
 * now stages this one marker row. `tag_value = ''` is refused by `validateTag`
 * for every type, so no real destination can equal it. It is the staged
 * generation "zero destinations": it cools off like any change, retires the live
 * set when due, and is itself retired — never promoted to `live`. */
const REMOVAL_MARKER = { tag_type: "other_url", tag_value: "" } as const;
const isMarker = (r: { tag_value: string }): boolean => r.tag_value === REMOVAL_MARKER.tag_value;

/** The set that reflects the creator's current INTENT: the pending staged set
 * if one exists, else the live set. A resubmission equal to this is a no-op. */
async function currentIntentKey(db: DbLike, passportId: string): Promise<string> {
  return tagKey(await currentIntent(db, passportId));
}

/** The current staged-or-live intent as a tag list (not just its key). */
async function currentIntent(db: DbLike, passportId: string): Promise<{ tag_type: string; tag_value: string }[]> {
  const { results } = await db
    .prepare("SELECT tag_type, tag_value, status FROM payout_tags WHERE passport_id = ? AND status IN ('staged','live') ORDER BY id")
    .bind(passportId)
    .all<{ tag_type: string; tag_value: string; status: string }>();
  const staged = results.filter((r) => r.status === "staged");
  const chosen = staged.length > 0 ? staged.filter((r) => !isMarker(r)) : results.filter((r) => r.status === "live");
  return chosen.map((r) => ({ tag_type: r.tag_type, tag_value: r.tag_value }));
}

/**
 * THE LEDGER DRAFTS FOR A PAYOUT CHANGE — the thing the passkey signs.
 *
 * Built SERVER-SIDE from the validated set, and built identically on the
 * challenge leg and the append leg, so the two action hashes match. That is what
 * stops a client taking a challenge for one destination and appending another:
 * the assertion is bound to the hash of exactly these drafts, and the drafts are
 * derived from the tags the server just validated, never from anything the
 * client names.
 *
 * One event per tag entering and per tag leaving the set, in a DETERMINISTIC
 * order (removals first, each side sorted), because two calls that produce
 * different draft ORDER produce different action hashes and the assertion stops
 * verifying.
 */
export function payoutTagActions(
  before: { tag_type: string; tag_value: string }[],
  after: { tag_type: string; tag_value: string }[],
): CreatorAction[] {
  const key = (t: { tag_type: string; tag_value: string }): string => `${t.tag_type}\u0000${t.tag_value}`;
  const beforeKeys = new Set(before.map(key));
  const afterKeys = new Set(after.map(key));
  const removed = before.filter((t) => !afterKeys.has(key(t))).sort((a, b) => (key(a) < key(b) ? -1 : 1));
  const added = after.filter((t) => !beforeKeys.has(key(t))).sort((a, b) => (key(a) < key(b) ? -1 : 1));
  return [
    ...removed.map((t) => ({ event_type: "payout_tag_removed" as const, payload: { tag_type: t.tag_type, tag_value: t.tag_value } })),
    ...added.map((t) => ({ event_type: "payout_tag_added" as const, payload: { tag_type: t.tag_type, tag_value: t.tag_value } })),
  ];
}

/** What a payout change WOULD do — the shape both legs of the signing flow need. */

  unchanged: boolean;
  /** The ledger drafts this change appends (empty iff `unchanged`). */
  actions: CreatorAction[];
  /** The set that will be STAGED — deduped and trimmed, and exactly the set the
   * drafts above were computed over, so what is signed and what is stored can
   * never be two different things. */
  after: { tag_type: string; tag_value: string }[];
  /** When the staged set starts rendering. */
  effective_at: string;
}

/**
 * Validate a proposed payout set and work out what it changes — WITHOUT writing
 * anything. The challenge leg calls this to learn which drafts to bind the
 * assertion to; `setPayoutTags` calls it again on the append leg and gets the
 * same answer, because it is a pure function of (stored intent, submitted set).
 */
export async function planPayoutTagChange(
  db: DbLike,
  passportId: string,
  tags: PayoutTagInput[],
  nowMs: number,
): Promise<PayoutTagPlan> {
  if (!/^psp_[0-9a-f]{16}$/.test(passportId)) throw new PayoutError("bad passport id");
  if (tags.length > MAX_TAGS) throw new PayoutError(`at most ${MAX_TAGS} tags`);
  for (const t of tags) validateTag(t);

  // DEDUPED. A submitted set with the same destination twice would otherwise
  // stage two identical rows and append two identical `payout_tag_added` events
  // — a creator's page listing one address twice, and a ledger that says the
  // money moved there twice. A set is a set.
  const seen = new Set<string>();
  const after: { tag_type: string; tag_value: string }[] = [];
  for (const t of tags) {
    const row = { tag_type: t.type, tag_value: t.value.trim() };
    const k = `${row.tag_type}\u0000${row.tag_value}`;
    if (seen.has(k)) continue;
    seen.add(k);
    after.push(row);
  }
  const before = await currentIntent(db, passportId);

  // THE WRITE GATE IS THE CHOKEPOINT, NOT A LOCAL `if`. It asks all three
  // standing questions (claimed? retracted? suppressed?) on one statement — the
  // same refusal every other mint path gets — so a future door inherits the rule
  // instead of re-deriving it. It throws VerifiedCreatorError (403); routes map
  // it alongside PayoutError.
  //
  // M35: but the gate only guards MOVING money to a NEW place. Turning payouts
  // OFF — removing a destination, or clearing the set — never sends money
  // anywhere, and a creator whose verification LAPSED (verified→claimed) must not
  // be trapped with a live payout address they can no longer take down. So the
  // gate is required only when the change ADDS a destination not already
  // staged/live; a removal-only (or unchanged) plan skips it.
  const beforeKeys = new Set(before.map((b) => `${b.tag_type}\u0000${b.tag_value}`));
  const addsNew = after.some((a) => !beforeKeys.has(`${a.tag_type}\u0000${a.tag_value}`));
  // #637b (D-0926(3)): the folded `verified` state lags an expired grant until the nightly sweep, so the add
  // also asks the grant's own expiry at nowMs — the same `isCurrentlyVerified` predicate the promote step uses.
  if (addsNew) {
    await requireVerifiedCreator(db, passportId);
    if (!isCurrentlyVerified(await passportStanding(db, passportId, nowMs))) throw new VerifiedCreatorError("verified creators only");
  }

  const effective_at = new Date(nowMs + PAYOUT_COOLING_OFF_MS).toISOString();
  if (tagKey(after) === tagKey(before)) return { unchanged: true, actions: [], after, effective_at };
  return { unchanged: false, actions: payoutTagActions(before, after), after, effective_at };
}

/**
 * A payout change, signed. `assertion` is a WebAuthn assertion over exactly the
 * drafts `planPayoutTagChange` produces (obtain the challenge from the generic
 * `POST /passkey/sign/start` leg with the same actions).
 */
export interface PayoutSigning {
  assertion: AssertionResponse;
  cfg: WebAuthnConfig;
}

/**
 * Replace a verified creator's payout tags. The new set STAGES with a cooling-
 * off; the previous live tags stay live until it elapses (so a display isn't
 * blanked mid-change). The follower alert is created IN THE SAME BATCH as the
 * staging (all-or-nothing — a swap can never be staged with its alert dropped).
 * An unchanged resubmission is a no-op (no re-staging, no re-alert) so the route
 * can't be looped to spam followers.
 *
 * SIGNED, AND IN THE LEDGER. This is where a fan's money goes, and until now it
 * was the one creator action that took neither. `payout_tag_added` /
 * `payout_tag_removed` had existed in EVENT_TYPES since S1 as vocabulary the
 * ledger REFUSED — no payload arm, no actor policy, no effects branch — so a
 * destination swap left no hash-chained row anywhere, while authorizing an
 * agency, granting likeness consent or registering a release, all of them worth
 * less, each required a passkey AND an append. A leaked session cookie was
 * enough to move the address; nothing after the fact could prove when it moved
 * or what it had been.
 *
 * Now the assertion binds the exact set of {tag_type, tag_value} entering and
 * leaving, and the ledger append, the staging rows and the follower alerts all
 * ride ONE atomic batch (appendCreatorSigned's `extraStatements`): there is no
 * interleaving in which the tags move without the row that says so, or the row
 * lands without the alert that warns the followers.
 *
 * `signing` is OPTIONAL ONLY SO THE ROUTE CAN BE WIRED SEPARATELY — see the note
 * on the unsigned path below. It is not a supported way to change a destination.
 *
 * CALLER-GATING: must be invoked only for the authenticated creator's own
 * passport (the HTTP layer enforces the creator session).
 */
export async function setPayoutTags(
  db: DbLike,
  passportId: string,
  tags: PayoutTagInput[],
  nowMs: number,
  now: () => string = () => new Date().toISOString(),
  buildAlerts?: BuildChangeAlerts,
  signing?: PayoutSigning,
): Promise<{ effective_at: string; change_id: string; unchanged: boolean }> {
  const plan = await planPayoutTagChange(db, passportId, tags, nowMs);
  if (plan.unchanged) {
    // No-op: the submitted set matches the current staged-or-live intent. Don't
    // re-stage, re-alert or append (this is the anti-spam guard for a looped
    // route, and an unchanged set is not a ledger event).
    return { effective_at: plan.effective_at, change_id: "", unchanged: true };
  }

  const ts = now();
  const changeId = crypto.randomUUID();
  // THE PROMOTER'S STANDING RULE, ON THIS PROMOTION PATH TOO (B24-599 / RG-A
  // RGA-599-01). A removal-only plan no longer carries the batch gate, so the
  // in-batch promote below must ask what `promoteDueTags` asks: real due rows go
  // live only for a currently-verified creator — read here with the same
  // `passportStanding`, and re-asked inside the batch with `VERIFIED_AT_SUBQUERY` at
  // `nowIso` (D-1079(2): the same instant, so grant expiry is asked in both).
  const promotable = isCurrentlyVerified(await passportStanding(db, passportId, nowMs));
  const nowIso = new Date(nowMs).toISOString();
  // #637a: the promoter's subset rule on this path too — a due set that only takes
  // destinations down retires the live rows it leaves out whatever the standing; its
  // staged rows are consumed by the staged-retire below.
  const subsetDue = !promotable && (await dueStrictlyWithinLive(db, passportId, nowIso));
  const alertStmts = buildAlerts ? await buildAlerts(passportId, changeId, { destinations: plan.after.length, effective_at: plan.effective_at }) : [];
  const stmts = [
    // THE STANDING RE-CHECK, INSIDE THE WRITE BOUNDARY (audit v11, D5).
    //
    // `planPayoutTagChange` above asks `requireVerifiedCreator`, and that is a
    // READ. Part 5 moved every MINT path onto this gate and the route comment
    // came to say the write gate had moved with it — it had not. Only the read
    // had. A downgrade or a suppression committing in the window between the two
    // (measured at 1.5–214 ms, docs/VERIFIED-MINT.md §2) still landed a signed
    // `payout_tag_added` and its staged rows. The serve gate and `promoteDueTags`
    // meant nothing RENDERED, so it looked contained — but the staged rows
    // survive, and they promote the moment the creator re-verifies. That is a
    // destination change authorised under standing the creator no longer had,
    // arriving a day or a month later, which is precisely the shape the
    // "STAGED IS NOT OPTIONAL" note on `retractAllPayoutTags` was written for.
    //
    // It is FIRST in the array, and it is a gate rather than a `WHERE` clause,
    // for the reason the mint paths give: a conditional statement that matches
    // no row does not fail a D1 batch — it silently does nothing while the signed
    // event commits around it. `verifiedMintGate` supplies the standing predicate
    // to a NOT NULL column, so a passport that fails it rolls the whole batch
    // back. Same gate, same three questions, same `write_admissions` row as every
    // other signed batch — which is what finally makes 0022's premise ("every
    // signed batch touches `write_admissions`") true of the money path too.
    //
    // M35, AT THE WRITE (B24-599): the gate rides only a plan that ADDS a
    // destination — the same condition `planPayoutTagChange` applies to its read.
    // A removal-only plan moves money nowhere, and a creator whose standing lapsed
    // must be able to take a destination down; the unconditional gate refused it.
    ...(plan.actions.some((a) => a.event_type === "payout_tag_added") ? [verifiedMintGate(db, passportId, changeId, () => ts)] : []),
    ...(subsetDue ? [retireLiveOutsideDue(db, passportId, nowIso)] : []),
    // MATERIALISE WHAT IS ALREADY EFFECTIVE BEFORE REPLACING IT (S-070 SJ-02).
    //
    // `pickRendering` says a staged set whose `effective_at` has ARRIVED is what
    // renders — it replaces the live set on promotion, so a reader is already
    // being shown it whether or not `sweepDuePayoutTags` has run. The retire
    // below then took that set away while leaving the older PHYSICAL `live`
    // rows standing, so the reader fell back to a destination the creator had
    // replaced a day earlier. Two ordinary sequential creator actions, no
    // concurrency and no outage: whether the old address came back depended
    // entirely on whether a six-hourly cron happened to fire in between, which
    // is the same answer arriving at two different destinations.
    //
    // So the effective set is promoted HERE, in the SAME batch as its
    // replacement's staging — the only place the two can be reconciled
    // atomically. Order is load-bearing and D1 runs a batch in order: retire the
    // outgoing live rows FIRST, guarded on a due staged set actually existing
    // (so an edit with nothing due leaves the live set exactly as it was), then
    // promote. This is `promoteDueTags`' pair of statements, minus its own
    // standing re-read: `planPayoutTagChange` above already required
    // `requireVerifiedCreator` only for a plan that ADDS, so the promote pair
    // carries the promoter's own rule (`promotable` above, `VERIFIED_AT_SUBQUERY`
    // at `nowIso` in the statement — grant expiry included, D-1079(2)): a due set with real rows is promoted — and the live set
    // it replaces retired — only under current standing. A due REMOVAL (marker
    // only) retires the live set whatever the standing, as `promoteDueTags` does.
    // A due set refused here is retired below with the other staged rows: the
    // new submission replaces it and it never goes live.
    db
      .prepare(
        `UPDATE payout_tags SET status = 'retired'
          WHERE passport_id = ? AND status = 'live'
            AND EXISTS (SELECT 1 FROM payout_tags s WHERE s.passport_id = ? AND s.status = 'staged' AND s.effective_at <= ?)
            AND (NOT EXISTS (SELECT 1 FROM payout_tags r WHERE r.passport_id = ? AND r.status = 'staged' AND r.effective_at <= ? AND r.tag_value <> '')
                 OR ${promotable ? `EXISTS ${VERIFIED_AT_SUBQUERY}` : "0"})`,
      )
      .bind(passportId, passportId, nowIso, passportId, nowIso, ...(promotable ? [passportId, nowIso] : [])),
    ...(promotable
      ? [db.prepare(`UPDATE payout_tags SET status = 'live' WHERE passport_id = ? AND status = 'staged' AND effective_at <= ? AND tag_value <> '' AND EXISTS ${VERIFIED_AT_SUBQUERY}`).bind(passportId, nowIso, passportId, nowIso)]
      : []),
    // Retire any previously-staged (not-yet-live) change — the newest staging wins.
    // This also retires a DUE removal marker, which the promote above skipped.
    // After the two statements above this reaches only NOT-YET-DUE staged rows:
    // the due ones are `live` and are the set the new staging will replace.
    db.prepare("UPDATE payout_tags SET status = 'retired' WHERE passport_id = ? AND status = 'staged'").bind(passportId),
    // The PLAN's set, not the raw submission: deduped, trimmed, and the exact
    // set the drafts (and therefore the signature) were computed over.
    // An EMPTY set stages the removal marker, so the removal cools off and lands.
    ...(plan.after.length > 0 ? plan.after : [REMOVAL_MARKER]).map((t) => db.prepare("INSERT INTO payout_tags (passport_id, tag_type, tag_value, effective_at, status, created_at) VALUES (?, ?, ?, ?, 'staged', ?)").bind(passportId, t.tag_type, t.tag_value, plan.effective_at, ts)),
    // The follower alert commits WITH the staging — never one without the other.
    ...alertStmts,
  ];

  // THE GATE'S REFUSAL IS A STANDING REFUSAL, AND MUST SAY SO. Without this the
  // NOT NULL surfaces as a bare constraint error and index.ts answers 400
  // "error" — the same wrong-shaped answer `uploadCodes` and the release paths
  // each had to fix. `isWriteAdmissionRefusal` narrows to exactly this gate's
  // failure (not any constraint), so a UNIQUE or a FOREIGN KEY still propagates
  // as itself.
  try {
    if (signing) {
      // ONE BATCH: the standing gate, the signed ledger events, the staging rows
      // and the alerts. appendCreatorSigned verifies the assertion against the
      // hash of exactly these drafts BEFORE anything is written, and refuses a
      // draft whose subject is not this passport or whose type is not
      // CREATOR_SIGNABLE.
      await appendCreatorSigned(db, passportId, buildCreatorDrafts(passportId, plan.actions), signing.assertion, signing.cfg, nowMs, now, stmts);
    } else {
      // THE UNSIGNED PATH — kept only so the HTTP route can be moved onto the
      // signed one in a separate change, and green in between. It stages the tags
      // with no passkey and no ledger row, which is precisely the hole the signed
      // path closes. Nothing in trust-core should call this once the route passes
      // `signing`; it is not a fallback and must not become one. It carries the
      // standing gate all the same — an ungated door is not made safe by being
      // deprecated.
      await db.batch(stmts);
    }
  } catch (e) {
    if (isWriteAdmissionRefusal(e)) throw new VerifiedCreatorError();
    throw e;
  }
  return { effective_at: plan.effective_at, change_id: changeId, unchanged: false };
}

/** The NOT-YET-DUE staged set being discarded and the live set reverted to —
 * the two inputs a cancel's compensating drafts are computed from. One
 * statement.
 *
 * `effective_at > now` IS LOAD-BEARING, not a nicety. A staged set whose
 * cooling-off has ELAPSED is already what the public read renders
 * (`readLivePayoutTags`/`pickRendering` pick a due staged set over the live one,
 * even before the cron promotes the rows). Cancelling such a set would change
 * the PUBLICLY-DISPLAYED destination while firing no "payout changed — verify
 * before you send" alert — the exact silent-swap the alert exists to prevent. So
 * cancel only ever touches a change that is NOT YET rendering; that is what makes
 * "the public set never changed, so no alert" true by construction, and it is
 * the same predicate `pendingStaged` (the set the Cancel control is offered for)
 * uses. A due-but-unpromoted set is removed through the normal signed
 * `setPayoutTags` path, which DOES alert. */
async function stagedAndLive(
  db: DbLike,
  passportId: string,
  nowMs: number,
): Promise<{ pending: boolean; staged: { tag_type: string; tag_value: string }[]; live: { tag_type: string; tag_value: string }[] }> {
  const { results } = await db
    .prepare("SELECT tag_type, tag_value, status FROM payout_tags WHERE passport_id = ? AND (status = 'live' OR (status = 'staged' AND effective_at > ?)) ORDER BY id")
    .bind(passportId, new Date(nowMs).toISOString())
    .all<{ tag_type: string; tag_value: string; status: string }>();
  return {
    // A not-yet-due removal marker is a pending change whose set is empty.
    pending: results.some((r) => r.status === "staged"),
    staged: results.filter((r) => r.status === "staged" && !isMarker(r)).map((r) => ({ tag_type: r.tag_type, tag_value: r.tag_value })),
    live: results.filter((r) => r.status === "live").map((r) => ({ tag_type: r.tag_type, tag_value: r.tag_value })),
  };
}

/**
 * What a CANCEL would append: the exact INVERSE of the not-yet-due staging being
 * discarded — remove what it added, re-add what it removed — walking the ledger
 * intent back to the live set. `unchanged` when there is nothing NOT-YET-DUE to
 * cancel, OR when the staged set equals the live set (the inverse is empty —
 * a degenerate resubmission that promotes to an identical live set, so there is
 * nothing to sign and the retire is a no-op). A pure function of stored state,
 * computed identically on the challenge leg and the append leg so the two action
 * hashes match and the assertion binds — the same server-builds-the-drafts
 * discipline `planPayoutTagChange` follows. The client never names these actions
 * (the generic creator-action door refuses payout types); it can only ask to
 * cancel, and the server derives what that means.
 */
const restoresOnly = (actions: CreatorAction[]): boolean => actions.every((a) => a.event_type === "payout_tag_added");

export async function planCancelStagedPayout(db: DbLike, passportId: string, nowMs: number): Promise<{ unchanged: boolean; actions: CreatorAction[] }> {
  if (!/^psp_[0-9a-f]{16}$/.test(passportId)) throw new PayoutError("bad passport id");
  const { pending, staged, live } = await stagedAndLive(db, passportId, nowMs);
  if (!pending) return { unchanged: true, actions: [] };
  const actions = payoutTagActions(staged, live);
  // staged == live: nothing to compensate and nothing worth signing. Treat as a
  // clean no-op rather than trying to sign an EMPTY draft set (appendCreatorSigned
  // refuses zero drafts) — the redundant staged rows simply promote to the same
  // live set.
  if (actions.length === 0) return { unchanged: true, actions: [] };
  // CANCELLING A REMOVAL ONLY RESTORES (B24-599 / RG-A RGA-599-02). A staged set
  // that adds nothing — a removal, which a creator without standing may stage —
  // cancels into `payout_tag_added` drafts alone: it keeps destinations that were
  // leaving, which is an add, so it takes the add's standing. A mixed cancel (the
  // M35 lapsed-cancel of a replacement staged under standing) stays ungated.
  // PayoutError, not VerifiedCreatorError: the cancel routes map PayoutError.
  // I16-RG-MONEY:9: "the add's standing" is BOTH halves of the add's gate (:217-220) — the folded state AND the
  // grant's own expiry at nowMs (`isCurrentlyVerified`), which the fold lags until the nightly sweep (#637b).
  if (restoresOnly(actions)) {
    try {
      await requireVerifiedCreator(db, passportId);
      if (!isCurrentlyVerified(await passportStanding(db, passportId, nowMs))) throw new VerifiedCreatorError("verified creators only");
    } catch (e) {
      if (e instanceof VerifiedCreatorError) throw new PayoutError(e.message, 403);
      throw e;
    }
  }
  return { unchanged: false, actions };
}

/**
 * CANCEL A PENDING STAGED PAYOUT CHANGE (M14). Retires the creator's staged
 * (not-yet-live) rows and reverts to the live set, which keeps rendering
 * untouched. This is the discard the money screen's "change pending" panel
 * offers before the cooling-off elapses — distinct from `retractAllPayoutTags`,
 * which zeroes live AND staged for hijack recovery.
 *
 * SIGNED, LIKE EVERY PAYOUT MUTATION. A cancel changes what will render in the
 * future (staged → live), so it takes a passkey for the same reason
 * `setPayoutTags` does: a stolen session cookie must not touch the payout set —
 * here, by repeatedly blocking a creator's legitimate pending change. The
 * assertion binds the exact inverse drafts, and the ledger append and the
 * staged-row retirement ride ONE atomic batch (`extraStatements`). `signing` is
 * REQUIRED — unlike `setPayoutTags`, this operation and its route land in the
 * same change, so there is no transitional unsigned path to keep.
 *
 * SAFE-DIRECTION — DELIBERATELY UNGATED ON STANDING (M35). It reverts to the
 * already-live, already-authorised set and introduces NO new rendering
 * destination (the re-added drafts only re-assert destinations that are ALREADY
 * live), so there is no `requireVerifiedCreator` and no `verifiedMintGate`: a
 * creator whose verification lapsed must still be able to cancel a pending
 * change. And NO follower alert — the public set never changes on a cancel (the
 * staged set never rendered), so fans have nothing to re-verify.
 *
 * CALLER-GATING: only ever for the authenticated creator's own passport (the
 * HTTP layer enforces the creator session).
 */
export async function cancelStagedPayoutTags(
  db: DbLike,
  passportId: string,
  nowMs: number,
  signing: PayoutSigning,
  now: () => string = () => new Date().toISOString(),
): Promise<{ unchanged: boolean }> {
  const plan = await planCancelStagedPayout(db, passportId, nowMs);
  // Nothing NOT-YET-DUE to cancel: no write, and the assertion (if any) is simply
  // not consumed. The /start leg returns `unchanged` in this case, so a
  // well-behaved client never signs; a direct POST that does is answered without
  // touching a row.
  if (plan.unchanged) return { unchanged: true };
  const stmts = [
    // A restore-only cancel re-asks the standing inside the batch (see the plan).
    ...(restoresOnly(plan.actions) ? [verifiedMintGate(db, passportId, `cancel:${passportId}`, now)] : []),
    // Retire ONLY the not-yet-due staged rows — the exact set the inverse drafts
    // were computed over. `effective_at > ?` (the same `nowMs` the plan used)
    // keeps the retire aligned with the signed drafts and, with the `status =
    // 'staged'` guard, idempotent against a concurrent `promoteDueTags`: whichever
    // runs first, the row is no longer a not-yet-due 'staged' row for the loser,
    // so the change is never applied twice. A due staged row (already rendering)
    // is deliberately out of range — it is never cancelled without an alert.
    db.prepare("UPDATE payout_tags SET status = 'retired' WHERE passport_id = ? AND status = 'staged' AND effective_at > ?").bind(passportId, new Date(nowMs).toISOString()),
  ];
  // ONE BATCH: appendCreatorSigned verifies the assertion against the hash of
  // exactly these inverse drafts BEFORE anything is written, and refuses a draft
  // whose subject is not this passport or whose type is not CREATOR_SIGNABLE.
  try {
    await appendCreatorSigned(db, passportId, buildCreatorDrafts(passportId, plan.actions), signing.assertion, signing.cfg, nowMs, now, stmts);
  } catch (e) {
    if (isWriteAdmissionRefusal(e)) throw new PayoutError("verified creators only", 403);
    throw e;
  }
  return { unchanged: false };
}

/**
 * THE CURRENTLY-VERIFIED PREDICATE, FOLDED INTO THE TAG READ ITSELF.
 *
 * A payout destination is the only thing on a passport that moves real money,
 * and it was the one thing whose SERVE path never asked whether the creator was
 * still verified. `planPayoutTagChange` gated the WRITE; `livePassport` (the
 * only gate the public presentation had) covers `retracted` and suppression and
 * says nothing at all about `claim_state`. So a `correction` /
 * `downgrade_verification` — which deliberately leaves `retracted = 0` — took
 * the badge away and left the addresses rendering under it. De-verification is
 * exactly the population verified-creator.ts warns is the one the old checks
 * missed.
 *
 * IT IS A CORRELATED `EXISTS` ON THE SAME STATEMENT, NOT A SECOND READ, for the
 * reason `livePassport` and `passportStanding` give: a gate whose COST depends
 * on its answer is an oracle behind byte-identical responses, and
 * `tests/invariants/suppression-oracles.test.ts` measures the statement
 * sequence of `GET /p/:id/presentation` for a suppressed subject against a
 * never-listed one. Folding the predicate in keeps the read one statement for
 * every input; a separate `SELECT claim_state` before it would not.
 *
 * It asks all three questions — `claim_state = 'verified'`, `retracted = 0`,
 * NOT suppressed — i.e. the same three `requireVerifiedCreator` asks on the
 * write side, so the served set and the settable set can no longer disagree.
 * `t` is the payout_tags alias every caller below binds it against.
 *
 * IT ALSO ASKS THE GRANT'S DEADLINE (I3-1768, D-1082(1), I32-FX1-VERIFIED): the
 * latest grant or renewal has not lapsed at the read's instant (standing.ts
 * `grantLiveAtSql`, the rule `VERIFIED_AT_SUBQUERY` is built from). The fold
 * (`verification-sweep.ts` appends `verification_expired` and `claim_state`
 * walks back to `'claimed'`) is cursor-windowed, so until it reaches a lapsed
 * creator this clause is what stops their tip destinations being served. Binds
 * ONE value, the read's instant, after the caller's own binds.
 */
const CURRENTLY_VERIFIED = `EXISTS (
             SELECT 1 FROM passports p
              WHERE p.passport_id = t.passport_id
                AND p.claim_state = 'verified'
                AND p.retracted = 0
                AND NOT EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = p.passport_id)
                AND NOT ${CREATOR_CLOSED_SQL("p.passport_id")}
                AND ${grantLiveAtSql("p.passport_id")})`; // D-0964(D)/D-0960(2) (I20-FX3A): a closed creator's tip destinations are not served

/**
 * Compromise-recovery: IMMEDIATELY retire ALL of a creator's payout tags (live
 * and staged), no cooling-off. Removing a payout destination is a
 * safe-direction change (like a correction — removal is fast, addition is
 * slow), so a creator who spots a bad/compromised tag can zero it out at once
 * and re-add later through the normal cooling-off. Fires the change alert.
 *
 * STAGED IS NOT OPTIONAL, and it is the half a hijack recovery needs: a tag the
 * attacker staged an hour before detection has its own clock running, and both
 * `promoteDueTags` (cron) and `readLivePayoutTags` (the public read) would
 * surface it the moment `effective_at` passes — a day AFTER the operator
 * resolved the case.
 *
 * ONE BATCH, ALERTS INCLUDED. This used to be a bare `.run()` followed by an
 * out-of-band `notifyChange`, so a failure between them retired the
 * destinations while dropping the "payout changed — verify before you send"
 * notice that tells followers something happened. The alert statements now ride
 * the SAME atomic batch as the retirement, exactly as `setPayoutTags` does:
 * either both land or neither does. Build them with
 * `alerts.buildPayoutChangeStatements` (kept as a caller-supplied builder, not
 * an import, so this module's import closure stays the narrow one the paywall
 * guard walks).
 *
 * …AND THE BATCH IS NOW CONSTANT-WIDTH, which is what makes that argument safe
 * instead of self-defeating. `buildPayoutChangeStatements` used to return ONE
 * INSERT PER SAVER with no cap, so a compromised creator with 200,000 followers
 * hitting this — the emergency brake, the one path where the creator has already
 * lost control of their account — produced a batch D1 refuses. The whole write
 * rolled back, the hijacker's live AND staged destinations survived, and the
 * victim had no working lever. The atomicity was real and it was the mechanism
 * of the failure: the bigger the audience, the more certainly the brake did not
 * work. It returns three statements now (a bounded first page of saver notices,
 * a fan-out marker the cron drains in pages, and the creator's own notice),
 * whatever the follower count.
 */
export async function retractAllPayoutTags(
  db: DbLike,
  passportId: string,
  changeId: string,
  buildAlerts?: BuildChangeAlerts,
  /** QUEUE #942(a) (D-0946(1)): the caller's audit row, committed in the retirement's own batch. */
  extraStatements: DbStatement[] = [],
  /** I18-FXA D32 (D-0949(1)): statements run in the same batch BEFORE the retirement (a count of what it is about
   * to retire); their results lead the returned batch results. */
  leadingStatements: DbStatement[] = [],
): Promise<unknown[]> {
  if (!/^psp_[0-9a-f]{16}$/.test(passportId)) throw new PayoutError("bad passport id");
  // Zero destinations, effective NOW: a retraction is the one payout change with
  // no cooling-off, and the creator's notice says so in those words.
  const alertStmts = buildAlerts ? await buildAlerts(passportId, changeId, { destinations: 0, effective_at: null }) : [];
  return db.batch([
    ...leadingStatements,
    db.prepare("UPDATE payout_tags SET status = 'retired' WHERE passport_id = ? AND status IN ('live','staged')").bind(passportId),
    ...alertStmts,
    ...extraStatements,
  ]);
}

/** Promote staged tags whose cooling-off has elapsed to 'live' (retiring the
 * prior live set). Idempotent; run on read or by cron.
 *
 * THE STANDING RECHECK IS THE POINT OF THE SECOND READ. A staged row records a
 * decision taken up to `PAYOUT_COOLING_OFF_MS` ago, and promotion used to be a
 * pure clock: `effective_at <= now` and the row went live, whatever had
 * happened to the creator in between. That is how a tag staged an hour before a
 * hijack was detected goes live a day AFTER the operator resolved the case, and
 * how a de-verified creator's destination quietly starts rendering again. A
 * staged row may never outlive the standing that authorised it, so the promoter
 * asks `passportStanding` — the one authoritative derivation — and simply does
 * not promote otherwise. It does NOT retire the rows: refusing to promote is
 * the safe direction and is reversible if the creator re-verifies, whereas
 * destroying a set on a transient lapse is not. Recovery's own
 * `retractAllPayoutTags` is what deletes an attacker's staging outright. */
/**
 * #637a (D-0920 option A): a due staged set that is a STRICT SUBSET of the live set adds no
 * destination — promoting it only takes destinations down — so it lands without the standing
 * check, the same reasoning as the removal marker (M35). `dueWithinLiveSql` is that fact as SQL,
 * bound into each statement that acts on it (INV-058); `dueStrictlyWithinLive` is the read.
 * `hi` bounds the due rows to the ones the read counted.
 */
function dueWithinLiveSql(hi: boolean): string {
  const due = `FROM payout_tags d WHERE d.passport_id = ? AND d.status = 'staged' AND d.effective_at <= ?${hi ? " AND d.id <= ?" : ""} AND d.tag_value <> ''`;
  // Both halves: the due set still EXISTS (a writer that consumed it in the window leaves nothing
  // to act on) and none of it is outside the live set.
  return `(EXISTS (SELECT 1 ${due}) AND NOT EXISTS (SELECT 1 ${due}
            AND NOT EXISTS (SELECT 1 FROM payout_tags l WHERE l.passport_id = d.passport_id AND l.status = 'live' AND l.tag_type = d.tag_type AND l.tag_value = d.tag_value)))`;
}

async function dueStrictlyWithinLive(db: DbLike, passportId: string, nowIso: string, hi?: number): Promise<boolean> {
  const { results } = await db
    .prepare(`SELECT tag_type, tag_value, status FROM payout_tags WHERE passport_id = ? AND (status = 'live' OR (status = 'staged' AND effective_at <= ?${hi === undefined ? "" : " AND id <= ?"} AND tag_value <> ''))`)
    .bind(passportId, nowIso, ...(hi === undefined ? [] : [hi]))
    .all<{ tag_type: string; tag_value: string; status: string }>();
  const k = (r: { tag_type: string; tag_value: string }): string => `${r.tag_type}\u0000${r.tag_value}`;
  const live = new Set(results.filter((r) => r.status === "live").map(k));
  const due = new Set(results.filter((r) => r.status === "staged").map(k));
  return due.size > 0 && [...due].every((x) => live.has(x)) && [...live].some((x) => !due.has(x));
}

/** #637a: retire the live rows the due subset leaves out, guarded by the subset fact itself. */
function retireLiveOutsideDue(db: DbLike, passportId: string, nowIso: string, hi?: number): DbStatement {
  const bound = hi === undefined ? "" : " AND d.id <= ?";
  const hiArgs = hi === undefined ? [] : [hi];
  return db
    .prepare(
      `UPDATE payout_tags SET status = 'retired' WHERE passport_id = ? AND status = 'live'
         AND NOT EXISTS (SELECT 1 FROM payout_tags d WHERE d.passport_id = payout_tags.passport_id AND d.status = 'staged' AND d.effective_at <= ?${bound} AND d.tag_value <> ''
                           AND d.tag_type = payout_tags.tag_type AND d.tag_value = payout_tags.tag_value)
         AND ${dueWithinLiveSql(hi !== undefined)}`,
    )
    .bind(passportId, nowIso, ...hiArgs, passportId, nowIso, ...hiArgs, passportId, nowIso, ...hiArgs);
}

export async function promoteDueTags(db: DbLike, passportId: string, nowMs: number): Promise<void> {
  const nowIso = new Date(nowMs).toISOString();
  const due = await db
    .prepare("SELECT COUNT(*) AS n, COALESCE(SUM(tag_value <> ''), 0) AS real, MAX(id) AS hi FROM payout_tags WHERE passport_id = ? AND status = 'staged' AND effective_at <= ?")
    .bind(passportId, nowIso)
    .first<{ n: number; real: number; hi: number | null }>();
  if ((due?.n ?? 0) === 0) return;
  // A due generation that is ONLY the removal marker adds nothing: retiring is the
  // safe direction, so it lands whatever the standing (M35). Anything else keeps
  // the standing re-check.
  const real = Number(due?.real ?? 0) > 0;
  if (real && !isCurrentlyVerified(await passportStanding(db, passportId, nowMs))) {
    // #637a: without standing, only a due set that takes destinations DOWN lands.
    const hiOnly = Number(due?.hi ?? 0);
    if (!(await dueStrictlyWithinLive(db, passportId, nowIso, hiOnly))) return;
    await db.batch([
      retireLiveOutsideDue(db, passportId, nowIso, hiOnly),
      db
        .prepare(`UPDATE payout_tags SET status = 'retired' WHERE passport_id = ? AND status = 'staged' AND effective_at <= ? AND id <= ? AND ${dueWithinLiveSql(true)}`)
        .bind(passportId, nowIso, hiOnly, passportId, nowIso, hiOnly, passportId, nowIso, hiOnly),
    ]);
    return;
  }
  // I3-1234 (INV-058): the batch acts on what the read ESTABLISHED, not on the
  // shape. Every statement is bound to the rows the read counted (`id <= hi`),
  // and a real due set re-asks the standing IN the statement
  // (`VERIFIED_AT_SUBQUERY` at `nowIso`, as `setPayoutTags` does), so a de-verification
  // landing between the read above and this batch leaves the live set and the
  // staged set both untouched. The live set is retired only if its replacement
  // is still there to be promoted.
  const hi = Number(due?.hi ?? 0);
  const gate = real ? ` AND EXISTS ${VERIFIED_AT_SUBQUERY}` : "";
  const gateArgs = real ? [passportId, nowIso] : [];
  await db.batch([
    db
      .prepare(
        `UPDATE payout_tags SET status = 'retired' WHERE passport_id = ? AND status = 'live' AND EXISTS (SELECT 1 FROM payout_tags s WHERE s.passport_id = ? AND s.status = 'staged' AND s.effective_at <= ? AND s.id <= ?)${gate}`,
      )
      .bind(passportId, passportId, nowIso, hi, ...gateArgs),
    db.prepare(`UPDATE payout_tags SET status = 'live' WHERE passport_id = ? AND status = 'staged' AND effective_at <= ? AND id <= ? AND tag_value <> ''${gate}`).bind(passportId, nowIso, hi, ...gateArgs),
    // The due removal marker is consumed, never promoted (B24-599).
    db.prepare("UPDATE payout_tags SET status = 'retired' WHERE passport_id = ? AND status = 'staged' AND effective_at <= ? AND id <= ? AND tag_value = ''").bind(passportId, nowIso, hi),
  ]);
}

export interface LivePayoutTag {
  type: TagType;
  value: string;
}

/**
 * READ-ONLY view of the currently-rendering tags — what `promoteDueTags`
 * followed by a plain read would show, without the write: if any staged set has cleared its
 * cooling-off it is what renders (it replaces the prior live set on
 * promotion); otherwise the live set renders. Safe on public, anonymous
 * GETs (a public route must never write); the actual row promotion happens
 * in the cron sweep (sweepDuePayoutTags) or on authenticated reads.
 *
 * ONE STATEMENT, AND THE STANDING GATE IS INSIDE IT. Both candidate sets are
 * fetched together and the "staged-and-due wins" rule is applied in JS, so the
 * public read costs exactly one round trip for every input and the
 * currently-verified predicate cannot be forgotten by a caller — there is no
 * ungated way to ask this question. `ownPayoutTags` is the deliberate,
 * authenticated exception, and it says so.
 */
export async function readLivePayoutTags(db: DbLike, passportId: string, nowMs: number): Promise<LivePayoutTag[]> {
  return pickRendering(await selectRenderable(db, passportId, nowMs, true));
}

interface RenderableRow {
  tag_type: TagType;
  tag_value: string;
  status: string;
}

/** The one SELECT behind both the public read and the creator's own read. When
 * `gated`, the currently-verified predicate rides the same statement. */
async function selectRenderable(db: DbLike, passportId: string, nowMs: number, gated: boolean): Promise<RenderableRow[]> {
  const { results } = await db
    .prepare(
      `SELECT tag_type, tag_value, status
         FROM payout_tags t
        WHERE t.passport_id = ?
          AND (t.status = 'live' OR (t.status = 'staged' AND t.effective_at <= ?))${gated ? `\n          AND ${CURRENTLY_VERIFIED}` : ""}
        ORDER BY id`,
    )
    .bind(passportId, new Date(nowMs).toISOString(), ...(gated ? [new Date(nowMs).toISOString()] : []))
    .all<RenderableRow>();
  return results;
}

/** A due staged set REPLACES the live one on promotion, so it is what renders. */
function pickRendering(rows: RenderableRow[]): LivePayoutTag[] {
  const staged = rows.filter((r) => r.status === "staged");
  // A due removal marker is a due EMPTY set: it wins, and renders nothing.
  const chosen = staged.length > 0 ? staged.filter((r) => !isMarker(r)) : rows.filter((r) => r.status === "live");
  return chosen.map((r) => ({ type: r.tag_type, value: r.tag_value }));
}

/**
 * WHY A DE-VERIFIED CREATOR IS NOT SHOWN A BLANK SCREEN.
 *
 * The creator console reads its own payout set to prefill the full-set
 * replacement form, and it used to read it off the PUBLIC presentation. Gating
 * the public read (correctly) would therefore have blanked the creator's own
 * money screen with no explanation the moment their verification lapsed — and a
 * screen that silently says "you have no payout destinations" to someone who
 * has four is a worse failure than the one being fixed: full-set replacement
 * over an empty box DELETES the set.
 *
 * So the two reads are split. The public one is gated and stays silent; this
 * one is authenticated (the caller's own passport, enforced at the route), is
 * NOT gated, and returns the truth plus whether fans can currently see it. The
 * console renders the destinations and the sentence "not currently displayed
 * publicly — your verification has lapsed".
 *
 * `display_reason` is deliberately COARSE. It never returns `suppressed` or
 * distinguishes it from a retracted/absent passport (standing.ts: that
 * distinction is the one oracle removal exists to deny), and it is only ever
 * computed for the caller's own passport, so it tells the subject nothing they
 * do not already know.
 */
export type PayoutDisplayReason = "verified" | "verification_lapsed" | "not_verified" | "unavailable";

export interface OwnPayoutTagView {
  /** What WOULD render publicly if standing allowed it — the creator's real set. */
  tags: LivePayoutTag[];
  /** Whether `GET /p/:id/presentation` is serving them right now. */
  publicly_displayed: boolean;
  display_reason: PayoutDisplayReason;
  /**
   * A change the creator has staged that has NOT yet taken over the public view
   * — the destinations that will render once the cooling-off elapses, and when
   * (M14). Null when nothing is pending. Only NOT-YET-DUE staged rows populate
   * it: a staged set past its `effective_at` is already what `tags` renders (via
   * `pickRendering`), so it is not reported here a second time. This is the one
   * place the creator can see a pending change before it goes live — and the set
   * the Cancel control (`cancelStagedPayoutTags`) discards.
   */
  pending: { tags: LivePayoutTag[]; effective_at: string } | null;
}

/** The creator's not-yet-due staged change, if any. All staged rows from one
 * change share an `effective_at` (a new staging retires the prior one), so there
 * is at most one generation. Own-passport read only — never a public path. */
async function pendingStaged(db: DbLike, passportId: string, nowMs: number): Promise<{ tags: LivePayoutTag[]; effective_at: string } | null> {
  const { results } = await db
    .prepare("SELECT tag_type, tag_value, effective_at FROM payout_tags WHERE passport_id = ? AND status = 'staged' AND effective_at > ? ORDER BY id")
    .bind(passportId, new Date(nowMs).toISOString())
    .all<{ tag_type: TagType; tag_value: string; effective_at: string }>();
  if (results.length === 0) return null;
  // A pending removal (the marker alone) reports an empty pending set.
  return { tags: results.filter((r) => !isMarker(r)).map((r) => ({ type: r.tag_type, value: r.tag_value })), effective_at: results[0]!.effective_at };
}

function displayReason(s: PassportStanding | null): PayoutDisplayReason {
  if (s === "verified") return "verified";
  if (s === "reverification_required") return "verification_lapsed";
  if (s === "claimed" || s === "unclaimed") return "not_verified";
  return "unavailable"; // retracted, suppressed, or no such passport — never distinguished
}

/** The creator's OWN payout set plus whether the public read is serving it.
 * CALLER-GATING: only ever for the authenticated creator's own passport. */
export async function ownPayoutTags(db: DbLike, passportId: string, nowMs: number): Promise<OwnPayoutTagView> {
  if (!/^psp_[0-9a-f]{16}$/.test(passportId)) throw new PayoutError("bad passport id");
  const standing = await passportStanding(db, passportId, nowMs);
  const tags = pickRendering(await selectRenderable(db, passportId, nowMs, false));
  const pending = await pendingStaged(db, passportId, nowMs);
  return { tags, publicly_displayed: isCurrentlyVerified(standing) && tags.length > 0, display_reason: displayReason(standing), pending };
}

/** Cron: promote every passport with a due staged set (bounded per sweep). */
export async function sweepDuePayoutTags(db: DbLike, nowMs: number, limit = 200): Promise<number> {
  const { results } = await db
    .prepare("SELECT DISTINCT passport_id FROM payout_tags WHERE status = 'staged' AND effective_at <= ? LIMIT ?")
    .bind(new Date(nowMs).toISOString(), limit)
    .all<{ passport_id: string }>();
  for (const r of results) await promoteDueTags(db, r.passport_id, nowMs);
  return results.length;
}
