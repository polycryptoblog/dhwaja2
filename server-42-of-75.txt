/**
 * F18 / master-list #32 — ACCOUNT CLOSURE, CONFISCATION AND APPEAL.
 *
 * CLOSING THE FOUNDER'S FAN ACCOUNT ENDS THE CREATOR SESSION (D-1009(1): I3-1422 stands; I3-1426 a). Every statement in
 * THIS module keys on a fan `account_id` (`account_closures`, `accounts.session_epoch`, `credit_ledger`, the fan's Stripe
 * rows) and none writes `passports`, a claim, `creator_closures` / `creator_suspensions` or `creator_sessions`; the
 * creator plane READS the closure instead: route-helpers.ts `creatorAdmissionForSubject` (I3-1422) refuses a creator
 * session whose `founder` account is suspended or terminated (pinned by test/i4-13r3-route-helpers.test.ts).
 * ONE EXCEPTION, D-1025(3) (I28-FX-ERASE): `terminateAccount` owes the founded passport's face-enrolment erasure
 * (`image_consent_revoke_outbox`, likeness.ts) in its flip's batch, as any termination does.
 *
 * THE ONE OPERATOR LEVER THAT EXISTED WAS ERASURE. `POST /ops/accounts/:id/delete`
 * destroys the account, under dual control, irreversibly — and it destroys the
 * evidence with it. The only other lever, `accounts.banned_at`, is set solely by
 * a chargeback dispute in `refunds.ts` and gates only money. So an operator who
 * had verified theft, market manipulation or economic abuse could erase, or do
 * nothing.
 *
 * THE POLICY, decided 2026-09-04, and this module is that policy in code:
 *
 *  1. TWO STAGES. `suspendAccount` is reversible and starts the appeal clock;
 *     `terminateAccount` is a separate act. Verified theft and market
 *     manipulation are the two grounds an operator is least likely to be sure
 *     of, and a wrong call must be recoverable.
 *  2. FREEZE ALL, FORFEIT ONLY WHAT WE INVENTED. On termination, promotional,
 *     referral, game and admin credits are forfeited; cash-backed credits and
 *     owed creator payouts remain a debt we owe (D-0619, D-0645), and
 *     collectibles stay recorded as theirs. WHAT THE CODE DOES WITH EACH:
 *       - the forfeit (`forfeitStmt`) is ONE `credit_ledger` append of the
 *         non-cash residual of ONE `account_id`, re-asked at commit. It reads
 *         only `credit_ledger` and `account_closures` and writes no other row.
 *       - cash-backed credits (`marketplace_proceeds` — creator proceeds —
 *         included) are retained and repaid by `repayTerminatedBalance`: operator
 *         triggered, to the last passkey-established payout address, never
 *         twice for one balance; with NO address on record it REFUSES and pays nothing
 *         (D-0644, D-0763), code `no_payout_address`, pointing the person to
 *         support — no collection path is built (D-0789).
 *       - earned creator money is `passport_id`-keyed in `sales` /
 *         `creator_payouts` (folded by `creator-payout.ts` `creatorEarnings`), in
 *         cents, and NO termination statement touches it: the weekly payout run
 *         keeps paying it to the creator's own passkey-bound binding under
 *         UNIQUE(passport_id, period_key) and the transfer idempotency key.
 *         Guarded by `test/b27-payout-owed-failed.test.ts` (#486).
 *  3. THIRTY DAYS, BY EMAIL, REVIEWED BY A PERSON. An appeal does not lift the
 *     suspension, but termination cannot execute while one is open and
 *     undecided, or before the deadline — enforced HERE rather than by an
 *     operator's memory.
 *  4. THE CATEGORY IS TOLD, THE EVIDENCE IS NOT. `category` is required; `note`
 *     is operator-only and no reader serves it to the account.
 */

export async function closureFor(db: DbLike, accountId: string): Promise<ClosureRow | null> {
  return (await db.prepare("SELECT * FROM account_closures WHERE account_id = ?").bind(accountId).first<ClosureRow>()) ?? null;
}

/**
 * IS THIS ACCOUNT FROZEN? Callers: war-match.ts (both sides of a match). The
 * predicate itself is FROZEN_CLOSURE_STATES_SQL (credits.ts), shared with every
 * SQL site that asks the same question inside its own statement (I3-0389).
 *
 * Deliberately NOT folded into `accountBanned`. That is the chargeback money
 * gate, its eight call sites are audited as such, and a closure is a different
 * fact; one name for two facts is how a later reader gets one of them wrong.
 *
 * `lifted` is not frozen — that is the whole point of a reversible suspension —
 * and a row's mere existence is never the test.
 */
export async function accountFrozen(db: DbLike, accountId: string): Promise<boolean> {
  const row = await db.prepare(`SELECT state FROM account_closures WHERE account_id = ? AND state IN ${FROZEN_CLOSURE_STATES_SQL}`).bind(accountId).first<{ state: string }>();
  return row !== null && row !== undefined;
}

/** What a sign-in door tells somebody who has just PROVED they hold the account:
 * the ground and the route, and nothing else (decision 4). Null when the account
 * is not closed. I3-0452 (D-0920 A): one shape per state, both carrying the
 * `account_id` docs/legal/APPEALS.md asks the person to quote; a TERMINATED
 * account carries no `appeal_deadline`, because `openAppeal` refuses every state
 * but `suspended` and a date that can no longer be met is not a route. */
export type ClosureNotice =
  | { error: "account_closed"; state: "suspended"; account_id: string; category: ClosureCategory; appeal_deadline: string; appeal_to: string }
  | { error: "account_closed"; state: "terminated"; account_id: string; category: ClosureCategory; appeal_to: string };
export async function closureNotice(db: DbLike, accountId: string, env?: Record<string, unknown> | null): Promise<ClosureNotice | null> {
  const row = await closureFor(db, accountId);
  if (!row || row.state === "lifted") return null;
  if (row.state === "terminated") return { error: "account_closed", state: "terminated", account_id: row.account_id, category: row.category, appeal_to: supportAddress(env) };
  return { error: "account_closed", state: "suspended", account_id: row.account_id, category: row.category, appeal_deadline: row.appeal_deadline, appeal_to: supportAddress(env) };
}

/** I3-0040 (D-1078(29); copy approved, sheet-2 W56, D-1365): the sent closure notice — what `closureNotice` tells a

export function closureNoticeSummary(n: { state: "suspended"; account_id: string; category: ClosureCategory; appeal_deadline: string; appeal_to: string } | { state: "terminated"; account_id: string; category: ClosureCategory; appeal_to: string }): string {

/** I3-0040: the closure notice's INSERT, predicated on THIS call's flip (`guard`: this account's row carries this call's
 * own `suspended_at` / `terminated_at`), so a lost race or a declined flip sends nothing. One notice per closure episode and state: `closure:<state>:<closure_id>`. */
function closureNoticeStmt(db: DbLike, accountId: string, state: "suspended" | "terminated", summary: string, ts: string, guard: string, guardBinds: readonly unknown[]): DbStatement {
  return db
    .prepare(
      `INSERT INTO notifications (account_id, passport_id, event_id, kind, summary, created_at)
       SELECT ?, '', ? || ? || ':' || closure_id, 'account_closed', ?, ? FROM account_closures WHERE ${guard}`,
    )
    .bind(accountId, CLOSURE_NOTICE_EVENT_PREFIX, state, summary, ts, ...guardBinds);
}

const dayMs = 86_400_000;

/**
 * QUEUE #486 / D-0645 — THE FORFEIT'S SCOPE, SPELLED AS THE EXPRESSION THAT
 * TAKES IT, so the scope and the taking cannot drift apart.
 *
 * This is `forfeiturePreview`'s own arithmetic — `MAX(0, total - MAX(0,
 * MIN(total, cashOriginSum)))` — as a correlated sub-select, so the committing
 * INSERT can re-ask it at its own instant instead of trusting a number read
 * beforehand. Bind ONE `accountId`.
 *
 * B26-CLOSE: FREEZE-NEUTRAL. It sums only rows whose `reason` is not a D-0744
 * freeze or release (`FREEZE_REASONS_SQL`), so an open dispute freeze neither
 * lowers the forfeit nor declines this compare-and-swap; see `forfeiturePreview`.
 */
const FORFEIT_SQL = `(SELECT MAX(0, COALESCE(SUM(credits), 0) - MAX(0, MIN(COALESCE(SUM(credits), 0), COALESCE(SUM(CASE WHEN origin IN ${CASH_ORIGINS_SQL} THEN credits ELSE 0 END), 0)))) FROM credit_ledger WHERE account_id = ? AND reason NOT IN ${FREEZE_REASONS_SQL})`;
/** #641: the preview's freeze-neutral cash-backed figure (`cashUnfrozen`),
 * re-derived in SQL over `FORFEIT_SQL`'s own population. A non-cash grant
 * cannot leave both this and `FORFEIT_SQL` unchanged; a freeze moves neither.
 * Binds ONE `accountId`. */
const UNFROZEN_CASH_SQL = `(SELECT MAX(0, MIN(COALESCE(SUM(credits), 0), COALESCE(SUM(CASE WHEN origin IN ${CASH_ORIGINS_SQL} THEN credits ELSE 0 END), 0))) FROM credit_ledger WHERE account_id = ? AND reason NOT IN ${FREEZE_REASONS_SQL})`;
/** I29-FANERASE H5 (I29-OPS): `forfeiturePreview`'s `retained_cash_backed` — max(0, min(balance, cash-origin sum)) over
 * every credit_ledger row of ONE account — as SQL, the one form `previewFigures` reads it through and an in-batch guard
 * (account_data.ts's retained-balance erasure validator) imports instead of re-typing. Binds ONE `accountId`. */
export const RETAINED_CASH_BACKED_SQL = `(SELECT MAX(0, MIN(COALESCE(SUM(credits), 0), COALESCE(SUM(CASE WHEN origin IN ${CASH_ORIGINS_SQL} THEN credits ELSE 0 END), 0))) FROM credit_ledger WHERE account_id = ?)`;

/**
 * SUSPEND. Reversible, starts the clock, and refuses to reopen a case that is
 * already open — a second `close` on a suspended account is an operator acting
 * on stale information, and silently overwriting the category would rewrite what
 * the person was told.
 */
export async function suspendAccount(
  db: DbLike,
  accountId: string,
  category: ClosureCategory,
  operator: string,
  note = "",
  now: () => string = () => new Date().toISOString(),
  nowMs: number = Date.now(),
  /** I3-0210 b (D-0976(2)): the Stripe rail an `account_compromise` suspension pauses collection on. Absent is
   * recorded (`skipped ... reason=stripe_not_wired`), never read as a pause. */
  billing?: TerminationBilling,
  /** I3-0040: the environment the sent notice's appeal address is read from (`supportAddress`, as `closureNotice`). */
  env?: Record<string, unknown> | null,
  /** I39-OPS (D-1158(6b)): the caller's rows (dual_control.ts coSignerAuditStmts), joined into this function's batch. */
  extra: readonly DbStatement[] = [],
): Promise<ClosureRow> {
  if (!isClosureCategory(category)) throw new ClosureError("category must be one of the enumerated grounds");
  const acct = await db.prepare("SELECT account_id FROM accounts WHERE account_id = ?").bind(accountId).first<{ account_id: string }>();
  if (!acct) throw new ClosureError("no such account", 404);
  const existing = await closureFor(db, accountId);
  if (existing && existing.state !== "lifted") throw new ClosureError(`this account is already ${existing.state}`, 409);
  const ts = now();
  // I3-0166 (I21-Q24, "clock starts on send"): the act's deadline, which the sign-in notice shows; the email leg moves it
  // to the SEND instant + APPEAL_WINDOW_DAYS when it hands this notice over (alerts.ts `closureNoticeSentStmts`).
  const deadline = new Date(nowMs + APPEAL_WINDOW_DAYS * dayMs).toISOString();
  // I3-0210 — WHAT IS STILL BILLING WHEN THE DOOR SHUTS.
  //
  // Suspension freezes the session (the session check, oidc.ts
  // `currentSessionEpoch`, carries its own FROZEN_CLOSURE_STATES_SQL clause;
  // `accountFrozen` here is called only by war-match.ts `joinQueue` and
  // `tryPairWithWaiter`) and says NOTHING about money. The store subscription
  // renews and the Stripe subscription charges, and the person cannot cancel either: they cannot
  // sign in, and `alerts.ts` settles every notification to a closed account
  // without sending it, so nothing reaches them on any rail. Before this line
  // there was no row anywhere stating that the charges continue — FAMILY-003's
  // exact shape, an exit that leaves a state nobody can find and no next actor.
  //
  // THE SAME PREDICATES BOTH ERASURE DOORS USE, imported rather than re-typed
  // (FAMILY-004 P1): `readStoreRails` excludes the Stripe-shaped `unknown` rail
  // that would otherwise be double-counted, and `readStripeSubscriptionIds`
  // reads `billing_entitlements`. A second hand-written copy of either query
  // here would drift from the doors that act on it.
  //
  // A RECORD, NOT A CANCELLATION. Cancelling on a REVERSIBLE suspension is a
  // product decision (an overturned appeal cannot un-cancel a store
  // subscription). I3-0210 b (D-0976(2)): an `account_compromise` suspension
  // PAUSES Stripe collection after the flip lands (`pauseCompromiseStripe`) and
  // a lift or an overturned appeal resumes exactly what it paused; store rails
  // and every other ground are still this record only.
  // Reading fails OPEN: a billing read that throws must not stop a suspension
  // an operator has already decided on, so the residual records what could not
  // be read instead of an assumed empty.
  let residual = "";
  try {
    const [rails, stripeSubs] = await Promise.all([readStoreRails(db, accountId), readStripeSubscriptionIds(db, accountId)]);
    residual = `store_rails=${rails.length > 0 ? rails.map((r) => r.rail).join("+") : "none"} stripe_subscriptions=${stripeSubs.length}`;
  } catch {
    residual = "store_rails=unread stripe_subscriptions=unread";
  }
  await db.batch([
    // D-1119(3) (I34-RJ-05, I34-W3-REVIEW): A RE-SUSPENSION NEVER ERASES THE EARLIER CLOSURE'S APPEAL RECORD, held or not.
    // The upsert below resets a lifted row for the new episode; first, in this batch and only while the row is still
    // that lifted episode, the episode's id, appeal outcome, decision reason and note move to audit_log
    // (`account:closure_superseded`, item_id = the account), so the history survives the reset.
    db
      .prepare(
        `INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind)
         SELECT ?, 'account:closure_superseded', json_object('closure_id', closure_id, 'category', category, 'suspended_at', suspended_at,
           'appeal_opened_at', appeal_opened_at, 'appeal_decided_at', appeal_decided_at, 'appeal_outcome', appeal_outcome,
           'appeal_decision_reason', appeal_decision_reason, 'note', note, 'lifted_at', lifted_at), ?, account_id, 'operator'
          FROM account_closures WHERE account_id = ? AND state = 'lifted'`,
      )
      .bind(operator, ts, accountId),
    db
      .prepare(
        `INSERT INTO account_closures (account_id, state, category, suspended_at, suspended_by, appeal_deadline, note, closure_id)
         VALUES (?, 'suspended', ?, ?, ?, ?, ?, 'clo_' || lower(hex(randomblob(16))))
         ON CONFLICT(account_id) DO UPDATE SET
           state = 'suspended', closure_id = excluded.closure_id, category = excluded.category, suspended_at = excluded.suspended_at,
           suspended_by = excluded.suspended_by, appeal_deadline = excluded.appeal_deadline, note = excluded.note,
           appeal_opened_at = NULL, appeal_decided_at = NULL, appeal_outcome = NULL, appeal_decision_reason = NULL,
           terminated_at = NULL, terminated_by = NULL, lifted_at = NULL, lifted_by = NULL
         WHERE account_closures.state = 'lifted'`,
      )
      // I3-1293: the episode id is minted IN the statement, the 0126 backfill's form.
      .bind(accountId, category, ts, operator, deadline, note.slice(0, 2000)),
    // PREDICATED ON THE UPSERT, like the audit row on termination. Two operators
    // suspending at once: the loser's upsert correctly no-ops on the `state =
    // 'lifted'` guard — and then wrote an audit line naming a ground no closure
    // row ever held, and the function re-read and returned the WINNER's row as
    // if it were the caller's own act (review, LOW/MEDIUM).
    db
      .prepare(
        `INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind)
         SELECT ?, 'account:suspend', ?, ?, ?, 'operator'
          WHERE EXISTS (SELECT 1 FROM account_closures WHERE account_id = ? AND state = 'suspended' AND suspended_at = ?)`,
      )
      .bind(operator, category, ts, accountId, accountId, ts),
    // I3-0210 — THE RESIDUAL ROW, predicated on THIS call's upsert exactly as
    // the audit line above it is. The loser of two concurrent suspensions must
    // not write a billing statement about a closure it did not open.
    //
    // STATE: the account is frozen and these subscriptions keep charging.
    // NEXT ACTOR: an operator, through the payment rail's own console — the
    // fan cannot reach any surface that would cancel, and cannot be told.
    db
      .prepare(
        `INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind)
         SELECT ?, 'account:suspend_billing_residual', ?, ?, ?, 'operator'
          WHERE EXISTS (SELECT 1 FROM account_closures WHERE account_id = ? AND state = 'suspended' AND suspended_at = ?)`,
      )
      .bind(operator, `still billing after suspension — ${residual}`.slice(0, 150), ts, accountId, accountId, ts),
    // I3-0040 (D-1078(29)): THE NOTICE IS SENT AT THE ACT, predicated on THIS call's upsert like the two audit rows.
    closureNoticeStmt(
      db,
      accountId,
      "suspended",
      closureNoticeSummary({ state: "suspended", account_id: accountId, category, appeal_deadline: deadline, appeal_to: supportAddress(env) }),
      ts,
      "account_id = ? AND suspended_at = ?",
      [accountId, ts],
    ),
    // D-0813(1) / J-IDN-01: an `account_compromise` closure exists because
    // somebody other than the owner may hold a live token. Kill it AT closure,
    // with the lever `liftClosure` pulls below, predicated on THIS call's upsert
    // like the two audit rows: a session held before the closure would
    // otherwise still open `fanIdForDataRights` (raw epoch) — export and erase.
    ...(category === "account_compromise"
      ? [
          db
            .prepare(
              `UPDATE accounts SET session_epoch = session_epoch + 1
                WHERE account_id = ? AND EXISTS (SELECT 1 FROM account_closures WHERE account_id = ? AND state = 'suspended' AND suspended_at = ?)`,
            )
            .bind(accountId, accountId, ts),
        ]
      : []),
    ...extra,
  ]);
  const row = await closureFor(db, accountId);
  // THE CALLER'S OWN ACT, or a 409. A re-read that finds somebody else's row
  // reports their ground and their note back to this operator as the result of
  // the button they pressed.
  if (!row || row.state !== "suspended" || row.suspended_at !== ts) {
    throw new ClosureError("this account's state changed while suspending — re-read the case", 409);
  }
  // I3-0210 b (D-0976(2)): after the flip is verified as THIS call's, so a lost race pauses nothing. Never throws.
  if (category === "account_compromise") await pauseCompromiseStripe(db, accountId, operator, billing, ts);
  return row;
}

/** D-1004(4) / D-0993(1): the reason a terminal appeal decision carries — trimmed, 8-500 characters, and with text a
 * reader can see (`hasVisibleText`). `null` when it is not one. Shared by decideAppeal's rule, liftClosure and
 * creator-close.ts liftCreatorSuspension (a lift over an open appeal IS that appeal's decision). */
export function appealReason(reason: unknown): string | null {
  const why = typeof reason === "string" ? reason.trim() : "";
  return why.length >= 8 && why.length <= 500 && hasVisibleText(why) ? why : null;
}
export const APPEAL_REASON_REFUSAL = "a reason of 8 to 500 characters is required";

/** LIFT. The operator's own reversal, and the same act an overturned appeal
 * performs. Refuses on a terminated account: termination is the end of the
 * two-stage path, and undoing it is an erasure-class decision, not this one. */
export async function liftClosure(
  db: DbLike,
  accountId: string,
  operator: string,
  now: () => string = () => new Date().toISOString(),
  /** I3-0210 b: the rail a compromise lift resumes collection on (see `suspendAccount`). */
  billing?: TerminationBilling,
  /** D-1004(4): REQUIRED when an appeal is open and undecided (the lift decides it `overturned`): `appealReason`'s rule,
   * stored in `appeal_decision_reason`. Ignored when no appeal is open. */
  reason?: string,
  /** I39-OPS (D-1158(6b)): the caller's rows (dual_control.ts coSignerAuditStmts), joined into this function's batch. */
  extra: readonly DbStatement[] = [],
): Promise<void> {
  const row = await closureFor(db, accountId);
  if (!row) throw new ClosureError("no such closure", 404);
  if (row.state === "terminated") throw new ClosureError("a terminated account cannot be lifted here", 409);
  if (row.state === "lifted") return; // idempotent
  const why = appealReason(reason);
  if (row.appeal_opened_at && !row.appeal_decided_at && why === null) throw new ClosureError(APPEAL_REASON_REFUSAL, 400);
  const ts = now();
  // I3-1932: every consequence is predicated on THIS call's flip (the row is
  // lifted at `ts` by `operator`), never on the read above — a lift that lost
  // to a terminate or to another operator's lift wrote an audit line and
  // reported success over a zero-row UPDATE.
  const lifted = { sql: "EXISTS (SELECT 1 FROM account_closures WHERE account_id = ? AND state = 'lifted' AND lifted_at = ? AND lifted_by = ?)", binds: [accountId, ts, operator] };
  const stmts: DbStatement[] = [
    // I3-1695: a lift while an appeal is open and undecided IS that appeal's
    // decision — the suspension it asked to reverse is reversed — so it is
    // recorded as `overturned` at the lift instant. Left undecided, the row read
    // "open, undecided" forever and nothing could ever decide it.
    db
      .prepare(
        `UPDATE account_closures SET state = 'lifted', lifted_at = ?, lifted_by = ?,
           appeal_outcome = CASE WHEN appeal_opened_at IS NOT NULL AND appeal_decided_at IS NULL THEN 'overturned' ELSE appeal_outcome END,
           appeal_decision_reason = CASE WHEN appeal_opened_at IS NOT NULL AND appeal_decided_at IS NULL THEN ? ELSE appeal_decision_reason END,
           appeal_decided_at = CASE WHEN appeal_opened_at IS NOT NULL AND appeal_decided_at IS NULL THEN ? ELSE appeal_decided_at END
         WHERE account_id = ? AND state = 'suspended'
           AND (appeal_opened_at IS NULL OR appeal_decided_at IS NOT NULL OR ? IS NOT NULL)`,
      )
      // D-1004(4): the last clause refuses the flip when an appeal opened after the read and no reason was given.
      .bind(ts, operator, why, ts, accountId, why),
    db
      .prepare(`INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, 'account:lift', ?, ?, ?, 'operator' WHERE ${lifted.sql}`)
      .bind(operator, row.category, ts, accountId, ...lifted.binds),
  ];
  // C-40: a compromise closure exists BECAUSE somebody other than the owner may
  // hold a live token; lifting it must not hand that token back its validity.
  // Bumping the epoch is the same forced-relogin lever `fan-auth.ts revoke` and
  // `oidc.ts bumpSessionEpoch` already use — every session, ours and the
  // hijacker's, ends and the real owner signs in fresh. Other grounds
  // (economic_abuse, chargeback_fraud, …) are the account holder's own past
  // conduct, not a third party in possession, so lifting those restores the
  // session exactly as `f18-account-closure.test.ts:141-142` already pins.
  if (row.category === "account_compromise") {
    // I4-RG-FAN-02: predicated on THIS call's flip (`lifted`), as decideAppeal
    // passes `decided`. A lift that lost its race must not revoke a passkey the
    // fan enrolled after the winning lift, nor end the session it signed in to.
    stmts.push(...revokeFanSignIn(db, accountId, lifted));
  }
  await db.batch([...stmts, ...extra]);
  const final = await closureFor(db, accountId);
  if (!final || final.state !== "lifted" || final.lifted_at !== ts || final.lifted_by !== operator) {
    throw new ClosureError("this account's state changed while lifting — re-read the case", 409);
  }
  // I3-0210 b: THIS call lifted it (verified above); resume what this episode's suspension paused. Never throws.
  if (row.category === "account_compromise") await resumeCompromiseStripe(db, accountId, operator, billing, row.suspended_at, ts);
}

/** Record that an appeal arrived. SINGLE operator, deliberately — requiring two
 * people to acknowledge an appeal makes the appeal harder to file, which is the
 * wrong direction for the one control that exists to protect the person. */
export async function openAppeal(db: DbLike, accountId: string, operator: string, now: () => string = () => new Date().toISOString()): Promise<void> {
  const row = await closureFor(db, accountId);
  if (!row) throw new ClosureError("no such closure", 404);
  if (row.state !== "suspended") throw new ClosureError(`an appeal can only be opened on a suspended account (this one is ${row.state})`, 409);
  if (row.appeal_opened_at) return; // idempotent — one appeal per closure
  const ts = now();
  // I3-1694: the published 30-day window (docs/legal/APPEALS.md) is enforced
  // here and again at commit (`appeal_deadline >= ?` below), so an appeal filed
  // after it cannot block termination. An operator who finds a late appeal
  // persuasive still has `liftClosure`.
  if (Date.parse(ts) > Date.parse(row.appeal_deadline)) throw new ClosureError(`the appeal window closed at ${row.appeal_deadline}`, 409);
  await db.batch([
    db.prepare("UPDATE account_closures SET appeal_opened_at = ? WHERE account_id = ? AND state = 'suspended' AND appeal_opened_at IS NULL AND appeal_deadline >= ?").bind(ts, accountId, ts),
    // Predicated: an appeal racing a lift no-ops the UPDATE, and an audit line
    // saying an appeal was recorded when none was is worse than no line — it is
    // the record somebody would later rely on to say the window was honoured.
    db
      .prepare(
        `INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind)
         SELECT ?, 'account:appeal_opened', ?, ?, ?, 'operator'
          WHERE EXISTS (SELECT 1 FROM account_closures WHERE account_id = ? AND appeal_opened_at = ?)`,
      )
      .bind(operator, row.category, ts, accountId, accountId, ts),
  ]);
  // I3-1932: the UPDATE can no-op (a lift or terminate landed first); success
  // is reported only when an appeal is on the row. Another operator's
  // concurrent appeal satisfies that and stays idempotent.
  const final = await closureFor(db, accountId);
  if (!final?.appeal_opened_at) {
    throw new ClosureError("this account's state changed while recording the appeal — re-read the case", 409);
  }
}

/** Decide it. `overturned` lifts in the SAME batch — a decision that says the
 * closure was wrong and then leaves the account suspended until somebody
 * remembers a second click is not a decision. */
export async function decideAppeal(
  db: DbLike,
  accountId: string,
  outcome: "upheld" | "overturned",
  operator: string,
  now: () => string = () => new Date().toISOString(),
  /** I3-1293: the closure EPISODE the decision was requested on. Given, a
   * row re-opened since (a different `closure_id`) is refused; omitted, the
   * episode read below is bound, so the write can never land on a later one. */
  closureId?: string,
  /** I3-0210 b: the rail an overturned compromise appeal resumes collection on (see `suspendAccount`). */
  billing?: TerminationBilling,
  /** I3-0098 b (D-0993(1) pick A, migration 0215): the operator's written reason. Stored in
   * `account_closures.appeal_decision_reason` and appended to the audit detail. JA-04: REQUIRED here, trimmed, 8-500
   * characters (creator-close.ts decideCreatorAppeal's rule), refused before any read — whatever the door. */
  reason?: string,
  /** I39-OPS (D-1158(6b)): the caller's rows (dual_control.ts coSignerAuditStmts), joined into this function's batch. */
  extra: readonly DbStatement[] = [],
): Promise<void> {
  if (outcome !== "upheld" && outcome !== "overturned") throw new ClosureError("outcome must be upheld or overturned");
  const why = appealReason(reason); // D-1004(4): the one rule, shared with liftClosure
  if (why === null) throw new ClosureError(APPEAL_REASON_REFUSAL);
  const row = await closureFor(db, accountId);
  if (!row) throw new ClosureError("no such closure", 404);
  if (!row.appeal_opened_at) throw new ClosureError("no appeal is open on this account", 409);
  if (row.appeal_decided_at) throw new ClosureError("this appeal is already decided", 409);
  // I3-1787: a decision is about a suspension that still stands. A row lifted
  // before `liftClosure` stamped the open appeal (I3-1695) still carries it
  // undecided; deciding it would record an outcome for a case already reversed.
  if (row.state !== "suspended") throw new ClosureError(`this closure is ${row.state} — there is no suspension left to decide an appeal on`, 409);
  if (closureId !== undefined && row.closure_id !== closureId) throw new ClosureError("this closure was re-opened since the decision was requested — decide the current appeal afresh", 409);
  const episode = row.closure_id;
  const ts = now();
  // CO-09 — EVERY CONSEQUENCE IS PREDICATED ON THE DECISION THAT WON, not on
  // the read above it. Only the first statement was guarded
  // (`appeal_decided_at IS NULL`), which is what made the decision itself
  // exactly-once; the audit row and the closure lift were unconditional. Two
  // authorized operators deciding the same case at once — the only way to reach
  // this, and not a fan-reachable race — therefore left `appeal_outcome =
  // 'upheld'` recorded beside `state = 'lifted'`: the LOSER's lift still ran.
  // The session gate reads `state`, so the account (and every collectible
  // action its suspension had frozen) came back while the retained decision
  // said the suspension stood, and the audit log carried a decision line for an
  // operation that decided nothing.
  //
  // THE PREDICATE IS THE DECISION'S OWN IDENTITY: `appeal_decided_at = ts AND
  // appeal_outcome = ?`. It is the `forfeitStmt` pattern one function below —
  // predicate on the state change THIS batch performed, never on a prior read.
  // Two calls that agree on the outcome are not a disagreement and either may
  // satisfy it; two that disagree cannot both, which is the whole property. No
  // new column, and nothing is weakened: the winning decision still lifts.
  const decided = { sql: "EXISTS (SELECT 1 FROM account_closures WHERE account_id = ? AND appeal_decided_at = ? AND appeal_outcome = ?)", binds: [accountId, ts, outcome] };
  const stmts: DbStatement[] = [
    db.prepare("UPDATE account_closures SET appeal_decided_at = ?, appeal_outcome = ?, appeal_decision_reason = ? WHERE account_id = ? AND closure_id IS ? AND appeal_decided_at IS NULL").bind(ts, outcome, why, accountId, episode),
    db
      .prepare(`INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, ?, ?, ?, ?, 'operator' WHERE ${decided.sql}`)
      .bind(operator, "account:appeal_decided", `${row.category}|${outcome}|${why}`, ts, accountId, ...decided.binds),
  ];
  if (outcome === "overturned") {
    stmts.push(
      db
        .prepare(`UPDATE account_closures SET state = 'lifted', lifted_at = ?, lifted_by = ? WHERE account_id = ? AND state = 'suspended' AND ${decided.sql}`)
        .bind(ts, operator, accountId, ...decided.binds),
    );
    // I3-0362/C-40, SAME LEVER AS `liftClosure`: an overturned appeal is the
    // other path back to `lifted`, and it restores an `account_compromise`
    // closure exactly as the operator's own `/lift` does -- a third party may
    // hold a live pre-compromise token, and reinstating the account must not
    // hand it back its validity. Predicated on this decision's own flip
    // (`decided.sql`), not on the read above it, for the same reason every
    // other consequence in this function is.
    if (row.category === "account_compromise") {
      stmts.push(...revokeFanSignIn(db, accountId, decided));
    }
  }
  await db.batch([...stmts, ...extra]);
  // The losing operator is TOLD. A silent no-op would leave them believing the
  // case went their way, which is the reconciliation error this finding is
  // about — the same 409 `appeal_decided_at` already gives a sequential second
  // decision, now also given to a concurrent one.
  const final = await db.prepare("SELECT appeal_decided_at, appeal_outcome FROM account_closures WHERE account_id = ?").bind(accountId).first<{ appeal_decided_at: string | null; appeal_outcome: string | null }>();
  if (!final || final.appeal_decided_at !== ts || final.appeal_outcome !== outcome) throw new ClosureError("this appeal is already decided", 409);
  // I3-0210 b: an overturn is the other path to `lifted` (the lift statement above is predicated on this decision).
  if (outcome === "overturned" && row.category === "account_compromise") await resumeCompromiseStripe(db, accountId, operator, billing, row.suspended_at, ts);
}

/**
 * THE FORFEITURE — decision 2, and the half a future change is most likely to
 * break quietly.
 *
 * The amount is `total - cashBacked`, which is exactly the value Dhwaja invented
 * and that is still there: `cashBackedBalance` already returns
 * `min(total, cash-origin sum)`, so the difference is non-negative by
 * construction and is zero when everything remaining is cash-backed. Computing
 * it as a difference rather than as a sum over the invented origins is what
 * keeps it honest after a platform spend has drawn on promotional value first —
 * summing the grants would forfeit value that had already been spent.
 *
 * IT IS AN APPEND, like everything else in `credit_ledger` — `kind: 'reversal'`,
 * never an UPDATE, so `append-only` holds and the balance stays a fold.
 *
 * QUEUE #486 / D-0645 — WHAT THIS TAKES, AND THE AMOUNT IS RE-ASKED AT COMMIT.
 *
 * WHAT IT TAKES. Only the `credit_ledger` rows of ONE `account_id`, and of those
 * only the non-cash residual. WHAT IT CANNOT REACH, and this is the half D-0645
 * turns on: earned creator money is not in this table. `sales.creator_net_cents`
 * and `creator_payouts.amount_cents` are `passport_id`-keyed and denominated in
 * CENTS (migrations/0058_credit_journal.sql:85-100, 0036_creator_payouts.sql:29),
 * and no statement here touches either. `marketplace_proceeds` — the one origin
 * the CHECK vocabulary NAMES for creator proceeds — is a member of
 * `CASH_ORIGINS`, so it lands in `cash` and is retained, not forfeited; but it
 * has ZERO writers in this tree, so that retention is exercised by no live path
 * and is a promise, not a behaviour.
 *
 * THE AMOUNT WAS NOT RE-ASKED, AND THAT IS THE DEFECT THIS CLAUSE CLOSES. The
 * bound `amount` comes from a `forfeiturePreview` SELECT taken before the batch,
 * and the only guard was EXISTS-on-the-flip: the flip landing said nothing about
 * whether the amount was still true. Of the eight origins with live writers,
 * four are cash origins that move `total` and `cash` TOGETHER and leave the
 * forfeit unchanged; a NON-CASH append does not. `platform_spend`
 * (`rewards.ts:443`) lowers `total` and leaves `cash` alone, so the true forfeit
 * falls by that amount while the bound literal does not — and the debit then
 * exceeds the invented balance, with the excess coming out of the CASH-BACKED
 * sub-balance, which is exactly the money D-0645 says is still owed. So the
 * INSERT re-derives the figure with `FORFEIT_SQL` and compare-and-swaps it: if
 * the scope moved, no row is written, and `terminateAccount`'s re-read says so.
 *
 * STILL A RESIDUAL, ON PURPOSE AND NOT BY OMISSION. `FORFEIT_SQL` keeps
 * `total - cash` rather than summing an enumerated forfeitable set. When the
 * cash-origin sum is negative the enumerated form forfeits MORE, and the
 * paragraph above already records why summing the grants over-forfeits. Moving
 * the formula moves money in the confiscating direction, which is a ruling, not
 * a refactor.
 *
 * `origin: 'fraud_hold'` is the closest of the eleven the CHECK admits: it is
 * the one origin class reserved for value withheld over abuse, and it is
 * non-cash, so the entry reduces the invented sub-balance rather than the
 * cash-backed one. It is a slightly wider name than "fraud" — five of the six
 * closure grounds are not fraud — and widening the CHECK to add a
 * `closure_forfeit` origin would be a table rebuild for a naming preference.
 * Recorded rather than done; `reason` carries the real ground.
 */
export function forfeitStmt(db: DbLike, accountId: string, amount: number, category: ClosureCategory, ts: string): DbStatement {
  // PREDICATED ON THE FLIP THAT AUTHORISED IT, not on a read taken beforehand.
  //
  // This was the only unguarded statement in the batch, and a review drove all
  // three consequences: a `lift` racing a terminate forfeited 900 credits from
  // an account that was live again; an OVERTURNED appeal forfeited 600 while the
  // row read `lifted` — against copy that says in as many words "Nothing was
  // forfeited"; and an appeal filed mid-terminate was recorded and then
  // terminated over, against a Terms sentence promising the limit is "enforced
  // by the system rather than by an operator's memory". It was enforced by a
  // read.
  //
  // The `SELECT … WHERE EXISTS` makes the insert conditional on the state change
  // this same batch performed — the `acceptGift` pattern. If the UPDATE did not
  // match, the row is not `terminated` at this `terminated_at`, and no ledger
  // entry is written.
  return db
    .prepare(
      `INSERT INTO credit_ledger (entry_id, account_id, kind, origin, credits, ref, idem, reason, at)
       SELECT ?, ?, 'reversal', 'fraud_hold', ?, ?, ?, ?, ?
        WHERE EXISTS (SELECT 1 FROM account_closures WHERE account_id = ? AND state = 'terminated' AND terminated_at = ?)
          AND ${FORFEIT_SQL} = ?`,
    )
    .bind(mintId("cle"), accountId, -amount, accountId, `closure:${accountId}`, `closure_forfeit:${category}`, ts, accountId, ts, accountId, amount);
}

/** What termination would take, without taking it — for the operator's confirm
 * screen. The two numbers are the whole policy: what is forfeited, and what
 * stays owed. */
export async function forfeiturePreview(db: DbLike, accountId: string): Promise<{ forfeit: number; retained_cash_backed: number }> {
  // I3-0952: an id that names no account is a 404, never the reassuring zero.
  const acct = await db.prepare("SELECT account_id FROM accounts WHERE account_id = ?").bind(accountId).first<{ account_id: string }>();
  if (!acct) throw new ClosureError("no such account", 404);
  const { forfeit, retained_cash_backed } = await previewFigures(db, accountId);
  return { forfeit, retained_cash_backed };
}

/** `forfeiturePreview`'s one read, plus `cash_unfrozen` — the figure
 * `terminateAccount`'s flip re-asks as `UNFROZEN_CASH_SQL` (#641). */
async function previewFigures(db: DbLike, accountId: string): Promise<{ forfeit: number; retained_cash_backed: number; cash_unfrozen: number }> {
  // I3-0341: ONE commit-time snapshot. This used to call `creditBalance()`
  // then `cashBackedBalance()` — two separate SELECT SUM round trips against
  // the same `credit_ledger` rows — so a ledger append (a spend, a refund
  // reversal) landing between them could put `total` and `cash` at two
  // different moments, and `total - cash` would no longer describe any
  // balance the account actually held. Both sums now come off the one query
  // `cashBackedBalance` already ran internally, read here directly so the two
  // numbers are read together or not at all.
  // CASH_ORIGINS_SQL, not five bound placeholders. The first version of this expanded
  // CASH_ORIGINS into `?,?,?,?,?` and bound them, which `tests/invariants/fidelity-suite.test.ts`
  // caught on the gate: an `IN (...)` built from a list, unchunked and with no stated bound, is
  // the shape that returns a 500 from D1 (>100 bound parameters) while passing every local test,
  // because better-sqlite3 allows ~32,766. This list is a fixed five-value vocabulary and
  // `credits.ts` already carries it spelled as a SQL literal for exactly this reason, with a test
  // pinning the two spellings together. Using it costs nothing and needs no exemption entry.
  //
  // B26-CLOSE — THE FORFEIT IS FREEZE-NEUTRAL; THE RETAINED FIGURE IS NOT. A
  // D-0744 freeze debits the whole grant from `total` AND `cash`, so over the
  // raw sums the forfeit is `total - max(F, min(total, cash))`: an open freeze
  // LOWERED it whenever the frozen purchase was partly spent, and termination
  // is once, so the non-cash residual then sat on a terminated account for
  // good. The forfeit is taken over the ledger WITHOUT freeze/release rows (the
  // same population `FORFEIT_SQL` re-asks at commit); `retained_cash_backed`
  // stays net of an open freeze, because disputed money is not repaid.
  // Still ONE query (I3-0341).
  const row = await db
    .prepare(
      `SELECT COALESCE(SUM(credits), 0) AS bal, COALESCE(SUM(CASE WHEN origin IN ${CASH_ORIGINS_SQL} THEN credits ELSE 0 END),0) AS cash, ` +
        `COALESCE(SUM(CASE WHEN reason IN ${FREEZE_REASONS_SQL} THEN credits ELSE 0 END),0) AS frz, ` +
        `COALESCE(SUM(CASE WHEN reason IN ${FREEZE_REASONS_SQL} AND origin IN ${CASH_ORIGINS_SQL} THEN credits ELSE 0 END),0) AS frz_cash, ` +
        `${RETAINED_CASH_BACKED_SQL} AS retained FROM credit_ledger WHERE account_id = ?`,
    )
    .bind(accountId, accountId)
    .first<{ bal: number; cash: number; frz: number; frz_cash: number; retained: number }>();
  const total = row?.bal ?? 0;
  const cashRaw = row?.cash ?? 0;
  const totalUnfrozen = total - (row?.frz ?? 0);
  const cashUnfrozen = Math.max(0, Math.min(totalUnfrozen, cashRaw - (row?.frz_cash ?? 0)));
  // `Math.max(0, …)` IS LOAD-BEARING, and the case is the FIRST enumerated
  // ground: after a chargeback claw the balance is routinely NEGATIVE, and
  // `total - cash` is then negative too. Without the floor the preview and the
  // operator's confirm screen would render a negative forfeit, and the
  // termination would APPEND A POSITIVE CREDIT — handing money to the account
  // it is closing. A mutation pass found nothing testing it.
  return { forfeit: Math.max(0, totalUnfrozen - cashUnfrozen), retained_cash_backed: row?.retained ?? 0, cash_unfrozen: cashUnfrozen };
}

/**
 * TERMINATE. Permanent, and it refuses in the two cases that make the appeal

 *
 *   - an appeal is OPEN AND UNDECIDED, and
 *   - the deadline has not passed.
 *
 * Both are checked here rather than at the route, because the route is not the
 * only caller a future version will have, and "the operator waits 30 days" is
 * not a control.
 */
/**

 * the SAME batch and under the SAME guard as the owed `image_consent_revoke_outbox` row by every writer that owes it on
 * closing a creator (creator-close.ts closeCreatorAccount and terminateCreator, terminateAccount below for a founding
 * account) and by creator-close.ts `oweClosedCreatorFaceErasures`. A closure or founder termination with no such row
 * was written before the rule, and only those are swept. Moved here from creator-close.ts by I28-FX-INT: creator-close.ts
 * imports this module, so terminateAccount could not import it back (tests/invariants/i13-arch-cycles.test.ts).
 */
export const CREATOR_FACE_ERASURE_OWED_ACTION = "creator:face_erasure_owed";
/** The mark, under exactly the guard of the owed row it accompanies. RETURNING item_id. */
export function faceErasureOwedMark(db: DbLike, passportId: string, ts: string, actorId: string, actorKind: string, guard: string, guardBinds: readonly unknown[]): DbStatement {
  return db

    )

export async function terminateAccount(
  db: DbLike,
  accountId: string,
  operator: string,
  now: () => string = () => new Date().toISOString(),
  nowMs: number = Date.now(),
  billing?: TerminationBilling,
  /** I3-0040: the environment the sent notice's appeal address is read from (`supportAddress`, as `closureNotice`). */
  env?: Record<string, unknown> | null,
  /** I39-OPS (D-1158(6b)): the caller's rows (dual_control.ts coSignerAuditStmts), joined into this function's batch. */
  extra: readonly DbStatement[] = [],
): Promise<{ forfeited: number; retained_cash_backed: number }> {
  const row = await closureFor(db, accountId);
  if (!row) throw new ClosureError("this account is not suspended", 404);
  if (row.state === "terminated") throw new ClosureError("this account is already terminated", 409);
  if (row.state === "lifted") throw new ClosureError("this account's suspension was lifted — suspend it again before terminating", 409);
  if (row.appeal_opened_at && !row.appeal_decided_at) throw new ClosureError("an appeal is open and undecided on this account", 409);
  if (Date.parse(row.appeal_deadline) > nowMs) throw new ClosureError(`the appeal window runs until ${row.appeal_deadline}`, 409);

  const ts = now();
  const { forfeit, retained_cash_backed, cash_unfrozen } = await previewFigures(db, accountId);
  // QUEUE #419 (D-0790): WHAT IS STILL BILLING WHEN THE ACCOUNT IS CLOSED FOR
  // GOOD. The same predicates suspension and both erasure doors use, read
  // before the batch and failing OPEN: a billing read that throws never stops
  // a termination an operator has decided on.
  let billingResidual: string | null = null;
  try {
    const [rails, subs] = await Promise.all([readStoreRails(db, accountId), terminationStripeSubscriptionIds(db, accountId)]);
    if (rails.length > 0 || subs.length > 0) {
      // Store rails cannot be cancelled server-side: RECORDED as still renewing,
      // never marked cancelled.
      const store = rails.length > 0 ? `${rails.map((r) => r.rail).join("+")} (not cancellable server-side; renews until cancelled in the store)` : "none";
      billingResidual = `store_rails=${store} stripe_subscriptions=${subs.length}`;
    }
  } catch {
    billingResidual = "store_rails=unread stripe_subscriptions=unread";
  }
  // EVERY REFUSAL ABOVE IS RE-ASKED IN THE UPDATE'S OWN WHERE.
  //
  // The checks above are for the message — an operator is owed the specific
  // reason — but they are reads, and between a read and a write a suspension can
  // be lifted, an appeal can arrive, and an appeal can be overturned. All three
  // were demonstrated. The write is the only thing that can be atomic with
  // itself, so it carries the whole rule: still suspended, no undecided appeal,
  // and past the deadline.
  const flip = db
    .prepare(
      `UPDATE account_closures
          SET state = 'terminated', terminated_at = ?, terminated_by = ?
        WHERE account_id = ? AND state = 'suspended'
          AND (appeal_opened_at IS NULL OR appeal_decided_at IS NOT NULL)
          AND appeal_deadline <= ?
          AND ${FORFEIT_SQL} = ?
          AND ${UNFROZEN_CASH_SQL} = ?`,
    )
    // The deadline is compared against the WALL CLOCK, not against `ts`. They
    // agree in production, but `ts` is the recording clock — callers pass a
    // monotonic sequence for ordering — and a deadline check must be about the
    // time it actually is, not about the string we are about to store.
    .bind(ts, operator, accountId, new Date(nowMs).toISOString(), accountId, forfeit, accountId, cash_unfrozen);
  // #641 (D-0810 (15)): the flip re-asks BOTH previewed figures INSIDE its
  // batch. A non-cash grant committed between `forfeiturePreview` and here
  // moves `FORFEIT_SQL` or `UNFROZEN_CASH_SQL` (a grant that leaves one still moves
  // the other), the flip matches nothing, every statement predicated on it
  // declines, and the re-read below refuses with nothing taken.
  const stmts: DbStatement[] = [
    flip,
    // The audit row records BOTH numbers, because "what we kept" is as much a
    // fact about the act as "what we took", and an operator reading this later
    // needs to be able to answer a lawyer's question with it. It is predicated
    // on the flip for the same reason the forfeiture is: an audit line saying we
    // terminated an account we did not is worse than no line.
    db
      .prepare(
        `INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind)
         SELECT ?, 'account:terminate', ?, ?, ?, 'operator'
          WHERE EXISTS (SELECT 1 FROM account_closures WHERE account_id = ? AND state = 'terminated' AND terminated_at = ?)`,
      )
      .bind(operator, `${row.category}|forfeit=${forfeit}|retained=${retained_cash_backed}`, ts, accountId, accountId, ts),
    // I3-0040 (D-1078(29)): THE NOTICE IS SENT AT THE ACT, predicated on THIS call's flip like the audit row above.
    closureNoticeStmt(
      db,
      accountId,
      "terminated",
      closureNoticeSummary({ state: "terminated", account_id: accountId, category: row.category, appeal_to: supportAddress(env) }),
      ts,
      "account_id = ? AND terminated_at = ?",
      [accountId, ts],
    ),
  ];
  // QUEUE #419: the residual rides IN the flip's batch, predicated on it exactly
  // as `account:terminate` is (`account:suspend_billing_residual`'s shape).
  if (billingResidual !== null) {
    stmts.push(
      db
        .prepare(
          `INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind)
           SELECT ?, 'account:terminate_billing_residual', ?, ?, ?, 'operator'
            WHERE EXISTS (SELECT 1 FROM account_closures WHERE account_id = ? AND state = 'terminated' AND terminated_at = ?)`,
        )
        .bind(operator, billingResidual.slice(0, 200), ts, accountId, accountId, ts),
    );
  }
  // D-1025(3) (I28-FX-ERASE, I28-JN-03): terminating the account that founds a passport ends that creator too
  // (route-helpers.ts creatorAdmissionForSubject, I3-1422), so it OWES the passport's face-enrolment erasure as a creator
  // termination does (creator-close.ts terminateCreator): in this batch, guarded on the flip having landed AND on
  // `founder` still linking that passport to this account (the value read here, INV-058); drained by likeness.ts.
  const founded = (await db.prepare("SELECT passport_id FROM founder WHERE account_id = ?").bind(accountId).all<{ passport_id: string }>()).results;
  // I28-FX-INT: the owed-mark rides under the SAME guard, so creator-close.ts oweClosedCreatorFaceErasures (which also
  // sweeps founder terminations written before D-1025(3)) does not owe this passport a second time.
  const owedGuard =
    "EXISTS (SELECT 1 FROM account_closures WHERE account_id = ? AND state = 'terminated' AND terminated_at = ?) AND EXISTS (SELECT 1 FROM founder WHERE passport_id = ? AND account_id = ?)";
  for (const f of founded) {
    stmts.push(
      imageConsentOwedOnRemovalStatement(db, f.passport_id, ts, owedGuard, [accountId, ts, f.passport_id, accountId]), faceRevokeOwedStatement(db, f.passport_id, ts, "founder account termination", owedGuard, [accountId, ts, f.passport_id, accountId]), // I65-HOLD F1
      faceErasureOwedMark(db, f.passport_id, ts, operator, "operator", owedGuard, [accountId, ts, f.passport_id, accountId]),
    );
  }
  if (forfeit > 0) stmts.push(forfeitStmt(db, accountId, forfeit, row.category, ts));
  await db.batch([...stmts, ...extra]);
  // DID IT ACTUALLY LAND? A no-op UPDATE is the race the WHERE exists to catch,
  // and reporting `{forfeited: 900}` over one is the same class of lie as
  // announcing a refused sign-out. Re-read and say so.
  const after = await closureFor(db, accountId);
  if (after && after.state === "suspended" && after.suspended_at === row.suspended_at && after.appeal_opened_at === row.appeal_opened_at) {
    // #641: the refusal predicates still hold, so the flip declined on the
    // re-asked balance figures (`FORFEIT_SQL` / `UNFROZEN_CASH_SQL`).
    throw new ClosureError("this account's balance changed while terminating — the account was NOT terminated and NOTHING was forfeited; re-read the case", 409);
  }
  if (!after || after.state !== "terminated" || after.terminated_at !== ts) {
    throw new ClosureError("this account's state changed while terminating — nothing was taken; re-read the case", 409);
  }
  // QUEUE #419 (D-0790): the termination HAS LANDED; now stop the money. After
  // the commit, so a provider failure can neither block nor roll it back, and
  // BEFORE the forfeit re-check below, which throws over a landed termination.
  // `cancelTerminatedStripe` never throws.
  await cancelTerminatedStripe(db, accountId, operator, billing, ts);
  // AND DID THE FORFEIT LAND? The flip and the forfeit are two statements with
  // two different guards, and after QUEUE #486 the forfeit's guard can decline
  // on its own: the balance moved between the preview and the batch. Without
  // this, a declined forfeit returns `{forfeited: 900}` over a row that was
  // never written — the same class of lie the re-read above exists to stop, one
  // statement further down the same batch. Read by `idem`, which is
  // `closure:<account>` and UNIQUE (migrations/0058_credit_journal.sql:73).
  if (forfeit > 0) {
    const debited = await db.prepare("SELECT credits FROM credit_ledger WHERE idem = ?").bind(`closure:${accountId}`).first<{ credits: number }>();
    if (!debited || debited.credits !== -forfeit) {
      throw new ClosureError("this account's balance changed while terminating — the account IS terminated and NOTHING was forfeited; re-read the case", 409);
    }
  }
  return { forfeited: forfeit, retained_cash_backed };
}

/** QUEUE #419 (D-0790, D-0810 (15)): every Stripe subscription termination
 * cancels — the fan (Premium) set `readStripeSubscriptionIds` reads, PLUS the
 * creator (Pro) subscriptions of the passport this account founds (`founder`,
 * the one account-to-passport identity limb; see `payoutDestinationFor`). */
async function terminationStripeSubscriptionIds(db: DbLike, accountId: string): Promise<string[]> {
  const [fan, pro] = await Promise.all([
    readStripeSubscriptionIds(db, accountId),
    db
      .prepare(
        "SELECT DISTINCT e.stripe_subscription_id AS id FROM billing_entitlements e JOIN founder f ON f.passport_id = e.subject_id " +
          "WHERE f.account_id = ? AND e.subject_kind = 'creator' AND e.stripe_subscription_id IS NOT NULL AND e.stripe_subscription_id != ''",
      )
      .bind(accountId)
      .all<{ id: string }>(),
  ]);
  return [...new Set([...fan, ...pro.results.map((r) => r.id)])];
}

/** QUEUE #419 (D-0790): the Stripe rail `terminateAccount` cancels on. Absent
 * (a caller that does not pass it) is recorded as `reason=stripe_not_wired`,
 * never as a cancel. */
export interface TerminationBilling {
  stripe: StripeConfig | null;
  fetchImpl?: typeof fetch;
}

export interface TerminationCancelOutcome {
  cancelled: number;
  failed: number;
  skipped: number;
}

const STRIPE_CANCEL_ACTION = "account:terminate_stripe_cancel";

/**
 * Cancel every live Stripe subscription of a TERMINATED account. Never throws:
 * every outcome is an `account:terminate_stripe_cancel` audit row an operator
 * reads at `/ops/audit`. IDEMPOTENT: a subscription with an `ok sub=<id>` row
 * is never called or audited again, and that row is written under NOT EXISTS
 * so two concurrent runs leave exactly one. A `failed` row is the retry's input
 * (`retryTerminationStripeCancel`).
 */
async function cancelTerminatedStripe(
  db: DbLike,
  accountId: string,
  operator: string,
  billing: TerminationBilling | undefined,
  ts: string,
  /** D-1010(6): the scheduled retry's bounds — `keep` drops a subscription at STRIPE_PAUSE_RETRY_MAX failures, and
   * `terminal` is written once when nothing is left owed, so the episode leaves the window. D-1061(1): a subscription
   * `keep` dropped, or an exhausted read, is a STRIPE_CANCEL_BY_HAND_ACTION row instead, never `terminal`. */
  retry?: { keep: (sub: string) => boolean; terminal: string; readExhausted: boolean },
): Promise<TerminationCancelOutcome> {
  const out: TerminationCancelOutcome = { cancelled: 0, failed: 0, skipped: 0 };
  const record = async (detail: string, once: boolean): Promise<void> => {
    try {
      await db
        .prepare(
          `INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind)
           SELECT ?, '${STRIPE_CANCEL_ACTION}', ?, ?, ?, 'operator'
            WHERE EXISTS (SELECT 1 FROM account_closures WHERE account_id = ? AND state = 'terminated')` +
            (once ? ` AND NOT EXISTS (SELECT 1 FROM audit_log WHERE item_id = ? AND action = '${STRIPE_CANCEL_ACTION}' AND detail = ?)` : ""),
        )
        .bind(operator, detail, ts, accountId, accountId, ...(once ? [accountId, detail] : []))
        .run();
    } catch {
      // The termination stands; an unwritable audit row cannot undo it.
    }
  };
  // D-1061(1): what the retry's bound dropped is a cancel nobody made — one STRIPE_CANCEL_BY_HAND_ACTION row each.
  const byHand = async (what: string[]): Promise<void> => {
    try {
      await db.batch(
        what.map((w) =>
          db
            .prepare(
              `INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, '${STRIPE_CANCEL_BY_HAND_ACTION}', ?, ?, ?, 'system'
                WHERE EXISTS (SELECT 1 FROM account_closures WHERE account_id = ? AND state = 'terminated') AND ${byHandOnceSql(STRIPE_CANCEL_BY_HAND_ACTION, false)}`,
            )
            .bind(operator, `cancel_by_hand ${w}`, ts, accountId, accountId, accountId, `cancel_by_hand ${w}`),
        ),
      );
    } catch {
      // The termination stands; an unwritable audit row cannot undo it.
    }
  };
  if (retry?.readExhausted) {
    await byHand(["reason=subscriptions_unread"]);
    return out;
  }
  let pending: string[];
  let dropped: string[] = [];
  try {
    const [subs, done] = await Promise.all([
      terminationStripeSubscriptionIds(db, accountId),
      db
        .prepare(`SELECT detail FROM audit_log WHERE item_id = ? AND action = '${STRIPE_CANCEL_ACTION}' AND detail LIKE 'ok sub=%'`)
        .bind(accountId)
        .all<{ detail: string }>(),
    ]);
    const cancelled = new Set(done.results.map((r) => r.detail.slice("ok sub=".length)));
    pending = subs.filter((s) => !cancelled.has(s) && (retry?.keep(s) ?? true));
    dropped = subs.filter((s) => !cancelled.has(s) && !pending.includes(s));
  } catch {
    await record("failed reason=subscriptions_unread", false);
    out.failed = 1;
    return out;
  }
  if (pending.length === 0) {
    if (retry && dropped.length > 0) await byHand(dropped.map((s) => `sub=${s}`));
    else if (retry) await record(retry.terminal, true);
    return out;
  }
  const cfg = billing?.stripe ?? null;
  if (!cfg) {
    await record(`skipped n=${pending.length} reason=${billing ? "stripe_not_configured" : "stripe_not_wired"}`, false);
    out.skipped = pending.length;
    return out;
  }
  // #352a (D-0941(2a)): ONE OPS_CANCEL_TIMEOUT_MS deadline over every cancel.
  // RGM-09 (D-0942(1)): the cancels are sent CONCURRENTLY under it, so a hung
  // subscription cannot spend the budget a responsive one needed. A miss is
  // `failed sub=<id>` — the retry's input; rows are written in `pending` order.
  const deadline = opsCancelDeadline();
  try {
    const settled = await Promise.allSettled(pending.map((sub) => cancelStripeSubscription(cfg, sub, billing?.fetchImpl, deadline)));
    for (const [i, sub] of pending.entries()) {
      const r = settled[i];
      const ok = r?.status === "fulfilled" && r.value === true;
      await record(`${ok ? "ok" : "failed"} sub=${sub}`, ok);
      if (ok) out.cancelled += 1;
      else out.failed += 1;
    }
  } finally {
    deadline.clear();
  }
  return out;
}

/** QUEUE #419: the operator's RETRY of a termination's Stripe cancel — the
 * subscriptions with no `ok` row, and only on a terminated account. */
export async function retryTerminationStripeCancel(
  db: DbLike,
  accountId: string,
  operator: string,
  billing: TerminationBilling,
  now: () => string = () => new Date().toISOString(),
): Promise<TerminationCancelOutcome> {
  const row = await closureFor(db, accountId);
  if (!row || row.state !== "terminated") throw new ClosureError("this account is not terminated", 409);
  return cancelTerminatedStripe(db, accountId, operator, billing, now());
}

/**
 * I3-0210 b (D-0976(2) "pause billing on suspension"; D-0577 "suspendAccount reaches the billing rails").
 *
 * An `account_compromise` suspension freezes the session, so the fan cannot reach any surface that would cancel,
 * and the Stripe (Premium) subscription kept charging. `pauseCompromiseStripe` sets Stripe's `pause_collection`
 * (`behavior=void`: no invoice is collected while paused) on every fan subscription `readStripeSubscriptionIds`
 * reads, the set termination cancels; `resumeCompromiseStripe` unsets it on lift or an overturned appeal, ONLY for
 * the subscriptions THIS episode's pause recorded as `ok` (never one paused by anyone else).
 *
 * NEVER THROWS, like `cancelTerminatedStripe`: the closure stands whatever the provider does. Every outcome is an
 * audit row an operator reads at `/ops/audit` — `ok sub=<id>`, `failed sub=<id>`, or `skipped n=<k>
 * reason=stripe_not_wired|stripe_not_configured` — bound to the closure state THIS call produced (`suspended_at` for
 * the pause, `lifted_at` for the resume), so a row never describes an episode that did not happen. Store rails
 * (Apple/Google) have no server-side pause; they stay the `account:suspend_billing_residual` record.
 */
export const STRIPE_PAUSE_ACTION = "account:suspend_stripe_pause";
export const STRIPE_RESUME_ACTION = "account:lift_stripe_resume";
/** D-1003(1) JA-01b: the pause CLAIM, written (`attempt sub=<id>`, same episode bound) BEFORE Stripe is asked, so a lift
 * that lands while the call is in flight still resumes it. Its own action: the outcome rows above keep their meaning. */
export const STRIPE_PAUSE_ATTEMPT_ACTION = "account:suspend_stripe_pause_attempt";
/** I31 RG-01 (D-1073(11)): the RESUME claim, `attempt sub=<id>`, written under the resume's bound (the account still lifted
 * at THIS `lifted_at`) BEFORE Stripe is asked; a subscription whose claim is refused is not asked. Its own action, as the
 * pause claim's. */
export const STRIPE_RESUME_ATTEMPT_ACTION = "account:lift_stripe_resume_attempt";

/** Pause (`paused`) or resume collection under the door's one `OpsCancelDeadline` — stripe.ts's own exports
 * (I29-BILLING note: this file used to carry a hand copy of their body). true only on a 2xx. */
const setStripeCollectionPaused = (cfg: StripeConfig, subscriptionId: string, paused: boolean, fetchImpl: typeof fetch, dl: OpsCancelDeadline): Promise<boolean> =>
  (paused ? pauseStripeSubscription : resumeStripeSubscription)(cfg, subscriptionId, fetchImpl, dl);

async function stripeCollectionStep(
  db: DbLike,
  accountId: string,
  operator: string,
  billing: TerminationBilling | undefined,
  ts: string,
  mode: "pause" | "resume",
  subs: () => Promise<string[]>,
  bound: { sql: string; binds: unknown[] },
): Promise<TerminationCancelOutcome> {
  const action = mode === "pause" ? STRIPE_PAUSE_ACTION : STRIPE_RESUME_ACTION;
  const out: TerminationCancelOutcome = { cancelled: 0, failed: 0, skipped: 0 };
  /** "refused": the episode bound no longer holds (D-1004(3): a pause outcome refused this way is resumed below). */
  const record = async (detail: string): Promise<"written" | "refused" | "error"> => {
    try {
      const r = await db
        .prepare(`INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, '${action}', ?, ?, ?, 'operator' WHERE ${bound.sql}`)
        .bind(operator, detail, ts, accountId, ...bound.binds)
        .run();
      return (r.changes ?? 0) > 0 ? "written" : "refused";
    } catch {
      // The closure stands; an unwritable audit row cannot undo it.
      return "error";
    }
  };
  let pending: string[];
  try {
    pending = await subs();
  } catch {
    await record("failed reason=subscriptions_unread");
    out.failed = 1;
    return out;
  }
  // D-1062(1): both retry passes write their own closing row (by-hand or NOTHING_OWED) after this step returns.
  if (pending.length === 0) return out;
  const cfg = billing?.stripe ?? null;
  if (!cfg) {
    await record(`skipped n=${pending.length} reason=${billing ? "stripe_not_configured" : "stripe_not_wired"}`);
    out.skipped = pending.length;
    return out;
  }
  // D-1003(1) JA-01b: a pause is claimed per subscription before the call, under THIS episode's bound; a subscription
  // whose claim did not land (the episode was lifted first, or the write failed) is not asked at all. I31 RG-01
  // (D-1073(11)): a resume is claimed the same way, under `resumeBound` (an account_compromise suspension since the lift,
  // or a termination -> not asked; D-1074(2)).
  {
    const claimed: string[] = [];
    const claimAction = mode === "pause" ? STRIPE_PAUSE_ATTEMPT_ACTION : STRIPE_RESUME_ATTEMPT_ACTION;
    for (const sub of pending) {
      try {
        const r = await db
          .prepare(`INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, '${claimAction}', ?, ?, ?, 'operator' WHERE ${bound.sql}`)
          .bind(operator, `attempt sub=${sub}`, ts, accountId, ...bound.binds)
          .run();
        if (r.changes > 0) claimed.push(sub);
      } catch {
        // unclaimed -> not asked
      }
    }
    out.skipped = pending.length - claimed.length;
    pending = claimed;
    if (pending.length === 0) return out;
  }
  const dl = opsCancelDeadline();
  const lifted: string[] = [];
  const resuspended: { sub: string; ok: boolean }[] = [];
  try {
    const settled = await Promise.allSettled(pending.map((sub) => setStripeCollectionPaused(cfg, sub, mode === "pause", billing?.fetchImpl ?? fetch, dl)));
    for (const [i, sub] of pending.entries()) {
      const r = settled[i];
      const ok = r?.status === "fulfilled" && r.value === true;
      // D-1004(3): a PAUSE outcome the episode bound refuses means the suspension was lifted while Stripe was being
      // asked; the lift's resume may have reached Stripe first, so this pause (applied, or of unknown outcome) is undone.
      const outcome = await record(`${ok ? "ok" : "failed"} sub=${sub}`);
      if (outcome === "refused" && mode === "pause") lifted.push(sub);
      // I31 RG-01: a RESUME outcome `resumeBound` refuses means the account left that lift while Stripe was being asked
      // (an account_compromise suspension since, or a termination; account_closures is one row per account); a compromise
      // suspension's pause may have reached Stripe first (re-paused below).
      if (outcome === "refused" && mode === "resume") resuspended.push({ sub, ok });
      if (ok) out.cancelled += 1;
      else out.failed += 1;
    }
  } finally {
    dl.clear();
  }
  if (lifted.length > 0) {
    const dl2 = opsCancelDeadline();
    try {
      const undone = await Promise.allSettled(lifted.map((sub) => setStripeCollectionPaused(cfg, sub, false, billing?.fetchImpl ?? fetch, dl2)));
      for (const [i, sub] of lifted.entries()) {
        const ok = undone[i]?.status === "fulfilled" && (undone[i] as PromiseFulfilledResult<boolean>).value === true;
        try {
          await db
            .prepare(`INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) VALUES (?, '${STRIPE_RESUME_ACTION}', ?, ?, ?, 'operator')`)
            .bind(operator, `${ok ? "ok" : "failed"} sub=${sub} reason=pause_after_lift`, ts, accountId)
            .run();
        } catch {
          // best-effort record; the resume itself was asked
        }
      }
    } finally {
      dl2.clear();
    }
  }
  // I31 RG-01 (D-1073(11)): a resume this step sent whose bound no longer held when Stripe answered — a refused resume
  // outcome (recorded first, `reason=lift_superseded`), or a pause_after_lift undo above — may have landed after a NEWER
  // suspension's pause. So the account is re-checked AFTER Stripe answered: an account_compromise closure (the only
  // category that pauses, suspendAccount) gets the pause again through this same step, at its `suspended_at` and bound to
  // it still being suspended there (claimed, outcome rows; a `failed` row is the pause pass's work owed, a lift landing
  // meanwhile undoes it as above). Lifted again or terminated: the claim is refused and nothing is asked (that lift owns
  // the resume). Another category: nothing to pause (that suspension never paused).
  for (const { sub, ok } of resuspended) {
    try {
      await db
        .prepare(`INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) VALUES (?, '${STRIPE_RESUME_ACTION}', ?, ?, ?, 'operator')`)
        .bind(operator, `${ok ? "ok" : "failed"} sub=${sub} reason=lift_superseded`, ts, accountId)
        .run();
    } catch {
      // best-effort record; the re-check below still runs
    }
  }
  const repause = [...resuspended.map((r) => r.sub), ...lifted];
  if (repause.length > 0) {
    try {
      const live = await closureFor(db, accountId);
      if (live?.category === "account_compromise" && live.suspended_at)
        await stripeCollectionStep(db, accountId, operator, billing, live.suspended_at, "pause", async () => repause, {
          sql: "EXISTS (SELECT 1 FROM account_closures WHERE account_id = ? AND state = 'suspended' AND suspended_at = ?)",
          binds: [accountId, live.suspended_at],
        });
    } catch {
      // never throws. A closure read that fails here re-pauses nothing, and no pass reads the rows above for it.
    }
  }
  return out;
}

/** I3-0210 b: pause the fan's Stripe subscriptions after THIS call's compromise suspension (`suspended_at = ts`). */
export async function pauseCompromiseStripe(db: DbLike, accountId: string, operator: string, billing: TerminationBilling | undefined, ts: string): Promise<TerminationCancelOutcome> {
  return stripeCollectionStep(db, accountId, operator, billing, ts, "pause", () => readStripeSubscriptionIds(db, accountId), {
    sql: "EXISTS (SELECT 1 FROM account_closures WHERE account_id = ? AND state = 'suspended' AND suspended_at = ?)",
    binds: [accountId, ts],
  });
}

/** I3-0210 b: resume the subscriptions the episode suspended at `suspendedAt` paused, after THIS call's lift
 * (`lifted_at = ts`): D-1004(3), every one with ANY recorded pause attempt — an `attempt` claim, an `ok` or a `failed`
 * row — whatever its outcome (a `failed` pause may have been applied with its answer lost; resuming an unpaused
 * subscription is harmless). */
export async function resumeCompromiseStripe(db: DbLike, accountId: string, operator: string, billing: TerminationBilling | undefined, suspendedAt: string, ts: string): Promise<TerminationCancelOutcome> {
  return stripeCollectionStep(db, accountId, operator, billing, ts, "resume", () => pausedAtEpisode(db, accountId, suspendedAt), resumeBound(accountId, ts));
}

/** D-1074(2) (I31-RJ-02 FX-02): THE RESUME BOUND of both resume doors (the lift's `resumeCompromiseStripe` and the
 * resume-retry pass), for the claim, the outcome and the closing rows. A resume of the lift at `liftedAt` (of an
 * account_compromise suspension: the only ground that pauses billing, `suspendAccount` -> `pauseCompromiseStripe`) goes
 * ahead while the account is still lifted at that instant, or while its row (one per account) holds a LATER episode on
 * any other ground, suspended or lifted since: that suspension paused nothing this resume could undo. Refused: a later
 * `account_compromise` episode (its own pause and lift own the subscription; a refused outcome is re-paused in
 * `stripeCollectionStep`), and a termination. Binds `accountId`, `liftedAt`. */
function resumeBound(accountId: string, liftedAt: string): { sql: string; binds: unknown[] } {
  return { sql: resumeBoundSql("?", "?"), binds: [accountId, liftedAt] };
}
/** resumeBound's predicate over two SQL expressions; the resume-retry pass's selector applies it to each episode it
 * reads (I31-FX3, D-1075), so it selects exactly the episodes its step may act on. */
const resumeBoundSql = (account: string, liftedAt: string): string =>
  `EXISTS (SELECT 1 FROM account_closures WHERE account_id = ${account} AND ((state = 'lifted' AND lifted_at = ${liftedAt}) OR (state IN ('suspended', 'lifted') AND category <> 'account_compromise')))`;

/** The subscriptions the episode suspended at `suspendedAt` recorded ANY pause attempt for (resumeCompromiseStripe's set). */
async function pausedAtEpisode(db: DbLike, accountId: string, suspendedAt: string | null): Promise<string[]> {
  const { results } = await db
    .prepare(`SELECT action, detail FROM audit_log WHERE item_id = ? AND action IN ('${STRIPE_PAUSE_ACTION}', '${STRIPE_PAUSE_ATTEMPT_ACTION}') AND created_at = ? ORDER BY rowid`)
    .bind(accountId, suspendedAt)
    .all<{ action: string; detail: string }>();
  const tried = new Set<string>();
  for (const { action, detail } of results) {
    if (action === STRIPE_PAUSE_ATTEMPT_ACTION && detail.startsWith("attempt sub=")) tried.add(detail.slice("attempt sub=".length));
    else if (action === STRIPE_PAUSE_ACTION && (detail.startsWith("ok sub=") || detail.startsWith("failed sub="))) tried.add(detail.slice(detail.indexOf("sub=") + "sub=".length));
  }
  return [...tried];
}

/** S-2: attempts per subscription per suspension episode before the retry stops asking Stripe (the failed rows stay
 * on `/ops/audit` for an operator). */
export const STRIPE_PAUSE_RETRY_MAX = 5;
/** S-2: the retry's per-pass budget. POPULATION (I25): each of the FOUR selectors in `retryCompromiseStripePauses`
 * takes at most this many rows per pass, oldest first — the PAUSE pass counts `account_closures` rows with `state =
 * 'suspended'` whose episode (`suspended_at`) still owes a pause by its STRIPE_PAUSE_ACTION / ATTEMPT audit rows, ordered
 * by `suspended_at`; the RESUME pass counts lift episodes (the live `state = 'lifted'` row, and, I31-FX3, an earlier lift
 * read from its audit rows) that resumeBound still lets it act on and whose `lifted_at` still owes a resume by its
 * STRIPE_RESUME_ACTION rows (a `failed` row, or, D-1010(6) RG-05, a `skipped ... reason=stripe_not_*`
 * row the lift wrote while Stripe was unconfigured), ordered by `lifted_at`; the TERMINATION-CANCEL pass (D-1010(6),
 * JN-02 item 3) counts rows with `state = 'terminated'` carrying a `skipped ... reason=stripe_not_*`
 * STRIPE_CANCEL_ACTION row, ordered by `terminated_at`; the ERASURE-CANCEL pass counts `account_delete:stripe_cancel`
 * audit rows `skipped ... reason=stripe_not_* subs=<ids>` (one per erasure, item_id the erasure's pseudonym) and, D-1064(6),
 * erasures whose configured erase-time cancel wrote `failed sub=<id>` with no `ok sub=<id>`, ordered by the erasure's first
 * such row's `audit_id`. One row is one account (or one erasure), whatever its number of subscriptions. A TERMINATION
 * cancel that FAILED with Stripe configured and was never skipped is NOT in the termination population: that is the
 * operator's `retryTerminationStripeCancel` (QUEUE #419), unchanged. An episode carrying a terminal row (STRIPE_PAUSE_NOTHING_OWED /
 * STRIPE_PAUSE_READ_EXHAUSTED) or, D-1061(1), a by-hand row (STRIPE_PAUSE_BY_HAND_ACTION at the pause episode,
 * STRIPE_CANCEL_BY_HAND_ACTION at a termination; D-1062(1), STRIPE_RESUME_BY_HAND_ACTION at the lifted episode,
 * ERASURE_STRIPE_CANCEL_BY_HAND_ACTION at an erasure) is not in its population, so the next owed account is reached on a later pass
 * (test/i25-money2-retry.test.ts "BOUNDARY"). */
export const STRIPE_PAUSE_RETRY_ACCOUNTS = 20;
/** JA-01: the terminal rows the retry writes at an episode — nothing owed (every failed subscription paused, exhausted
 * at STRIPE_PAUSE_RETRY_MAX, or no longer on the account), or the subscription read itself failed
 * STRIPE_PAUSE_RETRY_MAX times. The selector never takes an episode carrying either again. */
export const STRIPE_PAUSE_NOTHING_OWED = "skipped n=0 reason=nothing_owed";
export const STRIPE_PAUSE_READ_EXHAUSTED = "skipped n=0 reason=read_retries_exhausted";
/** D-1061(1) (the fan twin of creator-close.ts CREATOR_STRIPE_PAUSE_BY_HAND_ACTION): a compromise pause the retry's
 * bound DROPPED — a subscription at STRIPE_PAUSE_RETRY_MAX failures with no `ok`, or the subscription read failed that
 * often — is not "nothing owed": the pause pass closes the episode with one `pause_by_hand sub=<id>` (or `pause_by_hand
 * reason=subscriptions_unread`) row per drop at its `suspended_at`, for an operator to pause at Stripe by hand (listed on
 * the console's #/levers from GET /ops/audit?action=<this>, apps/admin/src/main.ts). Never STRIPE_PAUSE_NOTHING_OWED or
 * STRIPE_PAUSE_READ_EXHAUSTED; the selector never takes an episode carrying one again. */
export const STRIPE_PAUSE_BY_HAND_ACTION = "account:suspend_stripe_pause_by_hand";
/** D-1061(1) (twin of creator-close.ts CREATOR_STRIPE_CANCEL_BY_HAND_ACTION): the same for the termination-cancel pass
 * — `cancel_by_hand sub=<id>` / `cancel_by_hand reason=subscriptions_unread`, Stripe may still be billing. */
export const STRIPE_CANCEL_BY_HAND_ACTION = "account:terminate_stripe_cancel_by_hand";
/** D-1062(1): the same for the RESUME pass — a resume after a lift the retry's bound dropped (a subscription at
 * STRIPE_PAUSE_RETRY_MAX failures whose last resume row is not `ok`, or the pause-set read failed that often) is a
 * `resume_by_hand sub=<id>` / `resume_by_hand reason=subscriptions_unread` row at the episode's `lifted_at`, bound to
 * that lifted episode: the fan's Premium may still be paused at Stripe. Never STRIPE_PAUSE_NOTHING_OWED or
 * STRIPE_PAUSE_READ_EXHAUSTED; the resume selector never takes an episode carrying one again. */
export const STRIPE_RESUME_BY_HAND_ACTION = "account:lift_stripe_resume_by_hand";
/** D-1064(6): every by-hand row is written ONCE per episode — the clause each by-hand INSERT carries (binds: item_id,
 * detail, then the episode instant when `atEpisode`). Episode-bound rows (pause at `suspended_at`, resume at `lifted_at`)
 * key on that instant; a termination's or an erasure's rows on the item alone, since their selectors never take an item
 * carrying any by-hand row again. Two passes that select the same episode before either writes leave one row each. */
export function byHandOnceSql(action: string, atEpisode: boolean): string {
  return `NOT EXISTS (SELECT 1 FROM audit_log x WHERE x.item_id = ? AND x.action = '${action}' AND x.detail = ?${atEpisode ? " AND x.created_at = ?" : ""})`;
}

/**
 * S-2 (I23-RG-MONEY, D-0995(11)) — THE RETRY OF A FAILED COMPROMISE PAUSE. The pause's audit rows ARE its outbox: a
 * `failed sub=<id>` (or `failed reason=subscriptions_unread`) row at the LIVE episode's `suspended_at`, with no
 * `ok sub=<id>` beside it, is work still owed. Each pass takes up to STRIPE_PAUSE_RETRY_ACCOUNTS still-suspended
 * accounts that carry one, re-reads their Stripe subscriptions, and re-runs THE SAME step `pauseCompromiseStripe`
 * runs (`stripeCollectionStep`, bound to the same episode) for every subscription without an `ok` row and with
 * fewer than STRIPE_PAUSE_RETRY_MAX failures. Its rows carry the episode's `suspended_at`, so a later lift resumes a
 * subscription a retry paused exactly as it resumes one the first call paused. A lifted account is never retried
 * for a PAUSE (the binding asks `state = 'suspended' AND suspended_at = ?`); D-1005(b): its failed RESUME is retried by
 * the second pass below (bound by `resumeBound`, I31-FX3: an earlier lift episode too). NEVER THROWS per account.
 * Scheduling it (index.ts `scheduled()`) is the hub's hand-off (I23-FX-TC HANDOFFS).
 */
export async function retryCompromiseStripePauses(
  db: DbLike,
  billing: TerminationBilling | undefined,
  opts: { limit?: number; now?: () => string } = {},
): Promise<TerminationCancelOutcome & { accounts: number }> {
  const limit = Math.max(1, Math.min(100, opts.limit ?? STRIPE_PAUSE_RETRY_ACCOUNTS));
  const now = opts.now ?? (() => new Date().toISOString());
  const out = { cancelled: 0, failed: 0, skipped: 0, accounts: 0 };
  // D-1004(3): besides an unpaired `failed` row, a compromise suspension with NO pause record at all (no `attempt`, no
  // outcome: a claim write that failed, or a pause that never ran) is owed its first pause too (the first OR branch
  // below), and so is one where ANY Stripe subscription has neither an `ok` nor a `failed` row at the episode (a second
  // subscription whose claim write failed beside another's `ok`, I24-RJ-01 i24j-3); `failed` subscriptions are the
  // first EXISTS's. I25 (JA-05): an episode the deployment could not ask at all (`skipped ... reason=stripe_not_*`) IS
  // selected — a subscription with only a `skipped` row is owed — so it is paused once Stripe is configured; a pass
  // with no Stripe config asks nothing and writes nothing (`if (!billing?.stripe)` in each loop below).
  const { results } = await db
    .prepare(
      `SELECT c.account_id AS account_id, c.suspended_at AS suspended_at FROM account_closures c
        WHERE c.state = 'suspended' AND c.suspended_at IS NOT NULL
          AND (EXISTS (SELECT 1 FROM audit_log a WHERE a.item_id = c.account_id AND a.action = '${STRIPE_PAUSE_ACTION}'
                        AND a.created_at = c.suspended_at
                        AND ((a.detail LIKE 'failed sub=%'
                              AND NOT EXISTS (SELECT 1 FROM audit_log o WHERE o.item_id = a.item_id AND o.action = a.action AND o.created_at = a.created_at AND o.detail = 'ok' || substr(a.detail, 7) AND o.rowid > a.rowid))
                          OR (a.detail = 'failed reason=subscriptions_unread'
                              AND a.rowid = (SELECT MAX(l.rowid) FROM audit_log l WHERE l.item_id = a.item_id AND l.action = a.action AND l.created_at = a.created_at))))
            OR (c.category = 'account_compromise'
                AND NOT EXISTS (SELECT 1 FROM audit_log p WHERE p.item_id = c.account_id AND p.action IN ('${STRIPE_PAUSE_ACTION}', '${STRIPE_PAUSE_ATTEMPT_ACTION}')
                                  AND p.created_at = c.suspended_at))
            OR (c.category = 'account_compromise'
                AND EXISTS (SELECT 1 ${stripeSubscriptionRowsSql("c.account_id")}
                              AND NOT EXISTS (SELECT 1 FROM audit_log q WHERE q.item_id = c.account_id AND q.action = '${STRIPE_PAUSE_ACTION}'
                                                AND q.created_at = c.suspended_at
                                                AND q.detail IN ('ok sub=' || be.stripe_subscription_id, 'failed sub=' || be.stripe_subscription_id)))))
          AND NOT EXISTS (SELECT 1 FROM audit_log d WHERE d.item_id = c.account_id AND d.action = '${STRIPE_PAUSE_ACTION}'
                            AND d.created_at = c.suspended_at AND d.detail IN (?, ?))
          -- D-1127(5): an erasure tombstones the closure and each of its audit rows to its own unlinkable del_ id (account_data.ts
          -- deleteAccount), so no row under the tombstone's id is a hand close or a pause record; a tombstone names no live account and is never owed.
          AND NOT EXISTS (SELECT 1 FROM audit_log h WHERE h.item_id = c.account_id AND h.action = '${STRIPE_PAUSE_BY_HAND_ACTION}' AND h.created_at = c.suspended_at)
          AND c.account_id NOT GLOB 'del_*'
        ORDER BY c.suspended_at ASC LIMIT ?`,
    )
    .bind(STRIPE_PAUSE_NOTHING_OWED, STRIPE_PAUSE_READ_EXHAUSTED, limit)
    .all<{ account_id: string; suspended_at: string }>();
  for (const r of results) {
    // I25 JA-05: with no Stripe config the step could only append `skipped ... stripe_not_*` again, every pass; the
    // episode's own rows already record it as owed, so the pass counts it and writes nothing.
    if (!billing?.stripe) {
      out.skipped += 1;
      continue;
    }
    try {
      const { results: rows } = await db
        .prepare(`SELECT detail FROM audit_log WHERE item_id = ? AND action = '${STRIPE_PAUSE_ACTION}' AND created_at = ? ORDER BY rowid`)
        .bind(r.account_id, r.suspended_at)
        .all<{ detail: string }>();
      // I31 RG-01: `ok` holds the subscriptions whose LAST pause row is `ok` — a re-pause after a resume that landed late
      // (stripeCollectionStep, reason=lift_superseded) can fail after the episode's first `ok`, and is owed again.
      const ok = new Set<string>();
      const failures = new Map<string, number>();
      let unread = 0;
      for (const { detail } of rows) {
        if (detail === "failed reason=subscriptions_unread") unread += 1;
        else if (detail.startsWith("ok sub=")) ok.add(detail.slice("ok sub=".length));
        else if (detail.startsWith("failed sub=")) {
          ok.delete(detail.slice("failed sub=".length));
          failures.set(detail.slice("failed sub=".length), (failures.get(detail.slice("failed sub=".length)) ?? 0) + 1);
        }
      }
      // JA-01: a read that failed STRIPE_PAUSE_RETRY_MAX times asks nothing. D-1061(1): what the bound dropped (that read,
      // or a subscription at STRIPE_PAUSE_RETRY_MAX failures) is a pause nobody made — STRIPE_PAUSE_BY_HAND_ACTION rows
      // close the episode, never nothing_owed; only an episode with nothing dropped and nothing owed gets the terminal row.
      const readExhausted = unread >= STRIPE_PAUSE_RETRY_MAX;
      let owesNothing = false;
      let gaveUp: string[] = [];
      const owed = async (): Promise<string[]> => {
        const open = readExhausted ? [] : (await readStripeSubscriptionIds(db, r.account_id)).filter((sub) => !ok.has(sub));
        const subs = open.filter((sub) => (failures.get(sub) ?? 0) < STRIPE_PAUSE_RETRY_MAX);
        gaveUp = readExhausted ? ["reason=subscriptions_unread"] : open.filter((sub) => !subs.includes(sub)).map((sub) => `sub=${sub}`);
        owesNothing = subs.length === 0;
        return subs;
      };
      const bound = { sql: "EXISTS (SELECT 1 FROM account_closures WHERE account_id = ? AND state = 'suspended' AND suspended_at = ?)", binds: [r.account_id, r.suspended_at] };
      const step = await stripeCollectionStep(db, r.account_id, "system:stripe_pause_retry", billing, r.suspended_at, "pause", owed, bound);
      if (owesNothing && gaveUp.length > 0) {
        await db.batch(
          gaveUp.map((what) =>
            db
              .prepare(`INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, '${STRIPE_PAUSE_BY_HAND_ACTION}', ?, ?, ?, 'system' WHERE ${bound.sql} AND ${byHandOnceSql(STRIPE_PAUSE_BY_HAND_ACTION, true)}`)
              .bind("system:stripe_pause_retry", `pause_by_hand ${what}`, r.suspended_at, r.account_id, ...bound.binds, r.account_id, `pause_by_hand ${what}`, r.suspended_at),
          ),
        );
      } else if (owesNothing) {
        await db
          .prepare(`INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, '${STRIPE_PAUSE_ACTION}', ?, ?, ?, 'operator' WHERE ${bound.sql}`)
          .bind("system:stripe_pause_retry", STRIPE_PAUSE_NOTHING_OWED, r.suspended_at, r.account_id, ...bound.binds)
          .run();
      }
      out.accounts += 1;
      out.cancelled += step.cancelled;
      out.failed += step.failed;
      out.skipped += step.skipped;
    } catch {
      out.failed += 1; // one account's fault never stops the pass
    }
  }
  // D-1005(b) (JA-01): a RESUME that failed after a lift — the lift's own (`failed sub=<id>` / `failed
  // reason=subscriptions_unread` at `lifted_at`) or the pause_after_lift undo's (at `suspended_at`) — is owed too, while
  // resumeBound holds for that `lifted_at`. A subscription is owed while its LAST resume row is not `ok` and it
  // has fewer than STRIPE_PAUSE_RETRY_MAX failures; the terminal rows above close the episode exactly as for a pause.
  // I31-FX3 (D-1075, FX2C-02b): a later suspension's upsert overwrites the row's `suspended_at` and clears `lifted_at`
  // (suspendAccount), so an EARLIER lift episode is also read from its own audit rows: `lifted_at` is an `account:lift`
  // row's or an overturning `account:appeal_decided` row's instant, `suspended_at` the last `account:suspend` row of the
  // account written before it (NULL when audit pruning took that row: the rows at `lifted_at` still say what is owed);
  // the live episode comes from the row alone. An episode is selected only while resumeBound holds for it (not after a
  // later account_compromise suspension or a termination, D-1074(2)), so a refused episode never takes a LIMIT slot.
  // I31 B11 (I31-RJ-05 residual): audit retention can prune the lift row while the failed resume row written after it
  // survives; that row seeds the episode itself (`lifted_at` = its instant) unless the closure row is lifted at that instant.
  // Its `suspended_at` is read before the FIRST row at that instant, so a retry's row written after a later suspension,
  // and an episode the second arm also reads, give the same tuple (UNION keeps one). A pause_after_lift undo row at a
  // `suspended_at` seeds an episode of its own only where resumeBound holds; it is ordered first and its rows land at that
  // instant, which the lift's own episode reads.
  const { results: lifts } = await db
    .prepare(
      `SELECT e.account_id AS account_id, e.suspended_at AS suspended_at, e.lifted_at AS lifted_at FROM (
          SELECT c.account_id AS account_id, c.suspended_at AS suspended_at, c.lifted_at AS lifted_at FROM account_closures c
           WHERE c.state = 'lifted' AND c.lifted_at IS NOT NULL
          UNION
          SELECT l.item_id, (SELECT s.created_at FROM audit_log s WHERE s.item_id = l.item_id AND s.action = 'account:suspend' AND s.rowid < l.rowid ORDER BY s.rowid DESC LIMIT 1), l.created_at
            FROM audit_log l WHERE (l.action = 'account:lift' OR (l.action = 'account:appeal_decided' AND substr(l.detail, instr(l.detail, '|')) LIKE '|overturned|%'))
             AND NOT EXISTS (SELECT 1 FROM account_closures o WHERE o.account_id = l.item_id AND o.lifted_at = l.created_at)
          UNION
          SELECT f.item_id, (SELECT s.created_at FROM audit_log s WHERE s.item_id = f.item_id AND s.action = 'account:suspend'
                               AND s.rowid < (SELECT MIN(g.rowid) FROM audit_log g WHERE g.item_id = f.item_id AND g.action = f.action AND g.created_at = f.created_at)
                             ORDER BY s.rowid DESC LIMIT 1), f.created_at
            FROM audit_log f WHERE f.action = '${STRIPE_RESUME_ACTION}'
             AND (f.detail LIKE 'failed sub=%' OR f.detail = 'failed reason=subscriptions_unread' OR f.detail LIKE 'skipped n=% reason=stripe_not_%')
             AND NOT EXISTS (SELECT 1 FROM account_closures o WHERE o.account_id = f.item_id AND o.lifted_at = f.created_at)
        ) e
        WHERE ${resumeBoundSql("e.account_id", "e.lifted_at")}
          AND EXISTS (SELECT 1 FROM audit_log a WHERE a.item_id = e.account_id AND a.action = '${STRIPE_RESUME_ACTION}'
                        AND a.created_at IN (e.lifted_at, e.suspended_at)
                        AND (a.detail LIKE 'failed sub=%' OR a.detail = 'failed reason=subscriptions_unread'
                             OR (a.created_at = e.lifted_at AND a.detail LIKE 'skipped n=% reason=stripe_not_%')))
          AND NOT EXISTS (SELECT 1 FROM audit_log d WHERE d.item_id = e.account_id AND d.action = '${STRIPE_RESUME_ACTION}'
                            AND d.created_at = e.lifted_at AND d.detail IN (?, ?))
          AND NOT EXISTS (SELECT 1 FROM audit_log h WHERE h.item_id = e.account_id AND h.action = '${STRIPE_RESUME_BY_HAND_ACTION}' AND h.created_at = e.lifted_at)
        ORDER BY e.lifted_at ASC LIMIT ?`,
    )
    .bind(STRIPE_PAUSE_NOTHING_OWED, STRIPE_PAUSE_READ_EXHAUSTED, limit)
    .all<{ account_id: string; suspended_at: string | null; lifted_at: string }>();
  for (const r of lifts) {
    if (!billing?.stripe) {
      out.skipped += 1; // I25 JA-05, the resume pass's site: no Stripe config -> counted, nothing written
      continue;
    }
    try {
      const { results: rows } = await db
        .prepare(`SELECT detail FROM audit_log WHERE item_id = ? AND action = '${STRIPE_RESUME_ACTION}' AND created_at IN (?, ?) ORDER BY rowid`)
        .bind(r.account_id, r.lifted_at, r.suspended_at)
        .all<{ detail: string }>();
      const subOf = (detail: string): string => (detail.slice(detail.indexOf("sub=") + "sub=".length).split(" ")[0] as string);
      const last = new Map<string, "ok" | "failed">();
      const failures = new Map<string, number>();
      let unread = 0;
      let notAsked = false; // D-1010(6) RG-05: the lift could not ask Stripe at all, so its whole set is owed
      for (const { detail } of rows) {
        if (detail === "failed reason=subscriptions_unread") unread += 1;
        else if (detail.startsWith("skipped n=") && detail.includes(" reason=stripe_not_")) notAsked = true;
        else if (detail.startsWith("ok sub=")) last.set(subOf(detail), "ok");
        else if (detail.startsWith("failed sub=")) {
          last.set(subOf(detail), "failed");
          failures.set(subOf(detail), (failures.get(subOf(detail)) ?? 0) + 1);
        }
      }
      // D-1062(1): what the bound DROPPED (a read that failed STRIPE_PAUSE_RETRY_MAX times, or a subscription at that many
      // failures whose last row is not `ok`) is a resume nobody made — STRIPE_RESUME_BY_HAND_ACTION rows close the
      // episode; only an episode with nothing dropped and nothing owed gets the NOTHING_OWED row.
      const resumeReadExhausted = unread >= STRIPE_PAUSE_RETRY_MAX;
      let resumeOwesNothing = false;
      let resumeGaveUp: string[] = [];
      const owed = async (): Promise<string[]> => {
        const open = resumeReadExhausted ? [] : [...new Set([...(unread > 0 || notAsked ? await pausedAtEpisode(db, r.account_id, r.suspended_at) : []), ...failures.keys()])];
        const subs = open.filter((sub) => last.get(sub) !== "ok" && STRIPE_PAUSE_RETRY_MAX > (failures.get(sub) ?? 0));
        resumeGaveUp = resumeReadExhausted ? ["reason=subscriptions_unread"] : open.filter((sub) => last.get(sub) !== "ok" && !subs.includes(sub)).map((sub) => `sub=${sub}`);
        resumeOwesNothing = subs.length === 0;
        return subs;
      };
      const liftBound = resumeBound(r.account_id, r.lifted_at);
      const step = await stripeCollectionStep(db, r.account_id, "system:stripe_resume_retry", billing, r.lifted_at, "resume", owed, liftBound);
      if (resumeOwesNothing && resumeGaveUp.length > 0) {
        await db.batch(
          resumeGaveUp.map((what) =>
            db
              .prepare(`INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, '${STRIPE_RESUME_BY_HAND_ACTION}', ?, ?, ?, 'system' WHERE ${liftBound.sql} AND ${byHandOnceSql(STRIPE_RESUME_BY_HAND_ACTION, true)}`)
              .bind("system:stripe_resume_retry", `resume_by_hand ${what}`, r.lifted_at, r.account_id, ...liftBound.binds, r.account_id, `resume_by_hand ${what}`, r.lifted_at),
          ),
        );
      } else if (resumeOwesNothing) {
        await db
          .prepare(`INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, '${STRIPE_RESUME_ACTION}', ?, ?, ?, 'operator' WHERE ${liftBound.sql}`)
          .bind("system:stripe_resume_retry", STRIPE_PAUSE_NOTHING_OWED, r.lifted_at, r.account_id, ...liftBound.binds)
          .run();
      }
      out.accounts += 1;
      out.cancelled += step.cancelled;
      out.failed += step.failed;
      out.skipped += step.skipped;
    } catch {
      out.failed += 1; // one account's fault never stops the pass
    }
  }
  // D-1010(6) (JN-02 item 3): a termination CANCEL the deployment could not ask (`skipped ... reason=stripe_not_*`) is
  // owed until a configured pass cancels it; the same STRIPE_PAUSE_RETRY_MAX / terminal-row bounds as the pause.
  const { results: terms } = await db
    .prepare(
      `SELECT c.account_id AS account_id FROM account_closures c
        WHERE c.state = 'terminated'
          AND EXISTS (SELECT 1 FROM audit_log a WHERE a.item_id = c.account_id AND a.action = '${STRIPE_CANCEL_ACTION}'
                        AND a.detail LIKE 'skipped n=% reason=stripe_not_%')
          AND NOT EXISTS (SELECT 1 FROM audit_log d WHERE d.item_id = c.account_id AND d.action = '${STRIPE_CANCEL_ACTION}' AND d.detail IN (?, ?))
          AND NOT EXISTS (SELECT 1 FROM audit_log h WHERE h.item_id = c.account_id AND h.action = '${STRIPE_CANCEL_BY_HAND_ACTION}')
        ORDER BY c.terminated_at ASC LIMIT ?`,
    )
    .bind(STRIPE_PAUSE_NOTHING_OWED, STRIPE_PAUSE_READ_EXHAUSTED, limit)
    .all<{ account_id: string }>();
  for (const r of terms) {
    if (!billing?.stripe) {
      out.skipped += 1; // no Stripe config -> counted, nothing written (the skipped row already records it owed)
      continue;
    }
    try {
      const { results: rows } = await db
        .prepare(`SELECT detail FROM audit_log WHERE item_id = ? AND action = '${STRIPE_CANCEL_ACTION}'`)
        .bind(r.account_id)
        .all<{ detail: string }>();
      const { failures, unread } = tallyFailures(rows);
      const step = await cancelTerminatedStripe(db, r.account_id, "system:stripe_cancel_retry", billing, now(), {
        keep: (sub) => (failures.get(sub) ?? 0) < STRIPE_PAUSE_RETRY_MAX,
        terminal: STRIPE_PAUSE_NOTHING_OWED, // D-1061(1): a dropped cancel or an exhausted read is a by-hand row instead
        readExhausted: unread >= STRIPE_PAUSE_RETRY_MAX,
      });
      out.accounts += 1;
      out.cancelled += step.cancelled;
      out.failed += step.failed;
      out.skipped += step.skipped;
    } catch {
      out.failed += 1;
    }
  }
  // D-1010(6) (JN-02 item 3): an ERASURE cancel skipped the same way. The account and its billing rows are gone, so the
  // owed set is the `subs=` list the skipped row recorded; outcomes are written under the same pseudonymous item_id.
  // D-1064(6) (I30-RJ-05 c5): an erasure whose CONFIGURED cancel failed at erase time (account_data.ts
  // eraseAccountWithSubscriptions, `failed sub=<id>` with no `ok sub=<id>`) is owed the same way — erasure has no operator
  // retry — and its owed set adds every subscription its own `failed`/`ok` rows name. One row per erasure (GROUP BY).
  const { results: erased } = await db
    .prepare(
      `SELECT a.item_id AS item_id, MIN(a.audit_id) AS first_id FROM audit_log a
        WHERE a.action = '${ERASURE_STRIPE_CANCEL_ACTION}' AND a.item_id IS NOT NULL
          AND (a.detail LIKE 'skipped n=% reason=stripe_not_%'
            OR (a.detail LIKE 'failed sub=%'
                AND NOT EXISTS (SELECT 1 FROM audit_log o WHERE o.item_id = a.item_id AND o.action = a.action AND o.detail = 'ok' || substr(a.detail, 7))))
          AND NOT EXISTS (SELECT 1 FROM audit_log d WHERE d.item_id = a.item_id AND d.action = a.action AND d.detail IN (?, ?))
          AND NOT EXISTS (SELECT 1 FROM audit_log h WHERE h.item_id = a.item_id AND h.action = '${ERASURE_STRIPE_CANCEL_BY_HAND_ACTION}')
        GROUP BY a.item_id ORDER BY first_id ASC LIMIT ?`,
    )
    .bind(STRIPE_PAUSE_NOTHING_OWED, STRIPE_PAUSE_READ_EXHAUSTED, limit)
    .all<{ item_id: string; first_id: number }>();
  for (const r of erased) {
    const cfg = billing?.stripe ?? null;
    if (!cfg) {
      out.skipped += 1;
      continue;
    }
    try {
      const ts = now();
      const record = async (detail: string, once: boolean): Promise<void> => {
        await db
          .prepare(
            `INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT 'system:stripe_cancel_retry', '${ERASURE_STRIPE_CANCEL_ACTION}', ?, ?, ?, 'operator'` +
              (once ? ` WHERE NOT EXISTS (SELECT 1 FROM audit_log WHERE item_id = ? AND action = '${ERASURE_STRIPE_CANCEL_ACTION}' AND detail = ?)` : ""),
          )
          .bind(detail, ts, r.item_id, ...(once ? [r.item_id, detail] : []))
          .run();
      };
      const { results: rows } = await db
        .prepare(`SELECT detail FROM audit_log WHERE item_id = ? AND action = '${ERASURE_STRIPE_CANCEL_ACTION}'`)
        .bind(r.item_id)
        .all<{ detail: string }>();
      const { failures } = tallyFailures(rows);
      const done = new Set(rows.filter((x) => x.detail.startsWith("ok sub=")).map((x) => x.detail.slice("ok sub=".length)));
      // The owed set: the `subs=` list a skipped row recorded, plus (D-1064(6)) every subscription a `failed sub=` row names.
      const named = rows.flatMap(({ detail }) =>
        detail.startsWith("failed sub=")
          ? [detail.slice("failed sub=".length)]
          : (/^skipped n=\d+ reason=stripe_not_\S+ (?:.* )?subs=(\S+)/.exec(detail)?.[1]?.split(",").filter((x) => x.length > 0) ?? []),
      );
      const pending = [...new Set(named)].filter((sub) => !done.has(sub) && (failures.get(sub) ?? 0) < STRIPE_PAUSE_RETRY_MAX);
      // D-1062(1): a named subscription with no `ok` and STRIPE_PAUSE_RETRY_MAX failures is a cancel nobody made.
      const erasureDropped = [...new Set(named)].filter((sub) => !done.has(sub) && !pending.includes(sub));
      out.accounts += 1;
      if (pending.length === 0 && erasureDropped.length > 0) {
        await db.batch(
          erasureDropped.map((sub) =>
            db
              .prepare(
                `INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT 'system:stripe_cancel_retry', '${ERASURE_STRIPE_CANCEL_BY_HAND_ACTION}', ?, ?, ?, 'system'
                  WHERE NOT EXISTS (SELECT 1 FROM audit_log WHERE item_id = ? AND action = '${ERASURE_STRIPE_CANCEL_BY_HAND_ACTION}' AND detail = ?)`,
              )
              .bind(`cancel_by_hand sub=${sub}`, ts, r.item_id, r.item_id, `cancel_by_hand sub=${sub}`),
          ),
        );
        continue;
      }
      if (pending.length === 0) {
        await record(STRIPE_PAUSE_NOTHING_OWED, true);
        continue;
      }
      const deadline = opsCancelDeadline();
      try {
        const settled = await Promise.allSettled(pending.map((sub) => cancelStripeSubscription(cfg, sub, billing?.fetchImpl, deadline)));
        for (const [i, sub] of pending.entries()) {
          const ok = settled[i]?.status === "fulfilled" && (settled[i] as PromiseFulfilledResult<boolean>).value === true;
          await record(`${ok ? "ok" : "failed"} sub=${sub}`, ok);
          if (ok) out.cancelled += 1;
          else out.failed += 1;
        }
      } finally {
        deadline.clear();
      }
    } catch {
      out.failed += 1;
    }
  }
  return out;
}

/** The `account_delete:stripe_cancel` rows `eraseAccountWithSubscriptions` writes (account_data.ts) and the retry reads. */
export const ERASURE_STRIPE_CANCEL_ACTION = "account_delete:stripe_cancel";
/** D-1062(1): an erasure's Stripe cancel the retry's bound dropped (a named subscription with no `ok` row and
 * STRIPE_PAUSE_RETRY_MAX failures) — one `cancel_by_hand sub=<id>` row under the erasure's pseudonymous item_id, for an
 * operator to cancel at Stripe by hand (the erased fan may still be billed). Never STRIPE_PAUSE_NOTHING_OWED; the erasure
 * selector never takes an erasure carrying one again. Listed on the console's #/levers (apps/admin/src/main.ts). */
export const ERASURE_STRIPE_CANCEL_BY_HAND_ACTION = "account_delete:stripe_cancel_by_hand";

/** Per-subscription `failed sub=<id>` counts and the `failed reason=subscriptions_unread` count of one episode's rows. */
function tallyFailures(rows: { detail: string }[]): { failures: Map<string, number>; unread: number } {
  const failures = new Map<string, number>();
  let unread = 0;
  for (const { detail } of rows) {
    if (detail === "failed reason=subscriptions_unread") unread += 1;
    else if (detail.startsWith("failed sub=")) {
      const sub = detail.slice("failed sub=".length).split(" ")[0] as string;
      failures.set(sub, (failures.get(sub) ?? 0) + 1);
    }
  }
  return { failures, unread };
}

/**
 * QUEUE #487 / D-0644(b) — WHERE THE MONEY GOES, AND WHY NO OPERATOR CAN SAY.
 *
 * Jason: "operator, the last payout address the account had is where it goes".
 * An operator TRIGGERS the repayment; an operator does not NAME its
 * destination. D-0639 left this door unmounted because a destination the
 * operator supplies, at a deployment where `withDualControl` takes its
 * `required<=1` fast path, is a withdrawal door wearing a repayment's name. The
 * ruling does not put a control on top of that field. It DELETES the field, by
 * binding the destination to a value only the ACCOUNT could have established.
 * So this resolver reads record, never argument: it takes an `accountId` and
 * NOTHING else, and there is deliberately no parameter, no option and no
 * fallback through which a caller can influence what it returns.
 *
 * `actor_kind = 'creator_passkey'` IS THE CLAUSE THAT MAKES THAT TRUE, and it
 * is load-bearing rather than descriptive. MEASURED in this tree:
 * `creator-payout.ts:415-421` is the only writer of a `bound` decision and it
 * passes `actorKind: "creator_passkey"`, reached only through the passkey
 * activation that migration 0036's header calls out ("a stolen session can
 * start an OAuth flow but cannot make an attacker's account the payout
 * destination"). The only operator-writable limb is `unbindPayoutAccount`
 * (`creator-payout.ts:454-457`), whose `actorKind` may be `'operator'` and
 * whose decision is always `'unbound'`. The row set this clause admits is
 * therefore one NO OPERATOR CAN ADD TO — which is the executable form of "not
 * an operator-supplied destination", and the reason the clause is here rather
 * than a `status = 'active'` read of `creator_connect`.
 *
 * "THE LAST ADDRESS THE ACCOUNT HAD" IS PAST TENSE, so this reads the decision
 * HISTORY and orders it, rather than reading the binding's current `status`. An
 * address that was later revoked is still an address the account had; a
 * terminated account's binding will commonly be revoked, and treating that as
 * "no address" would refuse the very population the ruling is about.
 *
 * THE ACCOUNT -> PASSPORT LIMB IS `founder`, AND THAT IS NOT A DESIGN CHOICE —
 * IT IS THE WHOLE OF WHAT THE SCHEMA CARRIES. `connect_binding_decisions` and
 * `creator_connect` are `passport_id`-keyed; `terminateAccount` is
 * `accounts.account_id`-keyed. Across all 101 migrations the ONLY column
 * relating a Dhwaja `account_id` to a `passport_id` as IDENTITY rather than as a
 * fan-to-creator relation is `founder.account_id` (`0011_founder.sql:22-27`,
 * commented "the founder's FAN account"), and `founder` is a SINGLETON
 * (`singleton INTEGER PRIMARY KEY CHECK (singleton = 1)`). Twenty-two other
 * tables carry both columns and every one of them is a fan's action toward a
 * creator. So this resolver returns null for every account but at most one, BY
 * CONSTRUCTION, and the refusal below is the normal outcome rather than the
 * edge. Widening the limb needs a migration and a product decision about who
 * may assert that an account and a passport are the same person; neither is
 * decided here.
 *
 * TWO SUBSTITUTES WERE AVAILABLE AND ARE REFUSED BY NAME, because improvising a
 * destination is precisely how D-0644(b) turns back into the field it deleted:
 *   - matching `accounts.email` to `creator_contacts.email`. A shared string is
 *     not a link. `accounts.email` is provider-asserted at OAuth sign-in and is
 *     not proof that the holder controls the passport, and paying on it would
 *     let anyone who can set an email choose the destination.
 *   - `payout_tags` (`0008_billing.sql:46`). A creator-published tip tag — btc,
 *     cashapp, a throne URL — is where FANS are invited to send money directly.
 *     Dhwaja does not pay through it, there is no rail here that could, and
 *     adopting one would be a new money path decided by a builder.
 */
export interface PayoutDestination {
  /** Stripe's opaque handle for the destination — `acct_…`, never a bank detail. */
  stripe_account_id: string;
  passport_id: string;
  bound_at: string;
  bind_epoch: number;
}

/** The WHERE/ORDER that DEFINES "the last payout address this account had",
 * written once so the resolver and the commit-time re-ask below cannot drift
 * into two different answers. Binds ONE `accountId`. */
const DESTINATION_SOURCE_SQL = `FROM connect_binding_decisions d
      JOIN founder f ON f.passport_id = d.passport_id
     WHERE f.account_id = ?
       AND d.decision = 'bound'
       AND d.actor_kind = 'creator_passkey'
     ORDER BY d.bind_epoch DESC, d.decided_at DESC
     LIMIT 1`;

/** The same row folded to one comparable value, for the compare-and-swap inside
 * the committing INSERT. Both halves are load-bearing: the epoch alone would let
 * a re-bind to a DIFFERENT account pass, and the account id alone would let a
 * re-bind of the SAME account at a new epoch pass. */
const LAST_DESTINATION_KEY_SQL = `(SELECT d.bind_epoch || ':' || d.stripe_account_id ${DESTINATION_SOURCE_SQL})`;

function destinationKey(d: PayoutDestination): string {
  return `${d.bind_epoch}:${d.stripe_account_id}`;
}

export async function payoutDestinationFor(db: DbLike, accountId: string): Promise<PayoutDestination | null> {
  return (
    (await db
      .prepare(
        `SELECT d.stripe_account_id AS stripe_account_id, d.passport_id AS passport_id, d.decided_at AS bound_at, d.bind_epoch AS bind_epoch
     ${DESTINATION_SOURCE_SQL}`,
      )
      .bind(accountId)
      .first<PayoutDestination>()) ?? null
  );
}

/**
 * D-0619 — REPAYMENT. The other half of decision 2, and it was missing for as
 * long as decision 2 has existed.
 *
 * `terminateAccount` forfeits only what Dhwaja invented and says, in the
 * docblock at the top of this file, that "cash-backed credits and owed creator
 * payouts remain a debt we owe". It then writes that debt into ONE PLACE: the
 * `retained=` substring of the termination `audit_log.detail`. A substring is
 * not a liability an operator can query, reconcile, age or discharge, and until
 * this function existed no statement anywhere in the tree read the retained
 * amount back out into a payment. The product recognised an obligation and
 * shipped no instrument to settle it.
 *
 * WHERE THE MONEY ALREADY IS, AND WHY NOTHING HAS TO MOVE IT FIRST. The forfeit
 * is `total - cash` (`forfeiturePreview`), appended with the non-cash origin
 * `fraud_hold`, so after termination the account's ENTIRE remaining
 * `credit_ledger` balance IS the retained cash-backed amount. Repayment is
 * therefore not a transfer between two stores of value — it is the discharge of
 * a balance that is already sitting on the one ledger this codebase keeps.
 *
 * ONE MONEY PATH, NOT TWO. This is an append to `credit_ledger`, the same
 * primitive `forfeitStmt` uses above and `refunds.ts` uses for a clawback. The
 * two alternatives were examined and neither can hold this:
 *   - `creator_payouts` is `passport_id`-keyed, ISO-week-keyed and reachable
 *     only through an ACTIVE Stripe Connect binding. A terminated FAN account
 *     has no passport row, no binding and no destination.
 *   - `credit_reversals` carries `purchase_id` UNIQUE — it reverses one named
 *     purchase. A retained balance is a fold over many grants and spends and is
 *     not attributable to any single purchase.
 *
 * `origin: 'refund_reversal'` IS LOAD-BEARING AND IT IS NOT A NAMING CHOICE. It
 * must be one of `CASH_ORIGINS`, because the retained amount is
 * `min(total, cash)`. A non-cash debit (the `fraud_hold` the forfeit uses) would
 * reduce `total` and leave `cash` untouched, so `retained_cash_backed` would
 * still report the full pre-repayment figure and the SAME BALANCE COULD BE
 * REPAID AGAIN. Of the five cash origins it is the only one that already means
 * "backing returned to the person"; `reason` carries the real ground.
 *
 * THE FIAT LEG IS OUT OF BAND, AND THAT IS STATED RATHER THAN HIDDEN. There is
 * no outbound rail in this codebase that can push money to an account:
 * `stripe.ts` exports `createConnectTransfer` (Connect, passport-keyed) and no
 * refund-creation call at all. Inventing one would be a second money path. So
 * the operator moves the money by whatever rail applies and passes its
 * reference here, and `externalRef` is REQUIRED: a discharge with no evidence
 * the money moved is a way to make a debt disappear without paying it. The
 * precedent is `migrations/0099_payout_void.sql`, which exists for exactly this
 * case — an out-of-band payment the books must be told about so the automated
 * side stops counting it as owed.
 *
 * NOT REVERSIBLE, AND THE REMEDY IS NAMED. `credit_ledger` is append-only, so a
 * repayment cannot be withdrawn. A repayment made in error is corrected by a
 * compensating grant, which restores the balance but NOT as cash-backed value —
 * the operator must re-establish that deliberately. The committing WHERE (the
 * retained balance re-derived, one row per external reference) means the error
 * can never be a DOUBLE payment of one balance, which is the irreversible
 * direction that actually costs money. A LATER repayment is legitimate only for
 * value that became retained after the first — the credits a won dispute
 * releases (RG-634-05, B26-CLOSE).
 */
export async function repayTerminatedBalance(
  db: DbLike,
  accountId: string,
  operator: string,
  externalRef: string,
  now: () => string = () => new Date().toISOString(),
  env?: Record<string, unknown> | null,
  /** I39-OPS (D-1158(6b)): the caller's rows (dual_control.ts coSignerAuditStmts), joined into this function's batch. */
  extra: readonly DbStatement[] = [],
): Promise<{ repaid: number; ref: string; destination: string }> {
  // THE REFERENCE IS CHECKED BEFORE THE ACCOUNT IS, so a caller that cannot
  // evidence the payment learns that without first learning anything about the
  // account's closure state.
  const ref = typeof externalRef === "string" ? externalRef.trim() : "";
  if (!ref || ref.length > 200 || !hasVisibleText(ref)) throw new ClosureError("a repayment needs the external payment reference it was paid under", 400);

  const row = await closureFor(db, accountId);
  if (!row) throw new ClosureError("this account is not suspended", 404);
  // A REPAYMENT DOOR ON A LIVE ACCOUNT IS A WITHDRAWAL DOOR. `suspended` is
  // reversible and `lifted` is live again — in both, the balance is still the
  // account holder's to spend, and draining it to an operator-supplied
  // reference would be taking it, not returning it. Only `terminated` is a
  // state in which the person can no longer reach their own money.
  if (row.state !== "terminated") {
    throw new ClosureError(
      row.state === "lifted"
        ? "this account's suspension was lifted — it holds its own balance again and there is nothing to repay"
        : "this account is suspended, not terminated — nothing is retained yet",
      409,
    );
  }

  // QUEUE #509 / D-0644(b) — NO ADDRESS ON RECORD IS A REFUSAL, NOT A PROMPT.
  //
  // This is the sub-question the ruling does not reach, and it is the one the
  // affected population actually sits in: a FAN account holding cash-backed
  // credits has no payout address anywhere in the schema, and neither does any
  // creator account, because no column joins an `account_id` to a `passport_id`
  // (see `payoutDestinationFor`). The only safe answer is to REFUSE BY NAME.
  // Asking the operator for a destination, defaulting to one, or accepting one
  // "just this once" would re-create the exact field D-0644(b) deleted and
  // D-0639 refused to mount — and it would do it precisely where nobody is
  // watching, on the account that has no record to check the answer against.
  //
  // IT REFUSES BEFORE THE AMOUNT IS READ, so a caller with no destination
  // learns nothing about what the account is holding.
  const destination = await payoutDestinationFor(db, accountId);
  if (!destination) {
    throw new ClosureError(
      "this account has no payout address on record — Dhwaja was never told where this account's money goes, and a repayment to an address an operator supplies is the withdrawal door D-0639 refused to mount. The balance is STILL OWED (D-0645) and there is no collection path (D-0789): point the person to support at " +
        supportAddress(env) +
        ", where a chargeback goes — not to this door",
      409,
      "no_payout_address",
    );
  }

  // B26-CLOSE / RG-634-05 — KEYED PER PAYOUT, NOT PER ACCOUNT. A repayment made
  // while a D-0744 freeze is open leaves the frozen credits out; when the
  // dispute is WON they are owed again, and a once-per-account key refused
  // them forever. The FIRST repayment keeps `closure_repay:<account>`; a LATER
  // one is keyed by its own external reference, `closure_repay:<account>:<ref>`.
  // The amount is still the retained balance re-derived in the INSERT's WHERE,
  // so a replayed win (no second release) or a replayed request pays nothing.
  // A reference already paid under is refused (read here for the message, and
  // re-asked in the INSERT): a retried request is not a second payment.
  const refCol = `closure_repay:${ref}`;
  const prior = await db
    .prepare("SELECT EXISTS (SELECT 1 FROM credit_ledger WHERE idem = ?) AS repaid, EXISTS (SELECT 1 FROM credit_ledger WHERE account_id = ? AND ref = ?) AS used")
    .bind(`closure_repay:${accountId}`, accountId, refCol)
    .first<{ repaid: number; used: number }>();
  if (prior?.used) {
    throw new ClosureError("this account has already been repaid under this payment reference — a retried request pays nothing; a later repayment carries the reference of its own payment", 409);
  }
  const idem = prior?.repaid ? `closure_repay:${accountId}:${ref}` : `closure_repay:${accountId}`;

  // D-0756(2) / RG-B26-02 — A LATER REPAYMENT PAYS CASH-BACKED MONEY ONLY. The
  // first repayment pays `MAX(0, MIN(total, cash))` (`retainedSql`, D-0645).
  // After it, `cash` can sit ABOVE `total` (a non-cash `platform_spend` drew on
  // purchased value), and `MIN(total, cash)` would then turn any later NON-CASH
  // grant into payable cash. So a later repayment is also capped by the
  // cash-origin credits appended SINCE the last repayment (`rowid` is the
  // ledger's insertion order; `credit_ledger` is append-only): a released
  // freeze is cash and counts, a promotional grant is not and does not.
  const cashSinceSql = `(SELECT COALESCE(SUM(CASE WHEN origin IN ${CASH_ORIGINS_SQL} THEN credits ELSE 0 END), 0) FROM credit_ledger WHERE account_id = ? AND rowid > (SELECT MAX(rowid) FROM credit_ledger WHERE account_id = ? AND substr(reason, 1, 18) = 'closure_repayment:'))`;
  const retainedSql = `(SELECT MAX(0, MIN(COALESCE(SUM(credits), 0), COALESCE(SUM(CASE WHEN origin IN ${CASH_ORIGINS_SQL} THEN credits ELSE 0 END), 0))) FROM credit_ledger WHERE account_id = ?)`;
  const payableSql = prior?.repaid ? `MAX(0, MIN(${retainedSql}, ${cashSinceSql}))` : retainedSql;
  const payableBinds = prior?.repaid ? [accountId, accountId, accountId] : [accountId];
  // Read BEFORE the preview, so an interleaving that lands after the preview is
  // caught by the INSERT's own WHERE, which re-asks `payableSql`.
  const cap = prior?.repaid
    ? ((await db.prepare(`SELECT ${payableSql} AS cap`).bind(...payableBinds).first<{ cap: number }>())?.cap ?? 0)
    : Infinity; // a bare global, not a member read: env-inventory reads an upper-case member as an env key

  const { retained_cash_backed: retained } = await forfeiturePreview(db, accountId);
  const amount = Math.min(retained, cap);
  if (amount <= 0) {
    throw new ClosureError(prior?.repaid ? "this account's retained balance has already been repaid" : "this account has no retained cash-backed balance to repay", 409);
  }

  // THE AMOUNT IS RE-DERIVED INSIDE THE COMMITTING STATEMENT, and the read
  // above is only for the message and the 409.
  //
  // `forfeiturePreview` is a SELECT; the append below is a different statement
  // at a different instant, and `credit_ledger` has other writers (a spend, a
  // chargeback claw, a marketplace credit). Between the two, the retained
  // balance can move — and a chargeback claw can move it DOWN, which is the
  // case that matters, because paying out the pre-claw figure pays money the
  // account no longer holds. So the INSERT's own WHERE re-asks the question:
  // it recomputes `payableSql` — `MAX(0, MIN(total, cash))`, the expression
  // `forfeiturePreview` uses, and for a later repayment also the cash appended
  // since the last one — and matches it against the amount we are about to
  // pay. If the balance moved, no row is written, and the re-read below
  // refuses. We never pay a number that stopped being true.
  //
  // `CASH_ORIGINS_SQL`, not five bound placeholders — the same reason
  // `forfeiturePreview` gives: an `IN (...)` built from a list is the shape
  // that returns a 500 from D1 past 100 bound parameters while passing every
  // local test.
  const ts = now();
  const entryId = mintId("cle");

  // ONE BATCH. A repayment that can half-apply — the money marked returned with
  // no audit line, or an audit line over a balance still sitting there — is the
  // defect this function exists to avoid, not a tolerable edge of it.
  await db.batch([
    // NO BALANCE IS PAID TWICE, and the guarantee is this WHERE, not the reads
    // above: `retainedSql = amount` means a second payer racing the first finds
    // the balance already moved and writes nothing; `NOT EXISTS … ref` means
    // one external payment is recorded once; and `idem` (UNIQUE,
    // migrations/0058_credit_journal.sql:73; `closure_repay:<account>` for the
    // first, `…:<ref>` after — B26-CLOSE) turns a same-key collision into a
    // constraint failure that rolls the whole batch back.
    db
      .prepare(
        `INSERT INTO credit_ledger (entry_id, account_id, kind, origin, credits, ref, idem, reason, at)
         SELECT ?, ?, 'reversal', 'refund_reversal', ?, ?, ?, ?, ?
          WHERE EXISTS (SELECT 1 FROM account_closures WHERE account_id = ? AND state = 'terminated')
            AND ${payableSql} = ?
            AND ${LAST_DESTINATION_KEY_SQL} = ?
            AND NOT EXISTS (SELECT 1 FROM credit_ledger WHERE account_id = ? AND ref = ?)`,
      )
      // THE DESTINATION IS RE-ASKED HERE, at its own call site, for the reason
      // the amount is: `payoutDestinationFor` is a SELECT and this is a
      // different statement at a different instant. A creator who re-binds —
      // or has a binding revoked and re-established — between the two must not
      // be paid to the address that WAS last. If the answer moved, no row is
      // written and the landing re-read below refuses.
      .bind(entryId, accountId, -amount, refCol, idem, `closure_repayment:${row.category}`, ts, accountId, ...payableBinds, amount, accountId, destinationKey(destination), accountId, refCol),
    // PREDICATED ON THE APPEND THIS BATCH PERFORMED — the `forfeitStmt` pattern,
    // one statement earlier in the same batch rather than a read taken before
    // it. An audit line saying we repaid an account we did not is the same class
    // of lie as a forfeiture line over a termination that never landed.
    db
      .prepare(
        `INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind)
         SELECT ?, 'account:repay_retained', ?, ?, ?, 'operator'
          WHERE EXISTS (SELECT 1 FROM credit_ledger WHERE entry_id = ?)`,
      )
      // `dest=` IS THE RESOLVED ADDRESS, never an operator's. It is recorded so
      // a later reader can answer "where did this go" from the books rather
      // than from the operator's account of it.
      // By THIS append's `entry_id`, not by `idem`: a concurrent repayment's row
      // can carry the same idem, and an audit line over it would be a lie.
      .bind(operator, `${row.category}|repaid=${amount}|ref=${ref}|dest=${destination.stripe_account_id}`, ts, accountId, entryId),
    // #489a / D-0920 (0197): THE CASH LEFT. The ledger row above says what the ACCOUNT no longer holds; this register
    // row says that money went out, under which external reference and on which rail (the destination is a Stripe
    // Connect account, `payoutDestinationFor`). Its cents are read from THIS append's own row, by `entry_id`, so it
    // states the amount that committed — and no row lands when the append did not. `reconcileCashOutflows` below is
    // the identity between the two.
    db
      .prepare(
        `INSERT INTO cash_outflows (source_ref, kind, rail, external_ref, cents, recorded_at)
         SELECT entry_id, 'repayment', 'stripe', ?, -credits * ${CREDIT_BACKING_CENTS}, ? FROM credit_ledger WHERE entry_id = ?`,
      )
      .bind(ref, ts, entryId),
    ...extra,
  ]);

  // DID IT ACTUALLY LAND? The WHERE can decline for two reasons an operator must
  // not be left guessing about — the closure stopped being `terminated`, or the
  // balance moved — and in both the correct answer is that NOTHING was paid.
  // Reporting a repayment over a no-op is `terminateAccount`'s own failure mode
  // and gets its own re-read for the same reason.
  // Read by `entry_id` (B26-CLOSE): by `idem`, a losing concurrent repayment
  // found the WINNER's row and reported a payment it never made.
  const landed = await db.prepare("SELECT credits FROM credit_ledger WHERE entry_id = ?").bind(entryId).first<{ credits: number }>();
  if (!landed || landed.credits !== -amount) {
    throw new ClosureError("this account's balance or closure state changed while repaying — nothing was paid; re-read the case", 409);
  }
  return { repaid: amount, ref, destination: destination.stripe_account_id };
}

/** #489a / D-0920: what `reconcileCashOutflows` found. `ok` is false on any mismatch; over an EMPTY population it is
 * true and `checked` carries the zeros, so a caller states the counts beside the verdict, never the boolean alone. */
export interface CashOutflowReconcile {
  ok: boolean;
  /** repayment ledger entries examined (live ledger rows) plus register rows examined. */
  checked: { ledger_repayments: number; register_rows: number; erased_with_account: number };
  /** a repayment ledger entry with no register row: money recorded as gone from the account, with no record it left. */
  ledger_without_outflow: string[];
  /** a register row whose ledger entry is gone and was NOT erased with its account. */
  outflow_without_ledger: string[];
  /** both exist and the cents disagree with credits x CREDIT_BACKING_CENTS. */
  cents_mismatch: string[];
  /** #489a (I16-W06C H-489-payout): the kind 'payout' identity — every `creator_payouts` row with status 'paid' has
   * exactly one register row with source_ref = payout_id and cents = amount_cents, and every payout register row
   * names a payout that is paid. Ids are payout ids.
   * I17 J5a (RG-MONEY:6, D-0938(1a)): a VOIDED payout (settled outside Stripe) is balanced by the register row
   * `voidPayout` writes with the operator-stated cents (0 < cents <= amount_cents, else `cents_mismatch`); a voided
   * payout with no row is `voided_without_outflow`, and a payout register row may name a paid OR a voided payout. */
  payouts: {
    checked: { paid_payouts: number; voided_payouts: number; register_rows: number };
    paid_without_outflow: string[];
    voided_without_outflow: string[];
    outflow_without_payout: string[];
    cents_mismatch: string[];
  };
}

/**
 * #489a / D-0920 — THE INTERNAL RECONCILE IDENTITY between the account ledger and the cash-outflow register (0197).
 * For kind 'repayment': every `credit_ledger` row with `reason` starting `closure_repayment:` has exactly one
 * `cash_outflows` row with `source_ref = entry_id` and `cents = -credits x CREDIT_BACKING_CENTS`; every register row
 * names a ledger entry that exists, or one erased with its account (`ledger_erased_at`, stamped by
 * `account_data.ts deleteAccount` in the batch that deletes the ledger). No processor pull (D-0920). Payout rows
 * (kind 'payout', written by migration 0201's `creator_payouts_paid_register` trigger) are checked as `payouts`,
 * against `creator_payouts` (D-0928(2); I16-W08A H-489-payout, landed by I16-L10).
 */
export async function reconcileCashOutflows(db: DbLike): Promise<CashOutflowReconcile> {
  const ledger = (
    await db
      .prepare("SELECT l.entry_id, l.credits, o.cents FROM credit_ledger l LEFT JOIN cash_outflows o ON o.source_ref = l.entry_id AND o.kind = 'repayment' WHERE substr(l.reason, 1, 18) = 'closure_repayment:'")
      .all<{ entry_id: string; credits: number; cents: number | null }>()
  ).results;
  const register = (
    await db
      .prepare("SELECT o.source_ref, o.ledger_erased_at, EXISTS (SELECT 1 FROM credit_ledger l WHERE l.entry_id = o.source_ref) AS live FROM cash_outflows o WHERE o.kind = 'repayment'")
      .all<{ source_ref: string; ledger_erased_at: string | null; live: number }>()
  ).results;
  const ledgerWithout = ledger.filter((r) => r.cents === null).map((r) => r.entry_id);
  const centsMismatch = ledger.filter((r) => r.cents !== null && r.cents !== -r.credits * CREDIT_BACKING_CENTS).map((r) => r.entry_id);
  const orphan = register.filter((r) => !r.live && r.ledger_erased_at === null).map((r) => r.source_ref);
  const erased = register.filter((r) => !r.live && r.ledger_erased_at !== null).length;
  const paid = (
    await db
      .prepare("SELECT p.payout_id, p.amount_cents, o.cents FROM creator_payouts p LEFT JOIN cash_outflows o ON o.source_ref = p.payout_id AND o.kind = 'payout' WHERE p.status = 'paid'")
      .all<{ payout_id: string; amount_cents: number; cents: number | null }>()
  ).results;
  const voided = (
    await db
      .prepare("SELECT p.payout_id, p.amount_cents, o.cents FROM creator_payouts p LEFT JOIN cash_outflows o ON o.source_ref = p.payout_id AND o.kind = 'payout' WHERE p.status = 'voided'")
      .all<{ payout_id: string; amount_cents: number; cents: number | null }>()
  ).results;
  const payoutRegister = (
    await db
      .prepare("SELECT o.source_ref, EXISTS (SELECT 1 FROM creator_payouts p WHERE p.payout_id = o.source_ref AND p.status IN ('paid', 'voided')) AS paid FROM cash_outflows o WHERE o.kind = 'payout'")
      .all<{ source_ref: string; paid: number }>()
  ).results;
  const payouts = {
    checked: { paid_payouts: paid.length, voided_payouts: voided.length, register_rows: payoutRegister.length },
    paid_without_outflow: paid.filter((r) => r.cents === null).map((r) => r.payout_id),
    voided_without_outflow: voided.filter((r) => r.cents === null).map((r) => r.payout_id),
    outflow_without_payout: payoutRegister.filter((r) => !r.paid).map((r) => r.source_ref),
    cents_mismatch: [
      ...paid.filter((r) => r.cents !== null && r.cents !== r.amount_cents).map((r) => r.payout_id),
      ...voided.filter((r) => r.cents !== null && (r.cents <= 0 || r.cents > r.amount_cents)).map((r) => r.payout_id),
    ],
  };
  const payoutsOk =
    payouts.paid_without_outflow.length === 0 && payouts.voided_without_outflow.length === 0 && payouts.outflow_without_payout.length === 0 && payouts.cents_mismatch.length === 0;
  return {
    ok: ledgerWithout.length === 0 && centsMismatch.length === 0 && orphan.length === 0 && payoutsOk,
    checked: { ledger_repayments: ledger.length, register_rows: register.length, erased_with_account: erased },
    ledger_without_outflow: ledgerWithout,
    outflow_without_ledger: orphan,
    cents_mismatch: centsMismatch,
    payouts,
  };
}

/** The list the operator console reads: open cases, soonest deadline first. */

/**
 * ONE ATOMIC BATCH, AND THE FAN-DAY IS CLAIMED BY A TOKEN (ADR-0006 §2,
 * O-12/O-23, migration 0079).
 *
 * WHAT WAS HERE. A point read of `last_seen_at` decided "is this a new UTC day
 * for this pair?", the caller incremented the region counter on the strength of
 * it, and the day the increment landed on came from a timestamp minted before
 * that read. Two requests in the same millisecond both read yesterday, both
 * reported true, and both incremented — so `fan_days`, whose UNIT IS DISTINCT
 * FANS, could count one person twice. The old header called that acceptable
 * "because nothing gates on it"; `active_fans(day) := Σ_country fan_days` now
 * makes it the day's whole active population (ADR §4), and a K-floor computed
 * over a population one person can inflate is not a floor.
 *
 * THE CLAIM. `ts` and a CSPRNG `token` are minted AFTER the two gate reads and
 * immediately before the batch, so the day written is the day of an instant
 * committed in one D1 round-trip — that round-trip is the request path's whole
 * lateness window, and the build grace (ADR §1) rests on it. The batch is:
 *
 *   0. the sighting bump — the shipped "later sightings bump last_seen_at +
 *      count" semantics, unchanged. A FIRST encounter matches nothing here and
 *      takes its count of 1 from the INSERT below.
 *   0b. the Codex row itself, on a first sighting. It no longer carries the
 *      claim, so its ON CONFLICT is DO NOTHING (statement 0 does the bump).
 *   1. the day claim, ON ITS OWN TABLE `creator_day_claims` (migration 0081):
 *      `claimed_day`/`claim_token` are written only when this day is strictly
 *      newer than the day already claimed (or none is).
 *   2. the geo increment, reading its day FROM THE CLAIM ROW and gated on the
 *      token.
 *   3. the `fan_active` outbox + scopes rows, gated on the same token AND on
 *      the fan's account still existing (`outboxStatements`).
 *
 * A losing token selects zero rows in 2 and 3, so a second sighting on a day
 * already claimed writes neither a fan-day nor an event. Two requests in the
 * same millisecond cannot both win: the upsert in 1 is one statement and the
 * later writer's token overwrites, but only one of them is the token that is
 * still in the row when 2 and 3 read it — and the batch is atomic, so a Worker
 * death leaves a full claim or none.
 *
 * THE CLAIM IS NOT IN THE CODEX (0081, FINDING-0010-F-1). It was two columns on
 * `creator_encounters` until 0081, and `clearCodex` (`DELETE FROM
 * creator_encounters WHERE account_id = ?`, the fan's own one-tap wipe) deleted
 * it — so {seen, clear} repeated N times minted N fan-days and N `fan_active`
 * events for one account on one day, past the geo day gate and past K. The
 * claim is the fan-day ledger, not the fan's sighting history: the fan clears
 * what they see, and what the creator is counted for does not re-arm. 0079's
 * two columns survive on `creator_encounters` and nothing writes or reads
 * them any more.
 *
 * THE COUNTRY NEVER MEETS THE FAN. It arrives as a normalised two-letter code
 * and is bound to statement 2, whose table has no account column; `regions.ts`
 * is still the only module that owns the shape of that table and still takes no
 * account id anywhere (its header's rule, unchanged).
 */
export async function recordEncounter(
  db: DbLike,
  accountId: string,
  passportId: string,
  source: string,
  now: () => string = () => new Date().toISOString(),
  /** `CF-IPCountry`, raw — normalised here. Absent is `ZZ`, a real bucket. */
  country?: string,
  /** `ANALYTICS_FAN_SECRET`. Undefined is refused by `fanKey` (ADR-0006 §2):
   * an encounter that cannot be counted is not silently uncounted. */
  fanSecret?: string,
  pubEnv?: AnalyticsTickEnv,
  pubCtx?: { waitUntil(p: Promise<unknown>): void },
): Promise<{ written: boolean }> {
  const NO = { written: false };
  if (!accountId || !PSP_RE.test(passportId) || !CODEX_SOURCES.has(source)) return NO;
  if (!(await loggingEnabled(db, accountId))) return NO;
  // Resolved through the one gated point-read: a SUPPRESSED subject is silently
  // not logged, exactly like a passport id that never existed. Writing the row
  // would put a delisted person's name back on a surface (`GET /account/codex`)
  // the moment the fan opened their Codex.
  const psp = await livePassport(db, passportId);
  if (!psp) return NO; // only ever log real, live, un-delisted creators
  const ts = now();
  const token = randomHex(16);
  const key = await fanKey(fanSecret, accountId, passportId);
  // E-17: `audience_regions` has no account column (regions.ts's header — this
  // module is the only writer and the only place an account-quality gate CAN
  // sit). Its only floor was GEO_DAY_MIN_ACTIVE, a population of ANY signed-in
  // views — inflatable at zero cost with throwaway accounts, unlike every
  // qualified-fan-gated board in rankings.ts (AUDIT #17). `creator_encounters`
  // stays unqualified on purpose (this file's header) because it is the fan's
  // own sighting history; the geo increment is not a sighting history, it is
  // an anonymous aggregate, so gating it here costs the fan nothing they can
  // see and closes the same hole rankings.ts already closed for every board.
  // I23 (I3-0459, D-0993(1) census pick 1; the same rule as I3-0455): a free passkey is enough to be
  // QUALIFIED, so ~10 throwaway accounts could still make a small country appear. The counter also needs
  // a COSTED signal (`fan_costed_qualification`: verified email, OTP email, an acquisition) that is at
  // least FOUNDING_FAN_MIN_AGE_DAYS old at this encounter (`fanCostedAgedEnoughExpr`, one `?` = now).
  const qualified =
    (await isQualifiedFan(db, accountId)) &&
    (await db.prepare(`SELECT 1 AS x FROM (SELECT ? AS account_id) qf WHERE ${fanCostedAgedEnoughExpr("qf")}`).bind(accountId, ts).first()) !== null;
  // The token predicate, spelled ONCE and bound three times: statement 2 and
  // both statements of 3 must ask the identical question of the identical row.
  const wonClaim = "SELECT 1 FROM creator_day_claims WHERE account_id = ? AND passport_id = ? AND claim_token = ?";
  const stmts: DbStatement[] = [
    db
      .prepare("UPDATE creator_encounters SET encounter_count = encounter_count + 1, last_seen_at = ? WHERE account_id = ? AND passport_id = ?")
      .bind(ts, accountId, passportId),
    db
      .prepare(
        // I3-1768 (D-1084(1), I32-FX3-VERIFIED): first_state is the class the fan was SHOWN at this sighting — a
        // `verified` row whose latest grant lapsed by `ts` (standing.ts grantLiveAtSql) and the sweep has not folded
        // reads `claimed` here, as it does on /p/:id (standing.ts CLAIM_STATE_AT_SQL, which passport-view.ts imports) and in listCodex.
        `INSERT INTO creator_encounters (account_id, passport_id, first_seen_at, last_seen_at, encounter_count, first_source, first_state)
         VALUES (?, ?, ?, ?, 1, ?, ${shownClassValueSql()})
         ON CONFLICT(account_id, passport_id) DO NOTHING`,
      )
      .bind(accountId, passportId, ts, ts, source, psp.claim_state, passportId, ts, psp.claim_state),
    db
      .prepare(
        `INSERT INTO creator_day_claims (account_id, passport_id, claimed_day, claim_token)
         VALUES (?, ?, substr(?, 1, 10), ?)
         ON CONFLICT(account_id, passport_id) DO UPDATE SET
           claimed_day = excluded.claimed_day,
           claim_token = excluded.claim_token
         WHERE creator_day_claims.claimed_day IS NULL OR creator_day_claims.claimed_day < excluded.claimed_day`,
      )
      .bind(accountId, passportId, ts, token),
    // THE DAY COMES OFF THE CLAIM ROW, not off `ts`. They are equal for the
    // winner by construction; reading the column is what makes "the cell the
    // token claimed" and "the cell incremented" the same cell by definition.
    // E-17: only a QUALIFIED fan's win moves the counter — an unqualified
    // account still wins the day claim (statement 3) and still gets its Codex
    // entry (statements 0/0b), it just cannot move a number a creator reads.
    ...(qualified
      ? [
          db
            .prepare(
              `INSERT INTO audience_regions (passport_id, day, country, fan_days)
         SELECT ?, claimed_day, ?, 1 FROM creator_day_claims
          WHERE account_id = ? AND passport_id = ? AND claim_token = ?
         ON CONFLICT(passport_id, day, country) DO UPDATE SET fan_days = fan_days + 1`,
            )
            .bind(passportId, normalizeCountry(country), accountId, passportId, token),
        ]
      : []),
    // I3-0378 / I3-1412: `fan_active` feeds the retention fold, a number a
    // creator reads, so it takes the SAME `qualified` gate as the geo counter
    // above — a throwaway account's win claims the day and keeps its Codex
    // entry but emits nothing.
    ...(qualified
      ? outboxStatements(
          db,
          { kind: "fan_active", passportId, occurredDay: occurredDay(ts), fanKey: key, dims: { source } },
          { accountId, extraWhere: { sql: `EXISTS (${wonClaim})`, params: [accountId, passportId, token] } },
        )
      : []),
  ];
  const span = outboxSpan(db);
  const spanRes = await db.batch([span.open, ...stmts, span.close]);
  const seqs = outboxSpanSeqs(spanRes[0], spanRes[spanRes.length - 1]);
  if (pubEnv && pubCtx) publishAfterCommit(pubEnv, pubCtx, seqs);
  return { written: true };
}

// --- LIVE-RELEASE STOCK CEILING (QUEUE #482) --------------------------------
// The creation meter (`releaseCreateGate`, 50 / 24 h, D-0649) bounds FLOW. This
// bounds STOCK: how many LIVE releases (`registered` or `published`; a
// `withdrawn` one frees its slot) one passport may hold. Stock is the multiplier
// every per-release ceiling is multiplied by (`credentials-routes.ts` mints:
// 20/hour x 2 actor buckets x MAX_ISSUE per release), and it is what a leaked
// session or an abandoned account leaves behind; the flow meter alone lets it
// grow by 50 a day forever. Unlimited-supply editions are untouched (D-0615).
//
// SIZE, AND ITS REASON: 500 = ten full days of the ratified flow ceiling. It
// binds only a subject that has run the creation meter at its maximum for ten
// days without withdrawing anything -- the abuse shape -- while a real catalogue
// sits far below it and can always withdraw to make room. Tiered through the
// SAME seam as the flow cap (`releaseCreateTier`), so a tier's number is a
// table edit here, never a door change. Unratified: flagged for Jason.
//
// BOTH CREATING DOORS (`finishCollectibleMint` here, `finishDrop` in drop.ts)
// use these three functions and nothing else: the witness gives a typed 409
// before the ceremony; the batch gate is the guarantee at commit.

export const RELEASE_STOCK_CAP_DEFAULT = 500;
const RELEASE_STOCK_CAP_BY_TIER: Readonly<Record<string, number>> = {};

/** A tier -> its live-release ceiling; unknown or non-positive -> the default. */
export function releaseStockCapForTier(tier: string): number {
  const cap = RELEASE_STOCK_CAP_BY_TIER[tier];
  return typeof cap === "number" && Number.isFinite(cap) && cap > 0 ? cap : RELEASE_STOCK_CAP_DEFAULT;
}

export async function releaseStockCap(db: DbLike, passportId: string): Promise<number> {

/** The one spelling of "a live release of this passport", used by the witness
 * and by the batch gate so the two can never count different things. */
const LIVE_RELEASES_FROM = "FROM releases WHERE passport_id = ? AND status IN ('registered','published')";

/** The typed refusal both creating doors answer with (HTTP 409). */
export class ReleaseStockError extends Error {
  override name = "ReleaseStockError";
  readonly status = 409 as const;
  constructor(public readonly limit: number) {
    super("live release limit reached");
  }
}

/** WITNESS: refuse before any work when the passport already holds `cap` live
 * releases. Racy by itself; `releaseStockGate` is what makes it true at commit. */
export async function assertReleaseStockAvailable(db: DbLike, passportId: string): Promise<number> {

  const r = await db.prepare(`SELECT COUNT(*) AS n ${LIVE_RELEASES_FROM}`).bind(passportId).first<{ n: number }>();

}

/** GUARANTEE: appended to the signed batch AFTER the release insert, it fails

/** Complete a collectible mint: rebuild the identical draft, then commit the
 * signed event + the release/fingerprint row inserts + an auto-publish flip in ONE
 * atomic appendCreatorSigned batch (all-or-nothing). Auto-publish makes the
 * collectible a live listing immediately (withdrawable via /account/releases/:id/
 * withdraw); the publish flip is scoped to this passport so it can never touch
 * another creator's row. */
export async function finishCollectibleMint(
  db: DbLike,
  passportId: string,
  input: CollectibleInput,
  assertion: AssertionResponse,
  cfg: WebAuthnConfig,
  nowMs: number,
  mintFpId: () => string,
  now: () => string = () => new Date().toISOString(),
  pubEnv?: AnalyticsTickEnv,
  pubCtx?: { waitUntil(p: Promise<unknown>): void },
): Promise<CollectibleResult> {
  await requireVerified(db, passportId);
  const stockCap = await assertReleaseStockAvailable(db, passportId);
  const { draft, statements } = await buildCollectibleParts(db, passportId, input, mintFpId, now);
  const drafts: DraftEvent[] = [draft];
  const extra: DbStatement[] = [...statements];
  const publishedAt = now();
  extra.push(db.prepare("UPDATE releases SET status = 'published', updated_at = ? WHERE release_id = ? AND passport_id = ?").bind(publishedAt, input.collectible_id, passportId));
  // ADR-0006 §2: the `release_published` row rides the SAME signed batch as the
  // auto-publish flip, predicated on this call's own stamp.
  extra.push(db.prepare(outboxSelectSql("release_published", { table: "releases", where: "release_id = ?", extraWhere: "status = 'published' AND updated_at = ?", scopeIdExpr: "release_id" })).bind(publishedAt, input.collectible_id, publishedAt));
  // I3-2132: same predicate as the outbox row above it.
  extra.push(
    db
      .prepare(
        `INSERT INTO release_status_history (release_id, passport_id, status, at)
          SELECT release_id, passport_id, status, updated_at FROM releases
           WHERE release_id = ? AND passport_id = ? AND updated_at = ?`,
      )
      .bind(input.collectible_id, passportId, publishedAt),
  );
  extra.push(releaseStockGate(db, passportId, stockCap, nowMs));

  // #862: the ledger door brackets its batch and hands back the outbox seqs THIS commit wrote.
  let seqs: number[] = [];
  try {
    await appendCreatorSigned(db, passportId, drafts, assertion, cfg, nowMs, now, extra, undefined, (own?: readonly number[]) => {
      seqs = [...(own ?? [])];
    });
  } catch (e) {
    // Asked FIRST: the stock gate fails with a NOT NULL, which the broad match
    // below would report as an id collision.
    if (isStockCeilingRefusal(e)) throw new ReleaseStockError(stockCap);
    // The standing re-check (audit item #29) rides this batch via
    // `buildCollectibleParts` → `insertStatements`, and it fails with a NOT
    // NULL. Asked before the broad match, which would otherwise call a standing
    // refusal an id collision.
    if (isWriteAdmissionRefusal(e)) throw new ReleaseError("registering a release requires a verified creator", 403);
    if (e instanceof Error && /UNIQUE|constraint/i.test(e.message)) throw new ReleaseError("a collectible with that id already exists", 409 as 400);
    throw e;
  }
  if (pubEnv && pubCtx) publishAfterCommit(pubEnv, pubCtx, seqs);
  return { collectible_id: input.collectible_id, published: true };
}

const LIST_LIMIT = 60;

export async function collectCard(
  db: DbLike,
  accountId: string,
  passportId: string,
  now: () => string = () => new Date().toISOString(),
  /** `ANALYTICS_FAN_SECRET`; undefined is refused by `fanKey` (ADR-0006 §2). */
  fanSecret?: string,
  pubEnv?: AnalyticsTickEnv,
  pubCtx?: { waitUntil(p: Promise<unknown>): void },
): Promise<void> {
  const acct = await db.prepare("SELECT 1 AS x FROM accounts WHERE account_id = ?").bind(accountId).first();
  if (!acct) throw new AccountError("no such account", 401);
  const card = await cardData(db, passportId, Date.parse(now()));
  if (!card) throw new AccountError("no card for that creator", 404);
  // Per-account cap (bounds table growth + list fan-out). Only consulted when
  // adding a new card — a re-collect of an existing one is always a no-op.
  const already = await db.prepare("SELECT 1 AS x FROM card_collections WHERE account_id = ? AND passport_id = ?").bind(accountId, passportId).first();
  if (already) return;
  // THE CAP IS IN THE INSERT, not a COUNT read before it — counted-then-inserted
  // it is overshot by the width of a concurrent burst (measured: 10003 cards
  // against a cap of 10000 with four collects held on that read). Same shape and
  // same fix as `saveCreator` and `requestCreator`.
  // THE ANALYTICS ROW RIDES THIS BATCH (ADR-0006 §2) — the event exists iff the
  // collect does. `DbLike.batch` returns no `changes` (db.ts:20), so the cap is
  // read off the table by the re-read below.
  //
  // AND THE EVENT IS GATED ON THE COLLECT NOT ALREADY EXISTING, evaluated INSIDE
  // this transaction and AHEAD of the INSERT (`DbLike.batch` is ordered and
  // all-or-nothing — db.ts:19-20). The `already` pre-read above sits OUTSIDE the
  // transaction and closes the SEQUENTIAL repeat only: two concurrent collects
  // both read no row, both arrive here, `(account_id, passport_id)` is the
  // PRIMARY KEY (migrations/0005_social.sql:43) so the second INSERT is silently
  // no-op'd, and the cap predicate — which used to be this event's ONLY guard —
  // is true for both of them. That was two `card_collected` rows, and two votes
  // on four public boards, for one card the fan holds once.
  //
  // THE ORDER IS ALSO WHAT MAKES THE CAP TERM TRUE AT THE BOUNDARY: evaluated
  // after the INSERT it counted the row it had itself just written, so the
  // collect that FILLED the cap landed and emitted nothing at all.
  const ts = now();
  const key = await fanKey(fanSecret, accountId, passportId);
  // The INSERT's own admission condition, spelled the same way and bound the same
  // way so the two cannot drift: not already collected, and under the cap.
  const admits = {
    sql: "NOT EXISTS (SELECT 1 FROM card_collections WHERE account_id = ? AND passport_id = ?) AND (SELECT COUNT(*) FROM card_collections WHERE account_id = ?) < ?",
    params: [accountId, passportId, accountId, MAX_COLLECTION_PER_ACCOUNT],
  };
  const span = outboxSpan(db);
  const spanRes = await db.batch([
    span.open,
    // THE EVENT CARRIES THE INSERT'S OWN ADMISSION CONDITION (FINDING-0010-A-3;
    // W3-15-R6 at accounts.ts:312-315) — `saveCreator`'s shape and reason
    // exactly: at the cap, and on a collect the PK refuses, the `card_collected`
    // row and the `analytics_fan_scopes` row would otherwise commit anyway.
    ...outboxStatements(db, { kind: "card_collected", passportId, occurredDay: occurredDay(ts), fanKey: key }, { accountId, extraWhere: admits }),
    db
      .prepare(
        `INSERT INTO card_collections (account_id, passport_id, created_at)
       SELECT ?, ?, ? WHERE (SELECT COUNT(*) FROM card_collections WHERE account_id = ?) < ?
         AND EXISTS (SELECT 1 FROM accounts WHERE account_id = ?)
       ON CONFLICT(account_id, passport_id) DO NOTHING`,
      )
      // I3-0190 (D-0581, D-1109(3), INV-061): the account is re-asked INSIDE the batch, so a collect racing an erasure that
      // committed after the existence read above leaves no card_collections row under the erased id.
      .bind(accountId, passportId, ts, accountId, MAX_COLLECTION_PER_ACCOUNT, accountId),
    // #839a collect door (I16-FXC H4, I16-W13S): the collect's UTC day is its COMMIT statement's, recorded beside the row
    // this batch inserted (matched on the request instant it carries), as `saveCreator` does for a save (accounts.ts).
    db
      .prepare(
        `INSERT OR IGNORE INTO community_settled_days (account_id, passport_id, kind, settled_day)
         SELECT account_id, passport_id, 'collect', ${TOMBSTONE_COMMIT_DAY} FROM card_collections WHERE account_id = ? AND passport_id = ? AND created_at = ?`,
      )
      .bind(accountId, passportId, ts),
    span.close,
  ]);
  const seqs = outboxSpanSeqs(spanRes[0], spanRes[spanRes.length - 1]);
  if (pubEnv && pubCtx) publishAfterCommit(pubEnv, pubCtx, seqs);
  const landed = await db.prepare("SELECT 1 AS x FROM card_collections WHERE account_id = ? AND passport_id = ?").bind(accountId, passportId).first();
  if (!landed) {
    // I3-0190: the INSERT's account term refused it — the account was erased under the batch, not the cap.
    if (!(await db.prepare("SELECT 1 AS x FROM accounts WHERE account_id = ?").bind(accountId).first())) throw new AccountError("no such account", 401);
    throw new AccountError("collection is full", 400);
  }
}

/** Uncollect (idempotent). The DELETE always runs; `card_uncollected` is
 * emitted only when there was a card to remove.
 *
 * THE GUARD IS IN THE BATCH, NOT IN THE PRE-READ (:wasCollected below). An
 * unconditional emit would let any client decrement a creator's collector stock
 * once per request by uncollecting a card they never held, so the emit carries
 * `EXISTS (… FROM card_collections …)` as its `extraWhere`, evaluated inside the
 * transaction and ahead of the DELETE. The `held` pre-read below is a skip-read
 * that avoids the work and decides nothing authoritative (ADR-0003 §11). It USED
 * to be the event's only guard — and this docblock used to cite `unsaveCreator`
 * for that, which stopped being true when W3-15-R6 moved that guard into the
 * batch (accounts.ts:368-373). That is a check-then-act window a same-instant
 * double-uncollect walked straight through: both callers read a row, both
 * emitted, and `card_collected − card_uncollected` went to −1 for a card that
 * was collected once. The DELETE is left unconditional. */
export async function uncollectCard(
  db: DbLike,
  accountId: string,
  passportId: string,
  now: () => string = () => new Date().toISOString(),
  fanSecret?: string,
  pubEnv?: AnalyticsTickEnv,
  pubCtx?: { waitUntil(p: Promise<unknown>): void },
): Promise<void> {
  const acct = await db.prepare("SELECT 1 AS x FROM accounts WHERE account_id = ?").bind(accountId).first();
  if (!acct) throw new AccountError("no such account", 401);
  const del = db.prepare("DELETE FROM card_collections WHERE account_id = ? AND passport_id = ?").bind(accountId, passportId);
  const held = await db.prepare("SELECT 1 AS x FROM card_collections WHERE account_id = ? AND passport_id = ?").bind(accountId, passportId).first();
  const ts = now();
  // QUEUE #793: the tombstone rides with EVERY DELETE of this door (accounts.ts communityTombstoneStmt).
  const tomb = communityTombstoneStmt(db, "collect", accountId, passportId);
  // #839a (I16-W13S): the settled day goes with the collect it settles, after the tombstone has read it.
  const unsettle = db.prepare("DELETE FROM community_settled_days WHERE account_id = ? AND passport_id = ? AND kind = 'collect'").bind(accountId, passportId);
  if (!held) {
    await db.batch([tomb, del, unsettle]);
    return;
  }
  const key = await fanKey(fanSecret, accountId, passportId);
  // THE GUARD IS RESTATED INSIDE THE BATCH, AHEAD OF THE DELETE. The `held`
  // pre-read above is outside the transaction, so two concurrent uncollects both
  // read a row and, before this, both emitted — a PHANTOM decrement of a stock
  // the `card_collections` detail can never account for. Serialised, exactly one
  // sees it. Verbatim transfer of `unsaveCreator` (accounts.ts:372-373).
  const wasCollected = { sql: "EXISTS (SELECT 1 FROM card_collections WHERE account_id = ? AND passport_id = ?)", params: [accountId, passportId] };
  const span = outboxSpan(db);
  const spanRes = await db.batch([span.open, ...outboxStatements(db, { kind: "card_uncollected", passportId, occurredDay: occurredDay(ts), fanKey: key }, { accountId, extraWhere: wasCollected }), tomb, del, unsettle, span.close]);
  const seqs = outboxSpanSeqs(spanRes[0], spanRes[spanRes.length - 1]);
  if (pubEnv && pubCtx) publishAfterCommit(pubEnv, pubCtx, seqs);
}


/** A page of `listCollection`, plus the cursor for the next one (null once the
 * fan's whole collection has been paged through). */
export interface CollectionPage {
  /** I25-Q1 (I3-1784, D-1059(16)): `warned` is the codex's own predicate (codex.ts `CODEX_WARNED_SQL`, imported — D-1060(5)), a strict boolean. */
  cards: (CardData & { warned: boolean })[];
  next: { created_at: string; passport_id: string } | null;
}

/** This fan's collection (most-recent first, paginated), each with live card
 * data. `before` is the (created_at, passport_id) cursor `listCollection`
 * itself returned as `next` — never a bare timestamp.
 *
 * A-012: `created_at` alone is not a total order — several collects can land
 * in the same instant (a batch import, a burst of taps in one second-wide
 * bucket), and `WHERE created_at < ?` drops the WHOLE tied group at a page
 * boundary, not just the rows already served: a card sharing the cursor's
 * exact timestamp is invisible to every later page forever, not merely
 * reordered. `passport_id` is the table's own second PK column
 * (`PRIMARY KEY (account_id, passport_id)`, migration 0005) — unique per
 * account row and never reused — so `(created_at, passport_id)` is a genuine
 * total order over one account's rows, and the SQLite row-value comparison
 * `(created_at, passport_id) < (?, ?)` walks it exactly like the single-column
 * cursor did, with no row skipped or repeated at the boundary.
 */
export async function listCollection(db: DbLike, accountId: string, before?: { created_at: string; passport_id: string }, nowMs: number = Date.now()): Promise<CollectionPage> {
  const acct = await db.prepare("SELECT 1 AS x FROM accounts WHERE account_id = ?").bind(accountId).first();
  if (!acct) throw new AccountError("no such account", 401);
  const { nowIso, provedSince } = freshProofBounds(nowMs);
  const warnedBinds = [nowIso, provedSince, nowIso, provedSince, (await servingConfirmedFraud(db)) ? 1 : 0, FRAUD_REVIEW_KIND, FRAUD_REVIEW_KIND];
  const q = before
    ? db
        .prepare(
          `SELECT e.passport_id, e.created_at, ${CODEX_WARNED_SQL} AS warned FROM card_collections e
            WHERE e.account_id = ? AND (e.created_at, e.passport_id) < (?, ?)
            ORDER BY e.created_at DESC, e.passport_id DESC LIMIT ?`,
        )
        .bind(...warnedBinds, accountId, before.created_at, before.passport_id, LIST_LIMIT)
    : db
        .prepare(`SELECT e.passport_id, e.created_at, ${CODEX_WARNED_SQL} AS warned FROM card_collections e WHERE e.account_id = ? ORDER BY e.created_at DESC, e.passport_id DESC LIMIT ?`)
        .bind(...warnedBinds, accountId, LIST_LIMIT);
  const { results } = await q.all<{ passport_id: string; created_at: string; warned: number }>();
  // I3-2212: ONE batched read for the whole page (`cardDataMany`) instead of
  // one `cardData` call per row — same result, statement count independent of
  // page size.
  const cards = await cardDataMany(db, results.map((r) => r.passport_id));
  const out: (CardData & { warned: boolean })[] = [];
  for (const r of results) {
    const card = cards.get(r.passport_id);
    if (card) out.push({ ...card, warned: r.warned === 1 }); // a since-retracted card silently drops out
  }
  const last = results[results.length - 1];
  const next = results.length === LIST_LIMIT && last ? { created_at: last.created_at, passport_id: last.passport_id } : null;
  return { cards: out, next };
}

/** Rows removed per table per run: bounded like every other hygiene step. */
export const ORPHAN_SWEEP_LIMIT = 500;

/** Delete up to `limit` rows per census table whose account no longer exists, and up to `limit` per
 * MONEY_ORPHAN_SWEEPS table that is also past the money-record period at `nowMs`.
 * Returns the number removed. One statement per table, each independent. */
export async function sweepOrphanAccountRows(db: DbLike, limit: number = ORPHAN_SWEEP_LIMIT, nowMs: number = Date.now()): Promise<number> {
  let removed = 0;
  for (const s of ORPHAN_SWEEPS) {
    removed += (await db.prepare(s.sql).bind(limit).run()).changes;
  }
  const cutoff = await moneyRecordCutoffIso(db, nowMs);
  for (const s of MONEY_ORPHAN_SWEEPS) {
    removed += (await db.prepare(s.sql).bind(cutoff, limit).run()).changes;
  }
  return removed;
}
