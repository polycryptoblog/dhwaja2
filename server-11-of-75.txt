/** I3-0544 — one season key's shape, mirroring `rewards.ts` SEASON_RE / `seasonKey`. */
export const SEASON_KEY_RE = /^\d{4}-Q[1-4]$/;

/** I3-0544 — the `count` seasons BEFORE the one containing `nowMs`, newest
 * first, in `rewards.ts seasonKey`'s UTC-quarter format — the picker's options. */
export function pastSeasonKeys(nowMs: number, count: number): string[] {
  const d = new Date(nowMs);
  let y = d.getUTCFullYear();
  let q = Math.floor(d.getUTCMonth() / 3) + 1;
  const out: string[] = [];
  for (let i = 0; i < Math.max(0, Math.trunc(count)); i++) {
    q -= 1;
    if (q === 0) {
      q = 4;
      y -= 1;
    }
    out.push(`${y}-Q${q}`);
  }
  return out;
}

/**
 * THE SYNTHETIC 401, AND WHY IT IS MARKED.
 *
 * Both private helpers below short-circuit an authenticated call when there is
 * no session token: they answer 401 WITHOUT MAKING A REQUEST. That is right —
 * sending a bearer-less call to a gated route only wastes a round trip — but
 * the envelope it produces is a statement about this client, not about anything
 * a server said, and it used to be indistinguishable from a real server 401
 * with an unreadable body.
 *
 * That indistinguishability had a cost. `backup-email.ts` separates "you're
 * signed out" from "that code is wrong" by the machine token
 * `{error:"unauthorized"}` that Hono's gate sends — a token this envelope can
 * never carry, because nothing generated it. So a fan whose session had expired
 * pressed "Confirm this address" and was told "That code didn't work… Send a
 * new code and try that": a transport/auth fact rendered as a verdict about the
 * credential, telling them to burn one of a capped number of sends over a
 * request that never left the browser. The same shape told them a removal that
 * never happened had left "the address still on the account and still works".
 *
 * `noSession` is set at the ONE place the envelope is invented, so a caller can
 * tell "we did not ask" from "the server answered 401" without guessing from a
 * body. It is present only on the synthetic envelope; a real response never
 * carries it.
 */
export interface NoSessionEnvelope {
  /** True ONLY on the locally-invented 401 above: no request was made. */
  noSession?: true;
}
export type ReadEnvelope<T> = { status: number; data: T | null; message?: string } & NoSessionEnvelope;
/** I3-0792 (I23-W1-HUB H-4, web half; I23-L05): a refusal's optional plain `message` (trust-core route-helpers.ts
 * `refusal(error, message?)`, e.g. RATE_LIMIT_MESSAGE on every 429 since phase 2). Present only on a NON-ok answer whose
 * body carried a non-empty string; never a server `error` code. A view may print it as text (textContent / xml()). */
export type SendEnvelope<T> = { ok: boolean; status: number; data: T | null; message?: string } & NoSessionEnvelope;

/** The refusal sentence a non-ok body carries, or null: a non-empty string `message` of at most 300 characters. */
export function refusalMessage(body: unknown): string | null {
  const m = body !== null && typeof body === "object" ? (body as { message?: unknown }).message : undefined;
  return typeof m === "string" && m.trim() !== "" && m.length <= 300 ? m : null;
}

/** D-1003(2): the server codes of its GENERIC limiters (route-helpers.ts `refusal("rate limit exceeded" | "rate_limited",
 * RATE_LIMIT_MESSAGE)`); their per-network sentence is shown. */
const GENERIC_LIMIT_ERRORS: ReadonlySet<string> = new Set(["rate limit exceeded", "rate_limited"]);
/** D-1003(2): a 429 whose `error` is a cap's own code (any other string — "report limit reached for today", "no Scouts
 * left this month — …") names a known per-account or per-creator cap, so the envelope carries no `message` and the
 * screen keeps its own specific copy (refusalText's fallback). */
export function namesKnownCap(status: number, body: unknown): boolean {
  const e = status === 429 && body !== null && typeof body === "object" ? (body as { error?: unknown }).error : undefined;
  return typeof e === "string" && e.trim() !== "" && !GENERIC_LIMIT_ERRORS.has(e);
}

/**
 * D-0996(2) (I3-0792, I23-FX-INT): THE ONE SHARED REFUSAL RENDERER for a web screen that called the shared API helper
 * (`send` -> SendEnvelope). The server's own `message` (refusalMessage above: non-ok bodies only) is shown verbatim;
 * otherwise the screen's own copy. An UNCERTAIN outcome (status 0 or 5xx) always keeps the screen's copy, because only
 * the screen knows whether a retry is safe. Census of the screens routed through it: I23-FX-INT REPORT.md.
 */
export function refusalText(r: { status: number; message?: string }, fallback: string): string {
  return r.status !== 0 && r.status < 500 && typeof r.message === "string" && r.message.length > 0 ? r.message : fallback;
}

/** I24-W1-REFUSAL (D-0996(2)): the same rule for a screen that renders an OUTCOME CARD (removal.ts, report.ts,

export type RefusalCard = { body: string };

export function refusalOutcome<T extends RefusalCard>(r: RefusalAnswer, out: T): T {

/**
 * The account-closure notice, exactly as `services/trust-core/src/closures.ts`
 * `closureNotice` builds it: state, account id, category, deadline and appeal
 * address. The operator note, the detection signal and the reporter are
 * deliberately absent server-side, and this type is the client's record of that
 * boundary. `appeal_deadline` is on a `suspended` notice only: the server's
 * `terminated` notice has none (closures.ts `ClosureNotice`; QUEUE #660a, D-0942).
 * `account_id` is the id docs/legal/APPEALS.md asks the person to quote (D-1073(6));
 * it is carried when present and its absence does not void the notice.
 */
export interface ClosureNoticeDto {
  state: string;
  category: string;
  appeal_deadline?: string;
  appeal_to: string;
  account_id?: string;
}

/**
 * Parse a response body into a closure notice, or null. PURE and exported so
 * the guard can drive it directly.
 *
 * The status is checked as well as the token: `account_closed` is a 403 on
 * every door that emits it, and a body carrying that string under any other
 * status is not a notice this client will act on.
 */
export function closureNoticeFrom(status: number, body: unknown): ClosureNoticeDto | null {
  if (status !== 403 || body === null || typeof body !== "object") return null;
  const b = body as Record<string, unknown>;
  if (b.error !== "account_closed") return null;
  const str = (v: unknown): string | null => (typeof v === "string" && v.length > 0 ? v : null);
  const state = str(b.state);
  const category = str(b.category);
  const appeal_deadline = str(b.appeal_deadline);
  const appeal_to = str(b.appeal_to);
  // Every field the state carries, or nothing. A half-notice would render a
  // deadline with no address to send an appeal to, which is worse than the
  // generic refusal. The deadline is required for `suspended` only: the server's
  // `terminated` notice carries none (QUEUE #660a, D-0942).
  if (!state || !category || !appeal_to) return null;
  if (state === "suspended" && !appeal_deadline) return null;
  const account_id = str(b.account_id);
  return { state, category, ...(appeal_deadline ? { appeal_deadline } : {}), appeal_to, ...(account_id ? { account_id } : {}) };
}

/**
 * QUEUE #417 (D-0580) — the closed-account data-rights credential. The 403
 * `account_closed` answer from the OIDC, passkey and OTP sign-in doors carries
 * `rights_session` (`fan-account-routes.ts closedSignInBody`) — to a browser, the CSRF HANDLE of a restricted session

export class Api {
  private readonly base: string;
  private readonly fetchImpl: typeof fetch;
  /** PC-06 — one outstanding evidence POST per case ref. A hung upload (see
   * `API_TIMEOUT_MS` above) invites exactly one retry shape: the fan sees no
   * progress, picks the file again (or double-fires the file input) while the
   * first request is still in flight. The evidence route has no idempotency
   * key and no content dedup (`services/trust-core/src/evidence.ts`), so two
   * requests that both land both consume a slot of the 4-screenshot cap for
   * ONE screenshot. A second call for a case ref already in flight rides the
   * SAME promise instead of opening a second POST. This does not (and cannot,
   * from this file alone) stop a request the client gave up ON TIMEOUT from
   * still being accepted server-side after the abort — that needs an
   * idempotency key the route does not read; see VERDICTS.md PC-06. */
  private readonly evidenceInFlight = new Map<string, Promise<{ ok: boolean; status: number; error: string | null }>>();
  constructor(private readonly opts: ApiOpts) {
    this.base = opts.baseUrl.replace(/\/+$/, "");
    // Native `fetch` invoked as a method (`this.fetchImpl(...)`) throws
    // "Illegal invocation" in browsers — its `this` must be the global, not
    // this client. Wrap so the default keeps `this` correct (tests inject their
    // own fetchImpl, so the bare form only bit in a real browser).
    this.fetchImpl = opts.fetchImpl ?? ((input, init) => fetch(input, init));
  }

  /** Attach an abort deadline to a RequestInit. `cancel()` MUST be called on
   * every path out (success, parsed failure, and thrown) or the timer leaks. */
  private withDeadline(init: RequestInit): { init: RequestInit; cancel: () => void; readJson: (res: Response) => Promise<{ data: unknown; timedOut: boolean }> } {
    const controller = new AbortController();
    let expire: () => void = () => undefined;
    const expired = new Promise<void>((r) => { expire = r; });
    const timer = setTimeout(() => { controller.abort(); expire(); }, API_TIMEOUT_MS);
    // I3-1255b — THE DEADLINE COVERS THE BODY. The caller clears the timer only
    // AFTER this read, and the read is raced against the deadline itself, so a
    // reply that starts and never finishes (or a fetch double that ignores the
    // abort signal) settles as `timedOut` instead of hanging.
    const readJson = (res: Response): Promise<{ data: unknown; timedOut: boolean }> =>
      Promise.race([
        res.json().then(
          (data: unknown) => ({ data, timedOut: false }),
          () => ({ data: null, timedOut: controller.signal.aborted }),
        ),
        expired.then(() => ({ data: null, timedOut: true })),
      ]);
    return { init: { ...init, signal: controller.signal }, cancel: () => clearTimeout(timer), readJson };
  }

  /** The public share URL for a passport (#11/#18: the bio link) — built ONLY
   * from the build-time constant base + a shape-validated id; null otherwise
   * so a hostile id can never become a rendered link. */
  shareUrl(passportId: string): string | null {
    return /^psp_[0-9a-f]{16}$/.test(passportId) ? `${this.base}/p/${passportId}` : null;
  }

  private async getJson<T>(path: string, auth = false): Promise<T | null> {
    return (await this.getJsonStatus<T>(path, auth)).data;
  }

  /** getJson, keeping the HTTP status. Only the plan page needs it: it has to
   * tell "billing is switched off on this deployment" (503 — the fail-closed
   * state) apart from "you're signed out" (401) and from a plain read failure,
   * and a `null` body collapses all three into one indistinguishable answer.
   * Status 0 means the request never completed (offline / DNS / CORS). */
  private async getJsonStatus<T>(path: string, auth = false, bearer?: string): Promise<ReadEnvelope<T>> {
    const headers: Record<string, string> = {};
    let sentTok: string | null = null; // I3-1256 — the fan bearer THIS request carried
    // `bearer` is the data-rights credential (QUEUE #417) and ONLY that: it is
    // sent with auth=false, so a 401 on it can never clear the fan session.
    // RG-01 (D-1010(1)): the data-rights credential is now the HANDLE of the restricted session trust-core put in the
    // fan cookie (csrf.ts finishSessionResponse), sent under the CSRF scheme like every other browser session.
    if (bearer) headers["Authorization"] = sessionAuthorization(bearer);
    else if (auth) {
      const tok = this.opts.session.get();
      if (!tok) return { status: 401, data: null, noSession: true }; // see NoSessionEnvelope
      headers["Authorization"] = sessionAuthorization(tok); // I25-W1-COOKIE: the CSRF handle; the session is the HttpOnly cookie
      sentTok = tok;
    }
    // I25-W1-COOKIE: credentials "include" sends the session cookie (the data-rights session is that cookie too, RG-01).
    const { init, cancel, readJson } = this.withDeadline({ method: "GET", headers, ...WITH_COOKIE });
    try {
      let res: Response;
      try {
        res = await this.fetchImpl(`${this.base}${path}`, init);
      } catch {
        return { status: 0, data: null };
      }
      const body = await readJson(res);
      if (body.timedOut) return { status: 0, data: null }; // I3-1255b — a body that never ended is "never completed"
      if (!res.ok) {
        // The body is read on the error path too, and ONLY to look for the machine
        // token below — the envelope still carries `data: null`, as every caller
        // expects, so nothing downstream can start rendering a server error string.
        const err = body.data as { error?: unknown } | null;
        this.forgetRefusedSession(sentTok, res.status, err?.error);
        this.recordClosure(res.status, err);
        // #963 (I25-W1-COOKIE): a read's refusal carries the server's sentence by the same rule as `send`.
        const message = namesKnownCap(res.status, err) ? null : refusalMessage(err);
        return { status: res.status, data: null, ...(message ? { message } : {}) };
      }
      return { status: res.status, data: body.data as T | null };
    } finally {
      cancel();
    }
  }

  /**
   * A TOKEN WE HAVE JUST BEEN TOLD IS DEAD IS DROPPED, HERE, ONCE.
   *
   * `FanSession.isSignedIn()` is presence-only — it shape-checks a stored
   * string and nothing more — so a token that expired, or that another device
   * revoked with "log out everywhere", left this client rendering the whole
   * signed-in shell while every authenticated call 401'd. The fan sees an app
   * that is signed in and does not work, and the one screen that could help
   * them ("Lost your passkey?") hangs off the SIGNED-OUT screen they can no
   * longer reach. Nothing anywhere cleared the token.
   *
   * The rule is exactly one string, and it is the one Hono's fan gate already
   * sends: a REAL 401 (we asked, and a server answered) carrying
   * `{"error":"unauthorized"}` on a call we attached a bearer to. That token
   * means one thing on this API — "the credential you presented is not a live
   * fan session" — and it is the same answer for expired, epoch-revoked,
   * deleted-account and forged, deliberately (see `FanCaller` in trust-core).
   *
   * It is NOT "any 401". `/fan/otp/verify` answers 401 `invalid_code` for a
   * wrong code and `/fan/passkey/register` answers 401 for a spent challenge —
   * neither says anything about the session, and neither may drop it. Nor is it
   * the SYNTHETIC 401 above: that envelope is invented here, no request was
   * made, and there is nothing to learn from it.
   *
   * I3-1256 — AND ONLY THE TOKEN THAT WAS REFUSED. The verdict is about the
   * bearer this request carried (`sentTok`), not about whatever the store holds
   * when the answer lands: a slow 401 for an old token must not clear a session
   * the fan signed back into (or another tab installed) meanwhile. Same rule as
   * `apps/agency/src/main.ts` (sessionGen) and Android (`bearerStillCurrent`).
   */
  private forgetRefusedSession(sentTok: string | null, status: number, error: unknown): void {
    if (sentTok !== null && status === 401 && error === "unauthorized" && this.opts.session.get() === sentTok) {
      this.opts.session.clear();
      try {
        this.opts.onSessionRefused?.();
      } catch {
        // best effort: the extension hand-off never blocks the client (extension-bridge.ts)
      }
    }
  }

  /**
   * THE CLOSURE NOTICE IS THE ONE 403 THIS CLIENT MUST NOT SWALLOW.
   *
   * `closures.ts closureNotice` builds `{error:"account_closed", state,
   * category, appeal_deadline, appeal_to}` and every fan door throws it
   * (`fan-auth.ts AccountClosedError`, `oidc.ts OidcAccountClosedError`, status
   * 403). Until now nothing in this client read a single field of it: the
   * envelope carries `data: null` on a read error and the sign-in doors return
   * a bare boolean, so a fan whose account was suspended or terminated was told
   * "Sign-in failed. Please try again." — advice that is false, and that hides
   * the appeal route and its deadline, which are the only two facts that can
   * still help them.
   *
   * IT IS RECORDED IN THE PRIMITIVE, NOT AT A DOOR. There are eleven server
   * call sites that can answer with it, across OAuth, passkey, OTP and the
   * authenticated reads. Putting the check in the two request wrappers means a
   * door added later carries it without being told (FAMILY-004 P1).
   */
  private closure: ClosureNoticeDto | null = null;
  /** QUEUE #417 — the data-rights bearer, held in MEMORY ONLY. It is never
   * handed to `opts.session` (that store is the ordinary fan session, which
   * every gated call attaches) and never written to storage: it opens three doors
   * (export, delete, and the cancel-only billing flow, QUEUE #660a), and only the
   * three methods below ever send it. A reload forgets it; signing
   * in again mints a fresh one. */
  private rights: string | null = null;

  /** The last `account_closed` notice any call received, or null. Reading it
   * does NOT clear it: the notice describes a state of the account, not of one
   * request, and it stays true until a call succeeds. `clearClosure()` is the
   * explicit reset (a successful sign-in on another account). */
  lastClosure(): ClosureNoticeDto | null {
    return this.closure;
  }
  clearClosure(): void {
    this.closure = null;
    this.rights = null;
  }
  /** D-0810(1) — THE ONE WAY A SESSION IS INSTALLED. A session landing means
   * some account just proved itself, so a closed account's notice and its
   * data-rights bearer from an earlier attempt in this tab are no longer this
   * tab's state: they are dropped here, at every door, not at one caller. */
  private adoptSession(token: string): void {
    this.opts.session.set(token);
    this.clearClosure();
  }
  /** I3-0986 — THE ROTATING CALLS (passkey revoke, OAuth unlink) hand back a
   * replacement for the bearer they SENT. It is adopted only if that bearer is
   * still the stored one: a fan who signed out while the call was in flight must
   * stay signed out, and a later account's session must not be replaced by the
   * earlier account's re-mint. Returns whether the token was installed. */
  private adoptRotated(sentBearer: string | null, token: string): boolean {
    if (sentBearer === null || this.opts.session.get() !== sentBearer) return false;
    this.adoptSession(token);
    return true;
  }

  /** I25-W1-COOKIE — THE BOOTSTRAP AND THE CROSS-TAB SYNC. The page keeps the CSRF handle in memory only, so on load
   * (and whenever the tab regains focus, main.ts) it asks trust-core for the handle of the HttpOnly session cookie
   * (GET /session/csrf?kind=fan). A live session's handle is installed; a 401 (no cookie, or a dead one — signed out in
   * another tab, logged out everywhere) clears this tab; anything else (offline, 5xx) changes nothing. A session set
   * or cleared in this tab while the answer was in flight wins. */
  async syncSession(): Promise<void> {
    const before = this.opts.session.get();
    const r = await this.getJsonStatus<{ csrf?: unknown }>(`/session/csrf?kind=${FAN_SESSION_KIND}`);
    if (this.opts.session.get() !== before) return;
    if (r.status === 200 && typeof r.data?.csrf === "string") {
      if (r.data.csrf !== before) this.opts.session.set(r.data.csrf);
    } else if (r.status === 401) this.opts.session.clear();
  }

  /** I25-W1-COOKIE — sign out of THIS browser: trust-core expires the HttpOnly cookie (POST /session/signout), and the
   * handle is dropped here whatever the answer, so the tab is signed out even offline. Revoking other devices stays
   * logoutAll(). */
  async signOut(): Promise<void> {
    if (this.opts.session.get() !== null) await this.send("POST", "/session/signout", undefined, true);
    this.opts.session.clear();
  }

  /** An account_closed answer REPLACES the held notice and bearer together, each
   * with this answer's own parse (null when it carries none), so a notice from an
   * earlier account never stands beside a later account's bearer (QUEUE #660a,
   * D-0942). Any other answer leaves both as they are. */
  private recordClosure(status: number, body: unknown): void {
    if (status !== 403 || body === null || typeof body !== "object" || (body as { error?: unknown }).error !== "account_closed") return;
    this.closure = closureNoticeFrom(status, body);
    this.rights = rightsSessionFrom(status, body);
  }

  private async send<T>(method: "POST" | "DELETE", path: string, body: unknown, auth = false, bearer?: string): Promise<SendEnvelope<T>> {
    const headers: Record<string, string> = { "content-type": "application/json" };
    let sentTok: string | null = null; // I3-1256 — see forgetRefusedSession
    if (bearer) headers["Authorization"] = sessionAuthorization(bearer); // see getJsonStatus — QUEUE #417, RG-01
    else if (auth) {
      const tok = this.opts.session.get();
      // THE SYNTHETIC 401 — see NoSessionEnvelope. No request is made, so this
      // envelope is a statement about THIS CLIENT, not about anything a server
      // said, and it is marked as such.
      if (!tok) return { ok: false, status: 401, data: null, noSession: true };
      headers["Authorization"] = sessionAuthorization(tok); // I25-W1-COOKIE — see getJsonStatus
      sentTok = tok;
    }
    const baseInit: RequestInit = { method, headers, ...WITH_COOKIE };
    if (body !== undefined) baseInit.body = JSON.stringify(body);
    const { init, cancel, readJson } = this.withDeadline(baseInit);
    try {
      let res: Response;
      try {
        res = await this.fetchImpl(`${this.base}${path}`, init);
      } catch {
        return { ok: false, status: 0, data: null };
      }
      const body = await readJson(res);
      if (body.timedOut) return { ok: false, status: 0, data: null }; // I3-1255b
      const data = body.data as T | null;
      this.forgetRefusedSession(sentTok, res.status, (data as { error?: unknown } | null)?.error);
      this.recordClosure(res.status, data);
      const message = res.ok || namesKnownCap(res.status, data) ? null : refusalMessage(data);
      return { ok: res.ok, status: res.status, data, ...(message ? { message } : {}) };

    } finally {
      cancel();
    }
  }

   * (refusalText). `ok` is true exactly when `oauthSignin` answers true; no request is made for a bad provider/token. */
  async oauthSigninResult(provider: "google" | "apple", idToken: string, nonce?: string, referralCode?: string): Promise<{ ok: boolean; status: number; message?: string }> {
    if (provider !== "google" && provider !== "apple") return { ok: false, status: 0 }; // path is a fixed literal, never user input
    if (typeof idToken !== "string" || idToken.length === 0) return { ok: false, status: 0 };

    // at birth (the server links it only for a just-created account, and ignores
    // a bad one — so a stale/invalid code never blocks sign-in).

    const r = await this.send<{ session?: string; created?: boolean }>("POST", `/account/oauth/${provider}/signin`, { id_token: idToken, ...(nonce ? { nonce } : {}), ...ref });
    if (r.ok && r.data?.session) {
      this.adoptSession(r.data.session);
      return { ok: true, status: r.status };

    return { ok: false, status: r.status, ...(r.message ? { message: r.message } : {}) };
  }

   * dashed `rpt_<uuid7>` the server mints) before it can form a path.
   *

   * — 413 / 409 / 415 / 404 / 401 and, above all, the fail-closed 503, which
   * must never be presented as an accepted upload.

  async uploadEvidence(caseRef: string, bytes: ArrayBuffer): Promise<{ ok: boolean; status: number; error: string | null }> {
    if (!CASE_REF_RE.test(caseRef)) return { ok: false, status: 404, error: "no such report" };
    const tok = this.opts.session.get();
    if (!tok) return { ok: false, status: 401, error: "unauthorized" };

    // opening a second POST (see the field doc on `evidenceInFlight`).

    if (pending) return pending;
    const { init, cancel, readJson } = this.withDeadline({
      method: "POST",
      headers: { Authorization: sessionAuthorization(tok), "content-type": "application/octet-stream" },
      ...WITH_COOKIE,

    });

      try {
        let res: Response;
        try {
          res = await this.fetchImpl(`${this.base}/report/${caseRef}/evidence`, init);
        } catch {
          return { ok: false, status: 0, error: null };
        }
        const body = await readJson(res);
        if (body.timedOut) return { ok: false, status: 0, error: null }; // I3-1255b
        const data = body.data as { error?: string } | null;
        return { ok: res.ok, status: res.status, error: typeof data?.error === "string" ? data.error : null };
      } finally {
        cancel();
      }
    })();

    } finally {

    }
  }

  /** The token-scoped status read. The request_id IS the credential, so this is
   * SESSIONLESS and shape-validated before it forms a request. A malformed id
   * is reported as the SAME 404 the server gives an unknown one — the caller
   * then renders one constant card for every non-200, so nothing here can
   * become a "did this person ask to be delisted?" probe. */
  async removalStatus(requestId: string): Promise<{ status: number; data: RemovalStatusDto | null; message?: string }> {
    // The reference rides in the BODY of a POST to a constant path, and the
    // request is built in exactly one place (`reference.ts`): a URL — path OR
    // query — is what access logs and the Workers invocation log record, and
    // the reference is a bearer credential that must reach neither.
    const req = referenceStatusRequest(requestId);
    if (!req) return { status: 404, data: null };
    const r = await this.send<RemovalStatusDto>("POST", req.path, req.body);
    // Mirror getJsonStatus semantics: data only on a 200, null on every
    // failure, so an error body can never start rendering as an answer.
    return { status: r.status, data: r.ok && r.status === 200 ? r.data : null, ...(r.message ? { message: r.message } : {}) };
  }

  /**
   * PC-01: the server pages this at LIST_LIMIT (200, `services/trust-core/
   * src/collections.ts`) newest-first and hands back a `next` cursor for the
   * rest — the ORIGINAL client read only page one and dropped `next` on the
   * floor. A collector who owns MORE than 200 cards got a silently truncated
   * list, and the one caller of this method (`app.ts`'s `viewerRelation`,
   * used to compute "have you already collected THIS card") would answer
   * "no" for anything older than their 200 most recent cards. Follow every
   * page the server offers. Bounded by MAX_PAGES so a server bug can never
   * turn this into an unbounded loop — 50 * 200 == MAX_COLLECTION_PER_ACCOUNT
   * (10000), the server's own ceiling, so a real account can never be
   * truncated by this cap.
   */
  async collection(): Promise<{ collection: unknown[]; partial?: true } | null> {
    type Page = { collection: unknown[]; next: { created_at: string; passport_id: string } | null };
    const MAX_PAGES = 50;
    const all: unknown[] = [];
    let before: { created_at: string; passport_id: string } | null = null;
    for (let i = 0; i < MAX_PAGES; i++) {
      const q: string = before ? `?before_created_at=${enc(before.created_at)}&before_passport_id=${enc(before.passport_id)}` : "";
      const page: Page | null = await this.getJson<Page>(`/account/collection${q}`, true);
      // A 200 whose body is not the page shape is a failed read, not a crash: an
      // unguarded spread threw here and took the whole #/me page down with it.
      if (page === null || !Array.isArray(page.collection)) return i === 0 ? null : { collection: all, partial: true }; // I3-1262a — a later page failing keeps what already loaded, and SAYS it is not the whole list
      all.push(...page.collection);
      if (!page.next) return { collection: all };
      before = page.next;
    }
    return { collection: all, partial: true }; // the page cap was hit with more offered
  }
  /** I3-0247 — the viewer's collect state for ONE passport needs ONE round
   * trip, not the whole walk above (up to MAX_PAGES serial requests blocking the
   * passport paint). This is page one only, in `collection()`'s shape; the
   * consumer (`views.ts viewerRelation` -> `inPage`) already reads a miss on a
   * FULL page (LIST_PAGE_CAP) as unknown (`null`), never as "not collected". */
  collectionFirstPage(): Promise<{ collection: unknown[] } | null> {
    return this.getJson<{ collection: unknown[] }>(`/account/collection`, true).then((page) => (page && Array.isArray(page.collection) ? { collection: page.collection } : null));

  /**

   *
   * THE STATUS RIDES ALONG. This used to answer a bare `boolean`, which

  async fanPasskeyAuth(fields: AssertionFieldsDto): Promise<PasskeyAuthAnswer> {
    const r = await this.send<{ ok?: boolean; session?: string; error?: string }>("POST", "/fan/passkey/auth", fields);
    if (r.ok && typeof r.data?.session === "string" && r.data.session !== "") {
      this.adoptSession(r.data.session);
      return { ok: true, status: r.status, signedIn: this.opts.session.get() === r.data.session };

  // THE INTENT IS AN ARGUMENT, NOT A GUESS, and that is the whole fix on this
  // client. One pair of routes serves two completely different acts:
  //
  //   • JOIN — a brand-new fan, no account yet. ANONYMOUS by construction: no
  //     bearer, because there is nothing to send.
  //   • ADD — put a(nother) passkey on the account already in hand. The last
  //     step of lost-passkey recovery, and the "Add a passkey" button on
  //     `#/security`. AUTHENTICATED by construction.
  //
  // Both legs used to default to `auth = false`, so the ADD act never sent the
  // bearer at all: the server saw an anonymous caller, minted a JOIN challenge,
  // and the finish created a BRAND-NEW EMPTY ACCOUNT whose session this client
  // then stored over the fan's real one. Not a race and not an expiry — the
  // "Add a passkey" button did that with a perfectly live session, every time,
  // and the fan was told "Passkey ready on this device."
  //
  // Naming the act at the call site is what stops that returning: `"join"` and
  // `"add"` disagree about the bearer, about whether a session may be stored,
  // and about what `created` is allowed to be.

  /** Begin a passkey JOIN (anonymous → account birth at finish) or an ADD (the
   * bearer goes up; the server binds the ceremony to that account). */
  fanPasskeyRegisterStart(intent: PasskeyIntent) {
    return this.send<FanRegistrationOptionsDto>("POST", "/fan/passkey/register/start", undefined, intent === "add");
  }
  /**
   * Finish registration with the attestation fields (base64url, as
   * finishFanRegister expects).
   *
   * A SESSION IS INSTALLED ONLY BY A JOIN THAT SAYS IT CREATED THE ACCOUNT.
   * The server no longer mints one on the add path at all, and this is the
   * client's own belt to that: `created` is the server's signal for "this was a
   * signup", and a signup is never the answer to "add a passkey to my account".
   * An add that comes back `created:true` installs nothing and is reported as a
   * failure — an empty account is not a recovery.
   */
  async fanPasskeyRegister(fields: AttestationFieldsDto, intent: PasskeyIntent): Promise<PasskeyRegisterAnswer> {
    const r = await this.send<{ ok?: boolean; session?: string; created?: boolean; error?: string }>("POST", "/fan/passkey/register", fields, intent === "add");
    const created = typeof r.data?.created === "boolean" ? r.data.created : null;
    if (r.ok && intent === "join" && created === true && typeof r.data?.session === "string" && r.data.session !== "") {
      this.adoptSession(r.data.session);
    }
    // I3-0934 (I4-06-R1): the server's error string rides along as `serverError`
    // so registerFailureCause can tell its three 401s apart; it is only ever
    // compared against known literals, never rendered.
    return { ok: r.ok, status: r.status, created, ...(r.noSession === true ? { noSession: true as const } : {}), ...(typeof r.data?.error === "string" ? { serverError: r.data.error } : {}), ...(r.message ? { message: r.message } : {}) };
  }

  // M2 — BACKUP EMAIL + OTP FALLBACK SIGN-IN. Six routes that were live,
  // hardened and rate-limited on the server with no client at all, which meant
  // a fan who lost their passkey lost the account.
  //
  // Two shapes again, and the split is the whole security story:
  //
  //   • SESSIONED ENROLMENT (`/account/email`, `/account/email/verify`,
  //     `DELETE /account/email`, `/account/auth-state`) — you can only make an
  //     address a way INTO an account while you are already inside it. That is
  //     why the OTP leg below is not an account-takeover surface.
  //   • ANONYMOUS FALLBACK (`/fan/otp/start`, `/fan/otp/verify`) — no bearer,
  //     deliberately: the fan reaching for these has no session left to send.
  //
  // Every one returns the raw {ok, status, data} envelope, never a soft `null`:
  // the callers are state machines that must tell 503 (mail switched off here)
  // from 429 from a wrong code, and a collapsed `null` makes an outage read
  // like an answer about an address.
  //
  // NO ACCOUNT-EXISTENCE ORACLE, CLIENT SIDE. `/fan/otp/start` answers the same
  // for an address that maps to an account and one that does not; nothing in
  // this client may branch on anything that could differ between those two
  // cases (there is nothing — the body is `{ok:true}` either way). The code is
  // never stored, and no address is ever logged or put in a URL.

  /** The fan's own enrolled sign-in methods (settings screen). */
  authState() {
    return this.getJsonStatus<AuthStateDto>(`/account/auth-state`, true);
  }
  /** Add (or replace) the backup address and mail a verification code. 503 when
   * the deployment has no mailer — fail closed, never silently "added". */
  addBackupEmail(address: string, currentCode?: string) {
    // I3-0215 (I16-W09B H3; landed I16-L11): the re-post after `confirm_current` carries the code mailed to the CURRENT address.
    return this.send<{ ok?: boolean; error?: string }>("POST", "/account/email", { address, ...(currentCode ? { current_code: currentCode } : {}) }, true);
  }
  /** Redeem the mailed code against the caller's OWN account → verified. */
  verifyBackupEmail(code: string) {
    return this.send<{ ok?: boolean; verified?: boolean; error?: string }>("POST", "/account/email/verify", { code }, true);
  }
  /** Back to passkey-only. The server REFUSES (409) when this would leave the
   * account with no way in; that refusal is a real explanation, not an error. */
  /** I16-RG-AUTH:4 (I16-FXB H2): removing a CONFIRMED address first answers `outcome: "confirm_current"` (a code
   * went to that address, nothing removed); the re-send carries that code as `current_code`. */
  removeBackupEmail(currentCode?: string) {
    return this.send<{ ok?: boolean; error?: string; outcome?: string }>("DELETE", "/account/email", currentCode ? { current_code: currentCode } : undefined, true);
  }
  /** The account's own active passkeys — what can sign in, and when each was
   * added and last used. Nothing else is stored about a fan credential, so
   * nothing else is offered: a client that renders a device NAME is rendering
   * something nobody knows. */
  fanPasskeys() {
    return this.getJsonStatus<FanPasskeysDto>(`/account/fan-passkeys`, true);
  }
  /**
   * Remove ONE passkey — the lost-device act, and the only one that severs.
   *
   * IT SIGNS EVERY DEVICE OUT, INCLUDING THIS ONE. That is the point (killing
   * the credential while the session it already minted keeps working for
   * another 30 days is not a severing), so the server hands back a replacement
   * session for the device that pressed the button, and storing it is what
   * keeps the fan signed in. `resigned` reports whether that token actually
   * landed in the store — never merely that a 200 came back — because a
   * successful revoke this client failed to re-session leaves the fan signed
   * out, and the card for that has to say so rather than say "done".
   */
  async revokeFanPasskey(credentialId: string): Promise<SendEnvelope<{ ok?: boolean; revoked?: string; remaining?: number }> & { resigned: boolean }> {
    const sentBearer = this.opts.session.get(); // I3-0986 — what this call carries; see adoptRotated
    const r = await this.send<{ ok?: boolean; revoked?: string; remaining?: number; session?: string }>("POST", "/account/fan-passkey/revoke", { credential_id: credentialId }, true);
    if (r.ok && typeof r.data?.session === "string") {
      this.adoptRotated(sentBearer, r.data.session);
      return { ...r, resigned: this.opts.session.get() === r.data.session };
    }
    return { ...r, resigned: false };
  }
  /**
   * #814a — remove ONE linked Google or Apple sign-in
   * (`DELETE /account/oauth/:provider`, fan-account-routes.ts). The body must
   * say `{confirm:true}` or the server refuses. Like a passkey revoke it ends
   * every session (the account's epoch moves), so the server hands back a
   * re-minted session for this device; `resigned` reports whether it landed.
   * A 409 is the server refusing to remove the account's last way in.
   */
  async unlinkOAuth(provider: "google" | "apple"): Promise<SendEnvelope<{ ok?: boolean; unlinked?: string; error?: string }> & { resigned: boolean }> {
    const sentBearer = this.opts.session.get(); // I3-0986 — see adoptRotated
    const r = await this.send<{ ok?: boolean; unlinked?: string; error?: string; session?: string }>("DELETE", `/account/oauth/${encodeURIComponent(provider)}`, { confirm: true }, true);
    if (r.ok && typeof r.data?.session === "string") {
      this.adoptRotated(sentBearer, r.data.session);
      return { ...r, resigned: this.opts.session.get() === r.data.session };
    }
    return { ...r, resigned: false };
  }
  /** Fallback sign-in, leg 1: maybe-mail a code. ANONYMOUS by construction — no
   * bearer is attached and no session is consulted, exactly like the public
   * reads above, because the caller is a fan with no session to offer. The response is
   * identical whether or not the address maps to an account; see the block
   * comment above. */
  fanOtpStart(address: string) {
    return this.send<{ ok?: boolean; error?: string }>("POST", "/fan/otp/start", { address });
  }
  /**
   * Fallback sign-in, leg 2: redeem address + code for a FULL fan session (the
   * server's design — `/fan/otp/verify` issues an ordinary session, not a
   * scoped one; it is safe because enrolment above required a live session).
   *
   * The status rides along so the caller can separate 503 (mail off here) and
   * 429 from the ONE uniform 401 the server returns for an unknown address, an
   * unverified address, and a wrong/expired/burned code alike. `signedIn` is
   * the token actually landing in the store, never merely a 200.
   */
  async fanOtpVerify(address: string, code: string): Promise<{ ok: boolean; status: number; signedIn: boolean; message?: string }> {
    const r = await this.send<{ ok?: boolean; session?: string; account_id?: string }>("POST", "/fan/otp/verify", { address, code });
    if (r.ok && typeof r.data?.session === "string") {
      this.adoptSession(r.data.session);
      return { ok: true, status: r.status, signedIn: this.opts.session.get() === r.data.session };
    }
    return { ok: false, status: r.status, signedIn: false, ...(r.message ? { message: r.message } : {}) };
  }

  // --- Creator Codex ("seen" log) ---
  codex(before?: string) {
    // M8: request the server's max page (200) rather than letting it default to
    // 50. The Codex header shows the FULL sighting count from /stats, so fetching
    // only 50 made a 60-sighting collector look like they'd lost ten. 200 covers
    // realistic collections; the view flags the (rare) cap honestly.
    const q = typeof before === "string" && before ? `?limit=200&before=${enc(before)}` : "?limit=200";
    return this.getJson<{ entries: CodexEntryDto[] }>(`/account/codex${q}`, true);
  }
  codexStats() {
    return this.getJson<CodexStatsDto>(`/account/codex/stats`, true);
  }
  codexSeen(id: string, source = "check") {
    return PSP_RE.test(id) ? this.send("POST", "/account/codex/seen", { passport_id: id, source }, true) : Promise.resolve({ ok: false, status: 400, data: null });
  }
  setCodexLogging(logging: boolean) {
    return this.send<{ logging: boolean }>("POST", "/account/codex/settings", { logging }, true);
  }
  clearCodex() {
    return this.send<{ removed: number }>("DELETE", "/account/codex", undefined, true);
  }
}

const PRACTICE_UNCERTAIN = "We couldn't confirm whether that went through. Nothing was staked either way — reload the page to see your queue and matches before trying again.";
/** war-match.ts ROSTER_REFUSAL's opening words: the server's one whole-roster refusal. */
const PRACTICE_ROSTER_REFUSED = "roster not currently playable";

async function joinWarPracticeOnce(deps: Pick<AppDeps, "api">, seed: () => string): Promise<{ ok: boolean; message: string }> {
  const shelf = await deps.api.warInstances();
  const instances = shelf.data && Array.isArray(shelf.data.instances) ? shelf.data.instances : null;
  if (!instances) return { ok: false, message: "We couldn't read your cards just now, so nothing was sent. Try again in a moment." };
  let clientSeed: string;
  try {
    clientSeed = seed();
  } catch {
    return { ok: false, message: "This browser couldn't produce the random number a match needs, so nothing was sent. Try again, or try another browser." };
  }
  const input = practiceJoinInput(instances, clientSeed);
  if (!input) return { ok: false, message: "A practice match needs 10 unstaked cards from 10 different creators." };
  let r = await deps.api.joinWarQueue(input);
  const refusal = (x: typeof r): string | null => {
    const e = (x.data as { error?: unknown } | null)?.error;
    return typeof e === "string" && e ? e : null;
  };
  if (!r.ok && r.status === 409 && refusal(r)?.startsWith(PRACTICE_ROSTER_REFUSED)) {
    const inGame = await practiceInGame(deps, instances);
    const retry = practiceJoinInput(instances, clientSeed, (p) => inGame.get(p) === true);
    if (retry && retry.roster.join() !== input.roster.join()) r = await deps.api.joinWarQueue(retry);
    else if (!retry && ![...inGame.values()].includes(null)) {
      return { ok: false, message: "Fewer than 10 of the creators on your shelf are in the game right now, so a practice match can't be fielded. Nothing was staked." };
    }
  }
  if (r.ok && r.data?.ok) return { ok: true, message: "You're in a practice match." };
  if (r.status === 0 || r.status >= 500) return { ok: false, message: PRACTICE_UNCERTAIN };
  const err = refusal(r);
  if (err === "account_suspended") return { ok: false, message: refusalText(r, "Your account can't play matches right now.") };
  if (err && !/^[a-z][a-z0-9_]*$/.test(err)) return { ok: false, message: refusalText(r, err) };
  return { ok: false, message: refusalText(r, "That didn't go through, so nothing was staked. Try again in a moment.") };
}

/** RGC-02 — which of the shelf's creators the server says are still in the game:
 * the PUBLIC presentation's `card_for_sale` (index.ts: card AND War opt-in AND
 * served ratings — the gate `loadRoster`/`snapshotStats` apply), read without the
 * fan bearer (`Api.presentation`), so no encounter is written. ABSENT MEANS NO;
 * an unreadable creator is `null`. Distinct unstaked creators in shelf order,
 * ten reads at a time, stopping once ten are confirmed. */
async function practiceInGame(deps: Pick<AppDeps, "api">, instances: InstanceDto[]): Promise<Map<string, boolean | null>> {
  const order: string[] = [];
  for (const c of instances) if (c && c.escrowed !== true && typeof c.passport_id === "string" && !order.includes(c.passport_id)) order.push(c.passport_id);
  const out = new Map<string, boolean | null>();
  let yes = 0;
  for (let i = 0; i < order.length && yes < PRACTICE_BRACKET; i += PRACTICE_BRACKET) {
    const chunk = order.slice(i, i + PRACTICE_BRACKET);
    const got = await Promise.all(chunk.map((p) => deps.api.presentation(p).catch(() => null)));
    chunk.forEach((p, k) => {
      const g = got[k];
      const v = g ? g.card_for_sale === true : null;
      out.set(p, v);
      if (v) yes++;
    });
  }
  return out;
}

/**
 * Render a trust-check result the ROUTE could not produce: an UNSUPPORTED input
 * (an empty box, a 2 KB paste, an unparseable link) has no subject, so there is
 * nothing to put in a hash and nothing to come back to. Everything that DOES
 * canonicalize now goes through `#/check/...` (see the `check` case above), so
 * this is the error path only.
 */
export function renderCheck(input: string, result: CheckResult, deps: AppDeps, reason?: string): string {
  // `reason` is the classifier's own sentence for an UNSUPPORTED input (empty
  // box, oversized paste, unparseable link). It was computed and discarded, so
  // all three rendered one line about "that link" — including for a fan who

  return shell(checkResultView(input, result, { ...(deps.install ? { install: deps.install } : {}), ...(reason ? { reason } : {}), ...(deps.ssrBase ? { howHref: `${deps.ssrBase}/how-it-works`, ssrBase: deps.ssrBase } : {}), ...(deps.exampleVerifiedUrl ? { exampleHref: deps.exampleVerifiedUrl } : {}) }), { signedIn: deps.session.isSignedIn(), route: "#/" });
}

 * rail existed for a flow nobody could start.
 *
 * WHAT THIS IS, EXACTLY. `/fan/otp/verify` issues a FULL fan session — not a
 * scoped "re-enrol a passkey" credential. That is the server's design and this
 * client matches it rather than inventing a narrower mode the server would not
 * enforce. It is defensible for one reason only: enrolment is OPT-IN FROM
 * INSIDE. An address becomes a way in only after the fan added it while
 * ALREADY SIGNED IN and proved the mailbox with a code, so email can never take
 * over an account that never enrolled one. That single fact is why the UX below
 * is allowed to exist, and it is why the enrolment screen says the cost out
 * loud, once, at the moment of the choice.
 *
 * PASSKEY STAYS PRIMARY. Email is presented as a backup everywhere and never as
 * a way to sign in; the recovery screen's last step is "register a new passkey
 * now", because that — not the OTP — is the act that actually recovers the fan.
 *
 * THE NO-ACCOUNT-EXISTENCE ORACLE, RESTATED AS A CLIENT RULE. `/fan/otp/start`
 * returns the identical body and status for an address that maps to an account
 * and one that does not, and `/fan/otp/verify` returns ONE 401 for an unknown
 * address, an unverified address and a wrong/expired/burned code alike. So:
 *
 *   - there is exactly ONE success copy for a start (`OTP_MAYBE_SENT`) and
 *     exactly ONE failure copy for a verify (`OTP_BAD_CODE`), and neither is
 *     ever varied by anything derived from the address;
 *   - nothing here ever probes an address before sending (no "check if this
 *     account exists" pre-flight), and nothing retries on the client's own
 *     initiative — a retry loop would eat the per-IP send cap that makes
 *     address-scanning expensive;
 *   - the code is never stored and the address is never logged, put in a hash,
 *     or placed in a URL.
 *
 * FAIL CLOSED, AND AN ERROR IS NEVER REASSURANCE. A 503 means the mailer is
 * unconfigured on this deployment — the feature is OFF. It must never render as
 * "no email on file" or as a sent code. A read of the fan's own auth state that
 * did not complete renders the UNREADABLE card, never the "no backup email"
 * fact.
 */

/** The address shape `fan-auth.ts` enforces. Checked here only to keep a typo
 * from spending a send-cap slot; the server check is the real one. Shape says
 * nothing about who has an account, so this is not an oracle. */
export const EMAIL_RE = /^[^\s@]{1,64}@[^\s@]{1,255}\.[^\s@]{2,24}$/;
/** `otp-policy.ts`: 6 digits, one use, 10 minutes, dead after 5 wrong tries. */
export const OTP_DIGITS = 6;
export const OTP_TTL_MIN = 10;
export const OTP_MAX_ATTEMPTS = 5;
export const OTP_CODE_RE = /^[0-9]{6}$/;

/** Trim only. Never lowercased here: the server lowercases, and a client that
 * "helpfully" rewrote what was typed would show one address and send another. */
export function cleanAddress(raw: string): string {
  return typeof raw === "string" ? raw.trim() : "";
}
export function looksLikeAddress(raw: string): boolean {
  return EMAIL_RE.test(cleanAddress(raw));
}
/** Digits only, so a pasted "123 456" or "123-456" still verifies. */
export function cleanCode(raw: string): string {
  return typeof raw === "string" ? raw.replace(/\D/g, "").slice(0, OTP_DIGITS) : "";
}
/**
 * Shape check before a code is submitted, and it EARNS ITS KEEP: a code dies
 * after ${OTP_MAX_ATTEMPTS} wrong tries, so an empty field or a half-typed code
 * posted straight to the server would spend one of the five the fan has. Shape
 * is public and identical for everyone, so this leaks nothing.
 */
export function looksLikeCode(raw: string): boolean {
  return OTP_CODE_RE.test(cleanCode(raw));
}

// ---------------------------------------------------------------------------
// Outcomes (pure). Every one maps an {ok,status,data} envelope to copy; none
// echoes a server string, so a new server error can never render raw.
// ---------------------------------------------------------------------------

export interface Outcome {
  kind: string;
  tone: "good" | "quiet";
  headline: string;
  body: string;
}

type Envelope = {
  ok: boolean;
  status: number;
  /** `step`: I3-0215 — "confirm_current" when a change of a verified address first needs the code mailed to it.
   * `reason` / `retry_after_seconds`: I3-0762 — which limit a 429 hit and how long until it clears (authenticated
   * backup-email routes only). Machine tokens and a number, never prose, so reading them echoes no server copy. */
  data: { error?: string; ok?: boolean; verified?: boolean; step?: string; outcome?: string; reason?: string; retry_after_seconds?: number } | null;
  noSession?: true;
};

/**
 * 401 IS THREE DIFFERENT ANSWERS on the sessioned legs, and only two of them
 * come from a server:
 *
 *  1. THE CLIENT NEVER ASKED. `Api.send` short-circuits an authenticated call
 *     with no token to `{ok:false,status:401,data:null}` and makes no request.
 *     It marks that envelope `noSession` (see NoSessionEnvelope in api.ts),
 *     because nothing else in it could ever distinguish it: it carries no body,
 *     so it could not carry the machine token below, and it fell straight past
 *     this separator into "That code didn't work" — a verdict about the fan's
 *     credential, from a request that never left the browser, telling them to
 *     spend one of a capped number of sends. The mark has to be made at the
 *     source; a client cannot infer "I did not ask" from the shape of a
 *     non-answer.
 *  2. HONO'S GATE. `{error:"unauthorized"}` — the session is gone or rejected.
 *     A machine token, not a sentence, so reading it is not echoing server copy.
 *  3. `FanAuthError` — 401 with prose when the CODE is wrong, expired or spent.
 *     Everything at 401 that is neither of the above is this.
 */
function isNoSession(r: Envelope): boolean {
  if (r.noSession === true) return true;
  return r.status === 401 && r.data?.error === "unauthorized";
}

const SIGNED_OUT: Outcome = {
  kind: "signed_out",
  tone: "quiet",
  headline: "You're signed out",
  body: "This needs the account it belongs to, so sign in first. Nothing was sent and nothing changed.",
};

const UNAVAILABLE: Outcome = { kind: "unavailable", tone: "quiet", headline: "Backup email is switched off here", body: MAIL_UNAVAILABLE };

/** I3-0215 (D-0920 A): what a fan is told when changing a CONFIRMED address first needs that address's code. */
export const CHANGE_CONFIRM_CURRENT = `To change a confirmed address we first check it's really you: we've sent a ${OTP_DIGITS}-digit code to the address you already confirmed. Enter it below within ${OTP_TTL_MIN} minutes and we'll then send a code to the new address. Nothing has changed yet — the address you confirmed still works.`;

/** I16-RG-AUTH:4 (D-0920 A, D-0929(9)): DELETE /account/email on a CONFIRMED address answers `outcome: "confirm_current"`
 * (fan-auth.ts removeFanEmail) — a code went to that address and nothing was removed. Says nothing about where the
 * code is typed: the field that re-sends it is the screen's (main.ts), not this card's. */
export const REMOVE_CONFIRM_CURRENT = `To remove a confirmed address we first check it's really you: we've sent a ${OTP_DIGITS}-digit code to that address, and it expires in ${OTP_TTL_MIN} minutes. Nothing has been removed yet — the address still works and still signs you in.`;

/** I3-0762: the 429 copy, by the limit the server named. A wait it named is shown (`{seconds}` / `{minutes}` are
 * filled at runtime, so every client carries the identical sentence); nothing else is guessed. */
export const CAP_COOLDOWN_WAIT = "Codes are spaced out so this can't be used to mail strangers. Nothing was sent just now — try again in {seconds} seconds.";
export const CAP_HOURLY_WAIT = "There's an hourly limit on codes, which is what stops this being a way to mail strangers. Nothing was sent just now — try again in about {minutes} minutes.";
/** I3-0215: a 401 on the change that is not the session gate — the code for the confirmed address was wrong. */
export const BAD_CURRENT_CODE = `Use the newest ${OTP_DIGITS}-digit code we sent to the address you already confirmed — a code works once and expires ${OTP_TTL_MIN} minutes after it is sent. Nothing has changed: the address you confirmed still works.`;

export function capOutcome(data: Envelope["data"]): Outcome {
  const wait = typeof data?.retry_after_seconds === "number" && Number.isFinite(data.retry_after_seconds) && data.retry_after_seconds > 0 ? Math.ceil(data.retry_after_seconds) : null;
  if (wait !== null && data?.reason === "cooldown") {
    const card: Outcome = { kind: "capped_cooldown", tone: "quiet", headline: "One code at a time", body: CAP_COOLDOWN_WAIT };
    return { ...card, body: card.body.replace("{seconds}", String(wait)) };
  }
  if (wait !== null && (data?.reason === "email_hourly_cap" || data?.reason === "ip_hourly_cap")) {
    const card: Outcome = { kind: "capped_hourly", tone: "quiet", headline: "That's enough codes for this hour", body: CAP_HOURLY_WAIT };
    return { ...card, body: card.body.replace("{minutes}", String(Math.max(1, Math.ceil(wait / 60)))) };
  }
  return { kind: "capped", tone: "quiet", headline: "That's enough codes for now", body: "There's an hourly limit on codes, which is what stops this being a way to mail strangers. Nothing was sent just now — wait a minute and ask again." };
}

/** POST /account/email — add the address and mail a code. */
export function enrolStartOutcome(r: Envelope): Outcome {
  if (r.ok && r.status === 200 && r.data?.step === "confirm_current") {
    return { kind: "confirm_current", tone: "good", headline: "Check the address you already confirmed", body: CHANGE_CONFIRM_CURRENT };
  }
  if (r.ok && r.status === 200) {
    return {
      kind: "code_sent",
      tone: "good",
      headline: "Check that inbox",
      body: `We've sent a ${OTP_DIGITS}-digit code. Enter it below within ${OTP_TTL_MIN} minutes. The address isn't a way into your account until you do — an unverified address signs nobody in.`,
    };
  }
  if (r.status === 503) return UNAVAILABLE;
  if (isNoSession(r)) return SIGNED_OUT;
  // D-1016(1): adding or changing the address needs a fresh sign-in (403 reauth_required, as the passkey add). The
  // caller signs this browser out before the card shows (main.ts), so the signed-out card is true as written.
  if (isReauthRequired(r.status, r.data?.error)) return { ...SIGNED_OUT, kind: "reauth" };
  if (r.status === 429) return capOutcome(r.data);
  // I3-0215: a 401 that is not the session gate is the CURRENT address code being wrong or expired.
  if (r.status === 401) return { kind: "bad_current_code", tone: "quiet", headline: "That code didn't work", body: BAD_CURRENT_CODE };
  if (r.status === 400) {
    return { kind: "bad_address", tone: "quiet", headline: "That doesn't look like an email address", body: "Check it and try again. Nothing was sent, and nothing on your account changed." };
  }
  // I3-2171 (I4-11-R4): no answer (status 0) or a 5xx is UNCONFIRMED — the
  // server may have sent the code before the response was lost.
  if (r.status === 0 || r.status >= 500) return { kind: "failed", tone: "quiet", headline: "We couldn't confirm that", body: "A code may or may not have been sent. Check that inbox; if nothing arrives in a few minutes, ask again. Your passkey is unaffected either way." };
  return { kind: "failed", tone: "quiet", headline: "We couldn't send that code", body: "Nothing was sent and nothing on your account changed. Try again in a moment — and your passkey is unaffected either way." };
}

/** POST /account/email/verify — redeem the code against your OWN account. */
export function enrolVerifyOutcome(r: Envelope): Outcome {
  if (r.ok && r.data?.verified === true) {
    return {
      kind: "verified",
      tone: "good",
      headline: "Backup email confirmed",
      body: "If you ever lose your passkey, this address can mail you a code to get back in. Your passkey is still how you sign in normally — nothing about it changed.",
    };
  }
  if (isNoSession(r)) return SIGNED_OUT;
  // A 200 THAT DID NOT SAY `verified` IS NOT A REASSURANCE. The server read the
  // code to answer at all, so it is spent — "the code you were sent is still
  // good until it expires" was a promise about a credential that no longer
  // works, and the fan would burn tries on it.
  if (r.ok) {
    return {
      kind: "unreadable",
      tone: "quiet",
      headline: "We couldn't read that answer",
      body: "Dhwaja answered, but not with anything we could read, so we can't tell you whether that address is confirmed. Treat the code you just entered as used — it won't work twice. Open Sign-in & recovery again to see what your account actually holds.",
    };
  }
  if (r.status === 401) {
    return {
      kind: "bad_code",
      tone: "quiet",
      headline: "That code didn't work",
      body: `Codes are ${OTP_DIGITS} digits, work once, and expire ${OTP_TTL_MIN} minutes after they're sent — an expired one and a mistyped one look the same from here. Send a new code and try that. Nothing on your account changed.`,
    };
  }
  if (r.status === 400) {
    return { kind: "no_code", tone: "quiet", headline: "Enter the code first", body: `The code is the ${OTP_DIGITS} digits in the email we sent. Nothing on your account changed.` };
  }
  return { kind: "failed", tone: "quiet", headline: "We couldn't check that code", body: "We didn't get an answer, so this says nothing about your code — and we can't tell you whether it was checked. If it was, it's spent: ask for a new one rather than retrying this one. Your passkey is unaffected either way." };
}

/** DELETE /account/email — back to passkey-only. */
/**
 * NOTE FOR THE NEXT EDITOR — why the never-confirmed case has no card of its
 * own. `data-act="email-remove"` now also appears in the PENDING enrolment
 * state (a fan who typed a stranger's address had no way to clear it), and the
 * obvious refinement is a second success card saying "it was never confirmed,
 * so nothing about signing in has changed". Both sentences below are already
 * TRUE of that state — an unverified address could not sign anyone in, and the
 * passkey was and remains the only way in — so the refinement buys precision,
 * not honesty, and it is not free: `apps/mobile-check/test/backup-email-cross-client.test.ts`
 * extracts every card in this module BY POSITION and compares the sequence with
 * the Swift and Kotlin mirrors, so an inserted card is a three-client change
 * plus a re-pin. Do it as that change, or not at all. The differing COST of the
 * two acts is stated where it matters — in the confirm dialog, in main.ts.
 */
export function removeOutcome(r: Envelope): Outcome {
  // I16-RG-AUTH:4: a 200 that says `confirm_current` removed NOTHING — it mailed a code to the confirmed address.
  if (r.ok && r.status === 200 && r.data?.outcome === "confirm_current") {
    return { kind: "remove_confirm_current", tone: "good", headline: "Check the address you're removing", body: REMOVE_CONFIRM_CURRENT };
  }
  if (r.ok && r.status === 200) {
    return {
      kind: "removed",
      tone: "good",
      headline: "Backup email removed",
      body: "That address can't sign in here any more, and any code already sent to it is dead. Your passkey is now the only way in — keep it safe.",
    };
  }
  if (r.status === 409) return { kind: "only_method", tone: "quiet", headline: "That's your only way in", body: ONLY_SIGNIN_METHOD };
  if (isNoSession(r)) return SIGNED_OUT;
  // I49-A050: removing the address needs a fresh sign-in (403 reauth_required, as the add); the caller signs out first.
  if (isReauthRequired(r.status, r.data?.error)) return { ...SIGNED_OUT, kind: "reauth" };
  // I16-RG-AUTH:4: a 401 that is not the session gate is the confirmed address's code being wrong or expired.
  if (r.status === 401) return { kind: "bad_current_code", tone: "quiet", headline: "That code didn't work", body: BAD_CURRENT_CODE };
  if (r.status === 404) {
    return { kind: "none", tone: "quiet", headline: "There's no backup email to remove", body: "This account doesn't have one — your passkey is already the only way in." };
  }
  // A DEFINITE REFUSAL (400/403/429) is an answer, not a dropped request: the
  // server was reached and said no, so nothing changed. Only 0/5xx/unmapped
  // failures below are genuinely unanswered.
  if (r.status === 400 || r.status === 403 || r.status === 429) {
    return { kind: "refused", tone: "quiet", headline: "Nothing changed", body: "That request was refused, so nothing changed — your backup email is exactly as it was. Open Sign-in & recovery again if you want to check." };
  }
  // A 2xx we could not read: the removal may well have happened.
  if (r.ok) {
    return {
      kind: "unreadable",
      tone: "quiet",
      headline: "We couldn't read that answer",
      body: "Dhwaja answered, but not with anything we could read, so we can't tell you whether the address was removed. Open Sign-in & recovery again to see what your account actually holds.",
    };
  }
  // AND NEITHER IS A FAILED REQUEST. "The address is still on the account and
  // still works" is a claim about server state, and this branch is reached for
  // a request that may never have been sent — it was what the sessionless 401
  // rendered before `noSession` existed.
  return { kind: "failed", tone: "quiet", headline: "We couldn't remove that", body: "We didn't get an answer, so we can't tell you whether that removal happened. Open Sign-in & recovery again and look at what your account holds — don't assume it either way." };
}

/**
 * POST /account/fan-passkey/revoke — the lost-device act.
 *
 * `resigned` IS PART OF SUCCESS, not a detail. A revoke signs every device out
 * on purpose, so a 200 this client failed to re-session leaves the fan signed
 * out — and "That passkey is gone" over a dead session is the same class of
 * lie as the copy this whole change exists to fix. Only a revoke that came back
 * WITH a session that actually landed in the store is reported as done.
 */
export function revokeOutcome(r: Envelope & { resigned?: boolean }): Outcome {
  if (r.ok && r.status === 200 && r.data?.ok === true && r.resigned === true) {
    return {
      kind: "passkey_revoked",
      tone: "good",
      headline: "That passkey is gone",
      body: "It can't sign in here again — not from that device and not from anywhere else. Every device that was signed in has been signed out, including the one you removed; you're still signed in here. Nothing else on your account changed.",
    };
  }
  // Ordered before the signed-out check on purpose: a 409 is the server
  // REFUSING to strand the account, which is a fact about the caller's own
  // account and the one refusal they can act on.
  if (r.status === 409) return { kind: "last_way_in", tone: "quiet", headline: "That's your only way in", body: LAST_WAY_IN };
  if (isNoSession(r)) return SIGNED_OUT;
  if (r.status === 404) {
    // ONE answer for every passkey this account cannot remove — already gone,
    // never existed, or somebody else's. The server collapses them deliberately
    // (a credential id is globally unique, so telling them apart would let
    // anyone test whether an id exists in Dhwaja) and so does this.
    return {
      kind: "no_such_passkey",
      tone: "quiet",
      headline: "There's no such passkey on this account",
      body: "Nothing was removed and nothing changed. It may already be gone — open Sign-in & recovery again to see what this account actually holds.",
    };
  }
  if (r.status === 503) {
    return {
      kind: "revoke_unavailable",
      tone: "quiet",
      headline: "We can't do that right now",
      body: "Removing a passkey signs every device out, and Dhwaja can't issue you a new session on this deployment at the moment — so it refused rather than sign you out with no way back in. Nothing was removed. Try again later.",
    };
  }
  // A 2xx WE COULD NOT READ IS NOT A SUCCESS AND NOT A FAILURE, and here it is
  // worse than usual: this device's session may have been ended by a revoke
  // that did happen, so the card must not claim either.
  if (r.status >= 200 && r.status < 300) {
    return {
      kind: "revoke_unreadable",
      tone: "quiet",
      headline: "We couldn't read that answer",
      body: "Dhwaja answered, but not with anything we could read, so we can't tell you whether that passkey was removed — and if it was, this device has been signed out along with the others. Open Sign-in & recovery again and look at what the account actually holds.",
    };
  }
  return {
    kind: "revoke_failed",
    tone: "quiet",
    headline: "We couldn't remove that passkey",
    body: "We didn't get an answer, so this says nothing about that passkey either way — don't assume it's gone and don't assume it's still there. Open Sign-in & recovery again and look at what the account actually holds.",
  };
}

/** POST /fan/otp/start — the anonymous maybe-send. */
export function otpStartOutcome(r: { ok: boolean; status: number; data: { error?: string } | null }): Outcome {
  // ONE success body, always. Do not add a branch here that could differ
  // between an address with an account and one without — there is nothing in
  // the response to branch on, and inventing one re-opens the oracle.
  if (r.ok && r.status === 200) return { kind: "maybe_sent", tone: "good", headline: "Check that inbox", body: OTP_MAYBE_SENT };
  if (r.status === 503) return UNAVAILABLE;
  if (r.status === 429) {
    return { kind: "capped", tone: "quiet", headline: "Too many codes from here", body: "There's an hourly cap on codes from one connection — it's what makes guessing addresses expensive. Nothing was sent just now. Wait a few minutes and ask again." };
  }
  if (r.status === 400) {
    return { kind: "bad_address", tone: "quiet", headline: "That doesn't look like an email address", body: "Check it and try again. Nothing was sent." };
  }
  return { kind: "failed", tone: "quiet", headline: "We couldn't send that code", body: "Nothing was sent. Try again in a moment — and if your passkey still works, use it instead." };
}

/** POST /fan/otp/verify — the anonymous redeem → a full fan session. */
export function otpVerifyOutcome(r: { ok: boolean; status: number; signedIn: boolean }): Outcome {
  if (r.ok && r.signedIn) {
    return {
      kind: "signed_in",
      tone: "good",
      headline: "You're in",
      body: "Now set up a passkey on this device. That's the part that actually gets you your account back — a code from an email is a spare key, and you shouldn't be carrying only a spare.",
    };
  }
  if (r.status === 503) return UNAVAILABLE;
  if (r.status === 429) {
    return { kind: "capped", tone: "quiet", headline: "Too many tries from here", body: "There's an hourly cap on sign-in attempts from one connection. Nothing about your account changed. Wait a few minutes and try again." };
  }
  if (r.status === 400) {
    return { kind: "bad_input", tone: "quiet", headline: "Enter the address and the code", body: `Both, exactly as they are in the email — the code is the ${OTP_DIGITS} digits.` };
  }
  // 401 and everything else that is not a session: ONE body. See OTP_BAD_CODE.
  if (r.status === 401) return { kind: "bad_code", tone: "quiet", headline: "That didn't work", body: OTP_BAD_CODE };
  // A 2xx THAT DID NOT SIGN ANYONE IN. The server got far enough to answer, so
  // the code was read and is spent; "Nothing changed. Try again in a moment."
  // sent the fan back to re-enter a dead code and spend one of five tries. This
  // branch cannot leak the oracle: the server answers 401, uniformly, for an
  // unknown address, an unverified address and a bad code alike, so a 2xx here
  // is reachable only for an address that HAS an account — and this copy names
  // no address and asserts nothing about one.
  // NOTE the predicate: `fanOtpVerify` only sets `ok:true` when a session token
  // actually landed in the store, so this case arrives as `ok:false` with a 2xx
  // status — keying on `ok` here would never fire.
  if (r.status >= 200 && r.status < 300) {
    return {
      kind: "no_session",
      tone: "quiet",
      headline: "That didn't sign you in",
      body: "Dhwaja answered, but not with anything that signs you in — so you're not signed in. Treat that code as used; it won't work twice. Ask for a new one, and if your passkey still works, use it instead.",
    };
  }
  return { kind: "failed", tone: "quiet", headline: "We couldn't sign you in", body: "The check didn't complete, so this isn't an answer about your code or your address — and we can't tell you whether the code was spent. Ask for a new one rather than retrying this one, and if your passkey still works, use it instead." };
}

/**
 * CANCELLING A CHANGE — `DELETE /account/email` pressed while the account is
 * waiting on a new address.
 *
 * IT IS THE SAME CALL AS THE REMOVAL, and the server decides which act it is
 * from the row rather than from anything this client sends: with a change in
 * flight it cancels the CHANGE and leaves the confirmed address alone; with no
 * change in flight it is the removal it has always been. So this screen offers
 * exactly one of the two labels, chosen from the panel it just read, and the
 * dangerous reading is never reachable from the state where the harmless one is
 * meant.
 *
 * `stillConfirmed` IS PART OF SUCCESS, for the same reason `resigned` is part
 * of a revoke's. A 200 is not by itself evidence of WHICH act happened — the
 * only way this client is wrong about that is if the change completed between
 * the read and the press, and then the call was a removal. So "Change
 * cancelled" is claimed only when the account, read back afterwards, still
 * holds a confirmed address; anything else says what it does not know.
 */
export function cancelPendingOutcome(r: Envelope, stillConfirmed: boolean | null): Outcome {
  if (r.ok && r.status === 200 && stillConfirmed === true) {
    return {
      kind: "change_cancelled",
      tone: "good",
      headline: "Change cancelled",
      body: "The address you were adding has been dropped, and nothing was mailed to it again. Your backup email is exactly as it was — same address, still confirmed, still able to mail you a code if you lose your passkey.",
    };
  }
  if (isNoSession(r)) return SIGNED_OUT;
  // I49-A050: the cancel is the same DELETE, so the same fresh-sign-in refusal; the caller signs out first.
  if (isReauthRequired(r.status, r.data?.error)) return { ...SIGNED_OUT, kind: "reauth" };
  if (r.status === 404) {
    return {
      kind: "change_none",
      tone: "quiet",
      headline: "There's no change to cancel",
      body: "This account isn't waiting on a new address, so nothing was cancelled and nothing was removed. Open Sign-in & recovery again to see what it actually holds.",
    };
  }
  // A DEFINITE REFUSAL (400/403/429): the server was reached and said no, so
  // the pending change (if any) is untouched — this is an answer, not silence.
  if (r.status === 400 || r.status === 403 || r.status === 429) {
    return {
      kind: "change_refused",
      tone: "quiet",
      headline: "Nothing changed",
      body: "That request was refused, so nothing was cancelled and nothing was removed. Open Sign-in & recovery again if you want to check what the account holds.",
    };
  }
  // A 200 THAT LEFT NO CONFIRMED ADDRESS BEHIND. Reachable only if the change
  // completed between the read and the press — in which case this was a
  // removal, not a cancel, and saying "cancelled" would be a claim about an act
  // that did not happen.
  // W-09 — AN UNREADABLE READ-BACK IS NOT THE FACT THAT THE ADDRESS IS GONE.
  // `stillConfirmed` used to be a Boolean and every caller derived it as
  // "panel === verified", so `authPanelState`'s THIRD value — Unreadable, which
  // it returns for a non-200, a missing body or a malformed field — arrived here
  // as `false` and fell into `change_gone` below, printing "this account no
  // longer shows a confirmed backup email" over a read that said NOTHING. That
  // is the law this file states one function up ("'No backup email on this
  // account' is a FACT and only an answered read may state it"), broken by the
  // caller rather than by the table. `null` is that third value, and it stops
  // here, above the row that makes a claim.
  if (r.ok && stillConfirmed === null) {
    return {
      kind: "change_unreadable",
      tone: "quiet",
      headline: "We couldn't read that answer",
      body: "Dhwaja answered the cancel, but we couldn't read this account back afterwards — so we can't tell you whether the change was called off or the address was removed. Open Sign-in & recovery again and look at what the account actually holds.",
    };
  }
  if (r.ok) {
    return {
      kind: "change_gone",
      tone: "quiet",
      headline: "We can't tell you that was cancelled",
      body: "Dhwaja answered, but this account no longer shows a confirmed backup email — so we won't tell you a change was cancelled when the read back says otherwise. Open Sign-in & recovery again and look at what the account actually holds.",
    };
  }
  return {
    kind: "change_failed",
    tone: "quiet",
    headline: "We couldn't cancel that",
    body: "We didn't get an answer, so we can't tell you whether the change was cancelled — don't assume it either way. Open Sign-in & recovery again and look at what the account actually holds.",
  };
}

// ---------------------------------------------------------------------------
// The settings panel state (GET /account/auth-state), pure.
// ---------------------------------------------------------------------------

export type AuthPanel =
  | { name: "signed-out" }
  | { name: "unreadable" }
  | { name: "none"; passkeys: number }
  | { name: "pending"; passkeys: number; masked: string }
  | { name: "verified"; passkeys: number; masked: string };

/**
 * Order matters, and the last clause is the fail-closed law:
 *
 *   1. signed out / 401 → the gate. Nothing is claimed about any account.
 *   2. anything that is not a parsed 200 → UNREADABLE. "No backup email on
 *      this account" is a FACT and only an answered read may state it; an
 *      outage that rendered it would tell a fan their recovery address was
 *      gone, which is the exact shape of error-as-reassurance this repo bans.
 *   3. A 200 WHOSE BODY DOES NOT CARRY THE ANSWER → also UNREADABLE. The check
 *      used to be `typeof read.data === "object"`, and `typeof [] === "object"`
 *      — so `{}` and `[]` both fell through to the real-state branch and
 *      rendered two facts nothing had said: "No backup email on this account"
 *      (email absent) AND "You have no passkey on this account" (the count
 *      defaulting to 0). Both fields must be present and of the right shape, or
 *      the read answered nothing.
 *   4. a 200 that carries them → the real state, email null / pending / verified.
 */
export function authPanelState(read: { status: number; data: AuthStateDto | null }, signedIn: boolean): AuthPanel {
  if (!signedIn || read.status === 401) return { name: "signed-out" };
  if (read.status !== 200) return { name: "unreadable" };
  // `typeof [] === "object"` and `typeof {} === "object"`, so the old shape
  // check admitted both — and an empty body then rendered TWO facts nothing had
  // said: "No backup email on this account" and "You have no passkey on this
  // account" (the passkey count defaulted to 0). A body that carries neither
  // field answered nothing, so it is UNREADABLE, not an empty account.
  const body: unknown = read.data;
  if (!body || typeof body !== "object" || Array.isArray(body)) return { name: "unreadable" };
  const d = body as Partial<AuthStateDto>;
  if (typeof d.passkey_count !== "number" || !Number.isFinite(d.passkey_count)) return { name: "unreadable" };
  if (!("email" in d)) return { name: "unreadable" };
  // CLAMPED, like both natives. Swift's `Int(Double)` TRAPS out of range, so
  // iOS bounds the count at 9999 and Kotlin matches it; leaving the web
  // unbounded made a 100 000-passkey body render differently on three clients
  // that are otherwise one table. No account has ten thousand passkeys, and a
  // number that says it does is not a count worth rendering faithfully.
  const passkeys = Math.min(9999, Math.max(0, Math.trunc(d.passkey_count)));
  const email = d.email;
  if (email === null || email === undefined) return { name: "none", passkeys };
  if (typeof email !== "object" || Array.isArray(email) || typeof email.address_masked !== "string") return { name: "unreadable" };
  return { name: email.verified === true ? "verified" : "pending", passkeys, masked: email.address_masked };
}

/**
 * THE ADDRESS A CHANGE IS WAITING ON, or null.
 *
 * Deliberately its OWN function rather than a field on `AuthPanel`.
 * `authPanelState` is the decision table the three clients are held to
 * statement-for-statement (`backup-email-cross-client.test.ts` parses the Swift
 * and Kotlin mirrors into a closed grammar and evaluates them against this one),
 * and that table answers a different question: "may this screen state anything
 * about the account at all?". A pending address is additional detail INSIDE an
 * answered read, so it is read separately and under the same fail-closed law —
 * an unanswered read, a body that is not an object, an email that is not an
 * object, or a `pending_masked` that is not a non-empty string all mean "no
 * change in flight is known", never "there is one".
 */
export function pendingMasked(read: { status: number; data: AuthStateDto | null }): string | null {
  if (read.status !== 200) return null;
  const body: unknown = read.data;
  if (!body || typeof body !== "object" || Array.isArray(body)) return null;
  const email: unknown = (body as Partial<AuthStateDto>).email;
  if (!email || typeof email !== "object" || Array.isArray(email)) return null;
  const pending: unknown = (email as { pending_masked?: unknown }).pending_masked;
  return typeof pending === "string" && pending !== "" ? pending : null;
}

/**
 * The device list (`GET /account/fan-passkeys`), under the same fail-closed law
 * as `authPanelState`: only an ANSWERED read may state what this account holds.
 * A failed read renders "we couldn't read your devices", never an empty list —
 * "you have no other passkeys" is exactly the sentence that would talk a fan
 * out of removing a lost one.
 */
export type DevicePanel = { name: "unread" } | { name: "devices"; devices: FanPasskeyDto[] };

export function devicePanelState(read: { status: number; data: FanPasskeysDto | null }): DevicePanel {
  if (read.status !== 200) return { name: "unread" };
  const body: unknown = read.data;
  if (!body || typeof body !== "object" || Array.isArray(body)) return { name: "unread" };
  const rows = (body as { passkeys?: unknown }).passkeys;
  if (!Array.isArray(rows)) return { name: "unread" };
  const devices: FanPasskeyDto[] = [];
  for (const row of rows) {
    // Every field checked: a row missing its id is a row with no button, and a
    // row whose dates are not strings is a row that would render "Invalid Date"
    // at the fan while they decide which device to cut off.
    if (!row || typeof row !== "object" || Array.isArray(row)) return { name: "unread" };
    const r = row as Partial<FanPasskeyDto>;
    if (typeof r.credential_id !== "string" || r.credential_id === "") return { name: "unread" };
    if (typeof r.created_at !== "string") return { name: "unread" };
    if (!(r.last_used_at === null || typeof r.last_used_at === "string")) return { name: "unread" };
    devices.push({ credential_id: r.credential_id, created_at: r.created_at, last_used_at: r.last_used_at });
  }
  return { name: "devices", devices };
}

/**
 * The state pill for an ADDRESS on your own account — its own class, and
 * deliberately NOT the `.badge.v` / `.badge.cau` verdict pills.
 *
 * DESIGN-SYSTEM.md §4/§11: the verdict hues are reserved, always mean a
 * statement about a creator's identity, and may not be borrowed. "Your backup
 * email is confirmed" is account plumbing; painting it Verified teal would put
 * the vetted-creator colour on a mailbox. Icon + label carry the meaning here
 * exactly as they do there — the hue is informational (§3 `--info`) and muted,
 * neither of which is in the verdict table.
 */
function authState(kind: "ok" | "wait", icon: string, label: string): string {
  return `<span class="authstate ${xml(kind)}"><span class="ico">${xml(icon)}</span>${xml(label)}</span>`;
}

/** A date, as the fan reads it: the calendar day out of an ISO timestamp, or
 * nothing at all if the string is not one. A row that cannot be dated is shown
 * WITHOUT a date rather than with "Invalid Date" — this screen is read by
 * somebody deciding which device to cut off. */
function onDay(iso: string | null): string | null {
  if (typeof iso !== "string") return null;
  const ms = Date.parse(iso);
  if (Number.isNaN(ms)) return null;
  return new Date(ms).toISOString().slice(0, 10);
}

/**
 * ONE DEVICE, AND EVERY WORD OF IT FROM THE ROW. Added-when, last-used-when,
 * and a button. No name, no icon, no guessed platform: `fan_passkeys` holds
 * none of those, and a plausible-looking device label on this screen is a
 * fabrication a frightened person would act on.
 */
function deviceRow(d: FanPasskeyDto): string {
  const added = onDay(d.created_at);
  const used = onDay(d.last_used_at);
  const when = added ? `Added ${added}` : "Added — date unreadable";
  const last = d.last_used_at === null ? "never used to sign in" : used ? `last signed in ${used}` : "last sign-in date unreadable";
  // THE BUTTON IS ALWAYS OFFERED AND THE SERVER IS ALWAYS THE AUTHORITY. The
  // "would this strand the account?" rule is a JOIN across active passkeys and
  // a verified email that only the server can evaluate atomically; a client
  // that re-implemented it to grey a button out would eventually disagree with
  // it, and the direction that costs a fan is hiding the control in a state
  // where the removal is in fact allowed — a stolen device they cannot remove.
  // So the refusal lives in one place, and its 409 carries LAST_WAY_IN.
  return `<li class="device"><span class="dwhen">${xml(when)}, ${xml(last)}</span><button type="button" data-act="revoke-passkey" data-cred="${xml(d.credential_id)}" class="ds-btn act danger">Remove this passkey</button></li>`;
}

function passkeyBlock(count: number, devices: DevicePanel): string {
  const n = count === 1 ? "1 passkey" : `${count} passkeys`;
  const note =
    count === 0
      ? "You have no passkey on this account. Add one — it's the way in that can't be phished, mailed to the wrong person, or read out of an inbox."
      : count === 1
        ? "One passkey means one device. If that device goes, so does this way in — add a second one on another device, or keep a backup email below."
        : "More than one device can sign you in. That's the shape you want.";
  // FAIL CLOSED, and it matters more here than on the email block: an empty
  // list rendered from a read that did not answer would tell a fan looking for
  // a stolen device's passkey that there is nothing to remove.
  const list =
    devices.name === "unread"
      ? `<p class="hint ${UNREAD_CLASS}">We couldn't read the passkeys on this account, so nothing is listed here. That's a connection problem, not an answer — it does not mean there are none. Try again.</p>`
      : devices.devices.length === 0
        ? `<p class="empty">No passkeys on this account.</p>`
        : `<p class="hint">${xml(DEVICE_UNNAMED)}</p>
<ul class="devices">${devices.devices.map((d) => deviceRow(d)).join("")}</ul>
<p class="hint">${xml(REVOKE_COST)} Dhwaja will refuse to remove the last way into this account, so you cannot lock yourself out from here.</p>`;
  return `<div class="authrow">
<p class="eyebrow">How you sign in</p>
<p class="authnow"><b>${xml(n)}</b> on this account</p>
<p class="hint">${xml(note)}</p>
${list}
<div class="sacts"><button type="button" data-act="add-passkey" class="ds-btn act primary">Add a passkey</button></div>
<span data-passkey-msg class="rmsg" aria-live="polite"></span>
<div data-passkey-out></div>
</div>`;
}

/**
 * #814a — THE GOOGLE/APPLE SIGN-INS LINKED TO THIS ACCOUNT, or none.
 *
 * Read under the same fail-closed law as `pendingMasked`: an unanswered read, a
 * body that is not an object, or an `oauth_providers` that is not an array all
 * mean "no linked sign-in is known" — and the screen then offers no Remove, so
 * nothing is offered that the server has not said exists. Only the two
 * providers the server can unlink (fan-identity-unlink.ts UNLINKABLE_PROVIDERS)
 * pass, each once.
 */
export function linkedProviders(read: { status: number; data: AuthStateDto | null }): Array<"google" | "apple"> {
  if (read.status !== 200) return [];
  const body: unknown = read.data;
  if (!body || typeof body !== "object" || Array.isArray(body)) return [];
  const list = (body as { oauth_providers?: unknown }).oauth_providers;
  if (!Array.isArray(list)) return [];
  return (["google", "apple"] as const).filter((p) => list.includes(p));
}

/** #814a — what removing a linked sign-in costs, said before the confirm. */
export const UNLINK_COST = "Removing it means that Google or Apple account can no longer sign you in here. It also signs every other device out; you stay signed in on this one. Your passkeys and backup email are not touched.";

/** #814a — the answer to `DELETE /account/oauth/:provider`. A 409 is the
 * server refusing to remove the last way in, and is said as such. */
export function unlinkOutcome(r: Envelope & { resigned?: boolean }, provider: "google" | "apple"): Outcome {
  const name = provider === "google" ? "Google" : "Apple";
  if (r.ok && r.status === 200 && r.resigned === true) {
    return { kind: "oauth_unlinked", tone: "good", headline: `${name} sign-in removed`, body: `That ${name} account can't sign in here any more. Every other device was signed out; you're still signed in here.` };
  }
  if (r.status === 409) {
    return { kind: "last_way_in", tone: "quiet", headline: "That's your only way in", body: `Your ${name} sign-in is the only way into this account right now, so we won't remove it — that would lock you out. Add a passkey or confirm a backup email first, then remove it.` };
  }
  if (isNoSession(r)) return SIGNED_OUT;
  if (r.status === 404) return { kind: "no_such_signin", tone: "quiet", headline: `There's no ${name} sign-in on this account`, body: "Nothing was removed and nothing changed. Open Sign-in & recovery again to see what this account actually holds." };
  return { kind: "unlink_failed", tone: "quiet", headline: `We couldn't confirm the ${name} sign-in was removed`, body: "We didn't get an answer we could read, so this says nothing either way. Open Sign-in & recovery again and look at what the account actually holds." };
}

/**
 * Web app → browser-extension tier hand-off.
 *
 * The extension has no session, no `storage` permission and no host permission
 * for the API, so it cannot read billing itself — and it should not be able to.
 * This is the whole of the other half: when the SPA has ALREADY read the
 * entitlement for its own screens, it hands the extension the two fields its
 * one-line readout needs, over `chrome.runtime.sendMessage(id, …)`. The
 * extension's manifest pins the allowed origin with `externally_connectable`,
 * and its worker re-checks the browser-reported sender origin.
 *
 * Deliberately narrow:
 *   - PUSH ONLY. There is no request direction and no reply that matters; the
 *     extension can never ask the web app for anything.
 *   - TWO FIELDS. Not the entitlement — no dates, no customer flag, no source,
 *     no purchased tier. Just "which tier" and "is a payment retrying".
 *   - BEST EFFORT. No extension id configured, no `chrome.runtime` on the page
 *     (the extension isn't installed), or a throwing call → nothing happens.
 *     Nothing on either side depends on it, and nothing protective does.
 */

export interface TierMark {
  tier: "free" | "premium" | "pro";
  grace: boolean;
}

const TIERS: readonly string[] = ["free", "premium", "pro"];
/** Chrome extension ids are 32 characters drawn from a–p. */
const EXTENSION_ID_RE = /^[a-p]{32}$/;

/**
 * Reduce an already-read entitlement to the mark. `null` in — or an entitlement
 * whose tier we don't recognise — means "we don't know", and nothing is sent:
 * the extension keeps showing nothing rather than a guess. A tier of `free` IS
 * sent, because that is how a cancelled subscription clears a stale line.
 */
export function tierMarkOf(ent: { tier?: unknown; in_grace?: unknown } | null | undefined): TierMark | null {
  const t = ent?.tier;
  if (typeof t !== "string" || !TIERS.includes(t)) return null;
  return { tier: t as TierMark["tier"], grace: ent?.in_grace === true };
}

interface RuntimeLike {
  sendMessage(id: string, message: unknown, cb?: () => void): void;
  lastError?: unknown;
}

/** The page's `chrome.runtime`, present only on origins the extension's
 * `externally_connectable` lists — so its absence is the normal case. */
function runtimeOf(host: unknown): RuntimeLike | null {
  const rt = (host as { chrome?: { runtime?: unknown } } | undefined)?.chrome?.runtime as RuntimeLike | undefined;
  return rt && typeof rt.sendMessage === "function" ? rt : null;
}

/**
 * I3-0986 d — SIGN-OUT / ACCOUNT SWITCH clears the extension's line. Distinct from "unknown" (which sends nothing,
 * above): a signed-out page KNOWS there is no tier to show, so it tells the worker to forget the last account's
 * mark. The worker's `parseTierMark(null)` (apps/extension/src/tier.ts) already reads `mark: null` as "no mark".
 * Returns whether a message was dispatched.
 */
export function clearTierInExtension(extensionId: string, host: unknown = globalThis): boolean {
  return dispatch({ type: "tier", mark: null }, extensionId, host);
}

function dispatch(message: { type: "tier"; mark: TierMark | null }, extensionId: string, host: unknown): boolean {
  if (typeof extensionId !== "string" || !EXTENSION_ID_RE.test(extensionId)) return false;
  const rt = runtimeOf(host);
  if (rt === null) return false;
  try {
    // The callback exists only so Chrome doesn't log an unchecked lastError
    // when the extension isn't installed; the answer is never used.
    rt.sendMessage(extensionId, message, () => void rt.lastError);
    return true;
  } catch {
    return false; // extension uninstalled mid-session, or the API threw
  }
}

/**
 * Hand the mark over. Returns whether a message was actually dispatched, which
 * is what the tests assert on; callers ignore it.
 */
export function shareTierWithExtension(mark: TierMark | null, extensionId: string, host: unknown = globalThis): boolean {
  if (mark === null) return false;
  return dispatch({ type: "tier", mark }, extensionId, host);
}

      ? `<ul class="collectedlist">${items.map((it) => `<li data-collected="${xml(it.passport_id)}">${xml(it.display_name)}${warnedMark(it.warned)}</li>`).join("")}</ul>`
      : `<div class="empty">No collected cards yet. Press Collect on a creator's card to add it here.</div>`;
    // I3-1262 a (I22-L03 landing WEBB H3): api.ts collection() marks a list cut short by a failed page or the
    // page cap; the panel says so instead of presenting it as the whole collection.
    if (partial) body += `<p class="hint" role="note">Showing the first ${items.length} cards; the rest didn't load. Reload to try again.</p>`;
  }
  return `<section class="freecollect" data-free-collection-panel>
<h3>Collected</h3>
${body}
</section>`;
}

import { canonicalize, PLATFORMS } from "@dhwaja/canonical";
import { ACQ_RE, LOOKUP_HANDLE_RE, PSP_RE, type CheckIdKind, type Route } from "./router.js";

/**
 * PS-114: every host the canonicalizer's platform table recognizes as a
 * platform's OWN domain (onlyfans.com, instagram.com, x.com, twitter.com...).
 * A dotted token that IS one of these verbatim is not a creator handle on
 * some platform -- it IS the platform -- so it must never win the
 * dotted-handle race below. Built from `PLATFORMS` (already exported by
 * @dhwaja/canonical) rather than a hand-kept list, so a host added to the
 * table later is covered here for free.
 */
const KNOWN_PLATFORM_HOSTS = new Set(Object.values(PLATFORMS).flatMap((p) => p.hosts));

/**
 * Classify whatever a fan types into the search box into an action. The box is
 * the front door, so it is deliberately forgiving: a Passport id, a user
 * handle, an acquisition id, or an external profile URL to trust-check.
 *
 * A URL is NOT navigated to — it is only ever canonicalized locally and turned
 * into a trust check (no open-redirect: we never put a user-supplied URL into
 * location, an href, or a fetch target here).
 */

export type SearchAction =
  | { kind: "route"; route: Route }
  | { kind: "check"; platform: string; handle: string; canonical: string; idKind: CheckIdKind }
  | { kind: "unsupported"; reason: string };

export function classifyInput(raw: string): SearchAction {
  const input = typeof raw === "string" ? raw.trim() : "";
  if (input.length === 0) return { kind: "unsupported", reason: "Type a handle, a profile link, or a Dhwaja id." };
  if (input.length > 2048) return { kind: "unsupported", reason: "That's too long to be a handle or link." };

  // Direct Dhwaja ids first (unambiguous prefixes).
  if (PSP_RE.test(input)) return { kind: "route", route: { name: "passport", id: input } };
  if (ACQ_RE.test(input)) return { kind: "route", route: { name: "collectible", id: input } };

  // A leading @ or a bare word that's a valid handle → the DISAMBIGUATION
  // screen, never straight to a fan page. This used to route to the social
  // profile, which was a P0-class trust bug (external review): typing
  // "@rileymoon" into the front door of an authenticity checker landed on
  // whichever COLLECTOR registered that handle — a page an impersonator could
  // squat — instead of a creator check. The lookup screen puts the creator
  // check first and labels the collector profile as exactly what it is.
  //
  // ---------------------------------------------------------------------
  // THE FRONT DOOR ACCEPTS THE HANDLES THAT EXIST, not the ones Dhwaja mints.
  //
  // This test used to be `HANDLE_RE` — the DHWAJA COLLECTOR namespace,
  // `[a-z0-9_]{3,20}` — with a guard that also rejected any input containing a
  // dot. `router.ts` had already widened the lookup ROUTE to `LOOKUP_HANDLE_RE`
  // and written down why (Instagram, TikTok and YouTube handles legally carry
  // dots and hyphens and run to 30 characters), and this end never followed. So
  // `mr.beast` and `mr-beast` fell through to the URL branch, canonicalized to
  // nothing (a bare host with no path is `url|norm`), and were answered with
  // "That link isn't a supported creator profile." — a sentence about a link, at
  // somebody who typed a handle. The one place a legal handle must never be
  // bounced is the front door of the checker.
  //
  // AND THE ORDER HAD TO CHANGE WITH IT. Once dots are legal in a handle, a
  // dotted token with no path is ambiguous: `mr.beast` is a handle and
  // `acme.com` looks like a domain. The handle reading wins, because the domain
  // reading has nowhere to go — a bare host with no path canonicalizes to
  // `url|norm`, which this function refuses two lines below. Trying the URL
  // branch first would therefore turn every dotted handle into an error and
  // rescue no domain at all. A SCHEME, A SLASH or WHITESPACE still means "this
  // is a link, not a handle", and those are the only signals that do.
  // ---------------------------------------------------------------------
  // I3-2017: validate the RAW (un-folded) token's SHAPE first, case-insensitively,
  // and lowercase only once it has passed. Lowercasing before the shape check let
  // a character illegal in either case (U+212A KELVIN SIGN, which JS
  // `.toLowerCase()` maps to "k") sail through as whatever it folds to.
  const rawHandle = input.startsWith("@") ? input.slice(1) : input;
  const handleShapeOk = new RegExp(LOOKUP_HANDLE_RE.source, "i").test(rawHandle);
  const handle = rawHandle.toLowerCase();
  // PS-114: a bare known-platform domain ("onlyfans.com") looks exactly like a
  // legal dotted handle ("mr.beast") to LOOKUP_HANDLE_RE, but it isn't one --
  // it's the platform itself. Routing it to the handle-disambiguation screen
  // produces nonsense guidance ("Pick the platform you saw @onlyfans.com
  // on..."). Exclude an exact match against a known platform host so it falls
  // through to the URL branch below, which correctly reports it as an
  // unsupported bare profile link instead.
  if (!/[:/\s]/.test(input) && handleShapeOk && !KNOWN_PLATFORM_HOSTS.has(handle)) {
    return { kind: "route", route: { name: "lookup", handle } };
  }

  // Otherwise treat it as an external profile URL to trust-check. Canonicalize
  // LOCALLY (no navigation, no fetch of the URL) — the same canonicalizer the
  // server's registry uses, so a local hash matches a stored one.
  const url = /^[a-z][a-z0-9+.-]*:\/\//i.test(input) ? input : `https://${input}`;
  try {
    const subject = canonicalize(url);
    if (subject.platform === "url" || subject.idKind === "norm") {
      return { kind: "unsupported", reason: "That link isn't a supported creator profile." };
    }
    // The ID KIND rides along: a check result is a routable place now, and
    // rebuilding its subject from (platform, id) alone would have to guess the
    // kind — which hashes differently for a case-significant `pid` and for a
    // Tier-2 link's `url`. A guess would answer a different question from the
    // one the fan asked. A kind outside the three the route can carry is
    // refused rather than coerced.
    const idKind: CheckIdKind | null = subject.idKind === "handle" || subject.idKind === "pid" || subject.idKind === "url" ? subject.idKind : null;
    if (idKind === null) return { kind: "unsupported", reason: "That link isn't a supported creator profile." };
    return { kind: "check", platform: subject.platform, handle: subject.id, canonical: subject.canonical, idKind };
  } catch {
    return { kind: "unsupported", reason: "That doesn't look like a handle, profile link, or Dhwaja id." };
  }
}
