import { constants as zlibConstants, inflateSync } from "node:zlib";

const { Z_SYNC_FLUSH } = zlibConstants;
import { PNG } from "pngjs";
import jpeg from "jpeg-js";

/**
 * Safe decode → 9x8 luminance matrix, for the trusted embedder container.
 *
 * This is the FIRST thing untrusted bytes hit, so it is the security-critical
 * seam: a crafted file must never balloon memory or hang. Guards, in order:
 *   1. byte cap (caller also caps the HTTP read; this is defence in depth)
 *   2. format allowlist by MAGIC bytes (PNG/JPEG only) — never by extension
 *   3. header-declared pixel cap, checked BEFORE the decoder allocates
 *      (pngjs/jpeg-js have no pre-decode size gate; a few-hundred-byte file can
 *      declare 20000x20000 and force a multi-GB alloc / interlaced bomb)
 *
 * The pipeline (decode → Rec.601 grayscale → area-average box-resample to 9x8)
 * is bit-for-bit identical to apps/archive-agent/src/imagefp.ts; a parity test
 * pins them so a container-minted matrix is directly comparable to an
 * agent-minted one. dHash itself is NOT computed here — the Worker owns the
 * single canonical dHash (services/image/src/phash.ts); the container only ever
 * emits the luminance matrix.
 */

export const DHASH_COLS = 9;
export const DHASH_ROWS = 8;
export const MAX_INPUT_BYTES = 16 * 1024 * 1024; // 16 MB encoded upload
// Decoded-pixel cap. This is the REAL memory bound: a solid-colour image
// compresses to a few KB (passing the byte cap) but decodes to width*height*4
// bytes of RGBA — so a loose cap is a compression-amplified decode bomb. We
// only downsample to a 9x8 matrix and run a face model on a small crop, so
// there is no reason to accept more than ~16 MP; that bounds the worst-case
// RGBA allocation to ~64 MB. Operators handling larger images should raise this
// AND size the container's memory to match. (The archive-agent uses a laxer cap
// for a creator's own trusted local files — this path is untrusted uploads.)
export const MAX_PIXELS = 16 * 1024 * 1024; // 16 MP decoded (~64 MB RGBA)

/** A decodable-image guard was tripped by an oversize / decode-bomb input. */
export class OversizeImage extends Error {
  override name = "OversizeImage";
}

interface DecodedImage {
  width: number;
  height: number;
  rgba: Uint8Array;
}

/** PNG declared pixel count from the IHDR (bytes 16..24) WITHOUT decoding. */
function pngIhdrPixels(bytes: Uint8Array): number | null {
  if (bytes.length < 24 || bytes[12] !== 0x49 || bytes[13] !== 0x48 || bytes[14] !== 0x44 || bytes[15] !== 0x52) return null;
  const w = (bytes[16]! << 24) | (bytes[17]! << 16) | (bytes[18]! << 8) | bytes[19]!;
  const h = (bytes[20]! << 24) | (bytes[21]! << 16) | (bytes[22]! << 8) | bytes[23]!;
  if (w <= 0 || h <= 0) return null;
  return w * h; // may overflow huge/negative → caught by the > MAX_PIXELS gate
}

/** Channels per pixel for each PNG colour type, or null for an undefined one. */
function pngChannels(colourType: number): number | null {
  return colourType === 0 ? 1 : colourType === 2 ? 3 : colourType === 3 ? 1 : colourType === 4 ? 2 : colourType === 6 ? 4 : null;
}

/** The bit depths each PNG colour type is allowed to carry. Rejecting an illegal
 * pair (colour 3 at depth 16, say) BEFORE the inflate below matters: pngjs bails
 * on those in about a millisecond, and without this check the gate would do
 * hundreds of milliseconds of decompression on the way to the same answer. */
const PNG_LEGAL_DEPTHS: Readonly<Record<number, readonly number[]>> = {
  0: [1, 2, 4, 8, 16], // greyscale
  2: [8, 16], // truecolour
  3: [1, 2, 4, 8], // palette
  4: [8, 16], // greyscale + alpha
  6: [8, 16], // truecolour + alpha
};

// Adam7 pass geometry: seven passes, each with its own origin and stride.
const ADAM7_X_START = [0, 4, 0, 2, 0, 1, 0];
const ADAM7_Y_START = [0, 0, 4, 0, 2, 0, 1];
const ADAM7_X_STEP = [8, 8, 4, 4, 2, 2, 1];
const ADAM7_Y_STEP = [8, 8, 8, 4, 4, 2, 2];

/**
 * How many raw (post-inflate) bytes a complete image of this shape must carry:
 * one filter byte per scanline plus the packed samples, summed over every Adam7
 * pass when the image is interlaced.
 *
 * INTERLACED USED TO BE "CANNOT TELL" HERE, and that hole was a live
 * decompression bomb. The gate returned early for interlaced images, so the
 * bounded inflate never ran, and `pngjs`'s interlaced branch uses **stock
 * `zlib.inflateSync` with no output limit** (`parser-sync.js`) — a 1x1
 * interlaced PNG (one pixel: every pixel cap waves it through) carrying a 1 GB
 * payload measured 3.1 s and +1046 MB of RSS, from a 1019 KB file. On the
 * container, which accepts 16 MB uploads, one request bought ~16 GB. Found by
 * the F20 regression review, in the F20 fix, one round after the same class of
 * bug was fixed on the non-interlaced path. See MANIFEST-F20.md §5.
 *
 * Adam7 sizing is exactly computable, so there is no need to punt: pass p covers
 * `ceil((w - xStart) / xStep)` columns and `ceil((h - yStart) / yStep)` rows,
 * and a pass with no columns or no rows contributes nothing at all (which is
 * what happens for images narrower or shorter than 8 px).
 */
function pngRawBytesNeeded(w: number, h: number, channels: number, depth: number, interlace: number): number {
  const rowBytes = (px: number): number => 1 + Math.ceil((px * channels * depth) / 8);
  if (interlace === 0) return h * rowBytes(w);
  let total = 0;
  for (let p = 0; p < 7; p++) {
    const passW = w > ADAM7_X_START[p]! ? Math.ceil((w - ADAM7_X_START[p]!) / ADAM7_X_STEP[p]!) : 0;
    const passH = h > ADAM7_Y_START[p]! ? Math.ceil((h - ADAM7_Y_START[p]!) / ADAM7_Y_STEP[p]!) : 0;
    if (passW === 0 || passH === 0) continue;
    total += passH * rowBytes(passW);
  }
  return total;
}

/**
 * IS THE IMAGE DATA ACTUALLY ALL THERE? — the precondition `pngjs` does not check.
 *
 * `PNG.sync.read` on a PNG whose chunk CRCs are valid but whose DEFLATE STREAM
 * IS SHORT reads past the end of the inflated data into an uninitialised buffer.
 * Measured by the archive agent's fuzz pass: the SAME bytes, in the same
 * process, microseconds apart, decoded successfully about 63% of the time and
 * threw `Unrecognised filter type - N` the rest — with a DIFFERENT N almost
 * every time (9, 8, 255, 222, 112, 60, 136, 58, 226 …). That varying N is
 * uninitialised memory being read as a filter byte; when it happens to land on
 * 0–4 the decode "succeeds" and the pixel rows are whatever was in that memory.
 *
 * ON THIS PATH the consequences are worse than on a creator's laptop, because
 * this is the container that eats UNTRUSTED UPLOADS:
 *
 *   1. IT IS A MEMORY ORACLE. The 9x8 luminance matrix this module returns is
 *      derived from the decoded pixels and travels back to the Worker. Upload
 *      one short-deflate PNG repeatedly and the matrices differ run to run —
 *      72 bytes of this container's heap per request, read by the uploader.
 *      The container also holds the shared-secret auth material for its own
 *      endpoint.
 *   2. THE FINGERPRINT STOPS BEING A FUNCTION OF THE FILE. The whole cascade
 *      downstream — near-duplicate matching, revocation, the "have we seen this
 *      image" question — assumes the same bytes hash the same way. They did not.
 *
 * We cannot fix `pngjs`, so the seam checks what it lacks: concatenate the IDAT
 * payloads, inflate, and require exactly the bytes a complete image needs
 * (`pngRawBytesNeeded` — Adam7 included). A short stream is CORRUPT, not
 * oversize — null, 422.
 *
 * Three things this function is careful about, each of which was got wrong once:
 *
 *   1. THE INFLATE IS BOUNDED, at `needed`. An unbounded one made this gate a
 *      worse bomb than the decoder it protects — at 16 MB uploads and a ~1000:1
 *      IDAT ratio, ~16 GB of allocation per request. Hitting the cap answers the
 *      question by itself.
 *   2. MORE data than needed is REFUSED, not accepted. pngjs refuses those
 *      itself, so it costs nothing — and saying "complete" is what hands a file
 *      on to pngjs's *interlaced* inflate, which is stock `zlib.inflateSync`
 *      with no bound at all. That was the same bomb, a second time.
 *   3. `Z_SYNC_FLUSH`, because pngjs uses it. A stream that carries every pixel
 *      but never terminates tidily is not an incomplete image, and refusing
 *      those turned working uploads into 422s for one review round.
 *
 * The IHDR is validated before any of that: an undefined colour type, bit depth,
 * colour/depth pair, compression method, filter method or interlace method is
 * refused without decompressing anything, because pngjs rejects those for free
 * and the inflate is the expensive part of this function.
 *
 * PINNED: this logic is duplicated in apps/archive-agent/src/imagefp.ts and the
 * parity test asserts both modules answer identically. Change one, change both.
 */
function pngDataIsComplete(bytes: Uint8Array): boolean {
  if (bytes.length < 33) return false;
  const depth = bytes[24]!;
  const colour = bytes[25]!;
  const compression = bytes[26]!;
  const filter = bytes[27]!;
  const interlace = bytes[28]!;
  // Reject what the spec does not define BEFORE doing any decompression: pngjs
  // refuses these in about a millisecond, and the inflate below is the expensive
  // part of this function.
  if (compression !== 0 || filter !== 0 || interlace > 1) return false;
  const channels = pngChannels(colour);
  if (channels === null || !PNG_LEGAL_DEPTHS[colour]!.includes(depth)) return false;
  const w = ((bytes[16]! << 24) | (bytes[17]! << 16) | (bytes[18]! << 8) | bytes[19]!) >>> 0;
  const h = ((bytes[20]! << 24) | (bytes[21]! << 16) | (bytes[22]! << 8) | bytes[23]!) >>> 0;
  if (w <= 0 || h <= 0) return false;
  if (w * h > MAX_PIXELS) return false; // keeps `needed` (and the inflate cap below) bounded standalone

  // Concatenate every IDAT payload, in order, bounded by the file itself.
  //
  // ME-02 (S-068). THE SAME WALK NOW COUNTS IHDR CHUNKS, because the bytes that
  // ADMIT this file must be the bytes that DECODE it.
  //
  // Everything above reads the header off a FIXED OFFSET — bytes 12..28, the
  // first chunk after the signature — and computes the pixel cap and `needed`
  // from it. `pngjs@7.0.0` does not agree that that header is authoritative:
  // `lib/parser.js:73-82` checks only that the FIRST chunk is IHDR, and
  // `_parseIHDR` (`:119-178`) is dispatched again for EVERY later IHDR chunk,
  // calling `this.metadata(...)` again each time. `lib/parser-sync.js:28-29`
  // keeps the LAST one, and `:82-88` derives the non-interlaced inflate's
  // `chunkSize`/`maxLength` from `metaData.width * bpp * depth` and
  // `metaData.height` — the REPLACEMENT header.
  //
  // Measured on this tree before this line existed: a 4x4 PNG carrying a second
  // IHDR declaring 8000x8000 passed the declared-pixel cap and this
  // completeness gate as a 4x4, and `PNG.sync.read` then returned an 8000x8000
  // image — 64 MP, ~256 MB of RGBA allocated — with the pixel cap firing only
  // AFTERWARDS, from the post-decode check. The cap that exists to run before
  // the allocation ran after it.
  //
  // A spec-valid PNG has exactly one IHDR and it is the first chunk, so
  // requiring that costs nothing and closes the disagreement at its root: there
  // is one header, the one this function and the caps above already read.
  let ihdrCount = 0;
  const parts: Uint8Array[] = [];
  let off = 8;
  let total = 0;
  while (off + 12 <= bytes.length) {
    const len = ((bytes[off]! << 24) | (bytes[off + 1]! << 16) | (bytes[off + 2]! << 8) | bytes[off + 3]!) >>> 0;
    const dataStart = off + 8;
    const dataEnd = dataStart + len;
    if (len > bytes.length || dataEnd + 4 > bytes.length) return false; // chunk table lies
    if (bytes[off + 4] === 0x49 && bytes[off + 5] === 0x48 && bytes[off + 6] === 0x44 && bytes[off + 7] === 0x52) {
      // ME-02: a second IHDR anywhere makes "the header" ambiguous, and the
      // decoder resolves that ambiguity differently from this gate. Refused —
      // and refused HERE, before `PNG.sync.read` is ever called.
      if (++ihdrCount > 1 || off !== 8) return false;
    }
    if (bytes[off + 4] === 0x49 && bytes[off + 5] === 0x44 && bytes[off + 6] === 0x41 && bytes[off + 7] === 0x54) {
      parts.push(bytes.subarray(dataStart, dataEnd));
      total += len;
      if (total > MAX_INPUT_BYTES) return false;
    }
    off = dataEnd + 4;
  }
  if (ihdrCount !== 1) return false; // exactly one, and it was the first chunk
  if (parts.length === 0) return false;

  const idat = new Uint8Array(total);
  let at = 0;
  for (const p of parts) {
    idat.set(p, at);
    at += p.length;
  }
  // BOUND THE INFLATE, or this gate is itself the bomb. `pngjs`'s NON-interlaced
  // inflate is bounded (it caps output at the image size), so an unbounded
  // `inflateSync` here would make a hardened path strictly more bombable than the
  // one it replaced: a 1x1 PNG with a 1029:1 IDAT measured 1 GB of RSS and 1.9 s,
  // against a 7 ms throw from pngjs alone.
  //
  // The only question this function asks is "at least `needed` bytes?", so the
  // inflate is capped AT `needed` and the cap being hit is itself the answer.
  // `needed` is bounded by the pixel cap re-checked above.
  //
  // `Z_SYNC_FLUSH` because a stream that carries every pixel but never cleanly
  // terminates — no adler32, or a flush an embedded encoder never finished — is
  // NOT an incomplete image, and pngjs's own inflate accepts it. Measured: three
  // such files decoded correctly and reproducibly before this gate existed and
  // were refused by the first version of it, which conflated "the zlib stream
  // did not end tidily" with "the pixels are missing". A genuinely short stream
  // still produces fewer than `needed` bytes and is still refused.
  const needed = pngRawBytesNeeded(w, h, channels, depth, interlace);
  if (needed <= 0) return false;
  try {
    return inflateSync(Buffer.from(idat), { maxOutputLength: needed, finishFlush: Z_SYNC_FLUSH }).length >= needed;
  } catch (e) {
    // MORE output than the image needs. `pngjs` refuses those itself — its
    // reader throws on leftover data as well as on a shortfall — so refusing
    // here costs nothing and buys the thing that actually matters: the inflate
    // never runs past `needed` bytes, on EITHER interlace setting. That is what
    // closes the interlaced bomb, because the gate saying "complete" is what
    // used to hand the file on to pngjs's own UNBOUNDED interlaced inflate.
    // (Both arms answer `false`; they are spelled separately because they are
    // different facts — "too much data" and "not decompressible at all" — and a
    // future change may want to tell them apart.)
    if ((e as NodeJS.ErrnoException).code === "ERR_BUFFER_TOO_LARGE") return false;
    return false; // a stream that will not inflate at all is one pngjs will read past
  }
}

/**
 * I3-0265: EXIF orientation. `jpeg-js` decodes the raw scan data and never
 * reads the EXIF APP1 orientation tag, so a pre-rotated phone-camera JPEG
 * (the common case: every browser/OS viewer applies the tag at display time)
 * decodes here upright-on-disk but rotated relative to what a human — and
 * every downstream matcher (dHash, the face embedder) — actually sees. That
 * silently defeats identity matching for any image whose EXIF orientation
 * isn't 1.
 *
 * This reads ONLY the orientation tag (APP1/Exif IFD0, tag 0x0112) by walking
 * the marker segments (JPEG) or scanning the first uncompressed pixel-adjacent
 * ancillary chunk isn't a PNG concept — PNG has no EXIF-orientation reader
 * here because pngjs's callers overwhelmingly ship orientation 1 (screenshots,
 * generated images) and no code path here claims to normalize PNG EXIF; only
 * JPEG orientation is read and applied. Returns 1 (identity) on any parse
 * failure or absent tag — fail OPEN to "no rotation", never throw: a
 * malformed/absent EXIF block must not turn a decodable image into a 422.
 */
function jpegExifOrientation(bytes: Uint8Array): number {
  // JPEG: after the SOI (FFD8), a sequence of marker segments. APP1 is FFE1.
  let off = 2;
  while (off + 4 <= bytes.length) {
    if (bytes[off] !== 0xff) break; // not a marker: stop, no orientation found
    const marker = bytes[off + 1]!;
    if (marker === 0xd8 || marker === 0xd9) {
      off += 2;
      continue;
    }
    // SOS (FFDA) starts entropy-coded scan data: no more markers to read.
    if (marker === 0xda) break;
    const segLen = (bytes[off + 2]! << 8) | bytes[off + 3]!;
    if (segLen < 2 || off + 2 + segLen > bytes.length) break;
    if (marker === 0xe1) {
      const segStart = off + 4;
      // "Exif" + two NUL bytes header (6 bytes) precedes the TIFF structure.
      if (
        bytes[segStart] === 0x45 &&
        bytes[segStart + 1] === 0x78 &&
        bytes[segStart + 2] === 0x69 &&
        bytes[segStart + 3] === 0x66 &&
        bytes[segStart + 4] === 0x00 &&
        bytes[segStart + 5] === 0x00
      ) {
        const tiff = segStart + 6;
        if (tiff + 8 > bytes.length) return 1;
        const little = bytes[tiff] === 0x49 && bytes[tiff + 1] === 0x49; // "II"
        const big = bytes[tiff] === 0x4d && bytes[tiff + 1] === 0x4d; // "MM"
        if (!little && !big) return 1;
        const u16 = (p: number): number => (little ? bytes[p]! | (bytes[p + 1]! << 8) : (bytes[p]! << 8) | bytes[p + 1]!);
        const u32 = (p: number): number => (little ? (bytes[p]! | (bytes[p + 1]! << 8) | (bytes[p + 2]! << 16) | (bytes[p + 3]! << 24)) >>> 0 : ((bytes[p]! << 24) | (bytes[p + 1]! << 16) | (bytes[p + 2]! << 8) | bytes[p + 3]!) >>> 0);
        const ifd0 = tiff + u32(tiff + 4);
        if (ifd0 + 2 > bytes.length) return 1;
        const count = u16(ifd0);
        for (let i = 0; i < count; i++) {
          const entry = ifd0 + 2 + i * 12;
          if (entry + 12 > bytes.length) break;
          const tag = u16(entry);
          if (tag === 0x0112) {
            const value = u16(entry + 8);
            return value >= 1 && value <= 8 ? value : 1;
          }
        }
        // The Exif APP1 itself carried no orientation tag: identity, and stop —
        // a second Exif APP1 is not a thing a real file has.
        return 1;
      }
      // W3-03 FIX2 item 3: a NON-Exif APP1 is not the end of the walk. This
      // `return 1` was unconditional, and an XMP APP1 (`http://ns.adobe.com/
      // xap/1.0/\0`) ahead of the Exif APP1 is the ORDINARY Adobe/phone JPEG
      // layout — so for those files orientation was never read at all and
      // I3-0265 survived. Fall through to the segment advance below and keep
      // walking to the APP1 whose payload begins "Exif\0\0".
    }
    off += 2 + segLen;
  }
  return 1;
}

/**
 * Apply EXIF orientation (1..8) to an already-decoded RGBA image, producing
 * pixels in DISPLAY orientation. Orientation 1 is a no-op (returns the same
 * object). The full 8-case table (mirror + rotate combinations) per the EXIF
 * spec; only rotation (5..8 involve a transpose+flip, 3/4/6/8 are used by
 * real cameras — but every case is implemented, not just the common ones, so
 * an unusual writer doesn't silently fall through to "no rotation").
 */
function applyExifOrientation(img: DecodedImage, orientation: number): DecodedImage {
  if (orientation === 1) return img;
  const { width: w, height: h, rgba } = img;
  const swapped = orientation >= 5 && orientation <= 8;
  const outW = swapped ? h : w;
  const outH = swapped ? w : h;
  const out = new Uint8Array(outW * outH * 4);
  for (let y = 0; y < h; y++) {
    for (let x = 0; x < w; x++) {
      let dx: number;
      let dy: number;
      switch (orientation) {
        case 2: // mirror horizontal
          dx = w - 1 - x;
          dy = y;
          break;
        case 3: // rotate 180
          dx = w - 1 - x;
          dy = h - 1 - y;
          break;
        case 4: // mirror vertical
          dx = x;
          dy = h - 1 - y;
          break;
        case 5: // mirror horizontal + rotate 270 CW
          dx = y;
          dy = x;
          break;
        case 6: // rotate 90 CW
          dx = h - 1 - y;
          dy = x;
          break;
        case 7: // mirror horizontal + rotate 90 CW
          dx = h - 1 - y;
          dy = w - 1 - x;
          break;
        case 8: // rotate 270 CW
          dx = y;
          dy = w - 1 - x;
          break;
        default:
          dx = x;
          dy = y;
      }
      const si = (y * w + x) * 4;
      const di = (dy * outW + dx) * 4;
      out[di] = rgba[si]!;
      out[di + 1] = rgba[si + 1]!;
      out[di + 2] = rgba[si + 2]!;
      out[di + 3] = rgba[si + 3]!;
    }
  }
  return { width: outW, height: outH, rgba: out };
}

/**
 * Decode PNG or JPEG (magic-sniffed). Returns null for unsupported/corrupt
 * input (caller → 422). Throws OversizeImage for a pixel-bomb (caller → 413) so
 * the two conditions stay distinguishable.
 */
function decodeImage(bytes: Uint8Array): DecodedImage | null {
  if (bytes.length === 0) return null;
  if (bytes.length > MAX_INPUT_BYTES) throw new OversizeImage(`input exceeds ${MAX_INPUT_BYTES} bytes`);
  // PNG magic: 89 50 4E 47
  if (bytes[0] === 0x89 && bytes[1] === 0x50 && bytes[2] === 0x4e && bytes[3] === 0x47) {
    const declared = pngIhdrPixels(bytes);
    if (declared === null) return null; // not a real IHDR → corrupt
    if (declared > MAX_PIXELS || declared < 0) throw new OversizeImage("PNG dimensions exceed pixel cap");
    // The precondition pngjs does not check — see `pngDataIsComplete`. A PNG
    // that does not carry its own pixels is corrupt (422), not oversize.
    // The precondition pngjs does not check — see `pngDataIsComplete`. A PNG
    // that does not carry its own pixels is corrupt (422), not oversize. The
    // gate answers for interlaced images too, which is what bounds THEIR
    // inflate: pngjs's interlaced branch uses stock zlib with no output limit.
    if (!pngDataIsComplete(bytes)) return null;
    let png: { width: number; height: number; data: Buffer };
    try {
      png = PNG.sync.read(Buffer.from(bytes));
    } catch {
      return null; // corrupt PNG body
    }
    if (png.width * png.height > MAX_PIXELS) throw new OversizeImage("PNG dimensions exceed pixel cap");
    return { width: png.width, height: png.height, rgba: new Uint8Array(png.data) };
  }
  // JPEG magic: FF D8
  if (bytes[0] === 0xff && bytes[1] === 0xd8) {
    let img: { width: number; height: number; data: Uint8Array };
    try {
      img = jpeg.decode(Buffer.from(bytes), { useTArray: true, maxMemoryUsageInMB: 256, maxResolutionInMP: 16 });
    } catch (e) {
      // jpeg-js throws on both corrupt data AND its own resolution/memory caps;
      // treat a cap message as oversize, anything else as corrupt.
      const msg = e instanceof Error ? e.message : "";
      if (/memory|resolution|maxResolution|maxMemory/i.test(msg)) throw new OversizeImage(`JPEG too large: ${msg}`);
      return null;
    }
    if (img.width * img.height > MAX_PIXELS) throw new OversizeImage("JPEG dimensions exceed pixel cap");
    const decoded: DecodedImage = { width: img.width, height: img.height, rgba: new Uint8Array(img.data) };
    // I3-0265: normalize to DISPLAY orientation before anything downstream
    // (grayscale, box-resample, the face embedder) ever sees the pixels.
    return applyExifOrientation(decoded, jpegExifOrientation(bytes));
  }
  return null; // unknown format
}

/** Rec.601 luminance of RGBA alpha-composited over opaque white (I3-0836). */
function toGrayscale(img: DecodedImage): Uint8Array {
  const out = new Uint8Array(img.width * img.height);
  for (let i = 0; i < out.length; i++) {
    const o = i * 4;
    const y = 0.299 * img.rgba[o]! + 0.587 * img.rgba[o + 1]! + 0.114 * img.rgba[o + 2]!;
    const a = img.rgba[o + 3]!;
    // I3-0836: composite over opaque white BEFORE hashing — the matrix is of the
    // image a viewer sees, so alpha-only content is structure and RGB under
    // alpha 0 is not. Luminance is linear, so compositing it equals compositing
    // each channel. Opaque pixels take the untouched path.
    out[i] = a === 255 ? Math.round(y) : Math.round((y * a + 255 * (255 - a)) / 255);
  }
  return out;
}

/** Area-average (box) resample of a grayscale image to targetW×targetH. */
function boxResize(gray: Uint8Array, w: number, h: number, targetW: number, targetH: number): Uint8Array {
  if (w < 1 || h < 1 || gray.length !== w * h) throw new Error("boxResize: bad source dimensions");
  const out = new Uint8Array(targetW * targetH);
  for (let ty = 0; ty < targetH; ty++) {
    const y0 = Math.floor((ty * h) / targetH);
    const y1 = Math.max(y0 + 1, Math.floor(((ty + 1) * h) / targetH));
    for (let tx = 0; tx < targetW; tx++) {
      const x0 = Math.floor((tx * w) / targetW);
      const x1 = Math.max(x0 + 1, Math.floor(((tx + 1) * w) / targetW));
      let sum = 0;
      for (let y = y0; y < y1; y++) {
        for (let x = x0; x < x1; x++) sum += gray[y * w + x]!;
      }
      out[ty * targetW + tx] = Math.round(sum / ((y1 - y0) * (x1 - x0)));
    }
  }
  return out;
}

/**
 * Full decode → 9x8 luminance matrix. Returns null when the format isn't
 * decodable (→ the caller mints an exact-hash-only fingerprint); throws
 * OversizeImage on a decode-bomb. The returned matrix is exactly what
 * services/image/src/phash.ts dHashes and what the near-duplicate stage
 * compares against.
 */
export function gray9x8FromBytes(bytes: Uint8Array): Uint8Array | null {
  const img = decodeImage(bytes);
  if (!img) return null;
  const gray = toGrayscale(img);
  return boxResize(gray, img.width, img.height, DHASH_COLS, DHASH_ROWS);
}

/**
 * D-0835(1) transition source: the PRE-I3-0836 matrix (alpha dropped, RGB
 * luminance only) for an image with at least one non-opaque pixel; null for an
 * opaque image (its matrix never changed) or an undecodable one. A match check
 * compares a stored phash minted before the fix against this, beside the
 * current `gray9x8FromBytes`. Retire once enrollments are re-fingerprinted.
 */
export function legacyGray9x8FromBytes(bytes: Uint8Array): Uint8Array | null {
  const img = decodeImage(bytes);
  if (!img) return null;
  const n = img.width * img.height;
  let translucent = false;
  for (let i = 0; i < n && !translucent; i++) translucent = img.rgba[i * 4 + 3]! !== 255;
  if (!translucent) return null;
  const gray = new Uint8Array(n);
  for (let i = 0; i < n; i++) {
    const o = i * 4;
    gray[i] = Math.round(0.299 * img.rgba[o]! + 0.587 * img.rgba[o + 1]! + 0.114 * img.rgba[o + 2]!);
  }
  return boxResize(gray, img.width, img.height, DHASH_COLS, DHASH_ROWS);
}

import http from "node:http";

/**
 * Container entrypoint. Cloudflare Containers deliver requests to the port the
 * process listens on; PORT is provided by the runtime (default 8080). Today
 * (see services/image/wrangler.jsonc) this process is NOT actually run inside
 * a Cloudflare Container — there is no Durable Object class to attach one to
 * — it is run as a plain Docker/HTTPS origin (option A in that file's
 * comment). That deployment shape (a bare container host: Docker, ECS,
 * Kubernetes, a VM behind systemd) sends SIGTERM on redeploy/restart/scale-
 * down, and Node's default SIGTERM behavior with no handler installed is an
 * immediate process exit — in-flight `/fingerprint` requests get an RST
 * instead of a response, and the caller (the image Worker, PS-094's own
 * bounded caller) sees a hard failure instead of a clean 5xx it could retry.
 * PS-082: install a handler so a normal stop drains in-flight requests first.
 */

/** PS-082: how long to wait for in-flight requests to drain before forcing
 * the process to exit. Bounded so a wedged connection cannot block a
 * deploy/restart forever. */
export const DEFAULT_SHUTDOWN_GRACE_MS = 10_000;

export interface GracefulShutdownOptions {
  /** Milliseconds to wait for server.close() to finish before forcing exit. */
  forceExitMs?: number;
  /** Injectable for tests — never call process.exit from a test process. */
  exit?: (code: number) => void;
  /** Injectable for tests. */
  log?: (msg: string) => void;
  /** Injectable for tests — avoid depending on the real `process` object. */
  signalSource?: NodeJS.EventEmitter;
}

/**
 * PS-082: on SIGTERM/SIGINT, stop accepting new connections and let in-flight
 * ones finish, then exit — instead of the default "no handler" behavior,
 * which drops in-flight requests immediately. A second signal, or the grace
 * period elapsing, forces the exit so a stuck socket can't block a restart.
 */
export function installGracefulShutdown(server: http.Server, options: GracefulShutdownOptions = {}): void {
  const forceExitMs = options.forceExitMs ?? DEFAULT_SHUTDOWN_GRACE_MS;
  const exit = options.exit ?? ((code: number) => process.exit(code));
  const log = options.log ?? ((msg: string) => console.log(msg));
  const signalSource: NodeJS.EventEmitter = options.signalSource ?? process;

  let shuttingDown = false;
  const handle = (signal: string) => {
    if (shuttingDown) {
      // Second signal: stop waiting, exit now.
      log(JSON.stringify({ at: "image-embedder", event: "shutdown_forced", signal }));
      exit(1);
      return;
    }
    shuttingDown = true;
    log(JSON.stringify({ at: "image-embedder", event: "shutdown_start", signal, grace_ms: forceExitMs }));
    const forceTimer = setTimeout(() => {
      log(JSON.stringify({ at: "image-embedder", event: "shutdown_grace_exceeded", signal }));
      exit(1);
    }, forceExitMs);
    forceTimer.unref?.();
    server.close((err) => {
      clearTimeout(forceTimer);
      log(JSON.stringify({ at: "image-embedder", event: "shutdown_complete", signal, error: err ? String(err) : null }));
      exit(err ? 1 : 0);
    });
  };

  signalSource.on("SIGTERM", () => handle("SIGTERM"));
  signalSource.on("SIGINT", () => handle("SIGINT"));
}

/** Build and start the server from process.env, with graceful shutdown wired
 * up. Exported (rather than only run as a side effect) so tests can exercise
 * it without spawning a subprocess. */
export function main(): http.Server {
  const port = Number(process.env.PORT ?? "8080") || 8080;
  const server = serverFromEnv(process.env);
  installGracefulShutdown(server);
  server.listen(port, () => {
    // One line, no request data ever logged.
    console.log(`image-embedder listening on :${port}`);
  });
  return server;
}

// Only run as a side effect when executed directly (`node dist/main.js`),
// never on import — this is what lets main.test.ts import
// `installGracefulShutdown` and `main` without requiring a real
// EMBEDDER_SHARED_SECRET / binding a real port as an import side effect.
if (import.meta.url === `file://${process.argv[1]}`) {
  main();
}

/**
 * The face embedding step — the only part of fingerprinting that needs a model.
 *
 * TWO paths:
 *   - REAL: an ONNX face-embedding model loaded from FACE_MODEL_PATH via
 *     onnxruntime-node (an optionalDependency; native, not bundled). This is the
 *     production path and is verified at deploy time — we cannot ship or run GPU
 *     weights in CI.
 *   - STUB: a deterministic, clearly-labelled embedding derived from the
 *     luminance matrix, so the whole pipeline runs end to end headless. It is
 *     NOT a biometric matcher — it must never be used in production, and
 *     resolveEmbedder() REFUSES it unless EMBEDDER_ALLOW_STUB is explicitly set
 *     (never set it in prod).
 *
 * A degenerate image (flat / near-constant) is reported as no-face: there is
 * nothing to enroll or search, and a degenerate vector would match everything.
 */

export const EMBED_DIM = 128;
const STDDEV_MIN = 1.0; // matches @dhwaja/media STRUCTURE_MIN_STDDEV

export interface Embedding {
  vector: number[]; // L2-normalized; [] when faceFound is false
  faceFound: boolean;
  /** I3-0831: faces the detector saw. The Worker enrols only a count of exactly 1;
   * the real model supplies it (the stub reports faceFound ? 1 : 0 and detects nothing). */
  faceCount: number;
}

export interface FaceEmbedder {
  /** I-15: server.ts's fingerprintBytes() copies this straight onto the
   * `/fingerprint` response as `model`, so a caller that checks can tell a
   * real face embedding from the stub without a separate /health poll. Keep
   * this the single source of truth for "which embedder produced this
   * vector" — never re-derive it from env at the call site. */
  readonly kind: "onnx" | "stub";
  embed(gray9x8: Uint8Array, rgbaOrBytes: Uint8Array): Promise<Embedding>;
}

function stddev(v: Uint8Array): number {
  if (v.length === 0) return 0;
  let mean = 0;
  for (const x of v) mean += x;
  mean /= v.length;
  let acc = 0;
  for (const x of v) acc += (x - mean) * (x - mean);
  return Math.sqrt(acc / v.length);
}

function l2normalize(v: number[]): number[] {
  let norm = 0;
  for (const x of v) norm += x * x;
  norm = Math.sqrt(norm);
  if (!(norm > 0)) return v.map(() => 0);
  return v.map((x) => x / norm);
}

/**
 * Deterministic stub embedding. Expands the 72-value luminance matrix into an
 * EMBED_DIM vector through a FIXED (seeded, code-constant) projection, then
 * L2-normalizes. Same image → same vector; different images → different
 * vectors. Good enough to exercise the cascade; useless as a real face matcher.
 */
function stubEmbedding(gray9x8: Uint8Array): number[] {
  const out = new Array<number>(EMBED_DIM).fill(0);
  // A deterministic, code-fixed projection: each output dim is a weighted sum of
  // the inputs with weights from a small LCG seeded per-dim. No RNG at runtime.
  for (let d = 0; d < EMBED_DIM; d++) {
    let seed = (d * 2654435761) >>> 0; // Knuth multiplicative hash → per-dim seed
    let acc = 0;
    for (let i = 0; i < gray9x8.length; i++) {
      seed = (seed * 1664525 + 1013904223) >>> 0; // LCG
      const w = seed / 0xffffffff - 0.5; // deterministic weight in [-0.5, 0.5)
      acc += w * (gray9x8[i]! / 255 - 0.5);
    }
    out[d] = acc;
  }
  return l2normalize(out);
}

class StubEmbedder implements FaceEmbedder {
  readonly kind = "stub" as const;
  async embed(gray9x8: Uint8Array): Promise<Embedding> {
    if (stddev(gray9x8) < STDDEV_MIN) return { vector: [], faceFound: false, faceCount: 0 };
    return { vector: stubEmbedding(gray9x8), faceFound: true, faceCount: 1 };
  }
}

/**
 * ONNX-backed embedder. Lazily requires onnxruntime-node so the container base
 * image doesn't need the native module unless a model is actually configured.
 * The exact pre-processing (crop/align/normalize to the model's input tensor)
 * is model-specific and wired here at deploy time; this class is the seam.
 */
class OnnxEmbedder implements FaceEmbedder {
  readonly kind = "onnx" as const;
  constructor(private readonly modelPath: string) {}
  async embed(gray9x8: Uint8Array, _bytes: Uint8Array): Promise<Embedding> {
    if (stddev(gray9x8) < STDDEV_MIN) return { vector: [], faceFound: false, faceCount: 0 };
    // Deploy-time: load the session once (cache), detect+align the face, build
    // the input tensor, run inference, L2-normalize the output embedding, and
    // set faceFound from the detector. Kept behind the optional dependency so
    // CI (which has no weights) never imports the native module.
    const ort = (await import("onnxruntime-node")) as unknown as { InferenceSession: { create(p: string): Promise<unknown> } };
    void ort; // referenced so the import isn't dead-code-eliminated
    throw new Error(`ONNX embedding not wired for model at ${this.modelPath}: supply the model-specific preprocessing + inference at deploy time`);
  }
}

/**
 * Choose the embedder from the environment. Production (or any run without
 * EMBEDDER_ALLOW_STUB) MUST have a real FACE_MODEL_PATH; the stub is refused so
 * a misconfigured deploy fails loudly instead of silently serving fake
 * embeddings that would corrupt the identity index.
 */
/**
 * Is the ONNX inference path actually wired?
 *
 * OnnxEmbedder.embed() throws unconditionally (see :103) — the class selects the
 * model file and stops short of the model-specific preprocessing + inference.
 * This is the ONE constant whoever supplies that work flips to `true`. Nothing
 * else in the tree decides it.
 *
 * serverFromEnv() reads this and REFUSES TO START when FACE_MODEL_PATH is set
 * and this is false (D-0567, 2026-09-19). resolveEmbedder's return is unchanged:
 * a configured ONNX deploy still resolves to `kind: "onnx"`, it just does not
 * get to boot and 500 every /fingerprint.
 */
export function onnxWired(): boolean {
  return false;
}

export function resolveEmbedder(env: Record<string, string | undefined>): FaceEmbedder {
  const modelPath = env.FACE_MODEL_PATH;
  if (modelPath) return new OnnxEmbedder(modelPath);
  const allowStub = env.EMBEDDER_ALLOW_STUB === "1" || env.EMBEDDER_ALLOW_STUB === "true";
  const isProd = env.NODE_ENV === "production";
  if (!allowStub || isProd) {
    throw new Error("no FACE_MODEL_PATH configured and the deterministic stub is refused (set EMBEDDER_ALLOW_STUB=1 only in dev/CI, never in production)");
  }
  return new StubEmbedder();
}

import { createHash, timingSafeEqual } from "node:crypto";
import http from "node:http";
import { gray9x8FromBytes, legacyGray9x8FromBytes, OversizeImage, MAX_INPUT_BYTES } from "./decode.js";
import { resolveEmbedder, onnxWired, type FaceEmbedder } from "./model.js";

/**
 * The trusted fingerprint server. It is the machine that mints authoritative
 * fingerprints from real bytes (this is what stops a client from fabricating
 * authorship — see F1). It NEVER writes bytes to disk or logs, and holds them
 * only for the duration of the request.
 *
 * Contract:  POST /fingerprint   body = raw image bytes
 *   200 { exact_hash, gray9x8:[72], vector:[]|[EMBED_DIM], face_found, face_count, model: "stub"|"onnx" }
 *   413 too large / decode-bomb     422 undecodable (not PNG/JPEG or corrupt)
 *   400 empty body      401 bad/absent shared secret
 *   503 no usable shared secret configured (fail closed — see below)
 *   GET /health → 200
 */

/**
 * Minimum length of `EMBEDDER_SHARED_SECRET`. Same floor as the sibling
 * `INTERNAL_TRIAGE_SECRET` gate in `services/image/src/http.ts`, and for the
 * same reason: a one-character "secret" is a secret an attacker guesses, and a
 * shell that expanded an unset variable to "" must not read as "configured".
 */
export const MIN_SHARED_SECRET_LEN = 16;

/**
 * PS-094: this process decodes on the Node main thread — pngjs/jpeg-js/zlib
 * are all synchronous, so one in-flight `/fingerprint` blocks the event loop
 * for the whole request. With no admission control, concurrent uploads pile
 * up unbounded: each holds up to MAX_INPUT_BYTES in memory AND queues its
 * full CPU-bound decode behind every other one, so tail latency grows with
 * concurrency with no signal to the caller that it should back off.
 * Bounding in-flight requests and answering 503 over the cap is the same
 * fail-closed shape already used for `not_configured` — reject loudly with a
 * distinct error rather than let the queue grow silently. Reversible (env
 * override), cheap (one counter), and does not touch the decode path itself.
 */
export const DEFAULT_MAX_CONCURRENT_REQUESTS = 4;

/** Is the configured secret usable at all? Absent, empty or too short → no,
 * and every door this gates then refuses. */
export function secretConfigured(expected: string | undefined): expected is string {
  return typeof expected === "string" && expected.length >= MIN_SHARED_SECRET_LEN && expected.trim() !== "" && !/TODO/i.test(expected);
}

/** I60-CONFIG C2: why secretConfigured refuses `expected` (its own four tests, in its order), or null when it accepts it.
 * The reason never carries anything derived from the value. */
export function secretRefusal(expected: string | undefined): string | null {
  if (typeof expected !== "string" || expected.length === 0) return "missing";
  if (expected.length < MIN_SHARED_SECRET_LEN) return `shorter than ${MIN_SHARED_SECRET_LEN} characters`;
  if (expected.trim() === "") return "blank (whitespace only)";
  if (/TODO/i.test(expected)) return "contains TODO (in any case)";
  return null;
}

export class UndecodableImage extends Error {
  override name = "UndecodableImage";
}

export interface RawFingerprint {
  exact_hash: string;
  gray9x8: number[];
  /** D-0835 (I3-0836 transition): the pre-fix alpha-DROPPED matrix, sent only
   * for an image with an alpha channel so the Worker can match enrollments
   * fingerprinted before the composite fix. Absent for opaque images. */
  gray9x8_legacy?: number[];
  vector: number[];
  face_found: boolean;
  /** I3-0831: the embedder's face count, copied verbatim (the Worker enrols only 1). */
  face_count: number;
  /** I-15: which embedder minted this vector ("stub" | "onnx"), so a caller
   * that bothers to look can tell a real face embedding from the
   * deterministic stub without polling /health out-of-band. Sourced from
   * FaceEmbedder.kind — the SAME value resolveEmbedder() already gates in
   * production — so tagging costs nothing new to trust. */
  model: "stub" | "onnx";
}

/** sha256 of the raw uploaded bytes. This is a CONTENT hash of an image file,
 * NOT subject-identity hashing — deliberately separate from
 * packages/canonical's subject hasher. This file is the sole hashing site in
 * the isolated embedder container and carries a narrow, pinned exemption in the
 * single-hasher invariant (see tests/invariants/single-hasher.test.ts). */
export function sha256HexOfBytes(bytes: Uint8Array): string {
  return createHash("sha256").update(bytes).digest("hex");
}

/**
 * Mint a fingerprint from raw bytes. Throws OversizeImage (→413) on a
 * decode-bomb and UndecodableImage (→422) when the bytes aren't a decodable
 * PNG/JPEG. The exact hash is sha256 of the bytes VERBATIM (byte-exact
 * re-upload = definitive match).
 */
export async function fingerprintBytes(bytes: Uint8Array, embedder: FaceEmbedder): Promise<RawFingerprint> {
  const gray = gray9x8FromBytes(bytes); // throws OversizeImage; null = undecodable
  if (!gray) throw new UndecodableImage("not a decodable PNG or JPEG");
  const exact_hash = sha256HexOfBytes(bytes);
  const { vector, faceFound, faceCount } = await embedder.embed(gray, bytes);
  const legacy = legacyGray9x8FromBytes(bytes); // D-0835: null for an opaque image
  return { exact_hash, gray9x8: Array.from(gray), ...(legacy ? { gray9x8_legacy: Array.from(legacy) } : {}), vector, face_found: faceFound, face_count: faceCount, model: embedder.kind };
}

/**
 * Constant-time shared-secret check. NO CONFIG → CLOSED.
 *
 * This used to `return true` when the secret was unset, justified as
 * "service-binding reachability is the boundary". That justification assumed a
 * shape this repo cannot deploy: `services/image/wrangler.jsonc` documents at
 * length that the Cloudflare Container binding needs a Durable Object class
 * that does not exist here, and that no fronting Worker exists either — so the
 * ONLY supported way to run this process today is (A): a public HTTPS origin
 * with `EMBEDDER_URL` + `EMBEDDER_SHARED_SECRET`. On a public origin
 * reachability is not a boundary; it is the internet.
 *
 * The failure it produced was silent and asymmetric. An operator who set the
 * secret on the Worker but not in the container's env published an
 * unauthenticated image-decode + face-embedding endpoint — the most
 * CPU-expensive operation in the product — behind one startup log line. The
 * mirror-image mistake (container has it, Worker doesn't) failed loudly with a
 * 401. Fail-closed means the quiet direction is the one that must stop.
 */
function secretOk(provided: string | undefined, expected: string | undefined): boolean {
  if (!secretConfigured(expected)) return false; // absent/short config never authorizes
  if (!provided) return false;
  // Hash both sides to a fixed 32 bytes before comparing, so the compare is
  // constant-time AND does not leak the secret's length via an early return.
  const a = createHash("sha256").update(provided).digest();
  const b = createHash("sha256").update(expected).digest();
  return timingSafeEqual(a, b);
}

/** Read the request body with a hard cap; resolves null if the cap is exceeded
 * (the socket is destroyed) so a large upload can't exhaust memory. */
function readCapped(req: http.IncomingMessage, cap: number): Promise<Uint8Array | null> {
  return new Promise((resolve) => {
    const chunks: Buffer[] = [];
    let total = 0;
    let done = false;
    const finish = (v: Uint8Array | null) => {
      if (!done) {
        done = true;
        resolve(v);
      }
    };
    req.on("data", (c: Buffer) => {
      total += c.length;
      if (total > cap) {
        req.destroy();
        finish(null);
        return;
      }
      chunks.push(c);
    });
    req.on("end", () => finish(new Uint8Array(Buffer.concat(chunks))));
    req.on("error", () => finish(null));
  });
}

export interface ServerOptions {
  embedder: FaceEmbedder;
  sharedSecret?: string | undefined;
  /** PS-094: cap on concurrent /fingerprint bodies being read+decoded. Over
   * the cap, the server answers 503 { error: "overloaded" } BEFORE reading
   * the body, rather than letting requests queue unbounded on the CPU-bound
   * decode path. Defaults to DEFAULT_MAX_CONCURRENT_REQUESTS. */
  maxConcurrentRequests?: number | undefined;
}

function json(res: http.ServerResponse, status: number, body: unknown): void {
  const s = JSON.stringify(body);
  res.writeHead(status, { "content-type": "application/json" });
  res.end(s);
}

export function createServer(opts: ServerOptions): http.Server {
  const maxConcurrent = opts.maxConcurrentRequests ?? DEFAULT_MAX_CONCURRENT_REQUESTS;
  let inFlight = 0;
  return http.createServer((req, res) => {
    // PS-091: minimal per-request telemetry — method, path, status, duration.
    // NEVER the body, headers, or secret (see the file-level "never logs"
    // contract): this is enough to tell a 401 storm (attack) apart from a
    // 500 spike (breakage) from the outside, which nothing here did before.
    const startedAt = Date.now();
    res.on("finish", () => {
      // I3-1757: log the PATH ONLY -- `req.url` is the raw request target
      // including the query string, and the query is caller-chosen, unbounded
      // content (length-bound only by the HTTP parser). Stripping it keeps
      // the file-level "never body/headers/secret" logging contract from
      // being defeated via the URL instead.
      const pathOnly = (req.url ?? "").split("?", 1)[0];
      console.log(
        JSON.stringify({
          at: "image-embedder",
          method: req.method,
          path: pathOnly,
          status: res.statusCode,
          duration_ms: Date.now() - startedAt,
        }),
      );
    });
    void (async () => {
      const isFingerprint = req.method === "POST" && req.url === "/fingerprint";
      if (isFingerprint) {
        // PS-094: bounded admission, checked before the body is read at all —
        // an attacker (or a slow client) cannot use queued-but-unread bodies
        // to grow memory past the cap either.
        if (inFlight >= maxConcurrent) return json(res, 503, { error: "overloaded" });
        inFlight++;
      }
      try {
        // `secret_configured` is on the liveness probe deliberately: the
        // misconfiguration this guards against is invisible from the outside,
        // and `curl /health` is the one thing an operator actually runs. It
        // discloses nothing an attacker can use — the door it describes is shut.
        // I11-FX2 (I11-FACEGUARD WO1): `wired` is true only for a real model with
        // inference wired (`onnxWired`), so a caller reading `model: "onnx"` does not
        // take a configured-but-unwired embedder for a working one.
        if (req.method === "GET" && req.url === "/health") return json(res, 200, { ok: true, service: "image-embedder", model: opts.embedder.kind, wired: opts.embedder.kind === "onnx" && onnxWired(), secret_configured: secretConfigured(opts.sharedSecret) });
        if (req.method !== "POST" || req.url !== "/fingerprint") return json(res, 404, { error: "not_found" });
        // Refuse BEFORE the auth compare, and say which failure it is: an
        // operator who gets 401 goes hunting for a bad header, while
        // `not_configured` names the actual mistake. Same shape as the sibling
        // internal seam in services/image/src/http.ts.
        if (!secretConfigured(opts.sharedSecret)) return json(res, 503, { error: "not_configured" });
        if (!secretOk(req.headers["x-embedder-secret"] as string | undefined, opts.sharedSecret)) return json(res, 401, { error: "unauthorized" });

        const bytes = await readCapped(req, MAX_INPUT_BYTES);
        if (bytes === null) return json(res, 413, { error: "too_large" });
        if (bytes.length === 0) return json(res, 400, { error: "empty_body" });

        const fp = await fingerprintBytes(bytes, opts.embedder);
        return json(res, 200, fp);
      } catch (e) {
        if (e instanceof OversizeImage) return json(res, 413, { error: "too_large" });
        if (e instanceof UndecodableImage) return json(res, 422, { error: "undecodable" });
        // Never leak internals (e.g. a model path) to the caller.
        return json(res, 500, { error: "internal" });
      } finally {
        if (isFingerprint) inFlight--;
      }
    })();
  });
}

/**
 * Vector-search entry-gate benchmark (the plan's hard gate for shipping
 * Find-the-Creator). It measures, on a synthetic corpus, whether an APPROXIMATE
 * nearest-neighbour index recovers the same identity a brute-force EXACT search
 * would — i.e. recall-vs-exact — plus a cost proxy (candidates scanned) and a
 * latency proxy. Ship the image search only if the ANN clears the recall floor
 * at scale; otherwise swap the engine behind the VectorStore interface (the
 * cascade's top-K + score-floor + live-row cross-check is unchanged either way).
 *
 * This module is engine-agnostic and pure: it models the property the real
 * Vectorize (or any ANN) must satisfy, so the same harness runs at 20k in CI
 * and at 10M from a script. No Math.random (seedable LCG) so runs reproduce.
 *
 * I-10: recall alone cannot fail an arbitrary floor — every query here was
 * historically a near-duplicate of an ENROLLED vector, so the harness could
 * not observe a false positive no matter what `floor` was set to. It now also
 * runs independent, out-of-corpus "impostor" queries and reports the fraction
 * that falsely clear the floor (`falsePositiveRate`), so a floor that accepts
 * strangers fails this gate. This does NOT calibrate the floor against a real
 * model's score distribution — that calibration is I-22, still open.
 */

export type Vec = Float64Array;

/** Deterministic PRNG (mulberry32) — reproducible corpora across runs/machines. */
export function rng(seed: number): () => number {
  let a = seed >>> 0;
  return () => {
    a |= 0;
    a = (a + 0x6d2b79f5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

/** A random unit vector (Gaussian-ish via sum of uniforms, then normalized). */
export function randomUnit(dim: number, r: () => number): Vec {
  const v = new Float64Array(dim);
  let norm = 0;
  for (let i = 0; i < dim; i++) {
    // Irwin–Hall(12) − 6 ≈ N(0,1)
    let s = 0;
    for (let k = 0; k < 12; k++) s += r();
    v[i] = s - 6;
    norm += v[i]! * v[i]!;
  }
  norm = Math.sqrt(norm) || 1;
  for (let i = 0; i < dim; i++) v[i]! /= norm;
  return v;
}

/** A near-duplicate of `base`: base + small noise, renormalized. Models a
 * re-encoded/cropped photo of the same person (high cosine to the enrolled). */
export function nearDuplicate(base: Vec, noise: number, r: () => number): Vec {
  const dim = base.length;
  const v = new Float64Array(dim);
  let norm = 0;
  for (let i = 0; i < dim; i++) {
    v[i] = base[i]! + noise * (r() * 2 - 1);
    norm += v[i]! * v[i]!;
  }
  norm = Math.sqrt(norm) || 1;
  for (let i = 0; i < dim; i++) v[i]! /= norm;
  return v;
}

export function cosine(a: Vec, b: Vec): number {
  let d = 0;
  for (let i = 0; i < a.length; i++) d += a[i]! * b[i]!;
  return d; // both unit vectors → dot = cosine
}

export interface Hit {
  id: number;
  score: number;
}

/** Brute-force exact top-K — the ground truth the ANN is graded against. */
export function exactTopK(corpus: Vec[], q: Vec, k: number): Hit[] {
  const hits: Hit[] = [];
  for (let i = 0; i < corpus.length; i++) hits.push({ id: i, score: cosine(corpus[i]!, q) });
  hits.sort((x, y) => y.score - x.score);
  return hits.slice(0, k);
}

/**
 * A small IVF (inverted-file) ANN: assign each vector to its nearest of
 * `nlist` random centroids; a query scans only the `nprobe` nearest lists. This
 * is the class of index a production ANN uses; here it stands in so the harness
 * measures recall-vs-exact and the candidate-scan cost the same way the real
 * engine would be measured.
 */
export class IvfIndex {
  private centroids: Vec[] = [];
  private lists: number[][] = [];
  constructor(private corpus: Vec[], private nlist: number, seed = 1) {
    const r = rng(seed);
    // Random centroids sampled from the corpus.
    const dim = corpus[0]?.length ?? 0;
    for (let c = 0; c < nlist; c++) {
      this.centroids.push(corpus.length > 0 ? corpus[Math.floor(r() * corpus.length)]! : new Float64Array(dim));
      this.lists.push([]);
    }
    for (let i = 0; i < corpus.length; i++) this.lists[this.nearestCentroid(corpus[i]!)]!.push(i);
  }

  private nearestCentroid(v: Vec): number {
    let best = 0;
    let bestScore = -Infinity;
    for (let c = 0; c < this.centroids.length; c++) {
      const s = cosine(this.centroids[c]!, v);
      if (s > bestScore) {
        bestScore = s;
        best = c;
      }
    }
    return best;
  }

  /** Returns top-K hits AND the number of candidates scanned (cost proxy). */
  query(q: Vec, k: number, nprobe: number): { hits: Hit[]; scanned: number } {
    // Rank centroids by proximity, probe the nearest nprobe lists.
    const ranked = this.centroids
      .map((c, idx) => ({ idx, s: cosine(c, q) }))
      .sort((a, b) => b.s - a.s)
      .slice(0, nprobe);
    const hits: Hit[] = [];
    let scanned = 0;
    for (const { idx } of ranked) {
      for (const i of this.lists[idx]!) {
        hits.push({ id: i, score: cosine(this.corpus[i]!, q) });
        scanned++;
      }
    }
    hits.sort((x, y) => y.score - x.score);
    return { hits: hits.slice(0, k), scanned };
  }
}

export interface BenchConfig {
  n: number; // corpus size
  dim: number; // embedding dimension
  queries: number; // number of near-duplicate queries
  noise: number; // near-duplicate perturbation
  k: number; // top-K
  nlist: number; // IVF lists
  nprobe: number; // lists probed per query
  floor: number; // cosine identity floor (mirrors EMBED_MIN_SCORE)
  seed: number;
}

export interface BenchResult {
  n: number;
  dim: number;
  queries: number;
  /** RJb-T2-BQ-01: fraction of queries where the ANN's RANK-1 hit is the true
   * enrolled id AND clears the floor (`cascadeWouldMatch`). The cascade acts on
   * the first match only, so a true id at rank 2..K is a MISS here — counting it
   * anywhere in top-K overstated what the real search delivers. */
  recall: number;
  /** Fraction where EXACT rank-1 would match (methodology check; ~1.0). */
  exactRecall: number;
  /** Mean candidates scanned per ANN query (cost proxy). */
  meanScanned: number;
  /** Mean brute-force comparisons for exact (cost baseline = n). */
  exactComparisons: number;
  /** I-10: fraction of INDEPENDENT, out-of-corpus impostor queries whose ANN
   * top-K falsely clears `floor` against some enrolled vector — the false-accept
   * rate the recall-only harness could not previously observe. */
  falsePositiveRate: number;
}

/** RJb-T2-BQ-01: what the cascade delivers for one query — its RANK-1 hit is
 * the true id and clears the floor. Hits are in descending score order (the
 * shape `exactTopK` and `IvfIndex.query` return). */
export function cascadeWouldMatch(hits: ReadonlyArray<Hit>, truthId: number, floor: number): boolean {
  const top = hits[0];
  return top !== undefined && top.id === truthId && top.score >= floor;
}

export function runBenchmark(cfg: BenchConfig): BenchResult {
  const r = rng(cfg.seed);
  const corpus: Vec[] = [];
  for (let i = 0; i < cfg.n; i++) corpus.push(randomUnit(cfg.dim, r));
  const index = new IvfIndex(corpus, cfg.nlist, cfg.seed);

  let annMatched = 0;
  let exactMatched = 0;
  let scannedTotal = 0;
  for (let query = 0; query < cfg.queries; query++) {
    const truthId = Math.floor(r() * cfg.n);
    const q = nearDuplicate(corpus[truthId]!, cfg.noise, r);

    const exact = exactTopK(corpus, q, cfg.k);
    if (cascadeWouldMatch(exact, truthId, cfg.floor)) exactMatched++;

    const { hits, scanned } = index.query(q, cfg.k, cfg.nprobe);
    scannedTotal += scanned;
    if (cascadeWouldMatch(hits, truthId, cfg.floor)) annMatched++;
  }

  // I-10: impostor queries are INDEPENDENT random unit vectors, never derived
  // from any corpus member, so a false accept here is a genuine identity
  // collision at the configured floor, not an artifact of the near-duplicate
  // construction above.
  let falsePositives = 0;
  for (let query = 0; query < cfg.queries; query++) {
    const impostor = randomUnit(cfg.dim, r);
    const { hits } = index.query(impostor, cfg.k, cfg.nprobe);
    if (hits.some((h) => h.score >= cfg.floor)) falsePositives++;
  }

  return {
    n: cfg.n,
    dim: cfg.dim,
    queries: cfg.queries,
    recall: annMatched / cfg.queries,
    exactRecall: exactMatched / cfg.queries,
    meanScanned: scannedTotal / cfg.queries,
    exactComparisons: cfg.n,
    falsePositiveRate: falsePositives / cfg.queries,
  };
}

/**
 * Find-the-Creator cascade. TWO things decide identity, and only two:
 *
 *   1. EXACT hash — sha256 of the normalized image. A byte-exact re-upload of
 *      an enrolled image is a definitive match (cheap, no GPU).
 *   2. EMBEDDING — a face/image embedding whose cosine similarity to an enrolled
 *      vector clears a high floor, AND whose returned image maps to a LIVE
 *      enrollment row. This is the only fuzzy identity signal.
 *
 * Perceptual hash is deliberately NOT a terminal matcher: dHash is coarse (two
 * flat/low-detail images collide within a small Hamming distance), so asserting
 * biometric IDENTITY on phash alone would misidentify people. phash is retained
 * only as a future COST optimization to pre-narrow embedding candidates — and
 * any such candidate must still be confirmed by the embedding + live-row check.
 *
 * PRIVACY: the query is EPHEMERAL — nothing about it (bytes, exact hash, phash,
 * query embedding) is written or logged. Only the match RESULT and the quota
 * counter persist.
 *
 * TRUST: the VectorStore is not trusted to assert identity. Its returned
 * passport_id/score are cross-checked: a match requires score ≥ floor AND the
 * returned image_id to resolve to a live enrollment row (whose passport_id is
 * authoritative). So a stale vector (deleted/withdrawn enrollment) or a buggy
 * store returning high scores cannot forge an identity.
 */

 * correct order, and must stay that way: the SQL check is what stops a stale
 * vector from disclosing a withdrawn identity, and an index-level filter can
 * never replace it.

 * revokeConsents` then kept the enrollment ROWS and the vectors, inert), so a

 * withdrawal that still leaves both behind is one whose vectors were not

 * consent revoked, until a retry erases them (`vectors_pending`, D-0570,

 * occupy every returned position, and a qualifying LIVE enrollment ranked
 * behind them was never even looked at: a deterministic false "no match",
 * reachable by nothing more hostile than a creator withdrawing old photos and
 * enrolling a new one. Both the fan-facing search and the S66 internal triage

 *

 * one query with a larger k, not a scan and not a loop — the resource bound is
 * `EMBED_TOPK * VECTOR_OVERFETCH` candidates, full stop. Widening this buys
 * recall past more retained entries and costs a larger single response;

/**
 * ===== THE SUPPRESSION ORACLE (D-0527) =====================================
 *
 * WHY THIS EXISTS. This cascade decided identity from ONE source of consent —
 * `IMAGE_DB.consents.revoked` — and nothing else. trust-core's suppression
 * sources (`request_suppressions`: a subject opted out of discovery;
 * `passports.retracted`: the passport itself was erased) live in a SEPARATE
 * D1 with no join path and, until this change, no seam to ask at all
 * (`services/image/wrangler.jsonc` bound no TRUST_DB and no suppression
 * channel — the `TRUST_CORE` service binding it already holds was reached
 * only by `revocation.ts`). A subject who asked to be delisted, or whose
 * passport was retracted, stayed fully matchable through image search: the
 * exact guarantee `dropSuppressed` (trust-core's rankings.ts) makes for every
 * discovery board, just never extended here.
 *
 * THE SEAM. `POST /internal/suppression-check` on trust-core, reached through
 * the SAME two shapes `revocation.ts`'s `SessionOracle` already resolves the
 * seam from (`env.TRUST_CORE` a Fetcher, else `env.TRUST_CORE_URL` an https
 * origin) — one convention in this service, not two. The machine credential
 * is derived from the image key (`IMAGE_SCOPE_KEY`, I3-1350), under a
 * DOMAIN-SEPARATED audience (`SUPPRESSION_CHECK_AUDIENCE` below, distinct
 * from `SESSION_CHECK_AUDIENCE`) so a credential minted for one internal seam
 * can never be replayed against the other.
 *
 * FAIL DIRECTION — DECIDED, NOT INHERITED BY ACCIDENT. `revocation.ts`'s
 * stance is: an unverifiable session is refused, never treated as live. The
 * mirror for suppression is NOT "refuse the whole request" (this is a search,
 * not a revocable session — trust-core being unreachable is not a reason to
 * 503 every /find call system-wide) but IS the same fail-closed direction
 * applied to what this oracle actually gates: `check()` returns `null` on
 * anything short of a well-formed 200, and `findIdentity` treats null as
 * "withhold every candidate this call was about to surface" — never "assume
 * not suppressed" and never "serve the match, log a warning". A suppression
 * that stops working when a network call fails is not a suppression; the
 * candidate is dropped, `findIdentity` moves on as if nothing matched, and
 * the fan sees an ordinary "no match" — never a distinguishable error, which
 * would itself be a signal.
 *
 * WHAT THIS ORACLE DISCLOSES, DELIBERATELY BOUNDED. `check()` takes only
 * passport ids THIS cascade already produced as a live-enrollment candidate
 * (a store hit that already cleared the score floor AND the live-consent
 * row) — never an arbitrary caller-supplied id, never a sweep, never a
 * cursor. The wire request is a small bounded batch (trust-core enforces
 * `SUPPRESSION_CHECK_MAX_BATCH`) and the wire response is checked to name
 * only ids THIS call asked about (`unknownAnswer` below) — an id trust-core
 * names back that was never in `toQuery` makes the whole answer untrusted
 * (`null`), the same shape-checking `revocation.ts` applies to `live`/`tier`.
 * This is what keeps the seam from becoming a general "is passport X
 * suppressed" probe: it only ever answers about ids the cascade already
 * matched by face or exact hash, which is not a question an outside caller
 * can pose through this interface at all.
 *
 * THE CACHE IS SECONDS, NOT MINUTES — same bound and same reasoning as
 * `SessionOracle`: the maximum time a delisted subject stays matchable here
 * after trust-core suppressed them. Failures are never cached, for the same
 * reason: an outage must not pin a subject into either state for the TTL.
 */

// I25-INT (i13-arch-cycles): SuppressionOracle lives in owners.ts, the leaf both enroll.ts and owners.ts read it from,
// so neither imports this module back (cascade -> enroll/owners -> cascade was a cycle). Re-exported for every caller.
export type { SuppressionOracle } from "./owners.js";

/** Minimal service-binding shape, same convention as revocation.ts. */
interface SuppressionFetcher {
  fetch(input: string, init?: RequestInit): Promise<Response>;
}

/** How long one "not suppressed" or "suppressed" answer is reused, per id.
 * Seconds, deliberately — see the header. */
export const SUPPRESSION_CHECK_TTL_MS = 5_000;
/** A hung trust-core must not hold a find request open. */
const SUPPRESSION_CHECK_TIMEOUT_MS = 2_000;
/** Bounds the per-isolate cache, same eviction shape as SessionOracle. */
const SUPPRESSION_CACHE_MAX = 1_000;
/** cascade never asks about more than one candidate per findIdentity step
 * today (see the loop in findIdentity below); this is slack for a future
 * batched caller, not a size the current code exercises. */
const SUPPRESSION_CHECK_MAX_BATCH = 16;
// SUPPRESSION_CHECK_AUDIENCE is imported from @dhwaja/session above, beside
// SESSION_CHECK_AUDIENCE, rather than duplicated here. The duplicate this
// replaces was the W308-A-02 break: two bare literals in two workers under
// comments saying they MUST match, with nothing comparing them. Drift was
// invisible to every test in the tree and a silent total outage in production.

interface SuppressionCacheEntry {
  expiresAtMs: number;
  suppressed: boolean;
}

/**
 * Build the per-isolate suppression-oracle factory. Same shape as
 * `makeSessionOracleFactory` in revocation.ts: the returned function resolves
 * the transport from a Worker env per request, and answers null when the
 * seam is not configured.
 */
export function makeSuppressionOracleFactory(opts?: { nowMs?: () => number }): (env: Record<string, unknown>) => SuppressionOracle | null {
  const nowMs = opts?.nowMs ?? (() => Date.now());
  const cache = new Map<string, SuppressionCacheEntry>();

  return (env: Record<string, unknown>): SuppressionOracle | null => {
    const svc = env.TRUST_CORE as SuppressionFetcher | undefined;
    const url = typeof env.TRUST_CORE_URL === "string" && /^https:\/\//i.test(env.TRUST_CORE_URL) ? env.TRUST_CORE_URL : "";
    if (!svc && !url) return null; // no channel to the suppression source → fail closed
    const imageKey = typeof env.IMAGE_SCOPE_KEY === "string" ? env.IMAGE_SCOPE_KEY : undefined; // I3-1350: never SESSION_SECRET

    return {
      async check(passportIds: readonly string[]): Promise<ReadonlySet<string> | null> {
        const distinct = [...new Set(passportIds)];
        if (distinct.length === 0) return new Set();
        if (distinct.length > SUPPRESSION_CHECK_MAX_BATCH) return null; // never truncate silently — refuse the whole batch
        const now = nowMs();
        const result = new Set<string>();
        const toQuery: string[] = [];
        for (const id of distinct) {
          const hit = cache.get(id);
          if (hit && hit.expiresAtMs > now) {
            if (hit.suppressed) result.add(id);
          } else {
            toQuery.push(id);
          }
        }
        if (toQuery.length === 0) return result;
        const credential = await imageSeamCredential(imageKey, SUPPRESSION_CHECK_AUDIENCE);
        if (!credential) return null; // unusable image key → seam unavailable, never unauthenticated
        let res: Response;
        try {
          const doFetch = svc ? svc.fetch.bind(svc) : fetch;
          const target = svc ? "https://trust-core.internal/internal/suppression-check" : `${url.replace(/\/$/, "")}/internal/suppression-check`;
          res = await doFetch(target, {
            method: "POST",
            headers: { "content-type": "application/json", "X-Internal-Auth": credential },
            body: JSON.stringify({ passport_ids: toQuery }),
            redirect: "manual", // #611 (D-0930(3)): a 3xx is a non-200 refusal below, never followed
            signal: AbortSignal.timeout(SUPPRESSION_CHECK_TIMEOUT_MS),
          });
        } catch {
          return null; // unreachable / timed out / DNS — refuse, never admit
        }
        if (res.status !== 200) return null; // 401 (credential drift), 503 (unconfigured), 400, 5xx — all refuse
        let body: unknown;
        try {
          body = await res.json();
        } catch {
          return null;
        }
        if (typeof body !== "object" || body === null || Array.isArray(body)) return null;
        const b = body as { suppressed?: unknown };
        if (!Array.isArray(b.suppressed) || !b.suppressed.every((x) => typeof x === "string")) return null;
        const answeredSuppressed = new Set(b.suppressed as string[]);
        // Closed-world check: an id trust-core names that we did not ask
        // about is a malformed or hostile response, not a fact about a
        // subject — the whole batch is refused rather than partially trusted.
        for (const id of answeredSuppressed) if (!toQuery.includes(id)) return null;
        for (const id of toQuery) {
          const isSuppressed = answeredSuppressed.has(id);
          if (isSuppressed) result.add(id);
          if (cache.size >= SUPPRESSION_CACHE_MAX) {
            const oldest = cache.keys().next();
            if (!oldest.done) cache.delete(oldest.value);
          }
          cache.set(id, { expiresAtMs: now + SUPPRESSION_CHECK_TTL_MS, suppressed: isSuppressed });
        }
        return result;
      },
    };
  };
}

export type MatchStage = "exact" | "embedding" | "none";

export interface FindResult {
  matched: boolean;
  passport_id?: string;
  stage: MatchStage;

   * enrolled image (the SAME photo re-encoded), not merely the same face.
   * Corroboration; never the basis of a match. */
  near_duplicate?: boolean;
  /** I3-0141 (D-0981(6)): present (true) only when the exact photo is held by more
   * than one live passport. OPERATOR-ONLY: the fan door strips it (http.ts /find). */
  exact_conflict?: boolean;
  quota_remaining: number;
}

/** phash corroboration: is the query a perceptual near-duplicate of the stored
 * enrolled image? Fail-safe false on any bad/degenerate input.
 *
 * D-0835 TRANSITION (I3-0836): an enrollment fingerprinted before the
 * alpha-composite fix stores the dHash of the alpha-DROPPED matrix, so for a
 * query with an alpha channel the stored value is compared against BOTH the
 * current matrix and `legacy` (the container's `gray9x8_legacy`). */
function phashNearDuplicate(gray9x8: Uint8Array, storedPhash: string, legacy?: Uint8Array): boolean {
  if (!structureOk(gray9x8)) return false; // a degenerate hash matches everything
  try {
    if (hamming(dHash(gray9x8), storedPhash) <= PHASH_NEAR) return true;
    return legacy !== undefined && structureOk(legacy) && hamming(dHash(legacy), storedPhash) <= PHASH_NEAR;
  } catch {
    return false;
  }
}

/** I3-1818: the most candidate ids one live-row read binds (D1 caps bound
 * parameters per statement at 100). EMBED_TOPK * VECTOR_OVERFETCH fits in one. */
const CANDIDATE_READ_CHUNK = IN_CHUNK; // I28-IMAGE: the one IN-list rule (db.ts)


/** I24 (I3-0840, RJ-04): the exact stage's owners read like the embedding stage's candidates (`eligibleAboveFloor`):
 * live consent (D10) AND not suppressed (D-0527). Every live owner of the bytes, earliest enrolment first
 * (`created_at`, then `image_id`), is put to the oracle in one call; the earliest UNSUPPRESSED owner wins and
 * `conflict` counts unsuppressed owners only, so a suppressed creator holding the same bytes neither hides nor
 * blocks a live creator (and stays unobservable). "unavailable" = the oracle could not answer -> the caller REFUSES. */
export async function exactLookupEligible(
  db: DbLike,
  exactHash: string,
  suppression: SuppressionOracle,
  /** I21-Q13 (Jason, D-1030 (2): own-face search only): when set, ONLY this passport's live enrolments are read, so
   * another passport holding the same bytes is never read, never matched and never makes the answer ambiguous. */
  onlyPassportId?: string,
): Promise<{ passport_id: string; conflict: boolean } | null | "unavailable"> {
  if (onlyPassportId !== undefined) {
    const own = await db
      .prepare(`SELECT 1 AS one FROM enrollments e WHERE e.exact_hash = ? AND e.passport_id = ? AND ${MATCH_CONSENT} LIMIT 1`)
      .bind(exactHash, onlyPassportId)
      .first<{ one: number }>();
    if (!own) return null;
    const sup = await suppression.check([onlyPassportId]);
    if (sup === null) return "unavailable";
    return sup.has(onlyPassportId) ? null : { passport_id: onlyPassportId, conflict: false };
  }
  // I25 (JA-02): the owner read is `eligibleExactOwners` (owners.ts), shared with enroll.ts.
  const { live, unavailable } = await eligibleExactOwners(db, exactHash, suppression, undefined, MATCH_CONSENT);
  if (unavailable) return "unavailable";
  return live.length ? { passport_id: live[0] as string, conflict: live.length > 1 } : null;
}

/**
 * D-0835 / D-0839 LAZY RE-FINGERPRINT. An exact-hash hit means the query IS the
 * enrolled image's bytes, so the query's CURRENT fingerprints are the
 * enrollment's correct ones. When the query carries a legacy matrix (it has an
 * alpha channel), every live enrollment of these bytes whose stored phash is
 * still the LEGACY value is rewritten to the current one — the UPDATE binds the
 * old VALUE (`AND phash = ?`, INV-058), so only the statement that changed the
 * row re-embeds it — and its face vector is re-upserted from the query's
 * current vector (D-0839). A withdrawal that commits between that UPDATE and
 * the upsert would otherwise leave a resurrected vector, so consent is re-read
 * after the upsert and the vector erased again when it is gone. Best effort:
 * a failure here never changes the find's answer.
 */
async function refreshLegacyEnrollments(query: NormalizedImage, deps: { db: DbLike; vectors: VectorStore }): Promise<void> {
  const legacy = query.gray9x8Legacy;
  if (legacy === undefined) return;
  try {
    const oldPhash = dHash(legacy);
    const newPhash = dHash(query.gray9x8);
    if (oldPhash === newPhash) return;
    const { results } = await deps.db
      .prepare(`SELECT e.image_id, e.passport_id FROM enrollments e WHERE e.exact_hash = ? AND e.phash = ? AND ${MATCH_CONSENT}`)
      .bind(query.exactHash, oldPhash)
      .all<{ image_id: string; passport_id: string }>();
    for (const r of results) {
      const claimed = await deps.db
        .prepare(`UPDATE enrollments SET phash = ? WHERE image_id = ? AND phash = ? AND ${MATCH_CONSENT_ENROLLMENTS}`)
        .bind(newPhash, r.image_id, oldPhash)
        .run();
      // I3-0704: the second door into the index — the same shape predicate as enroll.
      if (claimed.changes !== 1 || !isEmbedding(query.vector) || vectorDegenerate(query.vector)) continue;
      // I26 (RG NEW-1): the intent is recorded BEFORE the upsert and dropped only once the vector is settled (kept, or
      // erased again below), as enroll.ts does; any throw in between leaves it for reconcile.ts, which can then name
      // a vector whose row an erasure removed in the window.
      await deps.db.prepare("INSERT OR IGNORE INTO vector_intents (image_id, created_at) VALUES (?, ?)").bind(r.image_id, new Date().toISOString()).run();
      await deps.vectors.upsert(r.image_id, r.passport_id, query.vector);
      const still = await deps.db.prepare(`SELECT 1 AS one FROM enrollments e WHERE e.image_id = ? AND ${MATCH_CONSENT}`).bind(r.image_id).first<{ one: number }>();
      if (!still) await deps.vectors.deleteByIds([r.image_id]);
      await deps.db.prepare("DELETE FROM vector_intents WHERE image_id = ?").bind(r.image_id).run();
    }
  } catch {
    // Left for the next exact hit: the row still holds the legacy value.
  }
}

export interface FindDeps {
  db: DbLike;
  vectors: VectorStore;
  nowMs: number;
  /** REQUIRED, not optional — see the "not an optional argument" precedent
   * this mirrors: rankings.ts made `dropSuppressed`'s db parameter mandatory
   * after an optional one let three call sites silently skip delisting when
   * one forgot to pass it. An oracle-less findCreator is a compile error here
   * for the same reason, not a silently-unfiltered cascade. */
  suppression: SuppressionOracle;
}

/** The identity cascade WITHOUT quota — the shared core behind the fan-facing
 * findCreator (which meters it) and the S66 internal triage seam (machine
 *-to-machine, metered by the triage sweep's own batching, not fan quota).
 * Stores NOTHING about the query — ephemerality is structural here.
 *
 * `deps.suppression` is REQUIRED for the same reason `deps.db` is: this
 * function fails closed on it, never falls back to skipping the check. */
export async function findIdentity(
  query: NormalizedImage,
  /** `onlyPassportId` (I21-Q13): the creator own-face door (http.ts POST /find) scopes BOTH stages to the caller's
   * passport; the triage seam passes none. */
  deps: { db: DbLike; vectors: VectorStore; suppression: SuppressionOracle; onlyPassportId?: string },
): Promise<Omit<FindResult, "quota_remaining">> {
  // 1. Exact hash — a byte-exact re-upload of an enrolled image.
  // D10: a match may ONLY be returned for an enrollment whose consent is still
  // LIVE. Without this, a creator whose consent was withdrawn still had their
  // passport_id returned to a fan holding their photo: an identity oracle the
  // rest of the product refuses. The live-consent EXISTS rides the same statement
  // everywhere a match is decided. A DELISTED creator is refused by the other
  // gate, the trust-core suppression oracle (D-0527, `deps.suppression`, at both
  // stages), not by consent: only a delisting granted on the subject's own request
  // also owes the image consent revoke (trust-core likeness.ts
  // imageConsentOwedOnRemovalStatement), and the match does not wait on it.
  // D-0527: an exact re-upload is still gated by trust-core suppression, over EVERY live owner of the bytes (I24,
  // I3-0840: `exactLookupEligible`). "unavailable" → the oracle is unreachable → REFUSE (never "assume clear"); the
  // whole call returns unmatched rather than retrying per-candidate against a channel we already know is down.
  const exact = await exactLookupEligible(deps.db, query.exactHash, deps.suppression, deps.onlyPassportId);
  if (exact === "unavailable") return { matched: false, stage: "none" };
  // Every owner suppressed (null) does not end the search: the same passport may have other live, non-exact
  // enrollments (or the query may simply coincide with another creator's hash in a degenerate corpus); fall
  // through to the embedding path exactly as a stale vector does.
  if (exact) {
    await refreshLegacyEnrollments(query, deps);
    // D-1024(1) (I28-IMAGE M4): the query IS the enrolled bytes, so a live enrolment of them that the reconciler found
    // without its vector is re-embedded from the query's vector (reembed.ts; the D-0839 precedent above). Best effort.
    await reembedOwed(deps.db, deps.vectors, { exactHash: query.exactHash }, query.vector).catch(() => 0);
    // I3-0840 (I23-RG-SEC, I23-FX-TC): two-matches-means-no-match governs the exact stage too. A byte-exact
    // enrolment is still one candidate: when ANOTHER eligible passport clears the embedding floor for the same
    // query, the photo is ambiguous and the answer is no match — otherwise an impostor who enrols a shared photo of
    // a creator's face wins every exact upload of it. No usable query vector -> no fuzzy evidence -> exact stands,
    // UNLESS the hash itself has two live unsuppressed owners (I23-FX-GATE, matching http.ts's degraded exact branches): two
    // matches is no match, so the vectorless stage never picks between them.
    if (structureOk(query.gray9x8) && !vectorDegenerate(query.vector)) {
      const elig = await eligibleAboveFloor(query.vector, deps);
      if (elig === null) return { matched: false, stage: "none" };
      if (elig.some((r) => r.passport_id !== exact.passport_id)) return { matched: false, stage: "none" };
    } else if (exact.conflict) return { matched: false, stage: "none" };
    return { matched: true, passport_id: exact.passport_id, stage: "exact", ...(exact.conflict ? { exact_conflict: true } : {}) };
  }

  // DEGENERACY FLOOR (query side). A flat / near-constant query produces a
  // hash and an embedding that sit close to EVERYTHING, so a "match" against it
  // is meaningless. Exact (above) still stands; the fuzzy identity path does not
  // run on a degenerate input. (A degenerate byte-exact re-upload already
  // matched at step 2.)
  if (!structureOk(query.gray9x8)) return { matched: false, stage: "none" };

  // 3. Embedding — the only fuzzy identity signal. The query embedding was
  //    minted by the trusted Container (never stored here); search the enrolled
  //    corpus and confirm the TOP result clears the floor AND maps to a live
  //    enrollment row. An empty vector means the Container found no face.
  const vector = query.vector;
  // A degenerate / absent query vector (near-zero / near-constant / no face)
  // matches everything — never query the store with one.
  if (vectorDegenerate(vector)) return { matched: false, stage: "none" };
  const elig = await eligibleAboveFloor(vector, deps);
  if (elig === null) return { matched: false, stage: "none" };
  // I3-0840 (D-0993(1)): TWO OR MORE distinct eligible people over the floor is
  // AMBIGUOUS, and ambiguous is no match — a wrong attribution is worse than a
  // miss, and "top score wins" let a polished enrolment of someone else's face
  // outrank them. Eligible = the same live-row + not-suppressed filter as the
  // match itself, so a suppressed or withdrawn creator never makes a query
  // ambiguous (that would make them observable). Several vectors of ONE passport
  // are one person, not an ambiguity.
  let best: { passport_id: string; phash: string } | undefined;
  for (const row of elig) {
    if (!best) best = row;
    else if (row.passport_id !== best.passport_id) return { matched: false, stage: "none" };
  }
  if (best) {
    return { matched: true, passport_id: best.passport_id, stage: "embedding", near_duplicate: phashNearDuplicate(query.gray9x8, best.phash, query.gray9x8Legacy) };
  }

  return { matched: false, stage: "none" };
}

/** The embedding stage's candidates: every above-floor match (score order) that maps to a live enrollment row with
 * live consent (D10) and is not suppressed (D-0527), as its row. null = the suppression oracle is unreachable (the
 * caller REFUSES). Shared by the embedding stage and the exact stage's ambiguity check (I3-0840), so both read one
 * definition of "eligible". */
async function eligibleAboveFloor(
  vector: number[],
  deps: { db: DbLike; vectors: VectorStore; suppression: SuppressionOracle; onlyPassportId?: string },
): Promise<{ passport_id: string; phash: string }[] | null> {
  // ME-10: over-fetch a bounded multiple of the limit, because the live-row +
  // live-consent filter below runs AFTER retrieval and retained inactive
  // vectors would otherwise consume every position.
  const matches = await deps.vectors.query(vector, EMBED_TOPK * VECTOR_OVERFETCH);
  // Negated compare so a non-numeric / NaN score from the (untrusted) store
  // FAILS CLOSED: `NaN < FLOOR` is false (falls through and would match), but
  // `!(NaN >= FLOOR)` is true → stop. Store is sorted desc, so the first
  // sub-floor score means nothing after it qualifies.
  const above: VectorMatch[] = [];
  for (const m of matches) {
    if (!(m.score >= EMBED_MIN_SCORE)) break;
    above.push(m);
  }
  if (!above.length) return [];
  // I3-1818: ONE live-row read and ONE suppression call for every above-floor
  // candidate, instead of one of each per candidate in turn — the response time
  // no longer grows with how many similar faces exist. The decision is the same:
  // the candidates that (a) map to a live enrollment row with live consent (D10;
  // a stale vector — row deleted / consent withdrawn — is skipped) and (b) are
  // not suppressed (D-0527; suppressed is a skip) are eligible; exactly one
  // eligible PASSPORT matches (I3-0840 below).
  const ids = [...new Set(above.map((m) => m.id))];
  const rows = new Map<string, { passport_id: string; phash: string }>();
  // I21-Q13: scoped to one passport -> only that passport's rows are candidates (one bound parameter more: IN_CHUNK
  // is 90, so the statement stays within D1's 100).
  const scoped = deps.onlyPassportId !== undefined;
  for (let i = 0; i < ids.length; i += CANDIDATE_READ_CHUNK) {
    const chunk = ids.slice(i, i + CANDIDATE_READ_CHUNK);
    const got = await deps.db
      .prepare(`SELECT e.image_id, e.passport_id, e.phash FROM enrollments e WHERE e.image_id IN (${chunk.map(() => "?").join(", ")}) AND ${MATCH_CONSENT}${scoped ? " AND e.passport_id = ?" : ""}`)
      .bind(...chunk, ...(scoped ? [deps.onlyPassportId] : []))
      .all<{ image_id: string; passport_id: string; phash: string }>();
    for (const r of got.results) rows.set(r.image_id, { passport_id: r.passport_id, phash: r.phash });
  }
  if (!rows.size) return [];
  // null → the oracle is unreachable → REFUSE the whole call (never "assume clear").
  const sup = await deps.suppression.check([...new Set([...rows.values()].map((r) => r.passport_id))]);
  if (sup === null) return null;
  const out: { passport_id: string; phash: string }[] = [];
  for (const m of above) {
    const row = rows.get(m.id);
    if (!row || sup.has(row.passport_id)) continue;
    out.push(row);
  }
  return out;
}

/** Fan-facing find: quota first (one unit per attempt), then the shared
 * identity cascade. Behavior identical to the pre-S66 findCreator. */
export async function findCreator(
  query: NormalizedImage,
  accountId: string,
  tier: Tier,
  deps: FindDeps,
): Promise<FindResult> {
  const q = await consumeQuota(deps.db, accountId, tier, deps.nowMs);
  const r = await findIdentity(query, { db: deps.db, vectors: deps.vectors, suppression: deps.suppression });
  return { ...r, quota_remaining: q.remaining };
}

 *
 * COMPLIANCE: an enrollment REQUIRES a consent_ref — with no consent record,
 * enrollImage refuses. A creator can delete ALL their enrolled images, which
 * removes the DB rows AND the vectors (the biometric embeddings) from the
 * VectorStore. Deletion is idempotent.

 * had a door. `consents.revoked` shipped in migration 0001 and five production

 * is not — it erases the enrollments and leaves the consent record LIVE and
 * REUSABLE, so the withdrawal was never recorded and the same consent_ref still
 * authorized a future enrollment. `revokeConsents` below is the withdrawal
 * door; erasure now revokes too, in the same atomic batch, because a creator
 * deleting their enrollments has plainly withdrawn.
 *
 * REVOCATION IS ONE-WAY THROUGH THIS MODULE. Nothing here sets `revoked` back
 * to 0, and `recordConsent` REFUSES to reuse a consent_ref that has been
 * withdrawn — re-consenting means a NEW consent record, which is what keeps the
 * audit trail honest ("they withdrew on the 3rd and re-consented on the 9th"
 * rather than a single row that has silently changed its mind).
 *
 * THE TWO CONSENTS ARE CONNECTED (I3-0603, D-0340 / D-0829). trust-core's

 * both withdrawal doors (`revokeConsents`, `deleteCreatorImages`) take a
 * `LikenessRevokePeer`; when one is passed, a `likeness_revoke_outbox` row is
 * written IN THE SAME BATCH as the local revoke (so the local revoke is
 * recorded whatever the peer does), the peer is called after that batch
 * commits, and the row is deleted only once the peer answers. A failed call
 * leaves the row; `drainLikenessRevokeOutbox` is the sweep that retries it.
 * The peer must be idempotent (a retry re-sends the same revoke). A caller
 * that passes NO peer gets no propagation — the doors that must pass one are

 * direction (trust-core's revoke withdrawing THIS consent) arrives as a plain

/** I3-0603: trust-core's likeness-consent revoke for one passport. RESOLVES
 * when trust-core has recorded the revoke (or already had it — the call must
 * be idempotent, because the sweep re-sends it); REJECTS on anything else. */
export type LikenessRevokePeer = (passportId: string) => Promise<void>;

/** The peer-owed record, written in the SAME batch as the local revoke. One
 * row per passport; a withdrawal that lands while one is owed bumps `gen`, so a
 * peer call that was already in flight cannot clear the newer debt (INV-058:
 * the clearing DELETE binds the `gen` read BEFORE its peer call). */
function likenessOwedStmt(db: DbLike, passportId: string, ts: string): DbStatement {
  return db
    .prepare("INSERT INTO likeness_revoke_outbox (passport_id, queued_at) VALUES (?, ?) ON CONFLICT(passport_id) DO UPDATE SET gen = gen + 1")
    .bind(passportId, ts);
}

/** Call the peer for one owed passport. The owed row is DELETED only when the
 * peer resolves; on a rejection it stays, with its attempt counted, for
 * `drainLikenessRevokeOutbox`. NEVER THROWS: the local revoke has already
 * committed, and a peer outage must not turn a recorded withdrawal into an
 * error answered to the person who withdrew. Returns whether the peer took it. */
export async function propagateLikenessRevoke(
  db: DbLike,
  peer: LikenessRevokePeer,
  passportId: string,
  now: () => string = () => new Date().toISOString(),
): Promise<boolean> {
  let gen: number | undefined;
  try {
    gen = (await db.prepare("SELECT gen FROM likeness_revoke_outbox WHERE passport_id = ?").bind(passportId).first<{ gen: number }>())?.gen;
  } catch {
    gen = undefined; // unreadable: the call still goes out, the row is simply not cleared
  }
  let ok = false;
  try {
    await peer(passportId);
    ok = true;
  } catch {
    ok = false;
  }
  try {
    if (ok && gen !== undefined) await db.prepare("DELETE FROM likeness_revoke_outbox WHERE passport_id = ? AND gen = ?").bind(passportId, gen).run();
    else if (!ok) await db.prepare("UPDATE likeness_revoke_outbox SET attempts = attempts + 1, last_attempt_at = ? WHERE passport_id = ?").bind(now(), passportId).run();
  } catch {
    // The owed row is left as it was: a success re-sent by the sweep is
    // idempotent, and a failure not counted is still retried.
  }
  return ok;
}

/** I3-0603 — THE SWEEP. Retries every owed likeness revoke, least recently
 * attempted first so a passport that keeps failing cannot starve the rest,
 * at most `limit` per run; `more` says rows remain for the next run. */
export async function drainLikenessRevokeOutbox(
  db: DbLike,
  peer: LikenessRevokePeer,
  now: () => string = () => new Date().toISOString(),
  limit = 50,
): Promise<{ attempted: number; synced: number; more: boolean }> {
  const cap = Math.max(1, Math.min(500, Math.floor(limit)));
  const rows = (
    await db
      .prepare("SELECT passport_id FROM likeness_revoke_outbox ORDER BY COALESCE(last_attempt_at, '') ASC, passport_id ASC LIMIT ?")
      .bind(cap + 1)
      .all<{ passport_id: string }>()
  ).results;
  const batch = rows.slice(0, cap);
  let synced = 0;
  for (const r of batch) if (await propagateLikenessRevoke(db, peer, r.passport_id, now)) synced++;
  return { attempted: batch.length, synced, more: rows.length > cap };
}

/** Rows-affected for ONE statement of a `batch()`.

 * D1 returns one `D1Result` per batched statement — `{ success, results, meta:
 * { changes } }` — and `d1()` (env.ts:34-38) hands that array back unadapted;
 * the executing test double mirrors the shape (test/doubles.ts `runBatch`). The
 * long-standing comment in this file that `batch()` reports no per-statement

function batchChanges(result: unknown): number | null {
  const meta = (result as { meta?: { changes?: unknown } } | null)?.meta;
  return typeof meta?.changes === "number" ? meta.changes : null;
}

function assertShapes(i: EnrollInput): void {
  if (!/^psp_[0-9a-f]{16}$/.test(i.passportId)) throw new EnrollError("bad passport id");
  if (!i.consentRef || i.consentRef.length > 128) throw new EnrollError("a consent record is required to enroll");
  if (!/^[0-9a-f]{64}$/.test(i.exactHash)) throw new EnrollError("bad exact hash");
  if (!/^[0-9a-f]{16}$/.test(i.phash)) throw new EnrollError("bad phash");
  // I3-0704: exactly EMBED_DIM finite numbers (control: adapters.ts isEmbedding).
  if (!isEmbedding(i.vector)) {
    throw new EnrollError("bad embedding vector");
  }
  // DEGENERACY FLOOR (enroll side). A degenerate embedding (near-zero /
  // near-constant) matches EVERYTHING once indexed — it must never enter the
  // corpus. (phash degeneracy can only be judged from the grayscale matrix's
  // structure, which the caller checks at normalize time — an all-zero phash is
  // NOT a reliable signal: a monotonic gradient legitimately produces one.)
  if (vectorDegenerate(i.vector)) throw new EnrollError("degenerate embedding refused (would match everything)");
}

  return `img_${crypto.randomUUID().replace(/-/g, "")}`;

/** Record a creator's consent to be enrolled/searchable. In production this is
 * written ONLY by a creator-session-gated action. I21-Q16: the row stores the accepted wording's version, its text hash
 * and the acceptance instant when `opts.wording` is given (POST /consent always gives it); docs/IMAGE-PIPELINE.md

 * "belongs to someone else" would turn this door into a probe for other

export async function recordConsent(
  db: DbLike,
  consentRef: string,
  passportId: string,
  now: () => string = () => new Date().toISOString(),
  /** I21-Q20 (Jason, D-1030 (1)): the creator's over-18 declaration. POST /consent (http.ts) refuses without it and
   * passes `over18Declared: true`; the row then carries `over_18_declared_at` (migration 0009).
   * I21-Q16: `wording` is the approved `consent_wordings` row the creator accepted (approvedConsentWording); the row
   * then carries `wording_version`, `wording_hash` and `accepted_at` (migration 0011).
   * The over_18_declared_at and accepted_at instants equal the row's created_at on a first post, and are the re-post instant when this passport's live row lacked them. */
  opts: { over18Declared?: boolean; wording?: { wording_version: string; text_hash: string } } = {},
): Promise<void> {
  if (!/^psp_[0-9a-f]{16}$/.test(passportId)) throw new EnrollError("bad passport id");
  if (!consentRef || consentRef.length > 128) throw new EnrollError("bad consent ref");
  const existing = await db.prepare("SELECT passport_id, revoked FROM consents WHERE consent_ref = ?").bind(consentRef).first<{ passport_id: string; revoked: number }>();
  if (existing && (existing.revoked === 1 || existing.passport_id !== passportId)) {
    throw new EnrollError("consent_ref unavailable — record a NEW consent reference");
  }
  const at = now();
  // D-1059(13) (I30-RG-PRIV I30-IMAGE-Q20): re-posting the declaration on THIS passport's live row that lacks it stores
  // it (the declaration instant); any other conflict changes nothing and is judged below exactly as before.
  // I21-Q16: the same for the accepted wording — a live row of this passport with no wording takes the one posted now;
  // a value already stored is never replaced (COALESCE keeps it).
  const w = opts.wording ?? null;
  const wrote = await db
    .prepare(
      "INSERT INTO consents (consent_ref, passport_id, created_at, over_18_declared_at, wording_version, wording_hash, accepted_at) VALUES (?, ?, ?, ?, ?, ?, ?)" +
        " ON CONFLICT(consent_ref) DO UPDATE SET over_18_declared_at = COALESCE(consents.over_18_declared_at, excluded.over_18_declared_at)," +
        " wording_version = COALESCE(consents.wording_version, excluded.wording_version), wording_hash = COALESCE(consents.wording_hash, excluded.wording_hash)," +
        " accepted_at = COALESCE(consents.accepted_at, excluded.accepted_at)" +
        " WHERE (consents.over_18_declared_at IS NULL OR consents.wording_version IS NULL) AND consents.passport_id = excluded.passport_id AND consents.revoked = 0",
    )
    .bind(consentRef, passportId, at, opts.over18Declared === true ? at : null, w ? w.wording_version : null, w ? w.text_hash : null, w ? at : null)
    .run();
  if (wrote.changes === 0) {
    // I3-1788. The SELECT above answered for a moment that had already passed
    // by the time this statement ran, and `ON CONFLICT DO NOTHING` cannot tell
    // "I re-recorded my own live ref" from "somebody else's FIRST write beat
    // mine": both are zero rows and both used to resolve. The second one left
    // the creator believing they had consented while the ref was bound to
    // another passport — and the refusal this docblock promises was never
    // applied to the only state that decides it, which is the table AFTER the
    // write. Same predicate, same message, applied to what is actually there.
    const bound = await db.prepare("SELECT passport_id, revoked FROM consents WHERE consent_ref = ?").bind(consentRef).first<{ passport_id: string; revoked: number }>();
    if (!bound || bound.revoked === 1 || bound.passport_id !== passportId) {

    }

/** I23-Q15d (Jason, D-1030: "(b) build it"; census pick: plumbing now, the verdict filled after I21-Q14). Would an
 * accepted photo be found? `threshold` is the calibrated setting (env.ts findableThresholdFromEnv); null -> 'unknown'.
 * Quality is the photo's structure (`structureStdDev` of the container's 9x8 luminance matrix, the cascade's own
 * query-side floor) and 0 when the face vector is degenerate (the cascade never queries one). Below the threshold
 * -> 'no', else 'yes'. http.ts /enroll omits the field while FIND_ENABLED is off. */
export type Findable = "yes" | "no" | "unknown";
export function findableVerdict(gray9x8: Uint8Array, vector: number[], threshold: number | null): Findable {
  if (threshold === null) return "unknown";
  const quality = vectorDegenerate(vector) ? 0 : structureStdDev(gray9x8);
  return quality < threshold ? "no" : "yes";
}

 *
 * `consentRef` narrows it to a single record; omitted, it withdraws EVERY live
 * consent this passport holds. Either way the `passport_id = ?` predicate rides
 * on the statement, so a named ref belonging to somebody else matches no row
 * and the answer is the same `{revoked: 0}` a ref that never existed gets — no
 * cross-creator probe, and no oracle in the count.
 *
 * IDEMPOTENT: the `AND revoked = 0` guard means a second call changes nothing
 * and reports 0. IRREVERSIBLE through this module: nothing sets `revoked` back
 * to 0, and `recordConsent` refuses to reuse a withdrawn ref.
 *
 * WHAT WITHDRAWAL DOES IMMEDIATELY: every
 * surface that can return a passport_id — the two exact-hash lookups in

 *
 * I3-0138 — WITHDRAWAL ALSO ERASES THE BIOMETRICS. The rows and the face
 * vectors used to be RETAINED as inert. Inert is not erased: a withdrawn
 * consent that leaves the embedding in the index has not undone the biometric
 * processing it withdrew, and "the cascade rejects it" is a live guard, not a
 * deletion.
 *
 * VECTORS FIRST, THEN ROWS (#609f — this paragraph used to say the opposite;
 * the code below is the record): the vectors of the covered enrollment ids are
 * erased BEFORE the batch (`eraseVectors`), by ID, because the rows are the only
 * record that can name them. The batch then carries the revoke and, only when
 * no vector delete failed, the DELETE of those rows (`dropRows`). On a vector
 * failure the revoke still lands and the rows are LEFT so a retry can enumerate
 * the ids. By ID, because the deployed Vectorize adapter's `deleteByPassport`
 * is a documented permanent no-op (env.ts) — only `deleteByIds` deletes
 * anything in production.
 *
 * THE VECTOR STORE IS A PARAMETER, not a lookup: this module has no env. A
 * caller that does not pass one still records the withdrawal but deletes no
 * rows: with enrollments to erase, the rows stay to name the vectors nothing
 * erased, `erased` is 0 and `vectorsErased` false — pending, never finished.

 * deleteCreatorImages counts before its own: `batch()` reports no
 * per-statement rows-affected. Nothing branches on it.
 */
export async function revokeConsents(
  db: DbLike,
  passportId: string,
  consentRef?: string,
  now: () => string = () => new Date().toISOString(),
  vectors?: VectorStore,
  peer?: LikenessRevokePeer,
  holdAsk?: FaceHoldAsk | null, // I65-IMAGE (D-1569): undefined = no ask (trust-core's own call); null, or any answer but "unheld", keeps the face data
  through?: string, // I65-FX-FACE R4 (D-1575 (4a)): trust-core's owed instant; only consents and enrolments created at or before it
): Promise<{ revoked: number; erased: number; vectorsErased: boolean; likenessSynced: boolean | null }> {
  if (!/^psp_[0-9a-f]{16}$/.test(passportId)) throw new EnrollError("bad passport id");
  if (consentRef !== undefined && (!consentRef || consentRef.length > 128)) throw new EnrollError("bad consent ref");
  if (through !== undefined && (consentRef !== undefined || !OWED_INSTANT_RE.test(through))) throw new EnrollError("bad owed instant");
  const cut = through ?? "9999-12-31T23:59:59.999Z"; // no cutoff = every row (an ISO instant sorts as text)
  const ts = now();
  const ids = (
    await (consentRef === undefined
      ? db.prepare("SELECT image_id FROM enrollments WHERE passport_id = ? AND created_at <= ?").bind(passportId, cut)
      : db.prepare("SELECT image_id FROM enrollments WHERE passport_id = ? AND consent_ref = ?").bind(passportId, consentRef)
    ).all<{ image_id: string }>()
  ).results.map((r) => r.image_id);
  const live = await (consentRef === undefined
    ? db.prepare("SELECT COUNT(*) AS n FROM consents WHERE passport_id = ? AND revoked = 0 AND created_at <= ?").bind(passportId, cut)
    : db.prepare("SELECT COUNT(*) AS n FROM consents WHERE passport_id = ? AND consent_ref = ? AND revoked = 0").bind(passportId, consentRef)
  ).first<{ n: number }>();
  // I3-0119 ORDERING, APPLIED HERE TOO — this is the module's SECOND erasure
  // door and it had the same shape. The enrollment rows are the ONLY record that can name these embeddings, so deleting them before the vectors turns
  // any vector failure into residue nothing can enumerate. Vectors first.
  //
  // WITHDRAWAL IS STILL NEVER BLOCKED BY IT. Refusing the withdrawal because a
  // vector store is unreachable is the worse failure for the only party this
  // door exists to serve, and http.ts rules on exactly that. So on a vector
  // FAILURE the revoke still lands — it is what makes the subject unsearchable
  // — and the enrollment ROWS are LEFT IN PLACE, which keeps the ids
  // enumerable for a retry instead of stranding the embeddings. `erased` then
  // reports 0 and `vectorsErased` false, so the door's `vectors_pending` is
  // true and the residue is NAMED (D-0570), not reported as a whole erasure.
  //
  // A store never PASSED (none bound, or I65-IMAGE: a hold or no hold answer) is the same case since I3-0138 (D-1082(14)):
  // nothing erased the vectors, so the rows stay to name them, `erased` is 0 and `vectorsErased` false — pending.
  const erasable = ids.length && holdAsk !== undefined && !(await faceEraseUnheld(holdAsk, passportId)) ? undefined : vectors;
  let vectorsErased = ids.length === 0; // nothing to erase is vacuously done
  if (erasable && ids.length) {
    try {
      await eraseVectors(erasable, ids);
      vectorsErased = true;
    } catch {
      // the rows stay (dropRows below), so the ids remain enumerable for a retry
    }
  }
  const dropRows = ids.length > 0 && vectorsErased;
  const revokeStmt =
    consentRef === undefined
      ? db.prepare("UPDATE consents SET revoked = 1, revoked_at = ? WHERE passport_id = ? AND revoked = 0 AND created_at <= ?").bind(ts, passportId, cut)
      : db.prepare("UPDATE consents SET revoked = 1, revoked_at = ? WHERE passport_id = ? AND consent_ref = ? AND revoked = 0").bind(ts, passportId, consentRef);
  const dropStmt =
    consentRef === undefined
      ? db.prepare("DELETE FROM enrollments WHERE passport_id = ? AND created_at <= ?").bind(passportId, cut)
      : db.prepare("DELETE FROM enrollments WHERE passport_id = ? AND consent_ref = ?").bind(passportId, consentRef);
  const stmts = dropRows ? [revokeStmt, dropStmt] : [revokeStmt];
  // I3-0603: the peer-owed record commits WITH the local revoke, never after it.
  if (peer) stmts.push(likenessOwedStmt(db, passportId, ts));
  await db.batch(stmts);
  const likenessSynced = peer ? await propagateLikenessRevoke(db, peer, passportId, now) : null;
  // JUDGE-W309A (2026-09-20). `erased` counts the enrollment ROWS this call
  // removed. It is NOT a statement about the VECTORS, and the door must not
  // read it as one. (The paragraph that used to stand here described
  // `deleteByIds` as duck-typed off an interface that did not declare it;
  // adapters.ts:68 declares it now, so that sentence had outlived its code and
  // is gone rather than left to be believed.)
  //
  // `vectorsErased` reports WHAT HAPPENED rather than what was passed. The
  // door's `vectors_pending` is derived from it, so a partial erasure can no
  // longer be reported as a whole one by the mere presence of a store object —
  // which is the exact failure D-0570 was written to prevent, reintroduced one
  // level down.
  return { revoked: live?.n ?? 0, erased: dropRows ? ids.length : 0, vectorsErased, likenessSynced };
}

/**
 * ME-08 (S-068). THE CHEAP PREREQUISITES, ASKED BEFORE THE EXPENSIVE WORK.
 *
 * `POST /find` was moved to consume quota BEFORE calling the Container back in
 * S-038 I-13 / PS-024, for exactly this reason, and it still does. `/enroll`
 * was never given the same treatment: it fingerprints the upload — the trusted
 * Container decodes the image and runs the face embedder, the most expensive
 * thing this service can do — and only afterwards reaches `enrollImage`, which
 * discovers with two cheap SQL reads that the creator has no live consent
 * record or is already at `MAX_ENROLLMENTS_PER_CREATOR`. A creator at the cap
 * could buy a decode + embed on every attempt, forever, and be refused every
 * time.
 *
 * This is a PRE-CHECK, not a replacement. `enrollImage` re-asks both facts
 * inside the write path and remains the authority — a consent withdrawal or a
 * concurrent enrollment landing between the two is still caught there, which
 * is why nothing here is removed from it. Cheap admission first, authoritative
 * admission at the write: the same two-layer shape `media.ts storeSignedMedia`
 * uses for its own volume cap, and the same one `warnings.ts` uses for its
 * gates.
 *
 * Throws `EnrollError` with the SAME messages `enrollImage` throws, so a caller
 * that checks here and a caller that does not cannot report different reasons
 * for the same refusal.
 */
export async function enrollPreconditions(db: DbLike, passportId: string, consentRef: string): Promise<void> {
  if (!/^psp_[0-9a-f]{16}$/.test(passportId)) throw new EnrollError("bad passport id");
  if (!consentRef || consentRef.length > 128) throw new EnrollError("a consent record is required to enroll");
  // D-1059(13): a consent with no over-18 declaration (a pre-0009 row) is not a live consent for enrolment.
  // D-1073(9): nor is one not on an approved wording (CONSENT_WORDING_APPROVED).
  const consent = await db
    .prepare(`SELECT c.passport_id, c.revoked, c.over_18_declared_at, ${CONSENT_WORDING_APPROVED} AS wording_ok FROM consents c WHERE c.consent_ref = ?`)
    .bind(consentRef)
    .first<{ passport_id: string; revoked: number; over_18_declared_at: string | null; wording_ok: number }>();
  if (!consent || consent.revoked === 1 || consent.passport_id !== passportId || consent.over_18_declared_at === null || consent.wording_ok !== 1) {
    throw new EnrollError("no live consent record for this passport");
  }
  const existingCount = await db.prepare("SELECT COUNT(*) AS n FROM enrollments WHERE passport_id = ?").bind(passportId).first<{ n: number }>();
  if ((existingCount?.n ?? 0) >= MAX_ENROLLMENTS_PER_CREATOR) {
    throw new EnrollError(`enrollment cap reached (${MAX_ENROLLMENTS_PER_CREATOR})`);
  }
}

/**
 * Enroll one consented image. CALLER-GATING CONTRACT: this must be invoked only
 * for the AUTHENTICATED creator's own passport (the HTTP layer enforces the
 * creator session — never call it with a passport the caller hasn't proven).
 * Consent is a real check: a live, non-revoked consent row bound to THIS
 * passport, with the over-18 declaration and on an approved wording (CONSENT_WORDING_APPROVED), must exist, else
 * enrollment is refused.
 */
export async function enrollImage(
  db: DbLike,
  vectors: VectorStore,
  /** I25 (JA-02): REQUIRED, so no door can skip it. Only a live-consent, UNSUPPRESSED other owner refuses. */
  suppression: SuppressionOracle,
  input: EnrollInput,
  now: () => string = () => new Date().toISOString(),
  holdAsk?: FaceHoldAsk | null, // I65-FX-FACE R7 (J3): undefined = no ask; held or no answer keeps a refused row's face data
): Promise<{ image_id: string }> {
  assertShapes(input);
  const consent = await db
    .prepare(`SELECT c.passport_id, c.revoked, c.over_18_declared_at, ${CONSENT_WORDING_APPROVED} AS wording_ok FROM consents c WHERE c.consent_ref = ?`)
    .bind(input.consentRef)
    .first<{ passport_id: string; revoked: number; over_18_declared_at: string | null; wording_ok: number }>();
  if (!consent || consent.revoked === 1 || consent.passport_id !== input.passportId || consent.over_18_declared_at === null || consent.wording_ok !== 1) {
    throw new EnrollError("no live consent record for this passport");
  }
  // I3-1703: re-uploading the EXACT same image is a no-op, not a new slot.
  // Idempotent on (passport_id, exact_hash) — a live row for this passport with
  // this exact_hash IS the enrollment the caller is asking for, so hand back
  // its id rather than growing the vector index and spending a cap slot on a
  // duplicate of something already enrolled and searchable. Scoped to the
  // CALLER's own passport (never cross-passport) — same scoping the rest of
  // this function already uses.
  const dup = await db.prepare("SELECT image_id FROM enrollments WHERE passport_id = ? AND exact_hash = ?").bind(input.passportId, input.exactHash).first<{ image_id: string }>();
  if (dup) {
    // D-1024(1) (I28-IMAGE M4): a live enrolment the reconciler found without its vector is re-embedded from this upload
    // of the same bytes (reembed.ts). Best effort: a failure leaves the debt and its intent for the next door or run.
    await reembedOwed(db, vectors, { exactHash: input.exactHash, passportId: input.passportId }, input.vector, now).catch(() => 0);
    return { image_id: dup.image_id };
  }
  // I3-0141 (D-0981(6)): one exact photo, one creator. Another passport already
  // holding this hash sends the attempt to operator review, never into the index.
  // I25 (JA-02, D-0527): owners are read like /find reads them (owners.ts). A suppressed or withdrawn owner
  // is not an owner here either, so the answer is the one a photo nobody holds gets; a suppressed owner
  // still gets its review row, silently. The oracle down -> every owner counts (fail closed).
  const owners = await eligibleExactOwners(db, input.exactHash, suppression, input.passportId);
  if (owners.live.length) return refuseIntoReview(db, input, owners.live[0] as string, now);
  const existingCount = await db.prepare("SELECT COUNT(*) AS n FROM enrollments WHERE passport_id = ?").bind(input.passportId).first<{ n: number }>();
  if ((existingCount?.n ?? 0) >= MAX_ENROLLMENTS_PER_CREATOR) {
    throw new EnrollError(`enrollment cap reached (${MAX_ENROLLMENTS_PER_CREATOR})`);
  }
  const id = imageId();
  // QUEUE #205 / ME-09 (I26-W1-IMAGE): the intent is recorded BEFORE the row, and deleted when the attempt is refused
  // or completes; an interrupted attempt leaves it for reconcile.ts (migration 0006).
  await db.prepare("INSERT OR IGNORE INTO vector_intents (image_id, created_at) VALUES (?, ?)").bind(id, now()).run();
  // Row first (the record of consent + provenance), then the vector — so a
  // partial failure never leaves a searchable vector with no consent row.
  //
  // I3-0268, CLAIM THEN ACT. The two reads above are CLAIMS and this is the
  // ACT, and they used to be four awaits apart with no transaction between
  // them: a withdrawal or a concurrent enrollment landing in the window was
  // written against a fact that had already stopped being true. The INSERT now
  // carries BOTH predicates itself, so the row lands only if they still hold AT
  // THE WRITE. The reads stay — they are what produces the specific refusal
  // message, and they are the cheap path.
  const wrote = await db
    .prepare(
      "INSERT INTO enrollments (image_id, passport_id, exact_hash, phash, consent_ref, created_at) " +
        `SELECT ?, ?, ?, ?, ?, ? WHERE EXISTS (SELECT 1 FROM consents c WHERE c.consent_ref = ? AND c.passport_id = ? AND c.revoked = 0 AND c.over_18_declared_at IS NOT NULL AND ${CONSENT_WORDING_APPROVED}) ` +
        "AND (SELECT COUNT(*) FROM enrollments WHERE passport_id = ?) < ? " +
        // I3-0141: the cross-passport fact rides the write too (claim then act). I25 (JA-02, INV-058): the
        // hidden owners are the VALUES the claim read, bound here; any other live owner still blocks.
        `AND NOT EXISTS (SELECT 1 FROM enrollments e WHERE e.exact_hash = ? AND e.passport_id <> ? AND ${LIVE_CONSENT} AND e.passport_id NOT IN (SELECT value FROM json_each(?)))`,
    )
    .bind(id, input.passportId, input.exactHash, input.phash, input.consentRef, now(), input.consentRef, input.passportId, input.passportId, MAX_ENROLLMENTS_PER_CREATOR, input.exactHash, input.passportId, JSON.stringify(owners.hidden))
    .run();
  if (wrote.changes === 0) {
    await db.prepare("DELETE FROM vector_intents WHERE image_id = ?").bind(id).run(); // nothing was written: no intent
    // Re-read to pick between the two refusals callers already know. No new
    // error string: a caller that lost a race must not be told something a
    // caller that never raced could never hear.
    const stillLive = await db
      .prepare(`SELECT 1 AS ok FROM consents c WHERE c.consent_ref = ? AND c.passport_id = ? AND c.revoked = 0 AND c.over_18_declared_at IS NOT NULL AND ${CONSENT_WORDING_APPROVED}`)
      .bind(input.consentRef, input.passportId)
      .first<{ ok: number }>();
    if (!stillLive) throw new EnrollError("no live consent record for this passport");
    const raced = await eligibleExactOwners(db, input.exactHash, suppression, input.passportId);
    if (raced.live.length) return refuseIntoReview(db, input, raced.live[0] as string, now);
    throw new EnrollError(`enrollment cap reached (${MAX_ENROLLMENTS_PER_CREATOR})`);
  }
  // I25 (JA-02): the review row a suppressed owner earns, written silently — a failure here must not change
  // the answer, or the answer would differ exactly when a hidden owner exists.
  if (owners.hidden.length) await queueReview(db, input, owners.hidden[0] as string, now).catch(() => undefined);
  // PS-072 / PS-084: a failed vector upsert (a Vectorize hiccup, a network
  // blip) used to leave the row above stranded forever — exact-hash
  // searchable, embedding-invisible, with no cleanup and no retry, because the
  // caller sees a 500 and has no way to reach the orphaned image_id. The row
  // is the compensating action's target, not a second write path: on a failed
  // upsert we remove the row we just inserted and rethrow, so a retry starts
  // clean (a new image_id) instead of piling up dead rows next to each retry.
  //
  // JA-02 (I22, ME-09 window c): a throwing upsert is AMBIGUOUS — the vector may
  // have landed before the throw. Erase it by id FIRST, and drop the row only
  // once that erase succeeded; if the erase throws too, the row is LEFT, because
  // it is the only record that names the vector (deleteCreatorImages and
  // revokeConsents enumerate ids from rows). The upsert's own error is rethrown.
  try {
    await vectors.upsert(id, input.passportId, input.vector);
  } catch (e) {
    if (holdAsk !== undefined && !(await faceEraseUnheld(holdAsk, input.passportId))) throw e; // I65-FX-FACE R7 (J3): kept under a hold
    let erased = false;
    try {
      await eraseVectors(vectors, [id]);
      erased = true;
    } catch {
      // row kept: see above
    }
    if (erased) await db.prepare("DELETE FROM enrollments WHERE image_id = ?").bind(id).run();
    throw e;
  }
  // I3-0268, THE OTHER SIDE OF THE SAME RACE, and the expensive one. An erasure
  // or a withdrawal that took its id snapshot between the INSERT above and this
  // upsert deleted the ROW and could not delete a vector that did not exist
  // yet. The embedding then lands in the index with no row, no consent and no
  // id recorded anywhere — not merely un-erased but UNENUMERABLE: no later
  // sweep, and no retry of the erasure door, can ever name it. Re-read the row
  // we wrote together with its consent; gone or revoked means our vector is
  // exactly that residue, so erase it here — the same compensating-action shape
  // as the catch above — and refuse.
  const landed = await db
    .prepare(`SELECT 1 AS ok FROM enrollments e WHERE e.image_id = ? AND EXISTS (SELECT 1 FROM consents c WHERE c.consent_ref = e.consent_ref AND c.passport_id = e.passport_id AND c.revoked = 0 AND ${CONSENT_WORDING_APPROVED})`)
    .bind(id)
    .first<{ ok: number }>();
  if (!landed) {
    // I65-FX-FACE R7 (J3): a row still present under a withdrawn consent is kept under a hold (sweepOwedFaceErasures takes it
    // after the release); a row already gone leaves nothing to keep, so its vector is erased whatever the hold says (I3-0268).
    if (holdAsk !== undefined && (await db.prepare("SELECT 1 AS ok FROM enrollments WHERE image_id = ?").bind(id).first<{ ok: number }>()) && !(await faceEraseUnheld(holdAsk, input.passportId))) throw new EnrollError("no live consent record for this passport");
    await eraseVectors(vectors, [id]);
    await db.prepare("DELETE FROM enrollments WHERE image_id = ?").bind(id).run();
    throw new EnrollError("no live consent record for this passport");
  }
  await db.prepare("DELETE FROM vector_intents WHERE image_id = ?").bind(id).run(); // row and vector both landed
  return { image_id: id };
}

/**
 * Creator right-to-deletion: withdraw their consent AND remove all enrolled
 * images + their vectors.
 *
 * ERASURE IMPLIES WITHDRAWAL, and the two land in ONE atomic batch. Deleting
 * the enrollments while leaving the consent record live left the creator's
 * recorded position as "I consent to biometric processing" the instant after
 * they asked for their biometrics to be destroyed, and left the consent_ref
 * valid for a future enrollment. A batch (a transaction in D1 and in the test
 * double) means there is no window in which one happened and the other did not.
 *
 * REVOKE IS THE FIRST STATEMENT IN THE BATCH deliberately: it is the one that
 * makes the subject unsearchable, since every match path requires
 * `revoked = 0`. If a future refactor breaks the batch apart, this order is the
 * one that fails safe.
 *
 * VECTORS BEFORE ROWS (I3-0119, 2026-09-22 — this paragraph said the opposite
 * and the opposite is what produced the defect). "A leftover orphan vector is
 * harmless because the cascade rejects any match with no live row" answers the
 * SEARCHABILITY question and was true. It is not the question an erasure is
 * asked. The rows are the ONLY record in this system that can NAME these
 * embeddings, so committing their deletion first meant that any failure of the
 * vector call left face embeddings in the index that nothing — no retry of this
 * door, no later sweep, no operator — can ever enumerate. Inert is not erased,
 * and unenumerable is worse than inert. The vectors go first now: a store that
 * throws does so with every row still present, so the ids are re-derivable and
 * a retry of this idempotent door finishes the job.
 *
 * THE ROW DELETE IS BOUND TO THE IDS THE SNAPSHOT CLAIMED, not to the whole
 * passport. That is what makes `removed` true, and it is also what stops the
 * racer class: an enrollment landing between the snapshot and the batch is no
 * longer deleted-without-its-vector — its row SURVIVES (unsearchable, since the
 * revoke rode the same batch) and PASS TWO erases it vector-first like any
 * other. Two passes, never a loop; whatever is still present after pass two is
 * REPORTED as `residual` rather than swallowed.
 *
 * THE COUNTS. `consents_revoked` is still read before the batch — it is a
 * report, nothing branches on it. `removed` is NOT: it is read from the DELETE
 * statement's own `meta.changes` (see `batchChanges`), because the claim that
 * `batch()` reports no per-statement rows-affected stopped being true and the
 * number was being handed to a DATA SUBJECT as the answer to "how much of me
 * did you destroy".
 */
export async function deleteCreatorImages(
  db: DbLike,
  vectors: VectorStore,
  passportId: string,
  now: () => string = () => new Date().toISOString(),
  peer?: LikenessRevokePeer,
): Promise<{ removed: number; consents_revoked: number; vectors_erased: boolean; residual: number; likeness_synced: boolean | null }> {
  if (!/^psp_[0-9a-f]{16}$/.test(passportId)) throw new EnrollError("bad passport id");
  // TE-03: `vectors.deleteByPassport` reads as the vector-removal step, but the
  // real Vectorize adapter (services/image/src/env.ts) documents it as a
  // permanent NO-OP — Vectorize has no metadata-filtered delete. Before this,
  // the ACTUAL deployed deletion was a duck-typed `deleteByIds` call bolted
  // onto the HTTP route (http.ts), which independently re-queried these same
  // ids and ran after this function returned — a second, untested path this
  // function's own doc comment never disclosed depending on, and exactly what
  // let the erasure tests pass against a double whose `deleteByPassport` does
  // real work while the production one does nothing (TE-03). Querying the ids
  // HERE and deleting by id HERE, the same duck-typed way http.ts already did,
  // makes this function's "removes the vectors" claim true on its own, for any
  // caller, without a second file's cooperation.
  const live = await db.prepare("SELECT COUNT(*) AS n FROM consents WHERE passport_id = ? AND revoked = 0").bind(passportId).first<{ n: number }>();
  // Runs in the VECTOR phase, before any row is removed, like every other
  // vector call here. On the deployed Vectorize adapter it is a documented
  // permanent no-op (env.ts:54) — Vectorize has no metadata-filtered delete —
  // so it is the by-id loop below that does the work in production.
  await vectors.deleteByPassport(passportId);
  let removed = 0;
  let revoked = false;
  for (let pass = 0; pass < 2; pass++) {
    const ids = (await db.prepare("SELECT image_id FROM enrollments WHERE passport_id = ?").bind(passportId).all<{ image_id: string }>()).results.map((r) => r.image_id);
    // VECTORS BEFORE ROWS: a throw here leaves every row in place, so the ids
    // stay enumerable and a retry of this door finishes the erasure.
    if (ids.length) await eraseVectors(vectors, ids);
    const chunks: string[][] = [];
    // I28-IMAGE M2: the row DELETE binds passport_id + one chunk, so it chunks by IN_CHUNK (db.ts), not by the
    // vector store's VECTOR_ERASE_CHUNK (100 ids + passport_id = 101 binds, over D1's cap).
    for (let i = 0; i < ids.length; i += IN_CHUNK) chunks.push(ids.slice(i, i + IN_CHUNK));
    if (!chunks.length) chunks.push([]); // the revoke still has to land
    for (const chunk of chunks) {
      const stmts: DbStatement[] = [];
      // REVOKE IS THE FIRST STATEMENT OF THE FIRST BATCH, for the reason the
      // docblock gives: it is the one that makes the subject unsearchable.
      if (!revoked) {
        const ts = now();
        stmts.push(db.prepare("UPDATE consents SET revoked = 1, revoked_at = ? WHERE passport_id = ? AND revoked = 0").bind(ts, passportId));
        // I3-0603: the peer-owed record rides the batch that carries the revoke.
        if (peer) stmts.push(likenessOwedStmt(db, passportId, ts));
        // I3-0141: the review queue (migration 0005) names this passport and a
        // photo hash; erasure removes every row naming it on either side.
        stmts.push(db.prepare("DELETE FROM exact_hash_reviews WHERE requesting_passport_id = ? OR existing_passport_id = ?").bind(passportId, passportId));
        revoked = true;
      }
      if (chunk.length) {
        stmts.push(
          db.prepare(`DELETE FROM enrollments WHERE passport_id = ? AND image_id IN (${chunk.map(() => "?").join(", ")})`).bind(passportId, ...chunk),
        );
      }
      if (!stmts.length) continue;
      const res = await db.batch(stmts);
      if (chunk.length) removed += batchChanges(res[stmts.length - 1]) ?? chunk.length;
    }
    if (!ids.length) break;
  }
  const left = await db.prepare("SELECT COUNT(*) AS n FROM enrollments WHERE passport_id = ?").bind(passportId).first<{ n: number }>();
  const residual = left?.n ?? 0;
  // `vectors_erased` is FALSE exactly when rows this passport owns are still
  // present after two passes: those rows' embeddings are still in the index,
  // and a caller that reports a whole erasure on that state is reporting a lie
  // to a data subject. It is reachable — an enrollment that keeps landing
  // inside the sweep window reaches it.
  const likeness_synced = peer ? await propagateLikenessRevoke(db, peer, passportId, now) : null;
  return { removed, consents_revoked: live?.n ?? 0, vectors_erased: residual === 0, residual, likeness_synced };
}

/**
 * I65-IMAGE (Jason D-1569 "Yes, ask trust-core first (Recommended)"; W0-13, amendment 8). THE HOLD ASK.
 * Before this service erases a creator's face data (enrolment rows and their vectors) at a door the creator opened, it
 * asks trust-core (POST /internal/face-hold, env.ts faceHoldAskFromEnv) whether an open legal hold names the passport or
 * the account that owns it. RESOLVES "unheld" (erase as before) or "held" (trust-core has written image:face_erase_owed;
 * its sweep arm erases through POST /internal/face-erase once no hold remains); REJECTS when no answer could be had.
 * Held and no answer both keep the data: the withdrawal is still recorded at once, and the rows stay, which is what keeps
 * the erasure owed (a withdrawn row is enumerable) for sweepOwedFaceErasures below.
 */

 * consent is withdrawn (UNREVOKED_CONSENT false). That is the set a held or unanswered /consent/revoke or DELETE

export async function eraseWithdrawnFaceData(db: DbLike, vectors: VectorStore, passportId: string, opts: { reviews?: boolean } = {}): Promise<number> {
  if (!/^psp_[0-9a-f]{16}$/.test(passportId)) throw new EnrollError("bad passport id");
  const ids = (await db.prepare(`SELECT e.image_id FROM enrollments e WHERE e.passport_id = ? AND NOT ${UNREVOKED_CONSENT}`).bind(passportId).all<{ image_id: string }>()).results.map((r) => r.image_id);
  // I65-FX-FACE R7 (D-1576, I65-Q3): review rows go ONLY when trust-core's release arm says a held DELETE /enrollments kept
  // them (`opts.reviews`, from its image:face_erase_owed reviews marker), and only those made by the time the consents were
  // withdrawn; an unheld withdrawal keeps them until DELETE /enrollments, as at i65-head-1c. Not counted in the rows removed.
  if (opts.reviews === true) await db.prepare(`DELETE FROM exact_hash_reviews WHERE (requesting_passport_id = ? OR existing_passport_id = ?) AND ${REVIEWS_KEPT_BY_DELETE_SQL}`).bind(passportId, passportId, passportId).run();
  if (!ids.length) return 0;
  await eraseVectors(vectors, ids);
  let removed = 0;
  for (let i = 0; i < ids.length; i += IN_CHUNK) {
    const chunk = ids.slice(i, i + IN_CHUNK);
    const res = await db.batch([db.prepare(`DELETE FROM enrollments WHERE passport_id = ? AND image_id IN (${chunk.map(() => "?").join(", ")})`).bind(passportId, ...chunk)]);

/** I65-FX-FACE R7 (D-1576): a review row made by the time the passport's consents were last withdrawn (the held DELETE
 * /enrollments revoked them all at once). Binds the passport id once. */
const REVIEWS_KEPT_BY_DELETE_SQL = "created_at <= (SELECT MAX(wc.revoked_at) FROM consents wc WHERE wc.passport_id = ? AND wc.revoked = 1)";

/** I65-FX-FACE R5: every passport a face-data erasure is owed for: a withdrawn enrolment. */
const OWED_PASSPORTS_SQL = `SELECT e.passport_id AS pid FROM enrollments e WHERE NOT ${UNREVOKED_CONSENT}`;

/** I65-FX-FACE R5: a run's start key in passport-id space, from the run clock (any start is fair; the clock varies it). */
export function faceEraseRunStartKey(nowMs: number): string {
  let a = Math.floor(nowMs) >>> 0;
  let b = Math.floor(nowMs / 4294967296) >>> 0;
  a = Math.imul(a ^ (a >>> 16), 0x7feb352d) >>> 0;
  b = Math.imul((b ^ a) ^ ((b ^ a) >>> 15), 0x846ca68b) >>> 0;
  return `psp_${a.toString(16).padStart(8, "0")}${b.toString(16).padStart(8, "0")}`;
}

/** Passports per cron run (control: sweepOwedFaceErasures). */
export const FACE_ERASE_SWEEP_LIMIT = 50;

/**
 * I65-IMAGE (W0-13, amendment 8): the durable retry. Every passport with a withdrawn enrolment still present is a face-data
 * erasure owed by a door: a hold kept it, trust-core could not be asked, or the vector store failed. Each run asks again
 * for at most FACE_ERASE_SWEEP_LIMIT of them; "unheld" erases (eraseWithdrawnFaceData), "held" leaves it (trust-core's
 * image:face_erase_owed is written once, by the ask), no answer leaves it for the next run. Idempotent.
 */
export async function sweepOwedFaceErasures(
  db: DbLike,
  vectors: VectorStore,
  ask: FaceHoldAsk | null,
  limit = FACE_ERASE_SWEEP_LIMIT,
  startKey: string = faceEraseRunStartKey(Date.now()), // I65-FX-FACE R5: where this run starts in passport order, wrapping
): Promise<{ asked: number; erased: number; kept: number; more: boolean }> {
  const cap = Math.max(1, Math.min(500, Math.floor(limit)));
  // I65-FX-FACE R5 (I65-RG-W1-01, D-1575 (5)): a held (or unanswered) passport takes no erase slot, so `cap` held ones
  // cannot starve an unheld one; at most 2*cap asks per run; and the walk starts at a per-run key and wraps, so more held
  // passports than that are passed over in turn, never the same ones first every run.
  const window = 2 * cap;
  const from = (await db.prepare(`SELECT DISTINCT pid FROM (${OWED_PASSPORTS_SQL}) WHERE pid >= ? ORDER BY pid ASC LIMIT ?`).bind(startKey, window + 1).all<{ pid: string }>()).results;
  const wrap = (await db.prepare(`SELECT DISTINCT pid FROM (${OWED_PASSPORTS_SQL}) WHERE pid < ? ORDER BY pid ASC LIMIT ?`).bind(startKey, window + 1 - from.length).all<{ pid: string }>()).results;
  const rows = [...from, ...wrap];
  let asked = 0;
  let slots = 0;
  let erased = 0;
  let kept = 0;
  for (const { pid } of rows.slice(0, window)) {
    if (slots >= cap) break;
    asked++;
    if (!(await faceEraseUnheld(ask, pid))) {
      kept++;
      continue;
    }
    slots++;
    try {
      erased += await eraseWithdrawnFaceData(db, vectors, pid);
    } catch {
      kept++; // the rows stay; the next run retries
    }
  }
  return { asked, erased, kept, more: rows.length > asked };
}

  bind(...values: unknown[]): DbStatement { return new Stmt(this.s.bind(...values)); }

/** The decode/embed Container — a service binding (`EMBEDDER`) or an HTTPS URL

export function fingerprinterFromEnv(env: Record<string, unknown>): Fingerprinter | null {
  const svc = env.EMBEDDER as Fetcher | undefined;
  const url = typeof env.EMBEDDER_URL === "string" && /^https:\/\//i.test(env.EMBEDDER_URL) ? env.EMBEDDER_URL : "";
  if (!svc && !url) return null;
  const secret = typeof env.EMBEDDER_SHARED_SECRET === "string" ? env.EMBEDDER_SHARED_SECRET : "";
  return {
    async fingerprint(bytes: Uint8Array): Promise<RawFingerprint> {
      const doFetch = svc ? svc.fetch.bind(svc) : fetch;
      const target = svc ? "https://embedder.internal/fingerprint" : `${url.replace(/\/$/, "")}/fingerprint`;
      const headers: Record<string, string> = { "content-type": "application/octet-stream" };
      if (secret) headers["X-Embedder-Secret"] = secret;
      let res: Response;
      try {
        res = await doFetch(target, { method: "POST", headers, body: bytes, redirect: "manual", signal: AbortSignal.timeout(EMBEDDER_FETCH_TIMEOUT_MS) }); // #611: a 3xx is !ok
      } catch (e) {
        // Timed out, aborted, or unreachable — a refusal like any other, not a 500.
        throw new FingerprintError("embedder unreachable or timed out", 504);
      }
      if (!res.ok) throw new FingerprintError(`fingerprint status ${res.status}`, res.status);
      const j = (await res.json()) as { exact_hash?: unknown; gray9x8?: unknown; gray9x8_legacy?: unknown; vector?: unknown; face_found?: unknown; face_count?: unknown };
      // The Container is trusted, but its output is still shape-checked before use.
      if (typeof j.exact_hash !== "string" || !/^[0-9a-f]{64}$/.test(j.exact_hash)) throw new FingerprintError("bad exact_hash from container", 502);
      if (!isGray9x8(j.gray9x8)) throw new FingerprintError("bad gray9x8 from container", 502);
      // D-0835: the legacy (alpha-dropped) matrix is OPTIONAL — absent for an
      // opaque image — but when present it passes the SAME shape check.
      if (j.gray9x8_legacy !== undefined && !isGray9x8(j.gray9x8_legacy)) throw new FingerprintError("bad gray9x8_legacy from container", 502);
      // I3-0704: [] (no face) or exactly EMBED_DIM finite numbers (control:
      // adapters.ts isEmbedding) — a container on another model dimension is refused
      // here, before its vector can reach the index or a query.
      if (!Array.isArray(j.vector) || (j.vector.length !== 0 && !isEmbedding(j.vector))) throw new FingerprintError("bad vector from container", 502);
      return {
        exactHash: j.exact_hash,
        gray9x8: new Uint8Array(j.gray9x8 as number[]),
        ...(j.gray9x8_legacy !== undefined ? { gray9x8Legacy: new Uint8Array(j.gray9x8_legacy as number[]) } : {}),
        vector: j.vector as number[],
        faceFound: j.face_found === true,
        // I3-0831: a count is kept only when it is a non-negative integer; anything
        // else is absent, which /enroll refuses (fail closed).
        ...(typeof j.face_count === "number" && Number.isInteger(j.face_count) && j.face_count >= 0 ? { faceCount: j.face_count } : {}),
      };
    },
  };
}

/**
 * The Find-the-Creator HTTP surface.
 *
 * AUTH: a Bearer IMAGE token (I3-1350, D-0920) that trust-core mints for a live
 * session, verified with the derived IMAGE_SCOPE_KEY (`verifyImageToken`). This
 * worker never holds SESSION_SECRET, so a session token is refused here.
 * ENROLL + consent + revoke + delete are CREATOR-gated
 * (a creator acts only on their own passport, from the session). FIND is
 * CREATOR-gated (a fan token is 403), scoped to the caller's own enrolments and quota-limited (I21-Q13).
 *
 * A VALID SIGNATURE IS NOT A LIVE SESSION, and this service learned that the
 * expensive way. It used to verify the MAC, read `sub.kind`, and never read
 * `sub.ep` at all — it holds no copy of trust-core's `accounts.session_epoch` /
 * `creator_sessions`, so it could not. Every revocation lever in Dhwaja bumps
 * an epoch (account recovery's force_logout, `/account/logout-all`,
 * `/account/creator-logout-all`, passkey revocation), so every one of them
 * stopped at trust-core's edge: a hijacked creator's attacker kept enrolling
 * BIOMETRIC FACE VECTORS against the victim's passport for the remainder of the
 * 30-day session TTL, minutes after the victim was told they were locked out.
 * Every gated route now resolves its caller through `caller()` below, which
 * asks trust-core's `/internal/session-check` seam (see revocation.ts) and
 * REFUSES when it cannot get an answer — 503, never a served request. An
 * unverifiable epoch is not a valid one.
 *
 * THE FAN TIER COMES FROM THAT SAME ANSWER, live. It used to be read off a
 * `tier` claim in the token that no production call site ever stamped, so every
 * paying fan was metered as free; stamping it would have been worse (a 30-day
 * unrevocable entitlement in a worker that cannot see a cancellation). It is
 * now the `effectiveFanTier` trust-core itself reads, and a cancellation
 * reaches the quota in seconds.
 *
 * BYTES IN, FINGERPRINTS OUT OF OUR MACHINE: callers upload raw image bytes.
 * The TRUSTED Container (the Fingerprinter seam) decodes them and mints the
 * authoritative fingerprints — the caller never supplies a fingerprint, so it
 * can no longer fabricate authorship (F1). Bytes are read under a hard cap and
 * never persisted.
 *
 * DEGRADES HONESTLY: the Container is required to process an upload at all
 * (absent → 503). With the Container but no vector store, enroll is 503 and
 * find runs the EXACT stage from D1 alone; the fuzzy embedding stage runs only
 * when the vector store is wired and the Container found a face.
 */

/** Why a caller was refused. `unauthorized` covers no token, a bad token, the
 * wrong kind, AND a revoked one — one answer, exactly as trust-core gives a
 * stale credential, so a bare token holder learns nothing about the account. */
type CallerRefusal = "unauthorized" | "unavailable";
type CallerResult = { ok: true; id: string; tier: Tier } | { ok: false; reason: CallerRefusal };

/**
 * Resolve the authenticated caller of a gated route: signature, kind, AND
 * liveness, in that order.
 *
 * The liveness read is not skippable and not cacheable here — `revocation.ts`
 * owns the (seconds-long) cache and returns null for anything it could not
 * establish. Null becomes `unavailable`, which the handlers answer 503; a
 * live:false answer becomes `unauthorized`, which is the same 401 a stranger
 * gets. Fail closed in both directions.
 */
async function caller(
  c: { req: { header: (k: string) => string | undefined }; env: Record<string, unknown> },
  deps: ImageDeps,
  kind: "creator" | "fan",
  nowSec: number,
  /** I3-1712 (D-0993(1)): "fresh" on every door that WRITES (consent, withdrawal,
   * enrolment, erasure, quota-row erasure) — a just-revoked token is refused at
   * once; "cached" on reads (/find, GET /quota) — the 5 s window is accepted. */
  freshness: "fresh" | "cached",
): Promise<CallerResult> {
  const m = /^Bearer\s+(.+)$/.exec(c.req.header("Authorization") ?? "");
  if (!m) return { ok: false, reason: "unauthorized" };
  const token = m[1]!;
  const sub = await verifyImageToken(deps.imageScopeKey(c.env), token, nowSec);
  if (!sub || sub.kind !== kind) return { ok: false, reason: "unauthorized" };
  const oracle = deps.sessions(c.env);
  if (!oracle) return { ok: false, reason: "unavailable" }; // seam unconfigured → refuse, never serve unchecked
  const state = await oracle.check(token, { fresh: freshness === "fresh" });
  if (!state) return { ok: false, reason: "unavailable" }; // trust-core unreachable → refuse
  if (!state.live) return { ok: false, reason: "unauthorized" }; // epoch bumped: logged out everywhere / recovered
  return { ok: true, id: sub.id, tier: state.tier };
}

/** The refusal a `caller()` result maps to. 503 says "we could not check", which
 * is a different fact from "you may not" and must never be flattened into 401 —
 * a client that sees 401 signs the user out; a client that sees 503 retries. */
function refuse(c: { json: (b: unknown, s: 401 | 503) => Response }, reason: CallerRefusal): Response {
  return reason === "unavailable" ? c.json({ error: "session_check_unavailable" }, 503) : c.json({ error: "unauthorized" }, 401);
}

const JSON_BODY_CAP = 64 * 1024; // the control endpoints' advertised body cap

/** Read a JSON control body under the cap WITHOUT ever buffering past it.
 *
 * The cap used to be applied AFTER `req.text()`, which drains the entire stream
 * first: a chunked body that omits or understates Content-Length was fully
 * resident in the isolate before the check at the next line ever ran, so the
 * declared-length precheck was the only real bound and a caller simply omitted
 * it (I3-1344 / REV:PR-11). This is the same streaming abort `readImageBytes`
 * below has always used, at the JSON cap instead of the image cap: the reader
 * is cancelled the moment the RUNNING total exceeds it.
 *
 * The cap is counted in BYTES here, where it counted UTF-16 code units before —
 * strictly tighter for multibyte input, and the same unit the Content-Length
 * precheck above it has always used. Returns null over cap (→ the caller's 400). */
async function readJsonText(req: Request, cap: number = JSON_BODY_CAP): Promise<string | null> {
  const declared = Number(req.headers.get("content-length"));
  if (Number.isFinite(declared) && declared > cap) return null;
  const body = req.body;
  if (!body) {
    const raw = await req.text(); // no stream to bound: an absent body is empty
    return raw.length > cap ? null : raw;
  }
  const reader = body.getReader();
  const chunks: Uint8Array[] = [];
  let total = 0;
  for (;;) {
    const { done, value } = await reader.read();
    if (done) break;
    if (value) {
      total += value.length;
      if (total > cap) {
        await reader.cancel();
        return null;
      }
      chunks.push(value);
    }
  }
  const out = new Uint8Array(total);
  let off = 0;
  for (const c of chunks) {
    out.set(c, off);
    off += c.length;
  }
  return new TextDecoder().decode(out);
}

async function safeJson(req: Request): Promise<Record<string, unknown> | null> {
  const raw = await readJsonText(req);
  if (raw === null) return null;
  try {
    const v = JSON.parse(raw) as unknown;
    return v && typeof v === "object" && !Array.isArray(v) ? (v as Record<string, unknown>) : null;
  } catch {
    return null;
  }
}

/** Read the raw image body under a hard cap. Streams the body and aborts as
 * soon as the running total exceeds the cap, so a chunked / absent / understated
 * Content-Length can't force the whole (up to platform-limit) body into memory
 * before the check. Returns null on over-cap (→ 413). */
async function readImageBytes(req: Request): Promise<Uint8Array | null> {
  const declared = Number(req.headers.get("content-length"));
  if (Number.isFinite(declared) && declared > MAX_IMAGE_BYTES) return null;
  const body = req.body;
  if (!body) {
    const buf = new Uint8Array(await req.arrayBuffer());
    return buf.length > MAX_IMAGE_BYTES ? null : buf;
  }
  const reader = body.getReader();
  const chunks: Uint8Array[] = [];
  let total = 0;
  for (;;) {
    const { done, value } = await reader.read();
    if (done) break;
    if (value) {
      total += value.length;
      if (total > MAX_IMAGE_BYTES) {
        await reader.cancel();
        return null;
      }
      chunks.push(value);
    }
  }
  const out = new Uint8Array(total);
  let off = 0;
  for (const c of chunks) {
    out.set(c, off);
    off += c.length;
  }
  return out;
}

/** Map a Container refusal to a caller-facing code.
 *
 * I3-1819: this used to collapse EVERY non-413 status — a timeout, the
 * container being overloaded or unreachable, a malformed upstream response —
 * into the same 422 `image_unprocessable` a genuinely corrupt upload gets.
 * The fan's quota unit is already spent by the time this runs either way, so
 * the status code is their only remaining signal for "retry, this wasn't your
 * photo" vs. "your photo itself is the problem" — and the two were
 * indistinguishable. RETRYABLE_FINGERPRINT_STATUSES (env.ts) is the one
 * classification both this function and env.ts's own throws answer to. */
function fingerprintErrorResponse(c: { json: (b: unknown, s: 413 | 422 | 503) => Response }, e: FingerprintError): Response {
  if (e.status === 413) return c.json({ error: "image_too_large" }, 413);
  if (RETRYABLE_FINGERPRINT_STATUSES.has(e.status)) return c.json({ error: "image_processing_unavailable" }, 503);
  return c.json({ error: "image_unprocessable" }, 422);
}

/** Constant-time string compare (per-char XOR over the longer length) so the
 * internal-secret check leaks nothing through timing. */
function timingSafeEqual(a: string, b: string): boolean {
  const len = Math.max(a.length, b.length);
  let diff = a.length === b.length ? 0 : 1;
  for (let i = 0; i < len; i++) {
    diff |= (a.charCodeAt(i) || 0) ^ (b.charCodeAt(i) || 0);
  }
  return diff === 0;
}

/** I3-0027 interim guard (D-0879(2)): bound on the embedder `/health` read
 * below, so a hung Container cannot hold the readiness answer open. */
const EMBEDDER_HEALTH_TIMEOUT_MS = 3_000;

/**
 * I3-0027 interim guard (D-0879(2)): is the face embedder behind the
 * Fingerprinter seam a REAL, wired model? Asks the Container's own `GET /health`
 * (embedder/src/server.ts — `wired` is `kind === "onnx" && onnxWired()`) and
 * answers true ONLY for a 2xx whose body says `wired: true` (#882: it used to
 * read `model: "onnx"`, the embedder's NAME, which a configured-but-unwired
 * model also reports). `model: "onnx"` without `wired: true`, the stub, no
 * Container configured, a timeout, a non-2xx or an unparseable body are all false. Transport resolution is the same as
 * env.ts `fingerprinterFromEnv`: the `EMBEDDER` binding, else an https
 * `EMBEDDER_URL`. D-0911(4): `POST /find` and `POST /enroll` refuse unless it is
 * true, and `POST /internal/triage-find` runs the embedding stage only when it
 * is true (the exact-hash stage runs either way). trust-core's
 * `faceTriageArmed` reads it from `/health?ready=1` for its log line only.
 */
export async function embedderWiredFromEnv(env: Record<string, unknown>, timeoutMs: number = EMBEDDER_HEALTH_TIMEOUT_MS): Promise<boolean> {
  const svc = env.EMBEDDER as { fetch: (u: string, i?: RequestInit) => Promise<Response> } | undefined;
  const url = typeof env.EMBEDDER_URL === "string" && /^https:\/\//i.test(env.EMBEDDER_URL) ? env.EMBEDDER_URL : "";
  if (!(svc && typeof svc.fetch === "function") && !url) return false;
  const ctrl = new AbortController();
  const timer = setTimeout(() => ctrl.abort(), timeoutMs);
  const probe = (async (): Promise<boolean> => {
    const res = svc && typeof svc.fetch === "function"
      ? await svc.fetch("https://embedder.internal/health", { method: "GET", redirect: "manual", signal: ctrl.signal })
      : await fetch(`${url.replace(/\/$/, "")}/health`, { method: "GET", redirect: "manual", signal: ctrl.signal }); // #611: a 3xx reads !res.ok -> false
    if (!res.ok) return false;
    const body = (await res.json()) as { wired?: unknown } | null;
    return body?.wired === true;
  })();
  const timedOut = new Promise<boolean>((resolve) => ctrl.signal.addEventListener("abort", () => resolve(false)));
  try {
    return await Promise.race([probe, timedOut]);
  } catch {
    return false;
  } finally {
    clearTimeout(timer);
    probe.catch(() => {});
  }
}

export function makeImageApp(deps: ImageDeps): Hono<{ Bindings: Record<string, unknown> }> {
  const nowMs = deps.nowMs ?? (() => Date.now());
  const app = new Hono<{ Bindings: Record<string, unknown> }>();

  // I3-2065 (image limb; trust-core's shape, index.ts I3-1512): bare `/health`
  // stays the static LIVENESS answer. `?ready=1` is READINESS: one D1 read,
  // 503 when it fails. The other seams have no cheap read, so they are
  // reported as configured/absent only — never as a reason to answer 503.
  app.get("/health", async (c) => {
    if (c.req.query("ready") === undefined) return c.json({ ok: true, service: "image", version: "0.2.0" });
    let db = false;
    try {
      db = (await deps.db(c.env).prepare("SELECT 1 AS one").first<{ one: number }>())?.one === 1;
    } catch {
      db = false;
    }
    const checks = {
      db,
      fingerprinter: deps.fingerprinter(c.env) !== null,
      vectors: deps.vectors(c.env) !== null,
      sessions: deps.sessions(c.env) !== null,
      suppression: deps.suppression(c.env) !== null,
      likeness_peer: (deps.likenessPeer?.(c.env) ?? null) !== null,
      // I3-0027: a real (non-stub) embedder answered. Reported, never a reason
      // for 503 (a stub dev deploy is still ready). /find and /enroll refuse,
      // and triage-find skips the embedding stage, unless it is true (D-0911(4)).
      embedder_wired: await embedderWiredFromEnv(c.env),
    };
    return c.json({ ok: db, service: "image", version: "0.2.0", checks }, db ? 200 : 503);
  });

  // Creator records consent (the WS-I consent record) before enrolling. Creator
  // session gated → the consent binds to THAT passport, never a claimed one.
  app.post("/consent", async (c) => {
    // I23-Q13 (Jason, D-1030: "10 a minute"): the per-IP limit is checked FIRST, before the session read and the body.
    // An absent binding passes (env.ts consentRateLimiterFromEnv; wrangler.jsonc declares it in every block).
    const limiter = consentRateLimiterFromEnv(c.env);
    // I3-1216 (D-1078(14)): the header names the caller only on a request that came through Cloudflare (`request.cf`);
    // anything else shares one off-edge key, as trust-core's route-helpers.ts clientIp does.
    const viaEdge = (c.req.raw as { cf?: unknown }).cf != null;
    if (limiter && !(await limiter.limit({ key: viaEdge ? (c.req.header("cf-connecting-ip") ?? "") : "offedge" })).success) {
      return c.json({ error: "rate limit exceeded", message: RATE_LIMIT_MESSAGE }, 429);
    }
    const who = await caller(c, deps, "creator", Math.floor(nowMs() / 1000), "fresh");
    if (!who.ok) return refuse(c, who.reason);
    const body = await safeJson(c.req.raw);
    const ref = typeof body?.consent_ref === "string" ? body.consent_ref : "";
    if (!ref) return c.json({ error: "consent_ref required" }, 400);
    // I21-Q20 (Jason, D-1030 (1)): no consent without the over-18 declaration; nothing is written without it.
    if (body?.over_18 !== true) return c.json({ error: "over_18 declaration required" }, 400);
    // I21-Q16 (Jason, D-1030): no consent except on a current approved wording (`consent_wordings`, image 0011, ships
    // empty); unknown, draft and retired versions are refused and nothing is written.
    const version = typeof body?.wording_version === "string" ? body.wording_version : "";
    try {
      const wording = await approvedConsentWording(deps.db(c.env), version);
      if (!wording) return c.json({ error: "consent wording not approved" }, 400);
      await recordConsent(deps.db(c.env), ref, who.id, undefined, { over18Declared: true, wording });
      return c.json({ ok: true });
    } catch (e) {
      return e instanceof EnrollError ? c.json({ error: e.message }, 400) : c.json({ error: "internal" }, 500);
    }
  });

  /**
   * WITHDRAW consent to biometric processing — the door that did not exist.
   *
   * `consents.revoked` has gated five production reads since migration 0001,
   * and until now the only thing in the repo that ever set it was a test: a
   * creator who withdrew had no mechanism, and `DELETE /enrollments` (erasure)
   * left the consent record live and reusable, so the withdrawal was never
   * recorded. This is the withdrawal right; that one is the erasure right.
   *
   * SCOPED BY THE SESSION, never by the body: the passport comes from the
   * creator token, so a caller can only withdraw their OWN consents. A
   * `consent_ref` narrows it to one record; omitted, every live consent this
   * passport holds is withdrawn. A ref belonging to someone else matches no row
   * and reports `revoked: 0` — the same answer a ref that never existed gets,
   * so the count is not a probe for other creators' consent-record ids.
   *
   * IDEMPOTENT (a second call reports 0 and changes nothing) and IRREVERSIBLE
   * through this door: nothing un-revokes, and `recordConsent` refuses to reuse
   * a withdrawn ref, so re-consent means a NEW consent record. That is what
   * keeps the audit trail honest.
   *
   * IT DOES NOT REACH TRUST-CORE'S LIKENESS CONSENT. `likeness_consent_granted`
   * / `likeness_consent_revoked` (services/trust-core/src/likeness.ts) live in a
   * DIFFERENT database, gate releases and likeness generation, and make no call
   * into this service. The two "consent" systems are unconnected today; this
   * route actuates THIS service's flag and nothing else.
   */
  app.post("/consent/revoke", async (c) => {
    const who = await caller(c, deps, "creator", Math.floor(nowMs() / 1000), "fresh");
    if (!who.ok) return refuse(c, who.reason);
    // THE BODY IS PARSED STRICTLY HERE, and not through `safeJson`, because
    // withdrawal is IRREVERSIBLE and "no consent_ref" means "withdraw
    // EVERYTHING". `safeJson` answers null for an empty body, a truncated body
    // and an oversized body alike, so reading its null as "all" would turn a
    // client's truncated `{"consent_ref": "x` into a full, permanent withdrawal
    // of every consent the creator holds. Only an ABSENT body means all; a
    // present body that does not parse, or names a consent_ref that is not a
    // non-empty string, is a 400 and changes nothing.
    // Same cap `safeJson` applies, through the same streaming reader
    // (`readJsonText`): a declared over-length is refused before a byte is
    // read, and an absent or understated Content-Length is refused mid-stream
    // rather than after the whole body is resident (I3-1344). Null is over-cap
    // ONLY — an absent body reads as "" and still means "withdraw everything".
    const raw = await readJsonText(c.req.raw);
    if (raw === null) return c.json({ error: "payload too large" }, 400);
    let ref: string | undefined;
    if (raw.trim() !== "") {
      let parsed: unknown;
      try {
        parsed = JSON.parse(raw) as unknown;
      } catch {
        return c.json({ error: "invalid json" }, 400);
      }
      if (typeof parsed !== "object" || parsed === null || Array.isArray(parsed)) return c.json({ error: "invalid json" }, 400);
      const given = (parsed as Record<string, unknown>).consent_ref;
      if (given !== undefined && (typeof given !== "string" || given === "")) return c.json({ error: "consent_ref must be a non-empty string" }, 400);
      if (typeof given === "string") ref = given;
    }
    try {
      // I3-0138 (INVAUDIT item 1, 2026-09-20). This call used to pass three arguments, so `revokeConsents`'s VectorStore parameter was `undefined`
      // and the biometric vectors survived every withdrawal — the erasure was
      // built, proven as a unit, and never reached by the door. It also dropped `erased` from the response, so wiring the store alone would have made
      // the erasure real and still reported nothing.
      //
      // The store being unconfigured does NOT refuse the withdrawal the way
      // `/enroll` refuses an enrollment (:339). Refusing an enrollment protects
      // the index; refusing a WITHDRAWAL would leave the person unable to
      // withdraw at all, which is the worse failure for the only party this
      // route exists to serve. Instead the withdrawal proceeds and the residue
      // is NAMED: `vectors_pending` is true while enrollment rows still name
      // vectors no erase has removed (no store bound, or its erase failed; the
      // rows stay for the retry, I3-0138 D-1082(14)), including on a repeat
      // withdrawal that revokes nothing new, so a partial erasure is never
      // reported as a whole one (D-0570).
      const store = deps.vectors(c.env) ?? undefined;
      const r = await revokeConsents(deps.db(c.env), who.id, ref, undefined, store, deps.likenessPeer?.(c.env) ?? undefined, deps.faceHoldAsk ? deps.faceHoldAsk(c.env) : undefined); // I65-IMAGE: held or no answer -> vectors_pending
      // JUDGE-W309A: derived from what the erasure actually DID, not from
      // whether a store object was handed in. `VectorStore` does not declare
      // `deleteByIds`, so "a store was passed" was never evidence the vectors
      // went — and a partial erasure reported as a whole one is the one
      // outcome D-0570 ruled unacceptable.
      // I3-0603: `likeness_pending` names a trust-core revoke still owed (the
      // peer failed; the outbox sweep retries it) — never a failed withdrawal.
      return c.json({ ok: true, revoked: r.revoked, erased: r.erased, vectors_pending: !r.vectorsErased, likeness_pending: r.likenessSynced === false });
    } catch (e) {
      return e instanceof EnrollError ? c.json({ error: e.message }, 400) : c.json({ error: "internal" }, 500);
    }
  });

  // Enroll one consented image. The creator uploads the IMAGE BYTES; the trusted

/**

 * before matching, so the gate decodes too (repeatedly, so a double-encoded
 * `%2566ind` fails closed rather than open) and drops trailing slashes.
 */
function routedPath(pathname: string): string {
  let p = pathname;
  for (let i = 0; i < 5 && p.includes("%"); i++) {
    let next: string;
    try {
      next = decodeURIComponent(p);
    } catch {
      break;
    }
    if (next === p) break;
    p = next;
  }
  return p.length > 1 ? p.replace(/\/+$/, "") : p;
}

/**
 * Perceptual hash (dHash) — a PURE function over a normalized grayscale matrix.
 * The Container decodes + resizes the raw image to a 9x8 grayscale luminance
 * matrix (that decode is the platform primitive, like sha256/fetch elsewhere);
 * dHash then compares horizontally-adjacent pixels to a 64-bit hash. Small
 * crops/recompressions produce a small Hamming distance, so the cascade's phash
 * stage catches near-duplicates the exact hash misses, before the expensive
 * embedding stage.
 *
 * Format: input is 9 columns x 8 rows = 72 luminance bytes, row-major. Each of
 * the 8 rows yields 8 bits (col[i] < col[i+1]). Bit order: row 0 first, MSB is
 * the first comparison. Returned as 16-hex-char (64-bit) string.
 */

export const DHASH_COLS = 9;
export const DHASH_ROWS = 8;
export const DHASH_LEN = DHASH_COLS * DHASH_ROWS; // 72

export function dHash(gray: Uint8Array | number[]): string {
  if (gray.length !== DHASH_LEN) {
    throw new Error(`dHash expects ${DHASH_LEN} grayscale values (9x8), got ${gray.length}`);
  }
  const bits: number[] = [];
  for (let row = 0; row < DHASH_ROWS; row++) {
    for (let col = 0; col < DHASH_COLS - 1; col++) {
      const a = gray[row * DHASH_COLS + col]!;
      const b = gray[row * DHASH_COLS + col + 1]!;
      bits.push(a < b ? 1 : 0);
    }
  }
  // 64 bits → 16 hex chars.
  let hex = "";
  for (let i = 0; i < 64; i += 4) {
    const nibble = (bits[i]! << 3) | (bits[i + 1]! << 2) | (bits[i + 2]! << 1) | bits[i + 3]!;
    hex += nibble.toString(16);
  }
  return hex;
}

const POPCOUNT = new Uint8Array(256);
for (let i = 0; i < 256; i++) POPCOUNT[i] = (i & 1) + POPCOUNT[i >> 1]!;

/** Hamming distance between two 64-bit hex dHashes (0..64). */
export function hamming(a: string, b: string): number {
  if (a.length !== 16 || b.length !== 16 || !/^[0-9a-f]+$/.test(a) || !/^[0-9a-f]+$/.test(b)) {
    throw new Error("hamming expects two 16-hex-char dHashes");
  }
  let dist = 0;
  for (let i = 0; i < 16; i++) {
    dist += POPCOUNT[parseInt(a[i]!, 16) ^ parseInt(b[i]!, 16)]!;
  }
  return dist;
}

import type { DbLike } from "./db.js";

/**
 * Per-account daily quota for Find-the-Creator queries. A query consumes one
 * unit BEFORE the search runs; when the daily allowance is spent the query is
 * refused. Tier comes from the account record (not a client claim).
 */

export type Tier = "free" | "premium";

export const DAILY_LIMIT: Record<Tier, number> = {
  free: 20,
  premium: 100,
};

/** The quota window. Also the retention bound of `pruneExpiredQuota` below. */
export const QUOTA_WINDOW_MS = 24 * 60 * 60 * 1000;
const DAY_MS = QUOTA_WINDOW_MS;

export class QuotaError extends Error {
  override name = "QuotaError";
}

export interface QuotaState {
  used: number;
  limit: number;
  remaining: number;
}

function limitFor(tier: Tier): number {
  const limit = DAILY_LIMIT[tier];
  // Reject an unknown tier — an undefined limit would otherwise DISABLE the
  // quota entirely (count >= undefined is always false). Fail closed.
  if (typeof limit !== "number") throw new QuotaError(`unknown tier: ${String(tier)}`);
  return limit;
}

/**
 * Consume one query unit, or throw QuotaError if exhausted. Atomic: the
 * increment is a single guarded UPDATE (`WHERE count < limit`) whose
 * rows-affected gates the consume, so two concurrent queries at the last unit
 * cannot both succeed. `nowMs` MUST be a server clock (never client-supplied)
 * and `tier` MUST come from the account record — the HTTP layer enforces both.
 */
export async function consumeQuota(db: DbLike, accountId: string, tier: Tier, nowMs: number): Promise<QuotaState> {
  const limit = limitFor(tier);
  const nowIso = new Date(nowMs).toISOString();
  const cutoffIso = new Date(nowMs - DAY_MS).toISOString();

  // Window reset AND guarded increment in ONE statement — no check-then-act
  // window at the day boundary (two concurrent resets could otherwise each
  // clobber the other to count 1 and overdraw). window_start is ISO-8601
  // (lexicographically ordered), so "expired" is `window_start < cutoffIso`.
  // Fresh row → count 1; expired window → reset to 1; otherwise increment while
  // under the limit, else the upsert-WHERE matches no row → changes 0 → denied.
  const res = await db
    .prepare(
      `INSERT INTO image_quota (account_id, window_start, count) VALUES (?, ?, 1)
       ON CONFLICT(account_id) DO UPDATE SET
         window_start = CASE WHEN image_quota.window_start < ? THEN ? ELSE image_quota.window_start END,
         count        = CASE WHEN image_quota.window_start < ? THEN 1  ELSE image_quota.count + 1 END
       WHERE image_quota.window_start < ? OR image_quota.count < ?`,
    )
    .bind(accountId, nowIso, cutoffIso, nowIso, cutoffIso, cutoffIso, limit)
    .run();
  if (res.changes === 0) throw new QuotaError(`daily image-search limit reached (${limit})`);
  const after = await db.prepare("SELECT count FROM image_quota WHERE account_id = ?").bind(accountId).first<{ count: number }>();
  const used = after?.count ?? limit;
  return { used, limit, remaining: Math.max(0, limit - used) };
}

/** Read-only quota state (no consumption). */
export async function quotaState(db: DbLike, accountId: string, tier: Tier, nowMs: number): Promise<QuotaState> {
  const limit = limitFor(tier);
  const row = await db.prepare("SELECT window_start, count FROM image_quota WHERE account_id = ?").bind(accountId).first<{ window_start: string; count: number }>();
  if (!row || nowMs - Date.parse(row.window_start) >= DAY_MS) return { used: 0, limit, remaining: limit };
  return { used: row.count, limit, remaining: Math.max(0, limit - row.count) };
}

export interface QuotaEraseResult {
  removed: number;
  /** Set when a LIVE row was kept: the instant its window expires (after which
   * the hourly sweep deletes it). Absent when nothing is kept. */
  retained_until?: string;
}

/**
 * I3-0022 (D-0819(6)): the fan's own erasure of their `image_quota` row. The row
 * IS the counted usage of the current window, so a LIVE row is never deleted —
 * deleting it made consumeQuota open a fresh window (a reset of the daily limit).
 * One statement deletes the row only when its window has expired under
 * consumeQuota's own strict predicate (`window_start < nowMs - QUOTA_WINDOW_MS`),
 * so a row a concurrent consume just reset is never deleted; a kept row is
 * reported with the instant it expires, and pruneExpiredQuota deletes it then.
 */
export async function eraseQuotaRow(db: DbLike, accountId: string, nowMs: number): Promise<QuotaEraseResult> {
  if (!Number.isFinite(nowMs)) throw new QuotaError(`invalid erase clock: ${String(nowMs)}`);
  const cutoffIso = new Date(nowMs - QUOTA_WINDOW_MS).toISOString();
  const r = await db.prepare("DELETE FROM image_quota WHERE account_id = ? AND window_start < ?").bind(accountId, cutoffIso).run();
  if (r.changes > 0) return { removed: r.changes };
  const live = await db.prepare("SELECT window_start FROM image_quota WHERE account_id = ?").bind(accountId).first<{ window_start: string }>();
  if (!live) return { removed: 0 };
  return { removed: 0, retained_until: new Date(Date.parse(live.window_start) + QUOTA_WINDOW_MS).toISOString() };
}

/** Rows deleted per statement, and statements per run: one run deletes at most
 * QUOTA_SWEEP_BATCH * QUOTA_SWEEP_MAX_BATCHES rows; the rest wait for the next. */
export const QUOTA_SWEEP_BATCH = 500;
export const QUOTA_SWEEP_MAX_BATCHES = 20;

export interface QuotaSweepResult {
  deleted: number;
  batches: number;
  /** true when the run stopped at its bound with expired rows possibly left. */
  more: boolean;
}

/**
 * QUEUE #659 / #130 I-24 (D-0817(9)): scheduled retention for `image_quota`.
 * Deletes exactly the rows whose window has EXPIRED under consumeQuota's own
 * predicate (`window_start < nowMs - QUOTA_WINDOW_MS`, strict, as above). Such a
 * row is already dead to every reader: consumeQuota resets it to count 1 exactly
 * as it inserts a missing row, and quotaState reports used 0 for both — so the
 * delete is unobservable (test/b28-quota-retention.test.ts proves it by reading
 * a swept and an unswept copy). The outer `window_start < ?` re-checks inside the
 * same statement, so a row a concurrent consume just reset is never deleted.
 * `nowMs` is the cron's scheduledTime (server clock), never a caller value.
 */
export async function pruneExpiredQuota(
  db: DbLike,
  nowMs: number,
  opts: { batch?: number; maxBatches?: number } = {},
): Promise<QuotaSweepResult> {
  const batch = opts.batch ?? QUOTA_SWEEP_BATCH;
  const maxBatches = opts.maxBatches ?? QUOTA_SWEEP_MAX_BATCHES;
  if (!(Number.isInteger(batch) && batch > 0 && Number.isInteger(maxBatches) && maxBatches > 0)) {
    throw new QuotaError(`invalid sweep bound: batch=${batch} maxBatches=${maxBatches}`);
  }
  if (!Number.isFinite(nowMs)) throw new QuotaError(`invalid sweep clock: ${String(nowMs)}`);
  const cutoffIso = new Date(nowMs - QUOTA_WINDOW_MS).toISOString();
  let deleted = 0;
  for (let i = 1; i <= maxBatches; i++) {
    const r = await db
      .prepare(
        `DELETE FROM image_quota
         WHERE account_id IN (SELECT account_id FROM image_quota WHERE window_start < ? LIMIT ?)
           AND window_start < ?`,
      )
      .bind(cutoffIso, batch, cutoffIso)
      .run();
    deleted += r.changes;
    if (r.changes < batch) return { deleted, batches: i, more: false };
  }
  return { deleted, batches: maxBatches, more: true };
}

 * enrollImage (enroll.ts) records an intent (`vector_intents`, migration 0006) before the enrolment ROW, writes the

 * grace period therefore names an INTERRUPTED attempt, and the Vectorize index — which cannot be listed — is never
 * scanned: this pass works only from those ids.

 *     id (`deleteByIds` of an absent id is a no-op), so no embedding outlives its row.        "vector with no row"
 *   - a row whose consent is WITHDRAWN (a withdrawal whose vector erase failed keeps its rows to name the vectors):
 *     the vector is erased by id, then the row and the intent go. D-1024(1): an erased enrolment's vector is always
 *     removed — whatever the inventory says, so no inventory is needed for it.                     "withdrawn row"

 *     row is NEVER deleted. The photo is never stored, so the vector cannot be rebuilt here: the debt is recorded in
 *     `vector_reembed` (migration 0008) and reembed.ts rebuilds it from the next upload of the same bytes.
 *     "row with no vector" (I28-IMAGE M4; this pass REMOVED the row until then)
 *   - a live row and its vector: consistent; the intent (and any re-embed debt) is dropped.
 * A `vector_reembed` row whose enrolment is gone (an erasure took it) is purged on every run.
 * An id whose check could not be made (no inventory, or a store call threw) keeps its intent for the next run.

 * in-flight enrolment, and Vectorize's eventual consistency after an upsert, out of the pass.

export const RECONCILE_GRACE_MS = 60 * 60 * 1000;
export const RECONCILE_LIMIT = 100;

const marks = (ids: string[]): string => ids.map(() => "?").join(", ");

export interface ReconcileResult {

  vectors_erased: number;

  rows_removed: number;

export async function reconcileVectors(
  db: DbLike,
  vectors: VectorStore | null,
  inventory: VectorInventory | null,
  nowMs: number,
  /** I65-IMAGE (D-1569, W0-12): `holdAsk` set (index.ts scheduled) = a withdrawn row is erased only for an "unheld" passport. */
  opts: { graceMs?: number; limit?: number; holdAsk?: FaceHoldAsk | null } = {},
): Promise<ReconcileResult> {
  const limit = Math.max(1, Math.min(500, Math.floor(opts.limit ?? RECONCILE_LIMIT)));
  const cutoff = new Date(nowMs - (opts.graceMs ?? RECONCILE_GRACE_MS)).toISOString();
  const out: ReconcileResult = { checked: 0, vectors_erased: 0, rows_removed: 0, reembed_owed: 0, consistent: 0, deferred: 0, more: false };
  // A re-embed debt whose enrolment is gone (an erasure took the row) is dropped: it names nothing any more.
  await db.prepare("DELETE FROM vector_reembed WHERE image_id NOT IN (SELECT image_id FROM enrollments)").run();
  if (!vectors) return out; // no index bound: nothing can be checked or erased, every intent waits
  const intents = (
    await db.prepare("SELECT image_id FROM vector_intents WHERE created_at <= ? ORDER BY created_at ASC, image_id ASC LIMIT ?").bind(cutoff, limit + 1).all<{ image_id: string }>()
  ).results.map((r) => r.image_id);
  out.more = intents.length > limit;
  const ids = intents.slice(0, limit);
  out.checked = ids.length;
  if (!ids.length) return out;
  // live: image_id -> whether the row's consent is unrevoked (UNREVOKED_CONSENT, owners.ts); absent = no row.
  const live = new Map<string, boolean>();
  const owner = new Map<string, string>();
  for (const c of chunks(ids)) {
    const got = await db.prepare(`SELECT e.image_id, e.passport_id, ${UNREVOKED_CONSENT} AS live FROM enrollments e WHERE e.image_id IN (${marks(c)})`).bind(...c).all<{ image_id: string; passport_id: string; live: number }>();
    for (const r of got.results) {
      live.set(r.image_id, r.live === 1);
      owner.set(r.image_id, r.passport_id);
    }
  }
  const done: string[] = [];
  const noRow = ids.filter((id) => !live.has(id));
  if (noRow.length) {
    try {
      await vectors.deleteByIds(noRow);
      out.vectors_erased += noRow.length;
      done.push(...noRow);
    } catch {
      out.deferred += noRow.length;
    }
  }
  // D-1024(1): a withdrawn enrolment's vector is always removed. VECTORS BEFORE ROWS (I3-0119).
  let withdrawn = ids.filter((id) => live.get(id) === false);
  // I65-IMAGE (Jason D-1569, W0-12): a withdrawn row is a creator's face data. Its passport is asked first; a hold or no
  // answer keeps the row and its intent (deferred), and enroll.ts sweepOwedFaceErasures erases it once trust-core says unheld.
  if (withdrawn.length && opts.holdAsk !== undefined) {
    const unheld = new Set<string>();
    for (const p of new Set(withdrawn.map((id) => owner.get(id) ?? ""))) if (await faceEraseUnheld(opts.holdAsk, p)) unheld.add(p);
    out.deferred += withdrawn.filter((id) => !unheld.has(owner.get(id) ?? "")).length;
    withdrawn = withdrawn.filter((id) => unheld.has(owner.get(id) ?? ""));
  }
  if (withdrawn.length) {
    try {
      await vectors.deleteByIds(withdrawn);
      await db.batch(chunks(withdrawn).map((c) => db.prepare(`DELETE FROM enrollments WHERE image_id IN (${marks(c)})`).bind(...c)));
      out.vectors_erased += withdrawn.length;
      out.rows_removed += withdrawn.length;
      done.push(...withdrawn);
    } catch {
      out.deferred += withdrawn.length;
    }
  }
  const rowIds = ids.filter((id) => live.get(id) === true);
  let present: Set<string> | null = null;
  if (rowIds.length && inventory) {
    try {
      present = await inventory.existing(rowIds);
    } catch {
      present = null;
    }
  }
  if (rowIds.length && !present) out.deferred += rowIds.length;
  if (present) {
    const missing = rowIds.filter((id) => !present!.has(id));
    out.consistent += rowIds.length - missing.length;
    done.push(...rowIds.filter((id) => present!.has(id)));
    if (missing.length) {
      // D-1024(1): a LIVE enrolment is never deleted for a missing vector; the debt is recorded with the intent's drop.
      const ts = new Date(nowMs).toISOString();
      await db.batch(
        chunks(missing).flatMap((c) => [
          db.prepare("INSERT OR IGNORE INTO vector_reembed (image_id, created_at) SELECT value, ? FROM json_each(?)").bind(ts, JSON.stringify(c)),
          db.prepare(`DELETE FROM vector_intents WHERE image_id IN (${marks(c)})`).bind(...c),
        ]),
      );
      out.reembed_owed += missing.length;
    }
  }
  for (const c of chunks(done)) {
    await db.batch([db.prepare(`DELETE FROM vector_intents WHERE image_id IN (${marks(c)})`).bind(...c), db.prepare(`DELETE FROM vector_reembed WHERE image_id IN (${marks(c)})`).bind(...c)]);
  }
  return out;
}

/**
 * D-1024(1) (I28-IMAGE M4): THE RE-EMBED DOOR. reconcile.ts records a live enrolment whose vector the index lost in
 * `vector_reembed` (migration 0008) instead of deleting it. The photo is never stored, so the vector is rebuilt from the
 * next upload of the SAME bytes (same exact_hash): the creator re-enrolling it (enroll.ts) or an exact-hash find
 * (cascade.ts). Same shape as every other door into the index: the shape predicate first (I3-0704), the intent before
 * the upsert (QUEUE #205), the live row re-read after it (I3-0268), and the vector erased again when the row or its
 * consent went in the window. Returns how many enrolments were re-embedded. A throw leaves the intent and the debt:
 * the reconciler then sees either a vector (consistent, debt dropped) or none (debt kept).
 */
export async function reembedOwed(
  db: DbLike,
  vectors: VectorStore,
  of: { exactHash: string; passportId?: string },
  vector: number[],
  now: () => string = () => new Date().toISOString(),
): Promise<number> {
  if (!isEmbedding(vector) || vectorDegenerate(vector)) return 0;
  const { results } = await db
    .prepare(
      `SELECT e.image_id, e.passport_id FROM enrollments e JOIN vector_reembed r ON r.image_id = e.image_id WHERE e.exact_hash = ? AND (? IS NULL OR e.passport_id = ?) AND ${MATCH_CONSENT}`,
    )
    .bind(of.exactHash, of.passportId ?? null, of.passportId ?? null)
    .all<{ image_id: string; passport_id: string }>();
  let done = 0;
  for (const r of results) {
    await db.prepare("INSERT OR IGNORE INTO vector_intents (image_id, created_at) VALUES (?, ?)").bind(r.image_id, now()).run();
    await vectors.upsert(r.image_id, r.passport_id, vector);
    const still = await db.prepare(`SELECT 1 AS one FROM enrollments e WHERE e.image_id = ? AND ${MATCH_CONSENT}`).bind(r.image_id).first<{ one: number }>();
    if (!still) await vectors.deleteByIds([r.image_id]);
    else done++;
    await db.batch([db.prepare("DELETE FROM vector_reembed WHERE image_id = ?").bind(r.image_id), db.prepare("DELETE FROM vector_intents WHERE image_id = ?").bind(r.image_id)]);
  }
  return done;
}

import { imageSeamCredential, SESSION_CHECK_AUDIENCE } from "@dhwaja/session";
import type { Tier } from "./quota.js";

/**
 * THE REVOCATION READ — is this session still live, and what is it entitled to?
 *
 * WHY THIS FILE EXISTS. This worker verifies the same HMAC-signed tokens
 * trust-core issues, and for a long time it stopped there: `bearerSubject`
 * checked the MAC and the caller checked `sub.kind`, but NOTHING read `sub.ep`,
 * because this service has no copy of `accounts.session_epoch` /
 * `creator_sessions` and no channel to ask. Every revocation lever in Dhwaja
 * works by bumping an epoch — `recovery.ts`'s force_logout and reset_passkeys,
 * `/account/logout-all`, `/account/creator-logout-all`, `revokeFanPasskey` and
 * `revokePasskey` — so ALL of them stopped at trust-core's edge. A hijacked
 * creator whose passport had just been recovered was told they were safe while
 * the attacker's token kept opening `POST /enroll` and `POST /consent` for the
 * rest of the 30-day session TTL, enrolling BIOMETRIC FACE VECTORS against the
 * victim's passport. This module is the channel that was missing.
 *
 * THE SEAM. `POST /internal/session-check` on trust-core, reached either
 * through a service binding (`TRUST_CORE`, anything with `.fetch()`) or an
 * https origin (`TRUST_CORE_URL`) — the same two shapes, in the same order,
 * that `fingerprinterFromEnv` already resolves the embedder from, so there is
 * one convention in this service and not two. The machine credential is
 * computed from IMAGE_SCOPE_KEY (I3-1350, D-0920; landed by I16-L06): the key
 * trust-core DERIVES from its SESSION_SECRET (`deriveImageScopeKey`), so this
 * worker never holds SESSION_SECRET itself; trust-core's
 * `internalServiceCredential` computes the same value for the image-seam
 * audiences (@dhwaja/session `imageSeamCredential`).
 *
 * FAIL CLOSED IS THE INVARIANT, and it is the whole point of the module:
 *   • no transport configured, or no usable IMAGE_SCOPE_KEY → the factory
 *     returns null and every session-gated route answers 503. The image
 *     surface refuses to serve sessions it cannot check rather than serving
 *     them unchecked.
 *   • transport configured but unreachable, slow, or answering anything other
 *     than a well-formed 200 → `check()` returns null, and the caller must
 *     REFUSE. An unverifiable epoch is not a valid one. A stale-but-cached
 *     answer is never invented to paper over an outage.
 *
 * THE CACHE IS SECONDS, NOT MINUTES, and that is a correctness bound rather
 * than a tuning knob: it is the maximum time a revoked token keeps working
 * here after trust-core has killed it. Five seconds costs one extra subrequest
 * per session per five seconds and keeps the propagation delay well inside
 * "the operator clicked force-logout and it took effect". FAILURES ARE NEVER
 * CACHED — an outage must not pin a subject into a refusal (or a decision) for
 * the TTL; every request retries and every request fails closed on its own.
 *
 * The cache lives in the FACTORY's closure, not at module scope: `index.ts`
 * builds one factory per isolate (so the cache is shared across requests, which
 * is the point), while each test builds its own (so no test can be answered
 * from another test's cache).
 */

/** What the seam answers about one token. `tier` is meaningful for fan
 * sessions; a creator session always reads `free` (tier meters fan queries). */
export interface SessionState {
  live: boolean;
  tier: Tier;
}

export interface SessionOracle {
  /** The live state of `token`, or null when trust-core could not be asked.
   * Null means REFUSE — never "assume live". `fresh: true` (I3-1712, D-0993(1))
   * skips the cache READ, so a mutation door sees a revocation at once instead
   * of up to SESSION_CHECK_TTL_MS later; reads keep the cached answer. */
  check(token: string, opts?: { fresh?: boolean }): Promise<SessionState | null>;
}

/** Minimal service-binding shape (avoids coupling to a workers-types revision). */
interface Fetcher {
  fetch(input: string, init?: RequestInit): Promise<Response>;
}

/** How long one answer is reused. Seconds, deliberately — see the header. */
export const SESSION_CHECK_TTL_MS = 5_000;
/** A hung trust-core must not hold a biometric request open. A timeout is a
 * refusal like any other unreachable answer. */
const SESSION_CHECK_TIMEOUT_MS = 2_000;
/** Bound the per-isolate cache so a token flood cannot grow it without limit.
 * Eviction is insertion-order (Map semantics) — oldest entry first. */
const MAX_CACHED = 1_000;

interface CacheEntry {
  expiresAtMs: number;
  state: SessionState;
}

/**
 * Build the per-isolate oracle factory. The returned function resolves the
 * transport from a Worker env on each request and answers null when the seam
 * is not configured — which the HTTP layer turns into a 503.
 */
export function makeSessionOracleFactory(opts?: { nowMs?: () => number }): (env: Record<string, unknown>) => SessionOracle | null {
  const nowMs = opts?.nowMs ?? (() => Date.now());
  const cache = new Map<string, CacheEntry>();

  return (env: Record<string, unknown>): SessionOracle | null => {
    const svc = env.TRUST_CORE as Fetcher | undefined;
    const url = typeof env.TRUST_CORE_URL === "string" && /^https:\/\//i.test(env.TRUST_CORE_URL) ? env.TRUST_CORE_URL : "";
    if (!svc && !url) return null; // no channel to the revocation state → fail closed
    const imageKey = typeof env.IMAGE_SCOPE_KEY === "string" ? env.IMAGE_SCOPE_KEY : undefined; // I3-1350: never SESSION_SECRET

    return {
      async check(token: string, opts?: { fresh?: boolean }): Promise<SessionState | null> {
        const hit = opts?.fresh === true ? undefined : cache.get(token);
        if (hit && hit.expiresAtMs > nowMs()) return hit.state;
        // `imageSeamCredential` applies the ONE image-key predicate
        // (`usableImageScopeKey`): an absent or placeholder key returns null
        // here, so the seam is unavailable rather than unauthenticated.
        const credential = await imageSeamCredential(imageKey, SESSION_CHECK_AUDIENCE);
        if (!credential) return null;
        let res: Response;
        try {
          const doFetch = svc ? svc.fetch.bind(svc) : fetch;
          const target = svc ? "https://trust-core.internal/internal/session-check" : `${url.replace(/\/$/, "")}/internal/session-check`;
          res = await doFetch(target, {
            method: "POST",
            headers: { "content-type": "application/json", "X-Internal-Auth": credential },
            body: JSON.stringify({ token }),
            redirect: "manual", // #611: a 3xx carrying the credential is never followed; it reads status !== 200 -> null
            signal: AbortSignal.timeout(SESSION_CHECK_TIMEOUT_MS),
          });
        } catch {
          return null; // unreachable / timed out / DNS — refuse, never admit
        }
        if (res.status !== 200) return null; // 401 (credential drift), 503 (unconfigured), 5xx — all refuse
        let body: unknown;
        try {
          body = await res.json();
        } catch {
          return null;
        }
        // The answer is shape-checked before it is trusted, exactly as the
        // container's fingerprint is: a proxy or a misrouted origin returning
        // some other 200 JSON must not read as "live".
        if (typeof body !== "object" || body === null || Array.isArray(body)) return null;
        const b = body as { live?: unknown; tier?: unknown };
        if (typeof b.live !== "boolean") return null;
        if (b.tier !== "free" && b.tier !== "premium") return null;
        const state: SessionState = { live: b.live, tier: b.tier };
        if (cache.size >= MAX_CACHED) {
          const oldest = cache.keys().next();
          if (!oldest.done) cache.delete(oldest.value);
        }
        cache.set(token, { expiresAtMs: nowMs() + SESSION_CHECK_TTL_MS, state });
        return state;
      },
    };
  };
}
