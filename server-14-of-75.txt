/** I3-1785 — " (until YYYY-MM-DD)" from the server's exact cutoff, or "" when absent/unparseable (never a guessed date). */
function graceUntilSuffix(until: string | null | undefined): string {
  const ms = typeof until === "string" ? Date.parse(until) : Number.NaN;
  return Number.isFinite(ms) ? ` (until ${new Date(ms).toISOString().slice(0, 10)})` : "";
}

/**

 * states of mock 67. It lives in its OWN module on purpose.
 *
 * `views.ts` renders verdicts (checkCard, the passport views, the badge) and is
 * region-scanned by `tests/invariants/paywall.test.ts`, where exactly one region
 * (`fn billingStateCopy`) may name an entitlement. Rather than widen that
 * exemption, every entitlement-shaped read for this page is parked here: this
 * module renders no verdict, is imported by no verdict path, and is reachable
 * only from the router (`app.ts`, static copy) and the DOM controller
 * (`main.ts`, which does the fetch). Nothing here can influence what a fan is
 * told about anyone.
 *
 * Three product rules the copy encodes and must keep encoding:
 *   - PROTECTION IS NEVER PAYWALLED. Premium buys convenience, not trust. The
 *     honest block is load-bearing; softening it is a product regression.
 *   - SEEN SOONER IS NOT COUNTS FOR MORE (product decision 2026-08-15). A
 *     supporter's report is looked at earlier by an operator — queue ORDER,
 *     bounded by `PRIORITY_COMPARABILITY_STEP` in
 *     `services/trust-core/src/reports.ts`, and only among reports whose EARNED
 *     priority was already comparable. It buys no weight: not the corroboration
 *     threshold behind a Caution, not the per-reporter cap, not the priority
 *     score, not whether anything is published. The page must sell the first
 *     sentence and must never imply the second — advertising "your reports

 *   - FAIL CLOSED. Billing unconfigured (503) is a designed state with its own

  premium: { month: "US$3/mo", year: "US$30/yr", monthlyTotal: "US$36", saving: "US$6", savingPct: "17%" },
  pro: { month: "US$12/mo", year: "US$99/yr", monthlyTotal: "US$144", saving: "US$45", savingPct: "31%" },

export type PlanStateName = "signed-out" | "free" | "active" | "grace" | "lapsed" | "unavailable" | "processing";

/**
 * What the page's one button does.
 *
 * `legacy-portal` is GONE (item 14). It existed for the pre-M7 subscriber who
 * paid through the second web processor; that rail was deleted, so there is no
 * second portal to open. The shape it used to catch — paying, but with no
 * Stripe customer — did not go away though: it is now the STORE subscriber, who
 * bought Premium inside the iOS or Android app. Apple and Google own that
 * subscription and Dhwaja cannot manage it, so the page renders NO button and
 * says where to go instead. Offering Stripe checkout to them would open a
 * SECOND subscription and bill them twice, which is why `checkout` is never
 * offered on an active/grace state.
 */
export type PlanActionKind = "checkout" | "portal" | "signin" | "none";

export interface PlanAction {
  kind: PlanActionKind;
  label: string;
  /** Only meaningful for `checkout`. */
  interval: "month" | "year";
}

export interface PlanState {
  name: PlanStateName;
  /** "Premium" for a fan session, "Creator Pro" for a creator session. */
  product: "Premium" | "Creator Pro";
  tone: "quiet" | "good";
  headline: string;
  body: string;
  /** The "your protection is untouched" strip; "" when the state doesn't need it. */
  unaffected: string;
  /** Right-hand meta line (dates, price, "no reactivation fee"). */
  meta: string;
  /** The small print under the action. */
  fine: string;
  action: PlanAction;
}

/** Everything the state machine is allowed to see. Nothing else exists here. */
export interface PlanRead {
  /** HTTP status of GET /billing/entitlement. 0 = the request never completed. */
  status: number;
  ent: EntitlementDto | null;
  signedIn: boolean;
  /** Stripe's success URL carries ?upgraded=1 in the SEARCH string. */
  justUpgraded?: boolean;
  /**
   * FU-14 (S-068) — WHICH ADVERTISED INTERVAL THE FAN HAS CHOSEN. Absent means
   * ANNUAL, which is what every action on this page used to be hardcoded to.
   * The product supports both (`stripe.ts:799,844` reads a
   * STRIPE_PRICE_PREMIUM_MONTH price id; the web API sends either), this page
   * ADVERTISES both ($3/mo · $30/yr in the comparison table), and the sibling
   * creator surface already offers the choice as two radios feeding this same
   * field (`apps/creator-console/src/billing.ts:384-387`). Only the fan surface
   * had no way to pick, so the $3 plan it offers could not be bought.
   */
  interval?: "month" | "year";
}

/** The advertised price for the chosen interval — the one the button charges. */
function priceOf(price: { month: string; year: string }, interval: "month" | "year"): string {
  return interval === "month" ? price.month : price.year;
}

/** I3-0015 / D-0911(2): the "still stuck" sentence names the server-served support
 * address (`support_to`, index.ts GET /billing/entitlement) only when it is a plain
 * address; unset, absent or the operator placeholder it promises no contact. Plain text. */
function stuckLine(supportTo: unknown): string {
  const to = typeof supportTo === "string" ? supportTo.trim().slice(0, 200) : "";
  if (/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(to)) return `Still stuck after a few minutes? Contact support at ${to} with your Stripe receipt.`;
  return "Still stuck after a few minutes? Keep your Stripe receipt — no support inbox is set up yet, so there is no address to send it to.";
}

function iso(d: unknown): string {
  return typeof d === "string" ? d.slice(0, 10) : "";
}

/**
 * The whole state machine, pure. Order matters and is deliberate:
 *
 *   1. 503 — billing is switched off on this deployment. Wins over everything,
 *      including a tier we may have read a moment ago: if the surface is off we
 *      say so and offer no control (invariant 6, fail closed).
 *   2. signed out / 401 — the pricing is public, buying is not.
 *   3. any other unreadable answer — the SAME fail-closed card. Mock 67 pins
 *      this: "same card whether the keys are absent, the provider is
 *      unreachable, or an operator paused sales". Guessing "free" here would
 *      invent a state we did not read.
 *   4. in grace  → quiet grey, never red.
 *   5. paying    → active (a STORE subscriber gets no button — see PlanActionKind).
 *   6. purchased before, not now → lapsed.
 *   7. otherwise → free, which is not an error and not a trial.
 */
/** What Premium buys (I21-Q77, D-1030; H16). The one copy: the upgrade page, the web welcome note (views.ts
 * billingStateCopy), iOS `premiumBenefit` and Android `PREMIUM_BENEFIT` carry it whole (D-1059(16)). Copy approved (D-1365 W64). */
export const PREMIUM_BENEFIT =
  "You can save more creators. Your reports are seen sooner by an operator — a purchased place in the queue, never counting for more. Everything that keeps you safe is already free; Premium is a way to give back.";

export function planState(read: PlanRead): PlanState {
  const ent = read.ent;
  const pro = ent?.subject_kind === "creator";
  const product: PlanState["product"] = pro ? "Creator Pro" : "Premium";
  const price = pro ? PRICING.pro : PRICING.premium;
  // Anything that is not the literal "month" is annual: an unrecognised value
  // must never silently change what a click buys (FU-14).
  const interval: "month" | "year" = read.interval === "month" ? "month" : "year";

  if (read.status === 503) return unavailableState(product);
  if (!read.signedIn || read.status === 401) {
    return {
      name: "signed-out",
      product,
      tone: "quiet",
      headline: "Sign in to see your plan",
      body: "Prices and what each tier includes are on this page either way. You only ever need an account to buy — never to be protected.",
      unaffected: "",
      meta: "",
      fine: "Checks, verdicts and warnings work signed out, exactly as they do signed in.",
      action: { kind: "signin", label: "Sign in", interval: "year" },
    };
  }
  if (ent === null || read.status !== 200) return unavailableState(product);

  if (ent.in_grace) {
    const days = Math.max(0, Math.trunc(Number(ent.grace_days) || 0));
    const keep = `you keep every ${product} extra for ${days} more ${days === 1 ? "day" : "days"}`;
    const meta = (days === 1 ? "grace ends tomorrow" : `grace ends in ${days} days`) + graceUntilSuffix(ent.grace_until);
    // RJb-T2-AS-012 (D-0920 option b): only a Stripe customer can have had a
    // charge fail. Without one there was no charge to fail, so the card says the
    // period ended and has not been renewed — never that a payment was refused.
    if (!ent.has_customer) {
      return {
        name: "grace",
        product,
        tone: "quiet",
        headline: `Your ${product} period has ended`,
        body: `It hasn't been renewed yet. Nothing has changed so far: ${keep}. If it isn't renewed by then you go back to the free tier — no other consequence, ever.`,
        unaffected: "Either way, your protection and every verdict are untouched. Warnings, verification, proven accounts and your passport do not depend on this plan and never will.",
        meta,
        fine: "Grey, never red. A plan that ran out is an admin matter, not an accusation — and red on this page would be a lie about how serious it is.",
        action: { kind: "none", label: "", interval: "year" },
      };
    }
    return {
      name: "grace",
      product,
      tone: "quiet",
      headline: `We couldn't take this ${product} payment`,
      body: `The charge didn't go through. Nothing has changed yet: ${keep}. If it's still unpaid then you go back to the free tier — no other consequence, ever.`,
      unaffected: "Either way, your protection and every verdict are untouched. Warnings, verification, proven accounts and your passport do not depend on this payment and never will.",
      meta,
      fine: "Grey, never red. A card that expired is an admin problem, not an accusation — and red on this page would be a lie about how serious it is.",
      action: { kind: "portal", label: "Update payment ↗ Stripe", interval: "year" },
    };
  }

  if (ent.tier !== "free") {
    const through = iso(ent.paid_through);
    // RJb-T2-AS-013 / #609q (D-0920; I16-W09B H7, landed by I16-L10): a subscription the subscriber asked to cancel
    // carries `cancel_at` (entitlements-store.ts entitlementView, migration 0198). Read off the wire shape here; the
    // typed field in apps/web/src/api.ts is a hand-off.
    const endsOn = iso((ent as { cancel_at?: unknown }).cancel_at);
    // No Stripe customer + paying = bought in the app. The store holds the
    // subscription; Dhwaja can neither manage nor cancel it.
    const viaStore = !ent.has_customer;
    const extras = pro
      ? "Your caps are up: 50 likes a day, 5 named recognitions a day, 250 follows, one broadcast a day."
      // H16: sell only what actually exists. Premium's concrete perks are the
      // supporter queue position (reports seen sooner) and, since I21-Q77
      // (D-1030, I30-PREMIUM), a higher saved-creator cap (accounts.ts
      // fanSavedCap); everything protective is already the free tier. (Flair,
      // early drop access and Codex themes were advertised but not built —
      // removed rather than sold as vapor.)
      // D-1059(16): PREMIUM_BENEFIT is the one copy; iOS, Android and the web welcome note carry it whole
      // (apps/web/test/i30-fx-client.test.ts, apps/mobile-check/test/i30-fx-client.test.ts).
      : PREMIUM_BENEFIT;
    return {
      name: "active",
      product,
      tone: "good",
      headline: read.justUpgraded ? `Welcome to ${product}` : `You're on ${product}`,
      body: `${extras} Everything protective was already yours, and none of it moved.`,
      unaffected: "",
      meta: endsOn ? `Your plan ends on ${endsOn} and won't renew` : through ? `renews ${through}` : "",
      fine: viaStore
        ? `You bought ${product} inside the app, so Apple or Google holds this subscription — manage or cancel it in your App Store or Google Play subscription settings, not here. There is deliberately no button: buying again on this page would start a SECOND subscription on top of the one you already pay for.`
        : `Manage opens Stripe's portal — change the card, switch monthly/annual, download invoices, or cancel. Cancelling there leaves ${product} running until the date you already paid through.`,
      action: viaStore ? { kind: "none", label: "", interval: "year" } : { kind: "portal", label: "Manage your subscription and receipts ↗ Stripe", interval: "year" },
    };
  }

  // M10: Stripe's success URL returns the fan here with ?upgraded=1, but the
  // webhook that flips the entitlement to paid can lag behind the redirect by a
  // few seconds. In that window the entitlement still reads free, and without
  // this branch the fan who JUST PAID was shown "Go Premium" — a button that
  // would start a second checkout. Show a self-clearing "confirming…" state
  // instead (the controller re-checks the entitlement while it's up). Guarded on
  // tier === free so a confirmed payment always wins to the active card above.
  if (read.justUpgraded && ent.tier === "free") {
    return {
      name: "processing",
      product,
      tone: "quiet",
      headline: `Confirming your ${product} payment…`,
      body: `Thanks — Stripe has your payment. It can take a moment for the confirmation to reach Dhwaja; this page updates itself the instant it does, so there's nothing you need to do.`,
      unaffected: "Your protection was never waiting on this — checks, verdicts and warnings have worked the whole time.",
      meta: "usually a few seconds",
      fine: `If ${product} hasn't appeared after a minute, refresh the page. You haven't been charged twice — there's no button on this screen, so it will never quietly start a second checkout. ${stuckLine(ent.support_to)}`,
      action: { kind: "none", label: "", interval: "year" },
    };
  }

  if (ent.purchased_tier !== null && ent.purchased_tier !== undefined) {
    const through = iso(ent.paid_through);
    return {
      name: "lapsed",
      product,
      tone: "quiet",
      headline: through ? `${product} ended on ${through}` : `${product} ended`,
      body: pro
        ? "You're on the free caps again: 10 likes a day, 1 named recognition, 25 follows, one broadcast a week. Nothing you made was removed — your drops, collectibles, codes and recognitions all stand exactly as published."
        : "You're on the free tier again. Nothing you made was removed — your collection, your codes and your Codex all stand exactly as they were.",
      unaffected: "Your verification, your warnings and your proven accounts are unchanged. Money buys capacity here, never safety — so losing it costs you capacity, never safety.",
      meta: "no reactivation fee · no lost history",
      fine: `No countdown, no "you're missing out," no dimmed features dangling behind a lock. Someone who stops paying is a ${pro ? "creator" : "fan"}, full stop.`,
      action: { kind: "checkout", label: `Start ${product} again — ${priceOf(price, interval)}`, interval },
    };
  }

  return {
    name: "free",
    product,
    tone: "quiet",
    headline: "You're on Free",
    body: `Free isn't a trial and it isn't a lapsed state — it's the whole protective product. Checks, verdicts, warnings, reports, removal and your passport are yours permanently, at no cost. ${product} raises the ceilings on the tools, and nothing else.`,
    unaffected: "",
    // AS-048/O-06 (S-035): this meta sits beside the button on this card and
    // must never advertise a purchase the button can't start. That was fixed by
    // DELETING the monthly mention; FU-14 (S-068) fixes the other half — the
    // button can now start either interval, so the meta names the one that is
    // actually selected and the fan sees the charge before they leave the page.
    // It never names the OTHER interval, which is what AS-048 forbade.
    meta: `${priceOf(price, interval)} · billed ${interval === "month" ? "monthly" : "annually"}`,
    fine: "Checkout happens on Stripe, not here. You'll leave Dhwaja, pay on Stripe's page, and come back to this one. Dhwaja never sees or stores your card number.",
    action: { kind: "checkout", label: `Go ${product} — ${priceOf(price, interval)} →`, interval },
  };
}

/**
 * Invariant 6 made visible. One card for every reason billing can't run —
 * absent keys, an unreachable provider, an operator pausing sales — and NO
 * button, because a dead button reads as a bug and a queued checkout would be a
 * lie. It never suggests that anything protective is degraded, because nothing
 * protective depends on this.
 */
function unavailableState(product: PlanState["product"]): PlanState {
  return {
    name: "unavailable",
    product,
    tone: "quiet",
    headline: "Billing isn't set up yet — everything protective still works",
    body: `Payments aren't configured on this deployment, so ${product} can't be bought right now. This is the safe failure, not a broken page: when Dhwaja can't run something correctly it switches it off rather than half-run it. Your checks, your saved creators, your Codex, reports and every free feature are working normally.`,
    unaffected: "",
    meta: "",
    fine: 'Same card whether the keys are absent, the provider is unreachable, or an operator paused sales. No "try again in a moment," no queued checkout, and no card details collected in the meantime — the button is gone rather than dead.',
    action: { kind: "none", label: "", interval: "year" },
  };
}

/** The one action control for a state. `none` renders nothing at all. */
function actionHtml(state: PlanState): string {
  const a = state.action;
  if (a.kind === "none") return "";
  if (a.kind === "signin") return `<a href="#/signin" data-nav="#/signin" class="ds-btn nav primary">${xml(a.label)}</a>`;
  return `<button data-act="plan-action" data-plan-kind="${xml(a.kind)}" data-plan-interval="${xml(a.interval)}" class="ds-btn act primary">${xml(a.label)}</button>`;
}

/**
 * FU-14 — the monthly/annual choice for a card that actually sells. Checked on
 * `state.action.interval`, so a re-paint after a change reopens on the fan's own
 * choice rather than snapping back to the default.
 */
function intervalPicker(state: PlanState): string {
  const price = state.product === "Creator Pro" ? PRICING.pro : PRICING.premium;
  const on = state.action.interval;
  const opt = (value: "month" | "year", label: string, amount: string, note: string): string =>
    `<label class="pint"><input type="radio" name="tier_interval" value="${value}"${on === value ? " checked" : ""}> <span class="pl">${xml(label)}</span> <span class="amt">${xml(amount)}</span> <span class="d">${xml(note)}</span></label>`;
  return `<div class="pints" role="group" aria-label="Billing interval">${opt("month", "Monthly", price.month, "Billed every month. Cancel any time; it runs to the end of the month you paid for.")}${opt("year", "Annual", price.year, `Twelve monthly payments would be ${price.monthlyTotal}, so this saves ${price.saving} (${price.savingPct}) — paid up front, in one charge.`)}</div>`;
}

/** The state card that sits at the top of the plan page. */
export function planStateView(state: PlanState): string {
  const unaffected = state.unaffected ? `<p class="unaffected">🛡 ${xml(state.unaffected)}</p>` : "";
  // PS-103: textClamp() already escapes (packages/card-render/src/svg.ts) —
  // wrapping its output in xml() again double-escapes (V12-MANIFEST.md #11).
  const meta = state.meta ? `<span class="planmeta">${textClamp(state.meta, 120)}</span>` : "";
  const acts = actionHtml(state);
  // FU-14 — THE INTERVAL CHOICE, ON THE SELLING CARDS ONLY. A non-selling state
  // (active, grace, portal, signed-out, 503, processing) carries no checkout
  // button, and an input beside no button is a dead control — the same reason
  // those cards carry no button at all. Mirrors the creator console's
  // `pro_interval` radios; the controller re-paints on change, so the price on
  // the button can never disagree with the option the fan has selected.
  const pick = state.action.kind === "checkout" ? intervalPicker(state) : "";
  const actRow = acts || meta ? `<div class="planacts">${acts}${meta}</div>` : "";
  return `<div class="plancard s-${xml(state.name)} t-${xml(state.tone)}">
<div class="planhead">${xml(state.headline)}</div>
<p class="planbody">${xml(state.body)}</p>
${unaffected}${pick}${actRow}
<p class="planfine">${xml(state.fine)}</p>
<span data-plan-msg class="rmsg" aria-live="polite"></span>
</div>`;
}
