/**
 * THE READ SEAM, AND THE TWO THINGS IT USED TO THROW AWAY.
 *
 * It was `if (!r.ok) return null` — the status discarded and the body never
 * read — so all ~35 read failures on this console arrive at their call site as
 * one undifferentiated `null` and render as one undifferentiated sentence.
 * "Couldn't load the recovery queue" is the same words whether the seat was
 * refused (403), the server broke (500), or the request never left the browser
 * (0), and those are three different things for the operator to do next. The
 * envelope below is the one `apiPost` has returned all along; `apiGet` is the
 * form a caller uses when it needs to say WHICH failure happened, through
 * `failText`.
 *
 * `expect` IS THE CAST TURNED INTO A CHECK. `as T` is a claim about a body
 * nobody looked at: the compiler believes it and nothing executes it. W3-09
 * shipped the reversal-attribution lookup against the wrong envelope TWICE
 * behind a clean `tsc` — the screen read a key that was not there, rendered
 * `undefined` as "Nothing recovered", and typechecked both times. So a caller
 * names the top-level keys it is about to read and the seam verifies they are
 * present on the parsed body. A missing key is not an empty result and must
 * never render as one: it is this console disagreeing with the API about the
 * shape of an answer, which is a version mismatch, and `run`/`region` already
 * own that sentence (`UNDRAWABLE`). Hence the THROW — the check has to reach a
 * surface, and a `null` would have been absorbed by the very "couldn't load"
 * branch that cannot tell the operator the truth.
 */
interface ApiRead<T> {
  ok: boolean;
  status: number;
  data: T | null;
  /** Keys `expect` named that the parsed body does not carry. Empty when the
   * caller named none — an unchecked read is recorded as unchecked, never as
   * checked-and-clean. */
  missing: string[];
}

async function apiGet<T>(path: string, expect?: readonly string[]): Promise<ApiRead<T>> {
  try {
    const r = await fetch(`${OPS}${path}`, { credentials: "include", headers: { accept: "application/json" } });
    let data: T | null = null;
    // Read the body on a REFUSAL too: `/ops/*` refusals carry a written
    // sentence, and that sentence is the whole difference between "you can't do
    // this from a support seat" and "something is actually wrong".
    try { data = (await r.json()) as T; } catch { /* no body */ }
    const missing = r.ok ? missingKeys(data, expect) : [];
    return { ok: r.ok, status: r.status, data, missing };
  } catch {
    return { ok: false, status: 0, data: null, missing: [] };
  }
}

/** Which of `expect` the body does not carry. A non-object body (an Access
 * sign-in page parsed as JSON, a bare array, `null`) is missing ALL of them —
 * that is the case the `as T` cast was quietest about. */
function missingKeys(data: unknown, expect?: readonly string[]): string[] {
  if (!expect || expect.length === 0) return [];
  if (data === null || typeof data !== "object" || Array.isArray(data)) return [...expect];
  const o = data as Record<string, unknown>;
  return expect.filter((k) => !Object.prototype.hasOwnProperty.call(o, k));
}

/** The value-or-null form every read site on this console already speaks. It
 * keeps that contract on purpose — a refusal stays `null` and the call site's
 * own sentence still renders — and adds the one case a `null` cannot express:
 * a body of the wrong SHAPE throws, so it lands on `UNDRAWABLE` instead of
 * being read as an empty result. */
async function api<T>(path: string, expect?: readonly string[]): Promise<T | null> {
  const r = await apiGet<T>(path, expect);
  if (r.missing.length > 0) { envelopeMismatch(path, r.missing); return null; }
  // I25 (JN-02): a refused read keeps the call site's own sentence (`null`) and ALSO shows the server's, when it wrote one.
  const msg = refusalMessage(r.data);
  if (!r.ok && msg) toast(msg);
  return r.ok ? r.data : null;
}

/**
 * WHERE A SHAPE DISAGREEMENT SURFACES, AND WHY IT IS NOT A THROW.
 *
 * The first cut threw, so `run`/`region` would render `UNDRAWABLE` into the
 * screen that owns it — the better sentence, in the contexts where it reaches.
 * It does not reach all of them: several reads run from an `onclick` dispatched
 * as `void fn()`, where a rejection lands on nothing. An operator would press a
 * button and see NOTHING happen, which is the same class of wrong as the
 * "Nothing recovered" this whole check exists to stop. So the seam returns
 * `null` — every call site's existing `if (!d)` branch fires, in every context,
 * unchanged — and raises the standing notice, which is the only thing that can
 * say what the call site's own sentence cannot: the read did NOT fail, this
 * console and the API disagree about the shape of the answer.
 */
function envelopeMismatch(path: string, missing: string[]): void {
  setNotice("mismatch", `<div class="seatnote" role="alert"><span class="g">·</span><div>${esc(UNDRAWABLE)}</div><div class="muted mono" style="font-size:11px">${esc(path)} — missing ${esc(missing.join(", "))}</div><button class="ds-btn seatclose" data-act="dismiss-seatnote" aria-label="Dismiss">×</button></div>`);
}

async function apiPost<T>(path: string, body: unknown): Promise<{ ok: boolean; status: number; data: T | null }> {
  try {
    const r = await fetch(`${OPS}${path}`, { method: "POST", credentials: "include", headers: { "content-type": "application/json", accept: "application/json" }, body: JSON.stringify(body) });
    let data: T | null = null;
    try {
      data = (await r.json()) as T;
    } catch {
      // I3-1325: every /ops POST answers its 2xx with `c.json` (services/trust-core/src
      // has no 2xx text/empty POST answer), so a 2xx we cannot parse is an answer we
      // could not READ — never a success. Read as success, a dual-control route's
      // `{pending:true}` vanished and the first signer was told the action was done.
      if (r.ok) return unreadableReply<T>(path, r.status);
    }
    return { ok: r.ok, status: r.status, data };
  } catch {
    return { ok: false, status: 0, data: null };
  }
}

/** I3-1325: the answer `apiPost`/`apiPostRaw` give for a 2xx whose body is not JSON:
 * NOT ok, with a sentence `failText` prints, so no caller can read the missing
 * body as "not pending". The call site's own `did` sentence assumes a refusal
 * ("nothing changed"), which an unreadable 2xx does not establish, so the
 * standing notice says what is actually known — the server answered, the action
 * may or may not have run — and is the line an operator can trust. */
const UNREADABLE_REPLY = "the server answered but its reply couldn't be read — reload before acting again; a second sign-off may still be owed";
function unreadableReply<T>(path: string, status: number): { ok: false; status: number; data: T } {
  setNotice("mismatch", `<div class="seatnote" role="alert"><span class="g">·</span><div>${esc(`The server answered (${status}) but its reply couldn't be read, so this console cannot tell whether that action ran or is waiting for a second sign-off. Reload and check before pressing anything again.`)}</div><div class="muted mono" style="font-size:11px">${esc(path)}</div><button class="ds-btn seatclose" data-act="dismiss-seatnote" aria-label="Dismiss">×</button></div>`);
  return { ok: false, status, data: { error: UNREADABLE_REPLY } as unknown as T };
}

async function apiPostRaw<T>(path: string, text: string): Promise<{ ok: boolean; status: number; data: T | null }> {
  try {
    const r = await fetch(`${OPS}${path}`, { method: "POST", credentials: "include", headers: { "content-type": "text/plain", accept: "application/json" }, body: text });
    let data: T | null = null;
    try { data = (await r.json()) as T; } catch { if (r.ok) return unreadableReply<T>(path, r.status); /* I3-1325, as apiPost */ }
    return { ok: r.ok, status: r.status, data };
  } catch {
    return { ok: false, status: 0, data: null };
  }
}

async function apiDelete(path: string): Promise<{ ok: boolean; status: number; data: unknown }> {
  try {
    const r = await fetch(`${OPS}${path}`, { method: "DELETE", credentials: "include", headers: { accept: "application/json" } });
    // The body is read here for the SAME reason apiPost reads it: `/ops/*`
    // refusals carry a written sentence, and a purge that fails silently with a
    // status code is a purge an operator cannot diagnose.
    let data: unknown = null;
    try { data = await r.json(); } catch { /* no body */ }
    return { ok: r.ok, status: r.status, data };
  } catch {
    return { ok: false, status: 0, data: null };
  }
}

// Confirms for irreversible acts are rendered by the pure view modules and
// mounted here. The host is emptied (not hidden) on close so a confirm button

const modalHost = document.getElementById("modal") as HTMLElement;
const noticeHost = document.getElementById("notice") as HTMLElement;

/**
 * THE NOTICE HOST IS COMPOSED, NOT OVERWRITTEN.
 *
 * `#notice` had four unordered writers, each replacing the whole `innerHTML`:
 * the shape-disagreement notice (`envelopeMismatch`), `toast` (70 call sites in
 * this file), the seat notice in the capturing click handler, and `boot`'s
 * unconfirmed-seat banner. Whichever wrote last won, so any toast destroyed the
 * standing correction that `api<T>()` raises — leaving the operator on exactly
 * the sentence that correction exists to contradict — and one shape
 * disagreement destroyed the standing unconfirmed-seat banner, on a console
 * where an unknown tier is the whole reason `full-only` controls are not to be
 * trusted.
 *
 * The fix is the SLOT, not the four writers (FAMILY-004 P1 — the control in the
 * primitive): every writer goes through `setNotice`, the host renders the slots
 * that are set, and a writer can only ever replace ITS OWN slot. Order is
 * fixed, not arrival-ordered: a standing notice is not a queue entry, and the
 * two standing kinds outrank the transient one visually without either
 * suppressing the other. Dismissal clears the ONE notice dismissed.
 */
type NoticeKind = "mismatch" | "seat" | "transient";
const NOTICE_ORDER: readonly NoticeKind[] = ["mismatch", "seat", "transient"];
const notices = new Map<NoticeKind, string>();
function renderNotices(): void {
  // The wrapper is what makes dismissal per-notice: it is emitted HERE, by the
  // primitive, so a writer cannot forget it and a helper this file does not own
  // (`role.ts`'s `seatNotice`/`unconfirmedSeatBanner`) does not have to carry it.
  noticeHost.innerHTML = NOTICE_ORDER.filter((k) => notices.has(k))
    .map((k) => `<div data-notice="${k}">${notices.get(k)!}</div>`)
    .join("");
}
/** Set (or, with `null`, clear) one slot and re-render. The ONLY writer of
 * `noticeHost.innerHTML` outside `renderNotices` is nothing: there is no other. */
function setNotice(kind: NoticeKind, html: string | null): void {
  if (html === null) notices.delete(kind);
  else notices.set(kind, html);
  renderNotices();
}

/** Everything inside the host that can take focus, in document order. The
 * selector is the trap's POPULATION: a control this misses is a control Tab can
 * escape the dialog through, so it enumerates the focusable kinds rather than
 * the two the confirms happen to use today. */
const FOCUSABLE = 'a[href],button:not([disabled]),input:not([disabled]),select:not([disabled]),textarea:not([disabled]),[tabindex]:not([tabindex="-1"])';
function modalFocusables(): HTMLElement[] {
  return [...modalHost.querySelectorAll<HTMLElement>(FOCUSABLE)];
}

/** Where focus goes back to when the modal closes. Captured at OPEN, because by
 * close time the opener is gone from the document in the redraw cases. */
let modalReturn: HTMLElement | null = null;

/**
 * DIALOG SEMANTICS FOR EVERY CALL SITE, ESTABLISHED ONCE HERE.
 *
 * It was two lines: set `innerHTML`, focus the first `input,button`. No role,
 * no `aria-modal`, no focus trap, no focus restore — and the audit row that
 * reported this said the failure was one modal's, "unlike its sibling". The
 * sibling had the same gap. EVERY modal this console mounts comes through this
 * function (census: 13 `showModal(` call sites carrying 14 modal roots — one
 * site, the removals confirm, chooses between two), so the control belongs
 * here and not at any of them: a fix at the call sites is a fix that the next
 * modal added to this file does not get.
 *
 *  1. ROLE is NOT stamped here, and that is a decision, not an omission.
 *     Measured: all 15 modal roots in `apps/admin/src/` carry
 *     `role="alertdialog" aria-modal="true"` in their own markup — the two that
 *     did not (`main.ts:m-terminate`, `closures.ts:m-closure`) were fixed there.
 *     A fallback stamp here would have had a door count of ZERO (FAMILY-019),
 *     and a mutation proved it: swapping its role value to `region` left the
 *     guard GREEN because the branch never runs. Dead code does not defend a
 *     property; the walk in
 *     `test/i3-0427-i3-0553-rx-modal-and-envelope.test.ts` does, and it reddens
 *     when a root is added without a role. Retired explicitly rather than left
 *     noted (FAMILY-019 P2).
 *  2. TRAP. Tab and Shift+Tab wrap inside the host. Without it, Tab walks out
 *     of a destructive confirm into the page behind it, where the operator is
 *     typing into a form they cannot see and the "Cancel" they think they are
 *     on is a table row.
 *  3. RESTORE. Focus returns to whatever opened the dialog. Without it, focus
 *     falls to `body` on close and the next Tab starts at the top of the rail —
 *     an operator working a queue by keyboard loses their place on every
 *     confirm.
 */
function showModal(html: string): void {
  modalReturn = document.activeElement instanceof HTMLElement ? document.activeElement : null;
  modalHost.innerHTML = html;
  (modalFocusables()[0] ?? modalHost.querySelector<HTMLElement>("input,button"))?.focus();
}
function closeModal(): void {
  modalHost.innerHTML = "";
  const back = modalReturn;
  modalReturn = null;
  // `isConnected`: a confirm whose opener was a button in a table the handler
  // has since redrawn must not throw focus at a detached node.
  if (back && back.isConnected) back.focus();
}

// The trap. On the HOST, so it exists for every modal and dies with none of
// them. `preventDefault` is what makes it a trap rather than a hint: the
// browser's own Tab would have moved focus first.
modalHost.addEventListener("keydown", (e) => {
  if (e.key !== "Tab") return;
  const f = modalFocusables();
  if (f.length === 0) return;
  const first = f[0]!;
  const last = f[f.length - 1]!;
  const here = document.activeElement;
  e.preventDefault();
  if (e.shiftKey) (here === first ? last : f[Math.max(0, f.indexOf(here as HTMLElement) - 1)] ?? last).focus();
  else (here === last ? first : f[f.indexOf(here as HTMLElement) + 1] ?? first).focus();
});
document.addEventListener("keydown", (e) => {
  if (e.key === "Escape" && modalHost.innerHTML !== "") closeModal();
});

/**
 * UI-AUDIT F-A2 — THE ACCUSATION CONFIRM, and the two things the one it replaced
 * got wrong about the single most consequential button on this console.
 *
 * It was:
 *
 *   window.confirm("Publish this warning? It becomes a public, on-ledger
 *                   accusation. Retraction is possible but the record is
 *                   permanent.")
 *
 *  1. IT NAMED NOBODY. Every other destructive confirm in this file names the
 *     person — "Delist ${display_name} entirely?", "Rotate ${display_name}'s
 *     key?", "Force ${display_name} to sign out everywhere?" — and the one that
 *     publishes a PUBLIC ACCUSATION AGAINST A NAMED SUBJECT named nothing at
 *     all. The pipeline is a table of rows that look alike; an operator who
 *     clicks one row up publishes a permanent, on-ledger accusation against the
 *     wrong account, and this dialog gave them not one character to catch it
 *     with. The subject and the class are what the operator is actually deciding
 *     about, and they were the two things it omitted.
 *
 *  2. IT WAS FACTUALLY WRONG UNDER DUAL CONTROL. The handler's very next lines
 *     read `r.data?.pending` and answer "Recorded your sign-off (1/2). Awaiting
 *     another operator to publish." So when `approvals_required > 1` this press
 *     does NOT publish — and the dialog said "It becomes a public, on-ledger
 *     accusation," which is exactly the claim dual control exists to make
 *     false. An operator told that is either deterred from signing off at all,
 *     or — the worse direction — learns that this dialog overstates, and reads
 *     it as overstating on the day it does not.
 *
 * So it names the subject, it states the class, and it says what THIS press
 * does given the approval setting the screen has already read. Promise-based to
 * keep the call site's shape; the modal is the console's own, so the copy can
 * carry the subject as marked-up text rather than a native dialog's flat string.

/**
 * RJb-T2-AS-040 — what a dual-controlled press does, in the confirm an operator
 * reads BEFORE pressing it. Three confirms (re-list, recover a collectible,
 * reward policy) said "a second operator" unconditionally, while the server's
 * threshold is `approvals_required` (`GET /ops/settings`, the same read
 * `confirmPublish` is handed): at 1 the press takes effect immediately, and at 3
 * it is two more operators, not one. A failed read says it could not tell —
 * never a number the console did not read.
 */
async function signoffClause(effect: string, sameWhat: string): Promise<string> {
  const s = await api<OpsSettings>("/ops/settings");
  const n = typeof s?.approvals_required === "number" && Number.isInteger(s.approvals_required) ? s.approvals_required : null;
  if (n === null) return `This records YOUR sign-off. The console could not read how many operators must approve, so it cannot say whether this press ${effect} now or waits for more operators to submit ${sameWhat}.`;
  if (n <= 1) return `One operator sign-off is configured on this deployment, so this press ${effect} immediately — there is no second pair of eyes after it.`;
  return `This records YOUR sign-off; ${n} distinct operators must approve. It ${effect} only once ${n - 1 === 1 ? "another operator submits" : `${n - 1} more operators submit`} ${sameWhat}.`;
}

function confirmPublish(subject: string, wclass: string, approvalsRequired: number): Promise<boolean> {

  const whatThisDoes = dual
    ? `<p><b>This press records YOUR sign-off — it does not publish.</b> ${esc(String(approvalsRequired))} distinct operators must sign off, and the accusation goes public when the last one does. If you are the last, it publishes immediately.</p>`

  return new Promise((resolve) => {
    showModal(`<div class="scrim open" id="m-publish">
      <div class="confirm destructive" role="alertdialog" aria-modal="true" aria-labelledby="pub-h">

/**
 * I3-0555 — THE SHARED CONFIRM/PROMPT INSTRUMENT.
 *
 * 35 native `window.confirm`/`window.prompt` calls in this file bypassed the
 * console's own modal standard (`showModal`/`closeModal`, `confirmPublish`
 * above, `appealDecideConfirm`/`openTerminateConfirm`) — a native dialog is
 * dismissible by muscle memory, carries no `role="alertdialog"`, and (for
 * `window.prompt`) can silently truncate a shown-once secret. This is that
 * shared instrument, factored out of `confirmPublish`'s shape rather than
 * duplicating it a 36th time. Money or irreversible acts pass `retypeText`
 * (the id/name the operator must retype, `typedConfirmMatches`-checked, same
 * as `openTerminateConfirm`); everything else gets a plain Cancel/Confirm.
 */
let modalSeq = 0;
function confirmDialog(opts: {
  title: string;
  body: string;
  confirmLabel: string;
  tone?: "destructive" | "amber";
  retypeText?: string;
  /** I20-FXB (D-0960(12)): an optional checkbox; its state is written back to `check.checked` on confirm. */
  check?: { label: string; checked: boolean };
}): Promise<boolean> {
  const id = `m-confirm-${++modalSeq}`;
  const retype = opts.retypeText;
  const check = opts.check;
  return new Promise((resolve) => {
    showModal(`<div class="scrim open" id="${id}">
      <div class="confirm ${opts.tone ?? "destructive"}" role="alertdialog" aria-modal="true" aria-labelledby="${id}-h">
        <h3 id="${id}-h">${esc(opts.title)}</h3>
        ${opts.body}
        ${retype ? `<p><label>Type <code>${esc(retype)}</code> to confirm <input id="${id}-retype" style="width:100%;font-family:ui-monospace,Menlo,monospace" autocomplete="off" spellcheck="false"></label></p>` : ""}
        ${check ? `<p><label><input type="checkbox" id="${id}-check"${check.checked ? " checked" : ""}> ${esc(check.label)}</label></p>` : ""}
        <div class="row spaced">
          <button class="ds-btn" data-act="confirm-cancel">Cancel</button>
          <button class="ds-btn danger full-only" data-act="confirm-go">${esc(opts.confirmLabel)}</button>
        </div>
      </div>
    </div>`);
    const root = document.getElementById(id)!;
    const done = (v: boolean): void => { closeModal(); resolve(v); };
    root.querySelector<HTMLButtonElement>('button[data-act="confirm-cancel"]')!.onclick = () => done(false);
    root.querySelector<HTMLButtonElement>('button[data-act="confirm-go"]')!.onclick = () => {
      if (retype) {
        const typed = (document.getElementById(`${id}-retype`) as HTMLInputElement | null)?.value ?? "";
        if (!typedConfirmMatches(typed, retype)) { toast("Type it exactly to confirm."); return; }
      }
      if (check) check.checked = (document.getElementById(`${id}-check`) as HTMLInputElement | null)?.checked === true;
      done(true);
    };
  });
}

/** I3-0555 — the optional-reason half: an in-dialog textarea replacing
 * `window.prompt("Why? (optional)")`. Returns the trimmed text, or `null` on
 * Cancel/Escape — the same null the native prompt gave on Cancel, so callers
 * that already write `why && why.trim() ? {reason: why.trim()} : {}` need no
 * further change beyond `await`ing this in place of `window.prompt`. */
function reasonDialog(opts: { title: string; body: string; placeholder: string }): Promise<string | null> {
  const id = `m-reason-${++modalSeq}`;
  return new Promise((resolve) => {
    showModal(`<div class="scrim open" id="${id}">
      <div class="confirm" role="alertdialog" aria-modal="true" aria-labelledby="${id}-h">
        <h3 id="${id}-h">${esc(opts.title)}</h3>
        ${opts.body}
        <p><textarea id="${id}-reason" maxlength="2000" placeholder="${esc(opts.placeholder)}" style="width:100%;min-height:70px;font-family:inherit"></textarea></p>
        <div class="row spaced">
          <button class="ds-btn" data-act="reason-cancel">Cancel</button>
          <button class="ds-btn full-only" data-act="reason-go">Continue</button>
        </div>
      </div>
    </div>`);
    const root = document.getElementById(id)!;
    const done = (v: string | null): void => { closeModal(); resolve(v); };
    root.querySelector<HTMLButtonElement>('button[data-act="reason-cancel"]')!.onclick = () => done(null);
    root.querySelector<HTMLButtonElement>('button[data-act="reason-go"]')!.onclick = () => {

    };
  });

/** I3-0555 — the shown-once-secret half: a read-only input + Copy button
 * inside the house dialog, replacing `window.prompt(message, secret)` (a
 * native prompt can truncate a long value and offers no explicit copy
 * action). Resolves once the operator dismisses it; there is nothing to
 * cancel — the secret was already minted server-side. */
function secretDialog(opts: { title: string; body: string; secret: string }): Promise<void> {
  const id = `m-secret-${++modalSeq}`;
  return new Promise((resolve) => {
    showModal(`<div class="scrim open" id="${id}">
      <div class="confirm" role="alertdialog" aria-modal="true" aria-labelledby="${id}-h">
        <h3 id="${id}-h">${esc(opts.title)}</h3>
        ${opts.body}
        <p><input id="${id}-secret" readonly value="${esc(opts.secret)}" style="width:100%;font-family:ui-monospace,Menlo,monospace" onclick="this.select()"></p>
        <div class="row spaced">
          <button class="ds-btn" data-act="secret-copy">Copy</button>
          <button class="ds-btn" data-act="secret-done">Done</button>
        </div>
      </div>
    </div>`);
    const root = document.getElementById(id)!;
    const input = document.getElementById(`${id}-secret`) as HTMLInputElement | null;
    const done = (): void => { closeModal(); resolve(); };
    root.querySelector<HTMLButtonElement>('button[data-act="secret-copy"]')!.onclick = () => {
      input?.select();
      void navigator.clipboard?.writeText(opts.secret).catch(() => {});
    };
    root.querySelector<HTMLButtonElement>('button[data-act="secret-done"]')!.onclick = () => done();
  });
}

/** I3-0555 (D-0837(4)) — the remaining native-dialog call sites keep their
 * message text: the first paragraph becomes the dialog title, the rest the
 * body, every byte escaped (the message interpolates operator- and
 * creator-supplied names). */
function nativeText(message: string): { title: string; body: string } {
  const [head, ...rest] = message.split("\n\n");
  return { title: head ?? "", body: rest.map((para) => `<p style="white-space:pre-wrap">${esc(para)}</p>`).join("") };
}
/** In-page replacement for `window.confirm(message)`: true only on the go button. */
function confirmText(message: string, confirmLabel: string): Promise<boolean> {
  return confirmDialog({ ...nativeText(message), confirmLabel });
}
/** In-page replacement for `window.prompt(message)`: the trimmed text, or `null` on Cancel. */
function promptText(message: string): Promise<string | null> {
  return reasonDialog({ ...nativeText(message), placeholder: "" });
}

/** The copy a screen that could not be DRAWN gets. Distinct from a read failure
 * ("Couldn't load X"), because the two need different fixes: a read failure is
 * the API being unreachable, this is the API answering with something this build
 * cannot render. */
const UNDRAWABLE = "This screen could not be drawn. The ops API answered, but not with something this screen can read — check the API version matches this console.";


/**
 * RUN A VIEW SO THAT A THROW BECOMES A SCREEN.
 *
 * The router painted `Loading…` and then dispatched `void viewX()`, so a view
 * that threw left the rejection unhandled and the placeholder up FOREVER —
 * indistinguishable from a slow network, on the console whose own audit says an
 * operator has no other source of truth. A review then found the first fix was
 * only half of it: it took a PROMISE, and four of the views (`viewImports`,
 * `viewCreators`, `viewAccounts`, `viewTakedown`) are SYNCHRONOUS, so their
 * throws sailed past `p instanceof Promise`; and the inner region loaders and
 * tab handlers those screens dispatch were never routed through it at all,
 * leaving three regions of `#/imports` and the payout-monitor tab hung on their
 * own placeholders while the outer screen looked fine.
 *
 * So it takes a THUNK and wraps both. `where` is optional: an inner loader owns
 * one box, and writing the failure into that box rather than over the whole
 * screen is what keeps the rest of the operator's page usable.
 */
function run(f: () => void | Promise<void>, where?: string): void {
  // Only a top-level view (`where` unset) owns the whole screen and should
  // take focus; a region loader (`where` set) redraws one box inside an
  // already-focused page and must leave focus where the operator put it.
  void region(f, where).then(() => { if (where === undefined) app.focus(); });
}

/** The awaitable form. A screen with SEVERAL regions must be able to run them in
 * order without one region's failure stranding the next on its own placeholder:
 * `#/imports` draws history, metrics and roster, and a throw in the metrics half
 * left the roster below it hung forever (review, MEDIUM). Never rejects. */
async function region(f: () => void | Promise<void>, where?: string): Promise<void> {
  try {
    await f();
  } catch {
    const box = where ? document.getElementById(where) : null;
    if (box) box.innerHTML = `<p class="err">${esc(UNDRAWABLE)}</p>`;
    else fail(UNDRAWABLE);
  }
}

// Block clicks on high-liability controls when signed in as support. Capturing
// phase so it runs before the button's own handler. The server 403s these too —
// this is just so a support seat never sees a confusing failure.
document.addEventListener("click", (e) => {
  const el = e.target as HTMLElement | null;
  const dismiss = el?.closest('[data-act="dismiss-seatnote"]');
  if (dismiss) {
    // ONE notice, not the host: the dismissed notice is the one the operator
    // pressed ×  on, and a standing notice beside it is not theirs to clear.
    const kind = dismiss.closest("[data-notice]")?.getAttribute("data-notice") as NoticeKind | null | undefined;
    if (kind) setNotice(kind, null);
    else { notices.clear(); renderNotices(); }
    return;
  }
  if (ROLE === "full") return;
  const t = el?.closest(".full-only");
  if (t) {
    e.preventDefault();
    e.stopPropagation();
    // The replacement for the old `window.alert`: an inline notice that names
    // the action, names the seat, says the server agrees, and says what to do

    // An UNCONFIRMED seat gets the unconfirmed notice, not the support one:
    // telling an operator "you are signed in as SUPPORT TIER — the server
    // refuses it too" when we never learned their tier is a confident statement
    // we have no basis for, and it is the sentence that would send a full

    setNotice("seat", SEAT_CONFIRMED

  }
}, true);

/**
 * C-544: every `tr.rowlink` (8 tables — review queue, creators search,
 * seeding roster, import batches, accounts search, account detail, agencies,
 * fan reports) opened its detail on CLICK ONLY: no `tabindex`, no keyboard
 * equivalent, no role — a mouse-only control on an operator surface. ONE
 * helper carries the fix so all eight share the SAME activation rule
 * (FAMILY-004 P1: the control in the primitive, not eight hand-copied
 * tabindex/keydown pairs). `role="button"` + `tabIndex=0` make the row a
 * real focusable control; Enter and Space both activate, matching native
 * link/button semantics, and Space is prevented from also scrolling the page.
 */
function wireRowlinks(container: ParentNode, onOpen: (id: string, tr: HTMLElement) => void): void {
  container.querySelectorAll<HTMLElement>("tr.rowlink").forEach((tr) => {
    tr.style.cursor = "pointer";
    tr.tabIndex = 0;
    tr.setAttribute("role", "button");
    const activate = (): void => onOpen(tr.dataset.id ?? "", tr);
    tr.onclick = activate;
    tr.onkeydown = (e) => {
      if (e.key === "Enter" || e.key === " " || e.key === "Spacebar") {
        e.preventDefault(); // Space must not also scroll the page
        activate();
      }
    };
  });
}

/**
 * The operator side of the two-stage closure path. Most writes here go
 * through dual control on the server — suspend, terminate, lift — but
 * recording an appeal and upholding one (`console.ts:1899-1931`) are
 * deliberately single-operator: two people to acknowledge an appeal arrived
 * would make the appeal harder to file, and "upheld" changes no state. Only
 * an appeal OVERTURN runs through dual control. I3-1696: this comment used to
 * say "every write", which is not what the buttons below do.
 */
async function viewClosures(): Promise<void> {
  const m = /[?&]state=([a-z]+)/.exec(location.hash);
  // I3-1404: `suspended` is no longer a tab (it was `open` in another order); an old link lands on `open`.
  const state = m && (CLOSURE_TABS as readonly string[]).includes(m[1]!) ? m[1]! : "open";
  const d = await api<{ closures: ClosureRow[] }>(`/ops/closures?state=${state}`, ["closures"]);
  app.innerHTML = closuresScreen(d === null ? null : d.closures, state, ROLE, Date.now()) + closeAccountForm(ROLE) + creatorSuspendForm(ROLE);
  wireCreatorSuspension();

  const msg = (t: string): void => {
    const el = document.getElementById("cl-msg");
    if (el) el.textContent = t;
  };
  // I3-1380: lift/appeal/terminate fired `void act(...)` with the button still
  // enabled, so a slow connection let a second click double-submit. Disable
  // the clicked button for the life of its own request; re-enable it on
  // failure (the row is about to be redrawn by `viewClosures()` on success, so
  // there's nothing to re-enable there).
  const act = async (path: string, body: unknown, ok: string, btn?: HTMLButtonElement): Promise<void> => {
    if (btn) btn.disabled = true;
    const r = await apiPost<{ pending?: boolean; approvals?: number; required?: number }>(path, body);
    // A PENDING DUAL-CONTROL SIGN-OFF IS NOT A DONE ACT, and saying so is the
    // whole `operator-truthfulness` discipline: the console is the operator's
    // only source of truth about what happened.
    if (!r.ok) { if (btn) btn.disabled = false; toast(failText(r.status, r.data, "That didn't go through — nothing changed.")); return; }
    if (r.data?.pending) { toast(`Recorded your sign-off (${r.data.approvals ?? 1} of ${r.data.required ?? 2}). Nothing has changed yet — a second operator has to approve.`); }
    else toast(ok);
    await viewClosures();
  };

  // D-1004(4): a Lift over an OPEN appeal decides it (overturned), so it carries the decision's reason, 8-500 characters
  // (the server refuses it without one).
  app.querySelectorAll<HTMLButtonElement>('button[data-act="closure-lift"]').forEach((b) => { b.onclick = async () => {
    const url = `/ops/accounts/${encodeURIComponent(b.dataset.id ?? "")}/lift`;
    if (b.dataset.appealOpen !== "1") return void act(url, {}, "Suspension lifted.", b);
    const reason = await promptText(`Why — the reason for this decision (at least ${APPEAL_REASON_MIN} characters; required)`);
    if (reason === null) return;
    if (!appealReasonOk(reason) || reason.length > 500) { toast(`Write the reason for this decision (at least ${APPEAL_REASON_MIN} characters).`); return; }
    void act(url, { reason }, "Suspension lifted.", b);
  }; });
  app.querySelectorAll<HTMLButtonElement>('button[data-act="closure-appeal-open"]').forEach((b) => { b.onclick = () => void act(`/ops/accounts/${encodeURIComponent(b.dataset.id ?? "")}/appeal`, {}, "Appeal recorded.", b); });
  // #487a / #663: record an out-of-band repayment (typed confirm + reference).
  app.querySelectorAll<HTMLButtonElement>('button[data-act="closure-repay"]').forEach((b) => { b.onclick = () => void openRepayConfirm(b.dataset.id ?? ""); });
  // QUEUE #419: the counts are what happened at Stripe, said as counts.
  app.querySelectorAll<HTMLButtonElement>('button[data-act="closure-stripe-retry"]').forEach((b) => { b.onclick = async () => {
    b.disabled = true;
    const r = await apiPost<{ cancelled?: number; failed?: number; skipped?: number }>(`/ops/accounts/${encodeURIComponent(b.dataset.id ?? "")}/stripe-cancel/retry`, {});
    b.disabled = false;
    if (!r.ok) { toast(failText(r.status, r.data, "The retry did not run — nothing changed.")); return; }
    const d = r.data ?? {};
    toast(`Stripe: ${d.cancelled ?? 0} cancelled, ${d.failed ?? 0} still failing, ${d.skipped ?? 0} skipped (Stripe not configured). Every outcome is an audit row.`);
  }; });
  // I3-0098 — an appeal decision was one click with no confirm, beside a
  // REVERSIBLE suspend that has one. The `data-act` names stay exactly as the
  // row markup writes them; only what the click DOES changes.
  app.querySelectorAll<HTMLButtonElement>('button[data-act="closure-appeal-uphold"]').forEach((b) => { b.onclick = () => openAppealDecideConfirm(b.dataset.id ?? "", "upheld", act); });
  app.querySelectorAll<HTMLButtonElement>('button[data-act="closure-appeal-overturn"]').forEach((b) => { b.onclick = () => openAppealDecideConfirm(b.dataset.id ?? "", "overturned", act); });
  app.querySelectorAll<HTMLButtonElement>('button[data-act="closure-terminate"]').forEach((b) => { b.onclick = () => {
    const id = b.dataset.id ?? "";
    // I3-0099: terminate is IRREVERSIBLE (forfeits granted credits) while the
    // reversible suspend it follows requires retyping the account id inside a
    // modal that first reads the forfeiture preview. A single native
    // `window.confirm` — dismissible by muscle memory, and one of several
    // buttons an operator may be clicking down a list — gated the permanent
    // act more weakly than the one it follows. Route it through the same
    // considered-modal + typed-confirm + forfeiture-preview instrument.
    void openTerminateConfirm(id);
  }; });

  const open = app.querySelector<HTMLButtonElement>('button[data-act="closure-open"]');
  if (open) open.onclick = () => void openClosure();
}

/** I3-0099: the terminate confirm, gated the same way the suspend confirm
 * (openClosure/closeConfirm) already is — forfeiture preview read BEFORE the
 * modal, the account id must be retyped exactly, and a failed preview read
 * renders as unknown rather than a reassuring zero. Terminate is permanent
 * (forfeits granted credits) and must not be gated more weakly than the
 * reversible suspend that precedes it. */
async function openTerminateConfirm(id: string): Promise<void> {
  const pre = await api<{ forfeit: number; retained_cash_backed: number }>(`/ops/accounts/${encodeURIComponent(id)}/forfeiture`);
  showModal(`<div class="scrim open" id="m-terminate">
    <div class="confirm destructive" role="alertdialog" aria-modal="true" aria-labelledby="term-h">
    <h3 id="term-h">Terminate ${esc(id)} permanently?</h3>
    <p class="muted">This is <strong>not reversible</strong>. Granted credits are forfeited; purchased credits and money owed to a creator remain a debt we owe.</p>
    ${
      pre === null
        ? `<p class="muted">We couldn't read what this would forfeit — that is a read failure, not a zero.</p>`
        : `<p class="muted"><strong>${pre.forfeit}</strong> granted credits would be forfeited now, and <strong>${pre.retained_cash_backed}</strong> credits would <em>not</em> be forfeited — they stay on the ledger and both numbers are written onto the termination audit line. <strong>Nothing pays them out automatically.</strong> That balance is still a debt Dhwaja owes. ${REPAY_HOW_LINE} ${NO_PAYOUT_ADDRESS_LINE}</p>`
    }
    <p><label>Type the account id to confirm <input id="term-confirm" style="width:100%;font-family:ui-monospace,Menlo,monospace"></label></p>
    <p><button data-act="terminate-confirm" class="ds-btn danger full-only">Terminate</button> <button class="ds-btn" data-act="terminate-cancel">Cancel</button></p></div></div>`);
  const cancel = document.querySelector<HTMLButtonElement>('#m-terminate button[data-act="terminate-cancel"]');
  if (cancel) cancel.onclick = () => closeModal();
  const go = document.querySelector<HTMLButtonElement>('#m-terminate button[data-act="terminate-confirm"]');
  if (go) go.onclick = () => {
    const typed = (document.getElementById("term-confirm") as HTMLInputElement | null)?.value ?? "";
    if (!typedConfirmMatches(typed, id)) { toast("Type the account id exactly to confirm."); return; }
    closeModal();
    void (async () => {
      const r = await apiPost<{ pending?: boolean; approvals?: number; required?: number }>(`/ops/accounts/${encodeURIComponent(id)}/terminate`, { confirm: id });
      if (!r.ok) { toast(failText(r.status, r.data, "The account was not terminated.")); return; }
      if (r.data?.pending) toast(`Recorded your sign-off (${r.data.approvals ?? 1} of ${r.data.required ?? 2}). The account is NOT terminated yet — a second operator has to approve.`);
      // I3-1457: name the account and what was forfeited — "Account terminated."
      // told the operator nothing they could check against the audit line.
      else toast(`Account ${id} terminated${pre === null ? "" : ` — ${pre.forfeit} granted credits forfeited`}.`);
      await viewClosures();
    })();
  };
}

/** #487a / #663: the repayment confirm (closures.ts `repayConfirm`). The retained

  showModal(repayConfirm(id, pre === null ? null : pre.retained_cash_backed, ROLE));

  const go = document.querySelector<HTMLButtonElement>('#m-repay button[data-act="repay-confirm"]');
  if (go) go.onclick = () => {
    const typed = (document.getElementById("rp-confirm") as HTMLInputElement | null)?.value ?? "";

  };
}

function wireReadonlyToggle(): void {
  const send = async (on: boolean, btn: HTMLButtonElement): Promise<void> => {
    btn.disabled = true;
    const r = await apiPost<{ pending?: boolean; approvals?: number; required?: number; platform_readonly?: boolean }>("/ops/settings/platform-readonly", { on });

    if (!r.ok) return leverMsg("readonly-msg", failText(r.status, r.data, "That didn't go through — the platform is exactly as it was"));

    await viewConfig();
  };

  if (onBtn) onBtn.onclick = () => void (async () => {
    const ok = await confirmDialog({
      title: "Make the whole platform read-only?",

// I3-0751: "who did what to account X" is the screen's stated purpose
// ("Every operator action, newest first"), and the backend already answers it
// with exact-match `action`/`item_id` filters on `GET /ops/audit`

// whatever `AUDIT_ITEM_SHAPE` in admin.ts accepts); a malformed value 400s
// server-side and `failText` surfaces it rather than the UI guessing a shape.
let auditFilterAction = "";
let auditFilterItem = "";
// QUEUE #167 S-10: WHO. An operator's email or a stored actor id
// (`op_…`, `system:…`); the server resolves an email to that person's ids.
let auditFilterActor = "";
function auditQuery(extra: string): string {
  const params = [extra];
  if (auditFilterAction) params.push(`action=${encodeURIComponent(auditFilterAction)}`);
  if (auditFilterItem) params.push(`item_id=${encodeURIComponent(auditFilterItem)}`);
  if (auditFilterActor) params.push(`actor=${encodeURIComponent(auditFilterActor)}`);
  return `/ops/audit?${params.join("&")}`;
}

// `payload` carries the escalated report's machine-detected pointer
// (suspected_passport_id) — H19: the draft form reads it so a published
// impersonation warning can point fans at the real creator.
interface ReviewRow { item_id: string; item_type: string; state: string; subject: string; created_at: string; payload?: { suspected_passport_id?: string; distinct_reporters?: number; source?: string; earlier_provider_pid_differs?: boolean }; earlier_provider_pid_differs?: boolean | null }
interface ReviewDetail extends Omit<ReviewRow, "payload"> { payload: unknown; wording_id: string | null; second_look: number; disposition: string | null; disposition_note: string | null; updated_at: string; claim_signals?: ClaimSignals | null; oauth_identity?: { provider_pid: string | null; earlier_provider_pid_differs?: boolean | null } | null }
/** I38-OAUTHREV (INV-015): the list marker on an OAuth claim whose ceremony proved a different provider id than an
 * earlier proof of the same account or passport recorded (the review item's flag, set when it was queued). */
const OAUTH_PID_DIFFERS_LIST_NOTE = "different account id than an earlier proof";
/** I38-OAUTHREV: the detail sentence for the same case, read when the item is opened. DRAFT wording (REPORT.md ## JASON). */
const OAUTH_PID_DIFFERS_DETAIL = "The account that signed in has a different account id than an earlier proof of this account or passport recorded. The handle may have been released and registered by somebody else. Approve only if you can confirm this is the same creator.";
/** I41-ADMINUI (D-1193(4)): the list pill and the detail line when the server serves `earlier_provider_pid_differs: null`
 * (the item's own provider id is unknown, so no comparison was made). Wording approved by Jason, D-1245 (10) a. */
const OAUTH_PID_DIFFERS_UNKNOWN = "unknown whether the account id differs from an earlier proof";
/** I38-OAUTHREV: the OAuth review item's header line — the claimant proved the account by signing in, and this is the id it proved. */
function oauthClaimBlock(it: Pick<ReviewDetail, "item_type" | "state" | "payload" | "oauth_identity">): string {
  const mp = it.payload as Record<string, unknown> | undefined;
  if (it.item_type !== "claim" || it.state !== "open" || mp?.source !== "oauth_claim") return "";
  const oi = it.oauth_identity ?? null;
  const idLine = oi?.provider_pid ? `The provider account id it proved is <code>${esc(oi.provider_pid.slice(0, 128))}</code>.` : "No provider account id was recorded for this item.";
  const flag = oi?.earlier_provider_pid_differs;
  const differs = flag === true ? `<div class="ds-banner banner banner-warn" style="margin:8px 0"><strong>${esc(OAUTH_PID_DIFFERS_DETAIL)}</strong></div>` : flag === null ? `<div class="ds-banner banner" style="margin:8px 0"><strong>${esc(OAUTH_PID_DIFFERS_UNKNOWN)}</strong></div>` : "";
  return `<div class="ds-banner banner" style="margin:8px 0"><strong>OAuth claim.</strong> The claimant signed in to <code>${esc(String(mp.platform ?? "").slice(0, 40))}</code> as this account. ${idLine}</div>${differs}`;
}

/** I3-1328 (I4-05-R9 W-UI-1328): the rows already shown and the keyset `next`

let reviewItems: ReviewRow[] = [];

/** I40-LAND (I40-OPS H3, D-1169 answer 4): a held door (console.ts countersignHeldAnswer) answers 200 with
 * `countersign_pending: true` — a recorded sign-off, never a done act. Sentence approved as written (D-1193(8)). */
function heldText(ok: boolean, d: unknown): string | null {
  const x = d as { countersign_pending?: unknown; approvals?: unknown; required?: unknown } | null;
  if (!ok || x?.countersign_pending !== true) return null;
  return `Your sign-off is recorded — this is held for a second operator and has NOT happened yet. ${typeof x.approvals === "number" ? x.approvals : 1} of ${typeof x.required === "number" ? x.required : 2} required.`;
}

    // M51: a warning is keyed to an ACCOUNT subject hash (sha256:…), never a
    // passport id — a psp_ subject enqueues but can never be PUBLISHED
    // (warnings.ts), so a report filed against one is a dead end the operator
    // wouldn't discover until the publish step. Refuse it here, at intake.
    if (/^psp_/i.test(subject)) {
      if (msg) msg.textContent = "A report targets the impersonating account (its sha256:… subject hash), not a passport id — a psp_ subject can never become a published warning.";
      return;
    }
    const r = await apiPost<{ item_id: string }>("/ops/report", { subject, payload: note ? { note } : {} });
    if (msg) msg.textContent = r.ok ? `Filed ${r.data?.item_id ?? ""}.` : failText(r.status, r.data, "Couldn't file this report");
    if (r.ok) void viewReview();
  };
}

async function openReviewDetail(id: string): Promise<void> {
  const box = document.getElementById("detail");
  if (!box) return;

  // (the creator's own renewal request always does) and defaults to a year out
  // otherwise, which is the term this product has always issued.
  const payloadExpiry = typeof (it.payload as Record<string, unknown> | undefined)?.expires_at === "string" ? String((it.payload as Record<string, string>).expires_at).slice(0, 10) : "";

"").slice(0, 40))}</code> and confirm the code <code>${esc(String(mp.code).slice(0, 40))}</code> is publicly visible on the profile.       const kyc = (box.querySelector<HTMLInputElement>("#vkyc")?.value ?? "").trim();

    // when this decision needs a second sign-off (claim approvals run through
    // dual control) — this used to fall straight to "Approved." on any `r.ok`,
    // telling the operator a decision was final when it was still pending.
    if (dmsg) dmsg.textContent = !r.ok
      ? failText(r.status, r.data, "Couldn't approve this item")
      : r.data?.pending
        ? `Recorded your sign-off (${r.data.approvals ?? 1} of ${r.data.required ?? 2}). This is NOT approved yet — a second operator has to sign off.`
        : r.data?.event_type === "verification_renewed" ? "Renewed — the badge never lapsed." : "Approved.";
    // A pending sign-off leaves the item OPEN (a second operator still has to
    // act on it), so reloading the queue here would wipe the message above
    // before the operator could read it, on the one path where it matters most.
    if (r.ok && !r.data?.pending) void viewReview();
  };
  // I3-1468: `window.prompt` used to hold the reject reason.
  // `window.prompt(…) ?? ""` returns `null` on Cancel AND on Escape, and both
  // collapsed to `""`, so an operator who mis-clicked Reject and pressed
  // Escape to back out permanently rejected a real creator's claim with a
  // blank audit trail — and there is no unreject route. Worse, the native
  // prompt is not this console's UI at all: Escape (or a slow click through
  // to the confirm) silently drops every character typed, with no way back.
  // The reason now lives in an on-page `<textarea>` inside this box, the same
  // shape `enrolDecline` already uses — Escape only closes a modal
  // (`main.ts` `document`-level Escape handler), and this panel is not one,
  // so it cannot be cleared by the key that clears modals.
  const rjpanel = box.querySelector<HTMLElement>("#rjpanel");
  const rjreason = box.querySelector<HTMLTextAreaElement>("#rjreason");
  const rjCancel = box.querySelector<HTMLButtonElement>('button[data-act="reject-cancel"]');
  const rjConfirm = box.querySelector<HTMLButtonElement>('button[data-act="reject-confirm"]');
  if (rj) rj.onclick = () => {
    if (rjpanel) rjpanel.style.display = "block";
    rjreason?.focus();
  };
  if (rjCancel) rjCancel.onclick = () => {
    if (rjpanel) rjpanel.style.display = "none";
    if (rjreason) rjreason.value = "";
    if (dmsg) dmsg.textContent = "Cancelled — nothing was rejected.";
  };
  if (rjConfirm) rjConfirm.onclick = async () => {
    const reason = (rjreason?.value ?? "").trim();
    if (!reason) { if (dmsg) dmsg.textContent = "A reason is required: it is the only record of why this was refused. Nothing was rejected."; return; }
    const r = await apiPost(`/ops/review/${encodeURIComponent(id)}/reject`, { reason });
    if (dmsg) dmsg.textContent = heldText(r.ok, r.data) ?? (r.ok ? "Rejected." : failText(r.status, r.data, "Couldn't reject this item"));
    if (r.ok && !heldText(r.ok, r.data)) void viewReview();
  };
}

/**

 *

    if (r.ok) await openCreator(id, text);
    else if (crmsg) crmsg.textContent = text;
  };
  // B25-635: release a delisted creator's held balance (two operators; see delisting.ts).
  const rel = box.querySelector<HTMLButtonElement>('button[data-act="release-held"]');
  if (rel) rel.onclick = async () => {

    const text = await releaseHeldPress(id, c.display_name, (m) => confirmText(m, "Release"), apiPost);
    if (text !== null && crmsg) crmsg.textContent = text;
  };
}

function fmtRemain(ms: number): string {
  if (ms <= 0) return "elapsed";

}

/**
 * SAY WHEN THE THRESHOLD WAS REDUCED. `effectiveApprovals` caps the configured
 * requirement at the number of allowlisted people, because a 2-of-1 gate is an
 * outage wearing a control's clothes. That cap was reported only in JSON, so the
 * screen showed the reduced number as though it were the number somebody chose —
 * a safety lever silently doing less than configured, which is the failure the
 * posture object exists to prevent.
 */
function clampNote(p: ApprovalsPosture | undefined): string {
  if (!p) return "";
  // THE UNSATISFIABLE CASE FIRST, because it is the one that costs something.
  // A deployment can be reduced below its own threshold by LINKING two operators
  // into one person; nothing in the env count sees that, so this used to read
  // "healthy" while nine routes and the whole warnings pipeline were stalled.
  if (p.satisfiable === false) {
    return ` <strong class="err">This threshold cannot be met.</strong> ${p.required} sign-off${p.required === 1 ? "" : "s"} are required and this deployment has ${p.persons ?? 0} distinct ${(p.persons ?? 0) === 1 ? "person" : "people"} on record — every dual-controlled action will sit pending forever. Check the Operators table below: two ids linked into one person is the usual cause, and an operator who is not part of that person can split them.`;
  }
  if (p.clamped) return ` <strong>Asked for ${p.configured}; enforcing ${p.required}</strong> — this deployment has ${p.operators} allowlisted operator${p.operators === 1 ? "" : "s"}.`;
  return ` (${p.operators} allowlisted operator${p.operators === 1 ? "" : "s"})`;
}

/**

 *
 * An `op_` id is minted per ADDRESS. Both governance gates compare the PERSON
 * behind the id, so this table is the only place an operator can see how strong
 * their two-person control actually is. Rows marked "same mailbox" were joined by
 * a fact about mail delivery and cannot be split; the rest were joined by an
 * operator and can be.
 */
function operatorsPanel(roster: OperatorRoster | null): string {
  const ops = roster?.operators ?? [];
  if (ops.length === 0) return "";
  const groups = new Map<string, OperatorPerson[]>();
  for (const o of ops) groups.set(o.person_id, [...(groups.get(o.person_id) ?? []), o]);
  const rows = [...groups.values()]
    .map((g) => {
      const anchorId = g[0]?.op_id ?? "";
      const members = g
        .map((o) => {
          const mailbox = o.shares_mailbox ? ` <span class="pill">same mailbox</span>` : "";
          const unlink =
            g.length > 1 && !o.shares_mailbox
              ? ` <button class="ds-btn full-only" data-act="op-unlink" data-op="${esc(o.op_id)}">Split off</button>`
              : "";
          // #135E-26: the revocation lever (POST /ops/operators/revoke, dual control). A revoked id says so and offers nothing.

            : ` <button class="ds-btn full-only" data-act="op-revoke" data-op="${esc(o.op_id)}" data-label="${esc(o.label)}">Revoke…</button>` +

    if (!r.ok) { if (pipemsg) pipemsg.textContent = failText(r.status, r.data, "Couldn't publish this warning"); return; }
    if (r.data?.pending) { if (pipemsg) pipemsg.textContent = `Recorded your sign-off (${r.data.approvals}/${r.data.required}). Awaiting another operator to publish.`; return; }
    if (pipemsg) pipemsg.textContent = "Published.";
    void viewWarnings();
  }; });
  // Retraction is no longer one free-text prompt. `POST /ops/warnings/:id/retract`
  // REQUIRES a disposition from a closed set, because "we were wrong about this
  // person" and "this was a duplicate case" are different facts that three
  // separate counters read (reporter credibility, the creator's protection
  // record, the public warnings board) and used to be indistinguishable. A
  // typed-in enum is exactly the wrong control for that, so it is a select.
  box.querySelectorAll<HTMLButtonElement>('button[data-act="retract"]').forEach((b) => { b.onclick = () => showRetractForm(b.dataset.id ?? ""); });
  // I39-LAND (I39-RETRACT H6): POST /ops/warnings/:id/retract/countersign. Wording approved as written (D-1169(10)).
  box.querySelectorAll<HTMLButtonElement>('button[data-act="retract-countersign"]').forEach((b) => { b.onclick = async () => {
    if (!(await confirmText("Countersign this retraction?\n\nA different operator retracted this warning. Countersigning runs the retraction's caution suppression and screenshot purge.", "Countersign"))) return;
    const r = await apiPost<{ caution_suppressed?: boolean; evidence?: { held: boolean; purged: number } | null }>(`/ops/warnings/${encodeURIComponent(b.dataset.id ?? "")}/retract/countersign`, {});
    if (!r.ok) { if (pipemsg) pipemsg.textContent = failText(r.status, r.data, "Couldn't countersign this retraction"); return; }
    const ev = r.data?.evidence ?? null;
    // Reload first, then say it: the reload re-renders #pipemsg.
    await viewWarnings();
    const done = document.querySelector<HTMLElement>("#pipemsg");
    if (done) done.textContent = `Countersigned.${r.data?.caution_suppressed ? " Caution suppressed." : ""}${ev ? (ev.held ? " Screenshots kept: a legal hold is open." : ` ${ev.purged} screenshot(s) purged.`) : ""}`;
  }; });
}

  /** Mark, do not delete: the operator may still want to read what A said, and
   * an empty box would look like a screen that had never been dry-run. */
  const markPreviewFreshness = (): void => {
    const mark = document.getElementById("istale");
    if (!mark) return;
    mark.innerHTML = importPreview === null || previewIsCurrent()
      ? ""
      : `<span class="ds-chip chip err">STALE</span> This report is about the content that was validated, which is <b>no longer</b> what is in the box. Dry-run again before you apply.`;
  };
  if (kindSel) { kindSel.value = importKind; kindSel.onchange = () => { importKind = (kindSel.value as typeof importKind); markPreviewFreshness(); }; }
  if (ta) ta.oninput = () => markPreviewFreshness();
  if (fileIn) fileIn.onchange = async () => {
    const f = fileIn.files?.[0];
    if (!f) return;
    if (ta) ta.value = await f.text();
    markPreviewFreshness();
  };
  const msg = app.querySelector<HTMLElement>("#imsg");
  // Renamed off `run` — that name is now the module-level view wrapper.
  const runImport = async (apply: boolean) => {
    const text = ta?.value ?? "";
    const kind = importKind;
    if (!text.trim()) { if (msg) msg.textContent = "Paste or choose a JSONL file first."; return; }
    // The confirmation used to name only the kind and the ledger write, so it
    // read identically whether the displayed dry-run was about this content or
    // about something else entirely. It now states which of the three it is.
    const provenance = importPreview === null
      ? "\n\nThis content has NOT been dry-run. The screen above says to dry-run first."
      : previewIsCurrent()
        ? "\n\nThe dry-run report on screen is about exactly this content."
        : "\n\nWARNING: the dry-run report on screen is about DIFFERENT content. Nothing on this screen has validated what you are about to apply.";
    if (apply && !(await confirmText(`Apply this ${importKind} batch? Writes to the ledger. (You can retract the whole batch afterwards.)${provenance}`, "Apply batch"))) return;
    if (msg) msg.textContent = apply ? "Applying…" : "Validating…";
    // QUEUE #775 (D-0858(11)): an apply answers at most 500 records per call with
    // `complete`/`next`; re-POST the SAME text with `&batch=` until complete. A
    // batch the history marks "apply incomplete" is continued the same way
    // (`resumeApply`); the server refuses (409) a different file.
    let batch = apply && resumeApply?.kind === kind ? resumeApply.id : null;
    let sum: ImportReport | null = null;
    let calls = 0;
    let complete = false;
    while (calls < IMPORT_MAX_CALLS) {
      calls++;
      const r = await apiPostRaw<ImportReport>(`/ops/import/${kind}` + (apply ? "?apply=true" : "") + (batch ? `&batch=${encodeURIComponent(batch)}` : ""), text);
      if (!r.ok || !r.data) {
        if (msg) msg.textContent = failText(r.status, r.data, "That didn't go through") + (batch ? ` — batch ${batch} is not finished; choose the same file and press Apply to continue it.` : "");
        if (batch) resumeApply = { id: batch, kind };
        if (apply) run(loadImportHistory, "ihistory");
        return;
      }
      sum = sum
        ? { ...r.data, accepted: sum.accepted + r.data.accepted, rejected: sum.rejected + r.data.rejected, skipped: sum.skipped + r.data.skipped, records: [...sum.records, ...r.data.records] }
        : r.data;
      if (!apply || r.data.complete !== false) { complete = true; break; }
      batch = r.data.batch_id;
      if (msg) msg.textContent = `Applying batch ${batch} — ${r.data.next ?? "?"} records so far…`;
    }
    if (!sum) return;
    resumeApply = complete ? null : { id: sum.batch_id, kind };
    if (msg) msg.textContent = !apply ? "Dry-run complete — nothing written." : complete ? `Applied batch ${sum.batch_id}.` : `Stopped after ${IMPORT_MAX_CALLS} calls with batch ${sum.batch_id} unfinished — press Apply again with the same file to continue.`;
    // The report is now ABOUT something: the exact pair that was just sent.
    importPreview = { kind, text };
    renderImportReport(sum, apply);
    markPreviewFreshness();
    if (apply) run(loadImportHistory, "ihistory");
  };
  app.querySelector<HTMLButtonElement>('button[data-act="dryrun"]')!.onclick = () => void runImport(false);
  app.querySelector<HTMLButtonElement>('button[data-act="apply"]')!.onclick = () => void runImport(true);
  run(loadImportHistory, "ihistory");
  run(loadSeeding, "smetrics");
}

/** The register's filter + page state, held across re-renders of the roster so a
 * "next" does not silently reset the filters that produced the page. */
let seedFilters: SeedFilters = { tier: "", source: "", batch: "" };
let seedOffset = 0;
let seedMetrics: SeedingMetrics | null = null;

/**
 * The seeded-listing register. TWO READS, and the metrics one is what supplies
 * the filter vocabularies — see `seedFilterOptions` for why the options are
 * derived from the answer rather than restated in this console.
 */
async function loadSeeding(): Promise<void> {
  // TWO REGIONS, TWO BOXES, IN ORDER — and the first one's failure does not
  // strand the second. `await region(...)` rather than `run(...)` for exactly
  // that: the roster reads `seedMetrics`, so it must run AFTER the metrics
  // attempt, and it must run even when that attempt threw (it renders the
  // filters in their "not read" state, which is a state they already have).
  await region(async () => {
    const mbox = document.getElementById("smetrics");
    if (!mbox) return;
    seedMetrics = await api<SeedingMetrics>("/ops/seeding/metrics");
    mbox.innerHTML = seedingMetricsCard(seedMetrics);
  }, "smetrics");
  await region(loadSeedRoster, "sroster");
}

async function loadSeedRoster(): Promise<void> {
  const fbox = document.getElementById("sfilters");
  const rbox = document.getElementById("sroster");
  if (fbox) fbox.innerHTML = seedingFilters(seedFilters, seedMetrics);
  if (!rbox) return;
  const q = new URLSearchParams({ limit: String(SEED_PAGE), offset: String(seedOffset) });
  // ABSENT, NOT EMPTY. `seedingList` refuses an out-of-vocabulary tier or source
  // by returning nothing, and an empty string is out of vocabulary — so an
  // unset filter must not be sent at all.
  if (seedFilters.tier) q.set("tier", seedFilters.tier);
  if (seedFilters.source) q.set("source", seedFilters.source);
  if (seedFilters.batch) q.set("batch", seedFilters.batch);
  const on = (act: string, fn: () => void): void => {
    const b = document.querySelector<HTMLButtonElement>(`button[data-act="${act}"]`);
    if (b) b.onclick = fn;
  };
  // THE FILTER CONTROLS ARE BOUND BEFORE THE FETCH, not after it. They were
  // re-rendered above and re-bound below the `await`, so a click landing during
  // an in-flight load hit a button with no handler and did nothing — on a screen
  // whose only feedback is the table changing, that reads as a broken filter.
  // The paging buttons are inside the roster and are bound after it, because
  // they do not exist until it is drawn.
  on("seed-filter", () => {
    seedFilters = {
      tier: (document.getElementById("stier") as HTMLSelectElement | null)?.value ?? "",
      source: (document.getElementById("ssource") as HTMLSelectElement | null)?.value ?? "",
      batch: ((document.getElementById("sbatch") as HTMLInputElement | null)?.value ?? "").trim(),
    };
    // A NEW FILTER RESETS THE PAGE. Keeping the offset would show page three of
    // a result set that may have one page, i.e. an empty table under a filter
    // that matches rows — the most misleading state this control can produce.
    seedOffset = 0;
    void loadSeedRoster();
  });
  on("seed-clear", () => { seedFilters = { tier: "", source: "", batch: "" }; seedOffset = 0; void loadSeedRoster(); });
  const list = await api<SeedList>(`/ops/seeding/list?${q.toString()}`);
  rbox.innerHTML = seedingRoster(list, seedFilters);
  wireRowlinks(rbox, (id) => { location.hash = `#/creators?q=${encodeURIComponent(id)}`; });
  on("seed-prev", () => { seedOffset = Math.max(0, seedOffset - SEED_PAGE); void loadSeedRoster(); });
  on("seed-next", () => { seedOffset += SEED_PAGE; void loadSeedRoster(); });
}

/** QUEUE #775: the client's cap on apply calls for one Apply press (500 records each). */
const IMPORT_MAX_CALLS = 40;
/** QUEUE #775: the half-applied batch the next Apply continues (set from the history's
 * "Continue" or an interrupted loop), cleared when an apply completes. */
let resumeApply: { id: string; kind: string } | null = null;

/** I3-1207: the client's cap on resume calls for one Retract press. */
const RETRACT_MAX_CALLS = 100;
// RGC-03/RGM-15 + JN-NEW-1: the server flips status to 'retracted' on the FIRST
// chunk, so status alone can't tell a half-run retraction from a finished one.
// The server says which in `retraction_incomplete` (admin.ts ImportBatchRow);
// that field decides, so a reload or a second operator still sees "continue".
// This session's own memory is consulted ONLY when the field is absent (an old
// server), never over it.
const incompleteRetractions = new Set<string>();
/** I3-1421 (I34-W2-CLIENTS; DRAFT chip words): an `applying` batch is never shown as applied; any other status is shown as itself.
 * The list row and openBatch's header both draw this chip (I34-FX2-CLIENTS); openBatch also offers Retract on a STUCK
 * applying batch, which retractBatch accepts (trust-core importer.ts: the claim flips status unless applying/applied). */
function importStatusChip(b: ImportBatchRow): string {
  if (b.status === "applying") return b.stuck === true ? '<span class="ds-chip chip warn">stuck — no recent progress</span>' : '<span class="ds-chip chip wait">applying</span>';
  return b.status === "applied" ? '<span class="ds-chip chip ok">applied</span>' : `<span class="ds-chip chip">${esc(b.status)}</span>`;
}
const halfRetracted = (b: ImportBatchRow): boolean => b.retraction_incomplete ?? incompleteRetractions.has(b.batch_id);
async function openBatch(id: string): Promise<void> {
  const box = document.getElementById("ibatch");
  if (!box) return;
  const d = await api<{ detail: ImportBatchDetail }>(`/ops/imports/${encodeURIComponent(id)}`, ["detail"]);
  if (!d) { box.innerHTML = `<p class="err">Couldn't load batch.</p>`; return; }
  const b = d.detail.batch;
  const profs = d.detail.profiles.map((p) => `<tr><td>${esc(p.display_name ?? "—")}</td><td class="mono">${esc(p.passport_id)}</td><td><code>${esc(p.claim_state ?? "?")}</code>${p.retracted ? ' <span class="ds-chip chip">retracted</span>' : ""}</td></tr>`).join("");
  const dets = d.detail.determinations.map((t) => `<tr><td class="mono">${esc(t.subject ?? "?")}</td><td class="mono">${esc(t.item_id)}</td><td><code>${esc(t.state ?? "?")}</code>${t.disposition ? " · " + esc(t.disposition) : ""}</td></tr>`).join("");
  box.innerHTML = `<div class="card">
    <div class="row"><strong class="mono">${esc(b.batch_id)}</strong> <code>${esc(b.kind)}</code>
      ${b.status === "retracted" ? (halfRetracted(b) ? '<span class="ds-chip chip wait">retraction incomplete</span>' : '<span class="ds-chip chip">retracted</span>') : importStatusChip(b)}
      <span style="flex:1"></span>${b.status === "applied" || halfRetracted(b) || (b.status === "applying" && b.stuck === true) ? `<button class="ds-btn full-only" data-act="retract" data-id="${esc(b.batch_id)}">${halfRetracted(b) ? "Continue retracting…" : "Retract batch…"}</button>` : ""}</div>
    <p class="muted" style="margin:6px 0 12px">accepted ${b.accepted} · rejected ${b.rejected} · skipped ${b.skipped} · ${esc(fmtTs(b.created_at))}</p>
    ${d.detail.profiles.length ? `<h3 style="font-size:13px;margin:8px 0 6px">Profiles → passports (live state)</h3><table class="ds-table"><tbody>${profs}</tbody></table>${batchShowing(d.detail.profiles.length, d.detail.profiles_total, d.detail.truncated?.profiles)}` : ""}
    ${d.detail.determinations.length ? `<h3 style="font-size:13px;margin:14px 0 6px">Determinations → warning drafts (live state)</h3><table class="ds-table"><tbody>${dets}</tbody></table>${batchShowing(d.detail.determinations.length, d.detail.determinations_total, d.detail.truncated?.determinations)}` : ""}
    ${!d.detail.profiles.length && !d.detail.determinations.length ? `<p class="muted">This batch produced no surviving records.</p>` : ""}
    <p id="bmsg" class="muted" style="margin:8px 0 0"></p></div>`;
  const bmsg = box.querySelector<HTMLElement>("#bmsg");
  const rt = box.querySelector<HTMLButtonElement>('button[data-act="retract"]');
  if (rt) rt.onclick = async () => {
    if (!(await confirmText("Retract this whole batch? Still-unclaimed passports are retracted and queued warnings withdrawn. Passports a creator has since claimed are left intact.", "Retract batch"))) return;
    // QUEUE #746 I3-1207 (D-0850(3)): the server retracts in capped chunks and
    // answers `complete`; re-POST the same body while it says `false`, summing
    // each call's counts. An absent field is an old server: complete.
    rt.disabled = true;
    const sum = { retracted: 0, withdrawn: 0, kept: 0 };
    const counts = (): string => `${sum.retracted} passports, ${sum.withdrawn} warnings withdrawn, ${sum.kept} kept (claimed)`;
    let calls = 0;
    let complete = false;
    while (calls < RETRACT_MAX_CALLS) {
      calls++;
      const r = await apiPost<{ passports_retracted?: number; warnings_withdrawn?: number; passports_kept_claimed?: number; complete?: boolean }>(`/ops/import/${encodeURIComponent(id)}/retract`, {});
      if (!r.ok) {
        rt.disabled = false;
        if (bmsg) bmsg.textContent = failText(r.status, r.data, "That didn't go through") + (calls > 1 ? ` — after ${calls - 1} completed chunk(s): ${counts()} so far. Press Retract batch again to continue.` : "");
        return;
      }
      sum.retracted += Number(r.data?.passports_retracted ?? 0) || 0;
      sum.withdrawn += Number(r.data?.warnings_withdrawn ?? 0) || 0;
      sum.kept += Number(r.data?.passports_kept_claimed ?? 0) || 0;
      if (r.data?.complete !== false) { complete = true; break; }
      if (bmsg) bmsg.textContent = `Retracting — chunk ${calls}: ${counts()} so far…`;
    }
    const done = complete
      ? `Retracted — ${counts()}.`
      : `Stopped after ${RETRACT_MAX_CALLS} calls without the server reporting the batch complete — ${counts()} so far. Nothing is lost: press Retract batch again to continue.`;
    if (complete) incompleteRetractions.delete(id);
    else incompleteRetractions.add(id);
    void loadImportHistory();
    // openBatch redraws the card (and #bmsg with it), so the outcome is written after.
    await openBatch(id);
    const after = document.getElementById("bmsg");
    if (after) after.textContent = done;
    else toast(done);
  };
}

  on("csusp-suspend", async (btn) => {
    const id = val("cs-id");
    if (val("cs-confirm") !== id) return leverMsg("cs-msg", "Type the passport id exactly to confirm the suspension.");
    await post(btn, (p) => `/ops/creators/${p}/suspend`, { category: val("cs-cat"), note: val("cs-note"), confirm: id }, "suspended.");
  });
  on("csusp-lift", async (btn) => {
    const id = val("cs-id");
    if (!CREATOR_PASSPORT_RE.test(id)) return leverMsg("cs-msg", "A creator passport id is psp_ then 16 hex characters.");
    if (!(await confirmDialog({ title: `Lift the suspension on ${id}?`, body: "<p>The creator's actions and payouts resume.</p>", confirmLabel: "Lift" }))) return;
    // D-1004(4): over an open appeal the lift decides it, and the server requires the reason field's text.
    const reason = val("cs-appeal-reason");
    await post(btn, (p) => `/ops/creators/${p}/lift`, reason.trim() ? { reason } : {}, "suspension lifted.");
  });
  // I3-0400 (D-0533; I25-INT, I25-W1-MONEY2 H2): the TERMINATE stage — dual control creator:terminate; the server refuses
  // inside the appeal window or over an undecided appeal (creator-close.ts terminateCreator). Wording: existing strings only.
  on("csusp-terminate", async (btn) => {
    const id = val("cs-id");
    if (!CREATOR_PASSPORT_RE.test(id)) return leverMsg("cs-msg", "A creator passport id is psp_ then 16 hex characters.");
    if (!(await confirmDialog({ title: `Terminate ${id} permanently?`, body: "<p>This is <strong>not reversible</strong>.</p>", confirmLabel: "Terminate" }))) return;
    await post(btn, (p) => `/ops/creators/${p}/terminate`, { confirm: id }, "terminated.");
  });
  on("csusp-appeal", (btn) => post(btn, (p) => `/ops/creators/${p}/appeal`, {}, "appeal recorded."));
  for (const [act, outcome] of [["csusp-appeal-uphold", "upheld"], ["csusp-appeal-overturn", "overturned"]] as const) {
    on(act, async (btn) => {
      const id = val("cs-id");
      if (!CREATOR_PASSPORT_RE.test(id)) return leverMsg("cs-msg", "A creator passport id is psp_ then 16 hex characters.");
      // I3-0098/D-0995(1) (I23-FX-APPS): a creator appeal decision carries a written reason, 8-500 characters, the same
      // as the account appeal (closures.ts appealReasonOk); the server re-checks (I23-FX-HUB).
      const reason = val("cs-appeal-reason");
      if (!appealReasonOk(reason) || reason.length > 500) return leverMsg("cs-msg", `Write the reason for this appeal decision (${APPEAL_REASON_MIN}-500 characters). Nothing was sent.`);
      const ok = await confirmDialog({
        title: outcome === "upheld" ? `Uphold the suspension of ${id}?` : `Overturn the suspension of ${id}?`,
        body: outcome === "upheld" ? "<p>The suspension stays in place.</p>" : "<p>The suspension is lifted.</p>",
        confirmLabel: outcome === "upheld" ? "Uphold" : "Overturn",
      });
      if (ok) await post(btn, (p) => `/ops/creators/${p}/appeal/decide`, { outcome, reason }, outcome === "upheld" ? "appeal decided: upheld." : "appeal decided: overturned.");
    });
  }
}

function wireVerificationRef(): void {
  const form = document.getElementById("vref-form") as HTMLFormElement | null;
  form?.addEventListener("submit", (ev) => {
    ev.preventDefault();
    void (async () => {
      const id = vrefEventId();
      const out = document.getElementById("vref-result");
      if (!id || id.length > 128) { if (out) out.innerHTML = `<p class="err">Enter the grant's event id.</p>`; return; }
      const r = await apiGet<{ verification_ref?: { subject?: string; kyc_ref?: string; recorded_by?: string; updated_at?: string } }>(`/ops/verification-refs/${encodeURIComponent(id)}`);
      // A 404 is the server's own "no reference recorded"; anything else unread is unknown, never "none".
      if (r.status === 404) { if (out) out.innerHTML = `<p class="muted">No case reference is recorded for that event.</p>`; return; }
      const v = r.ok ? r.data?.verification_ref : undefined;
      if (!v) { if (out) out.innerHTML = `<p class="err">${esc(failText(r.status, r.data, "Couldn't read that reference — treat it as unknown"))}</p>`; return; }
      if (out) out.innerHTML = `<p class="sub">Subject <code>${esc(v.subject ?? "")}</code> · reference <code>${esc(v.kyc_ref ?? "")}</code> · recorded by <code>${esc(v.recorded_by ?? "")}</code> · updated ${esc(fmtTs(v.updated_at ?? ""))}</p>`;
      const subj = document.querySelector<HTMLInputElement>('#vref-set-form input[name="subject"]');
      if (subj && !subj.value) subj.value = v.subject ?? "";
    })();
  });
  const set = document.getElementById("vref-set-form") as HTMLFormElement | null;
  set?.addEventListener("submit", (ev) => {
    ev.preventDefault();
    void (async () => {
      const id = vrefEventId();
      const fd = new FormData(set);
      const subject = String(fd.get("subject") ?? "").trim();
      const kycRef = String(fd.get("kyc_ref") ?? "").trim();
      if (!id || id.length > 128) return leverMsg("vref-msg", "Enter the grant's event id above first.");
      if (kycRef && !/^psp_[0-9a-f]{16}$/.test(subject)) return leverMsg("vref-msg", "Recording a reference needs the subject's passport id (psp_ then 16 hex characters).");
      const r = await apiPost<{ ok?: boolean }>(`/ops/verification-refs/${encodeURIComponent(id)}`, { kyc_ref: kycRef, subject });
      if (!r.ok) return leverMsg("vref-msg", failText(r.status, r.data, "That didn't go through — the reference is unchanged"));
      leverMsg("vref-msg", kycRef ? "Reference saved." : "Reference withdrawn.");
    })();
  });
}
