import type { DbLike } from "./db.js";
import { logWarn } from "./log.js";
import { noteOperatorAlarm } from "./operator-alarm.js";

/**
 * A single-holder lease over a scheduled sweep.
 *
 * Cloudflare offers no exclusion between cron invocations — a retry, a
 * hand-triggered run, or a slow run overlapping the next tick all give you two
 * `scheduled()` handlers against one D1. The alert pipeline's DURABILITY was
 * already right (notification creation dedups on `(account_id, event_id)`), but
 * DELIVERY was not: both runs select the same `delivered = 0` / `emailed = 0`
 * rows and both hand them to APNs/FCM/the mailer before either marks them. The
 * fan gets the mail and the push twice. Digests are the same shape.
 *
 * The lease is one row per sweep name, and both interesting properties live in
 * one statement each:
 *
 *  - ACQUIRE is a single `INSERT … ON CONFLICT … WHERE expires_at <= now`. No
 *    read-then-write, so two racers cannot both see a free lease; SQLite/D1
 *    settles it and `changes` says who won.
 *  - RELEASE is `DELETE … WHERE name = ? AND holder = ?`. A run that has already
 *    lost its lease to expiry (it overran) cannot delete the lease its successor
 *    now holds.
 *
 * TTL IS THE FAILURE MODE, both ways. Too long and a run that dies holding the
 * lease (a Worker evicted mid-sweep runs no `finally`) blocks every later run
 * until it lapses — an outage. Too short and a healthy long sweep loses its
 * lease to the next invocation mid-flight, which is the double-send it was
 * bought to stop. So the default sits between the two clocks: far above a real
 * sweep, far below the 6-hourly cron period, so a dead run costs at most the
 * remainder of its own slot and never bleeds into the next one.
 *
 * I3-0646 / I3-1065: the paragraph above used to be the whole defence, and it
 * only holds BETWEEN legs — a single leg that itself overruns (a registry
 * rebuild, a digest send) keeps running inside an invocation whose lease has
 * already lapsed by wall clock, with nothing re-checking it mid-leg. `withCronLease`
 * now renews the row on a timer for as long as `body()` is in flight, so the
 * expiry that matters is "how long since the last renewal", not "how long
 * since acquire" — a slow leg keeps its lease exactly like a slow request
 * keeps a keepalive connection, and a genuinely dead run (no renewals at all)
 * still lapses on the original TTL.
 */
export const CRON_LEASE_TTL_MS = 30 * 60 * 1000;

export interface CronLease {
  name: string;
  holder: string;
}

/** An opaque per-run holder id. Only the run that acquired may release. */
export function newLeaseHolder(): string {
  return [...crypto.getRandomValues(new Uint8Array(12))].map((b) => b.toString(16).padStart(2, "0")).join("");
}

/**
 * Take the lease for `name`, or return null if a live one is held. Fail closed:
 * a caller that gets null must do nothing, not proceed unsynchronised.
 */
export async function acquireCronLease(
  db: DbLike,
  name: string,
  nowMs: number = Date.now(),
  ttlMs: number = CRON_LEASE_TTL_MS,
  holder: string = newLeaseHolder(),
): Promise<CronLease | null> {
  const nowIso = new Date(nowMs).toISOString();
  const expires = new Date(nowMs + ttlMs).toISOString();
  const res = await db
    .prepare(
      `INSERT INTO cron_leases (name, holder, acquired_at, expires_at) VALUES (?, ?, ?, ?)
         ON CONFLICT(name) DO UPDATE SET holder = excluded.holder, acquired_at = excluded.acquired_at, expires_at = excluded.expires_at
          -- AC-04 fix: STRICT. At expires_at == now the incumbent may still be
          -- mid-run; "<=" let a successor's upsert win at that exact instant,
          -- so both holders proceeded. The equality case is never the one a
          -- caller wants — a lease that is not yet strictly past its expiry is
          -- still live.
          WHERE cron_leases.expires_at < ?`,
    )
    .bind(name, holder, nowIso, expires, nowIso)
    .run();
  return res.changes === 1 ? { name, holder } : null;
}

/** Give the lease back early. A no-op if it already expired and was re-taken. */
export async function releaseCronLease(db: DbLike, lease: CronLease): Promise<void> {
  await db.prepare("DELETE FROM cron_leases WHERE name = ? AND holder = ?").bind(lease.name, lease.holder).run();
}

/**
 * Push the lease's expiry out from `nowMs`, but ONLY while this holder still
 * owns the row (I3-0646/I3-1065). `changes === 0` means the lease already
 * lapsed and someone else may hold it now — the caller must not treat that as
 * success. `withCronLease` below sets its guard's `lost` on a `false` here or a
 * throw (I38-TICK).
 */
export async function renewCronLease(
  db: DbLike,
  lease: CronLease,
  nowMs: number = Date.now(),
  ttlMs: number = CRON_LEASE_TTL_MS,
): Promise<boolean> {
  const nowIso = new Date(nowMs).toISOString();
  const expires = new Date(nowMs + ttlMs).toISOString();
  const res = await db
    .prepare("UPDATE cron_leases SET expires_at = ? WHERE name = ? AND holder = ? AND expires_at >= ?")
    .bind(expires, lease.name, lease.holder, nowIso)
    .run();
  return res.changes === 1;
}

/** How often, relative to the TTL, a long-running body renews its own lease.
 * A third of the TTL means a single missed renewal (a slow tick, a GC pause)
 * still leaves two more chances before the lease actually lapses. */
const RENEW_FRACTION = 3;

/** I38-TICK: the operator-alarm marker of a withCronLease run whose renewal was refused or threw (operator-alarm.ts). */
export const CRON_LEASE_LOST_MARKER = "CRON_LEASE_LOST";

/** I38-TICK: what a withCronLease body reads before each write step. `lost` turns true once a renewal is refused or
 * throws; it never turns back. */
export interface CronLeaseGuard {
  readonly lost: boolean;
}

/**
 * Run `body` under the lease, or skip it entirely if another run holds it.
 * Returns whether the body ran. The release rides a `finally` for the ordinary
 * case (including a thrown sweep); the TTL is what covers the case a `finally`
 * cannot — the Worker being evicted mid-sweep. While `body` is in flight the
 * lease is renewed on a timer (I3-0646/I3-1065) so a leg that runs past one
 * TTL window keeps its lease while its renewals succeed; a run that stops renewing (evicted, wedged) still
 * lapses on schedule because the renewal loop dies with it.
 *
 * I38-TICK: a renewal that returns false (another holder, or the row lapsed) or
 * throws sets `guard.lost`, stops the renewal loop, and writes one warn line
 * (`cron_lease_lost`, the job name in `detail`) carrying CRON_LEASE_LOST_MARKER,
 * which is handed to the operator alarm. The wrapper does NOT stop `body`: a
 * body stops only where it reads `guard.lost` before a write step. A thrown
 * renewal counts as lost because the run can no longer show it holds the lease.
 */
export async function withCronLease<T>(
  db: DbLike,
  name: string,
  body: (guard: CronLeaseGuard) => Promise<T>,
  nowMs: number = Date.now(),
  ttlMs: number = CRON_LEASE_TTL_MS,
): Promise<{ ran: false } | { ran: true; result: T; leaseLost: boolean }> {
  const lease = await acquireCronLease(db, name, nowMs, ttlMs);
  if (!lease) return { ran: false };
  const renewEveryMs = Math.max(1, Math.floor(ttlMs / RENEW_FRACTION));
  // A recursive `setTimeout`, not `setInterval`: each tick only reschedules
  // itself once its own renew call has settled, so a slow renew cannot pile
  // ticks up. `clearTimeout` in `finally` is what makes this cancellable —
  // an earlier version awaited a sleeping promise instead and `stopped =
  // true` never woke it, so `withCronLease` hung until the real TTL elapsed
  // on every call (caught by this file's own test run, not by inspection).
  let timer: ReturnType<typeof setTimeout> | undefined;
  // `stopped` is the cancellation token; `timer` alone is not enough. A
  // renewal that is already in flight when `body()` resolves has no pending
  // timer id to clear -- its `.finally(scheduleRenew)` continuation runs
  // AFTER this function's own `finally` below, and would otherwise schedule
  // a fresh timer nothing here holds a handle to. Checking `stopped` inside
  // `scheduleRenew` itself makes that continuation a no-op regardless of
  // when it runs, which a timer id alone cannot do.
  let stopped = false;
  let lost = false;
  const guard: CronLeaseGuard = {
    get lost() {
      return lost;
    },
  };
  // Called at most once: the loop below schedules no renewal once `lost` is set, and only one renewal is in flight.
  const markLost = (how: "refused" | "threw"): void => {
    lost = true;
    const f = { cid: lease.holder, hop: "cron" as const, msg: "cron_lease_lost", detail: `${name}: renewal ${how}`, marker: CRON_LEASE_LOST_MARKER };
    logWarn(f);
    noteOperatorAlarm(f);
  };
  const scheduleRenew = () => {
    if (stopped || lost) return;
    timer = setTimeout(() => {
      renewCronLease(db, lease, Date.now(), ttlMs)
        .then(
          (held) => {
            if (!held) markLost("refused");
          },
          () => markLost("threw"),
        )
        .finally(scheduleRenew);
    }, renewEveryMs);
  };
  scheduleRenew();
  try {
    const result = await body(guard);
    return { ran: true, result, leaseLost: lost };
  } finally {
    stopped = true;
    if (timer !== undefined) clearTimeout(timer);
    await releaseCronLease(db, lease);
  }
}

/**
 * Verify the processor webhook signature: HMAC-SHA256 of the RAW body under the
 * shared secret, hex, constant-time (delegated to crypto.subtle.verify). Same
 * message-authentication pattern as the Stripe webhook — NOT a subject hash, so
 * no single-hasher concern. False on any malformation.
 */
export async function verifyCryptoSignature(rawBody: string, signatureHex: string, secret: string | undefined, nowMs: number = Date.now()): Promise<boolean> {
  if (placeholder(secret) || (secret as string).length < 16) return false;
  if (typeof signatureHex !== "string" || !/^[0-9a-fA-F]{64}$/.test(signatureHex)) return false;
  let key: Awaited<ReturnType<typeof crypto.subtle.importKey>>;
  try {
    key = await crypto.subtle.importKey("raw", encoder.encode(secret as string), { name: "HMAC", hash: "SHA-256" }, false, ["verify"]);
  } catch {
    return false;
  }
  const sig = new Uint8Array(signatureHex.match(/../g)!.map((h) => parseInt(h, 16))) as unknown as ArrayBuffer;
  let ok: boolean;
  try {
    ok = await crypto.subtle.verify("HMAC", key, sig, encoder.encode(rawBody) as unknown as ArrayBuffer);
  } catch {
    return false;
  }
  if (!ok) return false;
  // I3-1719: the MAC has no timestamp of its own, so freshness is bound to the
  // SIGNED body's event `created_at`; absent or unparseable → refused.
  let createdMs = NaN;
  try {
    const obj = JSON.parse(rawBody) as Record<string, unknown>;
    const ev = (obj.event as Record<string, unknown> | undefined) ?? obj;
    const at = ev && typeof ev === "object" ? (ev as Record<string, unknown>).created_at : undefined;
    if (typeof at === "string" && /^\d{4}-\d{2}-\d{2}T/.test(at)) createdMs = Date.parse(at);
  } catch {
    return false;
  }
  if (!Number.isFinite(createdMs)) return false;
  return createdMs >= nowMs - CRYPTO_WEBHOOK_MAX_AGE_MS && createdMs <= nowMs + CRYPTO_WEBHOOK_MAX_SKEW_MS;
}

export interface CryptoEvent {
  type: string;
  data: Record<string, unknown>;
}

/** Parse + shape-validate a processor webhook body. Null on anything we cannot
 * fully understand (fail closed). */
export function parseCryptoEvent(rawBody: string): CryptoEvent | null {
  let obj: Record<string, unknown>;
  try {
    obj = JSON.parse(rawBody) as Record<string, unknown>;
  } catch {
    return null;
  }
  const ev = (obj.event as Record<string, unknown> | undefined) ?? obj; // some processors wrap in {event:{}}
  if (!ev || typeof ev !== "object") return null;
  const type = ev.type;
  const data = ev.data;
  if (typeof type !== "string" || type.length === 0 || type.length > 64) return null;
  if (!data || typeof data !== "object") return null;
  return { type, data: data as Record<string, unknown> };
}

/**
 * Map a verified event to a credit grant intent, or null if it is not a
 * confirmed/settled credit charge. PURE — the webhook does the granting.
 *
 * SIZED FROM SETTLED FIAT: credits = sum of the CONFIRMED payments' local (fiat)
 * value in cents (USD only), divided by the backing — NEVER the quoted/requested
 * amount. A volatility shortfall between quote and settlement credits only what
 * actually arrived; I21-Q81 (D-1030): an OVERPAYMENT is credited too — every
 * settled cent becomes credit (credits = settled, below), still refused past
 * MAX_CRYPTO_CREDITS here and past the lifetime crypto cap in `grantCredits`.
 * O-07 (QUEUE #163): with the rail off (CRYPTO_RAIL_ENABLED) no fan can reach a
 * crypto checkout and `apps/web/src/wallet.ts` says crypto is not offered; its
 * overpayment line (CRYPTO_OVERPAY_NOTE) renders only while the rail is on.
 */
export function cryptoGrantFromEvent(ev: CryptoEvent): CryptoGrantIntent | null {
  const meta = (ev.data.metadata as Record<string, unknown> | undefined) ?? {};
  return grantForSubject(ev, typeof meta.dhwaja_subject === "string" ? meta.dhwaja_subject : "");
}

/**
 * I3-1037: the webhook's mapper. The charge carries a SEALED subject
 * (`sealCryptoSubject`); open it under the webhook secret. A plaintext `acc_`
 * subject is still honoured for a charge created before sealing existed.
 */
export async function cryptoGrantFromSignedEvent(ev: CryptoEvent, keys: SubjectSealKeys | string | null): Promise<CryptoGrantIntent | null> {
  const meta = (ev.data.metadata as Record<string, unknown> | undefined) ?? {};
  const raw = typeof meta.dhwaja_subject === "string" ? meta.dhwaja_subject : "";
  const subjectId = raw.startsWith(SEAL_PREFIX) ? await openCryptoSubject(raw, keys) : raw;
  return subjectId ? grantForSubject(ev, subjectId) : null;
}

/** I3-0687: a SETTLED charge — the only event type this rail grants from. A
 * null intent for one of these is money that settled and was attributed to
 * nobody (the webhook records it); every other type legitimately grants nothing. */
export function isSettledChargeEvent(ev: CryptoEvent): boolean {
  return ev.type === "charge:confirmed" || ev.type === "charge:resolved";
}

function grantForSubject(ev: CryptoEvent, subjectId: string): CryptoGrantIntent | null {
  if (!isSettledChargeEvent(ev)) return null;
  const d = ev.data;
  const meta = (d.metadata as Record<string, unknown> | undefined) ?? {};
  if (meta.dhwaja_purpose !== "credits") return null;
  if (!ACCOUNT_RE.test(subjectId)) return null;
  // Review (LOW): pin the idempotency key to ONE field (the charge id), never a
  // d.id-else-d.code that could give a confirmed and a resolved event two
  // different keys for the same charge and bypass the UNIQUE dedup.
  const chargeId = canonicalChargeId(d.id); // I3-1675
  if (!chargeId) return null;
  // Settled fiat = Σ CONFIRMED payments' local value (USD), in cents. NO
  // pricing.local fallback (review MED): the quoted price is never credited — a
  // confirmed event whose payments don't parse as USD/CONFIRMED grants nothing,
  // rather than crediting money we cannot see was received.
  const payments = Array.isArray(d.payments) ? (d.payments as Array<Record<string, unknown>>) : [];
  let settledCents = 0;
  for (const p of payments) {
    if ((p.status as string) !== "CONFIRMED") continue;
    const value = p.value as { local?: { amount?: unknown; currency?: unknown } } | undefined;
    const local = value?.local;
    if (!local || local.currency !== "USD") continue;
    const amt = typeof local.amount === "string" ? Number(local.amount) : typeof local.amount === "number" ? local.amount : NaN;
    if (!Number.isFinite(amt) || amt <= 0) continue;
    settledCents += Math.round(amt * 100);
  }
  if (settledCents <= 0) return null;
  const settledCredits = Math.floor(settledCents / CREDIT_BACKING_CENTS);
  // I21-Q81 (D-1030): credit what SETTLED, overpayment included — a $60 settlement
  // on a $50 charge is 6000 credits, and nothing is left for the fan to reclaim.
  const credits = settledCredits;
  if (credits <= 0 || credits > MAX_CRYPTO_CREDITS) return null;
  // I3-1671: gross is the fiat the processor settled (reconciles with its statement).
  const grossCents = settledCents;
  return { subjectId, credits, externalTxnId: chargeId, grossCents };
}

/**
 * I3-0678 a (I25) — A LATER, LARGER SETTLEMENT OF THE SAME CHARGE IS GRANTED AS A DELTA, ONCE.
 *
 * `grantForSubject` sizes a grant from the CUMULATIVE settled fiat of a charge, and `grantCredits` dedups on
 * `credit_purchases.external_txn_id` (UNIQUE). Keyed on the bare charge id, an underpaid `charge:confirmed` (30 of 50
 * credits) followed by the `charge:resolved` that completes it was deduped away: the fan paid 50 and kept 30.
 *
 * THE KEY CHAIN. Each grant for a charge is the difference `cumulative - prior` (prior = every crypto purchase row
 * already written for this charge, `priorCryptoGrant`), written under `cryptoGrantKey(charge, prior)`: the bare charge
 * id while prior is 0 (so a row written before this existed is the chain's first link), else `<charge>+<prior>`. So
 *   - a redelivery of the same event computes the same delta at a LATER prior, or none: never a second grant;
 *   - two deliveries racing at the SAME prior write the SAME key, so external_txn_id UNIQUE lets one land and the
 *     other is `duplicate`; `grantCryptoDelta` then re-reads the prior and grants what is still owed, if anything.
 * `+` is in neither CHARGE_ID_RE nor a UUID, so a chain key can never be another charge's id. An overpayment is
 * credited (I21-Q81), so the chain grants up to the cumulative settled fiat and never past it. The rail stays OFF
 * (CRYPTO_RAIL_ENABLED); the webhook (index.ts, the hub) calls `grantCryptoDelta` (I25-W1-MONEY2 HANDOFFS).
 */
export interface CryptoPriorGrant {
  /** Credits settled on the chain so far: granted plus owed back past the lifetime cap (I54 M6 a), which sizes the delta. */
  credits: number;
  grossCents: number;
  /** Of those, the credits actually granted (credit_purchases), which the charge's intended credits bound (I54 M6 a). */
  grantedCredits: number;
}
const DELTA_SEP = "+";
/** The external_txn_id of the grant that follows `priorCredits` already granted on `chargeId`. */
export function cryptoGrantKey(chargeId: string, priorCredits: number): string {
  return priorCredits === 0 ? chargeId : `${chargeId}${DELTA_SEP}${priorCredits}`;
}
/** What has been settled on this charge so far: every crypto `credit_purchases` row of its key chain, plus (I54 M6 a)
 * every 'crypto_cap_excess' refund_owed row of it, so a re-delivery of a settlement whose excess is owed owes nothing new. */
export async function priorCryptoGrant(db: DbLike, chargeId: string): Promise<CryptoPriorGrant> {
  const chain = "rail = 'crypto' AND (external_txn_id = ? OR substr(external_txn_id, 1, ?) = ?)";
  const binds = [chargeId, chargeId.length + DELTA_SEP.length, chargeId + DELTA_SEP];
  const row = await db
    .prepare(
      `SELECT COALESCE(SUM(credits_granted), 0) AS credits, COALESCE(SUM(gross_cents), 0) AS gross, (SELECT COALESCE(SUM(credits), 0) FROM credit_settlement_refusals WHERE reason = 'crypto_cap_excess' AND ${chain}) AS owed, (SELECT COALESCE(SUM(gross_cents), 0) FROM credit_settlement_refusals WHERE reason = 'crypto_cap_excess' AND ${chain}) AS owed_gross FROM credit_purchases WHERE ${chain}`,
    )
    .bind(...binds, ...binds, ...binds)
    .first<{ credits: number; gross: number; owed: number; owed_gross: number }>();
  const granted = Number(row?.credits ?? 0);
  return { credits: granted + Number(row?.owed ?? 0), grossCents: Number(row?.gross ?? 0) + Number(row?.owed_gross ?? 0), grantedCredits: granted };
}
/** I26-W1-MONEY (JA-06): every grant of `chargeId`'s key chain — the bare id and each `<charge>+<prior>` — by the same
 * predicate as `priorCryptoGrant`, oldest first. A refund or dispute of the charge is keyed by the charge id and reverses
 * all of them (refunds.ts `reverseGrant`). A key that is itself a chain link (`+` in it) names one grant, not a chain. */
export async function cryptoChainKeys(db: DbLike, chargeId: string): Promise<{ key: string; grossCents: number }[]> {
  if (chargeId.includes(DELTA_SEP)) return [];
  const { results } = await db
    .prepare(
      "SELECT external_txn_id AS key, gross_cents AS gross FROM credit_purchases WHERE rail = 'crypto' AND (external_txn_id = ? OR substr(external_txn_id, 1, ?) = ?) ORDER BY at, purchase_id",
    )
    .bind(chargeId, chargeId.length + DELTA_SEP.length, chargeId + DELTA_SEP)
    .all<{ key: string; gross: number }>();
  return results.map((r) => ({ key: r.key, grossCents: Number(r.gross) }));
}
/** PURE: the grant still owed on `intent` (its cumulative credits) after `prior`, or null when nothing is owed. */
export function cryptoDeltaGrant(intent: CryptoGrantIntent, prior: CryptoPriorGrant): CryptoGrantIntent | null {
  const credits = intent.credits - prior.credits;
  if (!Number.isInteger(credits) || credits <= 0) return null;
  const externalTxnId = cryptoGrantKey(intent.externalTxnId, prior.credits);
  if (externalTxnId.length > 200) return null; // grantCredits' own bound; a charge id that long cannot chain
  return { subjectId: intent.subjectId, credits, externalTxnId, grossCents: Math.max(0, intent.grossCents - prior.grossCents) };
}
/** I54 M6 a (INV-077): the credits the fan meant to buy on this charge — the checkout's own `dhwaja_credits` metadata
 * (`buildCryptoChargeBody`), a whole number 1..MAX_CRYPTO_CREDITS, else 0 (nothing intended can be read). */
export function cryptoIntendedCredits(ev: CryptoEvent): number {
  const v = (ev.data.metadata as Record<string, unknown> | undefined)?.dhwaja_credits;
  const n = typeof v === "string" && /^[1-9][0-9]{0,6}$/.test(v) ? Number(v) : 0;
  return n <= MAX_CRYPTO_CREDITS ? n : 0;
}
/**
 * I54 M6 a (INV-077, W0-M-08; the whole-settlement edge is W0 ruling J6, DRAFT pending Jason). PURE: how a delta of
 * `deltaCredits` splits when the account's lifetime crypto cap has `roomCents` left and the charge may still grant
 * `intendedLeft` of its intended credits. Within the room: all of it is granted (an overpayment too, I21-Q81). Past it: the
 * intended credits still owed are granted and the rest is owed back; when those alone pass the room (or none are left),
 * the whole delta is owed back.
 */
export function cryptoCapSplit(deltaCredits: number, intendedLeft: number, roomCents: number): { grant: number; owed: number } {
  if (deltaCredits * CREDIT_BACKING_CENTS <= roomCents) return { grant: deltaCredits, owed: 0 };
  // Here the delta passes the room, so intendedLeft <= room also means intendedLeft < deltaCredits.
  return intendedLeft * CREDIT_BACKING_CENTS <= roomCents ? { grant: intendedLeft, owed: deltaCredits - intendedLeft } : { grant: 0, owed: deltaCredits };
}
/** The account's lifetime crypto backing still under `crypto_credit_cap_cents` (credits.ts CRYPTO_CAP_OK_SQL's terms). */
async function cryptoCapRoomCents(db: DbLike, accountId: string): Promise<number> {
  const r = await db
    .prepare(`SELECT ${CRYPTO_CREDIT_CAP_CENTS_SQL} - (SELECT COALESCE(SUM(credits_granted), 0) FROM credit_purchases WHERE account_id = ? AND rail = 'crypto') * ${CREDIT_BACKING_CENTS} AS room`)
    .bind(accountId)
    .first<{ room: number }>();
  return Number(r?.room ?? 0);
}
/** Re-reads after a lost race (`duplicate`, or the cap room taken meanwhile) before giving up. */
export const CRYPTO_DELTA_ATTEMPTS = 3;
/**
 * Grant what is still owed on a settled charge through `grant` (the caller's `grantCredits` call, given the DELTA
 * intent; it passes `owedExcess` on as `cryptoOwedExcess`, so the excess is owed in the granting batch). `reason:
 * "duplicate"` means nothing is owed (already granted); `"contended"` means every attempt lost a race and something may
 * still be owed — the caller answers a retryable status so the processor redelivers. I54 M6 a: a delta past the lifetime
 * crypto cap is split (`cryptoCapSplit`, against the intended credits of `opts.event`, `cryptoIntendedCredits`); a delta owed back whole is recorded here
 * (credits.ts cryptoCapOwedStmt, `reason: "crypto_cap_owed"`), and when that write lands nothing (the cap moved, the
 * account is gone) the delta goes to `grant` whole, which grants it or answers why not.
 */
export async function grantCryptoDelta(
  db: DbLike,
  intent: CryptoGrantIntent,
  grant: (delta: CryptoGrantIntent & { owedExcess?: number }) => Promise<{ granted: boolean; reason?: string }>,
  opts: { event?: CryptoEvent; now?: () => string } = {},
): Promise<{ granted: boolean; credits: number; reason?: string }> {
  const now = opts.now ?? (() => new Date().toISOString());
  for (let i = 0; i < CRYPTO_DELTA_ATTEMPTS; i++) {
    const prior = await priorCryptoGrant(db, intent.externalTxnId);
    const delta = cryptoDeltaGrant(intent, prior);
    if (!delta) return { granted: false, credits: 0, reason: "duplicate" };
    const intended = opts.event ? cryptoIntendedCredits(opts.event) : 0;
    const split = cryptoCapSplit(delta.credits, Math.max(0, intended - prior.grantedCredits), await cryptoCapRoomCents(db, intent.subjectId));
    // split.grant is the whole delta (nothing owed), the intended part (the rest owed in the granting batch), or 0.
    const asked: CryptoGrantIntent & { owedExcess?: number } =
      split.grant > 0 ? { ...delta, credits: split.grant, grossCents: delta.grossCents - split.owed * CREDIT_BACKING_CENTS, owedExcess: split.owed } : delta;
    if (split.grant === 0 && (await cryptoCapOwedStmt(db, { accountId: intent.subjectId, externalTxnId: delta.externalTxnId, grossCents: delta.grossCents, credits: delta.credits }, now()).run()).changes === 1)
      return { granted: false, credits: 0, reason: "crypto_cap_owed" };
    const r = await grant(asked);
    if (r.granted) return { granted: true, credits: asked.credits };
    // A grant refused by the cap after the room was read lost it to a concurrent grant: re-read and split again.
    if (r.reason !== "duplicate" && r.reason !== "crypto_cap") return { granted: false, credits: 0, ...(r.reason ? { reason: r.reason } : {}) };
  }
  return { granted: false, credits: 0, reason: "contended" };
}

const SEAL_PREFIX = "csub1.";
const SEAL_LABEL = "dhwaja/crypto-rail/subject-seal/v1";
/** AES-256-GCM key bytes for the subject seal: HMAC-SHA256(SUBJECT_SEAL_KEY, label) (D-0855(1) — never the processor's webhook secret). The token format lives in subject-seal.ts (#747). */
async function sealKey(secret: string): Promise<ArrayBuffer> {

}

export const CRYPTO_FETCH_TIMEOUT_MS = 10_000;

/**
 * The processor POST behind `createCryptoCharge` (its only production caller, after every gate above), under
 * stripe.ts's deadline idiom: one AbortController, aborted at CRYPTO_FETCH_TIMEOUT_MS, over the request and its body.
 * Null on any failure, an abort included. Exported so the deadline is testable while CRYPTO_RAIL_ENABLED is false.
 */
export async function requestCryptoCharge(cfg: CryptoConfig, body: string, fetchImpl: typeof fetch = fetch): Promise<string | null> {
  if (!cfg.apiKey) return null; // charge creation not configured
  const ctl = new AbortController();
  const timer = setTimeout(() => ctl.abort(), CRYPTO_FETCH_TIMEOUT_MS);
  try {
    const res = await fetchImpl(`${cfg.apiBase}/charges`, { redirect: "manual",
      method: "POST",
      headers: { "content-type": "application/json", "X-CC-Api-Key": cfg.apiKey, "X-CC-Version": "2018-03-22" },
      body,
      signal: ctl.signal,
    });
    if (!res.ok) return null;
    const j = (await res.json()) as { data?: { hosted_url?: unknown } };
    const url = j.data?.hosted_url;
    return typeof url === "string" && /^https:\/\//.test(url) ? url : null;
  } catch {
    return null;
  } finally {
    clearTimeout(timer);
  }
}

/**
 * I25-W1-COOKIE (I3-0714 a, D-1007): BROWSER SESSIONS LIVE IN AN HttpOnly COOKIE, NEVER IN PAGE STORAGE.
 * The design and every choice below are written down in docs/SESSION-COOKIES.md; this docblock names controls only.
 *
 * - THE COOKIE: one per session kind (`__Host-dhwaja_fan`, `__Host-dhwaja_creator`, `__Host-dhwaja_agency`), set by
 *   `finishSessionResponse` when a BROWSER request's JSON answer (any status, any depth — RG-01, RG-02) carries a session
 *   token. The token is moved out of the body into `HttpOnly; Secure; SameSite=Strict; Path=/`; host-only unless SESSION_COOKIE_DOMAIN names a domain
 *   (then `__Secure-` + `Domain=`). The body keeps the field, holding the CSRF HANDLE instead of the token.
 * - THE HANDLE (CSRF): `<kind>.<hex>` = scopedKeyedMac(SESSION_SECRET, CSRF_SCOPE, "kind:id:ep"). Useless without the
 *   cookie; stable across a renewal of the same session epoch; changes on logout-everywhere (the epoch moves). I7-Q4: a
 *   cookie verifies under SESSION_SECRET or a SESSION_SECRET_PREVIOUS entry (sessions.ts verifySessionRotating), and a
 *   handle is accepted under any of those secrets, so a key rotation signs nobody out.
 * - THE ONLY WAY THE COOKIE AUTHENTICATES: `Authorization: Dhwaja-Csrf <handle>` (`admitSessionRequest`). The cookie for
 *   the handle's kind is verified, the handle is compared in constant time with the one derived from the cookie's own
 *   subject, and only then is the header rewritten to `Bearer <cookie>` for the existing resolvers. A wrong handle is
 *   403 `csrf_failed`; a missing or dead cookie is 401 `unauthorized` and the cookie is cleared. A request without the
 *   scheme never reads the cookie, so a cross-site form or navigation cannot ride it.
 * - A BEARER DOES NOT BYPASS IT: an unsafe-method request with `Authorization: Bearer` that also carries a session
 *   cookie is 403 `csrf_required`; so is ANY `Authorization: Bearer` sent from one of our pages (`fromOurPage`), so a
 *   token left in pre-I25 page storage cannot be replayed from a page (JN-01 r1 / RG-03).
 * - NATIVE CLIENTS: a request with neither `Origin` nor `Sec-Fetch-Site` (headers page script cannot remove) is not a
 *   browser; its answers are left byte-identical, token in the body, exactly as before.
 */

export function readSessionCookie(req: Request, kind: SubjectKind, env: CookieEnv): string | null {

function refusal(status: 401 | 403, error: string, message: string, setCookie?: string): Response {

export type Admission =
  | { kind: "pass" }
  | { kind: "refuse"; response: Response }
  | { kind: "cookie"; request: Request; sessionKind: SubjectKind; token: string };

/** The request half. Runs before any route. */
export async function admitSessionRequest(req: Request, env: CookieEnv, nowSec: number): Promise<Admission> {
  const method = req.method.toUpperCase();
  if (method === "OPTIONS") return { kind: "pass" };
  const auth = (req.headers.get("authorization") ?? "").trim();
  const scheme = /^Dhwaja-Csrf\s+(\S+)$/i.exec(auth);
  if (scheme) {
    const m = HANDLE_RE.exec(scheme[1]!);
    if (!m) return { kind: "refuse", response: refusal(403, "csrf_failed", CSRF_FAILED_MESSAGE) };
    const kind = m[1] as SubjectKind;
    const secret = env.SESSION_SECRET;
    const cookie = readSessionCookie(req, kind, env);
    const s = usableSessionSecret(secret) && cookie ? ((await verifySessionRotating(env, cookie, nowSec))?.subject ?? null) : null;
    if (!s || s.kind !== kind || !cookie || !usableSessionSecret(secret)) {
      return { kind: "refuse", response: refusal(401, "unauthorized", SIGNED_OUT_MESSAGE, clearSessionCookie(kind, env)) };
    }
    // I7-Q4: the handle is accepted under ANY configured secret, the verifying one among them — a fan renewal re-mints the
    // cookie under SESSION_SECRET while the open page still holds the handle derived under the retired one.
    let handleOk = false;
    for (const k of sessionSecrets(env)) if (constantTimeEqual(await csrfHandle(k, s), scheme[1]!)) handleOk = true;
    if (!handleOk) {
      return { kind: "refuse", response: refusal(403, "csrf_failed", CSRF_FAILED_MESSAGE) };
    }
    const headers = new Headers(req.headers);
    headers.set("authorization", `Bearer ${cookie}`);
    return { kind: "cookie", request: new Request(req, { headers }), sessionKind: kind, token: cookie };
  }
  // JN-01 r1 / RG-03: a BEARER from one of our pages is refused, any method — a token lifted from pre-I25 page storage
  // cannot be replayed from a page. Same refusal and sentence as the cookie-beside-Bearer case below.
  if (/^Bearer\s/i.test(auth) && fromOurPage(req, env)) return { kind: "refuse", response: refusal(403, "csrf_required", CSRF_REQUIRED_MESSAGE) };
  if (!SAFE.has(method) && auth !== "" && carriesAnySessionCookie(req, env)) {
    return { kind: "refuse", response: refusal(403, "csrf_required", CSRF_REQUIRED_MESSAGE) };
  }
  return { kind: "pass" };
}

}

/** RG-02: every string in a JSON value (nested objects and arrays, depth-bounded) that verifies as a session token is
 * replaced by its CSRF handle, and its cookie is queued. */
const STRIP_DEPTH = 16;
async function stripTokens(v: unknown, nowSec: number, env: CookieEnv, cookies: [string, string][], depth: number): Promise<unknown> {
  if (typeof v === "string") {
    if (!TOKEN_SHAPE_RE.test(v)) return v;
    const r = await verifySessionRotating(env, v, nowSec);
    if (!r) return v;
    const s = r.subject;
    // I26 (I3-2055): the cookie lives as long as the token in it — a capped re-mint (fan-account-routes.ts revoke/unlink) ends early.
    const exp = sessionExpiresAt(v);
    cookies.push(["Set-Cookie", sessionCookie(s.kind, v, exp === null ? SESSION_TTL_SECONDS : Math.min(SESSION_TTL_SECONDS, exp - nowSec), env)]);
    return csrfHandle(r.secret, s);
  }
  if (v === null || typeof v !== "object" || depth >= STRIP_DEPTH) return v;
  if (Array.isArray(v)) {
    const out: unknown[] = [];
    for (const x of v) out.push(await stripTokens(x, nowSec, env, cookies, depth + 1));
    return out;
  }
  const out: Record<string, unknown> = {};
  for (const [k, x] of Object.entries(v as Record<string, unknown>)) out[k] = await stripTokens(x, nowSec, env, cookies, depth + 1);
  return out;
}

/** The response half. `admission` is what `admitSessionRequest` returned for this request. `stillLive` (RG-06): when it
 * says the cookie's session is still live by its own surface's predicate, a 401 from some other door leaves the cookie. */
export async function finishSessionResponse(req: Request, res: Response, admission: Admission, env: CookieEnv, nowSec: number, stillLive?: (token: string) => Promise<boolean>): Promise<Response> {
  const secret = env.SESSION_SECRET;
  if (!usableSessionSecret(secret)) return res;
  const json = (res.headers.get("content-type") ?? "").includes("application/json");
  if (admission.kind === "cookie") {
    if (res.status === 401 && json) {
      const body = (await res.clone().json().catch(() => null)) as { error?: unknown } | null;
      if (body?.error === "unauthorized" && !(stillLive && (await stillLive(admission.token)))) return withHeaders(res, res.body, [["Set-Cookie", clearSessionCookie(admission.sessionKind, env)]]);
    }
    if (admission.sessionKind === "fan" && res.status < 400) {
      const iat = sessionIssuedAt(admission.token);
      const s = iat !== null && nowSec - iat > FAN_SESSION_RENEW_SECONDS ? ((await verifySessionRotating(env, admission.token, nowSec))?.subject ?? null) : null;
      const cap = sessionExpiresAt(admission.token);
      const fresh = s && cap !== null ? await issueSession(secret, { kind: "fan", id: s.id, ep: s.ep, expCap: cap }, nowSec) : null;
      if (fresh && cap !== null) res = withHeaders(res, res.body, [["Set-Cookie", sessionCookie("fan", fresh, cap - nowSec, env)]]);
    }
  } else if (!isBrowserRequest(req) && res.status < 400) {
    const bearer = /^Bearer\s+(\S+)$/.exec((req.headers.get("authorization") ?? "").trim())?.[1];
    const iat = bearer ? sessionIssuedAt(bearer) : null;
    if (bearer && iat !== null && nowSec - iat > FAN_SESSION_RENEW_SECONDS) {
      const s = (await verifySessionRotating(env, bearer, nowSec))?.subject ?? null;
      const cap = sessionExpiresAt(bearer);
      const fresh = s?.kind === "fan" && cap !== null ? await issueSession(secret, { kind: "fan", id: s.id, ep: s.ep, expCap: cap }, nowSec) : null;
      if (fresh) res = withHeaders(res, res.body, [[RENEWED_HEADER, fresh]]);
    }
  }
  // RG-01: every status, not only 2xx — the closed-fan 403 `rights_session` (fan-account-routes.ts closedSignInBody) moves too.
  if (!isBrowserRequest(req) || !json) return res;
  const text = await res.clone().text();
  let body: unknown;
  try {
    body = JSON.parse(text);
  } catch {
    return res;
  }
  if (!body || typeof body !== "object") return res;
  const cookies: [string, string][] = [];
  const out = await stripTokens(body, nowSec, env, cookies, 0);
  if (cookies.length === 0) return res;
  const headers = new Headers(res.headers);
  headers.delete("content-length");
  headers.set("Cache-Control", "no-store");
  for (const [k, v] of cookies) headers.append(k, v);
  return new Response(JSON.stringify(out), { status: res.status, statusText: res.statusText, headers });
}

/**
 * Strict calendar-date utilities. A "date" here is a DAY (`YYYY-MM-DD`), never a
 * timestamp — parsing is UTC (DST-free, deterministic) and STRICT: it rejects
 * everything `Date.parse` loosely accepts (locale strings, timezone-laden ISO
 * timestamps, out-of-range months/days, and silent rollovers like `2020-02-30`
 * or a non-leap `2021-02-29`). Shared by the write path (release provenance) and
 * the read path (back-catalog anniversaries) so both agree on what a date is.
 */

const DATE_RE = /^(\d{4})-(\d{2})-(\d{2})$/;

export function isLeapYear(y: number): boolean {
  return (y % 4 === 0 && y % 100 !== 0) || y % 400 === 0;
}

/** Days in month `m` (1..12) of year `y`, leap-aware. */
export function daysInMonth(y: number, m: number): number {
  return [31, isLeapYear(y) ? 29 : 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31][m - 1] ?? 0;
}

export interface DateParts {
  y: number;
  m: number; // 1..12
  d: number; // 1..31
}

/** Parse a STRICT `YYYY-MM-DD` real calendar date, or null. Rejects a bad shape,
 * an out-of-range month/day, and a day that doesn't exist in that month/year
 * (incl. Feb 29 in a common year). */
export function parseStrictDate(s: unknown): DateParts | null {
  if (typeof s !== "string") return null;
  const m = DATE_RE.exec(s);
  if (!m) return null;
  const y = Number(m[1]);
  const mo = Number(m[2]);
  const d = Number(m[3]);
  if (mo < 1 || mo > 12) return null;
  if (d < 1 || d > daysInMonth(y, mo)) return null;
  return { y, m: mo, d };
}

/** Whether `s` is a strict, real `YYYY-MM-DD` calendar date. */
export function isStrictDate(s: unknown): boolean {
  return parseStrictDate(s) !== null;
}

/** UTC midnight ms for a strict `YYYY-MM-DD`, or null if it isn't one.
 *
 * D-65: `Date.UTC(y, ...)` remaps any 0 <= y <= 99 to 1900+y (the legacy
 * two-digit-year rule) — so `Date.UTC(99, 0, 1)` is 1999, not year 99. That
 * silently moved a date `parseStrictDate` had already accepted on its own
 * arithmetic (no `Date` object, so it never saw the remap) 100 years off, with
 * no null and no throw. Build the `Date` via `Date.UTC` for month/day layout,
 * then overwrite the year with `setUTCFullYear`, which does NOT remap 0..99. */
export function strictDateUtcMs(s: unknown): number | null {
  const p = parseStrictDate(s);
  if (!p) return null;
  const dt = new Date(Date.UTC(p.y, p.m - 1, p.d));
  dt.setUTCFullYear(p.y);
  return dt.getTime();
}

/** A UTC DAY, `YYYY-MM-DD` (regions.ts re-exports it; its day cells seal on it). */
export function dayOf(nowMs: number): string {
  return new Date(nowMs).toISOString().slice(0, 10);
}

/**
 * Narrow database interface: the subset of D1's API the trust core uses.
 * D1Database satisfies this structurally; tests satisfy it with a
 * better-sqlite3 double (test/sqlite-double.ts). Keeping the surface this
 * small is what makes the fidelity swap (real D1 in S4) cheap.
 */

export interface DbStatement {
  bind(...values: unknown[]): DbStatement;
  /** Returns rows-affected (D1 exposes meta.changes; the double returns
   * better-sqlite3's changes) for atomic guarded UPDATEs + sweeps. */
  run(): Promise<{ changes: number }>;
  all<T = Record<string, unknown>>(): Promise<{ results: T[] }>;
  first<T = Record<string, unknown>>(): Promise<T | null>;
}

export interface DbLike {
  prepare(sql: string): DbStatement;
  /** Atomic: all statements apply or none do (D1 semantics). */
  batch(statements: DbStatement[]): Promise<unknown[]>;
}

// --- Real D1 → DbLike adapter -------------------------------------------------
// The app is written against DbLike. A real D1Database is ALMOST structurally
// compatible, but `run()` differs: D1 reports rows-affected at `meta.changes`,
// while the DbLike contract (and every `res.changes` guard in the code — the
// last-admin guard, claim-verify, passkey-revoke, credential reservation, …)
// expects it at the top level. Casting D1 straight to DbLike therefore leaves
// `res.changes` UNDEFINED at runtime, silently breaking those guards on real D1.
// AND `meta.changes` IS NOT "ROWS MY STATEMENT WROTE". D1 reports it as a delta
// of SQLite's `total_changes()`, which counts rows written by AFTER TRIGGERS as
// well as the statement's own. On a trigger-bearing table a statement that wrote
// one row therefore reports 2, and a `changes === 1` guard reads its own success
// as a failure — S-006 X1 (Z-01) found four of those in `fan-auth.ts`, one of
// which executed a fan's "cancel this change" as a deletion of their verified
// address. WHICH WRITES ARE INFLATED IS `INFLATING_WRITES` BELOW, not a list in
// this comment: the old comment named four trigger tables and missed `accounts`,
// whose DELETE cascades into `subscriptions` (migrations/0083), and FK actions
// count in `total_changes()` exactly as trigger rows do. `INFLATING_WRITES` is
// held equal to the live schema, and every `.changes` read in src/ is held off
// it, by test/i4-05-r1-db-seam.test.ts (I3-1130). So: A GUARD ON AN INFLATING
// WRITE DOES NOT COUNT, IT ENUMERATES —
// `INSERT/UPDATE … RETURNING <pk>` executed with `.all()`, guarded on
// `results.length`, which no trigger can inflate. `changes` remains correct, and
// remains the right tool, everywhere else. `test/sqlite-double.ts` returns the
// same total_changes() delta so the double cannot hide the difference.
// This adapter lifts `meta.changes` to `{ changes }`; `all()`/`first()` already
// match D1's shape; `batch()` unwraps each statement to its underlying D1
// prepared statement so D1's ATOMIC batch runs the real statements. The
// real-D1 fidelity suite (docs/D1-FIDELITY.md) exercises this against workerd.

// --- I3-1899: D1's bind set, enforced at the adapter ---------------------------
/** What D1 binds: null, number, string, boolean (stored 1/0), ArrayBuffer (and a
 * view of one) as BLOB. `undefined` is a D1_TYPE_ERROR on real D1 while the
 * better-sqlite3 double silently binds it as NULL; the double also binds bigint
 * and flattens arrays. */
export type DbBindValue = string | number | boolean | null | ArrayBuffer | ArrayBufferView;

/** One bind value as D1 takes it, or a throw naming the parameter. Booleans
 * become 1/0 here — D1's own conversion — so the adapter binds the same value
 * over real D1 and over the double (which refuses a boolean). */
export function toD1BindValue(v: unknown, i: number): string | number | null | ArrayBuffer | ArrayBufferView {
  if (v === null || typeof v === "string" || typeof v === "number") return v;
  if (typeof v === "boolean") return v ? 1 : 0;
  if (v instanceof ArrayBuffer || ArrayBuffer.isView(v)) return v;
  throw new TypeError(`d1().bind: parameter ${i + 1} is ${v === undefined ? "undefined" : Array.isArray(v) ? "an array" : typeof v}, which D1 does not bind`);
}

class D1Stmt implements DbStatement {
  constructor(
    private readonly stmt: D1PreparedStatement,
    private readonly owner: D1Database,
  ) {}
  /** The underlying prepared statement, for `batch()` to pass to D1 directly. */
  get raw(): D1PreparedStatement {
    return this.stmt;
  }
  /** The D1Database this statement was prepared against — batch() refuses a
   * statement from another binding, as the test double refuses a foreign handle (I3-2148). */
  get ownerDb(): D1Database {
    return this.owner;
  }
  bind(...values: unknown[]): DbStatement {
    return new D1Stmt(this.stmt.bind(...values.map(toD1BindValue)), this.owner);
  }
  async run(): Promise<{ changes: number }> {
    // Real D1 reports rows-affected at `meta.changes`; the better-sqlite3 test
    // double (also passed through this adapter by the HTTP integration tests)
    // reports it top-level as `changes`. Accept either so the one adapter is
    // faithful over both, without runtime type-sniffing.
    const r = (await this.stmt.run()) as { meta?: { changes?: number }; changes?: number };
    return { changes: r.meta?.changes ?? r.changes ?? 0 };
  }
  async all<T = Record<string, unknown>>(): Promise<{ results: T[] }> {
    const r = await this.stmt.all<T>();
    return { results: r.results ?? [] };
  }
  async first<T = Record<string, unknown>>(): Promise<T | null> {
    return (await this.stmt.first<T>()) ?? null;
  }
}

/** Adapt a real D1Database to the DbLike the trust core is written against. */
export function d1(db: D1Database): DbLike {
  return {
    prepare: (sql: string) => {
      assertAppendOnlyStatement(sql);
      return new D1Stmt(db.prepare(sql), db);
    },
    // I3-1987 a (I22): each batch result carries `changes` lifted from `meta.changes`, exactly as `run()` does
    // above, keeping D1's own `results`/`meta`/`success` intact — a caller reading `r.changes` on a batch
    // result read `undefined` (so 0) on real D1 while the test double reported the true count.
    batch: async (statements: DbStatement[]) =>
      (await db.batch(
        statements.map((s) => {
          // Every batched statement must originate from THIS adapter's prepare()
          // so it carries a real D1 prepared statement. A foreign statement would
          // otherwise unwrap to `undefined` and fail opaquely inside D1's batch;
          // throw a clear error instead (defense in depth — no call site mixes
          // provenance today). A statement prepared against ANOTHER D1Database
          // (a second binding) is refused the same way (I3-2148).
          if (!(s instanceof D1Stmt) || s.ownerDb !== db) throw new Error("d1().batch: statements must come from the same d1() handle");
          return s.raw;
        }),
      )).map((r) => {
        const x = r as { meta?: { changes?: number }; changes?: number };
        return { ...x, changes: x?.meta?.changes ?? x?.changes ?? 0 };
      }),
  };
}

// --- Bound-parameter chunking -------------------------------------------------
/**
 * D1 CAPS BOUND PARAMETERS AT 100 PER QUERY. Cross it and D1 rejects the whole
 * statement with `D1_ERROR: too many SQL variables`, which surfaces as a 500 on
 * whatever route built the list — including public, unauthenticated ones.
 *
 * The better-sqlite3 double the suite runs on has a ~32k ceiling, so NOTHING in
 * `pnpm test` can reproduce this. It is reproduced in the real-D1 fidelity suite
 * (`test-fidelity/param-cap.test.ts`), which pins both the ceiling itself and
 * that this constant sits under it.
 *
 * This knowledge used to live in a comment in `rankings.ts` while three other
 * `IN (...)` builders repeated the bug (audit v10: `social.ts` x2,
 * `operators.ts`, `events.ts`). It is a shared helper now so the next builder
 * inherits the fix instead of the comment.
 *
 * 90, not 100: leaves room for a caller that binds its own extra parameters
 * alongside the chunk (the exact shape that made `discover()` fail at 100 + 1).
 */
export const IN_CHUNK = 90;

/**
 * Split `items` into chunks that are safe to expand into a single `IN (...)`.
 *
 * `extraBindings` is the number of parameters the caller binds OUTSIDE the IN
 * list in the same statement; the chunk size is reduced by that many so the
 * total still clears the cap. De-duplicates while preserving first-seen order —
 * an IN list never needs a repeat, and fewer holes is strictly safer.
 *
 * Returns `[]` for an empty input: a caller that must always issue a statement
 * (a constant-shape read, e.g. the delisting-oracle guard in `social.ts`) is
 * responsible for supplying its own sentinel BEFORE calling this.
 */
export function inChunks<T>(items: readonly T[], extraBindings = 0): T[][] {
  if (extraBindings < 0 || !Number.isInteger(extraBindings)) throw new Error("inChunks: extraBindings must be a non-negative integer");
  const size = IN_CHUNK - extraBindings;
  if (size < 1) throw new Error(`inChunks: extraBindings (${extraBindings}) leaves no room under the ${IN_CHUNK}-parameter budget`);
  const distinct = [...new Set(items)];
  const out: T[][] = [];
  for (let i = 0; i < distinct.length; i += size) out.push(distinct.slice(i, i + size));
  return out;
}

/** Group an integer with thousands separators WITHOUT Intl (avoids a locale
 * dependency in the worker), for a value that is already a bounded, non-negative
 * request count. */
function groupInt(n: number): string {
  const v = Number.isFinite(n) && n > 0 ? Math.floor(n) : 0;
  const s = String(v);
  let out = "";
  for (let i = 0; i < s.length; i++) {
    if (i > 0 && (s.length - i) % 3 === 0) out += ",";
    out += s[i];
  }
  return out;
}

/** Race `send` against `ms`; a timeout resolves `"ambiguous"` (JNOPS-NEW-01, D-0862(1),
 * as `alerts.ts` `withSendTimeout`): we stopped waiting on a request the provider may
 * still accept, so the caller settles the row as sent and never re-sends it. A timeout
 * says nothing about the RECIPIENT, so it is never `"permanent"`. The timer is cleared
 * once the race settles. */
async function withEmailTimeout(send: Promise<EmailOutcome>, ms: number): Promise<EmailOutcome> {
  let timer: ReturnType<typeof setTimeout> | undefined;
  try {
    return await Promise.race([
      send,
      new Promise<EmailOutcome>((resolve) => {
        timer = setTimeout(() => resolve("ambiguous"), ms);
      }),
    ]);
  } finally {
    if (timer !== undefined) clearTimeout(timer);
  }
}

function newCode(): string {
  // Same shape as the bio-code (50 bits, unambiguous base32, no modulo bias):
  // single-use, 1h TTL, issuance-limited — ample against online guessing.
  const bytes = crypto.getRandomValues(new Uint8Array(10));
  const alpha = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";
  let s = "";
  for (const b of bytes) s += alpha[b % 32];
  return s;
}

/** I3-0964 (D-0913(3)): THE ONE OTP/verification-code hash, for fan_otp,
 * creator_otp and creator_contacts.verify_code_hash. With a usable SESSION_SECRET
 * it is hex HMAC-SHA256 under that secret (`internalServiceCredential`,
 * packages/session) over `dhwaja/otp/<store>.v1:<code>`, so a database reader
 * cannot invert a 10^6 code space offline. With no usable secret it is the
 * legacy unkeyed sha256 hex (dual_control.ts withheldParamsKey's posture). This
 * function never refuses: every production door passes c.env.SESSION_SECRET, and
 * refusing a missing secret is the DOOR's job — the session-minting doors answer
 * 503 `sessions_unavailable`, and so do the creator recovery doors
 * (`/creator/recovery/start` and `/file`, index.ts, I15-FX1 RG-AUTH-01), which
 * mint no session. A door without that gate stores and compares the unkeyed form. */
export async function otpCodeHash(secret: unknown, store: OtpStore, raw: string): Promise<string> {
  if (usableSessionSecret(secret)) {
    const mac = await internalServiceCredential(secret, `dhwaja/otp/${store}.v1:${raw}`);
    if (mac !== null) return mac;
  }
  const d = await sha256Bytes(new TextEncoder().encode(raw));
  return [...d].map((b) => b.toString(16).padStart(2, "0")).join("");
}

         pending_email = CASE WHEN creator_contacts.email_verified = 1 THEN excluded.email ELSE NULL END,
         pending_created_at = CASE WHEN creator_contacts.email_verified = 1 THEN excluded.updated_at ELSE NULL END,
         verify_code_hash = excluded.verify_code_hash,
         verify_expires_at = excluded.verify_expires_at,
         verify_target = excluded.verify_target,
         updated_at = excluded.updated_at WHERE NOT (creator_contacts.email_verified = 1 AND creator_contacts.email IS NOT NULL AND NOT EXISTS (SELECT 1 FROM passkeys WHERE passport_id = creator_contacts.passport_id AND status = 'active')) RETURNING passport_id`,
    )
    .bind(passportId, addr, await otpCodeHash(otpSecret, "creator-email", code), new Date(nowMs + VERIFY_TTL_MS).toISOString(), addr, ts, ts).all<{ passport_id: string }>();
  if (upserted.results.length !== 1) throw new DigestError("add_passkey_first", 409);
  return { code };
}

/**
 * Confirm the emailed code: hash compare, expiry, atomic single-use flip.
 *
 * ONE CLAIM, THEN TWO STAMPS, MUTUALLY EXCLUSIVE, ALL GUARDED IN SQL — the same
 * structure `redeemOtp` uses on the fan side, and for the same reasons. The
 * claim is the single-use flip: it clears the hash under the hash's own WHERE
 * and RETURNS the columns the arm is read from, so the arm choice is made from
 * a row no second caller can still claim (S10-2(a), U-15 — the verdict is
 * enumerated rows, never `changes`).
 *
 *   1. THE SWAP. A row holding a pending replacement is promoted in ONE
 *      statement: the pending address becomes the address, the pending columns
 *      clear, and `email_verified` is re-stamped together — so a creator whose
 *      verified address is their way back in is never, at any instant, without
 *      one.
 *   2. FIRST ENROLMENT, unchanged: an unverified row becomes verified.
 *
 * Each is bound to `verify_target`, the address the outstanding code was
 * actually mailed to, so a code can only ever confirm its own recipient. Without
 * that binding a code mailed to B would stamp whatever the columns held at
 * redemption time — and a second `setCreatorEmail` refused by the daily cap has
 * already moved them.
 *
 * THE ARM WRITE IS BOUND TO THE CLAIM, NOT TO THE ROW'S SHAPE (RUN-0015 D-1,
 * `:279` and `:298`). The claim and the act are two statements — they have to be,
 * because the swap is atomic with the token purge — so the act cannot re-derive
 * its verdict from live state: it carries `AND pending_email = ?` / `AND email = ?`
 * bound to the values the claim RETURNED, and it is enumerated by its own
 * `RETURNING` with 409 on anything but one row. A concurrent
 * `POST /account/creator-contact`, a cancel, or the pending sweep landing in that
 * window makes the act match nothing and the confirmation refuse — where the
 * shape guards (`pending_email IS NOT NULL` / `IS NULL`) promoted whichever
 * address the window had left in the column, and answered `ok` when it had left
 * none.
 */
export async function verifyCreatorEmail(db: DbLike, passportId: string, code: string, nowMs: number, now: () => string = () => new Date().toISOString(), otpSecret?: unknown): Promise<{ displaced: string | null }> {
  if (typeof code !== "string" || !/^[A-Z2-9]{10}$/.test(code.trim().toUpperCase())) throw new DigestError("bad code", 400);
  const hash = await otpCodeHash(otpSecret, "creator-email", code.trim().toUpperCase());
  const ts = now();
  const nowIso = new Date(nowMs).toISOString();
  // S10-2(a) — THE CODE IS CLAIMED BY ONE STATEMENT AND THE CLAIM IS ENUMERATED,
  // NOT COUNTED (U-15, `db.ts:30-42`). The single-use flip is this UPDATE's own
  // WHERE clause — hash and expiry are compared IN SQL, never in JS between a
  // read and a write — and its verdict is `results.length` off `RETURNING`,
  // which no AFTER TRIGGER can inflate. `creator_contacts` bears no trigger
  // today, so the old `swap.changes + first.changes !== 1` guard was correct and
  // dormant; one trigger on this table would have turned a confirmed code into a
  // 409, which is what Z-01 found four times in `fan-auth.ts`.
  // The claim carries the columns the arm is chosen from. They are immutable for
  // the life of the code: `verify_target` is written by `setCreatorEmail` in the
  // same statement as the hash, so a matching hash always names its own
  // recipient, and a second `setCreatorEmail` replaces BOTH (which is what makes
  // "a code cannot confirm an address it was not mailed to" hold).
  const claimed = await db
    .prepare(
      `UPDATE creator_contacts SET verify_code_hash = NULL, verify_expires_at = NULL
       WHERE passport_id = ? AND verify_code_hash = ? AND verify_expires_at > ?
       RETURNING email, pending_email, verify_target, email_verified`,
    )
    .bind(passportId, hash, nowIso)
    .all<{ email: string | null; email_verified: number; pending_email: string | null; verify_target: string | null }>();
  if (claimed.results.length !== 1) throw new DigestError("code is wrong or expired", 409);
  const row = claimed.results[0]!;
  // 0051 — A CONFIRMATION RETIRES THE PASSPORT'S ONE-CLICK TOKENS, AND IT DOES
  // SO IN THE CONFIRMATION'S OWN BATCH. Each unspent `creator_digest_email`
  // token is a capability over whatever address the passport holds
  // (`subject_id = passport_id`, `unsubscribe.ts:172` — there is no per-address
  // key), so every one of them was handed to a mailbox that is about to stop
  // being the contact. S-010 W0 (claim `digest-purge-window`): the DELETE used
  // to be a statement AFTER the swap had already committed, and D1 spans no
  // transaction across two `.run()` calls, so a one-click spent in that window
  // switched off the NEW address's mail. One `db.batch` is one atomic act: there
  // is no instant at which the row shows the new address and a live old token.
  // ONLY THE SWAP ARM RETIRES, unchanged: `first` is a first-time confirmation
  // of the address already on the row, where nothing changed hands, and
  // `slice21-unsubscribe.test.ts:315-323` pins that arm as retiring nothing.
  // S10-2(b)'s "both arms" clause is refused here and reported as a
  // BRIEF-DELTA rather than taken by weakening that guard.
  if (row.pending_email !== null && row.pending_email === row.verify_target) {
    //   1. THE SWAP. A row holding a pending replacement is promoted in ONE
    //      statement: the pending address becomes the address, the pending
    //      columns clear, and `email_verified` is re-stamped together — so a
    //      creator whose verified address is their way back in is never, at any
    //      instant, without one. EVERY statement here is BOUND TO THE CLAIMED
    //      VALUES (`= ?` off the RETURNING row, never the live column): the
    //      address written is `row.pending_email`, and both the swap and the
    //      purge apply only while the row still holds the pair the claim named.
    //      A `setCreatorEmail`, a `removeCreatorEmail` or the pending sweep
    //      (`creator-recovery.ts:715`) landing in the window therefore makes the
    //      whole act match nothing — and the act is its own verdict: the swap
    //      RETURNS its key and 0 rows is a 409, so a confirmation can never
    //      report `ok` for a promotion that did not happen, and the purge can
    //      never retire the LIVE address's one-click tokens for a swap that did
    //      not happen (its EXISTS carries the same claim).
    const armed = (await db.batch([
      db
        .prepare(
          `DELETE FROM unsubscribe_tokens WHERE stream = 'creator_digest_email' AND subject_id = ?
             AND EXISTS (SELECT 1 FROM creator_contacts WHERE passport_id = ? AND pending_email = ?) AND (? = 0 OR EXISTS (SELECT 1 FROM passkeys WHERE passport_id = ? AND status = 'active'))`,
        )
        .bind(passportId, passportId, row.pending_email, row.email_verified, passportId),
      db // I3-0034 (I34-W3-PASSKEY H4): swapping out a VERIFIED address (the claim's email_verified) needs an active passkey, decided here and in the purge above, in the act's own batch
        .prepare(
          `UPDATE creator_contacts SET email = ?, pending_email = NULL, pending_created_at = NULL,
             email_verified = 1, verify_target = NULL, verify_code_hash = NULL, verify_expires_at = NULL, updated_at = ?
           WHERE passport_id = ? AND pending_email = ? AND email = ? AND (? = 0 OR EXISTS (SELECT 1 FROM passkeys WHERE passport_id = ? AND status = 'active'))
           RETURNING passport_id`,
        )
        .bind(row.pending_email, ts, passportId, row.pending_email, row.email, row.email_verified, passportId),
    ])) as Array<{ results?: unknown[] } | undefined>;
    // U-15 AT THE BATCH DOOR: the verdict is the swap's own RETURNED rows, which
    // no AFTER TRIGGER can inflate (`db.ts:30-43`). The element is real D1's
    // `D1Result` (`db.ts:75-85` hands D1's array back unadapted; the double
    // returns the same shape, `test/sqlite-double.ts` `runBatch`).
    if ((armed[1]?.results ?? []).length !== 1) throw new DigestError(row.email_verified === 1 && (await db.prepare("SELECT 1 AS x FROM passkeys WHERE passport_id = ? AND status = 'active'").bind(passportId).first()) === null ? "add_passkey_first" : "code is wrong or expired", 409); // which refusal to SAY only: the act above decided
    // KREF A-18 (I5 RGC-02) — the address this swap displaced, read off the
    // CLAIM and proven by the swap's own RETURNING above, never by a re-read.
    return { displaced: row.email_verified === 1 && row.email !== row.pending_email ? row.email : null };
  }
  //   2. FIRST ENROLMENT: an unverified row becomes verified — bound to the
  //      CLAIMED address the same way (`AND email = ?` off the RETURNING row),
  //      because on an unverified row `setCreatorEmail` replaces `email` IN
  //      PLACE (`:160`), so `pending_email IS NULL` alone stamps whatever
  //      address the window left behind. Enumerated by RETURNING, 409 on 0.
  if (row.pending_email === null && row.email !== null && row.email === row.verify_target) {
    const armed = await db
      .prepare(
        `UPDATE creator_contacts SET email_verified = 1, verify_target = NULL,
           verify_code_hash = NULL, verify_expires_at = NULL, updated_at = ?
         WHERE passport_id = ? AND pending_email IS NULL AND email = ?
         RETURNING passport_id`,
      )
      .bind(ts, passportId, row.email)
      .all<{ passport_id: string }>();
    if (armed.results.length !== 1) throw new DigestError("code is wrong or expired", 409);
    return { displaced: null };
  }
  // The code was live but names neither column — no address is confirmed and the
  // answer is the one every other failure gets. The claim above has already
  // spent it: a code whose target has moved is dead in either spelling.
  throw new DigestError("code is wrong or expired", 409);
}

/**
 * Take the contact address off the passport — the creator's twin of
 * `removeFanEmail`, and the control 0043 owed and did not have.
 *
 * WHY IT IS NEEDED NOW. Before 0043 a verified creator contact was a
 * notification channel, and a single `setCreatorEmail` call neutralised a
 * hostile one instantly by clearing `email_verified`. It is now an
 * AUTHENTICATION FACTOR — it files into the operator recovery queue as
 * `creator:<passport_id>`, the strongest provenance this product emits — and the
 * pending-address change deliberately stopped a replacement from clearing the
 * verified one. So anyone who held a creator session ONCE (a stolen bearer, a
 * borrowed device, a hijack window) could verify their own address and keep a
 * session-independent, epoch-independent door that `bumpCreatorEpoch`,
 * `operatorResetPasskeys` and logout-everywhere all leave standing. There has to
 * be a way to take it away.
 *
 * TWO OUTCOMES, in `removeFanEmail`'s order:
 *
 *   1. A PENDING change is cancelled first. That is the common case (a typo),
 *      it is not a removal at all, and doing it first means cancelling never
 *      trips the guard below.
 *   2. Otherwise the verified address is removed — GUARDED ON A REMAINING WAY
 *      IN. A passport with no active passkey and no verified address has no
 *      self-service door left at all, which is precisely the lockout 0043
 *      exists to end; removing the last one would be handing a creator the
 *      trapdoor with the label still on it. The guard rides INSIDE the DELETE,
 *      not in JS before it, so a concurrent passkey revoke cannot slip between
 *      the check and the write.
 */
export async function removeCreatorEmail(db: DbLike, passportId: string, now: () => string = () => new Date().toISOString()): Promise<{ removed: "pending" | "address" }> {
  const { removed } = await removeCreatorEmailDisplacing(db, passportId, now);
  return { removed };
}

/** `removeCreatorEmail`, also naming the VERIFIED address the removal cleared
 * (null for a pending-cancel or an unverified address) — the route's KREF A-18
 * notice input (I5 RGC-02). */
export async function removeCreatorEmailDisplacing(db: DbLike, passportId: string, now: () => string = () => new Date().toISOString()): Promise<{ removed: "pending" | "address"; displaced: string | null }> {
  const ts = now();
  const cancelled = await db
    .prepare("UPDATE creator_contacts SET pending_email = NULL, pending_created_at = NULL, verify_code_hash = NULL, verify_expires_at = NULL, verify_target = NULL, updated_at = ? WHERE passport_id = ? AND pending_email IS NOT NULL")
    .bind(ts, passportId)
    .run();
  if ((cancelled.changes ?? 0) > 0) return { removed: "pending", displaced: null };

  // KREF A-18 (I5 RGC-02) — a removed VERIFIED address is a displaced one, and
  // it is told (creator-contact-routes.ts, `/account/creator-contact/remove`).
  // The address is read in the REMOVAL'S OWN BATCH (one transaction), and counts
  // only when the UPDATE RETURNED its row — so the value is exactly the one this
  // removal cleared, never a separate read.
  const removal = (await db.batch([
    db.prepare("SELECT email FROM creator_contacts WHERE passport_id = ? AND email_verified = 1").bind(passportId),
    db
      .prepare(
        `UPDATE creator_contacts SET email = NULL, email_verified = 0, pending_email = NULL, pending_created_at = NULL,
           verify_code_hash = NULL, verify_expires_at = NULL, verify_target = NULL, updated_at = ?
         WHERE passport_id = ?
           AND (SELECT COUNT(*) FROM passkeys WHERE passport_id = ? AND status = 'active') >= 1
         RETURNING passport_id`,
      )
      .bind(ts, passportId, passportId),
  ])) as Array<{ results?: Array<{ email?: string | null }> } | undefined>;
  if ((removal[1]?.results ?? []).length === 0) {
    throw new DigestError("register a passkey first — this address is the only way back into this passport if you lose your device", 409);
  }
  // 0051 — AND THE LIVE ONE-CLICK TOKENS THAT ADDRESS HOLDS.
  //
  // Each unspent token is a capability that turns this passport's digest email
  // off, and it was handed to the mailbox that is being removed. Left behind,
  // whoever holds the OLD address keeps that lever for up to the TTL — including
  // over a NEW address confirmed afterwards. The act is benign and reversible,
  // but "somebody who no longer has any relationship with this passport can
  // still switch off its protection mail" is not a reach to leave open by
  // omission. Best effort, after the removal has already succeeded: a failure
  // here must not undo a removal the creator was told happened.
  await db.prepare("DELETE FROM unsubscribe_tokens WHERE stream = 'creator_digest_email' AND subject_id = ?").bind(passportId).run();
  return { removed: "address", displaced: removal[0]?.results?.[0]?.email ?? null };
}

const PUSH_TOKEN_RE = /^[A-Za-z0-9_:.\-]{16,512}$/;
export const MAX_PUSH_TOKENS = 5;
/** I25-FX-DATA (RG-07): the machine refusal `storeVerifiedCreatorPush` throws (409) when the list already holds MAX_PUSH_TOKENS. */
export const PUSH_DEVICE_LIMIT = "push_device_limit";

/** I3-0203 b (I25): how long a creator push-possession challenge may be spent. One push round trip. */
export const CREATOR_PUSH_CHALLENGE_TTL_MS = 5 * 60 * 1000;

/** The platform/token shape both creator push steps refuse on (400). Exported so the route checks it BEFORE
 * spending an allowance. */
export function pushTokenShape(platform: string, token: string): void {
  if (platform !== "apns" && platform !== "fcm") throw new DigestError("platform must be apns or fcm", 400);
  if (typeof token !== "string" || !PUSH_TOKEN_RE.test(token)) throw new DigestError("bad push token", 400);
}

/**
 * I3-0203 b (I25, census pick A): PROOF OF POSSESSION, the fan device flow's twin (`accounts.ts`
 * createDeviceChallenge / redeemDeviceChallenge). Step 1 records (passport, token, a fresh secret `code`)
 * for CREATOR_PUSH_CHALLENGE_TTL_MS; the route pushes `code` to the token and answers only the challenge id.
 * Step 2 is `registerCreatorPush`. A session alone can no longer attach a device: only whoever receives the
 * push holds the code.
 */
export async function createCreatorPushChallenge(
  db: DbLike,
  passportId: string,
  platform: string,
  token: string,
  nowMs: number = Date.now(),
  now: () => string = () => new Date().toISOString(),
): Promise<{ challenge_id: string; code: string }> {
  pushTokenShape(platform, token);
  const challenge_id = `cpc_${randomHex(20)}`;
  const code = randomHex(8);
  await db
    .prepare("INSERT INTO creator_push_challenges (challenge_id, passport_id, platform, push_token, code, created_at, expires_at) VALUES (?, ?, ?, ?, ?, ?, ?)")
    .bind(challenge_id, passportId, platform, token, code, now(), new Date(nowMs + CREATOR_PUSH_CHALLENGE_TTL_MS).toISOString())
    .run();
  return { challenge_id, code };
}

/**
 * I3-0203 b (I25): step 2 — the token is stored ONLY when the pushed code comes back. The challenge is SPENT
 * by its first redeem, right or wrong (`DELETE ... RETURNING` keyed by id AND the session's passport): a
 * wrong code burns it, so a code is never guessed at more than once per challenge. The code and the expiry
 * are judged on the RETURNED values (INV-058), and only then is the token written. false = nothing stored.
 */
export async function registerCreatorPush(
  db: DbLike,
  passportId: string,
  challengeId: string,
  code: string,
  nowMs: number = Date.now(),
  now: () => string = () => new Date().toISOString(),
  /** I26-Q7 (a): the creator-device-changed notice goes through this mailer when a NEW device is stored. */
  mailer?: Mailer,
): Promise<boolean> {
  if (typeof challengeId !== "string" || typeof code !== "string" || challengeId.length === 0 || code.length === 0) return false;
  const row = await db
    .prepare("DELETE FROM creator_push_challenges WHERE challenge_id = ? AND passport_id = ? RETURNING platform, push_token, code, expires_at")
    .bind(challengeId, passportId)
    .first<{ platform: string; push_token: string; code: string; expires_at: string }>();
  if (!row || row.code !== code || !(Date.parse(row.expires_at) > nowMs)) return false;
  // I26-Q7 (a): true only when a NEW device row was written; the call keeps its own line (i25 census, FAMILY-019).
  const added =
    await storeVerifiedCreatorPush(db, passportId, row.platform, row.push_token, now);
  if (added) await notifyCreatorDeviceChanged(db, mailer, passportId, "added", now);
  return true;
}

/** Cron prune: drop expired creator push challenges (the twin of `pruneDeviceChallenges`). */
export async function pruneCreatorPushChallenges(db: DbLike, beforeIso: string): Promise<number> {
  return (await db.prepare("DELETE FROM creator_push_challenges WHERE expires_at < ?").bind(beforeIso).run()).changes ?? 0;
}

/** THE token write, reached in production ONLY from `registerCreatorPush` above, after the code came back
 * (i25-creator-push-challenge.test.ts censuses its callers). Exported for test seeding. */
export async function storeVerifiedCreatorPush(db: DbLike, passportId: string, platform: string, token: string, now: () => string = () => new Date().toISOString()): Promise<boolean> {
  pushTokenShape(platform, token);
  // I25-FX-DATA (RG-07, I3-0203 b): NO SILENT EVICTION. This used to be insert-then-trim, so a session holder's own
  // devices, each proving possession, pushed the creator's real phone off the list. Now ONE conditional INSERT carries
  // the cap (the count and the write are one statement, so a burst cannot overshoot it — the reason the trim existed):
  // a sixth device is refused with 409 `push_device_limit` and the creator removes one first
  // (`DELETE /account/creator-push`). A token already held is not a sixth device: re-registering it stays ok.
  const r = await db
    .prepare(
      "INSERT OR IGNORE INTO creator_push_tokens (passport_id, platform, push_token, created_at) " +
        "SELECT ?, ?, ?, ? WHERE (SELECT COUNT(*) FROM creator_push_tokens WHERE passport_id = ?) < ?",
    )
    .bind(passportId, platform, token, now(), passportId, MAX_PUSH_TOKENS)
    .run();
  // I26-Q7: true = a NEW device row was written (the caller's "added" notice); false = the token was already held.
  if ((r.changes ?? 0) > 0) return true;
  const held = await db.prepare("SELECT 1 AS one FROM creator_push_tokens WHERE passport_id = ? AND push_token = ?").bind(passportId, token).first<{ one: number }>();
  if (!held) throw new DigestError(PUSH_DEVICE_LIMIT, 409);
  return false;
}

/** JA-05 / I3-0203 b (I26): a creator's push device as the creator sees it. Never the token. `last_used_at` is the
 * last ACCEPTED push send (migration 0220, stamped in deliverDigests' push loop), null if none since 0220. */
export interface CreatorPushDevice {
  id: string;
  platform: "apns" | "fcm";
  added_at: string;
  last_used_at: string | null;
}

/** The id shape `creatorPushDeviceId` mints; the remove door refuses anything else before a read. */
export const CREATOR_PUSH_DEVICE_ID_RE = /^pd_[0-9a-f]{24}$/;

/**
 * JA-05 (I26): a stable, one-way id for one `creator_push_tokens` row — the table has no id column and the token is a
 * secret the list must not return. Canonical's single hasher over a domain label, the passport, the platform and the
 * token: the same row always gets the same id, and the id does not reveal the token.
 */
export async function creatorPushDeviceId(passportId: string, platform: string, token: string): Promise<string> {
  const d = await sha256Bytes(new TextEncoder().encode(`dhwaja-creator-push-device-v1\n${passportId}\n${platform}\n${token}`));
  return `pd_${[...d.slice(0, 12)].map((b) => b.toString(16).padStart(2, "0")).join("")}`;
}

/** JA-05 (I26): the caller's OWN devices (keyed by passport_id), oldest first. At most MAX_PUSH_TOKENS rows exist per
 * passport (storeVerifiedCreatorPush refuses a sixth), so the read is unpaged. */
export async function listCreatorPushDevices(db: DbLike, passportId: string): Promise<CreatorPushDevice[]> {
  const rows = (
    await db
      .prepare("SELECT platform, push_token, created_at, last_used_at FROM creator_push_tokens WHERE passport_id = ? ORDER BY created_at ASC, platform ASC, push_token ASC")
      .bind(passportId)
      .all<{ platform: "apns" | "fcm"; push_token: string; created_at: string; last_used_at: string | null }>()
  ).results;
  const out: CreatorPushDevice[] = [];
  for (const r of rows) out.push({ id: await creatorPushDeviceId(passportId, r.platform, r.push_token), platform: r.platform, added_at: r.created_at, last_used_at: r.last_used_at ?? null });
  return out;
}

/** JA-05 (I26): remove ONE of the caller's own devices by the id the list gave — the door a creator who lost the phone
 * (and so the token) can use. The DELETE is scoped by passport_id AND the matched row's platform AND token, so an id
 * minted for another passport matches nothing here. False when no row of this passport has that id. */
export async function removeCreatorPushById(db: DbLike, passportId: string, id: string): Promise<boolean> {
  if (!CREATOR_PUSH_DEVICE_ID_RE.test(id)) return false;
  const rows = (await db.prepare("SELECT platform, push_token FROM creator_push_tokens WHERE passport_id = ?").bind(passportId).all<{ platform: string; push_token: string }>()).results;
  for (const r of rows) {
    if ((await creatorPushDeviceId(passportId, r.platform, r.push_token)) !== id) continue;
    const d = await db.prepare("DELETE FROM creator_push_tokens WHERE passport_id = ? AND platform = ? AND push_token = ?").bind(passportId, r.platform, r.push_token).run();
    return (d.changes ?? 0) > 0;
  }
  return false;
}

/**
 * I26-Q7 (a): the no-link security notice to the creator's VERIFIED contact address, sent like recovery.ts
 * `notifyFanTakeoverClaim` — through the template mailer, outside the alerts path, best effort (a send failure never
 * fails the act it reports). The outcome is RECORDED either way (FAMILY-003: no loss without a record): one audit
 * row, `creator:device_notice`, detail `<change> sent|undelivered|no_address`. Returns whether the provider took it.
 */
export async function notifyCreatorDeviceChanged(db: DbLike, mailer: Mailer | undefined, passportId: string, change: CreatorDeviceChange, now: () => string): Promise<boolean> {
  const to = (await db.prepare("SELECT email FROM creator_contacts WHERE passport_id = ? AND email_verified = 1 AND email IS NOT NULL").bind(passportId).first<{ email: string }>())?.email ?? null;
  const ts = now();
  const sent = to !== null && mailer !== undefined && (await mailer.send(CREATOR_DEVICE_CHANGED_TEMPLATE, to, { [change]: "1", when_utc: ts }).catch(() => false));
  await db
    .prepare("INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) VALUES (?, 'creator:device_notice', ?, ?, ?, 'creator')")
    .bind(passportId, `${change} ${to === null ? "no_address" : sent ? "sent" : "undelivered"}`, ts, passportId)
    .run();
  return sent;
}

export interface CreatorPushRemovalHold {
  device_id: string;
  confirm_by: "passkey" | "email_code" | "unavailable";
  expires_at: string;
}

/**
 * I26-Q7 (b): a removal asked for with the session alone is HELD, never applied. Picks how the owner confirms —
 * `passkey` when the passport holds an active one, else `email_code` (a one-time code mailed to the verified contact
 * address, stored as its otpCodeHash), else `unavailable` (no factor exists: the hold can only lapse) — writes ONE live
 * hold per device, keyed by the device id (asking again replaces it, with a fresh code and 24h clock), and mails the
 * `removal_requested` notice. null when this passport holds no device with that id (nothing is held).
 */
export async function holdCreatorPushRemoval(
  db: DbLike,
  mailer: Mailer | undefined,
  passportId: string,
  deviceId: string,
  nowMs: number,
  now: () => string = () => new Date().toISOString(),
  otpSecret?: unknown,
): Promise<CreatorPushRemovalHold | null> {
  if (!(await listCreatorPushDevices(db, passportId)).some((d) => d.id === deviceId)) return null;
  const pk = await db.prepare("SELECT COUNT(*) AS n FROM passkeys WHERE passport_id = ? AND status = 'active'").bind(passportId).first<{ n: number }>();
  const to = (await db.prepare("SELECT email FROM creator_contacts WHERE passport_id = ? AND email_verified = 1 AND email IS NOT NULL").bind(passportId).first<{ email: string }>())?.email ?? null;
  const confirm_by: CreatorPushRemovalHold["confirm_by"] = (pk?.n ?? 0) > 0 ? "passkey" : to !== null ? "email_code" : "unavailable";
  const code = confirm_by === "email_code" ? newCode() : null;
  const expires_at = new Date(nowMs + CREATOR_PUSH_REMOVAL_HOLD_MS).toISOString();
  await db
    .prepare(
      `INSERT INTO creator_push_pending_removals (passport_id, device_id, confirm_by, code_hash, created_at, expires_at) VALUES (?, ?, ?, ?, ?, ?)
       ON CONFLICT(passport_id, device_id) DO UPDATE SET confirm_by = excluded.confirm_by, code_hash = excluded.code_hash, created_at = excluded.created_at, expires_at = excluded.expires_at`,
    )
    .bind(passportId, deviceId, confirm_by, code === null ? null : await otpCodeHash(otpSecret, "creator-device", code), now(), expires_at)
    .run();
  if (code !== null && to !== null && mailer !== undefined) await mailer.send(CREATOR_DEVICE_CODE_TEMPLATE, to, { code }).catch(() => false);
  await notifyCreatorDeviceChanged(db, mailer, passportId, "removal_requested", now);
  return { device_id: deviceId, confirm_by, expires_at };
}

export type CreatorPushRemovalOutcome = "removed" | "not_found" | "expired" | "wrong_factor" | "bad_code" | "gone";

/**
 * I26-Q7 (b): the owner's confirmation. SINGLE-SPEND: the hold is DELETED ... RETURNING by the device and the
 * session's passport before anything is judged, so any attempt — right, wrong or late — burns it (a code is never guessed at
 * twice). Judged on the RETURNED row (INV-058): a lapsed hold removes nothing (`expired`); the factor must be the one the
 * hold named (`wrong_factor`); a code must match its hash (`bad_code`). `passkey` means the CALLER has already verified
 * an assertion bound to `creatorPushRemovalBinding(device_id)`. Only then is the device removed, and the `removed`
 * notice mailed.
 */
export async function confirmCreatorPushRemoval(
  db: DbLike,
  mailer: Mailer | undefined,
  passportId: string,
  deviceId: string,
  proof: { kind: "passkey" } | { kind: "email_code"; code: string },
  nowMs: number,
  now: () => string = () => new Date().toISOString(),
  otpSecret?: unknown,
): Promise<CreatorPushRemovalOutcome> {
  const row = await db
    .prepare("DELETE FROM creator_push_pending_removals WHERE passport_id = ? AND device_id = ? RETURNING device_id, confirm_by, code_hash, expires_at")
    .bind(passportId, deviceId)
    .first<{ device_id: string; confirm_by: string; code_hash: string | null; expires_at: string }>();
  if (!row) return "not_found";
  if (!(Date.parse(row.expires_at) > nowMs)) return "expired";
  if (row.confirm_by !== proof.kind) return "wrong_factor";
  if (proof.kind === "email_code" && (row.code_hash === null || (await otpCodeHash(otpSecret, "creator-device", proof.code)) !== row.code_hash)) return "bad_code";
  if (!(await removeCreatorPushById(db, passportId, row.device_id))) return "gone";
  await notifyCreatorDeviceChanged(db, mailer, passportId, "removed", now);
  return "removed";
}

/** I26-Q7 (b): the step-up binding a held removal's passkey confirmation is signed over (its own namespace): the
 * owner's signature says "remove THIS device", whichever hold of it is live. */
export const creatorPushRemovalBinding = (deviceId: string): string => `creator-push-remove:v1:${deviceId}`;

/** I26-Q7: the caller's live holds, for the console's pending state. Never the code. */
export async function listPendingPushRemovals(db: DbLike, passportId: string, nowMs: number): Promise<CreatorPushRemovalHold[]> {
  return (
    await db
      .prepare("SELECT device_id, confirm_by, expires_at FROM creator_push_pending_removals WHERE passport_id = ? AND expires_at > ? ORDER BY created_at ASC, device_id ASC")
      .bind(passportId, new Date(nowMs).toISOString())
      .all<CreatorPushRemovalHold>()
  ).results;
}

/** I26-Q7: cron prune of lapsed holds (handed to I30-INT for scheduled()). A lapse removes no device. */
export async function pruneExpiredPushRemovals(db: DbLike, nowIso: string): Promise<number> {
  return (await db.prepare("DELETE FROM creator_push_pending_removals WHERE expires_at <= ?").bind(nowIso).run()).changes ?? 0;
}

export async function setDigestsEnabled(db: DbLike, passportId: string, on: boolean, now: () => string = () => new Date().toISOString()): Promise<void> {
  const ts = now();
  await db
    .prepare(
      `INSERT INTO creator_contacts (passport_id, digests_enabled, created_at, updated_at) VALUES (?, ?, ?, ?)
       ON CONFLICT(passport_id) DO UPDATE SET digests_enabled = excluded.digests_enabled, updated_at = excluded.updated_at`,
    )
    .bind(passportId, on ? 1 : 0, ts, ts)
    .run();
}

/**
 * 0051 — the EMAIL-ONLY switch, and the way back from an RFC 8058 one-click.
 *
 * Deliberately a SECOND function rather than a second argument to
 * `setDigestsEnabled`: the two switches mean different things (this one stops
 * mail, that one stops the whole stream including push), and the route below
 * accepts them independently so a console screen that sends only one cannot
 * silently reset the other.
 *
 * It does NOT touch `digests_enabled`. Turning email back on while the master
 * switch is off leaves the creator with email on and the stream off, which is
 * exactly what those two settings say — the console renders the combination
 * rather than papering over it.
 */
export async function setDigestEmailEnabled(db: DbLike, passportId: string, on: boolean, now: () => string = () => new Date().toISOString()): Promise<void> {
  const ts = now();
  await db
    .prepare(
      `INSERT INTO creator_contacts (passport_id, digest_email_enabled, created_at, updated_at) VALUES (?, ?, ?, ?)
       ON CONFLICT(passport_id) DO UPDATE SET digest_email_enabled = excluded.digest_email_enabled, updated_at = excluded.updated_at`,
    )
    .bind(passportId, on ? 1 : 0, ts, ts)
    .run();
}

/** The creator's own digest-contact state — booleans only, never the raw email,
 * so the S98 onboarding wizard can read whether the contact step is genuinely
 * done from the SERVER (self-heals across devices) instead of a local mark. */
export interface ContactState {
  email_set: boolean;
  email_verified: boolean;
  /** 0043: a replacement address is waiting for its code. A BOOLEAN — this
   * payload deliberately carries no address at all. */
  email_change_pending: boolean;
  digests_enabled: boolean;
  /**
   * 0051 — the EMAIL half, separate from `digests_enabled` above. False here
   * with `digests_enabled` true is what an RFC 8058 one-click leaves behind: the
   * mail is off and the push is not. The creator has to be able to SEE that and
   * turn it back on, or the mailbox provider's button is a one-way door on a
   * setting the console claims to own.
   */
  digest_email_enabled: boolean;
  push_registered: boolean;
}

export async function contactState(db: DbLike, passportId: string): Promise<ContactState> {
  const c = await db
    .prepare("SELECT email, email_verified, digests_enabled, digest_email_enabled, pending_email FROM creator_contacts WHERE passport_id = ?")
    .bind(passportId)
    .first<{ email: string | null; email_verified: number; digests_enabled: number; digest_email_enabled: number; pending_email: string | null }>();
  const push = await db.prepare("SELECT COUNT(*) AS n FROM creator_push_tokens WHERE passport_id = ?").bind(passportId).first<{ n: number }>();
  return {
    email_set: typeof c?.email === "string" && c.email.length > 0,
    email_verified: c?.email_verified === 1,
    // A CHANGE IN FLIGHT IS A STATE THE CREATOR HAS TO BE ABLE TO SEE. Since
    // 0043 a replacement parks in `pending_email` and the verified address is
    // left alone, so a creator who typed a new one and never confirmed it would
    // otherwise see the screen unchanged and conclude the change failed — and
    // the console would have no way to say "we sent a code to the new address".
    // A BOOLEAN, never the address: this payload is asserted elsewhere to carry
    // no "@" at all, because it is read on a screen the creator does not
    // necessarily control the device for.
    email_change_pending: typeof c?.pending_email === "string" && c.pending_email.length > 0,
    // Digests default ON (a fresh creator with no row still gets the roundup once
    // a verified email exists); an explicit row reflects the stored toggle.
    digests_enabled: c ? c.digests_enabled === 1 : true,
    // Same default as above, and for the same reason: no row means nothing was
    // ever turned off.
    digest_email_enabled: c ? c.digest_email_enabled === 1 : true,
    push_registered: (push?.n ?? 0) > 0,
  };
}

export interface DigestContent {
  /**
   * `daily`/`weekly` are the periodic roundups. `alert` is an EVENT-driven
   * creator notice that does not wait for a period to close — today the
   * pre-expiry verification warning, whose whole value is arriving before a
   * deadline rather than after one. Its `period_key` names the CAUSE (see
   * migration 0023 and `noticeVerificationExpiring`).

  kind: "daily" | "weekly" | "alert";
  period_key: string;
  /**
   * Impersonation section — the reason a daily exists at all. Counts and
   * outcomes only: reporter identities never appear.
   *
   * OPTIONAL since the `alert` kind exists, and absent rather than zeroed on
   * those: an alert is not a reporting window, and "0 new reports naming you"
   * would be a measurement nobody took presented as one that was.
   */
  impersonation?: {
    /**
     * Reports naming this creator that a REVIEWER escalated, filed under the
     * impersonation category. Not "reports that exist": `suspected_passport_id`
     * is stamped by a substring-matching scan and is therefore attacker-choosable,
     * so an unfiltered count was a number a stranger could drive up and Dhwaja
     * would then mail to the victim. See `protection.ts`.
     */
    new_reports_naming_you: number;
    /**
     * WHY THOSE REPORTS NAME THIS CREATOR — the qualifier the count never
     * carried. One value for the whole population, the strongest basis any one
     * item rests on; null when the count is zero or the basis could not be
     * established. See `protection.ts NamingBasis`, which the console reads from
     * the same helper so the mailed sentence and the on-screen one cannot
     * describe the naming differently.
     *
     * OPTIONAL IN THE STORED SHAPE, not merely in the type. Digest rows written
     * before this field existed are read back and rendered by the same code
     * (`digestSummary`, the console's list), so every consumer has to treat its
     * absence as "unqualified" rather than as a defect.
     */
    naming_basis?: NamingBasis | null;
    /**
     * FACE MATCHES THAT NAMED THIS CREATOR — the S66 image pipeline, and nothing
     * to do with the handle matcher.
     *
     * It was called `machine_flags` and mailed to creators as "N machine-flagged
     * LOOKALIKES", which is the one word this codebase reserves for the
     * handle-similarity matcher (`findLookalike`, `exact_lookalike`,
     * `findDomainLookalike`). A creator reading it would reasonably conclude the
     * handle matcher had named accounts resembling their handle. It had not.
     *
     * THE COUNT IS SCOPED BY THE FACE STAGE'S OWN POINTER, not by
     * `suspected_passport_id`, and that correction is the substance of the
     * rename rather than a detail of it. `face-triage.ts` writes
     * `suspected_passport_id = COALESCE(suspected_passport_id, ?)` — it KEEPS
     * whatever the handle stage wrote — and the handle stage always runs first
     * (`reports.ts` stamps it at intake; the face sweep picks up items already
     * `pending`). So the shared column routinely names the creator the HANDLE
     * matcher chose. Counting on it meant: handle stage names A, image check
     * matches B, and A is mailed "an account our image check flagged as using
     * your likeness" about somebody else's face. `caution.ts` documents exactly
     * this decoupling for the public caution and refuses to publish across it;
     * `face-triage.ts` refuses to auto-draft across it. This read now refuses too.
     *
     * IT IS ZERO UNDER THE CURRENT CONFIGURATION, which is a config gate and not
     * an impossibility: the sweep is gated on the `face_triage_enabled` ops
     * setting plus an `IMAGE_SERVICE` binding and a `FACE_TRIAGE_SECRET`
     * (`index.ts`), and none are provisioned. The sentence below has to be right
     * before those three flip, not after.
     */
    face_flags: number;
    warnings_published_protecting_you: number;
    /** Relative PSA path when an active warning protects the creator — the
     * share-this link. Absolute URL is composed at delivery. */
    psa_path: string | null;

  /**

   * (D-1113(1), D-1117(3)): a zero and a count of 1..4 both become `null`. The same three populations reach the
   * creator's own dashboard as day cells (`saved_by`, `card_collectors`, `collectors_total`, roi.ts), withheld
   * below the floor by `analytics-cells.ts` (`c.value < K`), and a weekly cell is the STRICTLY NARROWER grain of the two
   * — one save in one week names one fan more sharply than one save ever.
   * Serving the exact integer here withheld nothing the dashboard withheld.
   */
  activity?: {
    new_saves: number | null;
    new_card_collectors: number | null;
    new_acquisitions: number | null;

  ratings?: Record<string, { percentile: number; bar: number }> | null;
  global?: { verified_creators: number; newly_verified_in_period: number };

   * THE RE-PROOF NAG. Weekly only, one number, and it exists because a control
   * proof now EXPIRES (`PROOF_TTL_MS`, account-standing.ts).

   * A creator who proved an account and then heard nothing for six months would
   * otherwise discover the lapse the way the worst version of this product would
   * tell them: by being dropped out of the registry, losing the report shield on
   * that handle, and having it fall off their own card — all silently, for doing
   * nothing wrong. An expiry a person cannot see coming is a trap, so the same
   * pass that made proofs expire owes them a warning.

   * `accounts_needing_reproof` counts the creator's ACTIVE accounts in the

  reproof?: { accounts_needing_reproof: number };

  proof_expiry?: { expires_at: string; days_remaining: number; accounts: number };

  verification_expiry?: { expires_at: string; days_remaining: number };

  operator_lapse?: { since: string; badge_moves_at: string };

   *

   * The CHALLENGER IS NOT NAMED, deliberately. Nothing has been adjudicated —

  correction_resolved?: { request_id: string; outcome: "resolved" | "declined" };

  contest_decided?: { item_id: string; outcome: "decided" };

  report_outcome?: { case_ref: string; outcome: "published" | "closed" };

   * written by `warnings.ts publishWarning` in the publication's own batch (`warningPublishedNoticeStmt`) to a claimed or
   * verified passport holding the warned account. The account's platform and handle and the approved sentence only: no

  warning_published?: { platform: string; handle: string; text: string };
  /**
   * I3-0034 — A PASSKEY WAS ADDED TO THIS PASSPORT. Written by
   * `passkeys.ts finishPasskeyRegistration` in the enrolment's own batch.
   * `via` says whether an existing passkey approved it; the hint is the
   * credential id's first 8 characters (the console lists the full id).
   */
  passkey_added?: { credential_hint: string; added_at: string; via: "first_enrolment" | "existing_passkey" };
  /**

  passkey_removed?: { credential_hint: string; removed_at: string };
  /**
   * I3-1200 — AN OPERATOR DID NOT APPROVE THE CREATOR'S VERIFICATION REQUEST.
   * The review item id and the decision instant, nothing else: the operator's
   * reason is an internal audit field (`review.ts rejectItem`) and is never
   * carried here. When a new request may be filed is served live by
   * `verification.ts verificationStatus` (`refile_after`), not copied into mail.
   * WRITTEN BY `review.ts rejectItem` (in its decision batch), which builds this
   * shape itself: review.ts cannot import this module without a cycle
   * (review -> digest -> account-standing -> settings -> review).
   * `text` (I34-FX2-CLIENTS) is review.ts VERIFICATION_REJECTED_NOTICE_TEXT, word for word; rows written before carry none.
   */
  verification_rejected?: { item_id: string; rejected_at: string; text?: string };
  /**
   * THE PAYOUT-DESTINATION CHANGE NOTICE — THE HIGHEST-URGENCY THING WE SEND.
   *
   * A payout tag is where a fan's money goes. A change to it stages behind a
   * 24-hour cooling-off (`payout.PAYOUT_COOLING_OFF_MS`) for one reason: so that
   * a creator whose session was briefly taken can SEE the swap and pull it
   * before it renders. The cooling-off buys the time; this notice is what the
   * creator spends it on.
   *
   * Until this existed the notice did not reach them. `alerts.ts
   * buildPayoutChangeStatements` fanned out one `notifications` row per row in
   * `saves` — the creator's FANS — and its docstring claimed it reached the
   * creator too "via a self-follow". Part 4b (#24) had deleted the founder
   * auto-save and the only remaining writer of `saves` is the fan-scoped
   * `saveCreator`, which needs an `accounts` row a creator does not have. Zero
   * savers meant zero statements: an attacker moved the destination and the
   * creator heard nothing, on any channel, for the whole 24 hours. Part 7
   * diagnosed precisely this in `alerts.ts` and fixed only the sibling
   * (`fanOutVerificationExpiring`), leaving the money path on the dead query.
   *
   * `destinations` is a COUNT, never the addresses. An alert email is the last
   * place a creator's payout addresses should be duplicated, and the count is
   * what makes the sentence actionable ("you now have 2") without turning the
   * mail into a second copy of the thing being protected. `effective_at` is the
   * deadline the creator is racing; `null` means the change was IMMEDIATE, which
   * is what a compromise-recovery retraction is. Both are `null` when the caller
   * did not hand down a plan — the notice then says only that the destinations
   * changed, which is still the whole alarm. Present only on `kind: 'alert'`.
   */
  payout_change?: { change_id: string; destinations: number | null; effective_at: string | null };
  /**
   * THE PAYOUT-ACCOUNT CHANGE NOTICE — the Stripe Connect twin of the line
   * above, and a strictly larger exposure than it.
   *
   * `payout_change` is about payout TAGS: where a FAN is invited to send money,
   * one fan at a time. This is about the connected Stripe account, which is
   * where DHWAJA ITSELF transfers a creator's entire cleared balance on the
   * weekly run. It shipped with no ledger row, no alert and no way to change it;
   * the row and the change path came first, and this is the part that makes them
   * worth having — a record nobody reads is not a control.
   *
   * IT IS WHAT THE COOLING-OFF IS FOR. `CONNECT_COOLING_OFF_MS` holds transfers
   * off a newly-bound destination for 24 hours precisely so a creator whose
   * session was briefly taken can pull it before anything moves. That window
   * only helps someone who KNOWS, and this is how they find out.
   *
   * `account_hint` is the last four characters of the `acct_…` id and never the
   * whole thing. Email is a lower-trust channel than the console: enough to
   * recognise an account you set up yourself, not enough to be worth
   * intercepting. The console shows the full id where the disconnect lever is.
   */
  /** `pending_retargeted` (bind arm only; `null` on unbind) — how many payout
   * rows already reserved for this creator will be paid to the NEWLY bound
   * account. `creator_payouts` has no destination column and the payout run
   * reads the binding live, so a confirm moves money that was reserved under
   * the previous one. 0 is a value, never omitted on the bind arm. */
  payout_account_change?: { action: "bound" | "unbound"; account_hint: string; live_at: string | null; pending_retargeted?: number | null };
  /** I3-0997: a payout row reached `paid`, or `failed` (stopped retrying; the
   * money stays reserved). Written by `payoutOutcomeNoticeStatement` in the SAME
   * batch as the transition, bound to it. */
  payout_outcome?: { outcome: "paid" | "failed"; amount_cents: number; period_key: string };
  empty?: boolean;
}

function count(db: DbLike, sql: string, ...binds: unknown[]): Promise<{ n: number } | null> {
  return db
    .prepare(sql)
    .bind(...binds)
    .first<{ n: number }>();
}

/**
 * Warning context for a whole sweep, computed ONCE (review finding #2 — this
 * scan is O(total warning history) and must not run per creator). `protected`
 * = passports with an active impersonation warning right now; `publishedInWindow`
 * = per-protected-passport count of impersonation warnings published in the
 * window. Pass it into buildDigestContent to avoid re-scanning.
 */
export interface WarningContext {
  protectedNow: Set<string>;
  publishedInWindow: Map<string, number>;
}

export async function buildWarningContext(db: DbLike, windowStartIso: string, windowEndIso: string): Promise<WarningContext> {
  const { results } = await db
    .prepare(
      `SELECT subject, event_type, payload, created_at, ${operatorSetPointerSql("trust_events")} AS ptr_operator FROM trust_events WHERE event_type IN ('warning_published','warning_retracted') ORDER BY seq ASC`,
    )
    .bind()
    .all<{ subject: string; event_type: string; payload: string; created_at: string; ptr_operator: number }>();
  const active = new Map<string, boolean>();
  const protects = new Map<string, string | null>(); // subject → protected passport of its latest publish
  // I3-1147 (I15-W2J): in-window publications per SUBJECT since its last retraction; a later
  // `warning_retracted` for the subject drops them, so a withdrawn warning is not counted as protecting anyone.
  const inWindow = new Map<string, string[]>();
  for (const r of results) {
    if (r.event_type === "warning_published") {
      active.set(r.subject, true);
      let pid: string | null = null;
      try {
        const p = JSON.parse(r.payload) as { warning_class?: unknown; protected_passport_id?: unknown };
        if (r.ptr_operator === 1 && p.warning_class === "impersonation" && typeof p.protected_passport_id === "string") pid = p.protected_passport_id;
      } catch {
        pid = null;
      }
      protects.set(r.subject, pid);
      if (pid && r.created_at >= windowStartIso && r.created_at < windowEndIso) {
        inWindow.set(r.subject, [...(inWindow.get(r.subject) ?? []), pid]);
      }
    } else {
      active.set(r.subject, false);
      inWindow.delete(r.subject);
    }
  }
  const publishedInWindow = new Map<string, number>();
  for (const pids of inWindow.values()) for (const pid of pids) publishedInWindow.set(pid, (publishedInWindow.get(pid) ?? 0) + 1);
  const protectedNow = new Set<string>();
  for (const [subject, isActive] of active) {
    const pid = protects.get(subject);
    if (isActive && pid) protectedNow.add(pid);
  }
  return { protectedNow, publishedInWindow };
}

/** RG2C-01 (D-1117(9), D-1118): whether `/p/:id/psa` would serve this passport at `atIso` — verified, not retracted,
 * the grant live at that instant (standing.ts `grantLiveAtSql`), not suppressed, AND an active warning still protects
 * it (`impostorCount`, the fold the route 404s on: index.ts `view.impostor_count === undefined`). Asked when the
 * digest is built and again when it is mailed, so a link that stopped working in between is not sent. */
async function psaServableAt(db: DbLike, passportId: string, atIso: string): Promise<boolean> {
  const row = await db
    .prepare(
      `SELECT 1 AS x FROM passports p WHERE p.passport_id = ? AND p.claim_state = 'verified' AND p.retracted = 0
        AND ${grantLiveAtSql("p.passport_id")}
        AND NOT EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = p.passport_id)`,
    )
    .bind(passportId, atIso)
    .first<{ x: number }>();
  if (row === null || row === undefined) return false;
  return (await impostorCount(db, passportId)) > 0;
}

/** A fan-population count for the weekly digest: served at ROI_K_FLOOR or above, else WITHHELD (`null`), 0
 * included (D-1113(1), D-1117(3)). Same rule and same constant as `roi.ts`'s `kFloor`
 * (`n >= ROI_K_FLOOR ? n : null`). */
function kFloorFans(n: number): number | null {
  return n >= ROI_K_FLOOR ? n : null;
}

export async function buildDigestContent(
  db: DbLike,
  store: ArtifactReader | null,
  passportId: string,
  kind: "daily" | "weekly",
  periodKey: string,
  windowStartIso: string,
  windowEndIso: string,
  warnCtx?: WarningContext,
  /** RG2C-01 (D-1117(9)), JN-02: the instant the digest is built (generateDigests passes its `nowMs`); the PSA link
   * and the ratings block are read at it. The window's end when a caller has none. */
  atIso: string = windowEndIso,
): Promise<DigestContent> {
  // Impersonation involving this creator, in the window. Counts only.
  //
  // ESCALATED AND IMPERSONATION-CATEGORY ONLY (2026-08-30 review). This counted
  // every `fan_report_items` row naming the passport, in any state and any
  // category — and `suspected_passport_id` is stamped by `findLookalike`, which
  // matches on SUBSTRING INCLUSION across every verified creator's handles. So
  // the number was attacker-driven: one report from any fan session, against any
  // unclaimed account whose handle merely contains the victim's, incremented a
  // figure Dhwaja then EMAILED to that victim as "new reports of accounts
  // impersonating you". It also counted items an operator had already dismissed
  // (soft states re-open on a new reporter) and reports filed under
  // `dead_code`/`scam`, which allege nothing about impersonation at all.
  // (The "substring inclusion" phrasing above predates slice 11, which bounded
  // containment at both ends and made it one kind of three. The lever is
  // narrower than the sentence implies; it is still a lever.)
  //
  // `escalated` is the bar because it is a state a REVIEWER moved the item into
  // and no filing can reach. The category half of the predicate is NOT closed
  // the same way — a reporter row can be added to an already-escalated item, so
  // a filer can supply the impersonation category after the fact. See
  // `protection.ts`, which carries the full statement of what this bar does and
  // does not buy. Shared with `/account/protection` through
  // `protection.ts`, so the mailed number and the on-screen signal cannot
  // describe different populations.
  // I3-1145 (D-0913(3)): NEW means escalated within this digest's window — a standing report is not re-announced daily.
  const reportCount = await escalatedImpersonationReports(db, passportId, windowStartIso, windowEndIso);
  // The same population, asked why. Skipped when there is nothing to qualify —
  // `namingBasis` would answer null anyway, and this sweep runs once per
  // claimed creator per period.
  const basis = reportCount > 0 ? await namingBasis(db, passportId) : null;
  const flags = await count(
    db,
    // `$.face.matched_passport_id` — the pointer the FACE stage wrote, never the
    // shared `suspected_passport_id` the handle stage owns. See `face_flags`.
    //
    // I-12: this used to stop at `face_state`, with none of the review bar its
    // sibling `reportCount` (above, via `escalatedImpersonationReports`)
    // enforces — so a machine's own `match_medium` guess, on a row nobody had
    // opened (or one an operator had already dismissed: soft states re-open on
    // a new reporter), was mailed to the victim as a flagged photo match. The
    // bar is now the same one `protection.ts`'s `ESCALATED_IMPERSONATION`
    // states: a REVIEWER moved the item to `escalated`, and a reporter filed it
    // under `impersonation` — `i.state`/`i.item_id` below are that predicate,
    // inlined rather than imported because this file owns no export from
    // `protection.ts` for the face-scoped shape.
    "SELECT COUNT(*) AS n FROM fan_report_items i WHERE json_extract(i.evidence, '$.face.matched_passport_id') = ? " +
      "AND i.face_state IN ('match_high','match_medium') AND i.face_checked_at >= ? AND i.face_checked_at < ? " +
      "AND i.state = 'escalated' AND EXISTS (SELECT 1 FROM fan_report_reporters r WHERE r.item_id = i.item_id AND r.category = 'impersonation')",
    passportId,
    windowStartIso,
    windowEndIso,
  );
  // Warnings published in the window whose pointer names this creator, and
  // whether one actively protects them now. Use the sweep-wide context when
  // provided (one scan for the whole sweep — review finding #2); otherwise
  // fall back to the standalone per-creator scans (tests / one-off calls).
  const ctx = warnCtx ?? (await buildWarningContext(db, windowStartIso, windowEndIso));
  const published = ctx.publishedInWindow.get(passportId) ?? 0;
  const psaLive = ctx.protectedNow.has(passportId);
  // AD-02: `psaLive` only asks whether an impersonation warning is currently
  // active — it says nothing about whether `/p/:id/psa` itself resolves.
  // `protection.ts`'s `protectionState` applies the SAME three conditions that
  // route 404s on (verified, not retracted, not suppressed) before handing out
  // a psa_path; this builder used to skip them, so a creator whose
  // verification lapsed between the warning and this sweep (ordinary
  // `verification_expired`, no retraction or suppression involved) was mailed
  // a fresh digest offering a PSA link that already 404s. Asked only when
  // `psaLive` is true — the common case skips the extra read entirely.
  // I32 B4 / FX2S-14 (I34-W2-EXPIRY-B), RG2C-01 (D-1117(9)): and the grant behind `verified` is live at the instant the
  // digest is built (`atIso`) — a lapsed grant the sweep has not folded 404s /p/:id/psa (protection.ts reads
  // CLAIM_STATE_AT_SQL). Delivery asks `psaServableAt` again at its own instant.
  const psaServable = psaLive ? await psaServableAt(db, passportId, atIso) : false;

  const content: DigestContent = {
    kind,
    period_key: periodKey,
    impersonation: {
      new_reports_naming_you: reportCount,
      naming_basis: basis,
      face_flags: flags?.n ?? 0,
      warnings_published_protecting_you: published,
      psa_path: psaLive && psaServable ? `/p/${passportId}/psa` : null,
    },
  };

  if (kind === "weekly") {
    // #850b (D-0932(1), D-0935(2)): the day object's count — QUALIFIED savers, on the day each was COUNTED. `counted_at`
    // is set in the same batch as the `fan_followed` row the day object folds (accounts.ts saveCreator,
    // sweepQualifiedFollows), so the digest and the creator's analytics serve one number from one event.
    const saves = await count(db, "SELECT COUNT(*) AS n FROM saves WHERE passport_id = ? AND counted_at >= ? AND counted_at < ?", passportId, windowStartIso, windowEndIso);
    const collectors = await count(db, "SELECT COUNT(*) AS n FROM card_collections WHERE passport_id = ? AND created_at >= ? AND created_at < ?", passportId, windowStartIso, windowEndIso);
    const acq = await count(
      db,
      "SELECT COUNT(*) AS n FROM acquisitions a JOIN releases r ON r.release_id = a.release_id WHERE r.passport_id = ? AND a.acquired_at >= ? AND a.acquired_at < ?",
      passportId,
      windowStartIso,
      windowEndIso,
    );
    // Each cell floored on ITS OWN population (INV-004 spatial clause 2) — the
    // three are independent counts over three tables with no total over them,
    // so no complementary term is owed. Suppression, not rounding: a fuzzed
    // count still moves with the fan.
    content.activity = {
      new_saves: kFloorFans(saves?.n ?? 0),
      new_card_collectors: kFloorFans(collectors?.n ?? 0),
      new_acquisitions: kFloorFans(acq?.n ?? 0),
    };
    if (store) {
      try {
        const r = await serveRatings(store, db, passportId, atIso);
        content.ratings = r ? (r.ratings as unknown as NonNullable<DigestContent["ratings"]>) : null;
      } catch {
        content.ratings = null;
      }
    }
    // I34-W2-EXPIRY-B (EXPIRY-A census: buildDigestContent's verified total): verified creators whose grant is live at the window's end.
    const total = await count(db, `SELECT COUNT(*) AS n FROM passports WHERE claim_state = 'verified' AND retracted = 0 AND ${grantLiveAtSql("passports.passport_id")}`, windowEndIso);
    const fresh = await count(
      db,
      "SELECT COUNT(DISTINCT subject) AS n FROM trust_events WHERE event_type = 'verification_granted' AND created_at >= ? AND created_at < ?",
      windowStartIso,
      windowEndIso,
    );
    content.global = { verified_creators: total?.n ?? 0, newly_verified_in_period: fresh?.n ?? 0 };

    // THE RE-PROOF NAG — see `DigestContent.reproof`. One statement: active
    // accounts of this passport that HAVE a proof filed by this passport and have
    // NO CURRENT one, i.e. exactly `accountProofStanding`'s `proven_stale`,
    // evaluated set-wise so the whole section costs one read however many
    // accounts a creator has.
    //
    // Both halves are bound to `a.passport_id`: a proof somebody ELSE filed for
    // an account this creator now holds is not a proof of theirs that lapsed —
    // it is a handle that changed hands, which is a different sentence and not
    // one to put in a nag. The freshness test is the one in `freshProofBounds`,
    // spelled out here (never assembled from a fragment) so
    // tests/invariants/ledger-contention.test.ts can read how this statement
    // talks about `oauth_challenges.status`: literal `= 'proven'`, never negated,
    // never a bound parameter.
    // The period boundary IS "now" for a digest: it is what every other count
    // in this section is measured against, and it keeps the section deterministic
    // under a fixed test clock instead of reaching for the wall clock mid-build.
    // An unparseable boundary degrades to real time rather than to 1970, which
    // would report every proof in the product as stale.
    const boundary = Date.parse(windowEndIso);
    const { nowIso, provedSince } = freshProofBounds(Number.isFinite(boundary) ? boundary : Date.now());
    const stale = await count(
      db,
      `SELECT COUNT(*) AS n FROM passport_accounts a
        WHERE a.passport_id = ? AND a.status = 'active'
          AND (EXISTS (SELECT 1 FROM claim_challenges c WHERE c.account_key = a.account_key AND c.status = 'used' AND c.passport_id = a.passport_id)
            OR EXISTS (SELECT 1 FROM oauth_challenges o WHERE o.account_key = a.account_key AND o.status = 'proven' AND o.passport_id = a.passport_id))
          AND NOT EXISTS (SELECT 1 FROM claim_challenges c2 WHERE c2.account_key = a.account_key AND c2.status = 'used' AND c2.passport_id = a.passport_id
                            AND (c2.valid_until > ? OR (c2.valid_until IS NULL AND COALESCE(c2.proved_at, c2.created_at) > ?)))
          AND NOT EXISTS (SELECT 1 FROM oauth_challenges o2 WHERE o2.account_key = a.account_key AND o2.status = 'proven' AND o2.passport_id = a.passport_id
                            AND (o2.valid_until > ? OR (o2.valid_until IS NULL AND COALESCE(o2.proved_at, o2.created_at) > ?)))`,
      passportId,
      nowIso,
      provedSince,
      nowIso,
      provedSince,
    );
    const needing = stale?.n ?? 0;
    if (needing > 0) content.reproof = { accounts_needing_reproof: needing };
  }
  return content;
}

/** DAILY digests are calm by decree: they exist only when the impersonation
 * section has something in it. Weekly always lands. */
export function isNotable(content: DigestContent): boolean {
  if (content.kind === "weekly") return true;
  // An alert exists BECAUSE something happened; there is nothing to be calm
  // about. (It never reaches `generateDigests` anyway — it is written by the
  // event that caused it — but the predicate must not answer "not notable" for
  // a kind it was not written for.)
  if (content.kind === "alert") return true;
  const i = content.impersonation;
  if (!i) return false;
  return i.new_reports_naming_you > 0 || i.face_flags > 0 || i.warnings_published_protecting_you > 0;
}

       WHERE p.claim_state IN ('claimed', 'verified') AND p.retracted = 0
         AND NOT EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = p.passport_id)
         AND ${creatorNotWipedSql("p.passport_id")}
         AND NOT EXISTS (SELECT 1 FROM creator_digests d WHERE d.passport_id = p.passport_id AND d.kind = ? AND d.period_key = ?)`;
/** I14-CRON: one statement — is any eligible creator still without this period's digest? The five-minute leg skips a finished weekly period with it instead of re-running the weekly call's proof scan and warning context. */
export async function periodHasPendingCreator(db: DbLike, kind: "daily" | "weekly", periodKey: string): Promise<boolean> {
  const row = await db.prepare(`SELECT 1 AS x ${DIGEST_OWED_CREATORS_FROM} LIMIT 1`).bind(kind, periodKey).first<{ x: number }>();
  return row !== null;
}
class DigestBudgetExhausted extends Error {}
/**
 * The meter: every statement and store read is charged BEFORE it is sent, and one that
 * would cross `budget` is refused (thrown), so a call never spends more than `budget`.
 * A creator stopped mid-way wrote nothing: its content build only reads, and its one
 * write (the INSERT below) is its last statement. The next run picks it up (NOT EXISTS).
 */
function digestMeter(db: DbLike, store: ArtifactReader | null, budget: number): { db: DbLike; store: ArtifactReader | null; spent: () => number } {
  let n = 0;
  const charge = (k: number): void => {
    if (n + k > budget) throw new DigestBudgetExhausted("digest generation budget spent");
    n += k;
  };
  const inner = new WeakMap<DbStatement, DbStatement>();
  const wrap = (st: DbStatement): DbStatement => {
    const w: DbStatement = {
      bind: (...v: unknown[]) => wrap(st.bind(...v)),
      run: async () => (charge(1), st.run()),
      all: async <T = Record<string, unknown>>() => (charge(1), st.all<T>()),
      first: async <T = Record<string, unknown>>() => (charge(1), st.first<T>()),
    };
    inner.set(w, st);
    return w;
  };
  return {
    db: { prepare: (sql: string) => wrap(db.prepare(sql)), batch: async (sts: DbStatement[]) => (charge(sts.length), db.batch(sts.map((x) => inner.get(x) ?? x))) },
    store: store === null ? null : { get: async (k: string) => (charge(1), store.get(k)), list: async (p: string) => (charge(1), store.list(p)) },
    spent: () => n,
  };
}

/** Generate the period's digests for VERIFIED creators that don't have one
 * yet. UNIQUE(passport_id, kind, period_key) makes this idempotent however
 * often cron fires; non-notable dailies are stored as {"empty":true} with both
 * channels pre-settled (the idempotency marker — never delivered, never shown). */
export async function generateDigests(
  db: DbLike,
  store: ArtifactReader | null,
  kind: "daily" | "weekly",
  periodKey: string,
  windowStartIso: string,
  windowEndIso: string,
  opts: { batch?: number; now?: () => string; nowMs?: number; maxPages?: number; budget?: number } = {},
): Promise<GenerateOutcome> {
  // I13-FX1: with a budget, every read and write below goes through the meter.
  const meter = opts.budget === undefined ? null : digestMeter(db, store, Math.max(0, opts.budget));
  // Filled as the call goes, so a budget stop still reports what was done.
  const out: GenerateOutcome = { generated: 0, notable: 0, proof_warnings: 0 };
  try {
    await generateDigestsUnmetered(meter?.db ?? db, meter !== null ? meter.store : store, kind, periodKey, windowStartIso, windowEndIso, opts, out);
    return meter !== null ? { ...out, spent: meter.spent() } : out;
  } catch (e) {
    if (!(e instanceof DigestBudgetExhausted) || meter === null) throw e;
    return { ...out, spent: meter.spent(), truncated: true };
  }
}
async function generateDigestsUnmetered(
  db: DbLike,
  store: ArtifactReader | null,
  kind: "daily" | "weekly",
  periodKey: string,
  windowStartIso: string,
  windowEndIso: string,
  opts: { batch?: number; now?: () => string; nowMs?: number; maxPages?: number },
  out: GenerateOutcome,
): Promise<void> {
  const now = opts.now ?? (() => new Date().toISOString());
  // RG2C-01 (D-1117(9)), JN-02: the instant this pass builds digests at — the caller's `nowMs`, else the clock that
  // stamps the rows. buildDigestContent reads the PSA link and the ratings block at it, not at the window's end.
  const atIso = opts.nowMs !== undefined && Number.isFinite(opts.nowMs) ? new Date(opts.nowMs).toISOString() : now();
  const batch = Math.max(1, Math.min(200, opts.batch ?? 50));
  // THE PRE-EXPIRY PROOF WARNING RIDES ON THE WEEKLY TICK.
  //
  // It is an `alert`, not a section of the weekly digest, for the same reason
  // the verification warning is one: it has a clock on it and must not wait for
  // a period boundary. But it needs a caller, and the weekly sweep is the one
  // that already owns the re-proof nag — putting it here keeps "everything about
  // control proofs expiring" in one module and costs one bounded scan per cron
  // tick rather than a second cron entry. Isolated: a failure here never stops
  // the digests being generated.
  let proofWarnings = 0;
  if (kind === "weekly") {
    try {
      const at = opts.nowMs ?? Date.parse(windowEndIso);
      proofWarnings = await noticeProofsExpiring(db, Number.isFinite(at) ? at : Date.now(), { now });
    } catch {
      proofWarnings = 0;
    }
  }
  out.proof_warnings = proofWarnings;
  const warnCtx = await buildWarningContext(db, windowStartIso, windowEndIso);
  const maxPages = Math.max(1, opts.maxPages ?? 40);
  for (let page = 0; page < maxPages; page++) {
  const { results: creators } = await db
    .prepare(
      // H17: CLAIMED creators receive protection digests too, not just verified
      // ones. The onboarding "turn on protection alerts" step is served to every
      // claimed creator and promises exactly this — impersonation warnings, report
      // outcomes, the weekly roundup. Digests are the ONLY channel that tells a
      // creator about impersonation activity against them, so excluding claimed
      // creators meant an actively-impersonated claimed creator heard nothing. The
      // content builder degrades cleanly for a claimed creator (no verification
      // deadline, no payout tags — the impersonation lines are what matter), and
      // isNotable still gates whether anything is actually delivered.
      // I14-Q3 (Jason (b), design pick: first claim event, no migration): the creator's claim instant, read in the
      // SAME statement as the page — the first `claim_proven` on its ledger (the event that moves a passport into
      // `claimed`; `verification_granted` requires `claimed` first, events.ts), stamped by `appendEvents`, never by a caller.
      `SELECT p.passport_id,
              (SELECT MIN(e.created_at) FROM trust_events e WHERE e.subject = p.passport_id AND e.event_type = 'claim_proven') AS claimed_at
       ${DIGEST_OWED_CREATORS_FROM}
       ORDER BY p.passport_id LIMIT ?`,
    )
    .bind(kind, periodKey, batch)
    .all<{ passport_id: string; claimed_at: string | null }>();

  if (creators.length === 0) break;
  for (const c of creators) {
    try {
      // I14-Q3: report only activity after the claim. A claim later than the window start moves the start to the
      // claim instant (and that creator's warning count is read over the clipped window, not the sweep-wide context);
      // a window ending at or before the claim is then empty (start >= end) and is still written below as
      // `{ empty: true }`, so the period is settled and the creator leaves the owed set — a skip with no row would be
      // re-paged every tick. No claim event on the ledger (state set without one) -> the window is not clipped.
      const claimMs = c.claimed_at ? Date.parse(c.claimed_at) : Number.NaN;
      const clipped = Number.isFinite(claimMs) && claimMs > Date.parse(windowStartIso);
      const startIso = clipped ? new Date(claimMs).toISOString() : windowStartIso;
      const content = await buildDigestContent(db, store, c.passport_id, kind, periodKey, startIso, windowEndIso, clipped ? undefined : warnCtx, atIso);
      const notable = isNotable(content);
      const stored = notable ? content : { empty: true };
      const ts = now();
      await db
        .prepare(`INSERT OR IGNORE INTO creator_digests (digest_id, passport_id, kind, period_key, content, created_at, emailed, pushed) SELECT ?, ?, ?, ?, ?, ?, ?, ? WHERE ${creatorNotWipedSql("?")}`)
        .bind(`dig_${crypto.randomUUID()}`, c.passport_id, kind, periodKey, JSON.stringify(stored), ts, notable ? 0 : 1, notable ? 0 : 1, c.passport_id)
        .run();
      out.generated++;
      if (notable) out.notable++;
    } catch (e) {
      // I13-FX1: a spent budget stops the CALL, not one creator.
      if (e instanceof DigestBudgetExhausted) throw e;
      // One creator's failure never stops the sweep. I14-FX2 (RG-CRON-S1): counted, so the caller keeps the
      // period owed (index.ts generateDigestPass: retried with backoff for DIGEST_FAILED_RETRY_WINDOW_MS, I14-FX6).
      out.failed = (out.failed ?? 0) + 1;
    }
  }
  if (creators.length < batch) break;
  }
}

  settled: number;
  /** I3-0014: rows this pass settled because the provider refused the ADDRESS

  emails_permanently_rejected: number;

   * `DELIVERY_RETRY_MAX_ATTEMPTS` (0145). A count, never a recipient. */
  emails_retry_exhausted: number;

  pushes_retry_exhausted: number;

/**
 * The one clause that says on WHAT the naming rests, appended to the count.
 *
 * The strongest basis reads as an attribution and the weakest reads as a
 * caveat, which is the whole point: "a person here matched it to you" and "a
 * machine saw a resemblance it cannot interpret" were the same sentence until
 * this shipped.
 *
 * An absent or unrecognised basis renders NOTHING — the sentence that was mailed
 * before this field existed. A digest row written by an older build, or a basis
 * the classifier could not establish, must not acquire a confidence it never
 * had.
 */
function basisClause(basis: NamingBasis | null | undefined): string {
  switch (basis) {
    case "operator":
      return " — someone here matched it to you by hand";
    case "identity":
      return " — matched by machine on a close copy of a handle or domain of yours: the same once padding and punctuation come off, a letter or two away, your name on another ending, or your name in another alphabet";
    case "likeness":
      return " — matched by our image check: the picture on it is one of yours, the same file or a near-copy, which is not proof of who is using it";
    case "resemblance":
      return " — matched on our weakest signal: something about it resembles you and the machine cannot tell what the resemblance means, which is why a person is looking";
    default:
      return "";
  }
}

import { VERIFICATION_REJECTED_NOTICE_TEXT } from "./review.js";
function digestSummary(content: DigestContent): string {
  const i = content.impersonation;
  const bits: string[] = [];
  // MONEY LEADS. A payout-destination change outranks every other line here: it
  // is the only one where the wrong answer is a theft, and the creator is racing
  // a 24-hour clock to stop it. Never the addresses themselves — a count and a
  // door, so the mail cannot become a second copy of the thing being protected.
  // THE LAPSE, second only to money. It shipped with no branch here at all, so
  // the one notice the whole grace-window decision rests on was delivered to
  // email and push as "all quiet this period" under the subject "your
  // verification is about to expire" — telling the creator to start a renewal
  // for a verification that is not expiring, while the actual fix (get the
  // operator live, or ask an operator to re-bind) was named nowhere. Measured in
  // adversarial review. It is exactly the failure the payout note below forbids:
  // a mail whose words point at the wrong lever teaches a creator to ignore the
  // alerts that matter.
  const ol = content.operator_lapse;
  if (ol) {
    bits.push(
      `Nobody is currently accountable for this character. The agency or person bound to it is no longer live, and your Verified badge comes off on ${ol.badge_moves_at.slice(0, 10)} unless that changes. Nothing else about your passport changes, and the badge comes back the moment it is re-verified.`,
    );
  }
  const pc = content.payout_change;
  if (pc) {
    const n = pc.destinations;
    const what = n === null ? "your Dhwaja payout destinations were changed" : n === 0 ? "all of your Dhwaja payout destinations were removed" : `your Dhwaja payout destinations were changed — ${n} destination${n === 1 ? "" : "s"} ${n === 1 ? "is" : "are"} now staged`;
    const when = pc.effective_at ? ` The new set starts being shown to fans on ${pc.effective_at.slice(0, 10)}; until then they keep seeing the previous one.` : " This took effect immediately.";
    bits.push(
      `${what}.${when} IF THIS WAS NOT YOU, sign in and put them back now — a staged change can still be replaced before it goes live, removing a destination takes effect at once, and the change is signed onto your ledger either way so there is a record of when it happened`,
    );
  }
  const pa = content.payout_account_change;
  if (pa) {
    if (pa.action === "bound") {
      const when = pa.live_at ? ` It can receive transfers from ${pa.live_at.slice(0, 10)} at the earliest — until then nothing has been sent to it, and you can still disconnect it.` : "";
      // Said on both sides of zero, so the sentence's PRESENCE discloses nothing
      // the count does not already say.
      const moved =
        typeof pa.pending_retargeted === "number"
          ? pa.pending_retargeted > 0
            ? ` ${pa.pending_retargeted} payout${pa.pending_retargeted === 1 ? "" : "s"} already reserved for you will be paid to this account instead of the previous one.`
            : " No payouts were already reserved, so nothing you have earned so far changes destination."
          : "";
      bits.push(
        `a Stripe account ending ${pa.account_hint} was connected as the destination Dhwaja pays your earnings to.${when}${moved} IF THIS WAS NOT YOU, open your console and disconnect it now — disconnecting takes effect at once, and the change is on your ledger either way so there is a record of when it happened`,
      );
    } else {
      bits.push(
        // #145 (FX-F sibling): rendered at DELIVERY, which can follow a re-bind — so past
        // tense and bounded, like the console's DISCONNECTED arm; never "is not sending".
        `the Stripe account ending ${pa.account_hint} was disconnected — from then, Dhwaja sent your earnings nowhere until an account was connected again. Nothing was lost: your cleared and pending balance was held, and it pays out to an account once one is connected and confirmed. IF THIS WAS NOT YOU, sign in and check which payout account is connected now: someone with access to your session can stop your payouts even though they cannot redirect them without your passkey`,
      );
    }
  }
  // I3-0997: the payout outcome, beside the other money lines.
  const po = content.payout_outcome;
  if (po) {
    const amt = `$${(po.amount_cents / 100).toFixed(2)}`;
    bits.push(
      po.outcome === "paid"
        ? `a payout of ${amt} (${po.period_key}) was sent to the Stripe account connected to receive your earnings`
        : `a payout of ${amt} (${po.period_key}) could not be sent to your connected Stripe account after repeated attempts. The money is still yours and still reserved for you; check the payout account in your console, and contact Dhwaja if it stays stuck`,
    );
  }
  // The deadline leads what is left, whatever else is in the digest: it is the
  // only other line here with a clock on it.
  const v = content.verification_expiry;
  if (v) {
    bits.push(
      `your Dhwaja verification expires on ${v.expires_at.slice(0, 10)} (${v.days_remaining <= 0 ? "today" : `in ${v.days_remaining} day${v.days_remaining === 1 ? "" : "s"}`}) — when it lapses you keep your passport and everything on it, but the Verified badge, minting, registering releases and the public display of your payout destinations stop until it is renewed. Start the renewal from the Verification page in your console; nothing about your badge changes while an operator reviews it`,
    );
  }
  // I3-1200: the decision on the creator's own verification request. No reason
  // text (internal), no refile instant (served live on the Verification page). One source with the alert row's body:
  // review.ts VERIFICATION_REJECTED_NOTICE_TEXT less its closing full stop (the summary joins its lines with " · ").
  if (content.verification_rejected) {
    bits.push(VERIFICATION_REJECTED_NOTICE_TEXT.slice(0, -1));
  }
  // The proof deadline sits beside the verification one, and says what LAPSING
  // costs rather than only that it is coming — a creator who does not know that
  // an expired proof stops shielding the handle from impersonation reports has
  // no way to price the two minutes it takes to renew.
  const pe = content.proof_expiry;
  if (pe) {
    bits.push(
      `the control proof on ${pe.accounts === 1 ? "one of your accounts" : `${pe.accounts} of your accounts`} expires on ${pe.expires_at.slice(0, 10)} (${pe.days_remaining <= 0 ? "today" : `in ${pe.days_remaining} day${pe.days_remaining === 1 ? "" : "s"}`}) — when it lapses that handle stops being shown to fans as confirmed and stops being shielded from impersonation reports. Renewing takes one short code in the profile's bio: open Accounts in your console`,
    );
  }
  // S-070 RJ-01 — THE CONTEST NOTICE SAYS WHAT HAPPENED.
  //
  // `noticeContestedAccount` stores a `contested_account` alert and the console
  // promises the incumbent has been told. This summary handled every other
  // cause and not this one, so a contest-only digest produced NO bits at all and
  // fell through to "all quiet this period" — then the dispatcher below, seeing
  // only `kind === "alert"`, mailed it under "your verification is about to
  // expire" and pushed the same. A creator whose account somebody has just
  // claimed was told to renew a verification that is not expiring.
  //
  // The wording distinguishes a RECEIVED PROOF under review from an
  // adjudicated result, names only the affected account, and offers the one
  // continuation that exists (Accounts) — it must never imply an outcome, and
  // it names no other account and no other passport (INV-025), which is the
  // same discipline the stored notice and the console row already keep.
  // I3-0034 — a passkey is signing authority: it leads.
  const pk = content.passkey_added;
  if (pk) {
    bits.push(
      `a passkey (${pk.credential_hint}…) was added to your passport${pk.via === "existing_passkey" ? ", approved with a passkey you already had" : ""}. If that wasn't you, revoke it on Passkeys in your console now and contact Dhwaja`,
    );
  }
  const pr = content.passkey_removed;
  if (pr) {
    bits.push(`a passkey (${pr.credential_hint}…) was removed from your passport and can no longer sign in or sign for you. If that wasn't you, contact Dhwaja now`);
  }
  const ca = content.contested_account;
  if (ca) {
    bits.push(
      `somebody proved control of your account ${ca.account_key} — an operator is reviewing a competing claim on it. Nothing on your passport has changed yet, and nothing about your verification is affected; open Accounts in your console to see the account this concerns`,
    );
  }
  // I6-DIGEST2: the two outcome notices S9-2 stores, each in its own words. No
  // request id, case id or account handle (INV-025) — the console lists them.
  const cr = content.correction_resolved;
  if (cr) {
    bits.push(cr.outcome === "resolved" ? "A correction you asked for has been resolved." : "A correction you asked for was declined.");
  }
  if (content.contest_decided) {
    bits.push("A decision was made on your contest.");
  }
  const ro = content.report_outcome;
  if (ro) {
    bits.push(ro.outcome === "published" ? "A warning was published about an account you reported." : "A report you filed has been closed.");
  }
  // I21-Q128: the approved sentence as stored, so the mail and the console say the same words.
  const wpn = content.warning_published;
  if (wpn) bits.push(wpn.text);
  if (i) {
    if (i.warnings_published_protecting_you > 0) bits.push(`an impersonation warning protecting you was published — share your PSA link`);
    // "naming you", not "impersonating you". The escalation a reviewer performed
    // adjudicated the REPORT; nobody adjudicated the substring-matched pointer
    // that says the report is about this creator. `protection.ts` reasons this
    // out for the boolean on the same screen and the email is catching up.
    if (i.new_reports_naming_you > 0)
      bits.push(`${i.new_reports_naming_you} report${i.new_reports_naming_you === 1 ? "" : "s"} naming you, now with a reviewer${basisClause(i.naming_basis)}`);
    // "image check", never "lookalike": see `face_flags`. And "a possible photo
    // match", never "using your likeness" — `match_medium` is defined by
    // `face-triage.ts` as "same face, different photo — LOOKALIKE/TWIN RISK",
    // the hedge that stops the machine drafting an accusation. Mailing it as a
    // flat assertion converts a deliberately non-actionable signal into an
    // actionable one. The console already says the neutral thing; this is the
    // email catching up to it.
    if (i.face_flags > 0) bits.push(`${i.face_flags} account${i.face_flags === 1 ? "" : "s"} our image check flagged as a possible photo match to you`);
  }
  // THE RE-PROOF NAG, WHICH WAS COMPUTED, STORED, AND SHOWN TO NOBODY. It has
  // been folded into `content.reproof` since proofs started expiring, and
  // neither this summary nor the console's digest list ever read it — so the
  // one warning a creator got before an account silently stopped being served
  // reached no channel at all. Count only, never the handles (that list is
  // exactly the set of a creator's weakest links, and an email is the wrong
  // place for it); the console's Accounts screen names them.
  if (content.reproof) {
    const n = content.reproof.accounts_needing_reproof;
    bits.push(`${n} of your accounts need their control proof renewed — they stop being shown to fans as confirmed until you do (Accounts, in your console)`);
  }
  if (content.activity) {
    // A withheld cell leaves the sentence entirely — it is not rendered as 0,
    // which would make "withheld" and "nothing happened" different strings and
    // hand the reader the 1..K-1 bound the floor exists to deny.
    const a = content.activity;
    const parts: string[] = [];
    if (a.new_saves !== null) parts.push(`${a.new_saves} new saves`);
    if (a.new_card_collectors !== null) parts.push(`${a.new_card_collectors} new card collectors`);
    if (a.new_acquisitions !== null) parts.push(`${a.new_acquisitions} new collectible acquisitions`);
    if (parts.length > 0) bits.push(parts.join(", "));
  }
  if (bits.length === 0) bits.push("all quiet this period");
  return bits.join(" · ");
}

/**
 * THE 14-DAY VERIFICATION WARNING, ADDRESSED TO THE CREATOR.
 *
 * Called by `verification-sweep.ts` while the grant is still good and inside
 * `EXPIRY_WARNING_MS`. Writes one `alert` digest, which the ordinary delivery
 * pass then sends over every channel the creator has enabled — verified email,
 * push, and the in-app list — and which `GET /account/digests` serves whether
 * or not any of those exist.
 *
 * DEDUP IS ON THE DEADLINE, exactly as the fan-out's synthetic event id was:
 * `period_key` is `verification-expiry:<expires_at>` under the UNIQUE
 * (passport_id, kind, period_key) that already makes the periodic sweep
 * idempotent. Fourteen ticks produce one notice; a renewal moves the deadline
 * and is the only thing that re-arms it.
 *
 * Returns 1 when a notice was created, 0 when one already existed — so the
 * sweep's `warned` counter stays a count of PEOPLE TOLD rather than of ticks.
 */
/**
 * X1 — tell the incumbent that somebody proved control of one of their accounts.
 *
 * Idempotent per (passport, account): the `period_key` names the CAUSE, so a
 * challenger who runs the ceremony repeatedly cannot use this as a notification
 * cannon against the person they are contesting. A second, genuinely later
 * contest of the same account therefore does not re-alert — which is the right
 * trade: the case is already open and an operator is already looking at it.
 */
export async function noticeContestedAccount(
  db: DbLike,
  passportId: string,
  accountKey: string,
  now: () => string = () => new Date().toISOString(),
): Promise<number> {
  // QUEUE #124-adjacent / SQ-01 (S-033 CCVIEWS, never fixed there — outside
  // that room's manifest). This keyed solely on `accountKey`, under the same
  // UNIQUE (passport_id, kind, period_key) that makes the periodic sweep
  // idempotent — so the FIRST contest of an account to reach this function
  // claimed `contested-account:<accountKey>` PERMANENTLY. `verifyContest`
  // (contested-accounts.ts) folds a repeat proof against the SAME open case
  // (`review_queue`: item_type='dispute', state='open') into it, which is the
  // intended "not a notification cannon" behaviour — but once an operator
  // resolves that case, a genuinely NEW later contest of the same account
  // opens a NEW case, and this INSERT OR IGNORE silently ate its notice: the
  // incumbent heard about the first contest and never about any later one.
  //
  // The fix scopes the key by the CASE, not just the account — the same
  // discipline `payoutChangeNoticeStatement` already uses (`payout-change:
  // <changeId>`, "a genuinely new change mints a new id and is a new
  // notice"). `verifyContest` does not thread its `submitRecoveryRequest`
  // `item_id` through to here, and widening that call chain reaches into
  // `contested-accounts.ts`/`recovery.ts`, which this room does not own — so
  // the case is looked up here instead, by the same one-line evidence
  // `verifyContest` already writes into the case's payload ("Proved control of
  // <accountKey> (listed here)"): the newest dispute against this passport
  // naming this account. A resolved case and its successor are two different
  // `item_id`s and therefore two different notices; repeat folds into the SAME
  // open case keep resolving to the same `item_id` and stay one notice, same
  // as before.
  //
  // I3-0031 (I11-DIGEST): `instr`, not LIKE. The old `payload LIKE '%Proved
  // control of <key> (listed here)%'` needle was 34 fixed bytes plus the key,
  // over D1's ~50-byte LIKE pattern ceiling for any real key, so on D1 this
  // threw after the case was filed. `instr` has no pattern limit and no
  // wildcards. The needle is the line as `JSON.stringify(payload)` stores it
  // (recovery.ts), so a key with a JSON-escaped character still matches.
  const needle = JSON.stringify(`Proved control of ${accountKey} (listed here)`).slice(1, -1);
  const kase = await db
    .prepare(
      `SELECT item_id FROM review_queue WHERE item_type = 'dispute' AND subject = ?
        AND json_extract(payload, '$.kind') = 'competing_claim' AND instr(payload, ?) > 0
        ORDER BY created_at DESC, item_id DESC LIMIT 1`,
    )
    .bind(passportId, needle)
    .first<{ item_id: string }>();
  // No case found (should not happen — the caller just created or folded into
  // one) falls back to the OLD account-only key rather than throwing: a notice
  // late is still better than a notice lost, and the idempotency this branch
  // gives up is the same one that was already there.
  const caseKey = kase?.item_id ?? "unknown";
  const content: DigestContent = {
    kind: "alert",
    period_key: `contested-account:${accountKey}:${caseKey}`,
    contested_account: { account_key: accountKey },
  };
  const r = await db
    .prepare(
      ALERT_DIGEST_INSERT_SQL,
    )
    .bind(`dig_${crypto.randomUUID()}`, passportId, content.period_key, JSON.stringify(content), now())
    .run();
  return r.changes > 0 ? 1 : 0;
}

/**
 * THE SWEEP THAT FINDS THEM. Bounded by `limit` NOTICES (not by rows read): the
 * scan is `proofsExpiringSoon`'s one read of the proofs current at now (#733), and capping
 * the writes is what keeps one bad day from turning into an unbounded batch of
 * inserts. Anything left over is picked up on the next tick — the window is
 * `PROOF_EXPIRY_WARNING_MS` wide and the cron runs four times a day, so there is
 * an enormous amount of room before a creator could be missed.
 *
 * One failure never stops the sweep: a creator whose insert throws is retried on
 * the next tick, exactly like `generateDigests`.
 */
export async function noticeProofsExpiring(
  db: DbLike,
  nowMs: number,
  opts: { limit?: number; now?: () => string } = {},
): Promise<number> {
  const now = opts.now ?? (() => new Date(nowMs).toISOString());
  const limit = Math.max(1, Math.min(500, opts.limit ?? 200));
  const rows = await proofsExpiringSoon(db, nowMs);
  // Group by (passport, deadline) — the dedup key — so the count in the notice
  // is "how many of your accounts lapse on that day", not one mail per handle.
  const byDeadline = new Map<string, { passport_id: string; expires_at: string; days_remaining: number; accounts: number }>();
  for (const r of rows) {
    const k = `${r.passport_id}|${r.expires_at}`;
    const cur = byDeadline.get(k);
    if (cur) cur.accounts++;
    else byDeadline.set(k, { passport_id: r.passport_id, expires_at: r.expires_at, days_remaining: r.days_remaining, accounts: 1 });
  }
  // ALREADY-WARNED GROUPS ARE EXCLUDED BEFORE THE BUDGET IS APPLIED (S-070
  // SJ-05). `written` used to be charged BEFORE an `INSERT OR IGNORE` that
  // changed nothing, so a group whose deadline notice already existed spent a
  // slot out of the same 200-group budget on every tick. With more eligible
  // groups than budget — and no continuation cursor, and no ORDER BY on the
  // scan to rotate which groups are seen first — a stable already-warned cohort
  // at the front of the result held the whole budget indefinitely and the
  // groups behind it reached their deadline with no warning, which is the one
  // outcome this sweep exists to prevent.
  //
  // ONE STATEMENT, RANGE-SCOPED, NOT CHUNKED PER PASSPORT. `period_key` is
  // `proof-expiry:<expires_at>` and ISO instants sort lexically, so the notices
  // that could possibly match a group in this window are exactly one contiguous
  // key range. A chunked `IN (…)` over the grouped passports would cost a
  // statement per 90 creators; this costs one whatever the population is, and it
  // reads the same rows the groups were derived from. The low bound is backed
  // off one day so a deadline that has just passed (`days_remaining` 0, the
  // grace the standing derivation still calls `expiring_soon`) is still matched
  // against its existing notice rather than looking un-warned forever.
  const keyLo = `proof-expiry:${new Date(nowMs - 86_400_000).toISOString()}`;
  const keyHi = `proof-expiry:${new Date(nowMs + PROOF_EXPIRY_WARNING_MS).toISOString()}`;
  const { results: existing } = await db
    .prepare("SELECT passport_id, period_key FROM creator_digests WHERE kind = 'alert' AND period_key >= ? AND period_key <= ?")
    .bind(keyLo, keyHi)
    .all<{ passport_id: string; period_key: string }>();
  const alreadyWarned = new Set(existing.map((r) => `${r.passport_id}|${r.period_key}`));
  let warned = 0;
  let written = 0;
  for (const g of byDeadline.values()) {
    if (written >= limit) break;
    // NOT a substitute for the `INSERT OR IGNORE`: that is still what makes a
    // concurrent tick idempotent. This only stops a settled group from being
    // CHARGED. A group whose write failed is absent from `alreadyWarned` on the
    // next tick and is retried, which is the fairness half of the repair.
    if (alreadyWarned.has(`${g.passport_id}|proof-expiry:${g.expires_at}`)) continue;
    written++;
    try {
      warned += await noticeProofExpiring(db, g.passport_id, g.expires_at, g.accounts, g.days_remaining, now);
    } catch {
      /* one creator's failure never stops the sweep; retried next run */
    }
  }
  return warned;
}

/**
 * The payout-ACCOUNT notice, as a statement so it can ride the SAME atomic batch
 * as the binding it is about — the shape `payoutChangeNoticeStatement` uses, and
 * for the same reason: a destination can never change with its alert dropped.
 *
 * Keyed on the ledger event id, so the `INSERT OR IGNORE` under
 * UNIQUE(passport_id, kind, period_key) makes a retried append idempotent rather
 * than sending a creator the same alarm twice.
 *
 * NOT SUPPRESSION-GATED, for the reason its sibling states: this goes to the
 * creator's own console and their own verified address, names nobody, and is the
 * one message a delisted creator most needs — "someone moved where your money
 * goes". Delisting is a fan-surface property, not a gag order on the subject.
 */
export function payoutAccountNoticeStatement(
  db: DbLike,
  passportId: string,
  args: { eventKey: string; action: "bound" | "unbound"; stripeAccountId: string; liveAt: string | null },
  now: () => string = () => new Date().toISOString(),
): DbStatement {
  const content: DigestContent = {
    kind: "alert",
    period_key: `payout-account:${args.eventKey}`,
    payout_account_change: {
      action: args.action,
      // Last four only — see the note on the content arm. Padded rather than
      // sliced blind so a short or malformed id cannot produce an empty hint
      // that reads as "an account ending ".
      account_hint: args.stripeAccountId.slice(-4).padStart(4, "•"),
      live_at: args.liveAt,
      // Placeholder — overwritten below, on the bind arm, by the correlated
      // sub-select json_set() splices into THIS row's own INSERT. Unbind pays
      // nobody and keeps the explicit null.
      pending_retargeted: null,
    },
  };
  const json = JSON.stringify(content);
  if (args.action === "bound") {
    // INV-058 CLAIM-THEN-ACT BINDING: `pending_retargeted` used to arrive as a
    // JS number computed by a SELECT the caller ran BEFORE this statement's
    // batch (creator-payout.ts activateConnect, pre-fix). Anything that moved
    // a `creator_payouts` row between that read and this INSERT's commit — a
    // weekly tick paying a row to the OLD account — left this notice (and the
    // audit row `appendPayoutAccountEvent` inserts beside it) stating a count
    // that was never true at commit (INV-058, RUN-0015 UNSTATED-0018).
    //
    // Fixed by moving the read INSIDE this statement: the correlated
    // `SELECT COUNT(*)` below executes as part of committing THIS row, in the
    // SAME `db.batch()` transaction as the audit row's own correlated COUNT
    // and the activation UPDATE. One batch is one SQLite transaction (see
    // `runBatch` in test/sqlite-double.ts and `d1().batch` in db.ts): every
    // statement in it reads one consistent snapshot, so this json_set() and
    // the audit row's `... || (SELECT COUNT...)` (creator-payout.ts) can never
    // disagree, and neither can be stale against the activate UPDATE landing
    // in the same commit.
    return db
      .prepare(
        "INSERT OR IGNORE INTO creator_digests (digest_id, passport_id, kind, period_key, content, created_at, emailed, pushed) " +
          "SELECT ?, ?, 'alert', ?, json_set(?, '$.payout_account_change.pending_retargeted', " +
          // D-0800/H2: a `failed` row (I3-2131 -- still owed, still retrying)
          // must count here the same as `pending`, or this notice disagrees
          // with the audit row (creator-payout.ts:675, which already counts
          // both) whenever a failed row exists at rebind time.
          `(SELECT COUNT(*) FROM creator_payouts WHERE passport_id = ? AND (status = 'pending' OR status = 'failed'))), ?, 0, 0 WHERE ${creatorNotWipedSql("?")}`,
      )
      .bind(`dig_${crypto.randomUUID()}`, passportId, content.period_key, json, passportId, now(), passportId);
  }
  return db
    .prepare(
      ALERT_DIGEST_INSERT_SQL,
    )
    .bind(`dig_${crypto.randomUUID()}`, passportId, content.period_key, json, now());
}

/**
 * Deliver unsettled digests. Email goes only to a VERIFIED, digest-enabled
 * contact; push to the passport's registered tokens; a missing contact or
 * disabled channel settles silently (in-app remains — the digest row itself
 * IS the in-app channel, served by GET /account/digests). Delivery failures
 * leave the flag unset for the next pass (retry, never duplicate: flags flip
 * only on success or on nothing-to-do).
 */

/**
 * PC-05 (org/07-findings/S-034-batch/OBS/PLAN.md, "PC-05, sized as the tuning
 * change it is"). Cheap backlog gauge over the same `(emailed = 0 OR pushed =
 * 0)` predicate `deliverDigests` drains — logged every tick so the actual
 * growth rate is visible in Workers Logs without guessing at a threshold.
 */
/**
 * THE UNSETTLED PREDICATE, once. Every door that asks "has this digest still got
 * work outstanding" interpolates THIS const rather than re-typing it — the
 * backlog gauge, the drain, and `digestNoticeState` below. The parentheses are
 * load-bearing (see `deliverDigests`: a bare `emailed = 0 OR pushed = 0` with a
 * clause appended re-associates and matches every unemailed row).
 */
const UNSETTLED_SQL = "(emailed = 0 OR pushed = 0)";

export async function pendingDigestCount(db: DbLike): Promise<number> {
  const row = await db.prepare(`SELECT COUNT(*) AS n FROM creator_digests WHERE ${UNSETTLED_SQL}`).first<{ n: number }>();
  return row?.n ?? 0;
}

/**
 * QUEUED IS NOT DELIVERED, and until this read existed nothing could tell them
 * apart (queue #194; I-20 deferred it as "a bigger, separate change").
 *
 * A notice is born `emailed = 0, pushed = 0` by every `notice*` writer above, and
 * the only signal any caller had was a stamp written on ANOTHER table beside the
 * `INSERT OR IGNORE` — unconditionally, so it said "we tried to queue", never
 * "it is out of the queue". This answers the queue question off the row itself.
 *
 * `settled` MEANS NO FURTHER WORK, NOT DELIVERED. `deliverDigests` writes the
 * same 1 for a mail that sent, for a creator with no verified address, for a
 * disabled channel and for a push with no registered token. Distinguishing those
 * needs a column the schema does not have (`migrations/0006_creator_channel.sql`
 * pins both flags to `CHECK (… IN (0,1))`), so this function does not claim it
 * and no caller may render this word as "told".
 */
export type DigestNoticeState = "none" | "queued" | "settled";

export async function digestNoticeState(db: DbLike, passportId: string, periodKey: string): Promise<DigestNoticeState> {
  const row = await db
    .prepare(`SELECT CASE WHEN ${UNSETTLED_SQL} THEN 1 ELSE 0 END AS unsettled FROM creator_digests WHERE passport_id = ? AND period_key = ?`)
    .bind(passportId, periodKey)
    .first<{ unsettled: number }>();
  if (!row) return "none";
  return row.unsettled === 1 ? "queued" : "settled";
}

export async function deliverDigests(
  db: DbLike,
  email: EmailSender,
  push: PushSender,
  publicBase: string | null,
  opts: { batch?: number; safetyArm?: number; fetchBudget?: number; statementBudget?: number; now?: () => string; consoleOrigin?: string | null; cid?: string; emailTimeoutMs?: number; budgetMs?: number; nowMs?: () => number } = {},
): Promise<DeliverOutcome> {
  // M86: the creator manages (or turns off) digests at the console's #/digests
  // screen, so that is the List-Unsubscribe target. Absent console origin →
  // no header rather than a broken one (fail closed; the alert stream in
  // email.ts carries its own to the fan app).
  const manageDigests = opts.consoleOrigin ? `${opts.consoleOrigin}/#/digests` : null;
  const now = opts.now ?? (() => new Date().toISOString());
  const batch = Math.max(1, Math.min(200, opts.batch ?? 50));
  const safetyArm = Math.min(Math.max(0, opts.safetyArm ?? DIGEST_SAFETY_ARM), batch);
  // I3-1032 a (I22): `suppressed` also reads a RETRACTED passport (the other way a creator leaves public view,
  // and the one generation already filters at its own read; a LEFT JOIN on the PK, no extra statement), so a roundup queued before the retraction settles
  // unsent like one queued before a suppression; `alert` rows still go.
  // I3-0939 (I15-W2J): `suppressed` rides the row read (a LEFT JOIN in `unsettled` below, no extra statement):
  // a roundup generated before the creator's removal ceremony settles unsent in the loop. `alert` rows are
  // safety mail and are not gated here.
  const cols =
    "digest_id, creator_digests.passport_id, kind, period_key, content, emailed, pushed, email_attempts, email_next_at, push_attempts, push_next_at, (rs.passport_id IS NOT NULL OR COALESCE(rp.retracted, 0) = 1) AS suppressed";
  // PARENTHESISED, AND THAT IS LOAD-BEARING. `AND` binds tighter than `OR`, so
  // appending `AND kind = 'alert'` to a bare `emailed = 0 OR pushed = 0` parses
  // as `emailed = 0 OR (pushed = 0 AND kind = 'alert')` — which matches every
  // unemailed row, orders the herd by `created_at` and returns exactly the FIFO
  // page the arm exists to bypass. It failed silently: the arm ran, cost its
  // statement, and promoted nothing. Caught by this slice's own test.
  //
  // I3-1288 (0145): a row whose email leg is backing off and whose push leg is
  // settled is NOT due, so it is not read at all — it must not head the FIFO
  // page until `email_next_at`. Its own AND term, so `idx_creator_digests_alert`'s
  // partial WHERE is still implied by the safety arm's query.
  const nowIso = now();
  // #735 (0151): the same for the push leg — a row is read only when one of its
  // UNSETTLED legs is DUE. `UNSETTLED_SQL` stays its own AND term (the partial
  // index's WHERE must still be implied).
  const unsettled = `FROM creator_digests LEFT JOIN request_suppressions rs ON rs.passport_id = creator_digests.passport_id LEFT JOIN passports rp ON rp.passport_id = creator_digests.passport_id WHERE ${UNSETTLED_SQL} AND ((pushed = 0 AND (push_next_at IS NULL OR push_next_at <= ?)) OR (emailed = 0 AND (email_next_at IS NULL OR email_next_at <= ?)))`;
  type DigestRow = { digest_id: string; passport_id: string; kind: string; period_key: string; content: string; emailed: number; pushed: number; email_attempts: number; email_next_at: string | null; push_attempts: number; push_next_at: string | null; suppressed: number };

  // THE SAFETY ARM — the fix this whole slice was measured into existence for.
  //
  // This read was `ORDER BY created_at ASC LIMIT 50` and nothing else: pure
  // FIFO, no priority ordering of any kind, called ONLY from the 6-hourly sweep
  // and last in it. The fan legs got `PRIORITY_SCAN`/`SAFETY_KINDS` in their own
  // slice; this rail — the one a payout-destination change travels on to reach
  // the creator whose money it is — was never given the equivalent, and
  // `DEPLOY.md`'s "bounded pull" gap was written about the other one.
  //
  // MEASURED, one payout-change alert behind a weekly herd:
  //     100 creators → sweep  3 =  18 hours   (inside the 24-hour window)
  //     500 creators → sweep 11 =  66 hours   (MISSED)
  //   1,000 creators → sweep 21 = 126 hours   (MISSED — five days)
  //
  // With this arm it is sweep 1 at every population — provided the lane is also
  // MONEY-FIRST inside itself, which the first draft was not: see
  // `MONEY_FIRST_DIGEST`, where a cohort of proof-expiry notices reburied the
  // payout alert at sweep 9. The claim holds for a payout notice unless more
  // than `DIGEST_SAFETY_ARM` OTHER payout notices are already waiting, which is
  // a deployment-wide payout event rather than a queue. `generateDigests` fans out per creator at a
  // period boundary, so the herd is sized by the creator population and the FIFO
  // head is exactly where a fresh alert lands behind it.
  //
  // `kind` ALREADY CARRIES THE SPLIT and no new column is needed: migration
  // 0023's own comment says "'daily'/'weekly' are the periodic roundups.
  // 'alert' is an event-driven creator notice", which is precisely "has a clock
  // on it" versus "is news". `payout_account_change`, `payout_change`,
  // `proof_expiry` and `operator_lapse` all ride `alert`.
  //
  // TWO STATEMENTS, NOT A SORT. `ORDER BY (kind = 'alert') DESC, created_at ASC`
  // over the unsettled set was measured at 17-19 ms per read at 200,000 rows —
  // it sorts everything to find the few. The arm plus migration 0052's partial
  // index is 0.012 ms and flat. (The FIFO head read below is still a full scan
  // at that depth, ~17 ms: the `emailed = 0 OR pushed = 0` predicate defeats
  // `idx_creator_digests_undelivered` entirely. That is a pre-existing cost this
  // slice measures and names rather than changes.)
  // MONEY FIRST, INSIDE THE FAST LANE — and this was a real defect in the first
  // draft of this arm, not a refinement.
  //
  // `kind = 'alert'` is not one deadline. `payout-change:` and `payout-account:`
  // carry the 24-HOUR cooling-off window; `verification-expiry:`,
  // `proof-expiry:` and `operator-lapse:` are 14-day and 7-day horizons. And the
  // long-horizon ones arrive in COHORTS: `verification-sweep.ts`
  // (SWEEP_APPEND_LIMIT), `character-lapse-sweep.ts` (LAPSE_NOTICE_LIMIT) and
  // `noticeProofsExpiring` each emit up to 200 in ONE tick, against a fast lane
  // of `DIGEST_SAFETY_ARM` rows.
  //
  // So ordering the arm by `created_at` alone reburied the payout notice inside
  // the lane. MEASURED: 300 roundups + one tick of proof-expiry notices (200) +
  // one payout alert behind them → sweep 9 = 54 hours, against a 24-hour
  // window. The arm was buying two sweeps out of the twenty it needed.
  //
  // `period_key` carries the cause with a stable prefix (0023's own design: "its
  // period_key names the CAUSE"), so the tier is expressible in SQL without a
  // new column and without parsing `content`. The sort is over the ALERT subset
  // the partial index already restricts to, not over the backlog.
  const alerts = safetyArm > 0
    ? (await db.prepare(`SELECT ${cols} ${unsettled} AND kind = 'alert' ORDER BY ${MONEY_FIRST_DIGEST}, creator_digests.created_at ASC LIMIT ?`).bind(nowIso, nowIso, safetyArm).all<DigestRow>()).results
    : [];
  const { results: head } = await db
    .prepare(`SELECT ${cols} ${unsettled} ORDER BY creator_digests.created_at ASC LIMIT ?`)
    .bind(nowIso, nowIso, batch)
    .all<DigestRow>();
  // Alerts first, then the FIFO head, deduped by id — the two arms overlap
  // whenever the backlog is shallower than both, and a row carried twice would
  // be mailed twice.
  const seen = new Set<string>();
  const rows: DigestRow[] = [];
  for (const r of [...alerts, ...head]) {
    if (seen.has(r.digest_id)) continue;
    seen.add(r.digest_id);
    rows.push(r);
    if (rows.length >= batch) break;
  }

  const out: DeliverOutcome = { emails_sent: 0, pushes_sent: 0, settled: 0, emails_permanently_rejected: 0, emails_retry_exhausted: 0, pushes_retry_exhausted: 0, emails_ambiguous: 0, emails_suppressed: 0 };
  // THE BOUNDS THIS LEG DID NOT HAVE. Reserved BEFORE each row rather than
  // discovered after it, the same stance both fan legs take: a row not reached
  // is untouched and still unsettled, so the next sweep resumes exactly here.
  // Stopping is a decision; being killed by the platform mid-row is not.
  const fetchBudget = Math.max(1, opts.fetchBudget ?? DIGEST_FETCH_BUDGET);
  const statementBudget = Math.max(1, opts.statementBudget ?? DIGEST_STATEMENT_BUDGET);
  // The two page reads above.
  let statements = safetyArm > 0 ? 2 : 1;
  let fetches = 0;
  /** I3-0204: what ONE row's EMAIL leg can spend, plus the push-token read,
   * reserved up front and counted from the statements below: contact read,
   * one-click mint, still-current check, stale-token DELETE, email settle (either
   * the `emailed = 1` UPDATE or the transient back-off UPDATE), push-token read;
   * and the one mail. The PUSH leg's cost depends on how many tokens the read
   * returns (up to MAX_PUSH_TOKENS fetches plus a DELETE per dead token plus the
   * settle), so it is reserved after that read, exactly, or not spent at all. */
  // #791a: + the email claim and its release (the release rides the settle / back-off batch).
  // I23-FX-GATE: + a permanent refusal's suppression-list write (after the settle; see DIGEST_ROW_STATEMENTS).
  const ROW_STATEMENTS = DIGEST_ROW_STATEMENTS;
  const ROW_FETCHES = 1;
  // I3-0204: a wall clock as well as the two counts, as both fan legs carry
  // (`alerts.ts` DELIVERY_BUDGET_MS). Checked before each row and before each
  // push leg; a row or leg not reached stays unsettled and the next sweep
  // resumes it.
  const clock = opts.nowMs ?? (() => Date.now());
  const deadline = clock() + Math.max(0, opts.budgetMs ?? DIGEST_DELIVERY_BUDGET_MS);
  let stop = false;
  for (const row of rows) {
    if (stop || clock() >= deadline || statements + ROW_STATEMENTS > statementBudget || fetches + ROW_FETCHES > fetchBudget) break;
    statements += ROW_STATEMENTS;
    fetches += ROW_FETCHES;
    let content: DigestContent | { empty: true };
    try {
      content = JSON.parse(row.content) as DigestContent;
    } catch {
      content = { empty: true };
    }
    if ((content as { empty?: boolean }).empty || (row.kind !== "alert" && Number(row.suppressed) === 1)) {
      await db.prepare("UPDATE creator_digests SET emailed = 1, pushed = 1 WHERE digest_id = ?").bind(row.digest_id).run();
      out.settled++;
      continue;
    }
    const c = content as DigestContent;
    const contact = await db
      .prepare("SELECT email, email_verified, digests_enabled, digest_email_enabled FROM creator_contacts WHERE passport_id = ?")
      .bind(row.passport_id)
      .first<{ email: string | null; email_verified: number; digests_enabled: number; digest_email_enabled: number }>();
    const enabled = (contact?.digests_enabled ?? 1) === 1;
    // 0051 — TWO SWITCHES, NOT ONE, and the email leg is the only one that reads
    // the second. `digests_enabled` is the creator's own master control over the
    // whole stream (email AND push, which is why the push leg below still reads
    // it alone). `digest_email_enabled` is what RFC 8058 one-click writes, and it
    // must be email-only: a mailbox provider's unsubscribe button says "stop
    // these emails", and a creator whose phone went quiet because of it would
    // have no way to connect the two.
    //
    // I3-0165 (D-0882(2)): and `digest_email_enabled` is the ROUNDUP opt-out. It
    // stops roundups only; a `kind = 'alert'` row (payout change, payout account,
    // passkey added/removed, contested account, proof expiry, lapse, ...) is
    // safety mail and follows `digests_enabled` alone — the fan rail's
    // TRANSACTIONAL_KINDS exemption (`alerts.ts`, F-11), applied to this rail.
    const emailEnabled = enabled && (row.kind === "alert" || (contact?.digest_email_enabled ?? 1) === 1);

    // EMAIL: verified + enabled only; otherwise settled with nothing sent.
    // I3-1288: and only when its retry is due (a row read for its push leg alone
    // carries a future `email_next_at`).
    if (row.emailed === 0 && (row.email_next_at === null || row.email_next_at <= nowIso)) {
      // I3-0014 e: the per-address suppression list (email.ts, 0216) on this rail too. A listed address is not handed
      // to the provider and its row settles like a permanent rejection (the email question is closed); a permanent
      // rejection below records the address. No key (no SESSION_SECRET) -> no list, as on the fan rail.
      const supKey = emailEnabled && contact?.email && contact.email_verified === 1 ? suppressionKeyOf(email) : null;
      const addrHash = supKey && contact?.email ? await emailAddressHash(supKey, contact.email) : null;
      if (addrHash && contact?.email && (await suppressedAddressHashes(db, [addrHash], retiredSuppressionLookup(email, [contact.email]))).has(addrHash)) {
        await db.prepare("UPDATE creator_digests SET emailed = 1 WHERE digest_id = ? AND emailed = 0").bind(row.digest_id).run();
        out.emails_suppressed++;
      } else if (emailEnabled && contact?.email && contact.email_verified === 1) {
        const summary = digestSummary(c);
        // RG2C-01 (D-1117(9)): the stored link is asked again at the delivery instant and left out when it no longer serves.
        const psa = c.impersonation?.psa_path && publicBase && (await psaServableAt(db, row.passport_id, nowIso)) ? `\n\nShare your PSA (points fans to your real account): ${publicBase}${c.impersonation.psa_path}` : "";
        // 'alert' USED TO MEAN EXACTLY ONE THING, and both templates hardcoded
        // the verification-expiry wording. It now carries the payout-change
        // notice too, and mailing "your verification is about to expire" over a
        // destination swap would be worse than silence: it teaches the creator
        // that the one alert meaning MONEY IS MOVING is the one they already
        // ignore. The subject line is chosen by the CAUSE, not by the kind.
        // M86: every solicited digest carries a manage/unsubscribe footer and,
        // when the console origin is known, a List-Unsubscribe header.
        const manageFoot = manageDigests ? `\n\nManage or turn off these emails: ${manageDigests}` : "";
        // RFC 8058 one-click, minted per mail. Best effort: a mint that throws
        // must not cost a creator the payout-change notice, so a failure falls
        // back to the RFC 2369 console link this mail already carried.
        let oneClick: string | null = null;
        if (publicBase) {
          try {
            const minted = await mintUnsubscribeTokens(db, "creator_digest_email", [row.passport_id], now());
            const tok = minted.get(row.passport_id);
            // I3-1284 (2) [hand-off from I4-01-R4]: the token is keyed on the
            // passport, not the address. An email confirm-swap that committed
            // AFTER the contact read above (:1940) but BEFORE this mint leaves
            // `oneClick` pointing at a token mailed to the OLD address once the
            // mail below goes out under the NEW one. Re-check the contact is
            // still the one just read before handing out the token; if not,
            // drop the just-minted token rather than mail a stale one.
            const still = tok
              ? await db.prepare("SELECT 1 AS x FROM creator_contacts WHERE passport_id = ? AND email = ? AND email_verified = 1").bind(row.passport_id, contact.email).first()
              : null;
            if (tok && !still) await db.prepare("DELETE FROM unsubscribe_tokens WHERE token = ?").bind(tok).run();
            oneClick = tok && still ? unsubscribeUrl(publicBase, tok) : null;
          } catch (e) {
            logError({ cid: opts.cid ?? "unknown", hop: "cron", msg: "digest_one_click_mint", err: e instanceof Error ? e.name : typeof e });
          }
        }
        // THE SUBJECT LINE IS CHOSEN BY THE CAUSE, NOT BY THE KIND — a third
        // cause now rides on `alert`, and mailing "your verification is about to
        // expire" over a lapsing control proof would send a creator to the wrong
        // screen to fix the wrong thing.
        const bodyText =
          c.passkey_added
            ? `${summary}\n\nSign in to your Dhwaja creator console and open Passkeys to check it.`
            : c.passkey_removed
            ? `${summary}\n\nSign in to your Dhwaja creator console and open Passkeys to check which passkeys you still have.`
            : c.payout_account_change
            ? `${summary}\n\nSign in to your Dhwaja creator console and open Earnings to check it.`
            : c.payout_outcome
            ? `${summary}\n\nSign in to your Dhwaja creator console and open Earnings to see it.`
            : c.payout_change
            ? `${summary}\n\nSign in to your Dhwaja creator console and open Payout to check them.`
            : c.proof_expiry
              ? `${summary}\n\nSign in to your Dhwaja creator console and open Accounts to renew it.`
              : c.operator_lapse
              ? `${summary}\n\nSign in to your Dhwaja creator console and open Verification. Fixing this means getting the agency or person bound to this character live again — an operator can re-bind it only while a listing is unclaimed, so contact Dhwaja if that is what you need.`
              // S-070 RJ-01 — a FIFTH cause on `alert`, and the rule the comment
              // above states applies to it exactly: the body is chosen by the
              // CAUSE. Sending a renewal instruction over an account contest
              // points the creator at the wrong screen for the wrong event.
              : c.contested_account
              ? `${summary}\n\nSign in to your Dhwaja creator console and open Accounts. You do not need to do anything right now — this is a notice that an operator is reviewing it.`
              : c.verification_rejected
              ? `${summary}\n\nSign in to your Dhwaja creator console and open Verification.`
              : c.correction_resolved
              ? `${summary}\n\nSign in to your Dhwaja creator console and open Accounts to see where things stand. You do not need to do anything unless it looks wrong.`
              : c.contest_decided
              ? `${summary}\n\nSign in to your Dhwaja creator console and open Accounts to see what changed. You do not need to do anything unless it looks wrong.`
              : c.report_outcome
              ? `${summary}\n\nSign in to your Dhwaja creator console and open Report an impostor to see the status of the reports you filed.`
              : c.warning_published
              ? summary
              : c.kind === "alert"
                ? `${summary}\n\nSign in to your Dhwaja creator console to start the renewal.`
                : `Your ${c.kind} Dhwaja roundup (${c.period_key}):\n\n${summary}${psa}\n\nSign in to see the full digest.`;
        const unsubHeader = oneClick ?? manageDigests;
        // #791a (I16-W04C; migration 0184): THE CLAIM, immediately before the provider call — the fan alert mail's
        // 739-F claim (alerts.ts, 0148) on this rail. ONE statement re-checks the digest is still unsent and stamps a
        // claim only when none is held or the held one is older than DIGEST_EMAIL_CLAIM_LEASE_MS; only a digest it
        // RETURNS is sent. A Worker that dies after the send and before the settle leaves the claim, so the next sweep
        // finds it held and mails nothing; the settle and the transient back-off delete it in their own batch.
        const claimAt = now();
        const { results: claimed } = await db
          .prepare(
            `INSERT INTO creator_digest_email_claims (digest_id, claimed_at)
             SELECT ?, ? WHERE EXISTS (SELECT 1 FROM creator_digests WHERE digest_id = ? AND emailed = 0)
             ON CONFLICT(digest_id) DO UPDATE SET claimed_at = excluded.claimed_at WHERE creator_digest_email_claims.claimed_at < ?
             RETURNING digest_id`,
          )
          .bind(row.digest_id, claimAt, row.digest_id, new Date(Date.parse(claimAt) - DIGEST_EMAIL_CLAIM_LEASE_MS).toISOString())
          .all<{ digest_id: string }>();
        const releaseClaim = db.prepare("DELETE FROM creator_digest_email_claims WHERE digest_id = ?").bind(row.digest_id);
        // Held by another pass (or a crashed one, inside the lease): no mail and no settle here.
        if (claimed.length === 1) {
          const outcome = await withEmailTimeout(sendOnce(email, {
            to: contact.email,
            subject:
              // A FOURTH cause on `alert`, and the same rule the comment above
              // states: the subject is chosen by the CAUSE. Mailing "your payout
              // destinations were changed" over a Connect binding would send a
              // creator to the Payout screen to look for a change that is on
              // Earnings, and the two are different levers.
              c.passkey_added
                ? "Dhwaja — a passkey was added to your passport"
                : c.passkey_removed
                ? "Dhwaja — a passkey was removed from your passport"
                : c.payout_account_change
                ? c.payout_account_change.action === "bound"
                  ? "Dhwaja — a new account was connected to receive your earnings"
                  : "Dhwaja — your payout account was disconnected"
                : c.payout_outcome
                ? c.payout_outcome.outcome === "paid"
                  ? "Dhwaja — a payout was sent"
                  : "Dhwaja — a payout could not be sent"
                : c.payout_change
                ? "Dhwaja — your payout destinations were changed"
                : c.proof_expiry
                  ? "Dhwaja — renew a control proof before it expires"
                  : c.operator_lapse
                  ? "Dhwaja — nobody is accountable for this character"
                  : c.contested_account
                  ? "Dhwaja — someone has claimed one of your accounts"
                  : c.verification_rejected
                  ? "Dhwaja — your verification request was not approved"
                  : c.correction_resolved
                  ? c.correction_resolved.outcome === "resolved"
                    ? "Dhwaja — a correction you asked for has been resolved"
                    : "Dhwaja — a correction you asked for was declined"
                  : c.contest_decided
                  ? "Dhwaja — a decision was made on your contest"
                  : c.report_outcome
                  ? c.report_outcome.outcome === "published"
                    ? "Dhwaja — a warning was published about an account you reported"
                    : "Dhwaja — a report you filed has been closed"
                  : c.warning_published
                  ? "Dhwaja — a warning was published about your account"
                  : c.kind === "alert"
                    ? "Dhwaja — your verification is about to expire"
                    : c.kind === "daily"
                      ? "Dhwaja roundup — something needs your attention"
                      : "Your weekly Dhwaja roundup",
            // I3-0163 (D-0882(2)): the session-free one-click URL in the BODY as
            // well as the header; a click (a GET) reaches `GET /unsubscribe/:token`
            // (index.ts), which serves `unsubscribeConfirmPage` (unsubscribe.ts) and
            // changes nothing until its button POSTs.
            text: `${bodyText}${manageFoot}${oneClick ? `\nTurn off roundup emails (no sign-in needed): ${oneClick}` : ""}`,
            // I3-0543 (D-0920 option B): the styled part, in the shared frame (email.ts wrapHtml). The body is the text
            // part's own words, escaped; the footer carries the SAME manage URL and one-click URL as the text part.
            html: wrapHtml(
              textToHtml(bodyText),
              [
                manageDigests ? `<a href="${escapeHtml(manageDigests)}">Manage or turn off these emails</a>.` : "",
                oneClick ? `<a href="${escapeHtml(oneClick)}">Turn off roundup emails</a> (no sign-in needed).` : "",
              ].filter((x) => x).join(" "),
            ),
            // One-click REPLACES the console fragment in the header rather than
            // joining it — RFC 8058 §3 wants one https URI, and the fragment link
            // answers a POST with a 405. The human's link stays in `manageFoot`.
            ...(unsubHeader ? { listUnsubscribe: `<${unsubHeader}>` } : {}),
            ...(oneClick ? { listUnsubscribePost: true } : {}),
          }), opts.emailTimeoutMs ?? DIGEST_EMAIL_TIMEOUT_MS);
          // I3-0014 AT THE SECOND CALL SITE. A PERMANENT rejection settles this row
          // exactly as an acceptance does, because the row's EMAIL QUESTION IS
          // CLOSED: re-offering the same message to the same address cannot turn
          // into an acceptance. Leaving it `emailed = 0` is what made a dead
          // address a standing charge against this service's sender reputation on
          // every sweep, forever. Only `"transient"` is retried — which is the
          // whole of what the boolean this replaced could ever have meant.
          if (outcome !== "transient") {
            await db.batch([db.prepare("UPDATE creator_digests SET emailed = 1 WHERE digest_id = ?").bind(row.digest_id), releaseClaim]);
          }
          // I3-0014 e: a permanent refusal puts the address on the list, so the next row for it costs no send. After
          // the settle committed: a list write that fails is logged and never turns the settled row into a failure.
          if (outcome === "permanent" && addrHash) {
            try {
              await recordEmailSuppressions(db, [addrHash], nowIso);
            } catch (e) {
              logError({ cid: opts.cid ?? "unknown", hop: "cron", msg: "digest_suppression_record_failed", err: e instanceof Error ? e.name : typeof e });
            }
          }
          if (outcome === "accepted") out.emails_sent++;
          if (outcome === "permanent") out.emails_permanently_rejected++;
          if (outcome === "ambiguous") out.emails_ambiguous++;
          // I3-1288 (0145): "transient" backs off on the fan legs' schedule
          // (`alerts.ts` DELIVERY_RETRY_*): next_at = now + min(BASE << attempts, CAP),
          // and the MAX-th transient SETTLES the row instead of retrying forever.
          if (outcome === "transient") {
            await db.batch([
              db
                .prepare(
                  "UPDATE creator_digests SET emailed = CASE WHEN email_attempts + 1 >= ? THEN 1 ELSE 0 END, " +
                    "email_next_at = strftime('%Y-%m-%dT%H:%M:%fZ', ?, '+' || MIN(? << email_attempts, ?) || ' seconds'), " +
                    "email_attempts = email_attempts + 1 WHERE digest_id = ? AND emailed = 0",
                )
                .bind(DELIVERY_RETRY_MAX_ATTEMPTS, nowIso, DELIVERY_RETRY_BASE_S, DELIVERY_RETRY_CAP_S, row.digest_id),
              releaseClaim,
            ]);
            if ((row.email_attempts ?? 0) + 1 >= DELIVERY_RETRY_MAX_ATTEMPTS) out.emails_retry_exhausted++;
          }
        }
      } else {
        await db.prepare("UPDATE creator_digests SET emailed = 1 WHERE digest_id = ?").bind(row.digest_id).run(); // nothing to send → settled
      }
    }

    // PUSH: a short "it's ready" ping to every registered token; one success
    // settles the channel (a dead token among several must not wedge it).
    // #735 (0151): and only when its retry is due (a row read for its email leg
    // alone carries a future `push_next_at`).
    if (row.pushed === 0 && (row.push_next_at === null || row.push_next_at <= nowIso)) {
      const { results: tokens } = enabled
        ? await db.prepare("SELECT platform, push_token FROM creator_push_tokens WHERE passport_id = ?").bind(row.passport_id).all<{ platform: "apns" | "fcm"; push_token: string }>()
        : { results: [] as { platform: "apns" | "fcm"; push_token: string }[] };
      // I3-0204: reserve THIS row's push leg exactly — one fetch and at most one
      // dead-token DELETE per token, plus the settle — or leave it unsettled
      // (`pushed = 0`, read again next sweep) and stop the pass here.
      if (clock() >= deadline || statements + tokens.length + 1 > statementBudget || fetches + tokens.length > fetchBudget) {
        stop = true;
        continue;
      }
      statements += tokens.length + 1;
      fetches += tokens.length;
      if (tokens.length === 0) {
        await db.prepare("UPDATE creator_digests SET pushed = 1 WHERE digest_id = ?").bind(row.digest_id).run();
      } else {
        let any = false;
        let transient = false;
        // I3-0204: the row's tokens are pushed CONCURRENTLY (at most
        // MAX_PUSH_TOKENS), so one row costs one provider round-trip of wall
        // clock, not five; outcomes are read back in token order below.
        const oks = await Promise.all(tokens.map((t) => push(
            { device_id: `creator:${row.passport_id}`, platform: t.platform, push_token: t.push_token },
            // Name-free by construction, like every other push: the creator's
            // own passport id used to ride along in the provider payload.
            // Still name-free: "your verification" and "your payout
            // destinations" say nothing about WHICH creator this device belongs
            // to, nor what the destinations are, which is the property the push
            // payload has to keep.
            {
              account_id: "",
              kind: c.passkey_added
                ? "digest_passkey_added"

          } else if (outcome === "retry") transient = true;

          }

          // whose tokens were ALL dead is not a retry: they are deleted, and the
          // next pass settles the row on zero tokens.
          await db
            .prepare(
              "UPDATE creator_digests SET pushed = CASE WHEN push_attempts + 1 >= ? THEN 1 ELSE 0 END, " +
                "push_next_at = strftime('%Y-%m-%dT%H:%M:%fZ', ?, '+' || MIN(? << push_attempts, ?) || ' seconds'), " +

  if (out.emails_retry_exhausted > 0) {

  }

    logError({ cid: opts.cid ?? "unknown", hop: "cron", msg: "digest_push_retry_exhausted", failed: out.pushes_retry_exhausted, kind: "transient" });
  }
  // I6MAIL-04: an ambiguous send is settled above (no re-send) — ONE line per
  // pass says how many, so a provider that accepts-then-hangs is visible.
  if (out.emails_suppressed > 0) {
    logError({ cid: opts.cid ?? "unknown", hop: "cron", msg: "digest_email_suppressed_settled", count: out.emails_suppressed, kind: "suppressed" });
  }
  if (out.emails_ambiguous > 0) {
    logError({ cid: opts.cid ?? "unknown", hop: "cron", msg: "digest_email_ambiguous_settled", count: out.emails_ambiguous, kind: "ambiguous" });
  }
  if (out.emails_permanently_rejected > 0) {
    logError({
      cid: opts.cid ?? "unknown",
      hop: "cron",
      msg: "digest_permanent_rejection",
      failed: out.emails_permanently_rejected,
      kind: "permanent",
    });
  }
  return out;
}

/**
 * The creator's own digests, newest first, empties filtered out.
 *
 * I3-2153: the empty-placeholder filter used to run AFTER the LIMIT, in
 * application code -- so on a quiet stretch (several {"empty":true} rows in a
 * row, the common case for a low-activity creator) the newest `limit` rows
 * fetched could be ALL placeholders, pushing genuinely notable older digests
 * off the list entirely even though they still exist. The placeholder is
 * always written as the literal two-key object `{"empty":true}`
 * (`JSON.stringify({ empty: true })`, stable key order for a single-key
 * object), so it can be excluded in SQL and LIMIT then applies to real rows
 * only.
 */
export async function listDigests(db: DbLike, passportId: string, limit = 10): Promise<Array<{ digest_id: string; kind: string; period_key: string; content: DigestContent; created_at: string }>> {
  const { results } = await db
    .prepare("SELECT digest_id, kind, period_key, content, created_at FROM creator_digests WHERE passport_id = ? AND content != '{\"empty\":true}' ORDER BY created_at DESC LIMIT ?")
    .bind(passportId, Math.max(1, Math.min(50, limit)))
    .all<{ digest_id: string; kind: string; period_key: string; content: string; created_at: string }>();
  const out: Array<{ digest_id: string; kind: string; period_key: string; content: DigestContent; created_at: string }> = [];
  for (const r of results) {
    try {
      const c = JSON.parse(r.content) as DigestContent & { empty?: boolean };
      if (c.empty) continue;
      out.push({ digest_id: r.digest_id, kind: r.kind, period_key: r.period_key, content: c, created_at: r.created_at });
    } catch {
      /* skip malformed */
    }
  }
  return out;
}

/** I3-1673: rows per retention page, and pages per call -- the bounded shape
 * `pruneNotifications` (alerts.ts) already uses for the same reason: a DELETE
 * with an unbounded WHERE can run long against a big table, so this pages
 * oldest-first and stops a page short of a full page. */
export const CREATOR_DIGEST_PRUNE_PAGE = 5000;
export const CREATOR_DIGEST_PRUNE_PAGES = 20;

/** I3-1673: how long a `creator_digests` row is kept. The sibling fan-facing
 * table (`notifications`, alerts.ts `NOTIFICATION_RETENTION_DAYS`) is pruned
 * after 180 days; `creator_digests` had NO retention sweep at all -- every
 * digest ever generated, notable or the `{"empty":true}` placeholder, stayed
 * forever, and `listDigests`'/`pendingDigestCount`'s own scans grow with it.
 * 180 days matches the sibling window and is far past any UI or delivery
 * window that reads this table. */
export const CREATOR_DIGEST_RETENTION_DAYS = 180;

/**
 * Retention for `creator_digests` -- the table that had none (I3-1673).
 *
 * Bounded pages, oldest first, exactly like `pruneNotifications` (alerts.ts):
 * `digest_id IN (SELECT ... ORDER BY created_at ASC LIMIT ?)`, at most `pages`
 * of them; a short page ends the call. Returns the rows removed. Wiring this
 * into the daily cron sweep (alongside `pruneNotifications`) is outside this
 * file's ownership for this item -- see the REAL -- WRONG OWNER patch in this
 * room's report.
 */
export async function pruneCreatorDigests(db: DbLike, cutoffIso: string, opts: { page?: number; pages?: number } = {}): Promise<number> {
  const page = opts.page ?? CREATOR_DIGEST_PRUNE_PAGE;
  const pages = opts.pages ?? CREATOR_DIGEST_PRUNE_PAGES;
  let removed = 0;
  for (let i = 0; i < pages; i++) {
    const res = await db
      .prepare("DELETE FROM creator_digests WHERE digest_id IN (SELECT digest_id FROM creator_digests WHERE created_at < ? ORDER BY created_at ASC LIMIT ?)")
      .bind(cutoffIso, page)
      .run();
    const n = res.changes ?? 0;
    removed += n;
    if (n < page) break;
  }
  return removed;
}

export function dailyPeriod(nowMs: number): { key: string; startIso: string; endIso: string } {
  const end = new Date(nowMs);
  const dayStart = Date.UTC(end.getUTCFullYear(), end.getUTCMonth(), end.getUTCDate());
  // The DAILY digest covers YESTERDAY (a completed day), keyed by that day.
  const start = dayStart - 24 * 60 * 60 * 1000;
  return { key: new Date(start).toISOString().slice(0, 10), startIso: new Date(start).toISOString(), endIso: new Date(dayStart).toISOString() };
}

/** I3-1767: how many past days a single catch-up call will look back over.
 * Bounded so a long-dead cron cannot make one tick loop unboundedly; a gap
 * wider than this is a longer outage than the daily digest is meant to
 * survive automatically. */
export const DAILY_BACKFILL_MAX_DAYS = 14;

/**
 * I3-1767: catch-up for the daily digest.
 *
 * `dailyPeriod(nowMs)` only ever names YESTERDAY relative to the moment it is
 * called. The cron caller (index.ts) called it once per tick with the CURRENT
 * `nowMs` and generated only that one day -- so if the cron did not fire at
 * all for over a day (a deploy, an outage), the day(s) in between were never
 * named by any call and `generateDigests` never ran for them: gone for good,
 * with no backfill.
 *
 * This returns every one of the last `maxDays` calendar days (oldest first,
 * so a caller that runs them in order backfills chronologically) that does
 * not yet have ANY `creator_digests` row for kind='daily' at that period_key.
 * Presence of any row is a reasonable "already swept" signal here because
 * `generateDigests` (as of I3-1282) pages through every eligible creator for
 * a period in one call (bounded by `maxPages`), so a period that was ever
 * swept to completion has a row for every eligible creator, not a partial
 * set; a day with zero eligible creators keeps being returned harmlessly
 * (the caller's `generateDigests` for it is a cheap no-op).
 *
 * Wiring this into the cron caller in place of the single `dailyPeriod(nowMs)`
 * call is outside this file's ownership for this item -- see the REAL --
 * WRONG OWNER patch in this room's report.
 */
export async function dailyPeriodsToBackfill(
  db: DbLike,
  nowMs: number,
  maxDays: number = DAILY_BACKFILL_MAX_DAYS,
): Promise<Array<{ key: string; startIso: string; endIso: string }>> {
  const out: Array<{ key: string; startIso: string; endIso: string }> = [];
  for (let i = maxDays; i >= 1; i--) {
    const p = dailyPeriod(nowMs - (i - 1) * 24 * 60 * 60 * 1000);
    const row = await db.prepare("SELECT 1 AS x FROM creator_digests WHERE kind = 'daily' AND period_key = ? LIMIT 1").bind(p.key).first<{ x: number }>();
    if (!row) out.push(p);
  }
  return out;
}

export function weeklyPeriod(nowMs: number): { key: string; startIso: string; endIso: string } {
  // ISO week of the CURRENT moment; window = the prior completed 7 days
  // ending at the most recent Monday 00:00 UTC. Generated once per week
  // (UNIQUE key) whenever the first sweep of the new week runs.
  const d = new Date(nowMs);
  const day = (d.getUTCDay() + 6) % 7; // Mon=0
  const mondayStart = Date.UTC(d.getUTCFullYear(), d.getUTCMonth(), d.getUTCDate()) - day * 24 * 60 * 60 * 1000;
  const start = mondayStart - 7 * 24 * 60 * 60 * 1000;
  // ISO week number of the covered week's Thursday (standard trick).
  const thu = new Date(start + 3 * 24 * 60 * 60 * 1000);
  const jan1 = Date.UTC(thu.getUTCFullYear(), 0, 1);
  const week = Math.ceil(((thu.getTime() - jan1) / (24 * 60 * 60 * 1000) + 1) / 7);
  return { key: `${thu.getUTCFullYear()}-W${String(week).padStart(2, "0")}`, startIso: new Date(start).toISOString(), endIso: new Date(mondayStart).toISOString() };
}
