/**
 * F10 / #52 — WHERE A CREATOR'S AUDIENCE ACTS FROM, and nothing narrower.
 *
 * The privacy argument lives in `0068_audience_regions.sql`; the short version
 * is that the table has no account column and a fan's country is never held
 * anywhere but a local variable between reading a header and incrementing a
 * counter.
 *
 * THIS MODULE MUST NEVER LEARN WHO ACTED, and since ADR-0006 §2 it cannot: the
 * only writer is the encounter path's token-claim batch (`codex.ts
 * recordEncounter`), which owns the account id it needs for the claim and binds
 * this table's increment to a row it never reads back. This module is now the
 * READER and the shape, and it takes no account id in any signature.
 *
 * ---------------------------------------------------------------------------
 * THREE THINGS EARLIER VERSIONS GOT WRONG, kept because each is why a piece of
 * the current rule is shaped the way it is.
 * ---------------------------------------------------------------------------
 *
 * 1. THE UNIT WAS ACTIONS, THEN FAN-DAYS, AND NEITHER WAS A FAN.
 *    v1 counted raw actions, so a creator's only Icelandic fan opening their
 *    share page five times published `IS`. v2 counted FAN-DAYS and deduped them
 *    with a point-read of `creator_encounters.last_seen_at` before the write —
 *    two concurrent requests both read the old day and both incremented, so a
 *    column defined as "one fan, once" could count one fan twice, and the day
 *    the increment landed on came from a timestamp minted before that read.
 *    NOW the unit is DISTINCT FANS per (creator, country, UTC day), and it is a
 *    CLAIM rather than a count: the Worker mints a CSPRNG `claim_token`, one
 *    atomic batch writes `(claimed_day, claim_token)` on the pair's
 *    `creator_encounters` row only when the day advances, and the geo increment
 *    is `SELECT … WHERE claim_token = ?token` off that same row (ADR-0006 §2,
 *    O-12/O-23, migration 0079). Two requests in the same millisecond cannot
 *    both win, and the day comes off the row that won rather than off the
 *    clock that lost. `fan_days` keeps its column name and means distinct fans.
 *
 * 2. THE FLOOR WAS OVER A MONTH, SO IT COULD NOT BE STICKY.
 *    `REGION_FLOOR = 20` floored fan-days per calendar month, and a month is
 *    not a thing that seals: a country under the floor in mid-September was
 *    over it by the 30th, and the difference between the two reads was that
 *    country's whole withheld history. The floor also had to be far above the
 *    k-floor precisely because one person could supply many fan-days.
 *    NOW the floor is `K_ANONYMITY_FLOOR` and it applies PER SEALED DAY-CELL
 *    (ADR-0006 §3, §6, O-4). A day seals at `day + 1` 00:00 UTC and is never
 *    re-opened, so what a day serves it serves forever — stickiness is a
 *    property of the unit rather than a cache. K is the right number again
 *    because the population it floors is distinct accounts once more. A period
 *    view is Σ over the day-cells that were served and is NEVER RE-FLOORED:
 *    re-flooring a sum is what let a month hand back a day it had withheld.
 *
 * 3. THE TOTAL WAS THE OTHER OPERAND OF A SUBTRACTION, AND THE REPAIR WAS A
 *    RULE THIS MODULE INVENTED FOR ITSELF.
 *    v1 withheld `other` and served `total` beside the buckets, so
 *    `total − Σ buckets` was the hidden residual exactly. v2 fixed it by
 *    withholding the total whenever anything was folded — sound in isolation,
 *    and pass-3 BREAK 3 is that per-widget rules do not compose: a floor here
 *    and a nesting rule elsewhere never see each other, and the creator reads
 *    the difference.
 *    NOW THERE IS NO SUBTRACTION CHANNEL AND NO LOCAL RULE. The served set is
 *    `serveDay`'s output over a declared cell graph (ADR-0006 §0 P1): the
 *    countries PARTITION `active_fans` — `active_fans(day) := Σ_country
 *    fan_days` by definition, not a second population beside them (§2, O-17) —
 *    withheld countries pool into `other`, complementary suppression keeps the
 *    unseen mass of that equation at ≥ K and never leaves it as the equation's
 *    only unknown, and every served total is Σ over its served children. On top
 *    of it sits the ADR §6 geo day rule: a day under `GEO_DAY_MIN_ACTIVE`
 *    serves NO country rows at all, whatever any individual country's count is.
 *
 * `geoServedSet` is that graph and that call, exported, so this reader and
 * `analytics-read.ts`'s day object serve the identical set. Two spellings of
 * one floor is how two surfaces end up disagreeing about one fan.
 */

/** ISO-3166-1 alpha-2, and the bucket everything else falls into. */
const COUNTRY_RE = /^[A-Z]{2}$/;
export const UNKNOWN_COUNTRY = "ZZ";

/**
 * Cloudflare's own non-answers. `XX` is "could not determine", `T1` is Tor.
 * Both map to ZZ rather than keeping their own buckets: a `T1` column on a
 * creator's dashboard would tell them which of their fans browse over Tor,
 * which is a fact about a person's threat model and none of a creator's
 * business.
 */
const CF_NON_ANSWERS = new Set(["XX", "T1"]);

export function normalizeCountry(raw: unknown): string {
  if (typeof raw !== "string") return UNKNOWN_COUNTRY;
  const up = raw.trim().toUpperCase();
  if (!COUNTRY_RE.test(up) || CF_NON_ANSWERS.has(up)) return UNKNOWN_COUNTRY;
  return up;
}

const DAY_MS = 24 * 60 * 60 * 1000;

/**
 * Thirteen months, not twelve: a creator comparing September to last September
 * needs both, and nothing beyond that comparison justifies keeping a permanent
 * record of where somebody's audience used to be.
 */
export const RETENTION_MONTHS = 13;

/** One country's served cell. `fans` — distinct fans, per (1). */
export interface GeoCountry {
  country: string;
  fans: number;
}

/** `serveDay`'s output for one day's countries, and nothing raw. */
export interface GeoServed {
  /** Σ_country fan_days, when that cell is served. */
  active_fans: number | null;
  /** Served country cells only, largest first. */
  countries: GeoCountry[];
  /** The pooled residual, when the pool itself is served. */
  other: number | null;
  /** True when the served cells do not enumerate the day. */
  withheld: boolean;
}

/**
 * THE ONE PLACE THE GEO RULE IS SPELLED (ADR-0006 §6 geo row), for one day.
 *
 * The graph is exactly `analytics-cells.ts buildDayGraph`'s geo block, and
 * `test/s002-regions.test.ts` asserts it cell-for-cell against a full day
 * object: geo names no cell outside this partition, so serving it alone gives
 * the same answer as serving it inside the whole object, and `analytics-read.ts`
 * can call this instead of rebuilding it.
 */
export function geoServedSet(countries: Readonly<Record<string, number>>): GeoServed {
  const names = Object.keys(countries).sort();
  let activeFans = 0;
  for (const cc of names) activeFans += countries[cc] as number;

  const cells: Cell[] = [{ id: cellId.activeFans, kind: "population", value: activeFans }];
  const relations: Relation[] = [];
  if (names.length > 0) {
    for (const cc of names) cells.push({ id: cellId.country(cc), kind: "population", value: countries[cc] as number });
    cells.push({ id: cellId.geoOther, kind: "population", value: 0 }); // every country is enumerated, so the base is 0
    relations.push({
      type: "partition",
      parent: cellId.activeFans,
      children: names.map((cc) => cellId.country(cc)),
      other: cellId.geoOther,
      // The geo day rule: under ten active fans the breakdown is not withheld,
      // it does not exist — so nothing else in the algebra can recover it.
      ...(activeFans < GEO_DAY_MIN_ACTIVE ? { blocked: true } : {}),
    });
  }

  const set = serveDay(cells, { relations });
  const served: GeoCountry[] = [];
  for (const cc of names) {
    const v = set.served.get(cellId.country(cc));
    if (v !== undefined) served.push({ country: cc, fans: v });
  }
  served.sort((a, b) => b.fans - a.fans || (a.country < b.country ? -1 : 1));
  const active = set.served.get(cellId.activeFans) ?? null;
  return {
    active_fans: active,
    countries: served,
    other: set.served.get(cellId.geoOther) ?? null,
    // Not "something was folded into `other`" — that is legible from `other`
    // itself. This is "the rows you can see do not account for the day", which
    // is the thing a creator would otherwise fill in with arithmetic.
    withheld: active === null || served.length < names.length,
  };
}

/**
 * I3-1337: a day's `audience_regions` cells are served only from D+1 00:30 UTC.
 * Its writers commit a decision made before midnight up to that long after it —
 * the same bound as `analytics-read.ts` BUILD_GRACE_MS, which the frozen
 * dashboards honour; kept as its own constant here (that module imports half of
 * the analytics graph) and pinned equal by `i4-04-r3-regions-grace.test.ts`.
 */
export const REGIONS_PUBLICATION_GRACE_MS = 30 * 60 * 1000;

/** One SEALED UTC DAY of geo, served. `day` is D = T−1; see `audienceRegions`. */
export interface AudienceRegions extends GeoServed {
  /** The day these cells describe, `YYYY-MM-DD`. */
  day: string;
}

export const REGION_BASIS =
  "Counted in DISTINCT FANS per country per UTC day: one person, one day, counted once however many times they act, claimed once by the request that won the day. Dhwaja never stores anyone's country, so a country is where somebody acted from and not where they live. The running day is never included, each day is floored on its own and stays that way.";

/**
 * The creator's own read: the SEALED DAY D = T − 1, and nothing else.
 *
 * TODAY IS EXCLUDED, and that is still the differencing control. A read that
 * included the running day would let a creator read, have one known fan act,
 * read again, and take the delta as that person's country — reconstructing from
 * two permitted reads exactly the row the schema refuses to store. D is sealed
 * (ADR-0006 §3), so every read of this (creator, T) returns the same bytes at
 * every hour and there is no pair of reads to difference.
 *
 * THE FLOOR IS NOT APPLIED HERE. It is applied by `serveDay` over the declared
 * graph, which is the only thing entitled to decide what leaves — see (3).
 */
export async function audienceRegions(db: DbLike, passportId: string, nowMs: number = Date.now()): Promise<AudienceRegions & { basis: string }> {
  // I3-1337: T−1 only once its writers' grace has passed; before it, T−2.
  const D = dayOf(nowMs - REGIONS_PUBLICATION_GRACE_MS - DAY_MS);
  const { results } = await db
    .prepare("SELECT country, fan_days FROM audience_regions WHERE passport_id = ? AND day = ? ORDER BY country ASC")
    .bind(passportId, D)
    .all<{ country: string; fan_days: number }>();
  const countries: Record<string, number> = {};
  for (const r of results) countries[r.country] = r.fan_days;
  // I3-0757: the counting basis rides with the read it explains (served by
  // `GET /account/regions` unchanged, rendered under the console's table).
  return { day: D, ...geoServedSet(countries), basis: REGION_BASIS };
}

/**
 * Cron: drop days past the retention window.
 *
 * The cut is the first day of the month RETENTION_MONTHS-1 back, so exactly
 * thirteen months INCLUDING the current one survive. The first version was off
 * by one and kept fourteen — and the test stepped over the only value that
 * would have shown it.
 */
export async function sweepOldRegions(db: DbLike, nowMs: number = Date.now()): Promise<number> {
  const d = new Date(nowMs);
  const cut = new Date(Date.UTC(d.getUTCFullYear(), d.getUTCMonth() - (RETENTION_MONTHS - 1), 1));
  const res = await db.prepare("DELETE FROM audience_regions WHERE day < ?").bind(dayOf(cut.getTime())).run();
  return res.changes ?? 0;
}

/**
 * Registry bucket builder. Runs inside trust-core (the only service bound to
 * the trust DB) and writes bucket files the registry service serves verbatim.
 *
 * VERBATIM is why this file carries gates other public surfaces apply at READ
 * time. `services/registry` has no DB binding: whatever bytes land in R2 are
 * what a fan is shown until the next rebuild. Every other artifact (discovery,
 * Most Targeted, ratings, top-fans) is re-filtered by its serving function
 * against `request_suppressions`; this one cannot be, so the filter has to
 * happen HERE.
 *
 * Privacy properties enforced here:
 *  - Entries are {h, c} ONLY: the hash of a canonical subject and its class.
 *    No plaintext, and crucially NO passport ref on CREATOR entries — a shared
 *    passport id would (a) link a creator's accounts across buckets and
 *    (b) mark real entries (decoys can't share an id). Deep-linking to a
 *    passport after a match is an S7 concern, designed there with privacy in
 *    mind, not baked into the public bucket. So real and decoy CREATOR entries
 *    are byte-identical in shape. The ONE deliberate exception is the
 *    find-the-real-one pointer on WARNING entries (`r`, below): warnings are
 *    public-by-design (decoys take the class and the pointer too, at the real
 *    rate — see `decoyPointer`), and the protected creator's
 *    passport id + display name are exactly what the feature publishes — the
 *    pointer marks nothing that wasn't already marked and links only the
 *    (public) warning to the (public) verified profile it protects.
 *  - A SUPPRESSED subject is not published at all. A removal-request
 *    suppression means "gone from every public surface"; before this gate the
 *    hash stayed in the bucket while `/p/:id` 404'd, which made one
 *    unauthenticated, edge-cached bucket read a "did this person ask to be
 *    delisted?" oracle — the exact question the removal feature exists to make
 *    unanswerable. Suppressed account hashes still take part in PRECEDENCE
 *    (`precedence` below) so dropping them can never promote a warning or a
 *    caution onto a hash that creator identity used to win. Dropping a subject
 *    costs one of the bucket's `churn` departure slots, so an unusual burst of
 *    removals in one bucket drains over a few builds — see THE CHURN QUOTA.
 *    The gate reads the SUPPRESSION **and** the `suppressed_account_keys`
 *    snapshot (migration 0028), because the first resolves through
 *    `passport_accounts.passport_id` and `account_transferred` moves that row:
 *    one transfer off the delisted passport used to put the hash back in the
 *    buckets as `unclaimed` on the very next build.
 *  - Every bucket is padded to the same ENTRY COUNT (`padTarget`) with decoys
 *    whose hashes are indistinguishable from real SHA-256 hashes and whose
 *    class mix follows the global real distribution. Uniform count means
 *    per-bucket entry count reveals nothing about per-bucket real count.
 *    Decoys are DERIVED FROM A SECRET, NOT MINTED FRESH (see `decoyHash`):
 *    freshly-random decoys made two consecutive builds intersect down to
 *    exactly the real set.
 *  - Every bucket is ALSO padded to the same BYTE LENGTH (`padBodies`), and
 *    that is a separate property from the entry count. An entry's class is a
 *    WORD on the wire — "unclaimed" is nine characters, "claimed" seven,
 *    "verified" eight — a confirmed entry carries `p` and a warning carries `r`,
 *    so the per-bucket MIX set the body length even though the COUNT was
 *    uniform. Measured: 12 distinct lengths across 16 buckets on a live build,
 *    and 15 of 16 on a 300-subject fixture holding no warnings at all. A bucket
 *    is a cacheable GET with a stable body, so an attacker builds the
 *    size -> prefix table once and from then on a passive on-path observer reads
 *    the prefix off every user's check — the one thing the whole
 *    hash-prefix design keeps from the network. This used to be written down as
 *    an accepted residual on the grounds that "bucket contents are public
 *    anyway", which answers a different question: what an observer who FETCHES a
 *    bucket learns, not what an observer who only sees its SIZE learns.
 *    RESIDUAL, stated rather than hidden: uniformity is over the bytes this
 *    builder writes. A compressing intermediary re-opens a narrower version of
 *    the channel, because compressed length still depends on content. The filler
 *    is high-entropy (`fillerFor`) so that a naive `Content-Encoding: gzip` in
 *    front of the registry cannot simply squeeze the padding back out, but no
 *    claim is made that the compressed lengths are uniform — they are not, and
 *    the fix for that is a compression-off posture at the edge, not a builder
 *    parameter.
 *  - EVERY BUCKET SHEDS EXACTLY `churn` PUBLISHED HASHES, EVERY BUILD. Not
 *    "as many as the events required", not "as many as the events required plus
 *    a rotation" — exactly `churn`, whatever happened. See THE CHURN QUOTA.
 *  - AND AT LEAST `churn - realChurn` OF THEM ARE DECOYS, EVERY BUILD. The
 *    count being constant is not enough: real departures are not only removal
 *    requests, and anyone who can get a listing admitted can un-list it, so an
 *    attacker who owned every real departure slot owned the whole crowd and the
 *    residue after subtracting his own hashes WAS the delisted subject. Real
 *    departures are capped at `realChurnFor(pad)`; the balance is always paid
 *    out of decoys retired under the secret seed. See DECOY_CHURN_SHARE.
 *  - Width and the pad ladder follow a MONOTONE COUNT OF ADMITTED LISTINGS, not
 *    the current real count, so a removal can neither trigger nor delay a rung
 *    crossing or a width change. See `LadderState`.
 *
 * THE CHURN QUOTA — read this before changing anything below.
 *
 * A build is not one event. `wrangler.jsonc` rebuilds on `17 * / 6 * * *`, so
 * every published build carries six hours of accumulated listings and removals,
 * and `chooseWidth` holds width 1 — SIXTEEN buckets — until 256,000 subjects.
 * Those events collide in a bucket constantly, so the property has to be stated
 * per bucket and over a batch. The observable, for someone archiving
 * `/v1/bucket/:prefix`, is exactly three numbers per bucket per build:
 * ARRIVALS, DEPARTURES and PUBLISHED SIZE. Size is uniform and published, and
 * arrivals - departures = the change in size, so the whole observable collapses
 * to DEPARTURES.
 *
 * So departures is the number that must not depend on what happened:
 *
 *     departures(bucket, build) = churn(padTarget), a published constant.
 *
 * Not symmetric in listings and removals — INDEPENDENT of both. Real removals
 * consume departure slots first (oldest request first); decoys are retired to
 * fill the rest. Arrivals are whatever refills the bucket to `padTarget`: the
 * real entries admitted this build, then freshly minted decoys.
 *
 * WHY IT HAS TO BE A CONSTANT, AND WHY THAT FORCES A DELISTING SLA. A bucket can
 * only shed a hash it published, and the only real hashes it may shed are the
 * ones that stopped being real (`b` of them). So
 *
 *     departures = b + (decoys retired),   decoys retired <= live decoys.
 *
 * For departures not to depend on `b`, it must hold at b = 0, so it is bounded
 * by the decoys the bucket is GUARANTEED to hold — and it must hold for every
 * history, since an observer keeps every build, so it is bounded by the
 * guaranteed decoy floor. That floor is reserved out of `padTarget`
 * (`capacityFor`). Taking the largest feasible constant is the best any
 * b-independent scheme can do, and it caps real departures at `churn` per
 * bucket per build: if more removals than that land in one bucket in one
 * window, the remainder waits for the next build. That is the published
 * DELISTING SLA (docs/REGISTRY-PRIVACY-CLAIMS.md, `LISTING-POLICY.md` 5.1),
 * and it is a consequence of the arithmetic, not a tuning choice.
 *
 * THE THREE WAYS THIS WAS GOT WRONG BEFORE, ALL THE SAME MISTAKE — the scheme
 * TUNED the churn so that removals hid inside it, and every tuning left a
 * regime where the churn went to zero (measured, 2026-08-19):
 *
 *  1. A STACK. Padding was "the first N stream elements that land in this
 *     bucket", so a listing popped the tail decoy and a removal pushed that
 *     exact decoy back. "The arriving hash was published before, so this was a
 *     removal, and the departing hash is the delisted subject's" scored 10 of
 *     10, recovering 6 of 6 delisted hashes with no dictionary.
 *  2. NET-GATED ROTATION. `rotate = R * |net change|`, skipped at net zero. One
 *     listing plus one removal in the same bucket rotated nothing and swapped
 *     the two real entries alone: `arrivals=1 departures=1`, the departure
 *     being the delisted subject's hash. 26 of 26 buckets, no false positives,
 *     zero firings on a listings-only control. An attacker who plants one
 *     listing in every bucket forces that collision wherever a removal lands:
 *     32 of 32 delisted hashes recovered exactly.
 *  3b. Numbered 3 below because it broke the same design. See DECOY_CHURN_SHARE
 *     for the fourth, which broke the CONSTANT itself rather than a tuning of
 *     it: the count was independent of what happened, and the CROWD was not.
 *  3. THE RUNG CROSSING. `rotate = min(R, floor(live/4), live - |delta|)`. When
 *     `padTargetFor` steps up a rung, `delta` is about +padTarget, so
 *     `live - |delta|` is hugely negative and rotation is zero, while `delta > 0`
 *     makes the mandatory move zero too. Departures collapse to exactly the real
 *     removals, and the rule needs only the PUBLISHED bucket sizes: "if a
 *     bucket file's entry count grew, every hash that left this build is a
 *     delisted subject." Measured on natural growth with no attacker at all:
 *     120 builds, 360 removals, the rule fired on 2 builds and recovered 6 of 6
 *     departures exactly, no false positives. An attacker grinds handles into
 *     one bucket (the canonicalizer is public) and chooses when the crossing
 *     happens.
 *
 *  4. THE BORROWED CROWD. Departures were a constant and the constant was paid
 *     REAL-FIRST with no floor under the decoy half, so an attacker who could
 *     mint departures could own the crowd. He plants `churn` ground handles per
 *     bucket, lets them publish, revokes them (`account_revoked`, reachable via
 *     CREATOR_SIGNABLE and AGENCY_ACTIONABLE); suppressions sort ahead of
 *     voluntary leavers, so the victim's removal still took its slot and his
 *     hashes took every other one. `retire` fell to zero, not one decoy left,
 *     and subtracting his own hashes left the delisted subject alone. Measured
 *     at kFloor 1000: 100% precision, exact hashes, ZERO firings on a
 *     listings-only control — the signature of 1, 2 and 3. Cost to him: ~240
 *     published-then-revoked accounts per build registry-wide.
 *
 * Each fix was correct for the case it was aimed at and left a regime the guard
 * did not simulate. A CONSTANT does not have regimes — but a constant COUNT is
 * a claim about a number and the property needed is a claim about a SET, which
 * is what defect 4 cost. `tests/invariants/registry-churn.test.ts` sweeps the
 * reachable parameter space — every (a, b) mix, both directions of the rung
 * ladder, width changes, mass removal, an empty registry, an attacker who times
 * the crossing, an attacker who owns the departures, and a store that dies
 * part-way through a write — and asserts two things everywhere: departures ==
 * churn, and at least `churn - realChurn` of them are decoys.
 *
 * WHAT THE PADDING DOES AND DOES NOT HIDE — read this before changing it.
 *
 *  (a) Against the SERVER, decoys are worth nothing. The server holds the seed
 *      and can regenerate every decoy, so a bucket fetch narrows a fan's query
 *      to that bucket's REAL entries. The k that matters for "the server never
 *      learns which subject you checked" is therefore realCount / 16^width,
 *      not `padTarget`. `chooseWidth` is the only lever on it.
 *  (b) Against an OUTSIDER reading bucket files, decoys hide which entries are
 *      real ONLY against an attacker who cannot guess candidate subjects. The
 *      subject hash is UNKEYED and the canonicalizer is public — it has to be,
 *      because the client hashes locally and the URL never leaves the device.
 *      So anyone can hash a handle they already know, or a dictionary of
 *      handles, and test membership. Padding does not change that and no
 *      parameter here can. Only a keyed hash / OPRF would, at the cost of the
 *      local-check property. Say "cannot be enumerated", never "cannot be
 *      recovered".
 *  (c) The served total object volume bounds registry size to a power-of-two
 *      band (`padTargetFor`), not to a few percent.
 *  (d) Against a LONG-LIVED archiver, two residuals survive and are written up
 *      with measured numbers in docs/REGISTRY-PRIVACY-CLAIMS.md: the AGE of a
 *      departing hash still leans towards "that was a real entry" (bounded by
 *      the churn quota, not closed by it), and a constant churn turns the decoy
 *      pool over whether or not anything happened — so an intersection across a
 *      long enough gap converges on the real set even in an idle registry, not
 *      only across a registry doubling. Both are measured and published.
 */

/** Refuses to publish. Fail closed: a build that cannot be made privacy-safe
 * is not published at all, and the previously published buckets keep serving. */
export class RegistryBuildError extends Error {
  override name = "RegistryBuildError";
}

/**
 * C-058 (docket #2): `buildRegistry` had exactly TWO live call sites —
 * `scheduled()`'s six-hourly cron and `POST /ops/registry/rebuild` — with NO
 * mutual exclusion between them anywhere in src/. Both read `manifest.json`
 * + the ladder object, compute `seq = ladder.seqFloor + 1` and write buckets
 * keyed only by `buckets/${width}/${prefix}.json` (not seq-scoped) before
 * writing `manifest.json` last. Two live runs that race — an operator firing
 * the lever while the cron tick is already mid-build, or an impatient
 * double-click of it — can leave a torn state: the SAME seq committed twice,
 * and bucket bytes that are a genuine mix of both runs' writes (measured:
 * `c058-wedge.test.ts`, S-026-batch/REBUILD). The NEXT build's own
 * tear-detection guard below ("DOES THE MANIFEST AND THE BUCKETS DESCRIBE
 * THE SAME BUILD?") correctly refuses that state — but nothing ever writes a
 * fresh ladder/manifest to leave it, since every later attempt throws before
 * reaching its own writes. So an interleaved write does not degrade one
 * build, it WEDGES EVERY BUILD AFTER IT, forever.
 *
 * THE FIX IS THE SAME SHAPE `cron-lease.ts` ALREADY GIVES the cron sweeps
 * (`alerts.ts`'s delivery lease, `scheduled()`'s own tick lease): a
 * single-holder, D1-backed lease with a bounded TTL, acquired here so BOTH
 * callers are covered without either one having to remember to ask for it.
 * REFUSED, not queued — matching `acquireCronLease`'s own contract ("a
 * caller that gets null must do nothing, not proceed unsynchronised") and
 * `alerts.ts`'s "another run holds the lease, this tick did nothing. Not an
 * error." A second caller refusing outright, loudly, is the record: the
 * cron's existing `catch` around this call already logs
 * "REGISTRY REBUILD FAILED" with the thrown message, and the operator route
 * reports and audits a distinct `rebuild_in_progress` rather than folding it
 * into a generic failure.
 *
 * A SEPARATE lease name from `scheduled`'s own tick lease and from
 * `alerts.ts`'s `DELIVERY_LEASE_TTL_MS`: this one lease must be held for
 * exactly the duration of ONE registry build (bucket writes to R2), which is
 * a different real-world duration than either of those sweeps, and sharing a
 * name would make an unrelated sweep's TTL govern registry publication.
 */
/**
 * THE LEASE IS GONE, AND WHAT REPLACED IT (ADR-0012 §2, I3-0246).
 *
 * A single-holder D1 lease with a 10-minute TTL bounded how long two builders
 * could overlap; it did not stop the second one from doing damage, because it
 * was taken once and never re-checked while the body wrote every bucket over
 * the live keys. A build that outran its TTL kept writing.
 *
 * The fence is now structural rather than temporal. Each build owns a
 * `registry_builds` row and an EPOCH; every step commits its cursor with
 * `UPDATE … WHERE seq = ? AND epoch = ?`, so a builder that has been taken over
 * affects no row and stops. Everything it wrote before that went under its own
 * `gen/<seq>/` prefix, which the pointer does not name, and the pointer itself
 * is flipped by compare-and-swap against the etag read at plan time. A
 * predecessor can therefore be arbitrarily slow, arbitrarily stale, and still
 * mutate nothing a reader can reach — which is the property the lease was
 * bought for and never actually delivered.
 *
 * `RegistryBuildBusyError` survives it: two callers still must not both START a
 * build, and the single-statement insert in `registry-gen.ts` is what decides
 * which one does.
 */

/** Thrown when another build is already open.
 * Distinct from `RegistryBuildError` so a caller can tell "refused because a
 * concurrent build is in flight" (benign, expected, retry later) apart from
 * every other build refusal (a data problem this build needs a human for). */
export class RegistryBuildBusyError extends RegistryBuildError {
  override name = "RegistryBuildBusyError";
}

/**
 * Bucket file format: `{v, entries:[{h,c,p?,r?}], z}`.
 *
 * STILL VERSION 1, deliberately. `p` (per-account confirmation) and `z` (the
 * byte-length filler) are both ADDITIVE and optional to a reader: every field a
 * v1 parser required is still there, in the same shape, and every client this
 * repo ships reads `entries` and ignores keys it does not know. Bumping the
 * version is the one thing versions are for — telling a reader it may NOT
 * proceed: since D-0708 every client gates on `v` (`versionAccepted` in
 * packages/registry-proof/src/registry-proof.ts), so a bump with no `min_v` retires
 * every reader that does not list the new version.
 *
 * WHICH DIRECTION THE NEW FIELD FAILS. `p` is present only where an account's
 * own control proof is CURRENT, so a reader that has never heard of it — an
 * older extension, a bucket written before this change — sees no `p` anywhere
 * and says the weaker, passport-level sentence about every entry. Absence
 * under-claims; it can never over-claim. That is the only direction an optional
 * trust field is allowed to fail.
 */
export const BUCKET_FORMAT_VERSION = 1;
/**
 * Manifest format. VERSION 2 adds `pad`, `churn`, `seq` and `frontier`, and
 * they are load-bearing rather than informational: the manifest is now the
 * builder's cross-build state, and a version-1 manifest cannot be carried
 * forward because nothing in it says how far the decoy stream has been spent.
 * A build that finds one REFUSES (see `loadPrevious`) — see DEPLOY.md 3d for
 * the one-time cutover.
 *
 * Every field is a pure function of the PUBLISHED history (the widths, the
 * rungs and the churn quota of every build so far), so an observer who could
 * read them would learn nothing they could not already compute from the bucket
 * files. That is the test any new manifest field has to pass — and it is a
 * property of the VALUES, not of the route surface.
 *
 * `built_at` IS THE ONE EXCEPTION, and it is served (Y6). It is a wall clock,
 * so it is NOT a pure function of published history — it is the moment the
 * build ran. It passes the test that actually matters, which is the weaker and
 * correct one: an observer learns nothing they could not already compute from
 * the bucket files. Every bucket is rewritten on every build and sheds exactly
 * `churnFor(padTarget)` hashes whether or not anything happened, so diffing the
 * buckets already yields a build census; `built_at` only sharpens its
 * resolution from "inside the last edge-cache window" to an instant, and no
 * subject-scale event is measured in seconds. `manifest.json` is written LAST
 * (after the ladder and every bucket), so a fresh clock can never advertise
 * buckets that are not yet at origin.
 *
 * The rule for the NEXT field is therefore unchanged and now has a stated
 * exception rather than a silent one: a field that is a pure function of
 * published history may be served; a field that is not must be argued for
 * against the bucket bytes, in writing, here.
 */
export const MANIFEST_FORMAT_VERSION = 3;
/**
 * D-0708 — the oldest READER format that can still read what this builder
 * writes, served as the manifest's `min_v` beside `v`. Clients accept a
 * manifest iff a version they understand lies in `[min_v, v]`
 * (`versionAccepted`), so: bump `v` and keep this for an additive change; raise
 * this to retire the readers below it. It never exceeds `MANIFEST_FORMAT_VERSION`.
 * A code constant, identical for every build, so it passes the manifest-field
 * test above with nothing to argue (docs/REGISTRY-PRIVACY-CLAIMS.md).
 */
export const MANIFEST_MIN_READER_VERSION = 3;
/** The format this builder READS as the generation to cut over from: v2's
 * objects live at unversioned keys (`buckets/<w>/<p>.json`). One v3 build later
 * they are unreachable, and the GC deletes them after one edge-cache TTL. */
const LEGACY_MANIFEST_VERSION = 2;
export const K_FLOOR = 1000;
/** ADR-0012: aligned to the clients' own `MAX_CLIENT_WIDTH` — a width the
 * shipped apps cannot address is a width no fan can check against. */
export const MAX_WIDTH = 4;

/**
 * The largest rung a bucket file may reach before the WIDTH grows instead
 * (ADR-0012's earlier-widening ruling). At 4,000 entries a bucket body is about
 * 356 KB, under the Android client's 512 KiB response cap; the rung above it is
 * not. Growing the width divides the same subjects over sixteen times as many
 * files instead of doubling one file, and it is driven by the listing
 * high-water exactly as the ladder is, so no removal can trigger or delay it.
 */
export const PAD_ENTRY_CEILING = 4000;

/**
 * Pad ladder ratio. `padTarget` is the smallest K_FLOOR * 2^j whose CAPACITY
 * (see `capacityFor`) holds the per-bucket listing high-water.
 *
 * COARSE ON PURPOSE. The previous rule rounded the fullest bucket up to the
 * nearest 250, which bounded nothing once buckets were large: measured, 20,000
 * real subjects published 24,000 entries (1.20x) and 50,000 published 52,000
 * (1.04x), so the public object volume *tracked* the private count. On a
 * doubling ladder the volume says only which power-of-two band the registry is
 * in — about one bit.
 */
export const PAD_LADDER_RATIO = 2;

/**
 * A bucket may fill at most `1 - 1/CAPACITY_SHARE` of `padTarget` with
 * non-decoy entries. The remainder is the RESERVED DECOY FLOOR, and it is what
 * makes a constant departure count possible at all: a bucket that has nothing
 * real to shed must still shed `churn` hashes, and those have to come from
 * somewhere. Reserving them out of the rung means the guarantee holds in EVERY
 * history rather than in the ones that happened to leave decoys behind.
 *
 * It also closes the old "residual at the top of a rung": rotation used to
 * collapse to nothing in a bucket down to its last three decoys, so a removal
 * there had a crowd of one. The floor is now padTarget/8 — 125 decoys at the
 * shipped kFloor — everywhere in the rung.
 *
 * The cost is padding volume: the ratio of published entries to real ones is
 * between 1.14x and 2.29x, about 1.58x averaged over a rung, against 1.44x for
 * an unreserved ladder.
 */
const CAPACITY_SHARE = 8;

/**
 * The churn quota: `padTarget / CHURN_SHARE` hashes leave every bucket every
 * build, floored at CHURN_FLOOR so a toy registry still has a crowd.
 *
 * IT IS THE ONE DIAL, AND IT SETS FIVE THINGS AT ONCE — pick it by measurement,
 * not by taste (the numbers are in docs/REGISTRY-PRIVACY-CLAIMS.md):
 *  - the total size of the crowd a real departure hides in (bigger is better),
 *  - the GUARANTEED-DECOY half of that crowd, which is the only part an
 *    attacker who can mint departures cannot consume (bigger is better) — this
 *    is the fifth thing, and it is new; see DECOY_CHURN_SHARE,
 *  - the steady-state listing throughput within a rung (bigger is better),
 *  - the delisting SLA under a burst (bigger is better),
 *  - the rate at which the decoy pool turns over when NOTHING is happening
 *    (bigger is worse: it is what an archiver's intersection attack eats).
 *
 * The first four want it large and the fifth wants it small. It was 1/64 while
 * the whole quota paid removals. Splitting the quota in half moved the knee to
 * 1/32, chosen so that the REAL half is exactly what the whole quota used to be
 * — `realChurnFor(pad) == the old churnFor(pad)` at every production rung — so
 * the published delisting SLA does not move at all and the second half is pure
 * gain. At the shipped kFloor of 1000: `floor(1000/32)` = 31 departures per
 * bucket per build, of which 15 may be real (240 registry-wide, 960 a day,
 * unchanged) and 16 are always decoys, against a reserved decoy floor of 125
 * per bucket. (This comment used to say 16 departures and 256 registry-wide
 * against a 1/64 quota; the code said 15 and 240. The doc was right.)
 *
 * WHAT THE WIDENING COSTS, measured on the doc's own series: the archiver's
 * long-horizon intersection. Two consecutive builds intersect to 96.90% of the
 * file instead of 98.50%, and over twenty builds the "still here, so real" rule
 * runs at 60.4% against a base rate of 41.3% instead of 46.9%. That is a
 * degradation to a bound the doc already declares unfixable and
 * horizon-limited. What it buys is the property the removal feature actually
 * rests on: the composed elimination + departure-age attack falls from 68.5%
 * (and 100% with exact hash recovery when the attacker sizes his wave to the
 * whole quota) to 19.7% — below the 29.6% that is published today as the
 * age residual with no attacker present at all.
 *
 * SCALING WITH `padTarget` RATHER THAN A FLAT NUMBER is deliberate: it keeps the
 * crowd, the throughput and the pool turnover a fixed fraction of the rung, so
 * the measured residuals do not drift as the registry grows. And it keeps the
 * quota a function of a PUBLISHED number, which is the property that matters —
 * anything the quota depends on is something an observer already has.
 *
 * The one place widening it shows up outside the crowd is the pad ladder:
 * `padTargetFor` condition 2 subtracts the quota from the rung, so a bigger
 * quota steps the ladder marginally earlier in the SMALL rungs where the
 * sqrt(mean) headroom is proportionally large. Measured: rung 2,000 now serves
 * a per-bucket mean of 817..1,716 instead of 831..1,745, so its padding ratio
 * band is 1.17x..2.45x (mean 1.65x) instead of 1.15x..2.41x (1.62x). From rung
 * 16,000 up the bands are identical — 1.14x..2.29x, 1.58x averaged over a rung.
 */
const CHURN_SHARE = 32;
const CHURN_FLOOR = 4;

/**
 * HOW THE QUOTA IS SPLIT, AND WHY IT HAS TO BE — the fourth defect.
 *
 * `churn` fixes the SIZE of the departure crowd. It says nothing about WHO is
 * in it, and until this constant existed the answer was "whoever an attacker
 * chose". `leaving` is every hash that stopped being real, which is not only
 * removal requests: a revoked account, a retracted passport and a lapsed proof
 * all land there, and every one of them is reachable by anybody who can get a
 * listing admitted and then un-list it. So an attacker planted `churn` handles
 * per bucket (handles are ground against the public canonicalizer offline),
 * let them publish, and revoked them. Suppressions sort ahead of voluntary
 * leavers, so a victim's removal still took its slot and his hashes took every
 * other one — `retire` fell to 0, NOT ONE DECOY DEPARTED, and subtracting the
 * hashes he had minted left exactly the delisted subject. Measured on the code
 * this constant fixes, at kFloor 1000: 100% precision, exact hashes, and ZERO
 * firings on a listings-only control — the signature of breaks 1, 2 and 3.
 *
 * The lesson is that "departures == churn" is a claim about a COUNT and the
 * property that was needed is a claim about a CROWD:
 *
 *   of the `churn` hashes a bucket sheds, at least `decoyChurnFor(pad)` are
 *   decoys retired under the secret seed — hashes no external party can cause
 *   to depart, predict, or recognise.
 *
 * So the quota is split. Real departures are capped at `realChurnFor(pad)` and
 * the balance is ALWAYS paid in decoys, which makes the decoy floor
 * unconsumable by construction rather than by luck:
 *
 *   dropped = min(realChurn, |leaving|) <= realChurn
 *   retire  = churn - dropped          >= churn - realChurn = decoyChurn
 *
 * WHAT IT COSTS: the delisting SLA, and nothing else. The total departure count
 * is unchanged, so every measured residual that depends on the RATE — the
 * departure-age signal, the archiver's intersection, the decoy pool's turnover
 * — is unchanged too. What changes is how fast a burst of removals drains:
 * `realChurn` per bucket per build instead of `churn`. That is a real cost, it
 * is published, and it is the one the doc's SLA section now states.
 *
 * 1/2 is the split, i.e. the crowd is at least half decoys. It is a floor, not
 * a target: on an ordinary build almost nothing is leaving, so `retire` is
 * almost always the whole quota. It is paired with CHURN_SHARE = 32 so that
 * `realChurnFor(pad)` comes out at EXACTLY the old whole-quota `padTarget/64`
 * at every production rung — the delisting SLA is unchanged by this fix, and
 * the guaranteed decoy half is bought with padding turnover instead. See
 * docs/REGISTRY-PRIVACY-CLAIMS.md for the measured sweep behind both numbers.
 */
const DECOY_CHURN_SHARE = 2;

/**
 * Decoy class mix is quantised to this many steps of the global real class
 * distribution. Quantisation is what keeps a decoy's CLASS stable: an
 * unquantised mix moves on every insertion, and a decoy whose class flipped
 * between two builds would drop out of an (h, c) intersection. 1/1024 is a
 * deliberate compromise — coarse enough that ordinary growth does not move a
 * boundary, fine enough that a class holding 0.1% of real entries still gets
 * decoys. A coarser quantum (1/32 was tried) rounds a small class to ZERO
 * decoys, and a class no decoy can take is a hard marker: every entry with it
 * is real.
 */
const CLASS_QUANTUM = 1024;

/**
 * Bucket bodies are padded up to a multiple of this many BYTES (`padBodies`).
 *
 * The quantum exists so the one number that IS shared by every bucket — the
 * padded length — is coarse. Per-bucket length is what fingerprints a prefix and
 * it is now constant; the global length still moves with the registry's overall
 * class mix, and rounding it to a 512-byte step means it moves in visible jumps
 * rather than tracking the mix byte for byte. It costs at most 512 bytes per
 * bucket against bodies that run ~90 KB at the shipped k-floor.
 *
 * Not a power-of-two LADDER like `padTargetFor`, because there is nothing to
 * hide here: an observer reads the padded length off any bucket they like. The
 * only property that matters is that they all read the SAME one.
 */
const BUCKET_BYTE_QUANTUM = 512;

/** Domain separation for the decoy stream. Changing this string re-mints every
 * decoy in the registry, which is exactly the intersection attack this design
 * closes — treat it as frozen, like a preimage format. */
const DECOY_LABEL = "dhwaja/registry-decoy/v1";

/** Minimum accepted REGISTRY_DECOY_SEED length (hex/base64 chars). 32 chars is
 * 128 bits at the worst plausible alphabet; below that a seed is guessable and
 * every decoy with it. */
export const DECOY_SEED_MIN_LEN = 32;
const SEED_PLACEHOLDER_RE = /TODO|CHANGE[-_ ]?ME|PLACEHOLDER|EXAMPLE|^0+$|^(.)\1*$/i;

/** A seed is usable only if it is long enough and is not an obvious stand-in.
 * Absent or placeholder => the build refuses; it never falls back to random. */
export function isUsableDecoySeed(v: unknown): v is string {
  return typeof v === "string" && v.length >= DECOY_SEED_MIN_LEN && v.trim() !== "" && !SEED_PLACEHOLDER_RE.test(v);
}

/**
 * REGISTRY_DECOY_SEED, or null when it is absent / a placeholder / too short.
 * Same shape as `checkpointKeysFromEnv`: unreadable config disables the
 * feature, never half-enables it. The caller must SKIP the rebuild on null —
 * `buildRegistry` refuses anyway, but skipping keeps the cron's other sweeps.
 */
export function registryDecoySeedFromEnv(env: Record<string, unknown>): string | null {
  const v = settingValue(env.REGISTRY_DECOY_SEED); // I63 C2 (I62-CC-15, I63-Q4): the trimmed seed, then the floor and placeholders
  return isUsableDecoySeed(v) ? v : null;
}

/** Creator trust classes — the only classes decoys may take. */
export type EntryClass = "verified" | "claimed" | "unclaimed";
/** Everything a bucket entry's class may be, including the accusatory
 * "warning" (impersonation) and the soft, guardrailed "caution" (unproven /
 * machine-suspected-lookalike). DECOYS TAKE BOTH (D-0707, QUEUE #156 PS-042): a
 * class no decoy takes labels every entry that has it as real. A decoy is still
 * never presented to a fan as reported or cautioned: every client shows an entry
 * only on FULL-hash equality with the subject it checked, a decoy hash is a
 * seeded digest no canonical subject hashes to, and `composeBucket` refuses any
 * decoy hash this build knows to be real. Precedence at a real hash: warning >
 * caution > creator-class. */
export type ServedClass = EntryClass | "warning" | "caution";

export interface RegistryEntry {
  h: string; // sha256 of canonical subject
  c: ServedClass;
  /**
   * PER-ACCOUNT CONFIRMATION, present (as `1`) only on a `claimed` or `verified`
   * entry whose OWN control proof is current for the passport that lists it.
   *
   * WHY IT IS ON THE WIRE AT ALL. `claim_state` is a fact about a PASSPORT and
   * every client rendered it as a fact about the ACCOUNT the fan just checked:
   * "This creator controls this account" (web), "Claimed on Dhwaja"
   * (extension), "Account control and identity confirmed by Dhwaja" (verified,
   * everywhere). With `servingPerAccountStanding` off — the default — an
   * operator/OSINT-seeded handle on a claimed passport is served `claimed`, and
   * nobody has ever confirmed that handle is theirs. The passport page has said
   * so for as long as `PublicAccount.confirmed` has existed ("! not confirmed",
   * and passport-view.ts states outright that whether the page tells the truth
   * about its own unconfirmed rows is not an operator preference); the CHECK
   * path had no way to say it, because the entry carried nothing to say it with.
   *
   * ONE BYTE OF PAYLOAD, on purpose: the entry is multiplied by `padTarget` and
   * again by 16^width, so this is `,"p":1` and not a word. It is a presence
   * flag rather than `p:0/1` for the same reason.
   *
   * DECOYS CARRY IT TOO, at the per-class rate of the real set (`confirmRates`,
   * `decoyProven`) and pinned at mint like the class. A flag only real entries
   * could carry would be a hard marker — every entry with it real, and (worse,
   * because it eliminates rather than confirms) every claimed/verified entry
   * WITHOUT it a smaller haystack. It is never emitted on `unclaimed`, real or
   * decoy: an unclaimed passport has proved nothing by definition, so on that
   * class the flag has no meaning and a decoy carrying it would be recognisable
   * as a decoy.
   */
  p?: 1;
  /**
   * WHAT THE SUBJECT IS — `"a"` (AI character) or `"b"` (brand character).
   * Absent means a person, which is what every entry meant before characters
   * existed.
   *
   * WHY IT IS ON THE WIRE. A character carries the same verdicts a person does
   * (slice 14: kind is an AXIS, not a verdict), and slice 15 disclosed it on the
   * passport page, the crawlable page, the PSA, the share card and the unfurl —
   * every surface EXCEPT the one where a fan actually checks. The extension and
   * both mobile apps answered "Verified on Dhwaja" about an AI persona and said
   * nothing else, which is the confusion the whole disclosure exists to remove,
   * on the surface most fans use.
   *
   * A ONE-CHARACTER VALUE — `,"k":"a"` rather than `,"k":"ai_character"` — for
   * the reason `p:1` is a presence flag: the entry is multiplied by `padTarget`
   * (at least a thousand) and again by `16^width`, so the wire spends a byte
   * where the model spends a word.
   *
   * ON EVERY CREATOR CLASS, INCLUDING `unclaimed` — and here it diverges from `p`
   * deliberately. `p` skips unclaimed because a flag about proof is meaningless
   * where nothing was proved. A KIND is meaningful on every class, and unclaimed
   * is where a character spends its early life (`subject_kind_declared` requires
   * it), so skipping it would blind the check exactly where an AI persona is most
   * likely to be taken for a person. Never on `warning`/`caution`: those entries
   * are keyed on the IMPOSTOR's hash and say nothing about a creator.
   *
   * DECOYS CARRY IT TOO (`kindRatesFor`, `decoyKind`), at the real per-class rate.
   * The rule is `p`'s, and it runs in BOTH directions: a value only real entries
   * could take is a hard marker, and a value only DECOYS could take is the worse
   * one, because elimination narrows a haystack faster than confirmation.
   *
   * STILL NOT A VERDICT on any client. No new `TrustStatus` member, no verdict
   * hue, no verdict icon — a verified character is exactly as verified as a
   * verified person.
   */
  k?: "a" | "b";
  /** Find-the-real-one: on a "warning" OR "caution" entry only, the VERIFIED
   * creator this subject targets/resembles — enough for every client to render a
   * redirect with zero extra fetches (an after-match fetch would tell the server
   * someone just matched this specific entry, which the prefix design exists to
   * prevent). p = passport id, n = display name (clamped, markup-stripped; clients
   * still escape). Never present on creator entries. DECOY warning and caution
   * entries carry one too, at the real per-class rate (`pointerRatesFrom`,
   * `decoyPointer`), or "has a pointer" would mark the real ones; a decoy's `p`
   * is never a real passport id (`stepStage` vetoes any that is) and no client
   * renders it, because no client ever matches a decoy hash (see `ServedClass`). */
  r?: { p: string; n: string };
  /**
   * FOUNDING RANK (I3-0526, D-1078(13), D-1082(2), I32-FX1-VERIFIED): on a `verified` entry only, the Founding
   * Creator number (1..FOUNDING_CAPACITY) of the passport that lists it, so a client can say 'Founding Creator #N'
   * where it says Verified. Read from `founding` with `foundingRank`'s gate (verified, not retracted), and never on a
   * grant that has lapsed at the build (realEntriesFrom publishes that one as `claimed`). Absent under-claims, like
   * `p` and `k`: a bucket written before the field existed reads as "not founding".
   * DECOYS CARRY IT TOO, at the real share of verified entries and drawn from the ranks real entries carry, weighted
   * the same (`foundingRatesFrom`, `decoyFounding`), keyed on the decoy HASH so a survivor minted before the field
   * existed gets the same answer on every later build, exactly as `w` is. A rank only decoys carried would mark them.
   */
  f?: number;
  /**
   * WARNING KIND (I3-0122, D-0992(9), D-0981(4)): on a `warning` entry only, the one-character code of the
   * warning's kind (`@dhwaja/schema` WARNING_KIND_WORDINGS: "i" impersonation, "f" confirmed fraud), which every
   * client maps to that kind's wording through `warningWordingFor`. Never on another class. DECOY warnings carry one
   * too, drawn at the real kind mix (`warningKindRateFrom`, `decoyWarningKind`) — a warning with no code, or a
   * code only decoys take, would mark the entry. "f" is drawn only while `serve_confirmed_fraud` is on, for
   * `caution`'s reason: a class the operator switched off has no real member to hide.
   */
  w?: WarningKindCode;
}

/**
 * The served manifest. `width` is what every client reads; the rest is the
 * builder's own cross-build state, published because it is derivable from the
 * published history anyway (see MANIFEST_FORMAT_VERSION).
 */
export interface BucketManifest {
  v: number;
  /** D-0708: see `MANIFEST_MIN_READER_VERSION`. Absent on manifests written before it. */
  min_v?: number;
  width: number;
  /**
   * WHERE THIS GENERATION'S OBJECTS ARE: the exact string `gen/<seq>/`
   * (ADR-0012 §1). The manifest is the ONLY mutable served object; buckets and
   * proofs are written once, under a prefix nothing points at until this
   * pointer names it, and are never overwritten. A reader resolves
   * `${prefix}buckets/<w>/<p>.json` when `v` is 3 and validates the prefix
   * against `^gen/\d+/$` and against `seq`, so a pointer cannot send it at
   * another build's bytes.
   *
   * It is a pure function of the published history (the seq is already here), so
   * it passes the test every manifest field has to pass.
   */
  prefix?: string;
  built_at: string;
  buckets: number;
  /** Uniform per-bucket entry count. Also readable by counting any bucket. */
  pad: number;
  /** The k-anonymity floor this build ran with. A documented constant, and the
   * only thing that says what `pad` MEANS — a rung is a power of two times it. */
  kfloor: number;
  /** The departure quota this rung sets: the number of published hashes every
   * bucket sheds on the NEXT build, whatever happens between now and then. A
   * pure function of `pad`, published so that the constant an observer can
   * already measure is also the constant we state. */
  churn: number;
  /** Monotone build counter. Drives the retirement stream, so it must not
   * rewind: a rolled-back copy would replay a build's retirement picks against a
   * different live set. Held here AND in the private ladder object, higher wins. */
  seq: number;
  /** Next unspent index of the global decoy stream. Advances by a block whose
   * size is a function of the published widths and rungs, so it is itself a
   * function of the published history. Rewinding it re-mints decoys that have
   * already been retired, which is the returning-hash channel — so it is held
   * here AND in the private ladder object, higher wins, AND the pair is checked
   * against `digest` below. */
  frontier: number;
  /**
   * A digest of the bucket bytes this manifest describes — the thing that says
   * "these counters belong to THESE buckets".
   *
   * "Higher wins" across the two counter copies defends against ONE of them
   * being rolled back. It cannot defend against both, and both is reachable
   * without an adversary: restore `manifest.json` and the ladder object from a
   * backup, leave the buckets alone, and the next build re-mints from a block
   * it has already spent — a decoy retired in pass 1 comes straight back in
   * pass 3 and never departs, so the bucket sheds FEWER than `churn`, which
   * proves it had fewer than `churn` real removals. With the digest, a manifest
   * that does not describe the buckets on the wire is detected and the build
   * REFUSES instead (unless the ladder object carries a strictly newer `seq`,
   * which is the signature of a torn write rather than a rollback — see
   * `buildRegistry`).
   *
   * It is a pure function of the published bytes, so it passes the test every
   * manifest field has to pass. OPTIONAL only so that the first build after
   * this change, which reads a manifest written without it, is not a hard stop;
   * every manifest this builder writes carries it.
   */
  digest?: string;
  /**
   * The Merkle root of this build's published bucket bytes — the one value a
   * client can check a SINGLE bucket against without downloading the rest.
   *
   * It passes the test every manifest field has to pass, twice over: it is a
   * pure function of the published bytes (like `digest`, which it does NOT
   * replace — `digest` is a chain only the builder reads, and it keeps doing
   * its rollback job unchanged), and it says nothing about how many of a
   * build's entries were real. OPTIONAL so a manifest written before this
   * change is not a hard stop.
   */
  root?: string;
  /**
   * The signature over `{width, seq, built_at, root}`, under a key trust-core
   * does not hold. ABSENT is an honest state and is published as such: signing
   * off, or the signer unreachable when this build ran. It is never dressed up
   * as signed. See `registry-sign.ts`.
   */
  sig?: RegistrySignature;
}

/**
 * The largest width whose buckets still hold >= kFloor entries of the LISTING
 * HIGH-WATER, capped at MAX_WIDTH, and never below 1.
 *
 * IT IS FED THE HIGH-WATER, NOT THE CURRENT REAL COUNT, and that is a privacy
 * decision, not an accounting one. A width change re-partitions every bucket
 * and is plainly visible; if removals could delay one, an attacker who can hold
 * the registry at the width threshold reads "the partition did not change this
 * build" as "somebody was delisted". The high-water only ever grows, so the
 * moment a width changes is a function of listings alone.
 *
 * WHAT IT COSTS. `hw` counts every listing ever admitted, so it exceeds the
 * live real count by the total ever removed, and the true anonymity set is the
 * live count divided by 16^width. A registry that is delisted wholesale
 * therefore keeps a width sized for the registry it used to be, and its real
 * per-bucket count sits below kFloor until growth catches up. That degrades
 * gracefully (it takes removals on the order of the whole registry to matter)
 * and the alternative — narrowing the partition when people leave — publishes a
 * registry-wide removal signal. Stated in docs/REGISTRY-PRIVACY-CLAIMS.md.
 *
 * WHAT IT DOES NOT PROMISE. Width 1 is the protocol floor (the wire format is
 * >= 1 hex char, and every client computes the prefix locally), so below
 * 16 * kFloor real subjects the floor is UNREACHABLE and the true anonymity set
 * is realCount / 16 — about 31 real entries at 500 subjects, 313 at 5,000. The
 * padding does not fill that gap, because the server can regenerate its own
 * decoys. State the floor as conditional: "at least kFloor real subjects share
 * your prefix once the registry passes 16 * kFloor; below that, all of it
 * divided by 16." Getting the unconditional claim would take a width-0 (single
 * bucket) format, which is a wire change across four clients, not a parameter.
 */
export function chooseWidth(listingHighWater: number, kFloor: number = K_FLOOR, padCeiling: number = PAD_ENTRY_CEILING): number {
  let w = 1;
  while (w < MAX_WIDTH && listingHighWater / 16 ** (w + 1) >= kFloor) w++;
  // AND GROW RATHER THAN LET A BUCKET FILE PASS THE RUNG CEILING (ADR-0012).
  // The rung this width would need is a function of the listing high-water
  // alone, exactly like the loop above, so this arm moves the width on the same
  // kind of number: nothing a removal can touch. It is monotone for the same
  // reason — `padTargetFor` is non-decreasing in the high-water.
  while (w < MAX_WIDTH && padTargetFor(listingHighWater / 16 ** w, kFloor, w) > Math.max(padCeiling, kFloor)) w++;
  return w;
}

/**
 * How many NON-DECOY entries a bucket of this rung may publish. The rest of the
 * rung is the reserved decoy floor that pays for the churn quota — see
 * CAPACITY_SHARE.
 */
export function capacityFor(padTarget: number): number {
  return padTarget - Math.ceil(padTarget / CAPACITY_SHARE);
}

/**
 * OR-01 (ADR-0012 §6): the part of a bucket's non-decoy capacity RESERVED for
 * safety entries — warnings and cautions — and therefore denied to creator
 * listings. `max(4, ceil(pad/64))` is 16 at the shipped k-floor: about 1.6% of a
 * bucket's creator capacity.
 *
 * IT IS A PUBLIC FUNCTION OF `pad`, so it is no new channel: an observer who can
 * read the rung off any bucket can compute it. The alternative designs both
 * leak. EVICTING an incumbent creator to make room has to take a real-churn slot
 * to keep departures constant, and a new warning arriving in the same build as a
 * departure then bounds that bucket's real occupancy at "at capacity" — one bit
 * per bucket per build of exactly what the padding exists to hide, plus a false
 * "not listed" for a real creator. STEPPING THE RUNG on every blocked safety
 * entry doubles the served volume on a private per-bucket count. So: reserve the
 * lane, keep a blocked safety entry `pending`, and step the rung only when one
 * is still pending after a committed build.
 */
const SAFETY_LANE_SHARE = 64;
const SAFETY_LANE_FLOOR = 4;

export function safetyReserveFor(padTarget: number): number {
  return Math.max(SAFETY_LANE_FLOOR, Math.ceil(padTarget / SAFETY_LANE_SHARE));
}

/** How many non-decoy entries a bucket may publish for CREATOR listings: the
 * rung's capacity less the reserved safety lane. Warnings and cautions may use
 * the full `capacityFor`. */
export function creatorCapacityFor(padTarget: number): number {
  return Math.max(0, capacityFor(padTarget) - safetyReserveFor(padTarget));
}

/** Departures per bucket per build at this rung. See CHURN_SHARE.
 *
 * CLAMPED TO THE RESERVED DECOY FLOOR, which is what makes the quota feasible
 * at EVERY rung rather than only at the ones production uses: a bucket cannot
 * shed more hashes than it is guaranteed to be able to shed. The clamp only
 * ever binds on a toy rung (`CHURN_FLOOR` of 4 needs a rung of at least 32);
 * at the shipped floor of 1000 the reserve is 125 against a quota of 15. */
export function churnFor(padTarget: number): number {
  return Math.min(padTarget - capacityFor(padTarget), Math.max(CHURN_FLOOR, Math.floor(padTarget / CHURN_SHARE)));
}

/**
 * The GUARANTEED number of decoy retirements in a bucket's `churn` departures:
 * the part of the quota no listing, no revocation and no removal request can
 * consume. See DECOY_CHURN_SHARE.
 *
 * It is a floor on the crowd, so it must never be the whole quota — a rung
 * whose quota cannot pay at least one real departure would never delist
 * anybody. `churn` is >= CHURN_FLOOR (4) at every rung production can reach;
 * the `c <= 1` arm exists only for absurd rungs a test can construct.
 */
export function decoyChurnFor(padTarget: number): number {
  const c = churnFor(padTarget);
  if (c <= 1) return 0;
  return Math.min(c - 1, Math.max(1, Math.ceil(c / DECOY_CHURN_SHARE)));
}

/** The cap on REAL departures per bucket per build — the delisting SLA's rate.
 * The rest of `churn` is `decoyChurnFor`. */
export function realChurnFor(padTarget: number): number {
  return churnFor(padTarget) - decoyChurnFor(padTarget);
}

/**
 * The uniform per-bucket entry count: the smallest rung of the doubling ladder
 * (kFloor, 2*kFloor, 4*kFloor, ...) that satisfies BOTH conditions below for a
 * registry whose per-bucket listing high-water AVERAGES `meanPerBucket`.
 *
 *  1. `capacityFor(rung) >= meanPerBucket` — the average bucket can admit its
 *     share of the listings and still hold the reserved decoy floor.
 *  2. `rung - churnFor(rung) >= meanPerBucket + z * sqrt(meanPerBucket)` — and
 *     so can the FULLEST one.
 *
 * CONDITION 2 IS THE PRIVACY-CRITICAL ONE and it is why the width is a
 * parameter. The ladder is driven by a registry-wide count divided by the
 * number of buckets, which is a mean; the constraint that actually has to hold
 * is about the fullest bucket, because a bucket with fewer decoys than its
 * churn quota cannot publish a constant departure count. Subject hashes spread
 * multinomially, so the fullest of 16^width buckets sits about
 * `sqrt(2 ln 16^width) + 3` standard deviations above the mean. Size the rung
 * for that here, out of PUBLIC numbers, and the fallback in `buildRegistry`
 * that reads the actual per-bucket real count — a number a removal moves —
 * becomes a ~6-sigma event instead of a routine one.
 *
 * `meanPerBucket` is the LISTING HIGH-WATER, not the live real count — same
 * reason as `chooseWidth`. A rung crossing is the most visible event this
 * builder publishes (every bucket in the registry changes size at once), so
 * what triggers it must not be something a removal can move.
 *
 * The cost is padding volume: published entries run about 1.58x the real count
 * averaged over a rung and 2.29x at its worst, against 1.44x for a ladder with
 * no reserve and no headroom. It shrinks as the registry grows — the headroom
 * is sqrt(mean) against a rung linear in it.
 */
export function padTargetFor(meanPerBucket: number, kFloor: number = K_FLOOR, width: number = 1): number {
  const z = Math.sqrt(2 * Math.log(16 ** width)) + 3;
  const fullest = meanPerBucket + z * Math.sqrt(Math.max(0, meanPerBucket));
  let target = Math.max(1, kFloor);
  while (capacityFor(target) < meanPerBucket || target - churnFor(target) < fullest) target *= PAD_LADDER_RATIO;
  return target;
}

/** The smallest rung of the ladder that HOLDS `n` entries outright (no capacity
 * reserve). Used only to keep a bucket from shrinking below what it already
 * publishes when a width change re-partitions the survivors. */
function rungAtLeast(n: number, kFloor: number): number {
  let target = Math.max(1, kFloor);
  while (target < n) target *= PAD_LADDER_RATIO;
  return target;
}

/** The smallest rung that holds `n` non-decoy entries AND still leaves its own
 * churn quota in decoys. The HARD feasibility floor — a bucket that cannot pay
 * its quota out of decoys cannot publish a constant departure count. */
function rungReserving(n: number, kFloor: number): number {
  let target = Math.max(1, kFloor);
  while (target - churnFor(target) < n) target *= PAD_LADDER_RATIO;
  return target;
}

/**
 * One decoy hash: a full 64-hex digest derived from the SECRET seed and a
 * stream index, so the same seed reproduces the same decoys on every build.
 *
 * WHY DERIVED AND NOT RANDOM (the whole point). Fresh random decoys made every
 * rebuild an independent sample: two builds six hours apart intersected to
 * exactly the real set — measured, 16,000 entries each, intersection 20, all
 * 20 real, no guessing. Derived decoys make the intersection the whole file
 * minus the churn quota.
 *
 * WHAT A WIDTH CHANGE DOES, AND WHAT IT NO LONGER DOES (D-0437, ADR-0012 ›
 * Amendments). The stream used to be global and a decoy's bucket decided by its
 * own hash, so rebuilding ONE ledger into TWO EMPTY STORES at two widths gave
 * the same decoys, only re-partitioned. This build mints each bucket from its
 * own contiguous sub-range of the same published block (`perBucketBlock`,
 * `decoyHashIn`), because a step mints one bucket and must not walk the whole
 * block to find its own indices — so that two-store property is RETIRED, and
 * the auditor showed that pair is unreachable in production anyway.
 * What holds is the pair the registry's own life takes: across two CONSECUTIVE
 * generations spanning a width change, every survivor is carried and re-bucketed
 * by its own PUBLISHED hash (they are read off the wire, never re-derived)
 * except the `churn` the quota sheds, and fresh mints come from a disjoint block
 * under the same secret seed. So the day the registry outgrows width 1 is still
 * not an intersection attack, and a width change still needs no re-indexing code.
 *
 * WHICH elements of the stream a build may mint from is a BLOCK, `[frontier,
 * frontier + block)`, and the frontier only ever moves forward (it is in the
 * published manifest). So a decoy that has been published and retired can never
 * be published again, because its index is below every future block. That used
 * to be maintained by a separate state object with a recovery path; it is now a
 * property of a published monotone counter, which cannot be lost, replayed or
 * silently mis-recovered.
 */
async function decoyHash(seed: string, i: number): Promise<string> {
  return sha256Hex(`${DECOY_LABEL}|h|${seed}|${i}`);
}

/**
 * Which live decoy a retirement takes: uniform over the bucket's LIVE decoys,
 * derived from the SECRET seed. Both properties are load-bearing.
 *
 *  - UNIFORM, NOT FIFO. A FIFO queue retires in arrival order, so no decoy
 *    younger than the queue's drain time can ever depart — and a departing hash
 *    younger than that is then certainly a REAL entry, i.e. a removal, with the
 *    delisted subject's hash in hand. Measured on a FIFO variant: 5 of 5
 *    removals of post-genesis listings identified with no false positives.
 *    Uniform retirement gives a decoy a memoryless (geometric) lifetime with no
 *    cut-off, so "a young hash departed" is weak evidence instead of proof.
 *    NOTE that a CONSTANT churn quota does not reintroduce the FIFO defect: the
 *    quota fixes the RATE at which decoys leave, and uniform selection keeps the
 *    lifetime geometric at that rate. A deterministic rate with a memoryless
 *    order has no cut-off to read a real entry off.
 *  - SECRET, NOT PUBLIC. If the retirement order were computable from the
 *    published hash, an observer could check every departure against that order
 *    and any departure out of order would be a real entry.
 *
 * Keyed by the monotone build counter and the index of the retirement within the
 * build. The counter is held in BOTH the manifest and the private ladder object
 * and the higher of the two wins, so a rolled-back copy of either cannot replay
 * a build's retirement picks against a different live set.
 */
async function retirementPick(seed: string, prefix: string, seq: number, r: number, liveCount: number): Promise<number> {
  const d = await sha256Hex(`${DECOY_LABEL}|v|${seed}|${prefix}|${seq}|${r}`);
  return parseInt(d.slice(0, 13), 16) % liveCount; // 52 bits — exact in a double
}

/**
 * A decoy's class, from a SEPARATE seeded digest. It must not be a function of
 * the decoy's own hash: that function would be public (this file is), so anyone
 * could recompute it per entry and every entry that disagreed would be real.
 */
async function decoyClass(seed: string, i: number, thresholds: readonly number[]): Promise<ServedClass> {
  const d = await sha256Hex(`${DECOY_LABEL}|c|${seed}|${i}`);
  const v = parseInt(d.slice(0, 8), 16) % CLASS_QUANTUM;
  // `thresholds` has one cut fewer than the classes it splits. A plan frozen by
  // the three-class builder (two cuts) still resumes and mints creator classes.
  for (let j = 0; j < thresholds.length; j++) if (v < thresholds[j]!) return DECOY_CLASS_ORDER[j]!;
  return DECOY_CLASS_ORDER[thresholds.length]!;
}

/** The order `classThresholdsFrom` cuts the quantum in. Warning and caution are
 * LAST so a two-cut plan (the three creator classes) is a prefix of it. */
const DECOY_CLASS_ORDER: readonly ServedClass[] = ["unclaimed", "claimed", "verified", "warning", "caution"];

/** Cumulative class thresholds in CLASS_QUANTUM steps, ordered as
 * `DECOY_CLASS_ORDER`: unclaimed < claimed < verified < warning < caution. The
 * mix follows the GLOBAL real distribution — per-bucket class counts must not
 * leak per-bucket real counts — and EVERY served class is guaranteed at least
 * one step, so no class is ever decoy-free. A decoy-free class is not a
 * rounding artefact: it labels every real entry that has it. That is why the
 * two safety classes are in the mix at all (D-0707, QUEUE #156 PS-042): before,
 * every `c:"warning"` entry on the wire was certainly real.
 *
 * THE SAME LAPLACE FLOOR FOR EVERY CLASS, not a raised one for the safety
 * classes: each counts as 1 + its real count, and each is clamped to at least
 * one step. `caution` is in the mix only while the operator serves it
 * (`serve_caution`); a class the registry does not serve has no real member to
 * hide, and a decoy of it would publish a class the operator switched off. */
function classThresholdsFrom(t: Tally, serveCaution: boolean): number[] {
  const counts = [
    1 + t.cls.unclaimed,
    1 + t.cls.claimed,
    1 + t.cls.verified,
    1 + (t.cls.warning ?? 0),
    ...(serveCaution ? [1 + (t.cls.caution ?? 0)] : []),
  ];
  const total = counts.reduce((a, b) => a + b, 0);
  const clamp = (v: number, lo: number, hi: number): number => Math.min(hi, Math.max(lo, v));
  const cuts: number[] = [];
  let prev = 0;
  // Cut j leaves at least one step for itself (lo) and one for each class after
  // it (hi). For three classes this is exactly the two-cut rule it replaced.
  for (let j = 0; j < counts.length - 1; j++) {
    prev = clamp(prev + Math.round((counts[j]! / total) * CLASS_QUANTUM), prev + 1, CLASS_QUANTUM - (counts.length - 1 - j));
    cuts.push(prev);
  }
  return cuts;
}

/** How often a DECOY warning or caution carries a find-the-real-one pointer, in
 * CLASS_QUANTUM steps: the real per-class rate, Laplace-floored both ways and
 * clamped to [1, QUANTUM-1] exactly as `confirmRatesFrom` does for `p`, so
 * neither "carries `r`" nor "does not" is ever a decoy-free population. */
interface PointerRates {
  warning: number;
  caution: number;
}

function pointerRatesFrom(t: Tally): PointerRates {
  const ptr = t.ptr ?? { warningSeen: 0, warningYes: 0, cautionSeen: 0, cautionYes: 0 };
  const rate = (yes: number, seen: number): number => Math.min(CLASS_QUANTUM - 1, Math.max(1, Math.round(((1 + yes) / (2 + seen)) * CLASS_QUANTUM)));
  return { warning: rate(ptr.warningYes, ptr.warningSeen), caution: rate(ptr.cautionYes, ptr.cautionSeen) };
}

  ["אבגדלמנרשתכ", "יו", "", " "], // Hebrew
  ["कनरसमपलवजदत", "ािीुेो", "", " "], // Devanagari
  ["กนมสปรลวทด", "าิีุ", "", ""], // Thai
];

/**
 * The display name of the decoy pointer whose id is `p` — a function of the
 * SECRET seed and `p` only, so a surviving decoy republishes byte-identically and
 * nothing outside the builder can recompute it. Exported for the shape test.
 */
export async function decoyDisplayName(seed: string, p: string, attempt = 0): Promise<string> {
  // Attempt 0's preimages carry no attempt suffix; attempts 1.. append `@<attempt>`.
  const at0 = attempt === 0 ? "" : `@${attempt}`;
  const hex = (await sha256Hex(`${DECOY_LABEL}|n${at0}|${seed}|${p}`)) + (await sha256Hex(`${DECOY_LABEL}|n2${at0}|${seed}|${p}`));
  const hex3 = await sha256Hex(`${DECOY_LABEL}|n3${at0}|${seed}|${p}`);
  let at = 0;
  const byte = (): number => {
    const b = parseInt(hex.slice(at % hex.length, (at % hex.length) + 2), 16);
    at += 2;
    return b;
  };
  let at3 = 0;
  const byte3 = (): number => {
    const b = parseInt(hex3.slice(at3 % hex3.length, (at3 % hex3.length) + 2), 16);
    at3 += 2;
    return b;
  };
  const pick = <T,>(xs: readonly T[]): T => xs[byte() % xs.length]!;
  const pick3 = (s: string): string => {
    const cs = [...s];
    return cs[byte3() % cs.length]!;
  };
  const token = (): string => {
    const r = byte();
    const syl = r < 128 ? 1 : r < 246 ? 2 : 3;
    let w = "";
    for (let j = 0; j < syl; j++) w += pick(NAME_ONSETS) + pick(NAME_NUCLEI) + (j === syl - 1 || byte() % 3 === 0 ? pick(NAME_CODAS) : "");
    return w.length < 2 ? w + pick(NAME_NUCLEI) : w;
  };
  const title = (w: string): string => w[0]!.toUpperCase() + w.slice(1);
  const digits = (): string => {
    const r = byte();
    if (r < 64) return String(1970 + (byte() % 40)).slice(r < 32 ? 2 : 0);
    return String(byte() % (r < 160 ? 100 : 1000));
  };
  const scriptWord = (s: number): string => {
    if (s === NAME_SCRIPTS.length) {
      let w = ""; // Hangul: 2-3 syllable blocks from NAME_HANGUL.
      for (let j = 0, n = 2 + (byte3() % 2); j < n; j++) w += pick3(NAME_HANGUL);
      return w;
    }
    const [on, nu, co] = NAME_SCRIPTS[s]!;
    let w = "";
    for (let j = 0, n = 2 + (byte3() % 2); j < n; j++) w += pick3(on) + (nu === "" ? "" : pick3(nu));
    if (co !== "" && byte3() % 2 === 0) w += pick3(co);
    return title(w);
  };
  const shape = byte();
  let name: string;
  if (shape < 72) name = `${title(token())} ${title(token())}`; // "Mira Tolvek"
  else if (shape < 100) name = title(token()); // "Zarvik"
  else if (shape < 124) name = `${token()}${byte() < 96 ? digits() : ""}`; // "zarvik92"
  else if (shape < 146) name = `${token()}${pick(["_", "_", ".", "-"])}${token()}${byte() < 40 ? digits() : ""}`; // "zar_vik"
  else if (shape < 166) name = `${title(token())}${title(byte() < 128 ? token() : pick(NAME_TAGS))}`; // "ZarvikPlays"
  else if (shape < 180) name = token().toUpperCase().slice(0, 8); // "ZARV"
  else if (shape < 204) name = `${title(token())} ${pick(NAME_TAGS)}`; // "Zarvik TV"
  else if (shape < 218) name = `${title(token())} ${title(token())} ${title(token())}`;
  else if (shape < 240) name = `${token()} ${token()}`; // "zarvik mol"
  else if (shape < 248) name = `${title(token())}${digits()}`; // "Zarvik2004"
  else name = `${title(token())} ${title(token())}`;
  const rare = byte3();
  if (rare < 5) {
    // ~2%: the whole name in a non-Latin script (one or two words, as the script separates them).
    const s = byte3() % (NAME_SCRIPTS.length + 1);
    const sep = s === NAME_SCRIPTS.length ? " " : NAME_SCRIPTS[s]![3];
    name = sep !== "" && byte3() % 2 === 0 ? `${scriptWord(s)}${sep}${scriptWord(s)}` : scriptWord(s);
  } else if (rare < 10) {
    // ~2%: an apostrophe — "O'Mira Tolvek" or "Mira's Tolvek"; a quarter curly.
    const ap = byte3() % 4 === 0 ? "’" : "'";
    const sp = name.indexOf(" ");
    name = sp > 0 && byte3() % 2 === 0 ? `${name.slice(0, sp)}${ap}s${name.slice(sp)}` : `${pick3(NAME_ELISIONS)}${ap}${title(name)}`;
  } else if (rare < 16) {
    // ~2.3%: the punctuation display names carry.
    const k = byte3() % 4;
    name = k === 0 ? `${name} & ${title(token())}` : k === 1 ? `${name} | ${pick(NAME_TAGS)}` : k === 2 ? `${name}!` : `~${name}~`;
  }
  if (byte() < 28) {
    // ~11%: one accented letter — a quarter of the time the first accentable one (a capital initial, "Émile"),
    // otherwise one of the first two, half the time among the vowels (as real accents mostly are); a third of
    // the time a second; a sixth emitted NFD.
    const cs = [...name];
    const all = cs.flatMap((ch, j) => (NAME_ACCENTS[ch.toLowerCase()] !== undefined ? [j] : []));
    const vowels = all.filter((j) => "aeiouy".includes(cs[j]!.toLowerCase()));
    const r = byte3() % 4;
    const at1 = r === 0 || r === 1 || vowels.length === 0 ? all : vowels;
    if (at1.length > 0) {
      const first = r === 0 ? 0 : byte3() % Math.min(2, at1.length);
      const js = [at1[first]!];
      if (at1.length > 1 && byte3() % 3 === 0) js.push(at1[(first + 1 + (byte3() % (at1.length - 1))) % at1.length]!);
      for (const j of js) {
        const v = pick3(NAME_ACCENTS[cs[j]!.toLowerCase()]!);
        cs[j] = cs[j] === cs[j]!.toLowerCase() ? v : v.toUpperCase();
      }
      name = cs.join("");
      if (byte3() % 6 === 0) name = name.normalize(NFD_FORM);
    }
  }
  if (byte() < 8) {
    const e = pick(NAME_EMOJI);
    name = byte3() % 4 === 0 ? `${e} ${name}` : `${name} ${e}`;
  }
  // The builder's own clamp for real names (`ptr_name` above): no markup, <= 64, trimmed.
  return name.replace(/[<>]/g, "").slice(0, 64).trim();
}

/**
 * NO DECOY NAME EQUALS A REAL ONE (QUEUE #646, D-0819(8), D-0820(4)). A fresh
 * decoy pointer's `n` is the first attempt of `decoyDisplayName` whose fold is
 * not in `agg.nameVeto` (filled by `stepStage`), up to DECOY_NAME_TRIES; if
 * every try collides the decoy publishes with no pointer, the D-0707 veto's own
 * degrade. A SURVIVING decoy is re-checked at every build (`composeBucket`): a
 * name a real creator has since taken is replaced the same way, `p` unchanged.
 * Deterministic: (seed, p, attempt) and the veto only.
 */
const DECOY_NAME_TRIES = 4;

/** D-0849(2), QUEUE #690: every Unicode Default_Ignorable_Code_Point (U+00AD, U+034F, U+200B-F, U+2060-206F, U+FE00-F, U+FEFF, ...) renders as nothing, so the fold strips all of them — derived from Unicode, not a hand list. */
// Hoisted, as social.ts NFKD_FORM is: env-inventory.test.ts reads `fn("UPPER")` as an env key.
const NFC_FORM = "NFC";
// I19-L01 (I19-W1B H3, D-0929(6)): the SAME class as name-fold.ts foldName — Default_Ignorable plus U+2800,
// U+1D159, U+FFFC, U+303F, which render as nothing too — so a real name carrying one vetoes the decoy it renders as.
const NAME_INVISIBLES_RE = /[\p{Default_Ignorable_Code_Point}\u2800\u{1D159}\uFFFC\u303F]/gu;
/** D-0823(5): NBSP and every other Unicode space separator (`\p{Zs}`) render as a space, so they compare as one. */
const NAME_SPACES_RE = /\p{Zs}/gu;
/** D-0849(5), QUEUE #693(3): a run of spaces renders as one space, so it compares as one. */
const NAME_SPACE_RUN_RE = / {2,}/g;

/**
 * What a reader SEES, compared caselessly (D-0820(4)): the real-name clamp
 * (`ptr_name` below: no `<>`, <= 64, trimmed) after stripping every
 * Default_Ignorable_Code_Point (D-0849(2)) and mapping every Unicode space separator (NBSP
 * included) to U+0020 (D-0823(5)), the 64 clamp, then collapsing a run of spaces to one (D-0849(5)), then full Unicode case mapping (upper then lower:
 * "ß"/"SS" and the Kelvin sign meet "ss"/"k") and NFC, so a decomposed accent
 * meets its composed form.
 */
function foldDisplayName(n: string): string {
  return n.replace(/[<>]/g, "").replace(NAME_INVISIBLES_RE, "").replace(NAME_SPACES_RE, " ").slice(0, 64).replace(NAME_SPACE_RUN_RE, " ").trim().toUpperCase().toLowerCase().normalize(NFC_FORM);
}

/** The first attempt of `p`'s name no real display name folds to, or undefined. */
async function decoyNameFor(seed: string, p: string, nameVeto: ReadonlySet<string>): Promise<string | undefined> {
  for (let a = 0; a < DECOY_NAME_TRIES; a++) {
    const n = await decoyDisplayName(seed, p, a);
    if (!nameVeto.has(foldDisplayName(n))) return n;
  }
  return undefined;
}

async function decoyNameCandidates(seed: string, p: string): Promise<string[]> {
  const out: string[] = [];
  for (let a = 0; a < DECOY_NAME_TRIES; a++) out.push(foldDisplayName(await decoyDisplayName(seed, p, a)));
  return out;
}

/**
 * A decoy warning's or caution's find-the-real-one pointer, or undefined — from
 * two more seeded digests (`|q|` decides presence, `|r|` the bytes), separate
 * from the hash and the class for the reason every decoy attribute is: a value
 * computed from the entry's own published bytes would be a public function of
 * them.
 *
 * `p` has the real shape (`psp_` + 16 hex, PASSPORT_ID_RE) and is NEVER a real
 * passport: a 64-bit seeded value is checked against `passports` by `stepStage`,
 * and a pointer that names one is dropped from the entry (`agg.pointerVeto`).
 * `n` is `decoyDisplayName`. RESIDUAL, ACCEPTED by D-0817(5) and declared in
 * docs/REGISTRY-PRIVACY-CLAIMS.md: a decoy pointer does not resolve, so an
 * observer who fetches `/p/<p>` for every pointer can still tell a
 * pointer-carrying decoy from a pointer-carrying real warning.
 */
async function decoyPointer(seed: string, i: number, cls: ServedClass, rates: PointerRates | undefined, nameVeto: ReadonlySet<string>, fresh?: string[]): Promise<{ p: string; n: string } | undefined> {
  if (rates === undefined || (cls !== "warning" && cls !== "caution")) return undefined;
  const q = await sha256Hex(`${DECOY_LABEL}|q|${seed}|${i}`);
  if (parseInt(q.slice(0, 8), 16) % CLASS_QUANTUM >= rates[cls]) return undefined;
  const d = await sha256Hex(`${DECOY_LABEL}|r|${seed}|${i}`);
  const p = `psp_${d.slice(0, 16)}`;
  fresh?.push(p);
  const n = await decoyNameFor(seed, p, nameVeto);
  return n === undefined ? undefined : { p, n };
}

/** How often a decoy of each class carries `p`, in CLASS_QUANTUM steps. */
interface ConfirmRates {
  claimed: number;
  verified: number;
}

/**
 * The per-class rate at which real entries carry the confirmation flag, in
 * CLASS_QUANTUM steps — the same shape, and the same reasoning, as
 * `classThresholds`.
 *
 * PER CLASS, NOT GLOBAL. A verified passport's accounts are proved at a very
 * different rate from a claimed one's, and a single global rate would make
 * "verified without `p`" (or "claimed with `p`") a population whose decoy share
 * did not match its real share — an attacker's first move is always to find a
 * bucket of the file where the two mixes disagree.
 *
 * BOTH ENDS ARE FLOORED (Laplace, and clamped to [1, QUANTUM-1]) so neither
 * "carries `p`" nor "does not" is ever a decoy-free population within a class.
 * A decoy-free population is a hard marker: every entry in it is real. That is
 * the same defect `classThresholds` floors against, one axis over.
 */
/**
 * `passports.subject_kind` → the one-byte wire value, or undefined for a person.
 *
 * The mapping is EXPLICIT rather than a slice or a first letter: a fourth kind
 * added to the schema would fall through to `undefined` here and be served as a
 * PERSON to every client — the exact silent failure
 * `tests/invariants/subject-kind-parity.test.ts` exists to prevent.
 *
 * That file did not cover this function until adversarial review said so: the
 * docblock here asserted it did, and the guard never imported this module. It
 * does now, and it pins the round trip in both directions.
 */
export function wireKind(subjectKind: unknown): "a" | "b" | undefined {
  return subjectKind === "ai_character" ? "a" : subjectKind === "brand" ? "b" : undefined;
}

/**
 * THE PER-CLASS DECOY KIND RATE — `confirmRates`'s construction, applied to the
 * subject kind, and NORMALISED so all three outcomes stay reachable.
 *
 * ─────────────────────────────────────────────────────────────────────────────
 * WHAT THIS REPLACES, AND WHY THE FIRST VERSION WAS WRONG.
 * ─────────────────────────────────────────────────────────────────────────────
 * The first draft invented a raised floor (`KIND_FLOOR = 8/1024`) on the
 * argument that characters are too rare for the existing 1/1024 floor to give a
 * real character any cover: "about 0.2 decoy characters in a 200-entry bucket".
 *
 * THE BUCKET IS NOT 200 ENTRIES. `K_FLOOR` is 1000 and `padTargetFor` never
 * returns less, so a production bucket holds AT LEAST a thousand — five times
 * the number that argument was computed against. At 1/1024 that is already about
 * one decoy of EACH kind per bucket, and ~31 of each registry-wide at width 1.
 * The hazard the raised floor was sold against was covered by the existing floor
 * by an order of magnitude, and the floor bought nothing while costing two real
 * defects, both measured in adversarial review:
 *
 *  1. A DECOY-ONLY MARKER. Below the floor no REAL entry carries what ~1.56% of
 *     decoys carry. Slice 15 publishes the character set on `/p/:id` and the
 *     crawlable page, so an attacker enumerates the public characters, hashes
 *     their accounts, and every remaining `k` in the file is a decoy with
 *     CERTAINTY. `RegistryEntry.p`'s docblock names this direction explicitly and
 *     calls it the WORSE one, "because it eliminates rather than confirms".
 *  2. THE KINDLESS ARM STARVED TO ZERO. `rate()` clamped each kind
 *     independently, so `rates.a + rates.b` could reach 1024 and `decoyKind`
 *     could never return `undefined`. Measured on 130 real subjects of which 129
 *     were characters: the single PERSON's entry was the only kindless entry in
 *     the whole published file, recoverable by exact hash.
 *
 * So the floor is gone and the arithmetic is now a proper three-way distribution.
 *
 * PER-CLASS, like `confirmRates` and for its stated reason — which is
 * distributional, not model-shaped: "a single global rate would make 'verified
 * without `p`' … a population whose decoy share did not match its real share —
 * an attacker's first move is always to find a bucket of the file where the two
 * mixes disagree." The real kind/class mix genuinely correlates (a character is
 * declared while UNCLAIMED, and slice 16's sweep walks lapsed ones back OUT of
 * verified), so a class-independent decoy rate would publish a joint
 * distribution the real data does not have. The first draft argued "kind is an
 * axis, not a verdict" here; that is true of the MODEL and irrelevant to the
 * MIX.
 *
 * NORMALISED TO EXACTLY `CLASS_QUANTUM`. Each of the three outcomes — `a`, `b`
 * and no kind at all — is floored at 1 (nothing may be unreachable, in either
 * direction: a kindless population no decoy can take is the same defect as a
 * class no decoy can take) and the remainder is settled on the largest share, so
 * the three always sum to the quantum and `decoyKind`'s range split cannot
 * starve an arm.
 */
interface KindRate {
  a: number;
  b: number;
}
type KindRates = Record<EntryClass, KindRate>;

function kindRatesFrom(tally: Tally): KindRates {
  const out = {} as KindRates;
  for (const cls of ["unclaimed", "claimed", "verified"] as const) {
    const t = tally.kind[cls];
    // Laplace, exactly as `confirmRates` does it: one pseudo-entry per outcome,
    // so no arm is ever zero even on an empty class. With three outcomes that is
    // three pseudo-entries.
    const seen = t.seen + 3;
    let a = Math.max(1, Math.round(((t.a + 1) / seen) * CLASS_QUANTUM));
    let b = Math.max(1, Math.round(((t.b + 1) / seen) * CLASS_QUANTUM));
    // The kindless arm gets the remainder and is floored too; whatever the floors
    // cost is taken off the LARGEST arm, which is the one that can afford it.
    let none = CLASS_QUANTUM - a - b;
    if (none < 1) {
      const deficit = 1 - none;
      if (a >= b) a -= deficit;
      else b -= deficit;
      none = 1;
    }
    out[cls] = { a, b };
  }
  return out;
}

/**
 * Whether a decoy carries a kind, and which — from a FOURTH seeded digest,
 * separate from the hash, the class and the `p` flag for the reason
 * `decoyProven` states: a value computed from the entry's own published bytes
 * would be a public function of them, so anyone could recompute it and every
 * entry that disagreed would be real.
 *
 * The three outcomes share one digest and are split by range. `kindRatesFor`
 * guarantees `a + b < CLASS_QUANTUM`, so the kindless arm is always reachable and
 * `b` is never starved by an `a` that ran past the quantum.
 */
async function decoyKind(seed: string, i: number, cls: EntryClass, rates: KindRates): Promise<"a" | "b" | undefined> {
  const r = rates[cls];
  const d = await sha256Hex(`${DECOY_LABEL}|k|${seed}|${i}`);
  const v = parseInt(d.slice(0, 8), 16) % CLASS_QUANTUM;
  if (v < r.a) return "a";
  if (v < r.a + r.b) return "b";
  return undefined;
}

/**
 * I3-0122: how often a DECOY warning carries "f" (confirmed fraud), in CLASS_QUANTUM steps — the real share,
 * Laplace-floored and clamped to [1, QUANTUM-1] as `pointerRatesFrom` does, so neither code is ever decoy-free.
 * ZERO while `serve_confirmed_fraud` is off: no real "f" entry exists then, and a decoy "f" would be a decoy-only
 * value, marking every entry that holds it.
 */
function warningKindRateFrom(t: Tally, serveConfirmedFraud: boolean): number {
  if (!serveConfirmedFraud) return 0;
  const wk = t.wk ?? { i: 0, f: 0 };
  return Math.min(CLASS_QUANTUM - 1, Math.max(1, Math.round(((1 + wk.f) / (2 + wk.i + wk.f)) * CLASS_QUANTUM)));
}

/** I3-0122: a decoy warning's kind code, from its own seeded digest keyed on the decoy HASH (so a survivor minted
 * before I23 with no code gets the same code on every later build, never a fresh draw). */
async function decoyWarningKind(seed: string, h: string, fRate: number): Promise<WarningKindCode> {
  if (fRate <= 0) return "i";
  const d = await sha256Hex(`${DECOY_LABEL}|w|${seed}|${h}`);
  return parseInt(d.slice(0, 8), 16) % CLASS_QUANTUM < fRate ? "f" : "i";
}

/** I3-0526: is `v` a Founding rank a bucket may carry (`badge.ts` validFoundingRank's bounds)? */
function isFoundingRank(v: unknown): v is number {
  return typeof v === "number" && Number.isInteger(v) && v >= 1 && v <= FOUNDING_CAPACITY;
}

/** I3-0526: the decoy Founding mix — `rate` in CLASS_QUANTUM steps of verified decoys that carry `f`, and the
 * `[rank, weight]` pairs real verified entries carried (weight = entries, so a founding creator with three accounts
 * weighs three). The real share, Laplace-floored and clamped to [1, QUANTUM-1] as `warningKindRateFrom` does. With
 * no real rank the list is empty and `decoyFounding` draws none (a decoy rank would be a decoy-only value). UNDEFINED
 * — the mix is off and the build publishes no `f` at all — when the tally predates the field (`fd` absent), because
 * reals carrying ranks over decoys that cannot would mark the reals. */
interface FoundingRates {
  rate: number;
  ranks: [number, number][];
}
function foundingRatesFrom(t: Tally): FoundingRates | undefined {
  if (!t.fd) return undefined;
  const ranks = Object.entries(t.fd.byRank)
    .map(([r, w]) => [Number(r), w] as [number, number])
    .filter(([r, w]) => isFoundingRank(r) && w > 0)
    .sort((a, b) => a[0] - b[0]);
  const yes = ranks.reduce((n, [, w]) => n + w, 0);
  return { rate: Math.min(CLASS_QUANTUM - 1, Math.max(1, Math.round(((1 + yes) / (2 + t.fd.verifiedSeen)) * CLASS_QUANTUM))), ranks };
}

/** I3-0526: a verified decoy's Founding rank, or undefined — two seeded digests keyed on the decoy HASH (whether,
 * then which rank by weight), so a survivor gets the same answer on every build while the mix holds. An empty rank
 * list draws nothing (the loop below never returns). */
async function decoyFounding(seed: string, h: string, fr: FoundingRates | undefined): Promise<number | undefined> {
  if (!fr) return undefined;
  const d = await sha256Hex(`${DECOY_LABEL}|f|${seed}|${h}`);
  if (parseInt(d.slice(0, 8), 16) % CLASS_QUANTUM >= fr.rate) return undefined;
  const total = fr.ranks.reduce((n, [, w]) => n + w, 0);
  let v = parseInt(d.slice(8, 16), 16) % Math.max(1, total);
  for (const [r, w] of fr.ranks) {
    if (v < w) return r;
    v -= w;
  }
  return undefined;
}

/** I3-0526: an entry's `f` as this build publishes it. With the mix absent (a plan frozen before I32-FX1) no entry
 * carries one, real or decoy. A REAL entry keeps the rank `realEntriesFrom` gave it (verified only). A DECOY survivor
 * that is `verified` keeps its rank while a real entry of THIS build still carries it (`fr.ranks`, the founding set live
 * at the build — D-1083(2)), else draws one by its hash (`decoyFounding`), so a decoy never keeps the rank of a holder
 * who has left; any other class carries none. */
async function withFounding(seed: string, e: RegistryEntry, isReal: boolean, fr: FoundingRates | undefined): Promise<RegistryEntry> {
  const { f, ...rest } = e;
  if (!fr) return rest;
  if (isReal) return e;
  if (e.c !== "verified") return rest;
  const df = isFoundingRank(f) && fr.ranks.some(([r]) => r === f) ? f : await decoyFounding(seed, e.h, fr);
  return df === undefined ? rest : { ...rest, f: df };
}

function confirmRatesFrom(t: Tally): ConfirmRates {
  // Laplace: two pseudo-entries per class, one of which carries the flag.
  const seen = { claimed: 2 + t.confirm.claimedSeen, verified: 2 + t.confirm.verifiedSeen };
  const yes = { claimed: 1 + t.confirm.claimedYes, verified: 1 + t.confirm.verifiedYes };
  const rate = (k: "claimed" | "verified"): number => Math.min(CLASS_QUANTUM - 1, Math.max(1, Math.round((yes[k] / seen[k]) * CLASS_QUANTUM)));
  return { claimed: rate("claimed"), verified: rate("verified") };
}

/**
 * Whether a decoy carries `p`, from a THIRD seeded digest — separate from the
 * hash and from the class for the same reason those two are separate from each
 * other: a flag computed from the entry's own published bytes would be a public
 * function of them (this file is public), so anyone could recompute it per entry
 * and every entry that disagreed would be real.
 *
 * `unclaimed` never carries it, real or decoy — see `RegistryEntry.p`.
 */
async function decoyProven(seed: string, i: number, cls: EntryClass, rates: ConfirmRates): Promise<boolean> {
  if (cls === "unclaimed") return false;
  const d = await sha256Hex(`${DECOY_LABEL}|p|${seed}|${i}`);
  return parseInt(d.slice(0, 8), 16) % CLASS_QUANTUM < rates[cls];
}

/**
 * The byte-length filler for one bucket: `n` bytes of seeded hex.
 *
 * HIGH-ENTROPY, NOT A RUN OF ONE CHARACTER. The padding's job is to make the
 * SERVED length uniform; a run of `0`s makes the served length uniform and the
 * COMPRESSED length exactly as revealing as before, which would leave the whole
 * fix depending on nothing in front of the registry ever turning gzip on. Hex
 * from the decoy seed cannot be squeezed to nothing. It is not a claim that the
 * compressed lengths are uniform (they are not — see the header).
 *
 * Derived from the seed and the build counter rather than random so a build is
 * reproducible from its inputs, exactly like `decoyHash`. It is NOT secret in
 * any load-bearing sense: it carries no information about the bucket, and an
 * observer already holds the bytes.
 */
async function fillerFor(seed: string, prefix: string, seq: number, n: number): Promise<string> {
  if (n <= 0) return "";
  let out = "";
  for (let i = 0; out.length < n; i++) out += await sha256Hex(`${DECOY_LABEL}|z|${seed}|${prefix}|${seq}|${i}`);
  return out.slice(0, n);
}

// ---------------------------------------------------------------------------
// CROSS-BUILD STATE — what there is of it, and where it lives.
//
// There used to be a decoy WINDOW: a non-served object carrying, per bucket, a
// frontier into the decoy stream, the list of live decoy indices, a retirement
// counter and a fingerprint of the real set. It had to be parsed, magnitude-
// bounded, re-partitioned on a width change, and RECOVERED from the published
// buckets when it went missing — a recovery that walked the decoy stream under
// a scan cap and silently dropped, and therefore re-minted, every decoy it did
// not reach. Measured at production parameters, one lost-window event re-minted
// 23% of a bucket's decoy pool in a single build.
//
// None of it was necessary. The published bucket files ARE the live set, they
// cannot lie about what is on the wire, and they are exactly the bytes the
// privacy property is stated over. So the builder now reads them:
//
//   * WHICH hashes are published        <- the bucket files themselves
//   * WHICH of them are decoys          <- the ones that are not OURS (`ours`
//                                          below: every hash this registry has
//                                          ever had a real reason to publish)
//   * a decoy's PINNED CLASS            <- the class it is published with
//   * how far the decoy stream is spent <- `frontier` in the manifest, monotone
//   * the retirement stream position    <- `seq` in the manifest, monotone
//
// That leaves exactly one number that cannot be read off the published bytes:
// the monotone count of admitted listings that drives the ladder (see
// `LadderState`). It is small, it is not served, and losing it is safe.
// ---------------------------------------------------------------------------
/**
 * The listing high-water. NOT SERVED, and the one piece of builder state that
 * is neither published nor derivable from what is.
 *
 * WHY IT IS NOT IN THE MANIFEST. It is a running count of every listing ever
 * admitted, so in a registry where removals are rare it is very close to the
 * live real count — which is the number the doubling ladder exists to keep OFF
 * the wire. Published volume is meant to say which power-of-two band the
 * registry is in and nothing finer.
 *
 * WHY LOSING IT IS SAFE. A build that cannot read it falls back to the capacity
 * of the rung BELOW the one currently published — the smallest value consistent
 * with the bytes on the wire. That under-counts, never over-counts, so the
 * ladder can only be slow to step, never spuriously early, and `padTarget`
 * cannot fall below what is already published (the survivors term in the rung
 * step pins it). The cost of a lost counter is some listing throughput until it
 * catches up. `seq` is carried alongside purely so a STALE copy (a restored
 * backup, a replayed write) is detected and treated as lost rather than believed.
 */
export const LADDER_KEY = "internal/registry-ladder.json";
const LADDER_VERSION = 2;

/**
 * THE PER-BUCKET LISTING HIGH-WATER (QUEUE #357; closes D-0439's residual).
 *
 * One saturating counter per MAX_WIDTH bucket — the finest partition any shipped
 * client can address (`MAX_WIDTH` is aligned to `MAX_CLIENT_WIDTH`) — holding the
 * largest number of NON-DECOY entries that bucket has ever been asked to carry.
 * CARRIED AT MAX_WIDTH AND PREFIX-SUMMED (`bhwPrefixSums`/`bhwAt`) so the
 * requirement of any bucket at any width <= MAX_WIDTH is one subtraction, with no
 * live per-bucket state read at all.
 *
 * WHY A HIGH-WATER AND NOT A COUNT — the whole closure is this sentence. The rung
 * needs a FEASIBILITY FLOOR: a bucket whose non-decoy entries exceed
 * `pad - churnFor(pad)` has fewer decoys than its churn quota and cannot publish a
 * constant departure count (`rungReserving`). The number the floor has to dominate
 * is a live per-bucket real count, which a removal lowers — read it and the rung
 * becomes a removal oracle (D-0439). A HIGH-WATER IS MONOTONE: a removal cannot
 * lower it, so `bhwAt(...) >= ` the bucket's live non-decoy count is an INVARIANT,
 * and the floor can be taken off the high-water while the rung never reads the
 * count. That is how the floor and removal-insensitivity hold at the same time,
 * which no single term could do.
 *
 * ADVANCED BY MAX, WHICH IS WHY RESUME IS BYTE-IDENTICAL. Each survey step folds
 * its units' leaves in with `Math.max`; re-running a step re-applies the same max,
 * so a build killed at any step replays to the same bytes (ADR-0012 Acceptance,
 * "resume byte-identical"). Nothing here is a read-modify-ADD.
 *
 * LOSING IT IS SAFE, and that is deliberate. When the vector is missing, the wrong
 * length, or stale by the same rule `hw` takes, the floor falls back to
 * `capacityFor(prev.padTarget)` — a PUBLISHED number that dominates any single
 * bucket's surviving non-decoy count, because `composeBucket` caps admissions at
 * `capacityFor`/`creatorCapacityFor` and the only uncapped contributor is the
 * carried survivor set. `rungReserving(capacityFor(p)) === p` exactly, so a lost
 * vector costs no spurious rung step: the floor then says only "do not shrink below
 * the published rung". The vector is a TIGHTENING of that bound, not a correctness
 * dependency.
 *
 * COST: a fixed 174,764 base64 characters inside the (unserved) ladder object,
 * whatever the registry holds. Fixed on purpose — a sparse map's byte length is
 * itself a function of the live listing distribution.
 */
const BHW_LEAVES = 16 ** MAX_WIDTH;

function decodeBhw(v: unknown): Uint16Array | null {
  if (typeof v !== "string") return null;
  let bin: string;
  try {
    bin = atob(v);
  } catch {
    return null;
  }
  if (bin.length !== BHW_LEAVES * 2) return null;
  const out = new Uint16Array(BHW_LEAVES);
  for (let i = 0; i < BHW_LEAVES; i++) out[i] = (bin.charCodeAt(2 * i) & 0xff) | ((bin.charCodeAt(2 * i + 1) & 0xff) << 8);
  return out;
}

/** Little-endian by construction rather than by the platform, so the bytes a
 * resumed build writes are the bytes the killed one would have written. */
function encodeBhw(v: Uint16Array): string {
  const parts: string[] = [];
  const chunk = 4096;
  for (let i = 0; i < BHW_LEAVES; i += chunk) {
    const codes: number[] = [];
    for (let j = i; j < Math.min(i + chunk, BHW_LEAVES); j++) {
      const n = v[j]!;
      codes.push(n & 0xff, (n >> 8) & 0xff);
    }
    parts.push(String.fromCharCode(...codes));
  }
  return btoa(parts.join(""));
}

/** The prefix sum over the MAX_WIDTH leaves. `Float64Array` and not `Int32Array`:
 * 65,536 saturated counters sum past 2^31 and every value here is an exact
 * integer well under 2^53. */
export function bhwPrefixSums(v: Uint16Array): Float64Array {
  const cum = new Float64Array(BHW_LEAVES + 1);
  for (let i = 0; i < BHW_LEAVES; i++) cum[i + 1] = cum[i]! + v[i]!;
  return cum;
}

/** The high-water of the `index`-th bucket AT `width`, answered from the
 * MAX_WIDTH carry in one subtraction. */
export function bhwAt(cum: Float64Array, width: number, index: number): number {
  const stride = 16 ** (MAX_WIDTH - width);
  return cum[(index + 1) * stride]! - cum[index * stride]!;
}

/** The largest per-bucket high-water at `width` — the number the feasibility floor
 * is taken against. `null` vector: the published fallback above. */
export function bucketListingFloor(cum: Float64Array | null, width: number, fallback: number): number {
  if (cum === null) return fallback;
  let m = 0;
  const n = 16 ** width;
  for (let i = 0; i < n; i++) {
    const v = bhwAt(cum, width, i);
    if (v > m) m = v;
  }
  return m;
}

/** MERGE BY MAX, into the vector the ladder object carries. Idempotent, so a
 * replayed survey step is a no-op on bytes. */
function foldBhw(base: Uint16Array | null, leaves: Map<string, number>): Uint16Array {
  const out = base === null ? new Uint16Array(BHW_LEAVES) : base;
  for (const [q, n] of leaves) {
    const i = parseInt(q, 16);
    if (!Number.isInteger(i) || i < 0 || i >= BHW_LEAVES) continue;
    const v = Math.min(0xffff, n);
    if (v > out[i]!) out[i] = v;
  }
  return out;
}

/** Read the ladder object's raw `bhw` field without re-imposing `loadLadder`'s
 * freshness rule: during a build the object carries THIS build's `seq`, which that
 * rule would call stale. The freshness decision is made once, at plan time. */
async function readBhw(store: ObjectStore): Promise<Uint16Array | null> {
  const raw = await store.get(LADDER_KEY);
  if (raw === null) return null;
  try {
    return decodeBhw((JSON.parse(raw) as { bhw?: unknown }).bhw);
  } catch {
    return null;
  }
}

interface LadderState {
  /** Monotone count of real entries ever ADMITTED to a published bucket. */
  hw: number;
  /** Whether the ladder object's per-bucket high-water vector passed the same
   * freshness rule `hw` takes. The VECTOR itself is not carried in the plan — it
   * is 175 KB and the plan blob is written on every step — it is read from the
   * store by the survey and rung steps (`readBhw`). */
  bhwFresh?: boolean;
  /** Floors for the two monotone counters the manifest also carries. Held in
   * BOTH objects on purpose: rewinding either one alone is caught by the other,
   * and rewinding the decoy frontier is the one edit that would republish
   * retired decoys — the exact channel this design exists to close. */
  seqFloor: number;
  frontierFloor: number;
}

const isNat = (v: unknown): v is number => typeof v === "number" && Number.isSafeInteger(v) && v >= 0;

/**
 * What the SERVED manifest says about the generation this build starts from.
 *
 * `prefix` is where that generation's objects live: `gen/<seq>/` for a v3
 * manifest, and "" for the v2 (unversioned-key) generation this builder is
 * cutting over from. The per-bucket bytes are not read here — a build reads one
 * bucket per step (`validateBucket`), which is the whole point of ADR-0012.
 */
interface PrevMeta {
  v: 2 | 3;
  prefix: string;
  width: number;
  padTarget: number;
  kFloor: number;
  seq: number;
  frontier: number;
  /** What the manifest says its buckets digest to, or null on a manifest
   * written before the field existed. */
  manifestDigest: string | null;
}

/**
 * Parse and bound the served manifest.
 *
 * FAIL CLOSED IN EVERY AMBIGUOUS CASE. A manifest this builder cannot read —
 * bad JSON, an older format version, a missing counter, a prefix that does not
 * name its own generation — refuses, because the alternative is a build that
 * re-mints decoys the registry has already retired, and a build in which
 * hundreds of previously-seen hashes come back at once labels every one of them
 * a decoy and, by elimination, the rest of the file as real. Refusing leaves the
 * previously published generation serving, which is stale but correct.
 *
 * A version-1 manifest is one of those: it carries no frontier, so there is no
 * safe way to continue the decoy stream from it. DEPLOY.md 3d has the one-time
 * cutover (it is a deliberate re-mint, taken once, with the intersection cost
 * understood — not something a cron may decide to do at 3am).
 */
function parseManifest(raw: string): PrevMeta {
  let m: { v?: unknown; width?: unknown; pad?: unknown; kfloor?: unknown; seq?: unknown; frontier?: unknown; digest?: unknown; prefix?: unknown };
  try {
    m = JSON.parse(raw) as typeof m;
  } catch {
    throw new RegistryBuildError("registry manifest unreadable — refusing to rebuild the decoy stream from scratch");
  }
  if (m.v !== MANIFEST_FORMAT_VERSION && m.v !== LEGACY_MANIFEST_VERSION) {
    throw new RegistryBuildError(
      `registry manifest is format v${String(m.v)}, this builder writes v${MANIFEST_FORMAT_VERSION} — refusing to continue a decoy stream whose frontier it cannot read (see DEPLOY.md 3d)`,
    );
  }
  const { width, pad, kfloor, seq, frontier } = m;
  const manifestDigest = typeof m.digest === "string" && /^[0-9a-f]{64}$/.test(m.digest) ? m.digest : null;
  if (!isNat(width) || width < 1 || width > MAX_WIDTH || !isNat(pad) || pad < 1 || !isNat(kfloor) || kfloor < 1 || !isNat(seq) || !isNat(frontier)) {
    throw new RegistryBuildError("registry manifest is malformed — refusing to rebuild the decoy stream from scratch");
  }
  const v = m.v === MANIFEST_FORMAT_VERSION ? 3 : 2;
  // The pointer has to name its OWN generation: a prefix that disagrees with
  // `seq` would send the public worker at another build's bytes under this
  // build's signature, which is P5 (signed generation coherence) inverted.
  if (v === 3 && m.prefix !== genPrefix(seq)) {
    throw new RegistryBuildError(`registry manifest v3 carries prefix ${String(m.prefix)}, which does not name generation ${seq} — refusing to rebuild`);
  }
  return { v, prefix: v === 3 ? genPrefix(seq) : "", width, padTarget: pad, kFloor: kfloor, seq, frontier, manifestDigest };
}

/**
 * ONE published bucket, re-validated. Every entry, not just the count: this
 * builder classifies these bytes into "ours" and "our decoys" and republishes
 * the survivors verbatim, so anything that gets in here is served to fans until
 * it is retired — which, for a hash the builder reads as a decoy, is a geometric
 * number of builds. The count check alone let a malformed or hostile object
 * carry an arbitrary class, an arbitrary find-the-real-one pointer, a duplicate,
 * or a hash belonging to a different bucket straight back onto the wire.
 * (It does NOT stop a well-formed forged hash: the builder cannot tell a forged
 * 64-hex string from one of its own decoys without walking the decoy stream,
 * which is the state-recovery this design deleted. Write access to the bucket
 * store is trusted, and that is written up in the doc.)
 */
function validateBucket(prefix: string, width: number, pad: number, raw: string | null, tornPadOk?: (n: number) => boolean): RegistryEntry[] {
  if (raw === null) throw new RegistryBuildError(`registry bucket ${prefix} is missing at width ${width} — refusing to rebuild`);
  let entries: RegistryEntry[];
  try {
    entries = (JSON.parse(raw) as { entries?: RegistryEntry[] }).entries ?? [];
  } catch {
    throw new RegistryBuildError(`registry bucket ${prefix} will not parse — refusing to rebuild`);
  }
  // A bucket that does not match the manifest's own rung is not a bucket this
  // builder wrote. Bounding it here is what keeps a tampered object from turning
  // one write to the store into unbounded work below.
  // I3-1362: a TORN v2 pad-growth publication (the in-place writer reserved a
  // newer seq, rewrote some buckets at the grown pad, and died before the
  // manifest) is the one other count accepted, and only when `stepSurvey` says
  // the store is torn; `tornPadOk` pins it to one uniform bounded value.
  if (!Array.isArray(entries) || (entries.length !== pad && !(tornPadOk?.(entries.length) ?? false))) {
    throw new RegistryBuildError(`registry bucket ${prefix} holds ${Array.isArray(entries) ? entries.length : "non-array"} entries, manifest says ${pad} — refusing to rebuild`);
  }
  const seenHere = new Set<string>();
  for (const e of entries) {
    if (typeof e !== "object" || e === null) throw new RegistryBuildError(`registry bucket ${prefix} holds a non-object entry — refusing to rebuild`);
    const { h, c, p, r, k, w, f } = e as { h?: unknown; c?: unknown; p?: unknown; r?: unknown; k?: unknown; w?: unknown; f?: unknown };
    if (typeof h !== "string" || !/^[0-9a-f]{64}$/.test(h)) throw new RegistryBuildError(`registry bucket ${prefix} holds a malformed hash — refusing to rebuild`);
    if (h.slice(0, width) !== prefix) throw new RegistryBuildError(`registry bucket ${prefix} holds a hash that belongs in bucket ${h.slice(0, width)} — refusing to rebuild`);
    if (seenHere.has(h)) throw new RegistryBuildError(`registry bucket ${prefix} holds a duplicate hash — refusing to rebuild`);
    seenHere.add(h);
    if (c !== "verified" && c !== "claimed" && c !== "unclaimed" && c !== "warning" && c !== "caution") {
      throw new RegistryBuildError(`registry bucket ${prefix} holds an entry with an unknown class — refusing to rebuild`);
    }
    if (p !== undefined) {
      // The confirmation flag is presence-only (`1`) and meaningful on the two
      // classes that can carry a per-account proof. On anything else it is a
      // forgery — and on `unclaimed` it would additionally mark the entry as a
      // decoy, since no real unclaimed entry may carry it.
      if (p !== 1) throw new RegistryBuildError(`registry bucket ${prefix} holds a malformed confirmation flag — refusing to rebuild`);
      if (c !== "claimed" && c !== "verified") throw new RegistryBuildError(`registry bucket ${prefix} holds a confirmation flag on a ${c} entry — refusing to rebuild`);
    }
    if (k !== undefined) {
      // A CLOSED TWO-VALUE VOCABULARY, and a placement rule — see the entry
      // shape. An unbounded value here is also body-size amplification into the
      // byte-length padding, which sizes every bucket from the longest.
      if (k !== "a" && k !== "b") throw new RegistryBuildError(`registry bucket ${prefix} holds a malformed subject kind — refusing to rebuild`);
      if (c === "warning" || c === "caution") throw new RegistryBuildError(`registry bucket ${prefix} holds a subject kind on a ${c} entry — refusing to rebuild`);
    }
    if (f !== undefined) {
      // I3-0526: a Founding rank, 1..FOUNDING_CAPACITY, on a verified entry only. Absent is legal (before I32-FX1).
      if (!isFoundingRank(f)) throw new RegistryBuildError(`registry bucket ${prefix} holds a malformed founding rank — refusing to rebuild`);
      if (c !== "verified") throw new RegistryBuildError(`registry bucket ${prefix} holds a founding rank on a ${c} entry — refusing to rebuild`);
    }
    if (w !== undefined) {
      // I3-0122: a closed vocabulary, on a warning only. Absent is legal (a bucket written before I23).
      if (!isWarningKindCode(w)) throw new RegistryBuildError(`registry bucket ${prefix} holds a malformed warning kind — refusing to rebuild`);
      if (c !== "warning") throw new RegistryBuildError(`registry bucket ${prefix} holds a warning kind on a ${c} entry — refusing to rebuild`);
    }
    if (r !== undefined) {
      // The pointer belongs to the two safety classes (real and decoy alike, at
      // the same per-class rate); on any other class it is a forgery.
      if (c !== "warning" && c !== "caution") throw new RegistryBuildError(`registry bucket ${prefix} holds a find-the-real-one pointer on a ${c} entry — refusing to rebuild`);
      const rp = r as { p?: unknown; n?: unknown };
      if (typeof rp.p !== "string" || !PASSPORT_ID_RE.test(rp.p) || typeof rp.n !== "string" || rp.n.length === 0 || rp.n.length > 64) {
        throw new RegistryBuildError(`registry bucket ${prefix} holds a malformed find-the-real-one pointer — refusing to rebuild`);
      }
    }
  }
  return entries;
}

/**
 * The largest listing high-water that a published rung is consistent with —
 * `strict`, the largest that would have chosen a SMALLER rung. Both ends are
 * used and both are exact, because they bracket the one number the builder
 * cannot read off the wire (see `LADDER_KEY`).
 */
function hwBoundFor(padTarget: number, kFloor: number, width: number, strict: boolean): number {
  const buckets = 16 ** width;
  const ok = (h: number): boolean => {
    const rung = padTargetFor(h / buckets, kFloor, width);
    return strict ? rung < padTarget : rung <= padTarget;
  };
  if (!ok(0)) return 0;
  let lo = 0;
  let hi = padTarget * buckets;
  while (lo < hi) {
    const mid = Math.floor((lo + hi + 1) / 2);
    if (ok(mid)) lo = mid;
    else hi = mid - 1;
  }
  return lo;
}

/** The listing high-water, or the floor implied by the published rung when it is
 * missing, unreadable or stale. Never throws: a lost counter is a throughput
 * cost, not a privacy one. */
async function loadLadder(store: ObjectStore, prev: PrevMeta | null): Promise<LadderState> {
  const floor = prev === null ? 0 : hwBoundFor(prev.padTarget, prev.kFloor, prev.width, true);
  const base: LadderState = { hw: floor, seqFloor: prev?.seq ?? 0, frontierFloor: prev?.frontier ?? 0 };
  const raw = await store.get(LADDER_KEY);
  interface LadderDoc { v?: unknown; seq?: unknown; hw?: unknown; frontier?: unknown }
  let d: LadderDoc | null = null;
  if (raw !== null) {
    try {
      d = JSON.parse(raw) as LadderDoc;
    } catch {
      d = null;
    }
  }
  if (d === null || d.v !== LADDER_VERSION || !isNat(d.hw) || !isNat(d.seq) || !isNat(d.frontier)) {
    if (raw !== null || prev !== null) {
      logError({ cid: "unknown", hop: "cron", msg: "registry_ladder_state_unreadable_fallback" });
    }
    return base;
  }
  // THE TWO MONOTONE COUNTERS TAKE THE HIGHER OF THE TWO COPIES. `seq` drives
  // the retirement stream and `frontier` drives which stream indices may be
  // minted, so rewinding either one republishes decoys that have already been
  // retired — the returning-hash channel. They live in the manifest AND here, so
  // rolling back one object alone (a restored backup, a replayed write) is
  // caught by the other rather than believed.
  base.seqFloor = Math.max(base.seqFloor, d.seq);
  base.frontierFloor = Math.max(base.frontierFloor, d.frontier);
  // THE HIGH-WATER IS NOT MONOTONE-SAFE THE SAME WAY, so it is trusted only when
  // the two objects agree exactly — anything else is a copy from another build,
  // and a high-water read out of one is a rung crossing (a registry-wide,
  // published, irreversible event) taken on a stale number.
  if (d.seq !== (prev?.seq ?? 0)) {
    logError({ cid: "unknown", hop: "cron", msg: "registry_ladder_state_stale_fallback" });
    return base;
  }
  // THE PER-BUCKET VECTOR TAKES THE SAME FRESHNESS RULE AS `hw`, and for the same
  // reason: a copy out of another build is a number this one would step the rung
  // on. Dropping it is safe — the floor falls back to `capacityFor(prev.padTarget)`
  // (see `BHW_LEAVES`), which is published and exactly sufficient.
  base.bhwFresh = decodeBhw((d as { bhw?: unknown }).bhw) !== null;
  // AND BOUNDED BY THE PUBLISHED RUNG — computed against the k-floor the last
  // build ran with, which the manifest carries, so that changing `kFloor` is an
  // operator action rather than something mistaken for a forged counter.
  const ceiling = prev === null ? 0 : hwBoundFor(prev.padTarget, prev.kFloor, prev.width, false);
  base.hw = Math.min(d.hw, Math.max(floor, ceiling));
  return base;
}

/**
 * How many indices of the decoy stream this build may spend, and therefore how
 * far the published frontier moves.
 *
 * IT IS A FUNCTION OF PUBLISHED NUMBERS ONLY — the widths and rungs of this
 * build and the last one, and the largest number of slots any bucket has to
 * fill (which an observer can compute by intersecting the two builds). It must
 * not depend on how many of those slots went to real entries rather than
 * decoys, because the frontier is published: a block sized to the actual mint
 * count would publish the listing count.
 *
 * THE BLOCK IS NOW PARTITIONED, one contiguous sub-range per bucket
 * (`perBucketBlock`), because a step mints ONE bucket and must not walk the
 * whole block to find the indices that land in it. The block's SIZE is the
 * function it always was, so the published frontier advance is unchanged; what
 * changed is that a minted hash takes its bucket's prefix and its TAIL from the
 * stream digest (`decoyHashIn`) instead of being filtered by the prefix its own
 * digest happened to have. Both are a uniform hash under the prefix, which is
 * what a real entry in that bucket is too.
 */
const MINT_BLOCK_FACTOR = 8;
const MINT_BLOCK_MIN = 16;

function mintBlockFor(width: number, maxSlotsPerBucket: number): number {
  return MINT_BLOCK_FACTOR * 16 ** width * Math.max(MINT_BLOCK_MIN, maxSlotsPerBucket);
}

/** One bucket's share of the block: the oversampling factor times the largest
 * number of slots any bucket has to fill. */
function perBucketBlock(maxSlotsPerBucket: number): number {
  return MINT_BLOCK_FACTOR * Math.max(MINT_BLOCK_MIN, maxSlotsPerBucket);
}

/** A decoy for bucket `prefix`: the stream digest at `i`, wearing that bucket's
 * prefix. The tail is a full-entropy digest, exactly as a real subject hash in
 * that bucket is. */
async function decoyHashIn(seed: string, i: number, prefix: string): Promise<string> {
  return `${prefix}${(await decoyHash(seed, i)).slice(prefix.length)}`;
}

/** `removal.ts`'s AUTO_OPERATOR, duplicated here (see `realEntriesFrom`). A
 * suppression carrying it had no human behind it, so it drains in the lane
 * below the operator-decided one. */
const SELF_SERVE_OPERATOR = "system:removal-proof";

/**
 * The digest of a published generation: a fold over its bucket LEAVES in prefix
 * order, domain-separated from the decoy stream. It has one job — telling a
 * manifest that describes these buckets from one that describes a different
 * build — and a leaf is a digest of the bucket's exact bytes (`bucketLeaf`), so
 * it is still a pure function of what is on the wire. Folding leaves rather than
 * bodies is what lets a resumable build compute it one step at a time.

 * A v2 generation is folded the same way, over its BODIES (`digestSeed(w,
 * false)`), because that is what its manifest committed to.
 */
function digestSeed(width: number, v3: boolean): string {
  return `${DECOY_LABEL}|${v3 ? "d3" : "d"}|${width}`;
}

async function digestStep(acc: string, prefix: string, part: string): Promise<string> {
  return sha256Hex(`${acc}|${prefix}|${part}`);
}

/**
 * Pad ONE bucket body to the build's byte-length target, with the balance taken
 * up by a trailing `z` field of seeded hex. See BUCKET_BYTE_QUANTUM for why the
 * length has to be uniform and why it is measured in BYTES, not characters.
 *
 * The target is frozen in the plan (the longest body this build produced,
 * rounded up to the next quantum) before any bucket is written, so a bucket
 * written in one step and a bucket written an hour later are the same length.
 */
const FILLER_OVERHEAD = new TextEncoder().encode(`,"z":""`).length;

function byteTargetFor(longestBody: number): number {
  return Math.ceil((longestBody + FILLER_OVERHEAD) / BUCKET_BYTE_QUANTUM) * BUCKET_BYTE_QUANTUM;
}

async function padBody(body: string, byteLen: number, target: number, seed: string, prefix: string, seq: number): Promise<string> {
  const n = target - byteLen - FILLER_OVERHEAD;
  if (n < 0) throw new RegistryBuildError(`registry bucket ${prefix} is ${byteLen} bytes against a frozen target of ${target} — refusing to publish an uneven generation`);
  // Every body ends with the object's closing brace; the field goes in front of
  // it. Hex is ASCII, so its character count IS its byte count.
  return `${body.slice(0, -1)},"z":"${await fillerFor(seed, prefix, seq, n)}"}`;
}


interface AccountRow {
  account_key: string;
  id_kind: string;
  passport_id: string;
  status: string;
  subject_hash: string;
  acct_vetted: number | null;
  acct_asserted: number | null;
  claim_state: EntryClass;
  subject_kind: string;
  retracted: number;
  sup_operator: string | null;
  sup_at: string | null;
  key_operator: string | null;
  key_at: string | null;
  /** I3-1341: the latest verification_granted/renewed payload, read only for a `verified` passport (else NULL). */
  grant_payload: string | null;
  /** I3-0526: the passport's Founding rank, or NULL; `realEntriesFrom` serves it on a `verified` entry only, which is
   * `foundingRank`'s gate (verified, not retracted — a retracted passport publishes nothing). */
  founding_rank: number | null;
}

/** What `realEntriesFrom` hands back — see the lane comment inside.
 *
 *  - `published` is what goes in the buckets.
 *  - `precedence` is every served-eligible account hash INCLUDING the suppressed
 *    ones, and exists so that removing a suppressed creator from the published
 *    set can never hand their hash to a warning or a caution (creator identity
 *    keeps winning at that hash even though nothing is served there).
 *  - `ours` is every hash this registry has ever had a real reason to publish —
 *    every account of every passport whatever its status, suppressed or not,
 *    retracted or not. It is how the builder tells a DECOY from a real entry
 *    that has stopped being real: a published hash that is not `ours` is a decoy
 *    of ours, and one that IS `ours` but is not in `published` is a departure
 *    waiting for a churn slot.
 *  - `removalOrder` ranks the waiting departures in three LANES in strict
 *    priority order, because the rate a lane drains at is the delisting SLA and
 *    a lane an outsider can fill is an SLA an outsider controls.
 *
 *      lane 0  OPERATOR-DECIDED suppressions — a human closed a removal
 *              request, or an operator flipped the switch.
 *      lane 1  SELF-SERVE suppressions — `removal.ts`'s bio-code fast path.
 *      lane 2  EVERYTHING ELSE that stopped being real — a revoked account, a
 *              retracted passport, a proof that lapsed.
 */
interface RealSet {
  published: RegistryEntry[];
  precedence: Set<string>;
  ours: Set<string>;
  removalOrder: Map<string, string>;
}

/**
 * Real entries: active accounts of non-retracted, NON-SUPPRESSED passports, for
 * ONE prefix range of subject hashes.
 *
 * SAME GATES, PREFIX-BOUNDED. The rows arrive from `readUnitRows`, which reads
 * the materialized `subject_hash` (migration 0097) instead of hashing every
 * account in the registry once per build (I3-0635), and carries the suppression
 * and proof state as columns instead of three whole-table reads.
 *
 * ---------------------------------------------------------------------------
 * WHICH ACCOUNTS MUST PROVE THEMSELVES
 * ---------------------------------------------------------------------------
 * Two exceptions to "prove one account -> the whole passport serves" shipped
 * first, both the same principle — an ASSERTED account is never served as
 * proven: DOMAINS (S76) and CREATOR/AGENCY-ADDED handles (S101). "Asserted" is
 * decided by the ADDING event's actor_kind (unforgeable), NOT the
 * client-supplied `relationship_basis` label — here read from the
 * build-maintained `acct_asserted`/`acct_vetted` columns, which are the same
 * `foldAsserted` sets (registry-gen.ts). A row the fold has not reached yet
 * (both columns NULL) is read as ASSERTED: inert until proven is the fail-safe
 * direction, and INV-016 is the reason.
 *
 * `perAccountProof` is the THIRD, and it swallows the other two: with the
 * per-account standing model on, EVERY active account needs its own current
 * proof, whoever put the row there. The unproven residue is NOT dropped — it is
 * served as `caution` by `cautionEntries` — which is why this is gated on the
 * same operator setting the caution class is.
 *
 * ---------------------------------------------------------------------------
 * WHAT "PROVEN" MEANS HERE
 * ---------------------------------------------------------------------------
 * A (PASSPORT, ACCOUNT) PAIR whose proof is CURRENT — not an account_key that
 * appears anywhere in the proof tables. Reading the pair, with a TTL, makes
 * handle transfer SELF-INVALIDATING: the moment the account hangs off a
 * different passport the old proof stops matching.
 *
 * SUPPRESSION is read here, not left to a serving function, because the
 * registry service serves these bytes verbatim with no DB to re-check against.
 */
function realEntriesFrom(rows: AccountRow[], proven: Set<string>, perAccountProof: boolean, nowMs: number): RealSet {
  // lane 0 = a human decided, lane 1 = the self-serve bio-code fast path. The
  // sentinel is `removal.ts`'s AUTO_OPERATOR; it is duplicated rather than
  // imported so the builder stays free of the removal module's dependency
  // graph, and the two are pinned equal in tests/invariants/registry-churn.test.ts.
  const lane = (operator: unknown, createdAt: unknown): string => `${String(operator ?? "") === SELF_SERVE_OPERATOR ? 1 : 0}|${String(createdAt ?? "")}`;
  const published: RegistryEntry[] = [];
  const precedence = new Set<string>();
  const ours = new Set<string>();
  const removalOrder = new Map<string, string>();
  for (const row of rows) {
    const h = row.subject_hash;
    ours.add(h); // every account we have ever had a reason to publish
    // THE UNION, never an intersection: this account's CURRENT passport is
    // suppressed, OR this account KEY was frozen by a suppression (whatever
    // passport it hangs off today — `account_transferred` moves the row, and one
    // transfer off the delisted passport used to put the hash back in the
    // buckets as `unclaimed` on the very next build). A suppression written
    // before migration 0028 has no snapshot row, so the passport leg still drops it.
    const at = row.sup_at !== null || row.sup_operator !== null ? lane(row.sup_operator, row.sup_at) : row.key_at !== null || row.key_operator !== null ? lane(row.key_operator, row.key_at) : undefined;
    // A removal REQUEST is the queue that has an SLA attached to it, so it sorts
    // first and by request time. Everything else that stops being real takes
    // what is left, in a stable order.
    if (at !== undefined) removalOrder.set(h, `${at}|${h}`);
    if (row.status !== "active" || row.retracted !== 0) continue;
    // A domain (DNS) and a generic `link` (id_kind "url", M3) are BOTH control-
    // proven separately, never on the passport's claim alone.
    const isAsserted = row.acct_asserted === null || row.acct_vetted === null ? true : row.acct_asserted === 1 && row.acct_vetted !== 1;
    const ownProof = proven.has(proofPairKey(row.passport_id, row.account_key));
    const needsOwnProof = needsOwnProofTerm({ kind: row.id_kind, asserted: isAsserted, perAccountProof }); // account-standing.ts needsOwnProof, the one spelling
    // ---------------------------------------------------------------------
    // THE SEEDED-LINK ARM — the one population that must be DOWNGRADED rather
    // than dropped. A `link` (id_kind "url") is unclaimed BY CONSTRUCTION when an
    // operator seeds it: there is no link proof path, so under a plain "unproven
    // => never served" rule it was never in any bucket, while `/p/:id` PUBLISHED
    // it — so Dhwaja printed a link on a creator's own page and then answered
    // "Dhwaja has no record for this account" to the fan who checked it.
    // `unclaimed` is not a weaker version of the drop, it is the CORRECT answer.
    // An ASSERTED link is a creator saying "this is mine" with nothing behind it;
    // a DOMAIN has a proof path that works; and with per-account standing ON this
    // arm is off, because the amber caution is the stronger answer.
    // ---------------------------------------------------------------------
    const seededLink = row.id_kind === "url" && !isAsserted && !perAccountProof;
    if (needsOwnProof && !ownProof && !seededLink) continue; // unproven: never served
    // NOTE the ordering: this `continue` runs BEFORE `precedence.add(h)`, so an
    // account that is required to prove itself and has not is also outside
    // creator-precedence — which is exactly what lets `cautionEntries` label it
    // amber. A seeded link does NOT take that path: it is published, so it must
    // be inside precedence or a warning could be served at a hash a creator
    // listing already owns.
    precedence.add(h); // creator identity keeps winning here even when suppressed
    if (at !== undefined) continue; // asked to be gone: not published at all
    // The class is the PASSPORT's, except where this account owes a proof it has
    // not paid — then it is `unclaimed`, whatever the passport is.
    // I3-1341 / I3-1768: a `verified` row whose latest grant or renewal has lapsed
    // at the build's instant publishes as `claimed` — the class the expiry sweep's
    // `verification_expired` fold gives it — so the cursor-windowed sweep not
    // having reached it yet never publishes a lapsed grant as verified.
    // `grant_payload` is non-NULL only on a `verified` row (readRange's CASE), so
    // this can only ever move verified -> claimed.
    const passportClass: EntryClass = grantLapsed(row.grant_payload, nowMs) ? "claimed" : row.claim_state;
    const cls: EntryClass = needsOwnProof && !ownProof ? "unclaimed" : passportClass;
    const kind = wireKind(row.subject_kind);
    // I3-0526: the Founding rank rides a `verified` entry only — so never a lapsed grant (passportClass above).
    const founding = cls === "verified" && isFoundingRank(row.founding_rank) ? { f: row.founding_rank } : {};
    published.push({ h, c: cls, ...(ownProof && cls !== "unclaimed" ? { p: 1 as const } : {}), ...(kind ? { k: kind } : {}), ...founding });
  }
  return { published, precedence, ours, removalOrder };
}

interface WarningRow {
  seq: number;
  event_id: string;
  subject: string;
  event_type: string;
  payload: string;
  ptr_pid: string | null;
  ptr_name: string | null;
  ptr_retracted: number | null;
  ptr_suppressed: number | null;
}

/**
 * Active warnings for one prefix range, derived from the ledger: a subject is
 * served as "warning" iff its latest warning event is a warning_published (not a
 * warning_retracted) AND its class is in the SERVED set. impersonation is always
 * served (the launch scope); confirmed_fraud only when an operator has flipped
 * the counsel-cleared ops setting; historical_fraud_association stays deferred.
 * These are the REAL warnings. Decoys take the class too (`decoyClass`), and a
 * fan is still only ever shown "reported" for an account that truly is: a
 * client shows an entry only on full-hash equality with the subject it checked.
 *
 * THE FIND-THE-REAL-ONE POINTER IS ENRICHED BY THE QUERY, not by a point read
 * per warning (I3-1343's N+1): `readUnitRows` LEFT JOINs the protected passport
 * and its suppression onto the publish event. The gate is re-checked at EVERY
 * build: the pointer is served while the operator set it (QUEUE #292 / D-0802,
 * `operatorSetPointerSql` in the JOIN), the protected passport
 * EXISTS, is not retracted, and is not suppressed — claim state does NOT gate it
 * (flag-5, 2026-08-14). A retraction or a removal-request suppression self-heals
 * out of the served data on the next rebuild without touching the ledger.
 * Display name is clamped and markup-stripped here; clients still treat it as
 * untrusted.
 */
function warningEntriesFrom(rows: WarningRow[], serveConfirmedFraud: boolean): { e: RegistryEntry; ord: number; cls: string; ev: string }[] {
  const served = new Set<string>(["impersonation"]);
  if (serveConfirmedFraud) served.add("confirmed_fraud");
  const active = new Map<string, boolean>();
  const firstSeen = new Map<string, number>();
  const cls = new Map<string, string>();
  const ptr = new Map<string, { p: string; n: string } | undefined>(); // subject → pointer (latest publish wins)
  const pubEvent = new Map<string, string>(); // subject → the LIVE publication's event_id (D-0442 keys its review to it)
  for (const r of rows) {
    if (!firstSeen.has(r.subject)) firstSeen.set(r.subject, Number(r.seq));
    if (r.event_type === "warning_published") {
      active.set(r.subject, true);
      pubEvent.set(r.subject, String(r.event_id));
      let payloadPid: unknown;
      try {
        const p = JSON.parse(r.payload) as { warning_class?: unknown; protected_passport_id?: unknown };
        cls.set(r.subject, String(p.warning_class));
        payloadPid = p.protected_passport_id;
      } catch {
        cls.set(r.subject, "");
        ptr.set(r.subject, undefined);
        continue;
      }
      // Defensive re-shape-check: a malformed pointer degrades to "no pointer",
      // never to a crash or a served junk value. The shape is the SHARED one
      // (PASSPORT_ID_RE) — a pointer this builder would serve but the SPA's
      // router would not route is a redirect-to-safety link that silently lands
      // on the home page, which is worse than no pointer.
      const ok = typeof payloadPid === "string" && PASSPORT_ID_RE.test(payloadPid) && r.ptr_pid === payloadPid && r.ptr_retracted === 0 && !r.ptr_suppressed;
      const name = ok ? String(r.ptr_name ?? "").replace(/[<>]/g, "").slice(0, 64).trim() : "";
      // An unrenderable pointer is worse than none.
      ptr.set(r.subject, ok && name.length > 0 ? { p: r.ptr_pid as string, n: name } : undefined);
    } else {
      active.set(r.subject, false); // a later retraction wins (seq ASC)
    }
  }
  const entries: { e: RegistryEntry; ord: number; cls: string; ev: string }[] = [];
  for (const [subject, isActive] of active) {
    if (!isActive || !served.has(cls.get(subject) ?? "")) continue;
    const m = /^sha256:([0-9a-f]{64})$/.exec(subject);
    if (!m) continue;
    const real = cls.get(subject) === "impersonation" ? ptr.get(subject) : undefined;
    // I3-0122: the served class's kind code (`served` holds only classes WARNING_KIND_CODE names).
    const w = WARNING_KIND_CODE[cls.get(subject) as keyof typeof WARNING_KIND_CODE];
    entries.push({
      e: real ? { h: m[1]!, c: "warning", r: real, w } : { h: m[1]!, c: "warning", w },
      ord: firstSeen.get(subject) ?? 0,
      cls: cls.get(subject) ?? "",
      ev: pubEvent.get(subject) ?? "",
    });
  }
  return entries;
}

function emptyTally(): Tally {
  return {
    cls: { unclaimed: 0, claimed: 0, verified: 0, warning: 0, caution: 0 },
    ptr: { warningSeen: 0, warningYes: 0, cautionSeen: 0, cautionYes: 0 },
    confirm: { claimedSeen: 0, claimedYes: 0, verifiedSeen: 0, verifiedYes: 0 },
    kind: { unclaimed: { seen: 0, a: 0, b: 0 }, claimed: { seen: 0, a: 0, b: 0 }, verified: { seen: 0, a: 0, b: 0 } },
    wk: { i: 0, f: 0 },
    fd: { verifiedSeen: 0, byRank: {} },
  };
}

function emptyAgg(): Agg {
  return {
    backfillSteps: 0, wantIn: 0, spilledOut: 0, spilledOutRetractions: 0, precedenceCollisions: 0, prevDigest: null,
    byWidth: {}, tally: emptyTally(), longest: 0, total: 0, realCount: 0, spilledIn: 0, spilledInWarnings: 0, spilledInCautions: 0, spilledInListings: 0, pointerVeto: [], nameVeto: [], blockedBuckets: 0,
    safetyPending: 0, safetyAdmitted: 0, safetyStages: [], digest: null, subRoots: [], root: null,
  };
}

/** The class/confirmation/kind mix a decoy is minted against, tallied ACROSS
 * STEPS: every decoy in the generation is minted against the same global mix,
 * which is what keeps per-bucket class counts from leaking per-bucket real
 * counts. The Laplace floors live in the three `…From` functions. */
function tallyInto(t: Tally, entries: RegistryEntry[]): void {
  for (const e of entries) {
    if (e.c === "warning" || e.c === "caution") {
      // The safety classes drive the decoy class mix and the decoy pointer rate
      // (D-0707); they carry no `p` and no `k`, real or decoy.
      t.cls[e.c] = (t.cls[e.c] ?? 0) + 1;
      t.ptr ??= { warningSeen: 0, warningYes: 0, cautionSeen: 0, cautionYes: 0 };
      if (e.c === "warning") {
        t.ptr.warningSeen += 1;
        if (e.r) t.ptr.warningYes += 1;
        t.wk ??= { i: 0, f: 0 };
        if (e.w === "f") t.wk.f += 1;
        else t.wk.i += 1;
      } else {
        t.ptr.cautionSeen += 1;
        if (e.r) t.ptr.cautionYes += 1;
      }
      continue;
    }
    t.cls[e.c] += 1;
    if (e.c === "claimed" || e.c === "verified") {
      const seen = e.c === "claimed" ? "claimedSeen" : "verifiedSeen";
      const yes = e.c === "claimed" ? "claimedYes" : "verifiedYes";
      t.confirm[seen] += 1;
      if (e.p === 1) t.confirm[yes] += 1;
    }
    const k = t.kind[e.c];
    k.seen += 1;
    if (e.k === "a") k.a += 1;
    else if (e.k === "b") k.b += 1;
    if (e.c === "verified" && t.fd) {
      t.fd.verifiedSeen += 1;
      if (isFoundingRank(e.f)) t.fd.byRank[String(e.f)] = (t.fd.byRank[String(e.f)] ?? 0) + 1;
    }
  }
}

/** Unit snapshot: what the survey froze for one previous-partition prefix.
 *  s: survivors, each with "is one of ours" and "is currently placed"
 *  w: arrivals, each with its admission lane (0 warning, 1 caution, 2 creator)
 *     and its order within that lane
 *  o: our other hashes here (occupied, so a decoy can never collide with one) */
interface StageUnit {
  s: [RegistryEntry, 0 | 1, 0 | 1][];
  w: [RegistryEntry, number, number][];
  o: string[];
}

function stageUnitKey(seq: number, prefix: string): string {
  return `${genPrefix(seq)}stage/u/${prefix}.json`;
}
function stageCautionKey(seq: number, hex: string): string {
  return `${genPrefix(seq)}stage/caution/${hex}.json`;
}
function stageLeavesKey(seq: number, subtree: number): string {
  return `${genPrefix(seq)}stage/leaves/${subtree}.json`;
}
function genBucketKey(seq: number, width: number, prefix: string): string {
  return `${genPrefix(seq)}buckets/${width}/${prefix}.json`;
}
function genProofKey(seq: number, width: number, prefix: string): string {
  return `${genPrefix(seq)}${proofObjectKey(width, prefix)}`;
}

interface Estimate { heap: number; d1: number; r2: number }

function fits(e: Estimate, b: StepBudget): boolean {
  return e.heap <= b.heapBytes && e.d1 <= b.d1Queries && e.r2 <= b.r2Ops;
}

/**
 * THE PREFLIGHT (ADR-0012 §4). Bounded counts predict this step's heap, D1 and
 * R2 use; over the declared budget it throws with ZERO writes, the served
 * generation is untouched, and `registry_capacity_exhausted` is the line to
 * alert on.
 */
function preflight(phase: string, seq: number, e: Estimate, b: StepBudget, cid: string): void {
  if (fits(e, b)) return;
  logError({
    cid, hop: "cron", msg: "registry_capacity_exhausted", kind: phase, seq,
    detail: `heap=${Math.round(e.heap / 2 ** 20)}/${Math.round(b.heapBytes / 2 ** 20)}MB d1=${e.d1}/${b.d1Queries} r2=${e.r2}/${b.r2Ops}`,
  });
  throw new RegistryCapacityError(
    `registry build step ${phase} needs ~${Math.round(e.heap / 2 ** 20)} MB heap, ${e.d1} D1 queries and ${e.r2} R2 ops against a declared step budget of ${Math.round(b.heapBytes / 2 ** 20)} MB, ${b.d1Queries} and ${b.r2Ops} — refusing to write anything`,
  );
}

interface RangeCounts { accounts: number; events: number; reports: number }

async function rangeCounts(db: DbLike, lo: string, hi: string): Promise<RangeCounts> {
  const r = await db
    .prepare(
      `SELECT (SELECT COUNT(*) FROM passport_accounts WHERE subject_hash >= ? AND subject_hash < ?) AS accounts,
              (SELECT COUNT(*) FROM trust_events WHERE event_type IN ('warning_published','warning_retracted') AND subject >= ? AND subject < ?) AS events,
              (SELECT COUNT(*) FROM fan_report_items WHERE subject_hash >= ? AND subject_hash < ?) AS reports`,
    )
    .bind(lo, hi, `sha256:${lo}`, `sha256:${hi}`, lo, hi)
    .first<RangeCounts>();
  return { accounts: Number(r?.accounts ?? 0), events: Number(r?.events ?? 0), reports: Number(r?.reports ?? 0) };
}

interface RangeData {
  accounts: AccountRow[];
  warnings: WarningRow[];
  reports: string[];
  proven: Set<string>;
  /** D-0442: publication event ids whose `fraud_at_proven_creator` review was APPROVED. */
  fraudApproved: Set<string>;
  /** D-0442: publication event ids that already carry a review item, in ANY state. */
  fraudReviewed: Set<string>;
}

/**
 * EVERY READ THIS BUILD MAKES OF THE LEDGER, BOUNDED BY ONE PREFIX RANGE
 * (ADR-0012 §5). Four queries, whatever the size of the registry and whatever
 * the number of impersonation pointers: the pointer enrichment is a LEFT JOIN
 * rather than two point reads per warning, and the proof pairs are bounded by
 * the same range through the accounts table rather than read whole.
 */
async function readRange(db: DbLike, lo: string, hi: string, nowMs: number): Promise<RangeData> {
  // D-1117(8) (I34-FX2-EXPIRY): the grant read is the grant IN FORCE at the build's instant (standing.ts
  // grantInForceOrderSql, imported at the file head), not the newest row.
  const { results: accounts } = await db
    .prepare(
      `SELECT a.account_key, a.id_kind, a.passport_id, a.status, a.subject_hash, a.acct_vetted, a.acct_asserted,
              p.claim_state, p.retracted, p.subject_kind,
              s.operator AS sup_operator, s.created_at AS sup_at,
              k.operator AS key_operator, k.created_at AS key_at,
              CASE WHEN p.claim_state = 'verified' THEN
                (SELECT g.payload FROM trust_events g
                  WHERE g.subject = a.passport_id AND g.event_type IN ('verification_granted', 'verification_renewed')
                  ${grantInForceOrderSql("g", instantSqlLiteral(nowMs))} LIMIT 1) END AS grant_payload,
              (SELECT fo.rank FROM founding fo WHERE fo.passport_id = a.passport_id) AS founding_rank
         FROM passport_accounts a
         JOIN passports p ON p.passport_id = a.passport_id
         LEFT JOIN request_suppressions s ON s.passport_id = a.passport_id
         LEFT JOIN suppressed_account_keys k ON k.account_key = a.account_key
        WHERE a.subject_hash >= ? AND a.subject_hash < ?
        ORDER BY a.rowid ASC`,
    )
    .bind(lo, hi)
    .all<AccountRow>();
  // The same freshness arithmetic `accountProofStanding` applies one account at
  // a time and `freshProofPairs` applies to the whole table — here to the rows
  // this step can actually publish. `proofIsCurrent` is imported, not restated.
  const { results: pairs } = await db
    .prepare(
      `SELECT c.passport_id AS passport_id, c.account_key AS account_key,
              COALESCE(c.proved_at, c.created_at) AS proved_at, c.valid_until AS valid_until
         FROM claim_challenges c
        WHERE c.status = 'used' AND c.purpose <> 'contest'
          AND c.account_key IN (SELECT account_key FROM passport_accounts WHERE subject_hash >= ? AND subject_hash < ?)
        UNION ALL
       SELECT o.passport_id AS passport_id, o.account_key AS account_key,
              COALESCE(o.proved_at, o.created_at) AS proved_at, o.valid_until AS valid_until
         FROM oauth_challenges o
        WHERE o.status = 'proven'
          AND o.account_key IN (SELECT account_key FROM passport_accounts WHERE subject_hash >= ? AND subject_hash < ?)`,
    )
    .bind(lo, hi, lo, hi)
    .all<{ passport_id: string; account_key: string; proved_at: string | null; valid_until: string | null }>();
  const proven = new Set<string>();
  for (const r of pairs) {
    if (typeof r.passport_id !== "string" || typeof r.account_key !== "string") continue;
    if (proofIsCurrent(r, nowMs)) proven.add(proofPairKey(r.passport_id, r.account_key));
  }
  const { results: warnings } = await db
    .prepare(
      `SELECT e.seq AS seq, e.event_id AS event_id, e.subject AS subject, e.event_type AS event_type, e.payload AS payload,
              p.passport_id AS ptr_pid, p.display_name AS ptr_name, p.retracted AS ptr_retracted,
              (SELECT 1 FROM request_suppressions rs WHERE rs.passport_id = p.passport_id) AS ptr_suppressed
         FROM trust_events e
         LEFT JOIN passports p
           ON e.event_type = 'warning_published' AND json_valid(e.payload) = 1
          AND p.passport_id = json_extract(e.payload, '$.protected_passport_id')
          AND ${operatorSetPointerSql("e")}
        WHERE e.event_type IN ('warning_published','warning_retracted') AND e.subject >= ? AND e.subject < ?
        ORDER BY e.seq ASC`,
    )
    .bind(`sha256:${lo}`, `sha256:${hi}`)
    .all<WarningRow>();
  const { results: reports } = await db
    .prepare("SELECT subject_hash FROM fan_report_items WHERE subject_hash >= ? AND subject_hash < ?")
    .bind(lo, hi)
    .all<{ subject_hash: string }>();
  // D-0442 (docket #49): the `fraud_at_proven_creator` decisions for the warned
  // subjects in this range, keyed by the PUBLICATION they decide (a re-publish
  // after a retraction is a new accusation and needs its own review).
  const { results: reviews } = await db
    .prepare(
      `SELECT json_extract(payload, '$.warning_event_id') AS ev, state, updated_at
         FROM review_queue
        WHERE item_type = 'dispute' AND json_valid(payload) = 1 AND json_extract(payload, '$.kind') = ?
          AND json_extract(payload, '$.warning_subject') >= ? AND json_extract(payload, '$.warning_subject') < ?`,
    )
    .bind(FRAUD_REVIEW_KIND, `sha256:${lo}`, `sha256:${hi}`)
    .all<{ ev: unknown; state: string; updated_at: string }>();
  // D-0810 (10), QUEUE #363: a publication is approved only when its LATEST
  // DECIDED review approves — `updated_at` is stamped by the deciding UPDATE
  // (review.ts approve/reject). Two decisions at the same latest instant that
  // disagree resolve to NOT approved: reject keeps the creator (D-0442 default).
  const fraudReviewed = new Set<string>();
  const latestDecided = new Map<string, { at: string; approved: boolean }>();
  for (const r of reviews) {
    if (typeof r.ev !== "string" || r.ev.length === 0) continue;
    fraudReviewed.add(r.ev);
    if (r.state !== "approved" && r.state !== "rejected") continue;
    const approved = r.state === "approved";
    const cur = latestDecided.get(r.ev);
    if (cur === undefined || r.updated_at > cur.at) latestDecided.set(r.ev, { at: r.updated_at, approved });
    else if (r.updated_at === cur.at && !approved) cur.approved = false;
  }
  const fraudApproved = new Set<string>();
  for (const [ev, d] of latestDecided) if (d.approved) fraudApproved.add(ev);
  return {
    accounts,
    warnings,
    proven,
    reports: reports.map((r) => r.subject_hash).filter((h) => /^[0-9a-f]{64}$/.test(h)),
    fraudApproved,
    fraudReviewed,
  };
}

/**
 * A LOST FENCE IS NOT AN ERROR AND IT IS NOT A RESULT — it is the end of this
 * builder's turn. It is thrown rather than returned because eight step
 * functions ended with a `commitStep(...)` whose boolean nobody read (INVAUDIT
 * A3): each of them would have carried on into the next phase on a build
 * another epoch owns. A throw cannot be ignored by forgetting to read it.
 */
class FencedSignal extends Error {
  override name = "FencedSignal";
}

interface Ctx {
  db: DbLike;
  store: ObjectStore;
  seed: string;
  plan: Plan;
  agg: Agg;
  row: BuildRow;
  epoch: number;
  cid: string;
  signer?: RegistrySigner | undefined;
  safetyChunk: number; // hashes per OR-01 bookkeeping statement (`SAFETY_CHUNK`, or opts.safetyChunk)
}

const wallIso = (): string => new Date().toISOString();

/** Commit this step's work: the cursor, the aggregates and the phase, under the
 * epoch fence. `false` means this builder was superseded and must stop — its own
 * writes went into its own generation prefix, which nothing points at. */
/**
 * THE LADDER OBJECT IS WRITTEN OUTSIDE THE FENCE, and this is why that is safe
 * (REG-10). It is an R2 object, so no D1 statement can carry it; what guards it
 * is the shape of the value, not the timing of the write.
 *
 * A superseded builder can only write its OWN `{seq, hw, frontier}`, and every
 * reader of that object takes `Math.max` against the MANIFEST's copy of both
 * monotone counters (`loadLadder`): a predecessor writing a LOWER seq or
 * frontier cannot lower either floor, because the pointer still carries the
 * winner's. The third number, `hw`, has no manifest copy — and is trusted only
 * when the ladder's `seq` equals the served manifest's, so a predecessor's write
 * makes it look STALE rather than wrong, and a stale `hw` falls back to the
 * floor the published rung implies. That costs listing throughput for one
 * generation and can never step the rung early, which is the only direction that
 * would matter. Both counters rolled back at once is the case the generation
 * markers catch at plan time.
 *
 * `assertStillOurs` narrows the window anyway: it is a fenced no-op advance
 * immediately before each ladder write, so a builder that has already been
 * superseded stops instead of writing. It cannot CLOSE the window — the two
 * systems cannot be written atomically — which is what the paragraph above is
 * for.
 */
/** THE ONLY WRITER OF THE LADDER OBJECT. Field order is fixed so two builds that
 * compute the same state write the same bytes; `bhw` is omitted, never null, when
 * there is no vector to carry. */
async function putLadder(store: ObjectStore, f: { seq: number; hw: number; frontier: number }, bhw: Uint16Array | null): Promise<void> {
  await store.put(LADDER_KEY, JSON.stringify({ v: LADDER_VERSION, seq: f.seq, hw: f.hw, frontier: f.frontier, ...(bhw === null ? {} : { bhw: encodeBhw(bhw) }) }));
}

async function assertStillOurs(c: Ctx): Promise<void> {
  if (!(await advanceBuild(c.db, c.plan.seq, c.epoch, {}, wallIso()))) {
    throw new FencedSignal(`registry build ${c.plan.seq} moved on from epoch ${c.epoch} — not writing the ladder object`);
  }
}

/** A fraud-class publication held by D-0442 that has no review item yet. */
interface HeldFraud {
  passportId: string;
  subject: string;
  cls: string;
  ev: string;
}

/**
 * D-0442: "the pending row is keyed by subject hash so the 6-hourly projection
 * re-derives instead of re-enqueueing". `warnings.ts publishWarning` writes the
 * review item at publish when its own lookup finds the proven creator (WARN49);
 * a publication it missed — a truncated lookup, a creator proven after the
 * publish, a warning older than WARN49 — is held here all the same, and this is
 * where its review item comes from, so every held creator has a NEXT ACTOR (the
 * operator queue) rather than a hold nobody can decide. Keyed by the
 * publication's event id: NOT EXISTS over EVERY state, so a replayed step writes
 * nothing twice and a decided (approved or rejected) review is never re-opened.
 * The audit line is written only when the row it names exists.
 */
async function deriveFraudReviews(db: DbLike, held: HeldFraud[], seq: number): Promise<void> {
  const ts = wallIso();
  const stmts = held.flatMap((h) => {
    const itemId = mintId("itm"); // the same id shape `review.ts enqueue` mints
    const payload = JSON.stringify({ kind: FRAUD_REVIEW_KIND, warning_class: h.cls, warning_subject: h.subject, warning_event_id: h.ev, derived_by: `registry_build:${seq}` });
    return [
      db
        .prepare(
          `INSERT INTO review_queue (item_id, item_type, state, subject, payload, created_at, updated_at)
           SELECT ?, 'dispute', 'open', ?, ?, ?, ?
            WHERE NOT EXISTS (SELECT 1 FROM review_queue WHERE item_type = 'dispute' AND json_valid(payload) = 1
                                 AND json_extract(payload, '$.kind') = ? AND json_extract(payload, '$.warning_event_id') = ?)
`,
        )
        .bind(itemId, h.passportId, payload, ts, ts, FRAUD_REVIEW_KIND, h.ev),
      db
        .prepare(
          `INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind)
           SELECT 'system:registry-build', 'enqueue:dispute', ?, ?, ?, 'system'
            WHERE EXISTS (SELECT 1 FROM review_queue WHERE item_id = ?)`,
        )
        .bind(itemId, ts, itemId, itemId), // I34-W3-SEAM-P (D-1119(8), D-1120(6)): the review-queue audit copy keeps only the item id
    ];
  });
  await db.batch(stmts);
}

async function commitStep(c: Ctx, a: Advance): Promise<void> {
  if (!(await advanceBuild(c.db, c.plan.seq, c.epoch, { ...a, agg: JSON.stringify(c.agg), resetFailures: true }, wallIso()))) {
    throw new FencedSignal(`registry build ${c.plan.seq} moved on from epoch ${c.epoch} — this builder stops`);
  }
}

// --- SURVEY ---------------------------------------------------------------

/**
 * PASS 1 — DEPARTURES, on the PREVIOUS partition, exactly `churn` per bucket.
 *
 * The previous partition, not this one: at a width change the two do not line
 * up, and the observer's diff is anchored on the buckets that were actually
 * published last time. Doing it here also makes the quota FEASIBLE by induction
 * — the last build left every bucket at least `churn` decoys (`capacityFor`),
 * whatever this build's width turns out to be.
 *
 * RETRACTED ACCUSATIONS LEAVE FIRST AND OUTSIDE THE QUOTA. A retracted warning
 * is `ours` and is not placed, so it used to land in `leaving` behind every
 * removal request and capped with them — and a kept entry is republished as the
 * OLD ENTRY OBJECT, so a fan kept being shown `c:"warning"`, pointer and all,
 * for a warning an operator had already withdrawn. A retraction has nothing to
 * hide (the subject's hash is already published and its class is already
 * `warning` in bytes anyone can fetch), so it is shed OUTSIDE the quota:
 * `dropped` and `retire` are untouched, the removal crowd is exactly the size it
 * was, and the guaranteed decoy half is exactly as unconsumable.
 *
 * Real departures are then taken by LANE and oldest request first, capped at
 * `realChurn` — NOT at `churn`. The gap is the guaranteed decoy retirement:
 * capping real departures at the full quota let anyone who could mint departures
 * (list a handle, then revoke it) take every slot but the victim's, leaving the
 * delisted subject alone in the diff. Anything over the cap stays published:
 * that is the delisting SLA. Decoys make up the balance, retired uniformly at
 * random from the bucket's live decoys under the secret seed (`retirementPick`),
 * so no outsider can cause, predict or recognise one.
 */
async function pass1(
  c: Ctx,
  prefix: string,
  entries: RegistryEntry[],
  ours: Set<string>,
  placedByHash: Map<string, RegistryEntry>,
  removalOrder: Map<string, string>,
): Promise<{ survivors: [RegistryEntry, 0 | 1, 0 | 1][]; spilledOut: number; spilledOutRetractions: number }> {
  const { churn, realChurn, seq } = c.plan;
  const leaving: RegistryEntry[] = [];
  const retracted: RegistryEntry[] = [];
  const decoys: RegistryEntry[] = [];
  const keep: RegistryEntry[] = [];
  for (const e of entries) {
    if (!ours.has(e.h)) decoys.push(e);
    else if (placedByHash.has(e.h)) keep.push(e);
    // A PUBLISHED ACCUSATION THAT STOPPED BEING ONE gets its own lane, ahead of
    // every removal and outside the real-departure cap. Caution rides the same
    // lane as warning, deliberately: an expired or killed caution is the same
    // act — an accusation withdrawn — and the same argument applies word for word.
    else if (e.c === "warning" || e.c === "caution") retracted.push(e);
    else leaving.push(e);
  }
  // Lane 0 (operator-decided suppression) < lane 1 (self-serve) < lane 2
  // (everything else that stopped being real); FIFO by request time inside a
  // lane. `removalOrder` carries the lane prefix; a leaver with no suppression
  // row is lane 2.
  leaving.sort((a, b) => {
    const ka = removalOrder.get(a.h) ?? `2|${a.h}`;
    const kb = removalOrder.get(b.h) ?? `2|${b.h}`;
    return ka < kb ? -1 : ka > kb ? 1 : 0;
  });
  decoys.sort((a, b) => (a.h < b.h ? -1 : a.h > b.h ? 1 : 0));
  const dropped = Math.min(realChurn, leaving.length);
  const spilledOut = leaving.length - dropped;
  // The retraction lane has no cap, so it always sheds all of itself and the
  // spill is zero. Written as the subtraction rather than as a literal so the
  // day somebody caps this lane the number an operator reads is "N accounts are
  // still being accused" instead of silence.
  const spilledOutRetractions = retracted.length - retracted.length;
  // Over the quota: still published, unchanged, waiting for the next build.
  for (let i = dropped; i < leaving.length; i++) keep.push(leaving[i]!);
  // >= churn - realChurn = decoyChurnFor(prev.padTarget), by construction.
  const retire = churn - dropped;
  if (retire > decoys.length) {
    // UNREACHABLE WHILE THE FEASIBILITY FLOOR HELD AT THE WIDTH THAT PUBLISHED
    // THIS BUCKET. `capacityFor` alone is not the whole story and the comment used
    // to say it was: it caps ADMISSIONS (`composeBucket`), while a width change
    // carries a parent's survivors into a child unconditionally, so a child's
    // non-decoy count is bounded by the rung's reserve term and by nothing else.
    // That term is `rungReserving(hwFloor(w))` in `resolveRung`, which guarantees
    // `pad - churnFor(pad) >= hwFloor(w) >= ` this bucket's non-decoy count, hence
    // `decoys >= churnFor(pad) = churn >= retire`. Fail closed rather than shed
    // fewer, because shedding fewer is exactly the signal this mechanism removes.
    throw new RegistryBuildError(`bucket ${prefix} holds ${decoys.length} decoys but owes ${retire} retirements — refusing to publish a short churn`);
  }
  for (let r = 0; r < retire; r++) {
    decoys.splice(await retirementPick(c.seed, prefix, seq, r, decoys.length), 1);
  }
  const survivors: [RegistryEntry, 0 | 1, 0 | 1][] = [];
  // A survivor that is still real is republished at its CURRENT class (a creator
  // who claimed or was verified since). Same hash, so it is neither an arrival
  // nor a departure — and an entry whose class changes is therefore certainly a
  // real one, which is declared in the doc.
  for (const e of keep) survivors.push([placedByHash.get(e.h) ?? e, 1, placedByHash.has(e.h) ? 1 : 0]);
  for (const e of decoys) survivors.push([e, 0, 0]);
  return { survivors, spilledOut, spilledOutRetractions };
}

async function stepSurvey(c: Ctx): Promise<void> {
  const { plan, agg } = c;
  const u = plan.unitWidth;
  const units = 16 ** u;
  const start = c.row.cursor;
  const prevPad = plan.prev?.padTarget ?? 0;
  const perUnit = plan.accounts / units;
  let k = Math.max(
    1,
    Math.min(units - start, Math.floor((c.plan.budget.r2Ops - 16) / 2), Math.floor((plan.budget.heapBytes - EST_BASE) / Math.max(1, perUnit * EST_ROW + prevPad * EST_ENTRY * 3))),
  );
  let counts = { accounts: 0, events: 0, reports: 0 };
  let lo = "";
  let hi = "";
  for (let attempt = 0; ; attempt++) {
    [lo, hi] = hexRange(start, start + k, u);
    counts = await rangeCounts(c.db, lo, hi);
    const est: Estimate = {
      heap: EST_BASE + counts.accounts * EST_ROW + counts.events * EST_EVENT + counts.reports * 100 + k * prevPad * EST_ENTRY * 3,
      d1: 8, // + the D-0442 review read and its one derive batch
      r2: 2 * k + (plan.flags.serveCaution ? 16 : 0),
    };
    if (fits(est, plan.budget) || k === 1 || attempt >= 4) {
      preflight("survey", plan.seq, est, plan.budget, c.cid);
      break;
    }
    k = Math.max(1, Math.floor(k / 2));
  }
  const data = await readRange(c.db, lo, hi, plan.nowMs);
  const cautionShards = new Map<string, RegistryEntry[]>();
  if (plan.flags.serveCaution) {
    for (let i = start; i < start + k; i++) {
      const hex = hexOf(i, u)[0]!;
      if (cautionShards.has(hex)) continue;
      const raw = await c.store.get(stageCautionKey(plan.seq, hex));
      cautionShards.set(hex, raw === null ? [] : (JSON.parse(raw) as RegistryEntry[]));
    }
  }
  // Group the range's rows by unit ONCE: a step reads a contiguous range with
  // four queries and then works unit by unit, so the query count per step is a
  // constant and never a function of the registry's size.
  const byUnit = <T>(rows: T[], keyOf: (r: T) => string): Map<string, T[]> => {
    const m = new Map<string, T[]>();
    for (const r of rows) {
      const p = keyOf(r);
      const cur = m.get(p);
      if (cur) cur.push(r);
      else m.set(p, [r]);
    }
    return m;
  };
  const accountsBy = byUnit(data.accounts, (r) => r.subject_hash.slice(0, u));
  const warningsBy = byUnit(data.warnings, (r) => r.subject.slice(7, 7 + u));
  const reportsBy = byUnit(data.reports, (h) => h.slice(0, u));

  // THE PER-BUCKET LISTING HIGH-WATER this step observes, at MAX_WIDTH (see
  // `BHW_LEAVES`). Folded into the ladder object by MAX after the unit loop, so a
  // replayed step writes the same bytes.
  const bhwLeaves = new Map<string, number>();
  // D-0442: held publications that carry no review item yet — derived once per
  // step after the unit loop (see `deriveFraudReviews`).
  const fraudToDerive: HeldFraud[] = [];

  for (let i = start; i < start + k; i++) {
    const p = hexOf(i, u);
    const rows = accountsBy.get(p) ?? [];
    const wrows = warningsBy.get(p) ?? [];
    const real = realEntriesFrom(rows, data.proven, plan.flags.perAccountProof, plan.nowMs);
    const realCreator = real.published;
    // INV-027, docket #4: WARNING > CREATOR > CAUTION. A PUBLISHED WARNING
    // OVERRIDES EVERYTHING — except the D-0442 hold below (a fraud-class warning
    // at a proven creator's hash waits for an approved review). `precedence` is
    // still the set that matters for CAUTION (it includes the SUPPRESSED creators, so a creator who asked to be
    // delisted does not thereby become cautionable).
    const creatorHashes = real.precedence;
    // THE ONE EXEMPTION, and it is INV-003's, not INV-027's: a hash whose creator
    // asked to be DELISTED is served NEITHER way. Subtracting the suppressed set
    // is what keeps a REMOVAL REQUEST from being converted into a published
    // ACCUSATION. Deriving it as `precedence MINUS published` is sound: the ONLY
    // `continue` between `precedence.add(h)` and `published.push(...)` is the
    // suppression one; every other reason a hash is unserved drops it BEFORE
    // `precedence.add`, so it is in neither set.
    const publishedHashes = new Set(realCreator.map((e) => e.h));
    const suppressedHashes = new Set([...creatorHashes].filter((h) => !publishedHashes.has(h)));
    // D-0442 (docket #49, Jason 2026-09-18): WHILE A FRAUD-CLASS WARNING AT A
    // PROVEN CREATOR'S OWN HASH IS UNDER REVIEW, THE REGISTRY SERVES THE CREATOR.
    // "Proven creator" = a PUBLISHED creator entry whose class is not `unclaimed`.
    // The warning is held (not served, not lost) until a named operator APPROVES
    // the `fraud_at_proven_creator` review of THAT publication; reject, or no
    // decision at all, keeps the creator. Nothing here reads a clock: there is no
    // timeout that delists. `impersonation` is untouched — the publish door
    // already refuses it at a registered creator account (warnings.ts).
    const provenCreatorHashes = new Set(realCreator.filter((e) => e.c !== "unclaimed").map((e) => e.h));
    const warnings = warningEntriesFrom(wrows, plan.flags.serveConfirmedFraud).filter((w) => {
      if (suppressedHashes.has(w.e.h)) return false;
      if (w.cls === "impersonation" || !provenCreatorHashes.has(w.e.h) || data.fraudApproved.has(w.ev)) return true;
      if (!data.fraudReviewed.has(w.ev)) {
        const acct = rows.find((r) => r.subject_hash === w.e.h && r.status === "active" && r.retracted === 0);
        if (acct) fraudToDerive.push({ passportId: acct.passport_id, subject: `sha256:${w.e.h}`, cls: w.cls, ev: w.ev });
      }
      return false;
    });
    const warnedHashes = new Set(warnings.map((w) => w.e.h));
    // Caution (soft class) — the real ones, gated behind serve_caution (default
    // ON: unset reads as served, D-0534, settings.ts servingCaution); decoys take the class only while it is served (`classThresholdsFrom`). Precedence WARNING > CAUTION > creator.
    const caution = (cautionShards.get(p[0]!) ?? []).filter((e) => e.h.slice(0, u) === p && !creatorHashes.has(e.h) && !warnedHashes.has(e.h));
    // WARNING > CREATOR, executed BEFORE placement rather than by ordering: the
    // placement map is last-wins with creators ordered last, so leaving the
    // creator entry in and relying on arrival order would still serve the creator
    // at the warned hash. The entry must be DROPPED.
    const creatorPlaced = realCreator.filter((e) => !warnedHashes.has(e.h));
    const collisions = realCreator.length - creatorPlaced.length;
    if (collisions > 0) {
      // Either an impersonator who got a listing before the warning caught up, or
      // a warning published against a real creator — both are things a human has
      // to look at, and neither is visible in the served bytes. The HASH is not
      // logged (INV-061): the count is the operator signal, the hash is not.
      logError({ cid: c.cid, hop: "cron", msg: "registry_precedence_collision", seq: plan.seq, count: collisions });
    }
    // ARRIVAL PRIORITY: safety classes are admitted FIRST — not "never wait": a
    // safety entry that does not fit even the reserved lane is spilled, recorded
    // `pending` and steps the rung next build (OR-01; `BuildResult.spilledInWarnings`
    // / `spilledInCautions`). lane 0 warnings, lane 1 cautions, lane 2 creator listings, each in a stable order of its own so
    // that a bucket assembled from several units admits in the same order
    // whatever the chunking.
    const placedRanked: [RegistryEntry, number, number][] = [
      ...warnings.map((w) => [w.e, 0, w.ord] as [RegistryEntry, number, number]),
      ...caution.map((e, idx) => [e, 1, idx] as [RegistryEntry, number, number]),
      ...creatorPlaced.map((e, idx) => [e, 2, idx] as [RegistryEntry, number, number]),
    ];
    const placedByHash = new Map(placedRanked.map(([e]) => [e.h, e]));
    // `ours` = every hash we have ever had a real reason to publish. Its
    // COMPLEMENT within a published bucket is exactly our decoys, which is how a
    // departure that is a delisted subject is told apart from a retired decoy.
    const ours = real.ours;
    for (const r of wrows) {
      const m = /^sha256:([0-9a-f]{64})$/.exec(r.subject);
      if (m) ours.add(m[1]!);
    }
    for (const h of reportsBy.get(p) ?? []) ours.add(h);
    for (const [e] of placedRanked) ours.add(e.h);

    let prevEntries: RegistryEntry[] = [];
    if (plan.prev) {
      const raw = await c.store.get(`${plan.prev.prefix}buckets/${plan.prev.width}/${p}.json`);
      // The same `torn` predicate the digest check below uses, decided BEFORE the
      // bucket is read: a torn pad growth used to refuse here, so that branch was
      // unreachable for exactly the case it exists for (I3-1362).
      const prevPad = plan.prev.padTarget;
      const torn = plan.prev.v === 2 && plan.ladder.seqFloor > plan.prev.seq;
      const tornPadOk = torn
        ? (n: number): boolean => n > prevPad && n <= Math.max(PAD_ENTRY_CEILING, 2 * prevPad) && (agg.tornPad === undefined || agg.tornPad === n)
        : undefined;
      prevEntries = validateBucket(p, plan.prev.width, prevPad, raw, tornPadOk);
      if (prevEntries.length !== prevPad) agg.tornPad = prevEntries.length;
      const part = plan.prev.v === 3 ? (await bucketLeaf(plan.prev.width, p, raw!)) ?? "" : raw!;
      agg.prevDigest = await digestStep(agg.prevDigest ?? digestSeed(plan.prev.width, plan.prev.v === 3), p, part);
    }
    const prevPublished = new Set(prevEntries.map((e) => e.h));
    const { survivors, spilledOut, spilledOutRetractions } = await pass1(c, p, prevEntries, ours, placedByHash, real.removalOrder);
    // THE LADDER IS DRIVEN BY LISTINGS ALONE. `wantIn` is what wants admitting
    // now; neither it nor `hw` can be moved by a removal, so neither can a rung
    // crossing or a width change.
    const wantIn = placedRanked.filter(([e]) => !prevPublished.has(e.h));
    const survivorHashes = new Set(survivors.map(([e]) => e.h));
    const wantHashes = new Set(wantIn.map(([e]) => e.h));
    const stage: StageUnit = {
      s: survivors,
      w: wantIn,
      o: [...ours].filter((h) => !survivorHashes.has(h) && !wantHashes.has(h)),
    };
    await c.store.put(stageUnitKey(plan.seq, p), JSON.stringify(stage));

    agg.wantIn += wantIn.length;
    agg.spilledOut += spilledOut;
    agg.spilledOutRetractions += spilledOutRetractions;
    agg.precedenceCollisions += collisions;
    tallyInto(agg.tally, realCreator);
    tallyInto(agg.tally, [...warnings.map((w) => w.e), ...caution]);
    // What the rung needs, at every width this build could choose: the fullest
    // bucket, the fullest by NON-DECOY count, and the emptiest (which sets the
    // mint block). Counted here because the survivors are in hand.
    for (let w = u; w <= MAX_WIDTH; w++) {
      const surv = new Map<string, number>();
      const nd = new Map<string, number>();
      for (const [e, isOurs] of survivors) {
        const q = e.h.slice(0, w);
        surv.set(q, (surv.get(q) ?? 0) + 1);
        if (isOurs) nd.set(q, (nd.get(q) ?? 0) + 1);
      }
      const children = 16 ** (w - u);
      let maxSurv = 0;
      let minSurv = surv.size < children ? 0 : Number.MAX_SAFE_INTEGER;
      let maxND = 0;
      for (const v of surv.values()) {
        if (v > maxSurv) maxSurv = v;
        if (v < minSurv) minSurv = v;
      }
      for (const v of nd.values()) if (v > maxND) maxND = v;
      if (minSurv === Number.MAX_SAFE_INTEGER) minSurv = 0;
      // AT MAX_WIDTH, the same `nd` map IS the per-leaf non-decoy count, and the
      // arrivals that want in are the listings that advance the high-water. Both
      // are listing-side; the high-water is the MAX of this against what the
      // vector already holds, so no departure can lower it.
      if (w === MAX_WIDTH) {
        for (const [q, v] of nd) bhwLeaves.set(q, Math.max(bhwLeaves.get(q) ?? 0, v));
        const arrivals = new Map<string, number>();
        for (const [e] of wantIn) {
          const q = e.h.slice(0, MAX_WIDTH);
          arrivals.set(q, (arrivals.get(q) ?? 0) + 1);
        }
        for (const [q, v] of arrivals) bhwLeaves.set(q, Math.max(bhwLeaves.get(q) ?? 0, (nd.get(q) ?? 0) + v));
      }
      const cur = agg.byWidth[String(w)] ?? { maxSurv: 0, maxND: 0, minSurv: Number.MAX_SAFE_INTEGER };
      agg.byWidth[String(w)] = { maxSurv: Math.max(cur.maxSurv, maxSurv), maxND: Math.max(cur.maxND, maxND), minSurv: Math.min(cur.minSurv, minSurv) };
    }
  }
  const done = start + k >= units;
  if (done && plan.prev) {
    // ---------------------------------------------------------------------
    // DO THE MANIFEST AND THE BUCKETS DESCRIBE THE SAME BUILD?
    //
    // The manifest carries a digest of the generation it was written with, so a
    // manifest that describes a different build is detectable. Under the
    // generation layout the only benign mismatch a v2 store can still show is a
    // TORN WRITE — the frontier was reserved before the buckets, so a crash after
    // they landed leaves a ladder object whose `seq` is strictly newer than the
    // manifest's. Anything else is a rollback, the frontier is genuinely lost,
    // and re-minting is worse than not publishing: a decoy retired in a previous
    // pass 1 comes straight back and never departs, so the bucket sheds FEWER
    // than `churn`, and a short churn PROVES the bucket had fewer than `churn`
    // real removals. Fail closed; the published generation keeps serving.
    // ---------------------------------------------------------------------
    const torn = plan.prev.v === 2 && plan.ladder.seqFloor > plan.prev.seq;
    if (plan.prev.manifestDigest !== null && plan.prev.manifestDigest !== agg.prevDigest && !torn) {
      throw new RegistryBuildError(
        "registry manifest does not describe the published buckets and no newer reservation exists — refusing to rebuild from a rolled-back decoy frontier (see DEPLOY.md 3e)",
      );
    }
  }
  // FOLD BY MAX. Read-modify-MAX and never read-modify-add: re-running this step
  // after a kill re-applies the same maxima and the object is byte-identical, which
  // is ADR-0012's resume property. The write takes the same fence as every other
  // ladder write (`assertStillOurs`).
  if (bhwLeaves.size > 0) {
    await assertStillOurs(c);
    await putLadder(c.store, { seq: plan.seq, hw: plan.ladder.hw, frontier: plan.ladder.frontierFloor }, foldBhw(await readBhw(c.store), bhwLeaves));
  }
  // The operator signal for a hold is its review item in the ops queue (derived
  // here when publish did not write one) — not a log line, and never the hash.
  if (fraudToDerive.length > 0) await deriveFraudReviews(c.db, fraudToDerive, plan.seq);
  await commitStep(c, { cursor: done ? 0 : start + k, phase: done ? "rung" : "survey" });
}

/**
 * THE RUNG AND THE WIDTH, as one pure function of the frozen plan quantities.
 *
 * EXTRACTED FROM `stepRung` AND EXPORTED so the rung terms are reachable without
 * a `Ctx` (QUEUE #359: nothing exercised this path). `stepRung` is its only
 * caller in src/; the guard is `tests/invariants/registry-build.test.ts`
 * › "stepRung > rungAt".
 *
 * Three terms, and each has to be there:
 *  - the LADDER term is the one that governs: capacity for the per-bucket
 *    listing high-water, which no removal can move.
 *  - the SURVIVOR term keeps a bucket from shrinking below what it already
 *    publishes, which is also what pins `padTarget` monotone within a width.
 *  - the RESERVE term is the feasibility floor for the one case induction does
 *    not cover: a width change hands a child bucket its parent's real entries
 *    against a rung sixteen times smaller. It reads private state, so it is
 *    deliberately the LOOSEST of the three and only ever RAISES the rung.
 *
 * THE RESERVE TERM IS THE FEASIBILITY FLOOR, AND IT IS TAKEN OFF A MONOTONE
 * HIGH-WATER (QUEUE #357; the residual is D-0439, ADR-0012 > Amendments).
 * `rungReserving(n)` guarantees `rung - churnFor(rung) >= n`: the bucket keeps at
 * least its churn quota in decoys. `n` has to dominate the bucket's non-decoy
 * count, and the LIVE count is a number a removal lowers — reading it made the
 * rung, and through the loop below the WIDTH, a one-bit removal oracle (D-0439),
 * and the one-line mitigation that stopped reading it (`rungAtLeast(maxND)`)
 * dropped the floor instead: 102 of 3,960 shapes then published a bucket with
 * fewer decoys than its quota (INVAUDIT A2). So `n` is `hwFloor(w)` — the largest
 * per-bucket LISTING HIGH-WATER at this width, carried at MAX_WIDTH and
 * prefix-summed (`BHW_LEAVES`). A high-water is monotone, so `hwFloor(w)` >= the
 * live non-decoy count of every bucket at `w` and the floor holds, while nothing
 * a removal can move is read at all. NOTHING IN THIS FUNCTION READS `maxND`.
 *
 * WHAT REMAINS, AND IT IS LOGGED. The SURVIVOR term is the last reader of live
 * per-bucket state. At a FIXED width it is a constant — every bucket sheds exactly
 * `churn`, so `maxSurv` is identically `pad_prev - churn` — and only a width
 * change, where survivors re-partition into children, can make it move. It is
 * therefore split out of `fromLadder` deliberately: `pad > fromLadder` is true
 * EXACTLY when it binds above the listing floor, which is the gate on
 * `registry_rung_reserve_step` in `stepRung`. The declared residual stays declared
 * and stays loud (D-0439 bound (3)).
 */
export function resolveRung(
  hwTarget: number,
  kFloor: number,
  prevWidth: number,
  padCeiling: number,
  bw: (w: number) => WidthAgg,
  safetyFallback: boolean,
  hwFloor: (w: number) => number,
): { width: number; pad: number; fromLadder: number } {
  const rungAt = (w: number): { pad: number; fromLadder: number } => {
    const mean = hwTarget / 16 ** w;
    let fromLadder = Math.max(padTargetFor(mean, kFloor, w), rungReserving(hwFloor(w), kFloor));
    // OR-01 (ii), the automatic fallback: a safety entry the reserved lane could
    // not admit, still pending after a published build, steps the rung.
    //
    // THE FIRST DECLARED RESIDUAL (D-0438, ADR-0012 › Amendments). "Width and the
    // pad ladder are a function of the listing high-water alone" is not true
    // without a qualifier: this term is private state, it can raise the rung, and
    // a raised rung can cross the file ceiling and take the WIDTH with it. What an
    // observer learns is ONE BIT PER BUILD — "some bucket could not admit a safety
    // entry last generation". It is NOT a removal oracle: the term is driven by
    // ARRIVALS the lane could not fit, and no departure can set it, clear it or
    // change its timing. Jason ruled for the fallback (ADR-0012 §6, option (ii))
    // against eviction, whose residual is a per-bucket bit about real occupancy
    // every build.
    if (safetyFallback) fromLadder *= PAD_LADDER_RATIO;
    // THE ONE LIVE-STATE TERM LEFT, and it is outside `fromLadder` so the log in
    // `stepRung` fires exactly when it binds. It keeps a bucket from shrinking
    // below what it already publishes, which is also what pins `padTarget`
    // monotone within a width.
    return { pad: Math.max(fromLadder, rungAtLeast(bw(w).maxSurv, kFloor)), fromLadder };
  };
  let width = Math.max(prevWidth, chooseWidth(hwTarget, kFloor, padCeiling));
  let r = rungAt(width);
  // EARLIER WIDENING (ADR-0012 ruling): a bucket file stays at or under the
  // 4,000-entry rung — about 356 KB, under the Android client's 512 KiB cap — so
  // when the plan would need more than that at this width, the width grows by
  // one instead. `chooseWidth` already applies the same ceiling to the listing
  // high-water; this loop covers the two terms above that the high-water does
  // not: the survivors of a width change and the reserve floor.
  while (width < MAX_WIDTH && r.pad > Math.max(padCeiling, kFloor)) {
    width++;
    r = rungAt(width);
  }
  return { width, pad: r.pad, fromLadder: r.fromLadder };
}

// --- RUNG (the one global freeze between the survey and any output) --------

async function stepRung(c: Ctx): Promise<void> {
  const { plan, agg } = c;
  const kFloor = plan.kFloor;
  const hwTarget = plan.ladder.hw + agg.wantIn;
  const bw = (w: number): WidthAgg => agg.byWidth[String(w)] ?? { maxSurv: 0, maxND: 0, minSurv: 0 };
  // THE FEASIBILITY FLOOR'S ONLY INPUT: the per-bucket listing high-water the
  // survey folded into the ladder object, prefix-summed once here. A stale vector
  // was already dropped at plan time by `loadLadder`'s freshness rule, so what is
  // here is this build's own; absent or the wrong length falls back to
  // `capacityFor(prev.padTarget)` — published, and exactly sufficient (BHW_LEAVES).
  const vec = await readBhw(c.store);
  const cum = vec === null ? null : bhwPrefixSums(vec);
  const floorFallback = plan.prev === null ? 0 : capacityFor(plan.prev.padTarget);
  const hwFloor = (w: number): number => bucketListingFloor(cum, w, floorFallback);
  // The rung and the width: `resolveRung` above (the ladder term, the listing
  // floor, and the one live-state term the log below declares).
  const r = resolveRung(hwTarget, kFloor, plan.prev?.width ?? 1, plan.padCeiling, bw, !!plan.safetyFallback, hwFloor);
  const width = r.width;
  const pad = r.pad;
  // AT MAX_WIDTH THE CEILING IS A WISH, AND IT SAYS SO. Above width 4 there is
  // no wider partition to move the volume into — MAX_WIDTH is aligned to the
  // clients' own MAX_CLIENT_WIDTH, so a fifth hex digit is a prefix no shipped
  // app can address. The rung therefore keeps growing and the bucket file goes
  // past the declared 4,000 entries (~356 KB), which is the size the Android
  // client's 512 KiB response cap is measured against. Publishing the bigger
  // file is the least-bad of the three: refusing the build stops publication
  // outright (warnings included), and capping the rung breaks the padding floor
  // every privacy property here rests on. So it publishes, and it ALARMS —
  // this is the line that says the client envelope is being exceeded, and the
  // remedy is a decision (raise MAX_WIDTH across the clients, or raise K_FLOOR),
  // never something this builder may take on its own.
  if (pad > Math.max(plan.padCeiling, kFloor)) {
    logError({
      cid: c.cid, hop: "cron", msg: "registry_bucket_ceiling_breached", seq: plan.seq, count: pad, cap: Math.max(plan.padCeiling, kFloor),
      kind: `w${width}`, detail: `rung ${pad} above the declared ${Math.max(plan.padCeiling, kFloor)}-entry file ceiling at MAX_WIDTH ${MAX_WIDTH} — clients may refuse the bucket file`,
    });
  }
  // THE FLOOR'S OWN PREMISE, NAMED AND LOGGED. `hwFloor(w) >= maxND(w)` holds by
  // construction — the survey folds this build's own per-leaf non-decoy counts into
  // the vector BEFORE this step reads it, and the fallback is a published cap that
  // dominates — so this never fires. It is here because the premise is what makes
  // the throw in `pass1` unreachable, and a premise nothing checks is how a
  // dropped floor stayed invisible for a whole build (D-0439, INVAUDIT A2). It
  // LOGS and does not throw: refusing the build stops publication outright,
  // warnings included, which the comment above rejects as the least acceptable
  // option.
  if (hwFloor(width) < bw(width).maxND) {
    logError({ cid: c.cid, hop: "cron", msg: "registry_listing_floor_short", seq: plan.seq, count: bw(width).maxND, cap: hwFloor(width), kind: `w${width}` });
  }
  if (pad > r.fromLadder) {
    // THE DECLARED RESIDUAL, AND LOUD ON PURPOSE (D-0439 bound (3), kept alive by
    // QUEUE #357's closure). The listing floor and the ladder are both in
    // `fromLadder`, so this is true only when the SURVIVOR term — the last term
    // that reads live per-bucket state — stepped the rung above what listings
    // alone asked for. It can happen only at a width change, where survivors
    // re-partition into children; at a fixed width `maxSurv` is identically
    // `pad_prev - churn` and cannot move. If it is happening routinely the ladder's
    // slack is wrong and the ladder, not this term, is what needs fixing.
    logError({ cid: c.cid, hop: "cron", msg: "registry_rung_reserve_step", seq: plan.seq, count: bw(width).maxSurv, cap: pad });
  }
  const maxSlots = Math.max(0, pad - bw(width).minSurv);
  const frontier = plan.ladder.frontierFloor;
  const block = mintBlockFor(width, maxSlots);
  const founding = foundingRatesFrom(agg.tally);
  const rung: Rung = {
    width,
    pad,
    nextChurn: churnFor(pad),
    cap: capacityFor(pad),
    creatorCap: creatorCapacityFor(pad),
    frontier,
    block,
    perBucket: perBucketBlock(maxSlots),
    hwTarget,
    thresholds: classThresholdsFrom(agg.tally, plan.flags.serveCaution),
    pointerRates: pointerRatesFrom(agg.tally),
    rates: confirmRatesFrom(agg.tally),
    kinds: kindRatesFrom(agg.tally),
    warningKinds: warningKindRateFrom(agg.tally, plan.flags.serveConfirmedFraud),
    ...(founding ? { founding } : {}),
    subtree: Math.min(width, SUBTREE_WIDTH),
  };
  // ---------------------------------------------------------------------
  // THE BLOCK IS RESERVED BEFORE IT IS SPENT, and the order matters more than
  // anything else in this file. Both monotone counters live in the manifest AND
  // in the ladder object and the higher wins, which covers rolling back EITHER —
  // but a crash after the buckets land used to roll back BOTH, so the next build
  // re-minted from a block it had already spent. The re-minted decoys are
  // skipped as already-occupied where they survived, but a decoy RETIRED in that
  // build's pass 1 is not occupied, so it was minted straight back and never
  // departed: departures fell BELOW `churn`, and a short churn proves the bucket
  // had fewer than `churn` real removals. So the frontier advance is COMMITTED
  // FIRST and the block is only then spent. A crash anywhere after this line
  // burns indices — the next build starts past them and mints hashes that have
  // never been published — which is the failure this design can absorb. `hw` is
  // deliberately NOT advanced here: it is the one counter whose over-count would
  // step the published rung, so it is written only once the buckets are down.
  // ---------------------------------------------------------------------
  await assertStillOurs(c);
  await putLadder(c.store, { seq: plan.seq, hw: plan.ladder.hw, frontier: frontier + block }, vec);
  // The marker says "this generation reserved a block", and it is what tells a
  // rolled-back counter pair from a fresh store: a marker newer than every
  // counter means the counters were restored from a backup while the generation
  // they reserved is still on the wire (see `planRegistryBuild`).
  await c.store.put(genMarkerKey(plan.seq), JSON.stringify({ seq: plan.seq, at: plan.builtAt, width, pad }));
  plan.rung = rung;
  logInfo({ cid: c.cid, hop: "cron", msg: "registry_rung", seq: plan.seq, count: rung.pad, cap: rung.creatorCap, churn: rung.nextChurn, kind: `w${width}` });
  // I3-1110 (D-0858(22)): the watch cover's fill order is complete only at width
  // 1 (apps/extension/src/cover.ts); every build at width >= 2 says so loudly
  // until the deferred cover ruling (D-0829(2)) is taken.
  if (width >= 2) logError({ cid: c.cid, hop: "cron", msg: "registry_width_cover_weakened", seq: plan.seq, count: width, kind: `w${width}` });
  await commitStep(c, { cursor: 0, phase: "stage", plan: JSON.stringify(plan) });
}

// --- THE PER-BUCKET FUNCTION ----------------------------------------------

interface Composed {
  entries: RegistryEntry[];
  body: string;
  byteLen: number;
  realCount: number;
  spilledIn: number;
  creatorBlocked: number;
  /** Spilled safety arrivals by class — `blocked` is the same set as hashes. */
  warningBlocked: number;
  cautionBlocked: number;
  blocked: string[];
  admitted: string[];
  squat: boolean;
  /** Every decoy pointer id this bucket would publish (fresh and surviving),
   * before the veto — what `stepStage` checks against `passports`. */
  decoyPointers: string[];
  /** Fresh decoy pointer ids minted here — whose name candidates `stepStage` checks (QUEUE #646). */
  freshPointers: string[];
  /** Surviving decoy pointers as published — whose shown name and candidates `stepStage` checks (D-0820(4)). */
  survivorNames: { p: string; n: string }[];
}

async function loadUnitsFor(c: Ctx, q: string, cache: Map<string, StageUnit>): Promise<StageUnit[]> {
  const u = c.plan.unitWidth;
  const W = c.plan.rung!.width;
  const keys = u <= W ? [q.slice(0, u)] : Array.from({ length: 16 ** (u - W) }, (_, i) => `${q}${hexOf(i, u - W)}`);
  const out: StageUnit[] = [];
  for (const p of keys) {
    let unit = cache.get(p);
    if (!unit) {
      const raw = await c.store.get(stageUnitKey(c.plan.seq, p));
      if (raw === null) throw new RegistryBuildError(`registry build ${c.plan.seq} lost its staged unit ${p} — refusing to publish a generation it cannot reproduce`);
      unit = JSON.parse(raw) as StageUnit;
      cache.set(p, unit);
    }
    out.push(unit);
  }
  return out;
}

/**
 * ONE BUCKET, from the frozen plan and the staged units alone — no clock, no
 * database, no dependence on which other buckets this step is doing. That is
 * what makes a killed build resumable: re-running this function writes the same
 * bytes, so the committed generation is byte-identical to an uninterrupted one.
 *
 * PASS 2 (admission) and PASS 3 (decoy mint) in one place.
 *
 * ADMISSION is capped twice: by the rung (a bucket refills to exactly `pad`) and
 * by `capacityFor`, so the bucket leaves at least `nextChurn` decoys behind for
 * the NEXT build to retire. Both caps read private state and are invisible on
 * the wire by construction: a slot a listing cannot take is taken by a decoy,
 * and the two are indistinguishable.
 *
 * THE RESERVED SAFETY LANE (OR-01, ADR-0012 §6) is the second cap, split: a
 * creator listing may fill up to `creatorCap = capacityFor(pad) - max(4,
 * ceil(pad/64))`, while a warning or a caution may use the full `capacityFor`.
 * `s` is a public function of `pad`, so it is no new channel; it costs about
 * 1.6% of creator capacity per bucket; and a safety entry that still does not
 * fit is recorded `pending` and steps the rung NEXT build. Nothing is evicted —
 * an eviction inside the real-churn quota would bound the bucket's real
 * occupancy at "at capacity" for anyone diffing it, which is one bit per bucket
 * per build of exactly what the padding exists to hide.
 */
async function composeBucket(c: Ctx, q: string, cache: Map<string, StageUnit>): Promise<Composed> {
  const { plan } = c;
  const rung = plan.rung!;
  const units = await loadUnitsFor(c, q, cache);
  const survivors: [RegistryEntry, 0 | 1, 0 | 1][] = [];
  const arrivals: [RegistryEntry, number, number][] = [];
  const occupied = new Set<string>();
  for (const unit of units) {
    for (const s of unit.s) if (s[0].h.startsWith(q)) survivors.push(s);
    for (const w of unit.w) if (w[0].h.startsWith(q)) arrivals.push(w);
    for (const h of unit.o) if (h.startsWith(q)) occupied.add(h);
  }
  const list: RegistryEntry[] = [];
  let nonDecoy = 0;
  let realCount = 0;
  // D-0707 (2): a decoy pointer that names a real passport is never published.
  const veto = new Set(c.agg.pointerVeto ?? []);
  const nameVeto = new Set(c.agg.nameVeto ?? []);
  const decoyPointers: string[] = [];
  const freshPointers: string[] = [];
  const survivorNames: { p: string; n: string }[] = [];
  for (const [e0, isOurs, placed] of survivors) {
    let e = e0;
    if (!isOurs && e.r) {
      decoyPointers.push(e.r.p);
      survivorNames.push(e.r);
      if (veto.has(e.r.p)) e = { h: e.h, c: e.c };
      else if (nameVeto.has(foldDisplayName(e.r.n))) {
        // D-0820(4): a real creator now holds this decoy's name. Same `p` (never re-aimed, D-0707), next free name, or no pointer.
        const n = await decoyNameFor(c.seed, e.r.p, nameVeto);
        e = n === undefined ? { h: e.h, c: e.c } : { h: e.h, c: e.c, r: { p: e.r.p, n } };
      }
    }
    // I3-0122: a decoy warning keeps its kind code through the pointer rewrites above, and one minted before I23
    // (no code) gets its hash-keyed code, so "has no `w`" never marks a decoy.
    if (!isOurs && e.c === "warning") e = { ...e, w: isWarningKindCode(e0.w) ? e0.w : await decoyWarningKind(c.seed, e.h, rung.warningKinds ?? 0) };
    e = await withFounding(c.seed, e, isOurs === 1, rung.founding);
    list.push(e);
    occupied.add(e.h);
    if (isOurs) nonDecoy++;
    if (placed) realCount++;
  }
  for (const [e] of arrivals) occupied.add(e.h);
  // Lane, then the lane's own stable order, then the hash: assembled from one
  // unit or from sixteen, a bucket admits in the same order.
  arrivals.sort((a, b) => a[1] - b[1] || a[2] - b[2] || (a[0].h < b[0].h ? -1 : a[0].h > b[0].h ? 1 : 0));
  const blocked: string[] = [];
  const admitted: string[] = [];
  let spilledIn = 0;
  let creatorBlocked = 0;
  let warningBlocked = 0;
  let cautionBlocked = 0;
  for (const [e, lane] of arrivals) {
    const safety = lane < 2;
    const limit = safety ? rung.cap : rung.creatorCap;
    if (list.length >= rung.pad || nonDecoy >= limit) {
      spilledIn++;
      if (safety) blocked.push(e.h);
      else creatorBlocked++;
      if (lane === 0) warningBlocked++;
      else if (lane === 1) cautionBlocked++;
      continue;
    }
    list.push(await withFounding(c.seed, e, true, rung.founding));
    nonDecoy++;
    realCount++;
    if (safety) admitted.push(e.h);
  }
  // PASS 3 — the decoy mint, out of this bucket's own sub-range of the reserved
  // block. A decoy costs four seeded digests (hash, class, proven, kind), each
  // from its own domain-separated preimage: a value computed from the entry's
  // own published bytes would be a public function of them, so anyone could
  // recompute it and every entry that disagreed would be real.
  const base = rung.frontier + parseInt(q, 16) * rung.perBucket;
  let need = rung.pad - list.length;
  for (let j = 0; need > 0; j++) {
    if (j >= rung.perBucket) {
      // A ~20-sigma shortfall, or a broken hash. Publish nothing and let the
      // previous generation keep serving.
      throw new RegistryBuildError(`decoy stream sub-block [${base}, ${base + rung.perBucket}) left ${need} slots of bucket ${q} unfilled`);
    }
    const i = base + j;
    const h = await decoyHashIn(c.seed, i, q);
    if (occupied.has(h)) continue;
    occupied.add(h);
    const cls = await decoyClass(c.seed, i, rung.thresholds);
    if (cls === "warning" || cls === "caution") {
      // A safety decoy carries no `p` and no `k` (no real one does) and a pointer
      // at the real per-class rate, never one `stepStage` found to be real.
      const r = await decoyPointer(c.seed, i, cls, rung.pointerRates, nameVeto, freshPointers);
      if (r) decoyPointers.push(r.p);
      const dw = cls === "warning" ? { w: await decoyWarningKind(c.seed, h, rung.warningKinds ?? 0) } : {};
      list.push(r && !veto.has(r.p) ? { h, c: cls, r, ...dw } : { h, c: cls, ...dw });
      need--;
      continue;
    }
    const dp = await decoyProven(c.seed, i, cls, rung.rates);
    const dk = await decoyKind(c.seed, i, cls, rung.kinds);
    const df = cls === "verified" ? await decoyFounding(c.seed, h, rung.founding) : undefined;
    list.push({ h, c: cls, ...(dp ? { p: 1 as const } : {}), ...(dk ? { k: dk } : {}), ...(df !== undefined ? { f: df } : {}) });
    need--;
  }
  list.sort((a, b) => (a.h < b.h ? -1 : a.h > b.h ? 1 : 0)); // decoys and reals interleave; order leaks nothing
  const body = JSON.stringify({ v: BUCKET_FORMAT_VERSION, entries: list });
  return {
    entries: list,
    body,
    byteLen: new TextEncoder().encode(body).length,
    realCount,
    spilledIn,
    creatorBlocked,
    warningBlocked,
    cautionBlocked,
    blocked,
    admitted,
    decoyPointers,
    freshPointers,
    survivorNames,
    // THE CAPACITY SQUAT, made loud instead of silent: a bucket held at its cap
    // swallows every later listing whose hash lands there, and subject hashes are
    // ground against the public canonicalizer offline, so an attacker picks the
    // bucket. It is NOT fixed by stepping the rung on "a bucket is full", because
    // that would put a registry-wide published event under a PRIVATE per-bucket
    // count — and a removal in that bucket frees a slot, so "the registry did not
    // double this build" would become a removal oracle.
    squat: creatorBlocked > 0 && nonDecoy >= rung.creatorCap,
  };
}

/** How many buckets this step can afford, from the frozen rung. */
function bucketsPerStep(c: Ctx, remaining: number, r2PerBucket: number, phase: string): number {
  const rung = c.plan.rung!;
  const perBucket = rung.pad * EST_ENTRY * 3;
  const k = Math.max(1, Math.min(remaining, Math.floor(c.plan.budget.r2Ops / r2PerBucket), Math.floor((c.plan.budget.heapBytes - EST_BASE) / Math.max(1, perBucket))));
  // stage spends one more D1 query than emit: the decoy-pointer veto.
  const est: Estimate = { heap: EST_BASE + perBucket * Math.min(k, 2) + rung.pad * EST_ENTRY, d1: phase === "stage" ? 6 : 5 /* emit: + the I3-1342 admission-row read */, r2: k * r2PerBucket };
  preflight(phase, c.plan.seq, est, c.plan.budget, c.cid);
  return k;
}

// --- STAGE (the byte-length survey) and EMIT (the bucket writes) -----------

async function stepStage(c: Ctx): Promise<void> {
  const { plan, agg } = c;
  const total = 16 ** plan.rung!.width;
  const start = c.row.cursor;
  const k = bucketsPerStep(c, total - start, 1, "stage");
  const cache = new Map<string, StageUnit>();
  const pointers: string[] = [];
  const fresh: string[] = [];
  const kept: { p: string; n: string }[] = [];
  const lens: number[] = [];
  for (let i = start; i < start + k; i++) {
    const composed = await composeBucket(c, hexOf(i, plan.rung!.width), cache);
    lens.push(composed.byteLen);
    pointers.push(...composed.decoyPointers);
    fresh.push(...composed.freshPointers);
    kept.push(...composed.survivorNames);
  }
  // QUEUE #646, D-0819(8), D-0820(4): NO DECOY NAME EQUALS A REAL DISPLAY NAME.
  // Every candidate name of every fresh pointer, and every surviving pointer's
  // shown name and candidates, in one query; a hit is skipped (fresh) or
  // replaced (surviving) for the whole generation. The SQL fold lower(trim())
  // equals `foldDisplayName` exactly on printable-ASCII names of <= 64 without
  // `<>` and without a doubled space (D-0849(5)); every other name is returned
  // raw and folded here. If the veto grew,
  // this step's buckets are re-composed, so `longest` measures the names emit
  // will write.
  if (fresh.length > 0 || kept.length > 0) {
    const cands = new Set<string>();
    for (const p of fresh) for (const n of await decoyNameCandidates(c.seed, p)) cands.add(n);
    for (const r of kept) {
      cands.add(foldDisplayName(r.n));
      for (const n of await decoyNameCandidates(c.seed, r.p)) cands.add(n);
    }
    const { results } = await c.db
      .prepare(
        "SELECT DISTINCT display_name AS n FROM passports WHERE lower(trim(display_name)) IN (SELECT value FROM json_each(?)) " +
          "OR display_name GLOB '*[^ -~]*' OR display_name GLOB '*[<>]*' OR display_name GLOB '*  *' OR length(display_name) > 64",
      )
      .bind(JSON.stringify([...cands]))
      .all<{ n: string }>();
    const had = new Set(agg.nameVeto ?? []);
    let grew = false;
    for (const r of results) {
      if (typeof r.n !== "string") continue;
      const f = foldDisplayName(r.n);
      if (!cands.has(f) || had.has(f)) continue;
      had.add(f);
      grew = true;
    }
    agg.nameVeto = [...had].sort();
    if (grew) {
      for (let i = start; i < start + k; i++) lens[i - start] = (await composeBucket(c, hexOf(i, plan.rung!.width), cache)).byteLen;
    }
  }
  for (const len of lens) if (len > agg.longest) agg.longest = len;
  // D-0707 (2): A DECOY POINTER NEVER POINTS AT A REAL ACCOUNT. One query per
  // step over every decoy pointer these buckets would publish; a hit is vetoed
  // for the whole generation, so emit (a pure function of plan and agg) drops it.
  // A 64-bit seeded id meeting a real one is not expected to happen; this is
  // what makes "never" a check instead of a probability. A surviving decoy is
  // re-checked every generation, so a passport minted later onto a published
  // decoy pointer loses it at the next build.
  if (pointers.length > 0) {
    const { results } = await c.db
      .prepare("SELECT passport_id FROM passports WHERE passport_id IN (SELECT value FROM json_each(?))")
      .bind(JSON.stringify(pointers))
      .all<{ passport_id: string }>();
    const veto = new Set(agg.pointerVeto ?? []);
    for (const r of results) veto.add(r.passport_id);
    agg.pointerVeto = [...veto].sort();
  }
  const done = start + k >= total;
  if (done) {
    // UNIFORM BYTE LENGTH, not just a uniform entry count. An entry's class is a
    // WORD on the wire and a warning carries a pointer, so the per-bucket MIX set
    // the body length even though the COUNT was uniform — and a bucket is a
    // cacheable GET with a stable body, so an attacker builds the size -> prefix
    // table once and then reads the prefix off every user's check. The target is
    // frozen HERE, from the longest body this generation produces, so that a
    // bucket written in one step and a bucket written an hour later are the same
    // length.
    plan.byteTarget = byteTargetFor(agg.longest);
  }
  await commitStep(c, { cursor: done ? 0 : start + k, phase: done ? "emit" : "stage", ...(done ? { plan: JSON.stringify(plan) } : {}) });
}

async function stepEmit(c: Ctx): Promise<void> {
  const { plan, agg } = c;
  const rung = plan.rung!;
  const W = rung.width;
  const total = 16 ** W;
  const leavesPer = 16 ** rung.subtree;
  const start = c.row.cursor;
  const k = bucketsPerStep(c, total - start, 3, "emit");
  const cache = new Map<string, StageUnit>();
  let leafBuf: string[] = [];
  let leafSubtree = Math.floor(start / leavesPer);
  // The leaves of a subtree are one object, appended to as the steps walk it.
  // A step that died before writing it left the object short; the cursor is then
  // rewound to where the object really ends and those buckets are re-emitted —
  // byte-identically, because `composeBucket` is a pure function of the plan.
  const head = start - leafSubtree * leavesPer;
  if (head > 0) {
    const raw = await c.store.get(stageLeavesKey(plan.seq, leafSubtree));
    leafBuf = raw === null ? [] : (JSON.parse(raw) as string[]);
    if (leafBuf.length < head) {
      await commitStep(c, { cursor: leafSubtree * leavesPer + leafBuf.length, phase: "emit" });
      return;
    }
    leafBuf = leafBuf.slice(0, head);
  }
  const blockedHere: string[] = [];
  let admittedHere: string[] = [];
  for (let i = start; i < start + k; i++) {
    const q = hexOf(i, W);
    const subtree = Math.floor(i / leavesPer);
    if (subtree !== leafSubtree) {
      await c.store.put(stageLeavesKey(plan.seq, leafSubtree), JSON.stringify(leafBuf));
      leafSubtree = subtree;
      leafBuf = [];
    }
    const composed = await composeBucket(c, q, cache);
    const body = await padBody(composed.body, composed.byteLen, plan.byteTarget!, c.seed, q, plan.seq);
    await c.store.put(genBucketKey(plan.seq, W, q), body);
    const leaf = await bucketLeaf(W, q, body);
    if (leaf === null) throw new RegistryBuildError(`registry bucket ${q} could not be committed to at width ${W}`);
    leafBuf.push(leaf);
    agg.total += composed.entries.length;
    agg.realCount += composed.realCount;
    agg.spilledIn += composed.spilledIn;
    agg.spilledInWarnings = (agg.spilledInWarnings ?? 0) + composed.warningBlocked;
    agg.spilledInCautions = (agg.spilledInCautions ?? 0) + composed.cautionBlocked;
    agg.spilledInListings = (agg.spilledInListings ?? 0) + composed.creatorBlocked;
    agg.safetyPending += composed.blocked.length;
    agg.safetyAdmitted += composed.admitted.length;
    if (composed.squat) {
      agg.blockedBuckets++;
      logError({ cid: c.cid, hop: "cron", msg: "registry_bucket_at_capacity", seq: plan.seq, count: composed.creatorBlocked, cap: rung.creatorCap, kind: `w${W}` });
    }
    // OR-01 bookkeeping is COLLECTED here and WRITTEN AFTER THE FLIP
    // (`applySafetyBookkeeping`). It used to be written in this step, outside any
    // commit condition, so a generation that was ABANDONED after emit still
    // deleted the `pending` rows a PUBLISHED generation had left — and the rung
    // step OR-01 promises ("still pending after one build") was deferred by a
    // whole generation, 6 h at the declared cadence, on the warning and caution
    // classes (REG-3). What a build did not publish must not move this state.
    blockedHere.push(...composed.blocked);
    admittedHere.push(...composed.admitted);
  }
  // I3-1342: `applySafetyBookkeeping` only UPDATEs rows that exist with the
  // admitted list, and its DELETE is blind to a hash with no row, so only an
  // admitted hash that HAS a `registry_safety_admission` row is carried. The list
  // used to hold every admitted safety arrival — at a first build with cautions
  // served, every unproven active account — re-serialized into `agg` on every
  // step and deduplicated with `Array.includes`, O(n^2).
  if (admittedHere.length > 0) {
    const { results } = await c.db
      .prepare("SELECT subject_hash FROM registry_safety_admission WHERE subject_hash IN (SELECT value FROM json_each(?))")
      .bind(JSON.stringify(admittedHere))
      .all<{ subject_hash: string }>();
    const tracked = new Set(results.map((r) => r.subject_hash));
    admittedHere = admittedHere.filter((h) => tracked.has(h));
  }
  // I3-1342: the split is STAGED per step (object store, not the OR-01 table —
  // REG-3) and agg carries only this step's cursor start, so agg_json no longer
  // grows with the number of turned-away hashes.
  if (blockedHere.length > 0 || admittedHere.length > 0) {
    await c.store.put(stageSafetyKey(plan.seq, start), JSON.stringify({ b: [...new Set(blockedHere)], a: [...new Set(admittedHere)] }));
    const stages = agg.safetyStages ?? [];
    if (!stages.includes(start)) stages.push(start);
    agg.safetyStages = stages;
  }
  await c.store.put(stageLeavesKey(plan.seq, leafSubtree), JSON.stringify(leafBuf));
  const done = start + k >= total;
  await commitStep(c, { cursor: done ? 0 : start + k, phase: done ? "tree" : "emit" });
}

// --- TREE, PROOFS, FINALIZE -----------------------------------------------

async function readLeaves(c: Ctx, subtree: number, expected: number): Promise<string[]> {
  const raw = await c.store.get(stageLeavesKey(c.plan.seq, subtree));
  const leaves = raw === null ? [] : (JSON.parse(raw) as string[]);
  if (leaves.length !== expected) {
    throw new RegistryBuildError(`registry build ${c.plan.seq} staged ${leaves.length} leaves for subtree ${subtree}, expected ${expected} — refusing to commit a generation it cannot commit to`);
  }
  return leaves;
}

/**
 * THE PUBLICATION COMMITMENT, one subtree at a time. A perfect Merkle tree over
 * the EXACT bytes of every bucket, in prefix order, so a client holding one
 * bucket and one path can check it against a single signed root without
 * downloading the other 16^width - 1 buckets.
 *
 * It is built in two levels — 256-leaf subtrees, then a tree over their roots —
 * because `buildTree` returns every leaf's whole path, and at width 4 that is
 * 65,536 paths of 16 hashes in one isolate. A perfect binary tree composes:
 * the nodes at level log2(256) ARE the subtree roots, so the root this produces
 * is the root the one-shot build produced.
 */
async function stepTree(c: Ctx): Promise<void> {
  const { plan, agg } = c;
  const rung = plan.rung!;
  const leavesPer = 16 ** rung.subtree;
  const subtrees = 16 ** (rung.width - rung.subtree);
  const start = c.row.cursor;
  const k = Math.max(1, Math.min(subtrees - start, Math.floor(plan.budget.r2Ops / 2), Math.max(1, Math.floor((plan.budget.heapBytes - EST_BASE) / (leavesPer * 400)))));
  preflight("tree", plan.seq, { heap: EST_BASE + leavesPer * 400 * 2, d1: 2, r2: k }, plan.budget, c.cid);
  for (let t = start; t < start + k; t++) {
    const leaves = await readLeaves(c, t, leavesPer);
    const tree = await buildTree(leaves);
    if (tree === null) throw new RegistryBuildError(`registry publication tree refused ${leaves.length} leaves — the bucket count is not a power of two`);
    agg.subRoots[t] = tree.root;
  }
  const done = start + k >= subtrees;
  if (done) {
    if (subtrees === 1) agg.root = agg.subRoots[0] ?? null;
    else {
      const top = await buildTree(agg.subRoots);
      if (top === null) throw new RegistryBuildError(`registry publication tree refused ${agg.subRoots.length} subtree roots`);
      agg.root = top.root;
    }
    // The generation digest: a fold over the leaves in prefix order. A leaf is a
    // digest of a bucket's exact bytes, so this is still a pure function of what
    // is on the wire — and unlike a fold over the bodies it can be computed one
    // step at a time. It is what tells a manifest that describes THIS generation
    // from one that describes a different build.
    let acc = digestSeed(rung.width, true);
    for (let t = 0; t < subtrees; t++) {
      const leaves = await readLeaves(c, t, leavesPer);
      for (let j = 0; j < leaves.length; j++) acc = await digestStep(acc, hexOf(t * leavesPer + j, rung.width), leaves[j]!);
    }
    agg.digest = acc;
  }
  await commitStep(c, { cursor: done ? 0 : start + k, phase: done ? "proofs" : "tree" });
}

async function stepProofs(c: Ctx): Promise<void> {
  const { plan, agg } = c;
  const rung = plan.rung!;
  const leavesPer = 16 ** rung.subtree;
  const subtrees = 16 ** (rung.width - rung.subtree);
  const start = c.row.cursor;
  const k = Math.max(1, Math.min(subtrees - start, Math.floor(plan.budget.r2Ops / leavesPer)));
  preflight("proofs", plan.seq, { heap: EST_BASE + leavesPer * 400 * 4, d1: 2, r2: k * leavesPer + 1 }, plan.budget, c.cid);
  const top = subtrees === 1 ? null : await buildTree(agg.subRoots);
  if (subtrees > 1 && top === null) throw new RegistryBuildError(`registry publication tree refused ${agg.subRoots.length} subtree roots`);
  for (let t = start; t < start + k; t++) {
    const leaves = await readLeaves(c, t, leavesPer);
    const tree = await buildTree(leaves);
    if (tree === null) throw new RegistryBuildError(`registry publication tree refused ${leaves.length} leaves`);
    for (let j = 0; j < leaves.length; j++) {
      const idx = t * leavesPer + j;
      const path = top === null ? tree.paths[j]! : [...tree.paths[j]!, ...top.paths[t]!];
      await c.store.put(genProofKey(plan.seq, rung.width, hexOf(idx, rung.width)), JSON.stringify({ v: PROOF_VERSION, seq: plan.seq, root: agg.root, path }));
    }
  }
  const done = start + k >= subtrees;
  await commitStep(c, { cursor: done ? 0 : start + k, phase: done ? "finalize" : "proofs" });
}

/**
 * THE FLIP. Everything this generation publishes is already in R2, under a
 * prefix nothing points at. Finalize signs the root, advances the high-water,
 * re-checks the epoch, and writes the pointer by COMPARE-AND-SWAP against the
 * etag read at plan time.
 *
 * A fenced predecessor fails one of those three and has written nothing a reader
 * can reach. That is what makes the ten-minute build lease unnecessary: the
 * lease bounded how long a builder could be wrong for, and the fence makes being
 * wrong harmless.
 */
/**
 * THE POINTER IS THE RECORD OF THE COMMIT, AND THE ROW ONLY CATCHES UP TO IT.
 *
 * The flip and the row update are two writes to two systems and no order makes
 * them one act, so the question is which one a resume BELIEVES. It used to
 * believe the row: flip, then `endBuild('committed')`, and a crash in between
 * left the pointer live on a generation whose row still said `open` — the next
 * tick re-ran finalize, lost the CAS against its own flip, and recorded
 * `abandoned` over a generation every reader was already being served
 * (INVAUDIT A3 ATTACK 2, which then let GC delete it).
 *
 * So the INTENT is written first, under the fence: the exact pointer bytes this
 * generation is about to publish go into the build row before the CAS. A resume
 * then reads the SERVED pointer and decides from it — if the live pointer is
 * this generation's bytes, the flip happened and the build is committed,
 * whatever the row says. The row is repaired to match; it is never the thing
 * that overrules what readers can see.
 */
/** One emit step's OR-01 split `{b: blocked, a: admitted-with-a-row}`, keyed by
 * the step's start cursor. Under `stage/`, so `gcStep` removes it once the
 * build's row is no longer open (never before `applySafetyBookkeeping` ran). */
function stageSafetyKey(seq: number, start: number): string {
  return `${genPrefix(seq)}stage/safety/${start}.json`;
}
/** I3-1342: hashes per bound json_each string in `applySafetyBookkeeping` —
 * ~67 B each, so ~335 KB per value; the batch is ceil(A/C)+ceil(B/C)+1 statements. */
export const SAFETY_CHUNK = 5_000;
/**
 * OR-01's state, written ONLY by a generation that reached the wire (REG-3).
 *
 * Three statements, all idempotent, so the commit path and the recovery path can
 * both run them: forget every hash this generation neither blocked nor admitted
 * (a withdrawn accusation stops being pending), mark the admitted ones, and
 * record the blocked ones as `pending` — keeping the ORIGINAL `since_seq` of a
 * row that was already pending, because "still pending after one build" is
 * measured from the build that first turned it away.
 */
async function applySafetyBookkeeping(c: Ctx): Promise<void> {
  const { plan, agg } = c;
  const blocked = new Set(agg.safetyBlockedHashes ?? []);
  const admitted = new Set(agg.safetyAdmittedHashes ?? []);
  for (const start of agg.safetyStages ?? []) {
    const raw = await c.store.get(stageSafetyKey(plan.seq, start));
    // A plain Error, never a RegistryBuildError: the pointer may already name this
    // generation, and a refusal would abandon it. Retried; the rows stay as the
    // previous generation left them rather than being rewritten from half a list.
    if (raw === null) throw new Error(`registry build ${plan.seq} staged no OR-01 split for emit step ${start}`);
    const split = JSON.parse(raw) as { b: string[]; a: string[] };
    for (const h of split.b) blocked.add(h);
    for (const h of split.a) admitted.add(h);
  }
  const nowIso = wallIso();
  const chunks = (xs: string[]): string[][] => {
    const out: string[][] = [];
    for (let i = 0; i < xs.length; i += c.safetyChunk) out.push(xs.slice(i, i + c.safetyChunk));
    return out;
  };
  const stmts: DbStatement[] = [];
  for (const part of chunks([...admitted])) {
    stmts.push(
      c.db
        .prepare("UPDATE registry_safety_admission SET state = 'admitted', updated_seq = ?, updated_at = ? WHERE subject_hash IN (SELECT value FROM json_each(?))")
        .bind(plan.seq, nowIso, JSON.stringify(part)),
    );
  }
  for (const part of chunks([...blocked])) {
    stmts.push(
      c.db
        .prepare(
          `INSERT INTO registry_safety_admission (subject_hash, state, since, since_seq, updated_seq, updated_at)
           SELECT value, 'pending', ?, ?, ?, ? FROM json_each(?) WHERE true
           ON CONFLICT(subject_hash) DO UPDATE SET
             state = 'pending',
             since = CASE WHEN registry_safety_admission.state = 'pending' THEN registry_safety_admission.since ELSE excluded.since END,
             since_seq = CASE WHEN registry_safety_admission.state = 'pending' THEN registry_safety_admission.since_seq ELSE excluded.since_seq END,
             updated_seq = excluded.updated_seq, updated_at = excluded.updated_at`,
        )
        .bind(plan.builtAt, plan.seq, plan.seq, nowIso, JSON.stringify(part)),
    );
  }
  // LAST: every row this generation blocked or admitted now carries its seq, so
  // "neither" is `updated_seq < seq` — no bound list of kept hashes. `<`, not
  // `<>`: a row a newer generation stamped is never this one's to forget.
  stmts.push(c.db.prepare("DELETE FROM registry_safety_admission WHERE updated_seq < ?").bind(plan.seq));
  await c.db.batch(stmts);
}

async function reconcileWithPointer(c: Ctx): Promise<BuildResult | null> {
  const intent = c.plan.pointerIntent;
  if (!intent) return null;
  const live = await c.store.get(MANIFEST_KEY);
  if (live === null || live !== intent) return null;
  // The pointer is ours: this build published, whatever the row says.
  // The flip is what makes this generation's OR-01 decisions real, and the turn
  // that flipped may have died before writing them.
  await applySafetyBookkeeping(c);
  if (!(await endBuild(c.db, c.plan.seq, "committed", null, wallIso(), c.epoch))) throw new FencedSignal(`registry build ${c.plan.seq} moved on from epoch ${c.epoch}`);
  logInfo({ cid: c.cid, hop: "cron", msg: "registry_commit_recovered", seq: c.plan.seq });
  return buildResultOf(c, JSON.parse(intent) as BucketManifest);
}

function buildResultOf(c: Ctx, manifest: BucketManifest): BuildResult {
  const { plan, agg } = c;
  return {
    manifest,
    totalEntries: agg.total,
    realCount: agg.realCount,
    padTarget: plan.rung!.pad,
    churn: plan.churn,
    realChurn: plan.realChurn,
    spilledIn: agg.spilledIn,
    spilledInWarnings: agg.spilledInWarnings ?? 0,
    spilledInCautions: agg.spilledInCautions ?? 0,
    spilledInListings: agg.spilledInListings ?? 0,
    spilledOut: agg.spilledOut,
    spilledOutRetractions: agg.spilledOutRetractions,
    blockedBuckets: agg.blockedBuckets,
    precedence_collisions: agg.precedenceCollisions,
    seq: plan.seq,
    safetyPending: agg.safetyPending,
    safetyAdmitted: agg.safetyAdmitted,
  };
}

/**
 * I3-0281 b (I22-W1-LEDGER): bytes per staged OR-01 hash that `applySafetyBookkeeping` holds at the flip (two
 * Sets, the staged splits, the json_each chunks). I26 #741 RE-MEASURED (I26-W1-REGISTRY harness
 * i26-741-flip-heap.measure.test.ts): 182-190 B a hash are LIVE when the bookkeeping batch is handed to the db
 * (the old figure, I6-REGHEAP:2), but the PEAK is higher -- by --max-old-space-size bisection, 100k hashes need
 * 60-64 MB and 300k need 112-116 MB, a marginal 252-294 B a hash. 300 B covers the peak, not the live set.
 */
export const EST_SAFETY_HASH = 300;
/** The flip runs alone after every staged step, so its allowance is twice a step's (96 MB at the default: the
 * measured passing cap), not the isolate. */
export const FLIP_HEAP_FACTOR = 2;

/** I3-0281 b: what the flip's bookkeeping will hold. `safetyPending + safetyAdmitted` counts every hash the emit
 * steps staged (admitted is counted before the tracked-row filter, so it is an upper bound). */
export function flipEstimate(agg: { safetyPending: number; safetyAdmitted: number; safetyStages?: number[] }): Estimate {
  return { heap: EST_BASE + (agg.safetyPending + agg.safetyAdmitted) * EST_SAFETY_HASH, d1: 4, r2: (agg.safetyStages?.length ?? 0) + 4 };
}

async function stepFinalize(c: Ctx): Promise<BuildResult | null> {
  const { plan, agg } = c;
  const rung = plan.rung!;
  // Did the flip already happen, in a turn that died before it could say so?
  const recovered = await reconcileWithPointer(c);
  if (recovered) return recovered;
  // I3-0281 b: the flip's O(M) bookkeeping is preflighted HERE, before the signer, the ladder write and the pointer
  // CAS -- once the pointer moves a refusal would strand a published generation, so a registry too large for it is
  // refused with zero writes and the served generation keeps serving (`registry_capacity_exhausted`, kind flip).
  preflight("flip", plan.seq, flipEstimate(agg), { ...plan.budget, heapBytes: plan.budget.flipHeapBytes ?? FLIP_HEAP_FACTOR * plan.budget.heapBytes }, c.cid);
  // A signature we could not obtain is published as ABSENT. Refusing to publish
  // would take the whole lookup path down for the length of a signer outage —
  // trading a working registry that says "unsigned" for no registry at all.
  const sig: RegistrySignature | null = c.signer ? await c.signer.sign({ width: rung.width, seq: plan.seq, built_at: plan.builtAt, root: agg.root ?? "" }) : null;
  // The second ladder write: the same reservation, plus the high-water this
  // build actually admitted. Idempotent with the first, monotone in both
  // counters, and before the pointer — a crash between the two leaves `hw` stale
  // rather than ahead, which costs throughput and never a rung crossing.
  await assertStillOurs(c);
  await putLadder(c.store, { seq: plan.seq, hw: rung.hwTarget - agg.spilledIn, frontier: rung.frontier + rung.block }, await readBhw(c.store));
  if (!(await advanceBuild(c.db, plan.seq, c.epoch, { phase: "finalize" }, wallIso()))) return null; // fenced: stop, write nothing
  const manifest: BucketManifest = {
    v: MANIFEST_FORMAT_VERSION,
    min_v: MANIFEST_MIN_READER_VERSION,
    width: rung.width,
    built_at: plan.builtAt,
    buckets: 16 ** rung.width,
    pad: rung.pad,
    kfloor: plan.kFloor,
    churn: rung.nextChurn,
    seq: plan.seq,
    frontier: rung.frontier + rung.block,
    ...(agg.digest ? { digest: agg.digest } : {}),
    ...(agg.root ? { root: agg.root } : {}),
    prefix: genPrefix(plan.seq),
    ...(sig ? { sig } : {}),
  };
  const pointer = JSON.stringify(manifest);
  // THE INTENT, under the fence and before the CAS: the bytes this generation is
  // about to publish. A turn that dies after the flip leaves this behind, and
  // the next one recognises its own pointer (`reconcileWithPointer`).
  plan.pointerIntent = pointer;
  await commitStep(c, { phase: "flip", plan: JSON.stringify(plan) });
  const etag = await versionedView(c.store).putIfMatch(MANIFEST_KEY, pointer, plan.manifestEtag);
  if (etag === null) {
    // Somebody else's generation is on the wire — unless it is OURS, flipped by
    // a turn that died before it could record it, which the reconcile above has
    // already answered. Ours is unreachable and will be collected; nothing we
    // wrote is served, and the winner's bytes stand.
    if (!(await endBuild(c.db, plan.seq, "abandoned", "pointer compare-and-swap lost", wallIso(), c.epoch))) {
      throw new FencedSignal(`registry build ${plan.seq} moved on from epoch ${c.epoch}`);
    }
    logError({ cid: c.cid, hop: "cron", msg: "registry_pointer_cas_lost", seq: plan.seq });
    return null;
  }
  // PUBLISHED, so the OR-01 state this generation decided is now the record.
  await applySafetyBookkeeping(c);
  if (!(await endBuild(c.db, plan.seq, "committed", null, wallIso(), c.epoch))) {
    // The flip HAPPENED: readers are on this generation. The row is what failed,
    // and the next turn's reconcile repairs it from the pointer.
    logError({ cid: c.cid, hop: "cron", msg: "registry_commit_unrecorded", seq: plan.seq });
  }
  const result: BuildResult = {
    manifest,
    totalEntries: agg.total,
    realCount: agg.realCount,
    padTarget: rung.pad,
    churn: plan.churn,
    realChurn: plan.realChurn,
    spilledIn: agg.spilledIn,
    spilledInWarnings: agg.spilledInWarnings ?? 0,
    spilledInCautions: agg.spilledInCautions ?? 0,
    spilledInListings: agg.spilledInListings ?? 0,
    spilledOut: agg.spilledOut,
    spilledOutRetractions: agg.spilledOutRetractions,
    blockedBuckets: agg.blockedBuckets,
    precedence_collisions: agg.precedenceCollisions,
    seq: plan.seq,
    safetyPending: agg.safetyPending,
    safetyAdmitted: agg.safetyAdmitted,
  };
  // The operator-facing numbers of a publication, in one line: how long the
  // generation took from plan to pointer, what is still waiting for a slot, and
  // what turned a listing away.
  logInfo({
    cid: c.cid, hop: "cron", msg: "registry_published", seq: plan.seq, churn: rung.nextChurn, count: agg.total,
    duration_ms: Math.max(0, Date.now() - Date.parse(c.row.started_at)), spilled: agg.spilledOut, deferred: agg.spilledIn,
    pending: agg.safetyPending, held: agg.blockedBuckets, kind: `w${rung.width}`,
    // I3-0867: `deferred` by lane, so the line tells a spilled warning from a
    // spilled listing. `detail` because log.ts allow-lists no per-lane key.
    detail: `deferred_warnings=${result.spilledInWarnings} deferred_cautions=${result.spilledInCautions} deferred_listings=${result.spilledInListings}`,
  });
  return result;
}

export interface BuildResult {
  manifest: BucketManifest;
  totalEntries: number;
  /** Real entries actually PUBLISHED (not the number eligible — see `spilledIn`). */
  realCount: number;
  /** The uniform per-bucket entry count actually used (a ladder rung). */
  padTarget: number;
  /** Departures per bucket this build. Published in the manifest. */
  churn: number;
  /** The cap on REAL departures per bucket this build. `churn - realChurn` is
   * the guaranteed decoy retirement — the part of the crowd nobody outside can
   * consume. Published here so an operator surface can show both numbers. */
  realChurn: number;
  /** Real entries that wanted a slot this build and did not get one, because
   * their bucket's arrivals were already spoken for. They are candidates again
   * next build. ALL LANES: it is exactly `spilledInWarnings + spilledInCautions
   * + spilledInListings`, and those three are what tell an operator a spilled
   * warning from a spilled listing (QUEUE #139). */
  spilledIn: number;
  /** Warnings that did not fit even the reserved safety lane (OR-01): a warning
   * IS spilled when its bucket is full, and waits a build as `pending`. */
  spilledInWarnings: number;
  /** Cautions spilled the same way. */
  spilledInCautions: number;
  /** Creator listings spilled (capacity squat or a full rung). */
  spilledInListings: number;
  /** Departures that wanted one of their bucket's `realChurn` slots and did not
   * get one. THIS IS THE DELISTING BACKLOG — an operator surface should watch
   * it, because it is the number the published SLA is about.
   *
   * It counts REMOVALS and the other quiet departures (a revoked account, a
   * retracted passport, a lapsed proof) only. A RETRACTED ACCUSATION is counted
   * separately below and, by construction, is never in this number — a stuck
   * retraction is an account still being accused after the accusation was
   * withdrawn, and it used to be reported inside a figure an operator reads as
   * "people waiting to be delisted". */
  spilledOut: number;
  /** Retracted accusations that did not get a departure slot. STRUCTURALLY ZERO
   * — retractions leave ahead of every other lane and outside the real-departure
   * cap — and reported anyway, so a future change that reintroduces a cap on
   * this lane shows up as a non-zero figure instead of as silence. */
  spilledOutRetractions: number;
  /** Buckets that were at their creator capacity and turned a listing away.
   * Non-zero means either the registry has outgrown its rung in one bucket or
   * somebody is squatting it. */
  blockedBuckets: number;
  /** The generation this build published. Its objects live under `gen/<seq>/`. */
  seq: number;
  /** Safety entries (warnings, cautions) the reserved lane could not admit, now
   * `pending` in `registry_safety_admission`. One still pending after this build
   * steps the rung at the next plan (OR-01 (ii)) — never an eviction. */
  safetyPending: number;
  /** Safety entries admitted into the reserved lane this build. */
  safetyAdmitted: number;
  /** Hashes where a real creator listing LOST to a published warning (INV-027,
   * docket #4: warning > creator > caution). Structurally these are the cases a
   * human must look at and the served bytes cannot show: either an impersonator
   * who obtained a listing before the warning caught up, or a warning published
   * against a genuine creator. Suppressed hashes are NOT counted — they are
   * served neither way (INV-003), so nothing lost to anything. */
  precedence_collisions: number;
}

// --- THE PLAN --------------------------------------------------------------

function requireSeed(seed: unknown): string {
  if (!isUsableDecoySeed(seed)) {
    throw new RegistryBuildError(`REGISTRY_DECOY_SEED absent, shorter than ${DECOY_SEED_MIN_LEN} chars, or a placeholder — registry build refused`);
  }
  return seed;
}

export interface RegistryBuildOpts {
  now?: () => string;
  /** The rung a bucket file may not pass before the WIDTH grows instead — the
   * same kind of knob as `kFloor`, and a guard exercises the rule at a toy rung
   * with it rather than seeding the 53k listings production needs to reach 4,000. */
  padCeiling?: number;
  /** How many objects one garbage-collection pass may delete. */
  gcBudget?: number;
  decoySeed?: string;
  kFloor?: number;
  nowMs?: number;
  signer?: RegistrySigner | undefined;
  budget?: Partial<StepBudget>;
  cid?: string;
  /** Hashes per OR-01 bookkeeping statement (default `SAFETY_CHUNK`) — the same
   * kind of knob as `padCeiling`: a guard exercises the chunking without seeding
   * five thousand turned-away warnings. */
  safetyChunk?: number;
}

export interface PlanHandle {
  seq: number;
  epoch: number;
  /** False when an open build was already there and was adopted rather than started. */
  started: boolean;
}

/**
 * PLAN — the only step that reads global state, and the only one that decides
 * anything. It freezes the clock, the operator settings, the previous
 * generation's pointer and etag, the ladder counters, the churn quota and the
 * unit partition into `plan_json`; every later step is a function of that plan
 * and of what the survey staged.
 *
 * Nothing is written before the preflight has passed and the previous generation
 * has been read: an over-budget or incoherent registry leaves the store exactly
 * as it found it.
 */
export async function planRegistryBuild(db: DbLike, store: ObjectStore, opts: RegistryBuildOpts & { takeover?: boolean } = {}): Promise<PlanHandle> {
  const seed = requireSeed(opts.decoySeed);
  const cid = opts.cid ?? "registry";
  const now = opts.now ?? (() => new Date().toISOString());
  const kFloor = opts.kFloor ?? K_FLOOR;
  const padCeiling = opts.padCeiling ?? PAD_ENTRY_CEILING;
  const builtAt = now();
  const parsed = Date.parse(builtAt);
  const nowMs = opts.nowMs ?? (Number.isFinite(parsed) ? parsed : Date.now());
  const budget: StepBudget = { ...DEFAULT_STEP_BUDGET, ...opts.budget };
  const open = await readOpenBuild(db);
  if (open) {
    const stale = Date.now() - Date.parse(open.updated_at) > STALE_OPEN_BUILD_MS;
    if (!opts.takeover && !stale) {
      throw new RegistryBuildBusyError(`registry build ${open.seq} is already in progress (phase ${open.phase}) — refusing to start a second one`);
    }
    // Fenced on the epoch the open row was READ at: if another invocation
    // claimed it in between, the takeover loses and this planner steps aside
    // rather than ending a build somebody else is advancing.
    if (!(await endBuild(db, open.seq, "abandoned", opts.takeover ? "superseded by a new plan" : "no advance within the takeover window", wallIso(), open.epoch))) {
      throw new RegistryBuildBusyError(`registry build ${open.seq} was claimed while this plan was taking it over — refusing to start a second one`);
    }
  }
  // Read the per-account gate BEFORE the real set, because it decides what the
  // real set IS. It is the same operator setting the caution class rides on, and
  // it is read once per build, never per account.
  const flags = {
    perAccountProof: await servingPerAccountStanding(db),
    serveConfirmedFraud: await servingConfirmedFraud(db),
    serveCaution: await servingCaution(db),
  };
  const counts = await db
    .prepare(
      `SELECT (SELECT COUNT(*) FROM passport_accounts) AS accounts,
              (SELECT COUNT(*) FROM passport_accounts a JOIN passports p ON p.passport_id = a.passport_id WHERE a.status = 'active' AND p.retracted = 0) AS active,
              (SELECT COUNT(*) FROM fan_report_items WHERE state = 'open') AS reports`,
    )
    .bind()
    .first<{ accounts: number; active: number; reports: number }>();
  const accounts = Number(counts?.accounts ?? 0);
  // The caution population is the one read this builder cannot bound by prefix
  // (`cautionEntries` is the caution module's, and it hashes its own rows), so
  // it is taken ONCE per build and sharded to R2 for the steps — and the
  // preflight refuses the whole build when that one read would not fit.
  const planHeap = EST_BASE + (flags.serveCaution ? (Number(counts?.active ?? 0) + Number(counts?.reports ?? 0)) * EST_ROW : 0);
  preflight("plan", 0, { heap: planHeap, d1: 12, r2: 24 }, budget, cid);
  const cur = await versionedView(store).getVersioned(MANIFEST_KEY);
  const markers = (await store.list(GEN_MARKER_PREFIX)).map(markerSeq).filter((s): s is number => s !== null);
  let prev: PrevMeta | null = null;
  if (cur === null) {
    // No manifest and no generation is a fresh registry. No manifest WITH a
    // generation on the wire is an incoherent store: rebuilding the decoy stream
    // from scratch there re-mints hashes that have already been published and
    // retired. A generation whose build this database knows was ABANDONED never
    // served anything, so it does not count.
    const abandoned = new Set<number>();
    for (const m of markers) {
      const row = await readBuild(db, m);
      if (row?.state === "abandoned") abandoned.add(m);
    }
    // I26 JN-04 (D-1016(3)): existence needs ONE page — a store with `listPage` (R2) reads one, not the whole prefix.
    const lp = (store as ObjectStore & { listPage?: ListPage }).listPage;
    const anyBucket = typeof lp === "function" ? (await lp("buckets/")).keys.length > 0 : (await store.list("buckets/")).length > 0;
    if (anyBucket || markers.some((m) => !abandoned.has(m))) {
      throw new RegistryBuildError("registry store holds buckets but no manifest — refusing to rebuild the decoy stream from scratch");
    }
  } else {
    prev = parseManifest(cur.value);
    if (prev.manifestDigest === null) {
      logError({ cid, hop: "cron", msg: "registry_manifest_no_bucket_digest_accepted_once" });
    }
  }
  const ladder = await loadLadder(store, prev);
  // A GENERATION NEWER THAN EVERY COUNTER. "Higher wins" across the manifest and
  // the ladder object survives ONE of them being rolled back; it cannot survive
  // both, and both is what restoring a backup does. The markers are written with
  // the frontier reservation and are not part of that pair, so a marker above
  // both counters says the counters were rewound while the generation they
  // reserved is still on the wire — the frontier is genuinely lost, and
  // re-minting a spent block is worse than not publishing.
  const newestMarker = markers.reduce((a, b) => Math.max(a, b), 0);
  if (newestMarker > ladder.seqFloor) {
    throw new RegistryBuildError(
      `registry generation ${newestMarker} reserved a decoy block that neither the manifest nor the ladder object knows about — refusing to rebuild from a rolled-back decoy frontier (see DEPLOY.md 3e)`,
    );
  }
  const seq = Math.max(ladder.seqFloor, await maxBuildSeq(db), newestMarker) + 1;
  // OR-01 (ii): a safety entry that a COMMITTED build left pending steps the
  // rung. An abandoned build's leftovers do not — a rung crossing is a published,
  // irreversible event and must not be triggered by a generation nobody saw.
  const pendingSafety = await db
    .prepare("SELECT 1 AS x FROM registry_safety_admission WHERE state = 'pending' AND since_seq < ? AND since_seq IN (SELECT seq FROM registry_builds WHERE state = 'committed') LIMIT 1")
    .bind(seq)
    .first<{ x: number }>();
  const plan: Plan = {
    seq,
    nowMs,
    builtAt,
    kFloor,
    padCeiling,
    flags,
    prev,
    manifestEtag: cur?.etag ?? null,
    ladder,
    // The survey works on the PREVIOUS partition, because that is where pass 1
    // has to happen. Without one, the units are sized from the account count so
    // that a step's reads stay bounded whatever this build's width turns out to be.
    unitWidth: prev ? prev.width : Math.max(1, Math.min(MAX_WIDTH, chooseWidth(accounts, kFloor, padCeiling))),
    churn: prev === null ? 0 : churnFor(prev.padTarget),
    // The part of `churn` real departures may take. The rest is paid in decoys in
    // EVERY history, which is what stops an attacker who can mint departures from
    // owning the whole crowd — see DECOY_CHURN_SHARE.
    realChurn: prev === null ? 0 : realChurnFor(prev.padTarget),
    safetyFallback: !!pendingSafety,
    budget,
    accounts,
    rung: null,
    byteTarget: null,
  };
  if (!(await insertBuild(db, seq, "backfill", JSON.stringify(plan), wallIso()))) {
    const other = await readOpenBuild(db);
    throw new RegistryBuildBusyError(`registry build ${other?.seq ?? "?"} is already in progress — refusing to start a second one`);
  }
  void seed;
  logInfo({ cid, hop: "cron", msg: "registry_planned", seq, count: accounts, kind: prev ? `v${prev.v}` : "first" });
  return { seq, epoch: 1, started: true };
}

// --- THE DRIVER ------------------------------------------------------------

/** The first write of every build: the SEQ reservation. The frontier is not
 * advanced here (the block is not known until the survey has counted the
 * emptiest bucket) — this write only says which generation is being assembled,
 * so a crash before the frontier reservation costs a burned seq and nothing else. */
async function stepReserve(c: Ctx): Promise<void> {
  const { plan } = c;
  await assertStillOurs(c);
  // A STALE OR MISSING VECTOR IS DROPPED HERE, once, at the only point that knows
  // (`loadLadder`'s freshness rule). Every later write carries whatever survives.
  await putLadder(c.store, { seq: plan.seq, hw: plan.ladder.hw, frontier: plan.ladder.frontierFloor }, plan.ladder.bhwFresh ? await readBhw(c.store) : null);
  if (plan.flags.serveCaution) {
    // The ONE unbounded read left, taken once per build and sharded so the steps
    // read only their own prefix. `cautionEntries` belongs to the caution module
    // (it applies the corroboration gate, the strong-kind gate and the
    // consent-of-the-named pointer rule), so it is called, not reimplemented.
    const entries = await cautionEntries(c.db, plan.nowMs);
    const shards = new Map<string, RegistryEntry[]>();
    for (const e of entries) {
      const hex = e.h[0]!;
      const cur = shards.get(hex);
      if (cur) cur.push(e);
      else shards.set(hex, [e]);
    }
    for (const [hex, list] of shards) await c.store.put(stageCautionKey(plan.seq, hex), JSON.stringify(list));
  }
  await commitStep(c, { cursor: 0, phase: "survey" });
}

/**
 * Fills `subject_hash` for up to `limit` rows that lack it (migration 0097).
 * Returns how many rows were filled; fewer than `limit` means none are left.
 * `account_key` is the primary key and no statement changes it, so a filled
 * value never goes stale.
 *
 * IT LIVES HERE, not beside the rest of the D1 plumbing in registry-gen.ts,
 * because it is a HASHING SITE: `tests/invariants/hash-positive-control.test.ts`
 * enumerates every module that imports a hash function from @dhwaja/canonical
 * and requires each to be classified and exercised, and this file is the one
 * that already is — the guard compares this builder's bucket entry hash and
 * prefix against canonical's for every corpus subject. Materializing the same
 * hash into a column is the same claim, so it is checked by the same test.
 */
export const HASH_BACKFILL_BATCH = 400;

async function backfillSubjectHashes(db: DbLike, limit: number = HASH_BACKFILL_BATCH): Promise<number> {
  const { results } = await db
    .prepare("SELECT account_key FROM passport_accounts WHERE subject_hash IS NULL LIMIT ?")
    .bind(limit)
    .all<{ account_key: string }>();
  if (results.length === 0) return 0;
  const stmts = [];
  for (const r of results) {
    stmts.push(db.prepare("UPDATE passport_accounts SET subject_hash = ? WHERE account_key = ? AND subject_hash IS NULL").bind(await subjectHash(r.account_key), r.account_key));
  }
  await db.batch(stmts);
  return results.length;
}

/**
 * The build's own maintenance of the two lookup columns migration 0097 adds,
 * in bounded pages: `subject_hash` (so a step reads only its prefix) and the
 * asserted/vetted fold (so no step reads the whole ledger). Both are idempotent,
 * so a killed step simply re-runs.
 *
 * BOUNDED, so a column that cannot converge fails the build loudly instead of
 * spinning: each page fills at most `HASH_BACKFILL_BATCH` rows and advances the
 * fold by at most one page, and a registry that needs more pages than this has
 * more accounts than the declared envelope allows anyway.
 */
const MAX_BACKFILL_STEPS = 20_000;

async function stepBackfill(c: Ctx): Promise<void> {
  if (c.agg.backfillSteps >= MAX_BACKFILL_STEPS) {
    throw new RegistryBuildError(`registry build ${c.plan.seq} could not finish its lookup-column backfill in ${MAX_BACKFILL_STEPS} pages — refusing to publish from a partial index`);
  }
  c.agg.backfillSteps += 1;
  const filled = await backfillSubjectHashes(c.db);
  const caught = await foldAssertedStep(c.db, wallIso());
  await commitStep(c, { phase: filled === 0 && caught ? "reserve" : "backfill" });
}

export interface AdvanceResult {
  seq: number | null;
  state: "none" | "open" | "committed" | "abandoned" | "busy";
  phase: string | null;
  steps: number;
  result: BuildResult | null;
}

/**
 * D1 calls ONE invocation of the step driver may spend (#741, D-0859(3)).
 * SOURCE: the ~1,000 binding calls per Worker invocation that
 * DEFAULT_STEP_BUDGET's docblock cites, which this leg SHARES with the rest of
 * its tick (alerts.ts:320 budgets up to 597 for delivery alone); 400 is what is
 * left. The step budget's `d1Queries` is per STEP, and before this nothing
 * bounded the steps per invocation but the deadline: I6-REGHEAP-04 counted
 * 1,546 D1 calls in one 33 s invocation. Counted through the db handle by
 * `meterDb`, one per binding call: run/all/first, and a batch (one call however
 * many statements it carries), which is how I6-REGHEAP counted the 1,546.
 */
export const REGISTRY_INVOCATION_D1_CALLS = 400;

/**
 * #797a: GC after a publish (or on an idle tick) runs on the SAME invocation. Its
 * worst D1 cost — registry-gen.ts `gcStep`: the previous-generation read, the
 * candidate list (LIMIT 16), one `gc_state` UPDATE per listed generation, the
 * legacy marker read and its INSERT — is this many calls, and `collect` skips the
 * pass when the invocation's meter has less than that left. Its store calls
 * (pointer read, list, pointer re-read, delete) are capped per invocation at
 * REGISTRY_INVOCATION_STORE_OPS, the same share of the ~1,000 binding calls the
 * D1 budget above takes; a capped pass resumes next tick (GC is hygiene).
 * I26 #797: that cap is the WHOLE invocation's — `meterStore` counts every store
 * call the plan, the steps and GC make; `runSteps` stops before a step whose
 * declared `r2Ops` (or the dearest step so far) would cross it, and GC gets only
 * what the steps left.
 */
export const REGISTRY_GC_D1_CALLS = 2 + 16 + 2;
export const REGISTRY_INVOCATION_STORE_OPS = 400;

interface D1Meter {
  calls: number;
  budget: number;
  storeCalls: number;
  storeBudget: number;
}

/** R2's `list` page: a whole-prefix `list` is one R2 op per this many keys (registry-gen.ts `registryR2Store.list`). */
const STORE_LIST_PAGE_KEYS = 1000;

/** I26 #797: one `storeCalls` per call of any store method (get/put/list/delete/conditional put); I26 JN-04
 * (D-1016(3)): a whole-prefix `list` is charged one per PAGE of keys it returned (at least one), as R2 spends them. */
function meterStore<S extends ObjectStore>(store: S, m: D1Meter): S {
  return new Proxy(store, {
    get(target, prop) {
      const v = Reflect.get(target, prop, target) as unknown;
      if (typeof v !== "function") return v;
      return (...args: unknown[]) => {
        m.storeCalls++;
        const r = (v as (...a: unknown[]) => unknown).apply(target, args);
        if (prop !== "list") return r;
        return Promise.resolve(r).then((keys) => {
          if (Array.isArray(keys)) m.storeCalls += Math.max(1, Math.ceil(keys.length / STORE_LIST_PAGE_KEYS)) - 1;
          return keys;
        });
      };
    },
  });
}

function meterDb(db: DbLike, m: D1Meter): DbLike {
  const inner = new WeakMap<DbStatement, DbStatement>();
  const wrap = (s: DbStatement): DbStatement => {
    const w: DbStatement = {
      bind: (...v: unknown[]) => wrap(s.bind(...v)),
      run: () => {
        m.calls++;
        return s.run();
      },
      all: <T = Record<string, unknown>>() => {
        m.calls++;
        return s.all<T>();
      },
      first: <T = Record<string, unknown>>() => {
        m.calls++;
        return s.first<T>();
      },
    };
    inner.set(w, s);
    return w;
  };
  return {
    prepare: (sql: string) => wrap(db.prepare(sql)),
    batch: (st: DbStatement[]) => {
      m.calls++;
      return db.batch(st.map((x) => inner.get(x) ?? x));
    },
  };
}

async function runSteps(db: DbLike, store: ObjectStore, seq: number, epoch: number, opts: RegistryBuildOpts, deadlineAt: number, maxSteps: number, meter?: D1Meter): Promise<AdvanceResult> {
  const seed = requireSeed(opts.decoySeed);
  const cid = opts.cid ?? "registry";
  let steps = 0;
  let worstStep = 0;
  let worstStore = 0;
  for (;;) {
    const row = await readBuild(db, seq);
    if (row?.state !== "open") return { seq, state: (row?.state as AdvanceResult["state"]) ?? "none", phase: row?.phase ?? null, steps, result: null };
    // THE FENCE, read side: another invocation has taken this build over, so
    // everything this one would write is into a generation nothing points at.
    if (row.epoch !== epoch) return { seq, state: "busy", phase: row.phase, steps, result: null };
    if (steps >= maxSteps || Date.now() >= deadlineAt) return { seq, state: "open", phase: row.phase, steps, result: null };
    const plan = JSON.parse(row.plan_json) as Plan;
    // THE INVOCATION'S D1 BUDGET: stop BEFORE a step that could cross it (the
    // step's declared allowance, or the dearest step this invocation has run).
    if (meter && meter.calls + Math.max(plan.budget.d1Queries, worstStep) > meter.budget) return { seq, state: "open", phase: row.phase, steps, result: null };
    // I26 #797: the same rule for the invocation's store calls (the step's declared r2Ops, or the dearest so far).
    if (meter && meter.storeCalls + Math.max(plan.budget.r2Ops ?? 0, worstStore) > meter.storeBudget) return { seq, state: "open", phase: row.phase, steps, result: null };
    const callsBefore = meter?.calls ?? 0;
    const storeBefore = meter?.storeCalls ?? 0;
    const agg: Agg = { ...emptyAgg(), ...(JSON.parse(row.agg_json) as Partial<Agg>) };
    const c: Ctx = { db, store, seed, plan, agg, row, epoch, cid, signer: opts.signer, safetyChunk: Math.max(1, opts.safetyChunk ?? SAFETY_CHUNK) };
    const t0 = Date.now();
    let result: BuildResult | null = null;
    try {
      // THE POINTER DECIDES WHETHER THIS BUILD IS DONE. A turn that flipped and
      // died before recording it leaves an `open` row whose generation every
      // reader is already on; believing the row here is what let the next turn
      // abandon a live generation (INVAUDIT A3 ATTACK 2).
      const already = await reconcileWithPointer(c);
      if (already) return { seq, state: "committed", phase: "done", steps, result: already };
      switch (row.phase as Phase) {
        case "backfill": await stepBackfill(c); break;
        case "reserve": await stepReserve(c); break;
        case "survey": await stepSurvey(c); break;
        case "rung": await stepRung(c); break;
        case "stage": await stepStage(c); break;
        case "emit": await stepEmit(c); break;
        case "tree": await stepTree(c); break;
        case "proofs": await stepProofs(c); break;
        case "finalize":
        case "flip":
          result = await stepFinalize(c);
          break;
        default: throw new RegistryBuildError(`registry build ${seq} is in an unknown phase ${row.phase}`);
      }
    } catch (e) {
      // RG-PRIVACY:12 / W11A:18: `msg` is the build row's stored reason (endBuild / recordFailure); a LOG line carries the class only (#857).
      const msg = e instanceof Error ? e.message : String(e);
      // Superseded: everything this builder wrote went into a generation nothing
      // points at, and the row belongs to the epoch that took it over.
      if (e instanceof FencedSignal) {
        logInfo({ cid, hop: "cron", msg: "registry_step_fenced", seq, kind: row.phase });
        return { seq, state: "busy", phase: row.phase, steps, result: null };
      }
      if (e instanceof RegistryBuildError) {
        // A refusal is a decision about the DATA, so retrying it changes nothing:
        // the generation is abandoned (it was never reachable) and the published
        // one keeps serving.
        // Fenced: a refusal reached by a builder that has been superseded must
        // not end the build its successor now holds (INVAUDIT A3 ATTACK 1).
        await endBuild(db, seq, "abandoned", msg, wallIso(), epoch);
        logError({ cid, hop: "cron", msg: "registry_build_refused", seq, kind: row.phase, err: e instanceof Error ? e.name : typeof e });
        throw e;
      }
      const n = await recordFailure(db, seq, epoch, msg, wallIso());
      logError({ cid, hop: "cron", msg: "registry_step_failed", seq, kind: row.phase, count: n, err: e instanceof Error ? e.name : typeof e });
      if (n >= MAX_STEP_FAILURES) await endBuild(db, seq, "abandoned", `${n} consecutive step failures: ${msg}`, wallIso(), epoch);
      throw e;
    }
    steps++;
    if (meter) worstStep = Math.max(worstStep, meter.calls - callsBefore);
    if (meter) worstStore = Math.max(worstStore, meter.storeCalls - storeBefore);
    logInfo({ cid, hop: "cron", msg: "registry_step", seq, kind: row.phase, count: steps, duration_ms: Date.now() - t0 });
    if (result) return { seq, state: "committed", phase: "done", steps, result };
  }
}

async function collect(db: DbLike, store: ObjectStore, nowMs: number, cid: string, budget: number, meter?: D1Meter): Promise<void> {
  if (meter && meter.budget - meter.calls < REGISTRY_GC_D1_CALLS) {
    logInfo({ cid, hop: "cron", msg: "registry_gc", count: 0, kind: "deferred_d1_budget" });
    return;
  }
  try {
    const gc = await gcStep(db, store, nowMs, wallIso(), budget, meter ? Math.max(0, meter.storeBudget - meter.storeCalls) : Infinity);
    if (gc.deleted > 0) logInfo({ cid, hop: "cron", msg: "registry_gc", count: gc.deleted });
  } catch (e) {
    // Collection is hygiene: an unreachable generation left in place costs
    // storage, never correctness.
    logError({ cid, hop: "cron", msg: "registry_gc_failed", err: e instanceof Error ? e.name : typeof e });
  }
}

/**
 * ONE INVOCATION'S WORTH of a staged build: take the open build over (which
 * fences whoever held it), run as many steps as fit the invocation's deadline, its D1 call budget (REGISTRY_INVOCATION_D1_CALLS)
 * and the per-step budget, and leave the cursor where the next tick picks it up.
 * With `plan: true` and no build open, it starts one.
 */
export async function advanceRegistryBuild(
  db: DbLike,
  store: ObjectStore,
  opts: RegistryBuildOpts & { plan?: boolean; deadlineMs?: number; maxSteps?: number; gcBudget?: number; d1CallBudget?: number; storeCallBudget?: number } = {},
): Promise<AdvanceResult> {
  const meter: D1Meter = { calls: 0, budget: opts.d1CallBudget ?? REGISTRY_INVOCATION_D1_CALLS, storeCalls: 0, storeBudget: opts.storeCallBudget ?? REGISTRY_INVOCATION_STORE_OPS };
  const mdb = meterDb(db, meter);
  const mstore = meterStore(store, meter);
  const cid = opts.cid ?? "registry";
  const deadlineAt = Date.now() + (opts.deadlineMs ?? DEFAULT_INVOCATION_MS);
  let open = await readOpenBuild(mdb);
  let epoch: number;
  if (!open) {
    if (!opts.plan) {
      await collect(mdb, mstore, opts.nowMs ?? Date.now(), cid, opts.gcBudget ?? 200, meter);
      return { seq: null, state: "none", phase: null, steps: 0, result: null };
    }
    const handle = await planRegistryBuild(mdb, mstore, opts);
    epoch = handle.epoch;
    open = await readBuild(mdb, handle.seq);
    if (!open) return { seq: handle.seq, state: "none", phase: null, steps: 0, result: null };
  } else {
    const claimed = await claimBuild(mdb, open.seq, open.epoch, wallIso());
    if (claimed === null) return { seq: open.seq, state: "busy", phase: open.phase, steps: 0, result: null };
    epoch = claimed;
  }
  const r = await runSteps(mdb, mstore, open.seq, epoch, opts, deadlineAt, opts.maxSteps ?? Number.MAX_SAFE_INTEGER, meter);
  if (r.result) await collect(mdb, mstore, opts.nowMs ?? Date.now(), cid, opts.gcBudget ?? 200, meter);
  return r;
}

/**
 * Build all bucket files + manifest into the store, run to completion: plan,
 * every step, finalize, in one call. This is the shape every existing caller and
 * harness uses; the cron uses `advanceRegistryBuild` instead, which does the same
 * work a few steps at a time.
 *
 * `decoySeed` is REQUIRED and is a secret (REGISTRY_DECOY_SEED). Absent, short
 * or placeholder => this throws and NOTHING is written: a build padded with
 * fresh random decoys re-opens the cross-build intersection attack, so no build
 * is strictly better than that build. The previously published generation keeps
 * being served in the meantime, which is stale but consistent.
 *
 * AN OPEN BUILD THAT DIED IS RESUMED, not restarted (T5). Its plan is frozen, so
 * finishing it publishes the generation it planned, byte for byte. An open build
 * that is being advanced right now is refused with `RegistryBuildBusyError`.
 */
export async function buildRegistry(db: DbLike, store: ObjectStore, opts: RegistryBuildOpts = {}): Promise<BuildResult> {
  requireSeed(opts.decoySeed);
  const open = await readOpenBuild(db);
  let seq: number;
  let epoch: number;
  if (open && (open.failures > 0 || Date.now() - Date.parse(open.updated_at) > STALE_OPEN_BUILD_MS)) {
    const claimed = await claimBuild(db, open.seq, open.epoch, wallIso());
    if (claimed === null) throw new RegistryBuildBusyError(`registry build ${open.seq} is already in progress — refusing to run a second one concurrently`);
    seq = open.seq;
    epoch = claimed;
  } else {
    const handle = await planRegistryBuild(db, store, opts);
    seq = handle.seq;
    epoch = handle.epoch;
  }
  const r = await runSteps(db, store, seq, epoch, opts, Number.MAX_SAFE_INTEGER, Number.MAX_SAFE_INTEGER);
  if (!r.result) {
    if (r.state === "busy") throw new RegistryBuildBusyError(`registry build ${seq} was taken over by another builder`);
    throw new RegistryBuildError(`registry build ${seq} did not publish (${r.state})`);
  }
  await collect(db, store, opts.nowMs ?? Date.now(), opts.cid ?? "registry", opts.gcBudget ?? 200);
  return r.result;
}

/** How old the SERVED generation is, in ms, or null when nothing is published. */
export async function registryPublicationAgeMs(store: ObjectStore, nowMs: number): Promise<number | null> {
  const raw = await store.get(MANIFEST_KEY);
  if (raw === null) return null;
  try {
    const m = JSON.parse(raw) as { built_at?: unknown };
    const at = typeof m.built_at === "string" ? Date.parse(m.built_at) : Number.NaN;
    return Number.isFinite(at) ? nowMs - at : null;
  } catch {
    return null;
  }
}

/**
 * THE PUBLICATION-AGE ALARM. The promise is "a generation starts every 6 hours
 * and is live within 12 hours, or an alarm fires" — this is the alarm, and it is
 * a log line an operator alerts on, the same mechanism the ledger-integrity
 * axes use (DEPLOY.md 3f). The threshold is `REGISTRY_STALE_AFTER_HOURS`, which
 * the ops console's own staleness banner already reads.
 */
export async function registryPublicationAlarm(store: ObjectStore, nowMs: number, cid: string): Promise<number | null> {
  const age = await registryPublicationAgeMs(store, nowMs);
  if (age !== null && age > PUBLICATION_ALARM_MS) {
    logError({ cid, hop: "cron", msg: "registry_publication_stale", duration_ms: age, cap: PUBLICATION_ALARM_MS });
  }
  return age;
}

/**
 * The five-minute tick's registry leg: advance whatever generation is open, and
 * check the publication age. It never plans — the six-hourly tick does that —
 * and it never throws: a storage fault costs this tick's progress and nothing
 * else, exactly as the six-hourly leg's own try/catch does.
 */
export async function registryAdvanceLeg(db: DbLike, store: ObjectStore, opts: RegistryBuildOpts & { deadlineMs?: number }): Promise<AdvanceResult | null> {
  const cid = opts.cid ?? "cron";
  try {
    await registryPublicationAlarm(store, opts.nowMs ?? Date.now(), cid);
    return await advanceRegistryBuild(db, store, { ...opts, plan: false });
  } catch (e) {
    logError({ cid, hop: "cron", msg: "registry_advance_failed", err: e instanceof Error ? e.name : typeof e });
    return null;
  }
}
