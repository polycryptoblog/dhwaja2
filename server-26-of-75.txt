#!/usr/bin/env node
// QUEUE #395 — regenerate `_LIVE-ITEMS.json`'s `owns` arrays from the per-room SCOPE.md files.
//
// `_LIVE-ITEMS.json` is hand-written: `grep -rln _LIVE-ITEMS scripts/ org/rooms/` returns
// NOTHING, so nothing has ever kept its `owns` arrays in step with an ownership move. D-0496
// moved files between rooms, the SCOPE.md files were updated, and this file was not — anything
// reading it for ownership gets the pre-move answer. The SCOPE.md `**Owns (the ONLY files you
// may edit):**` block is the authority a room is actually bound by (B2), so it is the source.
//
// Usage: node scripts/live-items-owns.mjs --root <dir> (--batch <batch-dir> | --all) [--check] [--write]
//                                          [--out <path>]
//   --all    QUEUE #498 (D-0870(6)): every batch org/00-context/intake-0003/BATCHES.json names,
//            as `<batch>-batch`; a named batch with no _LIVE-ITEMS.json is printed as SKIPPED.
//            --out is refused with --all (one path, many batches). Not wired into the gate.
//   --check  exit 1 when a room's owns differ from its SCOPE.md (prints every difference)
//   --write  rewrite the batch's _LIVE-ITEMS.json in place (otherwise nothing is written)
//   --out    write the regenerated JSON to <path> instead of in place
import fs from 'node:fs';
import path from 'node:path';
import { fileURLToPath } from 'node:url';

const argv = process.argv.slice(2);
const opt = (n, d = null) => { const i = argv.indexOf(`--${n}`); return i >= 0 ? argv[i + 1] : d; };
const flag = (n) => argv.includes(`--${n}`);

/**
 * The files a SCOPE.md actually grants. The marker is `**Owns ...:**` followed by a bullet
 * list of backticked paths — NOT a `## OWNS` heading; a parser looking for the heading finds
 * nothing in all ten W3-04 rooms and would report a clean diff over an empty population
 * (meta-guard §1). Returns null when the block is absent, never an empty array.
 */
export function scopeOwns(text) {
  // TWO DIALECTS EXIST IN THE TREE AND ONLY ONE WAS PARSED (R08, QUEUE #395).
  // W3-04..W3-08 write `**Owns (the ONLY files you may edit):**`; W3-01..W3-03 write
  // a `## OWNED FILES — ...` heading. Measured: 51 SCOPE.md carry the first, 41 the
  // second. Parsing only the first made all 30 rooms of W3-01..W3-03 "unresolved",
  // which the meta-guard below correctly reported as VACUOUS rather than CLEAN --
  // the guard held, but the population it protected was empty for three whole
  // batches. Both dialects are a marker line followed by a bullet list of backticked
  // paths terminated by a blank line, so both are read the same way.
  const m = text.match(/\*\*Owns[^*]*\*\*\s*\n([\s\S]*?)(?=\n\s*\n)/)
    || text.match(/^#{1,6}[ \t]*OWNED FILES\b[^\n]*\n([\s\S]*?)(?=\n\s*\n)/mi);
  if (!m) return null;
  const paths = [...m[1].matchAll(/`([^`]+)`/g)].map((x) => x[1].trim()).filter(Boolean);
  return [...new Set(paths)].sort();
}

export function regenerate(root, batch) {
  const bdir = path.join(root, 'org/07-findings', batch);
  const lp = path.join(bdir, '_LIVE-ITEMS.json');
  if (!fs.existsSync(lp)) throw new Error(`${lp} does not exist`);
  const items = JSON.parse(fs.readFileSync(lp, 'utf8'));
  const diffs = [];
  const unresolved = [];
  for (const entry of items) {
    const sp = path.join(bdir, entry.room, 'SCOPE.md');
    if (!fs.existsSync(sp)) { unresolved.push(`${entry.room}: no SCOPE.md — owns left as-is`); continue; }
    const owns = scopeOwns(fs.readFileSync(sp, 'utf8'));
    if (owns === null) { unresolved.push(`${entry.room}: SCOPE.md has no **Owns** block — owns left as-is`); continue; }
    const before = [...new Set(entry.owns ?? [])].sort();
    if (JSON.stringify(before) !== JSON.stringify(owns)) {
      diffs.push({
        room: entry.room,
        stale: before.filter((f) => !owns.includes(f)),
        missing: owns.filter((f) => !before.includes(f)),
        before: before.length, after: owns.length,
      });
    }
    entry.owns = owns;
  }
  return { items, diffs, unresolved, rooms: items.length };
}

/** QUEUE #498: every batch dir BATCHES.json names, in its order; throws when it is unreadable or names none. */
export function allBatches(root) {
  const bp = path.join(root, 'org/00-context/intake-0003/BATCHES.json');
  if (!fs.existsSync(bp)) throw new Error(`${bp} does not exist — --all has no batch list`);
  const names = JSON.parse(fs.readFileSync(bp, 'utf8')).map((b) => b && b.batch).filter(Boolean).map((b) => `${b}-batch`);
  if (!names.length) throw new Error(`${bp} names no batch — --all would check nothing`);
  return [...new Set(names)];
}

// MAIN-MODULE GUARD (R08, QUEUE #395). Everything below is the CLI. Without this
// guard the CLI ran on IMPORT: `import { scopeOwns } from './live-items-owns.mjs'`
// reached the `--batch is required` branch and called process.exit(2), so the module
// could not be imported by a test or by the caller QUEUE #395 asks us to wire it to.
// A tool that cannot be imported cannot be wired, and this is why it was wired to nothing.
// QUEUE #498 (D-0871(10), I10-FX1): REAL paths are compared, so a run through a symlink (a symlinked
// scripts dir or room folder) is still the entry point; path.resolve alone left it a silent no-op, exit 0.
const realOr = (p) => { try { return fs.realpathSync(p); } catch { return path.resolve(p); } };
const isMain = process.argv[1]
  && realOr(process.argv[1]) === realOr(fileURLToPath(import.meta.url));
// One batch: prints its disagreements and returns 'clean' | 'disagree' | 'vacuous'.
function report(res, batch) {
  for (const u of res.unresolved) console.error(`live-items-owns: UNRESOLVED ${batch} ${u}`);
  for (const d of res.diffs) {
    console.log(`${d.room}: owns ${d.before} -> ${d.after}`);
    for (const f of d.stale) console.log(`    stale (in _LIVE-ITEMS, not granted by SCOPE.md): ${f}`);
    for (const f of d.missing) console.log(`    missing (granted by SCOPE.md, absent here):    ${f}`);
  }
  console.log(`live-items-owns: ${batch} — ${res.rooms} room(s), ${res.diffs.length} disagree, ${res.unresolved.length} unresolved`);
  // meta-guard §1 — a check whose population can be EMPTY passes trivially. W3-01-batch's ten
  // SCOPE.md files carry no **Owns** block at all, so every room is unresolved and a bare
  // `diffs.length === 0` would report that batch CLEAN while checking nothing. Fail instead.
  if (res.unresolved.length === res.rooms && res.rooms > 0) return 'vacuous';
  return res.diffs.length === 0 ? 'clean' : 'disagree';
}
if (isMain) {
  const root = opt('root', process.cwd());
  const all = flag('all');
  const batch = opt('batch');
  if (all && batch) { console.error('live-items-owns: --all and --batch are exclusive'); process.exit(2); }
  if (!batch && !all) { console.error('live-items-owns: --batch <batch-dir> or --all is required'); process.exit(2); }
  if (all && opt('out')) { console.error('live-items-owns: --out takes one batch; with --all use --write (each batch in place) or nothing'); process.exit(2); }
  let batches;
  try { batches = all ? allBatches(root) : [batch]; } catch (e) { console.error(`live-items-owns: ${e.message}`); process.exit(2); }
  const verdicts = [];
  for (const b of batches) {
    if (all && !fs.existsSync(path.join(root, 'org/07-findings', b, '_LIVE-ITEMS.json'))) { console.log(`live-items-owns: ${b} — SKIPPED (no _LIVE-ITEMS.json)`); continue; }
    let res;
    try { res = regenerate(root, b); } catch (e) { console.error(`live-items-owns: ${e.message}`); process.exit(2); }
    const v = report(res, b);
    verdicts.push(v);
    if (flag('check')) { if (v === 'vacuous') console.error(`live-items-owns --check: VACUOUS — ${b}: all ${res.rooms} room(s) unresolved, nothing was compared`); continue; }
    const json = `${JSON.stringify(res.items, null, 1)}\n`;
    const out = opt('out');
    if (out) { fs.writeFileSync(out, json); console.log(`live-items-owns: wrote ${out}`); continue; }
    if (flag('write')) { fs.writeFileSync(path.join(root, 'org/07-findings', b, '_LIVE-ITEMS.json'), json); console.log(`live-items-owns: rewrote ${b} in place`); continue; }
    console.log('live-items-owns: nothing written (pass --write or --out)');
  }
  if (all) console.log(`live-items-owns --all: ${batches.length} batch(es) named, ${verdicts.length} checked, ${batches.length - verdicts.length} skipped`);
  if (flag('check')) {
    // --all over zero checked batches is the same empty population: refuse it.
    if (!verdicts.length) { console.error('live-items-owns --check: VACUOUS — no batch had a _LIVE-ITEMS.json'); process.exit(1); }
    process.exit(verdicts.every((v) => v === 'clean') ? 0 : 1);
  }
}

  // (CI and the cloud gate run in UTC), and corepack never waits on a [Y/n] prompt nobody can answer.
  const childEnv = {
    ...env, PATH: `${shims}:${dirname(node)}:${env.PATH || ""}`, COREPACK_ENABLE_DOWNLOAD_PROMPT: "0", TZ: "UTC",
    VITEST_MAX_THREADS: workers, VITEST_MAX_FORKS: workers,
  };
  return { home, node, cores, build, repo, gate, box, shims, snapDir, par, workers, slots, roomWorkers, childEnv, heartbeatMs: HEARTBEAT_MS, pollMs: POLL_MS, prepareCapMs, self: fileURLToPath(import.meta.url) };
}

export function defaultSh(cmd, args, opts = {}) {
  return execFileSync(cmd, args, { encoding: "utf8", stdio: ["ignore", "pipe", "pipe"], maxBuffer: 512 * 1024 * 1024, ...opts });
}
function firstTool(tools, sh) {
  for (const [cmd, args] of tools) {
    try { return sh(cmd, args).trim().split(/\s+/)[0]; } catch (e) { if (e?.code !== "ENOENT") throw e; }
  }
  throw new Error(`none of ${tools.map((t) => t[0]).join(", ")} is on PATH`);
}
/** md5 of a file's bytes by the system tool (GNU md5sum, else BSD md5 -q). */
export const md5File = (file, sh = defaultSh) => firstTool([["md5sum", [file]], ["md5", ["-q", file]]], sh);
/** sha256 of a file's bytes by the system tool (GNU sha256sum, else BSD shasum -a 256). */
export const sha256File = (file, sh = defaultSh) => firstTool([["sha256sum", [file]], ["shasum", ["-a", "256", file]]], sh);

export function defaultDeps() {
  return { sh: defaultSh, spawn: nodeSpawn, kill: (pid, sig) => process.kill(pid, sig), md5: (f) => md5File(f), sha256: (f) => sha256File(f), now: () => new Date(), log: (m) => console.log(`[${nowIso()}] ${m}`) };
}

/** 32-bit FNV-1a over UTF-8 bytes: a name suffix, not a digest of record. */
export function fnv1a(s) {
  let h = 0x811c9dc5;
  for (const b of Buffer.from(String(s), "utf8")) { h ^= b; h = Math.imul(h, 0x01000193) >>> 0; }
  return h.toString(16).padStart(8, "0");
}
/** Leg output names: the legacy `[^a-z0-9]+ -> _` name, plus a hash suffix when two legs collide or the name is long. */
export function legFileNames(legs) {
  const safe = legs.map((l) => l.replace(/[^a-z0-9]+/gi, "_"));
  return legs.map((l, i) => (safe.filter((s) => s === safe[i]).length > 1 || safe[i].length > LEG_NAME_MAX ? `${safe[i].slice(0, LEG_NAME_MAX)}-${fnv1a(l)}` : safe[i]));
}
export const sanitizeId = (id) => String(id).replace(/[^A-Za-z0-9._-]/g, "_");
/** A tree-relative destination: never absolute, never `..`, every segment printable ASCII (0x20-0x7E, so no Unicode look-alike such as U+017F reaches the Mac volume's case folding; FXR-1, D-1420), and no segment at ANY depth is .git, node_modules or .scratch in any case (the Mac volume is case-insensitive). */
export function validDest(d) {
  if (typeof d !== "string" || d === "" || d.includes("\\")) return false; // an absolute path has an empty first segment, refused below
  const parts = d.split("/");
  if (parts.some((p) => p === ".." || p === "" || p === ".")) return false;
  if (parts.some((p) => !PRINTABLE_ASCII_RE.test(p))) return false;
  return !parts.some((p) => RESERVED_DEST_SEGMENTS.includes(p.toLowerCase()));
}
/** Runner-side containment, checked before any byte is written: the realpath of the deepest existing path of <gate>/<dest> stays inside the gate copy, and no existing segment is a symlink. Null when contained, else why. */
export function containedDest(gate, dest) {
  if (!validDest(dest)) return `dest ${JSON.stringify(dest)} is refused (absolute, '..', .git, node_modules or .scratch, or a segment that is not printable ASCII)`;
  const root = realpathSync(gate);
  const walked = [];
  let at = gate;
  for (const part of dest.split("/")) {
    const next = join(at, part);
    let st;
    try { st = lstatSync(next); } catch { break; }
    walked.push([next, st]);
    at = next;
  }
  const real = realpathSync(at);
  if (real !== root && !real.startsWith(root + sep)) return `dest ${JSON.stringify(dest)} resolves to ${real}, outside the gate copy`;
  const link = walked.find(([, st]) => st.isSymbolicLink());
  if (link) return `dest ${JSON.stringify(dest)} crosses a symlink at ${link[0].slice(gate.length + 1)}`;
  return null;
}
/** A request's test file: non-empty, no leading '-', not absolute, no '..' segment, no backslash (SFX-16). */
export function validFile(f) {
  return typeof f === "string" && f !== "" && !f.startsWith("-") && !isAbsolute(f) && !f.includes("\\") && !f.split("/").includes("..");
}
/** A ref or base: a ref name or sha shape, never an option (SFX-16). */
export function validRef(r) {
  return typeof r === "string" && /^[A-Za-z0-9][A-Za-z0-9._/-]*$/.test(r) && !r.includes("..");
}
/** Gate legs: absent or "all", or a non-empty list of the gate's known legs and `vitest:<file> [<file>...]` legs (SFX-16). */
export function validLegs(legs) {
  if (legs === undefined || legs === "all") return true;
  if (!Array.isArray(legs) || legs.length === 0) return false;
  return legs.every((l) => typeof l === "string" && (ALL_LEGS.includes(l) || (l.startsWith("vitest:") && l.slice(7).split(" ").every(validFile))));
}
export function countBytes(buf, find) {
  const f = Buffer.from(find, "utf8");
  let n = 0;
  for (let i = buf.indexOf(f); i !== -1; i = buf.indexOf(f, i + f.length)) n++;
  return n;
}

const readText = (p) => { try { return readFileSync(p, "utf8"); } catch { return null; } };
const readJson = (p) => { const t = readText(p); if (t === null) return null; try { return JSON.parse(t); } catch { return null; } };
/** create-then-rename, so no reader ever sees a half-written file */
export function writeAtomic(p, text) { mkdirSync(dirname(p), { recursive: true }); writeFileSync(`${p}.tmp`, text); renameSync(`${p}.tmp`, p); }
const statePath = (box, id) => join(box, "runs", id, "state.json");

}

/** W0-02 + D-1414: any terminal record in runs/<id>/ means the request is never run again. */

/** queue/*.json, FIFO by file mtime then id. */
export function listQueue(box) {
  const q = join(box, "queue");
  let names;
  try { names = readdirSync(q); } catch { return []; }
  const out = [];
  for (const n of names) {
    if (!n.endsWith(".json")) continue;
    try { out.push({ id: n.slice(0, -5), path: join(q, n), mtimeMs: statSync(join(q, n)).mtimeMs }); } catch { /* claimed meanwhile */ }
  }
  return out.sort((a, b) => a.mtimeMs - b.mtimeMs || (a.id < b.id ? -1 : a.id > b.id ? 1 : 0));
}

/** Why a v2 request is refused before it runs, or null. */
export function refusal(box, id, req) {
  if (!req || typeof req !== "object") return "unreadable request";
  if (sanitizeId(id) !== id) return `id ${JSON.stringify(id)} changes under sanitize`;
  if (id === "." || id === "..") return `id ${JSON.stringify(id)} is not a request id`;
  if (req.id !== id) return `request id ${JSON.stringify(req.id)} is not its file name ${JSON.stringify(id)}`;
  if (existsSync(join(box, "runs", id))) return `duplicate id: runs/${id}/ exists`;
  if (existsSync(join(box, "queue", `${id}.claimed`))) return `duplicate id: queue/${id}.claimed exists`;
  if (!V2_KINDS.has(req.kind)) return `unknown kind ${JSON.stringify(req.kind)}`;
  for (const k of ["room", "run"]) if (req[k] != null && typeof req[k] !== "string") return `${k} must be a string`; // D-1422: both are copied into state.json and status.json
  if (req.ref !== undefined && !validRef(req.ref)) return `bad ref ${JSON.stringify(req.ref)}`;
  if (req.base !== undefined && !validRef(req.base)) return `bad base ${JSON.stringify(req.base)}`;
  if (req.kind === "gate") {
    // D-1418: the client's own gate request carries env {} and overlay [] and its --arg legs; anything more is refused
    if (req.files !== undefined) return "kind gate takes no files";
    if (req.cwd !== undefined) return "kind gate takes no cwd";
    if (req.overlay !== undefined && !(Array.isArray(req.overlay) && req.overlay.length === 0)) return "kind gate takes no overlay";
    if (req.env !== undefined && !(req.env && typeof req.env === "object" && !Array.isArray(req.env) && Object.keys(req.env).length === 0)) return "kind gate takes no env";
    if (!validLegs(req.legs)) return `bad legs ${JSON.stringify(req.legs)}`;
  }
  if (req.env !== undefined) { // SF2-12 (D-1421): the client's allowlist, for every kind, before req.env reaches the child env
    if (!req.env || typeof req.env !== "object" || Array.isArray(req.env) || Object.values(req.env).some((v) => typeof v !== "string")) return "env must be an object of string values";
    const bad = Object.keys(req.env).find((k) => !envAllowed(k));
    if (bad !== undefined) return `env key ${JSON.stringify(bad)} is not in the allowlist (${ENV_ALLOW.join(", ")}); TZ stays UTC`;
  }
  if (req.kind !== "gate" && !(Array.isArray(req.files) && req.files.length > 0 && req.files.every(validFile))) return "files must be a non-empty array of tree-relative paths (no leading '-', no '..')";
  if (req.args !== undefined && !(Array.isArray(req.args) && req.args.every((a) => typeof a === "string"))) return "args must be an array of strings";
  if (req.timeout_s !== undefined && !(Number.isInteger(req.timeout_s) && req.timeout_s > 0)) return "timeout_s must be a positive integer";
  if (req.cwd !== undefined && !validDest(req.cwd)) return `bad cwd ${JSON.stringify(req.cwd)}`;
  if (req.overlay !== undefined && !Array.isArray(req.overlay)) return "overlay must be a list"; // Lead 2 (SF2-11): the loop below would throw
  for (const o of req.overlay ?? []) if (!validDest(o?.dest)) return `bad overlay dest ${JSON.stringify(o?.dest)}`;
  if (req.kind === "mutant") {
    const ms = req.mutations;
    if (!Array.isArray(ms) || ms.length === 0) return "kind mutant needs a non-empty mutations list";
    for (const m of ms) if (!m || ["name", "file", "find", "replace"].some((k) => typeof m[k] !== "string") || m.find === "" || !validDest(m.file)) return `bad mutation ${JSON.stringify(m?.name ?? m)}`;
    if (new Set(ms.map((m) => m.name)).size !== ms.length) return "mutation names must be unique";
  }
  return null;
}

/** Why a queue entry without `v: 2` is refused, or null (item 9, SFX-16): only the runner's own legacy entry, in the legacy shape. */
export function legacyRefusal(box, id, req) {
  if (!req || typeof req !== "object") return "unreadable request";
  if (req.v !== undefined) return `unknown request version ${JSON.stringify(req.v)}`;
  if (req.legacy !== true) return "a queue entry without v: 2 must be the runner's legacy entry (legacy: true)";
  if (id === "." || id === "..") return `id ${JSON.stringify(id)} is not a request id`; // an id sanitize changes fails the file-name check below
  if (typeof req.id !== "string" || sanitizeId(req.id) !== id) return `request id ${JSON.stringify(req.id)} is not its file name ${JSON.stringify(id)}`;
  if (existsSync(join(box, "runs", id))) return `duplicate id: runs/${id}/ exists`;
  if (existsSync(join(box, "queue", `${id}.claimed`))) return `duplicate id: queue/${id}.claimed exists`;
  for (const k of LEGACY_FORBIDDEN) if (req[k] !== undefined) return `a legacy request carries no ${k}`;
  for (const k of ["room", "run"]) if (req[k] != null && typeof req[k] !== "string") return `${k} must be a string`; // D-1422
  if (req.ref !== undefined && !validRef(req.ref)) return `bad ref ${JSON.stringify(req.ref)}`;
  if (req.base !== undefined && !validRef(req.base)) return `bad base ${JSON.stringify(req.base)}`;
  if (!validLegs(req.legs)) return `bad legs ${JSON.stringify(req.legs)}`;
  if (req.repin !== undefined && typeof req.repin !== "boolean") return "repin must be a boolean";
  return null;
}

/** The parent's overall cap: tests get the child's own timeout_s plus a backstop; mutant gets (N+1) x timeout_s. */
export function capMs(req) {
  const t = (req.timeout_s ?? DEFAULT_TIMEOUT_S) * 1000;
  if (req.kind === "mutant") return (req.mutations.length + 1) * t;
  if (req.kind === "tests") return t + TESTS_BACKSTOP_MS;
  return null; // legacy and gate: LEG_TIMEOUT_MS per leg, unchanged
}

/** vitest's summary counts and the names of failing tests. */
export function parseVitest(out) {
  const num = (line, word) => { const m = new RegExp(`(\\d+) ${word}`).exec(line ?? ""); return m ? Number(m[1]) : 0; };
  const filesLine = /^\s*Test Files\s+(.*)$/m.exec(out)?.[1];
  const testsLine = /^\s*Tests\s+(.*)$/m.exec(out)?.[1];
  const errorsLine = /^\s*Errors\s+(.*)$/m.exec(out)?.[1];
  const failing = [...new Set([...out.matchAll(/^\s*FAIL\s+(\S.* > .+?)\s*$/gm)].map((m) => m[1]))];
  return {
    files: { failed: num(filesLine, "failed"), passed: num(filesLine, "passed"), skipped: num(filesLine, "skipped") },
    tests: { failed: num(testsLine, "failed"), passed: num(testsLine, "passed"), skipped: num(testsLine, "skipped"), todo: num(testsLine, "todo") },
    errors: num(errorsLine, "errors?"),
    ran: testsLine !== undefined,
    failing,
  };
}
/** Item 3: a run executed a test only when one passed or failed; skipped and todo tests did not run. */
export const ranTests = (p) => p.tests.passed + p.tests.failed > 0;
export const noTestMessage = (p, what = "the run") => `no test ran: ${what} had 0 passed and 0 failed (${p.tests.skipped} skipped, ${p.tests.todo} todo); check --files and any -t filter`;
/** A mutation's outcome from its run: red is a kill; error (no test ran, or an error) is never a kill. */
export function mutationOutcome(r) {
  if (r.timedOut) return { outcome: "error", reason: "timed out" };
  const p = parseVitest(r.out);
  if (p.tests.failed > 0) return { outcome: "red", parsed: p };
  if (!ranTests(p)) return { outcome: "error", reason: "no test ran", parsed: p }; // no Tests line at all counts 0 too
  if (p.errors > 0 || r.code !== 0) return { outcome: "error", reason: `exit ${r.code} with no failing test`, parsed: p };
  return { outcome: "green", parsed: p };
}
/** A start-time token for a pid (ps lstart), so a later start can tell a reused pid from the one recorded; "" when ps cannot say. */
export function startToken(sh, pid) {
  try { return sh("ps", ["-o", "lstart=", "-p", String(pid)]).trim(); } catch { return ""; }
}

/** The Node floor, as a problem line or null (SN-9, SN-12). D-1134 (I35): 22.23.4 is the October 2026 security release (CVE-2026-111497 Brotli; the eMudhra intermediates blocked). */
export function nodeFloorProblem(nodeV) {
  const [maj, min, pat] = String(nodeV).split(".").map(Number);
  if (maj === 22 && (min > 23 || (min === 23 && pat >= 4))) return null;
  return `Node ${nodeV} found; the repo needs Node >=${NODE_FLOOR} and <23 (brew upgrade node@22, then put it first on PATH).`;
}

function checkTools(cfg, sh) {
  const problems = [];
  const floor = nodeFloorProblem(process.versions.node);
  if (floor) problems.push(floor);
  let pnpm = null;
  for (const c of [["pnpm", ["--version"]], ["corepack", ["pnpm", "--version"]]]) {
    try { sh(c[0], c[1], { cwd: cfg.build, env: cfg.childEnv }); pnpm = c[0] === "pnpm" ? ["pnpm"] : ["corepack", "pnpm"]; break; } catch { /* next */ }
  }
  if (!pnpm) problems.push("pnpm not found: run `corepack enable` (comes with Node 22), or `npm install -g pnpm@10.28.0`.");
  try { sh("git", ["--version"]); } catch { problems.push("git not found (xcode-select --install)."); }
  try { sh("rsync", ["--version"]); } catch { problems.push("rsync not found."); }
  return { problems, pnpm };
}

/** macOS has no sha256sum/md5sum (GNU names some tests spawn): small shims that call Node or the BSD tools. */
export function installShims(cfg) {
  mkdirSync(cfg.shims, { recursive: true });
  const shim = (name, body) => { const f = join(cfg.shims, name); writeFileSync(f, `#!/bin/sh\n${body}\n`); chmodSync(f, 0o755); };
  // I27 (I27-PORT-TMP P5): BSD shasum is a Perl script and was slow enough to time out the >1 MiB
  // room-write-audit tests. The hashers are a small Node program (GNU output, -z, stdin, "-");
  // anything else (e.g. -c) falls back to shasum.
  const nodeSum = (algo) => {
    const f = join(cfg.shims, `gnusum-${algo}.cjs`);
    writeFileSync(f, `const c=require("crypto"),fs=require("fs");const a=process.argv.slice(2);let z=false;const files=[];
for(const x of a){if(x==="-z"){z=true;continue}if(x==="--"||x==="-b"||x==="--binary"||x==="-t"||x==="--text")continue;if(x.startsWith("-")&&x!=="-"){require("child_process").execFileSync("shasum",["-a","${algo === "sha256" ? 256 : 1}",...a],{stdio:"inherit"});return}files.push(x)}
if(!files.length)files.push("-");let rc=0;const out=[];
for(const f of files){try{const d=f==="-"?fs.readFileSync(0):fs.readFileSync(f);out.push(c.createHash("${algo}").update(d).digest("hex")+"  "+f)}catch(e){process.stderr.write("sha: "+f+": "+e.message+"\\n");rc=1}}
process.exitCode=rc;process.stdout.write(out.map((l)=>l+(z?"\\0":"\\n")).join(""));`); // no process.exit: on macOS a pipe write is async and exit would truncate it
    return `exec "${cfg.node}" "${f}" "$@"`;
  };
  shim("sha256sum", nodeSum("sha256"));
  shim("sha1sum", nodeSum("sha1"));
  shim("md5sum", 'if [ "$#" -eq 0 ]; then md5 -r | sed "s/\\$/  -/"; else for f in "$@"; do printf "%s  %s\\n" "$(md5 -q "$f")" "$f"; done; fi');
}

/** Steps 1-2 of the legacy run, for EVERY kind (W0-07): the copy's org/ is a function of (repo now, ref, base). */
export function prepareCopy(ctx) {
  const { req, cfg, sh, note, gate, id } = ctx;
  const ref = String(req.ref || "HEAD");
  if (!existsSync(join(gate, ".git"))) { note(`cloning ${cfg.build} -> ${gate}`); sh("git", ["clone", "--no-hardlinks", cfg.build, gate]); }
  sh("git", ["-C", gate, "reset", "--hard", "-q"]);
  sh("git", ["-C", gate, "clean", "-fdq"]); // removes leftovers of an interrupted run (node_modules is ignored, so it stays)
  sh("git", ["-C", gate, "fetch", "-q", "--force", "--tags", cfg.build, "+refs/heads/*:refs/remotes/build/*"]);
  const sha = sh("git", ["-C", gate, "rev-parse", `${ref}^{commit}`]).trim();
  sh("git", ["-C", gate, "checkout", "-q", "-f", "--detach", sha]);
  note(`checked out ${ref} = ${sha.slice(0, 12)}`);
  ctx.set({ sha, ref });
  // org records are current only in the repo; the batch's skill edits are current only in the build tree
  note("overlaying org/ from the repo, then org/skills and .claude from the requested ref");
  sh("rsync", ["-a", "--exclude", ".scratch*", "--exclude", "LAND-BACKUPS/", `${cfg.repo}/org/`, `${gate}/org/`]);
  sh("git", ["-C", gate, "checkout", "-q", sha, "--", "org/skills", ".claude"]);
  // QUEUE #971: org files the batch changed in the build tree (base..ref) win over the repo's copies
  if (req.base) {
    const changed = sh("git", ["-C", gate, "diff", "-z", "--name-only", "--diff-filter=AMR", `${req.base}^{commit}`, sha, "--", "org"]).split("\0").filter(Boolean); // SF2-13: -z, so a non-ASCII name is not quoted
    for (let i = 0; i < changed.length; i += 200) sh("git", ["-C", gate, "checkout", "-q", sha, "--", ...changed.slice(i, i + 200)]);
    note(`base ${req.base}: ${changed.length} org path(s) changed in base..ref taken from the ref`);
  }
  sh("git", ["-C", gate, "add", "-A", "org", ".claude"]);
  try { sh("git", ["-C", gate, "-c", "user.name=mac-gate", "-c", "user.email=mac-gate@local", "commit", "-qm", `mac-gate overlay for ${id}`]); } catch { /* nothing to commit */ }
  return sha;
}

/** Lead 1 (SF2-9, D-1421): an overlay source is a regular file reached from queue/<id>.overlay/ with no symlink at any segment, the root and the leaf included (copyFileSync follows links). Null when it is, else why. */
function overlaySourceRefusal(queueDir, id, dest) {
  const segs = [`${id}.overlay`, ...dest.split("/")];
  let at = queueDir;
  for (const [i, s] of segs.entries()) {
    at = join(at, s);
    let st;
    try { st = lstatSync(at); } catch { return `overlay source ${JSON.stringify(dest)} is missing from queue/${id}.overlay/`; }
    if (st.isSymbolicLink()) return `overlay source ${JSON.stringify(dest)} crosses a symlink at queue/${segs.slice(0, i + 1).join("/")}`;
    if (i === segs.length - 1 && !st.isFile()) return `overlay source ${JSON.stringify(dest)} is not a regular file`;
  }
  return null;
}

/** W0-15: the request's overlay files, applied after the overlay commit and committed, so ls-files sees new files. */
export function applyOverlay(ctx) {
  const { req, sh, gate, id, cfg, deps } = ctx;
  const applied = [];
  const queueDir = join(cfg.box, "queue");
  for (const o of req.overlay ?? []) { // every entry is checked before any directory or byte is written
    const why = containedDest(gate, o.dest) ?? overlaySourceRefusal(queueDir, id, o.dest);
    if (why) throw new Error(`overlay refused: ${why}`);
  }
  for (const o of req.overlay ?? []) {
    const to = join(gate, o.dest);
    mkdirSync(dirname(to), { recursive: true });
    copyFileSync(join(queueDir, `${id}.overlay`, o.dest), to);
    const md5 = deps.md5(to);
    if (o.md5 && md5 !== o.md5) throw new Error(`overlay ${o.dest}: md5 ${md5} is not the ${o.md5} read at submit`);
    applied.push({ src: o.src, dest: o.dest, md5 });
  }
  if (applied.length) {
    const dests = applied.map((a) => a.dest);
    // item 2: a gitignored dest is added with -f; an overlay equal to the ref stages nothing and commits empty; both are recorded
    const ignored = new Set(dests.filter((d) => { try { sh("git", ["-C", gate, "check-ignore", "-q", "--", d]); return true; } catch { return false; } })); // plain: check-ignore refuses literal pathspecs (SF3-8)
    const staged = new Set();
    const literal = { env: { ...cfg.childEnv, GIT_LITERAL_PATHSPECS: "1" } }; // FX3 (FX2R-8, D-1425): a dest such as ':a.ts' is a path, never pathspec magic
    for (let i = 0; i < dests.length; i += 200) {
      sh("git", ["-C", gate, "add", "-f", "-A", "--", ...dests.slice(i, i + 200)], literal);
      for (const n of sh("git", ["-C", gate, "diff", "--cached", "-z", "--name-only", "--", ...dests.slice(i, i + 200)], literal).split("\0")) if (n) staged.add(n); // -z: a '"' in a dest is not quoted
    }
    sh("git", ["-C", gate, "-c", "user.name=mac-gate", "-c", "user.email=mac-gate@local", "commit", "-q", "--allow-empty", "-m", `mac-gate request overlay ${id}`]);
    for (const a of applied) { a.ignored = ignored.has(a.dest); a.changed = staged.has(a.dest); }
  }
  const porcelain = sh("git", ["-C", gate, "status", "--porcelain"]);
  if (porcelain.trim()) throw new Error(`gate copy not clean after the overlay commit:\n${porcelain.trim()}`);
  ctx.note(`overlay: ${applied.length} file(s) committed; git status --porcelain empty`);
  return applied;
}

/** W0-07: pnpm install only when pnpm-lock.yaml's sha256 differs from the last install's. */
export function installIfNeeded(ctx, pnpm) {
  const { sh, gate, note, deps } = ctx;
  const stamp = join(gate, ".git", "mac-gate-lock.sha256");
  const want = deps.sha256(join(gate, "pnpm-lock.yaml"));
  const have = readText(stamp)?.trim();
  if (have === want && existsSync(join(gate, "node_modules"))) { note(`pnpm install skipped: pnpm-lock.yaml sha256 ${want.slice(0, 12)} unchanged`); return true; }
  ctx.set({ run_state: "installing" });
  note("pnpm install --frozen-lockfile");
  try { sh(pnpm[0], [...pnpm.slice(1), "install", "--frozen-lockfile"], { cwd: gate }); }
  catch (e) { note(`install FAILED: ${String(e.stderr || e.message).slice(-2000)}`); return false; }
  writeFileSync(stamp, `${want}\n`);
  return true;
}

function runLeg(ctx, name, args, outFile, cwd) {
  const { deps, cfg } = ctx;
  return new Promise((done) => {
    const t0 = Date.now();
    // A leg "vitest:<file> [<file>...]" reruns named test files on their own (light load), outside gate.mjs.
    const isFiles = args[0]?.startsWith("vitest:");
    const cmdArgs = isFiles ? [join(cwd, "node_modules/vitest/vitest.mjs"), "run", args[0].slice(7), ...args.slice(1)] : ["scripts/gate.mjs", ...args, "--out", outFile];
    const child = deps.spawn(cfg.node, cmdArgs, { cwd, env: { ...cfg.childEnv, NO_COLOR: "1", FORCE_COLOR: "0" } });
    let buf = "";
    child.stdout?.on("data", (d) => { buf += d; });
    child.stderr?.on("data", (d) => { buf += d; });
    const timer = setTimeout(() => { buf += `\n[mac-gate] leg timed out after ${LEG_TIMEOUT_MS / 60000} min, killed\n`; child.kill("SIGKILL"); }, LEG_TIMEOUT_MS);
    child.on("close", (code) => { clearTimeout(timer); done({ name, code: code ?? 137, secs: Math.round((Date.now() - t0) / 1000), out: buf }); });
  });
}

/** Steps 4-6 of the legacy run: optional re-pin loop, the legs, GATE-LOGS.md and SUMMARY.txt. */
export async function gateLegs(ctx, sha) {
  const { req, sh, note, gate, out, id, cfg } = ctx;
  const ref = String(req.ref || "HEAD");
  if (req.repin) {
    ctx.set({ run_state: "re-pinning" });
    const repinLog = join(out, "REPIN-LOG.txt");
    for (let i = 0; i < 80; i++) {
      let outText = "", code = 0;
      try { outText = sh(cfg.node, ["scripts/check-invariants.mjs"], { cwd: gate }); }
      catch (e) { outText = String(e.stdout || "") + String(e.stderr || ""); code = e.status ?? 1; }
      writeFileSync(join(out, "repin-last.txt"), outText);
      if (code === 0) { appendFileSync(repinLog, `GREEN after ${i}\n`); break; }
      const mf = join(gate, "fixtures/invariant-manifest.json");
      const m = JSON.parse(readFileSync(mf, "utf8"));
      let changed = 0;
      for (const [, f, h] of outText.matchAll(PIN_NEW_RE)) {
        if (m.files[f] !== h) { m.files[f] = h; changed++; appendFileSync(repinLog, `pin-new ${f} ${h}\n`); }
      }
      for (const [, f, exp, act] of outText.matchAll(REPIN_RE)) {
        if (m.files[f] === exp) { m.files[f] = act; changed++; appendFileSync(repinLog, `repin ${f} ${exp} -> ${act}\n`); }
      }
      if (outText.includes("per-file test count mismatch")) {
        for (const [, f, n] of outText.matchAll(TESTS_RE)) {
          m.tests ??= {};
          if (m.tests[f] !== Number(n)) { appendFileSync(repinLog, `tests ${f} ${m.tests[f]} -> ${n}\n`); m.tests[f] = Number(n); changed++; }
        }
      }
      if (!changed) { appendFileSync(repinLog, "STOP: unhandled failure (see repin-last.txt)\n"); break; }
      writeFileSync(mf, JSON.stringify(m, null, 2) + "\n");
    }
    copyFileSync(join(gate, "fixtures/invariant-manifest.json"), join(out, "invariant-manifest.json"));
    sh("git", ["-C", gate, "add", "fixtures/invariant-manifest.json"]);
    try { sh("git", ["-C", gate, "-c", "user.name=mac-gate", "-c", "user.email=mac-gate@local", "commit", "-qm", `mac-gate re-pin for ${id}`]); } catch { /* unchanged */ }
    note(`re-pin: ${readFileSync(repinLog, "utf8").trim().split("\n").pop()}`);
  }
  // the legs: quick ones first, then the 8 vitest shards PAR at a time, then the record legs
  const legs = Array.isArray(req.legs) ? req.legs : ALL_LEGS;
  ctx.set({ run_state: "running legs", legs: {} });
  const legState = {};
  const results = [];
  const legDir = join(out, "legs");
  mkdirSync(legDir, { recursive: true });
  const names = legFileNames(legs);
  const runOne = async (leg) => {
    const safe = names[legs.indexOf(leg)];
    const o = join(legDir, `${safe}.md`);
    legState[leg] = { state: "running", began: nowIso(ctx.deps.now()) }; ctx.set({ legs: legState });
    const r = await runLeg(ctx, leg, leg.split(" "), o, gate);
    writeFileSync(join(legDir, `${safe}.txt`), r.out);
    legState[leg] = { state: "done", exit: r.code, secs: r.secs }; ctx.set({ legs: legState });
    note(`${leg} exit ${r.code} in ${r.secs}s`);
    results.push({ leg, ...r, o });
  };
  const shards = legs.filter((l) => l.startsWith("--shard"));
  const before = legs.filter((l) => !l.startsWith("--shard") && ["--typecheck", "--lint", "--lint-arch"].includes(l));
  const after = legs.filter((l) => !l.startsWith("--shard") && !before.includes(l));
  for (const l of before) await runOne(l);
  const queue = [...shards];
  await Promise.all(Array.from({ length: Math.min(cfg.par, queue.length) }, async () => { while (queue.length) await runOne(queue.shift()); }));
  for (const l of after) await runOne(l);
  // one GATE-LOGS.md in leg order, then the gate's own summary
  const merged = join(out, "GATE-LOGS.md");
  writeFileSync(merged, "");
  for (const leg of legs) {
    const r = results.find((x) => x.leg === leg);
    if (r && existsSync(r.o)) appendFileSync(merged, readFileSync(r.o, "utf8"));
  }
  let summary = "";
  try { summary = sh(cfg.node, ["scripts/gate.mjs", "--summarize", "--out", merged], { cwd: gate }); }
  catch (e) { summary = String(e.stdout || "") + String(e.stderr || ""); }
  const table = results.sort((a, b) => legs.indexOf(a.leg) - legs.indexOf(b.leg)).map((r) => `${r.code === 0 ? "PASS" : "FAIL"}  ${r.leg}  (${r.secs}s)`).join("\n");
  writeFileSync(join(out, "SUMMARY.txt"), `mac-gate run ${id}: ref ${ref} = ${sha}\ncores ${cfg.cores}, shards in parallel ${cfg.par}\n\n${table}\n\n--- gate.mjs --summarize ---\n${summary}`);
  const red = results.filter((r) => r.code !== 0).map((r) => r.leg);
  const runState = red.length ? `done: RED (${red.join(", ")})` : "done: GREEN";
  ctx.set({ run_state: runState, state: "done", exit: red.length ? 1 : 0, finished: nowIso(ctx.deps.now()) });
  note(runState);
}

/** One vitest run in its own process group, killed as a group at its timeout; its pgid is recorded while it lives. */
export function runVitest(ctx, files, timeoutMs, label) {
  const { req, cfg, deps, gate } = ctx;
  const rw = cfg.roomWorkers ?? cfg.workers; // D-1466: slots x room workers stays within the cores
  const env = { ...cfg.childEnv, VITEST_MAX_THREADS: rw, VITEST_MAX_FORKS: rw, ...(req.env ?? {}), NO_COLOR: "1", FORCE_COLOR: "0", TZ: "UTC" };
  return new Promise((done) => {
    const t0 = Date.now();
    const child = deps.spawn(cfg.node, [join(gate, "node_modules/vitest/vitest.mjs"), "run", ...files, ...(req.args ?? [])], { cwd: join(gate, req.cwd ?? ""), env, detached: true });
    const prior = readState(cfg.box, ctx.id) ?? {};
    const pgids = [...(prior.pgids ?? []), child.pid];
    const starts = { ...(prior.starts ?? {}), [child.pid]: startToken(deps.sh, child.pid) };
    ctx.set({ pgids, starts, running_since: prior.running_since ?? deps.now().toISOString() }); // ms kept: the parent cap is measured from it
    let buf = "", timedOut = false, grace = null;
    child.stdout?.on("data", (d) => { buf += d; });
    child.stderr?.on("data", (d) => { buf += d; });
    const timer = setTimeout(() => {
      timedOut = true;
      buf += `\n[mac-gate] ${label} timed out after ${timeoutMs / 1000} s, process group killed\n`;
      try { deps.kill(-child.pid, "SIGTERM"); } catch { /* gone */ }
      grace = setTimeout(() => { try { deps.kill(-child.pid, "SIGKILL"); } catch { /* gone */ } }, KILL_GRACE_MS);
    }, timeoutMs);
    child.on("close", (code) => {
      clearTimeout(timer);
      if (grace) clearTimeout(grace);
      ctx.set({ pgids: (readState(cfg.box, ctx.id)?.pgids ?? []).filter((p) => p !== child.pid) });
      done({ code: code ?? 137, out: buf, timedOut, secs: Math.round((Date.now() - t0) / 1000) });
    });
  });
}

const porcelainOf = (ctx, paths = []) => ctx.sh("git", ["-C", ctx.gate, "status", "--porcelain", ...(paths.length ? ["--", ...paths] : [])], { env: { ...ctx.cfg.childEnv, GIT_LITERAL_PATHSPECS: "1" } }).replace(/\s+$/, ""); // FX3 (SF3-10): a mutation file is a path, never pathspec magic
const sectionOf = (title, text) => `===== ${title} =====\n${text}\n`;

export async function runTests(ctx, overlay) {
  const { req, out } = ctx;
  const t = (req.timeout_s ?? DEFAULT_TIMEOUT_S) * 1000;
  ctx.set({ run_state: "running tests" });
  const r = await runVitest(ctx, req.files, t, "tests");
  writeFileSync(join(out, "output.txt"), r.out);
  const parsed = parseVitest(r.out);
  const noTest = r.code === 0 && !ranTests(parsed); // read only when not timed out (the branch below)
  const dirty = porcelainOf(ctx);
  const result = { id: ctx.id, kind: req.kind, room: req.room, run: req.run, ref: req.ref, sha: readState(ctx.cfg.box, ctx.id)?.sha, files: req.files, args: req.args ?? [], cwd: req.cwd ?? "", env: req.env ?? {}, overlay, exit_code: r.code, timed_out: r.timedOut, secs: r.secs, ...parsed, ...(dirty ? { tree_dirty_after_run: dirty.split("\n") } : {}) };
  writeAtomic(join(out, "result.json"), JSON.stringify(result, null, 1) + "\n");
  if (r.timedOut) ctx.set({ state: "timeout", exit: 6, finished: nowIso(ctx.deps.now()) });
  else if (noTest) ctx.set({ state: "error", exit: 7, message: noTestMessage(parsed), finished: nowIso(ctx.deps.now()) });
  else ctx.set({ state: "done", exit: r.code === 0 ? 0 : 1, finished: nowIso(ctx.deps.now()) });
  ctx.note(`tests exit ${r.code}${r.timedOut ? " (timed out)" : ""} in ${r.secs}s`);
}

export async function runMutation(ctx, m, i, overlay, t) {
  const { gate, out, deps } = ctx;
  const rec = { name: m.name, file: m.file };
  const path = join(gate, m.file);
  if (!validDest(m.file) || !existsSync(path)) return { ...rec, outcome: "refused", reason: "file not found in the gate copy" };
  const outside = containedDest(gate, m.file);
  if (outside) return { ...rec, outcome: "refused", reason: outside };
  const bytes = readFileSync(path);
  const before = deps.md5(path);
  const ov = overlay.find((o) => o.dest === m.file);
  if (ov && before !== ov.md5) return { ...rec, outcome: "refused", reason: `before md5 ${before} is not the overlay md5 ${ov.md5}`, md5_before: before };
  if (!ov && porcelainOf(ctx, [m.file])) return { ...rec, outcome: "refused", reason: "the file differs from the ref", md5_before: before };
  const n = countBytes(bytes, m.find);
  if (n !== 1) return { ...rec, outcome: "refused", reason: `anchor occurs ${n} times (must be exactly 1)`, md5_before: before };
  const held = join(out, "mutations", `${i}.orig`);
  mkdirSync(dirname(held), { recursive: true });
  copyFileSync(path, held);
  const at = bytes.indexOf(Buffer.from(m.find, "utf8"));
  const line = bytes.subarray(0, at).toString("utf8").split("\n").length;
  let r, mutated;
  try {
    writeFileSync(path, Buffer.concat([bytes.subarray(0, at), Buffer.from(m.replace, "utf8"), bytes.subarray(at + Buffer.byteLength(m.find, "utf8"))]));
    mutated = deps.md5(path);
    ctx.note(`mutation ${m.name}: ${m.file}:${line} applied (md5 ${before} -> ${mutated})`);
    r = await runVitest(ctx, ctx.req.files, t, `mutation ${m.name}`);
  } finally {
    copyFileSync(held, path);
  }
  const restored = deps.md5(path);
  const porcelain = porcelainOf(ctx);
  appendFileSync(join(out, "output.txt"), sectionOf(`mutation ${m.name}`, r.out));
  const o = mutationOutcome(r);
  return { ...rec, line, outcome: o.outcome, ...(o.reason ? { reason: o.reason } : {}), md5_before: before, md5_mutated: mutated, md5_restored: restored, restored_equal: restored === before, porcelain_after: porcelain, exit_code: r.code, secs: r.secs, tests: o.parsed?.tests, failing: o.parsed?.failing ?? [] };
}

export async function runMutant(ctx, overlay) {
  const { req, out } = ctx;
  const t = (req.timeout_s ?? DEFAULT_TIMEOUT_S) * 1000;
  ctx.set({ run_state: "running baseline" });
  const b = await runVitest(ctx, req.files, t, "baseline");
  writeFileSync(join(out, "output.txt"), sectionOf("baseline", b.out));
  const baseline = { exit_code: b.code, timed_out: b.timedOut, secs: b.secs, ...parseVitest(b.out) };
  const mutations = [];
  let exit = 0, state = "done", message;
  if (b.code !== 0 || b.timedOut) {
    for (const m of req.mutations) mutations.push({ name: m.name, file: m.file, outcome: "refused", reason: "baseline red" });
    exit = 1;
  } else if (!ranTests(baseline)) {
    for (const m of req.mutations) mutations.push({ name: m.name, file: m.file, outcome: "refused", reason: "baseline ran no test" });
    state = "error"; exit = 7; message = noTestMessage(baseline, "the baseline");
  } else {
    for (const [i, m] of req.mutations.entries()) {
      ctx.set({ run_state: `running mutation ${i + 1}/${req.mutations.length}` });
      const rec = await runMutation(ctx, m, i, overlay, t);
      mutations.push(rec);
      if (rec.md5_restored !== undefined && rec.restored_equal !== true) { state = "error"; exit = 7; message = `restore mismatch on ${m.file}: ${rec.md5_before} -> ${rec.md5_restored}`; break; }
    }
  }
  const result = { id: ctx.id, kind: "mutant", room: req.room, run: req.run, ref: req.ref, sha: readState(ctx.cfg.box, ctx.id)?.sha, files: req.files, args: req.args ?? [], cwd: req.cwd ?? "", env: req.env ?? {}, overlay, baseline, mutations, ...(message ? { message } : {}) };
  writeAtomic(join(out, "result.json"), JSON.stringify(result, null, 1) + "\n");
  ctx.set({ state, exit, finished: nowIso(ctx.deps.now()), ...(message ? { message } : {}) });
  ctx.note(`mutant: baseline exit ${b.code}; ${mutations.map((x) => `${x.name}=${x.outcome}`).join(", ")}`);
}

/** The per-request child: `node <snapshot> --run-once <id>` reads queue/<id>.claimed. */
export async function runOnce(id, cfg = config(), deps = defaultDeps()) {
  let req;
  if (id) req = JSON.parse(readFileSync(join(cfg.box, "queue", `${id}.claimed`), "utf8"));
  else { req = JSON.parse(readFileSync(join(cfg.box, "request.json"), "utf8")); id = sanitizeId(req.id || "run"); } // legacy `--run-once`
  const prior = readState(cfg.box, id) ?? {};
  const ctx = childCtx(id, req, cfg, deps, slotGate(cfg, prior.slot ?? 1)); // the parent records the slot before the spawn (D-1466)
  ctx.set({ state: "preparing", run_state: "preparing", pgid: process.pid, starts: { ...(prior.starts ?? {}), [process.pid]: startToken(deps.sh, process.pid) }, began: prior.began ?? nowIso(deps.now()) });
  try {
    const { problems, pnpm } = checkTools(cfg, deps.sh);
    if (problems.length) throw new Error(problems.join("; "));
    const sha = prepareCopy(ctx);
    const overlay = req.v === 2 && req.kind !== "gate" ? applyOverlay(ctx) : [];
    if (!installIfNeeded(ctx, pnpm)) { ctx.set({ state: "failed", run_state: "failed (install)", exit: 7, message: "pnpm install failed (RUNNER.txt)", finished: nowIso(deps.now()) }); return; }
    ctx.set({ state: "running", prepared_at: nowIso(deps.now()) }); // the prepare cap ends here (D-1418)
    if (req.v !== 2 || req.kind === "gate") await gateLegs(ctx, sha);
    else if (req.kind === "tests") await runTests(ctx, overlay);
    else await runMutant(ctx, overlay);
  } catch (e) {
    ctx.note(`failed: ${String(e?.message || e).slice(0, 2000)}`);
    ctx.set({ state: "error", run_state: `failed: ${String(e?.message || e).slice(0, 300)}`, exit: 7, message: String(e?.message || e).slice(0, 2000), finished: nowIso(deps.now()) });
  }
}

// ---------------------------------------------------------------- the long-lived process ----------------

export function createRunner(cfg, deps = defaultDeps(), opts = {}) {
  const box = cfg.box;
  const q = (n) => join(box, "queue", n);
  const live = new Set(); // pgids of per-request children this process started
  const skipped = new Set();
  const slots = cfg.slots ?? 1;
  const running = new Map(); // D-1464: id -> { id, req, slot, pgid, began, done }, in claim order
  let stopping = false; // set by stop() at SIGINT/SIGTERM (D-1417(3))
  const status = { runner: "idle", host_cores: cfg.cores, parallel_shards: cfg.par, started: nowIso(deps.now()), heartbeat: nowIso(deps.now()), run: null, current: null, queue: [], runner_md5: opts.runnerMd5 ?? null, slots, running: [] };
  const exclusive = (req) => req?.v !== 2 || req.kind === "gate"; // the gate (legacy request.json or kind gate) runs alone
  const jobs = () => [...running.values()];
  const firstJob = () => jobs()[0] ?? null;

  const str = (v) => (typeof v === "string" ? v : "<invalid>"); // D-1422 (FX2R-10): status.json copies only strings, so a malformed entry never makes saveStatus throw
  const queueSummary = () => listQueue(box).map((e, i) => { const r = readJson(e.path) ?? {}; return { id: str(e.id), kind: r.v === 2 ? str(r.kind) : "gate", room: r.room === undefined || r.room === null ? null : str(r.room), pos: i + 1 }; });
  function refreshJob(j) {
    const st = readState(box, j.id) ?? {};
    if (exclusive(j.req)) status.run = { id: j.id, ref: st.ref ?? j.req.ref, sha: st.sha, state: st.run_state ?? "preparing", legs: st.legs ?? {}, began: st.began, ...(st.finished ? { finished: st.finished } : {}) };
  }
  function refresh() { for (const j of jobs()) refreshJob(j); }
  function setRunning() {
    status.runner = running.size ? "running" : "idle";
    const v2 = jobs().filter((j) => j.req.v === 2);
    status.current = v2.length ? { id: v2[0].id, kind: v2[0].req.kind, room: v2[0].req.room ?? null, began: v2[0].began } : null; // mac-run reads the first
    status.running = jobs().map((j) => ({ id: j.id, kind: j.req.v === 2 ? str(j.req.kind) : "gate", room: j.req.room == null ? null : str(j.req.room), slot: j.slot, began: j.began }));
  }
  function saveStatus() {
    refresh();
    status.queue = queueSummary();
    status.heartbeat = nowIso(deps.now());
    writeAtomic(join(box, "status.json"), JSON.stringify(status, null, 1) + "\n");
  }
  function killGroups(pgids, sig) { for (const pg of pgids) { if (!pg) continue; try { deps.kill(-pg, sig); } catch { /* gone */ } } }
  function livePgids() {
    const out = new Set(live);
    for (const j of jobs()) for (const p of readState(box, j.id)?.pgids ?? []) out.add(p);
    return [...out];
  }
  const killAll = (sig) => killGroups(livePgids(), sig);
  function killRequest(id, pgid, sig) { killGroups([pgid, ...(readState(box, id)?.pgids ?? [])], sig); }

  /** Item 5: why a recorded pid is NOT this runner's process any more (so it is never killed), or null when verified. */
  function unverified(id, st, pid, role) {
    let lstart, command;
    try {
      lstart = deps.sh("ps", ["-o", "lstart=", "-p", String(pid)]).trim();
      command = deps.sh("ps", ["-ww", "-o", "command=", "-p", String(pid)]).trim();
    } catch { return "no such process"; } // ps exits 1 for a pid that does not exist
    const mine = role === "child" ? `${command} `.includes(`--run-once ${id} `) : command.includes(join(slotGate(cfg, st.slot ?? 1), "node_modules/vitest/vitest.mjs")); // the slot recorded at the claim
    if (!mine) return `its command is not this runner's ${role}: ${command.slice(0, 200)}`;
    const rec = st.starts?.[pid];
    if (rec && rec !== lstart) return `its start time ${lstart} is not the recorded ${rec}`;
    return null;
  }
  /** W0-03 then W0-04, before the gate copy is touched. */
  function startup() {
    let runIds = [];
    try { runIds = readdirSync(join(box, "runs")); } catch { /* first start */ }
    for (const id of runIds) {
      const st = readState(box, id);
      if (!st || TERMINAL.has(st.state)) continue;
      const verified = [];
      for (const [pid, role] of [[st.pgid, "child"], ...(st.pgids ?? []).map((p) => [p, "vitest"])]) {
        if (!pid) continue;
        const why = unverified(id, st, pid, role);
        if (why) deps.log(`${id}: recorded pid ${pid} not killed (${why}); recorded as gone`);
        else verified.push(pid);
      }
      killGroups(verified, "SIGKILL");
    }
    let names = [];
    try { names = readdirSync(join(box, "queue")); } catch { /* none */ }
    for (const n of names) {
      if (!n.endsWith(".claimed")) continue;
      const id = n.slice(0, -8);
      if (terminalRecord(box, id)) continue;
      writeState(box, id, { id, state: "interrupted", exit: 7, message: INTERRUPTED_MSG, finished: nowIso(deps.now()) });
      deps.log(`${id}: ${INTERRUPTED_MSG}`);
    }
  }

  /** The legacy request.json enters the queue under its id, once; a recorded or interrupted id is skipped and logged.
   *  FX3 (D-1425): it is checked before it is queued; a refusal or a throw is recorded refused, in runs/<id>/state.json, or,
   *  when no usable id can be made from it, in queue/legacy-refused-<fnv1a of its text>.marker; neither is re-read as new. */
  function enqueueLegacy() {
    const raw = readText(join(box, "request.json"));
    let req = null;
    try { req = raw === null ? null : JSON.parse(raw); } catch { /* not a request */ }
    if (!req || req.v !== undefined || !req.id) return null;
    const mark = q(`legacy-refused-${fnv1a(raw)}.marker`); // not .json: listQueue reads every queue/*.json as an entry
    if (existsSync(mark)) return null;
    let id = null;
    try {
      const sid = sanitizeId(req.id);
      if (sid === "." || sid === "..") throw new Error(`id ${JSON.stringify(sid)} is not a request id`);
      id = sid;
      if (existsSync(q(`${id}.json`)) || existsSync(q(`${id}.claimed`))) return null;
      const rec = terminalRecord(box, id);
      if (rec) { if (!skipped.has(id)) { skipped.add(id); deps.log(`request.json ${id}: skipped, runs/${id}/ holds ${rec}`); } return null; }
      if (existsSync(join(box, "runs", id))) {
        writeState(box, id, { id, state: "interrupted", exit: 7, message: INTERRUPTED_MSG, finished: nowIso(deps.now()) });
        deps.log(`request.json ${id}: runs/${id}/ exists with no terminal record; marked interrupted, not rerun`);
        return null;
      }
      const why = legacyRefusal(box, id, { ...req, legacy: true });
      if (why) throw new Error(why);
      writeAtomic(q(`${id}.json`), JSON.stringify({ ...req, legacy: true }) + "\n");
      return id;
    } catch (err) {
      const reason = String(err?.message || err);
      if (id === null) writeAtomic(mark, JSON.stringify({ reason, at: nowIso(deps.now()) }) + "\n");
      else writeState(box, id, { id, state: "refused", exit: 7, message: `request.json refused: ${reason}`, finished: nowIso(deps.now()) });
      deps.log(`request.json${id === null ? "" : ` ${id}`}: refused: ${reason}`);
      return null;
    }
  }

  /** D-1414: the runner (which can delete on the Mac) prunes terminal queue entries older than 7 days; never runs/. */
  function prune() {
    let names = [];
    try { names = readdirSync(join(box, "queue")); } catch { return []; }
    const gone = [];
    const t = deps.now().getTime();
    for (const n of names) {
      if (n.startsWith(".") && n.endsWith(".tmp")) { // item 9: a client's stranded in-flight entry
        let age = 0;
        try { age = t - statSync(q(n)).mtimeMs; } catch { continue; }
        if (age > TMP_PRUNE_AGE_MS) { rmSync(q(n), { recursive: true, force: true }); gone.push(n); }
      }
      const m = /^(.*)\.(claimed|overlay|cancel|refused)$/.exec(n);
      if (!m) continue;
      const done = m[2] === "refused" || terminalRecord(box, m[1]);
      let age = 0;
      try { age = t - statSync(q(n)).mtimeMs; } catch { continue; }
      if (done && age > PRUNE_AGE_MS && !existsSync(q(`${m[1]}.json`))) { rmSync(q(n), { recursive: true, force: true }); gone.push(n); }
    }
    return gone;
  }

  function refuse(e, req, reason) {
    let text;
    try { text = JSON.stringify({ id: e.id, reason, at: nowIso(deps.now()), request: req }); }
    catch (err) { text = JSON.stringify({ id: e.id, reason, at: nowIso(deps.now()), request: null, unprintable: String(err?.message || err) }); } // Lead 2: a request JSON cannot print is still refused
    writeAtomic(q(`${e.id}.refused`), text + "\n");
    rmSync(e.path, { force: true });
    deps.log(`${e.id}: refused: ${reason}`);
  }

  async function runClaimed(id, req, slot, job) {
    const v2 = req.v === 2;
    writeState(box, id, { v: 2, id, kind: v2 ? req.kind : "gate", room: req.room ?? null, run: req.run ?? null, ref: req.ref, state: "claimed", began: nowIso(deps.now()), runner_pid: process.pid, slot });
    saveStatus();
    const child = deps.spawn(cfg.node, [opts.snapshot ?? cfg.self, "--run-once", id], { detached: true, stdio: "inherit", env: cfg.childEnv });
    job.pgid = child.pid;
    live.add(child.pid);
    let reason = null, detail = null;
    const cap = v2 ? capMs(req) : null;
    const prepCap = cfg.prepareCapMs ?? PREPARE_CAP_S * 1000;
    const watch = setInterval(() => {
      if (reason) return;
      const st = readState(box, id) ?? {};
      const t = deps.now().getTime();
      if (v2 && existsSync(q(`${id}.cancel`))) reason = "cancelled";
      // D-1417(2) as corrected by D-1418: the prepare cap runs from the claim and only while the request is claimed or preparing (installing included)
      else if ((st.state === "claimed" || st.state === "preparing") && t - Date.parse(st.began) > prepCap) { reason = "timeout"; detail = `prepare took over the ${prepCap / 1000} s prepare cap from the claim`; }
      else if (cap !== null) { const since = st.running_since; if (since && t - Date.parse(since) > cap) reason = "timeout"; }
      if (!reason) return;
      deps.log(`${id}: ${reason}; killing process group ${child.pid}`);
      killRequest(id, child.pid, "SIGTERM");
      setTimeout(() => killRequest(id, child.pid, "SIGKILL"), KILL_GRACE_MS);
    }, cfg.pollMs);
    await new Promise((res) => child.on("close", res));
    clearInterval(watch);
    live.delete(child.pid);
    const st = readState(box, id) ?? {};
    if (reason && !TERMINAL.has(st.state)) writeState(box, id, { state: reason, exit: reason === "cancelled" ? 5 : 6, message: `${reason}${detail ? ` (${detail})` : ""}: process group ${child.pid} killed`, pgids: [], finished: nowIso(deps.now()) });
    else if (stopping && !TERMINAL.has(st.state)) writeState(box, id, { state: "interrupted", exit: 7, message: INTERRUPTED_MSG, finished: nowIso(deps.now()) });
    else if (!TERMINAL.has(st.state)) writeState(box, id, { state: "failed", exit: 7, message: "the per-request child exited without a terminal record (RUNNER.txt)", finished: nowIso(deps.now()) });
    refreshJob(job);
  }
  /** Claims the job into a free slot; it leaves the running set however it ends, and the status is saved then. */
  function startJob(id, req) {
    const used = new Set(jobs().map((j) => j.slot));
    let slot = 1;
    while (used.has(slot)) slot++;
    const job = { id, req, slot, pgid: null, began: nowIso(deps.now()), done: null };
    running.set(id, job);
    setRunning();
    job.done = (async () => {
      try { await runClaimed(id, req, slot, job); }
      finally { running.delete(id); setRunning(); saveStatus(); }
    })();
    return job;
  }

  /** One poll: enqueue the legacy request, prune, then claim queue entries in FIFO order while a slot is free (D-1464).
   *  The head that cannot start stops the scan: a gate waits for every running job, and nothing starts while a gate runs.
   *  SLOTS=1 awaits its one job, as before I59; more slots start jobs and return the first id started without awaiting. */
  async function tick() {
    try { enqueueLegacy(); } catch (err) { deps.log(`request.json: not enqueued: ${err?.message || err}`); } // FX3 (SF3-1, D-1425): never a stalled queue
    try { prune(); } catch (err) { deps.log(`prune: ${err?.message || err}`); }
    let first = null;
    for (const e of listQueue(box)) {
      if (jobs().some((j) => exclusive(j.req))) break;
      const req = readJson(e.path);
      let why;
      try { why = req?.v === 2 ? refusal(box, e.id, req) : legacyRefusal(box, e.id, req); } // decided by v, never by the legacy flag (item 9)
      catch (err) { why = `refusal threw: ${err?.message || err}`; } // Lead 2 (SF2-11): a malformed entry is refused, never a stalled queue
      if (why) { refuse(e, req, why); continue; }
      if (exclusive(req) ? running.size > 0 : running.size >= slots) break; // FIFO: nothing claims past the head
      const claimed = q(`${e.id}.claimed`);
      try { renameSync(e.path, claimed); } catch (err) { if (err?.code === "ENOENT") continue; throw err; }
      if (req.v === 2 && existsSync(q(`${e.id}.cancel`))) {
        writeState(box, e.id, { v: 2, id: e.id, kind: req.kind, room: req.room ?? null, state: "cancelled", exit: 5, message: "cancelled before the claim", finished: nowIso(deps.now()) });
        deps.log(`${e.id}: cancelled before the claim`);
        continue;
      }
      const job = startJob(e.id, req);
      if (slots === 1) { await job.done; return e.id; }
      job.done.catch((err) => deps.log(`${e.id}: ${err?.message || err}`));
      first ??= e.id;
    }
    return first;
  }
  /** Resolves when every job running now has ended (tests; a clean shutdown). */
  const settle = () => Promise.all(jobs().map((j) => j.done.catch(() => {})));

  /** D-1417(3): at SIGINT/SIGTERM the running request is recorded interrupted (exit 7) before its groups are killed. */
  function stop() {
    stopping = true;
    for (const j of jobs()) { // D-1466: every running job, not the first
      const st = readState(box, j.id) ?? {};
      if (!TERMINAL.has(st.state)) writeState(box, j.id, { state: "interrupted", exit: 7, message: INTERRUPTED_MSG, finished: nowIso(deps.now()) });
    }
  }

  return { status, startup, tick, settle, enqueueLegacy, prune, saveStatus, killAll, stop, live, get current() { return firstJob(); }, get running() { return jobs(); } };
}

export function installSignalHandlers(runner, proc = process, timers = { setTimeout }) {

}

/** I62 O1 (D-1518: "#559 is un-deferred and runs from the Mac runner loop"; I62-W0 amendments 1-4, D-1523): a real
 *  `node <repo>/scripts/org-tick.mjs --root <repo> --json` in an ASYNC child (deps.spawn, never deps.sh: an execFileSync
 *  tick would stop the heartbeat timer and every room's `wait` would exit 4), at start and then every ORG_TICK_EVERY_MS by
 *  wall clock whether or not jobs are queued, never awaited by tick() or a gate, no queue slot and no gate copy. At most one
 *  child at a time (a start while one lives is skipped and logged); one still alive after ORG_TICK_CAP_MS is killed. Its
 *  exit and output go to the runner's log as one line carrying "source":"mac-runner" (W0-23). Both values are DRAFT (Jason). */
export const ORG_TICK_EVERY_MS = 3600 * 1000;
export const ORG_TICK_CAP_MS = 10 * 60 * 1000;
export const orgTickArgs = (cfg) => [join(cfg.repo, "scripts/org-tick.mjs"), "--root", cfg.repo, "--json"];

export function createOrgTick(cfg, deps = defaultDeps(), o = {}) {
  const { capMs = ORG_TICK_CAP_MS, timers = { setTimeout, clearTimeout } } = o;
  let child = null;
  function start() {
    if (child) { deps.log(`org-tick: skipped, the previous tick (pid ${child.pid}) is still running`); return false; }
    let c;
    try { c = deps.spawn(cfg.node, orgTickArgs(cfg), { cwd: cfg.repo, env: cfg.childEnv, stdio: ["ignore", "pipe", "pipe"] }); }
    catch (e) { deps.log(`org-tick: not started: ${e?.message || e}`); return false; }
    let out = "";
    let err = "";
    let capped = false;
    let ended = false;
    let cap = null;
    const finish = (code, signal, error) => {
      if (ended) return;
      ended = true;
      timers.clearTimeout(cap);
      child = null;
      const line = { source: "mac-runner", exit: code, signal };
      if (capped) line.capped = true;
      if (error) line.error = String(error?.message || error);
      try { line.tick = JSON.parse(out); } catch { if (out.trim()) line.output = out.trim().slice(-2000); }
      if (err.trim()) line.stderr = err.trim().slice(-2000);
      deps.log(`org-tick ${JSON.stringify(line)}`);
    };
    c.stdout?.on("data", (d) => { out += d; });
    c.stderr?.on("data", (d) => { err += d; });
    c.on("error", (e) => finish(null, null, e));
    c.on("close", (code, signal) => finish(code, signal));
    cap = timers.setTimeout(() => {
      capped = true;
      deps.log(`org-tick: still running after ${capMs / 1000} s; killing pid ${c.pid}`);
      try { c.kill("SIGKILL"); } catch (e) { deps.log(`org-tick: kill failed: ${e?.message || e}`); }
    }, capMs);
    child = c;
    return true;
  }
  return { start, get running() { return child; } };
}

/** I63 U1 (D-1537 I62-Q3 "remember and add skip"; I63-W0 amendments 1-3, D-1540): the start time of the last org-tick that
 *  spawned, kept in the build tree's git-ignored .scratch/mac-gate/ (never the repo). The VM writes that folder too, so the
 *  record is untrusted: anything but a regular file of at most ORG_TICK_RECORD_MAX_BYTES whose `lastTickMs` is a safe
 *  integer in [0, nowMs] reads as absent (null), and the runner ticks. */
export const orgTickRecordPath = (cfg) => join(cfg.build, ".scratch", "mac-gate", "org-tick-last.json");
export const ORG_TICK_RECORD_MAX_BYTES = 4096;
export function readOrgTickRecord(cfg, nowMs) {
  const p = orgTickRecordPath(cfg);
  let st;
  try { st = lstatSync(p); } catch { return null; }
  if (!st.isFile() || st.size > ORG_TICK_RECORD_MAX_BYTES) return null;
  const at = readJson(p)?.lastTickMs;
  return Number.isSafeInteger(at) && at >= 0 && at <= nowMs ? at : null;
}

/** Ticks at start and then every `everyMs` on its own timer. Each tick that spawns stamps the record with the time taken
 *  before its spawn. Only the START tick reads the record: one that began less than everyMs ago skips it and schedules the
 *  first tick at last + everyMs, then hourly (a runner restarting more often than hourly still ticks); the interval never
 *  reads it. stop() clears both timers (main, after its loop). */
export function startOrgTick(cfg, deps = defaultDeps(), o = {}) {
  const { everyMs = ORG_TICK_EVERY_MS, timers = { setInterval, clearInterval, setTimeout, clearTimeout } } = o;
  const ot = createOrgTick(cfg, deps, { ...o, timers });
  const tryStart = () => {
    try {
      const at = deps.now().getTime();
      if (!ot.start()) return;
      try { writeAtomic(orgTickRecordPath(cfg), `${JSON.stringify({ lastTickMs: at, lastTick: new Date(at).toISOString() })}\n`); }
      catch (e) { deps.log(`org-tick: record not written: ${e?.message || e}`); }
    } catch (e) { deps.log(`org-tick: ${e?.message || e}`); }
  };
  let iv = null;
  let first = null;
  const hourly = () => { iv = timers.setInterval(tryStart, everyMs); };
  const now = deps.now().getTime();
  const last = readOrgTickRecord(cfg, now);
  if (last !== null && now - last < everyMs) {
    const due = last + everyMs;
    deps.log(`org-tick: skipped at start, the last tick began ${Math.round((now - last) / 1000)} s ago (${new Date(last).toISOString()}); the next is due at ${new Date(due).toISOString()}`);
    first = timers.setTimeout(() => { tryStart(); hourly(); }, due - now);
  } else { tryStart(); hourly(); }
  return { ot, stop: () => { timers.clearTimeout(first); timers.clearInterval(iv); } };
}

/** Strict flag parser: repeatable flags collect, --files collects until the next --flag, --arg takes the next token whatever it is. */
export function parseFlags(argv, spec) {
  const out = { _: [] };
  for (let i = 0; i < argv.length; i++) {
    const a = argv[i];
    if (!a.startsWith("--")) { out._.push(a); continue; }
    const k = a.slice(2);
    const kind = spec[k];
    if (!kind) throw usage(`unknown flag ${a}`);
    if (kind === "bool") { out[k] = true; continue; }
    if (kind === "list") {
      const vals = [];
      while (i + 1 < argv.length && !argv[i + 1].startsWith("--")) vals.push(argv[++i]);
      if (!vals.length) throw usage(`${a} needs at least one value`);
      out[k] = [...(out[k] ?? []), ...vals];
      continue;
    }
    if (i + 1 >= argv.length) throw usage(`${a} needs a value`);
    const v = argv[++i];
    if (kind === "repeat") out[k] = [...(out[k] ?? []), v];
    else out[k] = v;
  }
  return out;
}

/** An overlay `<src>[=<dest>]`: src is a file in the build tree or in the repo report dir of <run>/<room>. */
export function resolveOverlay(cfg, spec, run, room, cwd) {
  const eq = spec.indexOf("=");
  const srcIn = eq === -1 ? spec : spec.slice(0, eq);
  const src = resolve(cwd, srcIn);
  let isFile = false;
  try { isFile = statSync(src).isFile(); } catch { /* missing */ }
  if (!isFile) throw new Refused(`overlay source ${srcIn} is not a file`);
  const repo = repoRoot(cfg);
  const reportDir = join(repo, "org/07-findings", run, room);
  const under = (root) => { const r = relative(root, src); return r !== "" && !r.startsWith("..") && !isAbsolute(r) ? r.split(sep).join("/") : null; };
  const relBuild = under(cfg.build);
  const relRepo = under(reportDir) !== null ? under(repo) : null;
  if (relBuild === null && relRepo === null) throw new Refused(`overlay source ${srcIn} is in neither the build tree nor org/07-findings/${run}/${room}/ in the repo`);
  // SFX-18: containment is by realpath too, so a symlink inside either tree that points out of it is refused
  const real = realpathSync(src);
  const realUnder = (root) => { let r; try { r = relative(realpathSync(root), real); } catch { return false; } return !r.startsWith(".."); }; // a missing root contains nothing
  if (!realUnder(cfg.build) && !realUnder(reportDir)) throw new Refused(`overlay source ${srcIn} resolves to ${real}, outside the build tree and org/07-findings/${run}/${room}/ in the repo`);
  const dest = eq === -1 ? (relBuild ?? relRepo) : spec.slice(eq + 1);
  if (!validDest(dest)) throw new Refused(`overlay dest ${JSON.stringify(dest)} is refused (absolute, '..', a .git, node_modules or .scratch segment in any case, or a segment that is not printable ASCII)`);
  return { src, dest };
}

export function readSpec(path) {
  const spec = readJson(path);
  if (!Array.isArray(spec) || spec.length === 0) throw new Refused(`--mutate ${path}: not a non-empty JSON list`);
  for (const m of spec) {
    if (!m || ["name", "file", "find", "replace"].some((k) => typeof m[k] !== "string")) throw new Refused(`--mutate ${path}: each entry needs string name, file, find, replace`);
    if (m.find === "") throw new Refused(`--mutate ${path}: ${m.name}: find is empty`);
    if (!validDest(m.file)) throw new Refused(`--mutate ${path}: ${m.name}: bad file ${JSON.stringify(m.file)}`);
  }
  if (new Set(spec.map((m) => m.name)).size !== spec.length) throw new Refused(`--mutate ${path}: names must be unique`);
  return spec.map(({ name, file, find, replace }) => ({ name, file, find, replace }));
}

//
// "Do not assume any of this — the point of the rule is the measurement." The brief's baseline
// was measured by hand across S-092's six room prompts: the longest common prefix is 14,096
// bytes = 56% of the average prompt, and it breaks at `## ROOM CONTRACT (org/rooms/<role>.md)`.
// This script is that measurement, mechanised, so the next person is not re-deriving it either.
//
//   node scripts/measure-prompt-prefix.mjs <run-dir> [<run-dir> ...] [--json] [--root <repo>]
//
// It reports, over every PROMPT.md under the named run directories:
//   - the longest common prefix across ALL of them (bytes, and as a % of the average prompt)
//   - the same WITHIN each role group (the `ROOM TYPE:` line in the TAIL), which is the honest
//     unit: a role's contract IS different per role, so a wave-wide prefix can only ever reach
//     the first role contract. A W3 batch is mostly blue-builders, so one role group covers
//     eight of ten rooms and the per-role number is the one that moves a bill.
//   - the section heading the prefix breaks AT, so a regression in assembly order is visible
//     rather than inferred, and a sha256 of the prefix so two runs can be compared exactly.

// A comparison LABEL for a prefix, not a canonical hash — deliberately not a crypto primitive.
// INV single-hasher forbids hashing outside packages/canonical, and its EXEMPT_FILES list would
// have taken this script ("repo scripts may hash"). I did not use it: eleven rooms in W3-01 were
// told not to reach for a guard's escape hatch to go green, and the orchestrator does not get a
// different rule. FNV-1a over the prefix bytes is enough to answer the only question asked of it
// — "is this the same prefix as the last run?" — and it canonicalises nothing.
function prefixLabel(buf) {
  let h = 0x811c9dc5;
  for (let i = 0; i < buf.length; i++) { h ^= buf[i]; h = Math.imul(h, 0x01000193) >>> 0; }
  return h.toString(16).padStart(8, "0");
}

// Byte-wise longest common prefix. Buffers, not strings: the brief's number is in BYTES, and a
// multi-byte character split mid-sequence would otherwise silently shorten every measurement.
function lcpBytes(buffers) {
  if (!buffers.length) return 0;
  if (buffers.length === 1) return buffers[0].length;
  const min = Math.min(...buffers.map((b) => b.length));
  let i = 0;
  outer: for (; i < min; i++) {
    const c = buffers[0][i];
    for (let k = 1; k < buffers.length; k++) if (buffers[k][i] !== c) break outer;
  }
  return i;
}

// The heading the prefix breaks AT: the last `## ` heading that is wholly inside the shared
// prefix, plus the first one after it — "identical through X, diverges before Y".
function breakPoint(buf, n) {
  const head = buf.slice(0, n).toString("utf8");
  const tail = buf.slice(n).toString("utf8");
  const heads = [...head.matchAll(/^## [^\n]+$/gm)].map((m) => m[0]);
  const next = (tail.match(/^## [^\n]+$/m) || [])[0] ?? null;
  return { lastShared: heads.length ? heads[heads.length - 1] : null, firstDivergent: next };
}

/**
 * S-005a decision S5a-7 (DISCREPANCY-0025) — the ONLY sanctioned way to run
 * Stryker in this repo.
 *
 * stryker.config.mjs runs `inPlace: true` (MUTATION room's finding: the
 * single-hasher runtime tripwire resolves workspace-package caller frames
 * back out of Stryker's sandbox copy, so a sandboxed run trips on legitimate

 * on a clean exit. On this box a bash call dies at ~170s; an interrupted run

 * stayed green over them. So:

 *      detached children die at this box's bash-call boundary); a timed-out
 *      call saved its progress — run the same command again to resume.
 *   2. `--status <slug>` reads the log for the "exit <code>" trailer and runs
 *      the leak scan.
 *   3. The leak scan refuses to declare a run finished while any tracked
 *      source file carries `// @ts-nocheck` outside node_modules/.stryker-tmp
 *      and prints the list — those files must be restored from the repo
 *      before any gate part is trusted (scripts/gate.mjs --typecheck refuses

 * set is measured WEEKLY in CI (.github/workflows/mutation-refresh.yml, `cron: "0 3 * * 0"`) via
 * scripts/mutation-run.mjs, not nightly and not by this file — stryker.config.mjs and

 *
 * Usage:
 *   node scripts/mutate-file.mjs <src-file> [<src-file>...] [--seconds=150]

function leakScan() {
  let out = "";
  try {
    out = execFileSync("grep", ["-rl", "--include=*.ts", "--include=*.tsx", "--include=*.mts",
      "--exclude-dir=node_modules", "--exclude-dir=.stryker-tmp", "--exclude-dir=dist", "--exclude-dir=.git",
      "--exclude-dir=org", "-e", "^// @ts-nocheck", "."], { encoding: "utf8" });
  } catch (e) {
    // grep exit 1 = no matches; any other status (2 = error) means the leak scan did not run.
    if (e.status !== 1) { console.error(`LEAK SCAN cannot be read: grep exited ${e.status}: ${String(e.stderr ?? e.message).trim()}`); process.exit(1); }
    out = e.stdout ?? "";
  }

    console.error(`LEAK: ${files.length} file(s) carry // @ts-nocheck — restore from the repo before trusting any gate part:\n${files.join("\n")}`);

/** Measure ONE subject: stryker on that file alone (STRYKER_MUTATE_SUBJECT narrows the config's testFiles to its own
 * suites, I24), against its own incremental cache. Returns { row } with either a note (not measured) or the score row
 * plus the Stryker-schema entry. */
function measure(file) {
  const slug = slugOf(file);
  const incPath = join(outDir, `${slug}.incremental.json`);
  const logPath = join(outDir, `${slug}.log`);
  console.log(`mutation-run: ${file} (incremental -> ${incPath})`);
  const r = spawnSync(
    "bash",
    [
      "-o",
      "pipefail",
      "-c",
      `npx stryker run --mutate '${file}' --incremental --incrementalFile '${incPath}' 2>&1 | tee -a '${logPath}'`,
    ],
    { cwd: ROOT, stdio: "inherit", env: { ...process.env, STRYKER_MUTATE_SUBJECT: file } },
  );
  // I3-1253: `--incrementalFile` is a CACHE stryker reads and rewrites -- a failed or interrupted
  // run (nonzero/null status) can leave the PREVIOUS successful run's incremental data sitting at
  // incPath untouched, and existsSync alone cannot tell "fresh" from "stale" apart. A nonzero exit
  // means this run's own measurement is unverified; refuse to score it rather than parse whatever
  // was already on disk.
  if (r.status !== 0) {
    return { file, note: `not measured: stryker exited ${r.status} — any incremental data on disk is unverified and may be a stale prior result` };
  }
  if (!existsSync(incPath)) {
    return { file, note: `not measured: no incremental output produced (stryker exit ${r.status})` };
  }
  let inc;
  try {
    inc = JSON.parse(readFileSync(incPath, "utf8"));
  } catch (e) {
    return { file, note: `not measured: cannot parse ${incPath}: ${e.message}` };
  }
  const entry = inc.files?.[file] ?? inc.files?.[`./${file}`];
  if (!entry || !Array.isArray(entry.mutants)) {
    return { file, note: `not measured: no mutants[] for ${file} in ${incPath}` };
  }
  let killed = 0;
  let survived = 0;
  let timeout = 0;
  let noCoverage = 0;
  for (const m of entry.mutants) {
    if (m.status === "Killed") killed++;
    else if (m.status === "Survived") survived++;
    else if (m.status === "Timeout") timeout++;
    else if (m.status === "NoCoverage") noCoverage++;
  }
  const denom = killed + survived + timeout + noCoverage;
  // I3-0476: matches Stryker's own mutationScore formula (mutation-testing-metrics'
  // calculateMetrics.ts: totalDetected = timeout + killed) -- a Timeout mutant is DETECTED, same
  // as Killed, not merely absent from survived/noCoverage. mutate-file.mjs relays Stryker's own
  // printed percentage verbatim; this recomputation now agrees with it on the same run.
  const score = denom === 0 ? 0 : ((killed + timeout) / denom) * 100;
  return { file, score, entry, floorRow: { score: Number(score.toFixed(2)), mutants: denom, killed, survived, timeout, noCoverage } };
}

 * Before the C-057 fix, `draftWarning` copied the intake report payload into the
 * draft, so a queued (unpublished) warning may carry a `protected_passport_id` or
 * `linkage_basis` that came from the reporter, not from an operator. The fix
 * stops new ones; this lists the old ones still waiting to publish.
 *
 * For each `review_queue` row in `queued_warning` whose payload carries either
 * field, the field is compared with the ORIGINATING report payload, which the
 * draft overwrote and which survives only in the `enqueue:report` audit line
 * (`review.ts` enqueue: `JSON.stringify(payload).slice(0, 500)`):
 *   - value differs from the report's (or the report had none) -> operator-set, not listed.
 *   - value equals the report's:
 *       protected_passport_id: the `warning:draft` audit line records ` protects=<psp>`
 *         exactly when the operator passed it -> present: operator-set (not listed);
 *         absent: INHERITED.
 *       linkage_basis: no audit line records who set it -> CANNOT_TELL.
 *   - no enqueue audit line, an unparseable (truncated) one, or no draft audit
 *     line for a pointer -> CANNOT_TELL.
 *
 * Usage:
 *   node scripts/ops/c057-inherited-drafts.mjs --db <sqlite file> [--json]
 *     Opens `file:<path>?immutable=1` with readOnly: true: SQLite refuses every
 *     write and creates no -shm/-wal file. A non-empty <path>-wal is refused
 *     (exit 2), because immutable mode would not read it.
 *   node scripts/ops/c057-inherited-drafts.mjs --print-sql
 *     Prints the one SELECT, for D1:
 *     wrangler d1 execute <DB> --remote --json --command "$(node scripts/ops/c057-inherited-drafts.mjs --print-sql)" > rows.json
 *   node scripts/ops/c057-inherited-drafts.mjs --rows rows.json [--json]
 *     Classifies rows exported by that command (wrangler's JSON array, or a bare row array).
 * Exit: 0 nothing listed, 1 at least one INHERITED or CANNOT_TELL row, 2 usage error.
 */
import { existsSync, readFileSync, realpathSync, statSync } from "node:fs";
import { resolve as resolvePath } from "node:path";
import { fileURLToPath, pathToFileURL } from "node:url";

export const QUERY = `SELECT q.item_id, q.subject, q.payload, q.queued_at,
  (SELECT a.detail FROM audit_log a WHERE a.item_id = q.item_id AND a.action = 'enqueue:report' ORDER BY a.audit_id ASC LIMIT 1) AS enqueue_detail,
  (SELECT a.detail FROM audit_log a WHERE a.item_id = q.item_id AND a.action = 'warning:draft' ORDER BY a.audit_id DESC LIMIT 1) AS draft_detail
FROM review_queue q
WHERE q.item_type = 'report' AND q.state = 'queued_warning'
  AND (json_extract(q.payload, '$.protected_passport_id') IS NOT NULL OR json_extract(q.payload, '$.linkage_basis') IS NOT NULL)
ORDER BY q.queued_at, q.item_id`;

function parseObject(text) {
  if (typeof text !== "string") return null;
  try {
    const v = JSON.parse(text);
    return v && typeof v === "object" && !Array.isArray(v) ? v : null;
  } catch {
    return null;
  }
}

/** One verdict per field present on the draft. */
export function classify(row) {
  const draft = parseObject(row.payload) ?? {};
  const origin = parseObject(row.enqueue_detail);
  const out = [];
  for (const field of ["protected_passport_id", "linkage_basis"]) {
    const value = draft[field];
    if (value === undefined || value === null) continue;
    let verdict;
    let evidence;
    if (row.enqueue_detail == null) {
      verdict = "CANNOT_TELL";
      evidence = "no enqueue:report audit line";
    } else if (origin === null) {
      verdict = "CANNOT_TELL";
      evidence = "enqueue:report audit line unparseable (truncated at 500 chars?)";
    } else if (origin[field] !== value) {
      verdict = "OPERATOR_SET";
      evidence = origin[field] === undefined ? "report payload had no value" : "differs from report payload";
    } else if (field === "linkage_basis") {
      verdict = "CANNOT_TELL";
      evidence = "equals report payload; no audit line records linkage_basis";
    } else if (row.draft_detail == null) {
      verdict = "CANNOT_TELL";
      evidence = "equals report payload; no warning:draft audit line";
    } else if (String(row.draft_detail).split(" ").includes(`protects=${value}`)) {
      verdict = "OPERATOR_SET";
      evidence = "warning:draft audit line records protects=";
    } else {
      verdict = "INHERITED";
      evidence = "equals report payload; warning:draft audit line has no protects=";
    }
    out.push({ item_id: row.item_id, subject: row.subject, queued_at: row.queued_at, field, value, verdict, evidence });
  }
  return out;
}

export function report(rows) {
  return rows.flatMap(classify).filter((f) => f.verdict !== "OPERATOR_SET");
}

function rowsFromExport(text) {
  const v = JSON.parse(text);
  if (Array.isArray(v) && v.length > 0 && v.every((x) => x && Array.isArray(x.results))) return v.flatMap((x) => x.results);
  if (Array.isArray(v)) return v;
  if (v && Array.isArray(v.results)) return v.results;
  throw new Error("--rows: expected wrangler d1 --json output or an array of rows");
}

async function main(argv) {
  const arg = (name) => {
    const i = argv.indexOf(name);
    return i >= 0 ? argv[i + 1] : undefined;
  };
  if (argv.includes("--print-sql")) {
    process.stdout.write(`${QUERY.replace(/\s+/g, " ")}\n`);
    return 0;
  }
  let rows;
  if (arg("--db")) {
    const path = resolvePath(arg("--db"));
    // `immutable=1`: SQLite takes no locks and builds no WAL index, so no
    // `-shm`/`-wal` file is created and read-only media works (REGB-2). It
    // also IGNORES a live `-wal`, so rows only there would be silently
    // missed; a non-empty `-wal` is therefore refused, not read.
    if (existsSync(`${path}-wal`) && statSync(`${path}-wal`).size > 0) {
      throw new Error(`${path}-wal is not empty: checkpoint the database (or export a clean copy) before running this report`);
    }
    const { DatabaseSync } = await import("node:sqlite");
    const db = new DatabaseSync(`${pathToFileURL(path).href}?immutable=1`, { readOnly: true });
    try {
      rows = db.prepare(QUERY).all();
    } finally {
      db.close();
    }
  } else if (arg("--rows")) {
    rows = rowsFromExport(readFileSync(arg("--rows"), "utf8"));
  } else {
    process.stderr.write("usage: c057-inherited-drafts.mjs --db <sqlite file> | --rows <wrangler json> | --print-sql  [--json]\n");
    return 2;
  }
  const listed = report(rows);
  if (argv.includes("--json")) {
    process.stdout.write(`${JSON.stringify(listed, null, 2)}\n`);
  } else {
    process.stdout.write(`C-057 inherited-draft report: ${rows.length} queued draft(s) carry a pointer or linkage_basis; ${listed.length} field(s) listed.\n`);
    for (const f of listed) process.stdout.write(`${f.verdict}\t${f.item_id}\t${f.field}=${f.value}\t${f.evidence}\n`);
    if (listed.length > 0) {
      process.stdout.write("Do not publish a listed draft as is: an operator re-decides it (withdraw it with disposition evidence_pending, because not_impersonation/wrong_subject would clear a matching suspected_passport_id keyed to the inherited pointer; if the warning is still warranted, draft it again from a new report item with the pointer/basis set explicitly).\n");
    }
  }
  return listed.length > 0 ? 1 : 0;
}

#!/usr/bin/env node
// persona-dedup.mjs — ADR-0011 D6: extract sections shared byte-identically
// across persona files into org/personas/_COMMON.md, and replace each
// occurrence with an `<!-- COMMON: <heading> -->` marker.
//
//   node scripts/persona-dedup.mjs --check   # fail if any persona carries a
//                                             # section _COMMON.md already owns verbatim
//   node scripts/persona-dedup.mjs --write   # extract once, rewrite personas (idempotent)
import { existsSync, readdirSync, readFileSync, writeFileSync } from "node:fs";
import { join, resolve } from "node:path";

const ROOT = resolve(process.cwd());
const PERSONAS_DIR = join(ROOT, "org/personas");
const COMMON_PATH = join(PERSONAS_DIR, "_COMMON.md");
const THRESHOLD = 0.9;

function walk(dir) {
  let out = [];
  for (const e of readdirSync(dir, { withFileTypes: true })) {
    const p = join(dir, e.name);
    if (e.isDirectory()) out = out.concat(walk(p));
    else if (e.name.endsWith(".md") && e.name !== "_COMMON.md") out.push(p);
  }
  return out;
}

function hasIdFrontmatter(text) {
  return /^---\nid:\s*/.test(text);
}

function splitSections(body) {
  const parts = body.split(/^## /m).slice(1);
  return parts.map((p) => {
    const nl = p.indexOf("\n");
    const heading = "## " + p.slice(0, nl).trim();
    const content = p.slice(nl).trim();
    return { heading, content };
  });
}

function loadPersonas() {
  if (!existsSync(PERSONAS_DIR)) { console.error(`persona-dedup: ${PERSONAS_DIR} cannot be read — no persona population`); process.exit(1); }
  const files = walk(PERSONAS_DIR);
  return files
    .map((f) => ({ file: f, text: readFileSync(f, "utf8") }))
    .filter((p) => hasIdFrontmatter(p.text));
}

function isMarker(content) {
  return /^<!-- COMMON: .*-->$/.test(content);
}

function computeCommonSections(personas) {
  const byHeading = new Map();
  const totalByHeading = new Map();
  for (const { text } of personas) {
    const body = text.replace(/^---\n[\s\S]*?\n---\n/, "");
    for (const { heading, content } of splitSections(body)) {
      // Idempotency (ADR-0011 D6 item 3): a section already reduced to its
      // `<!-- COMMON: heading -->` marker by a prior --write must not be
      // counted as a content variant -- otherwise a second run would treat
      // the marker text itself as the "common" body and clobber _COMMON.md.
      if (isMarker(content)) continue;
      totalByHeading.set(heading, (totalByHeading.get(heading) || 0) + 1);
      if (!byHeading.has(heading)) byHeading.set(heading, new Map());
      const m = byHeading.get(heading);
      m.set(content, (m.get(content) || 0) + 1);
    }
  }
  const common = new Map();
  for (const [heading, variants] of byHeading) {
    const total = totalByHeading.get(heading);
    let best = null;
    for (const [content, count] of variants) {
      if (!best || count > best.count) best = { content, count };
    }
    if (best && best.count / total >= THRESHOLD) {
      common.set(heading, { content: best.content, count: best.count, total });
    }
  }
  return common;
}

function marker(heading) {
  return `<!-- COMMON: ${heading} -->`;
}

function replaceCommonInBody(text, common) {
  const fmMatch = text.match(/^---\n[\s\S]*?\n---\n/);
  const fm = fmMatch ? fmMatch[0] : "";
  const body = fmMatch ? text.slice(fm.length) : text;
  const parts = body.split(/(^## .*$)/m);
  let out = fm + (parts[0] ?? "");
  for (let i = 1; i < parts.length; i += 2) {
    const headingLine = parts[i].trim();
    const sectionBody = parts[i + 1] ?? "";
    const content = sectionBody.trim();
    if (common.has(headingLine) && common.get(headingLine).content === content) {
      out += headingLine + "\n" + marker(headingLine) + "\n";
    } else {
      out += headingLine + "\n" + sectionBody;
    }
  }
  return out;
}

function writeCommonFile(common) {
  const headings = [...common.keys()];
  const body = headings
    .map((h) => `${h}\n${common.get(h).content}\n`)
    .join("\n");
  const headerLine = "<!-- org/personas/_COMMON.md -- ADR-0011 D6. Extracted ONCE by scripts/persona-dedup.mjs --write from the sections whose heading AND body are byte-identical across >= " + Math.round(THRESHOLD * 100) + "% of persona files that carry that heading. Persona files reference a section here with `<!-- COMMON: <heading> -->`. -->\n\n";
  return headerLine + body;
}

function table(common) {
  const rows = [...common.entries()].sort((a, b) => b[1].count - a[1].count);
  const lines = ["section | files sharing it | bytes saved x files"];
  for (const [heading, { content, count, total }] of rows) {
    const bytesSaved = Buffer.byteLength(content, "utf8");
    lines.push(`${heading} | ${count}/${total} | ${bytesSaved} x ${count} = ${bytesSaved * count}`);
  }
  return lines.join("\n");
}

const mode = process.argv.includes("--write") ? "write" : process.argv.includes("--check") ? "check" : null;
if (!mode) {
  console.error("usage: persona-dedup.mjs [--check|--write]");
  process.exit(2);
}

const personas = loadPersonas();
if (!personas.length) {
  console.error(`no persona files with id: frontmatter found under ${PERSONAS_DIR}`);
  process.exit(1);
}

if (mode === "check") {
  if (!existsSync(COMMON_PATH)) {
    console.log(`OK (nothing extracted yet): no ${COMMON_PATH}`);
    console.log(table(computeCommonSections(personas)));
    process.exit(0);
  }
  const commonText = readFileSync(COMMON_PATH, "utf8");
  const ownedHeadings = new Map();
  for (const { heading, content } of splitSections(commonText.replace(/^<!--[\s\S]*?-->\n\n/, ""))) {
    ownedHeadings.set(heading, content);
  }
  const violations = [];
  for (const { file, text } of personas) {
    const body = text.replace(/^---\n[\s\S]*?\n---\n/, "");
    for (const { heading, content } of splitSections(body)) {
      if (ownedHeadings.has(heading) && ownedHeadings.get(heading) === content) {
        violations.push(`${file}: carries "${heading}" verbatim -- _COMMON.md already owns it; should be ${marker(heading)}`);
      }
    }
  }
  if (violations.length) {
    console.error("FAIL: drift -- sections not deduped:");
    for (const v of violations) console.error(" - " + v);
    process.exit(1);
  }
  console.log(`OK: clean -- no persona file duplicates a section _COMMON.md owns (${ownedHeadings.size} owned sections, ${personas.length} personas checked)`);
  process.exit(0);
}

const common = computeCommonSections(personas);
if (!common.size) {
  // Idempotency: a prior --write already reduced every qualifying section to
  // its marker, so there is nothing left to extract -- that is success, not
  // failure, as long as _COMMON.md already exists from that prior run.
  if (existsSync(COMMON_PATH)) {
    console.log(`OK: nothing new to extract -- ${COMMON_PATH} already owns every qualifying section (idempotent no-op)`);
    process.exit(0);
  }
  console.error("nothing meets the >= 90% identical threshold -- nothing to extract");
  process.exit(1);
}
console.log(table(common));
// Merge with any sections a PRIOR --write already extracted (idempotency: a
// heading fully marked in this pass keeps its earlier canonical body instead
// of being dropped from _COMMON.md).
const merged = new Map(common);
if (existsSync(COMMON_PATH)) {
  const priorText = readFileSync(COMMON_PATH, "utf8");
  for (const { heading, content } of splitSections(priorText.replace(/^<!--[\s\S]*?-->\n\n/, ""))) {
    if (!merged.has(heading)) merged.set(heading, { content, count: 0, total: 0 });
  }
}
writeFileSync(COMMON_PATH, writeCommonFile(merged));
let changed = 0;
for (const { file, text } of personas) {
  const rewritten = replaceCommonInBody(text, common);
  if (rewritten !== text) {
    writeFileSync(file, rewritten);
    changed++;
  }
}
console.log(`OK: wrote ${COMMON_PATH} (${merged.size} sections); rewrote ${changed}/${personas.length} persona files`);

#!/usr/bin/env node
/**
 * PRE-FLIGHT — everything knowable before anything expensive runs.
 *
 * WHY THIS EXISTS, measured rather than asserted. In one session, green gate legs went red eight
 * times for reasons that had nothing to do with the code under test:
 *   - the gate log was staged and then deleted, so repo-files failed three invariant files;
 *   - appending to that TRACKED log between invariant shards moved treeDigest under them and the
 *     merge correctly refused all four;
 *   - two build-tree files deleted as "strays" turned out to be tracked;
 *   - a room's working copy of a shipped script sat in a report directory and tripped
 *     single-hasher, TWICE — the second time because a sync restored the file after deletion;
 *   - a mutation sandbox left a nested biome.json in scratch and `pnpm lint` died on it;
 *   - two unused imports;
 *   - four manifest pins moved without their hashes;
 *   - claims needed re-deriving after the manifest moved, twice.
 * Every one is a file-scan fact. Each cost a full re-run of a leg: the invariant shards are ~110 s
 * each plus a merge, the vitest shards eight more. A two-second mistake cost ten minutes, repeatedly.
 *
 * So: this runs in seconds, it fails CLOSED, and it names the fix. It does not replace any gate leg
 * — it refuses to let one start while a known-cheap failure is sitting there.
 *
 * INV-059 applies to this file as much as to anything it checks: a check that verified nothing says
 * NOT VERIFIED and exits non-zero. It never answers `ok` for work it did not do.
 */
import { execFileSync } from "node:child_process";
import { createHash } from "node:crypto";
import { existsSync, readdirSync, readFileSync, statSync } from "node:fs";
import { join, relative, resolve } from "node:path";

// RG3-05: ONE argv parser. A value flag takes `--flag=<v>` or `--flag <v>`; anything else — an unknown
// flag, a value flag with no value, a second positional — REFUSES (exit 2) before any check runs,
// because a flag this script ignores is a check it silently did not run (`--mirror-tree <p>` did).
const VALUE_FLAGS = new Set(["--mirror-tree", "--gate-log", "--repo"]);
const argv = { positional: [], json: false, "--mirror-tree": [], "--gate-log": [], "--repo": [] };
{
  const refuseArgs = (msg) => { console.error(`preflight: ${msg}; nothing was checked. Usage: node scripts/preflight.mjs [root] [--json] [--mirror-tree <path>]... [--gate-log <path>] [--repo <path>]`); process.exit(2); };
  const av = process.argv.slice(2);
  for (let i = 0; i < av.length; i++) {
    const a = av[i];
    if (!a.startsWith("--")) { argv.positional.push(a); continue; }
    if (a === "--json") { argv.json = true; continue; }
    const eq = a.indexOf("=");
    const name = eq < 0 ? a : a.slice(0, eq);
    if (!VALUE_FLAGS.has(name)) refuseArgs(`unrecognised flag ${a}`);
    const v = eq < 0 ? av[++i] : a.slice(eq + 1);
    if (v == null || v === "" || (eq < 0 && v.startsWith("--"))) refuseArgs(`${name} needs a value`);
    argv[name].push(v);
  }
  if (argv.positional.length > 1) refuseArgs(`one root only, got ${argv.positional.length} positionals (${argv.positional.join(", ")})`);
  if (argv["--gate-log"].length > 1) refuseArgs("--gate-log given more than once");
  if (argv["--repo"].length > 1) refuseArgs("--repo given more than once");
}
const ROOT = resolve(argv.positional[0] ?? ".");
const JSON_OUT = argv.json;
const results = [];
const add = (name, ok, detail, fix) => results.push({ name, ok, detail, fix });

function git(args) {
  return execFileSync("git", ["--no-optional-locks", ...args], { cwd: ROOT, encoding: "utf8", maxBuffer: 64 * 1024 * 1024 });
}

// ---------------------------------------------------------------- 1. tracked paths exist
// The exact failure repo-files.ts throws on, hoisted out of the expensive leg. A tracked path
// removed from disk is invisible until a vitest shard dies on it 110 seconds in.
try {
  const tracked = git(["ls-files"]).split("\n").filter(Boolean);
  const missing = tracked.filter((p) => !existsSync(join(ROOT, p)));
  add("tracked-paths-exist", missing.length === 0,
    missing.length ? `${missing.length} tracked path(s) missing on disk: ${missing.slice(0, 5).join(", ")}${missing.length > 5 ? " …" : ""}` : `${tracked.length} tracked paths present`,
    "restore with `git --no-optional-locks show :<path> > <path>`, or stage the deletion — do NOT delete a tracked file to tidy a diff (D-0463)");
} catch (e) {
  add("tracked-paths-exist", false, `git ls-files failed: ${String(e.message).split("\n")[0]}`, "run inside a git tree");
}

// ---------------------------------------------------------------- 2. nested configs in scratch
// A Stryker sandbox or a room's private tree copy leaves a biome.json somewhere under the root and
// `pnpm lint` dies with "nested root configuration" — a lint failure with no lint cause.
try {
  const nested = [];
  const walk = (dir, depth) => {
    if (depth > 6) return;
    let entries;
    try { entries = readdirSync(dir, { withFileTypes: true }); } catch { return; }

#!/usr/bin/env node
// scripts/private-copy.mjs — I55-ENV (D-1411, D-1412; W0-20 as ruled in D-1413).
//
//   node scripts/private-copy.mjs <src-tree> <dest> [--paths <p>...] [--with-org-findings]
//
// Copies the TRACKED files (`git ls-files`) of the named paths of <src-tree> into <dest>, which must not exist.
// Default paths: apps services packages tests fixtures docs scripts, plus package.json, vitest.config.ts,
// test-setup.ts (a setupFiles entry of vitest.config.ts; D-1418), tsconfig*.json (root) and pnpm-lock.yaml. Always excluded: any path segment .git, node_modules, .pnpm-store,
// .scratch, reports or sessions, and org/07-findings/ unless --with-org-findings. The root node_modules and every
// <d>/node_modules and <d>/<e>/node_modules that exists in the source are SYMLINKED, never copied.
// Floors (D-1412): refuses to start when the destination filesystem has under 1024 MB free, or would have
// under 1024 MB after the measured copy; aborts below 512 MB. ANY error once the dest exists (the abort, EACCES,
// ENOSPC, anything) removes only the dest it created and is a refusal (exit 1; I55 FX item 11). Prints the
// bytes copied, and that the copy has no .git.
import { spawnSync } from "node:child_process";
import { copyFileSync, existsSync, lstatSync, mkdirSync, readFileSync, readdirSync, readlinkSync, realpathSync, rmSync, statfsSync, symlinkSync } from "node:fs";
import { dirname, join, resolve } from "node:path";
import { fileURLToPath } from "node:url";

export const MB = 1024 * 1024;
export const START_FLOOR_MB = 1024;
export const ABORT_FLOOR_MB = 512;
export const RECHECK_BYTES = 4 * MB;
export const DEFAULT_PATHS = ["apps", "services", "packages", "tests", "fixtures", "docs", "scripts", "package.json", "vitest.config.ts", "test-setup.ts", "tsconfig*.json", "pnpm-lock.yaml"];
export const EXCLUDED_SEGMENTS = [".git", "node_modules", ".pnpm-store", ".scratch", "reports", "sessions"];
export const ORG_FINDINGS = "org/07-findings";
const ROOT_TSCONFIG_RE = /^tsconfig[^/]*\.json$/;
const NM_SKIP = new Set([".git", ".scratch", "node_modules", ".pnpm-store", "org", "sessions", "reports"]);

/** Does tracked path `f` fall under requested path `p`? (`tsconfig*.json` = a root-level tsconfig file). */
export function underPath(f, p) {
  if (p === "tsconfig*.json") return ROOT_TSCONFIG_RE.test(f);
  const q = p.replace(/\/+$/, "");
  return f === q || f.startsWith(`${q}/`);
}

export function excluded(f, withOrgFindings) {
  if (f.split("/").some((s) => EXCLUDED_SEGMENTS.includes(s))) return true;
  return !withOrgFindings && underPath(f, ORG_FINDINGS);
}

export function selectFiles(tracked, paths = DEFAULT_PATHS, withOrgFindings = false) {
  return tracked.filter((f) => paths.some((p) => underPath(f, p)) && !excluded(f, withOrgFindings));
}

export function freeBytesOf(path) {
  let p = resolve(path);
  while (!existsSync(p)) p = dirname(p);
  const s = statfsSync(p);
  return Number(s.bavail) * Number(s.bsize);
}

/** Every node_modules directory to symlink: root, <d>/ and <d>/<e>/ (never inside a skipped dir). */
export function nodeModulesDirs(src) {
  const out = [];
  if (existsSync(join(src, "node_modules"))) out.push("node_modules");
  const subdirs = (rel) => {
    try { return readdirSync(join(src, rel), { withFileTypes: true }).filter((e) => e.isDirectory() && !NM_SKIP.has(e.name)).map((e) => (rel ? `${rel}/${e.name}` : e.name)); } catch { return []; }
  };
  for (const d of subdirs("")) {
    if (existsSync(join(src, d, "node_modules"))) out.push(`${d}/node_modules`);
    for (const e of subdirs(d)) if (existsSync(join(src, e, "node_modules"))) out.push(`${e}/node_modules`);
  }
  return out;
}

export class CopyRefused extends Error {}

export function privateCopy({ src, dest, paths = DEFAULT_PATHS, withOrgFindings = false, freeBytes = freeBytesOf, log = console.log, tracked = null, copyFile = copyFileSync }) {
  const S = resolve(src);
  const D = resolve(dest);
  if (existsSync(D)) throw new CopyRefused(`${D} already exists; private-copy writes only a dest it creates`);
  let list = tracked;
  if (!list) {
    const r = spawnSync("git", ["-C", S, "--no-optional-locks", "ls-files", "-z"], { encoding: "utf8", maxBuffer: 256 * MB });
    if (r.status !== 0) throw new CopyRefused(`git ls-files failed in ${S}: ${(r.stderr || "").trim()}`);
    list = r.stdout.split("\0").filter(Boolean);
  }
  const files = [];
  let size = 0;
  let missing = 0;
  for (const f of selectFiles(list, paths, withOrgFindings)) {
    let st;
    try { st = lstatSync(join(S, f)); } catch { missing++; continue; }
    if (!st.isFile() && !st.isSymbolicLink()) { missing++; continue; }
    files.push({ f, link: st.isSymbolicLink(), bytes: st.isSymbolicLink() ? 0 : st.size });
    size += st.isSymbolicLink() ? 0 : st.size;
  }
  if (files.length === 0) throw new CopyRefused(`no tracked file of ${S} matches --paths ${paths.join(" ")} after the excludes; nothing copied`);
  const free0 = freeBytes(D);
  if (free0 < START_FLOOR_MB * MB) throw new CopyRefused(`${Math.floor(free0 / MB)} MB free on the destination filesystem, under the ${START_FLOOR_MB} MB start floor; nothing copied`);
  if (free0 - size < START_FLOOR_MB * MB) throw new CopyRefused(`the copy is ${size} bytes and would leave ${Math.floor((free0 - size) / MB)} MB free, under the ${START_FLOOR_MB} MB floor; nothing copied`);
  mkdirSync(dirname(D), { recursive: true });
  mkdirSync(D); // not recursive: EEXIST if the dest appeared since the check, so a dir this run did not create is never adopted (or removed)
  let copied = 0;
  let sinceCheck = 0;
  const links = [];
  let lsFilesTests = 0;
  try {
    for (const { f, link, bytes } of files) {
      const to = join(D, f);
      mkdirSync(dirname(to), { recursive: true });
      if (link) symlinkSync(readlinkSync(join(S, f)), to);
      else copyFile(join(S, f), to);
      copied += bytes;
      sinceCheck += bytes;
      if (sinceCheck >= RECHECK_BYTES) {
        sinceCheck = 0;
        const free = freeBytes(D);
        if (free < ABORT_FLOOR_MB * MB) throw new CopyRefused(`aborted: ${Math.floor(free / MB)} MB free, under the ${ABORT_FLOOR_MB} MB abort floor, after ${copied} bytes; removed the partial ${D}`);
      }
    }
    for (const nm of nodeModulesDirs(S)) {
      const parent = dirname(nm);
      if (parent !== "." && !existsSync(join(D, parent))) continue;
      symlinkSync(join(S, nm), join(D, nm));
      links.push(nm);
    }
    lsFilesTests = files.filter(({ f, link }) => !link && f.startsWith("tests/") && readFileSync(join(D, f), "utf8").includes("ls-files")).length;
  } catch (e) {
    rmSync(D, { recursive: true }); // only D, which this run created above; rmSync never follows the node_modules symlinks
    if (e instanceof CopyRefused) throw e;
    throw new CopyRefused(`copy failed after ${copied} bytes: ${e?.code ?? "error"}: ${e?.message ?? e}; removed the partial ${D}`);
  }
  log(`private-copy: copied ${copied} bytes in ${files.length} tracked files from ${S} to ${D}${missing ? ` (${missing} tracked path(s) absent from the working tree, skipped)` : ""}`);
  log(`private-copy: symlinked ${links.length} node_modules dir(s) from the source: ${links.join(" ") || "none"}`);
  log(`private-copy: the copy has no .git, so the ${lsFilesTests} copied test file(s) that enumerate by git ls-files cannot run in it`);
  return { bytes: copied, files: files.length, links, missing, lsFilesTests };
}

export function main(argv, { log = console.log, err = console.error } = {}) {
  const pos = [];
  let paths = null;
  let withOrg = false;
  for (let i = 0; i < argv.length; i++) {
    const k = argv[i];
    if (k === "--with-org-findings") withOrg = true;
    else if (k === "--paths") { paths = []; while (i + 1 < argv.length && !argv[i + 1].startsWith("--")) paths.push(argv[++i]); }
    else if (k.startsWith("--")) { err(`private-copy: unknown flag ${k}`); return 2; }
    else pos.push(k);
  }
  if (pos.length !== 2 || (paths && paths.length === 0)) {
    err("usage: node scripts/private-copy.mjs <src-tree> <dest> [--paths <p>...] [--with-org-findings]");
    return 2;

    if (e instanceof CopyRefused) { err(`private-copy: REFUSED: ${e.message}`); return 1; }
    err(`private-copy: FAILED: ${e?.message ?? e}`);
    return 1;
  }
}

function numOpt(name) {
  const v = opt(name);
  if (v === undefined) return undefined;
  // I34-TOOLS-C (I33 JA-05 limb 2): Number() also reads hex/octal/binary ("0x10" is 16), "" and " " (0) and "Infinity";
  // only plain decimal notation is a figure. A decimal fraction or negative still reaches the D-1092(2) refusal below.
  // I34-FX3-TOOLS (D-1123(4), I34-RG-W3T TC:7): a fraction is read in the TEXT, before Number() rounds it to a whole
  // number: a decimal figure written with a `.` (`1000.00000000000001`, `9007199254740991.4`, `1000.0`) is handed on as
  // text, which figureFault never accepts, so the D-1092(2) fraction-or-negative refusal below prints it. I34-FX3-SEAM
  // (D-1123(4)): so is an exponent that names a whole number (`5e3`, `1E3`), since only a plain integer is a figure; an
  // exponent past the safe range (`1e21`) or below one (`1e-9`) is read as a number, to keep its too-large or fraction
  // sentence.
  if (DECIMAL_FIGURE.test(v) && (v.includes(".") || (/[eE]/.test(v) && Number.isSafeInteger(Number(v))))) return v;
  const n = DECIMAL_FIGURE.test(v) ? Number(v) : Number.NaN;

}

// DERIVED, never typed: contract 0.8.2 makes a room declare every other report it opened as an
// `opened: <path> — <finding id> — <why>` header line, so the bytes are the sizes of those files
// on disk. Absent lines mean a room that opened nothing (0), not a figure we do not have --

function readBill() {

  const reportPath = join(ROOT, "org/07-findings", run, room, "REPORT.md");

  if (file === undefined || file === null) return null;
  let json;
  try { json = JSON.parse(readFileSync(file, "utf8")); } catch (e) { failFn(`--required ${file} is not a readable JSON file (${e.message}).`); }
  if (json === null || typeof json !== "object" || Array.isArray(json)) failFn(`--required ${file} is not a JSON object of <room>/FINDINGS.jsonl:<line> -> one line of required behaviour.`);
  const out = new Map();
  for (const [k, v] of Object.entries(json)) {
    if (typeof v !== "string" || !v.trim() || /[\r\n\u2028\u2029]/.test(v)) failFn(`--required ${file}: ${JSON.stringify(k)} is not one non-blank line of required behaviour.`);
    out.set(k, v.trim());
  }
  return out;
}
/** Refuses naming every finding with no required line: a brief never falls back to a chain summary or a mechanism (D-1157). */
export function assertRequired(items, failFn = fail) {
  const missing = items.filter((x) => !x.cite || typeof x.required !== "string" || !x.required.trim()).map((x) => x.cite ?? "(a finding with no room and row to cite)");
  if (missing.length) failFn(`no required-behaviour line for ${missing.length} finding(s): ${missing.join(", ")}. Give each in the --required <file> JSON object (D-1157: a brief states the behaviour the code must have and cites the finding; it never carries a chain summary or a mechanism).`);
}
const need = (v, path, where) => { if (v === undefined || v === null || (Array.isArray(v) && !v.length) || v === "") fail(`${where} has no ${path}.`); return v; };

function checkPaths(root, paths, where) {
  const dead = paths.filter((p) => !existsSync(join(root, p)));
  if (dead.length) fail(`${where} names ${dead.length} scope path(s) that do not exist under ${root}: ${dead.join(", ")}.`);
}

/** Every field CAMPAIGN.schema.json marks `required` (through properties, items and one `#/$defs/<name>` ref), then a non-blank control_to_test on every chain step (D-1147(2)). Presence only: an empty value is need()'s refusal. */
export function checkSchema(root, json, ref) {
  const file = join(root, CAMPAIGN_SCHEMA);
  if (!existsSync(file)) fail(`cannot check ${ref}: ${CAMPAIGN_SCHEMA} does not exist under ${root}.`);
  let schema;
  try { schema = JSON.parse(readFileSync(file, "utf8")); } catch (e) { fail(`${CAMPAIGN_SCHEMA} is not JSON (${e.message}).`); }
  if (json === null || typeof json !== "object" || Array.isArray(json)) fail(`${ref} is not a JSON object.`);
  const deref = (node) => {
    if (typeof node?.$ref !== "string") return node;
    const m = /^#\/\$defs\/(.+)$/.exec(node.$ref);
    if (!m || !schema.$defs?.[m[1]]) fail(`${CAMPAIGN_SCHEMA} carries a $ref this check cannot resolve: ${node.$ref}.`);
    return schema.$defs[m[1]];
  };
  const walk = (node, v, path) => {
    const s = deref(node);
    if (!s || v === null || typeof v !== "object") return;
    if (Array.isArray(v)) { if (s.items) for (const [i, x] of v.entries()) walk(s.items, x, `${path}[${i}]`); return; }
    for (const k of s.required ?? []) if (!Object.hasOwn(v, k)) fail(`${ref} is missing ${path ? `${path}.` : ""}${k}, which ${CAMPAIGN_SCHEMA} requires.`);
    for (const [k, sub] of Object.entries(s.properties ?? {})) if (Object.hasOwn(v, k)) walk(sub, v[k], path ? `${path}.${k}` : k);
  };
  walk(schema, json, "");
  if (!Array.isArray(json.chain)) fail(`${ref} chain is not an array of steps.`);
  json.chain.forEach((step, i) => {
    if (typeof step?.control_to_test !== "string" || !step.control_to_test.trim()) fail(`${ref} chain[${i}].control_to_test is empty: every chain step names the control it tests.`);
  });
}

/** I42-REDGEN (I37-B11): the chain's next_step graph. Every step_id is a unique non-blank string, every next_step is null, absent or a step_id, and the walk from chain[0] reaches every step once and ends (no dropped step, no loop). */
export function checkChainGraph(json, ref) {
  const steps = json.chain;
  const ids = steps.map((s, i) => {
    if (typeof s?.step_id !== "string" || !s.step_id.trim()) fail(`${ref} chain[${i}].step_id is not a non-blank string.`);
    return s.step_id;
  });
  const dup = ids.find((x, i) => ids.indexOf(x) !== i);
  if (dup !== undefined) fail(`${ref} chain step_id ${JSON.stringify(dup)} names two steps.`);
  steps.forEach((s, i) => {
    if (s.next_step !== null && s.next_step !== undefined && !ids.includes(s.next_step)) fail(`${ref} chain[${i}] (${ids[i]}) next_step ${JSON.stringify(s.next_step)} names no step of the chain.`);
  });
  const seen = new Set();
  let at = steps.length ? 0 : -1;
  while (at !== -1) {
    if (seen.has(at)) fail(`${ref} chain loops: step ${ids[at]} is reached twice from chain[0].`);
    seen.add(at);
    const n = steps[at].next_step;
    at = n === null || n === undefined ? -1 : ids.indexOf(n);
  }
  const lost = ids.filter((_, i) => !seen.has(i));
  if (lost.length) fail(`${ref} chain step(s) ${lost.join(", ")} are never reached from chain[0] by next_step.`);
}

/** The one shape a chain line takes (red-attacker.md `## CHAINS`): a bare line, em dashes, one outcome word, a summary. */
export const CHAIN_RE = /^CHAIN: ([A-Za-z0-9][A-Za-z0-9._-]*) — (\S+) — (\S.*)$/;
/** Once leading whitespace, list, quote, table and heading marks and emphasis are set aside: a line that starts with the word CHAIN, or with the word in any other case followed by a colon or by one token and an em dash (`Chain:`, `chain s1 —`; D-1148(2)). Mixed-case prose (`Chain outcomes are above.`) is not a chain line. */
/** The lead set aside before the first word (I42-REDGEN, I37-B1): any run of characters that are not a letter or a digit (marks, quotes, brackets, parens, non-ASCII bullets), list numbers and `[x]` task boxes. */
export const CHAIN_LEAD = String.raw`^(?:[^\p{L}\p{N}]|\d+[.)]|\[[xX]\])*`;
export const CHAIN_LIKE_RE = new RegExp(String.raw`${CHAIN_LEAD}(?:CHAIN\b|[Cc][Hh][Aa][Ii][Nn][\s*_\x60~]*(?::|[A-Za-z0-9\x60][^\s—]*[\s*_\x60~]*—))`, "u");
/** The same marks set aside, a first word `chain` in any case (not `Chains`, not `Chain-break`), and one of the four outcome words anywhere after it (D-1149(1)): `Chain s1: SUCCEEDS — x` and `**Chain s1**: PARTIAL` are chain lines; `Chain outcomes are above.` is not. */
export const CHAIN_VERDICT_LIKE_RE = new RegExp(String.raw`${CHAIN_LEAD}chain(?![\w-]).*\b(?:SUCCEEDS|PARTIAL|BLOCKED|UNDETECTED)\b`, "iu"); // I37-B3: outcome words in any case

/** Every `CHAIN: <id> — <outcome> — <summary>` line of a red REPORT.md, plus the FINDINGS rows that name each chain. A CHAIN-like line that is not in that shape refuses: skipping it would drop a chain (D-1147(2)). */
export function parseChains(reportText, findingsText = "", where = "the red report") {
  const chains = [];
  const lines = reportText.split("\n").map((raw) => raw.replace(/\r$/, ""));
  // I42-REDGEN (I37-B12): where each line sits: inside a ``` or ~~~ fence, and inside a `## CHAINS` section (that heading outside a fence, up to the next # or ## heading).
  let fence = null;
  let inChains = false;
  let prevText = false;
  const pos = lines.map((line) => {
    const f = /^ {0,3}(`{3,}|~{3,})/.exec(line);
    // I42-FX-REDGEN (I42-REDGEN-07): a fence closes only on the opener's character, at least its length, then only spaces or tabs (fenceStep's rule); a fence line with an info string is content.
    const close = /^ {0,3}(`{3,}|~{3,})[ \t]*$/.exec(line);
    if (fence) { if (close && close[1][0] === fence[0] && close[1].length >= fence.length) fence = null; prevText = false; return { fenced: true, inChains }; }
    if (f) { fence = f[1]; return { fenced: true, inChains }; }
    // I42-FX-REDGEN (I42-REDGEN-07): a setext underline (`---` or `===` under a text line) makes a heading, which ends the section as `#` and `##` do.
    const setext = prevText && /^ {0,3}(?:=+|-+)[ \t]*$/.test(line);
    if (/^#{1,2}\s/.test(line)) inChains = /^##\s+CHAINS\s*$/.test(line);
    else if (setext) inChains = false;
    prevText = line.trim() !== "" && !/^ {0,3}#{1,6}(?:\s|$)/.test(line);
    return { fenced: false, inChains };
  });
  const sectioned = pos.some((x) => x.inChains);
  lines.forEach((line, i) => {
    if (!CHAIN_LIKE_RE.test(line) && !CHAIN_VERDICT_LIKE_RE.test(line)) return;
    const m = CHAIN_RE.exec(line);
    if (!m) fail(`${where} line ${i + 1} starts with CHAIN but is not \`CHAIN: <id> — <outcome> — <summary>\` (a bare line, em dashes, one outcome word, a summary): ${JSON.stringify(line.slice(0, 200))}.`);
    if (pos[i].fenced) fail(`${where} line ${i + 1} is a chain line inside a fenced block: chains are read from the \`## CHAINS\` section outside fences, so this line is refused, not skipped: ${JSON.stringify(line.slice(0, 200))}.`);
    if (sectioned && !pos[i].inChains) fail(`${where} line ${i + 1} is a chain line outside the \`## CHAINS\` section: move it there or reword it, it is refused, not skipped: ${JSON.stringify(line.slice(0, 200))}.`);
    if (!RED_VERDICTS.includes(m[2])) fail(`${where}: chain ${m[1]} carries outcome ${JSON.stringify(m[2])}, not one of ${RED_VERDICTS.join(" | ")}.`);
    if (chains.some((c) => c.id === m[1])) fail(`${where}: chain ${m[1]} has two CHAIN: lines.`);
    chains.push({ id: m[1], verdict: m[2], summary: m[3].trim(), rows: [], lines: [] });
  });
  if (!chains.length) fail(`${where} carries no \`CHAIN: <id> — <outcome> — <summary>\` line: there is no chain to respond to or judge.`);
  findingsText.split("\n").forEach((l, i) => {
    if (!l.trim()) return;
    let row;
    try { row = JSON.parse(l); } catch { fail(`${where} FINDINGS.jsonl line ${i + 1} is not JSON.`); }
    if (row.chain === undefined) return;
    const c = chains.find((x) => x.id === String(row.chain));
    if (!c) fail(`${where} FINDINGS.jsonl row ${row.id ?? i + 1} names chain ${JSON.stringify(row.chain)}, which no CHAIN: line declares.`);
    // A row id is written by the red room: a line break in it would start a Blue or Purple scope line (D-1148(2)).
    c.rows.push(one(row.id ?? `line ${i + 1}`));
    c.lines.push(i + 1);
  });
  return chains;
}

function readRed(root, dir, red) {
  if (!ROOM_ID_RE.test(red)) fail(`--from ${JSON.stringify(red)} is not a room id.`);
  const base = `org/07-findings/${dir}/${red}`;
  const report = join(root, base, "REPORT.md");
  if (!existsSync(report)) fail(`red room ${red} has no ${base}/REPORT.md.`);
  const findings = join(root, base, "FINDINGS.jsonl");
  const chains = parseChains(readFileSync(report, "utf8"), existsSync(findings) ? readFileSync(findings, "utf8") : "", `${base}/REPORT.md`);
  // The campaign, when the red scope this script wrote names one: its blue and purple contracts flow through.
  const scope = join(root, "org/07-findings", dir, "_scopes", `${red}.md`);
  if (!existsSync(scope)) fail(`red room ${red} has no scope at org/07-findings/${dir}/_scopes/${red}.md: write it with \`red-stage-gen.mjs red\` (no --out-dir), so Blue and Purple read the campaign and the files the red room was given.`);
  const scopeText = readFileSync(scope, "utf8");
  const cm = /^Campaign: (.+?) \(`([^`]+)`\)$/m.exec(scopeText);
  // The red room's files, so the Blue and Purple rooms read (and match families against) the same scope.
  const sp = /^Scope \(paths\): (.+)$/m.exec(scopeText);
  if (!sp) fail(`${red}'s scope org/07-findings/${dir}/_scopes/${red}.md carries no \`Scope (paths):\` line, so Blue and Purple would have no files to read or match families against.`);
  let template = null;
  if (cm) {
    if (!existsSync(join(root, cm[2]))) fail(`${red}'s scope names campaign template ${cm[2]}, which does not exist.`);
    template = { id: cm[1], path: cm[2], json: JSON.parse(readFileSync(join(root, cm[2]), "utf8")) };
  }
  // I42-FX-REDGEN (D-1224, D-1228): under a campaign a CHAIN id is one of the campaign's chain[].step_id values; blue and purple both read chains here.
  if (template) {
    const steps = (Array.isArray(template.json?.chain) ? template.json.chain : []).map((s) => s?.step_id);
    const off = chains.filter((c) => !steps.includes(c.id)).map((c) => c.id);
    if (off.length) fail(`${base}/REPORT.md: chain id(s) ${off.join(", ")} are not steps of campaign ${template.id} (\`${template.path}\`), whose step ids are ${steps.map(String).join(", ") || "none"}: a CHAIN id is one of the campaign's step ids (D-1224); report an off-plan attack path under findings or open questions, not as a CHAIN.`);
  }
  return { base, chains, template, scopeLine: `Scope (paths): ${sp[1].trim()}\n\n` };
}

const blueId = (red, chain) => `${red}-BLUE-${chain.replace(/[^A-Za-z0-9._-]/g, "_")}`;

#!/usr/bin/env node
// register-ownership-audit.mjs — W3-11a-R5.
//
// Answers one question mechanically, for every item in every INTAKE-0003 batch:
// does the room the item is assigned to own a file that carries the item's SUBJECT?
//
// The batches were cut by the rule at org/00-context/intake-0003/WAVE-PLAN-0003.md:34
// ("A room owns files. Each defect goes to its first source file."), so `Owns:` is a
// function of files[0] alone. This script measures what that rule costs, and re-derives
// a candidate owner from where the item's subject tokens actually occur in the tree.
//
//   node scripts/register-ownership-audit.mjs                # coverage measurement
//   node scripts/register-ownership-audit.mjs --derive       # + symbol re-derivation
//   node scripts/register-ownership-audit.mjs --derive --calibrate  # + score against
//                                                            #   run-batch room verdicts
//   node scripts/register-ownership-audit.mjs --derive --json=out.json
//   node scripts/register-ownership-audit.mjs --batch=W3-18 --derive   # one batch
//
// No writes outside --json. Read-only against the tree and org/07-findings.

import fs from 'node:fs';

// subject tokens for a register row: what a reader would grep for
export function subjectTokens(row) {
  const txt = `${row.title || ''} \n ${row.mechanism || ''}`;
  const out = new Set();
  for (const m of txt.matchAll(/`([^`\n]{2,60})`/g)) out.add(m[1].trim());
  for (const m of txt.matchAll(/\b([A-Za-z0-9_\-]+\.(?:ts|tsx|js|mjs|cjs|kt|swift|sql|md|json|yml|yaml|html|css|xml|toml))\b/g)) out.add(m[1]);
  for (const m of txt.matchAll(/\b([a-zA-Z][a-z0-9]+(?:[A-Z][a-z0-9]+)+)\b/g)) out.add(m[1]);
  for (const m of txt.matchAll(/\b([A-Za-z][A-Za-z0-9]*(?:_[A-Za-z0-9]+)+)\b/g)) out.add(m[1]);
  for (const m of txt.matchAll(/(\/[a-z0-9][a-z0-9\-_]*(?:\/(?::?[a-z0-9\-_.]+))+)/gi)) out.add(m[1]);
  return [...out]
    .map((s) => s.replace(/^[`'"]+|[`'"]+$/g, '').trim())
    .filter((s) => s.length >= 4 && s.length <= 60);
}

// ================================================================ R08 / QUEUE #441
// THE CUT RULE. The old cut (WAVE-PLAN-0003.md:23 -- NOT :34, which is the batch table;
// rule 22(l)) keyed every row on files[0] alone. files[] is an EVIDENCE list emitted by the
// intake parser; its ORDER carries no information about which file must change. Measured on
// the 750 labelled items: when the mechanism text names files[0], 19.5% came back
// wrong_owner; when it names a DIFFERENT cited file, 44.9%. That gap is the defect.
//
// The fix site is the ONE file a room must be able to edit. Every other cited path is
// CONTEXT: evidence the room may read and must not be required to own.

const BASENAME_RE = /\b([A-Za-z0-9_.\-]+\.(?:ts|tsx|js|mjs|cjs|kt|swift|sql|md|json|yml|yaml|html|css|xml|toml|sh))\b/g;

/** Basenames the row's own prose names, lowercased. */
export function namedBasenames(row) {
  const txt = `${row.mechanism || ''} \n ${row.title || ''}`;
  const out = new Set();
  for (const m of txt.matchAll(BASENAME_RE)) out.add(m[1].toLowerCase());
  return out;
}

/**
 * fixSite(row, ownerOf) -> { site, why, context }
 *
 * Deterministic, total, and computed from the ROW ALONE plus the ownership table.
 * Never returns undefined for a row that cites at least one file, and never drops a row
 * (22(m2)): a row with no citation returns site=null and is dispatched by slice, still live.
 */
export function fixSite(row, ownerOf) {
  const files = row.files || [];
  if (!files.length) return { site: null, why: 'no-citation', context: [] };
  const named = namedBasenames(row);
  const mech = files.filter((f) => named.has(f.split('/').pop().toLowerCase()));
  const ctx = (site) => files.filter((f) => f !== site);

  if (mech.length) {
    // Prefer a mechanism-named file some room actually owns; else the first named one.
    const ownedNamed = mech.filter((f) => ownerOf.has(f));
    const site = (ownedNamed.length ? ownedNamed : mech)[0];
    return {
      site,
      why: ownedNamed.length ? (mech.length === 1 ? 'mechanism' : 'mechanism-multi') : 'mechanism-unowned',
      context: ctx(site),
    };
  }
  // The mechanism names none of the cited files. Fall back to the row's CENTRE OF GRAVITY:
  // the cited file that shares its directory with the most other cited files. A row citing
  // twelve paths under services/trust-core/src and one DEPLOY.md is a trust-core row, and
  // ordering alone would hand it to whoever owns DEPLOY.md. Ties go to the room owning the
  // most cited files, then to files[] order -- arbitrary, but deterministic and recorded.
  const cand = files.filter((f) => ownerOf.has(f));
  if (!cand.length) return { site: files[0], why: 'orphan', context: ctx(files[0]) };
  const dirCount = new Map();
  for (const f of files) { const d = f.split('/').slice(0, -1).join('/'); dirCount.set(d, (dirCount.get(d) || 0) + 1); }
  const perRoom = new Map();
  for (const f of cand) for (const o of ownerOf.get(f) || []) perRoom.set(o.room, (perRoom.get(o.room) || 0) + 1);
  const keyOf = (f) => {
    const d = f.split('/').slice(0, -1).join('/');
    const rooms = (ownerOf.get(f) || []).map((o) => perRoom.get(o.room) || 0);
    return [dirCount.get(d) || 0, rooms.length ? Math.max(...rooms) : 0, -files.indexOf(f)];
  };
  let site = cand[0], bk = keyOf(cand[0]);
  for (const f of cand.slice(1)) {
    const k = keyOf(f);
    if (k[0] > bk[0] || (k[0] === bk[0] && (k[1] > bk[1] || (k[1] === bk[1] && k[2] > bk[2])))) { site = f; bk = k; }
  }
  const dirTie = [...dirCount.values()].filter((v) => v === bk[0]).length > 1;
  return { site, why: dirTie ? 'centre-of-gravity-tie' : 'centre-of-gravity', context: ctx(site) };
}

/** The room the new cut assigns a row to. null = undecidable (never dropped). */
export function recutRoom(row, ownerOf, preferBatch) {
  const fs_ = fixSite(row, ownerOf);
  if (!fs_.site) return { room: null, ...fs_ };
  const owners = ownerOf.get(fs_.site) || [];
  if (!owners.length) return { room: null, ...fs_, why: fs_.why === 'no-citation' ? fs_.why : 'orphan' };
  // A path can be listed by rooms in several batches; keep the row in its own batch when it can.
  const same = owners.find((o) => o.batch === preferBatch);
  return { room: (same || owners[0]).room, batch: (same || owners[0]).batch, ...fs_ };
}

const MAX_DF = 60; // a token in more than this many files locates nothing

function deriveOwner(row) {
  buildIndex();
  const toks = subjectTokens(row);
  const simple = [], complex_ = [];
  for (const t of toks) (/^[A-Za-z_][A-Za-z0-9_]*$/.test(t) ? simple : complex_).push(t);
  const distinctive = simple.filter((t) => (df.get(t) || 0) > 0 && (df.get(t) || 0) <= MAX_DF);
  const score = new Map();
  const add = (f, w) => score.set(f, (score.get(f) || 0) + w);
  for (const t of distinctive) {
    const w = 1 / Math.log2(2 + (df.get(t) || 1));
    for (const [f, words] of index) if (words.has(t)) add(f, w);
  }
  for (const t of complex_) {

    for (const [f, txt] of rawText) if (txt.includes(t)) { hits.push(f); if (hits.length > MAX_DF) break; }
    if (hits.length && hits.length <= MAX_DF) { const w = 1.4 / Math.log2(2 + hits.length); for (const f of hits) add(f, w); }
  }
  const ranked = [...score].sort((a, b) => b[1] - a[1]);
  return {
    tokens: toks, usable: distinctive.length + complex_.length,
    top: ranked.slice(0, 5).map(([f, s]) => ({ file: f, score: +s.toFixed(3) })),
