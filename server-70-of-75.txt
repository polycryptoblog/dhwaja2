/**
 * Review queue + audit log. Every mutating console action goes through here
 * and leaves an audit row — audit is written in the SAME batch as the state
 * change, so an action can never land unaudited.
 *
 * Item types and their approve semantics:
 *  - claim        → claim_proven event (passport becomes claimed) PLUS, when the
 *                   item names an account_key, the per-account proof row that
 *                   makes the approval visible to `accountProofStanding` — the
 *                   event alone is invisible to every freshness reader
 *  - verification → verification_granted event (passport becomes verified)
 *  - report       → (impersonation) becomes a warning draft; see warnings.ts
 *  - dispute      → operator disposition; may drive a correction elsewhere
 *
 * Warnings are NOT approved here — they route through warnings.ts, which adds
 * the cooling-off/second-look/wording gate that reports and claims don't need.
 *
 * A DECISION THAT CLEARS SOMEBODY CLEARS THE MACHINE TOO. `rejectItem` on a
 * report writes the `caution_suppressions` kill-switch row in the same batch as
 * the rejection (see `cautionSuppressionStmt`); `warnings.ts retractWarning`
 * does the same for a not-a-finding disposition. Without that, saying "no"
 * handed the subject straight back to `cautionEntries`.
 */

export type ItemType = "claim" | "verification" | "report" | "dispute" | "caution_dispute";
/**
 * `'retracted'` was MISSING from this union while `warnings.ts retractWarning`
 * had been writing it to `review_queue.state` since the day it shipped — the TS
 * type and the database had simply drifted apart, and `rowToItem`'s unchecked
 * `as ItemState` cast is why nothing ever noticed. Added, not because a new
 * state was introduced, but because this one already existed.
 */
/** Every state a queue item can hold. VALUES, not just a type: `creator-recovery.ts`
 * partitions this set to answer a locked-out creator, and a type alone cannot be
 * iterated by the test that pins every member to an arm — which is how that read
 * shipped testing for a state from a DIFFERENT table's vocabulary. */
export const ITEM_STATES = ["open", "approved", "rejected", "queued_warning", "published", "retracted", "withdrawn"] as const;
export type ItemState = (typeof ITEM_STATES)[number];

/**
 * The TERMINAL DECISION on an item, as a closed set — matched by a CHECK on the
 * column since migration 0019. It used to be free-form `TEXT` with a stale

 * English sentence into it; that sentence now lives in `disposition_note`.
 */
export const ITEM_DISPOSITIONS = ["approved", "rejected", "published", "retracted", "withdrawn"] as const;
export type ItemDisposition = (typeof ITEM_DISPOSITIONS)[number];

function rowToItem(r: ItemRow): ReviewItem {
  return {
    item_id: r.item_id,
    item_type: r.item_type as ItemType,
    state: r.state as ItemState,
    subject: r.subject,
    payload: JSON.parse(r.payload) as Record<string, unknown>,
    created_at: r.created_at,
    queued_at: r.queued_at,
    second_look: r.second_look,
    second_look_by: r.second_look_by ?? null,
    wording_id: r.wording_id,
    disposition: r.disposition as ItemDisposition | null,
    disposition_note: r.disposition_note ?? null,
    result_event: r.result_event,
  };
}

export function auditStmt(
  db: DbLike,
  actor: string,
  action: string,
  itemId: string | null,
  detail: string,
  now: string,
  actorKind?: "operator" | "creator_passkey" | "agency" | "system",
) {
  // F-31: the column existed to distinguish who acted; the single writer
  // inlined 'operator' regardless, so every system-attributed line (actor ids
  // like "system:*", "AUTO_OPERATOR") was filed as a human decision, and
  // `/ops/audit` (admin.ts) projects the value verbatim. Sixth parameter,
  // defaulted to the prior literal so every existing call site is unchanged
  // unless it opts in.
  //
  // OP-09 (B27): "unless it opts in" was the hole — the system and agency callers
  // never did, so their rows read as operator decisions. With no explicit kind the
  // label is now DERIVED FROM THE ACTOR ID (auditActorKind below); an explicit kind
  // still wins.
  return db
    .prepare(
      "INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) VALUES (?, ?, ?, ?, ?, ?)",
    )
    .bind(actor, action, detail, now, itemId, actorKind ?? auditActorKind(actor));
}

/** One audit row's fields, for a module that builds the row itself (`auditStmtAfterWrite`). */
export interface AuditRow {
  actor: string;
  action: string;
  itemId: string | null;
  detail: string;
  at: string;
  actorKind?: "operator" | "creator_passkey" | "agency" | "system";
}

/** QUEUE #942(a) (D-0946(1)): `auditStmt`'s row as `INSERT ... SELECT ... WHERE changes() = 1`, so in a
 * `db.batch` it is written only when the statement directly before it changed exactly one row. */
export function auditStmtAfterWrite(db: DbLike, row: AuditRow): DbStatement {
  return db
    .prepare("INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, ?, ?, ?, ?, ? WHERE changes() = 1")
    .bind(row.actor, row.action, row.detail, row.at, row.itemId, row.actorKind ?? auditActorKind(row.actor));
}

/** I18-FXA (D-0949(1), no-op audit rows): placed DIRECTLY after a change's write in a `db.batch`, it inserts
 * nothing when that write changed exactly one row and otherwise violates `audit_log.detail NOT NULL`, aborting the
 * batch, so the caller's joined rows never record a write that changed nothing. reports.ts / agencies.ts keep
 * private copies of the same statement. */
export function changeGateStmt(db: DbLike): DbStatement {
  return db.prepare(
    "INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT 'system:change-gate', 'change-gate', NULL, '', NULL, 'system' WHERE changes() <> 1",
  );
}
const CHANGE_GATE_REFUSED_RE = /NOT NULL constraint failed:\s*audit_log\.detail\b/i;
/** Did this batch fail on `changeGateStmt` (the write before it changed no row)? */
export function isChangeGateRefusal(e: unknown): boolean {
  return e instanceof Error && CHANGE_GATE_REFUSED_RE.test(e.message);
}

/** The `actor_kind` an actor id implies: `system` / `system:*` (sweeps, the
 * removal auto-operator, creator-recovery's mail failures) and `req:*` (a
 * self-service requester folded into a recovery case — nobody on staff acted)
 * are `system`; `agency:*` is `agency`; anything else — operators are `op_<hex>`
 * (operators.ts) — stays `operator`. */
export function auditActorKind(actor: string): "operator" | "agency" | "system" {
  if (actor === "system" || actor.startsWith("system:") || actor.startsWith("req:")) return "system";
  if (actor.startsWith("agency:")) return "agency";
  return "operator";
}

/** Every state a review item can be decided OUT of. The lock's second key half;
 * a door passes the state it validated before deciding. `open` is where the five
 * cross-module doors contend; the warning lane's two later steps leave
 * `queued_warning` and `published`. */
export type DecidableState = "open" | "queued_warning" | "published";

/** What a decision DID. The queue's own closed set (migration 0019) plus
 * `drafted`, the one transition the queue records as a STATE (`queued_warning`)
 * rather than a disposition. */
export type DecisionDisposition = "approved" | "rejected" | "drafted" | "published" | "retracted" | "withdrawn";

/**
 * THE DECISION LOCK: an item is moved out of a given state AT MOST ONCE, and the
 * proof of that is a PRIMARY KEY, not a re-read.
 *
 * WHAT THIS REPAIRS (adversarial round, 2026-08-27). `approveItem` was a
 * textbook check-then-act: read the item, assert `state === 'open'`, append the
 * ledger event, THEN run a batch guarded only on the queue flip. Two operators
 * held at a barrier on the item read both passed the check and both appended.
 * On a VERIFIED passport both drafts become `verification_renewed` — which the
 * ledger accepts against `verified` and which LEAVES the passport `verified` —
 * so the state machine, the thing that catches this for a `claim` ("claim_proven
 * requires unclaimed"), had nothing to say. Measured: two `verification_renewed`
 * events for one item, one of them orphaned (no queue row points at it), two
 * contradictory audit lines, and two `verification_refs` rows claiming the same
 * `item_id`. The loser's `UPDATE … WHERE state='open'` matched nothing, exactly
 * as designed — but a conditional UPDATE that matches no row does not fail a
 * batch, so every UNCONDITIONAL rider beside it landed anyway.
 *
 * WHY A LOCK RATHER THAN A GUARD ON EACH RIDER. Guarding the riders fixes the
 * riders and leaves the second EVENT standing, and the event is the
 * consequential half: a ledger row is forever. The event is written by
 * `appendEvents`, not by this module, so the only way to roll it back is to fail
 * the batch it rides in — which is what a violated PK does. This is the idiom
 * `claim_consumed` (claims.ts) and `warning_publications` (warnings.ts) already
 * use for exactly-once, spelled ONCE here so every door in the product uses the
 * same one.
 *
 * WHY THE KEY IS THE TRANSITION, NOT THE ITEM (migration 0031). 0030 keyed this
 * on `item_id` alone and therefore could only ever cover doors that END an item
 * in one step — which is why it covered the two in this module and named the
 * rest as a remaining gap. The warning lane legitimately decides ONE item more
 * than once: a report is drafted, then published, then perhaps retracted, and
 * all three are real operator decisions on the same `item_id`. What is genuinely
 * at-most-once is LEAVING A GIVEN STATE, so that is the key: `(item_id,
 * from_state)`. Two doors that both read `state = 'open'` — an `approveItem` and
 * a `face-triage` cleanup rejection, say — both insert (item, 'open') and
 * exactly one survives, whichever module they live in; a later, legitimate
 * decision on the same item inserts a different row and is never blocked.
 *
 * IT ALSO CLOSES THE CHECK-THEN-ACT WINDOW, at no extra cost. Every door reads
 * the item, validates its state, and only then builds a batch. A door that read
 * a stale `open` and arrives after the item has moved on still collides with the
 * row the winner holds for `open` — so it fails and rolls back WHOLE rather than
 * landing its riders against somebody else's decision. The guarded `UPDATE …
 * WHERE state = '…'` beside it stays, but it is belt: a conditional UPDATE that
 * matches no row does NOT fail a batch, which is the entire defect.
 *
 * THE SIX DOORS, and the state each one takes:
 *   review.ts      approveItem            open           -> approved
 *   review.ts      rejectItem             open           -> rejected
 *   recovery.ts    resolveRecoveryCase    open           -> approved
 *   face-triage.ts the failed-draft cleanup  open        -> rejected
 *   warnings.ts    draftWarning           open           -> drafted
 *   warnings.ts    publishWarning         queued_warning -> published
 *   warnings.ts    retractWarning         published      -> retracted
 *   importer.ts    retractBatch           queued_warning -> withdrawn
 *
 * `eventId` is the event this decision authorised, or `null` when the door
 * appends none (I3-2007, D-0850(3)); every in-tree door passes it explicitly
 * (the default exists for the pinned `concurrency-guards.test.ts` lock probe). A door whose event id is
 * minted inside the same `appendEvents` batch passes `null` here and binds the
 * id with an `UPDATE review_decisions … = <sub-select>` rider in THAT batch
 * (approveItem, publishWarning, retractWarning). No `'__pending__'` marker and
 * no post-batch backfill: rows written before this change may still carry the
 * marker, and nothing reads `review_decisions.event_id` by value.
 */
export function reviewDecisionStmt(
  db: DbLike,
  itemId: string,
  fromState: DecidableState,
  disposition: DecisionDisposition,
  actor: string,
  now: string,
  eventId: string | null = null,
) {
  return db
    .prepare("INSERT INTO review_decisions (item_id, from_state, disposition, decided_by, decided_at, event_id) VALUES (?, ?, ?, ?, ?, ?)")
    .bind(itemId, fromState, disposition, actor, now, eventId);
}

/** True iff this error is the decision lock being lost — "somebody already moved
 * this item out of that state" — rather than a real fault. Exported so the doors
 * that write their batch directly (rather than through `tryAppendEvents`) all
 * translate it the same way instead of each carrying its own regex. */
export function isDecisionLost(e: unknown): boolean {
  return e instanceof Error && DECISION_LOST_RE.test(e.message);
}

/** The driver text a lost uniqueness constraint produces, in the two engines
 * this runs on (better-sqlite3 in test, D1 in production). Doors that append an
 * event get the translation for free from `tryAppendEvents`
 * (`LedgerRefusal.conflict`): `approveItem`, `warnings.ts` publish and retract.
 * Doors that write their batch directly do it themselves through
 * `isDecisionLost` — `rejectItem` here, plus `warnings.ts draftWarning`,
 * `recovery.ts` and `importer.ts`, so there is ONE regex rather than four.
 * Narrow on purpose — a CHECK, NOT NULL or FOREIGN KEY failure is a bug in the
 * batch we just built and must surface as an error, never as "somebody beat you
 * to it" (the same reasoning as `UNIQUENESS_LOST_RE` in events.ts). */
const DECISION_LOST_RE = /UNIQUE constraint failed|PRIMARY KEY must be unique/i;

/** A ledger/queue subject that names an ACCOUNT is `sha256:<64 hex>`; a subject
 * that names one of our own passports is `psp_…`. Only the former can be a
 * caution subject, because `caution_suppressions` is keyed by subject hash. */
const SUBJECT_HASH_RE = /^sha256:([0-9a-f]{64})$/;

/**
 * THE KILL-SWITCH RIDER: mark a subject hash as never-cautioned, in the SAME
 * BATCH as the operator decision that cleared it.
 *
 * WHAT THIS REPAIRS (audit HIGH, 2026-08-26). `cautionEntries` selects on
 * `fan_report_items.state = 'open'` and reads NOTHING about what an operator
 * decided, so two different ways of saying "no" put the machine accusation
 * straight back:
 *
 *   - `rejectItem` on an escalated fan report — an operator ruling THIS IS NOT
 *     AN IMPERSONATOR — deliberately returns the fan item to `'open'` (so future
 *     reporters are not swallowed by a stranded item), which is exactly the
 *     state the caution query wants. The next 6-hourly rebuild republished it.
 *   - `retractWarning` with a NOT-A-FINDING disposition — an operator saying the
 *     published accusation was WRONG — leaves the item `'retracted'`, which is
 *     in the soft-reopen set, so ONE new report flipped it back to `'open'` with
 *     the original band-10 reporter row still attached, and it republished.
 *
 * There was a manual escape (`dismissReport`, `suppressReport`) but nothing
 * automatic and nothing that prompted, and the rebuild runs in the gap.
 *
 * WHY THE EXISTING ROW AND NOT A NEW `cleared` ITEM STATE. Both were on the
 * table. A new terminal state means widening the `fan_report_items.state` CHECK
 * (a migration), teaching the reopen predicate, the queue filters and the ops
 * console about it — and it does NOT actually cover `rejectItem`, whose whole
 * point is to hand the item BACK to the open queue. `caution_suppressions`
 * already exists, is already the thing `cautionEntries` consults first, is
 * idempotent, and rides a batch. Removal must always be faster than addition:
 * this is one INSERT next to a decision that is already being audited.
 *
 * IT SUPPRESSES THE MACHINE CLASS ONLY. A warning can still be published against
 * the same subject through the operator-gated funnel; what stops is Dhwaja
 * publishing an amber accusation about someone a human declined to accuse.
 *
 * Lives here rather than in `caution.ts` for the same reason `auditStmt` does:
 * this module owns "an operator decision, its audit line, and the statements
 * that ride the same batch", and `caution.ts` already imports from here (the
 * reverse would be a cycle). `suppressCaution` builds its row from this too, so
 * there is exactly one spelling of the INSERT.
 */
export function cautionSuppressionStmt(
  db: DbLike,
  operator: string,
  subjectHashHex: string,
  reason: string | null,
  now: string,
) {
  return db
    .prepare(
      "INSERT INTO caution_suppressions (subject_hash, operator, reason, created_at) VALUES (?, ?, ?, ?) ON CONFLICT(subject_hash) DO NOTHING",
    )
    .bind(subjectHashHex, operator, reason === null ? null : String(reason).slice(0, 240), now);
}

/**
 * The same rider, taken from a review/ledger SUBJECT string — `[]` when the
 * subject names no account hash (a claim/verification item names a `psp_`, and
 * there is no caution to clear for one). Spreading an array keeps the call site
 * a single `...` inside the batch literal, so the atomicity is visible.
 */
export function clearCautionStmts(
  db: DbLike,
  operator: string,
  subject: string,
  reason: string,
  now: string,
) {
  const m = SUBJECT_HASH_RE.exec(subject);
  return m ? [cautionSuppressionStmt(db, operator, m[1]!, reason, now)] : [];
}

export async function getItem(db: DbLike, itemId: string): Promise<ReviewItem | null> {
  const row = await db
    .prepare("SELECT * FROM review_queue WHERE item_id = ?")
    .bind(itemId)
    .first<ItemRow>();
  return row ? rowToItem(row) : null;
}

/**
 * S-03: this used to be `SELECT *` with no LIMIT and no cursor — the one
 * screen an operator works from, growing linearly with the open backlog
 * (measured: 3000 rows, 1.94MB, in one response). Capped and keyset-paged
 * now, on `(created_at, item_id)` — a KEYSET, not an OFFSET, because a
 * decision moves an item out of `state = 'open'`, so an offset would skip
 * the row after whichever one just got decided. `item_id` breaks ties on
 * one `created_at` millisecond so paging is stable under concurrent
 * enqueues. The default and cap mirror `auditList`'s (admin.ts): existing
 * callers that pass neither `limit` nor `before` are unaffected below the
 * cap and, for the first time, bounded above it.
 */
export async function listOpen(
  db: DbLike,
  itemType?: ItemType,
  opts: { limit?: number; before?: { created_at: string; item_id: string } } = {},
): Promise<ReviewItem[]> {
  const limit = Math.max(1, Math.min(200, opts.limit ?? 200));
  const conditions: string[] = ["state = 'open'"];
  const params: unknown[] = [];
  if (itemType) {
    conditions.push("item_type = ?");
    params.push(itemType);
  }
  if (opts.before) {
    conditions.push("(created_at > ? OR (created_at = ? AND item_id > ?))");
    params.push(opts.before.created_at, opts.before.created_at, opts.before.item_id);
  }
  const q = db
    .prepare(`SELECT * FROM review_queue WHERE ${conditions.join(" AND ")} ORDER BY created_at ASC, item_id ASC LIMIT ?`)
    .bind(...params, limit);
  const { results } = await q.all<ItemRow>();
  return results.map(rowToItem);
}

/** Enqueue a new item (claims/verification apps/reports/disputes arrive here). */
export async function enqueue(
  db: DbLike,
  actor: string,
  item: { item_type: ItemType; subject: string; payload: Record<string, unknown>; import_batch_id?: string },
  now: () => string = () => new Date().toISOString(),
): Promise<string> {
  const id = mintId("itm");
  const ts = now();
  // RGP-L4 (I15-FX4, I3-0364): an item TAGGED to an import batch is written only
  // while that batch is still `applying`, decided IN this statement. Otherwise
  // `created_at` is NULL, the NOT NULL fails and the audit row below rolls back
  // with it (`importer.ts` IMPORT_BATCH_CLOSED_RE names this column), so an apply
  // racing a retraction that already finished cannot leave an 'open' item under
  // the retracted batch. The same term importer.ts RECORD_WHILE_APPLYING writes.
  const batchId = item.import_batch_id ?? null;
  const createdAt = batchId === null ? "?" : "CASE WHEN (SELECT status FROM import_batches WHERE batch_id = ?) = 'applying' THEN ? END";
  const createdBinds: unknown[] = batchId === null ? [ts] : [batchId, ts];
  await db.batch([
    db
      .prepare(
        `INSERT INTO review_queue (item_id, item_type, state, subject, payload, created_at, updated_at, import_batch_id) VALUES (?, ?, 'open', ?, ?, ${createdAt}, ?, ?)`,
      )
      .bind(id, item.item_type, item.subject, JSON.stringify(item.payload), ...createdBinds, ts, batchId),
    // R3-I29-Q3 (D-1037(16), "stop copying it and keep only the case id"), extended by I34-W3-REVIEW: the detail is the
    // item id, never the payload (the payload lives in review_queue.payload), for every item type but `report`, whose
    // detail scripts/ops/c057-inherited-drafts.mjs still reads as the originating report payload (open: I34-W3-REVIEW).
    auditStmt(db, actor, `enqueue:${item.item_type}`, id, item.item_type === "report" ? JSON.stringify(item.payload).slice(0, 500) : id, ts),
  ]);
  return id;
}

/**
 * What the operator may CORRECT on the item at approval time, rather than
 * having to reject a real application because a field is wrong.
 *
 * `expires_at` exists because `approveItem` refuses a verification with no
 * parseable deadline (correctly — see below) while `POST /ops/review/:id/approve`
 * read no body and the detail pane was a read-only `<pre>`. The refusal was
 * therefore a HARD DEAD END: `Approved.` → `Failed (400).`, forever, with no
 * field anywhere in the product that could supply the missing date. Accepting
 * the correction here is what turns the refusal back into the thing its own
 * comment claims it is — "the review queue is exactly where a human can fix it".
 *
 * The correction is merged into the item's payload for THIS approval and is
 * recorded in the audit row; it is not a queue-payload edit route (there is no
 * PATCH on a review item, deliberately — the payload is what the applicant
 * submitted).
 */
export interface ApproveOverrides {
  expires_at?: string;
  kyc_ref?: string;
  /** D-0442: the recorded reason a `fraud_at_proven_creator` approval (= delist) requires. */
  reason?: string;
}

/**
 * R-17 (QUEUE #168): THE TERM STARTS WHEN IT IS GRANTED. `requestVerification`
 * mints `expires_at` from the REQUEST clock (verification.ts), so an approval
 * that sat in the queue handed out a grant already mostly spent — 65 days of a
 * year after a 300-day backlog. With no operator override and a readable
 * `requested_at`, the requested TERM (`expires_at - requested_at`) is restarted
 * at the approval instant (never earlier than the request). An operator-typed
 * `expires_at` is the operator's decision and keeps its instant, as does an
 * item that carries no `requested_at`.
 *
 * D-1112(8) (I34-FX1-CORE, RGC-N1): EVERY DEADLINE IS WRITTEN IN ONE FORM, the
 * canonical UTC ISO of `exp` (`toISOString()`), never the typed text. An
 * override typed `2031-01-11`, `+05:30` or `+0530` is stored as the instant
 * `parseVerificationExpiry` read from it, which every door reads the same way
 * (standing.ts `grantDeadlineMs` / `grantLiveAtSql`).
 */
function renewalExpiry(exp: number, requestedAt: unknown, override: unknown, approvedAt: string): string {
  const typed = new Date(exp).toISOString();
  if (typeof override === "string" && override.length > 0) return typed;
  const req = typeof requestedAt === "string" ? Date.parse(requestedAt) : Number.NaN;
  const at = Date.parse(approvedAt);
  if (!Number.isFinite(req) || !Number.isFinite(at) || exp <= req) return typed;
  return new Date(Math.max(at, req) + (exp - req)).toISOString();
}

/** The id of the ledger row THIS batch just appended, read back inside the same
 * batch (binds: subject, event_type). events.ts runs `extraStatements` after
 * the ledger INSERT in one transaction, so the newest row of that subject and
 * type is ours — no other writer can interleave inside the batch. */
const THIS_EVENT_SQL = "(SELECT event_id FROM trust_events WHERE subject = ? AND event_type = ? ORDER BY seq DESC LIMIT 1)";

const ACCOUNT_MOVED_MESSAGE =
  "the account this claim names is no longer an active account of the passport it was filed against — it moved or was revoked while the item sat in the queue, so this claim can no longer be granted";

const PASSPORT_NOT_CLAIMABLE_MESSAGE =
  "the passport this claim names is no longer open to be claimed — it was claimed, retracted or delisted while the item sat in the queue, so this claim can no longer be granted";

/** I63-P3 R1 (D-1539 (f)): Jason's sentence, D-1553 / I63-Q6. The add arm's refusal (both arms below: !mayAdd and the
 * domainAddGate block); the claim arm's sentence above is false for an add (the passport is claimed). */
const ADD_NOT_GRANTABLE_MESSAGE =
  "Can't approve: this passport or its agency grant no longer allows adding this domain. Reject the request or ask the creator to start again.";

/** True iff approveItem's commit-time ownership gate refused (the proof row's
 * `account_key` was fed NULL because the account is no longer live-held). Pinned
 * to the exact column, like `isUnaccountableCharacter`. */
function isClaimAccountMoved(e: unknown): boolean {
  return e instanceof Error && /NOT NULL constraint failed:\s*claim_challenges\.account_key/i.test(e.message);
}

/** Approve a claim or verification item → writes the corresponding trust event. */
export async function approveItem(
  db: DbLike,
  actor: string,
  itemId: string,
  now: () => string = () => new Date().toISOString(),
  overrides: ApproveOverrides = {},
  /** QUEUE #19: `publishHook(...)`'s thunk; `tryAppendEvents` runs it once after the
   * batch commits, publishing a `claim` approval's `creator_claimed` outbox row now. */
  afterCommit?: () => void,
): Promise<{ event_id: string; event_type: string }> {
  const stored = await getItem(db, itemId);
  if (!stored) throw new ReviewError("no such item", 404);
  if (stored.state !== "open") throw new ReviewError(`item not open (state ${stored.state})`, 409);
  if (stored.item_type === "dispute" && stored.payload.kind === FRAUD_REVIEW_KIND) {
    return approveFraudAtProvenCreator(db, actor, stored, overrides.reason ?? "", now);
  }
  const item: ReviewItem = {
    ...stored,
    payload: {
      ...stored.payload,
      ...(typeof overrides.expires_at === "string" && overrides.expires_at.length > 0 ? { expires_at: overrides.expires_at } : {}),
      ...(typeof overrides.kyc_ref === "string" ? { kyc_ref: overrides.kyc_ref } : {}),
    },
  };

  // The account a manual claim is ABOUT (`requestManualClaim` always carries
  // one). Read here rather than at the batch, because it decides whether the
  // approval may happen at all — see the ownership check in the claim branch.
  const claimAccountKey = item.item_type === "claim" && typeof item.payload.account_key === "string" ? item.payload.account_key : "";
  // I38-OAUTHREV (INV-015): an approved OAuth review item PINS the provider id its ceremony proved, as the direct
  // exit does (oauth-claims.ts spent->proven flip): a proven oauth_challenges row carrying provider_pid, which is what
  // the next OAuth proof (priorPid) and seededOrUnpinnedHold read. A new row, not a flip of the ceremony's own: that
  // row is 'spent', and pruneOAuthChallenges deletes spent rows past retention, while a review can sit longer.
  const oauthPid =
    item.payload.source === "oauth_claim" && claimAccountKey.length > 0 && typeof item.payload.provider_pid === "string" && item.payload.provider_pid.length > 0 && item.payload.provider_pid.length <= 128
      ? item.payload.provider_pid
      : null;
  // I40-FX-OAR3 (D-1154(2) OAR-3): the same value rides the claim_proven event as `pid` (draft below).

  // I63-P3 R2 (D-1535, I63-W0 amendment 5): the roster apexes this add sits under at APPROVAL, for the
  // `domain_add:under_agency_apex` row in the batch below. Empty for every other item.
  let addRiderApexes: string[] = [];
  let draft: DraftEvent;
  if (item.item_type === "claim") {
    // ONE approval instant for both arms' accountable clause (D-1116(2)); the claim arm read it at its own query before.
    const atMs = Date.parse(now());
    const accountableAt = accountableSubjectSqlAt(atMs);
    // OWNERSHIP IS A FACT WITH A DATE ON IT (adversarial round, 2026-08-27).
    //
    // `requestManualClaim` proves the named account is on this passport, active,
    // and the passport unclaimed — AT REQUEST TIME. Approval is asynchronous by
    // design (a human has to go and look at a profile), so it can be days later,
    // and this function re-read NOTHING: it wrote `item.payload.account_key`
    // verbatim into a `claim_proven` event and into the per-account proof row
    // below. An account can MOVE in between — that is exactly what
    // `account_transferred` is for, the operator's adjudication of a disputed or
    // recovered handle — and when it does, the stale approval says something
    // false about somebody else's listing. Measured after adjudicating the
    // account onto its real owner while the item sat in the queue:
    //
    //   proof rows: [{passport_id: <the claimant>, account_key: <the account>,
    //                 status:'used', purpose:'claim', method:'manual_review'}]
    //   standing(claimant, who no longer holds it): proven_current
    //   standing(real owner):                       transferred
    //
    // Two harms in one row. The claimant holds a control proof for an account
    // they do not hold, AND the real owner is DOWNGRADED by it — from
    // `listed_unconfirmed` to `transferred`, the standing whose whole meaning is
    // "a proof for this exact account exists and it is not yours", i.e. the
    // hijack signal, hung on the person who actually holds the handle.
    //
    // REFUSED OUTRIGHT, not silently skipped. Writing the event while quietly
    // omitting the proof row would leave `claim_proven` on an append-only ledger
    // asserting control of an account the subject does not have — and would tell
    // the creator, through `manualClaimStatus`, that their claim succeeded. An
    // operator approving a claim whose account has moved is being asked a
    // question that no longer has a right answer; the honest reply is that this
    // item can no longer be granted (409), leaving it open for a rejection with
    // a reason, which is the record that belongs on it.
    if (item.payload.source === "domain_add") {
      // W05B-H-review-add-approve (D-0921(4)): the ADD door's review item is a domain for an
      // already-claimed passport. The domain is what is being granted, so it is not held yet
      // and the grant is `account_added`, never `claim_proven` (which a claimed passport
      // refuses). The proof row is the add challenge `routeDomainClaimToReview` already spent
      // ('used'); events.ts refuses a suppressed or already-registered key on its own.
      // I63-P3 R1 (D-1535; I63-W0 amendments 1, 2): the add door's two eligibility questions, asked again at the approval
      // instant. (a) claims.ts `subjectMayAddAccounts`, inline (claims.ts reaches review.ts). (b) the door's own
      // `domainAddGate` with the passport's LIVE roster apexes (agency-grant-read.ts; agencies.ts reaches review.ts).
      // Refused iff the gate blocks, so an ordinary add is asked exactly what the door asked it.
      const host = String(item.payload.handle ?? "");
      const mayAdd = await db
        .prepare(`SELECT 1 AS x FROM passports WHERE passport_id = ? AND retracted = 0 AND claim_state IN ('claimed','verified') AND (${accountableAt})`)
        .bind(item.subject)
        .first<{ x: number }>();
      if (!mayAdd) throw new ReviewError(ADD_NOT_GRANTABLE_MESSAGE, 409);
      const apexes = await liveRosterApexes(db, item.subject, atMs);
      if ((await domainAddGate(db, host, item.subject, apexes)).blocked) throw new ReviewError(ADD_NOT_GRANTABLE_MESSAGE, 409);
      addRiderApexes = apexes.filter((a) => host.endsWith(`.${a}`)); // a strict subdomain (the apex itself never ends in `.apex`)
      draft = {
        event_type: "account_added",
        subject: item.subject,
        actor_kind: "operator",
        actor_id: actor,
        payload: { platform: "domain", id_kind: "domain", id: host, relationship_basis: "creator_claimed", observed_at: now() },
      };
    } else {
      if (claimAccountKey.length > 0) {
        const held = await db
          .prepare("SELECT 1 AS x FROM passport_accounts WHERE account_key = ? AND passport_id = ? AND status = 'active'")
          .bind(claimAccountKey, item.subject)
          .first<{ x: number }>();
        if (!held) {
          throw new ReviewError(ACCOUNT_MOVED_MESSAGE, 409);
        }
      }
      // #609j: the PASSPORT half of claims.ts `stillClaimable`, re-read at approval: still
      // unclaimed, unretracted, answerable at the approval's instant (`accountableSubjectSqlAt`, D-1116(2)), and the account not
      // delisted (`suppressedKeyStmt`). Inline, not imported: claims.ts reaches review.ts
      // through account-standing.ts and settings.ts, so importing it here is a cycle.
      const open = await db
        .prepare(`SELECT 1 AS x FROM passports WHERE passport_id = ? AND claim_state = 'unclaimed' AND retracted = 0 AND (${accountableAt})`)
        .bind(item.subject)
        .first<{ x: number }>();
      const delisted = claimAccountKey.length > 0 ? await suppressedKeyStmt(db, claimAccountKey).first<{ x: number }>() : null;
      if (!open || delisted) throw new ReviewError(PASSPORT_NOT_CLAIMABLE_MESSAGE, 409);
      draft = {
        event_type: "claim_proven",
        subject: item.subject,
        actor_kind: "operator",
        actor_id: actor,
        payload: { method: String(item.payload.method ?? "manual"), ...(oauthPid !== null ? { pid: oauthPid } : {}) },
      };
    }
  } else if (item.item_type === "verification") {
    // Window item 2 — `kyc_ref` NO LONGER RIDES INTO THE EVENT. It was a
    // shape-free string typed by a human here and written straight into an
    // append-only log, at exactly the point where identity-document data
    // arrives. The event now records only that verification was granted; the
    // vendor case reference is written to the mutable `verification_refs` table
    // below, in the same atomic batch as the approval flip.
    //
    // THE EXPIRY IS NOW LOAD-BEARING, SO IT IS NO LONGER COERCED. This line was
    // `String(item.payload.expires_at ?? "")`, which turned a missing deadline
    // into an empty-string one and wrote it into an append-only row. Nothing
    // read `expires_at`, so nothing noticed — until `verification-sweep.ts`
    // started enforcing it, at which point every such grant is a creator the
    // sweep can neither honour nor expire (it counts them as
    // `unreadable_expiry` and LEAVES THEM VERIFIED, deliberately, rather than
    // mass-downgrading the legacy population). Refusing here is what stops the
    // backlog growing: a verification approved without a readable deadline is an
    // operator mistake, and the review queue is exactly where a human can fix
    // it. Same rule as the sweep and the payload validator — one function.
    const expiresAt = parseVerificationExpiry(item.payload.expires_at);
    if (expiresAt === null) {
      throw new ReviewError("verification item needs a parseable expires_at before it can be approved (supply one with the approval)", 400);
    }
    // GRANT OR RENEWAL — THE SAME REVIEW DECISION, DECIDED BY THE PASSPORT AND
    // NOT BY THE APPLICANT.
    //
    // `verification_renewed` shipped fully specified and completely unwritten:
    // the ledger guard, the fold branch (mirrored in fold.ts), the payload
    // validator, the actor policy and the fan-alert copy all existed, the sweep
    // read it, and nothing in the product ever appended one. Meanwhile nothing
    // enqueued a `verification` item either, so the queue that grants
    // verification could only be filled by hand. Expiry was a one-way door.
    //
    // The renewal path is THIS path, not a second one beside it. A renewal is
    // "the same review decision as the original grant, taken again"
    // (ACTOR_POLICY's own words), so it reuses the queue, the approval, the
    // audit row, the KYC-reference table and the expiry refusal above — and the
    // ONLY thing that differs is which event the ledger will accept, which is a
    // fact about the passport's current state rather than anything the
    // requester or the operator typed. `verification_granted` requires
    // `claimed` and `verification_renewed` requires `verified` (events.ts), so
    // choosing by state is also the only choice that can be appended at all:
    // renewing a still-verified creator this way means their badge never blinks
    // out and no downgrade window opens between the decision and the grant.
    const pass = await db
      .prepare("SELECT claim_state FROM passports WHERE passport_id = ? AND retracted = 0")
      .bind(item.subject)
      .first<{ claim_state: string }>();
    if (!pass) throw new ReviewError("verification item names no live passport", 400);
    const renewal = pass.claim_state === "verified";
    draft = {
      event_type: renewal ? "verification_renewed" : "verification_granted",
      subject: item.subject,
      actor_kind: "operator",
      actor_id: actor,
      payload: {
        expires_at: renewalExpiry(expiresAt, item.payload.requested_at, overrides.expires_at, now()),
      },
    };
  } else {
    throw new ReviewError(`item type ${item.item_type} is not approved here`, 400);
  }

  // appendEvents re-validates everything (state machine, actor policy, payload).
  // `verification_renewed` is NOT on APPEND_ALLOWED — like every other governed
  // type it has exactly one door, and this is it, passed through as a governed
  // type so a generic caller still cannot mint one. The actor stays `operator`,
  // which needs no privileged-actor authorization (only creator_passkey/agency/
  // system do), so the policy in packages/schema is the whole gate.
  const governed = draft.event_type === "verification_renewed" ? ["verification_renewed"] : [];
  const ts = now();
  // Window item 2: the KYC case reference is operator bookkeeping, keyed to the
  // event it describes. It rides the SAME batch as the approval flip and the
  // audit row, so a granted verification and its reference can never half-land.
  // Absent/empty ⇒ no row (fail closed: record nothing rather than an empty one).
  const kycRef = typeof item.payload.kyc_ref === "string" ? item.payload.kyc_ref.slice(0, 512) : "";
  // AN APPROVED MANUAL CLAIM MUST LEAVE A PROOF ROW (audit HIGH, 2026-08-26).
  //
  // This branch used to write the `claim_proven` event and NOTHING else, and the
  // event is not what any freshness reader looks at. `accountProofStanding` /
  // `freshProofPairs` (account-standing.ts) read exactly two tables —
  // `claim_challenges` with `status = 'used'` and `oauth_challenges` with
  // `status = 'proven'` — so an account whose control an operator PERSONALLY
  // confirmed had no row either of them could ever see. Measured after a fully
  // approved manual claim: standing `listed_unconfirmed`, the public passport row
  // `confirmed: false`, permanently. With `serve_caution` on that same account is
  // also served amber ("listed but not proven") and loses its report shield — so
  // a creator who did everything the product asked, in front of a human, ends up
  // labelled unconfirmed and reportable on her own passport page, forever.
  //
  // The fix is to record the proof where proofs are recorded. A manual claim
  // always carries the `account_key` it is about (`requestManualClaim`), and this
  // is a per-ACCOUNT control proof, so the row is passport-bound and account-bound
  // exactly like a bio-code or DNS one — which is what makes it survive the
  // pair-bound freshness reads rather than laundering a passport-level fact into
  // an account-level claim.
  //
  //  - `status = 'used'` — the value that IS a proof (never 'issued'/'spent').
  //  - `proved_at = ts`, `valid_until` left NULL — so the human-reviewed proof
  //    ages out on the same `PROOF_TTL_MS` policy as every other one (migration
  //    0014's "rows keep the FACT, code keeps the POLICY"). A proof that never
  //    lapses is the bug; an operator-confirmed one is not exempt.
  //  - `method = 'manual_review'` — the same word the ledger payload uses, so
  //    the audit column never claims a bio code was fetched.
  //  - `purpose = 'claim'` — migration 0027's vocabulary. The row is born `used`,
  //    and every verify door requires `issued`, so it is unspendable by
  //    construction; the value is written anyway so the column never carries the
  //    fail-closed 'unknown' on a row we minted deliberately.
  //  - `expires_at = ts` — a challenge deadline is meaningless for a row that was
  //    never issued; dating it at the approval instant means it is already past,
  //    which is the direction that cannot resurrect anything.
  //  - `caller` is left to its `'-'` default: no masked address issued this, an
  //    operator did, and the operator is named on the audit line beside it.
  //  - LIVE OWNERSHIP IS RE-ASKED AT COMMIT, AND A "NO" ABORTS THE BATCH
  //    (I3-1074 / I3-1291). The refusal above fires first; this closes the
  //    window between that read and this batch (an `account_transferred` or a
  //    revoke landing in between). It used to be `SELECT … WHERE EXISTS`, which
  //    inserts NOTHING when ownership is gone — and a batch does not fail on a
  //    zero-row INSERT, so `claim_proven` and the approval committed WITHOUT
  //    their proof row. Now `account_key` (NOT NULL) is fed NULL when the
  //    account is not live-held, the batch fails, and `isClaimAccountMoved`
  //    turns that into the same 409 as the read — the `accountablePassport`
  //    idiom (events.ts `isUnaccountableCharacter`).
  //
  // EVERYTHING BELOW RIDES `appendEvents(…, extraStatements)` — one batch with
  // the ledger row, fronted by the decision lock (`reviewDecisionStmt`). Before
  // this it was `appendEvents(…)` THEN a separate `db.batch`, which is what let
  // two operators land two events for one item; see the lock's docstring for the
  // measurement. The queue flip keeps its `WHERE state = 'open'` — belt to the
  // lock's braces.
  //
  // EVERY ID IS WRITTEN IN THIS ONE BATCH — NO `__pending__` PLACEHOLDER, NO
  // SECOND BATCH (I3-1880, I3-2007). The event id is minted inside
  // `appendEvents`, but events.ts runs `[...statements, ...extraStatements]` as
  // ONE batch, so when these riders run the ledger row is already inserted and
  // `THIS_EVENT_SQL` reads its id back. It used to be a placeholder here and a
  // backfill `db.batch` after the append: a worker dying between the two left
  // the KYC reference keyed `__pending__:<item>` (unreadable by
  // `verificationRef`, undeletable by `setVerificationRef`), the proof row and
  // the decision lock pointing at no event, and nothing ever sweeping them.
  const evBind = [item.subject, draft.event_type] as const;
  const proofChallengeId = mintId("chg");
  const extra: DbStatement[] = [
    reviewDecisionStmt(db, itemId, "open", "approved", actor, ts, null),
    db.prepare(`UPDATE review_decisions SET event_id = ${THIS_EVENT_SQL} WHERE item_id = ? AND from_state = 'open'`).bind(...evBind, itemId),
    db
      .prepare(`UPDATE review_queue SET state = 'approved', disposition = 'approved', updated_at = ?, result_event = ${THIS_EVENT_SQL} WHERE item_id = ? AND state = 'open'`)
      .bind(ts, ...evBind, itemId),
    // The audit detail names the EVENT TYPE, so "was this creator granted or
    // renewed, and did the operator correct the deadline on the way in?" is
    // answerable from the audit log alone. It no longer names the event ID —
    // that is not a loss of trail: the row this audit line names (`item_id`)
    // carries the id in `review_queue.result_event` and `review_decisions.
    // event_id`, and the alternative was to keep writing the audit row OUTSIDE
    // the batch, after the append, which is precisely the unguarded rider being
    // removed.
    auditStmt(
      db,
      actor,
      `approve:${item.item_type}`,
      itemId,
      `${draft.event_type}${typeof overrides.expires_at === "string" && overrides.expires_at.length > 0 ? ` expires_at=${overrides.expires_at.slice(0, 40)}` : ""}`,
      ts,
    ),
    ...(item.item_type === "verification" && kycRef.length > 0
      ? [
          db
            .prepare(
              `INSERT INTO verification_refs (event_id, subject, item_id, kyc_ref, recorded_by, recorded_at, updated_at) SELECT ${THIS_EVENT_SQL}, ?, ?, ?, ?, ?, ?`,
            )
            .bind(...evBind, item.subject, itemId, kycRef, actor, ts, ts),
        ]
      : []),
    ...(claimAccountKey.length > 0
      ? [
          db
            .prepare(
              "INSERT INTO claim_challenges (challenge_id, passport_id, account_key, platform_key, handle, code, status, created_at, expires_at, proved_at, method, purpose, result_event) " +
                "SELECT ?, ?, CASE WHEN EXISTS (SELECT 1 FROM passport_accounts a WHERE a.account_key = ? AND a.passport_id = ? AND a.status = 'active') THEN ? ELSE NULL END, " +
                `?, ?, ?, 'used', ?, ?, ?, 'manual_review', 'claim', ${THIS_EVENT_SQL}`,
            )
            .bind(
              proofChallengeId,
              item.subject,
              claimAccountKey,
              item.subject,
              claimAccountKey,
              String(item.payload.platform ?? ""),
              String(item.payload.handle ?? ""),
              String(item.payload.code ?? "").slice(0, 128),
              ts,
              ts,
              ts,
              ...evBind,
            ),
        ]
      : []),
    // I63-P3 R2 (D-1535; I63-W0 amendment 5): the roster visibility row is written WITH the add it lists, so a rejected
    // add leaves none. The direct door's writer is domain-claims.ts `verifyDomainAdd` (`usedApexes`).
    ...(addRiderApexes.length > 0 ? [auditStmt(db, "system", "domain_add:under_agency_apex", String(item.payload.handle ?? ""), addRiderApexes.join(","), ts)] : []),
    ...(oauthPid !== null
      ? [
          db
            .prepare(
              "INSERT INTO oauth_challenges (state, passport_id, account_key, platform_key, code_verifier, status, created_at, expires_at, result_event, proved_at, method, provider_pid, purpose) " +
                `VALUES (?, ?, ?, ?, '', 'proven', ?, ?, ${THIS_EVENT_SQL}, ?, 'oauth_review', ?, 'claim')`,
            )
            .bind(mintId("orv"), item.subject, claimAccountKey, String(item.payload.platform ?? ""), ts, ts, ...evBind, ts, oauthPid),
        ]
      : []),
  ];
  // THE OTHER ATOMIC GATE REFUSES BY THROWING, and this door did not translate
  // it. `accountablePassport` (events.ts) refuses a claim or a verification for
  // a character with nobody live answering for it by feeding NULL to
  // `passports.claim_state`; `tryAppendEvents` rethrows a NOT NULL constraint
  // unchanged, so approving a verification for such a character came back to the
  // operator as a bare 500 — on the console that had just finished explaining,
  // in words, why the character was stuck. An operator is authenticated and gets
  // the SPECIFIC reason (the uniform refusal exists to protect anonymous
  // callers, not to keep operators guessing).
  const appended = await tryAppendEvents(db, [draft], now, governed, extra, [], afterCommit).catch((e: unknown) => {
    if (isUnaccountableCharacter(e)) {
      throw new ReviewError("this subject is a character with nobody live answering for it — bind a live operator on the creator's page first, then approve", 409);
    }
    if (isClaimAccountMoved(e)) throw new ReviewError(ACCOUNT_MOVED_MESSAGE, 409);
    throw e;
  });
  if (!appended.ok) {
    // ONLY the lost decision lock means "another operator decided this first"
    // — that was the ORIGINAL rule, and it undercounted. S-12: a lost race can
    // ALSO surface as a ledger STATE refusal rather than the decision lock's
    // own PK, depending on which constraint the batch's driver reports first
    // (e.g. two operators approving one claim: the loser's `claim_proven`
    // trips `claim_proven requires unclaimed` before/instead of the
    // `review_decisions` PK). The route has no typed mapping for a bare
    // `LedgerError` (console.ts `mapError`), so this fell through to a bare
    // 500 "internal" — the loser is told the console failed, not that a
    // colleague beat them to it, and the natural responses (retry, escalate,
    // reject) are all wrong on a terminal queue.
    //
    // Distinguished from a GENUINE state refusal (a bad payload, a passport
    // that moved for an unrelated reason) by re-reading the item: the queue
    // only ever leaves `'open'` through one of the six decision-lock doors, so
    // if it is no longer `'open'` here, somebody already decided it — same
    // domain answer as the conflict branch. If it is STILL `'open'`, this is
    // a real fault and must keep propagating unchanged.
    //
    // A `state` refusal on an item that IS still open is the ledger saying no
    // to THIS approval (e.g. `claim_proven requires unclaimed` after another
    // item claimed the passport). That is a domain answer an authenticated
    // operator is owed in words (I3-1023): it used to rethrow the raw
    // LedgerError, which console.ts `mapError` has no branch for — a bare 500.
    // Logged too, so a refusal that IS our bug is still seen.
    if (appended.refusal.kind !== "conflict") {
      const current = await getItem(db, itemId);
      if (current && current.state !== "open") {
        throw new ReviewError("this item has already been decided by another operator", 409);
      }
      logError({ cid: itemId, hop: "http", msg: "approveItem: ledger refused approval", err: appended.refusal.error.name });
      throw new ReviewError(`the ledger refused this approval: ${appended.refusal.message}`, 409);
    }
    throw new ReviewError("this item has already been decided by another operator", 409);
  }
  const event = appended.events[0];
  return { event_id: event!.event_id, event_type: draft.event_type };
}

/** The KYC/vendor case reference behind a verification — OPERATOR-ONLY, and

export async function rejectItem(
  db: DbLike,
  actor: string,
  itemId: string,
  reason: string,
  now: () => string = () => new Date().toISOString(),
  opts: { recoveryDoor?: boolean } = {},
  // I18-FXA D37 (D-0949(1)): the caller's statements (recovery's hold release and its audit row) ride the
  // rejection's own batch, after the decision lock, so a failing one leaves the item NOT rejected.
  extraStatements: DbStatement[] = [],
): Promise<void> {
  const item = await getItem(db, itemId);
  if (!item) throw new ReviewError("no such item", 404);
  if (item.state !== "open") throw new ReviewError(`item not open (state ${item.state})`, 409);
  // RG-AUTH-02 (I15-FX4): A RECOVERY CASE HAS ONE DOOR OUT, and this is not it.
  // Every `dispute` except the WARN49 fraud review is a recovery case (the same
  // partition `approveItem` draws, which refuses them with 400): closing one here
  // skipped the I3-1030 case lease and the recovery reject's own steps
  // (recovery.ts rejectRecoveryCase — sign-off clear, hold release). Only that
  // door passes `recoveryDoor`; any other caller, and any dispute kind added
  // later, is refused.
  if (item.item_type === "dispute" && item.payload.kind !== FRAUD_REVIEW_KIND && opts.recoveryDoor !== true) {
    throw new ReviewError("this is a recovery case — reject it through /ops/recovery/:id/reject, which holds the case lease and releases its holds", 409);
  }
  // A REJECTION IS TERMINAL AND MUST CARRY A REASON.
  //
  // There is no unreject route and `approveItem`/`rejectItem` both require
  // `state === 'open'`, so this write permanently closes a real person's claim.
  // The admin console reached it through `window.prompt(…) ?? ""`, which returns
  // `null` on CANCEL and `""` on empty-OK — both collapsed to `""`, and the
  // route accepted `""`, so pressing Escape to back out of a mis-click rejected
  // the item with a blank audit trail. The client is fixed too, but the rule
  // belongs HERE, where every caller passes: `/ops/recovery/:id/reject` has
  // required a non-empty reason since it shipped and this is the same act.
  if (reason.trim().length === 0 || !hasVisibleText(reason)) throw new ReviewError("a rejection reason is required — it is the only record of why this was refused", 400);
  const ts = now();
  try {
    await db.batch([
      // THE SAME DECISION LOCK THE APPROVAL TAKES, for the same reason: everything
      // under it is a rider that does NOT fail a batch when the state flip matches
      // nothing. Two operators — or one operator approving while another rejects —
      // both read `state = 'open'`, and the loser used to write its audit line, its
      // fan-report reopen and its caution kill-switch row against an item somebody
      // else had already decided the other way. `review_decisions` makes "this item
      // was decided" a fact one writer can hold, whichever door they came through.
      reviewDecisionStmt(db, itemId, "open", "rejected", actor, ts, null),
      db
        .prepare("UPDATE review_queue SET state = 'rejected', disposition = 'rejected', updated_at = ? WHERE item_id = ? AND state = 'open'")
        .bind(ts, itemId),
      auditStmt(db, actor, `reject:${item.item_type}`, itemId, reason.slice(0, 500), ts),
      // S65 back-link: if a FAN-report item escalated into this review item,
      // rejecting the review returns it to the open fan queue instead of
      // stranding it in 'escalated' forever (review finding — a stranded item
      // silently swallowed all future reports of that subject). Keyed by
      // review_item_id: a no-op for claims/verifications/operator reports.
      db
        .prepare("UPDATE fan_report_items SET state = 'open', review_item_id = NULL, updated_at = ? WHERE review_item_id = ? AND state = 'escalated'")
        .bind(ts, itemId),
      // ...AND THE MACHINE STOPS ACCUSING THEM TOO (audit HIGH, 2026-08-26).
      //
      // The line above is deliberately generous — it hands the report BACK to the
      // open fan queue so a future victim is not swallowed by a stranded item. But
      // `open` is precisely the state `cautionEntries` publishes from, so for a
      // corroborated fan report this rejection USED TO RE-ARM the amber caution
      // (and, before the pointer gate, the "the verified X may be here →" naming)
      // on the very subject an operator had just declined to accuse. The rebuild
      // runs 6-hourly, so the window was hours, not a race.
      //
      // Every `report` item is a subject hash (`escalateReport` sets the review
      // subject to `sha256:<hash>`, and `publishWarning` refuses a `psp_` target),
      // so `clearCautionStmts` writes exactly one row here and nothing at all for
      // a claim/verification rejection. Same batch as the state flip and the audit
      // line: a decision that clears somebody cannot half-land.
      ...(item.item_type === "report" ? clearCautionStmts(db, actor, item.subject, `review rejected: ${reason}`, ts) : []),
      // I3-1200: THE CREATOR IS TOLD. A `verification` item's subject is the
      // requesting passport (`verification.ts requestVerification`); the notice
      // rides THIS batch, so a lost decision lock writes none, and it carries no
      // part of `reason` (internal audit text).
      ...(item.item_type === "verification" && item.subject.startsWith("psp_") ? [verificationRejectedNoticeStmt(db, item.subject, itemId, ts)] : []),
      // I3-0290 b (I22-W1-AUTH): a dispute closed here releases the holds naming it. The recovery door
      // passes its own release in `extraStatements` (recovery.ts rejectRecoveryCase), so it is skipped here.
      ...(item.item_type === "dispute" && opts.recoveryDoor !== true ? disputeHoldReleaseStmts(db, actor, itemId, "dispute rejected", ts) : []),
      ...extraStatements,
    ]);
  } catch (e) {
    // `rejectItem` writes its batch directly (it appends no event, so there is
    // no `tryAppendEvents` to do the translation), which means the lost lock
    // arrives here as the driver's own constraint error. Narrowly translated to
    // the domain answer — everything else is a real fault and is rethrown, so a
    // CHECK/NOT NULL/FK failure never masquerades as "somebody beat you to it".
    if (isDecisionLost(e)) {
      throw new ReviewError("this item has already been decided by another operator", 409);
    }
    throw e;
  }
}

function only(obj: Record<string, unknown>, keys: string[]): void {
  for (const k of Object.keys(obj)) if (!keys.includes(k)) throw new RewardPolicyError(`unknown field "${k}"`);

/** Validate an operator's policy. Unknown fields anywhere are refused. */
export function parsePolicy(raw: unknown): PolicySpec {
  if (typeof raw !== "object" || raw === null || Array.isArray(raw)) throw new RewardPolicyError("policy must be an object");
  const o = raw as Record<string, unknown>;
  only(o, ["passport_id", "trigger", "rule", "reward", "per_account_lifetime", "per_day"]);
  const passportId = o.passport_id === undefined || o.passport_id === null ? null : o.passport_id;
  if (passportId !== null && (typeof passportId !== "string" || !PSP_RE.test(passportId))) throw new RewardPolicyError("passport_id must be a passport id or null");
  if (typeof o.trigger !== "string" || !REWARD_TRIGGERS.includes(o.trigger as RewardTrigger)) throw new RewardPolicyError("bad trigger");
  if (!WIRED_TRIGGERS.includes(o.trigger as RewardTrigger)) throw new RewardPolicyError(`trigger "${o.trigger}" is not wired yet — a policy on it would grant nothing`);
  // A platform-wide policy (no creator) may only read creator-neutral leaves —
  // the rule is parsed against an empty passport, so `founding_fan`, `bond`,
  // `supporter`, `early_scout` and creator-scoped achievements have nothing to
  // bind to and the parser's grantability check refuses the last.
  let rule: EligibilityRule;
  try {
    rule = parseRule(o.rule, passportId ?? "", CREATOR_NEUTRAL_ACHIEVEMENTS);
  } catch (e) {
    if (e instanceof EligibilityError) throw new RewardPolicyError(`rule: ${e.message}`, e.status);
    throw e;
  }
  if (passportId === null && usesCreatorLeaf(rule)) throw new RewardPolicyError("a platform-wide policy cannot use a creator-scoped condition (bond, supporter, founding_fan, early_scout, save, acquisition)");
  const reward = parseReward(o.reward, passportId);
  const life = o.per_account_lifetime;
  const day = o.per_day;
  if (!Number.isInteger(life) || (life as number) < 1 || (life as number) > 1000) throw new RewardPolicyError("per_account_lifetime must be 1..1000");
  if (!Number.isInteger(day) || (day as number) < 1 || (day as number) > 100_000) throw new RewardPolicyError("per_day must be 1..100000");
  return { passportId: passportId as string | null, trigger: o.trigger as RewardTrigger, rule, reward, perAccountLifetime: life as number, perDay: day as number };
}

function usesCreatorLeaf(rule: EligibilityRule): boolean {
  if (rule.kind === "all" || rule.kind === "any") return rule.rules.some(usesCreatorLeaf);
  // `connected` (F8) joins `account_age` and `achievement` as PLATFORM-WIDE: a
  // constellation is a count of external platforms that vouch for the fan being
  // a person, and it says nothing about any particular creator — so a
  // platform-wide policy may use it, which is exactly the shape the anti-Sybil
  // rewards want.
  return rule.kind !== "account_age" && rule.kind !== "achievement" && rule.kind !== "connected";
}

function parseReward(raw: unknown, passportId: string | null): Reward {
  if (typeof raw !== "object" || raw === null || Array.isArray(raw)) throw new RewardPolicyError("reward must be an object");
  const o = raw as Record<string, unknown>;
  if (o.kind === "achievement") {
    only(o, ["kind", "achievement_id"]);
    const id = o.achievement_id;
    const m = typeof id === "string" ? CUSTOM_BADGE_RE.exec(id) : null;
    if (!m) throw new RewardPolicyError("achievement_id must be custom:<passport_id>:<slug>");
    // A custom badge belongs to the creator in its name — and a creator-scoped
    // policy may only mint ITS creator's badges; a platform policy mints none
    // (a platform badge would be a new namespace, not this slice).
    if (passportId === null || m[1] !== passportId) throw new RewardPolicyError("a policy can only grant its own creator's custom badge", 403);
    return { kind: "achievement", achievement_id: id as string };
  }
  if (o.kind === "credits") {
    only(o, ["kind", "credits"]);
    if (!Number.isInteger(o.credits) || (o.credits as number) < 1 || (o.credits as number) > MAX_CREDITS_PER_GRANT) throw new RewardPolicyError(`credits must be 1..${MAX_CREDITS_PER_GRANT}`);
    return { kind: "credits", credits: o.credits as number };
  }
  throw new RewardPolicyError("reward.kind must be achievement or credits");
}

    .prepare(
      `INSERT INTO reward_policies (policy_id, passport_id, trigger, rule, reward, per_account_lifetime, per_day, active, created_by, created_at)
       SELECT ?, ?, ?, ?, ?, ?, ?, 1, ?, ?
        WHERE (SELECT COUNT(*) FROM reward_policies WHERE trigger = ? AND active = 1) < ?
          AND NOT EXISTS (SELECT 1 FROM reward_policies WHERE active = 1 AND passport_id IS ? AND trigger = ? AND rule = ? AND reward = ? AND per_account_lifetime = ? AND per_day = ?)
       RETURNING policy_id`,
    )
    .bind(policyId, spec.passportId, spec.trigger, ruleJson, rewardJson, spec.perAccountLifetime, spec.perDay, operator, now(), spec.trigger, MAX_ACTIVE_POLICIES_PER_TRIGGER, spec.passportId, spec.trigger, ruleJson, rewardJson, spec.perAccountLifetime, spec.perDay);
  const [res] = (await db.batch(audit === undefined ? [insert] : [insert, auditStmtAfterWrite(db, audit)])) as { results?: unknown[] }[];
  if ((res?.results?.length ?? 0) === 0) {
    // W-15: the INSERT's cap and duplicate predicates share one statement, so
    // changes === 0 does not say which fired. Re-check the duplicate guard
    // (the race window this closes: two concurrent submissions of the same
    // spec both pass the `activeDuplicate` pre-check above, the loser's NOT EXISTS fails here)
    // before blaming the cap.
    if (await activeDuplicate(db, spec)) throw new RewardPolicyError("an identical policy is already active", 409);
    throw new RewardPolicyError(`at most ${MAX_ACTIVE_POLICIES_PER_TRIGGER} active policies per trigger`, 409);
  }
  return { policy_id: policyId };
}

/** Is an identical policy already live? The console asks BEFORE the dual-
 * control gate so a duplicate never banks a sign-off. */
export async function activeDuplicate(db: DbLike, spec: PolicySpec): Promise<boolean> {
  const row = await db
    .prepare("SELECT 1 AS x FROM reward_policies WHERE active = 1 AND passport_id IS ? AND trigger = ? AND rule = ? AND reward = ? AND per_account_lifetime = ? AND per_day = ?")
    .bind(spec.passportId, spec.trigger, JSON.stringify(spec.rule), JSON.stringify(spec.reward), spec.perAccountLifetime, spec.perDay)
    .first();
  return row !== null;
}

/** Is the per-trigger cap already full? Asked before the gate for the same reason. */
export async function triggerFull(db: DbLike, trigger: RewardTrigger): Promise<boolean> {
  const row = await db.prepare("SELECT COUNT(*) AS n FROM reward_policies WHERE trigger = ? AND active = 1").bind(trigger).first<{ n: number }>();
  return (row?.n ?? 0) >= MAX_ACTIVE_POLICIES_PER_TRIGGER;
}

/** QUEUE #942(a) (D07): `audit` as in createPolicy — same batch, gated on the UPDATE having switched one policy off. */
export async function deactivatePolicy(db: DbLike, policyId: string, now: () => string = () => new Date().toISOString(), audit?: AuditRow): Promise<boolean> {
  if (!RWP_RE.test(policyId)) throw new RewardPolicyError("no such policy", 404);
  const off = db.prepare("UPDATE reward_policies SET active = 0, deactivated_at = ? WHERE policy_id = ? AND active = 1 RETURNING policy_id").bind(now(), policyId);
  const [res] = (await db.batch(audit === undefined ? [off] : [off, auditStmtAfterWrite(db, audit)])) as { results?: unknown[] }[];
  return (res?.results?.length ?? 0) === 1;
}

export async function listPolicies(db: DbLike): Promise<PolicyView[]> {

              (SELECT COUNT(*) FROM reward_grants g WHERE g.policy_id = p.policy_id) AS grants
         FROM reward_policies p ORDER BY p.active DESC, p.created_at DESC LIMIT 200`,
    )
    .all<{ policy_id: string; passport_id: string | null; trigger: RewardTrigger; rule: string; reward: string; per_account_lifetime: number; per_day: number; active: number; created_by: string; created_at: string; deactivated_at: string | null; grants: number }>();

    try {
      return JSON.parse(text) as T;
    } catch {
      return null;
    }
  };

/**
 * Fire a trigger. For every active policy on it scoped to this creator or
 * platform-wide: evaluate, then ONE batch — the grant row (UNIQUE +
 * caps-in-WHERE) and the reward predicated on the grant row. Returns the
 * grants that landed. Never throws (logs and continues): the caller is a
 * redemption or a save, and a reward must not be able to fail it.
 */
export async function applyRewardPolicies(
  db: DbLike,
  trigger: RewardTrigger,
  ctx: TriggerContext,
  minters: Minters,
  nowMs: number,
  /** `ANALYTICS_FAN_SECRET`. Undefined is refused by `fanKey` (ADR-0006 §2):
   * a badge that cannot be counted is not silently uncounted. Threaded from
   * the one route call site, `credentials-routes.ts`. */
  fanSecret?: string,
  pubEnv?: AnalyticsTickEnv,
  pubCtx?: { waitUntil(p: Promise<unknown>): void },
): Promise<string[]> {
  const landed: string[] = [];
  try {
    if (!ACCOUNT_RE.test(ctx.accountId) || !PSP_RE.test(ctx.passportId) || !EVENT_REF_RE.test(ctx.eventRef)) return landed;
    // CREATOR STANDING (F4 review, IDOR / Sybil): the rail the referral grants
    // use — value moves only on a CURRENTLY verified, not retracted, not
    // delisted creator's activity. A retracted impersonator's leftover
    // credentials must not keep a policy (theirs or the platform's) minting.
    // I3-1768 (D-1085(1), I34-EXPIRY-A): verified at `nowMs` — a grant lapsed by then (standing.ts grantLiveAtSql) mints nothing.
    const standing = await db
      .prepare(
        `SELECT 1 AS x FROM passports p WHERE p.passport_id = ? AND p.claim_state = 'verified' AND p.retracted = 0 AND ${grantLiveAtSql("p.passport_id")} AND NOT EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = p.passport_id)`,
      )
      .bind(ctx.passportId, new Date(nowMs).toISOString())
      .first();
    if (!standing) return landed;
    const { results: policies } = await db
      .prepare(
        `SELECT policy_id, passport_id, rule, reward, per_account_lifetime, per_day FROM reward_policies
          WHERE trigger = ? AND active = 1 AND (passport_id IS NULL OR passport_id = ?) ORDER BY created_at LIMIT ?`,
      )
      .bind(trigger, ctx.passportId, MAX_ACTIVE_POLICIES_PER_TRIGGER)
      .all<{ policy_id: string; passport_id: string | null; rule: string; reward: string; per_account_lifetime: number; per_day: number }>();
    // I3-2186 (DECIDED, I4-06-R8): every policy of ONE dispatch is evaluated
    // against the state BEFORE this dispatch wrote anything — phase 1 reads,
    // phase 2 writes — so policy N's grant can never satisfy policy N+1 in the
    // same pass. A ladder still climbs, one triggering event per rung.
    // #932e (D-0920 option C): a grant is decided on the facts at evaluation and STANDS. Phase 2
    // does not re-check the rule, and a later change of the fan's state revokes nothing
    // (test/i16-w08a-reward-grant-stands.test.ts).
    const due: Array<{ p: (typeof policies)[number]; reward: Reward; snap: Awaited<ReturnType<typeof snapshot>> }> = [];
    for (const p of policies) {
      try {
        const rule = safeRule(p.rule, p.passport_id);
        const reward = safeReward(p.reward);
        if (!rule || !reward) continue; // a row that no longer parses grants nothing
        const snap = await snapshot(db, ctx.accountId, ctx.passportId, rule, nowMs);
        if (snap.satisfied) due.push({ p, reward, snap });
      } catch (e) {
        logError({ cid: "reward_policy", hop: "http", msg: "reward_policy_evaluate_failed", err: e instanceof Error ? e.name : typeof e });
      }
    }
    for (const { p, reward, snap } of due) {
      try {
        const grantId = minters.grantId();
        const entryId = minters.entryId();
        if (!RWG_RE.test(grantId) || !CLE_RE.test(entryId)) continue;
        const ts = new Date(nowMs).toISOString();
        const dayStart = ts.slice(0, 10) + "T00:00:00.000Z";
        // Per-creator bound (F4b): only a platform-wide policy has more than
        // one creator to share `per_day` between; a creator-scoped one gets its
        // own `per_day` as the creator bound, so the third predicate is always
        // present and never looser than the second.
        const perCreator = p.passport_id === null ? perCreatorPerDay(p.per_day) : p.per_day;
        const stmts: DbStatement[] = [
          db
            .prepare(
              `INSERT INTO reward_grants (grant_id, policy_id, account_id, event_ref, snapshot, granted_at, passport_id)
               SELECT ?, ?, ?, ?, ?, ?, ?
                WHERE (SELECT COUNT(*) FROM reward_grants WHERE policy_id = ? AND account_id = ?) < ?
                  AND (SELECT COUNT(*) FROM reward_grants WHERE policy_id = ? AND granted_at >= ?) < ?
                  AND (SELECT COUNT(*) FROM reward_grants WHERE policy_id = ? AND passport_id = ? AND granted_at >= ?) < ?
                  AND EXISTS (SELECT 1 FROM reward_policies WHERE policy_id = ? AND active = 1)
                  AND (SELECT COUNT(*) FROM reward_grants g WHERE g.account_id = ? AND g.policy_id IN
                        (SELECT q.policy_id FROM reward_policies q JOIN reward_policies me ON me.policy_id = ?
                          WHERE q.passport_id IS me.passport_id AND q.trigger = me.trigger AND q.rule = me.rule AND q.reward = me.reward)) < ?
                  AND ${NOT_FROZEN_GRANT_SQL}`,
            )
            // D-0759(a), I21-Q73: the grant row is refused for a suspended or terminated account, so
            // the reward below (guarded on this row) writes nothing either.
            // I3-1267: the policy must still be active IN this statement — a
            // deactivation committed after the policy SELECT refuses the row.
            // I3-2187 (DECIDED): the lifetime cap also counts this fan's grants
            // under every policy with the same creator/trigger/rule/reward, so
            // deactivate + re-create identical does not reset it.
            .bind(grantId, p.policy_id, ctx.accountId, ctx.eventRef, JSON.stringify(snap), ts, ctx.passportId, p.policy_id, ctx.accountId, p.per_account_lifetime, p.policy_id, dayStart, p.per_day, p.policy_id, ctx.passportId, dayStart, perCreator, p.policy_id, ctx.accountId, p.policy_id, p.per_account_lifetime, ctx.accountId),
          ...(await rewardStatements(db, reward, ctx.accountId, grantId, ts, entryId, ctx.passportId, fanSecret)),
        ];
        let seqs: number[] = [];
        try {
          const span = outboxSpan(db);
          const spanRes = await db.batch([span.open, ...stmts, span.close]);
          seqs = outboxSpanSeqs(spanRes[0], spanRes[spanRes.length - 1]);
        } catch (e) {
          const msg = e instanceof Error ? e.message : String(e);
          if (/unique/i.test(msg)) continue; // already granted for this event (redelivery)
          throw e;
        }
        if (pubEnv && pubCtx) publishAfterCommit(pubEnv, pubCtx, seqs);
        if ((await db.prepare("SELECT 1 AS x FROM reward_grants WHERE grant_id = ?").bind(grantId).first()) !== null) landed.push(p.policy_id);
      } catch (e) {
        logError({ cid: "unknown", hop: "queue", msg: "reward_policy_grant_failed", policy_id: p.policy_id, err: e instanceof Error ? e.name : typeof e });
      }
    }
  } catch (e) {
    logError({ cid: "unknown", hop: "queue", msg: "reward_policies_failed", kind: trigger, err: e instanceof Error ? e.name : typeof e });
  }
  return landed;
}

async function rewardStatements(
  db: DbLike,
  reward: Reward,
  accountId: string,
  grantId: string,
  ts: string,
  entryId: string,
  passportId: string,
  fanSecret: string | undefined,
): Promise<DbStatement[]> {
  const landedGuard = "EXISTS (SELECT 1 FROM reward_grants WHERE grant_id = ?)";
  if (reward.kind === "achievement") {
    // THE SECOND WRITER OF `achievements` EMITS ITS EVENT (FINDING-0010-A-1).
    // `social.ts awardAchievements` puts an `achievement_earned` in the awards
    // batch; this branch wrote the badge and nothing else, so INV-051's "the
    // event exists iff its fact does" held in one direction only — and because
    // the erasure fan-out's GLOB matches `custom:<psp>:<slug>`, the DECREMENT
    // was emitted at erasure regardless, driving the copy's
    // `achievements − achievements_erased` stock negative for that creator.
    //
    // The event's guard is `landedGuard` — the badge INSERT's OWN predicate,
    // the same string bound the same way — so the row exists iff the grant
    // does. The passport is the trigger's creator: a custom badge is
    // creator-scoped and `parseReward` refuses one whose `custom:<psp>:` does
    // not equal the policy's creator (`parseReward`), which is this same passport.
    //
    // I3-2143 / I3-0481: the event also requires that the badge row does NOT
    // exist yet, and it runs BEFORE the INSERT OR IGNORE in the same batch, so
    // it reads the pre-insert state: it lands iff the insert creates the row. A
    // repeat grant of a held badge lands its grant row and emits nothing.
    const createsBadge = `${landedGuard} AND NOT EXISTS (SELECT 1 FROM achievements WHERE account_id = ? AND achievement_id = ?)`;
    return [
      ...outboxStatements(
        db,
        { kind: "achievement_earned", passportId, occurredDay: occurredDay(ts), fanKey: await fanKey(fanSecret, accountId, passportId) },
        { accountId, extraWhere: { sql: createsBadge, params: [grantId, accountId, reward.achievement_id] } },
      ),
      db
        .prepare(`INSERT OR IGNORE INTO achievements (account_id, achievement_id, awarded_at) SELECT ?, ?, ? WHERE ${landedGuard}`)
        .bind(accountId, reward.achievement_id, ts, grantId),
    ];
  }
  // Origin `promotional`: never cash-backed (credits.ts CASH_ORIGINS). idem =
  // the grant row, so the journal refuses a second grant for one reward even
  // if reward_grants' UNIQUE were ever relaxed.
  return [
    db
      .prepare(
        `INSERT INTO credit_ledger (entry_id, account_id, kind, origin, credits, ref, idem, reason, at)
         SELECT ?, ?, 'grant', 'promotional', ?, ?, ?, 'reward_policy', ? WHERE ${landedGuard} AND ${NOT_FROZEN_GRANT_SQL}`,
      )
      .bind(entryId, accountId, reward.credits, grantId, `reward:${grantId}`, ts, grantId, accountId),
  ];
}

function safeRule(text: string, passportId: string | null): EligibilityRule | null {
  try {
    return parseRule(JSON.parse(text), passportId ?? "", CREATOR_NEUTRAL_ACHIEVEMENTS);
  } catch {
    return null;
  }
}
function safeReward(text: string): Reward | null {
  try {
    const r = JSON.parse(text) as Reward;
    if (r.kind === "achievement" && typeof r.achievement_id === "string" && CUSTOM_BADGE_RE.test(r.achievement_id)) return r;
    if (r.kind === "credits" && Number.isInteger(r.credits) && r.credits >= 1 && r.credits <= MAX_CREDITS_PER_GRANT) return r;
    return null;
  } catch {
    return null;
  }
}

/** Own reward grants for the export (policy + when — never the rule text).
 * I3-1835: EVERY row — a subject-access export is not paged, and the old
 * `LIMIT 500` dropped the newest rows with no signal. */
export async function listOwnGrants(db: DbLike, accountId: string): Promise<Array<{ policy_id: string; event_ref: string; granted_at: string }>> {
  const { results } = await db.prepare("SELECT policy_id, event_ref, granted_at FROM reward_grants WHERE account_id = ? ORDER BY granted_at, grant_id").bind(accountId).all<{ policy_id: string; event_ref: string; granted_at: string }>();
  return results;
}

/**
 * S24 — Dhwaja credits + referrals + leaderboards. Deliberately simple: no
 * token, no NFT, no external merchant coupling. Credits are rows in our DB;
 * referral rewards trigger on VERIFIED activity (a real credential redemption,
 * a real passport claim), never on signups; leaderboards are precomputed by
 * the cron sweep and only PUBLIC-profile accounts appear (public visibility is
 * the opt-in, same rule as S22 discovery).
 *
 * Anti-farming posture (credits will eventually buy things):
 *  - referee→referrer link is set ONLY at account birth (referee PK), so
 *    self-referral and link rewriting are impossible by construction;
 *  - fan rewards need the referee to redeem a credential issued by a VERIFIED
 *    creator. This is the structural anti-farming gate (review finding F1): a
 *    merely-claimed creator can mint credentials for free, so a colluding
 *    creator+sockpuppet cluster could otherwise farm credits at zero cost.
 *    Verified status is operator-vetted and revocable — farming now requires
 *    burning a real vetted identity, whose dead-token/self-dealing abuse
 *    signal (S20) is visible to ops;
 *  - the creator-claim reward likewise pays only for VERIFIED passports (a
 *    self-dealing claimed sockpuppet passport earns nothing);
 *  - every automated grant carries a deterministic UNIQUE ref, so races and
 *    retries can never double-grant;
 *  - per-referrer daily grant cap enforced with an atomic guarded increment
 *    (defers past-cap grants, never loses them — the ungranted flag stays 0
 *    and a later qualifying redemption retries), PLUS a lifetime per-referrer
 *    cap counted from the `referrals` rows, which survive a referee's erasure
 *    as pseudonymised rows precisely so the ceiling cannot be reset by asking
 *    to be forgotten (V14 finding #1 — see MAX_FAN_GRANTS_PER_REFERRER_LIFETIME);
 *  - spends are negative entries guarded by a balance subquery in the same
 *    INSERT statement (can't overdraw via races).
 */

async function requireAccount(db: DbLike, accountId: string): Promise<void> {
  const a = await db.prepare("SELECT 1 AS x FROM accounts WHERE account_id = ?").bind(accountId).first();
  if (!a) throw new RewardError("no such account", 401);

/** 10 base32 chars = 50 bits — unguessable enough that codes can be public
 * (a code only ever CREDITS its owner; knowing one grants nothing). */
export function mintReferralCode(rand: (n: number) => Uint8Array = (n) => crypto.getRandomValues(new Uint8Array(n))): string {
  const bytes = rand(7); // 56 bits → 10 chars (50 bits used) + remainder dropped
  let bits = 0;
  let val = 0;
  let out = "";
  for (const b of bytes) {
    val = (val << 8) | b;
    bits += 8;
    while (bits >= 5 && out.length < 10) {
      bits -= 5;
      out += B32[(val >>> bits) & 31];
    }
  }
  return out;
}

/** Get-or-create the caller's referral code (idempotent). */
export async function referralCode(db: DbLike, accountId: string, mint: () => string = mintReferralCode, now: () => string = () => new Date().toISOString()): Promise<string> {
  await requireAccount(db, accountId);
  const existing = await db.prepare("SELECT code FROM referral_codes WHERE account_id = ?").bind(accountId).first<{ code: string }>();
  if (existing) return existing.code;
  const code = mint();
  if (!CODE_RE.test(code)) throw new RewardError("bad code mint");
  try {
    await db.prepare("INSERT INTO referral_codes (account_id, code, created_at) VALUES (?, ?, ?)").bind(accountId, code, now()).run();
  } catch {
    // Lost a race (own account row, or an astronomically unlikely code
    // collision) — return whatever won.
    const raced = await db.prepare("SELECT code FROM referral_codes WHERE account_id = ?").bind(accountId).first<{ code: string }>();
    if (raced) return raced.code;
    throw new RewardError("could not mint code", 409);
  }
  return code;
}

/**
 * Link a NEW account to a referrer's code — called only from device
 * registration, in the same breath the account is born. Invalid/own codes are
 * IGNORED (registration must never fail because of a bad referral code).
 */
export async function linkReferral(db: DbLike, refereeAccount: string, code: string, now: () => string = () => new Date().toISOString()): Promise<boolean> {
  if (typeof code !== "string" || !CODE_RE.test(code.toUpperCase())) return false;
  const owner = await db.prepare("SELECT account_id FROM referral_codes WHERE code = ?").bind(code.toUpperCase()).first<{ account_id: string }>();
  if (!owner || owner.account_id === refereeAccount) return false;
  try {
    await db
      .prepare("INSERT INTO referrals (referee_account, referrer_account, code, created_at) VALUES (?, ?, ?, ?)")
      .bind(refereeAccount, owner.account_id, code.toUpperCase(), now())
      .run();
    return true;
  } catch {
    return false; // already linked (PK) — first link wins, never rewritten
  }
}

/**
 * Grant the fan referral rewards if due: called after every successful
 * redemption by `refereeAccount`, with the redeemed RELEASE as context.
 * Pays only when that release belongs to a VERIFIED creator — the structural
 * anti-farming gate (a claimed sockpuppet creator's free credentials earn
 * nothing; see module doc). Idempotent three ways — the granted flag is
 * flipped with a guarded UPDATE (which re-asks the verified-creator predicate
 * and the lifetime cap IN the write — BF-20), the credit entries carry UNIQUE
 * refs, and the whole grant is one atomic batch. The per-referrer daily cap is consumed
 * with an ATOMIC guarded increment before granting; past the cap the grant is
 * DEFERRED (flag stays 0; a later qualifying redemption retries), never lost.
 */
export async function grantFanReferralIfDue(
  db: DbLike,
  refereeAccount: string,
  releaseId: string,
  mintEntryId: () => string,
  nowMs: number = Date.now(),
  now: () => string = () => new Date().toISOString(),
): Promise<boolean> {
  const ref = await db
    .prepare("SELECT referrer_account FROM referrals WHERE referee_account = ? AND fan_reward_granted = 0")
    .bind(refereeAccount)
    .first<{ referrer_account: string }>();
  if (!ref) return false;

  // F18 / #32 — VALUE DOES NOT FLOW INTO A CLOSED ACCOUNT, at THIS door too.
  // `creditCreatorReferral` below carries this predicate twice (credits.ts:81,
  // called at :334 and folded into its own batch); this function carried it
  // nowhere, so a suspension or a termination landing on the REFERRER still
  // paid them. Same fast-path/enforcing-copy split the caps already use: this
  // read is the fast path, the guarded UPDATE below is the gate.
  // I21-Q73 (D-1030 (a): no rewards while suspended, no backfill on lift): a frozen referrer's referral is DROPPED, not
  // deferred — `dropWhileFrozen` marks it settled with no credit to either side, so a redemption after a lift pays nothing.
  const dropWhileFrozen = db
    .prepare(`UPDATE referrals SET fan_reward_granted = 1 WHERE referee_account = ? AND fan_reward_granted = 0 AND NOT (${NOT_FROZEN_GRANT_SQL})`)
    .bind(refereeAccount, ref.referrer_account);
  if (await accountFrozenForCredit(db, ref.referrer_account)) {
    await dropWhileFrozen.run();
    return false;
  }

  // The redeemed release must belong to a VERIFIED creator (operator-vetted), verified at `nowMs`: a grant lapsed by then
  // (standing.ts grantLiveAtSql) is not, before the expiry sweep folds it back (I3-1768, D-1085(1), I34-EXPIRY-A).
  const verified = await db
    .prepare(`SELECT 1 AS x FROM releases r JOIN passports p ON p.passport_id = r.passport_id WHERE r.release_id = ? AND p.claim_state = 'verified' AND p.retracted = 0 AND ${grantLiveAtSql("p.passport_id")}`)
    .bind(releaseId, new Date(nowMs).toISOString())
    .first();
  if (!verified) return false; // deferred — a later verified-creator redemption qualifies

  // LIFETIME cap, checked before the daily one so a referrer at the ceiling
  // does not burn a daily slot discovering it. Counted from the `referrals`
  // rows themselves rather than from a counter, because those rows now SURVIVE
  // the referee's erasure (pseudonymised, account_data.ts) — which is the whole
  // point: a cap a deletion request can reset is not a cap. Deferred, not
  // errored: the caller is a redemption, and a redemption must never fail
  // because somebody else's budget is spent.
  //
  // THIS READ IS A FAST PATH, NOT THE CAP. It is check-then-act — N referees of
  // the SAME referrer redeeming concurrently all read the same count, all pass,
  // and all grant, so the ceiling is overshot by the width of the burst. The
  // enforcing copy of the predicate is on the guarded flag flip in the batch
  // below, where the read and the write are one statement. Keeping this one
  // costs a statement and buys the property the comment above claims: a referrer
  // already at the ceiling does not consume a daily rate-limit slot to find out.
  const lifetime = await db
    .prepare("SELECT COUNT(*) AS n FROM referrals WHERE referrer_account = ? AND fan_reward_granted = 1")
    .bind(ref.referrer_account)
    .first<{ n: number }>();
  if ((lifetime?.n ?? 0) >= MAX_FAN_GRANTS_PER_REFERRER_LIFETIME) return false;

  // Per-referrer daily cap consumed as one atomic windowed counter (reset +
  // increment in a single statement — no check-then-act boundary race). A
  // consumed slot on a later batch failure is harmless (the grant retries
  // tomorrow). Past the cap the grant is deferred, not lost.
  if (!(await consumeRateLimit(db, `refgrant:${ref.referrer_account}`, nowMs, MAX_FAN_GRANTS_PER_REFERRER_DAY, DAY_MS))) {
    return false;
  }

  const ts = now();
  try {
    await db.batch([
      // THE CAP IS A PREDICATE ON THE FLIP, not a number somebody read a moment
      // ago. The `SELECT COUNT(*)` above is check-then-act: two referees of the
      // same referrer redeeming at the same instant both read `n = 9` against a
      // ceiling of 10 and both grant, so the lifetime cap — the anti-farming
      // bound that survives account deletion — is exceeded by the width of the
      // concurrent burst. Folding it into the guarded UPDATE makes the count and
      // the write ONE statement, which SQLite/D1 serialises: the loser's UPDATE
      // matches no row.
      //
      // …AND SO IS ELIGIBILITY (BF-20, QUEUE #136). The verified-creator read
      // above is a FAIL-FAST, not the gate: it was the only place the
      // anti-farming predicate lived, so a retraction or a downgrade landing
      // between it and this batch still paid both sides off a passport that was
      // no longer verified. `creditCreatorReferral` closed exactly this shape by
      // re-asking `claim_state = 'verified' AND retracted = 0` AS PART OF the
      // write; this is the same predicate, on the same join the read above uses,
      // in the one statement whose outcome the credits below are guarded on.
      db
        .prepare(
          `UPDATE referrals SET fan_reward_granted = 1
            WHERE referee_account = ? AND fan_reward_granted = 0
              AND (SELECT COUNT(*) FROM referrals r2 WHERE r2.referrer_account = ? AND r2.fan_reward_granted = 1) < ?
              AND EXISTS (SELECT 1 FROM releases r JOIN passports p ON p.passport_id = r.passport_id
                           WHERE r.release_id = ? AND p.claim_state = 'verified' AND p.retracted = 0 AND ${grantLiveAtSql("p.passport_id")})
              AND NOT EXISTS (SELECT 1 FROM account_closures WHERE account_id = ? AND state IN ${FROZEN_CLOSURE_STATES_SQL})
              AND EXISTS (SELECT 1 FROM accounts WHERE account_id = ?)`,
        )
        // I34-EXPIRY-A: the grant-live term re-asked in the flip, at the write's own instant `ts` (the pre-check alone is a TOCTOU).
        .bind(refereeAccount, ref.referrer_account, MAX_FAN_GRANTS_PER_REFERRER_LIFETIME, releaseId, ts, ref.referrer_account, ref.referrer_account),
      // …AND THE CREDITS ARE GUARDED ON THE FLIP HAVING HAPPENED. Batch
      // statements run in order inside one transaction, so by this point the row
      // is `1` exactly when the UPDATE above landed. Without the guard, a flip
      // refused by the cap would still have paid out — the classic shape where
      // the check is on one statement and the money is on the next.
      //
      // The predicate deliberately reads the FLAG rather than re-asking the cap:
      // the UPDATE has already incremented the count the cap counts, so
      // re-evaluating `< MAX` here would be off by exactly one and would refuse
      // the last legitimate grant. (That is the same in-batch ordering trap the
      // recovery audit line fell into; see recovery.ts.)
      //
      // A racer that flipped the flag BEFORE this batch also satisfies the
      // predicate — and then the UNIQUE `ref` below collides, the batch throws,
      // and everything rolls back. Double-grant stays impossible on both axes.
      db
        .prepare(
          `INSERT INTO credit_ledger (entry_id, account_id, kind, origin, credits, ref, idem, reason, at)
           SELECT ?, ?, 'grant', 'referral', ?, ?, ?, 'referral_fan', ? WHERE EXISTS (SELECT 1 FROM referrals WHERE referee_account = ? AND fan_reward_granted = 1)
              AND ${NOT_FROZEN_GRANT_SQL}`,
        )
        .bind(mintEntryId(), ref.referrer_account, REFERRAL_FAN_REWARD, `fanref:${refereeAccount}`, `fanref:${refereeAccount}`, ts, refereeAccount, ref.referrer_account),
      db
        .prepare(
          `INSERT INTO credit_ledger (entry_id, account_id, kind, origin, credits, ref, idem, reason, at)
           SELECT ?, ?, 'grant', 'referral', ?, ?, ?, 'referral_welcome', ? WHERE EXISTS (SELECT 1 FROM referrals WHERE referee_account = ? AND fan_reward_granted = 1)
              AND ${NOT_FROZEN_GRANT_SQL}`,
        )
        // D-0759(a): the REFEREE's welcome carried no closure predicate at all
        // (the flip above guards the referrer only). A terminated referee gets
        // nothing; the referrer's reward is unaffected.
        .bind(mintEntryId(), refereeAccount, REFERRAL_FAN_WELCOME, `fanwelcome:${refereeAccount}`, `fanwelcome:${refereeAccount}`, ts, refereeAccount, refereeAccount),
      // I21-Q73: LAST, so it reads the flip above — a freeze landing after the pre-read drops the referral too.
      dropWhileFrozen,
    ]);
  } catch {
    return false; // raced — the UNIQUE ref held; nothing granted twice
  }
  // DID WE GRANT? A D1 batch carries no portable `changes`, so the outcome is
  // read back — the same thing `recovery.ts` does after its guarded enqueue, and
  // the shape `creditCreatorReferral` reads its own guard back with. The
  // batch did not throw, so no racer landed a grant for this referee; the flag
  // is therefore 1 exactly when OUR flip passed BOTH guards (the cap and the
  // live verified-creator predicate). A `false` here is
  // DEFERRED, not lost: the caller's contract is that a later qualifying
  // redemption retries, and the daily slot this consumed comes back tomorrow.
  // I21-Q73: the flag is also 1 on a DROP, so "did we grant" is the referrer's own credit row (its idem is unique).
  const settled = await db.prepare("SELECT 1 AS x FROM credit_ledger WHERE idem = ?").bind(`fanref:${refereeAccount}`).first();
  return settled !== null;
}

  // code, and their account is never otherwise consulted. So the freeze — which

  // review used it to put invented credits back into an account that had just

  // (verified/not-retracted, referrer not frozen, caps not exhausted) are
  // fail-fast, NOT the gate — a revoke or freeze landing in the gap between
  // them and this batch must not still pay out. Both statements re-check the
  // live predicate AS PART OF the write, the same pattern buyCard uses for its
  // in-batch cap and grantFanReferralIfDue uses for its guarded flip: the
  // check and the write are ONE statement so nothing can land between them.
  // The lifetime/daily caps above stay pre-batch fast paths only — the PK on
  // `creator_referrals` already makes a re-grant of the SAME passport
  // impossible, which is the only race the caps themselves needed to survive.
  try {
    await db.batch([
      db
        .prepare(

        )

        // I34-EXPIRY-A: `ts` after the second passportId is the grant-live term's instant, the write's own clock.
        .bind(passportId, owner.account_id, code.toUpperCase(), ts, passportId, ts, owner.account_id, owner.account_id, owner.account_id, MAX_CREATOR_CLAIM_GRANTS_PER_REFERRER_LIFETIME, passportId, owner.account_id, owner.account_id, owner.account_id, owner.account_id, passportId),

        .prepare(

              AND ${NOT_FROZEN_GRANT_SQL}`,
        )
        .bind(mintEntryId(), owner.account_id, REFERRAL_CREATOR_CLAIM_REWARD, `claimref:${passportId}`, `claimref:${passportId}`, ts, passportId, owner.account_id, owner.account_id),
    ]);
  } catch (e) {
    // I3-0765 shape (I15-W3S): only the PK/idem collision is "already credited";
    // any other D1 failure (an outage) propagates to the door's 500, never a false 409.
    if (e instanceof Error && ALREADY_CREDITED_RE.test(e.message)) throw new RewardError("already credited for this passport", 409);
    throw e;
  }
  // DID THE GUARD PASS? Same "read it back" shape grantFanReferralIfDue uses
  // (a D1 batch carries no portable `changes`): the batch did not throw, so no
  // racer collided on the PK — but the WHERE EXISTS above may still have made
  // both inserts no-ops if eligibility flipped between the reads and here.
  const settled = await db.prepare("SELECT 1 AS x FROM creator_referrals WHERE passport_id = ? AND referrer_account = ?").bind(passportId, owner.account_id).first();
  if (!settled) {
    // I5-Q2: a link that formed after the pre-read refused the insert; answered as the pre-read answers it.
    if (await linkedPair(db, passportId, owner.account_id)) throw new RewardError("unknown code", 404);
    const after = await db.prepare("SELECT COUNT(*) AS n FROM creator_referrals WHERE referrer_account = ?").bind(owner.account_id).first<{ n: number }>();
    if ((after?.n ?? 0) >= MAX_CREATOR_CLAIM_GRANTS_PER_REFERRER_LIFETIME) throw new RewardError("referral cap reached", 429);
    throw new RewardError("referral credit unlocks at verified status", 403);
  }
}

/** I3-1678: the self-dealing / 2-ring pre-read (`creditCreatorReferral`); the guarded insert re-asks the same terms. */
async function linkedPair(db: DbLike, passportId: string, account: string): Promise<boolean> {
  const sql = `SELECT 1 AS x WHERE ${linkedSql("?", "?")} OR EXISTS (SELECT 1 FROM creator_referrals r WHERE ${linkedSql("r.passport_id", "?")} AND ${linkedSql("?", "r.referrer_account")})`;
  return (await db.prepare(sql).bind(passportId, account, account, account, account, passportId).first()) !== null;
}

/** I3-1678: is passport `psp` email-linked to fan account `acct`? A verified
 * creator contact address equal (case-insensitively) to the account's verified
 * fan address or its provider-asserted email. Both arguments are SQL
 * expressions (a bind slot or a column). */
function linkedSql(psp: string, acct: string): string {
  return `EXISTS (SELECT 1 FROM creator_contacts cc WHERE cc.passport_id = ${psp} AND cc.email_verified = 1 AND cc.email IS NOT NULL
    AND LOWER(cc.email) IN (SELECT LOWER(fe.address) FROM fan_email fe WHERE fe.account_id = ${acct} AND fe.verified_at IS NOT NULL
                            UNION SELECT LOWER(a.email) FROM accounts a WHERE a.account_id = ${acct} AND a.email IS NOT NULL))`;
}

export interface CreditSummary {
  /** The ONE balance (0058): the same number `#/wallet` shows. */
  balance: number;
  /** I3-0441: a `purchased` entry also carries its purchase's `ref`,
   * `gross_cents`, `rail` and `status` (credit_purchases), so the ledger view
   * can show what the fan paid; other entries carry none of the four. */
  entries: Array<{ delta: number; reason: string; origin: string; created_at: string; ref?: string; gross_cents?: number; rail?: string; status?: string }>;
  /** What the balance is made of, by origin class. */
  by_origin: Record<string, number>;
  /** What can go towards a tip or a store purchase — the cash-backed part
   * (credits.ts CASH_ORIGINS); the rest is promotional value for Dhwaja-side
   * spends only. */
  cash_backed: number;
}

/** The caller's balance + recent entries (owner-scoped). Reads the journal —
 * since 0058 there is no second ledger, so this and `GET /account/war/credits`
 * cannot disagree. `delta`/`created_at` keep S24's wire names for the clients. */
export async function creditSummary(db: DbLike, accountId: string): Promise<CreditSummary> {
  await requireAccount(db, accountId);
  const bal = await db.prepare("SELECT COALESCE(SUM(credits), 0) AS b FROM credit_ledger WHERE account_id = ?").bind(accountId).first<{ b: number }>();
  // I3-0441: the purchase row joins on (purchase_id, account_id) and only for
  // origin 'purchased' — a ref never reaches another account's purchase.
  const { results: rows } = await db
    .prepare(
      `SELECT cl.credits AS delta, cl.reason, cl.origin, cl.at AS created_at,
              CASE WHEN cl.origin = 'purchased' THEN cl.ref END AS ref,
              cp.gross_cents, cp.rail, cp.status
         FROM credit_ledger cl
         LEFT JOIN credit_purchases cp ON cp.purchase_id = cl.ref AND cl.origin = 'purchased' AND cp.account_id = cl.account_id
        WHERE cl.account_id = ? ORDER BY cl.at DESC, cl.entry_id DESC LIMIT 100`,
    )
    .bind(accountId)
    .all<{ delta: number; reason: string; origin: string; created_at: string; ref: string | null; gross_cents: number | null; rail: string | null; status: string | null }>();
  const results: CreditSummary["entries"] = rows.map(({ ref, gross_cents, rail, status, ...e }) => ({
    ...e,
    ...(ref !== null ? { ref } : {}),
    ...(gross_cents !== null ? { gross_cents } : {}),
    ...(rail !== null ? { rail } : {}),
    ...(status !== null ? { status } : {}),
  }));
  // Per-origin balances come from credits.ts (one query, one owner) — NOT
  // spelled here: founder.test.ts reads every GROUP BY in this module as a
  // ranking board, and an owner-scoped fold is not one.
  return { balance: bal?.b ?? 0, entries: results, by_origin: await balanceByOrigin(db, accountId), cash_backed: await cashBackedBalance(db, accountId) };
}

/**
 * Spend credits on a DHWAJA-SIDE thing (the seam future features draw on: AI
 * cards, cosmetics, Premium). Origin `platform_spend`: no creator is paid, so
 * this is the one spend promotional / referral value MAY fund — it is guarded
 * on the total balance only (credits.ts CASH_ORIGINS explains the split). The
 * balance guard lives INSIDE the insert — a concurrent spend can't overdraw.
 * Reasons are a fixed vocabulary, never client free text.
 */
export async function spendCredits(
  db: DbLike,
  accountId: string,
  amount: number,
  reason: "ai_card" | "cosmetic" | "premium",
  mintEntryId: () => string,
  now: () => string = () => new Date().toISOString(),
): Promise<void> {
  await requireAccount(db, accountId);
  if (!Number.isInteger(amount) || amount < 1 || amount > 1_000_000) throw new RewardError("bad amount");
  const res = await platformSpendStmt(db, accountId, amount, reason, mintEntryId(), now()).run();
  if (res.changes === 0) throw new RewardError("insufficient credits", 409);
}

/** `spendCredits`' one statement. `nonCashOnly` ANDs credits.ts NON_CASH_FLOOR_SQL into the SAME WHERE (I30-PREMIUM). */
function platformSpendStmt(db: DbLike, accountId: string, amount: number, reason: "ai_card" | "cosmetic" | "premium", entryId: string, ts: string, nonCashOnly = false) {
  return db
    .prepare(
      `INSERT INTO credit_ledger (entry_id, account_id, kind, origin, credits, ref, idem, reason, at)
       SELECT ?, ?, 'spend', 'platform_spend', ?, ?, NULL, ?, ?
        WHERE (SELECT COALESCE(SUM(credits), 0) FROM credit_ledger WHERE account_id = ?) >= ?${nonCashOnly ? ` AND ${NON_CASH_FLOOR_SQL}` : ""}`,
    )
    .bind(entryId, accountId, -amount, reason, reason, ts, accountId, amount, ...(nonCashOnly ? [accountId, amount] : []));
}

/** I16-S4-I3-0758 (D-1030; I30-PREMIUM) — the price of a promotional Premium week, in credits, and its length in days.
 * Settings (ops_settings, the two-person limits door console.ts OPS_LIMIT_SETTINGS); a stored value outside [MIN, MAX]
 * reads as the default, so the door refuses it. Defaults are default_pending_jason. */
export const PREMIUM_WEEK_PROMO_CREDITS_KEY = "premium_week_promo_credits";
export const PREMIUM_WEEK_PROMO_CREDITS_DEFAULT = 100;
export const PREMIUM_WEEK_PROMO_CREDITS_MAX = 1_000_000; // spendCredits' own amount ceiling
export const PREMIUM_PROMO_WEEK_DAYS_KEY = "premium_promo_week_days";
export const PREMIUM_PROMO_WEEK_DAYS_DEFAULT = 7;
export const PREMIUM_PROMO_WEEK_DAYS_MAX = 366;
async function wholeSetting(db: DbLike, key: string, max: number, def: number): Promise<number> {
  const v = await getSetting(db, key);
  const n = v !== null && /^[0-9]{1,9}$/.test(v.trim()) ? Number(v.trim()) : Number.NaN;
  return Number.isInteger(n) && n >= 1 && n <= max ? n : def;
}
export const premiumWeekPromoCredits = (db: DbLike): Promise<number> => wholeSetting(db, PREMIUM_WEEK_PROMO_CREDITS_KEY, PREMIUM_WEEK_PROMO_CREDITS_MAX, PREMIUM_WEEK_PROMO_CREDITS_DEFAULT);
export const premiumPromoWeekDays = (db: DbLike): Promise<number> => wholeSetting(db, PREMIUM_PROMO_WEEK_DAYS_KEY, PREMIUM_PROMO_WEEK_DAYS_MAX, PREMIUM_PROMO_WEEK_DAYS_DEFAULT);
/** The refusal a fan reads when their promotional credit is short (approved, sheet-2 W66, D-1365). */
export const PROMO_WEEK_SHORT_REFUSAL = "You don't have enough promotional credits for a week of Premium. Credits you paid for can't be spent on this.";

/** What the upgrade page offers: the price, the length, and whether the fan's non-cash value covers it. */
export async function promoWeekOffer(db: DbLike, accountId: string): Promise<{ cost: number; days: number; eligible: boolean }> {
  const cost = await premiumWeekPromoCredits(db);
  const days = await premiumPromoWeekDays(db);
  // The same NON_CASH_FLOOR_SQL the spend carries, so the control shows exactly when the spend would land (bar a race).
  const row = await db.prepare(`SELECT 1 AS x WHERE ${NON_CASH_FLOOR_SQL}`).bind(accountId, cost).first<{ x: number }>();
  return { cost, days, eligible: row !== null };
}

/**
 * I16-S4-I3-0758: spend promotional credits on a week of Premium. ONE batch: `spendCredits`' statement (reason
 * 'premium', origin platform_spend — no creator is paid) with the non-cash floor in its WHERE, then the
 * `fan_promo_premium` upsert gated on that ledger row having landed. Consecutive weeks extend from the later of now and
 * the current end. A short balance lands neither row and throws PROMO_WEEK_SHORT_REFUSAL (409).
 */
export async function buyPromoPremiumWeek(db: DbLike, accountId: string, mintEntryId: () => string, nowMs: number): Promise<{ spent: number; days: number; until_ms: number }> {
  await requireAccount(db, accountId);
  const cost = await premiumWeekPromoCredits(db);
  const days = await premiumPromoWeekDays(db);
  const span = days * 86_400_000;
  const entryId = mintEntryId();
  const ts = new Date(nowMs).toISOString();
  await db.batch([
    platformSpendStmt(db, accountId, cost, "premium", entryId, ts, true),
    db
      .prepare(
        `INSERT INTO fan_promo_premium (account_id, until_ms, updated_at)
         SELECT ?, ? + ?, ? WHERE EXISTS (SELECT 1 FROM credit_ledger WHERE entry_id = ?)
         ON CONFLICT(account_id) DO UPDATE SET until_ms = MAX(fan_promo_premium.until_ms, ?) + ?, updated_at = excluded.updated_at`,
      )
      .bind(accountId, nowMs, span, ts, entryId, nowMs, span),
  ]);
  const row = await db
    .prepare("SELECT p.until_ms FROM fan_promo_premium p WHERE p.account_id = ? AND EXISTS (SELECT 1 FROM credit_ledger WHERE entry_id = ?)")
    .bind(accountId, entryId)
    .first<{ until_ms: number }>();
  if (!row) throw new RewardError(PROMO_WEEK_SHORT_REFUSAL, 409);
  return { spent: cost, days, until_ms: row.until_ms };
}

// --- Leaderboards (cron-precomputed; public profiles only) ---

export const LEADERBOARD_LIMIT = 50;
const LEADERBOARDS_KEY = "rankings/leaderboards.json";

/**
 * AUDIT ITEM #34 — `supporter` HERE MEANS "MADE CREATOR REQUESTS", and nothing
 * else in the product means that by this word.
 *
 * The word carries seven referents across the tree. Two of them were crossed
 * right here: this board counts `creator_requests` while `scout` counts the
 * `early_supporter:<psp>` achievement, so *Early Supporter* achievements made
 * you a **Scout** and the **Supporters** board counted something else. A third
 * referent — a paying Premium fan — is the consumer brand name two screens
 * away, which is what made the collision land on trust rather than on trivia.
 *
 * THE KEY IS FROZEN, THE LABEL IS NOT. This string is a field name inside the
 * cron-precomputed R2 artifact AND inside `rankings/leaderboards-<season>.json`
 * season snapshots, which are immutable by design. Renaming it would blank the
 * board on every archived season permanently. So the wire value stays and the
 * label a human reads was corrected instead — `CAT_LABEL` in
 * `apps/web/src/views.ts` now says **"Top Requesters"**, and the achievement
 * that feeds `scout` is labelled **"Early Scout"**. If this key is ever
 * renamed, it needs an artifact rebuild plus a dual-read window for the frozen
 * seasons; it is not a find-and-replace.
 */
export type LeaderboardCategory = "collector" | "archive_collector" | "scout" | "supporter" | "referrer";

export interface LeaderboardEntry {
  handle: string;
  display_name: string;
  score: number;
  /**
   * INTERNAL, AND NEVER ON THE WIRE (S-070 SJ-12). The immutable identity the
   * entry was BUILT from. It is written into the stored artifact and stripped by
   * `withLiveConsent`, which is the only path from either artifact to any
   * caller, so no served board ever carries it.
   *
   * It exists because the artifact's only other identity is the HANDLE, and a
   * handle is not an identity: `handle_reservations` releases a vacated handle
   * after `HANDLE_RESERVATION_DAYS` (social.ts) and `deleteAccount` deletes the
   * reservation outright (account_data.ts), so anyone can later own it. With the
   * consent re-ask keyed on the handle, a DIFFERENT person taking a vacated
   * handle republished the ORIGINAL fan's frozen entry — their display name and
   * their score — now attached to the new owner. A frozen season snapshot is
   * written once and never rewritten, so that attribution was permanent.
   *
   * Optional so an artifact written before this change still parses. An entry
   * WITHOUT it is not publishable (`withLiveConsent`): its original identity
   * cannot be established, and the fail-closed direction for an identity we
   * cannot confirm is to stop naming the person.
   */
  account_id?: string;
}

export interface Leaderboards {
  generated_at: string | null;
  season: string; // e.g. "2026-Q3"
  categories: Record<LeaderboardCategory, { all_time: LeaderboardEntry[]; season: LeaderboardEntry[] }>;
}

interface RankingStore {
  put(key: string, value: string): Promise<void>;
  get(key: string): Promise<string | null>;
  /** AC-11 (S-039 REWARDS_TRU) — real create-if-absent, not get-then-put. The
   * concrete store (route-helpers.ts r2Store) already implements this as one
   * conditional R2 write (`onlyIf: { etagDoesNotMatch: "*" }`); it was never on
   * this narrower interface, so the one call site that needs "only if absent"
   * (freezing a season snapshot, below) hand-rolled it as a read then a write —
   * two cron ticks racing the same rollover could both pass the read and both
   * write, and the LATER write wins on the one artifact this module calls
   * immutable. Returns true iff this call's write landed. */
  putIfAbsent?(key: string, value: string): Promise<boolean>;
}

/** Season key for a UTC instant: calendar quarter. */
export function seasonKey(nowMs: number): string {
  const d = new Date(nowMs);
  return `${d.getUTCFullYear()}-Q${Math.floor(d.getUTCMonth() / 3) + 1}`;
}

/** First instant of the current season (UTC quarter start), ISO. */
export function seasonStart(nowMs: number): string {
  const d = new Date(nowMs);
  return new Date(Date.UTC(d.getUTCFullYear(), Math.floor(d.getUTCMonth() / 3) * 3, 1)).toISOString();
}

// Each category is (join table, timestamp column, score expression). All are
// GROUP BY account over indexed fan-scoped tables, run from cron only.
// THE FOUNDER IS OFF EVERY CATEGORY. The profile join is the `rankable_profiles`
// view (`0011_founder.sql`) — user_profiles minus the configured founder — so Top
// Collectors, Archive Collectors, Scouts, Supporters and Referrers all exclude
// them by construction rather than by five separate WHERE clauses, and a SIXTH
// category added here inherits it. A follower count that is really a
// registration count must never sit at #1 on a board fans read to find real
// creators, and the founder's own follows would poison the Scouts board the
// same way in reverse. `test/founder.test.ts` fails any aggregate in this
// module that reads a base identity table instead; founder.ts holds the list.
//
// Supporter deliberately counts CREATOR REQUESTS, not saves: S22 treats the
// save list ("who you follow") as private regardless of profile visibility,
// so a save count must never surface publicly (review F4). Requests are the
// public-by-design demand signal, so counting them expands nothing.
//
// EVERY CATEGORY DECLARES WHAT A DELISTED CREATOR DOES TO ITS SCORE. Four of the
// five count rows that point at a passport, and none of them filtered
// `request_suppressions`, so a person who asked to be removed kept influencing a
// public board forever (review LOW 6, this pass). The boards carry only fan
// handles and integers — nothing identifies the suppressed creator, which is why
// this is second-order and not an oracle — but the fan's OWN surfaces
// (`/account/requests`, `/account/achievements`) already drop those rows, so
// leaving the aggregate counting them made the two answers disagree about the
// same fact. `suppressed` is the correlated-subquery predicate that removes a
// delisted creator's contribution; `null` means the category counts nothing
// creator-linked and is the DELIBERATE, written answer, not an omission. A sixth
// category cannot be added without choosing one.
const CATEGORY_SQL: Record<LeaderboardCategory, { table: string; ts: string; score: string; suppressed: string | null }> = {
  collector: { table: "card_collections", ts: "created_at", score: "COUNT(*)", suppressed: "SELECT 1 FROM request_suppressions s WHERE s.passport_id = t.passport_id" },
  archive_collector: {
    table: "acquisitions",
    ts: "acquired_at",
    score: "COUNT(DISTINCT release_id)",
    suppressed: "SELECT 1 FROM releases r JOIN request_suppressions s ON s.passport_id = r.passport_id WHERE r.release_id = t.release_id",
  },
  // The passport rides INSIDE the achievement id (`early_supporter:psp_…`), so
  // the join is on the composed string rather than on a column.
  scout: { table: "achievements", ts: "awarded_at", score: "COUNT(*)", suppressed: "SELECT 1 FROM request_suppressions s WHERE t.achievement_id = 'early_supporter:' || s.passport_id" },
  supporter: { table: "creator_requests", ts: "created_at", score: "COUNT(*)", suppressed: "SELECT 1 FROM request_suppressions s WHERE s.passport_id = t.passport_id" },
  // Referrals are fan→fan. There is no creator in the row to suppress.
  referrer: { table: "referrals", ts: "created_at", score: "COUNT(*)", suppressed: null },
};

async function categoryBoard(db: DbLike, cat: LeaderboardCategory, sinceIso: string | null): Promise<LeaderboardEntry[]> {
  const c = CATEGORY_SQL[cat];
  const acctCol = cat === "referrer" ? "referrer_account" : "account_id";
  const extra =
    cat === "scout" ? "AND t.achievement_id LIKE 'early_supporter:%'" : cat === "referrer" ? "AND t.fan_reward_granted = 1" : "";
  // BUILD-TIME, and the CREATOR suppression can only be build-time: the served
  // artifact carries no passport id, so there is nothing for a live read filter
  // to match on the way out (`/rankings/*` can do that; a leaderboard row is a
  // handle and an integer). Stated consequence: a creator's suppression takes
  // effect on the next cron sweep, and a FROZEN season snapshot keeps whatever
  // it was built with — with no creator identifier in it to redact.
  //
  // THE FAN'S OWN CONSENT IS A DIFFERENT QUESTION and is NOT build-time-only:
  // the row IS the fan's handle, so `serveLeaderboards` / `serveSeasonSnapshot`
  // re-ask it live at every read. See `publishableHandles` below.
  const suppressed = c.suppressed ? `AND NOT EXISTS (${c.suppressed})` : "";
  const since = sinceIso ? `AND t.${c.ts} >= ?` : "";
  // AUDIT #17 — QUALIFIED ACCOUNTS ONLY, and MEMBERSHIP rather than a weight.
  //
  // These five boards are not one-account-one-vote: the score counts the FAN'S
  // OWN activity, so a Sybil that splits its collecting across ten accounts
  // ranks worse, not better. The exposure here is the other half of #17 — a
  // throwaway account standing on a public board at all, which costs nothing
  // today (a public profile is a handle away, and `supporter` counts creator
  // requests, which are free). So the instrument is the qualification gate, not
  // the age ramp the discovery boards use: you have to be a real, reachable
  // person to appear, and once you are, your own count is your own count. An
  // age-weighted "14.2 cards collected" would also simply be a false statement
  // about a fan's own collection.
  //
  // ONE MORE PREDICATE ON THE JOIN CHAIN THIS STATEMENT ALREADY MAKES — no
  // extra round trip, and `qualified_fan_accounts` composes with
  // `rankable_fan_accounts`, so the founder exclusion above is untouched and
  // test/founder.test.ts's structural guard still sees `rankable_profiles`.
  // A SIXTH category added here inherits the gate for free.
  const qualified = qualifiedFanJoin(`t.${acctCol}`);
  // BOTH CONSENTS, and the second one is not decoration. `visibility` is "people
  // may see my profile"; `affiliation_visibility` (migration 0017, item #12) is
  // "creators may name me beside them". Four of these five categories count rows
  // that point at a passport, so a board seat here IS a public affiliation — and
  // `serveLeaderboards` re-checks BOTH at the read (below), so a build that
  // required only the first would write names the read has to drop anyway. The
  // read is the authoritative check because the artifact is up to six hours
  // stale and a frozen season snapshot is stale forever; this predicate keeps
  // the artifact from carrying a name it is not entitled to in the first place.
  const sql = `SELECT p.handle AS handle, p.display_name AS display_name, p.account_id AS account_id, ${c.score} AS score
                 FROM ${c.table} t
                 ${qualified}
                 JOIN rankable_profiles p ON p.account_id = t.${acctCol} AND p.visibility = 'public' AND p.affiliation_visibility = 'public'
                WHERE 1=1 ${extra} ${suppressed} ${since}
                GROUP BY t.${acctCol}
                ORDER BY score DESC, handle ASC
                LIMIT ${LEADERBOARD_LIMIT}`;
  const stmt = sinceIso ? db.prepare(sql).bind(sinceIso) : db.prepare(sql);
  const { results } = await stmt.all<LeaderboardEntry>();
  return results;
}

/** Cron: precompute every category, all-time + current season. When the
 * calendar quarter has rolled since the last artifact, the OLD artifact is
 * first frozen as an immutable season snapshot (written once, never
 * rewritten) — the S24 "seasons freeze nothing" gap, closed. */
export async function buildLeaderboards(db: DbLike, store: RankingStore, nowMs: number, now: () => string = () => new Date().toISOString()): Promise<void> {
  const currentSeason = seasonKey(nowMs);

  // Season rollover: freeze the previous season's final standings. ATOMIC
  // create-if-absent (AC-11) — two concurrent cron ticks at the exact rollover
  // instant both read the same `prevRaw` and both reach this call, but only one
  // write can land; the loser's `putIfAbsent` returns false and is a no-op,
  // instead of both racing a get-then-put and the artifact's own "immutable"
  // claim losing to whichever write happened to land last.
  const prevRaw = await store.get(LEADERBOARDS_KEY);
  if (prevRaw) {
    // TWO FAILURES, TWO ANSWERS — and they used to share one `catch` (S-070
    // SJ-11). A CORRUPT outgoing artifact is not archivable and must not wedge
    // the job: swallow it and rebuild. A STORAGE FAILURE on the archive write is
    // the opposite case, because the very next thing this function does is
    // `store.put(LEADERBOARDS_KEY, …)` — it OVERWRITES the bytes it just failed
    // to preserve. One swallowed 503 followed by a successful promotion
    // destroyed the outgoing season's standings permanently: the next run reads
    // the NEW season out of the artifact it just wrote, never attempts the
    // outgoing archive again, and the service has no reconstruction path for
    // that exact snapshot.
    //
    // So the parse keeps its swallow and the WRITE does not. An archive failure
    // now propagates, which leaves the current-board object untouched — the
    // outgoing season is still in it, still the source the next tick archives
    // from, and the rollover simply completes on a later tick. That is the
    // ordering rule this family keeps paying for: never run the destructive step
    // after a step that can fail silently.
    let prev: Leaderboards | null = null;
    try {
      prev = JSON.parse(prevRaw) as Leaderboards;
    } catch {
      prev = null; // corrupt live artifact — rebuild below without freezing garbage
    }
    if (prev?.season && prev.season !== currentSeason) {
      const snapKey = `rankings/leaderboards-${prev.season}.json`;
      // Optional so a store double built before this fix (S-039 REWARDS_TRU,
      // AC-11) still typechecks; the ONE store that runs in production
      // (route-helpers.ts r2Store) always has it, which is what closes the
      // race. A double without it keeps the old get-then-put shape, which is
      // harmless there — nothing in a unit test races itself.
      if (store.putIfAbsent) await store.putIfAbsent(snapKey, prevRaw);
      else if ((await store.get(snapKey)) === null) await store.put(snapKey, prevRaw);
    }
  }

  const cats = Object.keys(CATEGORY_SQL) as LeaderboardCategory[];
  const out: Leaderboards = {
    generated_at: now(),
    season: currentSeason,
    categories: {} as Leaderboards["categories"],
  };
  const since = seasonStart(nowMs);
  for (const cat of cats) {
    out.categories[cat] = {
      all_time: await categoryBoard(db, cat, null),
      season: await categoryBoard(db, cat, since),

  }

const SEASON_RE = /^\d{4}-Q[1-4]$/;
/** THE IDENTITY THE CONSENT RE-ASK IS KEYED ON (S-070 SJ-12), validated with
 * the same discipline the handle filter used to carry: an artifact is not a
 * contract, so nothing parsed out of one reaches an `IN (...)` list
 * unvalidated. `console.ts:110`'s shape. The old `HANDLE_RE` filter is gone
 * with the handle-keyed lookup it guarded — a handle is not an identity. */
const ACCOUNT_RE = /^acc_[0-9a-f]{20,64}$/; // console.ts:110's shape, one definition of what an account id looks like

/** Every board emptied, shape preserved — what a board with no identifiable
 * entry serves. Used on the early-return path so an artifact written before
 * S-070 SJ-12 cannot slip past the identity filter by having nothing to ask
 * about. */
function stripped(cats: Array<[LeaderboardCategory, { all_time?: LeaderboardEntry[]; season?: LeaderboardEntry[] }]>): Leaderboards["categories"] {
  const out = {} as Leaderboards["categories"];
  for (const [cat] of cats) out[cat] = { all_time: [], season: [] };
  return out;
}

/**
 * THE LIVE OPT-OUT, RE-ASKED AT EVERY READ — which of these handles may still be
 * published on a board at all.
 *
 * WHY IT EXISTS. `visibility = 'public' AND affiliation_visibility = 'public'`
 * was enforced ONLY in `categoryBoard`, i.e. only when the six-hourly cron sweep
 * happened to run. Between sweeps the artifact kept publishing the handle and
 * display name of a fan who had made their profile private — and, because the
 * profile row is DELETED by `deleteAccount` (account_data.ts) while the artifact
 * is not, of a fan who had erased their account outright. Measured: the board
 * answered identically before and after a full account deletion.
 *
 * AND THE FROZEN SEASON SNAPSHOT MADE IT PERMANENT. `rankings/leaderboards-<season>.json`
 * is written once and never rewritten (deliberately — a season's final standings
 * are a historical fact), so with no read filter a deleted fan's handle was
 * published forever, with no build that could ever remove it. Freezing a
 * SCOREBOARD is not the same as freezing a CONSENT, and this is the seam where
 * the two were being confused.
 *
 * `db` IS REQUIRED ON BOTH READERS, and that is the same rule `serveDiscovery` /
 * `serveImpersonation` / `serveTopFans` are built on: the entire opt-out
 * guarantee for this surface rides on one argument, and an OPTIONAL argument is
 * a filter one deleted call-site away from being absent, with tsc clean and the
 * whole suite green. Deleting it is now a compile error.
 *
 * Bounded and constant-shaped: at most five categories x two boards x
 * LEADERBOARD_LIMIT handles, de-duplicated, chunked under D1's parameter cap
 * (`inChunks`) exactly like `serveTopFans` — the artifact's own size is not
 * something this function is entitled to assume.
 */
async function publishableAccounts(db: DbLike, accountIds: readonly string[]): Promise<Set<string>> {
  const live = new Set<string>();
  const wanted = accountIds.filter((a) => typeof a === "string" && ACCOUNT_RE.test(a));
  for (const chunk of inChunks(wanted)) {
    const placeholders = chunk.map(() => "?").join(",");
    const { results } = await db
      .prepare(`SELECT account_id FROM rankable_profiles WHERE visibility = 'public' AND affiliation_visibility = 'public' AND account_id IN (${placeholders})`)
      .bind(...chunk)
      .all<{ account_id: string }>();
    for (const r of results) live.add(r.account_id);
  }
  return live;
}

/**
 * Drop every entry whose fan has withdrawn a consent since the artifact was
 * built. Shape-preserving: a category the artifact does not carry is not
 * invented, an extra one is not deleted, and the score/order of everything that
 * survives is untouched. A withheld fan is simply ABSENT — the board never says
 * a name was withheld, for the same reason `publicTopFans` doesn't.
 */
async function withLiveConsent(db: DbLike, boards: Leaderboards): Promise<Leaderboards> {
  const cats = Object.entries(boards.categories ?? {}) as Array<[LeaderboardCategory, { all_time?: LeaderboardEntry[]; season?: LeaderboardEntry[] }]>;
  const named: string[] = [];
  for (const [, board] of cats) {
    for (const rows of [board?.all_time, board?.season]) {
      if (Array.isArray(rows)) for (const e of rows) if (e && typeof e.account_id === "string") named.push(e.account_id);
    }
  }
  if (named.length === 0) return { ...boards, categories: stripped(cats) };
  // THE CONSENT IS ASKED OF THE ACCOUNT, NOT OF THE HANDLE (S-070 SJ-12).
  // Asking it of the handle asks whoever owns the handle TODAY, and handles are
  // reusable — so a stranger claiming a vacated handle answered the question on
  // behalf of the fan whose record the entry actually is.
  const live = await publishableAccounts(db, named);
  const keep = (rows: LeaderboardEntry[] | undefined): LeaderboardEntry[] =>
    Array.isArray(rows)
      ? rows
          // An entry with no immutable identity (an artifact written before this
          // change) is NOT published: fail closed, exactly as the finding's
          // "artifact migration must address older entries whose original
          // identity cannot be established safely" requires. The next cron sweep
          // rewrites the live board with identities; a frozen season snapshot
          // predating this change stays unpublished, which is the safe half of
          // that trade.
          .filter((e) => e && typeof e.account_id === "string" && live.has(e.account_id))
          // AND THE INTERNAL ID NEVER LEAVES. `withLiveConsent` is the only path
          // from either artifact to any caller (`serveLeaderboards`,
          // `serveSeasonSnapshot`), so stripping here is sufficient and there is
          // no second door to forget.
          .map((e) => ({ handle: e.handle, display_name: e.display_name, score: e.score }))
      : [];
  const categories = {} as Leaderboards["categories"];
  for (const [cat, board] of cats) categories[cat] = { all_time: keep(board?.all_time), season: keep(board?.season) };
  return { ...boards, categories };
}

/** Serve a frozen historical season snapshot (null = no such snapshot), with
 * every fan's consent re-asked live — see `publishableHandles`. The stored
 * object stays immutable; the RESPONSE is not, which is why the route serves it
 * with an ordinary short max-age rather than `immutable`. */
export async function serveSeasonSnapshot(store: RankingStore, db: DbLike, season: string): Promise<Leaderboards | null> {
  if (typeof season !== "string" || !SEASON_RE.test(season)) return null;
  const raw = await store.get(`rankings/leaderboards-${season}.json`);
  if (!raw) return null;
  let parsed: Leaderboards;
  try {
    parsed = JSON.parse(raw) as Leaderboards;
  } catch {
    return null;
  }
  if (!parsed || !parsed.categories) return null;
  return await withLiveConsent(db, parsed);
}

/** Serve the precomputed leaderboards (O(1) read; absent → empty shape), with
 * every fan's consent re-asked live — see `publishableHandles`. */
export async function serveLeaderboards(store: RankingStore, db: DbLike, nowMs: number): Promise<Leaderboards> {
  const empty: Leaderboards = {
    generated_at: null,
    season: seasonKey(nowMs),
    categories: {
      collector: { all_time: [], season: [] },
      archive_collector: { all_time: [], season: [] },
      scout: { all_time: [], season: [] },
      supporter: { all_time: [], season: [] },
      referrer: { all_time: [], season: [] },
    },
  };
  const raw = await store.get(LEADERBOARDS_KEY);
  if (!raw) return empty;
  let parsed: Leaderboards;
  try {
    parsed = JSON.parse(raw) as Leaderboards;
  } catch {
    return empty;
  }
  if (!parsed || !parsed.categories) return empty;
  return await withLiveConsent(db, parsed);
}

/**
 * Protection summary for ONE creator — derived from the ledger the same way
 * registry-build / rankings derive active warnings (fetch the warning rows, fold
 * in JS). Only warnings whose payload `protected_passport_id` equals THIS creator
 * count. LEAD-roi-130 (I16-W02I): the read used to walk EVERY warning event on the
 * platform per creator read; it now reads only the events of subjects that have
 * EVER published a warning pointing here, found through migration 0163's
 * expression index (`WARNING_POINTER_EXPR`, byte-identical to it), then those
 * subjects' rows by `idx_trust_events_subject` (the unary `+` keeps the planner off
 * the platform-wide `event_type` index, which it otherwise prefers without stats).
 * A subject that never pointed here contributes to neither count, so the JS fold
 * is unchanged.
 */
/** Migration 0163's indexed expression, byte-identical (unqualified `payload` resolves to the sub-select's `w`). */
const WARNING_POINTER_EXPR = "(CASE WHEN json_valid(payload) = 1 THEN json_extract(payload, '$.protected_passport_id') END)";

export async function protectionSummary(db: DbLike, passportId: string, before?: string): Promise<CreatorRoi["protection"]> {
  // Which published warnings an operator later withdrew as NOT A FINDING. One
  // small grouped read of the publication rows, keyed by the ledger event id
  // they carry, so the ledger scan below stays exactly the derive-from-log walk
  // it always was and the disposition is joined in JS rather than in SQL.
  const notAFinding = await withdrawnAsNotAFinding(db, before, passportId); // roi130 (I16-L03 hand-off): bounded to this creator
  // THE DAY BOUND (ADR-0006 §1). `before` is the day object's own day `T`, and
  // `created_at` is an ISO-8601 UTC instant, so the lexicographic compare IS
  // "strictly before the start of T": '…T23:59:59.999Z' < '2026-09-06' and
  // '2026-09-06T00:00:00.000Z' is not. The ledger is append-only and a
  // retraction is itself an event, so the fold over `created_at < T` is a
  // function of (creator, T) — which is what lets protection stay a production
  // read instead of a store cell (S-1). Absent `before` the fold is unbounded,
  // which is the shape every non-day caller still wants.
  const { results } = await db
    .prepare(
      `SELECT event_id, subject, event_type, payload, ${operatorSetPointerSql("trust_events")} AS ptr_operator FROM trust_events WHERE +event_type IN ('warning_published','warning_retracted')` +
        ` AND subject IN (SELECT w.subject FROM trust_events w WHERE w.event_type = 'warning_published' AND ${WARNING_POINTER_EXPR} = ?)` +
        (before === undefined ? "" : " AND created_at < ?") +
        " ORDER BY seq ASC",
    )
    .bind(passportId, ...(before === undefined ? [] : [before]))
    .all<{ event_id: string; subject: string; event_type: string; payload: string; ptr_operator: number }>();
  const active = new Map<string, boolean>(); // subject → latest state
  const cls = new Map<string, string>();
  const ptr = new Map<string, string | undefined>(); // subject → protected pid (impersonation)
  const impostors = new Set<string>(); // distinct impostor subjects that ever protected this pid
  let warningsTotal = 0;
  for (const r of results) {
    if (r.event_type === "warning_published") {
      let warningClass = "";
      let protectedPid: string | undefined;
      try {
        const p = JSON.parse(r.payload) as { warning_class?: unknown; protected_passport_id?: unknown };
        warningClass = String(p.warning_class);
        if (r.ptr_operator === 1 && typeof p.protected_passport_id === "string" && PSP_RE.test(p.protected_passport_id)) protectedPid = p.protected_passport_id;
      } catch {
        warningClass = "";
      }
      cls.set(r.subject, warningClass);
      ptr.set(r.subject, protectedPid);
      active.set(r.subject, true);
      if (warningClass === "impersonation" && protectedPid === passportId && !notAFinding.has(r.event_id)) {
        warningsTotal++;
        impostors.add(r.subject);
      }
    } else {
      active.set(r.subject, false); // later retraction wins (seq ASC)
    }
  }
  let activeWarnings = 0;
  for (const [subject, isActive] of active) {
    if (isActive && cls.get(subject) === "impersonation" && ptr.get(subject) === passportId) activeWarnings++;
  }
  return { protected: activeWarnings > 0, active_warnings: activeWarnings, warnings_total: warningsTotal, impostors_named: impostors.size };
}

      // two days' objects. It is a population cell now (`readDayBuildInput`), so
      // `serveDay` applies the floor AND the running-total delta rule (analytics-cells
      // step 1c) with its served state. I3-1243 b (I22): withheld serves null (the type above), never a factual 0.
      supporters_total: cells.get(DAY_CELL.supportersTotal),
      supporters_active: null, // WITHHELD (no cell) — null, never a served 0 (I4-08-R5)
      supporters_core: null, // WITHHELD (no cell)
      window_days: ACTIVE_WINDOW_DAYS,
    },
  };
}

/**
 * The error's CLASS, never its message. The message on this path is composed by
 * the object store, and its keys are `frozen/day/<passport_id>/<day>.json`, so a
 * message that names the key it failed on names the creator. The class is drawn
 * from the error TYPE, so no message — present, future, or from a binding this
 * repo does not own — can carry an identifier through it. Local and unexported:
 * the same three lines FX2 added to `analytics-build.ts:28` for the same reason,
 * and the shared home for them is `log.ts`, which is not this room's to widen.
 */
function errClass(e: unknown): string {
  return e instanceof Error ? (e.name || "Error") : typeof e;
}

export interface SnapshotStore {
  put(key: string, value: string): Promise<void>;
  get(key: string): Promise<string | null>;
}

/**
 * The build hook. It takes the passport and the reader's clock and NOTHING
 * else — no database handle crosses into this module, so `roi.ts` still cannot
 * compute a dashboard, and the one path that can (`analytics-read.ts`) decides
 * for itself whether the grace has passed.
 */
export type DayBuildHook = (passportId: string, nowMs: number) => Promise<void>;

/**
 * WHO MAY BUILD (ADR-0006 §1, O-18/C5/C6). `build: false` is written out at the
 * agency routes and at `POST /account/intelligence/explain` rather than left
 * implicit: those paths serve-or-refuse, and a reviewer should be able to see
 * that they decided to, not that somebody forgot an argument.
 */
export interface ReadOptions {
  build?: DayBuildHook | false;
}

/** The creator-private dashboards that report a FAN POPULATION and are therefore
 * served day-frozen. Named as a closed set so a fourth one cannot be added
 * without deciding.
 *
 * `back-catalog` is deliberately NOT here, and the reason this comment used to
 * give was FALSE: it said the archive-request COUNT is not a per-fan population.
 * `archive_requests` is `PRIMARY KEY (account_id, passport_id)` — one row per
 * fan — so it is exactly a per-fan population, and while that sentence stood the
 * count was served live and unfloored on three routes.
 *
 * It stays out for the REGIONS reason instead (see the note below, and
 * `requests.ts archiveDemand`): the primitive was removed at the DATA MODEL. The
 * count now spans only WHOLE UTC DAYS, so the answer for a given creator and day
 * is the same at every hour and there is no pair of reads to difference —
 * strictly stronger than freezing against a reader-chosen instant, with no
 * snapshot to be partial, nothing to sweep and no 503. Freezing it as well would
 * add a fourth artifact nothing needs. */
// `regions` (F10 / #52) is DELIBERATELY NOT HERE, and an earlier version of
// this comment argued the opposite — wrongly. It said a live region read would
// let a creator difference one fan's country out of two reads. It would not:
// `regions.ts audienceRegions` sums only COMPLETED UTC days, so the answer for
// a given creator and month is the same at every hour and there is no pair of
// reads to difference. F10 removed the primitive at the data model instead of
// timing it, which is the stronger fix. Freezing it anyway meant writing a
// per-creator country breakdown to the bucket every day, for thirty days, that
// nothing ever read — pure cost and pure liability for a feature whose whole
// claim is that we never learn who came from where. The prefix stays in the
// SWEEP list for one retention cycle to clear what was already written.
export type FrozenAnalytics = "roi" | "attribution" | "campaigns";

/**
 * DAY-FROZEN ANALYTICS — the same privacy control `intelligence.ts
 * dailyIntelligence` states in its own header, generalised to the dashboards
 * that were recomputing live.
 *
 * WHY IT IS A CONTROL AND NOT A CACHE. Live recomputation is a DIFFERENCING
 * PRIMITIVE: a caller who knows a single event is about to happen (they told one
 * named fan to save them, or handed one person a code) times two reads around it
 * and attributes the whole delta to that person. Aggregation is only privacy
 * when the aggregate cannot be differenced down to one row. The k-floor these
 * modules apply does not help — it hides a small POPULATION, not a DELTA on a
 * large one.
 *
 * ---------------------------------------------------------------------------
 * WHAT THE FIRST VERSION GOT WRONG: THE READER CHOSE THE INSTANT
 * ---------------------------------------------------------------------------
 * It computed a day's snapshot on the FIRST READ of that UTC day. A creator who
 * simply does not read therefore chose both endpoints:
 *
 *     23:59:50  GET /account/roi   → computes and freezes day D
 *     23:59:55  one known fan saves them
 *     00:00:10  GET /account/roi   → computes and freezes day D+1
 *
 * Two snapshots TWENTY SECONDS APART, and the delta is one person — exactly
 * what the paragraph above says cannot happen. The docblock claimed "a delta
 * observed across two reads aggregates a whole day of activity"; it aggregated
 * whatever interval the reader felt like leaving between two visits.
 *
 * SO NO READ CHOOSES THE INSTANT. `frozenRead` gets, parses and serves, or
 * refuses; nothing in this module computes a dashboard, which is why there is
 * no throttle here. One function writes a day object, `analytics-read.ts
 * buildDayIfMissing`, through two doors: the cron's `analytics-build.ts
 * runBuildSweep` (yesterday, for agency-granted creators) and the creator's own
 * read through `opts.build` (`analytics-read.ts dayBuilder`: today, only after
 * `BUILD_GRACE_MS`). Both stamp it `generatedAtFor(T)`, the midnight of its day.
 *
 * A MISS IS A REFUSAL, NOT A LIVE ANSWER. `FrozenNotReady` is what a creator
 * gets before their first nightly build. Falling back to a live compute would
 * hand back the primitive at precisely the moment somebody is most likely to be
 * probing for it — a first read is free to make, and there is no rate limit on
 * not-yet-existing.
 *
 * FAIL CLOSED. A store that cannot be read throws out of here rather than
 * answering live, and the routers refuse (503) when the artifact binding is
 * absent, for the same reason.
 */
export class FrozenNotReady extends Error {
  override name = "FrozenNotReady";
  constructor() {
    super("this dashboard is rebuilt once a day and has not been built yet — check back tomorrow");
  }
}

/** The key a kind/passport/day snapshot USED to live under. LEGACY: nothing
 * writes it any more — a (creator, day) is ONE object, `bundleKey` below.
 * The retention sweep and the erasure purge do not call it: they find a deployed bucket's legacy objects by key prefix (`<kind>/` and `<kind>/<psp>/`, analytics-build.ts OBJECT_PREFIXES).
 * It is kept as a test helper (D-1304(4)): the spelling the tests plant those objects under. */
export function frozenKey(kind: FrozenAnalytics, passportId: string, day: string): string {
  return `${kind}/${passportId}/${day}.json`;
}

/** The kinds a bundle is projected to, AND the legacy per-kind prefixes the
 * retention sweep and the erasure purge still clear. */
export const FROZEN_KINDS: readonly FrozenAnalytics[] = ["roi", "attribution", "campaigns"];

/**
 * THE DAY IS ONE OBJECT.
 *
 * `frozen/day/<psp>/<YYYY-MM-DD>.json` holds all four payloads of one (creator,
 * UTC day) under one `generated_at`, written by ONE conditional put. Four
 * independently written keys could not give that: a fault between two of those
 * writes left a PARTIAL day, and two writers interleaving across them left two
 * readings of one fan population under one day key — differenceable in a single
 * request pair, which is the primitive this whole feature exists to remove
 * (RUN-0003 residuals (e)/(g), FINDING-0020). One key is one instant by
 * construction, in every interleaving, with no protocol to get right.
 *
 * `<psp>` BEFORE `<day>`, so `purgeFrozenAnalytics`'s `<prefix>/<psp>/` erasure
 * loop reaches it and `sweepFrozenSnapshots`'s trailing-day regex still matches
 * it. The cost is that a sweep cannot LIST today's bundles — see the eligibility
 * step in `analytics-build.ts` for what that costs per sweep.
 */
export function bundleKey(passportId: string, day: string): string {
  return `frozen/day/${passportId}/${day}.json`;
}

/** What a bundle can be projected to. `intel` is in here and not in
 * `FrozenAnalytics` because `dailyIntelligence` reads the same object: one
 * reader, so reader and builder cannot drift apart over four kinds again.
 *
 * `summary` is the FIFTH kind (S-002 S-4, ADR-0006 §4/§7): the Phase-1 metrics
 * floored by `serveDay`, which `GET /account/intelligence/summary` (B11) serves
 * and `projectForBrief` reads. It joins the "all kinds or none" rule below with
 * the other four — a bundle that carries four of five is CORRUPT and skipped,
 * for the same reason a bundle missing `intel` was (FINDING-0020). */
export type FrozenBundleKind = FrozenAnalytics | "intel" | "summary";

/** EVERY KIND A BUNDLE MUST CARRY, in one place. `readFrozenBundle` serves a day
 * only if all four are present, and `analytics-build.ts` checks the same list
 * against the object it is about to write, so "what a whole day is" is one fact
 * rather than a builder's shape and a reader's expectation of it. */
export const FROZEN_BUNDLE_KINDS: readonly FrozenBundleKind[] = [...FROZEN_KINDS, "intel", "summary"];

/** THE SAME PREDICATE AT BOTH DOORS (FAMILY-004 P1): the reader
 * (`readFrozenBundle` below) and the builder's skip-read
 * (`analytics-read.ts buildDayIfMissing`) must agree on what "whole" means, or
 * one can treat as built what the other would refuse to serve. A bundle is
 * whole only if it names the (creator, day) it claims to be AND carries a
 * generated_at AND every kind is a non-null object — checking `typeof body[k]
 * === "object"` alone (the old check) is satisfied by `{}` and `[]`, which is
 * FINDING-0023: the day was never asked whether it is FOR this creator and
 * this day, only whether its five slots are shaped like objects. */
export function isWholeBundle(body: Record<string, unknown>, passportId: string, day: string): boolean {
  if (typeof body.passport_id !== "string" || body.passport_id !== passportId) return false;
  if (typeof body.day !== "string" || body.day !== day) return false;
  if (typeof body.generated_at !== "string" || body.generated_at.length === 0) return false;
  return FROZEN_BUNDLE_KINDS.every((k) => isWholeKind(body[k], passportId));
}

/** I3-1959: one kind of a whole day — a non-array object that names the
 * bundle's own passport (every kind type carries `passport_id`). `{}`, `[]` and
 * another creator's object all fail. The builder's write-side assertion
 * (`analytics-read.ts`) should call this too. */
export function isWholeKind(v: unknown, passportId: string): boolean {
  return typeof v === "object" && v !== null && !Array.isArray(v) && (v as { passport_id?: unknown }).passport_id === passportId;
}

/**
 * How far back a read will look for a snapshot.
 *
 * Two days, not one: the builder runs on the artifact sweep, and a sweep that
 * fails or is late must not blank every dashboard in the product. Beyond that
 * the answer is `FrozenNotReady` rather than a stale number a creator would
 * read as current.
 */
export const FROZEN_LOOKBACK_DAYS = 2;

/**
 * Serve the newest BUNDLE for this passport and project one kind out of it.
 * NEVER COMPUTES.
 *
 * Walks back from today so a creator reading before the day's build still gets
 * yesterday's numbers rather than a refusal — and the instant they are reading
 * is one the schedule picked either way. THE WHOLE BUNDLE MOVES BACK TOGETHER,
 * so two kinds read for one (creator, day) request pair are two projections of
 * ONE instant and can never come from two different days (FINDING-0020).
 */
export async function readFrozenBundle<T>(store: SnapshotStore, kind: FrozenBundleKind, passportId: string, nowMs: number, opts?: ReadOptions): Promise<T> {
  if (!PSP_RE.test(passportId)) throw new IntelError("bad passport id");
  // BUILD-ON-READ, AND ONLY FOR THE CREATOR'S OWN READ (ADR-0006 §1, O-18).
  // The hook is `analytics-read.ts dayBuilder`, which returns immediately
  // before the day's build grace and does a guarded skip-read before it
  // computes; agency routes and `explain` pass `{ build: false }` and reach the
  // loop below unchanged. A build that THROWS must not blank a dashboard the
  // lookback can still answer, so it is logged and the loop runs anyway — which
  // is also what happens for every read before the grace.
  if (opts?.build) {
    try {
      await opts.build(passportId, nowMs);
    } catch (e) {
      // THE SUBJECT DOES NOT GO TO THE LOG (W3-10 item 1, FX2's `errClass`
      // correction applied at this end of the same channel). `passport_id` is
      // not on `ALLOWED_KEYS`, and `e.message` is composed by the object store,
      // whose keys are `frozen/day/<passport_id>/<day>.json` by construction
      // (`bundleKey`, :434) — so only the error's CLASS travels. Erasure reaches
      // the bucket and the DB; it does not reach a platform log line.
      logError({ cid: "unknown", hop: "http", msg: "frozen_day_build_failed_on_read", err: errClass(e) });
    }
  }
  for (let back = 0; back <= FROZEN_LOOKBACK_DAYS; back++) {
    const day = new Date(nowMs - back * DAY_MS).toISOString().slice(0, 10);
    const cached = await store.get(bundleKey(passportId, day));
    if (cached === null) continue;
    try {
      const body = JSON.parse(cached) as Record<string, unknown>;
      // ALL FOUR KINDS OR NONE OF THEM. A bundle that does not carry every kind
      // is CORRUPT, and corrupt is skipped rather than recomputed — recomputing
      // here would put the instant back in the reader's hands, and a caller who
      // can write garbage into the store could use that to get one on demand.
      // Skipping only the ABSENT kind was FINDING-0020 in its second form: a
      // bundle missing `intel` served intel from D-1 beside roi from D, one fan
      // population read twice in one request pair (regression lookback-sweep
      // d2). The day is judged whole before ANY kind is served out of it, so the
      // pair moves back to D-1 together.
      if (!isWholeBundle(body, passportId, day)) {
        // FINDING-0006's own "cheapest robust follow-up": detection, not a
        // self-healing recompute (that would hand the instant back to the
        // reader). One line, so an operator can find and purge it.
        // The DAY is loggable (`ALLOWED_KEYS`, and it is not per-entity); the
        // passport id and the bundle key are not, and the key IS the passport id
        // (`frozen/day/<psp>/<day>.json`). An operator finds the object by the
        // day plus the subject the erasure record names, never by this line.
        logError({ cid: "unknown", hop: "http", msg: "frozen_bundle_not_whole_on_read", day });
        continue;
      }
      return body[kind] as T;
    } catch {
      /* an unparseable body is the same case, and takes the same day-wide skip */
    }
  }
  throw new FrozenNotReady();
}

/** `readFrozenBundle` for the three dashboard kinds. Signature unchanged: every
 * route calls this and none of them knows the layout. */
export async function frozenRead<T>(store: SnapshotStore, kind: FrozenAnalytics, passportId: string, nowMs: number, opts?: ReadOptions): Promise<T> {
  return await readFrozenBundle<T>(store, kind, passportId, nowMs, opts);
}

/** `creatorRoi`, served from the frozen snapshot — the form every ROUTE must
 * use. The bare `creatorRoi` above stays exported for the BUILDER and for tests;
 * anything that ANSWERS A REQUEST goes through here and cannot compute. */
export async function dailyRoi(store: SnapshotStore, passportId: string, nowMs: number = Date.now(), opts?: ReadOptions): Promise<CreatorRoi> {
  const roi = await frozenRead<CreatorRoi>(store, "roi", passportId, nowMs, opts);
  // D-1073(3): a day object sealed before `creatorRoi` floored `requests` carries it raw; it is floored on the way out too
  // (D-1113(1): 0 included). A sealed `null` stays `null`: kFloor's comparison is false for it.
  return { ...roi, discovery: { ...roi.discovery, requests: kFloor(roi.discovery.requests as number) } };
}
