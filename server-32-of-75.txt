-- QUEUE #737 (RGOPS-737; D-0861(10); I6-FX1-MIG). DATA BACKFILL ONLY, IDEMPOTENT: no schema change.
-- 0147 skipped every release with ANY history row, so a release published before 0135 and withdrawn after it holds
-- one 'withdrawn' row and the as-of readers (back-catalog.ts, analytics-read.ts: COALESCE(latest row before T,
-- 'registered')) read it 'registered' before that withdrawal. For a release whose EARLIEST row is later than its
-- created_at and is a 'withdrawn', insert 'published' at created_at: the only state it can have been withdrawn from
-- that the readers do not already default to. An earliest 'published' row is left alone (its prior state reads
-- 'registered' by default). Re-run inserts nothing: the earliest row is then at created_at.
INSERT INTO release_status_history (release_id, passport_id, status, at)
  SELECT r.release_id, r.passport_id, 'published', r.created_at
    FROM releases r
   WHERE (SELECT h.status FROM release_status_history h WHERE h.release_id = r.release_id ORDER BY h.at ASC LIMIT 1) = 'withdrawn'
     AND (SELECT MIN(h.at) FROM release_status_history h WHERE h.release_id = r.release_id) > r.created_at;

-- 0148 (notifications.email_claimed_at), 0151 (creator_digests.push_next_at) and 0153 (agency_seats created_at,
-- last_used_at, revoked_at) wrote the ISO-instant CHECK as ONE 77-byte GLOB pattern. Real D1 caps a LIKE/GLOB
-- pattern at 50 bytes and SQLite tests that length before it looks at the value, so on D1 the CHECK raised
-- "LIKE or GLOB pattern too complex" on EVERY insert into those tables, NULL column included. The same pattern,
-- split at its fixed-width boundaries, is three short GLOBs over substr() (42, 33 and 2 bytes): a value matches
-- 'DDDD-DD-DDTHH:MM:SS*Z' exactly when its first 10 characters match the date, the next 9 'THH:MM:SS', and the
-- rest '*Z'. NULL stays allowed where it was (every term NULL -> the CHECK passes). 0148/0151/0153 stay as written.
--
-- notifications / creator_digests: a COLUMN SWAP, not a table rebuild. Each column is its table's last column
-- and carries no index, FK, trigger or view, so DROP COLUMN is allowed and the re-added column lands in the same
-- position: the CREATE TABLE text changes only in the CHECK. A rebuild would make every other *_at column of
-- those tables "created by >= 0147" under tests/invariants/i6-1651-timestamp-check-forward.test.ts and oblige new
-- CHECKs on them. SQLite evaluates only the CHECKs that name an UPDATEd column, so the copy into the tmp column
-- never runs the old pattern.
ALTER TABLE notifications ADD COLUMN email_claimed_tmp TEXT;
UPDATE notifications SET email_claimed_tmp = email_claimed_at WHERE email_claimed_at IS NOT NULL;
ALTER TABLE notifications DROP COLUMN email_claimed_at;
ALTER TABLE notifications ADD COLUMN email_claimed_at TEXT CHECK (substr(email_claimed_at, 1, 10) GLOB '[0-9][0-9][0-9][0-9]-[0-9][0-9]-[0-9][0-9]' AND substr(email_claimed_at, 11, 9) GLOB 'T[0-9][0-9]:[0-9][0-9]:[0-9][0-9]' AND substr(email_claimed_at, 20) GLOB '*Z');
UPDATE notifications SET email_claimed_at = email_claimed_tmp WHERE email_claimed_tmp IS NOT NULL;
ALTER TABLE notifications DROP COLUMN email_claimed_tmp;

ALTER TABLE creator_digests DROP COLUMN push_next_at;
ALTER TABLE creator_digests ADD COLUMN push_next_at TEXT CHECK (substr(push_next_at, 1, 10) GLOB '[0-9][0-9][0-9][0-9]-[0-9][0-9]-[0-9][0-9]' AND substr(push_next_at, 11, 9) GLOB 'T[0-9][0-9]:[0-9][0-9]:[0-9][0-9]' AND substr(push_next_at, 20) GLOB '*Z');

-- 0160_trust_events_no_replace — I3-0145 REPLACE limb + I3-0307 shield read (I12-FX2). ADDITIVE: one trigger and
-- one index, no table touched, no row read or written.
--
-- 0159 refuses UPDATE and DELETE of the ledger. A REPLACE conflict resolution (`INSERT OR REPLACE`, `REPLACE INTO`)
-- deletes every row the new one collides with on a UNIQUE key WITHOUT firing a delete trigger while recursive triggers
-- are off (real D1 reports `recursive_triggers = 0`): one raw statement colliding on seq, prev_hash and event_hash
-- removed three ledger rows. A BEFORE INSERT trigger runs before any conflict resolution, so this one refuses an insert
-- that collides with a stored row on ANY of the four UNIQUE keys, whatever OR-clause it carries. An append collides with
-- nothing and is untouched.
--
-- The message starts with SQLite's own `UNIQUE constraint failed: trust_events.<col>` on purpose: a lost race for the
-- chain head (events.ts `isHeadRace`, which reads `trust_events.seq` / `trust_events.prev_hash` out of the message) and
-- every other uniqueness reader keep classifying the refusal exactly as they classified the constraint it pre-empts.
--
-- The index: the per-passport shield read (account-visibility.ts `shieldedAccountKeys`) joins shield events to the
-- passport's CURRENT account keys by the payload's `account_key`. Without an index on that key the read scanned every
-- shield event on the platform on each passport view and card render. The expression is byte-identical to
-- account-visibility.ts `KEY_OF("payload")`, which is what lets SQLite match it.

-- Precondition: none (no data is read; the index build reads only shield events). Re-run: fails on the first CREATE
-- (no IF NOT EXISTS), loudly. A future rebuild of trust_events drops both with the table and must recreate them.

  BEGIN

      WHERE EXISTS (SELECT 1 FROM trust_events WHERE seq = NEW.seq);
    SELECT RAISE(ABORT, 'UNIQUE constraint failed: trust_events.prev_hash - trust_events is append-only: a stored event is never replaced')
      WHERE EXISTS (SELECT 1 FROM trust_events WHERE prev_hash = NEW.prev_hash);
    SELECT RAISE(ABORT, 'UNIQUE constraint failed: trust_events.event_hash - trust_events is append-only: a stored event is never replaced')
      WHERE EXISTS (SELECT 1 FROM trust_events WHERE event_hash = NEW.event_hash);
    SELECT RAISE(ABORT, 'UNIQUE constraint failed: trust_events.event_id - trust_events is append-only: a stored event is never replaced')
      WHERE EXISTS (SELECT 1 FROM trust_events WHERE event_id = NEW.event_id);
  END;

CREATE TRIGGER creator_payouts_paid_register AFTER UPDATE OF status ON creator_payouts

  VALUES (
    NEW.payout_id, 'payout', 'stripe', NEW.stripe_transfer_id, NEW.amount_cents,

    END

END;

  WHEN OLD.extended_at IS NOT NULL AND (NEW.extended_at IS NOT OLD.extended_at OR NEW.extended_by IS NOT OLD.extended_by

  BEGIN SELECT RAISE(ABORT, 'data_subject_requests: the deadline is extended once'); END;
