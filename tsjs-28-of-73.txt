/** The METHOD steps a compiled prompt delivered: Map of SKnn -> the step numbers of the top-level `N. ` lines under its
 *  `### Required procedure` (ending at the next `### ` or `## ` heading). The section ends at the next `## <letter>` heading.
 *  A METHOD section with no readable step maps to []. The first section of a repeated SK id is the one read. */
export function promptMethodSteps(ptext) {
  const t = blankFences(withoutScopePacket(String(ptext)));
  const out = new Map();
  for (const h of t.matchAll(/^## METHOD (SK\d+)\b[^\n]*$/gm)) {
    if (out.has(h[1])) continue;
    const after = h.index + h[0].length;
    const next = /^## [A-Za-z]/m.exec(t.slice(after));
    const sec = t.slice(after, next ? after + next.index : t.length);
    const rp = /^### Required procedure[^\n]*$/m.exec(sec);
    let steps = [];
    if (rp) {
      const body = sec.slice(rp.index + rp[0].length);
      const end = /^#{2,3} /m.exec(body);
      steps = [...new Set([...(end ? body.slice(0, end.index) : body).matchAll(/^(\d+)\. /gm)].map((m) => Number(m[1])))].sort((a, b) => a - b);
    }
    out.set(h[1], steps);
  }
  return out;
}

/** A glob OWNS grant (I33-COMPILE-06). `globGrantRe` is the ONE reading of a glob grant, used by compile-room-prompt.mjs
 *  (which keeps a glob matching a file listRepoFiles lists under --root: tracked or untracked, ignored files and
 *  LAND-BACKUPS records excluded) and by the write audit below (which credits a write matching it, over the same listing):
 *  `**` any path, `*` / `?` within one segment, `[..]` a class, `{a,b}` literal alternatives. null when the token is not a
 *  usable grant: no glob character, whitespace, absolute, `..`, or no literal directory before the first glob character
 *  (`*.md` would grant a file anywhere). */
export const GLOB_META_RE = /[*?[\]{}]/;
const globEsc = (x) => x.replace(/[.*+?^${}()|[\]\\/]/g, "\\$&");
export function globGrantRe(glob) {
  const g = String(glob ?? "").trim().replace(/^\.\//, "");
  if (!GLOB_META_RE.test(g) || /\s/.test(g) || g.startsWith("/") || /(^|\/)\.\.(\/|$)/.test(g)) return null;
  if (g.lastIndexOf("/", g.search(GLOB_META_RE)) <= 0) return null;
  let re = "";
  for (let i = 0; i < g.length; i++) {
    const c = g[i];
    if (c === "*") {
      if (g[i + 1] === "*") { i++; if (g[i + 1] === "/") { i++; re += "(?:[^/]+/)*"; } else re += ".*"; } else re += "[^/]*";
    } else if (c === "?") re += "[^/]";
    else if (c === "[") {
      const j = g.indexOf("]", i + 1);
      if (j < 0) return null;
      // I34-TOOLS-A (JA-04): a class never matches `/` (a shell glob's class cannot cross a segment): `[!x]` is `[^/x]`;
      // any other class (`[^x]`, a range such as `.-0`) is guarded by `(?!/)`.
      const body = g.slice(i + 1, j).replace(/\\/g, "\\\\");
      re += body.startsWith("!") ? `[^/${body.slice(1)}]` : `(?!/)[${body}]`;
      i = j;
    } else if (c === "{") {
      const j = g.indexOf("}", i + 1);
      if (j < 0) return null;
      re += `(?:${g.slice(i + 1, j).split(",").map(globEsc).join("|")})`;
      i = j;
    } else if (c === "]" || c === "}") return null;
    else re += globEsc(c);
  }
  return new RegExp(`^${re}$`);
}
/** The glob grants a compiled PROMPT.md records: the compiler's `Scope (glob grants ...)` line, and (a prompt compiled
 *  before I33-FX1) its `Scope (dropped ...)` entries with the reason `(glob; OWNS entry ...)`. A glob the compiler
 *  dropped for matching no file listRepoFiles lists under --root (tracked or untracked, ignored files and LAND-BACKUPS
 *  records excluded) carries another reason and is never read as a grant. */
export function globGrantsInPrompt(text) {
  const out = [];
  const kept = /^Scope \(glob grants[^)\n]*\):[ \t]*(.*)$/m.exec(String(text ?? ""));
  if (kept) for (const t of kept[1].matchAll(/`([^`\n]+)`/g)) out.push(t[1]);
  const dropped = /^Scope \(dropped[^)\n]*\):[ \t]*(.*)$/m.exec(String(text ?? ""));
  if (dropped) for (const t of dropped[1].matchAll(/`([^`\n]+)` \(glob; OWNS entry /g)) out.push(t[1]);
  return [...new Set(out)].filter((g) => globGrantRe(g));
}

function requiredSections() {

  if (!existsSync(CONTRACT)) { console.error(`config error: ${CONTRACT} not found — the required sections cannot be derived (FALLBACK is ${FALLBACK.length} headings and predates ## PACKET)`); process.exit(2); }

function noFamiliesReasonFor(promptFile) {
  try {
    const dir = dirname(dirname(promptFile)); // <batch>/<room>/PROMPT.md -> <batch>
    const logPath = join(dir, "_packets.jsonl");
    if (!existsSync(logPath)) return { log: false, reason: "" };

function findPromptFile(reportDir) {
  const direct = join(reportDir, "PROMPT.md");
  if (existsSync(direct)) return direct;
  const up = join(dirname(reportDir), "PROMPT.md");
  if (existsSync(up)) return up;
  const chain = [];
  for (let d = dirname(dirname(reportDir)); ; d = dirname(d)) {
    const parent = dirname(d);
    if (parent === d) return null;
    chain.push(d);
    if (basename(dirname(parent)) === "07-findings") break;

function roomTypeFor(fileArg, fallbackDir) {

// QUEUE #310 REG-7 (S-088 REG round 1): "regression"/"red" bare directory names are the same
// attack-room fallback #303 already gives ROOM_DIR when no PROMPT.md/header names a real room
// type -- a room dispatched with neither still owes the coverage duty. FXC-10 (D-1142(1)): the set is findings-vocab.mjs

// QUEUE #310 REG-7: split on a separator only OUTSIDE {brace} groups, so "{recovery,takedown}"
// is not mistaken for a scope-item boundary.
function splitTopLevel(str, sep) {
  const out = [];
  let depth = 0, cur = "";
  for (const ch of str) {
    if (ch === "{") depth++;
    else if (ch === "}") depth = Math.max(0, depth - 1);
    if (ch === sep && depth === 0) { out.push(cur); cur = ""; }
    else cur += ch;
  }
  out.push(cur);
  return out;
}
// QUEUE #310 REG-7: "{a,b}.ts" -> ["a.ts", "b.ts"]; a scope item with no braces passes through.
function expandBraceItem(item) {
  const m = item.match(/^(.*)\{([^}]*)\}(.*)$/);
  if (!m) return [item];
  const [, pre, inner, post] = m;
  return inner.split(",").map((alt) => `${pre}${alt.trim()}${post}`);
}
// QUEUE #310 REG-7: a scope item read cell counts as covered only when it is a line range/list
// ("1-40", "1-40, 90-120") or an explicit "SKIPPED: <reason>" -- an empty cell or free prose does not.
// QUEUE #310 JA-3 (S-088 JUDGE round 2): anchored -- the earlier unanchored form matched a
// stray digit anywhere in free prose ("not opened (0 lines)" satisfied it via the "0").
const COVERAGE_CELL_OK = /^\s*\d+(-\d+)?(\s*,\s*\d+(-\d+)?)*\s*$|^\s*SKIPPED:\s*\S/;
// QUEUE #310 REG-7: one COVERAGE row, table (`| file | cell |`) or bullet (`- file: cell` / `- file — cell`).
// A pasted scope line (free prose, no row markers) yields zero rows, so it can no longer satisfy
// the duty just by happening to contain the file names as a substring.
function coverageRows(body) {
  const rows = [];
  for (const line of body.split("\n")) {
    const t = line.match(/^\|\s*([^|]+?)\s*\|\s*([^|]*?)\s*\|/);
    if (t) { rows.push({ file: t[1].trim(), cell: t[2].trim() }); continue; }
    const b = line.match(/^\s*[-*]\s*([^:—]+?)\s*[:—]\s*(.+)$/);
    if (b) rows.push({ file: b[1].trim(), cell: b[2].trim() });
  }
  return rows;
}
// QUEUE #310 JA-3: a "^## COVERAGE" match inside a fenced code block (a dodge pasting the whole
// section as a fenced example rather than a real one) does not count -- walk line by line,
// toggling fence state on a ``` or ~~~ line, and only match a heading while NOT inside one.
function findUnfencedHeadingIndex(text, headingRe) {
  const lines = text.split("\n");
  let inFence = false, offset = 0;
  for (const line of lines) {
    if (!inFence && headingRe.test(line)) return offset;
    if (/^\s*(```|~~~)/.test(line)) inFence = !inFence;
    offset += line.length + 1;
  }
  return -1;
}

// §8.6: rooms wrote `diff`, `grep`, `code-read`, `web-search`, `diff+prepass` against a contract
// naming executed|read|inferred, and the validator that would catch it could not see the section.
const EVIDENCE_KINDS = ["executed", "read", "inferred"];
// QUEUE #160 / D-0419 D2 (validator half): resolve a CLAIMS files[] path against the tree the
// same way section 4 resolves a cited path — a small, self-contained resolver here since section
// 4's helpers (candidates/findingsIndex/inTree) are defined further down and CLAIMS is checked
// first; the direct-path case (the shape files[] entries actually use) is what's duplicated —
// cheaper than reordering the file's existing structure for the fuzzy-basename fallback CLAIMS
// doesn't need.
// QUEUE #374(b): `--root` used to win whenever the file EXISTED there, so a room's EDIT to an
// existing file could not be cited at its candidate line -- the line was out of range (or held
// other text) in the base copy and `--alt-root` was never consulted. `want` ({l1, l2, symbols})
// now picks the first copy the citation actually FITS (line in range, and a symbol in the range
// when symbols are given), `--root` before `--alt-root`; when no copy fits, the old order decides
// and the caller's out-of-range / no-symbol message is unchanged.
function resolveClaimPath(p, want) {
  const cands = [p, join(REPORT_DIR, p).replace(ROOT + "/", ""), join(REPORT_DIR, "..", p).replace(ROOT + "/", "")];
  const fits = (abs) => {
    if (!want) return true;
    if (statSync(abs).isDirectory()) return true;
    const lines = readFileSync(abs, "utf8").split("\n");
    if (want.l1 < 1 || want.l1 > lines.length) return false;
    if (!want.symbols || !want.symbols.length) return true;
    return symbolsMatch(lines.slice(want.l1 - 1, Math.max(want.l1, want.l2)).join("\n"), want.symbols);
  };
  let first = null;
  for (const c of cands) {
    for (const base of [ROOT, ALT].filter(Boolean)) {
      const r = join(base, c);
      if (!existsSync(r)) continue;
      if (fits(r)) return r;
      if (first === null) first = r;
    }
  }
  return first;
}
// [REG-02] (S-080 FIX5): strip `// ...` line comments (best-effort — a `//` immediately
// preceded by `:` survives, so `http://`-shaped text in code is not eaten) before testing a
// symbol, so a symbol present only in a comment no longer satisfies the CLAIMS check.
function stripLineComments(t) {
  return t.split("\n").map((line) => line.replace(/(?<!:)\/\/.*$/, "")).join("\n");
}
// [REG-02] word-boundary match instead of a plain substring, so a symbol that is itself a
// substring of a longer identifier (e.g. "fold" inside "uniFold") no longer satisfies the check.
function symbolsMatch(t, symbols) {
  const clean = stripLineComments(t);
  return symbols.some((s) => new RegExp("\\b" + s.replace(/[.*+?^${}()|[\]\\]/g, "\\$&") + "\\b").test(clean));
}

            }
            continue;
          }
          const l1 = Number(fm[2]);
          const l2 = fm[3] ? Number(fm[3]) : l1;
          const flines = readFileSync(abs, "utf8").split("\n");
          if (l1 < 1 || l1 > flines.length) {

            if (LEGACY) console.log(`warning (--legacy): ${msg}`); else problems.push(msg);
            continue;
          }
          if (symbols.length) {

            if (!symbolsMatch(slice, symbols)) {
              // [JN-2] (S-080 FIX6): same posture as the no-line branch above -- a CLAIMS
              // symbol mismatch warns for every contract version until the output contract
              // defines the rule, instead of failing only outside --legacy.
              console.log(`warning: ${label}: files[] range ${p}:${fm[2]}${fm[3] ? "-" + fm[3] : ""} contains none of symbols [${symbols.join(", ")}] (QUEUE #160/D2)`);
            }
          }
        }
      });
    }
  }
}

// (merged pair -- >=2 limbs); one entry can satisfy "per limb" for at most one limb.

// is a plain substring test -- CLAIMS naming "F-34" satisfied merged row "F-3" for free. Match on

// 3d. PRE-W304/VALENF: a `true` in CHECKS owes its evidence (_PROMPT-CORE.md `## Priority tags`,
// `<high>` — "A `true` in CHECKS carries its evidence beside it, not just a boolean"). Nothing
// checked the second sentence before this: a bare `{"x": true}` with no supporting line passed
// every existing gate. This one closes that gap.
//
// Scope: every CHECKS key whose value is exactly `true`, top-level or one level nested (the same
// `group.key` shape the false-scan above already descends — `parentKey.childKey`). `false`,
// numbers and strings are NOT in scope: `false` already fails by contract (S-010 #61, the block
// above); a number or a string ("report_opens": 0) is a measurement, not a certification of a
// state — the packet's own framing — so it owes nothing here.
//
// Matching: NORMALIZED, not exact-token. Case-insensitive, and `_`/`-`/`.`/whitespace between the
// key's own words are treated as interchangeable separators (zero or more of them), so
// `no_source_edits` matches a prose mention of "no source edits" or "no-source-edits" without
// forcing every room to spell the JSON key back verbatim in prose. It is NOT a per-word substring
// test: the whole ordered sequence of the key's words must appear together, so a report that only
// says "source" nowhere near "no"/"edits" does not satisfy `no_source_edits` — the packet's own
// warning about a too-loose match.
// REVISED after sampling (rollout step 3, PRE-W304/VALENF): a CONSECUTIVE-sequence regex missed
// real evidence whenever prose put a number or another word between two of the key's words
// ("Tests 50 passed" for a "..._tests_pass" key — "50" breaks a `[\s_.-]*` separator class that
// only allows whitespace/underscore/dot/hyphen). Loosened to: every word of the key must appear,
// AT A WORD BOUNDARY, SOMEWHERE in the same line/claim — order- and adjacency-independent. This
// still refuses a bare one-word match (the packet's own "source" inside "no_source_edits"
// example): a 3-word key still needs all 3 words present, just not glued together.
// Sampling (rollout step 3, PRE-W304/VALENF) found one recurring, systemic false positive: a
// "typecheck"-named CHECKS key traced to a real, executed `npx tsc -p <pkg> --noEmit` line every
// time (CORE's own "Typecheck one package" alias for exactly that command) -- 3 of 10 sampled
// reports hit this same gap. A single, narrow synonym for a name this tree already treats as
// interchangeable (not a general synonym dictionary, which would reopen the "too loose" risk).
const WORD_SYNONYMS = { typecheck: ["tsc"] };
function keyWords(key) {
  return String(key).split(/[_\-.\s]+/).filter(Boolean);
}
function keyMatchesText(key, s) {
  const words = keyWords(key);
  if (!words.length) return false;
  const t = String(s ?? "");
  // A real `\b` treats `_` as a word character, so "no" would never match inside the single
  // underscore-joined token "no_source_edits" -- exactly the shape a CHECKS key IS, and exactly
  // the shape most reports echo it back in prose. Boundary redefined as "not a letter or digit"
  // (so `_`/`-`/`.`/space/punctuation all count as a boundary) instead of relying on `\b`.
  // Trailing-inflection tolerant: "pass" also matches "passed"/"passing" (a suffix of letters
  // ONLY, no leading match — "bypass" does not satisfy "pass") -- sampling (rollout step 3) found
  // "Tests 50 passed" failing to satisfy a "*_tests_pass" key on exact-word matching alone.
  return words.every((w) => {
    const alts = [w, ...(WORD_SYNONYMS[w.toLowerCase()] || [])];
    return alts.some((a) => new RegExp("(?<![A-Za-z0-9])" + a.replace(/[.*+?^${}()|[\]\\]/g, "\\$&") + "[a-z]*(?![0-9])", "i").test(t));
  });
}
// "A command or a path:line" — reused, not reinvented: the path:line half is the same
// `EXT` shape section 4's own citation regex already accepts (ts/tsx/mjs/js/sql/md/json/jsonc/
// py/txt with a `:N` or `:N-M` suffix); the command half is the same invocation shape
// `findings-digest.mjs`'s `evidenceText()`/its EV_MAX splitter already treats as re-runnable
// (npx/pnpm/node/vitest/tsc/git, an explicit `timeout <n>`, or a `cmd -> observation` arrow).
// I43-ORGTOOLS (RJ07-notes): `jsonl` too, so `<room>/FINDINGS.jsonl:3` is a path:line.
const EVIDENCE_PATHLINE_RE = /[A-Za-z0-9_./-]+\.(?:ts|tsx|mjs|js|sql|md|jsonl|json|jsonc|py|txt|swift|kt|kts):\d+(?:-\d+)?/;
const EVIDENCE_COMMAND_RE = /\b(?:npx|pnpm|node|npm|tsc|vitest|git(?:\s+--no-optional-locks)?|timeout\s+\d+|curl|sed\s+-n|bash|sh)\b/;
function hasCommandOrPathLine(s) {
  const t = String(s ?? "");
  return EVIDENCE_PATHLINE_RE.test(t) || EVIDENCE_COMMAND_RE.test(t) || /(?:->|→|=>)/.test(t);
}

  const methodSteps = promptMethodSteps(ptext);
  if (methodSteps.size) {
    const want = (sk, n) => `\`method: ${sk} — step ${n} — ${METHOD_STEP_KINDS.join("|")} — <clause>\``;
    const seen = new Map();
    for (const [sk, steps] of methodSteps) if (!steps.length) pmProblem(`the prompt's ## METHOD ${sk} carries no numbered step under ### Required procedure — no method line can be checked for it`);
    for (const l of text.split("\n").filter((x) => /^method:/.test(x))) {
      const m = methodLineRe().exec(l);
      if (!m) { pmProblem(`malformed method line "${l}" — write ${want("SKnn", "<N>")} for a skill the prompt's ## METHOD sections carry (${[...methodSteps.keys()].join(", ")})`); continue; }
      const [, sk, nRaw, , clause] = m;
      const n = Number(nRaw);
      if (!methodSteps.has(sk)) { pmProblem(`method line names ${sk}, which no ## METHOD section of the prompt carries (${[...methodSteps.keys()].join(", ")}) — "${l}"`); continue; }
      if (!methodSteps.get(sk).includes(n)) { pmProblem(`method line names ${sk} step ${n}, but the prompt's ## METHOD ${sk} Required procedure has steps ${methodSteps.get(sk).join(", ") || "none"} — "${l}"`); continue; }
      const k = `${sk} ${n}`;
      seen.set(k, (seen.get(k) ?? 0) + 1);
      if (seen.get(k) === 2) pmProblem(`more than one method line for ${sk} step ${n} — exactly one is owed (${want(sk, n)})`);
      if (!statesClause(clause)) pmProblem(`method line for ${sk} step ${n} carries an empty or placeholder clause ("${clause}") — say what was done, or why it was skipped (${want(sk, n)})`);
    }
    for (const [sk, steps] of methodSteps) for (const n of steps) if (!seen.has(`${sk} ${n}`)) pmProblem(`no method line for ${sk} step ${n} (${want(sk, n)})`);
  }
}

// 6a. QUEUE #594 (I16): the record said which model ALIAS was asked for (`dispatch_model` on the

// `resolved_model: <the model id its own system prompt states>` (org/rooms/_OUTPUT-CONTRACT.md shape
// block). Its FAMILY (opus|sonnet|haiku, the only words --dispatch-model accepts) is compared with the
// `dispatch_model` of the room's LATEST packet-log row (a resume row counts — it is the dispatch the
// report was finished under), found in the `_packets.jsonl` of the nearest ancestor of the room's
// PROMPT.md directory. A stated id in another family, or one naming no family, is a problem wherever
// the line appears. ABSENCE binds from I17 on: the I16 rooms were compiled before the contract asked

// printed warning.
const RESOLVED_MODEL_FAMILIES = ["opus", "sonnet", "haiku"];
const RESOLVED_MODEL_BINDS_FROM_I = 17;
function dispatchRowFor(reportFile) {

    });

}

// 6b. QUEUE #368 / contract 0.8.2 (S-093, Jason's consumable-output rule D-0443): the batch
// `_ENV.md` used to say "THERE IS NO BUDGET LINE. Never delegate; never time-box." — which is
// why a room could spend sixty turns and hand back nothing. The replacement is OUTPUT-shaped,
// not a turn budget: every room prompt carries a `checkpoint:` turn (default 25), and by it the
// room must already have ONE consumable artifact on disk. A room MAY pass its checkpoint; it may
// not pass it SILENTLY. So: a report whose declared `turns:` exceeds its checkpoint must carry a
// `## CHECKPOINT` section recording what existed at it, and a `## CHECKPOINT` that records
// outstanding work must be accompanied by `## QUEUED` (item 1b) — one named line per unfinished
// thread with the `file:line` that decides it, so a downstream room reads that section alone
// instead of the whole report.
//
// NO NEW MACHINERY (the brief's own constraint): the checkpoint is read from the report header
// if the room echoed it, else from the sibling PROMPT.md the compiler wrote, else 25. The gate is
// the same declared-version posture #235 and #222 use — enforced once a report declares >= 0.8.2,
// a warning below that — so no report written before this rule landed becomes INVALID for it.
const CHECKPOINT_ENFORCED_VERSION = [0, 8, 2];
const CHECKPOINT_DEFAULT = 25;
function checkpointFor(reportFile, headerText) {
  // Trailing prose after the number is allowed: the compiled TAIL line carries the rule's own
  // one-line reminder after the value, and a room that echoes the line usually echoes that too.
  const own = headerText.match(/^checkpoint:\s*(\d+)\b/m);
  if (own) return { n: Number(own[1]), src: "report header" };
  // The compiler writes the room's PROMPT.md beside its REPORT.md (ADR-0007 B4). Read only the
  // TAIL-sized end of it: the checkpoint line lives in the per-compile TAIL, and a bounded read
  // here is the same discipline #235 asks of rooms.
  const promptPath = join(dirname(resolve(reportFile)), "PROMPT.md");
  if (existsSync(promptPath)) {
    try {
      const t = readFileSync(promptPath, "utf8");
      const m = t.match(/^\s*-?\s*checkpoint:\s*(\d+)\b/m);
      if (m) return { n: Number(m[1]), src: "sibling PROMPT.md" };
    } catch { /* unreadable prompt falls through to the default, same as an absent one */ }
  }
  return { n: CHECKPOINT_DEFAULT, src: "default" };
}

    const sec = (text.match(/^## CHECKPOINT\b[^\n]*\n([\s\S]*?)(?=\n## |(?![\s\S]))/m) || [])[1] ?? "";
    if (!sec.trim()) problems.push("`## CHECKPOINT` section is EMPTY — it must record what existed on disk at the checkpoint turn (`checkpoint: <N> — <artifact path(s)> | none`), QUEUE #368");

        }

      return [];
    })();
    // D-0732 (FX-O1): a scout's `open` is a recon anchor in its own declared vocabulary, not a
    // disposition, so the #416 reason-for-open rule does not bind it. The room type is read ONLY
    // from what the compiler recorded in PROMPT.md (`- ROOM TYPE:` / `=== ROOM:`, via roomTypeFor
    // with NO dir fallback), never from the directory name (FAMILY-022 P1): a room dir named
    // `scout` with no PROMPT.md is not exempt.
    const scoutOpenExempt = roomTypeFor(file, null) === "scout";
    if (roomWords.length) console.log(`NOTE (W3-11a-R2 §3): ${contractFile} declares its own findings status vocabulary (${roomWords.join(", ")}); this report's rows are checked against those words as well as the contract's seven.`);
    const allowedStatus = new Set([...CANON_STATUS, ...roomWords]);
    const seenIds = new Map();
    // QUEUE #507: the addressee population, counted so the check cannot pass by being empty.
    let woRows = 0, woAddressed = 0;
    // QUEUE #548: the register-link population, counted the same way and for the same reason.
    let linkRows = 0, linkDeclared = 0;
    // D-0768: `none: <what>` is the STATED value for a row that disposes of no register item.
    let linkNone = 0;
    let linkQueue = 0;
    const linkDivergent = [];
    readFileSync(findingsPath, "utf8").split("\n").forEach((l, idx) => {
      if (!l.trim()) return;
      const lineNo = idx + 1;
      let row;
      try { row = JSON.parse(l); } catch { return; } // section 9 already reports the parse failure
      if (!row || typeof row !== "object" || Array.isArray(row)) { r2Problem(`FINDINGS.jsonl:${lineNo} is not a JSON OBJECT — one object per line, \`{id, sev, file, line, mechanism, status, evidence}\``); return; }
      const id = row.id ?? "?";
      const missing = REQUIRED_KEYS.filter((k) => row[k] === undefined || row[k] === null || row[k] === "");
      if (missing.length) r2Problem(`FINDINGS.jsonl:${lineNo} ("${id}") is missing required key(s) ${missing.join(", ")} — the row shape is \`{id, sev, file, line, mechanism (<=${MECHANISM_MAX} chars), status, evidence}\` (_OUTPUT-CONTRACT.md FINDINGS ARE DATA)`);
      // RG-TOOLING:9 (I16-FXG): the row's `line` is the file:line verified at HEAD, so it must exist in its `file`
      // (CLAIMS files[] are range-checked the same way). A file the tree does not hold is not judged here.
      if (typeof row.file === "string" && Number.isInteger(row.line)) {
        const fp = row.file.replace(/:[\d,\u2013-]+$/, "");
        const abs = fp ? resolveClaimPath(fp, { l1: row.line, l2: row.line, symbols: [] }) : null;
        if (abs && !statSync(abs).isDirectory()) {
          const n = readFileSync(abs, "utf8").split("\n").length;
          // Bound through the FINDINGS-schema channel (r2Problem, with its dispatch floors); before I17 only COUNTED, so
          // a historical report's stdout names no row it did not name before (W311a-regression's line count).
          if (row.line < 1 || row.line > n) {
            if (CLAIM_SHAPE_BINDS) r2Problem(`FINDINGS.jsonl:${lineNo} ("${id}"): line ${row.line} is out of range for ${fp} (${n} lines) -- a row cites the file:line it verified at HEAD`);
            else findingsLineUnbound++;
          }
        }
      }
      if (typeof row.mechanism === "string" && row.mechanism.length > MECHANISM_MAX) r2Problem(`FINDINGS.jsonl:${lineNo} ("${id}") has a ${row.mechanism.length}-char \`mechanism\`; the contract's bound is ${MECHANISM_MAX}. The digest truncates past it, so the part a downstream room reads is the part you did not choose.`);
      // QUEUE #500: `sev` had no vocabulary, and the corpus carries 30+ spellings of it (`high`,
      // `HIGH`, `med`, `S2`, `code (T2)`, ...), so every `"sev":"high"` count this org quoted
      // under-reported. Normalised HERE, at write time, not mapped downstream: a map of the
      // spellings seen so far passes the next one nobody listed, which is #500's own defect one
      // spelling later. So ONLY the five canonical words pass; a known variant is refused WITH its
      // canonical spelling named, and anything else is refused with the five listed.
      if (row.sev !== undefined && row.sev !== null && row.sev !== "" && !SEV_CANON.includes(row.sev)) {
        const hint = typeof row.sev === "string" ? SEV_VARIANTS[row.sev.trim().toLowerCase()] : undefined;
        floorProblem(B23_C2_SHIPPED_AT, `FINDINGS.jsonl:${lineNo} ("${id}") has sev ${JSON.stringify(row.sev)}, which is not one of {${SEV_CANON.join(", ")}}${hint ? ` — write "${hint}"` : " and maps to none of them; choose one"} (QUEUE #500: one spelling per severity, or no severity count means anything).`);
      }
      if (row.status !== undefined && !allowedStatus.has(String(row.status))) {
        r2Problem(`FINDINGS.jsonl:${lineNo} ("${id}") has status "${row.status}", which is not one of this room's words {${[...allowedStatus].join(", ")}}. An off-vocabulary status is not a new word — it is a row the closure ledger cannot map, and one such row dropped a whole room out of the closure count (#400). File it as a CONTRACT VIOLATION against the vocabulary rule instead of minting one.`);
      }
      if (String(row.status) === "open" && !scoutOpenExempt) {
        // #416: `reason` is a FIELD. A room that explains itself in `mechanism` has satisfied a
        // reader and NOT satisfied the ledger, which reads the field and nothing else.
        if (row.reason === undefined || row.reason === null || row.reason === "") r2Problem(`FINDINGS.jsonl:${lineNo} ("${id}") has status "open" and no \`reason\` FIELD — _OUTPUT-CONTRACT.md requires \`reason\` alongside status, one of {${OPEN_REASONS.join(", ")}}. Prose in \`mechanism\` does not answer it: the ledger reads the field.`);
        else if (!OPEN_REASONS.includes(String(row.reason))) r2Problem(`FINDINGS.jsonl:${lineNo} ("${id}") has status "open" with reason "${row.reason}"; the contract's two reasons are {${OPEN_REASONS.join(", ")}} — \`wrong_owner\` is its OWN status word, not an open reason, because "whose file is this" is a different question from "is this decided yet".`);
      }
      // QUEUE #507 (D-0555): a `wrong_owner` row owes an ADDRESSEE -- the repo-relative path of
      // the file the fix belongs in. THE WORD WAS MISTAKEN FOR THE ROUTING. 223 register defects
      // sit located, described and undispatchable because the row shape had nowhere to name a
      // recipient: of 32 `wrong_owner` rows the judge sampled, 30 carried an actionable mechanism
      // and 0 carried an addressee, and the teachback found `reason` absent on 41 of 41.
      // THIS IS A PROBLEM, NOT A WARNING, and deliberately so: `r2Problem` is the same sink the
      // vocabulary and `reason` rules use, so a room briefed after this rule shipped gets
      // INVALID + exit 1, and only a run this repo has NAMED as pre-dating the rule downgrades.
      // A warning here is the mechanism by which 223 accumulated.
      if (requiresAddressee(row.status)) {
        woRows++;
        const raw = row[ADDRESSEE_KEY];
        const fault = addresseeFault(raw);
        if (fault) {
          r2Problem(`FINDINGS.jsonl:${lineNo} ("${id}") has status "wrong_owner" and ${ADDRESSEE_FAULTS[fault]} (fault: ${fault}). \`wrong_owner\` MEANS "the fix belongs in a file this room does not own" -- name that file. \`${ADDRESSEE_KEY}\` is a repo-relative path (a trailing \`:line\` is allowed and ignored), never a room id: the cut is re-derived every wave and \`scripts/register-ownership-audit.mjs\` keys ownership BY PATH, so a path routes and a room id expires. Prose in \`mechanism\` does not answer it -- \`scripts/route-wrong-owner.mjs\` reads the field.`);
        } else {
          const site = normalizeAddressee(raw);
          const found = [ROOT, ALT].filter(Boolean).some((base) => existsSync(join(base, site)));
          if (!found) r2Problem(`FINDINGS.jsonl:${lineNo} ("${id}") has status "wrong_owner" with \`${ADDRESSEE_KEY}\` "${site}", which exists in no tree checked (${[ROOT, ALT].filter(Boolean).join(", ")}). A path no tree carries is a path no room owns, so the row cannot be routed and is the same dead end as an absent field (rule 22(l): the \`path\` you verified at HEAD).`);
          else woAddressed++;
        }
      }
      // QUEUE #548 (THE REGISTER LINK). A FINDINGS row is a DISPOSITION of a register item and
      // named that item only in PROSE, inside `mechanism`. The link was therefore recovered by
      // string-scanning, and two defensible extraction methods over one corpus returned 282 and
      // 356 closed -- 26% apart. `iid` is the field; it is OPTIONAL, because a tooling or process
      // finding is legitimately about no register item, and making it mandatory would buy a
      // compliant `"n/a"` in 2,386 rows rather than a link.
      //
      // WHAT IS A PROBLEM is a DECLARED link the ledger cannot parse: the row then LOOKS linked
      // and is not, which is strictly worse than declaring nothing -- the same shape as an
      // `addressee` that no tree carries, above.
      linkRows++;
      const rawIid = row[REGISTER_KEY];
      const iidNone = isIidNone(rawIid);
      // QUEUE #992 (I33-FX1-TOOLS): `QUEUE #NNN` names the QUEUE row the row disposes of -- a declared link, not a fault.
      // I34-W3-SEAM-B (I33 B21): a row whose `iid` names QUEUE rows and no register id, as a string or an array -- the rows
      // closure-ledger.mjs counts `declared-queue` (declaredQueueIds, declaredRegisterIds: the same two parses).
      const iidQueue = declaredQueueIds(row).length > 0 && !declaredRegisterIds(row).length;
      // RG-548-NONE-NA under D-0769: a `none:` value the pre-fix literal accepted and the fix refuses
      // (a placeholder description such as `none: n/a`) is a NEW refusal, so it binds only after B27.
      const iidPlaceholder = !iidNone && typeof rawIid === "string" && /^none:\s*\S.{2,}$/.test(rawIid.trim());
      const iidFault = iidNone || iidPlaceholder || iidQueue ? null : registerIdFault(rawIid);
      if (iidPlaceholder) {
        postB27Problem(`FINDINGS.jsonl:${lineNo} ("${id}") declares \`${REGISTER_KEY}: ${String(rawIid).trim()}\` — a placeholder is the compliant blank with a prefix; \`none: <what>\` names what the row is about instead (e.g. \`none: QUEUE #548\`) (QUEUE #548, B27-RG-ORG RG-548-NONE-NA)`);
      } else if (iidQueue) {
        linkQueue++;
      } else if (iidNone) {
        linkNone++;
        const prose = normalizeRegisterIds(String(row.mechanism ?? "").match(REGISTER_ID_RE) ?? []);
        if (prose.length) linkDivergent.push(`${id}: declares ${JSON.stringify(String(rawIid).trim())}, \`mechanism\` names ${prose.join(",")}`);
      } else if (rawIid === undefined || rawIid === null) {
        // #548 (D-0768): REQUIRED after B27. A legacy alias (`item`, ...) is read, never written, so it does not satisfy this.
        postB27Problem(`FINDINGS.jsonl:${lineNo} ("${id}") has no \`${REGISTER_KEY}\` field. After B${POST_B27_CUTOFF_BATCH} every row names what it disposes of: \`I3-NNNN\` (or an array), or \`none: <what the row is about instead>\` (e.g. \`none: QUEUE #548\`) — _OUTPUT-CONTRACT.md THE REGISTER LINK. Legacy aliases (${["item", "register", "register_id", "registerId"].join(", ")}) are read, never written.`);
      }
      if (iidFault) {
        r2Problem(`FINDINGS.jsonl:${lineNo} ("${id}") declares \`${REGISTER_KEY}\` and ${REGISTER_FAULTS[iidFault]} (fault: ${iidFault}). \`${REGISTER_KEY}\` is the register item this row DISPOSES OF -- the same key REGISTER.jsonl joins on -- as \`I3-NNNN\`, or an array of them. A value the ledger cannot parse links to nothing while looking linked, so \`scripts/closure-ledger.mjs\` counts the row in the 2,386 that name no item.`);
      } else {
        const declaredIds = declaredRegisterIds(row);
        if (declaredIds.length) {
          linkDeclared++;
          // DIVERGENCE, which is the 282-vs-356 gap made visible per report instead of per
          // corpus: prose that names an item the declaration does not carry. Not a problem --
          // a row may legitimately mention a sibling or a duplicate -- but the reader is told,
          // because under the OLD linker that mention WAS the link.
          // The prose side is canonicalised by the SAME function as the declared side
          // (`declaredRegisterIds` calls it too), never by a hand-rolled `toUpperCase`. Two
          // canonicalisers is the D-0621 defect in miniature, and here it would corrupt the
          // NOTICE itself: a prose id upper-cased one way and a declared id the other makes
          // this report a divergence that is not real, or hide one that is. Every token here
          // matched `REGISTER_ID_RE`, so the total parse's poison-on-malformed arm cannot fire;
          // it is doing dedupe + case + stable order, from one place.
          const prose = normalizeRegisterIds(String(row.mechanism ?? "").match(REGISTER_ID_RE) ?? []);
          const extra = prose.filter((x) => !declaredIds.includes(x));
          if (extra.length) linkDivergent.push(`${id}: declares ${declaredIds.join(",")}, \`mechanism\` also names ${extra.join(",")}`);
        }
      }
      if (row.id !== undefined && row.id !== null && row.id !== "") {
        if (seenIds.has(row.id)) r2Problem(`FINDINGS.jsonl:${lineNo} repeats id "${row.id}" (first at line ${seenIds.get(row.id)}) — the digest and the closure ledger key on the id, so a duplicate silently overwrites or double-counts one of the two rows.`);
        else seenIds.set(row.id, lineNo);
      }
    });
    // QUEUE #507 NON-VACUITY. A check whose input can be EMPTY is the failure mode that let
    // `validate-room-report.mjs`s OWNS mechanism go 316 filings across 51 rooms without firing
    // once (W3-13-batch/FX-WRITE-AUDIT). So the empty case is not spelled like the clean one:
    // zero `wrong_owner` rows prints NOTHING (there was no population to check), and a
    // non-zero population always prints its two numbers, k of n, whether or not any failed.
    if (woRows > 0) console.log(`ADDRESSEE (QUEUE #507): ${woAddressed}/${woRows} wrong_owner row(s) carry a resolvable \`${ADDRESSEE_KEY}\` path.`);
    // QUEUE #548 NON-VACUITY, in the ADDRESSEE block's own shape: a file with no rows prints
    // NOTHING (there was no population), and a non-empty one always prints k of n.
    if (linkRows > 0) console.log(`REGISTER LINK (QUEUE #548): ${linkDeclared}/${linkRows} row(s) declare \`${REGISTER_KEY}\`; the rest are linked by scraping \`mechanism\`, which is the part of the closure figure two extraction methods disagree about (282 vs 356, 26%).`);
    if (linkQueue > 0) console.log(`REGISTER LINK QUEUE (QUEUE #992): ${linkQueue}/${linkRows} row(s) state \`${REGISTER_KEY}: QUEUE #NNN\` — the QUEUE row they dispose of, not a register item.`);
    if (linkNone > 0) console.log(`REGISTER LINK NONE (QUEUE #548): ${linkNone}/${linkRows} row(s) state \`${REGISTER_KEY}: none: <what>\` — about no register item, by declaration.`);
    for (const d of linkDivergent) console.log(`REGISTER LINK DIVERGENCE (QUEUE #548): ${d} — under the prose linker the mention WAS the link; the declaration now wins. If the row is about both, declare both.`);
  }
}

// 11. QUEUE #199 (D-0765): THE VALIDATOR RECOUNTS WHAT EXISTS. A report's stated totals are CLAIMS
// about FINDINGS.jsonl and CLAIMS, and six rooms' headline totals disagreed with their own tables
// (S-067..S-071 tally). There is no `## VERDICTS` section to recount (#494): the totals are the two
// HEADER lines `findings: <N> [(<status> <n>, ...)]` and `claims: <N>`, and the counts are taken
// from the file and the fenced array themselves. After B27 (D-0769) both lines are required -- a
// recount whose input can be absent passes by being absent -- and any disagreement is a PROBLEM;
// at or before B27 a stated disagreement is a warning and an absent line is counted only.
if (KIND === "REPORT") {
  const fStated = header.match(/^findings:\s*(\d+)\s*(?:\(([^)\n]*)\))?\s*$/m);
  const cStated = header.match(/^claims:\s*(\d+)\s*$/m);
  const fPath = join(REPORT_DIR, "FINDINGS.jsonl");
  const fRows = [];
  if (existsSync(fPath)) {
    for (const l of readFileSync(fPath, "utf8").split("\n")) {
      if (!l.trim()) continue;
      let r = null;
      try { r = JSON.parse(l); } catch { /* section 9 reports it; the line is still a row */ }
      fRows.push(r);
    }
  }
  const recount = (msg) => {
    if (POST_B27) problems.push(msg);
    else console.log(`warning (D-0769: not bound at or before B${POST_B27_CUTOFF_BATCH}): ${msg}`);
  };
  if (!fStated) postB27Problem(`header carries no \`findings: <N>\` line — after B${POST_B27_CUTOFF_BATCH} a report states how many FINDINGS rows it wrote and the validator recounts them (QUEUE #199, D-0765)`);
  else {
    const stated = Number(fStated[1]);
    console.log(`RECOUNT (QUEUE #199): findings stated ${stated}, FINDINGS.jsonl carries ${fRows.length} row(s)`);
    if (stated !== fRows.length) recount(`header states \`findings: ${stated}\` but FINDINGS.jsonl carries ${fRows.length} row(s) (QUEUE #199: a stated total is recounted, not trusted)`);
    if (fStated[2] !== undefined) {
      const statedBy = new Map();
      let parseOk = true;
      for (const part of fStated[2].split(",").map((x) => x.trim()).filter(Boolean)) {
        const pm = /^([a-z_-]+)\s+(\d+)$/.exec(part);
        if (!pm || statedBy.has(pm[1])) { parseOk = false; break; }
        statedBy.set(pm[1], Number(pm[2]));
      }
      if (!parseOk || statedBy.size === 0) recount(`header \`findings:\` breakdown "(${fStated[2]})" is not \`(<status> <n>, ...)\` with each status once (QUEUE #199)`);
      else {
        const countedBy = new Map();
        for (const r of fRows) { const st = r && typeof r === "object" && !Array.isArray(r) ? String(r.status) : "(unparseable)"; countedBy.set(st, (countedBy.get(st) ?? 0) + 1); }
        const off = [...new Set([...statedBy.keys(), ...countedBy.keys()])].sort().filter((k) => (statedBy.get(k) ?? 0) !== (countedBy.get(k) ?? 0)).map((k) => `${k} stated ${statedBy.get(k) ?? 0}, counted ${countedBy.get(k) ?? 0}`);
        if (off.length) recount(`header \`findings:\` breakdown disagrees with FINDINGS.jsonl: ${off.join("; ")} (QUEUE #199)`);
      }
    }
  }
  const claimsCounted = Array.isArray(CLAIMS_ARR) ? CLAIMS_ARR.length : 0;
  if (!cStated) postB27Problem(`header carries no \`claims: <N>\` line — after B${POST_B27_CUTOFF_BATCH} a report states how many CLAIMS entries it made and the validator recounts them (QUEUE #199, D-0765)`);
  else {
    const stated = Number(cStated[1]);
    console.log(`RECOUNT (QUEUE #199): claims stated ${stated}, CLAIMS carries ${claimsCounted}`);
    if (stated !== claimsCounted) recount(`header states \`claims: ${stated}\` but the CLAIMS array carries ${claimsCounted} entr${claimsCounted === 1 ? "y" : "ies"} (QUEUE #199)`);
  }
  // B27-RG-ORG RG-199-RECOUNT-HOLES: the recount above read the FIRST header line only, so a second,
  // contradictory `findings:` line (header or body, bold, any case) and a PROSE total ("9 findings
  // filed: 7 fixed, 2 open.") were never recounted -- and the prose total is #199's own defect.
  // Every stated total outside a code fence is recounted here; binding follows recount() (D-0769).
  const bodyLines = [];
  {
    let fence = false;
    text.split("\n").forEach((l, i) => {
      if (/^\s*(```|~~~)/.test(l)) { fence = !fence; return; }
      if (!fence) bodyLines.push({ l, at: i + 1 });
    });
  }
  const counted199 = { findings: fRows.length, claims: claimsCounted };
  for (const word of ["findings", "claims"]) {
    const re = new RegExp(`^[\\s*_>\`-]*${word}\\s*:\\s*[*_]*\\s*(\\d+)`, "i");
    const all = bodyLines.map((x) => ({ ...x, m: re.exec(x.l) })).filter((x) => x.m);
    const off = all.filter((x) => Number(x.m[1]) !== counted199[word]);
    if (off.length && !(all.length === 1 && (word === "findings" ? fStated : cStated))) recount(`${all.length} \`${word}:\` line(s) in the report and ${off.length} disagree with the ${counted199[word]} counted: ${off.map((x) => `line ${x.at} "${x.l.trim().slice(0, 60)}"`).join("; ")} (QUEUE #199: every stated total is recounted, not only the first header line)`);
  }
  const countedBy199 = new Map();
  for (const r of fRows) { const st = r && typeof r === "object" && !Array.isArray(r) ? String(r.status) : "(unparseable)"; countedBy199.set(st, (countedBy199.get(st) ?? 0) + 1); }
  const PROSE_TOTAL = /\b(\d+)\s+findings?\s+(?:rows?\s+)?(?:filed|written|recorded|reported)\b[\s:(,\u2014-]*((?:\d+\s+[a-z_]+(?:\s*(?:,|and)\s*)?)*)/gi;
  for (const x of bodyLines) {
    for (const pm of x.l.matchAll(PROSE_TOTAL)) {
      if (Number(pm[1]) !== fRows.length) recount(`line ${x.at} states "${pm[0].trim().slice(0, 60)}" but FINDINGS.jsonl carries ${fRows.length} row(s) (QUEUE #199: a prose total is recounted too)`);
      for (const bm of String(pm[2] ?? "").matchAll(/(\d+)\s+([a-z_]+)/gi)) {
        const k = bm[2].toLowerCase();
        if (!new Set([...CANON_STATUS_]).has(k)) continue;
        if (Number(bm[1]) !== (countedBy199.get(k) ?? 0)) recount(`line ${x.at} states "${bm[0]}" but FINDINGS.jsonl carries ${countedBy199.get(k) ?? 0} row(s) with status ${k} (QUEUE #199)`);
      }
    }
  }
}

// 12. #494 (rule 22(n), ORCHESTRATOR.md): A SCOPE-SHADOWED LIMB. A row's limbs are ruled separately, and a
// limb in a file the room does not own is still a limb. The rule was written down and run by no script.
// POPULATION: the `- fix site ...` lines of the compiled PROMPT's SCOPE PACKET -- the fix-site path and
// every backticked path in that line's `files:` segment (a token with a space, e.g. `tests/tooling/ (new
// test)`, names no file and is skipped). Each must be the `file` (line stripped) of a FINDINGS.jsonl row
// or appear verbatim in the report; else the limb is listed. Binds for runs bindsAfterB27 names, except
// I<n>-batch with n < 17 (D-0922(3): batches in flight when the rule landed are not failed retroactively). A packet
// with no fix-site line is NAMED as an empty population, never passed in silence (FAMILY-012).
const SCOPE_SHADOW_FROM_I = 17;
// a citation's line part: `:12`, `:12-19`, `:1346-1352,1404-1409`
const LIMB_LINE_RE = /:[\d,\u2013-]+$/;
if (KIND === "REPORT" && RUN_DIR && ROOM_DIR) {
  const shadowBinds = bindsByName(SCOPE_SHADOW_FROM_I);
  const pf = existsSync(promptFile) ? promptFile : findPrompt([...new Set([ROOT, ALT, REPORT_TREE].filter(Boolean))], RUN_DIR, ROOM_DIR);
  const packet = pf && existsSync(pf) ? scopePacketOf(readFileSync(pf, "utf8")) : null;
  if (shadowBinds && packet !== null) {
    const limbs = new Set();
    for (const line of packet.split("\n")) {
      if (!/^\s*- fix site\b/.test(line)) continue;
      const site = /^\s*- fix site[^:]*:\s*`([^`\s]+)`/.exec(line);
      if (site) limbs.add(site[1].replace(LIMB_LINE_RE, ""));
      const files = /;\s*files:\s*(.*?)(?:;\s*tests pinning:|$)/.exec(line);
      if (files) for (const m of files[1].matchAll(/`([^`]+)`/g)) if (!/\s/.test(m[1])) limbs.add(m[1].replace(LIMB_LINE_RE, ""));
    }
    if (limbs.size === 0) console.log(`NOTE (#494, rule 22(n)): the compiled PROMPT's SCOPE PACKET carries no \`- fix site\` line, so there is no cited-file population to check for scope-shadowed limbs (empty population, named).`);
    else {
      const rowFiles = new Set();
      const fp = join(REPORT_DIR, "FINDINGS.jsonl");
      if (existsSync(fp)) for (const l of readFileSync(fp, "utf8").split("\n")) {
        try { const r = JSON.parse(l); if (r && typeof r.file === "string") rowFiles.add(r.file.replace(LIMB_LINE_RE, "")); } catch { /* section 9 reports a bad line */ }
      }
      const shadowed = [...limbs].filter((f) => !rowFiles.has(f) && !text.includes(f));
      console.log(`SCOPE-SHADOW (#494, rule 22(n)): ${limbs.size - shadowed.length}/${limbs.size} cited file(s) ruled (a FINDINGS row's \`file\` or a mention in the report)`);
      if (shadowed.length) problems.push(`22(n) SCOPE-SHADOWED LIMB: ${shadowed.length} file(s) the scope packet's rows cite are in no FINDINGS.jsonl \`file\` and never mentioned in the report: ${shadowed.join(", ")} -- a limb outside your OWNS is still a limb (REAL -- WRONG OWNER with an addressee, or its own disposition), never silence (ORCHESTRATOR rule 22(n))`);

function shapeProblem(rec) {

async function signatureProblem(rec, keys) {

function plainName(s, what) {

/** Last INFLIGHT row per DIR, by file line order. dir = row.dir ?? row.room, except that a row with no `dir`
 *  that is not a dispatch takes the dir of the latest earlier dispatch with the same (room, attempt ?? 1)
 *  (an attempt-2 `returned` row carries no dir). A `resumed` row with neither `dir` nor `attempt` marks
 *  every dir of its room live: it cannot say which attempt it resumed. */
export function lastRowByDir(text) {
  const dispatchDir = new Map();
  const roomDirs = new Map();
  const last = new Map();
  const problems = [];
  let rows = 0;
  String(text ?? "").split("\n").forEach((raw, i) => {
    if (!raw.trim()) return;
    let row;
    try { row = JSON.parse(raw); } catch { problems.push(`INFLIGHT line ${i + 1}: not JSON`); return; }
    const kind = KINDS.find((k) => row && row[k] !== undefined);
    if (!kind || typeof row.room !== "string") {
      problems.push(`INFLIGHT line ${i + 1}: unknown row shape (keys: ${Object.keys(row ?? {}).join(", ") || "none"})`);
      return;
    }
    rows++;
    const key = `${row.room}\0${row.attempt ?? 1}`;
    const isDispatch = DISPATCH_KINDS.includes(kind);
    const dir = row.dir ?? (isDispatch ? row.room : dispatchDir.get(key) ?? row.room);
    if (isDispatch) dispatchDir.set(key, dir);
    const entry = { kind, at: row[kind], line: i + 1, room: row.room };
    if (kind === "resumed" && row.dir === undefined && row.attempt === undefined) {
      for (const d of roomDirs.get(row.room) ?? []) last.set(d, entry);
    }
    last.set(dir, entry);
    if (!roomDirs.has(row.room)) roomDirs.set(row.room, new Set());
    roomDirs.get(row.room).add(dir);
  });
  return { last, problems, rows };
}

/** `{kind:"resume"}` rows of the batch packet log, as dir -> [time ms] (NaN for an unparsable `at`). */
export function resumeTimes(text) {
  const out = new Map();
  for (const raw of String(text ?? "").split("\n")) {
    if (!raw.trim()) continue;
    let row;
    try { row = JSON.parse(raw); } catch { continue; }
    if (row?.kind !== "resume" || typeof row.dir !== "string") continue;
    if (!out.has(row.dir)) out.set(row.dir, []);
    out.get(row.dir).push(Date.parse(row.at));
  }
  return out;
}

/** Bounded lstat walk (never follows a symlink). Reaching the cap counts as recent: unknown is kept. */
export function recentlyModified(root, { now = Date.now(), recentMs = RECENT_MS, walkCap = WALK_CAP } = {}) {
  const stack = [root];
  let seen = 0;
  while (stack.length) {
    const p = stack.pop();
    let st;
    try { st = lstatSync(p); } catch { continue; }
    seen++;
    if (seen > walkCap) return { recent: true, why: `walk cap ${walkCap} reached` };
    if (now - st.mtimeMs < recentMs) return { recent: true, why: `${p} modified ${Math.round((now - st.mtimeMs) / 1000)} s ago` };
    if (st.isDirectory()) for (const e of readdirSync(p)) stack.push(join(p, e));
  }
  return { recent: false, why: `${seen} entries, none modified in the last ${Math.round(recentMs / 60000)} min` };
}

/** Decide, per INFLIGHT dir, keep or remove. Removes nothing. */
export function planClean({ run, inflightText, packetsText = "", tmpRoot = TMP_ROOT, now = Date.now(), recentMs = RECENT_MS, walkCap = WALK_CAP }) {
  const { last, problems, rows } = lastRowByDir(inflightText);
  const resumes = resumeTimes(packetsText);
  const decisions = [];
  let runDirLink = false;
  try { runDirLink = lstatSync(`${tmpRoot}/${runLower(run)}`).isSymbolicLink(); } catch { /* no run dir: each dir is "no such dir" */ }
  for (const [dir, r] of last) {
    const keep = (why) => decisions.push({ dir, path, remove: false, why });
    let path = null;
    try { path = roomTmpDir(run, dir, tmpRoot); } catch (e) { keep(e.message); continue; }
    if (dir.includes("/")) { keep("a nested dir; --clean removes only a top-level room dir"); continue; }
    if (runDirLink) { keep(`the run dir ${tmpRoot}/${runLower(run)} is a symlink; --clean removes only under a real run dir`); continue; }
    if (r.kind !== "returned") { keep(`last INFLIGHT row (line ${r.line}) is ${r.kind}`); continue; }
    const retMs = Date.parse(r.at);
    if (Number.isNaN(retMs)) { keep(`returned time ${JSON.stringify(r.at)} (line ${r.line}) is not a time`); continue; }
    const later = (resumes.get(dir) ?? []).find((t) => Number.isNaN(t) || t > retMs);
    if (later !== undefined) { keep(`a {kind:"resume"} packet row for ${dir} comes after its return (line ${r.line})`); continue; }
    let st;
    try { st = lstatSync(path); } catch { keep("no such dir"); continue; }
    if (!st.isDirectory()) { keep("not a real directory"); continue; }
    const rec = recentlyModified(path, { now, recentMs, walkCap });
    if (rec.recent) { keep(rec.why); continue; }
    decisions.push({ dir, path, remove: true, why: `returned (line ${r.line}); ${rec.why}` });
  }
  return { decisions, problems, rows };
}

/** The one removal: a path that equals roomTmpDir(run, dir, tmpRoot) exactly, and nothing else. */

}

export function main(argv, { log = console.log, err = console.error, env = process.env } = {}) {
  let o;
  try { o = parseArgv(argv); } catch (e) {
    err(`vm-preflight: ${e.message}\nusage: node scripts/vm-preflight.mjs --run <batch-dir> [--floor-mb 1024] [--clean] [--dry-run]`);
    return 2;
  }
  const tmpBefore = dfAvailMb(o.tmpRoot, o.dfFixture);
  const sess = dfAvailMb(SESSIONS, o.dfFixture);
  log(`df ${o.tmpRoot}: ${tmpBefore ?? "unknown"} MB free (floor ${o.floorMb} MB)`);
  log(`df ${SESSIONS}: ${sess ?? "unknown"} MB free (reported only; never a refusal)`);
  if (o.clean) {
    let repo;
    try { repo = repoCensusRoot({ from: import.meta.url, env }).root; } catch (e) { err(`vm-preflight: ${e.message}`); return 2; }
    const batch = join(repo, "org/07-findings", o.run);
    const inflight = join(batch, "_plan/INFLIGHT.jsonl");
    if (!existsSync(inflight)) { err(`vm-preflight: --clean needs ${inflight}, which does not exist; nothing removed`); return 2; }
    const packets = join(batch, "_packets.jsonl");
    const packetsText = existsSync(packets) ? readFileSync(packets, "utf8") : "";
    if (!packetsText) log(`NOTE: ${packets} is absent or empty; the resume-by-message check had no rows`);
    const plan = planClean({ run: o.run, inflightText: readFileSync(inflight, "utf8"), packetsText, tmpRoot: o.tmpRoot });
    for (const p of plan.problems) log(`PROBLEM: ${p}`);
    if (plan.rows === 0) log(`EMPTY: ${inflight} has 0 rows; no dir decided`);
    let removed = 0;
    for (const d of plan.decisions) {
      if (!d.remove) { log(`keep ${d.path ?? d.dir}: ${d.why}`); continue; }
      if (o.dryRun) { log(`would remove ${d.path}: ${d.why}`); continue; }
      cleanOne(d.path, { run: o.run, dir: d.dir, tmpRoot: o.tmpRoot });
      removed++;
      log(`removed ${d.path}: ${d.why}`);
    }
    const listed = new Set(plan.decisions.map((d) => d.dir));
    const runDir = `${o.tmpRoot}/${runLower(o.run)}`;
    let unlisted = [];
    try { unlisted = readdirSync(runDir).filter((n) => !listed.has(n)); } catch { /* no run dir yet */ }
    for (const n of unlisted) log(`keep ${runDir}/${n}: no INFLIGHT row names this dir`);
    log(`clean: ${removed} removed${o.dryRun ? " (dry run)" : ""}, ${plan.decisions.filter((d) => d.remove).length} removable of ${plan.decisions.length} INFLIGHT dir(s) from ${plan.rows} row(s)`);
  } else log("clean: not requested");
  const tmpAfter = o.clean && !o.dryRun ? dfAvailMb(o.tmpRoot, o.dfFixture) : tmpBefore;
  if (tmpAfter === null) { err(`vm-preflight: df could not measure ${o.tmpRoot}: do not dispatch, copy or run`); return 1; }
  if (tmpAfter < o.floorMb) {
    err(`vm-preflight: ${o.tmpRoot} has ${tmpAfter ?? "unknown"} MB free, under the ${o.floorMb} MB floor${o.clean ? " after cleaning" : ""}: do not dispatch, copy or run`);
    return 1;
  }
  log(`OK: ${o.tmpRoot} ${tmpAfter} MB free >= ${o.floorMb} MB`);
  return 0;
}
