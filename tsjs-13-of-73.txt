import { textClamp, xml } from "@dhwaja/card-render";

/**
 * The profile-creation screen (`#/profile`) — the fan's opt-in social identity.
 *
 * It lives in its own module for the same reason `upgrade.ts` and the M8 safety
 * clients do: it renders NO verdict, is imported by no verdict path, and reads
 * nothing about entitlements. A social handle is not a trust signal and must
 * never be able to influence what a fan is told about anybody.
 *
 * THE FOUNDER DISCLOSURE USED TO BE RENDERED HERE, because
 * `POST /account/profile` was the call that created the founder auto-follow's
 * SOCIAL edge — a mutual follow batched into the same write as the profile.
 * Audit item #24 removed the edge, so this screen makes no promise about
 * follows at all: creating a profile creates a profile.
 *
 * The server is the validator. The constraints below are mirrored only so the
 * form can stop an obviously-doomed request; every one of them is re-checked in
 * `social.ts validateProfileInput`, and this module never treats its own
 * agreement as permission.
 */

export const HANDLE_MAX = 20;
export const DISPLAY_NAME_MAX = 40;
export const BIO_MAX = 200;

export type Visibility = "public" | "private" | "friends";

export interface VisibilityChoice {
  key: Visibility;
  label: string;
  blurb: string;
}

  // public profiles only — these blurbs now say what each level actually
  // covers instead of understating both sides. I22-FX-WEB (I22-JN-06): Private names everything social.ts still
  // serves outside its `visible` gate (fan-account marker, the constellation, the handle-collision warning).
  { key: "public", label: "Public", blurb: "Anyone can see your profile, collection, achievements, Scouts and war record (if you turned it on); you can appear on Dhwaja's leaderboards and be suggested to fans who collect the same releases." },
  { key: "friends", label: "Friends", blurb: "Only people who follow each other can see your collection — a one-way follow is not enough." },
  { key: "private", label: "Private", blurb: "People can find your handle and see that it is a fan account, how many other platforms you have connected (and any of them you chose to show), and a warning if a listed creator uses the same handle; your name, bio, follower counts, featured items, collection, achievements and Scouts stay hidden." },

export interface ProfileSetupData {
  /** The fan's existing profile, or null when they have none yet. */
  profile: ProfileSelfDto | null;
  signedIn: boolean;
  /** `affiliation_visibility` from GET /account/settings (audit item #12).
   * `null`/absent ⇒ the fan has no profile yet, so the control is not drawn:
   * there is nothing to list them as. */
  affiliationVisibility?: string | null;
  /** TRUE when the settings READ failed, which is a different thing from "no

  affiliationUnreadable?: boolean;

/**
 * 0017 / AUDIT ITEM #12 — the separable affiliation lever, rendered as its OWN
 * control immediately under the visibility select.
 *
 * WHY IT IS NOT A FIELD IN THE FORM ABOVE. The form posts to
 * `POST /account/profile`, which is the profile writer; this posts to
 * `POST /account/settings/affiliation`, the same class of narrow consent valve
 * as the broadcast mute. Keeping them separate is not incidental — the bug was
 * that ONE control carried two consents, and merging the fix back into that
 * control's payload would re-couple exactly what this change decoupled. A fan
 * flipping this must never have to re-submit their handle to do it.
 *
 * The copy states the SURFACES, not the mechanism: "creator pages" is what a fan
 * is actually deciding about, and Fan of the Week / Top Fans are named so the
 * decision is informed rather than abstract.
 */
export function affiliationControl(current: string | null | undefined, unreadable = false): string {
  // AN UNREADABLE SETTINGS READ IS NOT "YOU HAVE NO PROFILE".
  //
  // `api.settings()` returns null for every failure — 401, 429, 5xx, offline,
  // unparseable body — and this control used to collapse all of them into the
  // same empty string as "no profile yet". The result was the one place in the
  // client that broke the fail-closed copy law it enforces everywhere else: on
  // the page a fan navigated to specifically to find their consent lever, the
  // lever silently did not exist, with no card, no error and no trace. It is
  // also the NEWEST consent control in the product — added precisely because
  // the fan previously had no door.
  if (unreadable) return `<div class="affil"><h3>Being listed on creator pages</h3>${unreadableCard("your listing setting")}</div>`;
  if (current !== "public" && current !== "private") return ""; // no profile ⇒ nothing to list
  const listed = current === "public";
  return `<div class="affil">
<h3>Being listed on creator pages</h3>
<p class="hint">Creators can show their supporters on their own Dhwaja page — "Fan of the Week" and "Top fans". This is separate from who can see <em>your</em> profile: turning it off leaves your page, your collection and your achievements exactly as they are.</p>
<p class="hint">It covers creator-owned pages only. Dhwaja's own <a href="#/leaderboards" data-nav="#/leaderboards">Leaderboards</a> — Top Collectors, Scouts, Top Requesters and the rest — list any <em>public</em> collector profile by handle whatever this is set to; to come off those, set "Who can see it" above to Private.</p>
<p class="affilstate">You are currently <b>${listed ? "listed on creator pages" : "not listed on creator pages"}</b>.</p>
<button data-act="affiliation" data-next="${listed ? "private" : "public"}" class="ds-btn act">${listed ? "Don't list me on creator pages" : "List me on creator pages"}</button>
<span data-affiliation-msg class="rmsg" aria-live="polite"></span>
</div>`;
}

/** Plain clamp for labels handed to handleLink, which escapes exactly once
 * itself (the clampPlain rule in views.ts — pre-escaping double-escapes). */
function plain(s: unknown, max: number): string {
  const raw = typeof s === "string" ? s : "";
  return raw.length > max ? raw.slice(0, max - 1) + "…" : raw;
}

/**
 * The fan's own follow graph. Two independent lists, each keeping its own
 * failure: an empty following list rendered from a failed read would tell a
 * fan they follow nobody, and "you follow nobody" is a fact only an answered
 * read may state (the fail-closed copy law, views.ts).
 *
 * Each row's button is emitted ONLY for a handle that matches the shared shape
 * — a malformed handle from the API becomes an inert row, never a data-handle
 * attribute the controller would post back.
 *
 * AUDIT ITEM #25: the FOLLOWERS list now carries a Remove button too. It was
 * the only relationship on the screen with no undo anywhere in the product —
 * `POST /account/unfollow` deletes `follower_id = me`, so nothing on any client
 * could remove somebody else's follow OF you, and a follower relationship was
 * permanent by construction for every account. The copy says "Remove", not
 * "Block": it deletes one edge and records nothing about the person, and they
 * are free to follow again.
 */
export function peopleView(following: FollowEntryDto[] | null, followers: FollowEntryDto[] | null, blocks?: FollowBlockDto[] | null): string {
  const row = (f: FollowEntryDto, side: "following" | "followers"): string => {
    const h = typeof f.handle === "string" ? f.handle.toLowerCase() : "";
    const act =
      HANDLE_RE.test(h)
        ? side === "following"
          ? `<button data-act="unfollow" data-handle="${xml(h)}" class="linkbtn">Unfollow</button>`
          : `<button data-act="remove-follower" data-handle="${xml(h)}" class="linkbtn" title="They stop following you and can't follow you again until you unblock them under Removed followers.">Remove</button>`
        : "";
    const when = typeof f.followed_at === "string" ? f.followed_at.slice(0, 10) : "";
    return `<li>${handleLink(f.handle, plain(f.display_name || f.handle, 40))} <span class="muted">${xml(when)}</span>${act}</li>`;
  };
  const fList = Array.isArray(following) ? following : [];
  const fHtml = !Array.isArray(following)
    ? unreadableCard("who you follow")
    : fList.length > 0
      ? `<ul class="following">${fList.slice(0, 200).map((f) => row(f, "following")).join("")}</ul>`
      : `<p class="empty">You don't follow anyone yet.</p>`;
  const rList = Array.isArray(followers) ? followers : [];
  const rHtml = !Array.isArray(followers)
    ? unreadableCard("your followers")
    : rList.length > 0
      ? `<ul class="followers">${rList.slice(0, 200).map((f) => row(f, "followers")).join("")}</ul>`
      : `<p class="empty">No followers yet.</p>`;
  return `<section class="people">
<h3>Following</h3>${fHtml}
<h3>Followers</h3>${rHtml}
<h3>Removed followers</h3><div data-follow-blocks="${blocks === undefined ? "pending" : "read"}">${blocks === undefined ? `<p class="muted">Loading…</p>` : removedFollowersView(blocks)}</div>
<span data-people-msg class="rmsg" aria-live="polite"></span>
</section>`;
}

export interface FollowBlockDto {
  handle: string;
  display_name?: string | null;
  blocked_at?: string;
}

/**
 * I3-1055 — the fan's "Removed followers" (per-fan follow blocks, GET
 * /account/follow-blocks). null is a FAILED read ("couldn't read"), never an
 * empty list; [] renders the plain empty line. Each row's Unblock button is
 * emitted only for a handle of the shared shape (the peopleView rule).
 */
export function removedFollowersView(blocks: FollowBlockDto[] | null): string {
  if (!Array.isArray(blocks)) return unreadableCard("your removed followers");
  if (blocks.length === 0) return `<p class="empty">You haven't removed anyone.</p>`;
  const row = (b: FollowBlockDto): string => {
    const h = typeof b.handle === "string" ? b.handle.toLowerCase() : "";
    const act = HANDLE_RE.test(h) ? `<button data-act="unblock-follower" data-handle="${xml(h)}" class="linkbtn" title="They can follow you again. Nothing else changes.">Unblock</button>` : "";
    const when = typeof b.blocked_at === "string" ? b.blocked_at.slice(0, 10) : "";
    return `<li>${handleLink(b.handle, plain(b.display_name || b.handle, 40))} <span class="muted">${xml(when)}</span>${act}</li>`;
  };
  return `<ul class="follow-blocks">${blocks.slice(0, 200).map(row).join("")}</ul>`;
}

/** Mirrors social.ts MAX_FEATURED — the server remains the arbiter; this only
 * stops an obviously-doomed request and words the cap in the copy. */
export const FEATURED_MAX = 12;

/** An acquisition id, as the featured route accepts it. */
const ACQ_ID_RE = /^acq_[0-9a-f]{20}$/;

/**
 * The featured-showcase picker over the fan's OWN acquisitions. Saving
 * REPLACES the whole showcase (the server's atomic semantics), and the copy
 * says so. A malformed acquisition id from the API gets no checkbox at all —
 * it could never be posted, so offering it would offer a doomed save.
 *
 * AUDIT ITEM #22 — A WITHDRAWN RELEASE GETS NO CHECKBOX EITHER, and this is the
 * dead end the item names. The picker was built from the UNFILTERED
 * `/account/acquisitions`, `setFeatured` validated only ownership, and
 * `featuredViews` then silently dropped withdrawn rows on the way out. So a fan
 * ticked a withdrawn item, got "Saved ✓ — your showcase is updated", and it
 * never rendered, with nothing anywhere saying why. Both ends are fixed: the
 * server now REFUSES a withdrawn item (409, the arbiter as always), and this
 * row is drawn as a disabled, labelled entry rather than removed — a fan who
 * remembers buying something must not find it simply missing from the list of
 * things they own.
 */
export function featuredPickerView(acquisitions: AcquisitionDto[] | null): string {
  if (!Array.isArray(acquisitions)) return `<section class="featpick"><h3>Featured showcase</h3>${unreadableCard("your acquisitions")}</section>`;
  const rows = acquisitions
    .slice(0, 200)
    .filter((a) => typeof a?.acquisition_id === "string" && ACQ_ID_RE.test(a.acquisition_id))
    .map((a) => {
      const label = typeof a.title === "string" && a.title.trim() !== "" ? a.title : a.release_id;
      const when = `<span class="muted">${xml(typeof a.acquired_at === "string" ? a.acquired_at.slice(0, 10) : "")}</span>`;
      if (a.release_status === "withdrawn") {
        return `<label class="featopt off"><input type="checkbox" name="featured" value="${xml(a.acquisition_id)}" disabled> ${textClamp(label, 40)} ${when} <span class="ds-chip quiet">no longer listed — can't be featured</span></label>`;
      }
      return `<label class="featopt"><input type="checkbox" name="featured" value="${xml(a.acquisition_id)}"> ${textClamp(label, 40)} ${when}</label>`;
    })
    .join("");
  if (!rows) return `<section class="featpick"><h3>Featured showcase</h3><p class="empty">Nothing to feature yet — redeem a claim credential first.</p></section>`;
  return `<section class="featpick">
<h3>Featured showcase</h3>
<p class="hint">Pick up to ${FEATURED_MAX} of your authenticated acquisitions to show on your public profile. Saving replaces your current showcase. Featuring an item publishes its verify link.</p>
<form data-act="featured-save">
${rows}
<button type="submit" class="ds-btn act primary">Save showcase</button>
</form>
<span data-featured-msg class="rmsg" aria-live="polite"></span>
</section>`;
}

export interface FeaturedSaveOutcome {
  saved: boolean;
  text: string;
}

/**
 * The showcase save's one line of feedback, pure. 403 is the one failure worth
 * its own sentence — the server refuses an item the caller doesn't hold — and
 * everything unknown collapses into a fixed line that never echoes a server
 * string.
 */
export function featuredSaveOutcome(r: { ok: boolean; status: number; data: { ok?: boolean } | null }): FeaturedSaveOutcome {
  if (r.ok && r.data?.ok === true) return { saved: true, text: "Saved ✓ — your showcase is updated." };
  if (r.status === 401) return { saved: false, text: "You're signed out — sign in and try again." };
  if (r.status === 403) return { saved: false, text: "You can only feature items you hold. Nothing was changed." };
  // AUDIT ITEM #22: the server's 409 for a withdrawn item. Its own sentence,
  // because it is the one refusal the fan can act on — and because the silent
  // version of this outcome ("Saved ✓" followed by nothing appearing) is
  // exactly what the item was filed about.
  if (r.status === 409) return { saved: false, text: "One of those releases is no longer listed by its creator, so it can't go in your showcase. Nothing was changed." };
  if (r.status === 400 || r.status === 413) return { saved: false, text: `Check your picks — up to ${FEATURED_MAX} items, no duplicates. Nothing was changed.` };
  // #877b: no answer arrived, so the save may have landed — the outcome is unknown, never "unchanged".
  if (r.status === 0) return { saved: false, text: "No answer arrived, so we can't tell whether your showcase was saved. Reload to check before saving again." };
  return { saved: false, text: "That didn't save — your showcase is unchanged. Try again in a moment." };
}

/** The unfollow failure line, pure. An unfollow that did not land must never

/** The FOLLOW failure line, pure. Separate from the unfollow one for the same

 * the server's "cannot follow yourself" / "create a profile first" class and

export function followFailureText(status: number, code?: string): string {

  if (status === 403) return "Set up your own collector profile first, then you can follow people.";
  if (status === 400) return "You can't follow yourself.";
  return "Couldn't follow just now — nothing changed. Try again.";
}

export interface ProfileSaveOutcome {
  saved: boolean;
  text: string;
}

/**
 * The form's one line of feedback, pure.
 *
 * Every failure collapses into a fixed sentence rather than echoing the
 * server's `error` string: a rendered server message is an uncontrolled sink
 * for a string this client did not write, and a new server error must never be
 * able to reach a fan's screen unreviewed. 409 is worth its own line only
 * because it is the one failure the person can act on.
 */
export function profileSaveOutcome(r: { ok: boolean; status: number; data: { ok?: boolean; code?: unknown } | null }): ProfileSaveOutcome {
  if (r.ok && r.data?.ok === true) return { saved: true, text: "Saved ✓" };
  if (r.status === 401) return { saved: false, text: "You're signed out — sign in and try again." };
  if (r.status === 409) {
    // M12: a 409 is either a handle collision OR a display-name refusal (a name
    // that reads as "✓ Verified" or platform staff). The server tells us which
    // with a fixed `code`; we map it to our own reviewed copy (never the raw
    // server string). A display-name refusal was previously blamed on the handle.
    if (r.data?.code === "display_name") return { saved: false, text: "That display name isn't allowed — it can't imply Dhwaja verification or platform-staff status. Try another." };
    // I16-W03B H5: the server refuses a bio that shows a verification mark or claims Dhwaja (social.ts:284, code "bio").
    if (r.data?.code === "bio") return { saved: false, text: "That bio isn't allowed — it can't show a verification mark or claim to be from Dhwaja. Try rewording it." };
    return { saved: false, text: "That handle is taken or reserved. Try another." };
  }
  if (r.status === 429) return { saved: false, text: "That's a lot of changes for one day — try again tomorrow." };
  if (r.status === 400 || r.status === 413) {
    return { saved: false, text: `Check the fields: handle is 3–${HANDLE_MAX} characters of a–z, 0–9 or _, display name is 1–${DISPLAY_NAME_MAX} characters, bio is up to ${BIO_MAX}, and neither may contain < or >.` };
  }
  if (r.status === 0) return { saved: false, text: "Couldn't reach Dhwaja. Check your connection and try again." };
  return { saved: false, text: "That didn't save. Try again in a moment." };
}

/**
 * WHERE THE REMOVAL REFERENCE IS ALLOWED TO GO.
 *
 * `rmv_<32hex>` is 128 CSPRNG bits minted by `removal.ts`, and it is the ONLY
 * credential for `POST /removal/status`. There is no account behind
 * it — deliberately, because the people who most need the removal valve have no
 * Dhwaja account — so whoever holds the string can read the request's state.
 * That makes it a bearer secret, and a bearer secret's PLACEMENT is part of the
 * security argument, not a formatting detail:
 *
 *   • IN THE BODY, NEVER IN A URL — not a path segment and not a query string.
 *     The status route used to be `GET /removal/status/:request_id`, which put
 *     the capability in the request PATH; with invocation logging enabled on
 *     the Workers side, request URLs (path included) are recorded, so anyone
 *     with log access could both read live status capabilities and see that a
 *     specific reference asked for delisting. A query string is worse still —
 *     it is the part every proxy and analytics pipeline greps. A JSON body
 *     reaches none of those places, so the reference rides ONLY in a body.
 *
 *   • IN THE SPA FRAGMENT, NEVER IN THE SEARCH STRING. A `#` fragment is never
 *     sent to a server at all, and — critically — is stripped from the `Referer`
 *     header a browser attaches when the page links out. Anything in `?…` rides
 *     along to every third party the page ever links to.
 *
 *   • NEVER IN AN OFF-ORIGIN HREF. An anchor whose href carries the reference
 *     must be same-origin and fragment-scoped, so the string cannot leave the
 *     browser by being clicked.
 *
 * Every construction site for the reference lives in this file so there is
 * exactly one place to audit and exactly one place a mutation can hide.
 * `api.ts` (the network path), `router.ts` (the route shape) and `removal.ts`
 * (the rendered link) all call in here rather than re-typing the string.
 */

/** A removal reference. Hex-only by construction, so no placement below ever
 * needs escaping or encoding — and anything that is not this shape never forms
 * a request at all. */
export const RMV_REF_RE = /^rmv_[0-9a-f]{32}$/;

export function isReference(v: unknown): v is string {
  return typeof v === "string" && RMV_REF_RE.test(v);
}

/**
 * The API request for the token-scoped status read.
 *
 * The reference is a BODY FIELD on a POST to a CONSTANT path. It must never be

/**
 * I7 #250 — a response body as text, or null past `maxBytes`. The cap is applied
 * WHILE STREAMING (and to a declared Content-Length before any read), so a
 * hostile registry cannot make this client buffer more than the cap. A response
 * with no stream (test doubles) falls back to `text()`; every caller still
 * applies its own length check to what comes back. Port of the extension's
 * `readCapped` (registry-client.ts, I3-2043).
 */
async function readCapped(res: Response, maxBytes: number): Promise<string | null> {
  const declared = (res as { headers?: Headers }).headers?.get?.("content-length");
  if (declared && Number(declared) > maxBytes) return null;
  const body = (res as { body?: ReadableStream<Uint8Array> | null }).body;
  if (!body || typeof body.getReader !== "function") return res.text().catch(() => null);
  const reader = body.getReader();
  const chunks: Uint8Array[] = [];
  let n = 0;
  try {
    for (;;) {
      const { done, value } = await reader.read();
      if (done) break;
      n += value.byteLength;
      if (n > maxBytes) {
        reader.cancel().catch(() => {});
        return null;
      }
      chunks.push(value);
    }
  } catch {
    return null;
  }
  const buf = new Uint8Array(n);
  let off = 0;
  for (const c of chunks) {
    buf.set(c, off);
    off += c.byteLength;
  }
  return new TextDecoder().decode(buf);
}

function parseBucketText(text: string): { entries: RegistryEntry[]; servedCount: number; droppedHashes: string[] } | null {
  // E-37 (web half, S-053), widened by I3-0032: the BUCKET ceiling, not the proof one.
  if (text.length > MAX_BUCKET_BYTES) return null;
  let body: { entries?: unknown } | null;
  try {
    body = JSON.parse(text) as { entries?: unknown } | null;
  } catch {
    return null;
  }
  if (!body || !Array.isArray(body.entries)) return null;
  // D-0708: a bucket format this reader does not understand, or none, is a failed check.
  if (!versionAccepted(body, BUCKET_VERSIONS_UNDERSTOOD)) return null;
  const droppedHashes: string[] = []; // valid hashes under an unrecognized class
  try {
    // I3-0236: the padding measure is the DISTINCT well-formed hashes served (any
    // class, case-folded), never raw rows — junk or a repeated hash must not buy
    // compliance with the k-floor. The builder never serves a duplicate (registry-build.ts:1800).
    const servedCount = new Set((body.entries as RegistryEntry[]).map((x) => (x && typeof x.h === "string" ? x.h.toLowerCase() : "")).filter((h) => h.length === 64 && HEX.test(h))).size;
    for (const x of body.entries as RegistryEntry[]) {
      // H-10/PS-105: fold BEFORE testing HEX — a non-lowercase hash must
      // hit droppedHashes/entries the same way the extension's sanitize()
      // and mobile-check's compare do, never fall through as "no record".
      const hh = x && typeof x.h === "string" ? x.h.toLowerCase() : undefined;
      if (x && typeof hh === "string" && HEX.test(hh) && hh.length === 64 && !["verified", "claimed", "unclaimed", "warning", "caution"].includes(x.c)) {
        droppedHashes.push(hh); // registry newer than this build
      }
    }
    const entries: RegistryEntry[] = body.entries
      .filter((x): x is RegistryEntry => {
        const raw = (x as RegistryEntry | null)?.h;
        const hh = typeof raw === "string" ? raw.toLowerCase() : undefined;
        return !!x && typeof hh === "string" && HEX.test(hh) && hh.length === 64 && ["verified", "claimed", "unclaimed", "warning", "caution"].includes((x as RegistryEntry).c);
      })
      .map((x) => {
        // Per-account confirmation: STRICTLY `1`, and only where it can be
        // true. `true`, "1", or a flag on an unclaimed entry are shapes
        // this client's own server never writes, so they are dropped
        // rather than believed — the direction that under-claims.
        const confirmed = x.p === 1 && (x.c === "claimed" || x.c === "verified") ? ({ p: 1 } as const) : undefined;
        // WHAT THE SUBJECT IS. Strictly the two bytes the server writes,
        // and only on a creator class — a kind on a warning entry is a
        // shape our own builder never produces (those hashes are the
        // ACCUSED account's and say nothing about a creator), so it is
        // dropped rather than believed. Anything else reads as a person,
        // which is what every entry meant before characters existed:
        // absence under-claims and can never over-claim.
        const kind = (x.k === "a" || x.k === "b") && x.c !== "warning" && x.c !== "caution" ? ({ k: x.k } as const) : undefined;
        // Find-the-real-one pointer: on warning OR caution entries, strict
        // shapes. Strip/clamp FIRST, then require a non-empty name — a name
        // that is all markup must yield NO pointer, not an empty label.
        // Off-shape degrades to "no pointer", never junk.
        // I3-0122: the warning's kind code, strictly "i"/"f" and only on a warning.
        const wk = x.c === "warning" && (x.w === "i" || x.w === "f") ? ({ w: x.w } as const) : undefined;
        const rr = x.r;
        const name = (x.c === "warning" || x.c === "caution") && rr && typeof rr === "object" && typeof rr.n === "string" ? rr.n.replace(/[<>]/g, "").slice(0, 64).trim() : "";
        if (name.length > 0 && rr && typeof rr.p === "string" && PSP_RE.test(rr.p)) {
          return { h: x.h.toLowerCase(), c: x.c, ...confirmed, ...kind, ...wk, r: { p: rr.p, n: name } };
        }
        return { h: x.h.toLowerCase(), c: x.c, ...confirmed, ...kind, ...wk };
      });
    return { entries, servedCount, droppedHashes };
  } catch {
    return null; // an entry that throws is an uninterpretable bucket: FAILED
  }
}

/**
 * I3-0817 — web half of the rollback floor (mirrors the extension's
 * check.ts seqFloor/publicationSeqGate; see that file for the full
 * rationale). A publication carries its own genuine signature over its own
 * genuine root even when it is a RESTORED OLDER one, so it verifies
 * perfectly; only a client that remembers the highest signed `seq` it has
 * seen can notice. The mark is raised only from a publication whose
 * signature verified under a key this build holds (an unsigned `seq` is
 * whatever the server says, and a poisoned mark would fail every later
 * genuine check), kept in memory per trust anchor and mirrored to
 * `localStorage` when that is reachable, so it outlives the tab. Inert on an
 * unarmed build: no signature ever verifies, so no mark is ever set.
 */
export const SEQ_HIGH_WATER_KEY = "registrySeqHighWater";
const seqHighWater = new Map<string, number>();
const storedSeqMark = new Map<string, number | null>();

/** Per trust anchor: this build's key set (every key, so a rotation inside
 * one build is still covered). A build shipping a different key set starts
 * a fresh memory. */
export function seqAnchorFor(keys: readonly PublishedKey[]): string {
  return keys
    .map((k) => `${k.key_id}:${k.pub}`)
    .sort()
    .join("|");
}

function seqLocalStorage(): Storage | null {
  try {
    return typeof localStorage !== "undefined" ? localStorage : null;
  } catch {
    return null;
  }
}

function loadStoredSeqMark(anchor: string): number | null {
  if (storedSeqMark.has(anchor)) return storedSeqMark.get(anchor)!;
  let v: number | null = null;
  const store = seqLocalStorage();
  if (store) {
    try {
      const raw = store.getItem(SEQ_HIGH_WATER_KEY);
      const rec = raw ? (JSON.parse(raw) as Record<string, unknown>) : null;
      const x = rec ? rec[anchor] : undefined;
      if (typeof x === "number" && Number.isSafeInteger(x) && x >= 0) v = x;
    } catch {
      /* an unreadable store is no memory, never a reason to fail the check */
    }
  }
  storedSeqMark.set(anchor, v);
  return v;
}

/** The floor a check is held to: the mark as it stood when the check
 * started (an input, not a live read — see check.ts's `seqFloor`). */
export function seqFloorFor(anchor: string): number | null {
  return seqHighWater.get(anchor) ?? null;
}

/** False when the accepted publication is OLDER than `floor` or than what an
 * earlier tab persisted (a rollback) — the caller reports `failed`. Raises
 * the mark from a signature-verified `seq`. */
export function publicationSeqGate(infoText: string, signatureVerified: boolean, floor: number | null, anchor: string): boolean {
  // I3-0703 (D-0913(3)): the mark is raised ONLY from a verified signature and is
  // keyed by this build's key set, so a mark for this anchor IS "this client has
  // verified a signature under these keys". An accept that no signature backed
  // (`unsigned`) is then a stripped signature: refused. A new key set is a new
  // anchor, which clears it. An unarmed build (empty anchor) never has a mark.
  if (!signatureVerified && anchor !== "" && (floor !== null || seqHighWater.has(anchor) || loadStoredSeqMark(anchor) !== null)) return false;
  const seq = parseRegistryInfo(infoText)?.seq;
  if (seq === undefined) return true;
  const stored = loadStoredSeqMark(anchor);
  if (floor !== null && seq < floor) return false;
  if (stored !== null && seq < stored) return false;
  const high = Math.max(seqHighWater.get(anchor) ?? -1, stored ?? -1);
  if (signatureVerified && seq > high) {
    seqHighWater.set(anchor, seq);
    const store = seqLocalStorage();
    if (store) {
      try {
        store.setItem(SEQ_HIGH_WATER_KEY, JSON.stringify({ ...(JSON.parse(store.getItem(SEQ_HIGH_WATER_KEY) ?? "{}") as Record<string, number>), [anchor]: seq }));
      } catch {
        /* memory still holds it for this tab */
      }
    }
  }
  return true;
}

/** Test seam: forget the in-memory marks (a fresh module load). */
export function forgetSeqHighWater(): void {
  seqHighWater.clear();
  storedSeqMark.clear();
}

    // only so a test can arm this client against a key it generated — the gate
    // had no seam and therefore no coverage.
    this.keys = opts.registryKeys ?? PUBLISHED_KEYS;
  }

  /**
   * The served prefix width, cached for `INFO_TTL_MS` per read (the extension's
   * `width()` rule, mirrored). Past the TTL it re-fetches `/v1/info`; when that
   * re-fetch fails to ARRIVE (network, non-2xx, unreadable body, bad JSON) it
   * degrades to the width it already had, so a registry that is briefly
   * unreachable does not turn a warm tab's checks into "failed" — the bucket
   * fetch that follows is the real reachability test. `infoReadAt` is NOT
   * touched on that path, so the stale build clock is dropped from the card
   * (see the `builtAt` gate) and the next check tries `/v1/info` again.
   *
   * A served width that is OUT OF RANGE is different: that is an answer, and a
   * refused one (a wider width is a deanonymization attempt) — it returns null
   * with no fallback, exactly as before.
   */
  private async width(noCache = false): Promise<number | null> {
    const t = this.now();
    // AG-11/SC-03: same negative-age guard as `bucket()` — a backward clock
    // correction must force a re-fetch, not extend the cache indefinitely.
    const infoAge = t - this.infoReadAt;
    if (this.cachedWidth !== null && infoAge >= 0 && infoAge < INFO_TTL_MS) return this.cachedWidth;
    let res: Response;
    try {
      res = await this.fetchImpl(`${this.base}/v1/info`, noCache ? NO_CACHE : { method: "GET" });
    } catch {
      this.lastInfoRefused = false;
      return this.cachedWidth;
    }
    if (!res.ok) {
      this.lastInfoRefused = false;
      return this.cachedWidth;
    }
    // THE RAW BYTES ARE KEPT alongside the two fields this client reads: the
    // publication signature covers `{width, seq, built_at, root}` AS SERVED, so
    // a verifier rebuilds the preimage from the served text rather than from a
    // summary of it. Re-serializing the parsed object is a different string.
    const raw = await readCapped(res, MAX_INFO_BYTES);
    if (raw === null) {
      this.lastInfoRefused = false;
      return this.cachedWidth;
    }
    let body: { width?: unknown; built_at?: unknown } | null = null;
    try {
      body = JSON.parse(raw) as { width?: unknown; built_at?: unknown };
    } catch {
      body = null;
    }
    // D-0825(1)/D-0849(1): a 200 whose body is JSON `null`/`0`/`false` or not
    // JSON at all names no manifest version, so it is refused exactly as a
    // version this reader does not understand is — the held manifest is
    // dropped too (mirrors apps/extension/src/registry-client.ts readWidth).
    if (!body) {
      this.cachedWidth = null;
      this.cachedBuiltAt = null;
      this.cachedInfoText = null;
      this.lastInfoRefused = true;
      return null;
    }
    // D-0708: a manifest format this reader does not understand (or none at
    // all) is an ANSWER — this client is retired for it — so the held manifest
    // is dropped too and the check reads `failed`, never a cached width. It runs
    // BEFORE the width check (D-0823(5)): a bump whose width is also bad or
    // absent is still a bump, and must not leave the old manifest held.
    if (!versionAccepted(body, MANIFEST_VERSIONS_UNDERSTOOD)) {
      this.cachedWidth = null;
      this.cachedBuiltAt = null;
      this.cachedInfoText = null;
      this.lastInfoRefused = true;
      return null;
    }
    if (!Number.isInteger(body.width) || (body.width as number) < 1 || (body.width as number) > MAX_CLIENT_WIDTH) {
      this.lastInfoRefused = true;
      return null;
    }
    this.lastInfoRefused = false;
    this.cachedInfoText = raw;
    this.cachedWidth = body.width as number;
    // SHAPE-CHECKED, NOT TRUSTED. This string is rendered, so a hostile or
    // broken registry must not be able to put arbitrary text on the verdict
    // card. An unparseable or absent value degrades to "no freshness line"
    // rather than to a rendered lie — the same fail-closed stance the width
    // check above takes.
    this.cachedBuiltAt = cleanBuiltAt(body.built_at, t);
    this.infoReadAt = t;
    return this.cachedWidth;
  }

  /**
   * Check an external profile URL. Canonicalizes locally, hashes locally,
   * fetches ONE bucket by prefix, and matches the full hash locally. Any
   * failure surfaces as "failed" (fail-closed, flag 23) — never a forged hit and never the reassuring "unknown".
   */
  async checkUrl(rawUrl: string): Promise<CheckResult> {
    let subject: CanonicalSubject;
    try {
      subject = canonicalize(rawUrl);
    } catch {
      return { status: "unsupported" };
    }
    if (subject.platform === "url" || subject.idKind === "norm") return { status: "unsupported" };
    return this.checkSubject(subject);
  }

  /**
   * Check a (platform, handle) pair directly — the v14 bare-handle lookup
   * screen's path. Exactly the same local canonicalize→hash→one-bucket
   * pipeline as checkUrl; the only difference is that the subject arrives as a
   * pair the viewer picked instead of a URL, so it goes through
   * canonicalAccount (the same normalizer a URL extraction lands in).
   */
  async checkAccount(platform: string, handle: string): Promise<CheckResult> {
    // A handle a real profile URL could carry (the same charset profileUrl
    // re-checks). canonicalAccount itself accepts any string for some
    // platforms, but a handle with spaces or punctuation can never be a live
    // profile — "unsupported", and nothing is fetched for it.
    if (typeof handle !== "string" || !/^[A-Za-z0-9._-]{1,64}$/.test(handle)) return { status: "unsupported" };
    let subject: { platform: string; id: string; canonical: string };
    try {
      const a = canonicalAccount(platform, "handle", handle);
      subject = { platform: a.platform, id: a.id, canonical: a.canonical };
    } catch {
      return { status: "unsupported" };
    }
    return this.checkSubject(subject);
  }

  /**
   * Check the subject a `#/check/...` route names — the same local
   * canonicalize -> hash -> one-bucket pipeline, entered from a ROUTE instead of
   * from a keystroke, so a check result can be revisited, refreshed and shared.
   *
   * It rebuilds the subject through `subjectForAccount`, the canonicalizer's own
   * "(platform, id_kind, id) -> subject" function — the same one the server's
   * fold uses for a stored account row — rather than re-deriving it here. A
   * second derivation of a subject is a second chance to disagree with the
   * registry about what was checked, and disagreeing quietly is how a fan gets
   * the wrong verdict about the right account.
   *
   * Anything the canonicalizer refuses is `unsupported`: a hand-typed fragment
   * must not produce a hash, and a hash it cannot produce cannot match.
   */
  async checkRouted(platform: string, idKind: CheckIdKind, id: string): Promise<CheckResult> {
    let subject: { platform: string; id: string; canonical: string };
    try {
      // `subjectForAccount` speaks the STORED id-kind vocabulary, where a
      // Tier-2 link is "link"; the route carries the CANONICAL kind, which for
      // the same subject is "url". One mapping, in one place.
      const a = subjectForAccount(platform, idKind === "url" ? "link" : idKind, id);
      subject = { platform: a.platform, id: a.id, canonical: a.canonical };
    } catch {
      return { status: "unsupported" };
    }
    if (subject.platform === "url") return { status: "unsupported" };
    return this.checkSubject(subject);
  }

  /**
   * Fetch the bucket for a hash, refreshing a STALE cached width once (M4).
   *
   * The width is cached for `INFO_TTL_MS` (see `width()`), so a width change
   * shows up on its own within ten minutes. Inside that window a cached-width
   * client still builds a prefix of the wrong length after an ops rebuild, which
   * the registry service 404s (`prefix.length !== manifest.width`) — and before
   * M4 that meant EVERY check on that tab failed until a full page reload. On a
   * 404 we re-read `/v1/info` once (bypassing the TTL by clearing the cache); if
   * the width actually moved, we retry with the new prefix. A 404 under an
   * UNCHANGED width is a genuine failure (a non-padding registry) and stands.
   */
  private async fetchBucketFor(hash: string): Promise<Response | null> {
    for (let attempt = 0; attempt < 2; attempt++) {
      const width = await this.width();
      if (width === null) return null;
      const prefix = bucketPrefix(hash, width);
      if (!HEX.test(prefix) || prefix.length !== width) return null;
      let res: Response;
      try {
        res = await this.fetchImpl(`${this.base}/v1/bucket/${prefix}`, { method: "GET" });
      } catch {
        return null;
      }
      if (res.status === 404 && attempt === 0) {
        const prev = this.cachedWidth;
        const prevBuiltAt = this.cachedBuiltAt;
        const prevReadAt = this.infoReadAt;
        const prevInfoText = this.cachedInfoText;
        this.cachedWidth = null; // force a re-read of /v1/info
        this.cachedBuiltAt = null; // …and the build clock read with it, so the two can never come from different reads
        this.cachedInfoText = null; // …and the bytes, for the same reason
        const fresh = await this.width();
        if (fresh !== null && fresh !== prev) continue; // width moved → retry with the new prefix
        // D-0849(1): only a TRANSIENT re-read failure restores the previous
        // manifest. A definitive refusal (bad/absent JSON, an unaccepted
        // version, an out-of-range width) has already forgotten it inside
        // width() and must stay forgotten here.
        if (fresh === null && !this.lastInfoRefused) {
          this.cachedWidth = prev; // couldn't re-read → keep what we had; the 404 stands
          this.cachedBuiltAt = prevBuiltAt;
          this.infoReadAt = prevReadAt;
          this.cachedInfoText = prevInfoText;
        }
        return res;
      }
      return res;
    }
    return null;
  }

  /**
   * THE PUBLICATION CHECK — slice 28, the client half of signed registry
   * publication: this client's fetchers, handed to the retry decision shared
   * with the extension (`resolvePublication` in `@dhwaja/registry-proof/registry-proof`). The rules
   * live in `@dhwaja/registry-proof`, the same module the iOS and Android bundle runs, so
   * nothing here holds a grammar, a verdict mapping or an ordering decision.
   *
   * "accept" when this build holds no key or the registry publishes no
   * signature — which is every deployment today — and otherwise a clean
   * verification is required. A registry that IS making a signed claim and
   * cannot back it up does not get to answer.
   *
   * THE RETRY: `mismatch` and `stale` are both reachable from an edge cache
   * filled before the last rebuild, and a stale bucket is indistinguishable
   * from an injected one (the bucket is the only artifact carrying no build
   * identifier). One uncached re-fetch of both artifacts tells them apart.
   */
  /** I3-0817: like the extension's `publicationIO` in check.ts, the returned
   * `signatureVerified()` is true only once a candidate signature has
   * verified under a key this build holds, IN THE PASS CURRENTLY RUNNING —
   * observed by wrapping `verify`, the one seam this module already owns,
   * rather than reading `resolvePublication`'s accept shape (which is
   * identical for unarmed, unsigned and genuinely verified). */
  private publicationIO(hash: string, prefix: string, bucketRaw: string): PublicationIO & { signatureVerified(): boolean; retried(): boolean } {
    const verify = webCryptoVerifier(this.keys);
    let sigOk = false;
    let retried = false;
    return {
      armed: armed(this.keys),
      // I8-FX3 T2-05: did the uncached retry run at all? A retry can return the SAME bytes as the first fetch
      // (only the proof edge copy was stale), so "the bytes differ" is not the witness.
      retried: () => retried,
      verify: async (preimage, sig) => {
        const out = await verify(preimage, sig);
        if (out === "ok") sigOk = true;
        return out;
      },
      signatureVerified: () => sigOk,
      digest: sha256Hex,
      // I3-1106: a signed publication older than 14 days from its signed
      // `built_at` is not accepted as verified (registry-proof.ts).
      nowMs: this.now(),
      firstBucketText: bucketRaw,
      prefix,
      // A width change means the prefix this check was built on no longer exists.
      rehash: hash,
      prefixFor: bucketPrefix,
      infoText: () => this.cachedInfoText,
      proof: (p, noCache) => this.fetchProof(p, noCache),
      // THE MANIFEST IS REFRESHED FIRST, and it is the reason the retry works at
      // all. `cachedInfoText` is served from cache for up to `INFO_TTL_MS`, so
      // after an honest rebuild this instance can hold build n-1's manifest
      // against build n's bucket, report `stale`, and — if only the other two
      // artifacts were re-fetched — get the identical `stale` and fail for the
      // rest of the window. (Before the TTL existed it was for the life of the
      // tab; adversarial review measured it at 100% of lookups, permanently, the
      // moment a key was provisioned.) Clearing the cache forces `width()` past
      // the TTL and past its stale-width fallback: a manifest that cannot be
      // re-read is a "fail" here, never a retry against the old bytes.
      // RGC-01/02 (I15-FX5): the witness resets at the first-pass/retry boundary,
      // as in the extension's check.ts:112 — it describes the ACCEPTING pass only.
      refreshInfo: () => {
        sigOk = false;
        this.cachedWidth = null;
        this.cachedInfoText = null;
        this.cachedBuiltAt = null;
        // S-073 S9: uncached, like the bucket and proof retry — an edge still
        // holding the previous manifest would otherwise answer `stale` again.
        return this.width(true);
      },
      bucketTextUncached: (p) => {
        retried = true;
        return this.fetchBucketUncached(p);
      },
    };
  }

  /** The inclusion proof for one bucket. Never cached here — see the retry. */
  private async fetchProof(prefix: string, noCache: boolean): Promise<string | null> {
    try {
      const res = await this.fetchImpl(`${this.base}/v1/proof/${prefix}`, noCache ? NO_CACHE : { method: "GET" });
      if (!res.ok) return null;
      const text = await readCapped(res, MAX_PROOF_BYTES);
      if (text === null) return null;
      // A proof is ~1.2 KB at the widest client width; far larger is not one.
      return text.length > MAX_PROOF_BYTES ? null : text;
    } catch {
      return null;
    }
  }

  /** Re-fetch one bucket's bytes with caching bypassed. See the retry. */
  private async fetchBucketUncached(prefix: string): Promise<string | null> {
    try {
      const res = await this.fetchImpl(`${this.base}/v1/bucket/${prefix}`, NO_CACHE);
      if (!res.ok) return null;
      const text = await readCapped(res, MAX_BUCKET_BYTES);
      if (text === null) return null;
      // E-37 (web half, S-053): the bucket read is capped at MAX_BUCKET_BYTES
      // (512 KiB, I3-0032 — the extension's registry-client.ts carries the same
      // 512 KiB bucket / 64 KiB proof split); this uncached re-fetch had no cap.
      // `readCapped` stops the DOWNLOAD at the cap (I7 #250); this length check is
      // the same rule as parseBucketText, kept for bodies with no stream.
      return text.length > MAX_BUCKET_BYTES ? null : text;
    } catch {
      return null;
    }
  }

  /**
   * I3-1795 — CACHED entry point. A fan re-checking the same page inside the
   * short TTL gets a very recent answer with no network round trip, the same
   * guarantee the extension's watch sweep already has (registry-client.ts
   * `bucketTtl`). A "failed" answer is never cached — same rule as the
   * extension never caching a miss — so a registry that is briefly
   * unreachable does not get pinned as "failed" for the whole window.
   */
  private async checkSubject(subject: { platform: string; id: string; canonical: string }): Promise<CheckResult> {
    const key = subject.canonical;
    const t = this.now();
    const cached = this.results.get(key);
    if (cached) {
      const age = t - cached.at;
      // AG-11/SC-03: same negative-age guard as width()/bucket() — a backward
      // clock correction must force a re-fetch, not extend the cache.
      if (age >= 0 && age < this.resultTtl) return cached.value;
    }
    const result = await this.checkSubjectFresh(subject);
    if (result.status !== "failed") this.results.set(key, { at: t, value: result });
    return result;
  }

  private async checkSubjectFresh(subject: { platform: string; id: string; canonical: string }): Promise<CheckResult> {

    const hash = await subjectHash(subject.canonical);
    // I3-0817: the rollback floor, fixed BEFORE the network (mirrors check.ts).
    const seqAnchor = seqAnchorFor(this.keys);
    const seqFloor = seqFloorFor(seqAnchor);

    const failed: CheckResult = { status: "failed", platform: subject.platform, handle: subject.id };
    // I8 #256: a retry-accepted bucket for this prefix inside the window stands in for
    // the whole fetch + publication pass (it was verified when it was remembered).
    const remembered = await this.acceptedBucketFor(hash);
    if (remembered) {
      if (!publicationSeqGate(remembered.infoText, remembered.sigOk, seqFloor, seqAnchor)) {
        // I8-FX3 T2-06: a refused entry is EVICTED, so the next check re-fetches (and heals) instead of
        // answering `failed` from the same stale entry until the window lapses. This answer stays failed.
        this.acceptedBuckets.delete(remembered.prefix);
        return failed;
      }
      return this.answerFromBucket(subject, hash, remembered.bucketText, remembered.infoText, failed);
    }
    let bucketRaw: string | null = null; // the exact served bytes, for the publication check
    try {
      // The registry service's actual bucket route ("/v1/bucket/<prefix>",
      // body {v, entries}) — the same protocol the extension speaks. (This
      // client previously fetched a "/v1/b/<prefix>.json" / {e} shape no
      // service serves, so every web check degraded to "unknown".) The width the
      // prefix is built from is refreshed once on a 404 (M4).
      const res = await this.fetchBucketFor(hash);
      // flag 23: registry unreachable / prefix unbuildable = FAILED, not "no record".
      if (res === null) return failed;
      // THE EXACT SERVED BYTES. A publication leaf commits to the bucket file
      // as it was sent, and `parseBucketText` builds fresh objects from named
      // fields — hashing a re-serialization would report an honest registry as
      // tampered with. Slice 28.
      if (res.ok) bucketRaw = await readCapped(res, MAX_BUCKET_BYTES);
    } catch {
      bucketRaw = null;
    }
    // flag 23: no well-formed response ever arrived — the check FAILED. AUDIT
    // A-1: the first bytes are checked for SHAPE only; nothing read from them
    // is kept, because they may not be the bytes that verify.
    if (bucketRaw === null || parseBucketText(bucketRaw) === null) return failed;

    // SLICE 28 — the publication signature, before any bytes are read as an
    // answer. See `publicationIO`. THE WIDTH THE BUCKET WAS FETCHED UNDER,
    // read from the cache rather than through `width()`: `fetchBucketFor` has
    // just left `cachedWidth` equal to exactly that (a TTL refresh, a stale
    // fallback and the M4 restore all end there), whereas a second `width()`
    // call could re-fetch `/v1/info` and hand the verifier a DIFFERENT width
    // from the one the prefix was built with.
    const width = this.cachedWidth;
    if (width === null) return failed;
    const pubIO = this.publicationIO(hash, bucketPrefix(hash, width), bucketRaw);
    const publication = await resolvePublication(pubIO);
    if (publication.action === "fail") return failed;
    // I3-0817: a publication older than one this client already saw is a
    // rollback, even though its own signature is clean — only a client that
    // remembers the highest verified `seq` can tell (mirrors check.ts).
    if (!publicationSeqGate(publication.infoText, pubIO.signatureVerified(), seqFloor, seqAnchor)) return failed;
    // I8 #256: a retry that ACCEPTED handed back fresh bytes — remember those (not the
    // first bytes) so the next check in this bucket does not re-pay the retry. I8-FX3 T2-05: keyed on
    // the retry HAVING RUN, not on the bytes differing — a stale proof over an honest bucket retries to
    // identical bytes and is just as worth remembering. Still only after the sequence gate above.
    if (pubIO.retried()) this.rememberAccepted(bucketPrefix(hash, width), publication.bucketText, publication.infoText, pubIO.signatureVerified());
    return this.answerFromBucket(subject, hash, publication.bucketText, publication.infoText, failed);
  }

  private async acceptedBucketFor(hash: string): Promise<{ prefix: string; bucketText: string; infoText: string; sigOk: boolean } | null> {
    if (this.acceptedBuckets.size === 0) return null;
    const width = await this.width();
    if (width === null) return null;
    let prefix: string;
    try {
      prefix = bucketPrefix(hash, width);
    } catch {
      return null;
    }
    const hit = this.acceptedBuckets.get(prefix);
    if (!hit) return null;
    const age = this.now() - hit.at;
    // Same negative-age guard as `results`: a backward clock correction forces a re-fetch.
    if (age < 0 || age >= this.resultTtl) {
      this.acceptedBuckets.delete(prefix);
      return null;
    }
    return { prefix, ...hit };
  }

  private rememberAccepted(prefix: string, bucketText: string, infoText: string, sigOk: boolean): void {
    if (this.resultTtl <= 0) return;
    this.acceptedBuckets.delete(prefix);
    if (this.acceptedBuckets.size >= 64) this.acceptedBuckets.delete(this.acceptedBuckets.keys().next().value as string);
    this.acceptedBuckets.set(prefix, { at: this.now(), bucketText, infoText, sigOk });
  }

  /** Everything from the VERIFIED bytes on (I8 #256: shared by the fresh pass and the remembered-retry path). */
  private answerFromBucket(subject: { platform: string; id: string; canonical: string }, hash: string, bucketText: string, infoText: string, failed: CheckResult): CheckResult {
    // AUDIT A-1: every reading below — entries, the k-floor served count and
    // the dropped classes — comes from the bytes that VERIFIED (on a retry, the
    // fresh bytes), and the build clock from the manifest they verified under.
    const verified = parseBucketText(bucketText);
    if (verified === null) return failed;
    const { entries, servedCount, droppedHashes } = verified;
    const builtAt = builtAtOf(infoText, this.now());
    // A too-small bucket means a registry NOT padding to its k-floor → treat as
    // untrusted. Distrust is a FAILED check (flag 23), not a miss — "Not
    // recognized" about an unchecked subject is exactly the reassurance the
    // fail-closed copy law bans. Measured on the DISTINCT well-formed hashes served
    // (`servedCount` above, I3-0236: any class, case-folded, so neither junk rows
    // nor a repeated hash buy compliance). This is a protocol-compliance check, not an anonymity
    // lever — see docs/REGISTRY-PRIVACY-CLAIMS.md.
    // (No servedCount > 0 guard: a well-formed EMPTY bucket under a configured
    // k-floor is still a non-padding registry — distrust it too, extension parity.)
    if (this.minBucket > 0 && servedCount < this.minBucket) return failed;
    // The subject IS listed but under a class this build can't interpret —
    // FAILED (parity with the extension + mobile core, flag 23).
    if (droppedHashes.includes(hash)) return failed;
    const hit = entries.find((e) => e.h === hash); // full-hash equality only
    // A full-hash MISS is "unknown" (Dhwaja has no record) — NOT "unclaimed",
    // which is a real registry class. Parity with the extension + mobile core.
    return {
      status: hit ? hit.c : "unknown",
      platform: subject.platform,
      handle: subject.id,
      // The confirmation rides the matched ENTRY and only the entry: asking the
      // server "is this one proven?" afterwards would tell it exactly which
      // subject matched, which is what the prefix design exists to prevent.
      ...(hit?.p === 1 ? { confirmed: true } : {}),
    // From the same entry and no second fetch — asking the server about a
    // specific subject after a local match is what the prefix design prevents.
    ...(hit && kindFromWire(hit.k) ? { kind: kindFromWire(hit.k)! } : {}),
      ...(hit && (hit.c === "warning" || hit.c === "caution") && hit.r ? { real: { p: hit.r.p, n: hit.r.n } } : {}),
      ...(hit && hit.c === "warning" && (hit.w === "i" || hit.w === "f") ? { warningKind: hit.w } : {}),
      // ON THE ANSWER ONLY, never on a failure. Every `failed` return above
      // leaves this absent deliberately: a build clock beside "we couldn't
      // check" would date data that was never read, which is the same class of
      // lie as reporting a miss for an unreachable registry (flag 23).
      //
      // …and never a clock older than the window, for the same reason. A stale
      // one under-claims rather than over-claims, but the line's whole job is to
      // be true, and a creator screenshotting a dispute deserves a card that
      // does not understate its own freshness by hours.
      // AG-11/SC-03: same negative-age guard as `width()`/`bucket()` — a backward
      // clock correction must not read as "fresh" here either.
      ...(builtAt && this.now() - this.infoReadAt >= 0 && this.now() - this.infoReadAt < INFO_TTL_MS ? { builtAt } : {}),
    };
  }
}

 *  2. NO DELISTING ORACLE. `/removal/start` answers with the same shape for a
 *     listed subject, an unlisted one, an already-suppressed one and an
 *     ineligible one — a decoy challenge is minted so the bodies match. The
 *     client therefore CANNOT and MUST NOT tell the fan which one they got. The
 *     code screen says "if this listing is ours and this account is yours, this
 *     removes it" — a conditional, because the client genuinely does not know.
 *
 *  3. THE STATUS PAGE IS TOKEN-SCOPED AND IS NOT A PROBE. `request_id` is the
 *     only credential. Every non-200 — unknown id, someone else's id, a
 *     malformed id, a throttle, an outage — renders ONE constant card
 *     (`STATUS_UNREADABLE`). An unreadable state value falls into the same card.
 *     Pinned by test: nothing about the response other than a recognised state
 *     on a 200 may change a single byte of what is shown.
 *
 *  4. A FAILURE IS NEVER REASSURANCE. "We couldn't find the code" does not
 *     become "you're probably fine"; it says what is still true — the listing is
 *     still up — and points at the door that always works (ask a human).
 */

/** A Dhwaja passport id, as printed on a listing. */
const PSP_SUBJECT_RE = /^psp_[0-9a-f]{16}$/;

/**
 * Read the removal subject the way the server wants it: a `psp_` id goes in
 * `passport_id`, anything else in `url`.
 *
 * Extracted from the DOM controller so it is testable. Deliberately does NOT
 * decide whether the link is a profile we list — that judgement is the
 * server's, and a client guess here would produce a confident, wrong rejection
 * on the one path a person in trouble has to be able to use. An empty subject
 * is passed through as an empty `url` so the server's 400 (which the state
 * machine renders) is what the person sees, rather than a client-side refusal
 * that never reached the queue.
 */
export function removalSubject(raw: unknown): { url?: string; passport_id?: string } {
  const v = typeof raw === "string" ? raw.trim() : "";
  return PSP_SUBJECT_RE.test(v) ? { passport_id: v } : { url: v };
}

/**

export function verifyRetryOffered(kind: VerifyKind): boolean {
  return kind !== "removed" && kind !== "queued";
}

/** What `removal.ts` (trust-core) returns as `held_balance`. Absent when nothing
 * is held, and then this module renders nothing about money at all. */
export interface HeldBalanceDto {
  held_cents: number;
  clearing_cents: number;
}

/** Cents → a display dollar string, spelled exactly as the creator console's
 * earnings screen spells it (`creator-console/src/earnings.ts dollarsFromCents`:
 * 2550 → "US$25.50", 2500 → "US$25"; #132PS-344: the currency is named), so the figure here matches the tile the
 * creator already reads. */
export function dollarsFromCents(cents: number): string {
  const n = Math.round(Number(cents) || 0);
  const abs = Math.abs(n);
  const s = abs % 100 === 0 ? `US$${abs / 100}` : `US$${(abs / 100).toFixed(2)}`;
  return n < 0 ? `−${s}` : s;
}

function heldPositive(h: unknown): h is HeldBalanceDto {
  if (!h || typeof h !== "object") return false;
  const o = h as Record<string, unknown>;
  const held = typeof o.held_cents === "number" && Number.isFinite(o.held_cents) ? o.held_cents : 0;
  const clearing = typeof o.clearing_cents === "number" && Number.isFinite(o.clearing_cents) ? o.clearing_cents : 0;
  return held > 0 || clearing > 0;
}

/**
 * The notice a removal owes the person when their passport still holds money
 * (D-0670 (b)): that a cleared balance exists, that it is HELD — not paid, not
 * forfeited — and how to claim it. The claim path is the existing "Ask a person"
 * door on this page (a manual request, read by the same people who decided the
 * removal); no new flow. Empty string when nothing is held.
 */
export function heldBalanceNotice(h: unknown): string {
  if (!heldPositive(h)) return "";
  const held = Math.max(0, Math.round(h.held_cents || 0));
  const clearing = Math.max(0, Math.round(h.clearing_cents || 0));
  const lead =
    held > 0
      ? `Your cleared balance of ${dollarsFromCents(held)} has not been paid out, and it is held: removing the listing stops payouts, so the weekly run will not send it.`
      : "Removing the listing stops payouts, so earnings you have not been paid out are held and the weekly run will not send them.";
  const tail = clearing > 0 ? ` A further ${dollarsFromCents(clearing)} is still inside the 14-day hold and will be held the same way once it clears.` : "";
  return `${lead}${tail} It is not lost and nobody else is paid it. To claim it, send a request through "Ask a person" on the removal page, choose "Something on it is wrong", and quote this reference — a person reads it, the same queue that handled the removal.`;
}

/**
 * The verify state machine, pure.
 *
 * `code_absent` is the branch that carries the module's whole privacy argument:
 * the server returns byte-identical `{ok:false, state:"open",
 * outcome:"code_absent"}` for a real challenge whose code wasn't on the page AND
 * for a decoy that never had a challenge at all (unlisted / already suppressed /
 * ineligible subject). The client must therefore have exactly ONE branch here.
 * Splitting it — even into two tones — would hand back the bit the server spent
 * a whole module refusing to give: whether a live control-proof exists for that
 * subject. So the copy is written to be true of both: it describes what we
 * looked for and did not find, and never asserts anything about the listing's
 * existence.
 */
export function removalVerifyOutcome(r: { ok: boolean; status: number; data: { ok?: boolean; state?: string; outcome?: string } | null }): VerifyOutcome {
  if (r.status === 429) {
    return { kind: "capped", suppressed: false, tone: "quiet", headline: "Too many checks just now", body: `${STILL_LISTED} We rate-limit this so it can't be used to hammer other people's profiles. Nothing about your request changed. Wait a little and press it again — your code is still valid until it expires.`, offerManual: true };
  }
  if (r.status === 409) {
    return { kind: "closed", suppressed: false, tone: "quiet", headline: "This request is already finished", body: "There's nothing left to do on it. Open its status with your reference to see where it got to — and if the answer isn't what you wanted, start a new request or ask a person.", offerManual: true };
  }
  if (r.status === 404 || !r.ok || !r.data) {
    return { kind: "failed", suppressed: false, tone: "quiet", headline: "We couldn't complete that check", body: `${STILL_LISTED} Nothing was changed and nothing was decided. Try again in a moment, or ask a person instead — that door needs no proof and no code.`, offerManual: true };
  }
  const state = r.data.state;
  const outcome = r.data.outcome;
  if (state === "auto_suppressed") {
    const money = heldBalanceNotice((r.data as { held_balance?: unknown }).held_balance);
    return {
      kind: "removed",
      suppressed: true,
      tone: "good",
      headline: "Done. Your listing is gone.",
      body: `The page is down and Dhwaja will not list that account again — the next time we build the directory from public sources, it is refused rather than re-added. You don't need to do anything else, you don't need an account, and nobody had to approve it. Keep your reference below if you ever want to check this again.${money ? ` ${money}` : ""}`,
      offerManual: false,
    };
  }
  if (state === "queued") {
    return {
      kind: "queued",
      suppressed: false,
      tone: "good",
      headline: "Proved. A person will finish it.",
      body: "You proved you control that account. We don't take a listing down automatically when it's been claimed, verified, carries more than one account, or when you asked us to correct or dispute something rather than remove it — proving one account isn't authority over a whole identity, and that restraint protects you as much as anyone. Someone will look at this by hand. Keep your reference below to check on it.",
      offerManual: false,
    };
  }
  if (outcome === "expired") {
    return {
      kind: "expired",
      suppressed: false,
      tone: "quiet",
      headline: "That code has expired",
      body: `${STILL_LISTED} Codes are short-lived on purpose so an old one can't be reused by someone else later. Start a fresh request to get a new code, or ask a person instead.`,
      offerManual: true,
    };
  }
  // The shared "the proof did not land" body. ONE branch, deliberately — and since I3-0900 (D-1078(15)) the server
  // also answers it when the profile did not load, so the copy is true of an outage too (CODE_ABSENT_LINE).
  return {
    kind: "code_absent",
    suppressed: false,
    tone: "quiet",
    headline: "We didn't find that code on the profile",
    body: `${STILL_LISTED} ${CODE_ABSENT_LINE}`,
    offerManual: true,
  };
}

export interface StatusCard {
  /** `unreadable` is the constant. */
  state: "open" | "auto_suppressed" | "resolved" | "declined" | "unreadable";
  tone: "good" | "quiet";
  headline: string;
  body: string;
}

/**
 * The single card every unreadable answer renders.
 *
 * It covers: a reference that was never issued, a real reference belonging to
 * somebody else, a malformed reference, a rate-limited caller, and an outage.
 * They must be indistinguishable, because the alternative is a page that
 * answers "did this person ask to be delisted?" one guess at a time. The copy
 * therefore states the ambiguity out loud rather than picking a likely reason —
 * saying "no such request" would be a claim we are not entitled to make.
 */
export const STATUS_UNREADABLE: StatusCard = {
  state: "unreadable",
  tone: "quiet",
  headline: "We can't show a status for that reference",
  body: "This is the same answer for a reference that was never issued, one that belongs to somebody else, and one we simply can't read right now — we don't say which, and we never will, because saying which would let anyone test whether a particular person asked to be delisted. Check the reference for typos. If you've opened this several times in a row, wait an hour. And if you've lost the reference, just send a fresh request: nothing is lost by asking twice.",
};

/**
 * Read a status response. `state` is the ONLY input on the success path — no
 * timestamps, no intent, nothing else may change which card is chosen — and an
 * unrecognised state falls into the constant card rather than being rendered
 * raw, so a future server state can never leak through this page.
 */
export function removalStatusCard(r: { status: number; data: (RemovalStatusDto & { held_balance?: unknown }) | null }): StatusCard {
  if (r.status !== 200 || !r.data) return STATUS_UNREADABLE;
  const card = statusCardFor(r.data.state);
  // QUEUE #518: a removal that left money held says so, on the two states that
  // mean the listing came down. Nothing else on the body reaches the card.
  const money = card.state === "auto_suppressed" || card.state === "resolved" ? heldBalanceNotice(r.data.held_balance) : "";
  return money ? { ...card, body: `${card.body} ${money}` } : card;
}

function statusCardFor(state: string): StatusCard {
  switch (state) {
    case "open":
      return {
        state: "open",
        tone: "quiet",
        headline: "Open — waiting for a person",
        body: "It's in the queue and it hasn't been decided. We don't publish a position in the queue or an estimate, because either one would be a guess. Nothing you need to do; nothing expires while you wait.",
      };
    case "auto_suppressed":
      return {
        state: "auto_suppressed",
        tone: "good",
        headline: "Removed",
        body: "The listing is down, and Dhwaja will not list that account again — the next directory build refuses it rather than re-adding it. Nothing further is needed from you.",
      };
    case "resolved":
      return {
        state: "resolved",
        tone: "good",
        headline: "Done — a reviewer has actioned this",
        body: "A person has finished with this request and carried out what it needed. We don't show their notes or their name here: this page is scoped to a reference, not to an account, so it only ever shows state.",
      };
    case "declined":
      return {
        state: "declined",
        tone: "quiet",
        headline: "Declined",
        body: "A person looked and did not action this request. That decision is not the end of the road: you can send a new request at any time, and if this was about something Dhwaja published, disputing it is always open to you.",
      };
    default:
      // Fail closed. An unknown state is not rendered.
      return STATUS_UNREADABLE;
  }
}

/**
 * The canonical subject key the SERVER would key this report under.
 *
 * `POST /report` deliberately answers with no item id — report volume and queue
 * state are operator-only signals — so the evidence step has no case to attach
 * to until it finds one. The only place a fan's own case ref is served is
 * `GET /account/reports`, whose rows carry `canonical`. Matching on that is
 * exact (reports.ts keys ONE item per subject hash), which is why we compute the
 * key rather than guessing "the newest row": a repeat report on a subject the
 * fan already reported does NOT move to the top of that list (the reporter row
 * keeps its original `created_at`), so "newest" would attach a screenshot to
 * somebody's unrelated case.
 *
 * The two branches MIRROR `fileReport` exactly: a known-platform profile, or —
 * failing that — the bare brand DOMAIN minted off the un-routable URL fallback.
 * Getting the second branch wrong is not dangerous, only useless: a mismatch
 * yields null, which renders the honest "screenshots aren't available for this
 * case" panel instead of a control that would 404.
 *
 * No hashing happens here. The single-hasher invariant is untouched: this is
 * canonicalization only, from `@dhwaja/canonical`, the same module the server
 * canonicalizes with.
 */
export function canonicalSubjectKey(rawUrl: string): string | null {
  let subject: CanonicalSubject | null;
  try {
    subject = canonicalize(rawUrl);
  } catch {
    subject = null;
  }
  // TP-05 (S-068). THE `link` FOLD IS GONE, BECAUSE INTAKE NEVER DID IT.
  //
  // `services/trust-core/src/reports.ts fileReport` folds a subject to its
  // host domain for exactly one population — `platform === "url" || idKind ===
  // "norm"` — and a generic `link` page (M3 Slice 4) is neither, so the SERVER
  // keys that report under the page's own `v1|link|url|host/path`. This
  // resolver folded it to `v1|domain|domain|host` anyway, a key intake never
  // wrote. Usually that just yielded null and the honest "screenshots aren't
  // available" panel. But when the same fan ALSO has an attachable case about
  // the host domain, the fold matched THAT case and bound the page screenshot
  // to it — the backend's membership check passes, because the case really is
  // the reporter's, so nothing downstream can detect the wrong selection and
  // an operator receives page-specific evidence filed under a broader domain
  // allegation about the same host.
  //
  // The two branches now mirror `fileReport` exactly: anything the
  // canonicalizer gives an identity to keeps it; only `url|norm` falls through
  // to the domain branch below, which is where the bare-root lookalike-domain
  // report has always lived and still does.
  if (subject && subject.platform !== "url" && subject.idKind !== "norm") return subject.canonical;
  const host = hostOf(rawUrl);
  if (!host) return null;
  try {
    return canonicalAccount("domain", "domain", host).canonical;
  } catch {
    return null;
  }
}

/** The hostname, mirroring reports.ts `domainFromUrl` (same length bound, same
 * scheme-tolerant parse). */
function hostOf(raw: string): string | null {
  const trimmed = typeof raw === "string" ? raw.trim() : "";
  if (!trimmed || trimmed.length > 2048) return null;
  const withScheme = /^[a-z][a-z0-9+.-]*:\/\//i.test(trimmed) ? trimmed : `https://${trimmed}`;
  try {
    return new URL(withScheme).hostname;
  } catch {
    return null;
  }
}

/**
 * Find the case ref for a subject in the fan's OWN list. Exact key match only —
 * never a fuzzy or positional match, because attaching a screenshot to the
 * wrong case would put a stranger's image in someone else's file.
 */
export function findCaseRef(rows: unknown, rawUrl: string): string | null {
  const want = canonicalSubjectKey(rawUrl);
  if (!want || !Array.isArray(rows)) return null;
  for (const row of rows as OwnReportDto[]) {
    if (row && row.canonical === want && typeof row.case_ref === "string" && row.case_ref.length > 0) return row.case_ref;
  }
  return null;
}

/** Per (item, account). MUST EQUAL evidence.ts `MAX_EVIDENCE_PER_REPORT` —
 * pinned against the server's own source by test, because a client that
 * promises five and is refused at four tells a person their upload failed for

export const MAX_EVIDENCE = 4;
/** 4 MiB. MUST EQUAL evidence.ts `MAX_EVIDENCE_BYTES` (which the route's body
 * limit also enforces) — pinned against the server's source by test. Checking
 * it client-side saves a doomed 4 MiB upload but is NOT the gate: the server

export const MAX_EVIDENCE_BYTES = 4 * 1024 * 1024;
/** The figure the COPY quotes, derived from the cap rather than typed out
 * beside it — a hand-written "4 MB" is exactly how a stated limit drifts away
 * from the enforced one after somebody changes the constant. */
export const MAX_EVIDENCE_MB = MAX_EVIDENCE_BYTES / (1024 * 1024);

export type EvidenceKind = "attached" | "too_large" | "too_many" | "bad_type" | "unreadable" | "no_case" | "signed_out" | "unavailable" | "failed";

export interface EvidenceOutcome {
  kind: EvidenceKind;
  /** THE load-bearing field. True ONLY when the server told us the bytes are
   * stored. Every failure — including the fail-closed 503 — is false, so no
   * caller can render "attached ✓" for something that was never accepted. */
  attached: boolean;
  headline: string;
  body: string;
}

const NOT_ATTACHED = "The screenshot was not attached.";

/**
 * The upload failure ladder, pure.
 *
 * The 503 branch is the one that matters most and is asserted on its own:
 * `evidence_unavailable` means the evidence store is not configured on this
 * deployment, and the route refuses the bytes rather than half-storing them.
 * A client that showed a tick there would be inventing evidence in an
 * accusatory case file. It says the opposite, loudly, and offers the report
 * without the screenshot — which is a real, complete report.
 */
export function evidenceOutcome(r: { ok: boolean; status: number; error?: string | null }): EvidenceOutcome {
  if (r.ok && r.status >= 200 && r.status < 300) {
    return { kind: "attached", attached: true, headline: "Screenshot attached", body: "Only an operator reviewing this case can open it. There is no public link to it, and it is deleted when the case closes." };
  }
  if (r.status === 503 && r.error === "evidence_scan_unavailable") {
    // F1: the scan vendor did not answer. TRANSIENT — the deployment is
    // configured and the next attempt will usually work. Not "switched off".
    return {
      kind: "unavailable",
      attached: false,
      headline: "Couldn't check the screenshot just now",
      body: `${NOT_ATTACHED} Every screenshot is checked before Dhwaja stores it, and the check didn't answer in time. Your report itself is filed and complete — try attaching the screenshot again in a moment.`,
    };
  }
  if (r.status === 429) {
    return { kind: "unavailable", attached: false, headline: "Too many attempts for now", body: `${NOT_ATTACHED} Screenshot uploads are limited per hour. Your report is filed; try the screenshot again later.` };
  }
  if (r.status === 503) {
    return {
      kind: "unavailable",
      attached: false,
      headline: "Screenshots are switched off here",
      body: `${NOT_ATTACHED} Evidence storage isn't configured on this deployment, so we refuse the file rather than half-store it — when Dhwaja can't do something correctly it switches it off instead of pretending. Your report itself is filed and complete; describe what the screenshot showed in the note instead.`,
    };
  }
  if (r.status === 413 || r.error === "too_large" || r.error === "image too large") {
    return { kind: "too_large", attached: false, headline: "That image is too big", body: `${NOT_ATTACHED} The limit is ${MAX_EVIDENCE_MB} MB per screenshot. A screen grab is usually well under that — a photo of a screen usually isn't. Crop it or take a fresh screenshot and try again.` };
  }
  if (r.status === 409) {
    return { kind: "too_many", attached: false, headline: `That's ${MAX_EVIDENCE} screenshots`, body: `${NOT_ATTACHED} You can attach up to ${MAX_EVIDENCE} images to one case, which is enough to show a profile, a message and a payment page. If something else matters, put it in the note.` };
  }
  if (r.status === 415) {
    return {
      kind: "bad_type",
      attached: false,
      headline: "That file isn't a PNG or JPEG we can read",
      body: `${NOT_ATTACHED} We check the file's actual contents, not its name — so renaming something to .png doesn't get it through, and a very large image can be refused even when it's under the size limit. Take a normal screenshot and try again.`,
    };
  }
  if (r.status === 400) {
    return { kind: "unreadable", attached: false, headline: "We couldn't read that file", body: `${NOT_ATTACHED} It arrived empty or damaged. Try attaching it again.` };
  }
  if (r.status === 401) {
    return { kind: "signed_out", attached: false, headline: "You're signed out", body: `${NOT_ATTACHED} Sign in again and you can attach it to the same case — your report is already filed.` };
  }
  if (r.status === 404) {
    return {
      kind: "no_case",
      attached: false,
      headline: "We can't attach to that case",
      body: `${NOT_ATTACHED} Either that case reference isn't one of yours or the case is already closed. This is the same answer for both, on purpose — we don't tell anyone which cases exist.`,
    };
  }
  return { kind: "failed", attached: false, headline: "That upload didn't go through", body: `${NOT_ATTACHED} Your report is unaffected — try the screenshot again in a moment. If it did arrive, sending the same screenshot again won't use up another slot.` };
}

/** The only client-side refusals, applied BEFORE a doomed upload. Both are
 * bounds the server enforces independently; returning null means "let the
 * server decide", which is the default for everything else including the file
 * type (magic bytes are the server's call, and a name-based guess here would be
 * a confident lie). */
export function evidencePrecheck(file: { size: number }, alreadyAttached: number): EvidenceOutcome | null {
  if (alreadyAttached >= MAX_EVIDENCE) return evidenceOutcome({ ok: false, status: 409 });
  if (!Number.isFinite(file.size) || file.size <= 0) return evidenceOutcome({ ok: false, status: 400 });
  if (file.size > MAX_EVIDENCE_BYTES) return evidenceOutcome({ ok: false, status: 413 });
  return null;
}

/** Is the picker spent? The count reaches the SAME cap the precheck and the
 * server use — never a separate literal that could drift below it and hide the
 * fourth slot, or above it and offer a fifth upload that 409s. */
export function evidenceSlotsSpent(attached: number): boolean {
  return attached >= MAX_EVIDENCE;
}

/**
 * The one line each attempt adds to the visible list.
 *
 * Extracted from the DOM controller so the wording is under test. `attached`
 * is the ONLY thing that decides which sentence is produced — the tick is
 * never rendered for an outcome the server did not accept, including the
 * fail-closed 503. The file name is attacker-supplied and is returned as TEXT
 * for a `textContent` assignment; nothing here builds markup.
 */
export function evidenceRowText(fileName: unknown, o: EvidenceOutcome): { text: string; cls: string } {
  // Clamped but NOT html-escaped: this string is assigned with `textContent`,
  // where escaping would render `&amp;` to the person who named the file.
  const raw = typeof fileName === "string" ? fileName : "";
  const name = raw.length > 120 ? `${raw.slice(0, 119)}\u2026` : raw;
  return o.attached ? { text: `${name} — attached \u2713`, cls: "ev ok" } : { text: `${name} — not attached \u00b7 ${o.headline}`, cls: "ev bad" };
}

export type ReportStatus = "received" | "reviewing" | "published" | "closed";

export interface LadderStep {
  key: ReportStatus;
  label: string;
  blurb: string;
  /** Position on the visible ladder. `closed` shares the last rung with
   * `published` because it is the OTHER way a case ends, not a later stage. */
  index: number;
}

// AS-129: `warnings.ts withdrawWarning` (:580) sends an escalated (reviewing)
// item back to `open`, and `reports.ts` (:735-739) re-opens a dismissed or
// retracted (closed) item for a fresh reporter — both silently, no
// notification (AS-030), and both land the fan back on "received".
// `listOwnReports` carries no field marking a case as reopened (api.ts
// `OwnReportDto`), so the client cannot tell a genuinely new case from a
// regressed one — the "received" blurb below has to be true of both, not just
// deny the case's own possible prior state.
export const REPORT_LADDER: readonly LadderStep[] = [
  {
    key: "received",
    label: "Received",
    blurb:
      "It's in the queue for an operator to review. Nothing is published automatically and nothing has been decided — a case can land back on this rung after being reviewed, or after closing, if it returns to the queue; that's not a step backward in what was found.",
    index: 0,
  },
  { key: "reviewing", label: "In review", blurb: "An operator has picked this case up and is looking at it by hand.", index: 1 },
  { key: "published", label: "Warning published", blurb: "A warning is live on that account. Anyone checking it now sees it. Thank you — this one mattered.", index: 2 },
  {
    key: "closed",
    label: "Closed",
    blurb:
      "This case ended without a published warning. We don't say why, and we don't distinguish between the reasons a case can close — every closed case reads exactly like this one. Closed doesn't mean the account is safe: it means this case produced no warning. Keep treating it as unverified.",
    index: 2,
  },
];

const LADDER_BY_KEY: Record<ReportStatus, LadderStep> = {
  received: REPORT_LADDER[0]!,
  reviewing: REPORT_LADDER[1]!,
  published: REPORT_LADDER[2]!,
  closed: REPORT_LADDER[3]!,
};

/**
 * Map a served status onto the ladder. THE COLLAPSE IS THE POINT.
 *
 * The server already folds dismissed / suppressed / retracted into "closed"
 * (reports.ts `fanReportStatus`). This function folds them again — plus every
 * unknown or future value, plus the raw internal states, plus junk — so that no
 * arrangement of server output can produce a client rung that distinguishes an
 * operator dismissal from a suppression. There is exactly one terminal rung and
 * exactly one string for it.
 */
export function reportStatusStep(status: unknown): LadderStep {
  if (status === "received" || status === "reviewing" || status === "published") return LADDER_BY_KEY[status];
  return LADDER_BY_KEY.closed;
}

/** Human label for a category we may or may not still ship. Unknown → the
 * neutral "Something else", matching the server's own defense-in-depth
 * re-validation, so a stale category can never render as raw text. */
export function categoryLabel(key: unknown): string {
  for (const c of CATEGORY_CHOICES) if (c.key === key) return c.label;
  return "Something else";
}

/**
 * The visible rungs. `published` and `closed` are the two ways a case ENDS, so
 * exactly one of them is ever shown — a closed case renders
 * Received → In review → Closed and never displays "Warning published" greyed
 * out, which would read as "we nearly did" about an account we decided nothing
 * about.
 */
function ladderHtml(step: LadderStep): string {
  const terminal = step.key === "closed" ? "closed" : "published";
  const cells = REPORT_LADDER.filter((s) => s.index < 2 || s.key === terminal)
    .map((s) => {
      const state = s.key === step.key ? "on" : s.index < step.index ? "done" : "todo";
      return `<li class="rung ${state}"${state === "on" ? ` aria-current="step"` : ""}><span class="rdot" aria-hidden="true"></span><span class="rlab">${xml(s.label)}</span></li>`;
    })
    .join("");
  return `<ol class="ladder" aria-label="Case progress">${cells}</ol>`;
}

/**
 * One case row.
 *
 * EVERY field here comes from `OwnReportDto` and there are only six of them.
 * There is no reporter, no reporter count, no priority, no operator note and no
 * evidence state, because none of those are in the payload and none of them may
 * ever be added to it. The status branch reads `reportStatusStep` and nothing
 * else — no second look at the raw string — so a terminal case cannot be
 * rendered two ways.
 */
export function reportRow(r: OwnReportDto): string {
  const step = reportStatusStep(r?.status);
  const ref = typeof r?.case_ref === "string" ? r.case_ref : "";
  const when = typeof r?.filed_at === "string" ? r.filed_at.slice(0, 10) : "";
  return `<li class="case s-${xml(step.key)}">
<div class="chead"><span class="ctarget"><span class="plat">${textClamp(r?.platform ?? "", 40)}</span> <span class="hnd">@${textClamp(r?.handle ?? "", 64)}</span></span><span class="cwhen">${xml(when)}</span></div>
<p class="ccat">You reported this as: <b>${xml(categoryLabel(r?.category))}</b></p>
${ladderHtml(step)}
<p class="cblurb">${xml(step.blurb)}</p>
<p class="cref">Case reference <code>${textClamp(ref, 44)}</code></p>
</li>`;
}

/** The "Your reports" page (mock 50). `reports === null` is an unreadable
 * answer, NOT an empty list — telling someone they have filed nothing when we
 * simply couldn't read would be the reassuring lie the copy law forbids. */
export function reportsView(reports: OwnReportDto[] | null): string {
  const head = `<section class="cases"><h2>Your reports</h2>
<p class="lede">Every report you've filed and where it got to. Reports are reviewed by a person; a warning is only ever published by an operator, never automatically and never on one report alone.</p>
<p class="anon">🛡 The account you reported is never told who reported it — not their name, not their handle, not how many people reported them. That is true on every case below and it does not change with the outcome.</p>
<p class="lede">You don't have to keep checking this page: when a warning is published or retracted, you're told — it appears under <a href="#/whats-new" data-nav="#/whats-new">What's new</a>. A case that closes without a warning sends nothing, so a case below with no notification is not a case nobody looked at.</p>`;
  const foot = `<p class="casesfine">Cases are shown newest first, up to the 50 most recent. Money can never change what a report counts for; a supporter account can only affect how soon a comparable case is looked at.</p></section>`;
  // A payload that isn't an array is UNREADABLE, not empty. Coerced here rather
  // than trusted: a malformed body must land on the honest "we couldn't load"
  // card, never throw and leave the page blank, and never read as "you have
  // filed nothing" (adversarial review — the same fail-soft rule views.ts uses).
  // The `unread` class is the machine-readable half of that rule: it is the one
  // marker `apps/web/test/failure-states.test.ts` looks for, so a failed render
  // and an empty render can never be byte-identical anywhere in the client.
  if (reports === null || !Array.isArray(reports)) {
    return `${head}<div class="empty unread">We couldn't load your reports just now. This is a display problem only — nothing about your cases has changed, and none of them have been closed or withdrawn. Try again in a moment.</div>${foot}`;
  }
  if (reports.length === 0) {
    return `${head}<div class="empty">You haven't filed any reports. If you run into an account pretending to be a creator, <a href="#/report" data-nav="#/report">report it</a> — it's free, it always will be, and a person reads it.</div>${foot}`;
  }
  // ORDER IS A CHANNEL. The rows are rendered in the order the server sent
  // them (newest first) and are NEVER re-sorted here. Sorting on anything the
  // client collapses — the raw `dismissed` / `suppressed` / `retracted` value
  // behind "Closed" — would hand back through position exactly the bit that
  // `reportStatusStep` exists to withhold, as surely as a different icon or a
  // different tone class would. Pinned by test.
  return `${head}<ul class="caselist">${reports.slice(0, 50).map(reportRow).join("")}</ul>${foot}`;
}

/** Step 3's screenshot panel. Rendered ONLY when a case ref was resolved — a
 * report with no readable case ref shows the honest note instead of an upload

/**
 * THE passport-id shape — one constant for the whole client, and by value the
 * same one the server's pointer validator and the other three clients use
 * (`PASSPORT_ID_RE` in services/trust-core/src/passport-view.ts;
 * `tests/invariants/safety-handoff-parity.test.ts` pins them together).
 *
 * It was three shapes in this repo, and the mismatch sat on the redirect a
 * frightened fan is offered after a Reported verdict: the registry builder and
 * every client accepted `psp_[0-9a-z]{8,64}` for the find-the-real-one pointer,
 * and THIS router — the thing that has to route it — required exactly 16 hex.
 * A widened id would have been served, rendered as "The real X is here →", and
 * silently collapsed to `{name:"home"}` on arrival. A shape mismatch must fail
 * at the validator, where the answer is "no pointer" and the warning still
 * stands, not at the router, where it is a broken promise.
 */
export const PASSPORT_ID_RE = /^psp_[0-9a-f]{16}$/;
export const PSP_RE = PASSPORT_ID_RE;
export const ACQ_RE = /^acq_[0-9a-f]{20}$/;
/** A release id, as `releases.ts getReleaseProvenance` accepts it (QUEUE #532). */
export const RELEASE_RE = /^rel_[0-9a-f]{16}$/;
// A reverse-code pool id (`rcp_<16hex>`, minted server-side). Validated before
// it ever routes so a junk fragment can't reach the offer/claim fetch.
export const POOL_RE = /^rcp_[0-9a-f]{16}$/;
// Dhwaja's OWN collector-username namespace (the `#/u/<handle>` page and the
// profile editor). The server enforces exactly this shape (social.ts), so it
// must stay strict — loosening it would route to a page that can never exist.
export const HANDLE_RE = /^[a-z0-9_]{3,20}$/;
// M5: the BARE-HANDLE creator-lookup front door accepts platform handles, not
// Dhwaja usernames — Instagram/TikTok/YouTube handles legally carry dots and
// hyphens and run to 30 chars, and the canonicalizer already accepts them, so
// the front door must not be the one place a legal handle is bounced to home.
// Kept separate from HANDLE_RE (above) so the collector namespace stays strict.
// Must start alphanumeric; 2..30 chars of [a-z0-9._-]. The lookup screen only
// canonicalizes the handle locally for a k-anon check — nothing is navigated to.
// PU-03 (S-070): the first character is `[a-z0-9_]`, not `[a-z0-9]`. An
// underscore-leading handle is a value this repository's OWN contracts already
// issue — `canonicalize("https://instagram.com/_nova")` and
// `canonicalAccount("instagram","handle","_nova")` both mint
// `v1|instagram|handle|_nova`, and HANDLE_RE above (the collector namespace)
// accepts it too — so the ADVERTISED front door was the one place a supported
// identifier was bounced, into "That link isn't a supported creator profile":
// a sentence about a link, at somebody who typed a handle. `.` and `-` stay
// out of the first position on purpose: a leading dot reads as a hostname and
// a leading hyphen as a flag, and neither is a handle any platform issues.
export const LOOKUP_HANDLE_RE = /^[a-z0-9_][a-z0-9._-]{1,29}$/;
/**
 * A CHECK RESULT IS A PLACE, and it used to be an event.
 *
 * `renderCheck` wrote the verdict straight into `root.innerHTML` with no hash
 * change, so the single most important screen in the product had no address:
 * Back left the page the fan came from, a refresh threw the verdict away, and
 * there was nothing to send someone. ("Kept separate from routeBody because a
 * check is an action result, not a routable location" — it is a location; it is
 * the location.)
 *
 * WHAT THE FRAGMENT MAY CARRY. A hash is never sent to a server (the reason this
 * whole app is hash-routed), so the subject stays on the device exactly as the
 * k-anonymous check promises. `#/lookup/<handle>` has carried a handle since
 * v14 on the same reasoning. Sharing such a link shares the subject with the
 * person you send it to, which is the fan's own act.
 *
 * WHY THE ID KIND IS IN THE ROUTE. Re-deriving the subject from (platform,
 * handle) alone means guessing the kind, and the three kinds hash DIFFERENTLY:
 * a YouTube channel id is case-significant (`pid`) while a handle is lowercased,
 * and a Tier-2 link is a whole URL (`url`). A guess would produce a different
 * hash from the check the fan actually ran and therefore a different verdict —
 * the exact class of quiet wrongness this route exists to make revisitable.
 */
export type CheckIdKind = "handle" | "pid" | "url";
const CHECK_ID_KINDS: ReadonlySet<string> = new Set(["handle", "pid", "url"]);
/** A platform key (or the synthetic `link`/`domain` buckets). */
export const CHECK_PLATFORM_RE = /^[a-z0-9_]{1,32}$/;
/** The canonicalizer's own ceiling for a stored link URL. */
const CHECK_ID_MAX = 2048;

// A taxonomy category key (matches the server's frozen vocabulary shape). A
// junk fragment never routes into a browse fetch.
export const CATEGORY_RE = /^[a-z_]{2,24}$/;
// S80 facet shapes: ISO 639-1 language (2 lowercase) / ISO 3166-1 alpha-2 region
// (2 uppercase). A junk facet is dropped, never routed.
export const LANG_RE = /^[a-z]{2}$/;
export const REGION_RE = /^[A-Z]{2}$/;
// A raw claim code (base32-ish, as issued). Validated before it ever routes so a
// junk fragment can't reach the redeem action.
export const CODE_RE = /^[A-Z2-7]{16,40}$/;
// A removal request id. It is the ONLY credential for the status page, so a
// junk fragment must never reach the status fetch — it routes to the bare
// "paste your reference" form instead (requestId: null), which is also what an
// unreadable reference lands on, so the two are the same page.
export const RMV_ID_RE = RMV_REF_RE;

/** #243b — a Creator War match id (war-match.ts `matchId`: `wm_` + 20 hex). */
export const WAR_MATCH_RE = /^wm_[0-9a-f]{20}$/;

/** Parse a location hash (e.g. "#/p/psp_...") into a Route. Unknown/malformed
 * hashes fall back to home — never throws, never trusts an unvalidated id. */
export function parseHash(hash: string): Route {
  const h = typeof hash === "string" ? hash.replace(/^#/, "") : "";
  const parts = h.split("/").filter((s) => s.length > 0);
  if (parts.length === 0) return { name: "home" };
  const [head, arg] = parts;
  switch (head) {
    case "p":
      return arg && PSP_RE.test(arg) ? { name: "passport", id: arg } : { name: "home" };
    case "u":
      return arg && HANDLE_RE.test(arg.toLowerCase()) ? { name: "profile", handle: arg.toLowerCase() } : { name: "home" };
    case "lookup":
      // The front-door disambiguation for a bare handle (v14): "@rileymoon"
      // must never silently land on a fan page in a trust product — this
      // screen puts the creator check first and labels the collector profile
      // as exactly that.
      return arg && LOOKUP_HANDLE_RE.test(arg.toLowerCase()) ? { name: "lookup", handle: arg.toLowerCase() } : { name: "home" };
    case "check": {
      // `#/check/<platform>/<idKind>/<encoded id>`. The id is percent-encoded by
      // `buildHash`, so it can hold the slashes and dots a link URL carries
      // without colliding with the route's own separator. Anything malformed is
      // home, never a half-built subject: a check nobody can reconstruct must
      // not render as a verdict about something else.
      const kind = parts[2];
      const raw = parts[3];
      if (!arg || !CHECK_PLATFORM_RE.test(arg) || !kind || !CHECK_ID_KINDS.has(kind) || !raw) return { name: "home" };
      let id: string;
      try {
        id = decodeURIComponent(raw);
      } catch {
        return { name: "home" }; // a malformed escape is not an id
      }
      // No whitespace and no control characters: the canonicalizer produces
      // neither, so a fragment carrying one is hand-built.
      if (id.length === 0 || id.length > CHECK_ID_MAX || /[\s\u0000-\u001f\u007f]/.test(id)) return { name: "home" };
      return { name: "check", platform: arg, idKind: kind as CheckIdKind, id };
    }
    case "c":
      return arg && ACQ_RE.test(arg) ? { name: "collectible", id: arg } : { name: "home" };
    case "release":
      return arg && RELEASE_RE.test(arg) ? { name: "release", id: arg } : { name: "home" };
    case "me":
      return { name: "me" };
    case "profile":
      // `#/profile` is the fan's OWN profile form; `#/u/<handle>` is somebody
      // else's public one. Two different surfaces, two different heads, so a
      // handle can never route into the editor.
      return { name: "profile-setup" };
    case "signin":
      // `#/signin/email` is the BACKUP path (a mailed one-time code), kept
      // under the sign-in head so it reads as what it is — a fallback on the
      // sign-in screen, not a second front door. Any other arg is the ordinary
      // provider screen, never a 404: an unroutable sign-in hash must still
      // land somewhere a person can sign in.
      return arg === "email" ? { name: "signin-email" } : { name: "signin" };
    case "security":
      return { name: "security" };
    case "codex":
      return { name: "codex" };
    case "redeem":
      return arg && CODE_RE.test(arg) ? { name: "redeem", code: arg } : { name: "home" };
    case "leaderboards":
      return arg && /^\d{4}-Q[1-4]$/.test(arg) ? { name: "leaderboards", season: arg } : { name: "leaderboards" };
    case "requested":
      return { name: "requested" };
    case "discover":
      return { name: "discover" };
    case "browse": {
      // arg may carry a facet query: "music?lang=es&region=US". Split it off, then
      // validate each facet to its ISO shape; anything junk becomes null (no filter).
      const [cat, query] = (arg ?? "").split("?");
      let lang: string | null = null;
      let region: string | null = null;
      if (query) {
        try {
          const q = new URLSearchParams(query);
          const l = q.get("lang");
          const r = q.get("region");
          if (l && LANG_RE.test(l)) lang = l;
          if (r && REGION_RE.test(r)) region = r;
        } catch {
          /* malformed query → no facets */
        }
      }
      return { name: "browse", category: cat && CATEGORY_RE.test(cat) ? cat : null, lang, region };
    }
    case "demand":
      return arg && PSP_RE.test(arg) ? { name: "demand", id: arg } : { name: "home" };
    case "codes":
      return arg && POOL_RE.test(arg) ? { name: "codes", poolId: arg } : { name: "home" };
    case "whats-new":
      return { name: "whats-new" };
    case "upgrade":
      return { name: "upgrade" };
    case "wallet":
      return { name: "wallet" };
    case "store":
      return { name: "store" };
    case "odds":
      return { name: "odds" };
    case "collection":
      return { name: "collection" };
    case "war":
      // #243b: `#/war/match/<wm_id>` is one match's replay; anything else under
      // `#/war/` is the matches page, never a half-built replay.
      if (arg === "match" && parts[2] && WAR_MATCH_RE.test(parts[2])) return { name: "war-match", id: parts[2] };
      // S-CW5. The match loop is six live server routes (war-routes.ts:104-166)
      // that no fragment could reach: the feature shipped with no door.
      return { name: "war" };
    case "settings":
      return { name: "settings" };
    case "safety":
      return { name: "safety" };
    case "report":
      return { name: "report" };
    case "reports":
      return { name: "reports" };
    case "claim":
      return { name: "claim" };
    case "remove":
      return arg === "dispute" ? { name: "remove", intent: "contest" } : { name: "remove" };
    case "removal":
      // A malformed reference is NOT an error state: it lands on the same page
      // with an empty field. Refusing it loudly would be a (weak) oracle about
      // reference shape, and the person holding a real reference has enough to
      // do already.
      return { name: "removal", requestId: arg && RMV_ID_RE.test(arg) ? arg : null };
    case "contest":
      return { name: "contest" };
    case "recovery":
      return { name: "recovery" };
    case "takeover":
      return { name: "takeover" };
    case "nominate":
      return { name: "nominate" };
    case "terms":
      return { name: "terms" };
    case "privacy":
      return { name: "privacy" };
    case "how-it-works":
      return { name: "how-it-works" };
    case "ledger":
      return { name: "ledger" };
    default:
      return { name: "home" };
  }
}

/**
 * WHERE A FAN GOES BACK TO AFTER SIGNING IN.
 *
 * The controller remembers the hash it bounced from; this decides whether that
 * hash may be returned to. It used to be `hash.startsWith("#/redeem/")` and
 * `#/me` for everything else — so the return location was captured and then
 * thrown away for every gated action except one:
 *
 *   - "Save the real <creator>" after a Flagged verdict — the highest-intent
 *     button in the product — dropped the fan on an empty collection page with
 *     the creator's name gone.
 *   - a creator's `#/codes/<pool>` gift link: sign in, land on `#/me`, and
 *     nothing on that page mentions the offer they were sent.
 *   - Collect, Follow and "ask for their back catalogue", all the same.
 *
 * PARSE, THEN REBUILD — never return the caller's string. `parseHash` validates
 * every id and handle shape and collapses anything it does not recognise to
 * `home`, and `buildHash` re-emits the canonical form, so a hand-crafted
 * fragment cannot ride through this function into `window.location`.
 *
 * Two names are refused: the sign-in screens themselves, which would bounce a
 * freshly signed-in fan back into a flow they have just completed. `home` is
 * not a destination worth returning to either — a fan who signed in from the
 * front page is better served by their collection, which is the fallback.
 */
const NO_RETURN: ReadonlySet<Route["name"]> = new Set<Route["name"]>(["signin", "signin-email", "home"]);

export function safeReturnHash(hash: string, fallback = "#/me"): string {
  if (typeof hash !== "string" || hash.length === 0) return fallback;
  const route = parseHash(hash);
  return NO_RETURN.has(route.name) ? fallback : buildHash(route);
}

/** Build the canonical hash for a Route (the inverse of parseHash). */
export function buildHash(route: Route): string {
  switch (route.name) {
    case "passport":
      return `#/p/${route.id}`;
    case "profile":
      return `#/u/${route.handle}`;
    case "lookup":
      return `#/lookup/${route.handle}`;
    case "check":
      // The id is percent-encoded: a Tier-2 link's id is a whole URL, so it
      // carries the `/` this route splits on. Encoding is what lets the check
      // result be a place without the fragment becoming ambiguous.
      return `#/check/${route.platform}/${route.idKind}/${encodeURIComponent(route.id)}`;
    case "collectible":
      return `#/c/${route.id}`;
    case "release":
      return `#/release/${route.id}`;
    case "me":
      return "#/me";
    case "profile-setup":
      return "#/profile";
    case "signin":
      return "#/signin";
    case "signin-email":
      return "#/signin/email";
    case "security":
      return "#/security";
    case "codex":
      return "#/codex";
    case "redeem":
      return `#/redeem/${route.code}`;
    case "leaderboards":
      return route.season ? `#/leaderboards/${route.season}` : "#/leaderboards";
    case "requested":
      return "#/requested";
    case "discover":
      return "#/discover";
    case "browse": {
      if (!route.category) return "#/browse";
      const q = new URLSearchParams();
      if (route.lang && LANG_RE.test(route.lang)) q.set("lang", route.lang);
      if (route.region && REGION_RE.test(route.region)) q.set("region", route.region);
      const qs = q.toString();
      return qs ? `#/browse/${route.category}?${qs}` : `#/browse/${route.category}`;
    }
    case "demand":
      return `#/demand/${route.id}`;
    case "codes":
      return `#/codes/${route.poolId}`;
    case "whats-new":
      return "#/whats-new";
    case "upgrade":
      return "#/upgrade";
    case "wallet":
      return "#/wallet";
    case "store":
      return "#/store";
    case "odds":
      return "#/odds";
    case "collection":
      return "#/collection";
    case "war":
      return "#/war";
    case "war-match":
      return `#/war/match/${route.id}`;
    case "settings":
      return "#/settings";
    case "safety":
      return "#/safety";
    case "report":
      return "#/report";
    case "reports":
      return "#/reports";
    case "claim":
      return "#/claim";
    case "remove":
      return route.intent === "contest" ? "#/remove/dispute" : "#/remove";
    case "removal":
      // Fragment-scoped, via the one builder — a reference never reaches a
      // query string, not even through the route round-trip.
      return referenceHref(route.requestId) ?? "#/removal";
    case "contest":
      return "#/contest";
    case "recovery":
      return "#/recovery";
    case "takeover":

    case "nominate":
      return "#/nominate";

      return "#/terms";
    case "privacy":
      return "#/privacy";
    case "how-it-works":
      return "#/how-it-works";
    case "ledger":
      return "#/ledger";

/**
 * The recovery request outcome, pure.
 *
 * `deduped` — the server folding your request into an already-open case for the
 * same passport — is deliberately NOT surfaced as a different message. Telling
 * a requester "somebody already reported this" is a small oracle about other
 * people's actions on an account that may not be theirs, and it changes nothing
 * about what happens next. Both land on the same "received" card.
 */
/**
 * M27 — the four words a recovery filing can be in, put into sentences.
 *
 * `with_an_operator` COVERS TWO WORLDS AND MUST KEEP COVERING THEM: a case that
 * is genuinely open, and a filing that matched no passport at all. That is not a
 * gap in this copy, it is the reason the reference is safe to hand out — a
 * distinct "nothing here" would answer, one OTP later, the exact question
 * `POST /creator/recovery/start` spends its whole implementation refusing to
 * answer (whether an address is a verified creator contact). So this sentence
 * must never promise a case exists, and the dead-end case is told by email
 * instead, to an inbox whose control was just proved.
 */
export function recoveryStatusOutcome(r: { ok: boolean; status: number; data: { status?: string; error?: string } | null }): { headline: string; body: string; tone: "good" | "quiet" } {
  if (r.status === 429) {
    return { tone: "quiet", headline: "Too many checks just now", body: "Give it an hour. Nothing about your case changed — this limit is on looking, not on filing." };
  }
  if (!r.ok || r.status !== 200 || typeof r.data?.status !== "string") {
    // A 404 here is NOT evidence the reference is wrong: the server answers
    // unknown, malformed and pruned references identically, on purpose. So this
    // says what is true (we can't show you anything) and not what is tempting
    // (no such reference).
    return { tone: "quiet", headline: "We can't show you anything for that reference", body: "Check for a typo. A reference is trk_ followed by thirty-two characters. If it is right and this keeps happening, reply to the email we sent you — a person can look it up." };
  }
  switch (r.data.status) {
    case "resolved":
      return { tone: "good", headline: "An operator has resolved it", body: "Whatever was going to be done has been done. If you still can't get in, file again and say so — that is new information, not a repeat." };
    case "closed":
      return { tone: "quiet", headline: "An operator closed it without action", body: "That usually means there wasn't enough to act on, or the account was already fine. If circumstances changed, file again and describe what's different." };
    case "mixed":
      return { tone: "quiet", headline: "Different outcomes", body: "This reference covered more than one passport and they didn't all end the same way. Reply to the email we sent you and a person will tell you which is which." };
    default:
      return { tone: "quiet", headline: "Still with an operator", body: "Nobody has made a decision yet. Recovery is a human review and there is no queue position to give you — checking more often will not move it. You'll hear back at the address you proved." };
  }
}

export function recoveryOutcome(r: { ok: boolean; status: number; data: { received?: boolean } | null }): RecoveryOutcome {
  if (r.ok && r.status === 200 && r.data?.received === true) {
    return {
      kind: "received",
      received: true,
      headline: "Received. A person takes it from here.",
      body: "Recovery is never automatic on Dhwaja and never self-service — if it were, the fastest way to steal an account would be to claim you'd lost it. An operator reads this, decides what should happen, and acts. You don't choose the remedy and neither do we from this form.",
    };
  }
  if (r.status === 401) {
    return { kind: "signed_out", received: false, headline: "Sign in first", body: "This form needs an account so a person has somebody to answer. It does not need to be the account in trouble — you can raise this on someone else's behalf. Nothing was sent." };
  }
  if (r.status === 404) {
    return { kind: "no_passport", received: false, headline: "We couldn't find that passport", body: "Check the psp_… id, or the platform and handle. Nothing was sent. If the passport is listed but this form can't find it, use remove or correct a listing instead — that path takes a plain profile link." };
  }
  if (r.status === 429) {
    return { kind: "capped", received: false, headline: "That's your recovery requests for today", body: "There's a daily limit so this queue can't be flooded. Nothing was sent just now. If you already sent one about this passport, it is in the queue and being read — sending more would not move it." };

 * Fan session HANDLE store (I25-W1-COOKIE, I3-0714 a). The fan session itself is an HttpOnly cookie trust-core sets on
 * sign-in (services/trust-core/src/csrf.ts; docs/SESSION-COOKIES.md): page script never sees it. What a sign-in hands

 * the cookie. Storage is INJECTED and the browser bootstrap passes MEMORY (main.ts): nothing about a session is written
 * to localStorage; a reload asks GET /session/csrf for the handle again (api.ts syncSession).

/** The Authorization value for a held handle (trust-core csrf.ts CSRF_SCHEME). */
export function sessionAuthorization(handle: string): string {

export interface KVStore {
  get(key: string): string | null;
  set(key: string, value: string): void;
  remove(key: string): void;
}

/** In-memory KV (default; also the artifact-safe fallback when no storage). */
export function memoryStore(): KVStore {
  const m = new Map<string, string>();
  return {
    get: (k) => m.get(k) ?? null,
    set: (k, v) => void m.set(k, v),
    remove: (k) => void m.delete(k),
  };
}

const KEY = "dhwaja.fan.session";
/** RG-03 (D-1010(1)): the key the pre-I25 build kept a live fan BEARER under in localStorage; main.ts removes it at start. */
export const LEGACY_SESSION_KEY = KEY;
// A handle is `fan.<hex>` (a pre-cookie token was `<b64url>.<b64url>`, which this shape also admits; the server refuses
// either one that is not the live cookie's handle) — validate the SHAPE before a stored value reaches a header.
const TOKEN_RE = /^[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+$/;

export class FanSession {
  constructor(private readonly store: KVStore) {}

  get(): string | null {
    const t = this.store.get(KEY);
    return t && TOKEN_RE.test(t) ? t : null;
  }

  set(token: string): void {
    if (typeof token === "string" && TOKEN_RE.test(token)) this.store.set(KEY, token);
  }

  clear(): void {
    this.store.remove(KEY);
  }

  isSignedIn(): boolean {
    return this.get() !== null;
  }
}

export function mintClientSeed(): string {
  const b = new Uint8Array(16);
  crypto.getRandomValues(b);
  if (b.every((x) => x === 0)) throw new Error("client seed unavailable: the random-number generator returned no entropy");
  return [...b].map((x) => x.toString(16).padStart(2, "0")).join("");
}

/**
 * Ruling (d), S-022/ODDS — "odds surface before any pack is SOLD, in a market
 * that REQUIRES it." No market/region signal reaches this surface today (the
 * pack routes read no geo header — `CF-IPCountry` is read for a DIFFERENT
 * feature, `fan-social-routes.ts`'s encounter recording, and is never plumbed
 * into `commitPack`/`openPack`; see ODDS/REPORT.md). Rather than invent a geo
 * lookup this surface has no signal to feed, the gate below applies the
 * disclosure to EVERY sale, unconditionally — a strict superset of "every
 * requiring market", so it can never under-disclose wherever a fan happens to
 * be, and it costs a fan in a non-requiring market nothing but a read.
 *
 * FAILS CLOSED ON THE OTHER HALF TOO: "no computable odds ⇒ no sale." A
 * commitment is the ONLY thing that can make the button appear (`odds` below
 * comes from nowhere else), and `commitPack` never mints one without a
 * disclosure (services/trust-core/src/packs.ts). So `odds === null` — commit
 * never landed, or landed without a disclosure the client can validate — is
 * treated as "packs are not for sale", not as "show the button anyway and let
 * the click fail".
 */
/** I3-0415 — a disclosed entry as this view reads it: the DTO's fields plus an optional creator name. */
export type PackOddsEntryView = PackOddsDisclosureDto["entries"][number] & { display_name?: string };
export function parsePackOdds(raw: unknown): (PackOddsDisclosureDto & { entries: PackOddsEntryView[] }) | null {
  if (!raw || typeof raw !== "object") return null;
  const o = raw as { week?: unknown; content_hash?: unknown; cards_per_pack?: unknown; entries?: unknown };
  if (typeof o.week !== "string" || typeof o.content_hash !== "string" || typeof o.cards_per_pack !== "number") return null;
  if (!Array.isArray(o.entries) || o.entries.length === 0) return null;
  const entries: PackOddsEntryView[] = [];
  for (const e of o.entries) {
    if (!e || typeof e !== "object") return null;
    const { passport_id, percent_bp, display_name } = e as { passport_id?: unknown; percent_bp?: unknown; display_name?: unknown };
    if (typeof passport_id !== "string" || !PSP_RE.test(passport_id) || typeof percent_bp !== "number") return null;
    // I3-0415 — an optional creator name rides beside the id (decoration only; the committed hash covers ids and
    // odds). Shape-checked and bounded; anything else is dropped and the id is shown, never a guess.
    const name = typeof display_name === "string" ? display_name.trim() : "";
    entries.push(name.length > 0 && name.length <= 80 && !/[\u0000-\u001f\u007f]/.test(name) ? { passport_id, percent_bp, display_name: name } : { passport_id, percent_bp });
  }
  // "SUM correctly": a disclosure whose basis points don't sum to 10000 is not
  // a disclosure this view will show as one — malformed odds are the same as
  // no odds (fail closed), never a percentages-don't-add-up screen.
  if (entries.reduce((s, e) => s + e.percent_bp, 0) !== 10000) return null;
  return { week: o.week, content_hash: o.content_hash, cards_per_pack: o.cards_per_pack, entries };
}

/** PS-276 — the odds + contents disclosure, rendered ABOVE the buy control so

    })
    .join("");
  return `<details class="oddsbox" data-pack-odds open>
<summary>Pack odds &amp; contents (${xml(String(o.entries.length))} creators, week ${xml(o.week)})</summary>
<p class="wbody">Each of the ${xml(String(o.cards_per_pack))} base cards in this pack is drawn independently, with replacement, at the chance shown below. <p class="fine">This week's lineup runs until Monday 00:00 UTC.</p>
</details>`;
}

/**
 * The reveal — the cards at once (calm), the variant marked, the new balance,
 * and a collapsed "verify this draw" panel.
 *
 * UI-AUDIT F-P4, second pass — THE CLAIM IS TRUE NOW, SO THE PANEL MAKES IT.
 *
 * The original copy said "The server committed to its secret seed BEFORE you
 * opened … nobody could have steered it after the fact." That was false:
 * `openPack` minted its seed inside the call, after the client seed arrived. The
 * first fix rewrote the copy to what was actually true, and recorded the real
 * commitment as an open decision rather than smuggling it in behind a sentence.
 *
 * The commitment now exists. `POST /account/war/pack/commit` fixes the seed and
 * publishes only its hash when the store page loads; the open reveals the seed,
 * and the controller — which has been holding that hash since before it
 * generated the fan's seed — checks the reveal against it. So the panel says the
 * strong thing, and every clause in it is now something the reader can check:
 *
 *   • the hash was in their browser before their seed existed;
 *   • the revealed seed hashes to it;
 *   • (I3-0235) NOT "the draw replays to these cards": no verifier ships, so
 *     the panel no longer says it;
 *   • the odds table hash matches the published week.
 *
 * AND IF THE CHECK FAILS, THE PANEL SAYS SO INSTEAD. `commitment_broken` renders
 * a warning in place of the fairness copy. A verification panel that would
 * reassure even when its own check failed is the same defect as a claim nobody
 * could check — worse, because it looks like diligence.
 */
/** I3-0532 — the broken-commitment reveal's report affordance: a Copy pack id
 * control (only for a safe token-shaped id; main.ts binds the click) and a
 * "Report this pack" line. `support_to` (I3-1166 rule) is linked as mailto ONLY
 * when it is a plain address — anything else (the operator placeholder, a value
 * carrying `?`/headers) is plain escaped text; absent -> "our support inbox". */
const PACK_ID_RE = /^[A-Za-z0-9_-]{1,64}$/;
const PLAIN_EMAIL_RE = /^[A-Za-z0-9._%+-]+@[A-Za-z0-9-]+(\.[A-Za-z0-9-]+)*\.[A-Za-z]{2,}$/;
function reportPackLine(packId: string, supportTo: unknown): string {
  const subject = `Pack commitment mismatch ${packId}`;
  const copy = PACK_ID_RE.test(packId) ? `<button type="button" data-act="copy-pack-id" data-pack-id="${xml(packId)}" class="ds-btn act">Copy pack id</button>` : "";
  const to = typeof supportTo === "string" ? supportTo.trim() : "";
  const inbox = to && PLAIN_EMAIL_RE.test(to)
    ? `<a href="mailto:${xml(to)}?subject=${xml(encodeURIComponent(subject))}">${xml(to)}</a>`
    : to ? xml(to) : "our support inbox";
  return `<p class="revnote">${copy} Report this pack to ${inbox} — use the subject “${xml(subject)}”.</p>`;
}

export function packRevealView(r: PackResultDto): string {
  // F-P2: every field is treated as unread until it proves otherwise. This whole
  // view is assigned with `resultHolder.innerHTML = packRevealView(r)`, so a
  // throw in here loses a pack the fan HAS PAID FOR — the credits are gone, the
  // cards are minted, and the screen that would have shown them never renders.
  const cards = (Array.isArray(r.cards) ? (r.cards as unknown[]) : []).filter((c): c is { passport_id: string; variant?: boolean } => !!c && typeof c === "object" && typeof (c as { passport_id?: unknown }).passport_id === "string");
  // I3-0439: fold repeats of the same passport+variant into one tile with a
  // ×N badge instead of N identical tiles — order-preserving on first sight.
  const cardCounts = new Map<string, { passport_id: string; variant: boolean; count: number }>();
  for (const c of cards) {
    const variant = c.variant === true;
    const key = `${c.passport_id}|${variant}`;
    const existing = cardCounts.get(key);
    if (existing) existing.count += 1;
    else cardCounts.set(key, { passport_id: c.passport_id, variant, count: 1 });
  }
  const grid = Array.from(cardCounts.values()).map((c) => cardImg(c.passport_id, c.variant, c.count)).join("");
  const withheld = typeof r.withheld === "number" && r.withheld > 0 ? r.withheld : 0;
  const hex = (v: unknown): string => (typeof v === "string" && /^[0-9a-fA-F]{8,128}$/.test(v) ? v : "");
  const seed = hex(r.server_seed);
  const seedHash = hex(r.server_seed_hash);
  const oddsHash = hex(r.odds_hash);
  const clientSeed = hex(r.client_seed);
  const committed = hex(r.committed_hash);
  const packId = typeof r.pack_id === "string" ? r.pack_id : "";

  let verify: string;
  if (r.commitment_broken === true) {
    // The one outcome that must never be dressed as a fairness panel.
    verify = `<details class="verify" open><summary>⚠ This draw does not match its commitment</summary>
<p><b>The seed we were given before this pack was opened does not match the seed revealed with it.</b> That is the one thing this check exists to catch, and it means the draw cannot be treated as verified. You were charged for this pack as usual, the cards it gave you are yours, and nothing here is being taken back — but please report this, with the pack id below, before opening another.</p>
<dl class="vpairs">
<dt>Hash we held before opening</dt><dd><code>${xml(committed || "—")}</code></dd>
<dt>Hash returned with the reveal</dt><dd><code>${xml(seedHash || "—")}</code></dd>
<dt>Pack id</dt><dd><code>${xml(packId)}</code></dd>
</dl>${reportPackLine(packId, r.support_to)}</details>`;
  } else if (seed && seedHash && oddsHash) {
    // I3-0235: link the week's published odds table (GET /war/pack/odds/<week>,
    // war-routes.ts) so the odds hash is checkable; a missing or junk week links
    // the current table's path as text, never a URL built from an unchecked value.
    const week = typeof r.edition_week === "string" && /^\d{4}-W\d{2}$/.test(r.edition_week) ? r.edition_week : "";
    const oddsPath = week ? `/war/pack/odds/${week}` : "/war/pack/odds";
    const oddsLink = apiBase()
      ? `<a href="${xml(apiBase())}${oddsPath}" target="_blank" rel="noopener noreferrer">the published table${week ? ` for ${xml(week)}` : ""}</a>`
      : `the published table at <code>${xml(oddsPath)}</code>`;
    const committedLine = committed
      ? `<p><b>Checked for you already:</b> this browser held <code>${xml(committed)}</code> before it generated your seed, and the seed revealed above hashes to exactly that. The rest is here so you can repeat the check yourself rather than take ours.</p>`
      : "";
    verify = `<details class="verify"><summary>Verify this draw</summary>
<p><b>The server fixed its secret seed before you opened, and published only the hash of it.</b> Your device then added a seed of its own, and this week's odds table was already published and hashed. Because the server was committed before it could see your seed, it could not have tried seeds until it liked the result.</p>
<p>Two things here are yours to check: the revealed server seed hashes to the commitment, and the odds table hash matches ${oddsLink}. A tool that re-runs the draw itself isn't available yet.</p>
${committedLine}
<dl class="vpairs">
<dt>Server seed (revealed)</dt><dd><code>${xml(seed)}</code></dd>

${clientSeed ? `<dt>Your seed</dt><dd><code>${xml(clientSeed)}</code></dd>` : ""}

<dt>Pack id</dt><dd><code>${xml(packId)}</code></dd>
</dl></details>`;
  } else {
    verify = `<details class="verify"><summary>Verify this draw</summary>
<p>The replay values for this pack didn't come back, so there is nothing here to check against. Your cards are unaffected — this is a missing readout, not a missing draw.</p></details>`;
  }

  const withheldNote = withheld > 0
    ? `<p class="revnote">${xml(String(withheld))} of this pack's draws ${withheld === 1 ? "was" : "were"} for a creator who has since left the game or asked to be delisted, so ${withheld === 1 ? "that card was" : "those cards were"} not created — and you were not charged for ${withheld === 1 ? "it" : "them"}. The draw itself is unchanged and still verifiable below.</p>`
    : "";
  return `<div class="reveal">
<div class="revealhead"><b>You opened a pack</b><span class="revbal">${typeof r.balance === "number" ? `Balance: ${xml(String(r.balance))} credits` : "Balance: not shown"}</span></div>
<div class="gcards">${grid}</div>
<p class="revnote">The ✦ Foil is your collector's variant. Tap any card to see the creator.</p>
${withheldNote}
${verify}
<button data-act="open-pack" data-confirm-tap="1" class="ds-btn act">Open another — ${xml(PACK_PRICE_TEXT)}</button>
</div>`;
}

/**
 * PS-238 — "the fan can see what they have spent on packs." Rung 1 held: the
 * data is `war_packs` rows the fan's own opens already write
 * (`packSpendHistory`, own-scoped by `account_id`); this renders them, nothing

 * matching `collectionView`'s F-W6 convention.

// I3-0366 — the server default (`packSpendHistory`, services/trust-core/src/packs.ts:902)
// returns at most 100 rows, newest first, and the route never sends a count of
// how many rows exist beyond that page. `rows.length === PACK_SPEND_HISTORY_PAGE`
// is therefore the only signal this view has that there may be more history
// than it was handed — and until now it rendered that exact case as a
// confident "N packs opened, N credits total", which is a LIFETIME claim the
// data cannot support once the page is full.
const PACK_SPEND_HISTORY_PAGE = 100;

// I3-1478: the history row shows a plain date and a readable week, never the
// ISO instant or the `2026-W36` key (gifts.ts `e.at.slice(0, 10)` convention).
function spendDate(at: string): string {
  return /^\d{4}-\d{2}-\d{2}/.test(at) ? at.slice(0, 10) : "";
}
function spendWeek(week: string): string {
  const m = /^(\d{4})-W(\d{2})$/.exec(week);
  return m ? `Week ${Number(m[2])}, ${m[1]}` : "";
}

export function packSpendHistoryView(rows: PackSpendRowDto[] | null): string {
  if (!Array.isArray(rows)) {
    return `<div class="empty">We couldn't read your spend history just now — this isn't "you've spent nothing", just a read that didn't land. Nothing has been removed and nothing has changed. Try again in a moment.</div>`;
  }
  if (rows.length === 0) {
    return `<div class="empty">No packs opened yet. <a href="#/store" data-nav="#/store">Open one</a> to see it here.</div>`;
  }
  const total = rows.reduce((s, r) => s + r.price_credits, 0);
  const lines = rows
    .map((r) => `<li class="spendrow"><span class="spendweek">${xml(spendWeek(r.edition_week))}</span><span class="spendamt">${xml(String(r.price_credits))} credits</span><span class="spendat">${xml(spendDate(r.opened_at))}</span></li>`)
    .join("");
  const truncated = rows.length >= PACK_SPEND_HISTORY_PAGE;

/** The seven Power Rating dimensions in engine/card order, as the public ratings
 * read keys them (trust-core ratings.ts RATING_DIMENSIONS == war-engine STAT_NAMES
 * order, war-match.ts snapshotStats). Restated: apps/web cannot import trust-core. */
export const RATING_KEYS = ["reach", "momentum", "demand", "collector_base", "archive_depth", "community", "longevity"] as const;

/** The seven bars (integers 1..7) out of a `GET /p/:id/card/ratings` body, or
 * null when any one is missing or malformed — a partial card is not a card. */
export function ratingBars(raw: unknown): number[] | null {
  const r = raw && typeof raw === "object" ? (raw as { ratings?: unknown }).ratings : null;
  if (!r || typeof r !== "object") return null;
  const bars: number[] = [];
  for (const k of RATING_KEYS) {
    const bar = (r as Record<string, { bar?: unknown } | undefined>)[k]?.bar;
    if (typeof bar !== "number" || !Number.isInteger(bar) || bar < 1 || bar > 7) return null;
    bars.push(bar);
  }
  return bars;
}

/** #243c (D-0920) — PURE: for each of the seven stats, how many roster cards
 * have it as their strongest bar (a card with a tie at the top covers every
 * tied stat). 0 = a NAKED stat: no card in the roster leans on it. A specialist
 * roster (every card strongest in one stat) leaves six at 0; a balanced one
 * leaves none. A malformed card (not seven integers) counts toward nothing. */
export function coverage(roster: readonly (readonly number[])[]): number[] {
  const out = [0, 0, 0, 0, 0, 0, 0];
  for (const bars of roster) {
    if (!Array.isArray(bars) || bars.length !== out.length || bars.some((b) => typeof b !== "number" || !Number.isFinite(b))) continue;
    const top = Math.max(...bars);
    bars.forEach((b, i) => {
      if (b === top) out[i] = (out[i] ?? 0) + 1;
    });
  }
  return out;
}

/** The seven-bar strip with its text equivalent. `null` = the ratings did not
 * all read: said so, never drawn as seven naked stats. */
export function coverageView(cov: readonly number[] | null, rosterSize: number): string {
  if (!cov || cov.length !== STAT_LABELS.length) {
    return `<div class="coverage" data-coverage><div class="empty">We couldn't read every card's Power Ratings just now, so your roster's strengths aren't shown. This isn't an answer about your roster.</div></div>`;
  }
  const size = Math.max(1, rosterSize);
  const bars = cov
    .map((n, i) => {
      const pct = Math.round((Math.min(n, size) / size) * 100);
      return `<li class="covbar${n === 0 ? " naked" : ""}" data-stat="${xml(STAT_LABELS[i] as string)}" data-cover="${n}"><span class="covtrack" aria-hidden="true"><span class="covfill" style="height:${pct}%"></span></span><span class="covlbl">${xml(STAT_LABELS[i] as string)}</span></li>`;
    })
    .join("");
  const strong = cov.flatMap((n, i) => (n > 0 ? [`${STAT_LABELS[i]} (${n} card${n === 1 ? "" : "s"})`] : []));
  const naked = cov.flatMap((n, i) => (n === 0 ? [STAT_LABELS[i] as string] : []));
  const text = `Strongest in: ${strong.length ? strong.join(", ") : "none"}. ${naked.length ? `Naked (no card strongest here): ${naked.join(", ")}.` : "No naked stats — every Power Rating has a card that leans on it."}`;
  return `<div class="coverage" data-coverage><h4>Your roster's coverage</h4><p class="wbody">Each match round is fought on one of the seven Power Ratings. A bar shows how many of your ${size} practice cards are strongest in that rating; an empty bar is a stat nobody on your roster leans on.</p><ul class="covbars" aria-hidden="true">${bars}</ul><p class="covtext" data-coverage-text>${xml(text)}</p></div>`;
}

interface StandingRow { name: string; href: string | null; wins: number; losses: number }
const isCount = (x: unknown): x is number => typeof x === "number" && Number.isInteger(x) && x >= 0;

/** #240a/#240b (D-0920 option 3) — the public weekly board. 404 is "not built
 * yet"; any other failure, or a body that does not parse, is the unread card —
 * never an empty board stated as this week's. The fan board renders only when
 * the server served it (`fans_withheld` false and `fans` an array). */
export function warStandingsView(read: { status: number; data: unknown } | undefined): string {
  if (!read) return "";
  const head = `<section class="wstandings" data-war-standings><h3>This week's standings</h3>`;
  if (read.status === 404) return `${head}<div class="empty">This week's standings haven't been published yet.</div></section>`;
  const unread = `${head}<div class="empty unread">We couldn't read this week's standings just now — this isn't an answer about who's winning.</div></section>`;
  const d = read.data as { week?: unknown; as_of?: unknown; creators?: unknown; fans?: unknown; fans_withheld?: unknown } | null;
  if (read.status !== 200 || !d || typeof d !== "object" || typeof d.week !== "string" || !/^\d{4}-W\d{2}$/.test(d.week) || typeof d.as_of !== "string" || !Array.isArray(d.creators) || typeof d.fans_withheld !== "boolean") return unread;
  const creators: StandingRow[] = [];
  for (const c of d.creators as unknown[]) {
    const r = c as { passport_id?: unknown; display_name?: unknown; wins?: unknown; losses?: unknown } | null;
    if (!r || typeof r.passport_id !== "string" || !PSP_RE.test(r.passport_id) || typeof r.display_name !== "string" || !isCount(r.wins) || !isCount(r.losses)) return unread;
    creators.push({ name: r.display_name, href: `#/p/${r.passport_id}`, wins: r.wins, losses: r.losses });
  }
  let fans: StandingRow[] | null = null;
  if (!d.fans_withheld) {
    if (!Array.isArray(d.fans)) return unread;
    fans = [];
    for (const f of d.fans as unknown[]) {
      const r = f as { handle?: unknown; wins?: unknown; losses?: unknown } | null;
      if (!r || typeof r.handle !== "string" || r.handle.length === 0 || !isCount(r.wins) || !isCount(r.losses)) return unread;
      fans.push({ name: r.handle, href: null, wins: r.wins, losses: r.losses });
    }
  }
  const list = (rows: StandingRow[]): string =>
    `<ol class="wboard">${rows.map((r) => `<li>${r.href ? `<a href="${xml(r.href)}" data-nav="${xml(r.href)}">${xml(r.name)}</a>` : `<span class="wfan">${xml(r.name)}</span>`}<span class="wrec">${r.wins} won · ${r.losses} lost</span></li>`).join("")}</ol>`;

  return `${head}<p class="wbody">Ranked matches only, week ${xml(d.week.replace(/^(\d{4})-W(\d{2})$/, (_m, y: string, w: string) => `${Number(w)}, ${y}`))}${asOf ? ` — as of ${xml(asOf)}` : ""}. export type ShelfSort = "newest" | "creator" | "rarity";
/** I3-0436 — "rarity" in shelf terms (no rarity field exists on a card): a foil
 * collector's variant first, then pack pulls, direct buys, starters. */
const SOURCE_RARITY: Record<string, number> = { pack: 1, direct: 2, starter: 3 };
const rarityRank = (it: InstanceDto): number => (it.variant === true ? 0 : (SOURCE_RARITY[it.source] ?? 4));
const acquiredDesc = (a: InstanceDto, b: InstanceDto): number => (a.acquired_at < b.acquired_at ? 1 : a.acquired_at > b.acquired_at ? -1 : 0);

/** I3-0436 (D-0920 B) — PURE: the shelf sorted (newest | creator | rarity, ties
 * newest first) and, when `creator` is a well-formed passport id, filtered to
 * that creator. Never mutates its input. */
export function sortShelf(instances: readonly InstanceDto[], sort: ShelfSort, creator: string | null = null): InstanceDto[] {
  const pick = creator && PSP_RE.test(creator) ? instances.filter((i) => i.passport_id === creator) : [...instances];
  const cmp =
    sort === "creator"
      ? (a: InstanceDto, b: InstanceDto) => (a.passport_id < b.passport_id ? -1 : a.passport_id > b.passport_id ? 1 : acquiredDesc(a, b))
      : sort === "rarity"
        ? (a: InstanceDto, b: InstanceDto) => rarityRank(a) - rarityRank(b) || acquiredDesc(a, b)
        : acquiredDesc;
  return pick.sort(cmp);
}

function shelfControls(instances: readonly InstanceDto[], sort: ShelfSort, creator: string | null): string {
  const sorts: { key: ShelfSort; label: string }[] = [
    { key: "newest", label: "Newest first" },
    { key: "creator", label: "By creator" },
    // R3-I31-I21-Q118 (D-1091(4)): the approved label; true of rarityRank above (a foil, `variant === true`, ranks 0).
    { key: "rarity", label: "Foils first" },
  ];
  const sortOpts = sorts.map((s) => `<option value="${s.key}"${s.key === sort ? " selected" : ""}>${s.label}</option>`).join("");
  const creators = [...new Set(instances.map((i) => i.passport_id))].filter((p) => PSP_RE.test(p)).sort();
  const creatorOpts = [`<option value=""${creator ? "" : " selected"}>All creators</option>`, ...creators.map((p) => `<option value="${xml(p)}"${p === creator ? " selected" : ""}>${xml(p)}</option>`)].join("");
  return `<div class="shelfctl" data-shelf-controls><label>Sort <select data-shelf-sort>${sortOpts}</select></label><label>Creator <select data-shelf-creator>${creatorOpts}</select></label></div>`;
}

/** I3-0529 (D-0858(13)) — the practice bracket: war-match.ts BRACKETS[0],
 * restated because apps/web cannot import trust-core. */
export const PRACTICE_BRACKET = 10;

/** I3-0529 — the roster a PRACTICE join fields, chosen from the fan's shelf in
 * shelf order under the server's own join rules (war-match.ts validateJoinShape
 * and loadRoster): no staked (escrowed) card, one card per creator, exactly
 * PRACTICE_BRACKET cards. `null` when the shelf cannot field one. The server
 * stays the gate. D-0861(7) (RGC-02): `inGame`, when given, skips a creator it
 * rejects exactly like a staked card — app.ts `joinWarPractice` passes the
 * served in-game answer after the server refused a roster. */
export function practiceRoster(instances: InstanceDto[], inGame?: (passportId: string) => boolean): string[] | null {
  const seen = new Set<string>();
  const roster: string[] = [];
  for (const c of instances) {
    if (!c || c.escrowed === true || typeof c.instance_id !== "string" || typeof c.passport_id !== "string") continue;
    if (seen.has(c.passport_id)) continue;
    if (inGame && !inGame(c.passport_id)) continue;
    seen.add(c.passport_id);
    roster.push(c.instance_id);
    if (roster.length === PRACTICE_BRACKET) return roster;
  }
  return null;
}

/** I3-0529 — the practice join body. `practice: true` and NO `ante_instance`
 * key: no match stakes a card (QUEUE #236: war-match.ts validateJoinShape
 * refuses any body naming one) and practice moves no credits. Champion = the
 * first card. */
export function practiceJoinInput(instances: InstanceDto[], clientSeed: string, inGame?: (passportId: string) => boolean): { roster: string[]; champion_index: number; client_seed: string; practice: true } | null {
  const roster = practiceRoster(instances, inGame);
  return roster ? { roster, champion_index: 0, client_seed: clientSeed, practice: true } : null;
}

/** #241 (D-0920, I16-W05C) — one OPEN recon window, as trust-core war-match.ts
 * `ReconView` serves it (restated: apps/web cannot import trust-core). */
export interface WarReconDto {
  match_id: string;
  deadline: string;
  you_committed: boolean;
  opponent_committed: boolean;
  champion_index: number;
  your_roster: string[];
  opponent_size: number;
  opponent_reveal: { index: number; passport_id: string; stats: number[] }[];
  /** I21-Q30 (D-1030): each side's Bond per roster card, roster order, as the match will use it. */
  your_bonds: number[];
  opponent_bonds: number[];
  /** D-1059(2) (RGM-N7): the opponent's flag addend per roster card (null for an unrevealed slot); absent from an
   * older worker. */
  opponent_flags?: (number | null)[];
}

/** Bond's stored scale (trust-core war-bond.ts BOND_MAX); a match may use less (its dials' bondMax). */
const BOND_SCALE_MAX = 10;

/** #241 — PURE, fail-closed: a recon body or null. Every field is checked; a
 * reveal card that is not seven integers 1..7 or whose slot is out of range
 * makes the WHOLE window unreadable (a partial scouting report is not one). */
export function parseWarRecon(raw: unknown): WarReconDto | null {
  if (!raw || typeof raw !== "object") return null;
  const r = raw as Record<string, unknown>;
  if (typeof r.match_id !== "string" || !WAR_MATCH_RE.test(r.match_id)) return null;
  if (typeof r.deadline !== "string" || !Number.isFinite(Date.parse(r.deadline))) return null;
  if (typeof r.you_committed !== "boolean" || typeof r.opponent_committed !== "boolean") return null;
  const roster = r.your_roster;
  if (!Array.isArray(roster) || roster.length < 2 || !roster.every((p) => typeof p === "string" && PSP_RE.test(p))) return null;
  const ci = r.champion_index;
  if (typeof ci !== "number" || !Number.isInteger(ci) || ci < 0 || ci >= roster.length) return null;
  const size = r.opponent_size;
  if (typeof size !== "number" || !Number.isInteger(size) || size < 2) return null;
  const rev = r.opponent_reveal;
  if (!Array.isArray(rev) || rev.length !== Math.floor(size / 2)) return null;
  const reveal: WarReconDto["opponent_reveal"] = [];
  for (const c of rev) {
    const x = c && typeof c === "object" ? (c as Record<string, unknown>) : null;
    if (!x || typeof x.index !== "number" || !Number.isInteger(x.index) || x.index < 0 || x.index >= size) return null;
    if (typeof x.passport_id !== "string" || !PSP_RE.test(x.passport_id)) return null;
    const st = x.stats;
    if (!Array.isArray(st) || st.length !== STAT_LABELS.length || !st.every((b) => typeof b === "number" && Number.isInteger(b) && b >= 1 && b <= 7)) return null;
    reveal.push({ index: x.index, passport_id: x.passport_id, stats: st as number[] });
  }
  const bonds = (b: unknown, n: number): number[] | null => (Array.isArray(b) && b.length === n && b.every((v) => typeof v === "number" && Number.isInteger(v) && v >= 0 && v <= BOND_SCALE_MAX) ? (b as number[]) : null);
  const yourBonds = bonds(r.your_bonds, roster.length);
  const oppBonds = bonds(r.opponent_bonds, size);
  if (!yourBonds || !oppBonds) return null;
  // D-1059(2): absent -> no flags shown (an older worker); present but off-shape -> the whole window is unreadable.
  const flags = r.opponent_flags;
  if (flags !== undefined && !(Array.isArray(flags) && flags.length === size && flags.every((f) => f === null || (typeof f === "number" && Number.isInteger(f))))) return null;
  return { match_id: r.match_id, deadline: r.deadline, you_committed: r.you_committed, opponent_committed: r.opponent_committed, champion_index: ci, your_roster: roster as string[], opponent_size: size, opponent_reveal: reveal, your_bonds: yourBonds, opponent_bonds: oppBonds, ...(flags !== undefined ? { opponent_flags: flags as (number | null)[] } : {}) };
}

/** #241 — the RECON step: half the opponent's roster (which half is fixed by
 * the match's committed seed), a lead picker over your own roster, and the
 * deadline. Silence is safe and said so: your current champion leads. The
 * commit control is `data-act="war-recon-commit"` (bound in main.ts; app.ts passes `recon`). An
 * unreadable body says it could not read the window — never "no window". */
/** I21-Q30: one side's Bond for its whole roster (both sides see both, D-1030). Wording approved (I30-WAR; D-1365 W63). */
const bondLine = (label: string, bonds: readonly number[], side: "you" | "opponent"): string =>
  `<p class="wbody" data-war-bonds="${side}">${xml(label)}: ${bonds.map((b, i) => `${i + 1}: ${b}`).join(" · ")}</p>`;

/** D-1059(2) (RGM-N7): a revealed card's flag beside its Bond, in I30-WAR's replay wording ("+ n (flag)"); none at 0. */
const reconFlag = (f: number | null | undefined): string =>
  typeof f === "number" && f !== 0 ? ` <span class="wflag" data-war-flag>${f < 0 ? "-" : "+"} ${Math.abs(f)} (flag)</span>` : "";

export function warReconView(raw: unknown, nowMs: number): string {
  const r = parseWarRecon(raw);
  if (!r) return `<div class="empty" data-war-recon>We couldn't read this match's scouting window just now — this isn't an answer about whether one is open.</div>`;
  const left = Math.max(0, Math.ceil((Date.parse(r.deadline) - nowMs) / 1000));
  const cards = r.opponent_reveal
    .map((c) => {
      const strip = c.stats.map((b, i) => `${STAT_LABELS[i]} ${b}`).join(" · ");
      return `<li class="wreveal-card"><span class="wslot">Card ${c.index + 1} of ${r.opponent_size}</span> <a href="#/p/${xml(c.passport_id)}" data-nav="#/p/${xml(c.passport_id)}">${xml(c.passport_id)}</a> <span class="wstats">${xml(strip)}</span> <span class="wbond">Bond ${r.opponent_bonds[c.index]}</span>${reconFlag(r.opponent_flags?.[c.index])}</li>`;
    })
    .join("");
  const lead = r.your_roster[r.champion_index]!;
  let act: string;
  if (r.you_committed) {
    act = `<p class="wbody">Your lead is locked in: card ${r.champion_index + 1} (${xml(lead)}). ${r.opponent_committed ? "Your opponent has locked in too — the match is resolving." : "Waiting for your opponent."}</p>`;
  } else if (left === 0) {
    act = `<p class="wbody">The scouting window has closed. The match resolves with card ${r.champion_index + 1} (${xml(lead)}) as your lead.</p>`;
  } else {
    const options = r.your_roster
      .map((p, i) => `<option value="${i}"${i === r.champion_index ? " selected" : ""}>Card ${i + 1} — ${xml(p)} (Bond ${r.your_bonds[i]})</option>`)
      .join("");
    act = `<label class="wlead">Your lead <select data-war-lead="${xml(r.match_id)}">${options}</select></label><button data-act="war-recon-commit" data-match="${xml(r.match_id)}" class="ds-btn act primary">Lock in my lead</button><span data-war-recon-msg class="rmsg" aria-live="polite"></span>`;
  }
  return `<div class="wcard t-quiet" data-war-recon="${xml(r.match_id)}"><div class="whead">Scout your opponent</div><p class="wbody">You can see ${r.opponent_reveal.length} of their ${r.opponent_size} cards. Pick the card that leads your side. ${r.you_committed || left === 0 ? "" : `You have ${left} seconds; if you don't choose, your current champion (card ${r.champion_index + 1}) leads.`}</p><ul class="wreveal">${cards}</ul>${bondLine("Their Bond, card by card", r.opponent_bonds, "opponent")}${bondLine("Your Bond, card by card", r.your_bonds, "you")}${act}</div>`;
}

/** RJb-T2-AS-017 (I16-W09C H-3): what became of the caller's last queue entry, or "" when unknown. */

/** I3-0434 (D-0920 B) — how a match was decided, in words. A match always has a
 * winner (war_matches.winner CHECK a|b); "coin" is the engine's last tiebreak
 * after cards and HP were tied (packages/war-engine MatchResult.by). An unknown
 * value renders nothing rather than a guess. */
export function decidedByText(by: unknown): string {
  switch (by) {
    case "wipeout":
      return "won by knocking out every card";
    case "hp":
      return "decided on remaining HP";
    case "cards":
      return "decided on cards left standing";
    case "coin":
      return "tied on cards and HP — decided by a coin toss";
    default:
      return "";
  }
}

/** One match row. Every field is escaped and every field is OPTIONAL server-side,
 * so a missing one renders as nothing rather than as "undefined". */
function matchRow(m: WarMatchDto): string {
  const raw = typeof m?.match_id === "string" ? m.match_id : "";

  const id = raw;

/** #243b — one round of a committed replay, shape-checked. `null` = malformed. */
interface ReplayRound { round: number; loser: 0 | 1 | null; rollA: number; rollB: number; totalA: number; totalB: number; damage: number; ko: boolean; stat: string; statA: number; statB: number; edgeA: number; edgeB: number; bondA: number; bondB: number; flagA: number; flagB: number; rerolls: number }
/** #243d — the seven Power Ratings in engine order (packages/war-engine STAT_NAMES),
 * worded for a fan. Restated because apps/web cannot import the engine. */
export const STAT_LABELS = ["Reach", "Momentum", "Demand", "Collectors", "Archive", "Community", "Longevity"] as const;
const isInt = (x: unknown): x is number => typeof x === "number" && Number.isInteger(x);
function replayRound(x: unknown): ReplayRound | null {
  if (!x || typeof x !== "object" || Array.isArray(x)) return null;
  const r = x as Record<string, unknown>;
  const attempts = r.attempts;
  if (!isInt(r.round) || !Array.isArray(attempts) || attempts.length === 0) return null;
  const last = attempts[attempts.length - 1] as Record<string, unknown> | null;
  if (!last || typeof last !== "object" || !isInt(last.rollA) || !isInt(last.rollB) || !isInt(last.totalA) || !isInt(last.totalB)) return null;
  if (!(r.loser === 0 || r.loser === 1 || r.loser === null) || !isInt(r.damage)) return null;
  // #243d — THE MATHS, CHECKED BEFORE IT IS SHOWN. The engine (war-engine
  // engine.ts resolveRound) credits the stat edge |statA - statB| to the higher
  // side and adds each side's Bond: total = roll + edge-if-higher + Bond. A
  // round whose parts do not add up to its totals is not a round this view will
  // present as "the maths" — it is unread, like any other malformed round.
  if (!isInt(r.statIndex) || r.statIndex < 0 || r.statIndex >= STAT_LABELS.length) return null;
  if (!isInt(r.statA) || !isInt(r.statB) || !isInt(r.edge) || !isInt(r.bondA) || !isInt(r.bondB)) return null;
  if (r.edge !== Math.abs(r.statA - r.statB)) return null;
  const edgeA = r.statA > r.statB ? r.edge : 0;
  const edgeB = r.statB > r.statA ? r.edge : 0;
  // Q642 (D-1030): a flagged match's rounds carry flagA/flagB (war-engine RoundLog), added where Bond is; absent = 0.
  if ((r.flagA !== undefined && !isInt(r.flagA)) || (r.flagB !== undefined && !isInt(r.flagB))) return null;
  const flagA = isInt(r.flagA) ? r.flagA : 0;
  const flagB = isInt(r.flagB) ? r.flagB : 0;
  if (last.totalA !== last.rollA + edgeA + r.bondA + flagA || last.totalB !== last.rollB + edgeB + r.bondB + flagB) return null;
  const stat = STAT_LABELS[r.statIndex] as string;
  return { round: r.round, loser: r.loser, rollA: last.rollA, rollB: last.rollB, totalA: last.totalA, totalB: last.totalB, damage: r.damage, ko: r.ko === true, stat, statA: r.statA, statB: r.statB, edgeA, edgeB, bondA: r.bondA, bondB: r.bondB, flagA, flagB, rerolls: attempts.length - 1 };
}

/**
 * #243b — A FINISHED MATCH, PLAYED BACK FROM ITS COMMITTED REPLAY
 * (`GET /account/war/match/:id/replay`: the stored `{engine, inputs, log}` the
 * match's replay_hash commits to). Static — no realtime state, nothing re-rolled
 * here: every round's winner, the two dice of the attempt that counted (rollA /
 * rollB) and the totals after modifiers (totalA / totalB), then the result.
 *
 * THE FAIL-CLOSED LAW (views.ts `unreadable()`): a 404 is the server saying
 * there is no such match on this account; any other failure, and ANY round or
 * result that does not parse, renders the unread card — never a partial replay
 * that could show a match ending differently from how it did.
 */
export function warReplayView(matchId: string, read: { status: number; data: unknown }): string {
  const head = `<section class="store"><span class="eyebrow">Creator War</span><h2>Match replay</h2>`;
  const back = `<p><a href="#/war" data-nav="#/war" class="ds-btn nav">← Your matches</a></p>`;
  if (read.status === 404) return `${head}<div class="empty">There's no match ${xml(matchId)} on your account.</div>${back}</section>`;
  const unread = `${head}<div class="empty unread">We couldn't read this match's replay just now — this isn't an answer about how it went. Nothing has changed. Try again in a moment.</div>${back}</section>`;
  if (read.status !== 200 || !read.data || typeof read.data !== "object") return unread;
  const log = (read.data as { log?: unknown }).log as { rounds?: unknown; result?: unknown } | undefined;
  if (!log || typeof log !== "object" || !Array.isArray(log.rounds) || log.rounds.length === 0) return unread;
  const rounds = log.rounds.map(replayRound);
  if (rounds.some((r) => r === null)) return unread;
  const res = log.result as Record<string, unknown> | undefined;
  if (!res || typeof res !== "object" || !(res.winner === 0 || res.winner === 1) || typeof res.by !== "string") return unread;
  const side = (s: 0 | 1): string => (s === 0 ? "Side A" : "Side B");
  const rows = (rounds as ReplayRound[])
    .map((r) => {
      const won = r.loser === null ? "Tie — no damage" : `${side(r.loser === 0 ? 1 : 0)} wins (${r.damage} damage${r.ko ? ", knockout" : ""})`;
      // #243d: each side's total shown as its parts — the die, the stat edge (on
      // the stat this round was fought on, both values) and the Bond.
      const maths = (roll: number, edge: number, bond: number, flag: number, total: number): string => `${roll} (die) + ${edge} (${r.stat} edge) + ${bond} (Bond)${flag === 0 ? "" : ` ${flag < 0 ? "-" : "+"} ${Math.abs(flag)} (flag)`} = ${total}`;
      const rerolled = r.rerolls > 0 ? ` · ${r.rerolls} reroll${r.rerolls === 1 ? "" : "s"} before this throw` : "";
      return `<li class="wround" data-round="${r.round}"><b>Round ${r.round}</b> · A rolled ${r.rollA} → total ${r.totalA} · B rolled ${r.rollB} → total ${r.totalB} · ${xml(won)}<span class="wmaths" data-round-maths>Fought on ${xml(r.stat)} (A ${r.statA}, B ${r.statB}). A: ${maths(r.rollA, r.edgeA, r.bondA, r.flagA, r.totalA)} · B: ${maths(r.rollB, r.edgeB, r.bondB, r.flagB, r.totalB)}${rerolled}</span></li>`;
    })
    .join("");
  return `${head}<p class="lede">Match ${xml(matchId)}, played back from its committed record.</p><ol class="wrounds">${rows}</ol><p class="wresult" data-replay-result><b>${side(res.winner)} won</b> by ${xml(res.by)} after ${rounds.length} round${rounds.length === 1 ? "" : "s"}.</p>${back}</section>`;
}
