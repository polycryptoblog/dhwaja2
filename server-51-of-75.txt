/**
 * "One action = the whole drop" (DIRECTION.md Cluster D). ONE creator passkey
 * signature that atomically:
 *   1. MINTS a collectible — a release_registered, auto-published so the drop is
 *      immediately live (a drop is a launch, not a private timestamp).
 *   2. (optional) FIRES a one-way signed broadcast announcing it — the same
 *      passkey-signed creator_broadcast path, so it's provably-real and reaches
 *      savers via the alerts fold. Respects the Creator Pro frequency cap (never
 *      a cap bypass).
 *   3. (optional) ATTACHES a reverse-code pool (uploaded codes or Dhwaja
 *      vouchers), gated on save / this-collectible-acquisition / achievement.
 *
 * Everything commits in ONE appendCreatorSigned batch (all-or-nothing): the
 * signed release/broadcast EVENTS plus the release rows, the publish flip, and
 * the pool + codes ride the same transaction. The signature binds the ledger
 * events (release + broadcast); the pool is a creator-owned side-effect that
 * commits atomically with them.
 *
 * Scope note: the "optional artistic signature" from the roadmap is a separate,
 * still-unbuilt item — a drop mints the collectible's art (cover + fingerprints);
 * a signature-on-image tool is not part of this unifier.
 */

/** A drop is a launch, so its announcement is public. Named once and used by
 * BOTH legs: the /start and /finish drafts must be byte-identical or their action
 * hashes diverge and the passkey assertion fails. */
const DROP_ANNOUNCE_AUDIENCE = "public" as const;

export class DropError extends Error {
  override name = "DropError";
  constructor(
    message: string,
    public readonly status: 400 | 401 | 403 | 404 | 409 = 400,
  ) {
    super(message);
  }
}

export interface DropAnnounce {
  message: string;
  broadcast_id?: string; // server-minted at /start; the client echoes it back to /finish (it's signed)
}

/** I3-1111 — the reverse-code pool is not a ledger draft, so the signature
 * commits to it through `binding` (passkeys.ts actionHashForDrafts): both legs
 * derive the identical string from the pool INPUT the creator approved, so a
 * finish leg carrying a different pool than the one signed at /start fails the
 * challenge. Absent pool -> no binding (hash unchanged from before). */
function dropCodesBinding(codes: DropPoolInput | undefined): string | undefined {
  return codes === undefined ? undefined : `drop-codes:v1:${stableStringify(codes)}`;
}

export interface DropInput {
  collectible: ReleaseInput; // the release to mint (NEW only — no versioning in a drop)
  announce?: DropAnnounce; // optional signed broadcast
  codes?: DropPoolInput; // optional reverse-code pool
}

/** STRICT KEYS (QUEUE #524, I3-1772). Exactly the fields `DropInput` and
 * `DropAnnounce` declare; anything else is refused rather than silently carried
 * inward. Landable only because the route now strips its own transport field
 * (`assertion`) before calling in (`POST /account/drop`, releases-routes.ts). */
const DROP_INPUT_KEYS: Readonly<Record<keyof DropInput, true>> = { collectible: true, announce: true, codes: true };
const DROP_ANNOUNCE_KEYS: Readonly<Record<keyof DropAnnounce, true>> = { message: true, broadcast_id: true };

function ensureNewCollectible(input: DropInput): ReleaseInput {
  if (!input || typeof input !== "object") throw new DropError("drop input required");
  for (const k of Object.keys(input)) if (!Object.hasOwn(DROP_INPUT_KEYS, k)) throw new DropError(`unexpected field: ${k}`);
  if (input.announce !== undefined) {
    if (!input.announce || typeof input.announce !== "object") throw new DropError("bad announce");
    for (const k of Object.keys(input.announce)) if (!Object.hasOwn(DROP_ANNOUNCE_KEYS, k)) throw new DropError(`unexpected field: announce.${k}`);
  }
  const rel = input.collectible;
  if (!rel || typeof rel !== "object") throw new DropError("collectible required");
  if (rel.supersedes_hash) throw new DropError("a drop mints a NEW collectible; version an existing release separately", 400);
  return rel;
}

/**
 * Begin a drop: validate every part, build ALL the signable drafts (release +
 * optional broadcast), and return ONE assertion challenge bound to them. The
 * broadcast_id is minted here and echoed so the finish leg signs the identical
 * action. Fails fast (before the ceremony) on an over-cap broadcast or bad input.
 */
export async function startDrop(
  db: DbLike,
  passportId: string,
  input: DropInput,
  cfg: WebAuthnConfig,
  nowMs: number,
  now: () => string = () => new Date().toISOString(),
): Promise<{ options: Awaited<ReturnType<typeof startAssertionForAction>>; release_id: string; broadcast_id: string | null; message: string | null }> {
  const rel = ensureNewCollectible(input);
  await requireVerified(db, passportId); // minting a collectible requires a verified creator
  validateRelease(rel);
  await assertReleaseStockAvailable(db, passportId); // QUEUE #482 stock ceiling, before the ceremony

  const drafts: DraftEvent[] = await buildReleaseParts(db, passportId, rel, () => "fp_start", now).then((p) => [p.draft]);

  let broadcast_id: string | null = null;
  let message: string | null = null;
  if (input.announce) {
    await assertBroadcastAllowed(db, passportId, nowMs); // fast-fail the cap before the ceremony
    message = sanitizeMessage(input.announce.message);
    broadcast_id = newBroadcastId();
    // Window item 3 — audience is REQUIRED on the payload and set explicitly
    // here: a drop is a launch, so its announcement is public. It used to be
    // omitted and read back as public by a default-at-read branch that no longer
    // exists (broadcast-audience.ts). Both legs must build the identical payload
    // or the two action hashes diverge and the assertion fails.
    drafts.push(...buildCreatorDrafts(passportId, [{ event_type: "creator_broadcast", payload: { broadcast_id, message, audience: DROP_ANNOUNCE_AUDIENCE } }]));
  }

  // Validate the reverse-code block now (cheap, no mint) so a bad pool doesn't
  // surface only after the passkey ceremony. The pool is not a ledger draft, so
  // it is rebuilt fresh at finish; the signature commits to its INPUT through
  // `dropCodesBinding` (I13-AUTHFIX, QUEUE #881).
  //
  // The pool's OWN verification gate is fast-failed here for the same reason:
  // an unverified creator should learn it before the ceremony, not after. It is
  // NOT the `requireVerified` above — that one belongs to the collectible and
  // would stop covering the pool the moment a drop could carry one without a
  // release. `finishDrop` re-checks; this leg writes nothing.
  if (input.codes) {
    // The passport id is what lets this leg refuse an `achievement` ref that
    // belongs to ANOTHER creator (audit item #23) — a pure suffix/allowlist
    // test, so the fast-fail stays read-free.
    validateDropPoolInput(input.codes, passportId);
    await requireVerifiedCreator(db, passportId);
  }

  const options = await startAssertionForAction(db, passportId, drafts, cfg, nowMs, now, undefined, dropCodesBinding(input.codes));
  return { options, release_id: rel.release_id, broadcast_id, message };
}

export interface DropResult {
  release_id: string;
  published: boolean;
  broadcast_id: string | null;
  pool_id: string | null;
  generated_codes: string[];
}

/**
 * Complete a drop: rebuild the drafts identically, assemble every side-effect as
 * extraStatements (release rows + publish flip + pool + codes), and commit them
 * with the signed events in ONE atomic appendCreatorSigned batch.
 */
export async function finishDrop(
  db: DbLike,
  passportId: string,
  input: DropInput,
  assertion: AssertionResponse,
  cfg: WebAuthnConfig,
  nowMs: number,
  mintFpId: () => string,
  now: () => string = () => new Date().toISOString(),
  pubEnv?: AnalyticsTickEnv,
  pubCtx?: { waitUntil(p: Promise<unknown>): void },
): Promise<DropResult> {
  const rel = ensureNewCollectible(input);
  await requireVerified(db, passportId);
  const stockCap = await assertReleaseStockAvailable(db, passportId);

  // Collectible: signable draft + release/fingerprint row inserts (status
  // 'registered'), then a publish flip in the same batch so the drop goes live.
  const { draft: releaseDraft, statements: releaseStmts } = await buildReleaseParts(db, passportId, rel, mintFpId, now);
  const drafts: DraftEvent[] = [releaseDraft];
  const extra: DbStatement[] = [...releaseStmts];
  const publishedAt = now();
  extra.push(db.prepare("UPDATE releases SET status = 'published', updated_at = ? WHERE release_id = ? AND passport_id = ?").bind(publishedAt, rel.release_id, passportId));
  // ADR-0006 §2: the `release_published` row rides the SAME signed batch as the
  // flip, predicated on this call's own stamp — a drop that fails to append
  // emits nothing.
  extra.push(db.prepare(outboxSelectSql("release_published", { table: "releases", where: "release_id = ?", extraWhere: "status = 'published' AND updated_at = ?", scopeIdExpr: "release_id" })).bind(publishedAt, rel.release_id, publishedAt));
  // I3-2132: same predicate as the outbox row above it — the history entry
  // exists iff the flip above actually landed.
  extra.push(
    db
      .prepare(
        `INSERT INTO release_status_history (release_id, passport_id, status, at)
          SELECT release_id, passport_id, status, updated_at FROM releases
           WHERE release_id = ? AND passport_id = ? AND updated_at = ?`,
      )
      .bind(rel.release_id, passportId, publishedAt),
  );

  // Optional announcement — the broadcast draft MUST match what /start signed.
  let broadcast_id: string | null = null;
  if (input.announce) {
    await assertBroadcastAllowed(db, passportId, nowMs); // read check; the CLAIM is below, just before the append
    const message = sanitizeMessage(input.announce.message);
    if (typeof input.announce.broadcast_id !== "string" || !/^bcast_[0-9a-f]{16}$/.test(input.announce.broadcast_id)) throw new DropError("bad or missing broadcast_id (echo the one from /start)");
    broadcast_id = input.announce.broadcast_id;
    // Identical payload to /start, audience included — see the note there.
    drafts.push(...buildCreatorDrafts(passportId, [{ event_type: "creator_broadcast", payload: { broadcast_id, message, audience: DROP_ANNOUNCE_AUDIENCE } }]));
  }

  // Optional reverse-code pool — built fresh (pool id + code ids minted here),
  // riding the same atomic batch.
  let pool_id: string | null = null;
  let generated: string[] = [];
  if (input.codes) {
    // The authoritative check, at the action: a live `passports` read whose
    // result is the only thing `buildDropPool` accepts in place of a passport
    // id. Read here rather than reused from /start, because the two legs are
    // separate requests and verification can be revoked between them.
    const verified = await requireVerifiedCreator(db, passportId);
    const built = buildDropPool(db, verified, rel.release_id, input.codes, nowMs, now);
    pool_id = built.pool_id;
    generated = built.generated;
    extra.push(...built.statements);
  }

  // ATOMIC broadcast-cap claim, taken as late as possible — everything above can
  // still refuse the drop, and a claim taken earlier is a window a creator loses
  // to a rejected pool. Handed straight back if the append fails.
  if (broadcast_id) await claimBroadcastSlot(db, passportId, nowMs);

  // QUEUE #482: the stock ceiling at COMMIT, after the release insert above.
  extra.push(releaseStockGate(db, passportId, stockCap, nowMs));

  // ONE signature commits the collectible + announcement + codes atomically.
  let events: Awaited<ReturnType<typeof appendCreatorSigned>>;
  // #862: the ledger door brackets its batch and hands back the outbox seqs THIS commit wrote.
  let seqs: number[] = [];
  try {
    events = await appendCreatorSigned(db, passportId, drafts, assertion, cfg, nowMs, now, extra, dropCodesBinding(input.codes), (own?: readonly number[]) => {
      seqs = [...(own ?? [])];
    });
  } catch (e) {
    if (broadcast_id) await releaseBroadcastSlot(db, passportId);
    // Asked FIRST: a NOT NULL the broad match below would call an id collision.
    if (isStockCeilingRefusal(e)) throw new ReleaseStockError(stockCap);
    // The standing re-check (audit item #29) rides this batch twice — once from
    // `insertStatements` for the collectible, once from `buildDropPool` for the
    // codes — and fails it with a NOT NULL, which the broad match below would
    // report as an id collision. Ask the precise question first.
    if (isWriteAdmissionRefusal(e)) throw new DropError("verified creators only", 403);
    if (e instanceof Error && /UNIQUE|constraint/i.test(e.message)) throw new DropError("a release with that id already exists", 409);
    throw e;
  }

  // Record the broadcast cap AFTER the append (the event itself drives fan
  // delivery via the alerts fold; this is bookkeeping). The broadcast draft is
  // always pushed last, so it's the final appended envelope.
  if (broadcast_id) {
    const bev = events[events.length - 1];
    // BF-21 — A POST-COMMIT FAILURE MUST NOT TURN A COMMITTED DROP INTO A
    // REJECTION.
    //
    // Everything above has already committed: the release is published, the
    // announcement is on the ledger, and `generated` holds LIVE voucher codes.
    // This insert is bookkeeping that happens afterwards, and when it threw the
    // exception escaped — the route answered HTTP 400 and the console treated
    // the whole drop as a definitive failure, so the creator never received the
    // codes. The product DELIBERATELY provides no later raw-code listing route,
    // so that response was the only handoff those codes will ever have, and an
    // ordinary retry cannot recover it: the ceremony is spent and the release is
    // already published. A bookkeeping outage destroyed a creator's issued
    // vouchers.
    //
    // NOTHING IS WEAKENED BY SWALLOWING IT, and `broadcast.ts` says so on
    // `recordBroadcast` itself: "losing it costs the audit row, not the cap —
    // `assertBroadcastAllowed` reads the claim as well as the log, so the window
    // stays closed either way." The cap is held by `claimBroadcastSlot`, taken
    // atomically BEFORE the append. So the choice here is between losing an
    // audit row and losing live vouchers, and it is not close.
    //
    // It is not silent either: the audit row below is the operator-visible
    // trail, written the same way `index.ts logFrozenSettlement` writes its
    // own post-settlement anomaly — and if THAT write fails too, that failure
    // is logged and still does not fail the drop.
    if (bev) {
      try {
        await recordBroadcast(db, passportId, broadcast_id, bev.event_id, now);
      } catch (e) {
        try {
          await db
            .prepare("INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) VALUES (?, 'broadcast:log_write_failed', ?, ?, ?, 'system')")
            .bind(
              `creator:${passportId}`,
              JSON.stringify({ broadcast_id, event_id: bev.event_id, release_id: rel.release_id, err: e instanceof Error ? e.name : typeof e }).slice(0, 500), // RG-PRIVACY:13: the class, as removal.ts #858
              now(),
              passportId,
            )
            .run();
        } catch {
          // The trail is best-effort; the committed drop is not.
        }
      }
    }
  }

  if (pubEnv && pubCtx) publishAfterCommit(pubEnv, pubCtx, seqs);
  return { release_id: rel.release_id, published: true, broadcast_id, pool_id, generated_codes: generated };
}

import { sha256Hex } from "@dhwaja/canonical";
import { canonicalJson } from "./anchoring.js";
import type { DbLike, DbStatement } from "./db.js";

/**
 * Dual-control seam — approver-ready, single-operator by default.
 *
 * High-liability, hard-to-reverse actions (publishing a warning, flipping the
 * confirmed_fraud serving gate) can be made to require N DISTINCT operators to
 * sign off before they execute. The number is config-driven
 * (OPS_APPROVALS_REQUIRED). Since slice 29 the default is 2, clamped down to the
 * number of allowlisted PEOPLE so a solo deployment is not bricked — see
 * `effectiveApprovals`. At an effective threshold of 1 the seam is inert, the
 * action runs immediately, and the pending_actions table is never touched.
 *
 * Model at threshold N >= 2: each distinct operator who invokes the action for
 * the same (actionType, paramsKey) records one row in pending_actions. When the
 * count of DISTINCT sign-offs (including the current caller) reaches N, the
 * action executes exactly once and its pending rows are cleared. Re-signing by
 * the same operator is idempotent (INSERT OR IGNORE on the composite PK), so an
 * operator can never satisfy a 2-of-N gate alone.
 *
 * This is a governance gate, not a concurrency lock: the default path (N=1) has
 * zero overhead, and the multi-approver path targets the single-digit-operator
 * reality Dhwaja launches with. A true distributed lock is a later concern if
 * the operator pool ever gets large and racy.
 */

/** Distinct PEOPLE who must sign off on a guarded action when the env says
 * nothing. Clamped to [1, 5]; unparseable or non-integral → this default. */
export const DEFAULT_APPROVALS_REQUIRED = 2;

/**
 * How many DISTINCT PEOPLE must sign off, given this deployment's config.
 *
 * ---------------------------------------------------------------------------
 * THE DEFAULT IS 2 (slice 29), AND THE CLAMP IS WHY THAT IS SAFE
 * ---------------------------------------------------------------------------
 * It defaulted to 1 for the life of the seam, which meant every deployment that
 * never thought about it ran nine high-liability `/ops` actions — publishing a
 * warning, flipping two serving gates, deleting and transferring accounts,
 * resolving a recovery case — on one signature. A default that has to be
 * discovered to be safe is not a default.
 *
 * But raising it naively BRICKS a single-operator deployment, permanently and
 * with no way out from inside the product. Measured before this change:
 * `pending_actions` counts distinct sign-offs, one operator's count is pinned at
 * 1 forever by the composite primary key, the threshold is read from ENV ONLY
 * (there is no `ops_settings` row and so no operator route that can lower it),
 * and the table has no TTL. All nine routes become unreachable — including
 * `POST /ops/settings/serve-caution` and `POST /ops/settings/face-triage`, the
 * two switches that turn features on for the first time — until somebody
 * redeploys with a var change.
 *
 * So the requirement is CLAMPED TO THE NUMBER OF ALLOWLISTED OPERATORS. A
 * 2-of-1 gate is not a security control; it is an outage wearing one. Clamping
 * grants nothing an attacker did not already have: shrinking `OPERATOR_EMAILS`
 * needs the DEPLOY credential, which can set `OPS_APPROVALS_REQUIRED` directly.
 * And it is visible — `GET /ops/settings` reports the configured value, the
 * effective one, and the operator count, so a deployment can never quietly
 * believe it has two-person control when it has one person.
 *
 * The warnings two-person rule is NOT this seam and is not clamped: it is
 * always on, has no knob, and a single-operator deployment simply cannot
 * publish an accusation. That is the intended reading of the promise on
 * `/how-it-works` and it is unchanged here.
 */
export function approvalsRequired(env: Record<string, unknown>): number {
  return effectiveApprovals(env).required;
}

/**
 * F18 / #32 — TWO ACTIONS ARE NOT CLAMPED, and both are written into the Terms
 * as "two of our operators, not one".
 *
 * `approvalsRequired` clamps the configured N down to the number of operators,
 * so a single-operator deployment is not deadlocked. That is right for every
 * action whose promise is internal. It is wrong for one we PUBLISH:
 * `TERMS.md §7.2` tells a person that terminating their account takes two
 * people, and a review pointed out that with one address on the allowlist it
 * took one — an unclamped promise wired to a clamped seam.
 *
 * This is the warnings precedent, and this module's own docblock states it: the
 * two-person rule there "is always on, has no knob, and a single-operator
 * deployment simply cannot publish an accusation." Terminating an account and
 * forfeiting what it holds is at least that consequential. Suspension is NOT on
 * this list: it is reversible, it takes nothing, and making it undoable-only
 * would leave one operator unable to stop an active fraud.
 *
 * S-070 OW-01 — THIS IS NOW EXPORTED BECAUSE THE READ MODEL HAS TO USE IT TOO.
 * It was an inline expression inside `withDualControl`, so the ENFORCING side
 * knew that `account:terminate` needs `max(2, N)` and the QUEUE that operators
 * read did not. With the supported `OPS_APPROVALS_REQUIRED=1`, a correctly
 * pending termination rendered "1/1" — a denominator saying the action was
 * complete when the gate would (correctly) refuse to execute it, and the exact
 * kind of handoff error a second operator relies on that column to avoid. One
 * threshold calculation, two callers; the intentional minimum is untouched and
 * the gate is not relaxed by a single character.
 */
export function requiredApprovalsFor(env: Record<string, unknown>, actionType: string): number {
  // W3-09 regression attacker + judge, both independently (2026-09-20).
  //
  // THE CLAMP IS A DEPLOYMENT CONCESSION, NOT A POLICY. `approvalsRequired`
  // caps the threshold at the number of allowlisted operators so a one-operator
  // deployment is not bricked. For most actions that is the right trade. For an
  // action whose ENTIRE safeguard is "a human decides", it silently removes the
  // safeguard: at one operator — or two allowlisted addresses of one person —
  // `withDualControl` takes its `required <= 1` fast path and the primitive is
  // handed a literal `dualControlSatisfied: true`, which it cannot second-guess
  // because a boolean is all it knows.
  //
  // `brand_contest:admit` is exactly that action. Its filing door is ANONYMOUS
  // and takes `filer_passport_id` from the request BODY, defended at
  // `claim-routes.ts:275` on the explicit ground that an operator under dual
  // control decides. The clamp removed the only thing that sentence relies on,
  // so ONE operator could attach a contested brand host to a passport that
  // never asked — and the named filer is never notified, only the incumbent.
  //
  // Added here rather than special-cased at the route because the route already
  // did everything right: it keyed the params on lane AND incumbent, and it
  // asked `withDualControl`. The threshold was the thing that lied.
  //
  // `payout:release_held` (B25-635, D-0750 (3)): an operator paying a removed
  // creator's held balance is money out on one person's word unless two
  // different operators sign, whatever the deployment's setting. `payout:void` (D-0929(8)) is the same
  // money decision from the other side: one operator cannot void an open payout alone.
  // D-0931(1): `operator:review` — a person review is what lets that person's money sign-offs count, so it takes two.
  // D-0995(9): `payout:rearm` (console.ts /ops/payouts/:id/rearm) sends a parked payout again — the same money decision
  // on the same held row as void and release_held, so it takes two as they do.
  // D-1010(5) (I25 INT-M2): `creator:terminate` ends a creator as `account:terminate` ends a fan, so it takes two as that does.
  // I29-OPS (D-1030 I23-Q3 (a), I4-Q5, I28-Q6, I16-S5-n138C-36, I25-Q2, I21-Q29): unbinding a payout destination, lifting a
  // chargeback ban, correcting a fan's credits, releasing a dispute freeze, returning a parked payout, setting or clearing a
  // creator fraud finding, and opening or releasing a legal hold each take two distinct operators on every deployment.
  // I29-FX-OPS (D-1037(11), RG-OPS OPS:8 / OPS:13): clearing a creator's fraud-banner flag, and every write on the numeric
  // limits door (each OPS_LIMIT_SETTINGS key is a retention period or a money limit, console.ts), take two as well.
  const UNCLAMPED = new Set(["account:chargeback_ban_lift", "account:terminate", "account:repay", "brand_contest:admit", "caution:unsuppress", "connect:unbind", "creator:fraud_banner_clear", "creator:fraud_finding", "creator:terminate", "credit:correct", "dispute_freeze:release", "legal_hold:open", "legal_hold:release", "operator:review", "payout:rearm", "payout:release_held", "payout:return", "payout:void", "recovery:reentry_grant", "settings:limits", "transfer:no_address", ...COUNTERSIGN_HOLD_ACTIONS]);
  return UNCLAMPED.has(actionType) ? Math.max(2, approvalsRequired(env)) : approvalsRequired(env);
}

/** I40-OPS (D-1169(4), W0 amendment 3): while a retraction's countersign is pending on a subject, the one-operator doors
 * touching that subject's evidence or caution run under these actions (console.ts `underCountersignHold`), UNCLAMPED: two
 * distinct persons on every deployment, so a one-person deployment's door waits for the countersign. Not in
 * REVIEWED_PERSON_ACTIONS, as `caution:unsuppress` is not. */
export const COUNTERSIGN_HOLD_ACTIONS = [
  "countersign_hold:evidence_delete",
  "countersign_hold:fan_report_suppress",
  "countersign_hold:fan_report_dismiss",
  "countersign_hold:warning_withdraw",
  "countersign_hold:review_reject",
  "countersign_hold:caution_suppress",
] as const;

export interface ApprovalsPosture {
  /** What the env asked for, after clamping to [1,5]. */
  configured: number;
  /** What is actually enforced — `configured`, capped by `operators`. */
  required: number;
  /** Allowlisted operator addresses. 0 when the allowlist is unreadable. */
  operators: number;
  /** True when `required < configured`, i.e. the deployment cannot satisfy it. */
  clamped: boolean;
}

export function effectiveApprovals(env: Record<string, unknown>): ApprovalsPosture {
  const raw = env.OPS_APPROVALS_REQUIRED;
  // STRICT DIGITS. `Number.parseInt` happily reads "1e1" as 1 and "1.5" as 1, so
  // a deployment that meant ten, or that fat-fingered a decimal, silently got
  // the weakest possible gate while `clamped` reported nothing was reduced. A
  // value this parser cannot read is a value nobody chose: fall back to the
  // default (fail closed) rather than to 1. An explicit "0" still floors to 1 —
  // that is a deliberate "off" written by whoever holds the deploy credential.
  const n = typeof raw === "string" && /^\d+$/.test(raw.trim()) ? Number(raw.trim()) : typeof raw === "number" && Number.isInteger(raw) && raw >= 0 ? raw : NaN;
  const configured = Number.isFinite(n) ? Math.max(1, Math.min(5, n)) : DEFAULT_APPROVALS_REQUIRED;
  // Counted from the same string `accessConfigFromEnv` parses, rather than from
  // `operator_identities`, on purpose: the identities table records who has
  // LOGGED IN, so a deployment with two allowlisted operators of whom one has
  // never signed in would clamp to 1 and quietly drop to single control on the
  // day the second one was needed most.
  //
  // COUNTED AS PEOPLE, NOT ADDRESSES. The gate below counts DISTINCT PERSONS,
  // and this slice is the reason those differ: two allowlisted addresses that
  // canonicalize together are one human everywhere else in the system. Counting
  // strings here left the two halves measuring different things, and the failure
  // was the exact one the clamp exists to prevent — an alias pair reported
  // `operators: 2, clamped: false` while the gate could never see more than one
  // person, so all nine routes were unreachable and `GET /ops/settings` called
  // the deployment healthy. `canonicalizeEmail` is the same fold `operatorActorId`
  // applies at mint time, so the two now agree by construction.
  //
  // What env still cannot see is a MANUAL link (`POST /ops/operators/link`),
  // which can only merge further — so this count is an upper bound on people and
  // the residual is written down in the slice manifest rather than papered over.
  const emailsRaw = typeof env.OPERATOR_EMAILS === "string" ? env.OPERATOR_EMAILS : "";
  const operators = new Set(
    emailsRaw
      .split(",")
      .map((e) => e.trim().toLowerCase())
      .filter((e) => e.length > 0 && !/todo/i.test(e))
      .map((e) => canonicalizeEmail(e)),
  ).size;
  // An unreadable allowlist used to fall through to 1 — a fail-OPEN in a module
  // whose whole job is a gate. It is unreachable today (the `/ops` middleware
  // 503s on the same condition before any handler runs), but the strict branch
  // costs nothing and does not depend on a guarantee enforced in another module.
  const required = operators > 0 ? Math.min(configured, operators) : configured;
  return { configured, required, operators, clamped: required < configured };
}

export interface DualControlOutcome<T> {
  /** True iff the action ran on this call. */
  executed: boolean;
  /** True iff this call recorded a sign-off but the threshold isn't met yet. */
  pending: boolean;
  /** Distinct sign-offs now on record (including this caller). */
  approvals: number;
  /** Sign-offs required. */
  required: number;
  /** #131AF-12: on a `payout:%` action still pending, live sign-offs banked but not counted because their person carries no review (migration 0200). */
  unreviewed?: number;
  /** The wrapped action's return value, present only when executed. */
  result?: T;
}

/**
 * Run `execute` under the dual-control policy for (actionType, paramsKey).
 *
 * paramsKey is the stable identity of the SPECIFIC action instance (e.g. the
 * review item id for a publish, or "serve_confirmed_fraud=1" for a serving
 * flip) — two operators must sign the SAME paramsKey for their sign-offs to
 * combine. It is stored verbatim WHERE IT FITS, because the operator dashboard
 * renders this column as the human-readable "Key" of what is being approved, and
 * `recovery.ts clearResolveSignoffs` matches it by PREFIX. A blanket hash would
 * blind the first and break the second.
 *
 * ============================================================================
 * THE TRUNCATION WAS A DUAL-CONTROL BYPASS
 * ============================================================================
 *
 * The sentence that used to sit here said the key "is stored verbatim (bounded)
 * … these keys are already short, non-sensitive identifiers." **Two of them are
 * not short, and both carry an operator-supplied account key:**
 *
 *  - `console.ts`'s `account:transfer` builds `` `${accountKey}->${to}` ``, and
 *    an account key is a canonical account reference bounded at ~2048
 *    characters — the ledger's own `LINK_URL_MAX`;
 *  - `recovery:resolve`'s REQUEST-origin key is
 *    `` `${itemId}|${recoveryPlanKey(plan, kind)}` ``, whose `k=` component is
 *    the `contested_account_key` — bounded at 512 characters by `validatePlan`,
 *    and (until slice 21 hoisted the shape check above the gate) bounded by
 *    nothing at all at the moment the key was built. Two resolves of the same
 *    case with the same actions and destination but different contested keys
 *    agreeing in their first few characters collided the same way — on the route
 *    that hands a hijacked account to a new passport.
 *
 * With a bare `slice(0, 200)`, two DIFFERENT transfers whose keys agree in their
 * first 200 characters become the SAME `pending_actions` row. So a second
 * operator approving transfer A can execute transfer B: one sign-off, spent on a
 * decision nobody signed for. That is the exact property dual control exists to
 * provide, defeated by a bound nobody re-read after the keys got longer.
 *
 * The fix keeps the readable prefix and makes the REMAINDER unambiguous: a key
 * that fits is stored as-is; a key that does not is stored as a readable prefix
 * plus a digest of the WHOLE key. Two keys agreeing in their first 136
 * characters and differing anywhere after it now differ here too. `sha256Hex`
 * from `@dhwaja/canonical` is the one hasher (invariant 1); this is not a
 * subject and is not derived from one, so it does not route through
 * `subjectHash` — the same reasoning `manual-claims.ts` records for hashing a
 * bearer token.
 */
/** The stored `params_hash`, bounded to the column's 200-character budget
 * WITHOUT letting two different decisions collide.
 *
 * Short keys — every one whose operator-supplied components are small — are
 * unchanged, so the dashboard still shows what is being approved and
 * `clearResolveSignoffs`'s prefix match still works (its `${itemId}|%` prefix is
 * far inside the 136 characters kept below). A long key keeps a readable
 * 136-character prefix and gains a 63-character digest of the WHOLE key, which
 * is what makes it unique.
 *
 * `#` as the separator, and the honest version of why: it is a CONVENTION, not
 * an enforced property. No key shape in this repo uses `#` today, but two of
 * them embed operator-supplied account keys that are only length-bounded, so a
 * key containing one is expressible. It costs nothing, because forging
 * `prefix#digest` for a colliding decision needs a sha256 preimage — the
 * separator is a readability aid for the dashboard, and the digest is what
 * carries the uniqueness.
 */
/** I43-DUALCTL (D-1246 "8. a", D-1249(1)): the plan-key component for free text (a reason, a note) or a whole body an
 * executor reads: `d=` + `sha256Hex` (packages/canonical) over `canonicalJson` of the value the executor receives (the
 * route's own normalisation, none added here). A changed reason changes the key, so the action answers pending and the
 * second operator approves again; the text never reaches `pending_actions.params_hash` or the sign-off audit detail. */
export async function planTextDigest(v: unknown): Promise<string> {
  return `d=${await sha256Hex(canonicalJson(v ?? null))}`;
}
export async function boundedKey(paramsKey: string): Promise<string> {
  if (paramsKey.length <= 200) return paramsKey;
  return `${paramsKey.slice(0, 136)}#${(await sha256Hex(paramsKey)).slice(0, 63)}`;
}

/**
 * A SIGN-OFF EXPIRES. `pending_actions` had no TTL and no sweep; before slice 29
 * that was a curiosity (the default was one approver, so the table was never
 * written), and after it every deployment banks a row for every gated action
 * and nothing ever removes a row whose second approver never came. Two
 * consequences, one of them a control defect: the table grows without bound
 * and `GET /ops/settings` loaded all of it; and a sign-off banked in March
 * could be spent in September by an approver who never saw the March context.
 *
 * THE GATE APPLIES THE TTL ITSELF. `pruneStaleSignoffs` is hygiene — it runs on
 * the cron and only reclaims rows. The DISTINCT-person count below and
 * `listPending` both filter on `requested_at >= cutoff`, so a stale sign-off
 * never counts whether or not the sweep has run. Anything else would make the
 * control depend on a background job being alive, which is exactly the shape
 * `tests/invariants/i13-cron-reachability.test.ts` exists to distrust.
 *
 * `requested_at` is an ISO-8601 UTC string written by this module's `now()`,
 * so the comparison is lexicographic and needs no parsing in SQL. A `now()` that
 * yields something `Date.parse` cannot read throws here (`RangeError` from
 * `toISOString`) rather than computing a cutoff that admits everything — fail
 * closed, loudly.
 */
export const PENDING_SIGNOFF_TTL_MS = 7 * 24 * 60 * 60 * 1000;

/** How many `pending_actions` rows the dashboard read will load. Past this the
 * response says `truncated`; it does not silently render a partial queue as a
 * whole one (the same rule `operatorPersons` follows for the roster). */
export const PENDING_LIST_MAX = 1000;

export function signoffCutoff(nowIso: string): string {
  return new Date(Date.parse(nowIso) - PENDING_SIGNOFF_TTL_MS).toISOString();
}

/** Cron hygiene: reclaim sign-offs the gate already refuses to count. Returns
 * the number of rows removed. Deleting a row the gate would count is the one
 * thing this must never do, so the cutoff is the SAME function the gate uses. */
export async function pruneStaleSignoffs(db: DbLike, nowMs: number): Promise<number> {
  const cutoff = signoffCutoff(new Date(nowMs).toISOString());
  const res = await db.prepare("DELETE FROM pending_actions WHERE requested_at < ?").bind(cutoff).run();
  return res.changes ?? 0;
}

/**
 * THE SIGN-OFF AUDIT ROW NAMES THE PENDING ACTION, NEVER THE ACCOUNT (D-0810(7),
 * #378, write-time). Every `account:*` key is, or embeds, the subject account's
 * id (`account:delete` is the bare id; close/lift/terminate/chargeback_ban_lift
 * too; reactivate/transfer embed an account key). An erasure pseudonymises its own
 * rows (D-0543) but cannot reach an earlier sign-off's `detail`, so the id is kept
 * out of it here: the row says `<actionType> key-withheld`. The co-signer match and
 * the dashboard read the key from `pending_actions`, which the claim below consumes
 * and `pruneStaleSignoffs` expires; the executed action writes its own audit row.
 * Every other action keeps its key verbatim UNLESS it names an account id
 * (D-0813(3), below). `test/b27-fxb-378-signoff-withheld.test.ts`, `test/b27fx3-378-view-collectible.test.ts`.
 */
export function signoffDetail(actionType: string, key: string): string {
  return actionType.startsWith("account:") || NAMES_AN_ACCOUNT.test(key) ? `${actionType} key-withheld` : `${actionType} ${key}`;
}

/** D-0813(3): ANY key that names an account id is withheld, whatever its action
 * type (`collectible:recover` keys on `<kind>:<item>:<owner>:<to_account>:<ref>`).
 * The shape is `mintId("acc")` (`accounts.ts`): `acc_` + at least 16 hex. */
const NAMES_AN_ACCOUNT = /acc[_#][0-9a-f]{16,}/i;

/** I39-OPS (D-1158(6b)): the co-signer record of one dual-control execution, for the caller to join into the batch that
 * writes the action's effect. One `dual_control:cosigner` row per consumed sign-off whose op id is not the executing
 * operator's (`signers` is what `withDualControl` hands `execute`), actor = the executor, item_id NULL and the detail
 * `<actionType> co-signed by <op id>`: the action and an op id, never the key (`signoffDetail`'s withholding). On the
 * fast path `signers` is `[actor]`, so the list is empty and nothing is written. */
export function coSignerAuditStmts(db: DbLike, actionType: string, executor: string, signers: readonly string[], ts: string): DbStatement[] {
  return [...new Set(signers)].filter((s) => s !== executor).map((s) => auditStmt(db, executor, "dual_control:cosigner", null, `${actionType} co-signed by ${s}`, ts));
}

/**
 * THE STORED KEY NEVER HOLDS A RAW ACCOUNT ID (QUEUE #670, D-0817(8)). Every
 * `acc_<hex>` in a params key is replaced, before the key is bounded and
 * written to `pending_actions.params_hash`, by `acc#` + 32 hex of a one-way
 * digest. D-0825(1) (J-670-unkeyed; honours D-0817(8)'s "keyed digest"): the
 * digest is HMAC-SHA256 under the environment's `SESSION_SECRET`, which
 * `withDualControl` receives in `env` (label `dhwaja-dc-params-v2`), so holding a
 * raw id does not link a sign-off banked before its erasure. WITH NO SECRET
 * CONFIGURED (absent, empty or not a string) it falls back to the fixed-label
 * form — canonical's single hasher over `dhwaja-dc-params-v1` + id, the
 * D-0810(2)/D-0811(4) precedent (`ratelimit.ts` `refusalKeyForAudit`) — which
 * anyone holding the raw id can recompute. Rotating `SESSION_SECRET` changes
 * every token, so a sign-off banked before the rotation lapses like the raw-key
 * rows below. Deterministic, so two operators signing the same decision still meet on one
 * key and two decisions naming different accounts still differ; an erasure
 * therefore leaves no raw id behind in a sign-off banked before it. Prefix
 * readers are untouched: `recovery:resolve` (`<itemId>|`) and
 * `brand_enrol:approve` (`<id>|scrutiny|`) prefixes carry no account id.
 * A row banked under the raw key before this shipped no longer matches and
 * lapses by `PENDING_SIGNOFF_TTL_MS`; the co-signer signs again.
 */
export async function withheldParamsKey(paramsKey: string, secret?: string): Promise<string> {
  // INJECTIVE (QUEUE #670, D-0823(6)): the literal text is escaped FIRST — `~` as
  // `~~`, and every literal `acc#` (any case) as `acc~#` — so a digest token
  // `acc#<hex>` in the output can only have come from an `acc_` id, and typing
  // the withheld form of an id never meets the key that id produces. The escape
  // touches neither `acc_` ids nor keys without `~`/`acc#`, so those are unchanged.
  // Ids are replaced in ONE pass over maximal matches (never `split(id)`, which
  // rewrote a shorter id inside a longer one that shares its prefix).
  const escaped = paramsKey.replace(/~/g, "~~").replace(/acc#/gi, (m) => `${m.slice(0, 3)}~#`);
  const digest = new Map<string, string>();
  const enc = new TextEncoder();
  const mac = secret ? await crypto.subtle.importKey("raw", enc.encode(secret) as unknown as ArrayBuffer, { name: "HMAC", hash: "SHA-256" }, false, ["sign"]) : null;
  for (const id of new Set(paramsKey.match(/acc_[0-9a-f]{16,}/gi) ?? [])) {
    const hex = mac
      ? [...new Uint8Array(await crypto.subtle.sign("HMAC", mac, enc.encode(`dhwaja-dc-params-v2\n${id}`) as unknown as ArrayBuffer))].map((b) => b.toString(16).padStart(2, "0")).join("")
      : await sha256Hex(`dhwaja-dc-params-v1\n${id}`);
    digest.set(id, `acc#${hex.slice(0, 32)}`);
  }
  return escaped.replace(/acc_[0-9a-f]{16,}/gi, (id) => digest.get(id) ?? id);
}

/**
 * I3-0897 / I3-1075 — A SIGN-OFF COUNTS ONLY WHILE ITS OPERATOR IS STILL ON THE
 * ALLOWLIST. Removing an address from `OPERATOR_EMAILS` is how an operator is
 * revoked (`auth.ts accessConfigFromEnv` refuses them from then on), so a
 * sign-off they banked before the removal must stop counting at the same
 * moment. The quorum COUNT and the claiming DELETE both carry this one
 * predicate. The allowlist is parsed exactly as `accessConfigFromEnv` parses it
 * (trim, lowercase) and compared with `operator_identities.email`, which
 * `operatorActorId` stores in that same form. An id with NO identity row keeps
 * its old treatment (no email to test); every id the console mints has one.
 */
function liveOperatorsJson(env: Record<string, unknown>): string {
  const raw = typeof env.OPERATOR_EMAILS === "string" ? env.OPERATOR_EMAILS : "";
  return JSON.stringify(raw.split(",").map((e) => e.trim().toLowerCase()).filter((e) => e.length > 0));
}
const STILL_ALLOWLISTED = (col: string): string =>
  `NOT EXISTS (SELECT 1 FROM operator_identities oi WHERE oi.op_id = ${col} AND oi.email NOT IN (SELECT value FROM json_each(?)))`;

/**
 * #138CM-04 / #135E-26 (I16, D-0920) — A SIGN-OFF COUNTS ONLY WHILE ITS OPERATOR IS
 * NOT REVOKED. `revokeOperator` (operators.ts) writes `operator_revocations` and
 * discards the person's banked sign-offs in one batch, and the /ops middleware
 * refuses a revoked id — but a sign-off INSERT already past the middleware when the
 * revocation commits lands after the discard, and nothing else stopped it counting
 * for the whole PENDING_SIGNOFF_TTL_MS. One predicate, interpolated at every door
 * that counts a quorum: the gate's COUNT, the claiming DELETE...RETURNING, and the
 * two readers that show approvals "the way the gate counts" (`listPendingPage`,
 * `liveSignoffsByPrefix`). No bind parameter, so each door's binds are unchanged.
 */
const NOT_REVOKED = (col: string): string => `NOT EXISTS (SELECT 1 FROM operator_revocations orv WHERE orv.op_id = ${col})`;
/**
 * #131AF-12 (D-0920, D-0927(3); migration 0200) — ON A `payout:%` ACTION A SIGN-OFF
 * COUNTS ONLY WHEN ITS PERSON WAS REVIEWED BY ANOTHER PERSON. COALESCE(person_id,
 * op_id) makes two never-linked addresses of one human two "persons"; a review row
 * (`operator_person_reviews`) vouches for the requester's CURRENT person, so a
 * later link/unlink voids it, and a reviewer who resolves to that same person
 * vouches for nothing. One clause, interpolated at the count AND the claim.
 */
const PERSON_OF = (col: string): string => `COALESCE((SELECT person_id FROM operator_identities WHERE op_id = ${col}), ${col})`;
// D-0931(1) (I16-RG-MONEY:3; I66 P1): …AND ONLY WHILE EVERY OPERATOR WHO SIGNED IT IS UNREVOKED — `revokeOperator` leaves
// the reviews the revoked person signed, so the review stops counting here instead. SIGNER_OK holds one signer to both
// rules (another person, unrevoked). Exclusive fallback (CASE, never OR): a row with `signers` (migration 0285) counts
// only when no signer fails SIGNER_OK; a row without (written before 0285) is decided by `reviewed_by` alone.
const SIGNER_OK = (s: string): string => `${PERSON_OF(s)} <> opr.person_id AND ${NOT_REVOKED(s)}`;
const PERSON_REVIEWED = (col: string): string =>
  `EXISTS (SELECT 1 FROM operator_person_reviews opr WHERE opr.op_id = ${col} AND opr.person_id = ${PERSON_OF(col)} AND CASE WHEN opr.signers IS NULL THEN ${SIGNER_OK("opr.reviewed_by")} ELSE NOT EXISTS (SELECT 1 FROM json_each(opr.signers) sg WHERE NOT (${SIGNER_OK("sg.value")})) END)`;
/** D-0931(2) (I16-RG-MONEY:2): money out — every `payout:%` action AND `account:repay` (UNCLAMPED, "so one operator cannot pay alone").
 * I21-Q48 (D-1030 (a); I30-PREMIUM): and the six irreversible people-and-access actions, spelled as their withDualControl
 * call sites in console.ts spell them (:2604, :2643, :2814, :3370, :1992, :1138).
 * D-1059(4) (I30-RG-MONEY RGM-N3): and the six two-person money doors, spelled as their console.ts withDualControl call
 * sites spell them (credits.ts CREDIT_CORRECTION_ACTION is "credit:correct"). A `recovery:resolve` that unbinds a payout
 * account is asked as `connect:unbind` (withDualControl's gate action, thresholdActionType).
 * D-1070(2) (I30 OPEN-ROWS B1, I30-RJ-03 c6): and five of the six UNCLAMPED non-money actions, spelled as their console.ts
 * withDualControl call sites at i31-base spell them (:4772 brand_contest:admit, :3187 creator:fraud_banner_clear, :3156 and
 * :3170 creator:fraud_finding, :3120 legal_hold:open, :4049 recovery:reentry_grant). Census: UNCLAMPED minus this list
 * minus `payout:*` is now exactly `operator:review`, which stays OUT: its door (console.ts:1207) is the only writer of
 * `operator_person_reviews` (operators.ts reviewOperatorPerson), so requiring reviewed signers there leaves a deployment
 * with no review unable ever to write one (I31-MAILNOTICE probe; open for a ruling). */
export const REVIEWED_PERSON_ACTIONS: readonly string[] = [
  "account:repay",
  "account:close",
  "account:terminate",
  "creator:terminate",
  "account:delete",
  "agency_domain:release",
  "operator:revoke",
  "credit:correct",
  "dispute_freeze:release",
  "connect:unbind",
  "account:chargeback_ban_lift",
  "legal_hold:release",
  "settings:limits",
  "brand_contest:admit",
  "creator:fraud_banner_clear",
  "creator:fraud_finding",
  "legal_hold:open",
  "recovery:reentry_grant",
];
export const needsReviewedPerson = (actionType: string): boolean => actionType.startsWith("payout:") || REVIEWED_PERSON_ACTIONS.includes(actionType);
const reviewed = (actionType: string, col: string): string => (needsReviewedPerson(actionType) ? ` AND ${PERSON_REVIEWED(col)}` : "");
/**
 * A database without migration 0166 has no revocation table and so no revocation —
 * nothing could have written one: `isOperatorRevoked`'s missing-table rule
 * (operators.ts), applied here. EXACTLY that error falls back to the statement
 * without the predicate; any other fault throws as before.
 */
async function withRevocationClause<T>(run: (clause: (col: string) => string) => Promise<T>): Promise<T> {
  try {
    return await run((col) => ` AND ${NOT_REVOKED(col)}`);
  } catch (e) {
    if (!/no such table: operator_revocations/i.test(e instanceof Error ? e.message : String(e))) throw e;
    return run(() => "");
  }
}

/** I3-1857 — the item a sign-off is about, for `GET /ops/audit?item_id=`. Never
 * an account id and never for an account-subject action: the same withholding
 * `signoffDetail` applies to the detail column (D-0810(7), D-0813(3)). */
function signoffItemId(actionType: string, itemId: string | null): string | null {
  if (itemId === null || itemId === "") return null;
  return actionType.startsWith("account:") || NAMES_AN_ACCOUNT.test(itemId) ? null : itemId;
}

/** I29-OPS (CASHLOTS H6): a digest (40 hex) of one execution's consumed sign-offs — the action, the key and each
 * `requested_by@requested_at`, order-free. A derived name for a set of rows, not a minted id. The claim's DELETE ... RETURNING hands each row to one execution only, so two
 * executions never share it; an executor that re-runs (a redelivery) re-presents it, and an idempotency key bound to it
 * refuses the second write. */
export async function claimIdFor(actionType: string, key: string, signoffs: readonly string[]): Promise<string> {
  const digest = await sha256Hex(`${actionType}\n${key}\n${[...signoffs].sort().join("\n")}`);
  return digest.slice(0, 40);
}

export async function withDualControl<T>(
  db: DbLike,
  env: Record<string, unknown>,
  actor: string,
  actionType: string,
  paramsKey: string,
  /** D-0926(6): receives the op ids whose sign-offs this execution consumed (the actor alone on the fast path), and
   * (I29-OPS) `claim`, an id of exactly those consumed sign-offs, for an executor that binds an idempotency key to it. */
  execute: (signers: readonly string[], claim: string) => Promise<T>,
  now: () => string = () => new Date().toISOString(),
  itemId: string | null = null,
): Promise<DualControlOutcome<T>> {
  // F18 / #32 — TWO ACTIONS ARE NOT CLAMPED, and both are written into the
  // Terms as "two of our operators, not one".
  //
  // `approvalsRequired` clamps the configured N down to the number of
  // operators, so a single-operator deployment is not deadlocked. That is right
  // for every action whose promise is internal. It is wrong for one we PUBLISH:
  // `TERMS.md §7.2` tells a person that terminating their account takes two
  // people, and a review pointed out that with one address on the allowlist it
  // took one — an unclamped promise wired to a clamped seam.
  //
  // This is the warnings precedent, and this module's own docblock states it:
  // the two-person rule there "is always on, has no knob, and a single-operator
  // deployment simply cannot publish an accusation. That is the intended
  // reading of the promise on /how-it-works." Terminating an account and
  // forfeiting what it holds is at least that consequential, so a
  // single-operator deployment simply cannot do it. Suspension is NOT on this
  // list: it is reversible, it takes nothing, and making it undoable-only would
  // leave one operator unable to stop an active fraud.
  // I29-LOOSE L6 (I29-OPS J1, D-1030 I23-Q3): a recovery resolution that unbinds a payout account takes what the Connect
  // unbind door (`connect:unbind`, UNCLAMPED) takes; every other resolution keeps `recovery:resolve`'s threshold.
  // D-1059(4): the reviewed-person rule is asked of the same gate action, so an unbinding resolution takes it too.
  const gateAction = await thresholdActionType(db, actionType, itemId, paramsKey);
  const required = requiredApprovalsFor(env, gateAction);
  const secret = typeof env.SESSION_SECRET === "string" && env.SESSION_SECRET !== "" ? env.SESSION_SECRET : undefined;
  const key = await boundedKey(await withheldParamsKey(paramsKey, secret));

  // Fast path: single-operator. Seam inert; pending_actions untouched.
  if (required <= 1) {
    const result = await execute([actor], await claimIdFor(actionType, key, [`${actor}@${crypto.randomUUID()}`])); // no row was banked: each fast-path run is its own claim
    return { executed: true, pending: false, approvals: 1, required: 1, result };
  }

  const ts = now();
  // Record THIS operator's sign-off + audit it. A RE-SIGN REFRESHES THE ROW:
  // this was `INSERT OR IGNORE` on the composite PK, which under a TTL meant an
  // operator whose sign-off had lapsed could never renew it — their row kept
  // the stale `requested_at`, the gate kept refusing it, and the only way out
  // was the sweep deleting the row first. Found by the sitting-0 tests, not by
  // reasoning. Upserting the timestamp is the honest record: the operator
  // signed again, now.
  await db.batch([
    db
      .prepare(
        "INSERT INTO pending_actions (action_type, params_hash, requested_by, requested_at) VALUES (?, ?, ?, ?) " +
          "ON CONFLICT(action_type, params_hash, requested_by) DO UPDATE SET requested_at = excluded.requested_at",
      )
      .bind(actionType, key, actor, ts),
    auditStmt(db, actor, "dual_control:signoff", signoffItemId(actionType, itemId), signoffDetail(actionType, key), ts),
  ]);

  // DISTINCT PEOPLE, NOT DISTINCT IDS. `requested_by` is an `op_` id, and an
  // `op_` id is minted per ADDRESS — so one human under two addresses used to
  // satisfy a 2-of-N gate alone, exactly the defect the warnings two-person
  // rule had. COALESCE keeps an id with no identity row as its own person,
  // which can only ever make the gate stricter. See `operators.ts`.
  // …AND ONLY LIVE SIGN-OFFS. A row older than `PENDING_SIGNOFF_TTL_MS` does
  // not count, sweep or no sweep — see the constant's docblock. The operator
  // who just signed is inside the window by construction (their row carries
  // `ts`), so a lapsed co-signer leaves the count at 1 and the action pending;
  // the second approver simply signs again and the two fresh rows combine.
  const cutoff = signoffCutoff(ts);
  const live = liveOperatorsJson(env);
  const distinct = (
    await withRevocationClause((rev) =>
      db
        .prepare(
          "SELECT COUNT(DISTINCT COALESCE((SELECT person_id FROM operator_identities WHERE op_id = p.requested_by), p.requested_by)) AS n " +
            `FROM pending_actions p WHERE p.action_type = ? AND p.params_hash = ? AND p.requested_at >= ? AND ${STILL_ALLOWLISTED("p.requested_by")}${rev("p.requested_by")}${reviewed(gateAction, "p.requested_by")}`,
        )
        .bind(actionType, key, cutoff, live)
        .first<{ n: number }>(),
    )
  )?.n ?? 0;

  if (distinct < required) {
    if (!needsReviewedPerson(gateAction)) return { executed: false, pending: true, approvals: distinct, required };
    const banked = (
      await withRevocationClause((rev) =>
        db
          .prepare(
            "SELECT COUNT(DISTINCT COALESCE((SELECT person_id FROM operator_identities WHERE op_id = p.requested_by), p.requested_by)) AS n " +
              `FROM pending_actions p WHERE p.action_type = ? AND p.params_hash = ? AND p.requested_at >= ? AND ${STILL_ALLOWLISTED("p.requested_by")}${rev("p.requested_by")}`,
          )
          .bind(actionType, key, cutoff, live)
          .first<{ n: number }>(),
      )
    )?.n ?? 0;
    return { executed: false, pending: true, approvals: distinct, required, unreviewed: Math.max(0, banked - distinct) };
  }

  // CLAIM BEFORE EXECUTE, NOT AFTER (INV-057 / QUEUE #81). The old order —
  // execute() then a SEPARATE later DELETE — left a window the width of
  // execute() itself (for account:delete, an erasure across many tables)
  // during which pending_actions still held >= required LIVE sign-offs. A
  // third call landing inside that window recomputed the same distinct
  // count above, saw the threshold met AGAIN, and re-executed against an
  // action already committed: the two-key ceremony spent twice on one
  // erasure.
  //
  // The fix moves the clear to where the decision is made and makes it ONE
  // statement: DELETE ... RETURNING both consumes the rows and reports
  // exactly which ones it consumed, so this is the same enumerate-by-
  // RETURNING exactly-once pattern `digest.ts` (verify-claims, :202) and
  // `fan-auth.ts` (:398,:518) already use — no lock invented (db.ts:39-46).
  // SQLite/D1 admit no other writer between "which rows qualify" and "they
  // are gone" inside one statement, so two calls racing for the same
  // sign-offs cannot both spend them: the first DELETE to run empties the
  // key; the second's RETURNING comes back empty (or short) and it does
  // NOT execute — closing the window instead of narrowing it.
  const claimed = await withRevocationClause((rev) =>
    db
      .prepare(
        `DELETE FROM pending_actions WHERE action_type = ? AND params_hash = ? AND requested_at >= ? AND ${STILL_ALLOWLISTED("pending_actions.requested_by")}${rev("pending_actions.requested_by")}${reviewed(gateAction, "pending_actions.requested_by")} ` +
          "RETURNING requested_by, requested_at, " +
          "COALESCE((SELECT person_id FROM operator_identities WHERE op_id = requested_by), requested_by) AS person",
      )
      .bind(actionType, key, cutoff, live)
      .all<{ requested_by: string; requested_at: string; person: string }>(),
  );
  const claimedDistinct = new Set(claimed.results.map((r) => r.person)).size;

  const restore = (): Promise<unknown> =>
    claimed.results.length === 0
      ? Promise.resolve()
      : db.batch(
          claimed.results.map((r) =>
            db
              .prepare(
                "INSERT INTO pending_actions (action_type, params_hash, requested_by, requested_at) VALUES (?, ?, ?, ?) " +
                  "ON CONFLICT(action_type, params_hash, requested_by) DO NOTHING",
              )
              .bind(actionType, key, r.requested_by, r.requested_at),
          ),
        );

  if (claimedDistinct < required) {
    // Lost the race: another call's DELETE already emptied this key (it is
    // the one executing, or already has) — whatever THIS delete returned
    // was never enough to execute on its own. Put it back: a genuine
    // not-yet-consumed sign-off must not vanish because it lost a race it
    // was never part of. ON CONFLICT DO NOTHING so a fresher re-sign that
    // landed after our DELETE keeps its own timestamp.
    await restore();
    return { executed: false, pending: true, approvals: claimedDistinct, required };
  }

  // This call owns the one execution — the rows are already gone, so there
  // is nothing left to double-spend even if execute() runs long (the
  // erasure case) or a caller replays mid-flight.
  //
  // D-0646 — AND A FAILED EXECUTION KEEPS THEM GONE. There is deliberately no
  // catch arm: the wipe IS the claim DELETE above (:405), which committed
  // before execute() was entered, so no ordering exists in which the action
  // refuses and the sign-offs survive. A catch arm that wiped would BE that
  // ordering.
  //
  // What it replaced: `restore()` ran here too, putting every claimed row back
  // with its ORIGINAL `requested_at` and leaving `pending_actions` holding >=
  // `required` LIVE sign-offs with nothing pending — a state no non-throwing
  // path produces, and the one in which ONE operator then executed alone. The
  // approvers consented to an action that did not happen; re-spending that
  // consent on an attempt they never saw is not dual control. A second attempt
  // costs two FRESH sign-offs, and the re-sign after a transient fault is the
  // accepted price of that (D-0646). Telling a policy refusal from an
  // infrastructure fault needs callers to mark the difference and is a
  // separate question — filed, not built.
  //
  // `restore()` keeps its other caller at :436, which is a different event: a
  // claim that LOST the race never ran the action at all.
  const result = await execute(
    claimed.results.map((r) => r.requested_by),
    await claimIdFor(actionType, key, claimed.results.map((r) => `${r.requested_by}@${r.requested_at}`)),
  );
  return { executed: true, pending: false, approvals: claimedDistinct, required, result };
}

/** Pending sign-offs for the dashboard: what's awaiting more approvers. */
export interface PendingAction {
  action_type: string;
  params_hash: string;
  approvals: number;
  first_requested_at: string;
  requested_by: string[];
  /** S-070 OW-01 — THE THRESHOLD THIS ROW'S OWN ACTION NEEDS, from
   * `requiredApprovalsFor`, which is the same function the gate enforces with.
   * Optional on the type only because `listPending`/`listPendingPage` have no
   * `env` to compute it from; `withPendingRequirements` fills it and the
   * console read calls that. A renderer must show THIS number, never the
   * deployment-wide `approvals_required`, because they differ for exactly the
   * action where being wrong matters most. */
  required?: number;
}

/** S-070 OW-01 — stamp each pending row with its own action's threshold. Kept
 * beside `requiredApprovalsFor` so the two can never drift, and applied by the
 * read route rather than inside `listPendingPage` (which is deliberately
 * env-free and is also called by tests and by the sweep). */
export function withPendingRequirements(pending: PendingAction[], env: Record<string, unknown>): PendingAction[] {
  return pending.map((p) => ({ ...p, required: requiredApprovalsFor(env, p.action_type) }));
}

/**
 * I29-LOOSE L6 / I29-FX-OPS (D-1037(15)): the action whose threshold the gate applies. A `recovery:resolve` whose plan
 * unbinds a payout account (recovery-unbind.ts `recoveryResolveUnbindsPayout`, imported statically: this is a paywall
 * verdict module) takes what the Connect unbind door (`connect:unbind`, UNCLAMPED) takes; every other action is itself.
 * ONE function for the gate (`withDualControl`) and the pending list (`withLivePendingRequirements`), so the number an
 * operator reads is the number the gate enforces.
 */
export async function thresholdActionType(db: DbLike, actionType: string, itemId: string | null, paramsKey: string): Promise<string> {
  return actionType === "recovery:resolve" && itemId !== null && (await recoveryResolveUnbindsPayout(db, itemId, paramsKey)) ? "connect:unbind" : actionType;
}

/**
 * LOOSE H5 (I29-FX-OPS): `withPendingRequirements` with the recovery plan read, for the console's pending list. A
 * `recovery:resolve` row's stored key is `<case id>` or `<case id>|<plan>` (console.ts, `recoveryPlanKey`), never an
 * account key, so the case id is the text before the first `|` and the key is what the gate read. A key longer than
 * `boundedKey`'s 200 characters is stored truncated with a digest; its plan may be cut, which the predicate reads as
 * unbinding (fail closed), so the list can show the stricter number for such a row, never the weaker one.
 */
export async function withLivePendingRequirements(db: DbLike, pending: PendingAction[], env: Record<string, unknown>): Promise<PendingAction[]> {
  const out: PendingAction[] = [];
  for (const p of pending) {
    const bar = p.params_hash.indexOf("|");
    const itemId = p.action_type === "recovery:resolve" ? (bar >= 0 ? p.params_hash.slice(0, bar) : p.params_hash) : null;
    out.push({ ...p, required: requiredApprovalsFor(env, await thresholdActionType(db, p.action_type, itemId, p.params_hash)) });
  }
  return out;
}

export async function listPending(db: DbLike, nowIso: string = new Date().toISOString()): Promise<PendingAction[]> {
  return (await listPendingPage(db, nowIso)).pending;
}

/**
 * The dashboard read, with the two bounds the gate has: only LIVE sign-offs
 * (same cutoff as `withDualControl`), and at most `PENDING_LIST_MAX` rows —
 * newest first, so what is truncated is the oldest of a queue that is already
 * pathological, and `truncated` says so rather than the page pretending to be
 * the whole table.
 */
export async function listPendingPage(db: DbLike, nowIso: string = new Date().toISOString()): Promise<{ pending: PendingAction[]; truncated: boolean }> {
  // `approvals` MUST BE COUNTED THE WAY THE GATE COUNTS IT. One row is banked per
  // `op_` id, but `withDualControl` above counts DISTINCT PERSONS — so before
  // slice 29 the two agreed by construction and now they do not. Counting rows
  // here showed "2 of 2" on a gate that would never fire, next to the same
  // human's display name printed twice. The person is resolved in SQL with the
  // same COALESCE the gate uses; `requested_by` still lists every address, so
  // the console can show who signed without implying they were two signatures.
  const cutoff = signoffCutoff(nowIso);
  // A KEY IS WHOLE OR ABSENT. The first cut capped ROWS newest-first, which
  // could cut one action's sign-offs across the boundary and render "1 of 3"
  // for a key the gate counted at 2 (review persona 89). So the cap is on
  // KEYS: the newest `PENDING_LIST_MAX` (action_type, params_hash) pairs by
  // their latest live sign-off, then every live row of those keys.
  const { results: keys } = await withRevocationClause((rev) =>
    db
      .prepare(
        `SELECT action_type, params_hash, MAX(requested_at) AS latest FROM pending_actions WHERE requested_at >= ?${rev("pending_actions.requested_by")} ` +
          "GROUP BY action_type, params_hash ORDER BY latest DESC, action_type, params_hash LIMIT ?",
      )
      .bind(cutoff, PENDING_LIST_MAX + 1)
      .all<{ action_type: string; params_hash: string; latest: string }>(),
  );
  const truncated = keys.length > PENDING_LIST_MAX;
  const kept = keys.slice(0, PENDING_LIST_MAX);
  if (kept.length === 0) return { pending: [], truncated };
  // QUEUE #127 OR-02: THE KEY PREDICATE IS IN THE SQL. This read used to fetch
  // EVERY live row and filter to the kept keys in JS, so its cost was the whole
  // live table however small the page. It now JOINs the same newest-keys
  // subquery (same ORDER BY, LIMIT PENDING_LIST_MAX), so it returns only rows of
  // at most PENDING_LIST_MAX keys, each whole (every live row of a kept key).
  // The two statements can see different key sets if a sign-off lands between
  // them; the intersection with `kept` below keeps the page to the keys the
  // `truncated` verdict was computed over, and a key is still whole or absent.
  const keep = new Set(kept.map((k) => `${k.action_type}\u0000${k.params_hash}`));
  const { results: all } = await withRevocationClause((rev) =>
    db
      .prepare(
        "SELECT p.action_type AS action_type, p.params_hash AS params_hash, p.requested_by AS requested_by, p.requested_at AS requested_at, " +
          "COALESCE((SELECT person_id FROM operator_identities WHERE op_id = p.requested_by), p.requested_by) AS person " +
          `FROM pending_actions p JOIN (SELECT action_type, params_hash, MAX(requested_at) AS latest FROM pending_actions WHERE requested_at >= ?${rev("pending_actions.requested_by")} ` +
          "GROUP BY action_type, params_hash ORDER BY latest DESC, action_type, params_hash LIMIT ?) k " +
          "ON k.action_type = p.action_type AND k.params_hash = p.params_hash " +
          `WHERE p.requested_at >= ?${rev("p.requested_by")} ORDER BY p.action_type, p.params_hash, p.requested_at`,
      )
      .bind(cutoff, PENDING_LIST_MAX, cutoff)
      .all<{ action_type: string; params_hash: string; requested_by: string; requested_at: string; person: string }>(),
  );
  const results = all.filter((r) => keep.has(`${r.action_type}\u0000${r.params_hash}`));
  const byKey = new Map<string, PendingAction>();
  const personsByKey = new Map<string, Set<string>>();
  for (const r of results) {
    const k = `${r.action_type}\u0000${r.params_hash}`;
    const persons = personsByKey.get(k) ?? new Set<string>();
    persons.add(r.person);
    personsByKey.set(k, persons);
    const existing = byKey.get(k);
    if (existing) {
      existing.requested_by.push(r.requested_by);
    } else {
      byKey.set(k, {
        action_type: r.action_type,
        params_hash: r.params_hash,
        approvals: 0,
        first_requested_at: r.requested_at,
        requested_by: [r.requested_by],
      });
    }
  }
  for (const [k, v] of byKey) v.approvals = personsByKey.get(k)?.size ?? 0;
  return { pending: [...byKey.values()], truncated };
}

/** QUEUE #178 AS-056 — THE BANKED SIGN-OFF, MADE VISIBLE. Every LIVE sign-off
 * whose params key starts with `prefix` for one action type, grouped per key,
 * with `approvals` counted the way the gate counts them (distinct persons, same
 * cutoff, same COALESCE as `withDualControl`). `substr(...) = ?` rather than
 * LIKE: a key is operator/route-built text and must not be read as a pattern.
 * Read-only: nothing here can bank, spend or count toward a gate. */
export interface BankedSignoff {
  params_hash: string;
  approvals: number;
  signed: { op_id: string; at: string }[];
}
export async function liveSignoffsByPrefix(db: DbLike, actionType: string, prefix: string, nowIso: string): Promise<BankedSignoff[]> {
  const cutoff = signoffCutoff(nowIso);
  const { results } = await withRevocationClause((rev) =>
    db
      .prepare(
        "SELECT params_hash, requested_by, requested_at, COALESCE((SELECT person_id FROM operator_identities WHERE op_id = requested_by), requested_by) AS person " +
          `FROM pending_actions WHERE action_type = ? AND substr(params_hash, 1, ?) = ? AND requested_at >= ?${rev("pending_actions.requested_by")} ` +
          "ORDER BY params_hash, requested_at LIMIT 200",
      )
      .bind(actionType, prefix.length, prefix, cutoff)
      .all<{ params_hash: string; requested_by: string; requested_at: string; person: string }>(),
  );
  const byKey = new Map<string, { row: BankedSignoff; persons: Set<string> }>();
  for (const r of results) {
    const e = byKey.get(r.params_hash) ?? { row: { params_hash: r.params_hash, approvals: 0, signed: [] }, persons: new Set<string>() };
    e.row.signed.push({ op_id: r.requested_by, at: r.requested_at });
    e.persons.add(r.person);
    byKey.set(r.params_hash, e);
  }
  return [...byKey.values()].map((e) => ({ ...e.row, approvals: e.persons.size }));
}

/** I38-OPSUNDO: an operator WITHDRAWS their own banked sign-off. A `pending_actions` row is by construction not yet
 * executed (the claim DELETE in `withDualControl` consumes the rows it executes on), so removing it means the request
 * needs a fresh sign-off to reach its threshold. `requestedBy` names whose sign-off; it must be the caller's own op id,
 * or the call is refused (403) and nothing is written. One batch: the DELETE keyed to the caller, `changeGateStmt`
 * (aborts when no row of the caller's was removed -> 404) and the `dual_control:withdraw` audit line, whose detail is
 * `signoffDetail` so an account id stays withheld exactly as on the sign-off row. */
export class SignoffWithdrawError extends Error {
  override name = "SignoffWithdrawError";
  constructor(message: string, public readonly status: 400 | 403 | 404) {
    super(message);
  }
}
export async function withdrawSignoff(db: DbLike, actor: string, actionType: string, paramsHash: string, requestedBy: string = actor, now: () => string = () => new Date().toISOString()): Promise<{ withdrawn: true }> {
  if (typeof actionType !== "string" || actionType === "" || actionType.length > 64 || typeof paramsHash !== "string" || paramsHash === "" || paramsHash.length > 200 || typeof requestedBy !== "string") {
    throw new SignoffWithdrawError("bad_request", 400);
  }
  if (requestedBy !== actor) throw new SignoffWithdrawError("you can only withdraw your own sign-off", 403);
  const ts = now();
  try {
    await db.batch([
      db.prepare("DELETE FROM pending_actions WHERE action_type = ? AND params_hash = ? AND requested_by = ?").bind(actionType, paramsHash, actor),
      changeGateStmt(db),
      auditStmt(db, actor, "dual_control:withdraw", null, signoffDetail(actionType, paramsHash), ts),
    ]);
  } catch (e) {
    if (isChangeGateRefusal(e)) throw new SignoffWithdrawError("no banked sign-off of yours on this request", 404);
    throw e;
  }
  return { withdrawn: true };
}

/**
 * The LEDGER SEAM for the weekly edition-set anchor (ratings.ts
 * `anchorCompletedWeeks`).
 *
 * It lives here rather than in ratings.ts because that module is deliberately
 * kept out of the ledger writer path — it takes the append as an injected
 * callback so `events.ts` stays the only module that writes `trust_events` —
 * and it is not inlined in index.ts because it is the thing a race test has to
 * exercise. A copy of it in a test would prove nothing about what ships.
 *
 * ANCHOR ONCE. `week` is the PRIMARY KEY of `edition_set_anchors` and the row
 * rides the SAME atomic batch as the event (`extraStatements`, the
 * `appendCreatorSigned` idiom), so two overlapping sweeps cannot both commit:
 * the loser fails the constraint, the whole batch — event included — rolls
 * back, and `tryAppendEvents` hands that back as a `conflict` refusal. Returning
 * `null` for it is what makes the caller SKIP the week instead of anchoring it
 * twice. Anything else (a validator refusal, contention) is a real failure and
 * propagates: an anchor must never be silently dropped.
 */
export function editionSetAnchorAppender(
  db: DbLike,
  now: () => string = () => new Date().toISOString(),
): (week: string, setHash: string, editionsCount: number) => Promise<{ event_id: string } | null> {
  return async (week, setHash, editionsCount) => {
    const lock = db
      .prepare("INSERT INTO edition_set_anchors (week, set_hash, editions_count, anchored_at) VALUES (?, ?, ?, ?)")
      .bind(week, setHash, editionsCount, now());
    const r = await tryAppendEvents(
      db,
      [
        {
          event_type: "edition_set_anchored",
          subject: `sha256:${setHash}`,
          actor_kind: "system",
          actor_id: "system:editions-sweep",
          payload: { week, set_hash: setHash, editions_count: editionsCount },
        },
      ] as DraftEvent[],
      now,
      [],
      [lock],
      ["system"],
    );
    if (r.ok) return { event_id: r.events[0]!.event_id };
    if (r.refusal.kind === "conflict") return null; // another sweep already anchored this week
    throw r.refusal.error;
  };
}

export const MAX_LEAVES = 8;
export const MAX_DEPTH = 2;
const RELEASE_ID_RE = /^rel_[0-9a-f]{16}$/;
const PSP_RE = /^psp_[0-9a-f]{16}$/;
// The same shape reverse-codes.ts ACHIEVEMENT_REF_RE admits (lowercase, ≤64),
// so a rule can never name an id no writer can mint.
const ACHIEVEMENT_RE = /^(?=.{1,64}$)[a-z0-9_]+(?::[a-z0-9_]+){0,3}$/;
const DAY_MS = 24 * 60 * 60 * 1000;

export type EligibilityRule =
  | { kind: "save" }
  | { kind: "acquisition"; release_id: string }
  | { kind: "achievement"; achievement_id: string }
  | { kind: "bond"; min_level: number }
  | { kind: "founding_fan" }
  | { kind: "early_scout" }
  | { kind: "supporter"; tier: SupporterTier }
  | { kind: "account_age"; min_days: number }
  | { kind: "connected"; min_providers: number }
  | { kind: "all"; rules: EligibilityRule[] }
  | { kind: "any"; rules: EligibilityRule[] };

export type LeafKind = Exclude<EligibilityRule["kind"], "all" | "any">;
export const LEAF_KINDS: readonly LeafKind[] = ["save", "acquisition", "achievement", "bond", "founding_fan", "early_scout", "supporter", "account_age", "connected"];

export class EligibilityError extends Error {
  override name = "EligibilityError";
  constructor(
    message: string,
    public readonly status: 400 | 403 = 400,
  ) {
    super(message);
  }
}

/** The platform-wide achievement ids a creator may gate on without owning
 * them (mirrors reverse-codes.ts CREATOR_NEUTRAL_ACHIEVEMENTS — kept in one
 * place there; this module asks the same question by suffix). */
export function achievementGrantableBy(achievementId: string, passportId: string, neutral: ReadonlySet<string>): boolean {
  if (!ACHIEVEMENT_RE.test(achievementId)) return false;
  if (neutral.has(achievementId)) return true;
  if (!PSP_RE.test(passportId)) return false;
  // A creator's custom badge (`custom:<psp>:<slug>`, reward-policy.ts) carries
  // its owner in the middle and NOTHING else decides it — a slug that happens
  // to end in another passport's id must not fall through to the suffix rule.
  if (achievementId.startsWith("custom:")) return achievementId.startsWith(`custom:${passportId}:`);
  // Every other creator-scoped id carries its owner as a suffix
  // (`completion:psp_A` must not be grantable by psp_B).
  return achievementId.endsWith(passportId);
}

/**
 * Parse an untrusted rule for a creator. Unknown fields anywhere → refused;
 * unknown kinds → refused; bounds enforced. Cross-creator refs are refused with
 * ONE sentence that confirms nothing about the guessed ref.
 */
export function parseRule(raw: unknown, passportId: string, neutralAchievements: ReadonlySet<string>): EligibilityRule {
  const counter = { leaves: 0 };
  return parseNode(raw, passportId, neutralAchievements, 0, counter);
}

function only(obj: Record<string, unknown>, keys: string[]): void {
  for (const k of Object.keys(obj)) if (!keys.includes(k)) throw new EligibilityError(`unknown field "${k}" in eligibility rule`);
}

function parseNode(raw: unknown, passportId: string, neutral: ReadonlySet<string>, depth: number, counter: { leaves: number }): EligibilityRule {
  if (typeof raw !== "object" || raw === null || Array.isArray(raw)) throw new EligibilityError("eligibility rule must be an object");
  const o = raw as Record<string, unknown>;
  const kind = o.kind;
  if (typeof kind !== "string") throw new EligibilityError("eligibility rule needs a kind");
  if (kind === "all" || kind === "any") {
    only(o, ["kind", "rules"]);
    if (depth >= MAX_DEPTH) throw new EligibilityError(`eligibility rule nests deeper than ${MAX_DEPTH}`);
    if (!Array.isArray(o.rules) || o.rules.length === 0) throw new EligibilityError("all/any needs a non-empty rules array");
    const rules = o.rules.map((r) => parseNode(r, passportId, neutral, depth + 1, counter));
    return { kind, rules };
  }
  counter.leaves += 1;
  if (counter.leaves > MAX_LEAVES) throw new EligibilityError(`eligibility rule has more than ${MAX_LEAVES} conditions`);
  switch (kind) {
    case "save":
    case "founding_fan":
    case "early_scout":
      only(o, ["kind"]);
      return { kind };
    case "acquisition": {
      only(o, ["kind", "release_id"]);
      if (typeof o.release_id !== "string" || !RELEASE_ID_RE.test(o.release_id)) throw new EligibilityError("acquisition rule needs a release id");
      // Ownership of the release is a DB fact the caller checks (createPool
      // already does); the id shape is all a pure parser can assert.
      return { kind, release_id: o.release_id };
    }
    case "achievement": {
      only(o, ["kind", "achievement_id"]);
      if (typeof o.achievement_id !== "string" || !ACHIEVEMENT_RE.test(o.achievement_id)) throw new EligibilityError("achievement rule needs an achievement id");
      if (!achievementGrantableBy(o.achievement_id, passportId, neutral)) {
        throw new EligibilityError("you can only gate on your own achievements or a platform-wide one", 403);
      }
      return { kind, achievement_id: o.achievement_id };
    }
    case "bond": {
      only(o, ["kind", "min_level"]);
      if (!Number.isInteger(o.min_level) || (o.min_level as number) < 1 || (o.min_level as number) > 10) throw new EligibilityError("bond rule needs min_level 1..10");
      return { kind, min_level: o.min_level as number };
    }
    case "supporter": {
      only(o, ["kind", "tier"]);
      if (typeof o.tier !== "string" || !SUPPORTER_TIERS.includes(o.tier as SupporterTier)) throw new EligibilityError("supporter rule needs tier active|core");
      return { kind, tier: o.tier as SupporterTier };
    }
    case "account_age": {
      only(o, ["kind", "min_days"]);
      if (!Number.isInteger(o.min_days) || (o.min_days as number) < 1 || (o.min_days as number) > 3650) throw new EligibilityError("account_age rule needs min_days 1..3650");
      return { kind, min_days: o.min_days as number };
    }
    case "connected": {
      // F8 CONSTELLATION — how many external platforms vouch for this fan.
      // Bounded by the provider list itself: a rule that can never be satisfied
      // is a gate a creator sets by accident and a fan can never pass.
      only(o, ["kind", "min_providers"]);
      if (!Number.isInteger(o.min_providers) || (o.min_providers as number) < 1 || (o.min_providers as number) > FAN_PROVIDER_KEYS.length) {
        throw new EligibilityError(`connected rule needs min_providers 1..${FAN_PROVIDER_KEYS.length}`);
      }
      return { kind, min_providers: o.min_providers as number };
    }
    default:
      throw new EligibilityError(`unknown eligibility kind "${kind}"`);
  }
}

/** A stored rule (JSON text) back to a rule, or null — a stored rule that no
 * longer parses is nobody's eligibility (fail closed), never a 500. */
export function ruleFromStored(text: string | null | undefined, passportId: string, neutral: ReadonlySet<string>): EligibilityRule | null {
  if (typeof text !== "string" || text.length === 0 || text.length > 4096) return null;
  try {
    return parseRule(JSON.parse(text), passportId, neutral);
  } catch {
    return null;
  }
}

/** The legacy three-kind pool shape as a rule (so ONE evaluator serves the
 * pools created before F4 and the drop unifier's defaults). */
export function ruleFromLegacy(kind: string, ref: string | null): EligibilityRule | null {
  if (kind === "save") return { kind: "save" };
  if (kind === "acquisition" && typeof ref === "string" && RELEASE_ID_RE.test(ref)) return { kind: "acquisition", release_id: ref };
  if (kind === "achievement" && typeof ref === "string" && ACHIEVEMENT_RE.test(ref)) return { kind: "achievement", achievement_id: ref };
  return null;
}

export interface LeafResult {
  kind: LeafKind;
  satisfied: boolean;
}
export interface EligibilitySnapshot {
  rule: EligibilityRule;
  satisfied: boolean;
  evaluated_at: string;
  /** Every leaf that was EVALUATED (short-circuit means some may be absent)
   * and how it came out — no ids, no counts, no free text. */
  leaves: LeafResult[];
}

/** Does the fan qualify? Bounded reads only; short-circuits. */
export async function evaluate(db: DbLike, accountId: string, passportId: string, rule: EligibilityRule, nowMs: number): Promise<boolean> {
  return (await snapshot(db, accountId, passportId, rule, nowMs)).satisfied;
}

/** The same evaluation, with the record of what was true — the #17 seam. */
export async function snapshot(db: DbLike, accountId: string, passportId: string, rule: EligibilityRule, nowMs: number): Promise<EligibilitySnapshot> {
  const leaves: LeafResult[] = [];
  const satisfied = PSP_RE.test(passportId) && typeof accountId === "string" && accountId.length > 0 ? await evalNode(db, accountId, passportId, rule, nowMs, leaves) : false;
  return { rule, satisfied, evaluated_at: new Date(nowMs).toISOString(), leaves };
}

async function evalNode(db: DbLike, accountId: string, passportId: string, rule: EligibilityRule, nowMs: number, leaves: LeafResult[]): Promise<boolean> {
  if (rule.kind === "all") {
    for (const r of rule.rules) if (!(await evalNode(db, accountId, passportId, r, nowMs, leaves))) return false;
    return true;
  }
  if (rule.kind === "any") {
    for (const r of rule.rules) if (await evalNode(db, accountId, passportId, r, nowMs, leaves)) return true;
    return false;
  }
  const ok = await evalLeaf(db, accountId, passportId, rule, nowMs);
  leaves.push({ kind: rule.kind, satisfied: ok });
  return ok;
}

async function exists(db: DbLike, sql: string, binds: unknown[]): Promise<boolean> {
  return (await db.prepare(sql).bind(...binds).first()) !== null;
}

async function evalLeaf(db: DbLike, accountId: string, passportId: string, rule: Exclude<EligibilityRule, { kind: "all" } | { kind: "any" }>, nowMs: number): Promise<boolean> {
  switch (rule.kind) {
    case "save":
      return exists(db, "SELECT 1 AS x FROM saves WHERE account_id = ? AND passport_id = ? LIMIT 1", [accountId, passportId]);
    case "acquisition":
      // EVIDENCE, not property (0061 / F4b): redeemed by this fan, themselves —
      // whoever holds the item now. The redeemer keeps what the redemption
      // earned after gifting it away; a gifted-in holder never qualifies.
      // I3-2147 / I3-1978: bound to THIS creator's release. A rule naming a
      // foreign or nonexistent release can never be satisfied, whichever door
      // stored it (question settings never checked ownership at write).
      return exists(
        db,
        "SELECT 1 AS x FROM acquisitions a JOIN releases r ON r.release_id = a.release_id AND r.passport_id = ? WHERE a.release_id = ? AND a.redeemed_by = ? LIMIT 1",
        [passportId, rule.release_id, accountId],
      );
    case "achievement":
      return exists(db, "SELECT 1 AS x FROM achievements WHERE account_id = ? AND achievement_id = ? LIMIT 1", [accountId, rule.achievement_id]);
    case "bond":
      return (await getBond(db, accountId, passportId, nowMs)).level >= rule.min_level;
    case "founding_fan":
      return exists(db, "SELECT 1 AS x FROM founding_fan_ranks WHERE passport_id = ? AND account_id = ? LIMIT 1", [passportId, accountId]);
    case "early_scout":
      return exists(db, "SELECT 1 AS x FROM achievements WHERE account_id = ? AND achievement_id = ? LIMIT 1", [accountId, `early_supporter:${passportId}`]);
    case "supporter":
      return meetsTier(await supporterTier(db, accountId, passportId, nowMs), rule.tier);
    case "account_age": {
      const cutoff = new Date(nowMs - rule.min_days * DAY_MS).toISOString();
      // A NULL or future created_at fails the <= (unknown age = too new).
      return exists(db, "SELECT 1 AS x FROM accounts WHERE account_id = ? AND created_at <= ? LIMIT 1", [accountId, cutoff]);
    }
    case "connected":
      // A COUNT of platforms, never WHICH — the leaf cannot express "connected
      // on Twitch", deliberately, because a creator who could gate on one
      // platform could learn a fan is on it by watching who passes. It is also
      // PLATFORM-BLIND for the reason `entitlements.ts:10` is tier-blind: this
      // is a proof-of-personhood count, and turning it into a preference for
      // one platform's audience is a different product decision entirely.
      return (await connectionCount(db, accountId)) >= rule.min_providers;
    default:
      return false;
  }
}

/** A one-line, fan-safe description of a rule for the offer page — names
 * kinds, never ids. */
export function describeRule(rule: EligibilityRule): string {
  const leaf = (r: EligibilityRule): string => {
    switch (r.kind) {
      case "save":
        return "saved this creator";
      case "acquisition":
        return "redeemed the linked collectible";
      case "achievement":
        return "earned the linked achievement";
      case "bond":
        return `a Bond of level ${r.min_level} or more with this creator`;
      case "founding_fan":
        return "a Founding Fan slot for this creator";
      case "early_scout":
        return "the Early Scout badge for this creator";
      case "supporter":
        return r.tier === "core" ? "a core supporter of this creator" : "an active supporter of this creator";
      case "account_age":
        return `an account at least ${r.min_days} day${r.min_days === 1 ? "" : "s"} old`;
      case "connected":
        return `an account connected to at least ${r.min_providers} platform${r.min_providers === 1 ? "" : "s"}`;
      // A nested combinator is parenthesised, so "A and (B or C)" cannot read as
      // "(A and B) or C" (F4 review, Business Logic Skeptic).
      case "all":
        return r.rules.map((x) => (x.kind === "all" || x.kind === "any" ? `(${leaf(x)})` : leaf(x))).join(" and ");
      case "any":
        return r.rules.map((x) => (x.kind === "all" || x.kind === "any" ? `(${leaf(x)})` : leaf(x))).join(" or ");
      default:
        return "the creator's conditions";
    }
  };
  return leaf(rule);
}

/**
 * Transactional email — a fail-closed sender seam with a generic HTTP adapter,
 * mirroring `push.ts`. Used as a SECOND Save-Creator alert channel (off the hot
 * request path: the alerts cron drives it), so it never blocks a user request.
 *
 * The adapter posts to a provider-agnostic JSON API (`EMAIL_API_URL` +
 * `EMAIL_API_KEY`), shaped like the common transactional providers
 * (from/to/subject/text/html + optional headers). Absent/placeholder config →
 * the sender is disabled (null sender): notifications stay pullable and are
 * emailed once the secret is supplied. Never the trust boundary.
 */

import { DISPUTE_FINGERPRINT_SCOPE, previousSessionSecrets, scopedKeyedMac } from "./sessions.js"; // not account_data.ts: that reaches billing (paywall invariant, I23-L05)

export interface EmailMessage {
  to: string;
  subject: string;
  text: string;
  html?: string;
  /** RFC 2369 List-Unsubscribe value (a <mailto:…> or <https://…> in brackets). */
  listUnsubscribe?: string;
  /**
   * RFC 8058 one-click. Set ONLY when `listUnsubscribe` is an https URL that
   * accepts an unauthenticated POST and unsubscribes with no further
   * interaction — i.e. this Worker's `POST /unsubscribe/:token`, never an app
   * or console fragment (a POST to a Pages asset is a 405, and a provider that
   * gets a non-2xx from a one-click URL marks the sender down).
   *
   * The adapter emits the header only alongside `listUnsubscribe`, so this flag
   * can never produce a lone `List-Unsubscribe-Post` — which RFC 8058 §3 calls
   * out as invalid and which some providers treat as a spam signal.
   */
  listUnsubscribePost?: boolean;
}

/** Returns true when the provider accepted the message. */
export type EmailSender = (msg: EmailMessage) => Promise<boolean>;

/**
 * I3-0014. What the provider said about ONE attempt, for the one caller that can
 * act on the difference.
 *
 * `accepted` is a 2xx. `permanent` is a refusal that re-offering the same message
 * to the same address cannot turn into an acceptance — a dead domain, a mailbox
 * that is gone, a recipient the provider has on its own suppression list. Every
 * other outcome is `transient`: a 5xx, a rate limit, a socket error — or
 * `ambiguous`: WE stopped the attempt (this rail's own abort, or a caller's race
 * giving up) after the request may have reached the provider, so whether it was
 * accepted cannot be known. Mail cannot be recalled, and the provider offers no
 * idempotency key (D-0837(1)), so a caller settles `ambiguous` as SENT: at most
 * once (D-0858(3), QUEUE #739 limb 739-F).
 *
 * A BOOLEAN COULD NOT CARRY THIS, and that is the whole defect: `deliverPendingEmail`
 * leaves a failed row `emailed = 0` so the next pass retries it, which is right for
 * a hiccup and wrong forever for a dead address — 288 passes a day against a
 * mailbox that will never accept, which is what costs money and sender reputation.
 */
export type EmailOutcome = "accepted" | "transient" | "permanent" | "ambiguous";

/**
 * A sender that can also say WHY a send failed.
 *
 * DELIBERATELY A SECOND ENTRY POINT ON THE SAME FUNCTION, not a widened
 * `EmailSender` return type — and W3-09a-FR1 SETTLED that as the permanent
 * answer by executing the widening rather than reasoning about it. With
 * `EmailSender` returning `Promise<EmailOutcome>`, `tsc -p services/trust-core`
 * reports exactly two src errors (`digest.ts` and this file) and SIXTY-THREE in
 * `test/` — while the two call sites that would silently INVERT report nothing
 * at all: `alerts.ts` then coerced with `ok ? "accepted" : "transient"` (that
 * coercion now lives only in `sendOnce` below), and
 * `creator-contact-routes.ts:264` branched on `if (!ok)`; a truthy `"transient"`
 * makes both of them report a refusal as a success, with no compiler error to
 * catch it. A widening whose two dangerous sites are the two the compiler is
 * silent about is not a migration, so the boolean stays as the adapter surface
 * and `sendOnce` below is the one way to send when the answer matters.
 */
export interface ClassifyingEmailSender extends EmailSender {
  sendClassified(msg: EmailMessage): Promise<EmailOutcome>;
}

/** True when this sender can classify a failure (see `ClassifyingEmailSender`). */
export function isClassifyingSender(s: EmailSender): s is ClassifyingEmailSender {
  return typeof (s as Partial<ClassifyingEmailSender>).sendClassified === "function";
}

/**
 * THE ONE WAY TO SEND WHEN THE ANSWER MATTERS (I3-0014, W3-09a-FR1).
 *
 * Every consumer that has a RETRY DECISION to make calls this, and nothing
 * re-types the probe locally. `alerts.ts` used to re-type it privately, which is
 * how the digest leg at `digest.ts:1988` came to be the one call site the
 * control never reached (FAMILY-004 clause 4: a control at one of N doors).
 * The primitive carries it instead.
 *
 * A sender that cannot classify keeps its old meaning EXACTLY: `false` is a
 * hiccup, a hiccup is retried, and NOTHING is inferred from a bare `false` —
 * a caller that cannot tell has not told us anything.
 */
export function sendOnce(sender: EmailSender, msg: EmailMessage): Promise<EmailOutcome> {
  if (isClassifyingSender(sender)) return sender.sendClassified(msg);
  return sender(msg).then((ok) => (ok ? "accepted" : "transient"));
}

/** D-1004(2): the operator alert a mail CONFIGURATION refusal raises, on either rail (mailer.ts re-exports it). */
export const MAILER_CONFIG_ALERT_MARKER = "MAILER_CONFIG_ERROR";

/**
 * Map one HTTP status from the provider-agnostic rail (`httpEmailFromEnv`) to an outcome.
 *
 * D-1004(2): only a RECIPIENT-level dead signal suppresses an address, and a bare status on this rail carries none —
 * a 400/403/422 is as often OUR sender, key scope or payload as the recipient, and reading it as a dead address put a
 * whole alerts or digest wave on the suppression list (alerts.ts / digest.ts act on `permanent`). So no status here
 * is `permanent`: 2xx is accepted, everything else is retried (`transient`), and a 4xx other than 408/429 is ALSO a
 * configuration refusal (`httpConfigError`) that raises the MAILER_CONFIG_ERROR operator alert. The Postmark rail,
 * whose ErrorCode does name the recipient, is `classifyPostmarkStatus` below.
 */
export function classifyEmailStatus(status: number): EmailOutcome {
  if (status >= 200 && status < 300) return "accepted";
  return "transient";
}
/** D-1004(2): a refusal on the agnostic rail that re-sending cannot clear (a 4xx other than 408 / 429) is OURS to fix. */
export function httpConfigError(status: number): boolean {
  return status >= 400 && status < 500 && status !== 408 && status !== 429;
}

/** D-1004(2), D-1005(a): the ONLY Postmark refusal that says an ADDRESS is dead — ErrorCode 406 (inactive recipient).
 * Every other refusal (401, ErrorCode 300 and any other 422 code, 5xx) suppresses nothing. */
export const POSTMARK_DEAD_ADDRESS_CODES: ReadonlySet<number> = new Set([406]);
/** D-1004(2): Postmark refusals that are OUR configuration, not the message: HTTP 401, ErrorCode 10 (bad or missing
 * server token), 300 (invalid email request — a fault in the request we built, D-1005(a)), 400 (sender signature not
 * found), 401 (sender signature not confirmed). They raise an operator alert. */
export const POSTMARK_CONFIG_CODES: ReadonlySet<number> = new Set([10, 300, 400, 401]);
export function classifyPostmarkStatus(status: number, errorCode: number | undefined): EmailOutcome {
  if (status >= 200 && status < 300) return "accepted";
  if (status >= 400 && status < 500 && errorCode !== undefined && POSTMARK_DEAD_ADDRESS_CODES.has(errorCode)) return "permanent";
  return "transient";
}
export function postmarkConfigError(status: number, errorCode: number | undefined): boolean {
  return status === 401 || (errorCode !== undefined && POSTMARK_CONFIG_CODES.has(errorCode));
}

/** A38/F45: `EMAIL_FROM` must be a shape-valid address (RFC 5322 is not
 * re-implemented here — just enough shape to catch a typo’d/placeholder
 * value the way `mailer.ts:232` already does for a recipient), and when
 * `APP_ORIGIN` is itself shape-valid, `EMAIL_FROM`'s domain must equal
 * `APP_ORIGIN`'s host. Two independently configured senders with no assertion
 * that they share a domain removes the one heuristic an operator (or a
 * suspicious fan) can apply unaided — fail closed, same stance as every
 * other origin-shaped var in `env-config.ts`. */
const EMAIL_SHAPE_RE = /^[^\s@]+@([^\s@]+)$/;

function fromMatchesOrigin(from: string, appOrigin: string | null): boolean {
  const m = EMAIL_SHAPE_RE.exec(from);
  if (!m) return false;
  if (!appOrigin) return true; // nothing to check the domain against
  let host: string;
  try {
    host = new URL(appOrigin).host;
  } catch {
    return true; // unreachable given appOriginFromEnv's own shape check
  }
  // W2 (S-041): F-45's first form demanded EXACT host equality, which would have
  // disabled email for the ordinary correct deployment — a dedicated sending
  // subdomain (`noreply@mail.example.com` for `https://example.com`) is standard
  // practice for deliverability, not a misconfiguration, and `configHealth` would
  // have reported a healthy config as broken. The property F-45 is actually
  // about is ALIGNMENT: a fan (or an operator) looking at the sender should be
  // able to tell it belongs to this origin. So: equal, or a subdomain of it.
  // An UNRELATED domain is still refused, which is the case that mattered.
  const dom = (m[1] ?? "").toLowerCase();
  const h = host.toLowerCase();
  return dom === h || dom.endsWith(`.${h}`);
}

/** Build email config from env, or null (→ email disabled) if any secret is
 * missing/placeholder. */
export function emailConfigFromEnv(env: Record<string, unknown>): EmailConfig | null {
  // I63 C2 (I62-CC-12): each setting through settingValue (padding trimmed; blank or TODO is unset).
  const apiUrl = settingValue(env.EMAIL_API_URL);
  const apiKey = settingValue(env.EMAIL_API_KEY);
  const from = settingValue(env.EMAIL_FROM);
  if (apiUrl === null || apiKey === null || from === null) return null;
  // A38: read through the one shape rule every other origin-shaped var uses,
  // instead of casting APP_ORIGIN raw — a misshapen value (two trailing
  // slashes, a path, no scheme) now omits the manage-alerts link instead of
  // building a broken one, and one trailing slash is taken off as padding
  // (I62-RG-W1-03), matching `stripeConfigFromEnv`'s and `/ops/config`'s reading of it.
  const appOrigin = appOriginFromEnv(env);
  // F45: fail closed on a sender that cannot possibly be Dhwaja's, same as a
  // missing secret — this is the one rail-vs-rail check named in F-45; the
  // POSTMARK_FROM half is `mailer.ts`'s to add.
  if (!fromMatchesOrigin(from, appOrigin)) return null;
  return { apiUrl, apiKey, from, ...(appOrigin ? { appOrigin } : {}) };
}

}

/**
 * The generic HTTP email adapter. POSTs one message and reports success on a
 * 2xx. Never throws to the caller (the alerts loop also guards). `List-
 * Unsubscribe` is sent as a header when present (deliverability + good
 * citizenship for solicited alerts), with `List-Unsubscribe-Post` beside it for
 * RFC 8058 one-click.
 *
 * THIS IS THE ONLY RAIL THAT CAN CARRY EITHER. `mailer.ts`'s Postmark adapter
 * posts a fixed body literal with no `Headers` field and the `Mailer` interface
 * has no header channel, so the report-outcome mail it sends gets no
 * unsubscribe header at all. Widening that seam is a separate change; the gap
 * is named in SLICE21-MANIFEST.md rather than left to be discovered.
 */
// ANTI-015 finding: `push.ts:147` (`fetchWithDeadline`) and `mailer.ts:236`
// both wrap their provider call in an `AbortController`; this rail's fetch had
// none. It has one (`attempt` below).
//
// HAVING AN AbortController IS NOT THE PROPERTY, and this block used to claim it
// was. The property is that a retriggered send produces ONE delivery, and that
// needs the abort to land BEFORE the caller stops waiting. (Since 739-F both
// this abort and the caller's race give `"ambiguous"`, which `deliverPendingEmail`
// settles as sent, so the next pass no longer re-sends; the ordering below still
// matters for any caller that retries a `"transient"`.) Before 739-F the race
// read a timeout as `"transient"` and the next pass sent a second copy. If this rail's deadline is not STRICTLY below
// the caller's, the first request is still open when that second copy goes out
// and the provider may deliver BOTH — and mail cannot be recalled. Both
// constants were 10_000, a tie whose winner is the scheduler's, which is why
// this is one second under `EMAIL_SEND_TIMEOUT_MS`'s default. Asserted
// behaviourally, against a hanging fixture provider, by
// `test/w311-regression-w3-11-r6-email.test.ts`.
//
// A caller cannot end its race before this abort: alerts.ts floors
// `DeliverOptions.sendTimeoutMs` at `EMAIL_SEND_TIMEOUT_MS`
// (`emailRaceDeadlineMs`, I3-2008).
export const EMAIL_HTTP_TIMEOUT_MS = 9_000;

export function httpEmailFromEnv(env: Record<string, unknown>, fetchImpl: typeof fetch = fetch): EmailSender {
  const cfg = emailConfigFromEnv(env);
  if (!cfg) return nullEmailSender;
  const attempt = async (msg: EmailMessage): Promise<EmailOutcome> => {
    const ctl = new AbortController();
    const timer = setTimeout(() => ctl.abort(), EMAIL_HTTP_TIMEOUT_MS);
    try {
      const body: Record<string, unknown> = { from: cfg.from, to: msg.to, subject: msg.subject, text: msg.text };
      if (msg.html) body.html = msg.html;
      if (msg.listUnsubscribe) {
        const headers: Record<string, string> = { "List-Unsubscribe": msg.listUnsubscribe };
        // NESTED ON PURPOSE — see `listUnsubscribePost`. A one-click marker with
        // no URL to post to is invalid RFC 8058 and a deliverability liability,
        // so the flag alone can never emit it.
        if (msg.listUnsubscribePost) headers["List-Unsubscribe-Post"] = "List-Unsubscribe=One-Click";
        body.headers = headers;
      }
      const res = await fetchImpl(cfg.apiUrl, { redirect: "manual",
        method: "POST",
        headers: { authorization: `Bearer ${cfg.apiKey}`, "content-type": "application/json" },
        body: JSON.stringify(body),
        signal: ctl.signal,
      });
      // D-1004(2): the operator alert is the marker line (DEPLOY.md "MAILER_CONFIG_ERROR"); status only, never the recipient.
      if (httpConfigError(res.status)) logError({ cid: "unknown", hop: "http", msg: MAILER_CONFIG_ALERT_MARKER, marker: MAILER_CONFIG_ALERT_MARKER, kind: "email_http", status: res.status });
      return classifyEmailStatus(res.status);
    } catch {
      // Nothing the provider said about the RECIPIENT, so never permanent. OUR
      // abort (keyed on our own signal, whatever the rejection's shape) fired
      // after the request was handed over: `ambiguous`, settled as sent. Any
      // other throw (DNS, refused connection) is `transient`.
      return ctl.signal.aborted ? "ambiguous" : "transient";
    } finally {
      clearTimeout(timer);
    }
  };
  // The boolean seam every existing consumer already holds, with the classifying
  // entry point hung beside it. `deliverPendingEmail` probes for the second with
  // `isClassifyingSender`; everyone else keeps the first and sees no change.
  const send = (async (msg: EmailMessage): Promise<boolean> => (await attempt(msg)) === "accepted") as ClassifyingEmailSender & SuppressionKeyed;
  send.sendClassified = attempt;
  // I3-0014 e: the deployment's suppression key rides the sender every delivery path already holds, so no caller
  // needs a second env read to honour the suppression list. Absent secret -> no key -> no list (dev/test only).
  const secret = env.SESSION_SECRET;
  if (typeof secret === "string" && secret.length > 0) {
    send.suppressionKey = secret;
    // D-1003(3): the keys SESSION_SECRET rotation retired (SESSION_SECRET_PREVIOUS), consulted on LOOKUP only.
    send.previousSuppressionKeys = previousSessionSecrets(env);
  }
  return send;
}

/**
 * I3-0014 a/e (D-0993(1) census pick) — THE PER-ADDRESS SUPPRESSION LIST (`email_suppressions`, migration 0216).
 * A PERMANENT provider rejection settles its own row (alerts.ts); this list is what makes the NEXT message to the
 * same dead address cost no send. The table holds a KEYED hash, never the address (INV-061): HMAC-SHA256 of the
 * trimmed, lowercased address under a key derived from the session secret with a fixed scope label
 * (sessions.ts `scopedKeyedMac`, trust-core's one registered scoped keyed-MAC body), so a list of addresses cannot be hashed to match it.
 */
export interface SuppressionKeyed {
  suppressionKey?: string;
  /** D-1003(3): retired keys (sessions.ts previousSessionSecrets). A list row written under one still suppresses. */
  previousSuppressionKeys?: readonly string[];
}
/** The message prefix that domain-separates a suppression hash from a dispute fingerprint under the ONE derived key
 * (`DISPUTE_FINGERPRINT_SCOPE`, already a reserved label the session credential door refuses — packages/session
 * RESERVED_KEY_LABELS): a fingerprint's message starts with a provider (`google`/`apple`), never with this. */
export const EMAIL_SUPPRESSION_SCOPE = "email-suppression:v1";
/** The key a sender carries — an `EmailSender` (httpEmailFromEnv) or a template mailer (mailer.ts, JA-11). */
export function suppressionKeyOf(sender: EmailSender | object): string | null {
  const k = (sender as SuppressionKeyed).suppressionKey;
  return typeof k === "string" && k.length > 0 ? k : null;
}
/** D-1003(3): the lookup's retired half — the sender's retired keys and, aligned index for index with the hashes passed
 * to `suppressedAddressHashes`, each hash's address. */
export interface RetiredSuppressionLookup {
  keys: readonly string[];
  addresses: readonly string[];
}
export function retiredSuppressionLookup(sender: EmailSender | object, addresses: readonly string[]): RetiredSuppressionLookup {
  const k = (sender as SuppressionKeyed).previousSuppressionKeys;
  return { keys: Array.isArray(k) ? k.filter((x): x is string => typeof x === "string" && x.length > 0) : [], addresses };

export async function emailAddressHash(secret: string, address: string): Promise<string> {
  return scopedKeyedMac(secret, DISPUTE_FINGERPRINT_SCOPE, `${EMAIL_SUPPRESSION_SCOPE}\n${address.trim().toLowerCase()}`);
}
interface SuppressionDb {
  prepare(sql: string): { bind(...v: unknown[]): { all<T>(): Promise<{ results: T[] }>; run(): Promise<unknown> } };
}
/** The subset of `hashes` on the list. ONE statement whatever the count. D-1003(3): with `retired`, a hash also counts as
 * listed when its address's hash under any retired key is (a rotation does not un-suppress a dead address); the answer
 * is still a subset of `hashes`. Writes stay under the current key only (`recordEmailSuppressions`). */
export async function suppressedAddressHashes(db: SuppressionDb, hashes: readonly string[], retired?: RetiredSuppressionLookup): Promise<Set<string>> {
  if (hashes.length === 0) return new Set();
  const standsFor = new Map<string, string>(hashes.map((h) => [h, h]));
  for (const key of retired?.keys ?? []) {
    for (const [i, h] of hashes.entries()) {
      const address = retired?.addresses[i];
      if (typeof address !== "string") continue;
      const old = await emailAddressHash(key, address);
      if (!standsFor.has(old)) standsFor.set(old, h);
    }
  }
  const { results } = await db
    .prepare("SELECT address_hash FROM email_suppressions WHERE address_hash IN (SELECT value FROM json_each(?))")
    .bind(JSON.stringify([...standsFor.keys()]))
    .all<{ address_hash: string }>();
  return new Set(results.map((r) => standsFor.get(r.address_hash)).filter((h): h is string => typeof h === "string"));
}
/** I24-Q5: one address's list hashes — under the current key first, then under each retired key (the hashes
 * `suppressedAddressHashes` would match for it), de-duplicated. */
export async function emailSuppressionHashesFor(secret: string, retiredKeys: readonly string[], address: string): Promise<string[]> {
  const out = [await emailAddressHash(secret, address)];
  for (const k of retiredKeys) {
    const h = await emailAddressHash(k, address);
    if (!out.includes(h)) out.push(h);
  }
  return out;
}
/** The audit action an operator's lift writes (item_id NULL, detail `{rows, hash}` — the current-key hash, never the address). */
export const EMAIL_SUPPRESSION_LIFT_ACTION = "email_suppression:lift";
/**
 * I24-Q5 (D-1030 (a); I30-PREMIUM) — an operator takes ONE address off the dead-email list. `hashes` is
 * `emailSuppressionHashesFor`'s list (the current-key hash first). ONE batch: the audit row, written only while one of
 * the hashes is listed (its WHERE reads them ahead of the DELETE), then the DELETE of exactly those rows. Returns how
 * many were listed; 0 deletes nothing and audits nothing (the caller answers not found).
 */
export async function liftEmailSuppression(db: DbLike, hashes: readonly string[], actor: string, nowIso: string): Promise<number> {
  if (hashes.length === 0) return 0;
  const list = JSON.stringify([...new Set(hashes)]);
  const listed = "SELECT 1 FROM email_suppressions WHERE address_hash IN (SELECT value FROM json_each(?))";
  const found = (await db.prepare(`SELECT COUNT(*) AS n FROM (${listed})`).bind(list).first<{ n: number }>())?.n ?? 0;
  if (found === 0) return 0;
  await db.batch([
    db
      .prepare(
        `INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, ?, json_object('rows', (SELECT COUNT(*) FROM (${listed})), 'hash', ?), ?, NULL, 'operator' WHERE EXISTS (${listed})`,
      )
      .bind(actor, EMAIL_SUPPRESSION_LIFT_ACTION, list, hashes[0], nowIso, list),
    db.prepare("DELETE FROM email_suppressions WHERE address_hash IN (SELECT value FROM json_each(?))").bind(list),
  ]);
  return found;
}
/** Upsert each hash as permanently rejected at `nowIso` (first_seen kept, last_seen moved). ONE statement. */
export async function recordEmailSuppressions(db: SuppressionDb, hashes: readonly string[], nowIso: string): Promise<void> {
  if (hashes.length === 0) return;
  await db
    .prepare(
      `INSERT INTO email_suppressions (address_hash, reason, first_seen, last_seen)
         SELECT value, 'permanent', ?, ? FROM json_each(?) WHERE true
       ON CONFLICT(address_hash) DO UPDATE SET last_seen = excluded.last_seen`,
    )
    .bind(nowIso, nowIso, JSON.stringify([...new Set(hashes)]))
    .run();
}

 * never sold in an app — so with `creator_subscriptions` gone, `billing_entitlements`
 * is the single source and there is nothing left to reconcile. That is a real
 * simplification, not a hidden second rail: the shape of the shadowing bug needs
 * two sources, and the creator side no longer has two.
 */

export type SubjectKind = "fan" | "creator";

export interface StoredEntitlement {
  subject_id: string;
  subject_kind: SubjectKind;
  tier: "premium" | "pro";
  paid_through: string;
  source: string;
  stripe_customer_id: string | null;
  stripe_subscription_id: string;
  /** I21-Q61 (0224): set when the grant ended on a refund or a dispute — no grace on read. */
  revoked_at: string | null;
}

/** The subject's MOST GENEROUS subscription row — the one that decides what
 * they can do. `subject_kind` is matched in SQL so a creator id can never read
 * a fan entitlement even if the id spaces somehow collided. */
const SELECT =
  "SELECT subject_id, subject_kind, tier, paid_through, source, stripe_customer_id, stripe_subscription_id, revoked_at " +
  "FROM billing_entitlements WHERE subject_id = ? AND subject_kind = ? ORDER BY paid_through DESC LIMIT 1";

/** The effective raw row for a subject, or null. */
export async function readEntitlement(db: DbLike, subjectId: string, kind: SubjectKind): Promise<StoredEntitlement | null> {
  if (!subjectId) return null;
  return db.prepare(SELECT).bind(subjectId, kind).first<StoredEntitlement>();
}

/** Row → the pure module's shape. An unparseable paid_through is treated as
 * ABSENT (free), never as valid-forever. */
function pureRow(row: StoredEntitlement | null): EntitlementRow | null {
  if (!row) return null;
  const ms = Date.parse(row.paid_through);
  if (!Number.isFinite(ms)) return null;
  return { tier: row.tier, paidThroughMs: ms, revoked: row.revoked_at != null };
}

/** The Stripe rail's contribution alone (grace included), or "free". */
function stripeGrant<T extends "premium" | "pro">(row: StoredEntitlement | null, want: T, nowMs: number): T | "free" {
  const pure = pureRow(row);
  return pure && pure.tier === want ? effectiveTier({ tier: want, paidThroughMs: pure.paidThroughMs, revoked: pure.revoked } as EntitlementRow & { tier: T }, nowMs) : "free";
}

/**
 * A fan's effective tier: the MOST GENEROUS of the Stripe (web) grant and the
 * native STORE subscription. A lapsed Stripe row must not shadow a live Apple
 * or Google purchase — see the header.
 */
export async function effectiveFanTier(db: DbLike, accountId: string, nowMs: number): Promise<"free" | "premium"> {
  const fromStripe = stripeGrant(await readEntitlement(db, accountId, "fan"), "premium", nowMs);
  if (fromStripe === "premium") return "premium";
  if ((await promoPremiumUntil(db, accountId)) > nowMs) return "premium"; // I16-S4-I3-0758: a week bought with promotional credits
  return currentTier(db, accountId, nowMs);
}

/**
 * I16-S4-I3-0758 (I30-PREMIUM; migration 0257): the instant (epoch ms) a promotional Premium week runs to, 0 when none
 * was bought. A third most-generous rail of `effectiveFanTier`, written only by rewards.ts buyPromoPremiumWeek. It is
 * no subscription: `hasLiveSubscription` does not read it, so a fan on a promo week can still subscribe. A database
 * without 0257 has no such table and so no week: exactly that error reads as 0 (the dual_control.ts
 * withRevocationClause rule); any other fault throws.
 */
export async function promoPremiumUntil(db: DbLike, accountId: string): Promise<number> {
  try {
    const row = await db.prepare("SELECT until_ms FROM fan_promo_premium WHERE account_id = ?").bind(accountId).first<{ until_ms: number }>();
    return row?.until_ms ?? 0;
  } catch (e) {
    if (!/no such table: fan_promo_premium/i.test(e instanceof Error ? e.message : String(e))) throw e;
    return 0;
  }
}

/**
 * A creator's effective tier — the single read behind every Pro cap. ONE rail
 * since item 14 (Creator Pro is sold on Stripe and nowhere else), so this is
 * the Stripe grant with its grace and nothing else. If a second creator rail is
 * ever added — a creator IAP, say — it must come back through THIS function as
 * a most-generous fold, never as a caller that learns a tier another way.
 */
export async function effectiveCreatorTier(db: DbLike, passportId: string, nowMs: number): Promise<"free" | "pro"> {
  return stripeGrant(await readEntitlement(db, passportId, "creator"), "pro", nowMs);
}

/**
 * Does this subject have a LIVE subscription right now — one that is still
 * inside its paid period on either rail?
 *
 * This is the double-charge guard, and it is deliberately NOT `tier !== free`.
 * The read-time tier includes the 7-day grace, so a subject who cancelled was
 * "not free" for another week and the checkout route 409'd them; if the
 * subscription had been deleted outright there was nothing to reactivate in
 * the Portal either, so they simply could not give us money. 409 must mean
 * "you already have one", not "you are inside grace".
 */
export async function hasLiveSubscription(db: DbLike, subjectId: string, kind: SubjectKind, nowMs: number): Promise<boolean> {
  const pure = pureRow(await readEntitlement(db, subjectId, kind));
  if (pure && nowMs < pure.paidThroughMs) return true; // paid period, grace excluded
  // The store rail is fan-only. A creator has no second source to double-charge
  // against, so there is nothing further to ask.
  return kind === "fan" ? (await currentTier(db, subjectId, nowMs)) !== "free" : false;
}

/**
 * BF-05 — THE ONE COMBINED ANSWER `/account/billing` OWES A PHONE.
 *
 * `billing.ts billingStatus` folds the NATIVE store rails and nothing else, and
 * that is the route the iOS and Android apps read
 * (`StoreManager.swift refresh()` falls back to `api.billing()["tier"]`;
 * `ApiClient.kt billing()`). iOS then hides its subscribe control on exactly
 * that flag. So a fan PAYING ON STRIPE read `free` on their phone and was
 * offered a second recurring subscription for the same account — while the web
 * checkout route deliberately refuses that same fan 409 `already_subscribed`.
 * Two Dhwaja surfaces, opposite answers, about one fan's money.
 *
 * WHY THE FIX IS HERE AND NOT IN THE APPS. A client fix ships only to phones
 * that update, and the native purchase paths (`StoreManager.swift`,
 * `BillingManager.kt`) check no combined entitlement at all — they act on what
 * this route says. Making the route truthful fixes every already-installed
 * build.
 *
 * THREE FACTS, FROM ONE RESOLVER, which is the whole point of the row:
 *  - `tier` / `status` / `current_period_end` / `rail` all describe the SAME
 *    grant — the one actually holding the fan up — never a period borrowed from
 *    a revoked row while the tier came from a live one;
 *  - `provider` NAMES that grant's rail, so a client never has to infer it;
 *  - `can_purchase` is `hasLiveSubscription` — literally the same predicate the
 *    web 409 uses, grace deliberately excluded — so the two doors cannot drift.
 *
 * ADDITIVE ON THE WIRE. Every field `BillingStatus` carried is still there and
 * still means what it meant, so an un-updated client keeps working; the new
 * fields are what an updated one needs to stop selling a duplicate and to point
 * at the right management surface.
 */
export interface CombinedBillingStatus extends BillingStatus {
  /** The rail supplying the reported entitlement, or `none` when free. */
  provider: "stripe" | StoreRail | "none";
  /** May a NEW subscription be started for this account? Exactly the web
   * checkout's own rule (`hasLiveSubscription`), so a native purchase screen
   * and the web 409 answer the same question the same way. */
  can_purchase: boolean;
  /** True when the live grant is the STRIPE one — i.e. cancelling or changing
   * the card happens in the Dhwaja web portal, NOT in the App Store or Play
   * subscription settings the native copy otherwise points at. */
  manage_on_web: boolean;
  /** BF-05: is the grant REPORTED ABOVE inside its grace window? Derived from
   * that same grant, so a canceled row on another rail can no longer make a
   * live one look lapsed (or hide that a live one is about to lapse). */
  in_grace: boolean;
}

export async function combinedBillingStatus(db: DbLike, accountId: string, nowMs: number): Promise<CombinedBillingStatus> {
  const store = await billingStatus(db, accountId, nowMs);
  const row = await readEntitlement(db, accountId, "fan");
  const pure = pureRow(row);
  const stripeLive = effectiveTier(pure, nowMs) === "premium";
  const storeLive = store.tier === "premium";
  const stripeEnd = row?.paid_through ?? null;

  // MOST GENEROUS WINS, the same rule `effectiveFanTier` applies and the same
  // one `billingStatus` applies between the two stores — now applied one level
  // up, between Stripe and the store fold. A tie goes to the grant reaching
  // furthest forward, so the date a fan is shown is the date their access
  // actually ends.
  const stripeWins = stripeLive && (!storeLive || (stripeEnd ?? "") > (store.current_period_end ?? ""));

  let provider: CombinedBillingStatus["provider"];
  let status = store.status;
  let periodEnd = store.current_period_end;
  if (stripeWins) {
    provider = "stripe";
    status = "active";
    periodEnd = stripeEnd;
  } else if (storeLive || store.status !== "none") {
    // A live store grant, or — when nothing is live — the store lapse the fan
    // needs to be told about. `billingStatus` already picked which.
    provider = store.rail ?? "unknown";
  } else if (row) {
    // No store row at all and a Stripe row that is past its grace: report the
    // Stripe lapse rather than a bare `none`, which would read as "you never
    // subscribed" to someone who did.
    provider = "stripe";
    status = "expired";
    periodEnd = stripeEnd;
  } else {
    provider = "none";
  }

  return {
    tier: stripeLive || storeLive ? "premium" : "free",
    status,
    current_period_end: periodEnd,
    has_customer: store.has_customer || !!row?.stripe_customer_id,
    rail: store.rail,
    provider,
    can_purchase: !(await hasLiveSubscription(db, accountId, "fan", nowMs)),
    manage_on_web: stripeWins,
    // The grace of the grant this answer is ABOUT. For Stripe that is the pure
    // module's own `inGrace`; for a store rail it is the same window `tierFor`
    // applies — past the period end, still inside the grace — asked of the row
    // `billingStatus` selected, never of some other rail's row.
    in_grace: stripeWins
      ? inGrace(pure, nowMs)
      : status === "active" && !!periodEnd && Number.isFinite(Date.parse(periodEnd)) && nowMs >= Date.parse(periodEnd) && nowMs < Date.parse(periodEnd) + STORE_GRACE_MS,
  };
}

export interface EntitlementView {
  subject_kind: SubjectKind;
  /** What the subject is entitled to RIGHT NOW ("free" when lapsed past grace). */
  tier: "free" | "premium" | "pro";
  /** The purchased tier, even when currently lapsed (drives the lapse copy). */
  purchased_tier: "premium" | "pro" | null;
  paid_through: string | null;
  /** Past paid_through but inside the 7-day grace — the QUIET banner, never red. */
  in_grace: boolean;
  /** M11: whole days LEFT in the grace window (counts down; 0 once it lapses). */
  grace_days: number;
  /** M11: the exact instant access ends if not renewed (ISO), or null. */
  grace_until: string | null;
  has_customer: boolean;
  source: string | null;
  /** RJb-T2-AS-013 / #609q: the subscriber asked to cancel — the plan ENDS (does not renew) at this ISO instant;
   * null when it renews, or when the tier is not supplied by that Stripe subscription. */
  cancel_at: string | null;
}

/** The subject's own billing state, for the account UI (mock 36) and for the
 * mobile/extension read-only entitlement check. Never exposes Stripe ids. */
export async function entitlementView(db: DbLike, subjectId: string, kind: SubjectKind, nowMs: number): Promise<EntitlementView> {
  const row = await readEntitlement(db, subjectId, kind);
  const pure = pureRow(row);
  // EVERY rail this subject kind has — the view must agree with
  // effectiveFanTier / effectiveCreatorTier or the account screen lies. Fans
  // have a store rail; creators (item 14) have only Stripe.
  const store = kind === "fan" ? await currentTier(db, subjectId, nowMs) : "free";
  const fromStripe = effectiveTier(pure, nowMs);
  // AS-047: which rail actually supplies the tier the caller reads below. A
  // lapsed Stripe row must not lend its (stale) paid_through / has_customer to
  // a tier the STORE rail is the one supplying — that dressed a live Apple/
  // Google subscription as a Stripe one, offered a portal to a dead
  // subscription, and hid the store-management fine print for exactly the
  // mixed-history fan who needs it (upgrade.ts keys its store branch off
  // `!has_customer`). `viaStore` is only possible for fans (creators have one
  // rail, item 14).
  const viaStore = fromStripe === "free" && store !== "free";
  // RJb-T2-AS-013 / #609q: read only when that Stripe subscription is what supplies the tier.
  const cancel =
    fromStripe !== "free" && row?.stripe_subscription_id
      ? await db.prepare("SELECT cancel_at FROM stripe_subscription_cancel WHERE subscription_id = ? AND cancel_at IS NOT NULL").bind(row.stripe_subscription_id).first<{ cancel_at: string }>()
      : null;
  return {
    subject_kind: kind,
    tier: fromStripe !== "free" ? fromStripe : store,
    purchased_tier: row?.tier ?? null,
    // The store rail keeps no paid_through of its own here (see billing.ts's
    // currentTier — it folds every rail into a bare tier); showing NO date is
    // correct per the finder's own acceptance criteria #2, not a regression.
    paid_through: viaStore ? null : (row?.paid_through ?? null),
    // Only show the quiet lapse banner when the Stripe grace is what is
    // actually holding the subject up; a live STORE subscription is not a lapse
    // and must not be dressed as one.
    in_grace: inGrace(pure, nowMs) && store === "free",
    // M11: COUNT DOWN. This was `round(GRACE_MS/day)` — a constant 7 that never
    // moved, so a fan three days from losing access saw the same "7 days" they
    // saw on day one. Report the days LEFT until the grace cutoff (clamped ≥ 0),
    // and the exact cutoff date, so the banner can say "3 days left (until …)".
    grace_days: pure ? Math.max(0, Math.ceil((pure.paidThroughMs + graceMsFor(pure) - nowMs) / 86_400_000)) : 0,
    grace_until: pure ? new Date(pure.paidThroughMs + graceMsFor(pure)).toISOString() : null,
    // A live store grant is never dressed as a Stripe customer — that is what
    // sent a store subscriber to a Stripe portal for a subscription that isn't
    // there (AS-047).
    has_customer: viaStore ? false : !!row?.stripe_customer_id,
    source: viaStore ? "store" : (row?.source ?? null),
    cancel_at: cancel?.cancel_at ?? null,
  };
}

/** The Stripe customer id backing the portal link, or null (→ 409). Any of the
 * subject's subscriptions can supply it — they share one Stripe customer, and
 * the newest grant may be the one that arrived without it. */
export async function customerFor(db: DbLike, subjectId: string, kind: SubjectKind): Promise<string | null> {
  if (!subjectId) return null;
  const row = await db
    .prepare(
      "SELECT stripe_customer_id FROM billing_entitlements WHERE subject_id = ? AND subject_kind = ? AND stripe_customer_id IS NOT NULL ORDER BY paid_through DESC LIMIT 1",
    )
    .bind(subjectId, kind)
    .first<{ stripe_customer_id: string }>();
  return row?.stripe_customer_id ?? null;
}

/**
 * ORDERING GUARD, in two parts. `last_event_at` is Stripe's `created`, which
 * has ONE-SECOND granularity — and Stripe routinely emits `.updated` and
 * `.deleted` inside the same second, so a plain `>=` resolved a causally
 * ordered pair by ARRIVAL order instead. `last_event_rank` is the secondary
 * key: created(1) < updated(2) < deleted(3), so within one second the terminal
 * event wins no matter which arrives first.
 *
 * Residual, named rather than papered over: two `.updated` events sharing a
 * second still resolve by arrival order. Stripe exposes no sequence number and
 * the payloads carry nothing else monotonic, so there is no tie-break left to
 * take; the exposure is one second of a status flip, which the next event or
 * the read-time grace absorbs.
 */
/** RJb-T2-AS-013 / #609q (0198): the subscription's cancel state, ordered by the same event rule as the
 * entitlement (`NEWER`), so a stale `updated` never re-arms or clears a newer answer. */
const CANCEL_STATE =
  "INSERT INTO stripe_subscription_cancel (subscription_id, cancel_at, last_event_at, last_event_rank, updated_at) VALUES (?, ?, ?, ?, ?) " +
  "ON CONFLICT(subscription_id) DO UPDATE SET cancel_at = excluded.cancel_at, last_event_at = excluded.last_event_at, " +
  "last_event_rank = excluded.last_event_rank, updated_at = excluded.updated_at WHERE ";

const NEWER = (t: string) =>
  `(excluded.last_event_at > ${t}.last_event_at OR (excluded.last_event_at = ${t}.last_event_at AND excluded.last_event_rank >= ${t}.last_event_rank))`;

/** I21-Q61 (D-1030): the event types whose `lapse`/`revoke` outcome means money was taken back (stripe.ts
 * stripeOutcome: a full refund, a dispute that withdrew funds, a lost dispute). A `lapse` from any other type is a
 * subscription-state lapse (a card failure among them) and keeps the read-time grace. */
const REVERSAL_EVENTS: ReadonlySet<string> = new Set(["charge.refunded", "charge.dispute.created", "charge.dispute.closed"]);

/** I21-Q61 / D-1037(6): at instant `at` (an ISO expression, `?` or a column; it appears ONCE) the grant is unrevoked and
 * its read-time grace (entitlements.ts GRACE_MS) still runs: paid_through + GRACE is past `at` — so a grant lapsed by a
 * card failure or a cancel, and not yet free. (A grant paid past `at` satisfies it too; callers that need the lapsed
 * ones only ask it where no grant is paid past `at`.) */
const GRACE_DAYS = GRACE_MS / 86_400_000;
const IN_GRACE_AT = (t: string, at: string): string =>
  `(${t}revoked_at IS NULL AND ${t}paid_through > strftime('%Y-%m-%dT%H:%M:%fZ', ${at}, '-${GRACE_DAYS} days'))`;
/** The grant still GIVES the product at `at` (`at` appears TWICE): paid past it, or IN_GRACE_AT. The reversal
 * statements that name ONE row (LAPSE, REVOKE_BY_SUBSCRIPTION, APPLY_UNMATCHED_REVERSALS) match on THIS and write
 * `paid_through = MIN(paid_through, at)`, so a refund or lost dispute after a card-failure lapse ends the grace at the
 * reversal and still never lengthens a grant. REVOKE_BY_CUSTOMER uses the two halves separately (see there).
 * `t` is the column prefix (`""`, `"billing_entitlements."`). */
const GRANTS_AT = (t: string, at: string): string => `(${t}paid_through > ${at} OR ${IN_GRACE_AT(t, at)})`;

/** Event-type precedence inside one `created` second. Unknown/invoice → 0. */
const EVENT_RANK: Record<string, number> = {
  "customer.subscription.created": 1,
  "customer.subscription.updated": 2,
  "customer.subscription.deleted": 3,
};
function rankOf(eventType: string): number {
  return EVENT_RANK[eventType] ?? 0;
}

/** The rank a row carries once `customer.subscription.deleted` has written it.
 * Deletion is TERMINAL at Stripe — a resubscribe mints a new subscription id —
 * so a row stamped with it can never be extended again. See EXTEND. */
const DELETED_RANK = EVENT_RANK["customer.subscription.deleted"]!;

/** Create or RAISE the grant for ONE subscription. Only a paying status ever
 * reaches here, so a row's existence means money (or a trial) was granted. */
const UPSERT =
  "INSERT INTO billing_entitlements (subject_id, subject_kind, tier, paid_through, source, stripe_customer_id, stripe_subscription_id, last_event_at, last_event_rank, updated_at) " +
  "SELECT ?, ?, ?, ?, ?, ?, ?, ?, ?, ? " +
  // H3: never RESURRECT a deleted subject. `deleteAccount` erases the fan's
  // billing_entitlements rows, but a later renewal webhook carries
  // `metadata.dhwaja_subject` and this UPSERT would re-create the row (tier
  // pro again) for an account that no longer exists — a phantom paid
  // entitlement bound to a Stripe customer for an erased person. Only insert
  // when the subject still exists (a fan account, or a creator passport). An
  // existing row still updates via ON CONFLICT; this only blocks re-creation.
  "WHERE (? = 'fan' AND EXISTS (SELECT 1 FROM accounts WHERE account_id = ?)) " +
  "   OR (? = 'creator' AND EXISTS (SELECT 1 FROM passports WHERE passport_id = ?)) " +
  // Per SUBSCRIPTION (the composite PK in `0008_billing.sql`): a second
  // subscription is a second row, so it can
  // neither overwrite nor shorten the first.
  "ON CONFLICT(subject_id, stripe_subscription_id) DO UPDATE SET " +
  "tier = excluded.tier, paid_through = excluded.paid_through, source = excluded.source, " +
  // A later event may omit the customer; COALESCE keeps the known id so the
  // portal link never breaks.
  "stripe_customer_id = COALESCE(excluded.stripe_customer_id, billing_entitlements.stripe_customer_id), " +
  // I21-Q61: a grant paid again is no longer a revoked one, so it regains the read-time grace.
  "revoked_at = NULL, " +
  "last_event_at = excluded.last_event_at, last_event_rank = excluded.last_event_rank, updated_at = excluded.updated_at " +
  `WHERE ${NEWER("billing_entitlements")}`;

/**
 * END one subscription's grant. UPDATE-ONLY — a non-paying status can never
 * bring a row into existence, which is what stops a never-paid subscription
 * from collecting 7 days of grace (see stripeOutcome). Scoped to the SUBJECT'S
 * OWN subscription, so cancelling the monthly cannot revoke the annual. Only
 * ever shortens (`paid_through > ?`).
 */
const LAPSE =
  "UPDATE billing_entitlements SET paid_through = MIN(paid_through, ?), " +
  // I21-Q61: non-NULL only for a lapse that IS a reversal (stripe.ts chargeReversal's expanded-invoice path, `reversal`).
  "revoked_at = COALESCE(?, revoked_at), " +
  "stripe_customer_id = COALESCE(?, stripe_customer_id), " +
  "last_event_at = ?, last_event_rank = ?, updated_at = ? " +
  `WHERE subject_id = ? AND stripe_subscription_id = ? AND ${GRANTS_AT("", "?")} ` +
  "AND (last_event_at < ? OR (last_event_at = ? AND last_event_rank <= ?))";

/**
 * BF-07 — WHAT A `customer.subscription.deleted` MUST LEAVE BEHIND, and why
 * `LAPSE` could not leave it.
 *
 * `EXTEND` refuses a row whose `last_event_rank` is already `DELETED_RANK`.
 * That is the ENTIRE defence against a late `invoice.payment_succeeded` — the
 * last open invoice settling a day or two after a cancel is the ordinary way
 * this arrives — resurrecting a subscription that is gone at Stripe. But the
 * only statement that stamped that rank was `LAPSE`, and `LAPSE` is
 * SHORTEN-ONLY (`paid_through > ?`). So whenever deletion had no paid time left
 * to take away it matched ZERO ROWS, wrote no rank, and was still marked
 * processed:
 *
 *   - the grant had already expired, or an earlier failure / refund /
 *     chargeback had already lapsed it to at or before the cancel instant;
 *   - the deletion was delivered BEFORE the creation it cancels — Stripe does
 *     not order deliveries, and this module already reasons about exactly that
 *     interleaving for the lapse read.
 *
 * The invoice then extended application access for a cancelled subscription:
 * tier went free -> premium again, from an invoice, with nothing live anywhere.
 *
 * TERMINAL SEPARATES THE TWO JOBS `LAPSE` HAD CONFLATED. Recording that this
 * subscription is over is not the same act as shortening paid time, and it must
 * happen whether or not there is any to shorten. `MIN(paid_through, ?)` keeps
 * the shorten-only DIRECTION — a deletion can still never extend anyone — while
 * the rank is written unconditionally. Every ordering guard `LAPSE` carried is
 * carried here unchanged, so a redelivered old cancel still cannot end a grant
 * bought after it, and a resubscribe (a NEW subscription id, hence a new row)
 * is untouched.
 */
const TERMINAL =
  "UPDATE billing_entitlements SET paid_through = MIN(paid_through, ?), " +
  "stripe_customer_id = COALESCE(?, stripe_customer_id), " +
  "last_event_at = ?, last_event_rank = ?, updated_at = ? " +
  "WHERE subject_id = ? AND stripe_subscription_id = ? " +
  "AND (last_event_at < ? OR (last_event_at = ? AND last_event_rank <= ?))";

/**
 * BF-07, the deletion-before-create half. The row the cancel needed to stamp
 * does not exist yet, so there is nothing to UPDATE — and `LAPSE` is
 * deliberately update-only, which is right and must stay: letting a non-paying
 * status CREATE a row is how a never-paid subscription would collect seven days
 * of grace (see `stripeOutcome`).
 *
 * A TOMBSTONE IS NOT A GRANT. It is inserted at `TERMINAL_PAID_THROUGH` — the
 * epoch — so every read that could turn it into access answers no:
 *   - `effectiveTier` / `inGrace` add the 7-day grace to `paid_through`, and
 *     1970 + 7 days is still 1970: no tier, no grace, on any clock;
 *   - `readEntitlement` orders by `paid_through DESC`, so a tombstone sorts
 *     BELOW every real grant and can never shadow one;
 *   - `hasLiveSubscription` is `nowMs < paidThroughMs`: false, so a cancelled
 *     subject may re-purchase IMMEDIATELY, which is finding #12's decision and
 *     is preserved exactly.
 * What it does carry is `last_event_rank = 3` and the cancel's `last_event_at`,
 * which is precisely what `EXTEND` and the `UPSERT`'s `NEWER` guard read. The
 * late `created` is stamped EARLIER at Stripe than the cancel, so `NEWER`
 * refuses it; the late invoice is refused by the rank.
 *
 * Guarded on subject existence exactly like `UPSERT` (H3): a tombstone for an
 * ERASED account is still a row bound to a person who is gone, and erasure
 * deletes these rows.
 */
const TERMINAL_PAID_THROUGH = "1970-01-01T00:00:00.000Z";

const TOMBSTONE =
  "INSERT INTO billing_entitlements (subject_id, subject_kind, tier, paid_through, source, stripe_customer_id, stripe_subscription_id, last_event_at, last_event_rank, updated_at) " +
  "SELECT ?, ?, ?, ?, 'stripe', ?, ?, ?, ?, ? " +
  "WHERE ((? = 'fan' AND EXISTS (SELECT 1 FROM accounts WHERE account_id = ?)) " +
  "    OR (? = 'creator' AND EXISTS (SELECT 1 FROM passports WHERE passport_id = ?))) " +
  "AND NOT EXISTS (SELECT 1 FROM billing_entitlements b WHERE b.subject_id = ? AND b.stripe_subscription_id = ?)";

/**
 * END the grant a REVERSED CHARGE was paying for, scoped by CUSTOMER — the
 * FALLBACK reversal, taken when the payment is not one we recorded.
 *
 * WHY A SECOND REVOCATION EXISTS AT ALL. `LAPSE` names (subject, subscription),
 * and a Charge or a Dispute names neither: Stripe does not expand
 * `charge.invoice` (or `dispute.charge`) in webhook deliveries, and the
 * subscription/subject linkage lives nowhere else on a Charge. So the precise
 * path above was correct, fail-closed, and never fired on a real delivery —
 * money was taken back and the raised tier caps kept running for it. The one
 * identity a bare charge DOES carry is `customer`, and these rows already carry
 * it (that is how EXTEND scopes itself), so the customer is what a reversal is
 * scoped to when nothing better is on the wire.
 *
 * WHAT SUPERSEDED IT, AND WHY IT STAYS. `billing_payments` (0039) now records
 * the charge id and the PaymentIntent id of each paid invoice against the grant
 * they bought, so most reversals resolve to REVOKE_BY_SUBSCRIPTION instead —
 * narrower, unambiguous, and the only thing that could ever close the DISPUTE
 * case (a Dispute carries no customer at all, so this statement could never
 * fire on a chargeback). This one is still reached by every reversal with no
 * recorded linkage: a subscription paid for before 0039, an invoice whose
 * payment ids we could not read, and any refund whose charge we never saw.
 * Deleting it would re-open the refund hole for the entire installed base.
 *
 * WHAT IT CAN DO, exhaustively: shorten ONE existing row to the reversal
 * instant. UPDATE-only (it can never bring a grant into existence, so it can
 * never resurrect an erased subject's row or hand a never-paid subscription the
 * 7-day grace), `paid_through > ?` (it can only ever shorten — a reversal can
 * never extend anyone, which is also what makes "a won dispute restores
 * nothing" structural rather than a promise), and the same `last_event_at` /
 * `last_event_rank` ordering guard every other write carries, so a redelivered
 * old refund cannot end a grant bought after it.
 *
 * `last_event_rank` is deliberately NOT written. A charge event ranks 0 — it is
 * not part of the subscription state machine's precedence ladder — and writing
 * that over a row stamped 3 by `customer.subscription.deleted` would erase the
 * TERMINAL mark EXTEND reads, handing a late invoice the resurrection this
 * module already closed. Leaving the column alone can only ever keep a guard
 * stricter, never looser, because 0 is the floor.
 *
 * THE BLAST RADIUS, which is the whole design question. One Stripe customer may
 * hold MORE THAN ONE subscription — the header says so and finding #2 tests it
 * (an annual beside a monthly) — and a refunded charge cannot say which of them
 * the money bought. Revoking a customer's unrelated, still-paid subscription
 * because one charge was reversed would be a worse defect than the one this
 * closes: it takes the product from someone who is paying for it. So the
 * statement acts ONLY when the customer has exactly ONE revocable grant, and
 * otherwise changes nothing.
 *
 * That check is IN the statement, as a correlated count over the same
 * conditions the UPDATE itself matches on, rather than a read the caller takes
 * first. A check-then-act would have a window in which a second subscription
 * lands between the count and the write — and the write is customer-scoped, so
 * that window is precisely the one that revokes two grants instead of one.
 * Inside one statement there is no window, and because the counted set is
 * exactly the matched set, "the count is 1" and "at most one row can change"
 * are the same fact.
 *
 * RESIDUAL, named: a customer whose ONE live grant is not the grant the
 * reversed charge paid for — they refund an old, already-ended subscription
 * while a newer one runs — has the newer one ended. Nothing on a Charge
 * distinguishes the two (no price, no period, no subscription), the row keeps
 * no purchase-time to compare against, and the alternative (refuse whenever the
 * customer has ever held a second subscription) would drop the revocation for
 * every cancel-then-resubscribe, which is the ordinary shape of the account a
 * fraudster leaves behind. It ends at the reversal with no grace (revoked_at,
 * I21-Q61), on the customer's own grant, and the live subscription at Stripe re-establishes
 * paid_through through the priced path at its next subscription event or
 * renewal invoice — the same remedy the won-dispute decision relies on.
 */
// D-1037(6): the revocable grant is the customer's ONE PAID grant when it has exactly one; only when it has NONE is it
// the ONE grant still in its grace (IN_GRACE_AT) — a refund after a card-failure lapse. A paid grant beside an ended one
// (cancel-then-resubscribe) is still the paid one; two of either kind stands the statement down, as before.
const REVOKE_BY_CUSTOMER =
  "UPDATE billing_entitlements SET paid_through = MIN(paid_through, ?), revoked_at = ?, last_event_at = ?, updated_at = ? " +
  "WHERE stripe_customer_id = ? " +
  "AND (last_event_at < ? OR (last_event_at = ? AND last_event_rank <= ?)) " +
  "AND (CASE (SELECT COUNT(*) FROM billing_entitlements o WHERE o.stripe_customer_id = ? AND o.paid_through > ?) " +
  "WHEN 1 THEN paid_through > ? " +
  `WHEN 0 THEN ${IN_GRACE_AT("", "?")} AND (SELECT COUNT(*) FROM billing_entitlements g WHERE g.stripe_customer_id = ? AND ${IN_GRACE_AT("g.", "?")}) = 1 ` +
  "ELSE 0 END)";

/**
 * END exactly the grant a reversed payment bought — the narrow revocation, and
 * the one taken whenever the payment is one we recorded.
 *
 * WHY IT EXISTS. The customer-scoped statement above closed refunds and could
 * not close chargebacks: Stripe's Dispute carries no customer at all, so there
 * was nothing to bind it to. `billing_payments` (0039) records the charge id
 * and the PaymentIntent id of every paid subscription invoice against the
 * (subject, subscription) they paid for, so a bare `dispute.charge` resolves to
 * one grant here rather than to nothing. Scoped by the full primary key of
 * `billing_entitlements` — plus `subject_kind`, matched in SQL as every other
 * read of this table does — so it can reach exactly one row and no other, which
 * is what makes it strictly better than the customer scope: it is immune to the
 * ambiguity that makes REVOKE_BY_CUSTOMER stand down (a customer holding an
 * annual beside a monthly), because a charge id is not ambiguous.
 *
 * WHAT IT CAN DO, exhaustively: shorten ONE named row to the reversal instant.
 * UPDATE-only (it can never bring a grant into existence, so a linkage pointing
 * at an ERASED subject's deleted row resurrects nothing), `paid_through > ?`
 * (shorten-only — a reversal can never extend anyone, which is what makes "a
 * won dispute restores nothing" structural), and the same last_event ordering
 * guard every other write carries, so a redelivered old chargeback cannot end a
 * grant bought after it.
 *
 * `last_event_rank` is deliberately NOT written, for the same reason
 * REVOKE_BY_CUSTOMER does not write it: a charge event ranks 0, and writing
 * that over a row stamped 3 by `customer.subscription.deleted` would erase the
 * TERMINAL mark EXTEND reads and hand a late invoice the resurrection this
 * module already closed. (The LAPSE statement DOES write the rank, correctly —
 * it is only ever driven by the subscription state machine, whose ranks are the
 * ladder. A reversal is not part of that ladder, which is precisely why it gets
 * its own statement instead of borrowing LAPSE.)
 */
const REVOKE_BY_SUBSCRIPTION =
  "UPDATE billing_entitlements SET paid_through = MIN(paid_through, ?), revoked_at = ?, last_event_at = ?, updated_at = ? " +
  `WHERE subject_id = ? AND subject_kind = ? AND stripe_subscription_id = ? AND ${GRANTS_AT("", "?")} ` +
  "AND (last_event_at < ? OR (last_event_at = ? AND last_event_rank <= ?))";

/**
 * Record WHAT PAID for a grant, so a later refund or chargeback can find it.
 *
 * INSERT-ONCE, never updated: the row is a fact about a payment that has
 * already happened. The `NOT EXISTS` guard is what keeps a redelivery harmless
 * — the same invoice arriving under a second event id (a Stripe replay, a
 * dashboard resend) must not raise a PRIMARY KEY failure, because this
 * statement rides in the SAME batch as the entitlement write and a throw here
 * would roll that write back and answer 5xx for an event that is entirely fine.
 * Checking inside the statement rather than in a read beforehand also removes
 * the window a check-then-act would leave open.
 *
 * A row is written only for a grant this batch has already resolved
 * unambiguously (see the caller), so a linkage can never point at a subject the
 * invoice merely claimed.
 */
const RECORD_PAYMENT =
  "INSERT INTO billing_payments (invoice_id, stripe_charge_id, stripe_payment_intent_id, subject_id, subject_kind, stripe_subscription_id, stripe_customer_id, at, amount_cents, currency) " +
  "SELECT ?, ?, ?, ?, ?, ?, ?, ?, ?, ? WHERE NOT EXISTS (SELECT 1 FROM billing_payments WHERE invoice_id = ?)";

/**
 * I3-1068 (D-0831(1) option B, D-0840(4); table 0138). A reversal that arrives
 * BEFORE the invoice recording its payment linkage — no linkage, no customer —
 * is RECORDED here in the batch that answers it `ignored`, and APPLIED when the
 * linkage lands. APPLY/DELETE run in BOTH batches (the invoice's, after
 * RECORD_PAYMENT, and the reversal's, after RECORD), keyed by that batch's own
 * charge / payment-intent ids, so whichever of the two commits second finds the
 * other's row: no read taken before either batch decides it. APPLY carries
 * REVOKE_BY_SUBSCRIPTION's guards (never lengthens, never overrides a later
 * event; a charge event ranks 0); DELETE retires every row now resolved to a
 * linkage, whether or not the grant still had time to shorten.
 */
const RECORD_UNMATCHED_REVERSAL =
  "INSERT INTO billing_reversals_unmatched (stripe_charge_id, stripe_payment_intent_id, reversal_at, event_id, received_at) VALUES (?, ?, ?, ?, ?) " +
  "ON CONFLICT (event_id) DO NOTHING";
const APPLY_UNMATCHED_REVERSALS =
  "UPDATE billing_entitlements SET paid_through = MIN(billing_entitlements.paid_through, r.at), revoked_at = r.at, last_event_at = r.at, updated_at = ? " +
  "FROM (SELECT MIN(u.reversal_at) AS at, p.subject_id, p.subject_kind, p.stripe_subscription_id " +
  "FROM billing_reversals_unmatched u JOIN billing_payments p " +
  "ON (p.stripe_charge_id = u.stripe_charge_id OR p.stripe_payment_intent_id = u.stripe_payment_intent_id) " +
  "WHERE u.stripe_charge_id = ? OR u.stripe_payment_intent_id = ? " +
  "GROUP BY p.subject_id, p.subject_kind, p.stripe_subscription_id) AS r " +
  "WHERE billing_entitlements.subject_id = r.subject_id AND billing_entitlements.subject_kind = r.subject_kind " +
  `AND billing_entitlements.stripe_subscription_id = r.stripe_subscription_id AND ${GRANTS_AT("billing_entitlements.", "r.at")} ` +
  "AND (billing_entitlements.last_event_at < r.at OR (billing_entitlements.last_event_at = r.at AND billing_entitlements.last_event_rank <= 0))";
const DELETE_MATCHED_REVERSALS =
  "DELETE FROM billing_reversals_unmatched WHERE (stripe_charge_id = ? OR stripe_payment_intent_id = ?) " +
  "AND EXISTS (SELECT 1 FROM billing_payments p WHERE p.stripe_charge_id = billing_reversals_unmatched.stripe_charge_id " +
  "OR p.stripe_payment_intent_id = billing_reversals_unmatched.stripe_payment_intent_id)";

/** The grant a reversed payment bought, or null when we never recorded it
 * (every subscription paid for before 0039, and any invoice whose payment ids
 * we could not read). `= ?` is never true of NULL, so a reversal naming only
 * one of the two ids simply matches on that one. */
interface PaymentLink {
  subject_id: string;
  subject_kind: SubjectKind;
  stripe_subscription_id: string;
}
const FIND_PAYMENT =
  "SELECT subject_id, subject_kind, stripe_subscription_id FROM billing_payments WHERE stripe_charge_id = ? OR stripe_payment_intent_id = ? LIMIT 1";

/**
 * Extend forward ONLY, and only for a subscription we already know. An invoice
 * can never create an entitlement (it carries no Dhwaja subject) and can never
 * shorten one.
 *
 * TWO THINGS IT ALSO MUST NOT DO, both measured.
 *
 *  1. RESURRECT A DELETED SUBSCRIPTION. `customer.subscription.deleted` lapses
 *     the row to the cancel instant, and Stripe never un-deletes a subscription
 *     — a resubscribe is a NEW subscription id. But the row carried no terminal
 *     state, so a late `invoice.payment_succeeded` (the last open invoice
 *     settling two days after the cancel is the ordinary way this happens) moved
 *     paid_through back out to the period end: tier went free → pro again, from
 *     an invoice, with no live subscription anywhere. `last_event_rank` already
 *     records WHICH event last wrote the row, so "was deleted" is a fact the row
 *     already carries; the guard just has to read it.
 *  2. REACH A GRANT IT CANNOT NAME. The statement matched on subscription id
 *     alone while every other write is scoped to its subject. The PK is
 *     (subject_id, stripe_subscription_id), so two subjects CAN carry one
 *     subscription id — measured: a single invoice pushed a second, unrelated
 *     subject's paid_through a month forward. An invoice names no subject, so it
 *     is scoped by the only identity it does carry: `customer`.
 */
const EXTEND =
  "UPDATE billing_entitlements SET paid_through = ?, revoked_at = NULL, last_event_at = ?, updated_at = ? " +
  "WHERE stripe_subscription_id = ? AND stripe_customer_id = ? AND paid_through < ? AND last_event_at <= ? " +
  `AND last_event_rank < ${DELETED_RANK}`;

/**
 * The same statement for the one row an invoice can name but `= ?` can never
 * match: a grant that carries NO customer id at all.
 *
 * `stripe_customer_id = ?` is never true of NULL, so a row written by a `set`
 * whose customer Stripe did not supply was PERMANENTLY unreachable by every
 * invoice for its own subscription — measured `{applied:false,
 * reason:"unresolved"}` with no marker written, which is a 5xx by design, so
 * Stripe redelivers that invoice until it gives up and the subject is simply
 * never extended. Defensive (every `extend` outcome carries a customer id, and
 * stripe.ts refuses an invoice without one), but inherited rather than decided.
 *
 * The decision: ADOPT, and only where adoption cannot be wrong. A Stripe
 * subscription id belongs to exactly one customer, so an invoice saying
 * (sub_X → cus_Y) is authoritative about the row that holds sub_X — PROVIDED
 * there is exactly one such row. That proviso is the whole safety argument, and
 * it is checked by the caller before this runs: the reason EXTEND is scoped by
 * customer at all is that the PK is (subject_id, stripe_subscription_id) and two
 * SUBJECTS can carry one subscription id, so widening the match unconditionally
 * would re-open exactly the defect the scoping closed. Adopting into a
 * single, unambiguous, customerless row does not.
 */
const EXTEND_ADOPT =
  "UPDATE billing_entitlements SET paid_through = ?, revoked_at = NULL, last_event_at = ?, updated_at = ?, stripe_customer_id = ? " +
  "WHERE stripe_subscription_id = ? AND stripe_customer_id IS NULL AND paid_through < ? AND last_event_at <= ? " +
  `AND last_event_rank < ${DELETED_RANK}`;

export interface ApplyResult {
  applied: boolean;
  /** `unresolved` = the write was SUPPOSED to match a row and there was none.
   * The dedup marker is deliberately NOT written for it, and the route answers
   * 5xx so Stripe redelivers — a zero-row UPDATE must never be reported as a
   * success (that was a false 200: no retry, and a paying subject reading
   * `free` with nothing anywhere to say so). */
  reason: "applied" | "duplicate" | "ignored" | "unresolved";
}

/** ONLY a collision on the dedup table's primary key is a duplicate. The old
 * `/UNIQUE|constraint/i` also swallowed CHECK, NOT NULL and FK failures: the
 * whole batch (marker included) rolled back, the route answered 200, Stripe
 * never retried, and the event vanished with no signal anywhere. Everything
 * that is not this exact collision must surface. */
const DUPLICATE_EVENT_RE = /UNIQUE constraint failed:\s*stripe_events\.event_id/i;

/**
 * Apply a verified Stripe event's outcome ATOMICALLY and IDEMPOTENTLY.
 *
 * The dedup marker and the entitlement write commit in ONE batch: if the
 * marker landed alone, a redelivery would dedup-skip and a paying subject
 * would never be entitled (the exact bug S45 fixed in applySubscriptionEvent).
 * A racing duplicate delivery loses on the stripe_events PK, rolls the batch
 * back, and is reported as a no-op rather than a 500 — which is the correct
 * idempotent outcome.
 */
export async function applyStripeOutcome(
  db: DbLike,
  eventId: string,
  eventType: string,
  eventCreatedMs: number,
  outcome: StripeOutcome,
  now: () => string = () => new Date().toISOString(),
): Promise<ApplyResult> {
  const seen = await db.prepare("SELECT 1 AS x FROM stripe_events WHERE event_id = ?").bind(eventId).first();
  if (seen) return { applied: false, reason: "duplicate" };
  // #135SC-04: a config-dependent ignore (no prices / an unrecognised price) is
  // answered BEFORE the marker batch — no marker, `unresolved`, the route's 503
  // (index.ts `stripe_webhook_unresolved`), and Stripe redelivers once the price
  // config is fixed. A marker here would close a paid subscription for good.
  if (outcome.kind === "ignore" && outcome.retry === true) return { applied: false, reason: "unresolved" };

  // An INVOICE names no Dhwaja subject, so "we have never seen this
  // subscription" has to be settled BEFORE the batch: the answer decides
  // whether the dedup MARKER is written at all, and a marker written for an
  // unresolvable invoice is a redelivery we can never accept. This read is
  // therefore a decision about MARKING, not about the write — and its race is
  // one-sided and self-healing: if the `created` lands a moment later we
  // answer `unresolved`, Stripe redelivers, and the retry finds the row.
  let adoptCustomer = false;
  // THE GRANT THIS EVENT IS BOUND TO, when it can be named exactly: for an
  // `extend`, the row a payment linkage will be recorded against; for a
  // `revoke`, the row a recorded linkage resolves the reversal to. Null means
  // "not named", which is a fallback for a revoke and simply no linkage for an
  // extend — never a guess in either direction.
  let grant: PaymentLink | null = null;
  if (outcome.kind === "extend") {
    // Also reads the SUBJECT, because the linkage must be bound to the grant we
    // hold rather than to metadata the invoice carries: `subscription_details`
    // is Stripe's snapshot of what the subscription said, while this row is
    // what we actually granted. LIMIT 2 asks "is it unique?" — two subjects CAN
    // carry one subscription id (the PK is (subject_id, stripe_subscription_id)),
    // and an ambiguous pair records NO linkage rather than picking one.
    const { results } = await db
      .prepare("SELECT subject_id, subject_kind FROM billing_entitlements WHERE stripe_subscription_id = ? AND stripe_customer_id = ? LIMIT 2")
      .bind(outcome.subscriptionId, outcome.customerId)
      .all<{ subject_id: string; subject_kind: SubjectKind }>();
    if (results.length === 1) grant = { ...results[0]!, stripe_subscription_id: outcome.subscriptionId };
    if (results.length === 0) {
      // Before calling it unresolved: is there exactly ONE grant for this
      // subscription that simply carries no customer id? `= ?` can never match a
      // NULL, so such a row is unreachable by every invoice for its own
      // subscription — for good, since nothing else fills the column in. See
      // EXTEND_ADOPT for why exactly-one is what makes adopting safe. LIMIT 2 is
      // the cheapest way to ask "is it unique?".
      const { results: orphans } = await db
        .prepare("SELECT subject_id, subject_kind, stripe_customer_id FROM billing_entitlements WHERE stripe_subscription_id = ? LIMIT 2")
        .bind(outcome.subscriptionId)
        .all<{ subject_id: string; subject_kind: SubjectKind; stripe_customer_id: string | null }>();
      adoptCustomer = orphans.length === 1 && orphans[0]!.stripe_customer_id === null;
      // No grant for this subscription: the invoice may simply have overtaken
      // its customer.subscription.created (Stripe does not order deliveries).
      // Do NOT mark the event — an unwritten marker is what lets the redelivery
      // succeed once the subscription event lands.
      if (!adoptCustomer) return { applied: false, reason: "unresolved" };
      grant = { subject_id: orphans[0]!.subject_id, subject_kind: orphans[0]!.subject_kind, stripe_subscription_id: outcome.subscriptionId };
    }
  } else if (outcome.kind === "revoke" && (outcome.refs.chargeId || outcome.refs.paymentIntentId)) {
    // WHICH GRANT DID THIS REVERSED PAYMENT BUY? A read before the batch, and
    // unlike the lapse's old check-then-act it cannot drop a revocation: the
    // row it looks for is a fact about a payment that settled weeks ago, it is
    // written once and never changed, and if it is somehow not there yet the
    // answer degrades to the customer-scoped statement (or to doing nothing) —
    // both of which are guarded, so the failure direction is "narrower or no
    // action", never "wider or wrong row". Nothing about MARKING depends on it.
    grant = await db.prepare(FIND_PAYMENT).bind(outcome.refs.chargeId, outcome.refs.paymentIntentId).first<PaymentLink>();
  }
  // A LAPSE, by contrast, used to resolve its target the same way and then
  // decide whether to even ATTEMPT the guarded UPDATE — check-then-act, and a
  // cancel is the one outcome that must never be dropped. Measured with a
  // barrier held on that read (test/db-recorder.ts): a
  // `customer.subscription.deleted` whose target read ran while the matching
  // `customer.subscription.created` was in flight found nothing, skipped its
  // UPDATE, reported `ignored` — AND WROTE THE MARKER, so Stripe never
  // redelivered — leaving a cancelled subscriber entitled for the rest of the
  // period (measured: paid_through stayed 2026-10-01 where the sequential
  // order gave 2026-09-01). So the LAPSE now always rides in the batch. It is
  // UPDATE-only and self-guarded (see the constant), so running it against a
  // subject with no grant is a no-op by construction rather than by a prior
  // read, and the outcome is read back AFTER the write instead of guessed
  // before it.

  const ts = now();
  const eventAt = new Date(eventCreatedMs).toISOString();
  const rank = rankOf(eventType);
  const mark = db.prepare("INSERT INTO stripe_events (event_id, event_type, received_at) VALUES (?, ?, ?)").bind(eventId, eventType, ts);
  const writes: DbStatement[] = [mark];

  if (outcome.kind === "set") {
    writes.push(
      db
        .prepare(UPSERT)
        .bind(
          outcome.subjectId,
          outcome.subjectKind,
          outcome.tier,
          new Date(outcome.paidThroughMs).toISOString(),
          "stripe",
          outcome.customerId,
          outcome.subscriptionId,
          eventAt,
          rank,
          ts,
          // H3 resurrection guard params: (kind, id) for the fan branch and the
          // creator branch of the EXISTS check.
          outcome.subjectKind,
          outcome.subjectId,
          outcome.subjectKind,
          outcome.subjectId,
        ),
    );
    writes.push(
      db
        .prepare(CANCEL_STATE + NEWER("stripe_subscription_cancel"))
        .bind(outcome.subscriptionId, outcome.cancelAtMs == null ? null : new Date(outcome.cancelAtMs).toISOString(), eventAt, rank, ts),
    );
  } else if (outcome.kind === "lapse") {
    const paidThrough = new Date(outcome.paidThroughMs).toISOString();
    // BF-07: a TERMINAL lapse (`customer.subscription.deleted`) is not the same
    // act as an ordinary one. It must record that this subscription is over
    // even when there is no paid time left to shorten, and it must leave a mark
    // when the row it cancels has not been delivered yet.
    if (rank === DELETED_RANK) {
      writes.push(db.prepare(TERMINAL).bind(paidThrough, outcome.customerId, eventAt, rank, ts, outcome.subjectId, outcome.subscriptionId, eventAt, eventAt, rank));
      writes.push(
        db
          .prepare(TOMBSTONE)
          .bind(
            outcome.subjectId,
            outcome.subjectKind,
            outcome.subjectKind === "fan" ? "premium" : "pro",
            TERMINAL_PAID_THROUGH,
            outcome.customerId,
            outcome.subscriptionId,
            eventAt,
            rank,
            ts,
            outcome.subjectKind,
            outcome.subjectId,
            outcome.subjectKind,
            outcome.subjectId,
            outcome.subjectId,
            outcome.subscriptionId,
          ),
      );
    } else {
      writes.push(db.prepare(LAPSE).bind(paidThrough, REVERSAL_EVENTS.has(eventType) ? paidThrough : null, outcome.customerId, eventAt, rank, ts, outcome.subjectId, outcome.subscriptionId, paidThrough, paidThrough, eventAt, eventAt, rank));
    }
  } else if (outcome.kind === "revoke") {
    // Rides in the batch for the same reason the lapse does: a revocation is
    // the one thing that must never be dropped, and both statements are
    // self-guarded, so running one against nothing revocable is a no-op by
    // construction rather than by a prior read.
    //
    // NARROWEST FIRST. A recorded linkage names ONE grant, so it is preferred
    // over the customer scope whenever we have it — and it is not "downgraded"
    // to the customer scope when the named grant turns out to have ended
    // already: having positively identified what the money bought, revoking
    // something else instead would be exactly the blast radius the customer
    // scope's own "exactly one revocable grant" rule exists to avoid.
    const paidThrough = new Date(outcome.paidThroughMs).toISOString();
    if (grant) {
      writes.push(db.prepare(REVOKE_BY_SUBSCRIPTION).bind(paidThrough, paidThrough, eventAt, ts, grant.subject_id, grant.subject_kind, grant.stripe_subscription_id, paidThrough, paidThrough, eventAt, eventAt, rank));
    } else if (outcome.customerId) {
      writes.push(
        db
          .prepare(REVOKE_BY_CUSTOMER)
          .bind(paidThrough, paidThrough, eventAt, ts, outcome.customerId, eventAt, eventAt, rank, outcome.customerId, paidThrough, paidThrough, paidThrough, outcome.customerId, paidThrough),
      );
    }
    // No recorded linkage — a reversal delivered before its invoice, or one on
    // a subscription bought before 0039. With no customer nothing is revoked
    // now; with one, the customer scope above acts only on exactly one
    // revocable grant. Either way the reversal is RECORDED (I3-1068, see
    // RECORD_UNMATCHED_REVERSAL) and applied to the grant its linkage names if
    // and when that lands; APPLY never lengthens, so a grant the customer scope
    // already ended is left as it is. Answered as before (`ignored` + marker
    // when nothing landed): a retry cannot make the linkage appear sooner.
    if (!grant && (outcome.refs.chargeId || outcome.refs.paymentIntentId)) {
      const { chargeId, paymentIntentId } = outcome.refs;
      writes.push(
        db.prepare(RECORD_UNMATCHED_REVERSAL).bind(chargeId, paymentIntentId, paidThrough, eventId, ts),
        db.prepare(APPLY_UNMATCHED_REVERSALS).bind(ts, chargeId, paymentIntentId),
        db.prepare(DELETE_MATCHED_REVERSALS).bind(chargeId, paymentIntentId),
      );
    }
  } else if (outcome.kind === "extend") {
    const paidThrough = new Date(outcome.paidThroughMs).toISOString();
    writes.push(
      adoptCustomer
        ? db.prepare(EXTEND_ADOPT).bind(paidThrough, eventAt, ts, outcome.customerId, outcome.subscriptionId, paidThrough, eventAt)
        : db.prepare(EXTEND).bind(paidThrough, eventAt, ts, outcome.subscriptionId, outcome.customerId, paidThrough, eventAt),
    );
    // RECORD WHAT PAID, in this same batch (0039). Only for a grant this batch
    // named unambiguously, and only when the invoice carried a payment id — so
    // the linkage exists for exactly the money that can later be reversed, and
    // for no money we cannot account for. It rides here rather than in its own
    // write because a linkage that landed without the entitlement write (or the
    // reverse) would be a record of a payment we did not actually apply.
    if (outcome.payment && grant) {
      writes.push(
        db
          .prepare(RECORD_PAYMENT)
          .bind(
            outcome.payment.invoiceId,
            outcome.payment.chargeId,
            outcome.payment.paymentIntentId,
            grant.subject_id,
            grant.subject_kind,
            grant.stripe_subscription_id,
            outcome.customerId,
            ts,
            outcome.payment.amountCents,
            outcome.payment.currency,
            outcome.payment.invoiceId,
          ),
        // I3-1068: a reversal of THIS payment that arrived first is applied now.
        db.prepare(APPLY_UNMATCHED_REVERSALS).bind(ts, outcome.payment.chargeId, outcome.payment.paymentIntentId),
        db.prepare(DELETE_MATCHED_REVERSALS).bind(outcome.payment.chargeId, outcome.payment.paymentIntentId),
      );
    }
  }

  try {
    await db.batch(writes);
  } catch (e) {
    if (e instanceof Error && DUPLICATE_EVENT_RE.test(e.message)) return { applied: false, reason: "duplicate" };
    throw e; // any other constraint failure is a LOST WRITE — it must be loud
  }
  // A lapse with no row to lapse changed nothing (there was never a grant);
  // the event is still recorded so its redelivery stays cheap. Asked AFTER the
  // write, so a row that arrived while this cancel was in flight has already
  // been lapsed by the batch rather than missed by a read taken before it.
  if (outcome.kind === "lapse") {
    const landed = await db
      .prepare("SELECT 1 AS x FROM billing_entitlements WHERE subject_id = ? AND stripe_subscription_id = ? LIMIT 1")
      .bind(outcome.subjectId, outcome.subscriptionId)
      .first();
    return landed ? { applied: true, reason: "applied" } : { applied: false, reason: "ignored" };
  }
  // A customer-scoped revocation is read back the same way, and for the same
  // reason: the statement decides for itself whether it may act, so the only
  // honest way to report the outcome is to look at what it did. A row of this
  // customer now ending at exactly this event's reversal instant, stamped with
  // this event's time, is that revocation.
  //
  // `ignored` (200, marker written) covers BOTH remaining shapes, and neither
  // wants a redelivery: nothing revocable (the charge paid for a grant that has
  // already ended, or for a customer we never granted anything — the ordinary
  // case for a refunded one-off), and ambiguous (two live grants, which a retry
  // cannot disambiguate — a 5xx would only make Stripe re-ask the same
  // unanswerable question until it disables the endpoint, taking the whole
  // billing rail with it). The residual is that an ambiguous reversal is quiet;
  // the route logs only `unresolved`.
  if (outcome.kind === "revoke") {
    const paidThrough = new Date(outcome.paidThroughMs).toISOString();
    // Read back through whichever scope actually ran; a reversal that could be
    // scoped to neither wrote nothing and is `ignored` without a read at all.
    const landed = grant
      ? await db
          .prepare("SELECT 1 AS x FROM billing_entitlements WHERE subject_id = ? AND subject_kind = ? AND stripe_subscription_id = ? AND paid_through = ? AND last_event_at = ? LIMIT 1")
          .bind(grant.subject_id, grant.subject_kind, grant.stripe_subscription_id, paidThrough, eventAt)
          .first()
      : outcome.customerId
        ? await db
            .prepare("SELECT 1 AS x FROM billing_entitlements WHERE stripe_customer_id = ? AND paid_through = ? AND last_event_at = ? LIMIT 1")
            .bind(outcome.customerId, paidThrough, eventAt)
            .first()
        : null;
    return landed ? { applied: true, reason: "applied" } : { applied: false, reason: "ignored" };
  }
  // An EXTEND is also asked AFTER the write, for the same reason a lapse is: the
  // statement is self-guarded, so "did it land?" is a fact about the row. An
  // invoice against a DELETED subscription legitimately moves nothing, and
  // reporting that as `applied` would be the same false success the lapse
  // readback exists to prevent. `ignored` (not `unresolved`) is the honest
  // answer: 200, no redelivery — a deleted subscription will never become
  // extendable, so there is nothing for Stripe to retry into.
  if (outcome.kind === "extend") {
    const row = await db
      .prepare("SELECT paid_through FROM billing_entitlements WHERE stripe_subscription_id = ? AND stripe_customer_id = ? ORDER BY paid_through DESC LIMIT 1")
      .bind(outcome.subscriptionId, outcome.customerId)
      .first<{ paid_through: string }>();
    const covered = row !== null && row.paid_through >= new Date(outcome.paidThroughMs).toISOString();
    return covered ? { applied: true, reason: "applied" } : { applied: false, reason: "ignored" };
  }
  // An `ignore` outcome wrote nothing but the marker (a Stripe event type we
  // deliberately do not act on) — recorded so its redelivery stays cheap.
  return outcome.kind === "ignore" ? { applied: false, reason: "ignored" } : { applied: true, reason: "applied" };
}

/** Bound the dedup table (cron). Stripe stops retrying long before this. */
export async function pruneStripeEvents(db: DbLike, olderThanIso: string): Promise<number> {
  const res = await db.prepare("DELETE FROM stripe_events WHERE received_at < ?").bind(olderThanIso).run();
  return res.changes;
}

/** I21-Q72 (D-1030: "400 days"): how long an unmatched early-refund note (0138) is kept waiting for its payment. */
export const UNMATCHED_REVERSAL_RETENTION_MS = 400 * 24 * 60 * 60 * 1000;
/** Rows deleted per call, oldest first; the next tick takes the rest. */
export const UNMATCHED_REVERSAL_PRUNE_LIMIT = 500;

/** I21-Q72: delete `billing_reversals_unmatched` rows received more than UNMATCHED_REVERSAL_RETENTION_MS before
 * `nowMs`, at most `limit` per call, on `idx_bru_received` (0225). Cron (scheduled(), beside pruneStripeEvents). */
export async function pruneUnmatchedReversals(db: DbLike, nowMs: number, limit: number = UNMATCHED_REVERSAL_PRUNE_LIMIT): Promise<number> {
  const cutoff = new Date(nowMs - UNMATCHED_REVERSAL_RETENTION_MS).toISOString();
  const res = await db
    .prepare(
      "DELETE FROM billing_reversals_unmatched WHERE rowid IN (SELECT rowid FROM billing_reversals_unmatched WHERE received_at < ? ORDER BY received_at LIMIT ?)",
    )
    .bind(cutoff, limit)
    .run();
  return res.changes;
}

/** I21-Q37: the audit action of every outcome of cancelWebSubscriptionAfterStoreGrant
 * (`ok sub=<id>` | `failed sub=<id>` | `skipped n=<k> reason=stripe_not_configured`). */
export const WEB_CANCEL_AFTER_STORE_ACTION = "billing:web_cancel_after_store";

/**
 * I21-Q37 (D-1030: "apply it and cancel the web subscription at period end"). Called after a STORE subscription grant
 * lands (index.ts applyIapGrant): while the store rail reads premium, every LIVE Stripe fan grant of `accountId`
 * (paid_through in the future, no earlier `ok` row) is set to cancel at its period end
 * (stripe.ts setStripeCancelAtPeriodEnd), one audit row per subscription. Stripe unconfigured: one `skipped` row,
 * no call. Never throws: the store grant has already landed and stands whatever Stripe answers.
 */
export async function cancelWebSubscriptionAfterStoreGrant(
  db: DbLike,
  env: Record<string, unknown>,
  accountId: string,
  fetchImpl: typeof fetch = fetch,
  nowMs: number = Date.now(),
): Promise<{ cancelled: number; failed: number; skipped: number }> {
  const out = { cancelled: 0, failed: 0, skipped: 0 };
  const at = new Date(nowMs).toISOString();
  const record = async (detail: string): Promise<void> => {
    try {
      await db
        .prepare("INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) VALUES ('system', ?, ?, ?, ?, 'system')")
        .bind(WEB_CANCEL_AFTER_STORE_ACTION, detail, at, accountId)
        .run();
    } catch {
      // the store grant stands; an unwritable audit row cannot undo it
    }
  };
  try {
    if ((await currentTier(db, accountId, nowMs)) === "free") return out;
    const { results } = await db
      .prepare(
        "SELECT DISTINCT be.stripe_subscription_id AS sub FROM billing_entitlements be WHERE be.subject_id = ? AND be.subject_kind = 'fan' AND be.paid_through > ? " +
          "AND NOT EXISTS (SELECT 1 FROM audit_log a WHERE a.action = ? AND a.item_id = be.subject_id AND a.detail = 'ok sub=' || be.stripe_subscription_id)",
      )
      .bind(accountId, at, WEB_CANCEL_AFTER_STORE_ACTION)
      .all<{ sub: string }>();
    if (results.length === 0) return out;
    const cfg = stripeConfigFromEnv(env);
    if (!cfg) {
      await record(`skipped n=${results.length} reason=stripe_not_configured`);
      out.skipped = results.length;
      return out;
    }
    for (const { sub } of results) {
      const ok = await setStripeCancelAtPeriodEnd(cfg, sub, fetchImpl);
      await record(`${ok ? "ok" : "failed"} sub=${sub}`);
      if (ok) out.cancelled += 1;
      else out.failed += 1;
    }
  } catch {
    await record("failed reason=unread");
    out.failed += 1;
  }
  return out;
}

/** I25-Q3: a creator's Stripe (Creator Pro) subscription ids — the creator twin of account_data.ts
 * readStripeSubscriptionIds (same predicate, `subject_kind = 'creator'`), read by creator-close.ts to pause, resume
 * or cancel them. */
export async function readCreatorStripeSubscriptionIds(db: DbLike, passportId: string): Promise<string[]> {
  const { results } = await db
    .prepare(
      "SELECT DISTINCT stripe_subscription_id FROM billing_entitlements WHERE subject_id = ? AND subject_kind = 'creator' AND stripe_subscription_id IS NOT NULL AND stripe_subscription_id != ''",
    )
    .bind(passportId)
    .all<{ stripe_subscription_id: string }>();
  return results.map((r) => r.stripe_subscription_id);
}

/**
 * Billing entitlements read logic — the flag-17/24 rules, as pure logic.
 * (Decisions: claude/BILLING-AGENCY-DECISIONS.md; plan M7.)
 *
 * Pinned rules:
 *  - tiers: fan free/premium · creator free/pro
 *  - `paid_through` + 7-day grace, computed at READ time (never stored)
 *  - unknown/absent row ⇒ free (fail closed for features)
 *  - THE PAYWALL INVARIANT: only cap/tool gates may call this module.
 *    It must never be imported by registry-build, warnings, caution,
 *    verification, or any served-verdict path (invariant test in M7).
 *
 * Dependency-free; times are epoch ms from the caller.
 */

export const GRACE_MS = 7 * 24 * 60 * 60 * 1000;

export type FanTier = "free" | "premium";
export type CreatorTier = "free" | "pro";

export interface EntitlementRow {
  tier: "premium" | "pro";
  /** epoch ms the current paid period ends (from the billing webhook). */
  paidThroughMs: number;
  /** I21-Q61 (D-1030): the grant ended because its money was taken back (a full refund or a dispute; `revoked_at`,
   * 0224). Such a row gets NO grace; a card-failure lapse (absent/false) keeps the 7 days. */
  revoked?: boolean;
}

/** The grace a row gets past its paid_through: none once revoked (I21-Q61), else GRACE_MS. */
export function graceMsFor(row: EntitlementRow): number {
  return row.revoked === true ? 0 : GRACE_MS;
}

/** Effective tier at `nowMs`, honoring the 7-day grace (none for a revoked row). Null row ⇒ free. */
export function effectiveTier<T extends "premium" | "pro">(
  row: (EntitlementRow & { tier: T }) | null | undefined,
  nowMs: number,
): T | "free" {
  if (!row) return "free";
  return nowMs < row.paidThroughMs + graceMsFor(row) ? row.tier : "free";
}

/** True while past paid_through but inside grace (for the quiet lapse banner). */
export function inGrace(row: EntitlementRow | null | undefined, nowMs: number): boolean {
  if (!row) return false;
  return nowMs >= row.paidThroughMs && nowMs < row.paidThroughMs + graceMsFor(row);
}

/** The shipped S90 caps by tier — the ONLY thing money changes. */
export function recognitionCaps(tier: CreatorTier): {
  likeDaily: number;
  namedDaily: number;
  activeFollows: number;
} {
  return tier === "pro"
    ? { likeDaily: 50, namedDaily: 5, activeFollows: 250 }
    : { likeDaily: 10, namedDaily: 1, activeFollows: 25 };
}
