import type { CategoryBoard, DiscoveryRankings, RankedCreator } from "./rankings.js";
import { CATEGORIES, CATEGORY_LABELS, CONSENT_REQUIRED_CATEGORIES, type Category } from "./taxonomy.js";
import { HOW_PATH } from "./how-page.js";
import { settingValue } from "./env-config.js";
import { translator, type MessageKey } from "@dhwaja/i18n";

/** A locale-bound lookup. Every fixed sentence on these two pages comes from
 * `@dhwaja/i18n`; the catalogue returns TEXT and this file escapes it, so no
 * catalogue value can introduce markup. */
type Translate = (key: MessageKey, vars?: Readonly<Record<string, string>>) => string;

/**
 * SEO-indexable, server-rendered discovery pages — the crawlable counterpart to
 * the hash-routed SPA (which a crawler can't index). Same posture as the S64
 * `/p/:id` share page: a minimal, DEPENDENCY-FREE, SCRIPT-FREE HTML document that
 * a bot can read, with honest `<title>`/description/canonical/og and internal
 * links (`/discover` ↔ `/browse/<niche>` ↔ `/p/:id`) so the discovery graph is
 * crawlable. Every dynamic value is escaped; ids are shape-validated before they
 * form a link, so a stray board value can never inject markup or a bad href.
 *
 * These read the ALREADY-PRECOMPUTED S81 discovery artifact (never a live
 * aggregate), so the pages are cheap + edge-cacheable.
 */

const PSP_RE = /^psp_[0-9a-f]{16}$/;
const CAT_RE = /^[a-z_]{2,24}$/;

/** Is `k` a real, frozen taxonomy category? */
export function isCategoryKey(k: string): k is Category {
  return (CATEGORIES as readonly string[]).includes(k);
}

/** The human label for a category key (frozen vocab; escaped fallback to the key). */
export function categoryLabel(k: string): string {
  return isCategoryKey(k) ? CATEGORY_LABELS[k] : k;
}

/** Only claimed/verified creators reach these boards (rankings.ts serves no
 * unclaimed subject — discovery is a place a creator opts into by claiming),
 * so an unrecognised state renders NO pill rather than inventing one. */
const STATE_VERDICT: Record<string, VerdictKey> = { verified: "verified", claimed: "claimed" };

/** One crawlable creator row: the display name links to the server `/p/:id` page
 * ONLY when the id is well-formed; otherwise it's inert escaped text. The
 * verdict is a pill — colour AND icon AND text label, never colour alone. */
/** I3-0676 — a board score as the reader's locale writes a number: grouped
 * ("1,234,567", not "1234567") and at most the two decimals the fan-weighted
 * sum carries (`fanWeightSum` rounds to 2). An unknown locale tag falls back to
 * English rather than throwing out of the page. */
function scoreFormat(locale: string): Intl.NumberFormat {
  try {
    return new Intl.NumberFormat(locale, { maximumFractionDigits: 2 });
  } catch {
    return new Intl.NumberFormat("en", { maximumFractionDigits: 2 });
  }
}

// I3-0676/I3-2050: a score of exactly one takes the singular unit ("1 collector"),
// chosen by the locale's plural rule rather than `=== 1`.
function unitFor(score: number, unit: string, unitOne: string, nf: Intl.NumberFormat): string {
  return new Intl.PluralRules(nf.resolvedOptions().locale).select(score) === "one" ? unitOne : unit;
}

/** I3-0401 (D-1082(12)): a verified row's link to its `/is/` answer page, from the platform/handle `serveDiscovery`
 * attached live (rankings.ts answerAccounts); the shape is re-asserted here, as passport-view.ts answerLink does. A row
 * that is not verified, or carries no valid pair, gets nothing. */
const IS_PLATFORM_RE = /^[a-z0-9_]{1,24}$/;
const IS_HANDLE_RE = /^[a-z0-9][a-z0-9._-]{0,63}$/;
export function answerPageLink(c: RankedCreator, label: string): string {
  if (c?.claim_state !== "verified" || typeof c.platform !== "string" || typeof c.handle !== "string") return "";
  if (!IS_PLATFORM_RE.test(c.platform) || !IS_HANDLE_RE.test(c.handle)) return "";
  return ` <a class="isk" href="/is/${esc(c.platform)}/${esc(c.handle)}">${esc(label)}</a>`;
}

function creatorRow(c: RankedCreator, unit: string, unitOne: string, rank: number, nf: Intl.NumberFormat, isLabel: string): string {
  const name = esc(String(c?.display_name ?? "").slice(0, 64).trim());
  if (!name) return "";
  const pid = typeof c?.passport_id === "string" && PSP_RE.test(c.passport_id) ? c.passport_id : null;
  const label = pid ? `<a href="/p/${pid}">${name}</a>` : name;
  const key = STATE_VERDICT[typeof c?.claim_state === "string" ? c.claim_state : ""];
  const badge = key ? verdictPill(key, undefined, true) : "";
  const score = typeof c?.score === "number" && Number.isFinite(c.score) ? `<span class="score">${esc(nf.format(c.score))} ${esc(unitFor(c.score, unit, unitOne, nf))}</span>` : "";
  return `<li><span class="rank" aria-hidden="true">${rank}</span>${label}${badge}${score}${answerPageLink(c, isLabel)}</li>`;
}

function board(title: string, blurb: string, items: RankedCreator[] | undefined, unit: string, unitOne: string, nf: Intl.NumberFormat, isLabel: string, max = 25): string {
  const rows = (Array.isArray(items) ? items : []).slice(0, max).map((c, n) => creatorRow(c, unit, unitOne, n + 1, nf, isLabel)).filter(Boolean).join("");
  if (!rows) return "";
  return `<section><h2>${esc(title)}</h2><p class="note">${esc(blurb)}</p><ol class="rows">${rows}</ol></section>`;
}

/**
 * The NEWCOMER rail (F11 / #39) — UNNUMBERED and UNSCORED, and both omissions
 * are the feature.
 *
 * A rank number says "these are in order of merit", and this rail's order is a
 * rotation queue: least-exposed first, ties broken by a per-day hash. Printing
 * 1..8 beside it would invite a creator to try to climb something that cannot
 * be climbed, and would tell a viewer something false about what they are
 * looking at.
 *
 * The SCORE is worse than merely meaningless here. `newcomers` zeroes it on
 * purpose, so `creatorRow` would render "0 collects" — which reads as broken,
 * and on a rail whose membership rule is "under the size ceiling" any score at
 * all is a bound on that creator's audience they never published. So this row
 * carries a name and a verdict pill and nothing else, the shape `most_targeted`
 * already uses for the same reason.
 */
function newcomerRail(title: string, blurb: string, items: RankedCreator[] | undefined, isLabel: string, max = 8): string {
  const rows = (Array.isArray(items) ? items : [])
    .slice(0, max)
    .map((c) => {
      const name = esc(String(c?.display_name ?? "").slice(0, 64).trim());
      if (!name) return "";
      const pid = typeof c?.passport_id === "string" && PSP_RE.test(c.passport_id) ? c.passport_id : null;
      const label = pid ? `<a href="/p/${pid}">${name}</a>` : name;
      const key = STATE_VERDICT[typeof c?.claim_state === "string" ? c.claim_state : ""];
      return `<li>${label}${key ? verdictPill(key, undefined, true) : ""}${answerPageLink(c, isLabel)}</li>`;
    })
    .filter(Boolean)
    .join("");
  if (!rows) return "";
  // `ul`, not `ol`: the markup itself must not imply a ranking.
  return `<section><h2>${esc(title)}</h2><p class="note">${esc(blurb)}</p><ul class="rows unranked">${rows}</ul></section>`;
}

/** Niche chips linking to each `/browse/<category>` server page (internal links). */
function nicheChips(boards: CategoryBoard[] | undefined, m: Translate): string {
  // F-16: CONSENT_REQUIRED_CATEGORIES governs publication surfaces, not just
  // storage — a homepage/browse-page chip is unauthenticated discovery same
  // as the sitemap entry below, so it gets the same exclusion.
  const list = (Array.isArray(boards) ? boards : []).filter(
    (b) => b && typeof b.category === "string" && CAT_RE.test(b.category) && !CONSENT_REQUIRED_CATEGORIES.has(b.category as Category),
  );
  if (list.length === 0) return "";
  const chips = list.map((b) => `<li><a href="/browse/${esc(b.category)}" class="chip">${esc(categoryLabel(b.category))}</a></li>`).join("");
  return `<nav class="cats"><h2>${esc(m("disc.niche_heading"))}</h2><ul class="chips">${chips}</ul></nav>`;
}

export interface DiscoveryPageOpts {
  canonicalUrl?: string | null; // absolute PUBLIC_BASE_URL + path (route-derived, never Host)
  appHref?: string | null; // absolute APP_ORIGIN deep link into the interactive SPA
  /** Negotiated from `Accept-Language` by the route. Defaults to English, which
   * is also what every unknown or hostile value resolves to. */
  locale?: string;
  /** I3-0594 (I23): false on any deployment that is not production (`isProductionDeploy`) — every page is then
   * `noindex,nofollow` whatever it would otherwise be. Absent means production, the behaviour before the key existed. */
  production?: boolean;
}

/** I3-0594 (I23, D-0993(1) pick A): the ONE reading of the `DEPLOY_ENV` key. Production is exactly the literal
 * "production" once read through settingValue (I63 C2, I62-CC-08: padding trimmed); unset, empty or any other value
 * (staging, preview, a typo) is NOT production, so a forgotten key de-indexes a copy rather than indexing it. The route
 * passes `{ production: isProductionDeploy(env.DEPLOY_ENV) }`. */
export function isProductionDeploy(raw: unknown): boolean {
  return settingValue(raw) === "production";
}

type Robots = "index,follow" | "noindex,nofollow" | "noindex,follow";

function shell(title: string, description: string, canonicalUrl: string | null | undefined, appHref: string | null | undefined, body: string, locale: string, m: Translate, robots: Robots = "index,follow", production = true): string {
  // NB: include `_` — 8 of the 17 vocab categories (e.g. streaming_gaming) contain
  // one, so a class without it would silently drop the canonical/og/app links for
  // half the browse pages (the URLs are also esc()'d, so this is a validity gate,
  // not the XSS defence). `_` is not HTML-significant.
  const canon = canonicalUrl && /^https:\/\/[a-z0-9._\-/]+$/i.test(canonicalUrl) ? canonicalUrl : null;
  const app = appHref && /^https:\/\/[a-z0-9._\-/#]+$/i.test(appHref) ? appHref : null;
  const og = [
    `<meta property="og:title" content="${esc(title)}">`,
    `<meta property="og:description" content="${esc(description)}">`,
    `<meta property="og:type" content="website">`,
    canon ? `<meta property="og:url" content="${esc(canon)}">` : "",
    `<meta name="twitter:card" content="summary">`,
  ].filter(Boolean).join("\n");
  // The SAME string the shell's own hand-off uses — one key, not two copies that
  // a translator can make disagree on one page.
  const appLink = app ? `<a class="linkcta" href="${esc(app)}" rel="nofollow">${esc(m("shell.hand_off"))}</a>` : "";
  return renderShell({
    locale,
    title,
    description,
    // The discovery boards carry claimed/verified subjects only — people who
    // opted in by claiming — so these index. No unclaimed subject is ever
    // listed here, and absence from this list says nothing about anyone.
    // F-16: except a CONSENT_REQUIRED_CATEGORIES page — the reader-side gate
    // the finder found missing; the caller passes "noindex,nofollow" for it.
    robots: production ? robots : "noindex,nofollow",
    canonicalUrl: canon,
    headExtra: og,
    appHref: app,
    body: `${body}\n${appLink}`,
    footerNote: m("disc.footer_note"),
  });
}

/**
 * THE HEADLINE SAYS "CLAIMED AND VERIFIED", BECAUSE THAT IS WHAT THE RAILS ARE
 * (audit item 26).
 *
 * These pages used to open "Discover verified creators", promise "that
 * creator's verified Dhwaja profile", and ship both to the `<title>`, the meta
 * description and the OG card — while `rankings.ts` selects
 * `claim_state IN ('claimed','verified')` on Most Collected, Most Discovered,
 * Fastest Rising and every niche board. Verified is a STANDING on this product,
 * with its own ceremony, its own expiry and its own refusal
 * (`verified creators only`); calling a claimed creator verified on the one
 * surface a search engine indexes hands them a badge they did not earn and
 * devalues it for everyone who did. The page already contradicted itself: its
 * own footer says "Every creator listed here is claimed or verified".
 *
 * THE COPY MOVED, NOT THE RAILS. Narrowing the boards to verified-only would
 * empty them and would also be the wrong product — a claimed creator has proved
 * control of their accounts, which is exactly what discovery is for. Per-row
 * truth is unchanged and stays the authority: `creatorRow` renders each row's
 * real state as a `verdictPill`, colour AND icon AND label.
 *
 * BREAKOUT KEEPS ITS WORD. `rankings.ts` breakout() filters
 * `claim_state = 'verified'` and requires a recent `verification_granted`, so
 * "Just verified and already climbing" is a true sentence about that rail and
 * only that rail.
 */
/** The crawlable discovery homepage: the rails + niche links. */
export function renderDiscoverHtml(d: DiscoveryRankings | null, opts: DiscoveryPageOpts = {}): string {
  const locale = opts.locale ?? "en";
  const m = translator(locale);
  // ONE h1 STRING, built once and used by both arms. The empty state used to
  // repeat it; the two drifting apart is exactly how a corrected headline
  // half-lands, and a key does not fix that on its own — the variable does.
  const h1 = `<h1>${esc(m("disc.h1"))}</h1>`;
  const nf = scoreFormat(locale);
  const body = [
    h1,
    `<p class="lede">${esc(m("disc.lede"))}</p>`,
    nicheChips(d?.by_category, m),
    newcomerRail(m("disc.board_newcomers"), m("disc.board_newcomers_blurb"), d?.newcomers, m("pv.is_link")),
    board(m("disc.board_breakout"), m("disc.board_breakout_blurb"), d?.breakout, m("disc.unit_collects"), m("disc.unit_collect"), nf, m("pv.is_link")),
    board(m("disc.board_rising"), m("disc.board_rising_blurb"), d?.fastest_rising, m("disc.unit_this_week"), m("disc.unit_this_week"), nf, m("pv.is_link")),
    board(m("disc.board_collected"), m("disc.board_collected_blurb"), d?.most_collected, m("disc.unit_collectors"), m("disc.unit_collector"), nf, m("pv.is_link")),
    board(m("disc.board_discovered"), m("disc.board_discovered_blurb"), d?.most_discovered, m("disc.unit_fans"), m("disc.unit_fan"), nf, m("pv.is_link")),
  ].filter(Boolean).join("\n");
  const anyBoards = (d && (d.newcomers?.length || d.breakout?.length || d.fastest_rising?.length || d.most_collected?.length || d.most_discovered?.length || d.by_category?.length)) || false;
  const main = anyBoards ? body : `${h1}<p class="empty">${esc(m("disc.empty"))}</p>`;
  return shell(m("disc.title"), m("disc.description"), opts.canonicalUrl, opts.appHref, main, locale, m, "index,follow", opts.production !== false);
}

/**

 * never a page a search result could land on. Crawling them costs crawl budget

 * `*` and `$` are RFC 9309 pattern characters. Held against the census of

 */
export const ROBOTS_DISALLOW: readonly string[] = [

  "/editions/", "/health", "/leaderboards", "/ops/", "/proof/", "/rankings/", "/release/", "/requests/",
  "/reverse-codes/", "/taxonomy", "/u/", "/unsubscribe/", "/war/",
  // I25-GATEFIX: the session namespace answers JSON only (GET /session/csrf; POST /session/signout, /session/image-token).
  "/session/",
  // I23: two new JSON reads — the OpenTimestamps seal anchors (I3-0095 a) and the creator's profile (I3-0220).
  "/seal-anchors", "/p/*/profile$",
  "/p/*/json", "/p/*/profile-meta", "/p/*/archive", "/p/*/feed", "/p/*/presentation", "/p/*/card/",
  "/p/*/fan-of-the-week", "/p/*/top-fans", "/p/*/requests", "/p/*/answers", "/p/*/scouts", "/p/*/demand$",
];

/** robots.txt — everything but the JSON/API namespaces above, and the sitemap
 * when a base is set. */
export function renderRobotsTxt(base: string | null, opts: { production?: boolean } = {}): string {
  // I3-0594 (I23): a non-production deployment disallows everything and advertises no sitemap.
  if (opts.production === false) return "User-agent: *\nDisallow: /\n";
  const sm = base ? `\nSitemap: ${base}/sitemap.xml` : "";
  return `User-agent: *\nAllow: /\n${ROBOTS_DISALLOW.map((p) => `Disallow: ${p}`).join("\n")}${sm}\n`;
}

/** sitemap.xml of the STABLE discovery URLs: /discover + every niche /browse/<cat>.
 * Per-creator /p/:id pages are reachable by crawling those pages' links, so they
 * stay out of the sitemap (keeping it small + not an enumeration of the registry).
 * Requires an absolute base; the caller returns 404 when none is configured.
 *
 * THE `/is/<platform>/<handle>` ANSWER PAGES ARE DELIBERATELY NOT LISTED HERE,
 * and F15 turned that from a TODO into a decision.
 *
 * The old note proposed "a cron-built, gate-filtered `sitemap-is.xml`" when the
 * pages were wanted. Having looked at what that artifact would be: it should not
 * exist. A SITEMAP IS A COMPLETE, PUBLISHED, ARCHIVED LIST OF THE POPULATION —
 * third parties keep copies, that is what crawlers are — so presence on Monday
 * and absence on Tuesday becomes a durable, attributable signal, and the reasons
 * a row leaves are retraction, suppression and delisting. That is exactly the
 * bit `tests/invariants/suppression-oracles.test.ts` exists to protect: a
 * suppressed subject must be indistinguishable from a never-listed one, and
 * byte-equal 404s do not survive an observer who kept yesterday's list.
 *
 * The surfaces that DO publish subjects have neither property. `/is/…` 404s for
 * a suppressed subject, but only to somebody who already holds the handle, and
 * the answer never says whether it was ever listed. The boards publish a TOP 25,
 * where absence is overwhelmingly "not in the top 25". A full sitemap turns both
 * into an enumeration with a diff — a new class of exposure, not more of an
 * existing one. (An OPT-IN sitemap would close it, because absence would then be
 * ambiguous between never-opted-in, opted-out and delisted. Recorded rather than
 * built: it costs a consent surface to deliver a crawl path the links below
 * already deliver, and its honest default would be off.)
 *
 * AND THE DISTINCTION IS AUTHORITY, NOT ARCHIVING. Getting these pages crawled
 * IS getting them archived — by the index, by Common Crawl, by every SEO tool's
 * URL history — so "no artifact anybody can archive" would be false. What a
 * first-party sitemap adds is that it STATES the set: absence on Tuesday is
 * Dhwaja saying the URL is gone. A crawl-derived index is partial, lags by days
 * or weeks, and its absences are ambiguous between delisted, not-recrawled and
 * deranked. A delisting oracle needs a signal the observer can trust; that is
 * the property a sitemap hands over and a crawl does not.
 *
 * SO THE PAGES ARE DISCOVERED THE WAY THIS DOCBLOCK ALWAYS CLAIMED — by
 * crawling the links that point at them. That claim used to be false: a grep of
 * the whole tree for an `/is/` href found none, so the product's search-engine
 * front door had no inbound link and no sitemap entry and was reachable only by
 * typing the URL. F15 emits the link from `/p/:id` (`passport-view.ts
 * answerLink`), which completes the path — sitemap → /discover → /p/:id → /is/… —
 * and inherits the passport page's own gate rather than creating a list. The
 * honest ceiling on that path is recorded rather than hidden: the boards are
 * top-25, so internal reach is a few hundred passports, heavily overlapping.
 * Real discovery is external backlinks to `/p/:id` from creators' own bios,
 * which is the share-landing path the product already depends on. */
export function renderSitemapXml(base: string, boardsAsOf?: string | null): string {
  // `/how-it-works` first, and it is the only entry here that is not a board.
  // It is the page a fan who got a frightening answer needs and the one a search
  // engine should be able to find directly — the explainer used to live only
  // inside `/is/:platform/:handle`, which 404s for exactly the subject a
  // worried fan is asking about.
  // I3-1630 — `<lastmod>` on the BOARD pages only, as the UTC day of the
  // discovery artifact they render (`generated_at`), when the caller has it and
  // it is a real date. The explainer has no build date to state, so it gets
  // none rather than an invented one. Day precision: the boards change daily.
  const day = typeof boardsAsOf === "string" && /^\d{4}-\d{2}-\d{2}(?:T|$)/.test(boardsAsOf) && !Number.isNaN(Date.parse(boardsAsOf.slice(0, 10))) ? boardsAsOf.slice(0, 10) : null;
  const lm = day ? `<lastmod>${day}</lastmod>` : "";
  const boards = [`${base}/discover`, ...CATEGORIES.filter((k) => !CONSENT_REQUIRED_CATEGORIES.has(k)).map((k) => `${base}/browse/${k}`)];
  const body = `<url><loc>${esc(`${base}${HOW_PATH}`)}</loc></url>` + boards.map((u) => `<url><loc>${esc(u)}</loc>${lm}</url>`).join("");
  return `<?xml version="1.0" encoding="UTF-8"?><urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9">${body}</urlset>`;
}

/** A crawlable single-niche browse page. `category` is a validated vocab key. */
export function renderBrowseHtml(d: DiscoveryRankings | null, category: string, opts: DiscoveryPageOpts = {}): string {
  const locale = opts.locale ?? "en";
  const m = translator(locale);
  const label = categoryLabel(category);
  const board_ = (Array.isArray(d?.by_category) ? d!.by_category : []).find((b) => b && b.category === category) ?? null;
  const nf = scoreFormat(locale);
  const rows = (Array.isArray(board_?.top) ? board_!.top : []).slice(0, 25).map((c, n) => creatorRow(c, m("disc.unit_collectors"), m("disc.unit_collector"), n + 1, nf, m("pv.is_link"))).filter(Boolean).join("");
  const list = rows ? `<ol class="rows">${rows}</ol>` : `<p class="empty">${esc(m("browse.empty", { label }))}</p>`;
  const otherNiches = nicheChips(d?.by_category, m);
  const body = `<nav class="crumbs"><a href="/discover">${esc(m("browse.crumb"))}</a> › ${esc(label)}</nav>
<h1>${esc(m("browse.h1", { label }))}</h1>
<p class="lede">${esc(m("browse.lede", { label }))}</p>
${list}${otherNiches}`;
  // Same correction as the homepage, same reason (see DISCOVER_H1): byCategory
  // ranks `claim_state IN ('claimed','verified')`, so "the top verified X
  // creators" was a claim about people the board does not filter for.
  // I3-0633 (I23, D-0993(1) pick (a)): a board with no rows is a thin page — noindex, but its links still follow.
  // Page-level only; the sitemap keeps every board (the STABLE-sitemap design).
  const robots: Robots = isCategoryKey(category) && CONSENT_REQUIRED_CATEGORIES.has(category) ? "noindex,nofollow" : rows ? "index,follow" : "noindex,follow";
  return shell(m("browse.title", { label }), m("browse.description", { label }), opts.canonicalUrl, opts.appHref, body, locale, m, robots, opts.production !== false);
}

 * Domain control-proof claim (Cluster E foundation). A brand proves it controls
 * one of its passport's DOMAIN accounts by publishing a Dhwaja-issued token in a
 * DNS TXT record on the apex domain; we look it up over DNS-over-HTTPS (a fixed,
 * operator-configured resolver) and, on a match, write claim_proven.
 *
 * Mirrors the bio-code flow (claims.ts): the proof IS the authorization (no
 * login to start), the write goes through appendEvents (which enforces
 * unclaimed→claimed), single-use via claim_consumed in the same atomic batch,
 * and actor_kind is "system". The ONE difference is the verification channel:
 * DNS TXT via DoH instead of an HTTPS profile fetch. Fail-closed: no configured
 * resolver → the caller cannot verify (503 at the route).
 */

export const DOMAIN_CODE_TTL_MS = 24 * 60 * 60 * 1000; // 24h to publish DNS (propagation is slow)
export const MAX_DOMAIN_VERIFY_ATTEMPTS = 10;
export const MAX_DOMAIN_CHALLENGES_PER_DAY = 5; // per (account, caller) — see startDomainClaim
/** Per-PASSPORT day cap on the self-serve add door, taken before any read. Ten
 * is generous for a creator listing their own domains and small enough that
 * the door is never a bulk instrument. Keyed on the passport so extra sessions
 * do not multiply the budget. */
export const DOMAIN_ADD_CAP_PER_DAY = 10;
export const DOMAIN_CHALLENGE_WINDOW_MS = 24 * 60 * 60 * 1000;
/**
 * CONTINUITY WINDOW (S-007 X3(b)). How old an OPERATOR-SEEDED passport's domain
 * account may be, in LEDGER time, before a successful TXT proof stops minting a
 * session on its own and routes to a person instead.
 *
 * INV-018's sentence — "a handle that changes hands is self-invalidating" — at
 * the claim door. Nobody asked us to build the seeded passport; we listed it,
 * and everything attached to it is what WE observed about whoever held the
 * domain then. A TXT record proves who holds it TODAY. Those are the same fact
 * only while the two are close together, and 30 days is where this product
 * stops assuming they are.
 */
export const DOMAIN_CLAIM_CONTINUITY_DAYS = 30;
export const TXT_PREFIX = "dhwaja-domain-verification=";
const DOH_TIMEOUT_MS = 5000;
const DOH_MAX_BYTES = 64 * 1024;

export class DomainClaimError extends Error {
  override name = "DomainClaimError";
  constructor(
    message: string,
    public readonly status: 400 | 404 | 409 | 429 | 503 = 400,
  ) {
    super(message);
  }
}

/** A TXT lookup over DNS-over-HTTPS. Injectable so tests don't hit the network. */
export type TxtLookup = (host: string) => Promise<string[]>;
/** I3-0189 c: a DoH answer also says WHERE each TXT sat — `via[i]` is the owner name (DNS-normalised) of record `i`,
 * the queried name or a name on its CNAME chain. `dohTxtAnswer` always sets it; an injected lookup that does not
 * cannot claim a foreign owner, so `txtOwnerOutsideApex` reads its absence as "on the queried name". */
export type TxtAnswer = string[] & { via?: string[] };

/** A DNS name, comparably: lowercased, one trailing dot stripped. DNS names are
 * case-insensitive and a DoH resolver may return either the fully-qualified
 * `example.com.` or the bare `example.com`, so every name in this module is put
 * through here before it is compared to anything. Non-strings fold to "" (which
 * matches nothing) rather than throwing — the input is an untrusted JSON blob. */
function dnsName(n: unknown): string {
  return typeof n === "string" ? n.toLowerCase().replace(/\.$/, "") : "";
}

/** How far a CNAME chain may be followed inside ONE response. A brand pointing
 * its apex through a CDN is one or two hops; ten is far past anything real and
 * bounds the fixpoint loop below independently of the byte cap. */
const MAX_CNAME_DEPTH = 10;

/**
 * Build a TXT lookup bound to a FIXED resolver URL (operator-configured, e.g.
 * https://dns.google/resolve or https://cloudflare-dns.com/dns-query). The
 * queried host is only ever a query PARAM — the fetch target is always the
 * resolver, never the brand domain — so there is no SSRF surface here. Returns
 * [] on any resolver error/timeout (fail-closed: absence never verifies).
 *
 * ---------------------------------------------------------------------------
 * THE RESPONSE IS UNTRUSTED INPUT, AND IT DECIDES WHO OWNS A BRAND
 * ---------------------------------------------------------------------------
 * This parser used to walk `json.Answer` and accept EVERY `type: 16` record it
 * found, without reading `a.name`, without correlating `Question`, and without
 * looking at `Status`. Three things followed, all measured:
 *
 *   - a response carrying `Status: 3` (NXDOMAIN — the name does not exist)
 *     still yielded a token, because nothing read the status;
 *   - a TXT record at a COMPLETELY UNRELATED name in the same Answer section
 *     was returned as if it sat on the queried host;
 *   - nothing checked that the answer even belonged to the question asked.
 *
 * The resolver is operator-configured and fetched over HTTPS, so this is not a
 * "the resolver is hostile" story — it is that a DoH Answer section legitimately
 * carries records the caller did not ask for (CNAME chains, additional records,
 * and whatever a misconfigured or compromised intermediary adds), and this
 * function's ONE output decides whether a passport is handed to a brand.
 * `verifyDomainClaim` does `records.includes(want)` and writes `claim_proven` on
 * a hit; nothing downstream re-checks provenance. So the correlation has to
 * happen here.
 *
 * The rule now: `Status` must be 0 (NOERROR); `Question[0].name` must be the
 * name we asked for; and a TXT record counts only if ITS OWN `name` is the
 * queried host or a name reached from it by following CNAMEs WITHIN this
 * response. The CNAME walk is a fixpoint rather than a single pass because the
 * records are not ordered, and it is rooted at the queried name — an alias chain
 * that starts somewhere else contributes nothing.
 */
/** Read at most `cap` bytes of a DoH response body as text; `null` when the
 * body is longer than `cap` — the caller then fails closed exactly as it does
 * for an oversized `content-length`. Streams via the body reader and cancels
 * as soon as the cap is crossed, so an untrusted/misbehaving resolver cannot
 * force this lookup to buffer an unbounded response (mirrors
 * evidence-scan.ts's `readCapped`, same "operator-configured, HTTPS, still
 * untrusted" reasoning as this file's own header comment). Falls back to
 * `res.text()` only when the runtime hands back a body-less Response (the
 * test doubles in this suite construct plain `Response` objects). */
async function readCappedDohBody(res: Response, cap: number): Promise<string | null> {
  const body = res.body;
  if (!body) {
    const t = await res.text();
    return t.length > cap ? null : t;
  }
  const reader = body.getReader();
  const chunks: Uint8Array[] = [];
  let total = 0;
  for (;;) {
    const { done, value } = await reader.read();
    if (done) break;
    if (value) {
      total += value.byteLength;
      if (total > cap) {
        await reader.cancel().catch(() => undefined);
        return null;
      }
      chunks.push(value);
    }
  }
  const out = new Uint8Array(total);
  let off = 0;
  for (const c of chunks) {
    out.set(c, off);
    off += c.byteLength;
  }
  return new TextDecoder().decode(out);
}

/**
 * I3-0660 (I4-13-R2 P5): the DoH query with "we did not get an answer" as `null`
 * rather than `[]`, so a caller that must tell UNCHECKED from ABSENT (the agency
 * re-proof sweep, `AgencyTxtLookup`) can. `[]` is returned only for an ANSWER:
 * NOERROR with no matching TXT, or NXDOMAIN (DECIDED: the name does not exist,
 * which is an authoritative absence). `dohTxtLookup` below turns `null` into a thrown
 * DomainClaimError 503 (`dns_lookup_failed`) for the claim/brand/contact callers (I16-RG-CLIENT:2).
 */
export function dohTxtAnswer(resolverUrl: string, fetchImpl: typeof fetch = fetch): (host: string) => Promise<string[] | null> {
  return async (host: string): Promise<string[] | null> => {
    const u = new URL(resolverUrl);
    u.searchParams.set("name", host);
    u.searchParams.set("type", "TXT");
    const ctl = new AbortController();
    const timer = setTimeout(() => ctl.abort(), DOH_TIMEOUT_MS);
    try {
      const res = await fetchImpl(u.toString(), { headers: { accept: "application/dns-json" }, redirect: "manual", signal: ctl.signal });
      // manual mode surfaces a 3xx as a response; checked BEFORE !res.ok and
      // treated as "we did not get an answer" like any other non-2xx.
      if (res.status >= 300 && res.status < 400) return null;
      if (!res.ok) return null;
      const declared = Number(res.headers.get("content-length"));
      if (Number.isFinite(declared) && declared > DOH_MAX_BYTES) return null;
      // BREAK[E-49]: `content-length` is advisory (absent on chunked transfer,
      // or simply wrong) — the declared-size check above is not a bound on
      // what actually arrives. This used to be `await res.arrayBuffer()`,
      // which drains the ENTIRE body into memory before the byteLength check
      // below could ever run, exactly the whole-body-first read
      // evidence-scan.ts's `readCapped` (MAX_SCAN_RESPONSE_BYTES) exists to
      // avoid for the same "operator-configured but untrusted responder"
      // shape. `readCappedDohBody` mirrors that: it stops pulling and cancels
      // the stream the moment the cap is crossed, so an oversized or
      // never-ending body cannot be buffered here at all.
      const text = await readCappedDohBody(res, DOH_MAX_BYTES);
      if (text === null) return null;
      const json = JSON.parse(text) as {
        Status?: unknown;
        Question?: { name?: unknown; type?: unknown }[];
        Answer?: { name?: unknown; type?: unknown; data?: unknown }[];
      };
      // NOERROR only verifies. NXDOMAIN (3) is an answer of absence; anything
      // else (SERVFAIL=2, REFUSED=5, or a missing/!numeric field) is "we did
      // not get an answer". Neither may ever verify.
      if (json.Status === 3) return []; // NXDOMAIN: an answer — the name does not exist
      if (json.Status !== 0) return null;
      const asked = dnsName(host);
      // The answer has to be to the question we asked. A response whose
      // Question section names something else is not this lookup's answer,
      // whatever its Answer section contains.
      const question = Array.isArray(json.Question) ? json.Question[0] : undefined;
      if (!question || dnsName(question.name) !== asked || !asked) return null;

      const answers = Array.isArray(json.Answer) ? json.Answer : [];
      // Names the queried host legitimately resolves THROUGH, inside this one
      // response. Seeded with the queried name; grown by following CNAME (type
      // 5) edges from a name already in the set. Fixpoint, because a resolver
      // may list the chain in any order; depth-bounded, so a response full of
      // cyclic or fan-out aliases cannot spin.
      const chain = new Set<string>([asked]);
      for (let depth = 0; depth < MAX_CNAME_DEPTH; depth++) {
        let grew = false;
        for (const a of answers) {
          if (a.type !== 5 || typeof a.data !== "string") continue; // 5 = CNAME
          const from = dnsName(a.name);
          const to = dnsName(a.data);
          if (!to || !chain.has(from) || chain.has(to)) continue;
          chain.add(to);
          grew = true;
        }
        if (!grew) break;
      }

      const out: TxtAnswer = [];
      const via: string[] = [];
      for (const a of answers) {
        if (a.type !== 16 || typeof a.data !== "string") continue; // 16 = TXT
        // The record must sit on the queried name or on the alias chain rooted
        // at it. A TXT at any other name is somebody else's record.
        if (!chain.has(dnsName(a.name))) continue;
        // TXT data comes quoted, and long values arrive as multiple quoted
        // chunks ("a" "b"); concatenate chunk contents, else use the raw string.
        const chunks = a.data.match(/"([^"]*)"/g);
        out.push(chunks ? chunks.map((s) => s.slice(1, -1)).join("") : a.data);
        via.push(dnsName(a.name));
      }
      // NON-enumerable: the answer stays an ordinary string[] to every reader that compares it (deep equality, JSON).
      Object.defineProperty(out, "via", { value: via, enumerable: false });
      return out;
    } catch {
      return null; // timeout / network / parse error → no answer
    } finally {
      clearTimeout(timer);
    }
  };
}

/** I3-0189 c: true when no record equal to `want` sits on a name inside one of `apexes` (the apex itself or a name
 * under it) — every matching TXT was found off-apex, through the CNAME chain. False with no `via` (see TxtAnswer). */
export function txtOwnerOutsideApex(records: string[], want: string, apexes: readonly string[]): boolean {
  const via = (records as TxtAnswer).via;
  if (via === undefined) return false;
  const inApex = (n: string | undefined) => n !== undefined && apexes.some((a) => n === a || n.endsWith(`.${a}`));
  return !records.some((r, i) => r === want && inApex(via[i]));
}

export function dohTxtLookup(resolverUrl: string, fetchImpl: typeof fetch = fetch): TxtLookup {
  const q = dohTxtAnswer(resolverUrl, fetchImpl);
  return async (host: string): Promise<string[]> => {
    const answer = await q(host);
    // I16-RG-CLIENT:2: no answer is not an absent record; the verify door answers 503, never record_absent.
    if (answer === null) throw new DomainClaimError("dns_lookup_failed", 503);
    return answer;
  };
}

/**
 * D-0932(2) / D-0933(2): the TXT lookup of a verify door that has ALREADY charged `challengeId` one attempt
 * (`chargeAttempt`). A lookup that itself FAILS — DoH error, timeout, no answer; never an absent record, which is
 * `[]` — refunds that attempt (floored at 0, this challenge's counter only, as `refundAttemptOnContention`) and
 * answers 503 `dns_lookup_failed`, so a resolver outage cannot spend the ten-attempt budget.
 */
export async function lookupTxtUncharged(db: DbLike, challengeId: string, lookupTxt: TxtLookup, host: string): Promise<string[]> {
  try {
    return await lookupTxt(host);
  } catch (e) {
    await db.prepare("UPDATE claim_challenges SET attempts = attempts - 1 WHERE challenge_id = ? AND status = 'issued' AND attempts > 0").bind(challengeId).run();
    throw e instanceof DomainClaimError ? e : new DomainClaimError("dns_lookup_failed", 503);
  }
}

function newToken(): string {
  // 16 CSPRNG chars over 32 unambiguous symbols = 80 bits — a DNS record isn't
  // human-typed under time pressure, so we use more entropy than the bio-code.
  const bytes = crypto.getRandomValues(new Uint8Array(16));
  const alpha = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";
  let s = "";
  for (const b of bytes) s += alpha[b % 32];
  return s;
}

interface DomainAccountRow {
  account_key: string;
  passport_id: string;
  id_kind: string;
  id: string;
  status: string;
  claim_state: string;
  retracted: number;
}

/** Begin a domain claim for an existing UNCLAIMED domain account on a passport.
 * Issues a token to publish as a TXT record.
 *
 * `caller` is the SAME masked-IP prefix the bio-code flow stores (`claims.ts`
 * `callerKey`), and it matters here for the same two reasons it mattered there:
 * the per-day cap is keyed per (account, caller) so a stranger cannot spend the
 * real domain owner's budget on their behalf, and `idx_claim_live`'s partial
 * UNIQUE over (account_key, caller) means a stranger's live challenge no longer
 * occupies THE single live slot for the domain — before this, one anonymous
 * `POST /claim/domain/start` (caller '-') held the only slot and answered 409
 * to the legitimate owner for as long as the attacker cared to renew it. */
export async function startDomainClaim(
  db: DbLike,
  passportId: string,
  accountKey: string,
  now: () => string = () => new Date().toISOString(),
  nowMs: number = Date.now(),
  caller: string | null = null,
): Promise<DomainStartResult> {
  const acct = await db
    .prepare(
      `SELECT a.account_key, a.passport_id, a.id_kind, a.id, a.status, p.claim_state, p.retracted
       FROM passport_accounts a JOIN passports p ON p.passport_id = a.passport_id
       WHERE a.account_key = ? AND a.passport_id = ?`,
    )
    .bind(accountKey, passportId)
    .first<DomainAccountRow>();
  if (await passportAlreadyClaimedPublicly(db, passportId)) throw new DomainClaimError("passport already claimed", 409);
  if (!acct || (await accountIsShielded(db, passportId, accountKey))) throw new DomainClaimError("no such account on that passport", 404);
  if (acct.id_kind !== "domain") throw new DomainClaimError("not a domain account", 409);
  if (acct.retracted === 1) throw new DomainClaimError("passport retracted", 409);
  // I3-1091 (D-1078(22)): a REVOKED domain on an unclaimed passport may start a
  // fresh proof for recovery — `verifyDomainClaim` sends a match to a person
  // (`domainRecoveryStart`), never to `claim_proven`. Every other non-active
  // status is still refused.
  if (acct.status !== "active" && acct.status !== "revoked") throw new DomainClaimError("account not active", 409);
  if (acct.claim_state !== "unclaimed") throw new DomainClaimError("passport already claimed", 409);
  // NO APEX REFUSAL HERE — docket #37 (d)(ii), closing FINDING-0014-C-7 and
  // moving INV-054 off `open` at this door.
  //
  // What was here: `if (!isRegistrableApex(acct.id)) throw 400 "a domain claim
  // must be a registrable apex"`. `registrable()` splits on a curated suffix set
  // that reads `co`/`ai`/`me` as generic 2-letter TLDs — correctly, for `id.me` —
  // which makes the REAL registration suffixes `com.co` (Colombia) and `com.ai`
  // (Anguilla) invisible, so the genuine owner of `brand.com.co` failed the
  // comparison and was refused her own account with a 400. INV-054: a heuristic
  // whose contract declared its own failure direction acceptable may not sit
  // where the proof IS the authorization (INV-017). Deciding `com.co` from
  // `id.me` needs public-suffix DATA the tree does not carry, and adding the two
  // names by hand fixes two domains and leaves the class — so the heuristic is
  // taken OUT of the authorization position rather than patched.
  //
  // What replaces it: the challenge is issued for the name exactly as given, and
  // the split is re-asked in `verifyDomainClaim` AFTER the TXT match, where a
  // heuristically-ambiguous name routes to `routeDomainClaimToReview` instead of
  // `claim_proven`. The door's predicate is now DNS control of the exact name,
  // which is exact; the heuristic only decides proven→review versus
  // proven→granted, a bounded over-referral to a human.
  //
  // AND THE DOOR IS NOT WEAKER. `blog.shared.com` — the shared-host case the
  // apex predicate was added for — now has to prove DNS control of that exact
  // name AND then wait for a person, where before it was refused and could
  // simply be re-requested at the apex. Nothing mints unattended that did not
  // mint unattended before. The cost is a person's attention on the ambiguous
  // names, which is the direction INV-054 requires.
  //
  // The 404/409 state gates above still come FIRST, and now nothing follows
  // them: there is no 400 here at all, so the INV-003 ordering question this
  // comment used to answer no longer arises at this door.

  return issueDomainChallenge(db, passportId, accountKey, acct.id, caller, now, nowMs, "domain_claim");
}

/**
 * S101-for-domains (H18): prove control of a DOMAIN account on an ALREADY-claimed
 * or verified passport. `startDomainClaim` refuses a claimed passport (its write
 * is `claim_proven`, which the state machine only permits unclaimed→claimed), and
 * `startAccountProof` refuses a domain (it proves via a bio code, not DNS) — so a
 * creator who claimed with one account and later wants to add and prove a domain
 * had no door at all: the domain sat unproven forever, never served, an amber
 * caution with no recovery. This is the missing door.
 *
 * Session-gated at the route (the caller's own passport id is passed as
 * `passportId`, so the account/passport join proves ownership). It requires the
 * passport to be CLAIMED (not unclaimed — that is `startDomainClaim`'s job) and,
 * like `startAccountProof`, refuses only when the proof is already CURRENT;
 * every staler standing is a reason to run the ceremony, not to refuse it.
 * Completion (`verifyDomainProof`) writes NO ledger event — the used challenge
 * row IS the proof, exactly as the bio-code re-proof path (`verifyAccountProof`).
 */
export async function startDomainProof(
  db: DbLike,
  passportId: string,
  accountKey: string,
  now: () => string = () => new Date().toISOString(),
  nowMs: number = Date.now(),
  caller: string | null = null,
): Promise<DomainStartResult> {
  const acct = await db
    .prepare(
      `SELECT a.account_key, a.passport_id, a.id_kind, a.id, a.status, p.claim_state, p.retracted
       FROM passport_accounts a JOIN passports p ON p.passport_id = a.passport_id
       WHERE a.account_key = ? AND a.passport_id = ?`,
    )
    .bind(accountKey, passportId)
    .first<DomainAccountRow>();
  if (!acct) throw new DomainClaimError("no such account on your passport", 404);
  if (acct.id_kind !== "domain") throw new DomainClaimError("not a domain account", 409);
  if (acct.retracted === 1) throw new DomainClaimError("passport retracted", 409);
  // BREAK[R-18]: this used to be an un-relaxed `if (acct.status !== "active")
  // throw`, with no `reproof` arm — the F5 "path back" claims.ts:507-524 built
  // for a revoked/historical HANDLE (`startAccountProof`'s `reproof`) was never
  // mirrored here, so a revoked/historical DOMAIN account had no self-serve way
  // back at all: `startAccountProof` refuses every domain outright ("domains
  // prove control via DNS, not a bio code"), and this door — the one DNS-proof
  // door a domain COULD use — still hard-refused non-active status. The DNS
  // ceremony is status-blind (verifyDomainProof writes only a claim_challenges
  // row, no re-listing), so re-proving a revoked/historical row is exactly as
  // safe here as it is for a handle.
  const reproof = acct.status === "revoked" || acct.status === "historical";
  if (acct.status !== "active" && !reproof) throw new DomainClaimError("account not active", 409);
  // The inverse of startDomainClaim: an unclaimed passport proves its domain
  // through the CLAIM ceremony (which also claims the passport); this door is for
  // adding a proven domain to a passport that is already the creator's.
  if (acct.claim_state === "unclaimed") throw new DomainClaimError("claim the passport first (POST /claim/domain/*), then prove additional domains here", 409);
  // Already-current proof → nothing to do (mirror startAccountProof). A stale or
  // never-taken proof falls through and re-runs the ceremony. A reproof of a
  // revoked/historical row is never "already proven" — same reasoning as
  // claims.ts's own reproof arm: a still-current proof from BEFORE the
  // revocation must not stand in for one dated after it.
  // S-070 MA-01 — THE ADVERTISED EARLY RENEWAL IS NOW ACTUALLY PERMITTED.
  //
  // The console does not merely allow renewal inside the 14-day warning window,
  // it SELLS it: `views.ts:1284` tells the creator "this proof runs out on
  // <date> … renewing now costs you nothing" and `views.ts:1438` puts a
  // Re-prove button (the DNS one at the same line) under it, and `views.ts:246` links there from the banner
  // with "Renewing takes one short code in that profile's bio. Renew it →".
  // Every one of those paths arrived here and got a flat 409: the predicate was
  // `proven_current`, full stop, so the ONLY moment a renewal could start was
  // after the proof had already lapsed — which is precisely the interruption
  // the warning exists to let the creator avoid. Between the warning and the
  // deadline the advice was unfollowable.
  //
  // ONE WINDOW, NOT A SECOND DEFINITION. The window is
  // `PROOF_EXPIRY_WARNING_MS` via `deriveStanding`'s `expiring_soon`, which is
  // the same quantity the console chip, the digest nag and the expiry cron all
  // read — so the button, the email and this door can never disagree about
  // whether a creator is inside it.
  //
  // NOTHING IS WEAKENED. `expiring_soon` is `now < expires_at <= now + 14d`, so
  // a proof with a long life left still gets the same refusal it always did; a
  // proof with no readable deadline is not "expiring soon" and is still
  // refused. Starting a challenge writes a `claim_challenges` row and touches
  // no existing proof, so the current proof stays current for its whole
  // remaining life whether or not the new ceremony is finished or abandoned —
  // which is the property that makes renewing early cost nothing.
  const proof = await accountProofDetail(db, passportId, accountKey, nowMs);
  if (proof.standing === "proven_current" && !proof.expiring_soon && !reproof) throw new DomainClaimError("this domain is already proven", 409);

  return issueDomainChallenge(db, passportId, accountKey, acct.id, caller, now, nowMs, "domain_proof");
}

/**
 * THE ONE SENTENCE every add-eligibility refusal wears.
 *
 * `NOT_CLAIMABLE`'s doctrine, applied to a second door. Naming WHICH predicate
 * failed is what puts a delisting bit on the wire, and this door is reachable by
 * anyone with a creator session and a domain they control — including somebody
 * who just bought a lapsed one. "This domain was delisted at its owner's
 * request" is exactly the sentence `removal.ts` spends the whole product's
 * decoy apparatus making unsayable, and "it belongs to another passport" is a
 * listedness probe wearing a helpful tone.
 *
 * So the four refusals — the key is already registered, the key was suppressed,
 * the domain resembles a name listed under another passport (`domain-gate.ts`,
 * which matches at ANY claim state on the domains side, not only verified), the
 * passport is not in a state that can hold an account — are ONE string. Whoever
 * needs the specific reason is an operator looking at the passport in the
 * console, which is authenticated.
 *
 * WHERE THEY ARE ASKED MOVED, and the uniformity now does a different job. All
 * four live in `verifyDomainAdd`, behind the DNS proof, so the reader is the
 * domain's own controller rather than a prober; the string stays uniform to keep
 * four doors from drifting into four sentences a future caller could tell apart.
 */
export const NOT_ADDABLE = "this domain cannot be added to this passport";

/**
 * SELF-SERVE SECOND DOMAIN — start (slice 32).
 *
 * A claimed creator proves DNS control of another domain and it is appended to
 * their own passport. This is the START half: it issues a TXT token and nothing
 * else.
 *
 * ---------------------------------------------------------------------------
 * WHY THIS DOOR ASKS NOTHING ABOUT THE DOMAIN
 * ---------------------------------------------------------------------------
 * The first cut screened eligibility here — already-registered, suppressed,
 * lookalike — behind one uniform `NOT_ADDABLE`, and an adversarial review
 * showed the uniform sentence was decoration. Whatever the wording, the door
 * answered 200-with-a-token or 409, and "why 409" is recoverable from public
 * data for every reason but one. Already-listed is public (the k-anon lookup
 * says so). Resembles-a-brand is computable offline from the same public data.
 * So a 409 that neither explains IS the delisting bit — the one fact
 * `suppressed_account_keys` exists to keep, the bit `nominations.ts` spends
 * three absorbing writes an outcome to protect. Constant work would have
 * equalised the latency and left the 200/409 split standing, because a start
 * door that screens MUST answer whether it will issue.
 *
 * So it does not screen. Every eligibility question moved to `verifyDomainAdd`,
 * BEHIND THE PROOF. The bit is still answered there — it has to be, the caller
 * is owed a yes or no — but only to somebody who has just published a TXT
 * record on the domain, i.e. to the domain's actual controller, who is the one
 * party the delisting was never a secret from. A prober who does not control
 * the name cannot reach the question at all.
 *
 * What is left here is the caller's OWN passport state (theirs to know), a
 * per-passport day cap taken BEFORE any read so the door cannot be used as a
 * cheap anything, and the shared token issuer. The cost is real and it is the
 * point: a typo now costs a DNS ceremony to discover instead of a round trip.
 *
 * (The residual: `issueDomainChallenge`'s UNIQUE index is per (account_key,
 * caller), so a caller can only ever collide with their own open ceremony. No
 * cross-caller activity bit. Known and unchanged from the claim door.)
 */
export async function startDomainAdd(
  db: DbLike,
  passportId: string,
  rawDomain: string,
  now: () => string = () => new Date().toISOString(),
  nowMs: number = Date.now(),
  caller: string | null = null,
): Promise<DomainStartResult> {
  // A malformed domain is a 400 and NOT part of the uniform refusal: it is a
  // fact about the caller's own input, carries no bit about anybody else, and
  // collapsing it into `NOT_ADDABLE` would make a typo unfixable. It also costs
  // zero statements, which is why it may safely precede the throttle.
  let accountKey: string;
  let domainId: string;
  try {
    const canon = canonicalAccount("domain", "domain", rawDomain);
    accountKey = canon.canonical;
    domainId = canon.id;
  } catch {
    throw new DomainClaimError("that is not a usable domain", 400);
  }

  // A SUBDOMAIN IS ACCEPTED, AND THAT IS A DECISION WITH A COST.
  //
  // The "only the controller can reach the answer" argument rests on TXT
  // publication implying control of the name, and for a SUBDOMAIN it is weaker
  // than for an apex: a dangling CNAME (`blog.victim.com` pointing at an
  // unclaimed SaaS host) or a wildcard delegation hands TXT authority over that
  // name to whoever claims the host, and `canonicalAccount` keeps subdomains as
  // their own account keys (adversarial review, MEDIUM).
  //
  // The obvious rule — apex only — was WRITTEN, MEASURED AND REMOVED, because
  // this module cannot tell an apex from a subdomain. `registrable()` splits on
  // a CURATED public-suffix set, not the PSL, so it reads `netflix.uk.com` as a
  // subdomain of `uk.com` and `alice.github.io` as a subdomain of `github.io`.
  // Enforcing apex-only would have refused every registrant of every private
  // registry the curated list does not know — the same class of error as the
  // `regDomain` comparison `domain-gate.ts` had to drop — in exchange for
  // closing one takeover path. A heuristic is not a licence to refuse people.
  //
  // What DOES hold: `domainAddGate`'s subdomain rule refuses a subdomain of any
  // host listed under somebody else's passport, in both directions, so the
  // takeover only reaches a victim who is not listed here at all — which is the
  // same exposure the existing `domain_claim` door already carries. Recorded as
  // a known gap in `MANIFEST.md`, with operator import as the road for a
  // subdomain that matters.

  // THROTTLE BEFORE ANY READ (`nominations.ts` discipline). Nothing below is a
  // secret, but a door that mints challenge rows and, downstream, outbound DoH
  // lookups is not one to leave unmetered for an authenticated caller either.
  // Keyed on the passport, not the session: a creator with five sessions gets
  // one budget.
  if (!(await consumeRateLimit(db, `domadd:${passportId}`, nowMs, DOMAIN_ADD_CAP_PER_DAY))) {
    throw new DomainClaimError("too many domain attempts today", 429);
  }

  // The caller's own passport, and nothing else — the session owner's own state,
  // so it leaks nothing. `subjectMayAddAccounts` is the GATE'S OWN STRING
  // (`ACCOUNTABLE_SUBJECT_SQL`), not a hand-copy of it: see its docblock for the
  // clause every hand-copy in this repo has lost so far.
  if (!(await subjectMayAddAccounts(db, passportId, nowMs))) throw new DomainClaimError(NOT_ADDABLE, 409);

  return issueDomainChallenge(db, passportId, accountKey, domainId, caller, now, nowMs, "domain_add", passportId);
}

/**
 * The shared token-issue tail of all three domain-start doors — kept in ONE place so
 * the claim door and the proof door cannot drift on the rate cap or the
 * TOCTOU-free reap-then-insert (the same "two doors into claim_challenges must
 * not drift" discipline `claims.ts` enforces for the bio-code side).
 *
 * Sharing an issuer is also how a credential ends up spendable at a door it was
 * never meant for, so `purpose` is a REQUIRED parameter with no default (the
 * same rule `claims.ts issueChallenge` follows): `startDomainClaim` mints
 * `domain_claim`, `startDomainProof` mints `domain_proof`, `startDomainAdd`
 * mints `domain_add`, and each verify door below spends exactly one of them.
 */
export async function issueDomainChallenge(
  db: DbLike,
  /** NULL for `brand_enrol` (the one ceremony whose subject does not exist yet)
   * and for `agency_domain` (whose subject is an AGENCY, passed below). Migration
   * 0072 widened 0055's CHECK to name exactly those two purposes and to require
   * exactly one subject, so a door that passes null for any other reason — or
   * that passes BOTH — is refused by the schema rather than by a convention. */
  passportId: string | null,
  accountKey: string,
  domainHandle: string,
  caller: string | null,
  now: () => string,
  nowMs: number,
  purpose: ChallengePurpose,
  /** SLOT SCOPE, for the ONE door where the caller is authenticated.
   *
   * The live-challenge slot is `UNIQUE(account_key, caller) WHERE status =
   * 'issued'` (migration 0003) and `caller` is a masked IP prefix — or, when
   * `CF-Connecting-IP` is absent, `callerKey`'s SHARED `'-'` sentinel. On the
   * anonymous claim door that is the best available identity and its docblock
   * says so. On the SELF-SERVE ADD door it was wrong twice over: two creators
   * behind one NAT (or a deployment with no client-IP header at all) shared one
   * slot and one 5/day budget, so `409 a claim is already in progress for this
   * domain` versus `200 + token` varied with A STRANGER'S activity on a named
   * domain — and unlike the other two doors, this one accepts ANY syntactically
   * valid apex, so that slot could be squatted against arbitrary third-party
   * names (adversarial review, MEDIUM). Passing the passport id here scopes the
   * slot to the authenticated principal, which is the identity this door
   * actually has. */
  slotScope: string | null = null,
  /** THE AGENCY SUBJECT, non-null for exactly one purpose: `agency_domain`.
   *
   * It is a separate parameter rather than a reuse of `passportId` because the
   * two subjects are different KINDS of principal and the schema says so: 0072's
   * widened CHECK admits `(passport_id, NULL)`, `(NULL, agency_id)` for
   * `agency_domain`, or `(NULL, NULL)` for `brand_enrol`, and nothing else. A
   * caller that forgets it on the agency door mints a row the CHECK refuses,
   * which is the intended failure — a proof that cannot name its own subject
   * must not exist, because the whole point of the agency proof is that the
   * apex it unlocks is attributable to ONE agency. */
  agencyId: string | null = null,
  /** I13-AUTHFIX (#896, I12-FX7 D-0884(5)): a predicate over the ACTING principal,
   * evaluated INSIDE the INSERT (`INSERT ... SELECT ... WHERE guard`). When it is
   * false no row is written — so no live slot is taken and no token exists — and the
   * caller gets 403. Replaces a compensating UPDATE issued after the INSERT. */
  actorGuard: { sql: string; params: readonly (string | number)[] } | null = null,
  /** FXB-I16-W03D-12-start (I16-W13S H5): statements that commit IN the reap-then-insert batch, after the INSERT
   * (the agency start door's seat audit row), so a fault in one rolls the challenge back. A statement that must not
   * write when an `actorGuard` refuses the INSERT carries its own `WHERE changes() = 1` (seatAuditStatement's afterWrite). */
  extra: readonly DbStatement[] = [],
): Promise<DomainStartResult> {
  // Per-(account, caller) rate cap (bounds outbound DoH lookups + challenge
  // rows) — mirrors `claims.ts issueChallenge`, for the same reason: an
  // account-only cap is a budget a stranger spends on the victim's behalf.
  const key = slotScope ?? callerKey(caller);
  const windowStart = new Date(nowMs - DOMAIN_CHALLENGE_WINDOW_MS).toISOString();
  const recent = await db.prepare("SELECT COUNT(*) AS n FROM claim_challenges WHERE account_key = ? AND caller = ? AND created_at >= ?").bind(accountKey, key, windowStart).first<{ n: number }>();
  if ((recent?.n ?? 0) >= MAX_DOMAIN_CHALLENGES_PER_DAY) throw new DomainClaimError("too many claim attempts for this domain today", 409);

  const token = newToken();
  const id = mintId("chg");
  const ts = now();
  const expires = new Date(nowMs + DOMAIN_CODE_TTL_MS).toISOString();
  try {
    // Reap-then-insert in ONE atomic batch, exactly as the bio-code path does:
    // an expired-but-'issued' row must not hold the live slot past its TTL, and
    // doing the sweep in the same batch as the insert that needs the slot makes
    // it TOCTOU-free (see `claims.ts reapExpired`).
    const cols = "INSERT INTO claim_challenges (challenge_id, passport_id, agency_id, account_key, caller, platform_key, handle, code, status, created_at, expires_at, purpose)";
    const binds = [id, passportId, agencyId, accountKey, key, domainHandle, token, ts, expires, purpose];
    const res = await db.batch([
      reapExpired(db, accountKey, new Date(nowMs).toISOString()),
      actorGuard === null
        ? db
            // `agency_id` rides beside `passport_id`, and the table's CHECK (0072)
            // admits exactly one of them per purpose — so a passport-less challenge
            // can only exist for the two purposes that have another subject, and an
            // agency proof cannot be minted wearing a passport it does not have.
            .prepare(`${cols} VALUES (?, ?, ?, ?, ?, 'domain', ?, ?, 'issued', ?, ?, ?)`)
            .bind(...binds)
        : // RETURNING <pk> counted with `results.length`: a WHERE that selects no row
          // does not throw, and a batch element carries no `changes` (db.ts:64-85).
          db.prepare(`${cols} SELECT ?, ?, ?, ?, ?, 'domain', ?, ?, 'issued', ?, ?, ? WHERE ${actorGuard.sql} RETURNING challenge_id`).bind(...binds, ...actorGuard.params),
      ...extra,
    ]);
    if (actorGuard !== null && ((res[1] as { results?: unknown[] } | undefined)?.results ?? []).length !== 1) {
      throw new DomainClaimError("acting seat is no longer authorized", 403 as 400);
    }
  } catch (e) {
    // MATCHED ON THE EXACT INDEX, not on "constraint". `idx_claim_live` is what
    // means "somebody already has a live ceremony here"; migration 0055 added a
    // CHECK to this table, and the broad pattern turned a schema-level
    // programming error (a NULL passport_id for a purpose that may not have one)
    // into a domain answer about a stranger's activity (adversarial review, LOW).
    // A CHECK failure is a bug and must surface as one.
    if (e instanceof Error && /UNIQUE constraint failed:\s*claim_challenges/i.test(e.message)) {
      throw new DomainClaimError("a claim is already in progress for this domain", 409);
    }
    throw e;
  }
  return { challenge_id: id, host: domainHandle, record_name: domainHandle, record_value: `${TXT_PREFIX}${token}`, expires_at: expires };
}

export interface DomainVerifyResult {
  ok: boolean;
  outcome: "verified" | "record_absent" | "expired" | "unreachable" | "pending_review";
  passport_id?: string;
  /** Set ONLY on the `pending_review` outcome (S-007 X3(b)): the proof matched
   * but continuity did not, so a person decides. The caller polls this the way
   * a manual claimant polls theirs — `state` is the manual-claim poll vocabulary
   * so one client arm serves both. */
  state?: "pending_review";
  claim_ref?: string;
  event_id?: string;
  /** M33: how many verify attempts remain on THIS challenge after this one. Set
   * on the `record_absent` path so the creator is told they have a bounded
   * budget (and can fix the TXT record / wait for propagation) instead of
   * silently burning all ten tries and hitting a dead "too many attempts". */
  attempts_remaining?: number;
}

interface DomainChallengeRow {
  challenge_id: string;
  passport_id: string;
  account_key: string;
  platform_key: string;
  handle: string;
  code: string;
  status: string;
  attempts: number;
  created_at: string;
  expires_at: string;
  /** Migration 0027. Open `string` because it comes back from D1 unchecked; the
   * doors below compare it to a `ChallengePurpose`, which is the check. */
  purpose: string;
}

/** What the DoH lookup is allowed to see BEFORE it is spent — the attempt debit
 * comes first, so a caller cannot dodge it by aborting mid-flight, and the M33
 * "N tries left" hint is read back from the row AFTER the debit rather than
 * computed from a pre-read count (which a concurrent verify makes a lie). Only
 * the `record_absent` branch needs it, so only that branch pays for the read. */
async function attemptsRemainingAfterCharge(db: DbLike, challengeId: string): Promise<number> {
  const row = await db.prepare("SELECT attempts FROM claim_challenges WHERE challenge_id = ?").bind(challengeId).first<{ attempts: number }>();
  return Math.max(0, MAX_DOMAIN_VERIFY_ATTEMPTS - (row?.attempts ?? MAX_DOMAIN_VERIFY_ATTEMPTS));
}

/** The refusal shape both domain doors raise from the shared attempt debit, so
 * the route keeps catching `DomainClaimError` and answering 409 rather than
 * falling through to a generic 400. */
const domainFail = (message: string): Error => new DomainClaimError(message, 409);

/**
 * CONTINUITY (S-007 X3(b)) — may this proof mint on its own, or must a person
 * look at it?
 *
 * TRUE when the passport is NOT operator-seeded (no `seed_metadata` row),
 * whatever the account's age — I62 P3: D-1518 I61-Q1 "build wait-for-review",
 * ANSWERS-2026-11-04 #643 "non-seeded claims wait for review" (D-1521, D-1522:
 * the add door too) — and when it IS seeded and the account's `account_added`
 * is older, IN LEDGER TIME, than `DOMAIN_CLAIM_CONTINUITY_DAYS`. Only a seeded
 * passport whose domain account we wrote down recently keeps the direct path.
 *
 * THE CLOCK IS THE LEDGER'S, NOT THE PAYLOAD'S. `passport_accounts.observed_at`
 * and the `account_added` payload's `observed_at` are both CALLER-SUPPLIED
 * values — an importer states them — so a continuity window measured against
 * either is a window the party being screened gets to set. `trust_events
 * .created_at` is written by `appendEvents` at append time and is the only
 * timestamp here that is ours.
 *
 * POPULATION (FAMILY-004 (a)): the "operator-seeded" population is enumerated
 * from `seed_metadata` itself rather than from any list of seeding routes —
 * `seeding.ts:146` is the table's ONLY writer in the tree (grepped across
 * `services/**` and `packages/**`; `console.ts:2150` is a comment about the
 * table, not an INSERT), so the table IS the population and a future second
 * writer is covered without this predicate being touched.
 *
 * FAIL-CLOSED: seeded, but no datable `account_added` for the account → review.
 * The alternative is minting on an account whose age we cannot establish, which
 * is the exact case the rule exists for.
 */
export async function domainClaimNeedsReview(db: DbLike, passportId: string, accountId: string, nowMs: number): Promise<boolean> {
  // I3-0007 (D-0979, D-0981(9)): the claim doors' one hold, asked first.
  if (await seededOrUnpinnedHold(db, passportId)) return true;
  return domainContinuityNeedsReview(db, passportId, accountId, nowMs);
}

/** Q537: the CONTINUITY limb alone (unseeded passport, or seeded with the account not datable inside the window) — the one predicate
 * `domainClaimNeedsReview` (claim door) and `verifyDomainAdd` (add door) both ask. The add door does not ask the
 * I3-0007 hold, which D-0979 scopes to the self-serve CLAIM doors. */
export async function domainContinuityNeedsReview(db: DbLike, passportId: string, accountId: string, nowMs: number): Promise<boolean> {
  // ONE round trip: the seed marker and the ledger age are asked together, so
  // there is no window between them for either to change under the other.
  // The account is matched with `events.ts`'s own idiom (`:1497`):
  // `lower(json_extract(payload,'$.id'))`, scoped by subject and id_kind.
  const row = await db
    .prepare(
      `SELECT EXISTS (SELECT 1 FROM seed_metadata m WHERE m.passport_id = ?) AS seeded,
              (SELECT MIN(e.created_at) FROM trust_events e
                WHERE e.event_type = 'account_added' AND e.subject = ?
                  AND json_extract(e.payload, '$.id_kind') = 'domain'
                  AND lower(json_extract(e.payload, '$.id')) = lower(?)) AS added_at`,
    )
    .bind(passportId, passportId, accountId)
    .first<{ seeded: number; added_at: string | null }>();
  if (!row || !row.seeded) return true; // I62 P3 (#643, D-1521, D-1522): unseeded -> review, claim and add door, any age
  const addedAt = row.added_at ? Date.parse(row.added_at) : Number.NaN;
  if (!Number.isFinite(addedAt)) return true;
  return nowMs - addedAt > DOMAIN_CLAIM_CONTINUITY_DAYS * 24 * 60 * 60 * 1000;
}

/**
 * The review-routed exit of `verifyDomainClaim`. The TXT record MATCHED — this
 * caller controls the domain today — but continuity did not hold, so no
 * `claim_proven` event is written and no session can be minted from this
 * result: the passport stays `unclaimed` until a person approves the item.
 *
 * ONE-SHOT, like the minting exit. The challenge is consumed in the same batch
 * that enqueues, and the enqueue is GUARDED BY THE CONSUMPTION (`WHERE EXISTS`
 * over the row this batch just moved) rather than by a prior read — otherwise
 * K concurrent verifies of one live challenge would each spend a DoH lookup and
 * each enqueue, and what that buys is an operator's attention, the scarcest
 * per-victim resource in the product (`manual-claims.ts:154-176` is the same
 * argument at the sibling door, and this is its idiom, audit line included).
 * Admission is read back BY PRIMARY KEY because a D1 batch carries no portable
 * rowcount. A verify that lost the race to a verify that spent the SAME
 * challenge into review is answered `pending_review` with no `claim_ref`
 * (I63-P3 R4, D-1538 ruling 1); any other unadmitted enqueue answers
 * `challenge is not open`, the refusal this door gives a spent challenge.
 *
 * The item wears `manual-claims.ts:140-152`'s shape (`claim_ref_hash`, the
 * plaintext ref returned exactly once to the caller) with `method` =
 * `domain_review`, so the manual-claim poll arm — S7-1's consumption and epoch
 * rules included — applies to it unchanged.
 */
async function routeDomainClaimToReview(
  db: DbLike,
  chg: DomainChallengeRow,
  now: () => string,
  // RJ550b-W3-15-R7-1 / D-0921(4) (I16-W05B): the ADD door routes here too. `source` tells the approver which door
  // filed the item — an add item is a domain for an already-claimed passport, never a `claim_proven`.
  source: "domain_claim" | "domain_add" = "domain_claim",
  // I63-P3 R2 (D-1535): no roster rider here any more. The `domain_add:under_agency_apex` row is written by review.ts
  // `approveItem` in the approve batch, so a rejected add leaves none.
): Promise<DomainVerifyResult> {
  const ts = now();
  const claimRef = crypto.randomUUID().replace(/-/g, "") + crypto.randomUUID().replace(/-/g, "");
  const claimRefHash = await claimRefHashOf(claimRef);
  const itemId = mintId("itm");
  const payloadJson = JSON.stringify({
    source,
    method: "domain_review",
    platform: "domain",
    handle: chg.handle,
    account_key: chg.account_key,
    claim_ref_hash: claimRefHash,
  });
  await db.batch([
    // I26-W1-MONEY (I3-0696 sibling, I22 RJ-02): the spend is an UPSERT, as `verifyDomainProof` and the agency door
    // write theirs — a guarded `status = 'issued'` UPDATE lost a TXT record that MATCHED while the challenge was live
    // whenever hygiene flipped the row to `expired` (or deleted it) during the lookup. A row already `used` (a
    // concurrent verify won) is left alone, so `proved_at = ts` still binds the enqueue to THIS spend: one-shot holds.
    db
      .prepare(
        `INSERT INTO claim_challenges (challenge_id, passport_id, account_key, caller, platform_key, handle, code, status, attempts, created_at, expires_at, purpose, proved_at, method)
         VALUES (?, ?, ?, '-', 'domain', ?, ?, 'used', ?, ?, ?, ?, ?, 'domain_dns')
         ON CONFLICT(challenge_id) DO UPDATE SET status = 'used', proved_at = excluded.proved_at, method = excluded.method WHERE claim_challenges.status <> 'used'`,
      )
      .bind(chg.challenge_id, chg.passport_id, chg.account_key, chg.handle, chg.code, chg.attempts + 1, chg.created_at, chg.expires_at, chg.purpose, ts),
    db
      .prepare(
        `INSERT INTO review_queue (item_id, item_type, state, subject, payload, created_at, updated_at, import_batch_id)
         SELECT ?, 'claim', 'open', ?, ?, ?, ?, NULL
          WHERE EXISTS (SELECT 1 FROM claim_challenges c WHERE c.challenge_id = ? AND c.status = 'used' AND c.proved_at = ?)`,
      )
      .bind(itemId, chg.passport_id, payloadJson, ts, ts, chg.challenge_id, ts),
    // Guarded by the ITEM's own existence, exactly as `manual-claims.ts:203-208`
    // guards its own: a refused enqueue must leave no audit line saying it was
    // queued.
    db
      .prepare(
        `INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind)
         SELECT 'system', 'enqueue:claim', ?, ?, ?, 'system' WHERE EXISTS (SELECT 1 FROM review_queue q WHERE q.item_id = ?)`,
      )
      .bind(itemId, ts, itemId, itemId), // I34-W3-SEAM-P (D-1119(8), D-1120(6)): the review-queue audit copy keeps only the item id
  ]);
  const mine = await db.prepare("SELECT 1 AS x FROM review_queue WHERE item_id = ?").bind(itemId).first();
  if (!mine) {
    // I63-P3 R4 (D-1538 ruling 1, D-1539 (d), I63-W0 amendment 6): a verify that lost the race for THIS challenge to a
    // verify that spent it into review gets the winner's answer, without a claim_ref (the winner's ref was returned
    // once and is stored hashed; none was stored for this call). The winner's item is the one on this passport, for
    // this host and door, created at the instant the challenge was spent. Anything else stays the 409.
    const raced = await db
      .prepare(
        `SELECT 1 AS x FROM claim_challenges c JOIN review_queue q ON q.subject = c.passport_id AND q.created_at = c.proved_at
          WHERE c.challenge_id = ? AND json_extract(q.payload, '$.handle') = c.handle AND json_extract(q.payload, '$.source') = ? LIMIT 1`,
      )
      .bind(chg.challenge_id, source)
      .first();
    if (raced) return { ok: false, outcome: "pending_review", state: "pending_review", passport_id: chg.passport_id };
    throw new DomainClaimError("challenge is not open", 409);
  }
  return { ok: false, outcome: "pending_review", state: "pending_review", claim_ref: claimRef, passport_id: chg.passport_id };
}

/**
 * I3-1091 (D-1078(22)): is this challenge a RECOVERY start — a fresh domain proof
 * for an unclaimed passport whose domain was revoked? True iff the account still
 * hangs off THIS passport and is `revoked`, the passport is unclaimed, unretracted
 * and accountable (`stillClaimable`'s clauses with `revoked` for `active`), the
 * key is not suppressed, and the latest revoking event is OLDER than the
 * challenge — a revocation that lands mid-challenge is an operator remediation,
 * not a recovery, and still refuses. The spent challenge row (`used`,
 * `proved_at`) is the fresh control proof the operator's
 * `account_reactivated{control_reproven}` reads (events.ts `freshControlProof`).
 */
async function domainRecoveryStart(db: DbLike, chg: DomainChallengeRow, nowMs: number): Promise<boolean> {
  const live = await db
    .prepare(
      `SELECT 1 AS x FROM passport_accounts a JOIN passports ON passports.passport_id = a.passport_id
        WHERE a.account_key = ? AND a.passport_id = ? AND a.status = 'revoked'
          AND passports.claim_state = 'unclaimed' AND passports.retracted = 0
          AND (${accountableSubjectSqlAt(nowMs)})`,
    )
    .bind(chg.account_key, chg.passport_id)
    .first<{ x: number }>();
  const suppressed = await suppressedKeyStmt(db, chg.account_key).first<{ x: number }>();
  const latest = await latestRevokingEvent(db, chg.account_key);
  return live !== null && suppressed === null && latest !== null && latest.created_at < chg.created_at;
}

/** I46-FX1 SCB-001 (QUEUE #1017, W0-14): THE CHALLENGE-STATUS REFUSALS `verifyDomainClaim` ASKS FIRST, exported so
 * /claim/domain/verify (claim-routes.ts) asks them BEFORE the age check: unknown or foreign-purpose throws
 * `no such challenge` 404, a non-domain channel `not a domain challenge` 409, a spent one `challenge is not open`
 * 409, and an expired one is flipped and returned as `expired`. Nothing is charged here. */
export async function openDomainClaimChallenge(db: DbLike, challengeId: string, nowMs: number): Promise<{ chg: DomainChallengeRow; expired: DomainVerifyResult | null }> {
  const chg = await db.prepare("SELECT * FROM claim_challenges WHERE challenge_id = ?").bind(challengeId).first<DomainChallengeRow>();
  if (!chg) throw new DomainClaimError("no such challenge", 404);
  // Purpose FIRST, and byte-identical to "no such challenge" (migration 0027):
  // a credential minted at any other door — the anonymous `/removal/start`, a
  // bio-code claim, the H18 domain re-proof — is not something this door has
  // ever heard of, and a distinguishable refusal would tell an anonymous caller
  // which challenge ids are real. The `platform_key` test below is now
  // unreachable (only `issueDomainChallenge` writes 'domain_claim', and it
  // hardcodes `platform_key = 'domain'`) and is kept as the channel half of the
  // same binding.
  if (chg.purpose !== "domain_claim") throw new DomainClaimError("no such challenge", 404);
  if (chg.platform_key !== "domain") throw new DomainClaimError("not a domain challenge", 409);
  if (chg.status !== "issued") throw new DomainClaimError("challenge is not open", 409);
  if (Date.parse(chg.expires_at) < nowMs) {
    await db.prepare("UPDATE claim_challenges SET status = 'expired' WHERE challenge_id = ? AND status = 'issued'").bind(challengeId).run();
    return { chg, expired: { ok: false, outcome: "expired" } };
  }
  return { chg, expired: null };
}

/** Verify a domain claim: look up the apex TXT records and match the issued
 * token. On a match, write claim_proven(method='domain_dns') atomically; a
 * recovery start (`domainRecoveryStart`) goes to a person instead. */
export async function verifyDomainClaim(
  db: DbLike,
  challengeId: string,
  lookupTxt: TxtLookup,
  now: () => string = () => new Date().toISOString(),
  nowMs: number = Date.now(),
  /** QUEUE #19: `publishHook(...)`'s thunk, run once after the batch commits. */
  afterCommit?: () => void,
): Promise<DomainVerifyResult> {
  const open = await openDomainClaimChallenge(db, challengeId, nowMs);
  if (open.expired) return open.expired;
  const chg = open.chg;
  // THE CAP IS DECIDED IN THE STATEMENT (claims.ts `chargeAttempt`), not by the
  // `chg.attempts` read above. This was `SELECT` → compare → an UNGUARDED
  // `UPDATE attempts = attempts + 1`, and five verifies held on that read with
  // one attempt left measured 5 outbound DoH lookups, `attempts = 14` against a
  // cap of 10, and the challenge still `issued` — so the next honest verify
  // flipped a LIVE challenge to `failed`.
  await chargeAttempt(db, challengeId, MAX_DOMAIN_VERIFY_ATTEMPTS, domainFail);

  const want = `${TXT_PREFIX}${chg.code}`;
  const records = await lookupTxtUncharged(db, challengeId, lookupTxt, chg.handle);
  // M33: what the creator has left AFTER this attempt, so a `record_absent`
  // answer can say "N tries left" rather than silently draining the budget.
  if (!records.includes(want)) return { ok: false, outcome: "record_absent", attempts_remaining: await attemptsRemainingAfterCharge(db, challengeId) };

  // THE GATE, RE-ASKED — `claims.ts stillClaimable` has the three measured
  // shapes and the reason it is asked only after the lookup. This door needs it
  // MORE than the bio-code one, not less: `DOMAIN_CODE_TTL_MS` is twenty-four
  // hours (DNS propagation is slow), so `startDomainClaim`'s gate is a day stale
  // by the time a brand comes back to verify.
  // I3-1091 (D-1078(22)): a RECOVERY start (`domainRecoveryStart`) is not
  // claimable — its account is revoked — and goes to a person instead: the proof
  // row is spent with `proved_at` and a review item is queued, no `claim_proven`.
  // Asked on every verify, ahead of the gate, so the statement sequence does not
  // depend on which refusal applies.
  const recovery = await domainRecoveryStart(db, chg, nowMs);
  if (!(await stillClaimable(db, chg.passport_id, chg.account_key, nowMs))) {
    if (recovery) return routeDomainClaimToReview(db, chg, now);
    throw new DomainClaimError(NOT_CLAIMABLE, 409);
  }

  // CONTINUITY, asked AFTER the proof and BEFORE the write. After, because the
  // answer is only owed to somebody who has just proved DNS control (a start-door
  // check would be a "we seeded this passport" oracle to anyone who can name it).
  // Before, because the whole point is that no `claim_proven` is written.
  //
  // THE APEX SPLIT IS ASKED HERE TOO, and this is the whole of docket #37
  // (d)(ii): `startDomainClaim` no longer refuses a name the curated suffix set
  // reads as a subdomain, so the question arrives at the one place it can be
  // asked without costing an owner her account — after she has proved DNS
  // control of the exact name. A false here sends a PROVEN claimant to a human,
  // never away. PURE FIRST: it decides on the name alone, so the seeded/ledger
  // round trip below is not spent when it already answers.
  if (!isRegistrableApex(chg.handle) || (await domainClaimNeedsReview(db, chg.passport_id, chg.handle, nowMs))) {
    return routeDomainClaimToReview(db, chg, now);
  }

  // Proven. Same atomic single-claim write as the bio-code path — and the same
  // door: `tryAppendEvents`, whose `LedgerRefusal` cannot express a lost head
  // race, so the old `/UNIQUE|constraint/i → record_absent` translation (which
  // told a brand whose TXT record WAS published that it was not) is unwritable.
  // A contention loss propagates and refunds the attempt it was charged.
  const ts = now();
  let r: Awaited<ReturnType<typeof tryAppendEvents>>;
  try {
    r = await refundAttemptOnContention(db, challengeId, () =>
      tryAppendEvents(
        db,
        [{ event_type: "claim_proven", subject: chg.passport_id, actor_kind: "system", actor_id: challengeId.slice(0, 128), payload: { method: "domain_dns" } }],
        now,
        [],
        [
          // One-shot lock AND eligibility gate in one statement — `claims.ts
          // claimConsumedGateStmt` states what rides the batch and what cannot.
          claimConsumedGateStmt(db, challengeId, chg.passport_id, chg.account_key, ts),
          // THE PROOF ROW MUST LAND WHENEVER THE EVENT DOES, and on THIS door
          // the cost of it not landing is the worst in the product: `id_kind =
          // 'domain'` sets `needsOwnProof` in `servingClassFor`, so a domain
          // with no proof row is `not_served` — a successful DNS claim silently
          // dropped the brand's listing out of the registry altogether. The
          // guarded `UPDATE … AND status = 'issued'` this replaces wrote nothing
          // whenever the cron or a concurrent verify had moved the row, and a
          // zero-row statement does not fail its batch. `claims.ts verifyClaim`
          // carries the full argument for the UPSERT, including why dropping the
          // status clause alone cannot cover the row the cron DELETED.
          //
          // The proof gets a DATE and a NAME (migration 0014). `method =
          // 'domain_dns'`, never 'bio_code': `claim_challenges` carries both
          // ceremonies and labelling a DNS proof as a bio code would put a false
          // statement in an audit column. `proved_at = ts` is the same instant the
          // `claim_proven` event above is stamped with. `valid_until` stays NULL —
          // a DNS TXT record has no deadline of its own, so the default TTL in
          // `account-standing.ts` is the honest policy and stays revisable in code
          // (the argument is written out in full at `claims.ts verifyAccountProof`).
          db
            .prepare(
              `INSERT INTO claim_challenges (challenge_id, passport_id, account_key, caller, platform_key, handle, code, status, attempts, created_at, expires_at, purpose, result_event, proved_at, method)
               VALUES (?, ?, ?, '-', 'domain', ?, ?, 'used', ?, ?, ?, 'domain_claim', (SELECT event_id FROM trust_events WHERE subject = ? AND event_type = 'claim_proven' ORDER BY seq DESC LIMIT 1), ?, 'domain_dns')
               ON CONFLICT(challenge_id) DO UPDATE SET status = 'used', result_event = excluded.result_event, proved_at = excluded.proved_at, method = excluded.method`,
            )
            // I4-07-R3 WO-D: `result_event` is bound IN THIS BATCH (the sub-select
            // reads the `claim_proven` row appended before extraStatements), so no
            // '__pending__' is committed and no second write backfills it.
            .bind(challengeId, chg.passport_id, chg.account_key, chg.handle, chg.code, chg.attempts + 1, chg.created_at, chg.expires_at, chg.passport_id, ts),
        ],
        ["system"],
        afterCommit,
      ),
    );
  } catch (e) {
    // The batch gate refused — the domain account was revoked or moved off this
    // passport between the read and the commit. Same sentence, same status.
    if (isUniformClaimRefusal(e)) throw new DomainClaimError(NOT_CLAIMABLE, 409);
    throw e;
  }
  // ONE refusal is "a prior verify of this same proof already claimed the
  // passport" — "requires unclaimed", or the lost claim_consumed PK. Any other
  // ledger fault is rethrown unchanged rather than reported as a missing TXT
  // record. Neither branch can be contention.
  if (!r.ok) {
    const alreadyClaimed = r.refusal.kind === "conflict" || /requires unclaimed/.test(r.refusal.message);
    if (!alreadyClaimed) throw r.refusal.error;
    // I4-04-R8 (sibling of I3-2061, `claims.ts verifyClaim`): the TXT record WAS
    // present, so this is never `record_absent`. A `claim_consumed` row for THIS
    // challenge exists only if a concurrent verify of this same proof committed,
    // so the claim succeeded: `verified`. No row: a different door claimed the
    // passport in the window — the door's own not-claimable refusal.
    const won = await db
      .prepare("SELECT result_event FROM claim_challenges WHERE challenge_id = ? AND status = 'used' AND EXISTS (SELECT 1 FROM claim_consumed WHERE challenge_id = ?)")
      .bind(challengeId, challengeId)
      .first<{ result_event: string | null }>();
    if (!won) throw new DomainClaimError(NOT_CLAIMABLE, 409);
    return { ok: true, outcome: "verified", passport_id: chg.passport_id, ...(won.result_event ? { event_id: won.result_event } : {}) };
  }
  const event = r.events[0];
  return { ok: true, outcome: "verified", passport_id: chg.passport_id, event_id: event!.event_id };
}

/**
 * Verify a domain PROOF (H18) — the completion for `startDomainProof`. Same TXT
 * lookup and match as `verifyDomainClaim`, but for a passport that is ALREADY
 * claimed: it writes NO `claim_proven` event and consumes NO one-shot claim.
 * The `status='used'` challenge row with `proved_at` set IS the proof — exactly
 * `verifyAccountProof`'s model — and `accountProofStanding` reads it to serve
 * the domain in the registry and shield it from reports.
 *
 * `passportId` is the caller's session passport: a challenge that is not on it
 * reads as "no such challenge" (404, indistinguishable) so this route cannot be
 * used to complete another passport's ceremony.
 */
export async function verifyDomainProof(
  db: DbLike,
  passportId: string,
  challengeId: string,
  lookupTxt: TxtLookup,
  now: () => string = () => new Date().toISOString(),
  nowMs: number = Date.now(),
): Promise<DomainVerifyResult> {
  const chg = await db.prepare("SELECT * FROM claim_challenges WHERE challenge_id = ?").bind(challengeId).first<DomainChallengeRow>();
  if (!chg) throw new DomainClaimError("no such challenge", 404);
  // Purpose before passport (both are the same 404 bytes, so the order is only
  // about reading order): this door spends `domain_proof` and nothing else, so
  // the one-shot `domain_claim` credential — which mints a SESSION at
  // `/claim/domain/verify` — can never be quietly downgraded into a re-proof, or
  // vice versa. Migration 0027.
  if (chg.purpose !== "domain_proof") throw new DomainClaimError("no such challenge", 404);
  if (chg.passport_id !== passportId) throw new DomainClaimError("no such challenge", 404); // not yours → indistinguishable
  if (chg.platform_key !== "domain") throw new DomainClaimError("not a domain challenge", 409);
  if (chg.status !== "issued") throw new DomainClaimError("challenge is not open", 409);
  if (Date.parse(chg.expires_at) < nowMs) {
    await db.prepare("UPDATE claim_challenges SET status = 'expired' WHERE challenge_id = ? AND status = 'issued'").bind(challengeId).run();
    return { ok: false, outcome: "expired" };
  }
  // Guarded in the statement — same defect, same fix, same shared helper as
  // `verifyDomainClaim` above.
  await chargeAttempt(db, challengeId, MAX_DOMAIN_VERIFY_ATTEMPTS, domainFail);

  const want = `${TXT_PREFIX}${chg.code}`;
  const records = await lookupTxtUncharged(db, challengeId, lookupTxt, chg.handle);
  // M33: what the creator has left AFTER this attempt (read back, not inferred).
  if (!records.includes(want)) return { ok: false, outcome: "record_absent", attempts_remaining: await attemptsRemainingAfterCharge(db, challengeId) };

  // Proven. The used row IS the proof (no ledger event, no claim_consumed), and
  // `proved_at` dates it at the verify instant (valid_until stays NULL so the
  // freshness TTL lives in account-standing.ts — see verifyAccountProof).
  // I3-0696 b (I22-W1-AUTH): an UPSERT, as `verifyAccountProof` and the domain-add
  // door write theirs — a guarded `status = 'issued'` UPDATE lost a matched TXT
  // record whenever the cron flipped the row to `expired` (or deleted it) during
  // the lookup. Every value is `chg`'s own and the row is born `used`, so it mints
  // nothing; a replay is still refused by the `status !== 'issued'` gate above,
  // and concurrent verifies are serialized by `chargeAttempt` before the lookup.
  const provedAt = new Date(nowMs).toISOString();
  await db
    .prepare(
      `INSERT INTO claim_challenges (challenge_id, passport_id, account_key, caller, platform_key, handle, code, status, attempts, created_at, expires_at, purpose, proved_at, method)
       VALUES (?, ?, ?, '-', 'domain', ?, ?, 'used', ?, ?, ?, 'domain_proof', ?, 'domain_dns')
       ON CONFLICT(challenge_id) DO UPDATE SET status = 'used', proved_at = excluded.proved_at, method = excluded.method`,
    )
    .bind(challengeId, chg.passport_id, chg.account_key, chg.handle, chg.code, chg.attempts + 1, chg.created_at, chg.expires_at, provedAt)
    .run();
  return { ok: true, outcome: "verified", passport_id: chg.passport_id };
}

/**
 * VERIFY A DOMAIN ADD — the proof, and the account it creates, in one batch.
 *
 * The account row does not exist yet, which is the whole difference from
 * `verifyDomainProof`. On a TXT match this appends `account_added` with
 * `actor_kind: "system"` — the one non-operator actor `events.ts` permits for a
 * domain — carrying the SAME `extraStatements` batch that flips the challenge to
 * `used` with `proved_at`. Either the account exists AND is proven, or neither
 * happened; there is no window in which a passport carries an unproven domain
 * this door put there.
 *
 * THE ACCOUNT IS MARKED "listed by Dhwaja", NOT "by creator", and that is an
 * under-claim taken deliberately. `provenance.ts` derives the "by creator" mark
 * from `actor_kind: "creator_passkey"`, which only `appendCreatorSigned` writes,
 * and `events.ts` refuses a creator-signed domain outright. Relaxing that gate
 * so a passkey signature could carry a domain would mean teaching the ledger
 * append to look for a spent challenge in its own batch — a real change to a
 * real gate, for a cosmetic mark. The load-bearing mark here is `confirmed`,
 * which the proof earns honestly. Written up as a known gap rather than
 * smuggled in.
 *
 * `passportId` is the session's, so a challenge on another passport reads as
 * "no such challenge" — indistinguishable, exactly as the sibling doors.
 */
export async function verifyDomainAdd(
  db: DbLike,
  passportId: string,
  challengeId: string,
  lookupTxt: TxtLookup,
  now: () => string = () => new Date().toISOString(),
  nowMs: number = Date.now(),
): Promise<DomainVerifyResult> {
  const chg = await db.prepare("SELECT * FROM claim_challenges WHERE challenge_id = ?").bind(challengeId).first<DomainChallengeRow>();
  if (!chg) throw new DomainClaimError("no such challenge", 404);
  if (chg.purpose !== "domain_add") throw new DomainClaimError("no such challenge", 404);
  if (chg.passport_id !== passportId) throw new DomainClaimError("no such challenge", 404);
  if (chg.platform_key !== "domain") throw new DomainClaimError("not a domain challenge", 409);
  if (chg.status !== "issued") throw new DomainClaimError("challenge is not open", 409);
  if (Date.parse(chg.expires_at) < nowMs) {
    await db.prepare("UPDATE claim_challenges SET status = 'expired' WHERE challenge_id = ? AND status = 'issued'").bind(challengeId).run();
    return { ok: false, outcome: "expired" };
  }
  await chargeAttempt(db, challengeId, MAX_DOMAIN_VERIFY_ATTEMPTS, domainFail);
  // THE DAY BUDGET IS CHARGED HERE TOO, from the same bucket the start door
  // spends. Metering only the start half was a hole with a number on it: each
  // challenge carries `MAX_DOMAIN_VERIFY_ATTEMPTS` (10) verifies, and every one
  // of them runs a DoH lookup and, past the TXT match, two paged scans of the
  // listed world — so one passport commanded ten times the gate executions its
  // ten starts were supposed to buy (adversarial review, MEDIUM). Charged AFTER
  // `chargeAttempt` so the per-challenge debit cannot be dodged by exhausting
  // the day budget first.
  if (!(await consumeRateLimit(db, `domadd:${passportId}`, nowMs, DOMAIN_ADD_CAP_PER_DAY))) {
    throw new DomainClaimError("too many domain attempts today", 429);
  }

  const want = `${TXT_PREFIX}${chg.code}`;
  const records = await lookupTxtUncharged(db, challengeId, lookupTxt, chg.handle);
  if (!records.includes(want)) return { ok: false, outcome: "record_absent", attempts_remaining: await attemptsRemainingAfterCharge(db, challengeId) };

  // ---- EVERY ELIGIBILITY QUESTION, ASKED HERE AND ONLY HERE ----------------
  //
  // `startDomainAdd` deliberately asks none of these (see its docblock): asking
  // them at start turns the door into a delisting oracle, because the answer is
  // "here is your token" or "no" and only one of the reasons for "no" is
  // secret. Asked HERE, the caller has just published a Dhwaja-issued token in
  // the domain's own DNS, so the only party who can reach these lines is the
  // party who controls the name — and none of these facts is a secret from
  // them.
  //
  // Asking at verify is also the only place worth asking. The window between
  // start and verify is up to `DOMAIN_CODE_TTL_MS` (24h), and every predicate
  // below can move inside it: a passport can be retracted, the key can be
  // registered by somebody else's ceremony, a delisting can land, a brand can
  // become verified. `verifyDomainClaim` re-asks `stillClaimable` for exactly
  // this reason; a screen taken 24 hours ago is not a screen.
  //
  // All four refuse with the SAME sentence and status. That uniformity is now
  // doing real work rather than decoration: the reader is the domain's
  // controller, so the point is no longer to hide the reason from a prober but
  // to keep the four doors from drifting into four different messages that a
  // future caller could tell apart.
  // THE OPERATOR LOG NAMES THE CHALLENGE, NOT THE SUBJECT. `refused suppressed
  // <domain>` would write the delisting fact, in clear, beside the delisted
  // name, into the Worker log stream and every logpush sink downstream — a
  // surface with a different audience and a different retention from the
  // authenticated console. No sibling suppression path in this repo logs at all
  // (`nominations.ts`, `removal.ts` and `claims.ts` contain no `console.error`).
  // This one keeps a reason CODE, because a refusal here means either a race or
  // a bug and an operator has to tell those apart; the code carries no subject,
  // and the challenge id is the join key back to the authenticated surface where
  // the domain already is (adversarial review, LOW/MEDIUM).
  const uniform = (why: string): never => {
    logError({ cid: challengeId, hop: "http", msg: "domain add refused", kind: why });
    throw new DomainClaimError(NOT_ADDABLE, 409);
  };
  // The gate's own string, not a hand-copy — `subjectMayAddAccounts`, built from
  // `ACCOUNTABLE_SUBJECT_SQL`. A CHARACTER passport whose operating agency has
  // lapsed, been retracted or asked to be delisted is still `claimed` and
  // un-retracted, and `account_added` applies no accountability CASE of its own,
  // so the hand-copy here let an unaccountable character self-serve a
  // publicly-served `confirmed` domain the claim door would have refused it
  // (adversarial review, MEDIUM).
  if (!(await subjectMayAddAccounts(db, passportId, nowMs))) uniform("passport ineligible");

  // Registered ANYWHERE — this passport or another. `account_added` refuses it
  // downstream too; refusing here keeps the two doors from drifting.
  const registered = await db.prepare("SELECT 1 AS x FROM passport_accounts WHERE account_key = ? LIMIT 1").bind(chg.account_key).first<{ x: number }>();
  if (registered) uniform("already registered");

  // Delisted at its subject's request — never re-listed. Same statement
  // `nominations.ts` and the importer use, so the three cannot disagree about
  // what "suppressed" means.
  const suppressed = await suppressedKeyStmt(db, chg.account_key).first<{ x: number }>();
  if (suppressed) uniform("suppressed");

  // THE LOOKALIKE GATE (`domain-gate.ts`). DNS proof answers "do you control
  // it", which was never the question worth asking: an attacker genuinely
  // controls `netflix-support.com`, and a verified passport's accounts inherit
  // its standing on the public page, which would make the laundered name
  // `confirmed` and, at `reports.ts`, unreportable.
  //
  // WHAT A HIT ACTUALLY DOES, corrected 2026-09-20 (W3-09 SCOUT,
  // `W309-SCOUT-J02-claims-hygiene`, D-0440/ANTI-024). This comment used to say
  // "a hit routes to an operator ... rather than to a refusal". It does not and
  // never did: the executable line is the `if (gate.blocked) uniform(...)`
  // below, which REFUSES, writes no queue row, and signals only through the
  // operator log.
  //
  // NO LINE NUMBER, AND THAT IS THE POINT. This comment has now been wrong
  // about its own neighbour THREE times: first "routes to an operator" (a
  // behaviour that never existed), then "two lines below" (it was six), then
  // ":1150" — which was true when written and was falsified by adding the
  // three lines that said it, in the same edit. An absolute `:NNNN`
  // self-reference inside a comment in the file it numbers invalidates itself
  // whenever the comment's own length changes. So this one names the
  // STATEMENT, which moves with it, and nothing that can rot.
  // The claim described an intention; the code has always refused. An operator
  // route now exists — `brand_contests` (0102, D-0561) — but it is a channel
  // the refused party FILES INTO after proving the contested host, not
  // something this line does for them. A refusal with somewhere to go
  // afterwards is still a refusal here.
  // F14 / HANDOFF §1.2: the roster authorization, computed from THIS passport's
  // own active agency grants. Empty for everybody not on a roster, so the gate
  // behaves exactly as it did for every other caller.
  // #873a: THIS request's clock (`nowMs`), not the wall clock — one request, one clock.
  const apexes = await agencyGatingApexes(db, passportId, nowMs);
  const gate = await domainAddGate(db, chg.handle, passportId, apexes);
  if (gate.blocked) uniform(gate.reason === "resembles" ? `resembles:${gate.kind}` : gate.reason);

  // REVIEW, NOT ADD, for the two cases DNS proof cannot settle (I16-W05B). Both return BEFORE the append, so no
  // `account_added` is written; the challenge is spent and an operator item is queued (`routeDomainClaimToReview`).
  // D-0921(4): a same-script look-alike of another passport's IDN domain is a review, never a silent refusal.
  const same = await sameScriptGate(db, chg.handle, passportId);
  if (same.hit) return routeDomainClaimToReview(db, chg, now, "domain_add");
  // RJ550b-W3-15-R7-1 (D-0920 option a): a name the apex split cannot call a registrable apex can be a dangling
  // CNAME whose TXT authority belongs to whoever claimed the target, so one TXT record does not list it — the
  // claim door's own disjunct (`verifyDomainClaim`). EXCEPT a host under a load-bearing roster apex (F14), whose
  // authorization is a DNS proof of that apex plus this passport's own grant: that population is left as it was
  // and named as an open question (it conflicts with F14's direct add; REPORT.md OPEN QUESTIONS).
  const rosterAuthorized = apexes.some((a) => chg.handle !== a && chg.handle.endsWith(`.${a}`));
  if (!isRegistrableApex(chg.handle) && !rosterAuthorized) return routeDomainClaimToReview(db, chg, now, "domain_add");
  // Q537 (Jason: "apply the same continuity check"): the claim door's continuity limb, asked on the add door after
  // the proof and before the write — an unseeded passport's add (I62 P3, D-1522), or a seeded one's that cannot be
  // dated inside the window, goes to review.
  // I62-FX-P3B HANDOFF: under P3 every unseeded add routes here, so a roster add reaches the operator through review;
  // I63-P3 R2 (D-1535): its `domain_add:under_agency_apex` row is written at approval (review.ts `approveItem`).
  if (await domainContinuityNeedsReview(db, chg.passport_id, chg.handle, nowMs)) return routeDomainClaimToReview(db, chg, now, "domain_add");
  // I3-0189 c (D-0993(1) census pick): the roster-apex carve-out holds only for a TXT that sits INSIDE the apex the
  // grant authorizes. A matching record reached through a CNAME that LEAVES it is a foreign party's record — review.
  if (!isRegistrableApex(chg.handle) && txtOwnerOutsideApex(records, want, apexes.filter((a) => chg.handle.endsWith(`.${a}`)))) {
    return routeDomainClaimToReview(db, chg, now, "domain_add");
  }

  // THE PROOF AND THE ROW COMMIT TOGETHER. The guarded `status='issued'` in the
  // extra statement is what serializes two concurrent verifies: the loser's
  // batch changes no challenge row, and `appendEvents` is asked to append an
  // `account_added` whose account_key the winner has already registered — which
  // `events.ts` refuses on its own uniqueness read. Neither can double-add.
  const provedAt = new Date(nowMs).toISOString();
  // THE PROOF ROW MUST LAND WHENEVER THE EVENT DOES — the same statement, and
  // the same reason, as the claim door's (`verifyDomainClaim`, the UPSERT in its
  // `extraStatements`). This spend rides the SAME batch as the `account_added`
  // append, and the guarded `UPDATE ... AND status = 'issued'` it replaces wrote
  // NOTHING whenever the row had left `issued` since the read a DoH lookup ago —
  // `pruneClaimChallenges` expires an `issued` row on `expires_at <= now` and
  // then deletes it, and its `NOT EXISTS claim_consumed` guard protects nothing
  // on this door, which writes no `claim_consumed` row. A zero-row statement
  // does not fail its batch, so the event landed and the proof did not, and
  // `accountProofStanding` reads proof ONLY from a `status = 'used'` row: the
  // passport carried a domain the registry would not serve.
  //
  // SINGLE-SPEND IS UNCHANGED. A replay is refused above (`status !== 'issued'`
  // -> 409), two concurrent verifies are serialized by `chargeAttempt`'s own
  // guarded UPDATE before the lookup, and `events.ts` refuses a second
  // `account_added` on the same account_key. This statement makes the proof
  // follow the event; it authorizes nothing the event did not.
  const spend = db
    .prepare(
      `INSERT INTO claim_challenges (challenge_id, passport_id, account_key, caller, platform_key, handle, code, status, attempts, created_at, expires_at, purpose, proved_at, method)
       VALUES (?, ?, ?, '-', 'domain', ?, ?, 'used', ?, ?, ?, 'domain_add', ?, 'domain_dns')
       ON CONFLICT(challenge_id) DO UPDATE SET status = 'used', proved_at = excluded.proved_at, method = excluded.method`,
    )
    .bind(challengeId, chg.passport_id, chg.account_key, chg.handle, chg.code, chg.attempts + 1, chg.created_at, chg.expires_at, provedAt);
  // AN ADD THAT USED A ROSTER AUTHORIZATION IS ENUMERABLE AFTERWARDS.
  //
  // The review's third finding is that the widening is PERMANENT and no F14
  // lever reaches what it minted: revoking the grant, lapsing the apex and
  // sweeping the row all leave the added account exactly where it is — which is
  // the same rule that stops an agency taking `nova.talentco.com` off an artist
  // for leaving, running in the other direction. That asymmetry is deliberate
  // and it stays, so the answer is not a new remedy but VISIBILITY: an operator
  // who later learns an apex was proven in bad faith needs the list of accounts
  // it let through, and without this row there is nothing to join on. Written
  // only when an authorization was actually in play, so the ordinary door is
  // byte-for-byte unchanged.
  // …and only when the authorization was actually LOAD-BEARING for THIS host —
  // the same `underHere` predicate the gate uses, so the audit row means "this
  // add sat under a roster apex" and not merely "this creator has an agency".
  const usedApexes = apexes.filter((a) => chg.handle !== a && chg.handle.endsWith(`.${a}`));
  const extra = usedApexes.length > 0 ? [spend, auditStmt(db, "system", "domain_add:under_agency_apex", chg.handle, usedApexes.join(","), provedAt)] : [spend];
  const r = await tryAppendEvents(
    db,
    [
      {
        event_type: "account_added",
        subject: passportId,
        actor_kind: "system",
        actor_id: "system:domain-add",
        payload: {
          platform: "domain",
          id_kind: "domain",
          id: chg.handle,
          relationship_basis: "creator_claimed",
          observed_at: provedAt,
        },
      },
    ],
    now,
    [],
    extra,
    // THE `system` MARK IS ASSERTED, NOT ASSUMED. `events.ts` refuses
    // `actor_kind: "system"` on the general path and requires the caller to name
    // it here, because "system events are emitted only by server code that
    // proved the triggering condition". The TXT match a few lines above IS that
    // condition, and passing this list is how this function says so. A handler
    // that forwarded a caller-supplied actor kind could never reach this line.
    ["system"],
  );
  if (!r.ok) {
    // The ledger is the last word, and it says the same sentence. `appendEvents`
    // refuses a suppressed key and an already-registered one on its own reads;
    // the block above screened both moments earlier, so reaching here means a
    // concurrent write landed between the screen and the batch — a race, not a
    // probe. Logged for the operator, uniform for the caller.
    //
    // BREAK[SC-10]: this used to log `r.refusal.message` VERBATIM. For the
    // "already registered" ledger refusal that message is
    // `` `account already registered: ${subject.canonical}` `` (events.ts) —
    // the canonical domain itself — which put the accepted domain identity
    // into the Worker log/logpush stream this same function's `uniform()`
    // helper, a few dozen lines above, deliberately keeps subject-free
    // ("THE OPERATOR LOG NAMES THE CHALLENGE, NOT THE SUBJECT"). The two
    // refusal branches must share one diagnostic policy: a reason CODE plus
    // the challenge id, which is the join key back to the authenticated
    // console surface where the domain already lives — never the ledger's
    // own free-text message.
    logError({ cid: challengeId, hop: "http", msg: "domain add refused", kind: r.refusal.kind });
    throw new DomainClaimError(NOT_ADDABLE, 409);
  }
  const event = r.events[0];
  await db.prepare("UPDATE claim_challenges SET result_event = ? WHERE challenge_id = ? AND status = 'used'").bind(event!.event_id, challengeId).run();
  return { ok: true, outcome: "verified", passport_id: passportId, event_id: event!.event_id };
}

/**
 * #132PS-211 (D-0920): the authenticated re-read of a live `domain_add` / `domain_proof` TXT challenge — the
 * domain sibling of `claims.ts rereadLiveChallenge`, same rules: the same live row handed back (never a new one),
 * only to the passport that started it (the route passes the SESSION's passport id). Both doors mint a row for the
 * caller's own passport whatever the eligibility answer (`startDomainAdd` asks none at start), so a hit or a miss
 * carries nothing about anyone else. `domain_claim` (anonymous) and `brand_enrol`/`agency_domain` are not re-readable.
 */
export async function rereadLiveDomainChallenge(db: DbLike, sessionPassportId: string, rawDomain: string, nowMs: number = Date.now()): Promise<DomainStartResult | null> {
  let accountKey: string;
  try {
    accountKey = domainAccountKey(rawDomain);
  } catch {
    return null;
  }
  const row = await db
    .prepare(
      `SELECT challenge_id, handle, code, expires_at FROM claim_challenges
        WHERE passport_id = ? AND account_key = ? AND purpose IN ('domain_add', 'domain_proof') AND status = 'issued' AND expires_at > ?
        ORDER BY created_at DESC LIMIT 1`,
    )
    .bind(sessionPassportId, accountKey, new Date(nowMs).toISOString())
    .first<{ challenge_id: string; handle: string; code: string; expires_at: string }>();
  if (!row) return null;
  return { challenge_id: row.challenge_id, host: row.handle, record_name: row.handle, record_value: `${TXT_PREFIX}${row.code}`, expires_at: row.expires_at };
}

/**
 * THE SELF-SERVE DOMAIN-ADD GATE (slice 32).
 *
 * `startDomainAdd` lets a claimed creator attach a SECOND domain to their own
 * passport by proving DNS control of it. DNS proof answers "do you control this
 * name" — and that was never the question worth asking, because an attacker can
 * genuinely control `netflix-support.com`. A verified passport's accounts
 * inherit its standing on the public page and in the registry bucket, so an
 * unguarded door turns "register a lookalike" into "publish a lookalike as
 * confirmed, under a listed creator's name, unreportable" (`reports.ts` answers
 * `real_account` for a listed key).
 *
 * ---------------------------------------------------------------------------
 * WHY THIS IS NOT `findDomainLookalike`
 * ---------------------------------------------------------------------------
 * The first cut reused the report-evidence detector. The two jobs want opposite
 * error directions: that detector writes a POINTER at a named creator onto a fan
 * report, so a false positive there is an accusation and it is tuned to
 * under-fire. This gate decides whether a stranger may self-attach a name; a
 * false positive costs one uniform refusal and a slower road (an operator can
 * still add the domain by import, with a human looking), a false negative mints
 * a laundered lookalike the product then vouches for.
 *
 * ---------------------------------------------------------------------------
 * WHY THE RULES ARE SCALED BY BRAND LENGTH — AND NOT BY ANCHORING
 * ---------------------------------------------------------------------------
 * The SECOND cut of this file was measured by two adversarial reviews and it was
 * miscalibrated in BOTH directions at once, from one root cause: it decided
 * strictness by where the brand sat in the string instead of by how much
 * accident was possible.
 *
 *   UNDER-FIRE. Containment was ANCHORED — the brand had to start or end a
 *   token — so filler on both sides walked: `mynetflixhub`, `getuberdriver`,
 *   `thepaypalstore`. A sweep of prefix+brand+suffix candidates put the escape
 *   rate at 86%. A 12-character cap on the filler made it worse in the direction
 *   that matters: `ubereatsdelivery` was blocked and `ubereatsdeliveryhelp` —
 *   the more convincing name — was not.
 *
 *   OVER-FIRE. The same anchored rule at a 4-character floor refused ordinary
 *   words that merely begin or end with a short brand: `metadata`, `youtuber`,
 *   `dreamhost`, `ryanair`, and `visalia` — a real Californian city, and the
 *   exact over-match `domain-match.ts`'s own docblock says the evidence detector
 *   was designed to avoid. Measured on a ~370k-word corpus: a 4-character brand
 *   refused ~115 of them, a 9-character brand refused ~1. (Every number BELOW
 *   this line uses the 274,937-word corpus the derivation ships against, so the
 *   two sets of figures are not directly comparable.)
 *
 * The fix was the same rule for both problems: accident is a function of how
 * much ordinary language the brand string occupies, so containment became
 * unanchored with no filler cap and was gated on a proxy for that — brand
 * LENGTH. That proxy is what the next section replaces; what survives from it is
 * the shape:
 *
 *     >= 4 chars   identity and bounded edit distance, always
 *     >= 4 chars   unanchored containment, UNLESS the brand is generic
 *     <= 3 chars   nothing (a three-letter brand matches half the internet)
 *
 * ---------------------------------------------------------------------------
 * AND WHY LENGTH IS NO LONGER THE TEST (slice 33)
 * ---------------------------------------------------------------------------
 * The length rule shipped with a 40% residual escape rate on that sweep, and
 * EVERY escape was a four-character brand — `visa`, `uber`, `meta`, `nike`,
 * `ebay`, which is exactly the set worth laundering. Measuring the thing that
 * length was standing in for showed the proxy was not merely coarse but
 * INVERTED:
 *
 *     brand      English words strictly containing it
 *     ---------  ------------------------------------
 *     xbox espn  0        four characters, and CHEAPER to protect than every
 *     hulu       1        five-character brand the length floor was already
 *     ebay ikea  8        protecting unconditionally
 *     nike       10
 *     visa       74
 *     ---------  ------------------------------------
 *     chase      35       five characters, protected by the old floor at a
 *     apple      46       cost the four-character brands above would have
 *     label      49       beaten
 *     dream      61
 *     ---------  ------------------------------------
 *     uber       151      genuinely expensive, whatever their length
 *     acer       161
 *     meta       457
 *
 * So containment is gated on GENERICNESS, measured once offline
 * (`scripts/derive-generic-skeletons.mjs`, 274,937 words) and frozen into
 * `generic-skeletons.ts`. A brand in that set keeps identity and edit distance
 * and loses containment; every other brand gets containment from four characters
 * up. Of 73 real brands measured, exactly ONE is in the set: `meta`.
 *
 * Measured effect on a prefix+brand+suffix sweep: over the FIVE brands that were
 * escaping (1,430 labels) 50.0% addable → 20.0%; over the full 73-brand roster
 * (1,825 labels) 30.2% → 1.3%. `meta` is the only brand still escaping in both.
 * The first number answers "did the change fix what was broken"; the second
 * answers "how good is the gate now". An earlier draft quoted only the first,
 * which states a rate measured on a deliberately biased sample as if it were the
 * overall one.
 *
 * AND THE SECOND NUMBER, STRATIFIED (QUEUE #185, W-14). An aggregate over the
 * roster hides WHERE the escapes are, and they are not spread evenly: they sit
 * in the generic set and in the SEAM-STRADDLE stratum — brands a substring of
 * whose own edge joins the filler into an ordinary word (`the│e·bay` reads
 * `thee│bay`, `i│stripe` reads `is│tripe`, `go│twitch` reads `got│witch`), so
 * `compound-words.ts`'s straddle clause excuses the occurrence as a collision.
 * Being a vocabulary word is NOT the predicate (verizon, mastercard and visa
 * are and do not escape; `w14-seam-vocab-brand-stratum.test.ts` pins that).
 * Measured on a DIFFERENT, reproducible sweep from the one above — the same 73
 * brands × 8 prefixes (`""`, get, my, the, new, go, shop, i) × 10 suffixes
 * (`""`, shop, store, support, help, app, online, login, hq, official) — so
 * the two sets of numbers are not comparable with each other.
 * `b27-w14-seam-stratum.test.ts` re-derives every line below from the live
 * gate and fails when one stops matching:
 *
 *     W14 all: brands=73 labels=5840 escaped=82

 * CEILING. It counts single dictionary WORDS containing the brand. Domain labels

 * SLICE 34 CLEARS ALL FOUR — `compound-words.ts` segments the label and drops an

/**
 * The comparison set for a host: its registrable label's skeleton PLUS the
 * skeleton of every hyphen/dot-separated token above the public suffix.
 *
 * The token half closes the private-suffix family. `netflix.uk.com` has no entry
 * in the curated public-suffix set, so its registrable label is `uk` — two
 * characters, which a length floor drops before it compares anything. Its tokens
 * are `netflix` and `uk`, and the first is the whole attack.
 *
 * AN EMPTY COMPARISON SET IS NOT A FAILURE. `null` is returned only when the
 * host has no registrable form at all (a bare public suffix, an IP literal,
 * junk). A host whose skeleton is empty or two characters — `中国.com`,
 * `मुकेश.com`, `hp.net`, `x.ai` — gets `all: []`, and the subdomain rule below
 * still runs against it. The previous version returned `null` for those and the
 * caller blocked, which refused every wholly non-Latin domain and every
 * short-label domain in exchange for nothing: a skeleton under four characters
 * cannot match any brand this gate will consider (adversarial review, HIGH).
 */
export function gateSkeletons(host: string): GateHost | null {
  const h = host.trim().toLowerCase().replace(/\.$/, "");
  const reg = registrable(h);
  if (!reg) return null;
  const all = new Set<string>();
  let slack = 0;
  let confusable = false;
  let unreadable = false;
  const lab = gateLabelSkeleton(reg.label);
  const labDropped = Math.max(0, lab.dropped - [...toUnicodeLabel(reg.label)].filter((c) => c === "-").length); // I3-1143: an ASCII hyphen is a separator, not a deletion
  slack = Math.max(slack, labDropped);
  confusable ||= lab.confusable || labDropped > 0;
  // C-012 FIX (partial — see the residual gap named below). A skeleton under
  // 3 chars was dropped from the comparison UNCONDITIONALLY, and that is
  // right for a label that got there honestly: a genuinely short label, or a
  // label from a script this module does not curate at all (CJK, Devanagari)
  // folds to nothing and SHOULD stay out, or every wholly non-Latin domain is
  // refused for nothing — the exact adversarial-review HIGH the module
  // docblock above already records and this fix must not reopen (`hp.net`,
  // `x.ai`, `中国.com`, `मुकेश.com` all still yield `all: []`, unchanged;
  // pinned by `slice32-domain-add.test.ts`'s "hp.net" assertion).
  //
  // But a NONZERO, SHORT residual (1-2 chars) plus `dropped > 0` is a
  // different fact: something in this label folded to real a-z0-9 AND
  // something else in it fell outside the curated map — the shape of an
  // attacker who built a longer visual copy of a brand out of mostly
  // uncurated look-alikes with a couple of curated ones caught in the mesh.
  // Dropping that residual instead of comparing it with `slack` forgiveness
  // (which exists for exactly this — see `GATE_MAX_SLACK`'s docblock) is the
  // gap: the fold does its job (evidence the label is NOT what it displays as
  // Latin text) and the length floor throws the evidence away before `slack`
  // gets to use it. A fully-dropped label (skeleton `""`) still gets nothing:
  // that is indistinguishable from a genuine non-Latin domain by any signal
  // this module has, and is the residual gap named in OPEN QUESTIONS, not
  // silently fixed here.
  if (lab.skeleton.length >= 3 || (lab.skeleton.length > 0 && labDropped > 0)) all.add(lab.skeleton);
  const labels = h.split(".").filter((l) => l.length > 0);
  for (const rawLbl of labels.slice(0, labels.length - publicSuffixLen(labels))) {
    // DECODE, THEN SPLIT. Splitting `xn--fiqs8s` on "-" yields the punycode
    // fragment `fiqs8s`, which is not a word anybody wrote and would be compared
    // as if it were one.
    for (const tok of toUnicodeLabel(rawLbl).split("-")) {
      const t = gateLabelSkeleton(tok);
      slack = Math.max(slack, t.dropped);
      confusable ||= t.confusable || t.dropped > 0;
      // Same C-012 rule as the registrable label, above.
      if (t.skeleton.length >= 3 || (t.skeleton.length > 0 && t.dropped > 0)) all.add(t.skeleton);
      // QUEUE #104 — THE EMPTY FOLD. Every label token passes through here,
      // the registrable label's included, so this is the one place a token
      // that folded to "" is seen.
      // I20-W1D (#946 (b)): judged on the fold WITHOUT the shared name-fold rows too, so a token the brand fold
      // could not read before (an all-Lisu label) stays refused as unreadable; the shared rows only ADD a comparison.
      if (tok.length > 0 && (t.skeleton === "" || gateLabelSkeleton(tok, false).skeleton === "") && !emptyFoldAdmitted(tok)) unreadable = true;
    }
  }
  return { host: h, regDomain: reg.domain, all: [...all], slack: Math.min(slack, GATE_MAX_SLACK), label: lab.skeleton, confusable, unreadable };
}

/**
 * QUEUE #104 — THE EMPTY FOLD, AND THE QUESTION ASKED OF IT INSTEAD.
 *
 * A token whose every code point falls outside the curated fold skeletonises
 * to "" and has nothing to compare, so before this it was admitted: an
 * all-Cherokee `ᏢᎪᎩᏢᎪᏞ.com` renders as PAYPAL and cleared the gate. No
 * threshold on the skeleton can separate it from `中国.com` — both are "" with
 * every code point dropped — which is why the C-012 fix above stopped short
 * of it and why a length rule would reopen the adversarial-review HIGH in the
 * `gateSkeletons` docblock.
 *
 * So the question changes. Not "what does it fold to" but "COULD THIS TOKEN
 * SPELL A LATIN BRAND AT ALL?", which is a property of the SCRIPT, answered
 * here by an ALLOWLIST and failing closed for everything else:
 *
 *   ADMITTED UNREAD — letters and marks of a script with no Latin letter
 *   coverage (CJK, kana, Hangul, the Indic scripts, Thai/Lao/Khmer/Myanmar,
 *   Arabic, Hebrew, Ethiopic, Mongolian, Yi…). A token written wholly in one
 *   of these cannot render as a Latin word, so there is nothing to protect.
 *   Arabic and Hebrew each have two or three letters that resemble l/o/i;
 *   no brand of four or more characters is spelled from those alone.
 *   CYRILLIC IS NOT ADMITTED (D-0810(13), QUEUE #104 fix round). It was, on
 *   the premise that its look-alikes are exactly the ones the fold maps; they
 *   are not — ѵ є ѻ ь ꙇ ѡ ү fold to "" and spell `vevo`/`vivo`/`webb`
 *   (B27-RG-CLIENTS probe 5). So a Cyrillic token that folds to "" is refused
 *   self-serve like the list below, and that includes ordinary words
 *   (`люди`, `пыль`): the ruling's cost, routed to the scrutiny lane.
 *
 *   NEUTRAL — hyphen, ZWNJ/ZWJ (U+200C/D), the kana prolonged-sound mark
 *   U+30FC (`コーヒー`) and Inherited combining marks. Neutral characters are
 *   allowed but never make a token admissible on their own.
 *
 *   REFUSED (everything else) — Cherokee, Lisu, Canadian syllabics, Osage,
 *   Tifinagh, Runic, Armenian, Georgian, the unmapped Greek look-alikes
 *   (η μ ω γ β), Latin letters the fold does not map, Common-script letterlikes
 *   (mathematical alphanumerics `𝐩𝐚𝐲`, ℓ, circled letters), digits of every
 *   script, emoji and symbols. Refused means `domainAddGate` returns
 *   `unparseable` — the gate cannot read the name — so the self-serve door
 *   refuses uniformly and the enrolment door routes to the scrutiny lane,
 *   where a person reads it. The road for a genuine Cherokee, Armenian,
 *   Georgian or Greek-only label is operator import, and that cost is named in
 *   the B27-DOMAIN report rather than discovered in a support ticket.
 *
 * Enumerating HOSTILE scripts is the game the fold's own docblock says you
 * lose by playing; enumerating SAFE ones loses only by refusing, which is the
 * direction this gate is built to fail in.
 */
const EMPTY_FOLD_SCRIPTS =
  /^[\p{Script=Han}\p{Script=Hiragana}\p{Script=Katakana}\p{Script=Hangul}\p{Script=Bopomofo}\p{Script=Devanagari}\p{Script=Bengali}\p{Script=Gurmukhi}\p{Script=Gujarati}\p{Script=Oriya}\p{Script=Tamil}\p{Script=Telugu}\p{Script=Kannada}\p{Script=Malayalam}\p{Script=Sinhala}\p{Script=Thai}\p{Script=Lao}\p{Script=Tibetan}\p{Script=Myanmar}\p{Script=Khmer}\p{Script=Arabic}\p{Script=Hebrew}\p{Script=Syriac}\p{Script=Thaana}\p{Script=Ethiopic}\p{Script=Mongolian}\p{Script=Yi}]$/u;
const LETTER_OR_MARK = /^[\p{L}\p{M}]$/u;
const NEUTRAL_JOINER = /^[-\u200c\u200d\u30fc]$/u;
const INHERITED_MARK = /^\p{Script=Inherited}$/u;
const MARK = /^\p{M}$/u;

/**
 * D-0813(4), #104 (J-104-sibling) — CROSS-SCRIPT LATIN LOOK-ALIKES ARE FOLDED,
 * NOT ADMITTED UNREAD. `EMPTY_FOLD_SCRIPTS` admits whole scripts on the premise
 * that no letter of theirs reads as Latin; a handful do, and a token spelled
 * from them (`ןסן` = lol, `וסוס` = lolo, `ㅇㅣㅇ` = olo, `ロロ` = oo) folded to ""
 * and cleared without being compared. Each such LETTER is mapped to the Latin
 * letter it shows as BEFORE `labelSkeleton`, so the host is compared exactly as
 * its Latin twin would be, and marked `confusable` (no seam forgiveness). No
 * script is refused wholesale: an ordinary CJK/Hangul/Hebrew/Arabic name still
 * clears unless it actually spells a listed brand.
 *
 * ONLY letters of an ADMITTED script are mapped. A code point the gate already
 * REFUSES (a digit of any script, `〇` U+3007 which is Nl) is left alone:
 * folding it to a letter would turn a refusal into a comparison.
 *
 * SOURCES. Unicode confusables.txt (UTS #39) via the data shipped in
 * `confusable_homoglyphs` 3.3.1 (`confusables.json`, generated from it): ଠ ഠ ဝ → o,
 * ሀ → u, ዐ → o, and the chains ロ↔口, エ↔工, ト↔卜, ㅣ↔丨, ㅇ↔ᄋ. The Latin target of
 * each chain, and the Hebrew (ו ן → l, ס → o) and Arabic (ا → l, ه → o) rows, are
 * the ones D-0813(4) enumerates; confusables.txt itself was unreachable from the
 * build environment (HTTP 403), so those rows cite the ruling, not a re-read.
 */
const CROSS_SCRIPT_LATIN: ReadonlyMap<string, string> = new Map([
  // Hebrew
  ["\u05d5", "l"], ["\u05df", "l"], ["\u05e1", "o"],
  // Arabic
  ["\u0627", "l"], ["\u0647", "o"],
  // Hangul compatibility + conjoining jamo
  ["\u3147", "o"], ["\u110b", "o"], ["\u3163", "l"], ["\u1175", "l"],
  // Katakana and their Han twins
  ["\u30ed", "o"], ["\u53e3", "o"], ["\u56d7", "o"], ["\u30a8", "i"], ["\u5de5", "i"], ["\u30c8", "t"], ["\u535c", "t"], ["\u4e28", "l"],
  // Indic / Myanmar / Ethiopic letters listed in confusables.txt
  ["\u0b20", "o"], ["\u0d20", "o"], ["\u101d", "o"], ["\u1200", "u"], ["\u12d0", "o"],
]);

/** Fold cross-script look-alikes, then the shared skeleton. A hyphen-separated
 * part is folded ONLY when EVERY code point of it is a mapped look-alike: that
 * is exactly the token that folded to "" and was admitted unread. A part that
 * merely CONTAINS one (`דוגמה`, an ordinary Hebrew word with a vav) is left as it
 * was — mapping one letter of it would hand `GATE_MAX_SLACK` the other four as
 * deletions and refuse an ordinary name as a typo of a four-letter brand
 * (measured: `דוגמה.com` against `lolo.com`). A part holding a digit or any
 * unmapped code point therefore also keeps its old verdict. `confusable` is set
 * when anything was mapped: the ASCII the skeleton sees is a REWRITE.
 * #658(b) (D-0954(4)): the test runs on each part's NFD form. IDNA's NFC composes
 * ᄋ U+110B + ᅵ U+1175 into the syllable 이 U+C774 before this function sees the
 * label, so the jamo rows above never matched the pair; NFD gives the two jamo
 * back. A part that is not wholly mapped after NFD is returned unchanged. */
const HANGUL_SPLIT_FORM = "NFD";

/**
 * I20-W1D (QUEUE #946 (b); D-0957(3), D-0958(1)): THE BRAND FOLD SHARES THE NAME FOLD'S TABLE (U+0237, the Latin and
 * Greek small capitals, Lisu) — applied inside `labelSkeleton` itself since I20-W2C (domain-match.ts
 * `sharedLatinFold`, the one implementation), so this gate, reports.ts and agencies.ts read the same letters.
 * `shareNameFold = false` passes through: the `unreadable` verdict in `gateSkeletons` is judged on the fold without
 * the shared rows, so an all-Lisu token stays refused.
 */
export function gateLabelSkeleton(label: string, shareNameFold = true): ReturnType<typeof labelSkeleton> {
  let mapped = false;
  const parts = toUnicodeLabel(label)
    .split("-")
    .map((part) => {
      const cps = [...part.normalize(HANGUL_SPLIT_FORM)];
      if (cps.length === 0 || !cps.every((ch) => CROSS_SCRIPT_LATIN.has(ch))) return part;
      mapped = true;
      return cps.map((ch) => CROSS_SCRIPT_LATIN.get(ch)!).join("");
    });
  const r = labelSkeleton(parts.join("-"), shareNameFold);
  return mapped ? { ...r, confusable: true } : r;
}

/** QUEUE #104 — may a token that folded to "" be admitted unread? See above. */
export function emptyFoldAdmitted(token: string): boolean {
  let letters = 0;
  for (const ch of token) {
    if (LETTER_OR_MARK.test(ch) && EMPTY_FOLD_SCRIPTS.test(ch)) {
      letters++;
      continue;
    }
    if (NEUTRAL_JOINER.test(ch) || (MARK.test(ch) && INHERITED_MARK.test(ch))) continue;
    return false;
  }
  return letters > 0;
}

/**
 * Does any of the target's comparison strings resemble this brand skeleton?
 * Three arms, most-deliberate first; the first hit wins (the kinds only ever
 * reach an operator, so ranking them precisely buys nothing).
 */
/** Given a target string, a predicate saying whether an occurrence at [lo, hi)
 * is a SEAM COLLISION between two ordinary words rather than a name somebody
 * chose. Supplied by the caller so the per-label analysis is paid for once and
 * reused across every brand tested against that label. */
export type SeamTest = (target: string) => (lo: number, hi: number) => boolean;

export function gateMatch(
  targets: readonly string[],
  brand: string,
  arm: GateArm = "domain",
  slack = 0,
  /** Omitted → every containment hit counts, which is the pre-slice-34 rule and
   * what the pure match tests exercise. `domainAddGate` always supplies one. */
  seam?: SeamTest,
): GateKind | null {
  if (brand.length < GATE_MIN_BRAND) return null;
  for (const t of targets) if (t === brand) return "same";

  // A VERIFIED name is treated like a domain. `GATE_NAME_MIN_CONTAIN` exists for
  // VOLUME — there are far more display names than domains, and a containment
  // neighbourhood around each would take a large slice of the namespace — but
  // that argument does not apply to the operator-vetted rung, which is bounded
  // and curated. It applied to all three rungs only because the arm did not
  // distinguish them, and adversarial review measured the cost: a verified
  // handle-only brand (`@visa`, `@nike`) had NO containment protection at all,
  // and handle-only is most of the registry. The slice's own headline test
  // asserted the property in the one world where it held — a listed domain.
  const minContain = arm === "name" ? GATE_NAME_MIN_CONTAIN : GATE_MIN_BRAND;
  if (brand.length >= minContain && !isGenericSkeleton(brand)) {
    // UNANCHORED, AND WITH NO CAP ON THE FILLER. Both of those were earlier
    // versions' mistakes: anchoring let `mynetflixhub` through, and a filler cap
    // let the LONGER, more convincing `netflixofficialstore` through while
    // blocking the shorter one.
    //
    // "A brand of five or more characters inside a label is not an accident" is
    // what this comment used to say, and slices 33 and 34 exist because it is
    // false in both directions: the floor is four (genericness decides, not
    // length), and an accident is common enough at every length that `seam`
    // below is what tells one from a name.
    for (const t of targets) {
      if (t.length <= brand.length) continue;
      // EVERY occurrence, not the first: a label can carry one occurrence that
      // the seam explains and another it does not, and the unexplained one is
      // the whole reason this arm exists.
      let explained: ((lo: number, hi: number) => boolean) | null = null;
      for (let i = t.indexOf(brand); i >= 0; i = t.indexOf(brand, i + 1)) {
        if (seam) {
          explained ??= seam(t);
          if (explained(i, i + brand.length)) continue;
        }
        return "contains";
      }
    }
  }

  // I3-1142 — THE SAME RUNG RULE AS CONTAINMENT, ABOVE. `GATE_NAME_MIN_TYPO`
  // is a VOLUME floor for unvetted display names; the verified rung is bounded
  // and operator-vetted, so it earns the edit-distance arm from the domain
  // arm's floor. At 7 every verified handle-only brand of 4-6 characters
  // (`@visa`, `@nike`) had no typo protection at all.
  const minTypo = arm === "name" ? GATE_NAME_MIN_TYPO : GATE_MIN_BRAND;
  if (brand.length >= minTypo) {
    for (const t of targets) {
      // `slack` forgives exactly what the skeleton fold deleted — see
      // `GATE_MAX_SLACK`. It is zero for every plain-ASCII host, which is nearly
      // all of them, so this widens nothing that was not already mangled.
      const maxD = (brand.length <= 6 ? 1 : 2) + slack;
      // C-005 FIX. This USED to be `Math.abs(t.length - brand.length) > maxD`
      // then `editDistance(brand, t, maxD)` — comparing the BRAND to the WHOLE
      // token. That is the right comparison for a bare typosquat
      // (`netfl1x.com`) and the wrong one the moment filler is added on either
      // side: `netlfixsupportteam.com` transposes two letters of `netflix`
      // (edit distance 2, one below `maxD`) and the token is 19 characters
      // against a 7-character brand — a length gap of 12, so the OLD
      // prefilter discarded it before `editDistance` ever ran. The exact
      // substring is gone too (transposition breaks it), so the containment
      // arm above never fires either: one edit plus filler defeated both arms
      // at once, and it is cheap to construct WHATEVER the brand or the
      // filler is, because the filter that killed it had nothing to do with
      // resemblance.
      //
      // `minWindowDistance` finds the best-aligned CONTIGUOUS window of `t` —
      // free to start and end anywhere in `t` — so filler before or after the
      // near-miss costs nothing and only the edits INSIDE the window count.
      // It subsumes the old whole-token compare (the whole string is one
      // valid window), so nothing this arm caught before stops being caught.
      // `d === 0` is deliberately excluded: that is an EXACT substring
      // occurrence, which is the containment arm's decision alone — it may
      // have declined on purpose (a generic brand like `meta`, or a
      // seam-explained compound), and this arm re-flagging the identical
      // occurrence as "typo" would silently overrule that decision. Only an
      // ACTUAL edit — a mismatch this arm exists for — reaches here.
      if (t.length + maxD < brand.length) continue; // too short to ever close the gap
      const d = minWindowDistance(t, brand);
      if (d >= 1 && d <= maxD) return "typo";
    }
  }
  return null;
}

/**
 * Minimum edit distance (Damerau-Levenshtein — substitution, insertion,
 * deletion, and adjacent transposition, matching `editDistance`'s own ops)
 * between `brand` and ANY contiguous substring of `target` — free to start
 * and end at any position in `target`. This is what makes a near-miss
 * embedded in filler as cheap to catch as a bare one: the filler before and
 * after the matched window costs nothing, only the edits inside it do.
 *
 * Standard technique (semi-global / "free end" alignment): row 0 is seeded to
 * all zeros instead of `0..n`, so aligning an EMPTY brand prefix against a
 * window ending at any column `j` is free — the window's start is not fixed.
 * The answer is the minimum over every column of the last row, not just the
 * final one — the window's end is not fixed either.
 *
 * UNBOUNDED BY DESIGN, not banded: both a domain label and a token above the
 * public suffix are capped at 63 characters (DNS), so the full `brand.length
 * x target.length` matrix is at most 63x63 — thousands of cells, not the
 * scan-of-30000-rows cost this module is careful about elsewhere. Banding for
 * a saving that small would be a second thing to get wrong.
 */
function minWindowDistance(target: string, brand: string): number {
  const m = brand.length;
  const n = target.length;
  if (m === 0) return 0;
  let prevPrev = new Array<number>(n + 1).fill(0);
  let prev = new Array<number>(n + 1).fill(0); // row 0: free start, every end column costs 0
  for (let i = 1; i <= m; i++) {
    const curr = new Array<number>(n + 1);
    curr[0] = i; // brand[0..i) against an empty window: i insertions
    for (let j = 1; j <= n; j++) {
      const cost = brand[i - 1] === target[j - 1] ? 0 : 1;
      let v = Math.min(prev[j]! + 1, curr[j - 1]! + 1, prev[j - 1]! + cost);
      if (i > 1 && j > 1 && brand[i - 1] === target[j - 2] && brand[i - 2] === target[j - 1]) {
        v = Math.min(v, prevPrev[j - 2]! + 1);
      }
      // I3-1104 (I23-W1-TRUST hand-off 8): the same LOOKALIKE_PAIRS op `editDistance` has (cost 1, either direction),
      // so the creation gate no longer passes `rnirvo` against a short verified brand the report scan flags.
      for (const [pair, single] of LOOKALIKE_PAIRS) {
        if (i > 1 && target[j - 1] === single && brand[i - 2] === pair[0] && brand[i - 1] === pair[1]) v = Math.min(v, prevPrev[j - 1]! + 1);
        if (j > 1 && brand[i - 1] === single && target[j - 2] === pair[0] && target[j - 1] === pair[1]) v = Math.min(v, prev[j - 2]! + 1);
      }
      curr[j] = v;
    }
    prevPrev = prev;
    prev = curr;
  }
  let best = Infinity;
  for (let j = 0; j <= n; j++) if (prev[j]! < best) best = prev[j]!;
  return best;
}

/** A listed name reduced to the same skeleton alphabet the domain side uses, so
 * a handle (`@netflix`, `net_flix`) and a display name (`Net Flix`) compare
 * against a domain label. `labelSkeleton` drops the non-alphanumerics, which is
 * exactly the folding a domain label already gets. */
export function nameSkeleton(raw: string): string {
  return labelSkeleton(raw.trim().toLowerCase()).skeleton;
}

/** Is `target` the listed host itself, or a subdomain of it?
 *
 * THIS IS NOT `regDomain === regDomain`, and the difference is the whole point.
 * The previous version compared computed registrable domains, and the curated
 * public-suffix heuristic collapses every registrant of a private registry into
 * one: `bob.uk.com` and `alice.uk.com`, `bob.github.io` and `alice.github.io`,
 * every `.com.co` and every `*.pages.dev`. First creator to self-add locked out
 * every other tenant of that registry, permanently, with a sentence that names
 * nothing (adversarial review, HIGH). Comparing the ACTUAL hosts keeps the case
 * the rule was written for — a stranger attaching `shop.netflix.com` when
 * `netflix.com` is listed to somebody else — and drops the co-tenants. */
export function isUnderListedHost(target: string, listed: string): boolean {
  // BOTH DIRECTIONS. `shop.netflix.com` when somebody else holds `netflix.com`
  // is the obvious one; `example.com` when somebody else holds
  // `blog.example.com` is the same conflict seen from the other end, and a
  // one-directional rule would clear it.
  return target === listed || target.endsWith(`.${listed}`) || listed.endsWith(`.${target}`);
}

/** Read one arm to exhaustion, a page at a time, REMEMBERING the first hit and
 * still reading every page (see the body — the page count must not answer
 * whether or where the host matched). Returns the first hit, or `"exhausted"`, or `"capped"` when the page budget ran out
 * before the rows did — which BLOCKS, because a gate that silently compared
 * against a prefix of the world is indistinguishable from one that found
 * nothing. */
/**
 * The precomputed brand list for a row, or null to fold it live.
 *
 * THE INPUT CHECK IS THE WHOLE SAFETY ARGUMENT. An entry is used only when it
 * says it was computed from exactly the value this row currently holds, so a
 * cache that is stale, orphaned, truncated or hand-edited cannot change a
 * verdict — it can only fail to save work.
 */
function cachedBrands(input: string | null | undefined, skeletons: string | null | undefined, host: string): string[] | null {
  if (typeof input !== "string" || typeof skeletons !== "string" || input !== foldInput(host)) return null;
  try {
    const v: unknown = JSON.parse(skeletons);
    return Array.isArray(v) && v.every((x) => typeof x === "string") ? (v as string[]) : null;
  } catch {
    return null;
  }
}

/** The precomputed name skeleton, or null to fold it live. Same input check,
 * same reason. */
function cachedName(input: string | null | undefined, skeletons: string | null | undefined, name: string): string | null {
  if (typeof input !== "string" || typeof skeletons !== "string" || input !== foldInput(name)) return null;
  try {
    const v: unknown = JSON.parse(skeletons);
    return typeof v === "string" ? v : null;
  } catch {
    return null;
  }
}

export async function scanPages<R>(
  page: (cursor: string | null) => Promise<R[]>,
  cursorOf: (row: R) => string,
  hit: (row: R) => GateHit | null,
): Promise<GateHit | "exhausted" | "capped"> {
  // THE PAGE COUNT MUST NOT DEPEND ON THE HIT (QUEUE #428, D-0764). Returning at
  // the first hit made the number of statements answer WHERE in the listed world
  // the host matched, and whether it matched at all. So the first hit is kept
  // and the scan keeps paging (evaluating `hit` on every row) until the rows
  // run out or the cap: the count is a function of the population alone. A hit
  // found before the cap still beats "capped" — a match is a match.
  let cursor: string | null = null;
  let found: GateHit | null = null;
  for (let p = 0; p < DOMAIN_GATE_MAX_PAGES; p++) {
    const rows = await page(cursor);
    // D-0810(13): the WORK must not depend on the hit either. `hit` runs on
    // every row; the first non-null is kept (B27-RG-CLIENTS probe 2: skipping
    // it after a hit made a miss ~76x slower than a hit over 6,000 rows).
    for (const r of rows) {
      const h = hit(r);
      if (!found && h) found = h;
    }
    if (rows.length < DOMAIN_GATE_PAGE) return found ?? "exhausted";
    cursor = cursorOf(rows[rows.length - 1]!);
  }
  return found ?? "capped";
}

/**
 * THE GATE. Blocks unless the domain resembles nothing listed under anybody
 * else's passport.
 *
 * Two arms:
 *
 *   DOMAINS — every domain account on an un-retracted passport at ANY claim
 *   state and ANY account status. Unclaimed is included because unclaimed is the
 *   DEFAULT state of a seeded creator, so excluding it hid most of the registry
 *   from the gate while the laundering payoff — a `confirmed` lookalike on the
 *   ATTACKER'S verified passport — did not depend on the victim's state at all
 *   (adversarial review, HIGH). Historical rows are included because a name that
 *   used to be theirs is the recycling case.
 *
 *   NAMES — every un-retracted passport's handles and display names, for the
 *   brand whose identity here is a handle rather than a domain, which is most of
 *   them. Stricter thresholds (see the constants): a display name is not a
 *   namespace, and a four-character one has an edit-distance neighbourhood full
 *   of ordinary words. Unclaimed subjects are matched on IDENTITY ONLY — see the
 *   note at the row predicate for why that particular line.
 *
 * ONLY THE REGISTRABLE LABEL IS USED AS A BRAND, not every token of a listed
 * host — unless that label is too short to compare, in which case the tokens are
 * the fallback (a brand LISTED as `netflix.uk.com`). Harvesting every token from
 * the brand side made generic words into global brands: one listed
 * `shop.gymshark.com` made `shop` a brand and refused ~141 English words for
 * every creator on the platform, and an attacker could weaponise it by adding
 * one domain whose label carried ten generic tokens (adversarial review, HIGH).
 *
 * Rows belonging to `passportId` itself are skipped throughout: attaching a
 * second domain that looks like your OWN first one is the ordinary case this
 * door was built for.
 */
/**
 * F14 / HANDOFF §1.2 — apexes this passport is authorized to sit UNDER.
 *
 * EMPTY FOR EVERYBODY BY DEFAULT, which is what keeps "the gate is not widened"
 * literally true: with an empty set this function is byte-for-byte the gate it
 * was, for every existing caller. The set is computed from the passport's own
 * active grants (`agencies.ts agencyGatingApexes`) and never handed in by a
 * caller.
 *
 * WHAT IT RELAXES, EXACTLY. Two things, both conditioned on `host` being a
 * STRICT SUBDOMAIN of an apex in the set:
 *
 *   1. THE SUBDOMAIN RULE, for the row that IS that apex. `isUnderListedHost`
 *      is what refuses `nova.talentco.com` because `talentco.com` is listed to
 *      somebody else, and the authorization is precisely the missing evidence.
 *
 *   2. A MATCH ON THE APEX'S OWN BRAND, ON THE APEX HOLDER'S OWN ROWS. The first
 *      draft relaxed rule 1 alone and the door stayed shut: `nova.talentco.com`
 *      CONTAINS `talentco`, so the brand arm refused the identical host one
 *      predicate later, with `kind: "same"` instead of `kind: "subdomain"`. A
 *      relaxation that cannot let its own motivating case through is not a
 *      relaxation, it is a comment.
 *
 * THE SECOND DRAFT NEUTRALISED THE BRAND STRING GLOBALLY, AND THAT WAS A HIGH.
 * Skeletons are a SHARED namespace: `brandsOfHost` yields `netflix` for
 * `netflix.io` as readily as for `netflix.com`, and `labelSkeleton` exists
 * precisely to collapse `pаypal.com` (Cyrillic а, a ten-dollar registration)
 * onto `paypal`. An adversarial review executed it end to end — agency proves
 * `xn--pypal-4ve.com`, creator signs `manage_accounts`, creator adds
 * `login.xn--pypal-4ve.com`, and the gate went `{blocked: true, matched:
 * "paypal.com"}` → `{blocked: false}`, landing a homoglyph login host on a
 * verified passport. The NAMES arm was worse: an unrelated creator whose
 * DISPLAY NAME is "Netflix", holding no domain at all, was neutralised too.
 * That inverts this gate's own thesis — its docblock says DNS control "was
 * never the question worth asking", and a bare DNS proof was switching off a
 * third party's brand across a whole zone.
 *
 * SO THE EXCUSE IS ANCHORED TO THE APEX HOLDER, not to the string. In the
 * domain arm the brand is ignored only on the row that IS the authorized apex;
 * in the names arm only on rows belonging to the passport that LISTS it. The
 * motivating case survives untouched — `talentco.com` is listed, and its
 * holder's display name is "Talent Co", which is exactly why both arms needed
 * the excuse — while `netflix.io` buys nothing, because the `netflix.com` row
 * and the "Netflix" display name belong to somebody else. An apex nobody has
 * listed anchors nothing and neutralises nothing.
 *
 * And because the set only applies when `host` really does end in `.<apex>`,
 * `nova-talentco.com` and `talentco.evil.com` — lookalikes rather than
 * subdomains — get an empty set and are refused with the authorization exactly
 * as they are without it.
 */
export async function domainAddGate(db: DbLike, host: string, passportId: string, underApexes: readonly string[] = []): Promise<GateVerdict> {
  const t = gateSkeletons(host);
  if (!t) return { blocked: true, reason: "unparseable" };
  // QUEUE #104: a token that folded to nothing in a script that can spell a
  // Latin brand is a name this gate cannot read — see `emptyFoldAdmitted`.
  // Decided by the caller's OWN input before any statement, so it answers
  // nothing about the listed world (D-0764's property is about the scans).
  if (t.unreadable) return { blocked: true, reason: "unparseable" };

  // ONE ANALYSIS PER TARGET STRING, FOR THE WHOLE SCAN. The segmentation passes
  // depend on the label and not on the brand, and this scan tests thousands of
  // brands against the same handful of target strings — so the memo is what
  // makes the seam test affordable at all. Built on demand: a gate call that
  // never reaches the containment arm never segments anything.
  // NO SEAM TEST FOR A REWRITTEN HOST, and this is the direction the fold has to
  // run in. `labelSkeleton` maps Cyrillic `е` to `e`, strips accents and folds
  // leet digits, EXPRESSLY so a homoglyph label collapses onto the brand it
  // imitates and can be caught. Handing that collapsed string to the seam test
  // turns the same fold into evidence of innocent English: `gеtslacksupport.com`
  // in Cyrillic, and `g3tslacksupport.com` in leet, both skeletonise to
  // `getslacksupport` and were cleared as `gets│lack│support` (adversarial
  // review, MEDIUM). Whoever typed a Cyrillic е did not compound two English
  // words. Folding must only ever ADD suspicion, so a rewritten host gets the
  // strict rule.
  const seam: SeamTest | undefined = t.confusable ? undefined : makeSeamTest();

  // THE APEXES THIS HOST IS ACTUALLY UNDER, and the brand strings they yield.
  // Both are empty unless `host` is a STRICT subdomain of a proven apex, which
  // is what confines everything below to the one case F14 exists for. The apex
  // itself is never in `underHere` (`t.host !== a`), so the apex is never
  // addable through this door — an authorization to sit under a name is not an
  // authorization to take it.
  // `endsWith("." + a)` IS what makes the apex unreachable: a host equal to the
  // apex does not end in `.<apex>`. An earlier draft also spelled out `t.host
  // !== a` and the docblocks credited apex-safety to THAT clause; a mutation
  // pass proved it dead over 400,169 host/apex pairs (zero behavioural
  // difference), so the clause is gone and the credit is written where it
  // belongs — a comment that names the wrong guard is how a real guard gets
  // deleted later by someone tidying up.
  const underHere = underApexes
    .map((a) => a.trim().toLowerCase().replace(/\.$/, ""))
    .filter((a) => a.length > 0 && t.host.endsWith(`.${a}`));
  const authorizedBrands = new Set<string>(underHere.flatMap((a) => brandsOfHost(a)));
  // WHO HOLDS THE APEX. The names arm has no host to compare, so the anchor
  // there is the PASSPORT that lists one of these apexes — the only party whose
  // display name and handles the authorization has anything to say about. One
  // statement, and only when there is an apex in play at all: a passport with no
  // agency never reaches it.
  // CHUNKED rather than listed as bounded: `underHere` is one entry per proven
  // apex this host sits under, and a 253-byte name admits ~126 label suffixes,
  // so nothing structural holds it under D1's 100-parameter ceiling.
  const apexHolders = new Set<string>();
  for (const chunk of underHere.length > 0 ? inChunks(underHere) : []) {
    const { results } = await db
      .prepare(`SELECT passport_id FROM passport_accounts WHERE id_kind = 'domain' AND id IN (${chunk.map(() => "?").join(",")})`)
      .bind(...chunk)
      .all<{ passport_id: string }>();
    for (const row of results) apexHolders.add(row.passport_id);
  }

  const domains = await scanPages<DomainRow>(
    async (cursor) =>
      (
        await db
          .prepare(
            // `bi_input` / `bi_skeletons` are the PRECOMPUTED fold (F13). The
            // input travels with them so the row-level function can check that
            // the cache still describes THIS row and fold live when it does
            // not — a stale entry is ignored, never trusted.
            `SELECT a.id AS bhost, a.account_key, a.passport_id,
                    b.input AS bi_input, b.skeletons AS bi_skeletons
               FROM passport_accounts a
               JOIN passports p ON p.passport_id = a.passport_id
          LEFT JOIN brand_index b ON b.kind = 'domain' AND b.key = a.account_key
              WHERE a.id_kind = 'domain' AND p.retracted = 0 AND a.passport_id <> ?
                AND a.account_key > IFNULL(?, '')
              ORDER BY a.account_key
              LIMIT ?`,
          )
          // I3-0853 — A SEEKABLE CURSOR. `(? IS NULL OR a.account_key > ?)` is
          // not a range the planner can seek, so every page re-walked from the
          // start; `> IFNULL(?, '')` is ('' is below every key). Rows read per
          // page fall to the page itself only with a domain-only index to seek
          // in — NEEDS-MIGRATION idx_accounts_domain_key (I4-05-R1 REPORT);
          // until it lands the PK walk reads what it read before.
          .bind(passportId, cursor, DOMAIN_GATE_PAGE)
          .all<DomainRow>()
      ).results,
    (r) => r.account_key,
    (r) => {
      // THE SUBDOMAIN RULE RUNS FIRST, and before any skeleton is required.
      // Previously a listed host whose label was too short to skeletonise
      // (`hp.com`, `x.com`, `qq.com`) was skipped entirely, taking this rule
      // with it — so `shop.hp.com` was addable by a stranger, which the docblock
      // called "the least subtle attack of the lot" (adversarial review, MEDIUM).
      const listed = r.bhost.trim().toLowerCase().replace(/\.$/, "");
      if (isUnderListedHost(t.host, listed)) {
        // AUTHORIZED TO SIT UNDER THIS APEX? Then the subdomain rule steps aside
        // for THIS row — the row that is the apex named in the authorization,
        // and no other. A row that is a DIFFERENT subdomain of the same apex is
        // not an authorization, and neither is a row that merely resembles one.
        if (!underHere.includes(listed)) return { kind: "subdomain", matched: r.bhost, matched_passport_id: r.passport_id };
      }
      // THE INDEX, OR THE FOLD. `cachedBrands` returns the precomputed list
      // only when the cached input is this row's current host; anything else —
      // missing, stale, truncated, unparseable — folds live. So the ceiling
      // moves and the verdict does not.
      for (const bs of cachedBrands(r.bi_input, r.bi_skeletons, r.bhost) ?? brandsOfHost(r.bhost)) {
        // The apex's OWN brand, on the apex's OWN row, is not evidence against a
        // host inside that apex's zone. `underHere.includes(listed)` is the
        // anchor and it is load-bearing: without it, proving `netflix.io`
        // neutralised the brand `netflix` against the listed `netflix.com` too
        // (review, HIGH). Every other row is tested exactly as before.
        if (underHere.includes(listed) && authorizedBrands.has(bs)) continue;
        const kind = gateMatch(t.all, bs, "domain", t.slack, seam);
        if (kind) return { kind, matched: r.bhost, matched_passport_id: r.passport_id };
      }
      return null;
    },
  );
  // BOTH SCANS RUN ON EVERY ADD (QUEUE #428, D-0764). The names scan used to
  // run only when the domains scan came back clean, so a host resembling a
  // listed DOMAIN cost one statement fewer than every other host — at every door
  // that calls this gate — and that statement answered the very bit the uniform
  // refusal withholds. The domains verdict is decided FIRST below, exactly as
  // before; only the early return is gone. `w309-regression-j02-uniform-refusal`
  // pins the delta at 0 and `b27-domain-gate-uniform-scan` pins it per arm.
  const names = await scanPages<NameRow>(
    async (cursor) =>
      (
        await db
          .prepare(
            // I3-0467 — KEYSET ON THE TUPLE, LIMITED PER ARM. The key used to be
            // the computed string `passport_id || '|' || name` over the union,
            // which no index serves, so every page re-materialised and re-sorted
            // the whole union. Now each arm seeks its own index in
            // (passport_id, name) order and stops at the page size; the next
            // page of the union is inside the next page of each arm, so the
            // outer sort sees at most two pages. `display_name`/`id` are NOT
            // NULL, so the row-value comparison is total.
            `SELECT n.name AS name, n.passport_id AS passport_id, n.claim_state AS claim_state,
                    b.input AS bi_input, b.skeletons AS bi_skeletons
               FROM (
               SELECT * FROM (
               SELECT p.display_name AS name, p.passport_id AS passport_id, p.claim_state AS claim_state
                 FROM passports p
                WHERE (p.passport_id, p.display_name) > (?, ?) AND p.retracted = 0 AND p.passport_id <> ?
                ORDER BY p.passport_id, p.display_name
                LIMIT ?)
               UNION ALL
               SELECT * FROM (
               SELECT a.id AS name, a.passport_id AS passport_id, p2.claim_state AS claim_state
                 FROM passport_accounts a
                 JOIN passports p2 ON p2.passport_id = a.passport_id
                WHERE (a.passport_id, a.id) > (?, ?) AND a.id_kind = 'handle'
                  AND p2.retracted = 0 AND a.passport_id <> ?
                ORDER BY a.passport_id, a.id
                LIMIT ?)
             ) n
          LEFT JOIN brand_index b ON b.kind = 'name' AND b.key = n.passport_id || '|' || n.name
             ORDER BY n.passport_id, n.name
             LIMIT ?`,
          )
          .bind(...nameCursor(cursor), passportId, DOMAIN_GATE_PAGE, ...nameCursor(cursor), passportId, DOMAIN_GATE_PAGE, DOMAIN_GATE_PAGE)
          .all<NameRow>()
      ).results,
    (r) => JSON.stringify([r.passport_id, r.name]),
    (r) => {
      const bs = cachedName(r.bi_input, r.bi_skeletons, r.name ?? "") ?? nameSkeleton(r.name ?? "");
      // Same neutralisation as the domain arm, anchored the same way. It IS
      // needed here — an agency called "Talent Co" has a display name that folds
      // to the same `talentco` its apex does, so relaxing only the domain arm
      // would refuse the roster subdomain on the agency's own name instead — but
      // only for the passport that actually LISTS the apex. Unanchored, a
      // stranger's display name "Netflix" was neutralised by a proof of
      // `netflix.io` (review, HIGH).
      if (apexHolders.has(r.passport_id) && authorizedBrands.has(bs)) return null;
      const kind = gateMatch(t.all, bs, r.claim_state === "verified" ? "verified-name" : "name", t.slack, seam);
      if (!kind) return null;
      // UNCLAIMED SUBJECTS GET IDENTITY MATCHING AND NOTHING ELSE.
      //
      // The population matters in both directions. Unclaimed is the DEFAULT
      // state of a seeded creator, so an unclaimed handle-only creator — the
      // largest group in the registry — had no protection at all here, while the
      // harm does not depend on their state: the laundered name lands on the
      // ATTACKER'S verified passport and inherits the ATTACKER'S standing
      // (adversarial review, HIGH). But that group is also, by volume, most of
      // the names in the world, and giving thousands of unvetted display names a
      // containment and edit-distance neighbourhood each would refuse a large
      // slice of the namespace on the strength of names nobody has checked.
      //
      // So an unclaimed name matches only when the WHOLE registrable label of the
      // domain is that name — `bigcreator.com` against a listed `@bigcreator`,
      // and not `bigcreator-shop.com`, which a whole-TOKEN identity would also
      // have caught. One name, one domain, one refusal. Everything wider stays
      // on the claimed and verified rungs, where a human or a proof has been
      // involved.
      if (r.claim_state === "unclaimed" && bs !== t.label) return null;
      return { kind, matched: r.name, matched_passport_id: r.passport_id };
    },
  );
  if (domains === "capped") return { blocked: true, reason: "candidate_cap", arm: "domain" };
  if (domains !== "exhausted") return { blocked: true, reason: "resembles", ...domains };
  if (names === "capped") return { blocked: true, reason: "candidate_cap", arm: "name" };
  if (names !== "exhausted") return { blocked: true, reason: "resembles", ...names };

  return { blocked: false };
}

/**
 * The brand strings one listed host contributes. TWO, at most, and NEITHER is a
 * hyphen-token:
 *
 *   - the registrable label (`netflix.com` → `netflix`, `my-brand.co` →
 *     `mybrand`), and
 *   - the LEFTMOST label when it is six characters or more
 *     (`mrbeast.somehost.com` → `mrbeast`, `netflix.uk.com` → `netflix`).
 *
 * The second exists because a creator's name very often lives in a subdomain —
 * on a hosting provider, or under a private registry the curated suffix set does
 * not know — and label-only left exactly those creators unprotected while
 * over-protecting their host: with `mrbeast.somehost.com` listed,
 * `mrbeast-store.com` was addable and `somehost-support.com` was refused for
 * everybody (adversarial review, MEDIUM). Six characters is the floor because
 * that is where a generic label stops being generic.
 *
 * A LISTED SUBDOMAIN ALSO YIELDS ITS REGISTRABLE LABEL (I16 RJb-T2-J-02, D-0920),
 * at the same six-character floor, UNLESS its registrable domain is a listed
 * hosting provider (`HOSTING_PROVIDERS`), where only the leftmost-label rule
 * applies. Before this, `shop.gymshark.com` yielded `[]` — the registrable label
 * was skipped because the host is not its own apex, and `shop` is under the
 * floor — so `gymsharkoutlet.example` was addable while Gymshark was listed only
 * at `shop.gymshark.com`. The provider list is what tells a creator's own
 * domain from their host's: `alice.github.io` still does not make `github` a
 * brand. A provider missing from the list makes its label a brand for everybody
 * (the over-inclusive direction), which is the cost of a curated list and is
 * fixed by adding the provider.
 *
 * WHAT IS DELIBERATELY ABSENT IS TOKEN SPLITTING. An earlier draft fell back to
 * every hyphen/dot token whenever the registrable label was too short, and that
 * fallback was attacker-reachable: a private registry (`.uk.com`, `.com.co`,
 * `.eu.org` — all publicly registrable for a few dollars) yields a two-character
 * registrable label, so adding one domain named
 * `shop-store-music-photo-news-media-studio-fitness.uk.com` turned eight generic
 * words into global brands and refused that slice of the namespace for every
 * creator on the platform — at ten adds a day, and never blocking the poisoner,
 * whose own rows are skipped (adversarial review, HIGH). With no splitting, that
 * label contributes one 41-character string that matches nothing.
 */
/**
 * THE FOLD VERSION — bump this whenever any function feeding `brandsOfHost` or
 * `nameSkeleton` changes.
 *
 * `brand_index` caches the fold and checks that the cached INPUT still matches
 * the row. That covers stale DATA and misses stale ALGORITHM entirely, which is
 * a FAIL-OPEN in a fail-closed gate and was found by review before it shipped:
 * ship a fix that widens what counts as a brand, and for up to thirty days the
 * gate serves the OLD narrower fold for every already-indexed row — so the fix
 * silently does not take effect for exactly the rows it was written for. The
 * gate becomes weaker WITH the cache than without it, at the moment somebody
 * tightens it. Slices 32→34 changed these functions four times in a row, each
 * in response to an adversarial finding, so this is the normal case and not a
 * hypothetical.
 *
 * The version is composed into the stored input, so a bump invalidates every
 * entry at once and they refold over the next couple of sweeps.
 * `brand-fold-version.test.ts` pins the fold's source bytes and fails until this
 * is bumped — a version nobody is forced to change is a version nobody changes.
 */
export const BRAND_FOLD_VERSION = "7"; // "7": I20-W2C #946 (b) — the shared name-fold rows moved into labelSkeleton (nameSkeleton now folds a decoded xn-- label too)

/** The stored/compared key for a cached fold: the version and the input it was
 * computed from, so neither can drift without the entry being ignored. */
export function foldInput(input: string): string {
  return `${BRAND_FOLD_VERSION}\u0000${input}`;
}

/**
 * The brand list for a host, from the host alone.
 *
 * The one function `brand-index.ts` caches and the scan consults, so the cached
 * value and the live value are produced by the SAME code path — an index whose
 * contents were computed a different way would be a second implementation of a
 * security decision, quietly drifting.
 *
 * Returns `[]` for a host with no registrable form, which is what `gateSkeletons`
 * returning null means to the caller: nothing to compare, and the subdomain rule
 * (which does not need skeletons) still runs.
 */
export function brandsOfHost(host: string): string[] {
  const b = gateSkeletons(host);
  return b ? brandsOf(b) : [];
}

/**
 * HOSTING PROVIDERS (I16 RJb-T2-J-02, D-0920): registrable domains whose
 * subdomains belong to the provider's CUSTOMERS, so a listed customer subdomain
 * never makes the provider's own label a brand. Curated in-repo, exact
 * registrable domains (what `registrable()` returns), no wildcards. */
export const HOSTING_PROVIDERS: ReadonlySet<string> = new Set([
  "github.io", "gitlab.io", "bitbucket.io", "netlify.app", "vercel.app", "pages.dev", "workers.dev",
  "herokuapp.com", "fly.dev", "onrender.com", "glitch.me", "replit.app", "web.app", "firebaseapp.com",
  "appspot.com", "azurewebsites.net", "cloudfront.net", "amplifyapp.com", "myshopify.com", "wixsite.com",
  "squarespace.com", "wordpress.com", "blogspot.com", "tumblr.com", "substack.com", "medium.com",
  "ghost.io", "beehiiv.com", "webflow.io", "framer.website", "framer.app", "carrd.co", "notion.site",
  "weebly.com", "godaddysites.com", "square.site", "jimdosite.com", "mystrikingly.com", "bandcamp.com",
  "teachable.com", "thinkific.com", "kajabi.com", "gumroad.com", "neocities.org", "typepad.com",
]);

function brandsOf(b: GateHost): string[] {
  const out: string[] = [];
  // THE REGISTRABLE LABEL IS A BRAND ONLY WHEN THE LISTED HOST IS ITS APEX.
  // When the listed host is a strict subdomain, that registrable domain belongs
  // to whoever runs the hosting, not to the creator — and making it a brand
  // hands one provider's name to the gate as a global refusal: one creator
  // listed at `alice.github.io` would refuse `github-tools.com` and
  // `mygithubpages.com` for everybody, which is the token-harvesting harm
  // arriving through a different door. The cost is losing a brand when a real
  // brand is listed ONLY by a subdomain and never by its apex, which is the
  // unusual case; the apex row, when it exists, supplies the brand.
  if (b.host === b.regDomain && b.label.length >= GATE_MIN_BRAND) out.push(b.label);
  // …and a SUBDOMAIN of a registrable domain that is not a listed hosting provider
  // yields it too, at the six-character floor (D-0920; see the docblock above).
  else if (b.host !== b.regDomain && b.label.length >= 6 && !HOSTING_PROVIDERS.has(b.regDomain)) out.push(b.label);
  const labels = b.host.split(".").filter((l) => l.length > 0);
  if (labels.length > 1) {
    const left = gateLabelSkeleton(labels[0]!).skeleton;
    if (left.length >= 6 && left !== b.label) out.push(left);
  }
  return out;
}

// #116 (I16, D-0920) — SAME-SCRIPT CONFUSABLES.
//
// Everything above compares a label's LATIN skeleton, so two names written in
// the same non-Latin script — `中國.com` against a claimed `中国.com`, `カ` for
// `力`, `ज़` for `ज` — fold to "" and are never compared with each other. This
// is the second comparison D-0920 rules: a candidate's registrable label, folded
// within its own script family by the in-repo tables below (no dependency), is
// compared for EQUALITY against the registrable labels of IDN domains on
// CLAIMED, unretracted passports other than the caller's. A hit is a LANE
// signal, never a refusal: `verifyBrandEnrol` routes it to `scrutiny`, where a
// person reads both names. The tables are deliberately small and visual; a pair
// missing from them is a miss in the over-permissive direction and is fixed by
// adding the pair, while a pair wrongly present costs an operator a look.
// ---------------------------------------------------------------------------

/** The same-script comparison key of ONE label, or null when the label is not
 * written wholly in one supported family (a Latin label, a mixed label, an
 * unsupported script: the Latin skeleton arms above own those). */
export function sameScriptKey(label: string): { script: SameScript; key: string } | null {
  const cps = [...toUnicodeLabel(label).normalize(UNICODE_NFKD)];
  let family: SameScript | null = null;
  const out: string[] = [];
  for (const ch of cps) {
    if (SS_NEUTRAL.test(ch)) {
      if (/[0-9]/.test(ch)) out.push(ch);
      continue;
    }
    // A combining mark NFKD split off (the kana voicing marks U+3099/U+309A) is
    // kept as written: ガ stays distinct from カ.
    if (INHERITED_MARK.test(ch)) {
      out.push(ch);
      continue;
    }
    const f = SS_FAMILY.find(([, re]) => re.test(ch))?.[0];
    if (f === undefined || (family !== null && f !== family)) return null;
    family = f;
    let c = ch;
    if (f === "han-kana") {
      const code = c.codePointAt(0)!;
      if (code >= 0x3041 && code <= 0x3096) c = String.fromCodePoint(code + 0x60); // hiragana -> katakana
      c = KANA_FOLD.get(c) ?? c;
      c = HAN_FOLD.get(c) ?? c;
    } else if (f === "hangul") c = HANGUL_FOLD.get(c) ?? c;
    else c = DEVANAGARI_FOLD.get(c) ?? c;
    out.push(c);
  }
  if (family === null || out.length === 0) return null;
  return { script: family, key: out.join("") };
}

/** Rows read by one same-script scan. Past it the scan does not answer "no":
 * it returns `cap`, which the enrolment door also routes to `scrutiny`. */
export const SAME_SCRIPT_SCAN_MAX = 5000;

export type SameScriptVerdict =
  | { hit: false }
  | { hit: true; reason: "same_script"; script: SameScript; matched: string; matched_passport_id: string }
  | { hit: true; reason: "cap" };

/** Compare `host`'s registrable label, within its script family, against the
 * registrable labels of IDN (`xn--`) domains on claimed, unretracted passports
 * other than `passportId`. ONE statement for every caller, whatever the
 * candidate's script, so the round-trip count carries nothing. */
export async function sameScriptGate(db: DbLike, host: string, passportId: string): Promise<SameScriptVerdict> {
  const h = host.trim().toLowerCase().replace(/\.$/, "");
  const reg = registrable(h);
  const mine = reg ? sameScriptKey(reg.label) : null;
  const { results } = await db
    .prepare(
      `SELECT a.id AS host, a.passport_id
         FROM passport_accounts a
         JOIN passports p ON p.passport_id = a.passport_id
        WHERE a.id_kind = 'domain' AND a.id LIKE '%xn--%' AND p.retracted = 0
          AND p.claim_state IN ('claimed','verified') AND a.passport_id <> ?
        ORDER BY a.account_key
        LIMIT ?`,
    )
    .bind(passportId, SAME_SCRIPT_SCAN_MAX + 1)
    .all<{ host: string; passport_id: string }>();
  if (mine === null) return { hit: false };
  if (results.length > SAME_SCRIPT_SCAN_MAX) return { hit: true, reason: "cap" };
  for (const r of results) {
    if (r.host === h) continue;
    const theirs = registrable(r.host);
    const k = theirs ? sameScriptKey(theirs.label) : null;
    if (k !== null && k.script === mine.script && k.key === mine.key) {
      return { hit: true, reason: "same_script", script: k.script, matched: r.host, matched_passport_id: r.passport_id };
    }
  }
  return { hit: false };
}

/** The number of trailing labels that form the public suffix for this host.
 * Exported for `domain-gate.ts`, which needs the same split to enumerate the
 * tokens above the suffix without going through `domainKey` (whose null-on-empty
 * -skeleton behaviour the gate deliberately does not want). */
export function publicSuffixLen(labels: string[]): number {
  if (labels.length >= 2) {
    const last = labels[labels.length - 1]!;
    const second = labels[labels.length - 2]!;
    // A 2-label suffix only when the TLD is a real 2-letter ccTLD (NOT a generic
    // 2-letter TLD) AND the second-to-last label is a known registry word.
    if (/^[a-z]{2}$/.test(last) && !GENERIC_2L_TLD.has(last) && CCTLD_SECOND_LEVEL.has(second)) return 2; // paypal.CO.UK; bare co.uk (len 2) → registrable() null

  }
  return 1; // single-label TLD (com, io, …) or unknown → treat the last label as the suffix
}

export interface Registrable {
  label: string; // the registrable label — the "brand" part (one below the public suffix)

}

/** The registrable domain + its label for a host. `login.paypal.co.uk` →
 * { label: "paypal", domain: "paypal.co.uk" }. Returns null when the host is
 * itself only a public suffix (no registrable label) or is empty/an IP. */
function normHost(host: string): string {
  return host.trim().toLowerCase().replace(/\.$/, "");
}

export function registrable(host: string): Registrable | null {
  const h = normHost(host);
  if (!h || h.length > 253) return null;
  if (/^\d{1,3}(\.\d{1,3}){3}$/.test(h)) return null; // an IPv4 literal is not a lookalike
  const labels = h.split(".").filter((l) => l.length > 0);
  if (labels.length < 2) return null; // a bare TLD / single label has no registrable domain
  const suffixLen = publicSuffixLen(labels);
  if (labels.length <= suffixLen) return null; // host is only the public suffix
  const label = labels[labels.length - suffixLen - 1]!;
  const domain = labels.slice(labels.length - suffixLen - 1).join(".");
  return { label, domain };
}

// --- Punycode (RFC 3492) decode — xn-- label → Unicode ------------------------
const B = 36, TMIN = 1, TMAX = 26, SKEW = 38, DAMP = 700, INITIAL_BIAS = 72, INITIAL_N = 128;

function digitFromBasic(cp: number): number {
  if (cp >= 0x30 && cp <= 0x39) return cp - 0x30 + 26; // 0-9 → 26..35
  if (cp >= 0x61 && cp <= 0x7a) return cp - 0x61; // a-z → 0..25
  if (cp >= 0x41 && cp <= 0x5a) return cp - 0x41; // A-Z → 0..25
  return B; // invalid
}

function adaptBias(delta: number, numPoints: number, firstTime: boolean): number {
  let d = firstTime ? Math.floor(delta / DAMP) : delta >> 1;
  d += Math.floor(d / numPoints);
  let k = 0;
  while (d > ((B - TMIN) * TMAX) >> 1) {
    d = Math.floor(d / (B - TMIN));
    k += B;
  }
  return k + Math.floor(((B - TMIN + 1) * d) / (d + SKEW));
}

/** Decode a single punycode label (WITHOUT the `xn--` prefix) to a Unicode
 * string. Bounded (labels are ≤ ~63 chars); throws on malformed input, which
 * the caller treats as "leave it opaque". */
export function punycodeDecode(input: string): string {
  if (input.length > 63) throw new Error("punycode label too long");
  const output: number[] = [];
  const basic = input.lastIndexOf("-");
  for (let j = 0; j < Math.max(0, basic); j++) {
    const cp = input.charCodeAt(j);
    if (cp >= 0x80) throw new Error("bad basic code point");
    output.push(cp);
  }
  let idx = basic < 0 ? 0 : basic + 1;
  let i = 0, n = INITIAL_N, bias = INITIAL_BIAS;
  while (idx < input.length) {
    const oldi = i;
    let w = 1;
    for (let k = B; ; k += B) {
      if (idx >= input.length) throw new Error("truncated");
      const digit = digitFromBasic(input.charCodeAt(idx++));
      if (digit >= B) throw new Error("bad digit");
      if (digit > Math.floor((0x7fffffff - i) / w)) throw new Error("overflow");
      i += digit * w;
      const t = k <= bias ? TMIN : k >= bias + TMAX ? TMAX : k - bias;
      if (digit < t) break;
      w *= B - t;
      if (w > 0x7fffffff) throw new Error("overflow");
    }
    const outLen = output.length + 1;
    bias = adaptBias(i - oldi, outLen, oldi === 0);
    n += Math.floor(i / outLen);
    i %= outLen;
    if (n > 0x10ffff) throw new Error("code point out of range");
    output.splice(i++, 0, n);
  }
  return String.fromCodePoint(...output);
}

/** Decode an `xn--` label to Unicode; pass other labels through unchanged.
 * A decode failure returns the original label (stays opaque, no fold). */
export function toUnicodeLabel(label: string): string {
  if (!label.startsWith("xn--")) return label;
  try {
    return punycodeDecode(label.slice(4));
  } catch {
    return label;
  }
}

/** Map a single Unicode code point to its ASCII skeleton char (or "" to drop). */
function foldCodePoint(cp: number): string {
  if (cp >= 0x61 && cp <= 0x7a) return String.fromCharCode(cp); // a-z
  if (cp >= 0x30 && cp <= 0x39) return String.fromCharCode(cp); // 0-9
  if (cp >= 0x41 && cp <= 0x5a) return String.fromCharCode(cp + 0x20); // A-Z → a-z
  if (cp >= 0xff41 && cp <= 0xff5a) return String.fromCharCode(cp - 0xff41 + 0x61); // fullwidth a-z
  if (cp >= 0xff21 && cp <= 0xff3a) return String.fromCharCode(cp - 0xff21 + 0x61); // fullwidth A-Z
  if (cp >= 0xff10 && cp <= 0xff19) return String.fromCharCode(cp - 0xff10 + 0x30); // fullwidth 0-9
  const m = CONFUSABLE[cp];
  if (m !== undefined) return m;
  return ""; // unmapped (hyphen, symbols, other scripts) → dropped from the skeleton
}

/** The Unicode normalization form, as a NAMED CONSTANT rather than a literal
 * argument. `tests/invariants/env-inventory.test.ts` scans this tree for any
 * single-argument call whose sole argument is a SHOUTY string literal — the
 * accessor-helper idiom an earlier review used to evade it — and demands that
 * every match be a classified env key named in `DEPLOY.md`. `normalize("NFD")`
 * matches that shape exactly and is not an env key. The net is deliberately
 * broad and worth keeping broad, so the call site moves rather than the guard.
 * Do not inline this back. */
const UNICODE_NFD = "NFD";

/**
 * The confusable skeleton of a domain label: decode any punycode, fold each
 * confusable code point to ASCII, fold the common leet substitutions, and drop
 * separators — so a homoglyph / leet / hyphen variant collapses to the same
 * comparable string as the brand it imitates. Returns { skeleton, confusable }
 * where `confusable` is true when the fold changed a non-[a-z0-9] source char
 * (i.e. this label used a homoglyph, not just plain ASCII).
 */
export function labelSkeleton(label: string, shareNameFold = true): { skeleton: string; confusable: boolean; dropped: number; leetDigits: number } {
  const raw0 = toUnicodeLabel(label);
  // `shareNameFold = false` is the fold WITHOUT the shared rows: domain-gate.ts judges `unreadable` on it, so a token
  // only the shared rows can read (an all-Lisu label) stays refused there (b27-domain-empty-fold.test.ts).
  let raw = shareNameFold ? sharedLatinFold(raw0).out : raw0;
  for (const [re, to] of LIGATURES) raw = raw.replace(re, to);
  // ACCENTS ARE STRIPPED, NOT DROPPED. `é` used to hit `foldCodePoint`'s default
  // and vanish, so `mícrósóft` skeletonised to `mcrsft` — SHORTER than the brand
  // by three, which puts it outside the edit-distance window rather than inside
  // it, and made an accented rewrite of any longer brand invisible to both this
  // module's consumers (adversarial review, MEDIUM/HIGH). NFD splits a composed
  // letter into base + combining mark; dropping only the marks leaves the base
  // letter, which is what the eye reads and what the comparison wants.
  const uni = raw.normalize(UNICODE_NFD).replace(/[\u0300-\u036f]/g, "");
  let confusable = false;
  for (const ch of raw0) if (ch.codePointAt(0)! > 0x7f) confusable = true;
  let s = "";
  let dropped = 0;
  for (const ch of uni) {
    // `confusable` is decided ABOVE, from the pre-normalization string: any
    // non-ASCII source code point marks the label as a confusable/IDN rewrite,
    // whether it folds, decomposes or is dropped. It cannot be read off the
    // normalized string, because stripping the marks is exactly what removes the
    // evidence.
    const folded = foldCodePoint(ch.codePointAt(0)!);
    // COUNT WHAT WAS DELETED. A dropped code point SHORTENS the skeleton, and a
    // shorter skeleton falls OUTSIDE an edit-distance window rather than inside
    // it — so deletion is how a homoglyph ESCAPES rather than how it is caught.
    // The 24 Latin look-alikes mapped above closed one proof-of-concept and the
    // next review produced another in Cherokee; enumerating scripts is a game
    // you lose by playing. Reporting the count lets a consumer widen its budget
    // by exactly the amount the fold destroyed, which closes the whole class
    // rather than the code points somebody thought of.
    if (folded === "") dropped++;
    s += folded;
  }
  // Leet digits fold to their letter look-alikes. NOTE: the ASCII sequence folds
  // rn→m / vv→w are DELIBERATELY NOT applied — `rn` occurs in a huge number of
  // ordinary words (modern, corner, return, …) so folding it collides distinct
  // real brands (modern↔modem) and floods the queue; the genuine `rnicrosoft`
  // typosquat is already caught by the edit-distance kind instead.
  // COUNT WHAT WAS SUBSTITUTED, not just what was dropped. `confusable` only
  // marks a non-ASCII source char, so an all-ASCII leet rewrite (`paypa1`,
  // `7eleven`) sails through as `confusable === false` even though it folded
  // just as deliberately as a homoglyph — a caller deciding how much weight an
  // exact-skeleton match deserves needs this count to tell "genuinely plain
  // ASCII" apart from "ASCII that got leet-folded to collide with something".
  const leetDigits = (s.match(/[01357]/g) ?? []).length;
  s = s.replace(/0/g, "o").replace(/1/g, "l").replace(/3/g, "e").replace(/5/g, "s").replace(/7/g, "t");
  return { skeleton: s, confusable, dropped, leetDigits };
}

export interface DomainKey {
  regLabel: string; // registrable label, raw (lowercased)
  regDomain: string; // registrable domain, raw
  skeleton: string; // confusable skeleton of the registrable label
  tokens: string[]; // skeleton of each hyphen/dot-separated token ABOVE the public suffix
  confusable: boolean; // the registrable label used a homoglyph
  // The two fields below were computed by `labelSkeleton` all along and dropped
  // on the floor here (E-20/E-19, S-049-batch): `dropped` never reached the
  // caution path's `maxD`, and `leetDigits` didn't exist for a caller to gate
  // `exact_lookalike` on. Exposing them doesn't close either gap by itself —
  // the consumer in reports.ts/caution.ts still has to read them — but that
  // consumer is out of this room's ownership (domain-match.ts only); see
  // REPORT.md OPEN QUESTIONS.
  dropped: number; // code points the skeleton fold DELETED (see labelSkeleton)
  leetDigits: number; // ASCII digits the skeleton fold SUBSTITUTED (see labelSkeleton)
}

/** Reduce a host to its comparison key: registrable label + confusable skeleton,
 * plus the set of hyphen/dot-separated TOKENS above the public suffix — so a brand
 * that appears as a distinct word in a bigger host (`paypal-login`, `secure.paypal`,
 * `paypal.com.evil`) is caught as a token, WITHOUT an arbitrary-substring match that
 * would flag `visalia` for brand `visa`. Returns null for a bare suffix / IP / junk. */
export function domainKey(host: string): DomainKey | null {
  const reg = registrable(host);
  if (!reg) return null;
  const { skeleton, confusable, dropped, leetDigits } = labelSkeleton(reg.label);
  if (!skeleton) return null;
  const h = host.trim().toLowerCase().replace(/\.$/, "");
  const labels = h.split(".").filter((l) => l.length > 0);
  const suffixLen = publicSuffixLen(labels);
  const tokens: string[] = [];
  for (const rawLbl of labels.slice(0, labels.length - suffixLen)) {
    // DECODE FIRST, THEN SPLIT. Splitting the raw label on "-" cut `xn--d1aqf`
    // into ["xn", "", "d1aqf"] and skeletonised punycode fragments as if they
    // were words, so an IDN host contributed noise tokens and its real name
    // contributed none (adversarial review, LOW).
    const lbl = toUnicodeLabel(rawLbl);
    for (const tok of lbl.split("-")) {
      const sk = labelSkeleton(tok).skeleton;
      if (sk) tokens.push(sk);
    }
  }
  return { regLabel: reg.label, regDomain: reg.domain, skeleton, tokens, confusable, dropped, leetDigits };
}

/**
 * I3-1104 (I23, D-0993(1) pick): the two-letter LOOKALIKE PAIRS a reader takes for one letter. "rn" reads as "m" and
 * "vv" as "w", so `rnicrosoft` for `microsoft` is ONE edit, not two (substitute + insert), and stays inside the
 * short-brand budget of 1. ONE table for the report scan (reports.ts `editDistance`) and the creation gate
 * (domain-gate.ts `minWindowDistance`). An edit OP, not a fold: `domain-match.ts` deliberately does not fold rn->m in the skeleton,
 * so `modern`/`modem` stay distinct skeletons (no new exact collision) and only the fuzzy arm sees the pair.
 */
export const LOOKALIKE_PAIRS: readonly (readonly [pair: string, single: string])[] = [["rn", "m"], ["vv", "w"]];
