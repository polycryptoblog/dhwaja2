import { build } from "esbuild";
import { mkdirSync, readFileSync, writeFileSync } from "node:fs";
import { dirname, join, resolve } from "node:path";
import { fileURLToPath } from "node:url";

/**
 * Bundle the creator console into a single self-contained module (dist/app.js)
 * plus the static index.html. @dhwaja/canonical is inlined — no remote code,
 * no CDN. API_BASE is injected at build time and VALIDATED as a plain https
 * URL first (an unvalidated value with a quote could break out of the define
 * string; a user-shaped value could point the console at a hostile origin).
 */

const root = dirname(fileURLToPath(import.meta.url));
/**
 * THE CHOKEPOINT. Under a test runner, building into `dist/` is REFUSED
 * outright — not warned about, not tolerated. `pnpm test` used to run four of
 * these builds with no env on every invocation, so a real
 * `OPS_BASE=… node build.mjs` followed by the suite left the deploy artifact
 * rebuilt from the `dhwaja.example` defaults (exit 0, no warning), and the
 * console's `connect-src` named a domain we do not own. A test that wants a
 * build must say where to put it.
 */
if (process.env.VITEST && !process.env.BUILD_OUT_DIR) {
  throw new Error(
    "refusing to build into dist/ under a test runner: dist/ is the deploy artifact. " +
      "Set BUILD_OUT_DIR to a temp directory.",
  );
}

/**
 * OUTPUT DIRECTORY. `dist/` is the DEPLOY ARTIFACT — the checklist is
 * build-then-deploy and `wrangler pages deploy` ships exactly what is in it —
 * so nothing may write there implicitly. A caller that is not deploying MUST
 * pass BUILD_OUT_DIR and gets a directory of its own.
 *
 * This exists because `pnpm test` used to silently rebuild `dist/` from the
 * `dhwaja.example` defaults: a real `OPS_BASE=... node build.mjs` followed by
 * the test suite left the shipped bundle pointed at a domain we do not own,
 * exit 0, no warning. tests/invariants/build-artifacts.test.ts pins that a
 * test can never write into an app's dist/ again.
 */
const out = process.env.BUILD_OUT_DIR
  ? resolve(process.env.BUILD_OUT_DIR)
  : join(root, "dist");

const apiBase = process.env.API_BASE ?? "https://api.dhwaja.example";
const URL_RE = /^https:\/\/[a-z0-9.-]+(:\d+)?(\/[^\s"'`]*)?$/i;
if (!URL_RE.test(apiBase)) throw new Error(`API_BASE must be a plain https URL, got: ${apiBase}`);

/**
 * PUBLIC_BASE — the origin a FAN uses, which is not the API origin.
 *
 * The onboarding wizard tells a creator to paste `<base>/p/<passport>` "in every
 * bio", and it built that string from API_BASE. On any normal deployment those
 * are different hosts (`api.dhwaja.…` and the public site), so the product was
 * instructing creators to publish an api-subdomain URL across every platform
 * they are on, permanently — in the one place the whole "the link IS the proof"
 * story rests. The console also has no recovery route of its own and needs to
 * hand a locked-out creator a link to the fan app's, which lives on the same
 * public origin.
 *
 * Validated exactly like API_BASE (a plain https URL, so nothing user-shaped can
 * become a link target), and it DEFAULTS to API_BASE: a single-origin deployment
 * keeps working with no new variable, and the placeholder refusal below covers
 * it either way because it checks the value that actually gets compiled in.
 */
const publicBase = process.env.PUBLIC_BASE ?? apiBase;
if (!URL_RE.test(publicBase)) throw new Error(`PUBLIC_BASE must be a plain https URL, got: ${publicBase}`);

/**
 * THE SECOND CHOKEPOINT: a PLACEHOLDER build may never land in `dist/`.
 *
 * `dhwaja.example` is a reserved example domain nobody owns. A bundle built
 * against it cannot reach the API or the registry, so on a real deploy EVERY
 * check it makes fails — and a failed check is the state a user reads the least
 * clearly.
 *
 * The VITEST chokepoint above stops a TEST from overwriting the deploy artifact.
 * This one stops a developer from filling it with an unusable one: `node
 * build.mjs` with no env is a perfectly reasonable thing to run, it just may not
 * produce the thing DEPLOY.md ships. Anywhere else — a preview, a test, a local
 * sandbox — `BUILD_OUT_DIR` gets its own directory and the defaults are fine,
 * which is why this is scoped to `dist/`.
 *
 * IT RUNS BEFORE `mkdirSync`, DELIBERATELY. Refusing after the directory exists
 * leaves an EMPTY `dist/` behind — which reds the test suite and, worse, is a
 * directory `wrangler pages deploy` will happily publish as an empty site
 * (audit v10, B5). The refusal must write nothing at all.
 *
 * This block used to exist only in `apps/web` while five sibling builds carried
 * the same comment describing the same hazard and none of the code (audit v10,
 * B4). `tests/invariants/build-artifacts.test.ts` now decides which apps owe a
 * refusal by READING THE BUILT BYTES for the placeholder host, so an app cannot
 * opt out by naming its variables differently.
 */
const PLACEHOLDER_BASES = [["API_BASE", apiBase], ["PUBLIC_BASE", publicBase]];
if (!process.env.BUILD_OUT_DIR) {
  const placeholders = PLACEHOLDER_BASES.filter(([, val]) => /(^|\.)dhwaja\.example(\/|$|:)/i.test(val)).map(([name]) => name);
  if (placeholders.length > 0) {
    throw new Error(
      `refusing to build dist/ from the placeholder defaults (${placeholders.join(", ")} still point at dhwaja.example). ` +
        "dist/ is what `wrangler pages deploy` ships, and a bundle pointed at a domain nobody owns fails every check it makes. " +
        "Pass the real origins, or set BUILD_OUT_DIR to build a throwaway copy.",
    );
  }
}
mkdirSync(out, { recursive: true });

await build({
  entryPoints: { app: join(root, "src/main.ts") },
  outdir: out,
  bundle: true,
  format: "esm",
  target: "es2022",
  minify: true,
  legalComments: "none",
  define: { __API_BASE__: JSON.stringify(apiBase), __PUBLIC_BASE__: JSON.stringify(publicBase) },
});

// The shipped CSP names the API origin explicitly — same pattern as
// apps/admin/build.mjs and apps/agency/build.mjs. The checked-in index.html
// carries `connect-src 'self'` so an un-built or mis-built file fails CLOSED
// (the console can reach nothing rather than anything), and the one origin
// this app is allowed to talk to is appended here from the API_BASE that was
// validated above.
const CONNECT_SRC = "connect-src 'self'";
const html = readFileSync(join(root, "index.html"), "utf8");
if (!html.includes(CONNECT_SRC)) {
  throw new Error(`index.html must carry "${CONNECT_SRC}" for the API origin to be appended to it`);
}
if (!html.includes("form-action 'none'")) {
  throw new Error("index.html must carry form-action 'none' — it does not fall back to default-src");
}
writeFileSync(join(out, "index.html"), html.replace(CONNECT_SRC, `${CONNECT_SRC} ${new URL(apiBase).origin}`));
// `frame-ancestors` is IGNORED inside a <meta http-equiv> CSP — the spec makes
// it header-only, so every browser drops it and the copy in index.html has
// never protected anything. This app ships as static assets, so the only place
// a REAL header can come from is a Cloudflare Pages `_headers` file at the root
// of the output directory. The meta keeps the directives that do work there;
// this adds the one that does not. Two policies are enforced as an
// INTERSECTION, so this can only tighten, never loosen, the meta policy.
const HEADERS_FILE = "/*\n  Content-Security-Policy: frame-ancestors 'none'\n";
writeFileSync(join(out, "_headers"), HEADERS_FILE);

// eslint-disable-next-line no-console
console.log(`built creator console → ${out}/app.js + index.html (API_BASE=${apiBase}, PUBLIC_BASE=${publicBase})`);

/** #963 (I25-W1-COOKIE): the sentence a REFUSAL body carries (trust-core route-helpers.ts `refusal(error, message)`,
 * e.g. RATE_LIMIT_MESSAGE on a 429): a non-empty string `message` of at most 300 characters on a non-ok answer with a
 * real status below 500, else null. Every console 429 branch prints it in place of its own copy (main.ts). */
export function serverMessage(r: { ok: boolean; status: number; data: unknown }): string | null {
  if (r.ok || r.status === 0 || r.status >= 500 || r.data === null || typeof r.data !== "object") return null;
  const m = (r.data as { message?: unknown }).message;
  return typeof m === "string" && m.trim() !== "" && m.length <= 300 ? m : null;
}

/** What a first-passkey start answered, read once for every screen that adds a passkey (main.ts addPasskey):
 * `code` — the H3 answer, show the code field; `options` — options with no code asked (a server without H3);
 * `step_up` — 403 existing_passkey_required, the passport holds a passkey, so flows.ts registerPasskeyFlow's arm runs;
 * `refused` — anything else, with the server's status and code, and its whole answer (`answer`) for serverMessage. */
export type FirstPasskeyStart =
  | { kind: "code"; options: ServerRegistrationOptions; expires_at: string }
  | { kind: "options"; options: ServerRegistrationOptions }
  | { kind: "step_up" }
  | { kind: "refused"; status: number; error: string | null; answer: { ok: boolean; status: number; data: unknown } };

export async function startFirstPasskey(api: CreatorApi): Promise<FirstPasskeyStart> {
  const r = await api.passkeyRegisterStart();
  if (r.ok && r.data) {
    const ec = r.data.email_code;
    if (ec !== undefined && ec !== null && ec.sent === true && typeof ec.expires_at === "string") return { kind: "code", options: r.data, expires_at: ec.expires_at };
    return { kind: "options", options: r.data };
  }
  if (r.status === 403 && r.data?.error === "existing_passkey_required") return { kind: "step_up" };
  return { kind: "refused", status: r.status, error: typeof r.data?.error === "string" ? r.data.error : null, answer: r };
}

/** I34-W3-CLIENTS (I34-W2-PASSKEY H3, the contact door): while a verified address exists and no passkey, a change or a
 * verify of the contact address is answered 409 `add_passkey_first` with the server's sentence. flows.ts
 * setContactEmailFlow reads only `error` (errorCopy passes free text through and maps a code to a fallback), so that
 * answer is handed on with the sentence as `error` and the code kept as `code`. */
function addPasskeyFirstSentence<T>(r: ApiResult<T>): ApiResult<T> {
  const msg = r.status === 409 && r.data?.error === "add_passkey_first" ? serverMessage(r) : null;
  return msg !== null && r.data ? { ...r, data: { ...r.data, error: msg, code: "add_passkey_first" } } : r;
}

/**
 * I3-0100 — the account-closure notice, exactly as `services/trust-core/src/closures.ts`
 * `closureNotice` builds it: four fields and no fifth (the operator note, the detection
 * signal and the reporter are deliberately absent server-side). Same gate as
 * `apps/web/src/api.ts closureNoticeFrom`: a 403 carrying `account_closed` and EVERY field,
 * or nothing - a half notice would print a deadline with no address to appeal to.
 */
export interface ClosureNoticeDto {
  state: string;
  category: string;
  appeal_deadline?: string; // suspended only: a terminated notice has none (closures.ts closureNotice; QUEUE #660a, D-0942)
  appeal_to: string;
  account_id?: string; // D-1073(6): the id docs/legal/APPEALS.md asks the person to quote; carried when present
}
export function closureNoticeFrom(status: number, body: unknown): ClosureNoticeDto | null {
  if (status !== 403 || body === null || typeof body !== "object") return null;
  const b = body as Record<string, unknown>;
  if (b.error !== "account_closed") return null;
  const str = (v: unknown): string | null => (typeof v === "string" && v.length > 0 ? v : null);
  const state = str(b.state);
  const category = str(b.category);
  const appeal_deadline = str(b.appeal_deadline);
  const appeal_to = str(b.appeal_to);
  if (!state || !category || !appeal_to) return null;
  // I54 (P31 a): `appeal_deadline: null` is a suspension whose appeal window has not started (its notice was not delivered); kept,
  // and closureSentence gives it the no-deadline route. An absent or malformed deadline is still a half notice.
  if (state === "suspended" && !appeal_deadline && b.appeal_deadline !== null) return null;
  const account_id = str(b.account_id);
  return { state, category, ...(appeal_deadline ? { appeal_deadline } : {}), appeal_to, ...(account_id ? { account_id } : {}) };
}

/** The one sentence a closed creator reads: state, reason recorded, deadline, address. No inference of its own - it does not guess why and does not say whether an appeal will succeed. */
export function closureSentence(n: ClosureNoticeDto): string {
  const state = n.state === "suspended" ? "suspended" : n.state === "terminated" ? "closed" : n.state;
  const reason = Object.hasOwn(CLOSURE_CATEGORY_LABEL, n.category) ? (CLOSURE_CATEGORY_LABEL[n.category] as string) : n.category;
  const route = n.appeal_deadline ? `You can appeal until ${n.appeal_deadline.slice(0, 10)} by writing to ${n.appeal_to}.` : `You can write to ${n.appeal_to}.`;
  // D-1073(6) (approved label, D-1365 W65; the web's CLOSED_ACCOUNT_ID_LABEL): the id APPEALS.md tells the person to quote.
  const id = n.account_id ? ` Your account id (quote it when you email us): ${n.account_id}.` : "";
  return `This account is ${state}, so that can't be done. Reason recorded: ${reason}. ${route}${id}`;
}

interface ErrorBearing {
  error?: string;
}
/** `data` with a closure notice's `error` replaced by its sentence; any other body is returned untouched. */
function withClosureSentence<D extends ErrorBearing>(status: number, data: D | null): D | null {
  const notice = closureNoticeFrom(status, data);
  return notice && data ? { ...data, error: closureSentence(notice) } : data;
}

/** S-070 CN-01 — the console's application deadline for ONE request, headers
 * and body together. Long enough that an ordinary slow save is not cut off
 * (passkey ceremonies and registry reads are the slowest things here), short
 * enough that a stalled one cannot hold the module-wide ceremony guard for a
 * session. Not a server timeout and not a promise the write did not land. */
export const REQUEST_DEADLINE_MS = 30_000;

export class CreatorApi {
  private readonly base: string;
  private readonly fetchImpl: typeof fetch;
  constructor(private readonly opts: ApiOpts) {
    this.base = opts.baseUrl.replace(/\/+$/, "");
    // Bind the default to the global: native `fetch` invoked as a method
    // (`this.fetchImpl(...)`) throws "Illegal invocation" in every browser
    // because its `this` would be this client, not the global. The wrapper
    // keeps `this` correct. (Tests inject their own fetchImpl, so this only
    // bites in a real browser — which is why it slipped past the unit suite.)
    this.fetchImpl = opts.fetchImpl ?? ((input, init) => fetch(input, init));
  }

  /**
   * S-070 CN-01 — THE ONE DEADLINE BOTH NETWORK DOORS RUN UNDER.
   *
   * `send` carries JSON; `uploadMedia` posts raw bytes and cannot go through it,
   * so it is a SECOND door on the same property. It had no deadline at all —
   * neither leg — which is the wedge the block in `send` below describes, on the
   * slowest request this console makes. The control is one helper interpolated
   * at each door rather than re-typed at each, so the two cannot drift apart.
   */
  private deadline(): { init: RequestInit; aborted: () => boolean; cancel: () => void } {
    const ctl = typeof AbortController === "function" ? new AbortController() : null;
    const timer = ctl ? setTimeout(() => ctl.abort(), REQUEST_DEADLINE_MS) : null;
    return {
      init: ctl ? { signal: ctl.signal } : {},
      aborted: () => ctl?.signal.aborted === true,
      cancel: () => {
        if (timer !== null) clearTimeout(timer);
      },
    };
  }

  private async send<T>(method: "GET" | "POST" | "DELETE", path: string, body?: unknown, auth = true, cache?: "no-store" | "no-cache"): Promise<ApiResult<T>> {
    const headers: Record<string, string> = {};
    if (body !== undefined) headers["content-type"] = "application/json";
    if (auth) {
      const tok = this.opts.session.get();
      if (!tok) return { ok: false, status: 401, data: null };
      headers["Authorization"] = sessionAuthorization(tok); // I25-W1-COOKIE: the CSRF handle; the session is the HttpOnly cookie
    }
    // S-070 CN-01 — AN APPLICATION DEADLINE, COVERING THE BODY TOO.
    //
    // There was none, on either leg. That mattered far beyond this function:
    // `main.ts`'s `guarded` holds a single MODULE-WIDE `busy` flag for the whole
    // life of the call, and navigation does not enter that guard — so a creator
    // who submitted a slow save and then moved to Passkeys got a healthy,
    // fully-painted screen whose Register, Revoke and Sign out controls all
    // returned instantly from `guarded` and did nothing, with the request that
    // was holding them no longer represented anywhere on the page. With no
    // deadline, "the lifetime of that request" had no upper bound at all.
    //
    // THE BODY LEG IS THE HALF THAT IS EASY TO MISS: a response whose headers
    // arrive and whose body never completes hangs in `res.json()`, so a timeout
    // around `fetch` alone would have left exactly the same wedge. One signal
    // spans both.
    //
    // A TIMEOUT IS NOT PROOF THE WRITE DID NOT LAND, and this deliberately does
    // not pretend otherwise: it returns the SAME `{ ok: false, status: 0 }` the
    // network-failure arm already returns, which every caller in this console
    // already renders as "we could not read the result", never as "it did not
    // happen". Reconciling an uncertain mutation before any retry stays OP-04's
    // job; this only bounds the local wait.
    const dl = this.deadline();
    // I25-W1-COOKIE: "include" so the session cookie rides every call and a sign-in's Set-Cookie is kept.
    const init: RequestInit = { method, headers, ...WITH_COOKIE, ...(cache ? { cache } : {}), ...dl.init };
    if (body !== undefined) init.body = JSON.stringify(body);
    try {
      let res: Response;
      try {
        res = await this.fetchImpl(`${this.base}${path}`, init);
      } catch {
        return { ok: false, status: 0, data: null };
      }
      // S-070 CN-01, THE OTHER HALF: the deadline's OUTCOME has to cover the
      // body too, not only its signal. One signal does span both legs (:274),
      // but `.catch(() => null)` swallowed the abort and handed the caller
      // `{ ok: true, status: 200, data: null }` — a SUCCESS. `main.ts:318` then
      // renders an EMPTY inbox, which `main.ts:310-312` says in its own words is
      // the one sentence that must never stand in for "we could not load it".
      // An aborted body returns what the aborted HEADER leg already returns
      // (:281). A malformed body is untouched: real status, null data.
      let data: (T & { error?: string }) | null;
      try {
        data = (await res.json()) as (T & { error?: string }) | null;
      } catch {
        if (dl.aborted()) return { ok: false, status: 0, data: null };
        data = null;
      }
      // I3-0100: a closure notice becomes its own sentence in `error` (free text passes
      // through `errorCopy`), so every banner that already prints `r.data.error` names the
      // state, the reason, the deadline and the appeal address instead of the generic line.
      // Both doors that parse a body (this one and `uploadMedia`) run it.
      return { ok: res.ok, status: res.status, data: withClosureSentence(res.status, data) };
    } finally {
      dl.cancel();
    }
  }

  /** I25-W1-COOKIE — THE BOOTSTRAP AND THE CROSS-TAB SYNC: the handle of the HttpOnly creator cookie and the passport
   * it is for (GET /session/csrf?kind=creator). A live answer is installed; a 401 clears this tab; anything else
   * (offline, 5xx) changes nothing; a session set or cleared here while the answer was in flight wins. */
  async syncSession(): Promise<void> {
    const snap = this.opts.session.snapshot();
    const r = await this.send<{ csrf?: unknown; passport_id?: unknown }>("GET", "/session/csrf?kind=creator", undefined, false, "no-store");
    if (r.ok && typeof r.data?.csrf === "string" && typeof r.data.passport_id === "string" && PSP_RE.test(r.data.passport_id)) {
      if (r.data.csrf !== this.opts.session.get() || r.data.passport_id !== this.opts.session.passportId()) this.opts.session.setIfCurrent(snap, r.data.csrf, r.data.passport_id);
    } else if (r.status === 401) this.opts.session.clearIfCurrent(snap);
  }

  /** I25-W1-COOKIE — sign out of THIS browser: trust-core expires the HttpOnly cookie (POST /session/signout); the
   * handle is dropped here whatever the answer. Revoking every device stays logoutAll(). */
  async signOut(): Promise<void> {
    if (this.opts.session.get() !== null) await this.send("POST", "/session/signout");
    this.opts.session.clear();
  }

  /**
   * S91 — upload signed collectible art (raw bytes, NOT JSON). The server stores
   * it ONLY if sha256(bytes) matches a fingerprint the creator already signed into
   * a published release, so this is called AFTER the drop that signs the same hash.
   * A missing session fails soft as a 401-shaped result, like every write.
   */
  async uploadMedia(bytes: Uint8Array): Promise<ApiResult<{ ok: boolean; content_hash: string }>> {
    const tok = this.opts.session.get();
    if (!tok) return { ok: false, status: 401, data: null };
    // Copy into a fresh ArrayBuffer so a subarray view is sent whole, not the
    // backing buffer; content-type is octet-stream (the server derives the REAL
    // type from magic bytes and ignores this header).
    const body = bytes.slice();
    // THE SECOND DOOR, under the SAME deadline as `send` (`this.deadline()`).
    // It had none: an upload whose headers or body never completed held
    // `main.ts`'s module-wide `busy` flag open with no upper bound, on the
    // slowest request this console makes.
    const dl = this.deadline();
    try {
      let res: Response;
      try {
        res = await this.fetchImpl(`${this.base}/account/media`, { method: "POST", headers: { Authorization: sessionAuthorization(tok), "content-type": "application/octet-stream" }, ...WITH_COOKIE, body, ...dl.init });
      } catch {
        return { ok: false, status: 0, data: null };
      }
      let data: { ok: boolean; content_hash: string; error?: string } | null;
      try {
        data = (await res.json()) as { ok: boolean; content_hash: string; error?: string } | null;
      } catch {
        if (dl.aborted()) return { ok: false, status: 0, data: null };
        data = null;
      }
      // I3-0100: a closure notice becomes its own sentence in `error` (free text passes
      // through `errorCopy`), so every banner that already prints `r.data.error` names the
      // state, the reason, the deadline and the appeal address instead of the generic line.
      // Both doors that parse a body (this one and `uploadMedia`) run it.
      return { ok: res.ok, status: res.status, data: withClosureSentence(res.status, data) };
    } finally {
      dl.cancel();
    }
  }

/**

 * that can put a creator's name on a report a reviewer is holding, in the order
 * the server ranks them.
 *
 *   operator     a person here made the identification
 *   identity     a close copy of a handle or domain of theirs — equal after the

 *                another alphabet

 * fall through the renderer's default and read as unqualified, which is what a
 * value this client has never heard of actually means.

    integrity_withheld: boolean;

// --- S23/S88 — mirrors services/trust-core src/requests.ts ArchiveDemand.
//
// EVERY FIELD HERE IS A PRIVACY DECISION, not a report shape (RUN-0002). The
// count is a DISTINCT-FAN population, so the server day-lags it (today's
// requests arrive tomorrow) and withholds it below `floor` — `request_count` is
// `null` for every population from nobody up to floor-1, and a client that reads
// null as "nobody asked" is making a claim the server refused to make, which is
// what `withheld` exists to prevent. `floor` is diagnostic and MUST NOT be
// rendered; `through` is the last whole UTC day counted and is derived from the
// clock alone, so it carries no fan data and may be shown.
//
// A note's `day` is the day that TEXT was written — the field used to be called
// `created_at`, which said "when this fan requested" while carrying "when this
// text was last written", so clients rendering it by name mis-dated the request
// (FINDING-0014). There is one name for one fact now.
//
// `notes_withheld` (RUN-0006) is the SECOND gate and it is not the first one:
// the count includes the fans who erased today (a count that dropped when one
// person left would be the delta the day-lag removes), while the NOTES can only
// be the live fans' words, so they are gated on the LIVE population. `true`
// means the count is served and the notes are not. A renderer must answer it
// with the shared withheld sentence and never with "no notes", which would be
// false for a creator whose fans wrote some.
export interface ArchiveDemandDto {
  request_count: number | null;
  withheld: boolean;
  floor: number;
  through: string;
  notes: Array<{ note: string; day: string }>;
  notes_withheld: boolean;
}

/** I3-1785 — " (until YYYY-MM-DD)" from the server's exact cutoff, or "" when absent/unparseable (never a guessed date). */
export function graceUntilSuffix(until: unknown): string {
  const ms = typeof until === "string" ? Date.parse(until) : Number.NaN;
  return Number.isFinite(ms) ? ` (until ${new Date(ms).toISOString().slice(0, 10)})` : "";
}

/** I3-1166 — "Contact support" with the server-named destination as plain text
 * (the plan card's `fine` line is escaped text, so no link), bare when unnamed. */
export function supportSentence(to: unknown): string {
  const s = typeof to === "string" ? to.trim().slice(0, 200) : "";
  return s === "" ? "Contact support" : `Contact support at ${s}`;
}

/** I3-1251 (BF-06) — the interval an in-flight checkout was REPLAYED on, when it is
 * not the one the creator picked; null when it matches or the server said nothing. */
export function checkoutIntervalSwitched(requested: "month" | "year", data: { interval?: unknown } | null | undefined): "month" | "year" | null {
  const served = data?.interval;
  if (served !== "month" && served !== "year") return null;
  return served === requested ? null : served;
}

/**
 * The whole state machine, pure. Order matters and is deliberate:
 *
 *   1. 503 — billing is switched off on this deployment. Wins over everything,
 *      including a tier we may have read a moment ago: if the surface is off we
 *      say so and offer no control (invariant 6, fail closed).
 *   2. 401 — the console session expired underneath the page. Its own card, and
 *      never the fail-closed one: "billing isn't set up" would be a lie.
 *   3. any other unreadable answer → the SAME fail-closed card as (1). Mock 67
 *      pins this: one card whether the keys are absent, the provider is
 *      unreachable, or an operator paused sales. Guessing "free" here would
 *      invent a state we did not read, and guessing "active" would tell a
 *      creator they are paying on the strength of an answer the server never
 *      gave. "Unreadable" explicitly includes ANY non-200 status WITH a body:
 *      a 429 rate-limit envelope, a proxy's JSON, or a cached page that happens
 *      to parse must never rescue the status.
 *   4. in grace  → quiet grey, never red. Requires `in_grace === true` (a real
 *      boolean — the STRING "false" is truthy) AND tier "pro": grace is a
 *      failed-PRO-PAYMENT state, so {free, in_grace} is an incoherent answer and
 *      takes the fail-closed card at (3) rather than claiming Pro capacity for a
 *      creator who has none.
 *   5. paying    → active (no Stripe customer ⇒ no button; see PlanActionKind).
 *   6. purchased before, not now → lapsed.
 *   7. otherwise → free, which is not an error and not a trial.
 */
export function planState(read: PlanRead): PlanState {
  const interval: BillingInterval = read.interval === "month" ? "month" : "year";
  const ent = read.ent;

  if (read.status === 503) return unavailableState();
  if (read.status === 401) {
    return {
      name: "signed-out",
      tone: "quiet",
      headline: "Your console session has expired",
      body: "Sign in again to see or change your plan. You only ever need a session to buy — never to be protected: your passport, your account proofs and every warning protecting you carry on exactly as they are.",
      unaffected: "",
      meta: "",
      fine: "Your verdict, your warnings and your proven accounts are unchanged while you're signed out.",
      action: { kind: "signin", label: "Sign in again", interval },
    };
  }
  // Fail closed on the answer itself, not just on the status: this is a CREATOR
  // console, so a fan-shaped entitlement or an unknown tier is an answer we
  // cannot act on either. 200 is the only status that renders a plan.
  if (ent === null || read.status !== 200) return unavailableState();
  if (ent.subject_kind !== "creator") return unavailableState();
  if (ent.tier !== "free" && ent.tier !== "pro") return unavailableState();
  // `in_grace` decides whether we tell a creator a payment FAILED, so it is read
  // as a real boolean or not at all. It used to be tested for truthiness, and
  // the string "false" is truthy: an entitlement that said the payment was fine
  // rendered the card saying it was not.
  if (typeof ent.in_grace !== "boolean") return unavailableState();
  // AG-12: `has_customer` and `purchased_tier` used to be read truthily/loosely
  // while `subject_kind`/`tier`/`in_grace` were validated as closed sets a few
  // lines above -- the same "answer we cannot act on" standard applies to all
  // five fields. `has_customer: "false"` (truthy) would show a portal button
  // that goes nowhere; `has_customer: 0`/`""` would set `unmanageable` on a
  // real Pro subscriber, hiding the one manage control they have. Fail closed
  // exactly as the fields above do rather than coerce.
  if (typeof ent.has_customer !== "boolean") return unavailableState();
  if (ent.purchased_tier !== null && ent.purchased_tier !== "premium" && ent.purchased_tier !== "pro") {
    return unavailableState();
  }

  // GRACE IS A PRO-PAYMENT STATE. A well-formed 200 saying {creator, free,
  // in_grace: true} used to render "you keep every Pro cap for N more days" — a
  // paid-capacity claim on a tier that has no paid capacity. The server cannot
  // produce that pair (`effectiveTier` keeps returning the purchased tier for as
  // long as grace runs and only falls to "free" once it has expired), so it is
  // an incoherent answer, and an answer we cannot act on gets the fail-closed
  // card rather than a guess in either direction. Guessing "free" would hide a
  // failing payment the creator needs to fix; guessing "grace" claims capacity
  // they do not have.
  if (ent.in_grace && ent.tier !== "pro") return unavailableState();

  if (ent.in_grace) {
    const days = Math.max(0, Math.trunc(Number(ent.grace_days) || 0));
    const meta = (days === 1 ? "grace ends tomorrow" : `grace ends in ${days} days`) + graceUntilSuffix(ent.grace_until);
    // I16-W03A H2 (RJb-T2-AS-012, D-0920 option b; same split as apps/web/src/upgrade.ts): only a
    // Stripe customer can have had a charge fail. Without one the period ended and was not renewed —
    // the card never says a payment was refused.
    if (!ent.has_customer) {
      return {
        name: "grace",
        tone: "quiet",
        headline: "Your Creator Pro period has ended",
        body: `It hasn't been renewed yet. Nothing has changed so far: you keep every Pro cap for ${days} more ${days === 1 ? "day" : "days"} — ${PRO_CAPS_SENTENCE}. If it isn't renewed by then you go back to the free tier — no other consequence, ever.`,
        unaffected: "Either way, your protection and your verdict are untouched. Warnings, verification, proven accounts and your passport do not depend on this plan and never will.",
        meta,
        fine: "Grey, never red. A plan that ran out is an admin matter, not an accusation — and red on this page would be a lie about how serious it is.",
        action: { kind: "none", label: "", interval },
      };
    }
    return {
      name: "grace",
      tone: "quiet",
      headline: "We couldn't take this Creator Pro payment",
      body: `The charge didn't go through. Nothing has changed yet: you keep every Pro cap for ${days} more ${days === 1 ? "day" : "days"} — ${PRO_CAPS_SENTENCE}. If it's still unpaid then, you go back to the free caps — no other consequence, ever.`,
      unaffected: "Either way, your protection and your verdict are untouched. Warnings, verification, proven accounts and your passport do not depend on this payment and never will.",
      meta,
      fine: "Grey, never red. A card that expired is an admin problem, not an accusation — and red on this page would be a lie about how serious it is.",
      action: { kind: "portal", label: "Update payment ↗ Stripe", interval },
    };
  }

  if (ent.tier === "pro") {
    const through = iso(ent.paid_through);
    // Pro with no Stripe customer: there is exactly one rail that sells Pro, so
    // this should not happen. Show the true tier and NO control — a portal link
    // would go nowhere and a checkout button would open a second subscription.
    const unmanageable = !ent.has_customer;
    return {
      name: "active",
      tone: "good",
      headline: read.justUpgraded ? "Welcome to Creator Pro" : "You're on Creator Pro",
      body: `Your caps are up: ${PRO_CAPS_SENTENCE}. Everything protective was already yours, and none of it moved.`,
      unaffected: "",
      meta: through ? `renews ${through}` : "",
      fine: unmanageable
        ? `Your Pro is active, but this subscription has no billing profile we can open a portal for. ${supportSentence(ent.support_to)} and we'll sort it — there is deliberately no \u201cbuy Pro\u201d button here, because pressing it would start a second subscription on top of the one you already have.`
        : "Manage opens Stripe's portal — change the card, switch monthly/annual, download invoices, or cancel. Cancelling there leaves Pro running until the date you already paid through.",
      action: unmanageable ? { kind: "none", label: "", interval } : { kind: "portal", label: "Manage billing ↗ Stripe", interval },
    };
  }

  if (ent.purchased_tier !== null && ent.purchased_tier !== undefined) {
    const through = iso(ent.paid_through);
    return {
      name: "lapsed",
      tone: "quiet",
      headline: through ? `Creator Pro ended on ${through}` : "Creator Pro ended",
      body: `You're on the free caps again: ${FREE_CAPS_SENTENCE}. Nothing you made was removed — your drops, collectibles, codes and recognitions all stand exactly as published.`,
      unaffected: "Your verification, your warnings and your proven accounts are unchanged. Money buys capacity here, never safety — so losing it costs you capacity, never safety.",
      meta: "no reactivation fee · no lost history",
      fine: 'No countdown, no "you\'re missing out," no dimmed features dangling behind a lock. A creator who stops paying is a creator, full stop.',
      action: { kind: "checkout", label: `Start Pro again — ${priceOf(interval)}`, interval },
    };
  }

  return {
    name: "free",
    tone: "quiet",
    headline: "You're on Free",
    body: `Free isn't a trial and it isn't a lapsed state — it's the whole protective product. Your passport, your account proofs, every impostor warning protecting you and the verification review are yours permanently, at no cost. Right now you can send ${FREE_CAPS_SENTENCE}. Pro raises those four ceilings, and nothing else.`,
    unaffected: "",
    meta: interval === "month" ? `${PRO_PRICING.month} · or ${PRO_PRICING.year}, which saves ${PRO_PRICING.saving} a year` : `${PRO_PRICING.year} · or ${PRO_PRICING.month} if you'd rather pay monthly`,
    fine: "Checkout happens on Stripe, not here. You'll leave the console, pay on Stripe's page, and come back to this one. Dhwaja never sees or stores your card number.",
    action: { kind: "checkout", label: `Go Pro — ${priceOf(interval)} →`, interval },
  };
}

/**

/**
 * Reverse-code AUTHORING rules, mirrored on the client (S72/S73).
 *
 * The server module `services/trust-core/src/reverse-codes.ts` owns every rule
 * here; this file exists for ONE reason: `POST /account/reverse-codes/pool/:id/
 * codes` caps the RAW REQUEST BODY at 128 KiB and answers 413, and a creator
 * pasting a real list of a few thousand codes hits that cap routinely. A 413
 * arriving as a generic failure tells them nothing about what to do, so the
 * batch is counted and SIZED here — before anything is posted — and the view
 * says exactly how to split it.
 *
 * Two hard rules for everything in this file:
 *  1. It is a MIRROR, never an authority. The server re-validates every code,
 *     every title and every count; nothing here is trusted server-side, and a
 *     value that passes here can still be refused (the outcome adapters show
 *     the server's own words when it is).
 *  2. It never SPLITS OR RETRIES a batch behind the creator's back. Chunking an
 *     oversize upload into several automatic requests would defeat the body cap
 *     the server put there on purpose; we refuse, and hand the creator the exact
 *     number to split at.

/** How many malformed entries the plan reports individually. The creator needs
 * to FIND the bad lines, not be handed ten thousand of them. */
export const MALFORMED_SHOWN = 20;

export interface MalformedCode {
  /** 1-indexed line of the pasted/uploaded text, so the creator can go fix it. */
  line: number;
  value: string;
}

export interface CodeBatchPlan {
  /** Accepted, de-duplicated, order-preserved — exactly what would be posted. */
  codes: string[];
  /** Non-empty entries found in the input (before dedup or shape checks). */
  found: number;
  /** Dropped as repeats WITHIN this input (the pool de-dups again server-side). */
  duplicates: number;
  /** The first `MALFORMED_SHOWN` entries that fail the shape rule… */
  malformed: MalformedCode[];
  /** …and how many there are in total. */
  malformedTotal: number;
  /** Exact length of the JSON body this batch would send — what the 128 KiB
   * cap is measured against. */
  bytes: number;
  overCount: boolean;
  overSize: boolean;
  /** The largest prefix of `codes` that fits BOTH caps — the number to split at. */
  fitsPerRequest: number;
  /** True only when there is something to send and it is within both caps.
   * Malformed entries make the whole batch unsendable, because the server
   * rejects the batch as a unit rather than dropping the bad ones. */
  sendable: boolean;
}

/** Exact length of the request body `uploadReverseCodes` would send. The server
 * compares `(await c.req.text()).length` against the cap, so this is the same
 * measurement, not an estimate. */
export function uploadBodyLength(codes: readonly string[]): number {
  return JSON.stringify({ codes }).length;
}

/**
 * The largest prefix of `codes` that fits under BOTH the body cap and the
 * per-batch count cap — i.e. the number the creator should split at. Computed
 * on the serialized form (`{"codes":["A","B"]}`) so it matches the byte the
 * server counts. Never returns 0 for a non-empty list: a single code can't
 * exceed 128 KiB (codes are ≤64 chars), so 1 is always sendable.
 */
export function fitsPerRequest(codes: readonly string[]): number {
  let len = '{"codes":[]}'.length;
  let n = 0;
  for (const c of codes) {
    const add = c.length + 2 + (n > 0 ? 1 : 0); // "code" plus a separating comma
    if (n >= MAX_CODES_PER_BATCH || len + add > MAX_UPLOAD_BODY) break;
    len += add;
    n++;
  }
  return codes.length === 0 ? 0 : Math.max(1, n);
}

/**
 * Parse a pasted / uploaded code list into a plan. Entries are separated by
 * newlines, commas, semicolons and tabs. A PLAIN SPACE is deliberately NOT a
 * separator: a code cannot contain one, so `PROMO 20` is one creator's typo,
 * not two codes, and splitting it would quietly upload two values they never
 * wrote. It is reported as one malformed entry instead, with its line number,
 * so a rejection names the line rather than the batch.
 */
export function planCodeBatch(text: unknown): CodeBatchPlan {
  const lines = String(text ?? "").split(/\r\n|\r|\n/);
  const seen = new Set<string>();
  const codes: string[] = [];
  const malformed: MalformedCode[] = [];
  let found = 0;
  let duplicates = 0;
  let malformedTotal = 0;
  for (let i = 0; i < lines.length; i++) {
    for (const raw of lines[i]!.split(/[,;\t]+/)) {
      const v = raw.trim();
      if (v === "") continue;
      found++;
      if (v.length > MAX_CODE_LEN || !CODE_RE.test(v)) {
        malformedTotal++;
        if (malformed.length < MALFORMED_SHOWN) malformed.push({ line: i + 1, value: v });
        continue;
      }
      if (seen.has(v)) {
        duplicates++;
        continue;
      }
      seen.add(v);
      codes.push(v);
    }
  }
  const bytes = uploadBodyLength(codes);
  const overCount = codes.length > MAX_CODES_PER_BATCH;
  const overSize = bytes > MAX_UPLOAD_BODY;
  return {
    codes,
    found,
    duplicates,
    malformed,
    malformedTotal,
    bytes,
    overCount,
    overSize,
    fitsPerRequest: fitsPerRequest(codes),
    sendable: codes.length > 0 && malformedTotal === 0 && !overCount && !overSize,
  };
}

/** Validate a generate count the same way the server does (1..1000, integer).
 * Returns the reason it is refused, or null when it is fine. */
export function generateCountProblem(count: unknown): string | null {
  if (typeof count !== "number" || !Number.isFinite(count) || !Number.isInteger(count)) return "Enter a whole number of vouchers.";
  if (count < 1) return "Enter at least 1 voucher.";
  if (count > MAX_CODES_PER_BATCH) return `Dhwaja mints at most ${MAX_CODES_PER_BATCH} vouchers per batch. Generate ${MAX_CODES_PER_BATCH} now and run it again for the rest.`;
  return null;
}

}

/**
 * I3-0422 a — THE IN-PAGE CONFIRM AND PROMPT (creator console and agency console).
 *
 * `window.confirm`/`window.prompt` are the browser's own boxes: no title, no focus return, one wall of text, dismissed
 * by muscle memory, and the same look for "publish a reply" as for "remove every payout destination". This asks the
 * same question inside the page, with the cost sentences as their own paragraphs, in the DOM shape admin's
 * `confirmDialog`/`reasonDialog` use (apps/admin/src/main.ts) — `role="alertdialog"`, a button whose data-act is `confirm-go`
 * / `confirm-cancel`, a `-retype` input under `label code`, `reason-go` / `reason-cancel` with a textarea — so one test

/** Exact, trimmed, case-sensitive; nothing is confirmable against an empty expected text. */
export function retypeMatches(typed: unknown, expected: unknown): boolean {
  if (typeof expected !== "string" || expected.trim() === "") return false;

let seq = 0;
let queue: Promise<unknown> = Promise.resolve();

function enqueue<T>(run: () => Promise<T>): Promise<T> {
  const next = queue.then(run, run);
  queue = next.catch(() => undefined);
  return next;
}

/** The message's blank-line-separated paragraphs, each set as TEXT (never markup). */
function paragraphs(doc: Document, message: string): HTMLElement[] {
  return message
    .split(/\n\s*\n/)
    .map((p) => p.trim())
    .filter((p) => p.length > 0)
    .map((p) => {
      const el = doc.createElement("p");
      el.textContent = p;
      return el;
    });
}

function mount<T>(build: (box: HTMLElement, id: string, settle: (v: T) => void) => HTMLElement, cancelValue: T): Promise<T> {
  return enqueue(
    () =>
      new Promise<T>((resolve) => {
        const doc = document;
        const id = `app-dialog-${++seq}`;
        const holder = doc.createElement("div");
        holder.className = "dlg-scrim";
        holder.setAttribute("data-app-dialog", "");
        const box = doc.createElement("div");
        box.className = "dlg";
        box.setAttribute("role", "alertdialog");
        box.setAttribute("aria-modal", "true");
        box.setAttribute("aria-labelledby", `${id}-h`);
        holder.append(box);
        const trigger = doc.activeElement as HTMLElement | null;
        let done = false;
        const settle = (v: T): void => {
          if (done) return;
          done = true;
          doc.removeEventListener("keydown", onKey, true);
          holder.remove();
          if (trigger && typeof trigger.focus === "function" && trigger.isConnected) trigger.focus();
          resolve(v);
        };
        const onKey = (e: KeyboardEvent): void => {
          if (e.key === "Escape") {
            e.preventDefault();
            settle(cancelValue);
            return;
          }
          if (e.key !== "Tab") return;
          const f = [...box.querySelectorAll<HTMLElement>("button, input, textarea")];
          if (f.length === 0) return;
          const first = f[0]!;
          const last = f[f.length - 1]!;
          if (e.shiftKey && doc.activeElement === first) { e.preventDefault(); last.focus(); }
          else if (!e.shiftKey && doc.activeElement === last) { e.preventDefault(); first.focus(); }
        };
        const focusFirst = build(box, id, settle);
        holder.addEventListener("click", (e) => { if (e.target === holder) settle(cancelValue); });
        doc.addEventListener("keydown", onKey, true);
        doc.body.append(holder);
        focusFirst.focus();
      }),
  );
}

function heading(doc: Document, id: string, title: string): HTMLElement {

}

  b.className = primary ? "ds-btn primary" : "ds-btn";

    err.setAttribute("role", "alert");

      const p = doc.createElement("p");

      field.autocomplete = "off";

      label.append("Type ", code, " to confirm ", field);
      p.append(label);
      box.append(p, err);
    }
    const row = doc.createElement("div");
    row.className = "dlg-row";

/**
 * I3-1208 b (I23, D-0993(1) pick: WARN-ONLY) — THE DIRTY-FORM GUARD.
 *
 * The one-time-codes guard (`onetime.ts`) protects voucher values; every other console form lost
 * what the creator typed on a nav click, the back button, a reload or a closed tab, with no prompt.
 * This is the same idea generalised, and decided from the LIVE DOM rather than from bookkeeping:
 * a typed control is dirty when its value differs from its `defaultValue` (what the markup was
 * rendered with). A sent form's controls are marked clean (`markSent`), and a render replaces the
 * whole root, so a form that was sent or re-rendered is never dirty. WHEN a form counts as sent is
 * `markSubmitted`'s rule below (I34-FX2-CLIENTS).
 *
 * NOTHING IS STORED. No draft is kept anywhere (the ruling's reason: a creator's unsent words in
 * browser storage on a shared computer are a leak) — the guard only asks before the words are lost.
 */

/** The input types whose value a person SETS by hand: typed text, plus (I32 B12, I34-W2-CLIENTS) a date and a range
 * slider — the drop form's release date and the signature sliders, lost on a lapse before. Choices (checkbox, radio,
 * select, colour) and files are not; a password-like or hidden control is never one of these either. */
const TYPED_INPUT_TYPES = new Set(["", "text", "email", "url", "search", "tel", "number", "date", "range"]);

/** I3-1208 (I34-W2-CLIENTS): the marker main.ts's signature pad sets on a drawn, unsent stroke. */
export const UNSENT_INK_ATTR = "data-unsent-ink";

/** What the control read when it was rendered. A range with no `value` attribute reads its midpoint, not
 * `defaultValue`'s "", so an untouched slider is not mistaken for a moved one. */
export function renderedValue(el: Typed): string {
  if ((el.getAttribute("type") ?? "").toLowerCase() !== "range") return el.defaultValue;
  const probe = el.ownerDocument.createElement("input");
  for (const a of ["type", "min", "max", "step", "value"]) {
    const v = el.getAttribute(a);
    if (v !== null) probe.setAttribute(a, v);
  }
  return probe.value;
}

type Typed = HTMLInputElement | HTMLTextAreaElement;

function isTypedControl(el: Element): el is Typed {
  const tag = el.tagName.toLowerCase();
  if (tag === "textarea") return true;
  if (tag !== "input") return false;
  return TYPED_INPUT_TYPES.has(((el as HTMLInputElement).getAttribute("type") ?? "").toLowerCase());
}

/** The typed controls under `root` that hold unsent input — inside a `<form>` or not (I3-1208 b, I23-FX-APPS: the
 * Questions answer box and the account-contest note are typed and sent by a button, outside any form). Read-only and
 * disabled controls (a generated-codes box, a locked field) are not the creator's words. */
export function dirtyControls(root: ParentNode): Typed[] {
  const out: Typed[] = [];
  for (const el of Array.from(root.querySelectorAll("input, textarea"))) {
    if (!isTypedControl(el)) continue;
    if (el.readOnly || el.disabled) continue;
    if (el.value.trim() !== renderedValue(el).trim()) out.push(el);
  }
  return out;
}

export function hasDirtyForm(root: ParentNode): boolean {
  return dirtyControls(root).length > 0 || root.querySelector(`[${UNSENT_INK_ATTR}]`) !== null;
}

/** I3-1208 b (I34-RG-W2P CLIENTS:7 / W2P-N2; I34-FX2-CLIENTS): a form carrying this attribute counts as sent only once
 * its submit SUCCEEDED — its handler calls `markSent` on success, or its success re-renders it away — so a submit the
 * handler refuses leaves the typed words and the ink unsent, and leaving still asks. views.ts writes it as a literal on
 * #sign-form, #agency-form and #media-delete-form (views.ts is also compiled by trust-core, without the DOM lib this
 * module needs, so it does not import from here). */
export const SENT_ON_SUCCESS_ATTR = "data-sent-on-success";

/** The capture-phase submit wiring (main.ts). A form carrying SENT_ON_SUCCESS_ATTR is left untouched here; any other form
 * is still marked sent at submit (its handler does not mark it on success yet — the census of those forms is in
 * I34-FX2-CLIENTS REPORT.md). */
export function markSubmitted(form: Element): void {
  if (form.hasAttribute(SENT_ON_SUCCESS_ATTR)) return;
  markSent(form);
}

/** A form's words were sent: what it holds now is what was sent, so it is no longer unsent input. */
export function markSent(form: Element): void {
  for (const el of Array.from(form.querySelectorAll("input, textarea"))) {
    if (isTypedControl(el)) el.defaultValue = el.value;
  }
  for (const el of Array.from(form.querySelectorAll(`[${UNSENT_INK_ATTR}]`))) el.removeAttribute(UNSENT_INK_ATTR);

/** Cents → a display dollar string (e.g. 2550 → "US$25.50", 2500 → "US$25"; #132PS-344: the currency is named). */
export function dollarsFromCents(cents: number): string {
  const n = Math.round(Number(cents) || 0);
  const abs = Math.abs(n);
  const s = abs % 100 === 0 ? `US$${abs / 100}` : `US$${(abs / 100).toFixed(2)}`;
  return n < 0 ? `−${s}` : s;
}

/**
 * M-15 — the schema has no `failed` status a stuck transfer ever reaches
 * (`creator-payout.ts` writes only `pending`→`paid`; a refused transfer pushes
 * onto an in-memory `out.failed` that never reaches the row). Rather than the
 * copy asking the creator to count weekly runs against a date they'd have to
 * do the arithmetic on themselves, a `pending` row old enough that ordinary
 * retry cadence no longer explains it is flagged HERE, from data the console
 * already has. This is the client-side mitigation the finding calls for
 * "meanwhile" — the real fix (the server writing `failed`) belongs to
 * `services/trust-core/src/creator-payout.ts`, outside this file's ownership.
 */
export const PAYOUT_STALL_MS = 14 * 24 * 60 * 60 * 1000; // two weekly runs

/** The age, in ms, of the oldest still-`pending` row — or null when none is
 * pending. Used to decide whether "paying out now" is an ordinary wait or a
 * stall (M-15). */
function oldestPendingAgeMs(payouts: PayoutRowDto[], nowMs: number): number | null {
  let max: number | null = null;
  for (const r of payouts) {
    if (r.status !== "pending") continue;
    const at = Date.parse(String(r.created_at));
    if (!Number.isFinite(at)) continue;
    const age = nowMs - at;
    if (max === null || age > max) max = age;
  }
  return max;
}

/** R-14/S-039 follow-up: `payoutHistory` (creator-payout.ts) now caps at
 * `PAYOUT_HISTORY_LIMIT` (1000) rows, raised from 200 so no live creator
 * crosses it for decades — but `creatorEarnings.paid_cents` is still an
 * unbounded fold, so a row this table can't show would silently stop
 * reconciling with the "Paid out" tile above. Duplicated here rather than
 * imported: this package does not depend on services/trust-core, so the
 * value is a literal with this comment as its cross-reference, not a shared

const PAYOUT_HISTORY_LIMIT_HINT = 1000;

/** I3-1669 — an integer grouped with ASCII commas, independent of the viewer's
 * locale and ICU data (the `apps/web` `shortWhen` policy). `toLocaleString()`
 * rendered "1,000" or "1.000" by browser. */
export function groupThousands(n: number): string {
  const v = Math.trunc(Number(n) || 0);
  const digits = String(Math.abs(v)).replace(/\B(?=(\d{3})+(?!\d))/g, ",");
  return v < 0 ? `-${digits}` : digits;
}

function payoutHistory(rows: PayoutRowDto[], nowMs: number): string {
  if (rows.length === 0) return `<p class="ehint">No payouts yet — your first runs once you've connected an account and your cleared balance passes the minimum.</p>`;
  const body = rows
    .map((r) => {
      // D-0800/H4: a `failed` row (I3-2131) is still owed and still retrying --
      // rendered the same "delayed" way a slow `pending` row is, not as "failed".
      const stalled = (r.status === "pending" || r.status === "failed") && (() => {
        const at = Date.parse(String(r.created_at));
        return Number.isFinite(at) && nowMs - at > PAYOUT_STALL_MS;
      })();
      const label = stalled ? "delayed" : r.status;
      return `<tr><td>${esc(r.period_key)}</td><td class="num">${esc(dollarsFromCents(r.amount_cents))}</td><td><span class="pstatus ${esc(r.status)}${stalled ? " stalled" : ""}">${esc(label)}</span></td><td class="dt">${esc(String(r.created_at).slice(0, 10))}</td><td class="dt">${typeof r.paid_at === "string" && /^\d{4}-\d{2}-\d{2}/.test(r.paid_at) ? esc(r.paid_at.slice(0, 10)) : ""}</td></tr>`;
    })
    .join("");
  const cap = rows.length >= PAYOUT_HISTORY_LIMIT_HINT ? `<p class="ehint">Showing your most recent ${groupThousands(PAYOUT_HISTORY_LIMIT_HINT)} payouts. If you've been paid out longer than that, older rows aren't shown here yet.</p>` : "";

/**

 * `earningsView` read `data.earnings.connect_status` and `data.payouts.map`
 * directly. A 200 whose body was missing `earnings` (a partial deploy, a proxy

 *
 * Two coercions, both narrow on purpose. Neither invents a figure: a value that
 * is not a finite number becomes 0 AND sets `partial`, which the page then says
 * out loud rather than presenting a confident $0.
 */
function num(v: unknown): number | null {
  return typeof v === "number" && Number.isFinite(v) ? v : null;
}

interface SafeEarnings {
  e: EarningsDto;
  payouts: PayoutRowDto[];
  /** True when any figure or the status could not be read as its own type. The
   * page shows the numbers it has AND says some of them did not read — never a
   * silent zero on a money screen. */
  partial: boolean;
}

function readEarnings(data: CreatorEarningsDto): SafeEarnings {
  const raw = (data as { earnings?: unknown }).earnings;
  const src = raw && typeof raw === "object" ? (raw as Record<string, unknown>) : {};
  let partial = raw === null || typeof raw !== "object";
  const money = (k: keyof EarningsDto): number => {
    const n = num(src[k]);
    if (n === null) partial = true;
    return n ?? 0;
  };
  const STATUSES = ["none", "connecting", "pending", "active", "revoked"] as const;
  const st = src.connect_status;
  const connect_status = (STATUSES as readonly string[]).includes(st as string) ? (st as EarningsDto["connect_status"]) : "none";
  if (connect_status !== st) partial = true;
  const minRaw = num(src.min_payout_cents);
  const rows = (data as { payouts?: unknown }).payouts;
  const payouts = Array.isArray(rows) ? (rows.filter((r) => r && typeof r === "object") as PayoutRowDto[]) : [];
  if (rows !== undefined && rows !== null && !Array.isArray(rows)) partial = true;
  return {
    e: {
      cleared_cents: money("cleared_cents"),
      pending_cents: money("pending_cents"),
      paid_cents: money("paid_cents"),
      in_flight_cents: money("in_flight_cents"),
      lifetime_cents: money("lifetime_cents"),
      connect_status,
      payouts_live_at: typeof src.payouts_live_at === "string" ? src.payouts_live_at : null,
      // A missing minimum must not render as "$0 minimum" — that would read as
      // "any amount pays out", which is the opposite of true.
      min_payout_cents: minRaw ?? 2500,
    },
    payouts,
    partial: partial || minRaw === null,
  };
}

/** True when `s` is shaped like a machine error code rather than a sentence. */
export function isMachineCode(s: string): boolean {
  return MACHINE_CODE_RE.test(s.trim());
}

/** Plain copy for a server `error`: free text passes through, a code never does. */
export function errorCopy(code: unknown, fallback: string): string {
  if (typeof code !== "string" || code.trim() === "") return fallback;
  const k = code.trim();
  if (!isMachineCode(k)) return code;
  return Object.hasOwn(ERROR_COPY, k) ? (ERROR_COPY[k] as string) : fallback;
}

function errOf(data: { error?: string } | null, fallback: string): string {
  return errorCopy(data?.error, fallback);
}

/** The payout legs' views BRANCH on the raw code (views.ts payoutOutcome /
 * cancelStagedOutcome compare `r.error` to PASSKEYS_UNAVAILABLE_ERROR and
 * PAYOUT_VERIFIED_ONLY_ERROR), so those two flows carry it unmapped and the
 * view owns the copy. */
function codeOf(data: { error?: string } | null, fallback: string): string {
  return data?.error ?? fallback;
}

/** Plain copy for a THROWN authenticator/browser failure (I3-0764). Only the
 * exception's `name` is read, never its `message`: a browser's own text (a
 * WebAuthn DOMException carries a W3C URL) is not copy. Unknown -> `fallback`. */
const THROWN_COPY: Record<string, string> = {
  NotAllowedError: "Your device didn't approve that — it was cancelled or timed out. Nothing was changed. Try again when you're ready.",
  AbortError: "Your device didn't approve that — it was cancelled or timed out. Nothing was changed. Try again when you're ready.",
  WebAuthnClientError: "Your device didn't approve that — it was cancelled or isn't supported here. Nothing was changed.",
  InvalidStateError: "This device can't use that passkey for this step — it may already be registered. Nothing was changed.",
  SecurityError: "Your browser blocked the passkey step on this page. Nothing was changed.",
  NotSupportedError: "This browser or device can't do that passkey step. Nothing was changed.",
  ConstraintError: "This browser or device can't do that passkey step. Nothing was changed.",
};
export function thrownCopy(e: unknown, fallback: string): string {
  const name = typeof e === "object" && e !== null ? (e as { name?: unknown }).name : undefined;
  return typeof name === "string" && Object.hasOwn(THROWN_COPY, name) ? (THROWN_COPY[name] as string) : fallback;
}

/**
 * H11: is a failed FINISH leg INDETERMINATE (may have committed) rather than a
 * clean refusal? A refusal comes back as an app JSON error body with a 4xx/app-
 * 503 status — nothing committed. An indeterminate finish is one where we never
 * got an app verdict: the fetch threw (status 0 — offline, dropped, DNS), or an
 * edge/gateway answered 502/503/504 with no app body. For a finish that MINTS or
 * publishes, that difference is the difference between "safe to retry" and
 * "retrying double-mints (or loses the only copy of one-time codes)", so we bias
 * toward indeterminate whenever the server's own verdict did not come back.
 */
export function finishIndeterminate(fin: { ok: boolean; status: number; data: unknown }): boolean {
  if (fin.ok) return false;
  if (fin.status === 0) return true; // fetch threw — the request may still have landed
  if (fin.status >= 502 && fin.status <= 504 && fin.data == null) return true; // gateway error, no app answer
  return false;
}

/**
 * Register THIS device as a passkey for the signed-in creator.
 *
 * I3-0034 — A CREATOR WHO HOLDS A PASSKEY APPROVES THE NEXT ONE WITH IT. The
 * server answers a passport that already has a passkey with 403
 * `existing_passkey_required` and a `step_up` challenge (`passkeys.ts:214`);
 * this flow signs it with an existing passkey and asks again with that
 * assertion, so the step-up lives here, in the one register ceremony the
 * console runs (`main.ts` calls this). A first enrolment makes one start call.
 * I34-W3-SEAM-P (I34-W3-CLIENTS W3C-FLOWS-STALE): a start that answers `email_code` (the passport holds no passkey, so
 * the server mailed a code for a FIRST passkey) is never finished here without that code: the flow returns `code_asked`
 * (the options and the code's `expires_at`, null when the server named none) before the device ceremony, and sends no
 * finish, so the challenge is not spent; the caller paints the code view (main.ts addPasskey's code arm). Read on the
 * either start (I34-W3-SEAM-Z): the approved start after a step-up answers `email_code` too when the passport's passkeys
 * were revoked in between (passkeys.ts startPasskeyRegistration tests `existing.length === 0` before the step-up).
 * I34-FX3-PASSKEY (I34-JN-25 W3C-H3): a REFUSED start also hands back `start_refused` (status, code, the whole answer),
 * so main.ts addPasskey can give the first-passkey refusals (409 contact_email_unverified, 503 email_send_failed) the
 * zero-count door's words instead of this flow's generic `error`.
 */
export async function registerPasskeyFlow(
  api: CreatorApi,
  creds: CredentialsApi,
  passportId: string,
): Promise<FlowResult & { credential_id?: string; code_asked?: { options: NonNullable<Awaited<ReturnType<CreatorApi["passkeyRegisterStart"]>>["data"]>; expires_at: string | null }; start_refused?: { status: number; error: string | null; answer: { ok: boolean; status: number; data: unknown } } }> {
  let start = await api.passkeyRegisterStart();
  const stepUp = !start.ok && start.status === 403 && start.data?.error === "existing_passkey_required" ? start.data.step_up : undefined;
  if (stepUp) {
    let approval: AssertionPackage;
    try {
      approval = await performAssertion(stepUp, creds);
    } catch {
      return { ok: false, error: "Adding a passkey needs approval from a passkey you already have, and that approval didn't complete. Nothing was changed. Try again on a device that has your passkey — if you no longer have any, account recovery can reset them." };
    }
    start = await api.passkeyRegisterStart(approval);
  }
  if (!start.ok || !start.data) {
    return { ok: false, error: errOf(start.data, "could not start registration"), start_refused: { status: start.status, error: typeof start.data?.error === "string" ? start.data.error : null, answer: start } };
  }
  const ec = start.data.email_code;
  if (ec !== undefined && ec !== null) {
    return { ok: false, error: FIRST_PASSKEY_CODE_ASKED, code_asked: { options: start.data, expires_at: typeof ec.expires_at === "string" ? ec.expires_at : null } };
  }
  let att: AttestationPackage;
  try {
    att = await performRegistration(start.data, passportId, creds);
  } catch (e) {
    return { ok: false, error: thrownCopy(e, "Couldn't register this device. Nothing was changed. Try again.") };
  }
  const fin = await api.passkeyRegisterFinish(att);
  if (!fin.ok || !fin.data) {
    // B27 (#145): every finish leg runs the guard, not only the ones that mint.
    if (finishIndeterminate(fin)) return { ok: false, indeterminate: true, error: "We couldn't confirm this device was registered — the connection dropped after you approved it. It may already be registered. Reload the Passkeys page before trying again." };
    return { ok: false, error: errOf(fin.data, "could not finish registration") };
  }
  return { ok: true, credential_id: fin.data.credential_id };
}

/**
 * I3-0221 - the contact/recovery address is an authentication factor: once the
 * passport holds a passkey the server answers a change with 403
 * `existing_passkey_required` and a `step_up` bound to the NEW address
 * (`creator-contact-routes.ts contactChangeStepUp`). Sign it with an existing
 * passkey and ask again with that assertion - the same shape `registerPasskeyFlow`
 * answers above. No passkey: one call. While the address on file is VERIFIED and no passkey is active, the server
 * refuses the change 409 `add_passkey_first` (digest.ts setCreatorEmail, D-1110(8)) and api.ts addPasskeyFirstSentence
 * hands on its sentence as `error`; an unverified address still changes on the session alone.
 */
export async function setContactEmailFlow(api: CreatorApi, creds: CredentialsApi, email: string): Promise<FlowResult> {
  let r = await api.creatorContact(email);
  const stepUp = !r.ok && r.status === 403 && r.data?.error === "existing_passkey_required" ? r.data.step_up : undefined;
  if (stepUp) {
    let approval: AssertionPackage;
    try {
      approval = await performAssertion(stepUp, creds);
    } catch (e) {
      return { ok: false, error: thrownCopy(e, "Changing this address needs approval from one of your passkeys, and that approval didn't complete. Nothing was changed. Try again on a device that has your passkey.") };
    }
    r = await api.creatorContact(email, approval);
  }
  if (!r.ok || !r.data) return { ok: false, error: errOf(r.data, "could not send the code") };
  return { ok: true };
}

/**

 * factor change, so a passkey holder is answered 403 `existing_passkey_required`

/** One-way signed broadcast (S70): sign the server-echoed
 * {broadcast_id, message, audience}. flag 2 — the audience rides the SIGNED
 * draft, so the creator's passkey covers who the message was for; the flow
 * returns the SERVER's audience so the UI can state what actually went out. */
export async function broadcastFlow(api: CreatorApi, creds: CredentialsApi, message: string, audience: BroadcastAudienceDto = "public"): Promise<FlowResult & { broadcast_id?: string; audience?: BroadcastAudienceDto }> {
  const start = await api.broadcastStart(message, audience);
  if (!start.ok || !start.data) return { ok: false, error: errOf(start.data, "could not start broadcast") };
  let assertion: AssertionPackage;
  try {
    assertion = await performAssertion(start.data, creds);
  } catch (e) {
    return { ok: false, error: thrownCopy(e, "Couldn't sign that with your passkey. Nothing was changed. Try again.") };
  }
  // Finish with the ECHOED id + message + audience — the signature covers
  // exactly these, so re-sending anything else simply fails verification.
  const fin = await api.broadcastFinish(start.data.broadcast_id, start.data.message, start.data.audience, assertion);
  if (!fin.ok || !fin.data) {
    if (finishIndeterminate(fin)) return { ok: false, indeterminate: true, error: "We couldn't confirm your broadcast went out — the connection dropped mid-send. It may already have been sent. Check your broadcasts before sending again, or a duplicate could go out.", broadcast_id: start.data.broadcast_id };
    return { ok: false, error: errOf(fin.data, "broadcast was refused") };
  }
  return { ok: true, broadcast_id: fin.data.broadcast_id, audience: fin.data.audience ?? start.data.audience };
}

/**
 * F7 HANDSHAKE (#43) — answer a fan's question with a signature.
 *
 * The client sends WHICH handshake and WHICH of two dispositions and nothing
 * else; the server builds the draft from its own stored row on BOTH legs
 * (`DIRECTION.md:441` — the client never chooses what is signed). So the finish
 * leg re-sends only the echoed disposition, and a client that edited it simply
 * fails to verify.
 *
 * A `declined` answer is private to the asking fan and is never an accusation:
 * the console copy says so, and impersonation goes to the report path.
 */
export async function handshakeAnswerFlow(
  api: CreatorApi,
  creds: CredentialsApi,
  handshakeId: string,
  disposition: HandshakeDispositionDto,
): Promise<FlowResult> {
  const start = await api.handshakeAnswerStart(handshakeId, disposition);
  if (!start.ok || !start.data) return { ok: false, error: errOf(start.data, "could not start the answer") };
  let assertion: AssertionPackage;
  try {
    assertion = await performAssertion(start.data, creds);
  } catch (e) {
    return { ok: false, error: thrownCopy(e, "Couldn't sign that with your passkey. Nothing was changed. Try again.") };
  }
  // The ECHOED disposition, never the caller's copy.
  const fin = await api.handshakeAnswerFinish(start.data.handshake_id, start.data.disposition, assertion);
  if (!fin.ok) {
    if (finishIndeterminate(fin)) return { ok: false, indeterminate: true, error: "We couldn't confirm your answer went through — the connection dropped. Check your handshakes before answering again." };
    return { ok: false, error: errOf(fin.data, "the answer was refused") };
  }
  return { ok: true };
}

/** GENERIC signed creator action (S98): the two-leg ceremony for any

 * `creator_claimed` basis — the passkey-backed "I attest this is mine"). */

/**
 * M3 — set payout tags: the two-leg ceremony over the MONEY surface. The client
 * posts the FULL destination set to /start; the server plans the change, builds
 * its OWN ledger drafts and binds the challenge to them (the client never names
 * the actions — the generic creator-action door refuses payout types outright);
 * the finish re-sends the same set + the assertion.
 *
 * Three outcomes the caller must render differently, so the result carries the
 * failing LEG and STATUS alongside the error:
 *  - `unchanged: true` — the set already matches the staged-or-live intent;
 *    nothing was signed, staged or alerted (the server's anti-spam no-op).
 *  - a failure on `leg: "start"` or `"sign"` wrote NOTHING (start is read-only,
 *    and an unsigned set cannot stage) — safe to say "nothing changed".
 *  - a failure on `leg: "finish"` with status 0/5xx is an UNKNOWN OUTCOME: the
 *    change may have staged (with its follower alert). Never render it as
 *    "failed, try again".
 */
export async function payoutTagsFlow(
  api: CreatorApi,
  creds: CredentialsApi,
  tags: PayoutTagInputDto[],
): Promise<FlowResult & { unchanged?: boolean; effective_at?: string; leg?: "start" | "sign" | "finish"; status?: number }> {
  const start = await api.payoutTagsStart(tags);
  if (!start.ok || !start.data) return { ok: false, leg: "start", status: start.status, error: codeOf(start.data, "could not start the payout change") };
  if (start.data.unchanged === true) return { ok: true, unchanged: true };
  const opts = start.data;
  if (typeof opts.challenge !== "string" || typeof opts.rpId !== "string" || !Array.isArray(opts.allowCredentials) || typeof opts.userVerification !== "string") {
    // Neither a no-op nor a challenge we can sign — treat as a start failure
    // (nothing was written; a half-shaped challenge must never reach the
    // authenticator).
    return { ok: false, leg: "start", status: start.status, error: "the server's challenge was unreadable — nothing was changed" };
  }
  let assertion: AssertionPackage;
  try {
    assertion = await performAssertion({ challenge: opts.challenge, rpId: opts.rpId, allowCredentials: opts.allowCredentials, userVerification: opts.userVerification }, creds);
  } catch (e) {
    return { ok: false, leg: "sign", error: thrownCopy(e, "Couldn't sign that with your passkey. Nothing was changed. Try again.") };
  }
  const fin = await api.setPayoutTags(tags, assertion);
  if (!fin.ok || !fin.data) return { ok: false, leg: "finish", status: fin.status, error: codeOf(fin.data, "the payout change was refused") };
  return { ok: true, unchanged: fin.data.unchanged === true, effective_at: fin.data.effective_at };
}

/**
 * M14 — cancel a pending staged payout change: the same two-leg ceremony as
 * payoutTagsFlow, but the client names nothing (it asks only to discard the
 * staged set; the server derives the inverse drafts). Reverting to the live set
 * is safe-direction, so no follower alert and no verified-standing requirement.
 *
 * Same three outcomes, same leg/status discipline:
 *  - `unchanged: true` — nothing was staged, so there was nothing to cancel; no
 *    signature consumed, no write.
 *  - a failure on `leg: "start"`/`"sign"` wrote NOTHING.
 *  - a failure on `leg: "finish"` with status 0/5xx is an UNKNOWN OUTCOME: the
 *    cancel may have committed. Never render it as "failed, try again".
 */
export async function cancelStagedPayoutFlow(
  api: CreatorApi,
  creds: CredentialsApi,
): Promise<FlowResult & { unchanged?: boolean; leg?: "start" | "sign" | "finish"; status?: number }> {
  const start = await api.cancelStagedPayoutStart();
  if (!start.ok || !start.data) return { ok: false, leg: "start", status: start.status, error: codeOf(start.data, "could not start the cancellation") };
  if (start.data.unchanged === true) return { ok: true, unchanged: true };
  const opts = start.data;
  if (typeof opts.challenge !== "string" || typeof opts.rpId !== "string" || !Array.isArray(opts.allowCredentials) || typeof opts.userVerification !== "string") {
    return { ok: false, leg: "start", status: start.status, error: "the server's challenge was unreadable — nothing was changed" };
  }
  let assertion: AssertionPackage;
  try {
    assertion = await performAssertion({ challenge: opts.challenge, rpId: opts.rpId, allowCredentials: opts.allowCredentials, userVerification: opts.userVerification }, creds);
  } catch (e) {
    return { ok: false, leg: "sign", error: thrownCopy(e, "Couldn't sign that with your passkey. Nothing was changed. Try again.") };
  }
  const fin = await api.cancelStagedPayout(assertion);
  if (!fin.ok || !fin.data) return { ok: false, leg: "finish", status: fin.status, error: codeOf(fin.data, "the cancellation was refused") };
  return { ok: true, unchanged: fin.data.unchanged === true };
}

/** Mint a client-side release id (rel_ + 16 hex, CSPRNG — the server binds it
 * into the signed manifest). */
export function mintReleaseId(rand: (n: number) => Uint8Array = (n) => crypto.getRandomValues(new Uint8Array(n))): string {
  return "rel_" + [...rand(8)].map((b) => b.toString(16).padStart(2, "0")).join("");
}

function hex(b: Uint8Array): string {
  let s = "";
  for (const x of b) s += x.toString(16).padStart(2, "0");
  return s;
}

/** sha256 hex of a file's bytes via THE canonical hasher — the collectible's
 * exact_hash fingerprint. (No divergent hashing: sha256Bytes is canonical's.) */
export async function fileExactHash(bytes: Uint8Array): Promise<string> {
  return hex(await sha256Bytes(bytes));
}

/**
 * The whole drop in one signature (S75): start (server validates + mints the
 * broadcast_id + binds the challenge to ALL drafts) → sign → finish with the
 * IDENTICAL input plus the echoed broadcast_id. The release_id is client-minted
 * and part of the signed manifest, so it must be in `input` before calling.
 */
export async function dropFlow(
  api: CreatorApi,
  creds: CredentialsApi,
  input: DropInputDto,
): Promise<FlowResult & { release_id?: string; broadcast_id?: string | null; pool_id?: string | null; generated_codes?: string[] }> {
  const start = await api.dropStart(input);
  if (!start.ok || !start.data) return { ok: false, error: errOf(start.data, "could not start the drop") };
  let assertion: AssertionPackage;
  try {
    assertion = await performAssertion(start.data, creds);
  } catch (e) {
    return { ok: false, error: thrownCopy(e, "Couldn't sign that with your passkey. Nothing was changed. Try again.") };
  }
  // The finish leg re-sends the same input; the announce block (if any) must
  // carry the server-minted broadcast_id the signature covers.
  const finishInput: DropInputDto = input.announce && start.data.broadcast_id
    ? { ...input, announce: { ...input.announce, broadcast_id: start.data.broadcast_id } }
    : input;
  const fin = await api.dropFinish(finishInput, assertion);
  if (!fin.ok || !fin.data) {
    if (finishIndeterminate(fin)) return { ok: false, indeterminate: true, error: "We couldn't confirm your drop went through — the connection dropped after you signed. It may already be published and any vouchers already minted. Check your releases before trying again; retrying can publish a duplicate or mint a second set of one-time codes." };
    return { ok: false, error: errOf(fin.data, "the drop was refused") };
  }
  return { ok: true, release_id: fin.data.release_id, broadcast_id: fin.data.broadcast_id, pool_id: fin.data.pool_id, ...(fin.data.generated_codes ? { generated_codes: fin.data.generated_codes } : {}) };
}

/**
 * Lightweight collectible mint (release-lite): the simplest signed-mint ceremony —
 * a name + optional art, one signature, published live. When art bytes are given,
 * the flow BINDS the signed `art_hash` to the exact bytes it will upload (the
 * caller can't drift the signed hash from the stored image), runs the signed mint,
 * then uploads. Storage is a SEPARATE, soft step: a failed upload never unwinds the
 * already-signed mint — the collectible is still minted and provable, and the
 * identical bytes can be re-uploaded later (content-addressed, idempotent). So `ok`
 * reflects the MINT; `stored` (present only when art was supplied) reflects the bytes.
 */
export async function collectibleMintFlow(
  api: CreatorApi,
  creds: CredentialsApi,
  input: CollectibleInputDto,
  bytes?: Uint8Array,
): Promise<FlowResult & { collectible_id?: string; stored?: boolean; storeError?: string; retryable?: boolean }> {
  // Bind the signed art_hash to the EXACT bytes we will upload (if any).
  const bound: CollectibleInputDto = bytes ? { ...input, art_hash: await fileExactHash(bytes) } : input;
  const start = await api.collectibleMintStart(bound);
  if (!start.ok || !start.data) return { ok: false, error: errOf(start.data, "could not start the mint") };
  let assertion: AssertionPackage;
  try {
    assertion = await performAssertion(start.data, creds);
  } catch (e) {
    return { ok: false, error: thrownCopy(e, "Couldn't sign that with your passkey. Nothing was changed. Try again.") };
  }
  const fin = await api.collectibleMintFinish(bound, assertion);
  if (!fin.ok || !fin.data) {
    if (finishIndeterminate(fin)) return { ok: false, indeterminate: true, error: "We couldn't confirm the mint — the connection dropped after you signed. Your collectible may already be minted. Check your collectibles before minting again to avoid a duplicate." };
    return { ok: false, error: errOf(fin.data, "the mint was refused") };
  }
  const minted = { ok: true as const, collectible_id: fin.data.collectible_id };
  if (!bytes) return minted; // name-only: nothing to store

  const up = await api.uploadMedia(bytes);
  return {
    ...minted,
    stored: up.ok,
    // M38: the collectible is minted + provable and its art_hash is bound, so
    // re-uploading the SAME bytes is idempotent and succeeds once storage is
    // back — mark it retryable so the result view offers a REAL retry rather
    // than promising one in prose with no button behind it.
    ...(up.ok ? {} : { storeError: errorCopy(up.data?.error, up.status === 503 ? "media storage is not available yet" : "could not store the art"), retryable: true }),
  };
}

/**
 * S91 · Slice 2 — sign a composited image AND have Dhwaja store it. The order is
 * load-bearing: the drop's passkey signature covers `media[0].exact_hash =
 * sha256(bytes)`, and the upload only succeeds because that hash is now on a
 * PUBLISHED release of theirs. So we (1) FORCE the hash to match the exact bytes
 * we'll upload — the caller can't drift the signed hash from the stored bytes —
 * (2) run the signed drop, then (3) upload.
 *
 * Storage is a SEPARATE, softer step: if the drop succeeds but the upload fails
 * (network / bucket unprovisioned / cap), the collectible is still minted and
 * provable — the art just isn't hosted yet, and re-uploading the identical bytes
 * later is idempotent. So `ok` reflects the DROP; `stored` reflects the bytes.
 */
export async function signAndStoreArtFlow(
  api: CreatorApi,
  creds: CredentialsApi,
  input: DropInputDto,
  bytes: Uint8Array,
): Promise<FlowResult & { release_id?: string; broadcast_id?: string | null; pool_id?: string | null; generated_codes?: string[]; stored?: boolean; storeError?: string }> {
  // (1) Bind the signed manifest to the EXACT bytes we will upload. A single
  //     image fingerprint; anything the caller put in media[] is overridden.
  const exact_hash = await fileExactHash(bytes);
  const bound: DropInputDto = { ...input, collectible: { ...input.collectible, media: [{ media_kind: "image", exact_hash }] } };

  // (2) The signed drop (mints + optionally announces/attaches codes).
  const drop = await dropFlow(api, creds, bound);
  if (!drop.ok) return drop;

  // (3) Store the bytes — soft: a failure never unwinds the (already-signed) drop.
  const up = await api.uploadMedia(bytes);
  return {
    ...drop,
    stored: up.ok,
    ...(up.ok ? {} : { storeError: errorCopy(up.data?.error, up.status === 503 ? "media storage is not available yet" : "could not store the art (you can retry the upload later)") }),
  };
}

/**
 * THE GATE IN FRONT OF EVERY NAVIGATION THIS CONSOLE PERFORMS.
 *
 * The console hands the creator off to exactly two places it does not control:
 * a payment processor's checkout/portal, and an OAuth provider's consent screen.
 * Both assign a server-supplied string to `window.location.href`, which is a
 * script sink for `javascript:` and a phishing sink for anything else. Both used
 * an inline `^https://` prefix test, in `main.ts`, where nothing could import or
 * test them; the billing one had no test at all and the OAuth one only differed
 * by an `i` flag.
 *
 * One gate, in one importable place, for both. It lives here rather than in
 * `billing.ts` because that module's header commits it to holding nothing but
 * entitlement-shaped reads, and a general URL check is not one.
 *
 * The URLs come from our own API, so this is a belt to the server's braces — but
 * it is the last thing between a compromised or mis-templated response and a
 * navigation the creator will trust because the console performed it.
 *
 * Stricter than the prefix match it replaces, in ways a prefix cannot express:
 *   - parsed, not prefix-matched, so whitespace- and control-character-bearing
 *     values cannot pass on the strength of their first eight bytes;
 *   - no embedded credentials — `https://checkout.stripe.com@evil.example/`
 *     reads as Stripe to a person and resolves to evil.example in a browser,
 *     which is exactly the shape a hand-off must not accept.
 *
 * It deliberately does NOT pin an allow-list of hostnames: Stripe's checkout and
 * portal domains move, and the OAuth providers are configured server-side, so a
 * stale list here would fail a paying or claiming creator closed on a legitimate
 * hand-off. The scheme and the absence of credentials are what decide this.
 */
/**
 * THE RECOVERY DOOR'S URL, built from the build-time public origin.
 *
 * It lives HERE rather than inline in `main.ts` for the reason the header above
 * gives and the suite enforces: `main.ts` may not carry its own URL scheme test,
 * because an inline prefix check is exactly what this module replaced. This one
 * is not a hand-off gate — the value is a compiled-in constant, never a server
 * response — but it is a URL check, and one place is the rule.
 *
 * ORIGIN-SHAPED ONLY (no path, no query, no credentials): the console appends
 * `/#/recovery` to it, so anything with a path of its own would build a nonsense
 * link. Returns null for anything else, and the view renders no-link copy rather
 * than a dead link — the person following this one has already run out of other
 * options.
 */
export function publicRecoveryUrl(base: unknown): string | null {
  if (!isSafeHandoffUrl(base)) return null;
  const u = new URL(base);
  if (u.pathname !== "/" || u.search !== "" || u.hash !== "") return null;
  return `${u.origin}/#/recovery`;
}

export function isSafeHandoffUrl(url: unknown): url is string {
  if (typeof url !== "string" || url === "") return false;
  // `new URL` tolerates leading/trailing C0 and space, and browsers strip tabs
  // and newlines out of a URL entirely. A URL our own API emitted carries none
  // of them, and one that does is not an answer we act on.
  if (/[\u0000-\u0020\u007f]/.test(url)) return false;
  let parsed: URL;
  try {
    parsed = new URL(url);
  } catch {
    return false;
  }
  if (parsed.protocol !== "https:") return false;
  if (parsed.username !== "" || parsed.password !== "") return false;
  return parsed.hostname !== "";
}
