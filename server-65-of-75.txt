import type { DbLike } from "./db.js";

/**
 * I29-FX-OPS (D-1037(15)): a LEAF module (it imports only the DbLike type) holding the recovery plan's forced actions
 * and the "does this resolution unbind a payout account" predicate, so dual_control.ts (a paywall verdict module) can
 * import the predicate statically instead of `await import("./recovery.js")`. Moved verbatim from recovery.ts, which
 * imports `forcedFor` and re-exports both public names; `FORCED_ACTIONS`' rationale stays on recovery.ts's docblock
 * ("THE ACTIONS AN OPERATOR CANNOT LEAVE OUT").
 */
export type RecoveryAction = "force_logout" | "reset_passkeys" | "retract_payout_tags" | "revoke_connect_binding" | "revoke_agency_grants" | "evict_contact_channels" | "transfer_account" | "reactivate_account" | "revoke_fan_sign_in";

export const FORCED_ACTIONS: Readonly<Record<string, readonly RecoveryAction[]>> = {
  hijack: ["force_logout", "reset_passkeys", "retract_payout_tags", "revoke_connect_binding", "revoke_agency_grants", "evict_contact_channels"],
  // I3-2202 (D-0913(3) option A): a lost-passkey case exists to reset access, so
  // `approved` (which the creator reads as `resolved`, creator-recovery.ts) must
  // always mean the passkeys WERE reset — never an operator's untyped tick.
  lost_passkey: ["reset_passkeys"],
};

/** The actions this kind requires on top of whatever the operator named. */
export function forcedFor(kind: unknown): readonly RecoveryAction[] {
  return (typeof kind === "string" ? FORCED_ACTIONS[kind] : undefined) ?? [];
}

/**
 * I29-LOOSE L6 (I29-OPS J1; orchestrator ruling under D-1030 I23-Q3): does resolving case `itemId` under the signed
 * `recovery:resolve` key `paramsKey` run `revoke_connect_binding` (unbind the payout account)? dual_control.ts
 * `withDualControl` asks this and then requires what `connect:unbind` requires (two people on every deployment); every
 * other resolution keeps the deployment's threshold. The plan read is the one `resolveRecoveryCase` runs: the stored
 * actions (an operator-authored case) else the plan in the key (`<id>|a=...`, `recoveryPlanKey`, a request case), plus
 * the kind's forced actions. FAIL CLOSED: an unreadable payload, or a request case whose key carries no plan, answers
 * true (the gate cannot see what will run). No case at all answers false: `resolveRecoveryCase` refuses it.
 */
export async function recoveryResolveUnbindsPayout(db: DbLike, itemId: string, paramsKey: string): Promise<boolean> {
  const row = await db.prepare("SELECT payload FROM review_queue WHERE item_id = ? AND item_type = 'dispute'").bind(itemId).first<{ payload: string }>();
  if (!row) return false;
  let p: { kind?: unknown; actions?: unknown };
  try {
    p = JSON.parse(row.payload) as { kind?: unknown; actions?: unknown };
  } catch {
    return true;
  }
  let named: string[];
  if (Array.isArray(p.actions) && p.actions.length > 0) named = p.actions.map(String);
  else {
    const bar = paramsKey.indexOf("|");
    const m = bar >= 0 ? /^a=([^;]*);/.exec(paramsKey.slice(bar + 1)) : null;
    if (!m) return true;
    named = m[1] ? m[1].split(",") : [];
  }
  return [...named, ...forcedFor(p.kind)].includes("revoke_connect_binding");
}

/**
 * S96 — operator-governed account recovery & disputes.
 *
 * This module adds NO new primitive and NO new ledger event TYPE. It is pure
 * orchestration: a recovery case is a `dispute` review-queue item, and resolving
 * it composes the already-hardened, individually-audited primitives —
 *   • S94 operator force-logout  (bumpCreatorEpoch — invalidates live sessions)
 *   • S96 operator passkey reset (operatorResetPasskeys — break-glass revoke-all)
 *   • payout retraction          (retractAllPayoutTags — live AND staged)
 *   • agency-grant revocation    (agency_revoked, one per live grant)
 *   • S95 account transfer       (account_transferred ledger event)
 * — behind full-operator + dual-control + audit at the route. The queue's
 * disposition trail lives in review_queue + audit_log; the only ledger writes
 * are the grant revocations and the S95 transfer, both reusing EXISTING event
 * types (so no fold/golden churn here).
 *
 * The adjudicated path for: a lost passkey (reset so the owner re-enrolls), a
 * hijacked passport (force-logout + reset to lock the attacker out), and an
 * account dispute / competing claim (transfer the contested account to the
 * rightful passport, then lock the losing one down).
 *
 * WHAT "LOCK THE ATTACKER OUT" HAD TO GROW TO MEAN (invariant 3). The three
 * original actions covered sessions, devices and account ownership, and left
 * two authorities standing after a full lockdown: the payout destinations the
 * attacker set or STAGED (a staged one has its own clock and would surface days
 * later, by cron or by the public read), and any agency they authorized — the
 * worst of the two, because `reset_passkeys` revokes the only key that could
 * have revoked the grant, so the recovered creator became the one party unable
 * to stop it. Both are now FORCED for `kind === "hijack"`; see FORCED_ACTIONS.
 */

/** The four things that go wrong. EXPORTED since 0043 so the inbox-proven
 * filing path can validate a caller's `kind` BEFORE it burns a single-use
 * code — a refusal that arrives after the code is spent costs a locked-out
 * creator their one attempt and tells them it worked. */
export const RECOVERY_KINDS: ReadonlySet<string> = new Set(["lost_passkey", "hijack", "agency_dispute", "competing_claim"]);
const KINDS = RECOVERY_KINDS;
const ACTIONS: ReadonlySet<string> = new Set(["force_logout", "reset_passkeys", "retract_payout_tags", "revoke_connect_binding", "revoke_agency_grants", "evict_contact_channels", "transfer_account", "reactivate_account"]);
// Fixed EXECUTION order, independent of the order actions were requested in. The
// monotone lockdown steps run FIRST (they never fail on a live passport, and
// killing a hijacker's sessions/passkeys/destinations is the urgent part), then
// the fallible ledger writes run LAST. So the only reachable partial outcome is
// "locked down but not transferred" — the fail-SAFE direction — and a re-resolve of
// a case left un-approved by a mid-run failure converges instead of getting stuck:
// a grant already revoked is not in the live set the next run reads, the transfer
// skips when the account is already at the destination, and `reset_passkeys` on a
// retry is skipped only when no passkey is live (D-0931(5) superseding I12-FX7's D-0884(4) skip).
// NOT a no-op on retry: `force_logout` bumps the epoch AGAIN, so a retry signs out
// every session, including one the rightful owner opened between attempts.
//
// `retract_payout_tags` sits with the lockdown steps because it is one guarded
// UPDATE and cannot fail on a live passport; `revoke_agency_grants` sits with the
// ledger writes because it appends. Both are before `transfer_account`, so a
// transfer never hands a passport over with the attacker's money destinations or
// agency grants still attached.
//
// F5: `reactivate_account` runs LAST — after a transfer has put the key on the
// rightful passport — and appends `account_reactivated` with basis
// `control_reproven`, which the ledger refuses unless the creator proved
// control of the key within the reactivation window. A refusal surfaces as
// the resolve's 409 and the case stays open; every earlier step is idempotent,
// so re-resolving once the creator has re-proved converges.
// I3-0114: `evict_contact_channels` sits with the lockdown steps — it is
// guarded UPDATEs/DELETEs on the subject's own rows and cannot fail on a live
// passport — and BEFORE `transfer_account`, for the same reason the payout and
// grant steps are: a transfer must never hand a passport over still carrying
// the attacker's contact address, push devices or live recovery codes.
const EXEC_ORDER: RecoveryAction[] = ["force_logout", "reset_passkeys", "retract_payout_tags", "revoke_connect_binding", "revoke_agency_grants", "evict_contact_channels", "transfer_account", "reactivate_account"];

 *
 * Invariant: after a confirmed hijack recovery, NO authority the attacker
 * created may remain capable of moving money, acting for the creator, or
 * simply staying signed in as them. The three original lockdown levers
 * (`force_logout`, `reset_passkeys`, `transfer_account`) covered sessions,

/**
 * The operator's named spare, normalized the way an address is normalized before
 * it is looked up (trim + lowercase, `creator-recovery.normalizeAddress`).
 *
 * RE-DECLARED, NOT IMPORTED: `creator-recovery.ts` imports THIS module
 * (`submitRecoveryRequest`), so an import back would close a cycle.
 *
 * It deliberately does NOT re-implement that function's `EMAIL_RE`. A spare only
 * ever has an effect when it is EQUAL to the address already stored on the row,
 * and that address went through the real validator on its way in — so a
 * malformed spare simply never matches and evicts, which is the safe direction.
 * A second copy of an address regex would be a validator to keep in step for no
 * behaviour gained.
 */
function normalizeSpareAddress(v: unknown): string | null {
  if (typeof v !== "string") return null;
  const a = v.trim().toLowerCase();
  return a.length > 0 && a.length <= 320 ? a : null;
}

/**
 * A STABLE key fragment for a remediation plan, for the dual-control signature.
 * A request-origin case's plan is supplied at resolve time, so co-signers must
 * approve the SAME plan — not merely "resolve case X". This folds the plan into
 * the dual-control paramsKey. The security-relevant fields (the action set and
 * the transfer destination) come FIRST so they survive the paramsKey's length
 * cap; the long `contested_account_key` is last. Not a hash (the single-hasher
 * invariant confines SHA-256 to canonical/events) — a canonical string suffices
 * for equality, and truncating only the trailing account-key can't let two
 * genuinely different destinations/action-sets collide.
 */
export function recoveryPlanKey(plan: unknown, kind?: string): string {
  const i = (plan ?? {}) as Record<string, unknown>;
  const named = Array.isArray(i.actions) ? [...i.actions].filter((a) => typeof a === "string").map(String) : [];
  // THE KEY IS THE PLAN THAT WILL RUN, not the plan that was typed. `validatePlan`
  // forces the hijack lockdown actions in; if this hashed only what the operator
  // wrote, two co-signers who typed different subsets of the SAME executed plan
  // would produce two different keys and never combine — and worse, the key would
  // stop describing what actually executes, which is the one thing a dual-control
  // signature is for. Normalized the same way (dedup + sort) as there.
  const actions = [...new Set([...named, ...forcedFor(kind)])].sort();
  const to = typeof i.to_passport === "string" ? i.to_passport : "";
  // I3-0114 — AHEAD OF `k`, because it is security-relevant and `k` is the long
  // field the paramsKey cap truncates. `spare_contact_email` decides whether the
  // attacker's way back in survives the resolution, so two co-signers must be
  // approving the SAME spare, not merely the same action set.
  const spare = normalizeSpareAddress(i.spare_contact_email) ?? "";
  const k = typeof i.contested_account_key === "string" ? i.contested_account_key : "";
  // QUEUE #146 (D-0807): a fan plan's clear-email option is part of what two
  // co-signers approve, ahead of `k`; absent, the key is byte-identical to before.
  const ce = kind === FAN_TAKEOVER_KIND && i.clear_email === true ? "ce=1;" : "";
  return `a=${actions.join(",")};to=${to};spare=${spare};${ce}k=${k}`;
}

/** The remediation PLAN: the action set (+ transfer fields). Shared by the
 * operator-authored open path AND the operator's resolve-time plan for a
 * self-service request. `passportId` is the case subject (the from-passport), used
 * for the no-self-transfer check. Fail-closed. */
function validatePlan(passportId: string, raw: unknown, kind?: string): { actions: RecoveryAction[]; contested_account_key?: string; to_passport?: string; spare_contact_email?: string; clear_email?: true } {
  if (kind === FAN_TAKEOVER_KIND) return validateFanPlan(passportId, raw);
  const i = (raw ?? {}) as Record<string, unknown>;
  if (!Array.isArray(i.actions) || i.actions.length === 0) throw new RecoveryError("at least one action is required");
  const actions: RecoveryAction[] = [];
  for (const a of i.actions) {
    if (typeof a !== "string" || !ACTIONS.has(a)) throw new RecoveryError(`unknown action: ${String(a)}`);
    if (actions.includes(a as RecoveryAction)) throw new RecoveryError(`duplicate action: ${a}`);
    actions.push(a as RecoveryAction);
  }
  // FORCED, NOT DEFAULTED. A `hijack` resolution always retires the payout
  // destinations and revokes the agency grants — an operator cannot resolve one
  // without them (see FORCED_ACTIONS). Appended rather than reordered so the
  // stored payload still shows what the operator chose; EXEC_ORDER decides the
  // running order regardless.
  for (const a of forcedFor(kind)) if (!actions.includes(a)) actions.push(a);
  const out: { actions: RecoveryAction[]; contested_account_key?: string; to_passport?: string; spare_contact_email?: string } = { actions };
  // I3-0114. Validated even when `evict_contact_channels` is not in the set, so a
  // typo in the one field that decides whether the attacker keeps a door is a 400
  // at the operator rather than a silently ignored key. Fail-closed: anything
  // unusable is a refusal, never a quiet "spare nothing" — an operator who meant
  // to spare an address must not learn it failed from the creator's support
  // ticket.
  if (i.spare_contact_email !== undefined) {
    const spare = normalizeSpareAddress(i.spare_contact_email);
    if (!spare) throw new RecoveryError("spare_contact_email must be a non-empty address of at most 320 characters");
    out.spare_contact_email = spare;
  }
  if (actions.includes("reactivate_account") && !actions.includes("transfer_account")) {
    // F5: the key to re-list, on THIS passport (with a transfer, the same key
    // moves first and is re-listed on the destination).
    if (typeof i.contested_account_key !== "string" || i.contested_account_key.length === 0 || i.contested_account_key.length > 512) {
      throw new RecoveryError("reactivate_account requires contested_account_key (<=512 chars)");
    }
    out.contested_account_key = i.contested_account_key;
  }
  if (actions.includes("transfer_account")) {
    if (typeof i.contested_account_key !== "string" || i.contested_account_key.length === 0 || i.contested_account_key.length > 512) {
      throw new RecoveryError("transfer_account requires contested_account_key (<=512 chars)");
    }
    if (typeof i.to_passport !== "string" || !PSP.test(i.to_passport)) throw new RecoveryError("transfer_account requires a psp_ to_passport");
    if (i.to_passport === passportId) throw new RecoveryError("to_passport and passport_id are the same");
    out.contested_account_key = i.contested_account_key;
    out.to_passport = i.to_passport;
  }
  return out;
}

/**
 * SLICE 21 — the SHAPE half of `validatePlan`, exported so a route can run it
 * BEFORE the dual-control gate.
 *
 * `withDualControl` records the sign-off and its audit line before it calls
 * `execute()`, so a plan validated only inside `resolveRecoveryCase` banks an
 * approval for a decision that can never run: at `OPS_APPROVALS_REQUIRED >= 2`,
 * `{actions:["wizard"]}` against a real request-origin case returns
 * `{ok:true, pending:true}`, writes a `pending_actions` row nothing cleans up,
 * and 400s the moment a second approver arrives. Slice 20 fixed four sibling
 * routes and this one was missed — its plan is not merely IN the key, it IS the
 * decision being signed.
 *
 * It runs every check `validatePlan` does EXCEPT the self-transfer equality
 * (`to_passport === passportId`), which needs the case's own subject; the route
 * reads the queue row for its key anyway and passes it, so nothing is skipped in
 * practice. `resolveRecoveryCase` still validates in full — this is a pre-gate,
 * never a replacement, for the reason slices 14/15/18 each paid for: a rule
 * placed only at the door is a rule somebody adds a second door past.

export function assertRecoveryPlanShape(raw: unknown, kind?: string, passportId?: string): void {
  validatePlan(passportId ?? "", raw, kind);
}

/** Validate + normalize an OPERATOR-AUTHORED case payload (actions set up front).
 * Rejects unknown kinds/actions, bad ids, empty/duplicate action sets, and
 * missing per-action fields. Fail-closed. */
function validate(input: unknown): { passport_id: string; payload: RecoveryPayload } {
  const i = (input ?? {}) as Record<string, unknown>;
  if (typeof i.kind !== "string" || !KINDS.has(i.kind)) throw new RecoveryError("unknown recovery kind");
  if (typeof i.passport_id !== "string" || !PSP.test(i.passport_id)) throw new RecoveryError("passport_id must be a psp_ id");
  const plan = validatePlan(i.passport_id, i, i.kind);
  const payload: RecoveryPayload = { kind: i.kind as RecoveryKind, actions: plan.actions, origin: "operator" };
  if (plan.contested_account_key !== undefined) payload.contested_account_key = plan.contested_account_key;
  if (plan.to_passport !== undefined) payload.to_passport = plan.to_passport;
  if (plan.spare_contact_email !== undefined) payload.spare_contact_email = plan.spare_contact_email;
  if (i.note !== undefined) {
    if (typeof i.note !== "string" || i.note.length > 1000) throw new RecoveryError("note must be a string <= 1000 chars");
    payload.note = i.note;
  }
  return { passport_id: i.passport_id, payload };
}

/** Open a recovery case → a `dispute` review-queue item (subject = the affected
 * passport). Validated + audited. Does NOT execute anything.
 * I29-Q3 / RGC-N3 (I34-FX1-CORE): the `enqueue:dispute` audit detail is the case id only, as the request-intake creates
 * write it (openRecoveryRequest's two creates): review.ts `enqueue` copies the payload into the detail, and an operator
 * case's payload carries `spare_contact_email` and the operator's `note`. So the two rows are written here, in one
 * batch, without `enqueue`. */
export async function openRecoveryCase(
  db: DbLike,
  actor: string,
  input: RecoveryCaseInput | unknown,
  now: () => string = () => new Date().toISOString(),
): Promise<{ item_id: string }> {
  const { passport_id, payload } = validate(input);
  const item_id = mintId("itm");
  const ts = now();
  await db.batch([
    db
      .prepare("INSERT INTO review_queue (item_id, item_type, state, subject, payload, created_at, updated_at, import_batch_id) VALUES (?, 'dispute', 'open', ?, ?, ?, ?, NULL)")
      .bind(item_id, passport_id, JSON.stringify(payload), ts, ts),
    auditStmt(db, actor, "enqueue:dispute", item_id, item_id, ts),
  ]);
  return { item_id };
}

/**
 * SELF-SERVICE recovery request (S99). A signed-in fan/creator/agency describes a
 * problem with a passport they can't fix themselves (lost passkey, hijack,
 * dispute). This is a REQUEST, not an action: it enqueues a `dispute` case with
 * `origin='request'` and NO actions — it executes nothing. An operator triages it
 * and supplies the remediation plan at resolve time (`resolveRecoveryCase`).
 *
 * Anti-abuse, because the requester need NOT control the disputed passport (that's
 * the whole point): a signed-in session is required (the route enforces it and
 * passes the identity here), requests are rate-limited per requester per day, and
 * repeat requests against the same passport are DEDUPLICATED into one open case
 * (distinct requesters counted, not a new item each time) so the queue can't be
 * flooded. The requester cannot choose the actions or the transfer destination —
 * only the operator can, at resolve time.
 */
/**
 * WHERE A REQUEST CAME FROM, as a first-class fact rather than a note in the
 * description — because the dedupe folds cases together and the fold keeps the
 * FIRST filer's `kind`, `description` and `contact`.
 *
 * `"request"` is the historical value: a signed-in fan, creator or agency using
 * the self-service form. `"creator_inbox"` is a filing by somebody who proved
 * control of the passport's own verified contact address (creator-recovery.ts),
 * which is the strongest provenance this product can attach to a recovery case.
 *
 * THEY MUST NOT FOLD INTO EACH OTHER, and that is the whole reason this
 * parameter exists. Measured: an attacker with a throwaway fan account
 * pre-files `kind:"hijack"`, `description:"this was stolen from me, transfer it
 * to my passport"`, `contact:"attacker@…"`. The real creator later proves their
 * inbox and files `lost_passkey` with their own words — and the fold discards
 * their kind, their description and their contact, keeps the attacker's, and
 * appends `creator:<passport_id>` to the requester list. The operator then reads
 * the attacker's narrative endorsed by the strongest signal the system emits,
 * with the attacker's address as the reply-to, and `hijack` forces a payout
 * retraction on resolution.
 *
 * Separate origins mean separate cases: two rows an operator can compare, each
 * carrying its own filer's words.
 */
/**
 * WHERE A RECOVERY CASE CAME FROM, and the reason this discriminator exists.
 *
 * A case FOLDS into an open one for the same passport and kind, which is right
 * for two people reporting the same hijack and was wrong across origins: an
 * attacker pre-filing a `hijack` from a throwaway fan account captured the
 * creator's own inbox-proven filing as a corroborator, keeping the attacker's
 * description and contact (0043, review H1). The fold's guard predicate binds
 * this value, so a case can only ever absorb another of the SAME origin.
 *
 * `contest_proof` (X1) is the third: a creator who has DEMONSTRATED control of
 * an account listed on somebody else's passport. It must not fold into either
 * of the others — the evidence is different in kind, and an operator reading it
 * needs to know a proof landed rather than that somebody asserted something.
 */
export type RecoveryOrigin = "request" | "creator_inbox" | "contest_proof";

export async function submitRecoveryRequest(
  db: DbLike,
  requester: RecoveryRequester,
  input: RecoveryRequestInput,
  nowMs: number,
  now: () => string = () => new Date().toISOString(),
  origin: RecoveryOrigin = "request",
): Promise<{ item_id: string; deduped: boolean }> {
  const i = (input ?? {}) as RecoveryRequestInput;
  if (typeof i.kind !== "string" || !KINDS.has(i.kind)) throw new RecoveryError("unknown recovery kind");
  const description = typeof i.description === "string" ? i.description.trim() : "";
  if (!description || description.length > 1000) throw new RecoveryError("a description (1..1000 chars) is required");
  const contact = i.contact !== undefined ? String(i.contact).trim() : "";
  const evidence = typeof i.evidence === "string" && i.evidence.trim().length > 0 ? i.evidence.trim().slice(0, 500) : undefined;
  if (contact.length > 200) throw new RecoveryError("contact must be <= 200 chars");

  // Resolve the target passport: an explicit psp_ id, or a platform+handle we
  // canonicalize + look up. A plain 404 leaks nothing new ONLY for a passport
  // /p/:id would itself answer — and /p/:id 404s a SUPPRESSED one. So on the
  // public door (origin "request") a suppressed passport is ABSENT: see the
  // existence read below (QUEUE #382).
  let passportId: string | null = null;
  if (typeof i.target_passport_id === "string" && PSP.test(i.target_passport_id)) {
    passportId = i.target_passport_id;
  } else if (typeof i.target_platform === "string" && typeof i.target_handle === "string") {
    // canonicalAccount throws on an unknown/malformed platform+handle — treat an
    // uncanonicalizable target as simply not found (a 404, not a 500/log-spam).
    let key: string;
    try {
      key = canonicalAccount(i.target_platform, "handle", i.target_handle).canonical;
    } catch {
      throw new RecoveryError("no such passport", 404);
    }
    const row = await db.prepare("SELECT passport_id FROM passport_accounts WHERE account_key = ? AND status = 'active'").bind(key).first<{ passport_id: string }>();
    passportId = row?.passport_id ?? null;
  } else {
    throw new RecoveryError("target_passport_id, or target_platform + target_handle, is required");
  }
  if (!passportId) throw new RecoveryError("no such passport", 404);
  // QUEUE #382: ONE statement on both branches, so the read sequence is the same
  // for a suppressed passport as for a never-listed one. Before this the public
  // door answered a suppressed passport 200 + an enqueued case and a never-listed
  // one 404 — a suppression oracle any signed-in fan could ask. The SYSTEM
  // origins keep the plain read: `creator-recovery.ts` and `contested-accounts.ts`
  // swallow this refusal and answer uniformly, so they are not oracles, and a
  // suppressed creator's own inbox-proven case must still reach an operator.
  // D-0810(8): a RETRACTED passport is as absent as a suppressed one (both 404
  // at GET /p/:id, passport-view.ts livePassport); a signed-in creator filing
  // about ITS OWN passport is the owner and learns nothing, so it reads the
  // plain row.
  const owner = requester.kind === "creator" && requester.id === passportId;
  const exists = await db
    .prepare(
      origin === "request" && !owner
        ? "SELECT 1 AS x FROM passports p WHERE p.passport_id = ? AND p.retracted = 0 AND NOT EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = p.passport_id)"
        : "SELECT 1 AS x FROM passports WHERE passport_id = ?",
    )
    .bind(passportId)
    .first();
  if (!exists) throw new RecoveryError("no such passport", 404);

  // Rate-limit per requester (a fan and a creator session get separate buckets).
  // I3-1889 (JN-05, I23-FX-TC): an agency files as a seat (`<agencyId>:<memberId>`, index.ts), but its daily budget is
  // the AGENCY's — keyed on the agency id alone, so several admin seats share one REQUEST_DAILY_CAP, not one each.
  const rkey = `${requester.kind}:${requester.id}`;
  const budgetKey = requester.kind === "agency" ? `agency:${requester.id.split(":")[0]}` : rkey;
  if (!(await consumeRateLimit(db, `rcvreq:${budgetKey}`, nowMs, REQUEST_DAILY_CAP))) {
    throw new RecoveryError("too many recovery requests today — try again tomorrow", 429);
  }

  // Dedup: fold repeat requests for the same passport into the one open
  // request-origin case (count distinct requesters; don't spawn N items).
  const foldedId = await foldIntoOpenRequestCase(db, passportId, rkey, i.kind, now, origin, evidence);
  if (foldedId) return { item_id: foldedId, deduped: true };

  // No open request-origin case exists — create one. AUDIT ITEM #77: the SELECT
  // above and this create used to be check-then-act, so two SIMULTANEOUS
  // first-requests for the same passport (each finding nothing open) both
  // enqueued, splitting requester corroboration across two duplicate cases and
  // weakening the exact operator signal (request_count / distinct requesters)
  // this dedup exists to build. The INSERT is now guarded: it lands ONLY while
  // there is still no open request-origin dispute for the subject, and a loser
  // folds into the winner instead. json_extract reads the origin the payload
  // carries; the guard is request-origin-only so a concurrent OPERATOR case
  // never blocks a fan's request (matching the fold loop's own `continue`).
  const payload: RecoveryPayload = { kind: i.kind as RecoveryKind, actions: [], origin, description, requesters: [rkey], request_count: 1 };
  if (evidence) payload.evidence = [evidence];
  if (contact) payload.contact = contact;
  const payloadJson = JSON.stringify(payload);
  const itemId = mintId("itm");
  const ts = now();
  // Guarded on THIS origin, not on "request": an open inbox-proven case must not
  // block a fan's request and a fan's request must not block an inbox-proven
  // case. Each origin owns at most one open slot.
  // BF-16: the insertion guard asks the SAME question the fold above asks —
  // origin AND kind — because a guard narrower than its lookup is how a filing
  // that correctly refused to fold then failed to create, and fell through to
  // the unguarded fallback. Each (origin, kind) owns at most one open slot.
  const noOpenRequestCase =
    "NOT EXISTS (SELECT 1 FROM review_queue r WHERE r.item_type = 'dispute' AND r.state = 'open' AND r.subject = ? AND json_extract(r.payload, '$.origin') = ? AND json_extract(r.payload, '$.kind') = ?)";
  // THE AUDIT LINE IS GUARDED ON A DIFFERENT PREDICATE, AND IT HAS TO BE.
  //
  // Both statements used to carry `noOpenRequestCase`. Statements in a D1 batch
  // run IN ORDER inside one transaction, so by the time the second one is
  // evaluated the first has already inserted the open request-origin dispute —
  // which makes `NOT EXISTS (…)` FALSE. The audit row was therefore never
  // written for a SUCCESSFULLY-ENQUEUED case, and was written for exactly none
  // of them: the guard fired on success and on failure alike, so the one enqueue
  // path that reaches the review queue from an unauthenticated door left no
  // audit trail at all. (Verified by counting rows, not by reading the SQL.)
  //
  // The correct question for the audit row is "did OUR insert land?", which is
  // `EXISTS (a review_queue row with this item_id)` — true precisely when the
  // statement above won the slot, false when it lost to a concurrent first
  // request. `item_id` is freshly minted here and is the primary key, so nothing
  // else can satisfy it.
  const ourInsertLanded = "EXISTS (SELECT 1 FROM review_queue r WHERE r.item_id = ?)";
  await db.batch([
    db
      .prepare(
        `INSERT INTO review_queue (item_id, item_type, state, subject, payload, created_at, updated_at, import_batch_id)
         SELECT ?, 'dispute', 'open', ?, ?, ?, ?, NULL WHERE ${noOpenRequestCase}`,
      )
      .bind(itemId, passportId, payloadJson, ts, ts, passportId, origin, i.kind),
    db
      .prepare(
        `INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind)
         SELECT ?, 'enqueue:dispute', ?, ?, ?, ? WHERE ${ourInsertLanded}`,
      )
      // I29-Q3 (2026-10-21 answer: "stop copying it and keep only the case id"): the detail is the case id, never the
      // request's text — the description, contact and evidence live in the case payload, which pruneRecoveryCasePii reduces.
      .bind(`req:${rkey}`, itemId, ts, itemId, auditActorKind(`req:${rkey}`), itemId), // I19-FXC: the requester's real actor kind, never 'operator'
  ]);

  // Did OUR insert win the only slot? Read the row back rather than a batch
  // rowcount (D1 batch results carry no portable `changes`).
  const mine = await db.prepare("SELECT 1 AS x FROM review_queue WHERE item_id = ?").bind(itemId).first();
  if (mine) return { item_id: itemId, deduped: false };

  // A concurrent first-request beat us to the slot — fold into it, exactly as a
  // sequential repeat would have.
  const raced = await foldIntoOpenRequestCase(db, passportId, rkey, i.kind, now, origin, evidence);
  if (raced) return { item_id: raced, deduped: true };

  // The winner closed between our guard and this read (vanishingly rare). Fall
  // back to an unguarded create so the request is never silently dropped. Not
  // review.ts `enqueue`: its audit line copies the payload, and I29-Q3 keeps the
  // case id only — so the same two rows as above, without the slot guard.
  const fallback = mintId("itm");
  const fts = now();
  await db.batch([
    db
      .prepare("INSERT INTO review_queue (item_id, item_type, state, subject, payload, created_at, updated_at, import_batch_id) VALUES (?, 'dispute', 'open', ?, ?, ?, ?, NULL)")
      .bind(fallback, passportId, payloadJson, fts, fts),
    auditStmt(db, `req:${rkey}`, "enqueue:dispute", fallback, fallback, fts),
  ]);
  return { item_id: fallback, deduped: false };
}

/**
 * Fold a repeat request into the one OPEN request-origin recovery case for a
 * passport, counting the distinct requester. Returns the case id, or null when
 * no such case is open (the caller then creates one). Operator-origin cases are
 * deliberately skipped so a fan's request never disappears into an operator's.
 */
async function foldIntoOpenRequestCase(
  db: DbLike,
  passportId: string,
  rkey: string,
  kind: string,
  now: () => string,
  origin: RecoveryOrigin = "request",
  /** X1: one line per proven contest — appended, never dropped. */
  evidence?: string,
  /** BF-16: how many CAS retries this call has already spent. */
  attempt = 0,
): Promise<string | null> {
  const open = (await db.prepare("SELECT item_id, payload FROM review_queue WHERE item_type = 'dispute' AND state = 'open' AND subject = ?").bind(passportId).all<{ item_id: string; payload: string }>()).results;
  for (const r of open) {
    const p = JSON.parse(r.payload) as RecoveryPayload;
    // MATCHING ORIGIN ONLY. Operator cases were always skipped so a fan's
    // request never disappeared into one; the same argument applies between the
    // two REQUEST origins, and harder — see `RecoveryOrigin`. A fold across them
    // hands the first filer's words the second filer's provenance.
    if (p.origin !== origin) continue;
    // BF-16 — AND THE INCIDENT KIND, which this loop never asked about.
    //
    // The merge below keeps the FIRST report's kind, description and contact;
    // only requesters/count (and contest evidence) are added. So a creator who
    // filed `lost_passkey` and LATER discovered a `hijack` was acknowledged
    // against the old case while the operator still read the old kind and the
    // old narrative — and `FORCED_ACTIONS.hijack` (force_logout,
    // reset_passkeys, retract_payout_tags, revoke_connect_binding,
    // revoke_agency_grants, evict_contact_channels) derives from that retained
    // kind, so the escalation silently lost its mandatory protections, the
    // payout-destination revocation among them. This file's own comment already
    // promised deduplication "by passport and kind"; the query implemented half
    // of it.
    //
    // Same argument as the origin line above, and harder: a fold across kinds
    // hands the first filer's INCIDENT CLASS to the second filer's facts.
    if (p.kind !== kind) continue;
    const requesters = Array.isArray(p.requesters) ? p.requesters : [];
    if (!requesters.includes(rkey) && requesters.length < MAX_REQUESTERS) requesters.push(rkey);
    p.requesters = requesters;
    p.request_count = (p.request_count ?? requesters.length) + 1;
    if (evidence) {
      const lines = Array.isArray(p.evidence) ? p.evidence : [];
      // QUEUE #662: a full list drops its oldest line past the kept head, then takes the new one (was: refused it).
      if (!lines.includes(evidence)) {
        if (lines.length >= MAX_REQUESTERS) lines.splice(EVIDENCE_HEAD_KEEP, lines.length - MAX_REQUESTERS + 1);
        lines.push(evidence);
      }
      p.evidence = lines;
    }
    // BF-16 (extended) — COMPARE-AND-SWAP, AND CONSULT THE ROWCOUNT.
    //
    // This was a read, a JS merge, and a blind replacement of the WHOLE payload
    // guarded only on `state = 'open'`. Two consequences, both of which lose an
    // acknowledged submission:
    //
    //  1. Two concurrent valid filings each read the same payload and each
    //     wrote their own complete copy over it, so the later write ERASED the
    //     earlier filer's requester and evidence — while both filers were told
    //     their report had been recorded.
    //  2. If an operator CLOSED the case between the read and this update, the
    //     state predicate correctly protected the closed payload — and the
    //     zero-row result was then ignored, a duplicate was audited, and the
    //     CLOSED case was returned to the filer as the id of their filing. The
    //     predicate kept the payload safe; nothing kept the submission.
    //
    // `payload = ?` (the exact bytes we read) makes this a CAS: it lands only
    // if nothing changed underneath, and a loser retries from a fresh read.
    // When the retries are spent, or the case is no longer open, we return null
    // — which sends the caller to the guarded INSERT, i.e. the submission
    // becomes its own case rather than an acknowledgement of nothing.
    // I19-W1A (QUEUE #944(e1)): the swap and its request_dup row in ONE batch; the row is gated `changes() = 1`
    // on the swap, and the swap's verdict is read from the batch result.
    const ts = now();
    const [swap] = (await db.batch([
      db
        .prepare("UPDATE review_queue SET payload = ?, updated_at = ? WHERE item_id = ? AND state = 'open' AND payload = ?")
        .bind(JSON.stringify(p), ts, r.item_id, r.payload),
      auditStmtAfterWrite(db, { actor: `req:${rkey}`, action: "recovery:request_dup", itemId: r.item_id, detail: `${kind} +1`, at: ts }),
    ])) as { meta?: { changes?: number } }[];
    if (!swap?.meta?.changes) {
      // Somebody else moved this case. Re-read and merge onto what they left,
      // bounded so a hot case cannot spin. `attempt` counts THIS call's retries.
      if (attempt + 1 >= FOLD_MAX_ATTEMPTS) return null;
      return foldIntoOpenRequestCase(db, passportId, rkey, kind, now, origin, evidence, attempt + 1);
    }
    return r.item_id;
  }
  return null;
}

/** Open recovery cases (dispute items) for the queue view. */
/** RJ550b-W314-R7-1: the dispute rows that are recovery cases. An unparseable payload is kept (it is not a fraud
 * review, and JSON.parse below reports it rather than this filter hiding it). Bind FRAUD_REVIEW_KIND once. */
const NOT_FRAUD_REVIEW = "(CASE WHEN json_valid(payload) = 1 THEN json_extract(payload, '$.kind') END) IS NOT ?";

/** RJ550b-W314-R7-1: a WARN49 fraud review is a `dispute` row that no recovery door may act on. */
function isFraudReview(item: { payload: unknown }): boolean {
  return typeof item.payload === "object" && item.payload !== null && (item.payload as { kind?: unknown }).kind === FRAUD_REVIEW_KIND;
}

/** RJ550b-W312-R7-8: the queue view serves at most this many open cases, OLDEST first. Past it the answer
 * says `truncated: true` — the cases beyond are the NEWER open ones, which surface as older ones are decided;
 * nothing is concluded about them from this page. */
export const RECOVERY_LIST_MAX = 200;

export async function listRecoveryCases(db: DbLike): Promise<{ cases: Array<{ item_id: string; subject: string; payload: RecoveryPayload; queued_at: string | null }>; truncated: boolean }> {
  const { results: rows } = await db
    // RJ550b-W314-R7-1 (I16-W04C): a WARN49 fraud review rides `item_type = 'dispute'` (warnings.ts publishWarning,
    // registry-build.ts deriveFraudReviews) but is NOT a recovery case — review.ts rejectItem draws the same partition.
    .prepare(
      `SELECT item_id, subject, payload, created_at FROM review_queue
        WHERE item_type = 'dispute' AND state = 'open' AND ${NOT_FRAUD_REVIEW}
        ORDER BY created_at ASC, item_id ASC
        LIMIT ?`,
    )
    .bind(FRAUD_REVIEW_KIND, RECOVERY_LIST_MAX + 1)
    .all<{ item_id: string; subject: string; payload: string; created_at: string | null }>();
  const truncated = rows.length > RECOVERY_LIST_MAX;
  const results = truncated ? rows.slice(0, RECOVERY_LIST_MAX) : rows;
  return { cases: results.map((r) => ({ item_id: r.item_id, subject: r.subject, payload: JSON.parse(r.payload) as RecoveryPayload, queued_at: r.created_at })), truncated };
}

/**

/**
 * QUEUE #942(a) (D-0946(1), D-0948(4)): an audit row whose DETAIL is an SQL expression, for a row that
 * joins its change's batch but describes values only that batch knows (a count, an epoch). Same columns
 * and actor_kind derivation as review.ts `auditStmt`; the detail is cut at 150 like the callers' slices.
 */
function auditSqlStmt(db: DbLike, actor: string, action: string, itemId: string, detailSql: string, detailBinds: unknown[], at: string): DbStatement {
  return db
    .prepare(`INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, ?, substr(${detailSql}, 1, 150), ?, ?, ?`)
    .bind(actor, action, ...detailBinds, at, itemId, auditActorKind(actor));
}

/** I19-W1A (QUEUE #944(c4), JA-06): the `recovery:holds_released` row, placed DIRECTLY after the hold-release UPDATE
 * in a batch, is written only when that UPDATE closed at least one hold (`changes() > 0`, not `= 1`: holds are unique
 * per (kind, item_id), not per case `ref`, so one case can release several). Used at both doors (resolve, reject). */
function holdsReleasedStmt(db: DbLike, actor: string, itemId: string, detail: string, at: string): DbStatement {
  return db
    .prepare("INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, ?, ?, ?, ?, ? WHERE changes() > 0")
    .bind(actor, "recovery:holds_released", detail, at, itemId, auditActorKind(actor));
}

/**
 * I3-1030 (D-0913(3)): how long a resolver/rejector holds a recovery case. Long
 * enough for every EXEC_ORDER action (a Connect revoke is a network call), short
 * enough that a resolver that died mid-run frees the case within minutes. The
 * lease row lives in the generic single-holder lease table (`cron_leases`,
 * cron-lease.ts): CAS acquire, holder-only release, strict expiry takeover.
 */
export const RECOVERY_LEASE_TTL_MS = 5 * 60 * 1000;

/** Take the case's lease or refuse 409. Fail closed: no lease, no action. */
async function takeCaseLease(db: DbLike, itemId: string, now: () => string): Promise<CronLease> {
  const t = Date.parse(now());
  const lease = await acquireCronLease(db, `recovery-case:${itemId}`, Number.isFinite(t) ? t : Date.now(), RECOVERY_LEASE_TTL_MS);
  if (!lease) throw new RecoveryError("another operator is deciding this case right now; try again shortly", 409);
  return lease;
}

/**
 * RG-AUTH-03 (I15-FX3): the lease is HELD, not assumed. Called before every
 * action of `resolveRecoveryCase` and before its close. One CAS on the HOLDER:
 * while the row still names this run, its expiry moves to `now()` + TTL, so a
 * second resolver's acquire (strict `expires_at < now`) is refused from here on;
 * once another resolver has taken the row (it lapsed and was re-acquired),
 * `changes` is 0 and this run stops 409 before its next action instead of running
 * beside the new holder. Keyed on the holder, not on expiry: a lapsed row nobody
 * re-took means nobody else ran, and refusing it would only strand a half-run case.
 * (cron-lease.ts `renewCronLease` also requires `expires_at >= now`.)
 */
async function holdCaseLease(db: DbLike, lease: CronLease, now: () => string): Promise<void> {
  const t = Date.parse(now());
  const expires = new Date((Number.isFinite(t) ? t : Date.now()) + RECOVERY_LEASE_TTL_MS).toISOString();
  const res = await db.prepare("UPDATE cron_leases SET expires_at = ? WHERE name = ? AND holder = ?").bind(expires, lease.name, lease.holder).run();
  if (res.changes !== 1) {
    throw new RecoveryError("another operator took this case over after this run's lease lapsed; this run stopped before its next action — reload the case and check its state", 409);
  }
}

/**
 * Resolve (approve + execute) a recovery case. Loads the dispute item (must be
 * open), then runs the requested actions in the fixed safe order and marks the
 * item approved with a disposition summary. Each step writes its own audit row.
 *
 * NOT atomic across steps by construction: the epoch bump, the passkey reset, and
 * the transfer are separate writes (the transfer is its own ledger batch). The
 * lockdown steps run FIRST and are monotone (re-running is a no-op), and the
 * transfer runs LAST and is made idempotent (it skips when the account is already
 * at the destination). So a mid-run failure leaves at worst "locked down but not
 * transferred" — the fail-safe direction — and simply re-resolving the case (it's
 * still 'open' until the final UPDATE) converges without double-effect or getting
 * stuck on a "does not belong" ledger rejection.
 *
 * For a SELF-SERVICE request (origin='request', no stored actions), the operator
 * MUST supply the remediation `plan` here — the requester never chose the actions;
 * only the adjudicating operator does, at resolve time. An operator-authored case
 * (actions already in the payload) ignores any plan and uses its stored actions.
 *
 * MUST be called inside the caller's dual-control seam (the route wraps it); this
 * function assumes authorization has been established.
 */
export async function resolveRecoveryCase(
  db: DbLike,
  actor: string,
  itemId: string,
  now: () => string = () => new Date().toISOString(),
  plan?: unknown,
  buildPayoutAlerts?: BuildPayoutAlerts,
  revokeConnectBinding?: ConnectRevoker,
): Promise<RecoveryResult> {
  // I3-1030 (D-0913(3)): a short expiring lease on the CASE before its first
  // action. Taken before the case is even read, so the `open` check below runs
  // under it: a second operator is refused while it is held, and one that read
  // `open` before the first finished cannot run the lockdown a second time.
  const lease = await takeCaseLease(db, itemId, now);
  try {
    const item = await getItem(db, itemId);
    if (!item) throw new RecoveryError("no such recovery case", 404);
    if (item.item_type !== "dispute" || isFraudReview(item)) throw new RecoveryError("not a recovery case", 400);
    if (item.state !== "open") throw new RecoveryError(`case not open (state ${item.state})`, 409);
    // QUEUE #146: a fan takeover case has a fan ACCOUNT subject and exactly one plan.
    if ((item.payload as { kind?: unknown }).kind === FAN_TAKEOVER_KIND) return resolveFanTakeover(db, actor, itemId, item.subject, plan, now);

    const stored = item.payload as unknown as RecoveryPayload;
    const passportId = item.subject;
    const affected = await db.prepare("SELECT 1 AS x FROM passports WHERE passport_id = ?").bind(passportId).first();
    if (!affected) throw new RecoveryError("affected passport no longer exists", 404);

    // A request has no actions of its own — the operator supplies the plan now.
    // An operator-authored case already has its actions and takes no plan.
    let payload: RecoveryPayload = stored;
    if (!Array.isArray(stored.actions) || stored.actions.length === 0) {
      const p = validatePlan(passportId, plan, stored.kind);
      payload = { ...stored, actions: p.actions };
      if (p.contested_account_key !== undefined) payload.contested_account_key = p.contested_account_key;
      if (p.to_passport !== undefined) payload.to_passport = p.to_passport;
      if (p.spare_contact_email !== undefined) payload.spare_contact_email = p.spare_contact_email;
    }

    // RE-FORCED AT EXECUTION, not only at intake. `validatePlan` folds the hijack
    // lockdown in when a case is OPENED, but a case opened before this rule
    // existed carries the old, shorter action list in its stored payload — and
    // that payload is what runs. Re-applying here means the invariant is a
    // property of what EXECUTES, not of when the row was written.
    const requested = new Set<RecoveryAction>([...payload.actions, ...forcedFor(payload.kind)]);

    // REFUSED BEFORE THE FIRST ACTION RUNS, with the other plan validation, so the
    // refusal is atomic: a missing revoker discovered mid-loop would have already
    // bumped the epoch and revoked the passkeys, leaving a half-locked passport
    // and a case still open. `revoke_connect_binding` is FORCED for a hijack
    // (FORCED_ACTIONS), so a caller that cannot supply the lever cannot resolve
    // one at all — 500, because that is a wiring fault, not an operator's input.
    if (requested.has("revoke_connect_binding") && !revokeConnectBinding) {
      throw new RecoveryError("hijack recovery cannot run without a Connect revoker", 500);
    }

    const out: RecoveryResult = { item_id: itemId, actions_executed: [] };

    for (const action of EXEC_ORDER) {
      if (!requested.has(action)) continue;
      await holdCaseLease(db, lease, now);
      if (action === "transfer_account") {
        // Idempotent on retry: if the account is ALREADY at the destination (a prior
        // resolve committed the transfer but failed before marking the case
        // approved), skip the append instead of letting account_transferred throw
        // "does not belong" — otherwise a mid-run failure would strand the case
        // 'open' forever and never complete the lockdown steps.
        const owner = await db.prepare("SELECT passport_id FROM passport_accounts WHERE account_key = ?").bind(String(payload.contested_account_key)).first<{ passport_id: string }>();
        if (owner && owner.passport_id === payload.to_passport) {
          await auditStmt(db, actor, "recovery:transfer", itemId, `${payload.contested_account_key} already at ${payload.to_passport} (idempotent skip)`.slice(0, 150), now()).run();
        } else {
          // S95 primitive: emit the ledger-preserving account_transferred. Its own
          // guards (destination live, `from` owns the account, no self-transfer,
          // status preserved) run inside appendEvents and surface as LedgerError.
          // QUEUE #942(a): the audit row rides the transfer's own batch.
          const [ev] = await appendEvents(
            db,
            [{ event_type: "account_transferred", subject: String(payload.to_passport), actor_kind: "operator", actor_id: actor, payload: { account_key: String(payload.contested_account_key), from_passport: passportId } }],
            now,
            [],
            [auditStmt(db, actor, "recovery:transfer", itemId, `${payload.contested_account_key} ${passportId}->${payload.to_passport}`.slice(0, 150), now())],
          );
          out.transfer_event_id = ev!.event_id;
          out.transfer_from_passport = passportId;
          out.transfer_account_key = String(payload.contested_account_key);
        }
      } else if (action === "reactivate_account") {
        // F5. The key sits on the destination after a transfer, else on this
        // passport. Idempotent on retry: an already-active key is skipped rather
        // than refused (the ledger refuses reactivating an active key on purpose;
        // here that refusal would strand a case a prior run already completed).
        const key = String(payload.contested_account_key);
        const target = payload.to_passport && payload.actions.includes("transfer_account") ? String(payload.to_passport) : passportId;
        const row = await db.prepare("SELECT status, passport_id FROM passport_accounts WHERE account_key = ?").bind(key).first<{ status: string; passport_id: string }>();
        // BOUND TO THE CASE (F5 review, all four personas): the key must sit on
        // the passport this case is about (the destination after a transfer, else
        // the subject). A key that lives elsewhere is a refusal that keeps the
        // case open — never a re-listing on a passport nobody signed for, and
        // never an audit line naming the wrong one.
        if (!row) throw new RecoveryError(`reactivate_account: no such account ${key}`, 409);
        if (row.passport_id !== target) throw new RecoveryError(`reactivate_account: ${key} is on ${row.passport_id}, not this case's ${target}`, 409);
        if (row.status === "active") {
          await auditStmt(db, actor, "recovery:reactivate", itemId, `${key} already active on ${target} (idempotent skip)`.slice(0, 150), now()).run();
        } else {
          const latest = await latestRevokingEvent(db, key);
          if (!latest) throw new RecoveryError(`reactivate_account: no revoking event on record for ${key}`, 409);
          try {
            // QUEUE #942(a): the audit row rides the reactivation's own batch.
            const r = await reactivateAccount(db, actor, { account_key: key, basis: "control_reproven", reverses: latest.event_id, reason: `recovery:${itemId}`, expect_passport: target }, now, [
              auditStmt(db, actor, "recovery:reactivate", itemId, `${key} on ${target} (control_reproven, reverses ${latest.event_id})`.slice(0, 150), now()),
            ]);
            out.reactivation_event_id = r.event_id;
          } catch (e) {
            if (e instanceof ReactivationError) throw new RecoveryError(e.message, 409);
            throw e;
          }
        }
      } else if (action === "force_logout") {
        // #918 (D-0920 option B, superseding D-0888(2)'s retry skip): EVERY RUN BUMPS,
        // a retry included. A session minted between two runs may be the attacker's (a
        // creator token carries no issue time to tell it from the owner's), so a retry
        // ends every session again; the owner signs in once more.
        // QUEUE #942(a): the audit row rides the bump's batch, after the upsert, reading THIS bump's epoch.
        const forceLogoutAudit = [auditSqlStmt(db, actor, "recovery:force_logout", itemId, "'epoch ' || (SELECT session_epoch FROM creator_sessions WHERE passport_id = ?)", [passportId], now())];
        out.session_epoch = await bumpCreatorEpoch(db, passportId, now, true, forceLogoutAudit);  // INVOLUNTARY (0082): this is the door the claim-ref inertness rule is about.
      } else if (action === "reset_passkeys") {
        // D-0931(5) (I16-RG-AUTH:3, superseding D-0884(4)'s retry skip for this set): A RETRY ALSO REVOKES EVERY
        // PASSKEY ENROLLED AFTER THE CASE OPENED. #918 bumps the epoch on every run because a session minted between
        // runs may be the attacker's; a passkey that session enrolled would sign straight back in, so it goes too.
        // A retry is a case already holding its own `recovery:reset_passkeys` receipt; the step is skipped only when
        // no passkey is live at all (nothing to revoke, so no second epoch bump). Pre-case keys keep D-0884(4)'s rule:
        // revoked on a retry. The FIRST run is unchanged: it revokes every active key.
        const prior = await db.prepare("SELECT 1 AS x FROM audit_log WHERE item_id = ? AND action = 'recovery:reset_passkeys' LIMIT 1").bind(itemId).first();
        if (prior) {
          const live = await db
            .prepare("SELECT COUNT(*) AS n FROM passkeys WHERE passport_id = ? AND status = 'active'")
            .bind(passportId)
            .first<{ n: number }>();
          if ((live?.n ?? 0) === 0) {
            out.passkeys_revoked = 0;
            await auditStmt(db, actor, "recovery:reset_passkeys", itemId, "retry: no passkey live (idempotent skip)", now()).run();
            out.actions_executed.push(action);
            continue;
          }
        }
        // QUEUE #942(a): the receipt rides the reset's batch, FIRST, its counts computed over the keys about to be
        // revoked and the epoch about to be bumped — so a failed receipt leaves no committed reset to re-run.
        const r = await operatorResetPasskeys(db, passportId, undefined, [
          auditSqlStmt(
            db,
            actor,
            "recovery:reset_passkeys",
            itemId,
            "'revoked ' || (SELECT COUNT(*) FROM passkeys WHERE passport_id = ? AND status = 'active') || ', epoch ' || (COALESCE((SELECT session_epoch FROM creator_sessions WHERE passport_id = ?), 0) + 1)",
            [passportId, passportId],
            now(),
          ),
        ]);
        out.passkeys_revoked = r.revoked;
        // W3-03 FIX2 item 4: `operatorResetPasskeys` bumps the epoch ITSELF
        // (passkeys.ts:819, and it bumps FIRST so the only survivable partial
        // state fails closed), so the epoch this case ends on is ITS return
        // value, not whatever `force_logout` left in `out` one branch earlier.
        // Dropping it made the resolve RESULT and the audit detail below report
        // 1 while the database held 2 — the operator is told the sessions were
        // cut at an epoch that is already stale, and the record says the same.
        out.session_epoch = r.session_epoch;
      } else if (action === "retract_payout_tags") {
        // WHERE THE MONEY GOES, ZEROED — live AND staged, no cooling-off.
        // Retiring a destination is the safe direction (removal is fast, addition
        // is slow), and the STAGED half is the one a hijack needs: a tag the
        // attacker staged before detection has its own clock and would otherwise
        // go live a day AFTER the case was resolved, via the cron sweep or the
        // public read. The follower alert rides the same atomic batch as the
        // retirement (retractAllPayoutTags), so a fan who saved this creator is
        // told the destinations moved.
        // Destinations only: not the removal placeholder (payout.ts REMOVAL_MARKER).
        // I18-FXA D32 (RG-D32, D-0949(1)): the count is taken INSIDE the retirement's batch, by the two
        // statements that lead it (before the UPDATE), so the result and the audit row name what THIS batch
        // retired — a tag staged between a pre-read and the batch was retired uncounted.
        const live = "FROM payout_tags WHERE passport_id = ? AND status IN ('live','staged') AND tag_value <> ''";
        const [counted] = (await retractAllPayoutTags(db, passportId, `recovery:${itemId}`, buildPayoutAlerts, [], [
          db.prepare(`SELECT COUNT(*) AS n ${live}`).bind(passportId),
          auditSqlStmt(db, actor, "recovery:retract_payout_tags", itemId, `'retired ' || (SELECT COUNT(*) ${live})`, [passportId], now()),
        ])) as { results?: { n: number }[] }[];
        out.payout_tags_retired = Number(counted?.results?.[0]?.n ?? 0);
      } else if (action === "revoke_connect_binding") {
        // THE DESTINATION `retract_payout_tags` DOES NOT REACH. Payout tags are
        // the PUBLIC destinations; `creator_connect` is the one the weekly run
        // actually transfers to (creator-payout.ts:671), and it survived a full
        // lockdown. `unbindPayoutAccount` is passkey-gated at its only other
        // caller (index.ts:2325) — the key `reset_passkeys` has just revoked two
        // steps up — so during a hijack the rightful creator is precisely the
        // party who cannot reach it. Hence the operator, on the same resolution.
        //
        // The row goes to `revoked`, not `none`: earnings are HELD, not lost
        // (creator-payout.ts:332-336), and no Stripe call is made.
        //
        // IDEMPOTENT ON RETRY: the injected revoker returns `null` for the one
        // benign refusal ("no payout account to disconnect", creator-payout.ts:348)
        // — nothing bound is the state this action is trying to reach. Every other
        // throw propagates through the revoker untouched: the malformed-id 500
        // (:350) and the "changed while disconnecting" guard are real failures and
        // must stop the resolution rather than be reported as a clean lockdown.
        // The lever itself is INJECTED (ConnectRevoker) — see that type.
        // QUEUE #942(a): the unbound row rides the unbind's batch through `audit`. `joined` records that the
        // revoker built it; a revoker that returns a binding WITHOUT calling `audit` (one not yet forwarding
        // the fifth argument) gets the after-commit row below, so no unbind goes unrecorded.
        let joined = false;
        const r = await revokeConnectBinding!(db, passportId, { actorKind: "operator", actorId: actor, reason: "hijack recovery" }, now, (ctx) => {
          joined = true;
          return [auditStmt(db, actor, "recovery:revoke_connect_binding", itemId, `unbound ${ctx.stripe_account_id}`, now())];
        });
        // TWO NULLS, TWO FACTS (S10-3(a)). The revoker collapses the benign 409
        // onto `null`, so "this passport never had a binding" and "a row is there
        // holding a real account id, in a state `status IN ('active','pending')`
        // does not see" arrive here as the same value and used to leave the same
        // sentence in the log — the second is a row an operator must go and look
        // at, the first is nothing. The arms are separated HERE, by a read of the
        // row's own presence, because the error class never crosses the injection
        // boundary (the ConnectRevoker docblock at :578 is the rule, not an
        // accident). The read runs only on the null arm, and the null arm changed
        // nothing, so it cannot race its own action.
        //
        // There is no third arm: an uninjected revoker is refused at :654 with a
        // 500 before the first action runs, so `revoker: not injected` is
        // unreachable at this line and is not written.
        //
        // THE RECORD NAMES THE STATUS (RUN-0015 E-1). The read already selected
        // `status` and the expression that consumed it looked only at the row's
        // PRESENCE, so all three non-unbound worlds — a `connecting` ceremony, a
        // row already `revoked`, and anything a later migration adds to the CHECK
        // at 0036_creator_payouts.sql:21 — left one sentence that names none of
        // them. An operator reading "no payout account (409)" cannot tell which
        // row to go and look at; with the status on the line they can.
        //
        // The read runs AFTER the revoker, so on the `connecting` world it sees
        // the state the revoker's own burn left (`revoked`), and that burn's
        // record is its sibling `connect:flow_ended` row at the same instant
        // (creator-payout.ts, `unbindPayoutAccount`). This line describes the
        // state the resolution LEAVES; it is not a second claim about what ran.
        let detail: string;
        if (r) detail = `unbound ${r.stripe_account_id}`;
        else {
          const existing = await db.prepare("SELECT status FROM creator_connect WHERE passport_id = ?").bind(passportId).first<{ status: string }>();
          detail = existing ? `revoker refused: no payout account (409, creator_connect.status=${existing.status})` : "no active binding";
        }
        if (!(r && joined)) await auditStmt(db, actor, "recovery:revoke_connect_binding", itemId, detail, now()).run();
      } else if (action === "revoke_agency_grants") {
        // THE GRANT THE RECOVERED CREATOR COULD NOT REVOKE. `agency_revoked` was
        // creator_passkey-only, and `reset_passkeys` (two steps up) has just
        // revoked every one of the creator's passkeys — so without this the
        // attacker's agency kept acting on the passport and the rightful owner
        // was the single party in the system unable to stop it.
        //
        // REUSES THE EXISTING EVENT TYPE. One `agency_revoked` per live grant,
        // actor_kind 'operator' (schema ACTOR_POLICY was widened for exactly this
        // path), byte-identical in payload to a creator's own revoke — so the
        // derive-from-log fold ("later event wins") needs no new rule and
        // fixtures/chain.golden.json is untouched. Deterministic order (grants
        // come back sorted by agency_id) and idempotent on retry: a grant already
        // revoked is simply absent from the live set the next run reads.
        const grants = await liveGrantsForPassport(db, passportId);
        // I3-1205: `appendEvents` refuses a draft batch over LEDGER_DRAFT_BATCH_MAX
        // (events.ts:1945), so a cohort that large is revoked in chunks. A chunk
        // that fails leaves the earlier ones committed; the retry re-reads the
        // live set, which no longer holds them.
        //
        // QUEUE #942(a) (D-0948(3)): the audit row joins the LAST chunk's batch. A failure between chunks
        // still leaves the earlier chunks committed with no row (that remainder stays on #942). No grant
        // means no change, and the row is then written alone.
        const grantsAudit = () => auditStmt(db, actor, "recovery:revoke_agency_grants", itemId, `revoked ${grants.length}: ${grants.map((g) => g.agency_id).join(",")}`.slice(0, 150), now());
        for (let i = 0; i < grants.length; i += LEDGER_DRAFT_BATCH_MAX) {
          await appendEvents(
            db,
            grants.slice(i, i + LEDGER_DRAFT_BATCH_MAX).map((g) => ({ event_type: "agency_revoked" as const, subject: passportId, actor_kind: "operator" as const, actor_id: actor, payload: { agency_id: g.agency_id } })),
            now,
            ["agency_revoked"],
            i + LEDGER_DRAFT_BATCH_MAX >= grants.length ? [grantsAudit()] : [],
          );
        }
        out.agency_grants_revoked = grants.map((g) => g.agency_id);
        if (grants.length === 0) await grantsAudit().run();
        // S-CW4 (review F2): the WAR CONSENT the attacker may have signed joins
        // the same sweep, for the same reason the grant revocation exists — after
        // reset_passkeys the rightful owner cannot append war_opt_out themselves,
        // and a rogue agency's opt-in would otherwise keep cards of the victim
        // minting after the case closed. Same idiom as the grant revocation:
        // actor_kind 'operator' (ACTOR_POLICY names it for war_opt_out ONLY —
        // entry stays creator/agency), idempotent on retry (already opted-out →
        // no row). The creator re-enters by signing war_opt_in with a fresh
        // passkey — deliberately NOT automatic, like reauthorizing an agency.
        if ((await warOptState(db, passportId)).opted_in) {
          await appendEvents(
            db,
            [{ event_type: "war_opt_out" as const, subject: passportId, actor_kind: "operator" as const, actor_id: actor, payload: {} }],
            now,
            ["war_opt_out"],
            [auditStmt(db, actor, "recovery:war_opt_out", itemId, "war consent withdrawn with the grants", now())],  // QUEUE #942(a): same batch
          );
          out.war_opted_out = true;
        }
      } else if (action === "evict_contact_channels") {
        // I3-0114 — THE DOOR THE WHOLE LOCKDOWN LEFT STANDING.
        //
        // `creator_contacts.email` with `email_verified = 1` stopped being a
        // notification channel when 0043 landed: `creator-recovery.ts:577` resolves
        // exactly that predicate to file a `creator_inbox` case, which that module
        // calls "the strongest provenance this path can carry". It is reached with
        // NO session and NO passkey, so every lever above this one — the epoch bump,
        // the passkey reset, logout-everywhere — leaves it alone. MEASURED at HEAD
        // 88e2593: after a complete hijack resolution the attacker's address filed a
        // fresh case against the passport, ranked above the real owner's.
        //
        // `digest.removeCreatorEmail` is the creator-facing twin of this step and
        // CANNOT be reused here: its removal arm rides on
        // `(SELECT COUNT(*) FROM passkeys WHERE passport_id = ? AND status='active') >= 1`,
        // and `reset_passkeys` two steps up has just revoked every one — so the
        // reused call would 409 on every hijack. The operator lockdown is precisely
        // the case where "no way back in" is the accepted outcome; it is therefore
        // taken here and MADE VISIBLE at the end of this branch, not refused.
        //
        // THE SPARE, AND THE INSTANT IT IS MEASURED AGAINST (Jason: "spare one that
        // predates the hijack"). NOTHING IN THE SCHEMA CAN DECIDE THIS TODAY, and
        // saying so is the point rather than shipping a comparison that looks like
        // one. `creator_contacts` carries `created_at` (when the ROW was first
        // written, not when this address was proven) and `updated_at` (which moves
        // for a digest toggle, a pending clear and the attacker's own verification
        // alike). The remediation record carries `review_queue.created_at` — when
        // the case was FILED, which is necessarily AFTER the attacker's write — so
        // `updated_at < created_at` would spare the attacker EVERY time. There is no
        // in-schema pair that orders the address against the incident, and a guard
        // that cannot fire is worse than an absent one.
        //
        // So the anchor is named explicitly and it is the OPERATOR'S ADJUDICATION:
        // `spare_contact_email`, folded into the dual-control key so both co-signers
        // approve the same spare. Absent — the default, and what every existing
        // stored hijack payload carries — everything goes, which is the ruled
        // direction: the attacker keeping a route back is worse than the owner
        // having to use support. The incident timeline that would let the SYSTEM
        // answer this is filed as DESIGN NEEDED alongside this change.
        const evictTs = now();
        const contactRow = await db
          .prepare("SELECT email, email_verified FROM creator_contacts WHERE passport_id = ?")
          .bind(passportId)
          .first<{ email: string | null; email_verified: number }>();
        const heldAddress = contactRow?.email_verified === 1 && contactRow.email ? contactRow.email.trim().toLowerCase() : null;
        const spare = normalizeSpareAddress(payload.spare_contact_email);
        // A spare only ever spares the address the passport ACTUALLY holds. Naming
        // some other address is not a partial instruction to be interpreted — it is
        // an operator describing a row that is not there, and the safe reading of a
        // mismatch is that nothing was adjudicated.
        const spared = spare !== null && heldAddress !== null && spare === heldAddress;
        if (spared) out.contact_spared = heldAddress!;

        // QUEUE #942(a) (D-0948(4)): the three evictions below and their audit row are ONE batch, the audit
        // row FIRST with its counts computed in SQL over the rows about to be evicted (each count uses the
        // WHERE of the statement it describes), so a failed audit insert evicts nothing.
        //
        // (1) THE ADDRESS. Guarded on there being contact material to take, so the
        // count reports what was actually removed rather than "a row exists".
        const contactWhere = spared
          ? "passport_id = ? AND (pending_email IS NOT NULL OR verify_code_hash IS NOT NULL OR verify_target IS NOT NULL)"
          : "passport_id = ? AND (email IS NOT NULL OR pending_email IS NOT NULL OR verify_code_hash IS NOT NULL OR verify_target IS NOT NULL)";
        const contactSql = spared
          ? // A spared address keeps its verified value — but the UNPROVEN half of
            // the row is still whatever the attacker staged, and a pending change
            // left behind completes after the case closes, exactly as a staged
            // payout tag would have. It goes either way.
            `UPDATE creator_contacts SET pending_email = NULL, pending_created_at = NULL,
               verify_code_hash = NULL, verify_expires_at = NULL, verify_target = NULL, updated_at = ?
             WHERE ${contactWhere}`
          : `UPDATE creator_contacts SET email = NULL, email_verified = 0, pending_email = NULL, pending_created_at = NULL,
               verify_code_hash = NULL, verify_expires_at = NULL, verify_target = NULL, updated_at = ?
             WHERE ${contactWhere}`;
        const contactStmt = db.prepare(contactSql).bind(evictTs, passportId);

        // (2) THE DEVICES. NEVER SPARED, and the asymmetry is deliberate rather
        // than an oversight: a push registration is a device token the app
        // re-acquires on the owner's next launch at no cost, while an address is
        // the owner's only self-service door — so the "leave one in place" argument
        // that forces the spare question for the address has no force here, and the
        // attacker's handset is otherwise still receiving this creator's digests and
        // alerts after the case closed. There is also nothing on these rows that
        // could order them against the incident: `created_at` is when the device
        // registered, which an attacker's device satisfies as readily as the
        // owner's.
        const pushStmt = db.prepare("DELETE FROM creator_push_tokens WHERE passport_id = ?").bind(passportId);
        // I3-0203 b (I25-W1-IDENTITY H5): a pending push-possession challenge carries a push token too; it leaves in the
        // same batch (appended last, so the result indexes read below are unchanged).
        const pushChallengeStmt = db.prepare("DELETE FROM creator_push_challenges WHERE passport_id = ?").bind(passportId);
        // I30-OWNER (0265): a held push-device removal names one of the devices evicted here; it leaves too (appended last).
        const pushPendingStmt = db.prepare("DELETE FROM creator_push_pending_removals WHERE passport_id = ?").bind(passportId);

        // (3) THE CODES ALREADY IN FLIGHT, keyed on the ADDRESS because that is the
        // only thing `creator_otp` is keyed on — it carries no passport column.
        // Scoped to the address this step just EVICTED: a code for a SPARED address
        // sits in the owner's mailbox and only its holder can redeem it, so taking
        // it would be the eviction locking out the party it just decided to keep.
        // Unconsumed only; a spent row is a record, not a credential.
        const evictCodes = heldAddress !== null && !spared;
        const otpStmt = evictCodes ? db.prepare("DELETE FROM creator_otp WHERE address = ? AND consumed_at IS NULL").bind(heldAddress) : null;

        // The detail is the one the three `.run()` results used to build: `contacts N, push N, codes N[, spared X]`.
        const evictAudit = auditSqlStmt(
          db,
          actor,
          "recovery:evict_contact_channels",
          itemId,
          `'contacts ' || ${spared ? "0" : `(SELECT COUNT(*) FROM creator_contacts WHERE ${contactWhere})`}` +
            " || ', push ' || (SELECT COUNT(*) FROM creator_push_tokens WHERE passport_id = ?)" +
            ` || ', codes ' || ${evictCodes ? "(SELECT COUNT(*) FROM creator_otp WHERE address = ? AND consumed_at IS NULL)" : "0"}` +
            (spared ? " || ', spared ' || ?" : ""),
          [...(spared ? [] : [passportId]), passportId, ...(evictCodes ? [heldAddress] : []), ...(spared ? [heldAddress] : [])],
          now(),
        );
        const evictRes = (await db.batch([evictAudit, contactStmt, pushStmt, ...(otpStmt ? [otpStmt] : []), pushChallengeStmt, pushPendingStmt])) as { meta?: { changes?: number } }[];
        out.contacts_evicted = spared ? 0 : Number(evictRes[1]?.meta?.changes ?? 0);
        out.push_tokens_evicted = Number(evictRes[2]?.meta?.changes ?? 0);
        if (otpStmt) out.recovery_codes_evicted = Number(evictRes[3]?.meta?.changes ?? 0);

        // NEVER SILENT. A remediation that takes the last door has to say so where
        // somebody will act on it — the resolve RESULT the operator is looking at,
        // and a row of its own in the trail. The rightful owner cannot be told
        // directly: the only address we had is the one we just decided we cannot
        // trust, and mailing it would be the remediation notifying the attacker.
        // So the recourse is the operator's, and this is what makes it theirs.
        const liveKeys = await db
          .prepare("SELECT COUNT(*) AS n FROM passkeys WHERE passport_id = ? AND status = 'active'")
          .bind(passportId)
          .first<{ n: number }>();
        if (!spared && Number(liveKeys?.n ?? 0) === 0) {
          out.self_service_recovery_lost = true;
          await auditStmt(
            db,
            actor,
            "recovery:no_self_service_door",
            itemId,
            `${passportId} now has no verified contact address and no active passkey — the rightful owner can only return through an operator`.slice(0, 150),
            now(),
          ).run();
        }
      }
      out.actions_executed.push(action);
    }

    await holdCaseLease(db, lease, now);
    const ts = now();
    // THE CLOSE IS ONE BATCH, FRONTED BY THE SHARED DECISION LOCK (migration
    // 0031). Two things were wrong with the three separate `.run()`s this replaces.
    //
    // First, `res.changes !== 1` is a check on the flip ALONE: the audit line that
    // follows it was a separate write, so a lost race threw AFTER the flip and
    // before the audit — which is fine — while the reverse ordering of two
    // resolvers left the loser's `recovery:resolve` line against a case somebody
    // else had closed. Second, and the reason this is in the shared lock rather
    // than a tighter `WHERE`: `open` is a state FIVE doors leave, and `changes`
    // could only ever see the ones that came through this function. A console
    // `approveItem` or `rejectItem` on the same dispute item is exactly the
    // cross-module race migration 0030 named and 0031 closes, and the lock is what
    // makes the loser roll back WHOLE — flip, audit line and all.
    //
    // The lock deliberately sits HERE, at the close, and not before the actions.
    // Taking it up front would wedge the case forever on any mid-run failure (the
    // actions are re-runnable by design and the item stays `open` until this
    // batch), and there is no unreject route. What the lock therefore does NOT
    // cover is a second resolver's ACTIONS: those are idempotent/monotone by
    // construction (see the header) and re-running them is a no-op, so the cost of
    // losing here is wasted work, never a divergent case.
    try {
      await db.batch([
        reviewDecisionStmt(db, itemId, "open", "approved", actor, ts, out.transfer_event_id ?? null),
        // `disposition` is a CLOSED SET as of migration 0019 ('approved' | 'rejected'
        // | 'published' | 'retracted' | 'withdrawn'), so the summary sentence this
        // used to cram into it — the one writer of free text among seven — moves to
        // `disposition_note`. Same information on the same row, same batch, same
        // console read; it just stops being the value every counter has to parse
        // around. 0019 backfills existing `recovery: …` rows the same way.
        db
          .prepare("UPDATE review_queue SET state = 'approved', disposition = 'approved', disposition_note = ?, result_event = ?, updated_at = ? WHERE item_id = ? AND state = 'open'")
          .bind(`recovery: ${out.actions_executed.join("+")}`.slice(0, 240), out.transfer_event_id ?? null, ts, itemId),
        auditStmt(db, actor, "recovery:resolve", itemId, `${payload.kind} → ${out.actions_executed.join("+")}`.slice(0, 150), ts),
        // I3-0290 — THE CASE'S HOLDS DIE WITH THE CASE, IN THE CASE'S OWN BATCH.
        //
        // `openHold` (gifting.ts:623) takes a `ref` matching `RECOVERY_REF_RE`
        // and an operator working a recovery names the case id in it; nothing
        // then closed those holds when the case closed. `holdBlockingErasure`
        // (gifting.ts:719) reads `state = 'open' AND expires_at > now`, so a
        // hold outliving its case is a 409 on the fan's OWN erasure — a closed
        // case refusing a data-subject right, for up to the 90-day
        // `HOLD_MAX_TTL_MS` the cumulative bound admits (not forever: the sweep
        // and `expires_at` bound it, which is the only reason this is a hole and
        // not a permanent lock).
        //
        // IN THIS BATCH, not after it: the close is fronted by the shared
        // decision lock, so a resolver that LOST rolls back whole — flip, audit
        // line and hold release together. Releasing after the batch would free
        // the holds of a case this call did not close.
        //
        // `state = 'open'` in the WHERE, not a read above it: an `expired` hold
        // the sweep already settled keeps its own closure reason, and a
        // re-resolve releases nothing twice.
        db
          .prepare("UPDATE collectible_holds SET state = 'closed', closed_at = ?, closed_by = ?, closed_reason = 'recovery case resolved' WHERE ref = ? AND state = 'open'")
          .bind(ts, actor, itemId),
        holdsReleasedStmt(db, actor, itemId, "open collectible holds naming this case were closed with it", ts),
      ]);
    } catch (e) {
      // The same 409 this door has always given, now true for a loser that came
      // through ANY of the five doors out of `open`. Narrow: anything else is a
      // real fault in the batch we just built.
      if (isDecisionLost(e)) throw new RecoveryError("case was resolved concurrently", 409);
      throw e;
    }

    // I3-0955 — THE SECOND DOOR OUT OF `open` CLEARS ITS SIGN-OFFS TOO.
    //
    // `withDualControl` claims (DELETE ... RETURNING) only the ONE
    // `(action_type, params_hash)` it met the threshold on (dual_control.ts:405-412).
    // A recovery resolve is signed under SEVERAL keys for one case — the bare id
    // and one `<itemId>|<recoveryPlanKey>` per plan an operator typed
    // (console.ts:2318-2345) — so every key the executing call did NOT run on
    // survives the close, advertising a live approval for a case that can never
    // be resolved again (resolve requires `state = 'open'`). `rejectRecoveryCase`
    // has cleared them by prefix since AA-06; this is the SAME cleanup at the
    // other closing door, applied per site and proved per site (FAMILY-004
    // clause 4) rather than at one of the two.
    //
    // AFTER the close batch, never before: a resolver that LOST the shared
    // decision lock threw above, so a loser cannot delete the winner's — or a
    // still-live — sign-off for a case it did not close.
    await clearResolveSignoffs(db, itemId);

    // S9-2 (b) — TELL THE CREATOR WHO FILED THE COMPETING CLAIM.
    //
    // AFTER THE CLOSE BATCH, and that placement is the whole guard. This function
    // is only ever called inside `withDualControl`'s `execute()` callback
    // (console.ts:1824), so a `pending` first sign-off never reaches this line —
    // and a resolver that LOST the shared decision lock threw above, so the loser
    // does not tell a challenger they won a case somebody else decided. Both
    // ledger writes and the close have landed by here, so the notice describes a
    // decision that is true in the fold, not one that was merely attempted.
    // EXEC_ORDER is untouched.
    //
    // WHO: every `creator:<passport_id>` entry in `payload.requesters` (the shape
    // `submitRecoveryRequest` writes, bounded by MAX_REQUESTERS), minus the case
    // SUBJECT — the incumbent is the other party and already had their notice at
    // filing time (`noticeContestedAccount`); the challenger's notice must not
    // double as one to the person they contested. A `fan:` requester has no
    // passport and no creator surface: nothing is written and the drop is
    // recorded, never silently swallowed (FAMILY-003).
    //
    // WHAT: the case id and that a decision landed — never the other passport,
    // never the account key, never the action list (INV-025).
    if (payload.kind === "competing_claim") {
      const requesters = Array.isArray(payload.requesters) ? payload.requesters : [];
      const notifiable = requesters.filter((r) => r.startsWith("creator:") && r.slice("creator:".length) !== passportId);
      const skipped = requesters.length - notifiable.length;
      let told = 0;
      for (const r of notifiable) {
        try {
          told += await noticeContestDecided(db, r.slice("creator:".length), itemId, now);
        } catch (e) {
          await auditStmt(db, actor, "recovery:notice_failed", itemId, `contest_decided not written for ${r} — ${e instanceof Error ? e.name : "error"}`.slice(0, 150), now()).run();
        }
      }
      // THE ROW A READER CAN FIND AFTERWARDS — including when it is a row of
      // zero. "Nobody was told" is the answer a creator's complaint needs, and a
      // requester with no passport (a fan-origin filing) is the case where that
      // is correct rather than broken.
      await auditStmt(db, actor, "recovery:notify_requesters", itemId, `told ${told} of ${notifiable.length} creator requester(s); ${skipped} with no creator passport`.slice(0, 150), now()).run();
    }

    return out;
  } finally {
    await releaseCronLease(db, lease);
  }
}

/**
 * How long after a resolved recovery the creator keeps being told what it took.
 *
 * THE PANEL WAS GATED ON THE WRONG THING. `recoveryAftermathPanel` — the screen
 * that explains that recovery ALSO retired every payout destination and revoked
 * every agency grant — rendered only when `needs_reenroll` was true, i.e. zero
 * active passkeys and at least one revoked. Two holes fell straight out of that:
 *
 *   1. `reset_passkeys` was NOT in `FORCED_ACTIONS` at the time (only the payout
 *      retraction and the grant revocation were). An operator handling a
 *      hijack that came in through a stolen SESSION could reasonably tick
 *      `force_logout` alone, and then the creator's money screen and agency
 *      went quiet with no explanation anywhere — the exact scenario the panel
 *      exists to prevent. PS-023 closed this by forcing both lockdown levers
 *      for `hijack` too (see FORCED_ACTIONS); this panel-gating fix is kept
 *      because `needs_reenroll` is still the wrong signal for every OTHER
 *      recovery kind, where `reset_passkeys` stays an operator option.
 *   2. It deleted itself on first use. The panel says "register a device below
 *      and then reauthorize them"; registering the device makes `active > 0`,
 *      so the very next paint removed both the banner and the explanation —
 *      at the precise moment the creator started needing them.
 *
 * So the flag is keyed to THE RECOVERY, not to the passkey count, and it
 * outlives the first thing it asks for. Thirty days is the dismissal: long
 * enough to cover re-adding payout destinations through their cooling-off and
 * getting an agency re-granted, short enough that it is not permanent chrome.
 */
export const RECOVERY_AFTERMATH_WINDOW_MS = 30 * 24 * 60 * 60 * 1000;

/**
 * Was a recovery resolved for this passport recently enough that the creator
 * still needs the explanation?
 *
 * Derived from the audit row `resolveRecoveryCase` already writes, joined to the
 * case's subject — no new table, no acknowledgement column, and nothing that a
 * client can set. One statement.
 */
export interface RecoveryCleared {
  payout_destinations: boolean;
  agency_grants: boolean;
}
export async function recoveryAftermath(db: DbLike, passportId: string, nowMs: number): Promise<{ recent: boolean; resolved_at: string | null; cleared: RecoveryCleared }> {
  const none = { payout_destinations: false, agency_grants: false };
  if (!/^psp_[0-9a-f]{16}$/.test(passportId)) return { recent: false, resolved_at: null, cleared: none };
  // I3-2113: WHICH clean-up ran, over EVERY case resolved in the window (an
  // earlier hijack still explains missing destinations after a later limited
  // case). Read from `disposition_note` (`recovery: a+b`, written in the
  // resolve batch, slice 240 — the whole plan fits) rather than the audit
  // detail, whose slice 150 can cut the list. `instr` on `+name+` is an exact
  // token match: no LIKE wildcard.
  const plan = "('+' || replace(COALESCE(q.disposition_note, ''), 'recovery: ', '') || '+')";
  const row = await db
    .prepare(
      `SELECT MAX(a.created_at) AS created_at,
              MAX(instr(${plan}, '+retract_payout_tags+') > 0) AS payout,
              MAX(instr(${plan}, '+revoke_agency_grants+') > 0) AS agency
         FROM audit_log a
         JOIN review_queue q ON q.item_id = a.item_id
        WHERE a.action = 'recovery:resolve' AND q.subject = ? AND a.created_at >= ?`,
    )
    .bind(passportId, new Date(nowMs - RECOVERY_AFTERMATH_WINDOW_MS).toISOString())
    .first<{ created_at: string | null; payout: number | null; agency: number | null }>();
  if (!row?.created_at) return { recent: false, resolved_at: null, cleared: none };
  return { recent: true, resolved_at: row.created_at, cleared: { payout_destinations: Number(row.payout) === 1, agency_grants: Number(row.agency) === 1 } };
}

/** Reject a recovery case (reuses the review-queue reject, so a fan-report
 * backlink — if any — is restored). */
export async function rejectRecoveryCase(
  db: DbLike,
  actor: string,
  itemId: string,
  reason: string,
  now: () => string = () => new Date().toISOString(),
): Promise<void> {
  // I3-1030: the same case lease as `resolveRecoveryCase`: a reject cannot land
  // while a resolve is mid-actions.
  const lease = await takeCaseLease(db, itemId, now);
  try {
    const item = await getItem(db, itemId);
    if (!item) throw new RecoveryError("no such recovery case", 404);
    if (item.item_type !== "dispute" || isFraudReview(item)) throw new RecoveryError("not a recovery case", 400);
    // I15-FX4 RG-AUTH-02: this is the one door review.ts lets close a recovery case.
    // I18-FXA D37 (D-0949(1)): the hold release and its audit row ride `rejectItem`'s own batch (below the
    // decision lock), so a failing row leaves the case NOT rejected and its holds open, and a retry can run.
    const ts = now();
    await rejectItem(db, actor, itemId, reason, now, { recoveryDoor: true }, [
      db
        .prepare("UPDATE collectible_holds SET state = 'closed', closed_at = ?, closed_by = ?, closed_reason = 'recovery case rejected' WHERE ref = ? AND state = 'open'")
        .bind(ts, actor, itemId),
      holdsReleasedStmt(db, actor, itemId, "open collectible holds naming this case were closed with its rejection", ts),
    ]);
    await clearResolveSignoffs(db, itemId);
    // I3-0290, THE SECOND DOOR OUT OF `open`. A REJECTED case releases its holds
    // for the same reason a resolved one does: the case is closed and the hold
    // that named it is now refusing the fan's erasure on behalf of nothing. A
    // fix applied only at `resolveRecoveryCase` would leave the rejection path
    // exactly as broken (FAMILY-004 clause 4), so it is applied per site and
    // proved per site.
    //
    // The release rides `rejectItem`'s batch (above), under its decision lock: a losing rejector's batch
    // fails on the lock, so it never frees another decision's holds.
  } finally {
    await releaseCronLease(db, lease);
  }
}

/**
 * Clear every dual-control sign-off collected for a resolve that will now never
 * run — otherwise the ops pending list advertises an approval for an action
 * with no path back to executable.
 *
 * WHY TWO SPELLINGS. `recovery:resolve` is signed under TWO different
 * `params_hash` shapes, because the two case origins mean different things by
 * "the thing being approved":
 *
 *   operator-authored  →  `<itemId>`                    (plan is already stored)
 *   request-origin     →  `<itemId>|<recoveryPlanKey>`  (plan supplied at resolve)
 *
 * The composite exists for a real reason (co-signers must approve the SAME
 * remediation, not merely "resolve case X"), but this cleanup only ever deleted
 * the bare-id row, so a rejected self-service case left its sign-offs behind
 * until they lapsed: a sign-off expires at `PENDING_SIGNOFF_TTL_MS` (7 days,
 * dual_control.ts), and the gate and `listPending` apply that cutoff themselves
 * (`signoffCutoff`), sweep or no sweep. Not exploitable — resolve requires an
 * OPEN state and reject has no path back to open, so a stale row can never be
 * spent — but with two approvals required it is a week-long lie on the pending
 * dashboard, and the next
 * operator reading that list has to work out on their own that it means
 * nothing.
 *
 * A PREFIX MATCH, ESCAPED. Deleting `= id OR LIKE 'id|%'` covers both shapes and
 * every plan a co-signer might have signed for this case (there can be several:
 * two operators who typed genuinely different plans produce two different keys,
 * and all of them die with the case). `_` and `%` are LIKE wildcards and the
 * minted ids contain `_`, so the prefix is escaped — an unescaped `itm_ab…`
 * would also match `itmXab…`, and a cleanup that deletes another case's
 * sign-offs is a worse bug than the one being fixed.
 */
/**
 * AA-06 — the ESCAPED LIKE prefix `clearResolveSignoffs` sends to D1, bounded
 * short enough to stay under D1's real pattern-length ceiling.
 *
 * Exported ONLY for the byte-length regression test (AA-06); the DELETE is the
 * one real caller.
 *
 * `SIGNOFF_PREFIX_ID_BOUND` (40, not the 136 this used to be): AA-06's own
 * measurement puts D1's LIKE pattern ceiling at ~50 bytes (`aa_glob.mjs`). The
 * old 136-char bound sat 86+ bytes past the point D1 throws, so it protected
 * nothing an id anywhere near that long would need — the bound was the
 * failure, exactly as AA-06 says. `itm_` ids are ~24 characters via
 * mintId("itm") today (far inside 40, so this changes nothing for any id that
 * exists), and 40 keeps the escaped pattern (id + 1 escaped `_` + the `|%`
 * suffix, 43 bytes worst case) under D1's ceiling with margin — so a future
 * longer id fails LOUD on D1 (rejected statement) rather than SILENT here
 * (truncated prefix, stale sign-offs left behind). Real protection against a
 * longer id ever being minted is AA-02's static check on id length, not this
 * number; this bound only decides which of the two failure shapes a length
 * regression produces.
 */
export const SIGNOFF_PREFIX_ID_BOUND = 40;

/** Pure: the escaped `<key>|%` prefix `clearResolveSignoffs` binds to the LIKE.
 * `_` and `%` are LIKE wildcards and minted ids contain `_`, so the prefix is
 * escaped — an unescaped `itm_ab…` would also match `itmXab…`. */
export function boundedSignoffPrefix(itemId: string): string {
  const key = itemId.slice(0, SIGNOFF_PREFIX_ID_BOUND);
  return `${key.replace(/[\\%_]/g, (ch) => `\\${ch}`)}|%`;
}

async function clearResolveSignoffs(db: DbLike, itemId: string): Promise<void> {
  // The stored key's PREFIX. `withDualControl` no longer stores
  // `paramsKey.slice(0, 200)` — since slice 21 it stores `boundedKey(paramsKey)`,
  // which is the verbatim key when it fits and a prefix-plus-digest when it does
  // not. Both forms START with the item id, so this prefix match still reaches
  // every sign-off for this case in either form as long as the id fits inside
  // `SIGNOFF_PREFIX_ID_BOUND` — see `boundedSignoffPrefix` above for why that
  // bound is 40 and not the 136 it used to be.
  const key = itemId.slice(0, SIGNOFF_PREFIX_ID_BOUND);
  const prefix = boundedSignoffPrefix(itemId);
  await db
    .prepare("DELETE FROM pending_actions WHERE action_type = 'recovery:resolve' AND (params_hash = ? OR params_hash LIKE ? ESCAPE '\\')")
    .bind(key, prefix)
    .run();
}

/**
 * D-0820(5), D-0823(1): which address a `fan_email:removed` row removed, or a
 * `fan_email:swap` row replaced, without storing it —
 * `rm#` + 32 hex of canonical's single hasher (`sha256Bytes`, as ratelimit.ts
 * `refusalKeyForAudit`) over this label, the account id and
 * the stored (normalized) address. Keyed on the account id, so after an erasure
 * (account_data.ts re-keys the row's item_id) the tag no longer answers "was this
 * address ever removed". The removal door (fan-auth.ts removeFanEmail) writes it
 * into the row's detail, as does the swap arm of fan-auth.ts redeemOtp;
 * `fanEmailClearEligibility` compares the FIRST in-window row's tag to the
 * current address, so returning to the pre-chain address is not a replacement.
 * account_data.ts deleteAccount blanks it from both rows on erasure (D-0823(4)).
 */
export async function fanEmailRemovalTag(accountId: string, address: string): Promise<string> {
  const d = await sha256Bytes(new TextEncoder().encode(`dhwaja-fan-email-removed-v1\n${accountId}\n${address}`));
  return `rm#${[...d.slice(0, 16)].map((b) => b.toString(16).padStart(2, "0")).join("")}`;
}
/** Mail template for the owner notice (alias owed in mailer.ts — until it is
 * mapped, `send` returns false and the undelivered-channel audit line fires). */
export const FAN_TAKEOVER_TEMPLATE = "fan-takeover-claim";
export const FAN_TAKEOVER_PUSH_KIND = "takeover_claim";
export const FAN_TAKEOVER_PUSH_NOTICE =
  "Someone asked Dhwaja to review who controls this account. Nothing on it has changed; an operator reviews every request before anything happens.";
const ACC = /^acc_[0-9a-f]{20,64}$/;
const FAN_CLAIM_EMAIL_RE = /^[^\s@]{1,64}@[^\s@]{1,255}\.[^\s@]{2,24}$/;
const OPEN_FAN_CASE =
  "SELECT item_id FROM review_queue WHERE item_type = 'dispute' AND state = 'open' AND subject = ? AND json_extract(payload, '$.kind') = 'fan_takeover' ORDER BY created_at ASC, item_id ASC LIMIT 1";

/** The one plan a fan takeover case accepts. `subject` "" = shape-only pre-gate. */
function validateFanPlan(subject: string, raw: unknown): { actions: RecoveryAction[]; clear_email?: true } {
  if (subject !== "" && !ACC.test(subject)) throw new RecoveryError("a fan takeover case must name a fan account");
  const i = (raw ?? {}) as Record<string, unknown>;
  if (!Array.isArray(i.actions) || i.actions.length !== 1 || i.actions[0] !== FAN_REVOKE_ACTION) {
    throw new RecoveryError(`a fan takeover plan is exactly {"actions":["${FAN_REVOKE_ACTION}"]}`);
  }
  for (const k of Object.keys(i)) if (k !== "actions" && k !== "clear_email") throw new RecoveryError(`a fan takeover plan takes no ${k}`);
  if (i.clear_email !== undefined && typeof i.clear_email !== "boolean") throw new RecoveryError("clear_email is true or false");
  return i.clear_email === true ? { actions: [FAN_REVOKE_ACTION], clear_email: true } : { actions: [FAN_REVOKE_ACTION] };
}

export interface FanTakeoverClaimInput {
  address?: unknown;
  description?: unknown;
  contact?: unknown;
  /** Optional: when the filer says they lost control (ISO instant, not after now). */
  compromised_at?: unknown;
}

/**
 * FILE a fan takeover claim. Never changes the account; never grants. Returns the
 * account to notify (null when nothing matched) — the ROUTE must not let that
 * value shape its response.
 */
export async function fileFanTakeoverClaim(
  db: DbLike,
  input: FanTakeoverClaimInput | unknown,
  callerKey: string,
  nowMs: number,
  now: () => string = () => new Date(nowMs).toISOString(),
): Promise<{ notify: { account_id: string; item_id: string } | null }> {
  const i = (input ?? {}) as FanTakeoverClaimInput;
  const address = typeof i.address === "string" ? i.address.trim().toLowerCase() : "";
  if (!FAN_CLAIM_EMAIL_RE.test(address)) throw new RecoveryError("the email address the account used is required");
  const description = typeof i.description === "string" ? i.description.trim() : "";
  if (!description || description.length > 1000) throw new RecoveryError("a description (1..1000 chars) is required");
  const contact = typeof i.contact === "string" ? i.contact.trim() : "";
  if (!contact || contact.length > 200) throw new RecoveryError("a way to reach you (1..200 chars) is required");
  // Shape-only, before any lookup, so it answers the same for every address.
  let compromisedAt: string | null = null;
  if (i.compromised_at !== undefined && i.compromised_at !== null && i.compromised_at !== "") {
    const ms = typeof i.compromised_at === "string" && /^\d{4}-\d{2}-\d{2}(T[0-9:.]+Z)?$/.test(i.compromised_at) ? Date.parse(i.compromised_at) : Number.NaN;
    if (!Number.isFinite(ms) || ms > nowMs) throw new RecoveryError("compromised_at is a date (YYYY-MM-DD or an ISO UTC instant) not in the future");
    compromisedAt = new Date(ms).toISOString();
  }

  if (!(await consumeRateLimit(db, `fantko:c:${callerKey}`, nowMs, FAN_CLAIM_CALLER_DAILY_CAP))) {
    throw Object.assign(new RecoveryError("too many requests today — try again tomorrow", 429), { ipKeyed: true }); // D-1004(1): `fantko:c:<masked ip>` (route-helpers.ts errorBody)
  }

  const match = await db
    .prepare(
      `SELECT a.account_id FROM accounts a WHERE a.account_id IN (
         SELECT account_id FROM fan_email WHERE address = ? AND verified_at IS NOT NULL
         UNION SELECT account_id FROM oauth_identities WHERE lower(email) = ? AND email_verified = 1)
       ORDER BY a.created_at ASC, a.account_id ASC LIMIT 1`,
    )
    .bind(address, address)
    .first<{ account_id: string }>();
  // A miss is bound to an id no account can hold, so every statement below runs
  // and every guard evaluates false for it.
  const subject = match?.account_id ?? "acc_00000000000000000000";
  const ts = now();
  const itemId = mintId("itm");
  const line = `${ts} contact: ${contact}${compromisedAt ? ` (says control was lost at ${compromisedAt})` : ""} — ${description}`;
  // `compromised_at` on the payload is the CASE-OPENING claim's stated time
  // (D-0807's "earliest claim"); later filings carry theirs in their own line.
  const payload = JSON.stringify({ kind: FAN_TAKEOVER_KIND, origin: "fan_claim", actions: [], description, contact, compromised_at: compromisedAt, evidence: [line], request_count: 1 });
  // The per-target counter: counted for hit AND miss (a miss counts against the
  // sentinel), with no cap in the upsert, so the statement is the same on every
  // branch; the cap is a GUARD on the writes below.
  const tkey = `fantko:t:${subject}`;
  const cutoff = new Date(nowMs - RL_DAY_MS).toISOString();
  // `within` / `notified` choose the AUDIT line only (D-0810(3): every filing
  // folds); `notified` also gates the notice.
  const within = `(SELECT count FROM rate_limits WHERE rl_key = ?) <= ${FAN_CLAIM_TARGET_ATTACH_CAP}`;
  const notified = `(SELECT count FROM rate_limits WHERE rl_key = ?) <= ${FAN_CLAIM_TARGET_DAILY_CAP}`;
  await db.batch([
    db
      .prepare(
        `INSERT INTO rate_limits (rl_key, window_start, count) VALUES (?, ?, 1)
         ON CONFLICT(rl_key) DO UPDATE SET
           window_start = CASE WHEN rate_limits.window_start < ? THEN ? ELSE rate_limits.window_start END,
           count        = CASE WHEN rate_limits.window_start < ? THEN 1  ELSE rate_limits.count + 1 END`,
      )
      .bind(tkey, ts, cutoff, ts, cutoff),
    // FOLD into the open case (lands only when one was open BEFORE this batch —
    // it runs before the insert). D-0810(3): EVERY filing folds, past the attach
    // cap too; the count is unbounded and the evidence keeps FAN_CLAIM_EVIDENCE_RING
    // lines: the first EVIDENCE_HEAD_KEEP and the latest (QUEUE #662: the oldest line
    // AFTER the head is dropped), so a flood buries neither the earliest nor the newest claim.
    db
      .prepare(
        `UPDATE review_queue SET updated_at = ?, payload = json_set(CASE
           WHEN COALESCE(json_array_length(payload, '$.evidence'), 0) >= ${FAN_CLAIM_EVIDENCE_RING} THEN json_remove(payload, '$.evidence[${EVIDENCE_HEAD_KEEP}]')
           ELSE payload END,
           '$.request_count', COALESCE(json_extract(payload, '$.request_count'), 1) + 1, '$.evidence[#]', ?)
         WHERE item_id = (${OPEN_FAN_CASE})`,
      )
      .bind(ts, line, subject),
    db
      .prepare(
        `INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind)
         SELECT 'system:fan_claim', 'recovery:request_dup', 'fan_takeover +1', ?, r.item_id, 'system'
           FROM review_queue r WHERE r.item_id = (${OPEN_FAN_CASE}) AND r.updated_at = ?`,
      )
      .bind(ts, subject, ts),
    db
      .prepare(
        `INSERT INTO review_queue (item_id, item_type, state, subject, payload, created_at, updated_at, import_batch_id)
         SELECT ?, 'dispute', 'open', ?, ?, ?, ?, NULL
          WHERE EXISTS (SELECT 1 FROM accounts WHERE account_id = ?) AND NOT EXISTS (${OPEN_FAN_CASE})`,
      )
      .bind(itemId, subject, payload, ts, ts, subject, subject),
    db
      .prepare(
        `INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind)
         SELECT 'system:fan_claim', 'enqueue:dispute', 'fan_takeover', ?, ?, 'system' WHERE EXISTS (SELECT 1 FROM review_queue WHERE item_id = ?)`,
      )
      .bind(ts, itemId, itemId),
    db
      .prepare(
        `INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind)
         SELECT 'system:fan_claim', 'recovery:fan_claim_capped', 'per-account daily attach cap reached: filing kept in the recent ring, nobody notified', ?, (${OPEN_FAN_CASE}), 'system'
          WHERE EXISTS (SELECT 1 FROM accounts WHERE account_id = ?) AND NOT (${within})`,
      )
      .bind(ts, subject, subject, tkey),
    db
      .prepare(
        `INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind)
         SELECT 'system:fan_claim', 'recovery:fan_claim_unnotified', 'per-account daily notice cap reached: filing attached, nobody notified', ?, (${OPEN_FAN_CASE}), 'system'
          WHERE EXISTS (SELECT 1 FROM accounts WHERE account_id = ?) AND ${within} AND NOT (${notified})`,
      )
      .bind(ts, subject, subject, tkey, tkey),
  ]);
  const back = await db
    .prepare(`SELECT (${OPEN_FAN_CASE}) AS item_id, CASE WHEN ${notified} THEN 1 ELSE 0 END AS ok`)
    .bind(subject, tkey)
    .first<{ item_id: string | null; ok: number }>();
  return { notify: match && back?.item_id && back.ok === 1 ? { account_id: match.account_id, item_id: back.item_id } : null };
}

/**
 * TELL THE ACCOUNT, on every channel it has: each VERIFIED address (fan_email,
 * oauth_identities with email_verified) and each registered push device. The
 * notice names nobody and carries nothing the filer typed. A channel that could
 * not be reached, or an account with none, is an audit line on the case — a
 * notice is never dropped without a record.
 */
export async function notifyFanTakeoverClaim(
  db: DbLike,
  mailer: Mailer,
  push: PushSender,
  accountId: string,
  itemId: string,
  now: () => string = () => new Date().toISOString(),
): Promise<{ emailed: number; pushed: number; channels: number }> {
  const addrs = (
    await db
      .prepare(
        `SELECT address AS a FROM fan_email WHERE account_id = ? AND verified_at IS NOT NULL
         UNION SELECT lower(email) AS a FROM oauth_identities WHERE account_id = ? AND email_verified = 1 AND email IS NOT NULL`,
      )
      .bind(accountId, accountId)
      .all<{ a: string }>()
  ).results;
  const devices = (
    await db.prepare("SELECT device_id, platform, push_token FROM devices WHERE account_id = ?").bind(accountId).all<{ device_id: string; platform: "apns" | "fcm"; push_token: string }>()
  ).results;
  const filedUtc = now();
  let emailed = 0;
  let pushed = 0;
  for (const r of addrs) {
    const ok = await mailer.send(FAN_TAKEOVER_TEMPLATE, r.a, { filed_utc: filedUtc }).catch(() => false);
    if (ok) emailed++;
  }
  for (const d of devices) {
    const out = await push(d, { account_id: accountId, kind: FAN_TAKEOVER_PUSH_KIND, notice: FAN_TAKEOVER_PUSH_NOTICE }).catch(() => "retry" as const);
    if (out === true || out === "ok") pushed++;
  }
  const channels = addrs.length + devices.length;
  const undelivered = channels - emailed - pushed;
  const detail = `email ${emailed}/${addrs.length}, push ${pushed}/${devices.length}`;
  await auditStmt(
    db,
    "system:fan_claim",
    channels === 0 ? "recovery:fan_claim_no_channel" : undelivered > 0 ? "recovery:fan_claim_notice_undelivered" : "recovery:fan_claim_notified",
    itemId,
    detail,
    now(),
    "system",
  ).run();
  return { emailed, pushed, channels };
}

const NO_PREDICATE = { sql: "1", binds: [] as unknown[] };

/** C-40 FAN SIDE (D-0682, D-0777, D-0791): every live session (the epoch bump),
 * every passkey, every push device and every outstanding OTP goes, in the
 * caller's own batch, and the owner re-enrols. No provenance guess.
 * `oauth_identities` and `fan_email` are NOT touched, so the owner keeps a way
 * back in. `pred` is the caller's own flip predicate (`decided` on the appeal
 * arm); the default is none. THE ONE IMPLEMENTATION (#661, D-0807(5)): the
 * compromise lift and the overturned appeal (closures.ts) and this module's fan
 * takeover resolve all call this. It lives HERE, not in closures.ts, because recovery.ts is on the
 * verdict closure and closures.ts reaches billing (tests/invariants/paywall.test.ts);
 * b27-fan-takeover.test.ts pins that closures.ts reaches it through the import. */
export function revokeFanSignIn(db: DbLike, accountId: string, pred: { sql: string; binds: unknown[] } = NO_PREDICATE): DbStatement[] {
  return [
    db.prepare(`UPDATE accounts SET session_epoch = session_epoch + 1 WHERE account_id = ? AND ${pred.sql}`).bind(accountId, ...pred.binds),
    db.prepare(`UPDATE fan_passkeys SET status = 'revoked' WHERE account_id = ? AND status = 'active' AND ${pred.sql}`).bind(accountId, ...pred.binds),
    db.prepare(`DELETE FROM devices WHERE account_id = ? AND ${pred.sql}`).bind(accountId, ...pred.binds),
    db.prepare(`DELETE FROM fan_otp WHERE account_id = ? AND ${pred.sql}`).bind(accountId, ...pred.binds),
  ];
}

/**
 * QUEUE #146 (D-0807, D-0817(1)): may the fan plan's `clear_email` option act on this case?
 * Only an email that REPLACED an earlier one: a swap records a `fan_email:swap` audit
 * row stamped with the `created_at` it writes (fan-auth.ts redeemOtp); an email
 * enrolled within FAN_EMAIL_CLEAR_WINDOW_MS after a `fan_email:removed` row (fan-auth.ts
 * removeFanEmail) also counts (D-0819(1)) — unless it is the address the account
 * held before the FIRST swap or removal in that window (D-0823(1), amending
 * D-0820(5); both rows carry `fanEmailRemovalTag`). Such an email
 * qualifies when it took effect after the claim's stated compromise time, within
 * FAN_EMAIL_CLEAR_WINDOW_MS before the first claim (the case's `created_at`), or at
 * any time after the first claim. A first enrolment never qualifies. A row with no
 * audit line but the pre-change swap stamp (`verified_at = created_at`, both written
 * by the one swap UPDATE; an enrolment verifies later than it was added) keeps the
 * rule from before the audit row existed. Returns the judged row so the delete binds
 * that exact value.
 */
export async function fanEmailClearEligibility(
  db: DbLike,
  itemId: string,
  accountId: string,
): Promise<{ refusal: string | null; row: { address: string; created_at: string } | null }> {
  const [email, kase] = await Promise.all([
    db
      .prepare(
        "SELECT address, created_at, verified_at FROM fan_email WHERE account_id = ?",
      )
      .bind(accountId)
      .first<{ address: string; created_at: string; verified_at: string | null }>(),
    db
      .prepare("SELECT created_at, state, json_extract(payload, '$.compromised_at') AS compromised_at FROM review_queue WHERE item_id = ? AND subject = ? AND json_extract(payload, '$.kind') = 'fan_takeover'")
      .bind(itemId, accountId)
      .first<{ created_at: string; state: string; compromised_at: string | null }>(),
  ]);
  if (!kase) return { refusal: "no fan takeover case for this account", row: null };
  if (!email) return { refusal: "the account has no email to clear", row: null };
  // D-0823(1) (amends D-0820(5) and D-0821(3)): THE ADDRESS THAT DECIDES is the
  // one the account held BEFORE its FIRST swap or removal inside the window
  // [created_at - FAN_EMAIL_CLEAR_WINDOW_MS, created_at] of the current email.
  // Both rows carry that address's one-way tag (fan-auth.ts redeemOtp swap arm,
  // removeFanEmail), so the EARLIEST row in the window names it: the current
  // email is replaced iff such a row exists and it is not the current address's
  // tag — any chain of swaps, removals and re-adds that ends on another address
  // counts, and one that ends back on the first address does not. A first row
  // with no tag (written before the tags existed) keeps the old rule: replaced.
  // D-0825(1) (amends D-0824(2)): while THIS case is `open` the window is anchored
  // at the case — from FAN_EMAIL_CLEAR_WINDOW_MS before it opened, through now (no
  // upper bound) — so removing and re-adding an address while the owner's claim is
  // pending cannot age the swap out of the window. The lower bound is the EARLIER of
  // the case's and the current email's look-back, so D-0819(1)'s removal a few days
  // before the case's look-back still counts. Not open: D-0824(2)'s window,
  // [email.created_at - FAN_EMAIL_CLEAR_WINDOW_MS, email.created_at].
  const open = kase.state === "open";
  const set = Date.parse(email.created_at);
  const lo = (open ? Math.min(set, Date.parse(kase.created_at)) : set) - FAN_EMAIL_CLEAR_WINDOW_MS;
  const chain = await db
    .prepare(
      `SELECT detail FROM audit_log WHERE item_id = ? AND action IN ('fan_email:swap', 'fan_email:removed') AND created_at >= ?${open ? "" : " AND created_at <= ?"} ` +
        "ORDER BY created_at ASC, audit_id ASC LIMIT 1",
    )
    .bind(...[accountId, new Date(lo).toISOString(), ...(open ? [] : [email.created_at])])
    .first<{ detail: string | null }>();
  const firstTag = /\brm#[0-9a-f]{32}\b/.exec(chain?.detail ?? "")?.[0] ?? null;
  const replaced = Boolean(chain) && (firstTag === null || firstTag !== (await fanEmailRemovalTag(accountId, email.address)));
  // Pre-audit swaps only: when a chain row exists its tags decide (a swap back
  // to the pre-chain address also stamps verified_at = created_at).
  const legacySwap = !chain && email.verified_at !== null && email.verified_at === email.created_at;
  if (!replaced && !legacySwap) {
    return { refusal: `the email cannot be cleared: it is the account's first email (set at ${email.created_at}), not a replacement of an earlier one`, row: null };
  }
  const changed = Date.parse(email.created_at);
  const first = Date.parse(kase.created_at);
  // D-0810(3): the stated time is the first FILER's claim, so it is clamped to
  // no earlier than FAN_EMAIL_CLEAR_WINDOW_MS before the first claim — a
  // stranger cannot widen the look-back by stating 1970. (The ruling's other
  // bound, the account's creation, can never be later than this one AND
  // decisive: an account exists before its claim, and every instant in
  // [floor, first] is already admitted by `beforeFirst`.)
  const floor = first - FAN_EMAIL_CLEAR_WINDOW_MS;
  const raw = kase.compromised_at ? Date.parse(kase.compromised_at) : Number.NaN;
  const stated = Number.isFinite(raw) ? Math.max(raw, floor) : Number.NaN;
  const afterStated = Number.isFinite(stated) && changed > stated;
  const beforeFirst = changed >= first - FAN_EMAIL_CLEAR_WINDOW_MS && changed <= first;
  const afterFirst = replaced && changed > first;
  if (afterStated || beforeFirst || afterFirst) return { refusal: null, row: { address: email.address, created_at: email.created_at } };
  return {
    refusal: `the email cannot be cleared: it took effect at ${email.created_at}, which is ${kase.compromised_at ? `not after the first claim's stated compromise time (${kase.compromised_at}) and ` : ""}not within 30 days before the first claim (${kase.created_at})`,
    row: null,
  };
}

async function resolveFanTakeover(db: DbLike, actor: string, itemId: string, accountId: string, plan: unknown, now: () => string): Promise<RecoveryResult> {
  const p = validateFanPlan(accountId, plan);
  const exists = await db.prepare("SELECT 1 AS x FROM accounts WHERE account_id = ?").bind(accountId).first();
  if (!exists) throw new RecoveryError("affected fan account no longer exists", 404);
  // D-0807: the clear-email option is judged here, inside the dual-controlled
  // execute, and the delete below is bound to the judged (address, created_at).
  let judged: { address: string; created_at: string } | null = null;
  if (p.clear_email) {
    const e = await fanEmailClearEligibility(db, itemId, accountId);
    if (e.refusal || !e.row) throw new RecoveryError(e.refusal ?? "the email cannot be cleared", 409);
    judged = e.row;
  }
  const ts = now();
  const same = "SELECT 1 FROM fan_email WHERE account_id = ? AND address = ? AND created_at = ?";
  const clear: DbStatement[] = judged
    ? [
        // Audit FIRST, on the same predicate the delete uses, inside one
        // transaction: `cleared` is written exactly when the row is deleted.
        db
          .prepare(`INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, 'recovery:fan_email_cleared', ?, ?, ?, 'operator' WHERE EXISTS (${same})`)
          .bind(actor, `fan_email set ${judged.created_at} cleared (OAuth identities kept, D-0791)`, ts, itemId, accountId, judged.address, judged.created_at),
        db
          .prepare(`INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, 'recovery:fan_email_clear_skipped', 'the email changed after it was judged; not cleared', ?, ?, 'operator' WHERE NOT EXISTS (${same})`)
          .bind(actor, ts, itemId, accountId, judged.address, judged.created_at),
        db.prepare("DELETE FROM fan_email WHERE account_id = ? AND address = ? AND created_at = ?").bind(accountId, judged.address, judged.created_at),
      ]
    : [];
  try {
    // ONE transaction: the decision row (unique per item/from_state — a second
    // decider throws and rolls the revocations back with it), the state flip,
    // the revocation and the audit line.
    await db.batch([
      reviewDecisionStmt(db, itemId, "open", "approved", actor, ts, null),
      db
        .prepare("UPDATE review_queue SET state = 'approved', disposition = 'approved', disposition_note = ?, result_event = NULL, updated_at = ? WHERE item_id = ? AND state = 'open'")
        .bind(`recovery: ${FAN_REVOKE_ACTION}`, ts, itemId),
      ...revokeFanSignIn(db, accountId),
      ...clear,
      auditStmt(db, actor, "recovery:resolve", itemId, `${FAN_TAKEOVER_KIND} → ${FAN_REVOKE_ACTION}${judged ? "+clear_email" : ""}`, ts),
      // I3-0290 (I22-W1-AUTH): this arm closes the case, so it releases the holds naming it — the same keyed
      // statement and audit pair as resolveRecoveryCase's batch; `holdsReleasedStmt` must follow the UPDATE.
      db
        .prepare("UPDATE collectible_holds SET state = 'closed', closed_at = ?, closed_by = ?, closed_reason = 'recovery case resolved' WHERE ref = ? AND state = 'open'")
        .bind(ts, actor, itemId),
      holdsReleasedStmt(db, actor, itemId, "open collectible holds naming this case were closed with it", ts),
    ]);
  } catch (e) {
    if (isDecisionLost(e)) throw new RecoveryError("case was resolved concurrently", 409);
    throw e;
  }
  await clearResolveSignoffs(db, itemId);
  if (!judged) return { item_id: itemId, actions_executed: p.actions };
  // The `cleared` audit row is written iff the delete ran (same predicate, same
  // transaction), so it — not a re-read of fan_email — is the outcome.
  const cleared = await db.prepare("SELECT 1 AS x FROM audit_log WHERE item_id = ? AND action = 'recovery:fan_email_cleared' AND created_at = ?").bind(itemId, ts).first();
  return { item_id: itemId, actions_executed: p.actions, email_cleared: Boolean(cleared) };
}

// ---------------------------------------------------------------------------
// D-0921(2) / #157 RV-01 (I16-W04C; migration 0185): THE RE-ENTRY GRANT.
//
// A hijack resolution FORCES `reset_passkeys` and `evict_contact_channels` (FORCED_ACTIONS.hijack), so the rightful
// owner can be left with no passkey and — unless the operator spared it — no address. Where the operator DID spare
// the verified address, this is the door back: an operator-issued grant, approved by a SECOND operator (dual
// control: `issuedBy` and `approvedBy` must differ, and the table's CHECK says so too), single-use, valid
// REENTRY_GRANT_TTL_MS, mailed ONLY to that spared address. Nothing is ever sent to an evicted channel — it would
// reach the attacker — so a case with no spare (or whose spare is no longer the passport's verified address) is
// refused and nothing is mailed anywhere. The token exists only in the mail; the table holds its sha256.

/** The grant's stored digests: canonical's single hasher (`sha256Bytes`) over a domain-separated preimage, the same
 * construction this file's fan-email-removed marker uses (:1621). Neither input is a subject. */
async function reentryDigest(label: "token" | "address", value: string): Promise<string> {
  const d = await sha256Bytes(new TextEncoder().encode(`dhwaja-reentry-grant-v1\n${label}\n${value}`));
  return [...d].map((b) => b.toString(16).padStart(2, "0")).join("");
}

export async function issueReentryGrant(
  db: DbLike,
  sender: EmailSender,
  input: { itemId: string; issuedBy: string; approvedBy: string; consoleOrigin?: string | null },
  nowMs: number,
): Promise<ReentryGrantIssue> {
  const { itemId, issuedBy, approvedBy } = input;
  if (typeof issuedBy !== "string" || typeof approvedBy !== "string" || issuedBy.trim() === "" || approvedBy.trim() === "" || issuedBy === approvedBy) {
    throw new RecoveryError("a re-entry grant needs two different operators (issuer and approver)", 400);
  }
  const item = await getItem(db, itemId);
  if (item?.item_type !== "dispute" || isFraudReview(item)) throw new RecoveryError("no such recovery case", 404);
  const payload = item.payload as { kind?: unknown; spare_contact_email?: unknown };
  if (payload.kind !== "hijack" || item.state !== "approved") throw new RecoveryError("a re-entry grant follows a resolved hijack case only", 409);
  const spare = normalizeSpareAddress(payload.spare_contact_email);
  const contact = await db
    .prepare("SELECT email, email_verified FROM creator_contacts WHERE passport_id = ?")
    .bind(item.subject)
    .first<{ email: string | null; email_verified: number }>();
  const held = contact?.email_verified === 1 && contact.email ? contact.email.trim().toLowerCase() : null;
  // ONLY the spared address, and only while it is still the passport's verified address. No fallback, no notice.
  if (spare === null || held === null || spare !== held) throw new RecoveryError("this case spared no address that is still verified — a re-entry grant is never sent anywhere else", 409);
  const token = randomHex(32);
  const grantId = mintId("rgr");
  const issuedAt = new Date(nowMs).toISOString();
  const expiresAt = new Date(nowMs + REENTRY_GRANT_TTL_MS).toISOString();
  await db.batch([
    db
      .prepare(
        `INSERT INTO creator_reentry_grants (grant_id, item_id, passport_id, token_hash, delivered_to_hash, issued_by, approved_by, issued_at, expires_at)
         VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)`,
      )
      .bind(grantId, itemId, item.subject, await reentryDigest("token", token), await reentryDigest("address", held), issuedBy, approvedBy, issuedAt, expiresAt),
    auditStmt(db, issuedBy, "recovery:reentry_grant_issued", itemId, `grant ${grantId}, approved by ${approvedBy}, expires ${expiresAt}`.slice(0, 150), issuedAt),
  ]);
  const link = input.consoleOrigin ? `${input.consoleOrigin}/#/reentry?grant=${token}` : null;
  const text =
    "An operator has finished reviewing the takeover of your Dhwaja creator passport. Every passkey on it was removed, so you need a new one.\n\n" +
    (link ? `Use this link within 72 hours to add a passkey and get back in. It works once:\n${link}` : `Your one-time re-entry code (valid 72 hours, works once): ${token}`) +
    "\n\nIf you did not ask for this, ignore it — nothing changes unless the link is used.";
  let outcome: EmailOutcome;
  try {
    outcome = await sendOnce(sender, { to: held, subject: "Dhwaja — your way back into your creator passport", text, html: wrapHtml(textToHtml(text), "") });
  } catch {
    outcome = "transient";
  }
  if (outcome !== "accepted" && outcome !== "ambiguous") {
    // Not delivered: withdraw it, so no live grant exists that nobody holds; the operators may issue again.
    await db.prepare("DELETE FROM creator_reentry_grants WHERE grant_id = ? AND used_at IS NULL").bind(grantId).run();
  }
  return { grant_id: grantId, expires_at: expiresAt, outcome };
}

/**
 * Spend a re-entry grant: ONE claiming UPDATE bound to the row read (grant id AND token hash, still unused, not
 * expired), with its own RETURNING verdict (INV-058). The read before it re-checks that the address it was mailed to
 * is still the passport's verified address; a grant whose address was changed or evicted since issue is dead.
 * Every refusal is the same 404. The caller (the redeem route, a hub hand-off) mints the creator session.
 */
export async function redeemReentryGrant(db: DbLike, token: unknown, nowMs: number): Promise<{ passport_id: string }> {
  const none = () => new RecoveryError("no such re-entry grant", 404);
  if (typeof token !== "string" || !REENTRY_TOKEN_RE.test(token)) throw none();
  const nowIso = new Date(nowMs).toISOString();
  const tokenHash = await reentryDigest("token", token);
  const g = await db
    .prepare("SELECT grant_id, passport_id, delivered_to_hash FROM creator_reentry_grants WHERE token_hash = ? AND used_at IS NULL AND expires_at > ?")
    .bind(tokenHash, nowIso)
    .first<{ grant_id: string; passport_id: string; delivered_to_hash: string }>();
  if (!g) throw none();
  const contact = await db
    .prepare("SELECT email, email_verified FROM creator_contacts WHERE passport_id = ?")
    .bind(g.passport_id)
    .first<{ email: string | null; email_verified: number }>();
  const held = contact?.email_verified === 1 && contact.email ? contact.email.trim().toLowerCase() : null;
  if (held === null || (await reentryDigest("address", held)) !== g.delivered_to_hash) throw none();
  const { results } = await db
    .prepare("UPDATE creator_reentry_grants SET used_at = ? WHERE grant_id = ? AND token_hash = ? AND used_at IS NULL AND expires_at > ? RETURNING passport_id")
    .bind(nowIso, g.grant_id, tokenHash, nowIso)
    .all<{ passport_id: string }>();
  if (results.length !== 1) throw none();
  return { passport_id: results[0]!.passport_id };
}

export const REENTRY_GRANT_RETENTION_DAYS = 90;

/**
 * D-0939(7) retention: ONE bounded DELETE. The cutoff is 90 days before `nowMs`, so a live grant (unused,
 * `expires_at` in the future) can never match. Its cron call is in index.ts scheduled() (I17-J02 hand-off to I17-L01).
 */
export async function pruneReentryGrants(db: DbLike, nowMs: number): Promise<void> {
  const cutoff = new Date(nowMs - REENTRY_GRANT_RETENTION_DAYS * 86_400_000).toISOString();
  const dead = "((used_at IS NOT NULL AND used_at < ?) OR (used_at IS NULL AND expires_at < ?))";

    .prepare(`DELETE FROM creator_reentry_grants WHERE ${dead} AND grant_id IN (SELECT grant_id FROM creator_reentry_grants WHERE ${dead} ORDER BY grant_id LIMIT ${REENTRY_GRANT_PRUNE_BATCH})`)
    .bind(cutoff, cutoff, cutoff, cutoff)
    .run();
}

/**
 * I21-Q93: a `scheduled()` step (wired by I29-INT, INV-049) that reduces every recovery case — a `review_queue` row
 * with item_type 'dispute' whose payload kind is a RECOVERY_KINDS kind or `fan_takeover` (other dispute kinds are
 * not this file's) — decided (state not 'open') with `updated_at` before `beforeIso`: its payload loses
 * `description`, `requesters`, `contact`, `evidence` and `spare_contact_email`; kind, origin, actions, the row's
 * state/disposition and created_at/updated_at are kept. Only rows still holding one of those fields are rewritten,
 * so the count is the cases actually reduced and a second run changes nothing. Bounded on `item_id IN (... LIMIT ?)`.
 * I29-Q3 (2026-10-21 answer: "trim existing rows with the same 90-day rule."): another statement sets the `detail`
 * of every `enqueue:dispute` audit row of such a case (any state) created before `beforeIso` to the case id, so a line
 * written before intake stopped copying the request text keeps the case id only; it is not in the returned count,
 * touches at most `limit` rows a run (the rest are the next run's) and reads only rows whose detail is not already the
 * case id. audit_log_seal never covers `detail` (migration 0167), so the rewrite breaks no seal.
 * D-1118(6) (I34-W3-REVIEW): neither statement touches a case an open legal hold covers (recoveryCaseHeldSql below:
 * its subject passport or account, or a requester it names); the first run after release trims it.
 * I34-FX3-HOLDS: the audit statement runs FIRST and reaches every review item's `enqueue:<item_type>` copy, any item
 * type but `report`, whose copy keeps its body (D-1123(2), D-1119(8), D-1120(6)), oldest audit_id first; it skips an
 * item whose subject passport or account, or a requester it names, is held. A case keeps its requesters (and the rest
 * of its request details) while an `enqueue:dispute` copy of it is untrimmed, so that hold check still reads them
 * (RG3R-RECOVERY-HOLD-AUDIT-BACKLOG); a copy the LIMIT leaves for a later run defers its case's reduction to that run.
 */
export async function pruneRecoveryCasePii(db: DbLike, beforeIso: string, limit: number = RECOVERY_CASE_PII_PRUNE_LIMIT): Promise<number> {
  const kinds = JSON.stringify([...RECOVERY_KINDS, FAN_TAKEOVER_KIND]);
  // D-1118(6): an open hold on what the case `q` is about — its subject, a passport (creator and agency recovery) or an
  // account (fan_takeover), or a requester its payload names (`fan:<account_id>`, `creator:<passport_id>`). `held` is
  // legal-hold-sql.ts legalHoldOpenSql's text, inlined: recovery.ts imports no legal-hold module, and it is in the verdict
  // closure, where a dynamic import is refused (tests/invariants/paywall.test.ts).
  const held = (kind: "account" | "passport", id: string): string =>
    `EXISTS (SELECT 1 FROM legal_holds lh WHERE lh.released_at IS NULL AND lh.subject_kind = '${kind}' AND lh.subject_id = ${id})`;
  const recoveryCaseHeldSql = (q: string): string =>
    `(${held("passport", `${q}.subject`)} OR ${held("account", `${q}.subject`)} OR EXISTS (SELECT 1 FROM json_each(CASE WHEN json_valid(${q}.payload) = 1 THEN ${q}.payload ELSE '{}' END, '$.requesters') rq_lh WHERE (substr(rq_lh.value, 1, 4) = 'fan:' AND ${held("account", "substr(rq_lh.value, 5)")}) OR (substr(rq_lh.value, 1, 8) = 'creator:' AND ${held("passport", "substr(rq_lh.value, 9)")})))`;
  // D-1123(2): every non-report review item's enqueue copy, first (see the docblock). `action` is matched as the range
  // ('enqueue:', 'enqueue;') — every action that starts `enqueue:` — so idx_audit_action_id serves it.
  await db
    .prepare(
      `UPDATE audit_log SET detail = item_id
        WHERE audit_id IN (
          SELECT a.audit_id FROM audit_log a
           WHERE a.action > 'enqueue:' AND a.action < 'enqueue;' AND a.created_at < ? AND a.detail <> a.item_id
             AND EXISTS (SELECT 1 FROM review_queue q WHERE q.item_id = a.item_id AND q.item_type <> 'report' AND NOT ${recoveryCaseHeldSql("q")})
           ORDER BY a.audit_id LIMIT ?)`,
    )
    .bind(beforeIso, limit)
    .run();
  const r = await db
    .prepare(
      `UPDATE review_queue SET payload = json_remove(payload, '$.description', '$.requesters', '$.contact', '$.evidence', '$.spare_contact_email')
        WHERE item_id IN (
          SELECT item_id FROM review_queue q0
           WHERE item_type = 'dispute' AND state <> 'open' AND updated_at < ? AND json_valid(payload) = 1 AND NOT ${recoveryCaseHeldSql("q0")}
             AND json_extract(payload, '$.kind') IN (SELECT value FROM json_each(?))
             AND (json_type(payload, '$.description') IS NOT NULL OR json_type(payload, '$.requesters') IS NOT NULL
               OR json_type(payload, '$.contact') IS NOT NULL OR json_type(payload, '$.evidence') IS NOT NULL
               OR json_type(payload, '$.spare_contact_email') IS NOT NULL)
             AND NOT EXISTS (SELECT 1 FROM audit_log a0 WHERE a0.item_id = q0.item_id AND a0.action = 'enqueue:dispute' AND a0.detail <> a0.item_id)
           LIMIT ?)`,
    )
    .bind(beforeIso, kinds, limit)
    .run();
  return r.changes ?? 0;
}

/**
 * D-0745: the credits a reversal takes back. A LOST dispute takes
 * floor(credits_granted * disputed / gross) — floor, so the take-back never
 * exceeds the disputed share (the rounding is under one credit = one cent). A
 * missing or unusable amount, or one covering the whole charge, takes the
 * whole grant, which is the pre-D-0745 behaviour. I21-Q70 (D-1030 (a)): a
 * REFUND takes floor(credits_granted * refunded / gross) by the same rule, from
 * `refundedCents` (the fifth argument; a refund given none takes it all).
 */
export function clawFor(kind: ReverseInput["kind"], granted: number, grossCents: number, disputedCents: number | undefined, refundedCents?: number | undefined): number {
  const amount = kind === "dispute" ? disputedCents : kind === "refund" ? refundedCents : undefined;
  if (amount === undefined) return granted;
  if (!Number.isInteger(amount) || amount <= 0 || !Number.isInteger(grossCents) || grossCents <= 0 || amount >= grossCents) return granted;
  return Math.floor((granted * amount) / grossCents);
}

/** I28 RG-AM-01 (D-1025(8)): this dispute key is already RELEASED on a link of crypto charge `?` (binds: key, the charge
 * id, its length + 1, the charge id + "+"; the chain predicate of crypto-rail.ts `cryptoChainKeys`). An open delivered
 * after its won must not freeze a link that settled between them: the won consumed the key only on the links it saw. */
const CHAIN_KEY_RELEASED_SQL =
  "EXISTS (SELECT 1 FROM credit_ledger w JOIN credit_purchases c ON c.purchase_id = w.ref " +
  "WHERE w.idem = 'dispute_unfreeze:' || c.purchase_id || ':' || ? AND c.rail = 'crypto' AND (c.external_txn_id = ? OR substr(c.external_txn_id, 1, ?) = ?))";
/** D-1037(3): what a partial refund already took back from the purchase in scope (`credit_purchases` row), 0 when none. */
const PARTIAL_REFUND_CLAWED_SQL = "COALESCE((SELECT r.credits_clawed FROM credit_reversals r WHERE r.purchase_id = credit_purchases.purchase_id), 0)";
/** D-1037(3): the purchase in scope was reversed only by a PARTIAL refund — something is left and no lost dispute took it. */
const PARTIALLY_REFUNDED_SQL =
  "(EXISTS (SELECT 1 FROM credit_reversals r WHERE r.purchase_id = credit_purchases.purchase_id AND r.kind = 'refund' AND r.credits_clawed < credit_purchases.credits_granted) " +
  "AND NOT EXISTS (SELECT 1 FROM credit_ledger d WHERE d.ref = credit_purchases.purchase_id AND d.kind = 'reversal' AND d.reason = 'dispute'))";
function freezeStmt(db: DbLike, purchaseId: string, key: string, entryId: string, ts: string, chargeId: string | null = null): DbStatement {
  return db
    .prepare(
      "INSERT INTO credit_ledger (entry_id, account_id, kind, origin, credits, ref, idem, reason, at) " +
        `SELECT ?, account_id, 'reversal', 'refund_reversal', -(credits_granted - ${PARTIAL_REFUND_CLAWED_SQL}), purchase_id, ?, 'dispute_freeze', ? FROM credit_purchases ` +
        // D-1037(3) (RGM-03): a purchase a refund took back only IN PART (its one reversal row is kind 'refund' with
        // credits_clawed < credits_granted, and no lost dispute clawed it) is frozen for the REST, as a granted one is whole.
        `WHERE purchase_id = ? AND (status = 'granted' OR ${PARTIALLY_REFUNDED_SQL}) AND NOT EXISTS (SELECT 1 FROM credit_ledger WHERE idem = ?) ` +
        `AND NOT EXISTS (SELECT 1 FROM credit_ledger f WHERE ${OPEN_FREEZE_ROW_SQL})` +
        (chargeId !== null ? ` AND NOT ${CHAIN_KEY_RELEASED_SQL}` : ""),
    )
    .bind(entryId, freezeIdem(purchaseId, key), ts, purchaseId, freezeIdem(purchaseId, key), purchaseId, purchaseId, ...(chargeId !== null ? [key, chargeId, chargeId.length + 1, `${chargeId}+`] : []));
}
/** Release THIS dispute's freeze: the equal and opposite entry, read off the
 * freeze row itself; once only. */
function releaseStmt(db: DbLike, purchaseId: string, key: string, entryId: string, ts: string): DbStatement {
  return db
    .prepare(
      "INSERT INTO credit_ledger (entry_id, account_id, kind, origin, credits, ref, idem, reason, at) " +
        "SELECT ?, account_id, 'reversal', 'refund_reversal', -credits, ref, ?, 'dispute_unfreeze', ? FROM credit_ledger " +
        "WHERE idem = ? AND NOT EXISTS (SELECT 1 FROM credit_ledger WHERE idem = ?)",
    )
    .bind(entryId, unfreezeIdem(purchaseId, key), ts, freezeIdem(purchaseId, key), unfreezeIdem(purchaseId, key));
}
/** Release whichever freeze is open on the purchase (at most one — freezeStmt
 * refuses a second): every reversal, and a won with no dispute key. */
function releaseOpenStmt(db: DbLike, purchaseId: string, entryId: string, ts: string): DbStatement {
  return db
    .prepare(
      "INSERT INTO credit_ledger (entry_id, account_id, kind, origin, credits, ref, idem, reason, at) " +
        "SELECT ?, f.account_id, 'reversal', 'refund_reversal', -f.credits, f.ref, 'dispute_unfreeze' || substr(f.idem, 15), 'dispute_unfreeze', ? " +
        `FROM credit_ledger f WHERE ${OPEN_FREEZE_ROW_SQL} LIMIT 1`,
    )
    .bind(entryId, ts, purchaseId, purchaseId);
}

/** The credits an account holds frozen right now: the net of its open
 * freezes (a released freeze nets to zero). Read by `/account/war/credits`,
 * whose `credits` is already net of them. */
export async function frozenCredits(db: DbLike, accountId: string): Promise<number> {
  const row = await db
    .prepare(`SELECT COALESCE(SUM(credits), 0) AS n FROM credit_ledger WHERE account_id = ? AND reason IN ${FREEZE_REASONS_SQL}`)
    .bind(accountId)
    .first<{ n: number }>();
  return Math.max(0, -(row?.n ?? 0));
}

/**
 * I30-CONSOLE RGM-03-DISPLAY (D-1051(3): DISPLAY ONLY — it writes nothing and no freeze, ledger, guard or cash figure
 * reads it). When a dispute opens on a purchase and a partial refund then lands on it, the freeze keeps the whole grant
 * (D-1037(3), fail-closed) and the refund's claw lands too, so the refunded share is held twice and the balance reads
 * below its true figure until the dispute closes (-100 on a 500-credit purchase refunded 100). Per purchase with an OPEN
 * freeze (OPEN_FREEZE_ON_ACCOUNT_SQL) and a reversal row: what the freeze holds plus what the reversal took, past the
 * grant (frozen + clawed - granted), kept when above 0; 0 when the refund came first (the freeze then holds only the
 * rest). The row beside an open freeze is always a refund's: a LOST dispute releases the freeze in its own batch.
 * Served beside `frozen` on GET /account/war/credits as `held_twice` (index.ts; I30-INT).
 */
export interface RefundHeldTwice {
  total: number;
  purchases: { purchase_id: string; credits: number }[];
}
export async function refundHeldTwice(db: DbLike, accountId: string): Promise<RefundHeldTwice> {
  const rs = await db
    .prepare(
      "SELECT f.ref AS purchase_id, -f.credits + r.credits_clawed - p.credits_granted AS credits FROM credit_ledger f " +
        "JOIN credit_purchases p ON p.purchase_id = f.ref JOIN credit_reversals r ON r.purchase_id = f.ref " +
        `WHERE ${OPEN_FREEZE_ON_ACCOUNT_SQL} ORDER BY f.ref`,
    )
    .bind(accountId)
    .all<{ purchase_id: string; credits: number }>();
  const purchases = (rs.results ?? []).filter((x) => x.credits > 0);
  return { total: purchases.reduce((a, x) => a + x.credits, 0), purchases };
}

/** I28-Q6 (D-1031 6(b), for I29-OPS): an open freeze row `f` on the account `accountExpr` names (`?` or a column) —
 * `OPEN_FREEZE_ROW_SQL`'s release test (no `dispute_unfreeze` twin), keyed by the account instead of one purchase.
 * The ONE definition (I29-FX-INT, D-1037; FX-OPS H1, FX-ERASE H7): the release below, console.ts
 * freezeReleaseNamedGuardStmt and account_data.ts disputeOpenOnTombstoneSql interpolate it. */
export const OPEN_FREEZE_ON_ACCOUNT_FOR = (accountExpr: string): string =>
  `f.reason = 'dispute_freeze' AND f.account_id = ${accountExpr} AND NOT EXISTS (SELECT 1 FROM credit_ledger u WHERE u.idem = 'dispute_unfreeze' || substr(f.idem, 15))`;
/** `OPEN_FREEZE_ON_ACCOUNT_FOR` on a bound account (`?`). */
export const OPEN_FREEZE_ON_ACCOUNT_SQL = OPEN_FREEZE_ON_ACCOUNT_FOR("?");
export interface OpenDisputeFreeze {
  account_id: string;
  purchase_id: string;
  credits: number;
  /** The dispute the freeze is held under (`dispute_freeze:<purchase>:<key>`). */
  dispute_key: string;
  at: string;
}
/**
 * I28-Q6 (D-1031 6(b)): the dispute freezes open right now, oldest first — the support worklist behind I29-OPS's
 * `GET /ops/accounts/dispute-freezes`. At most `limit` (1-500, default 100) rows; `more` is true when rows exist past
 * the limit, so a caller never reads the page as the whole population.
 */
export async function listOpenDisputeFreezes(db: DbLike, limit = 100): Promise<{ rows: OpenDisputeFreeze[]; more: boolean }> {
  if (!Number.isInteger(limit) || limit < 1 || limit > 500) throw new RefundError("bad limit", 500);
  const n = limit;
  const rs = await db
    .prepare(
      "SELECT f.account_id, f.ref AS purchase_id, -f.credits AS credits, substr(f.idem, 17 + length(f.ref)) AS dispute_key, f.at FROM credit_ledger f " +
        "WHERE f.reason = 'dispute_freeze' AND NOT EXISTS (SELECT 1 FROM credit_ledger u WHERE u.idem = 'dispute_unfreeze' || substr(f.idem, 15)) " +
        "ORDER BY f.at, f.entry_id LIMIT ?",
    )
    .bind(n + 1)
    .all<OpenDisputeFreeze>();
  const rows = rs.results ?? [];
  return { rows: rows.slice(0, n), more: rows.length > n };
}
/**
 * I28-Q6 (D-1031 6(b)): an operator RELEASES every open dispute freeze on `accountId` — the statements I29-OPS batches
 * in its dual-control executor (action `dispute_freeze:release`). Built on `releaseOpenStmt`'s release (the equal and
 * opposite `dispute_unfreeze` row, idem `dispute_unfreeze` + the freeze's own suffix, so it lands once per freeze and a
 * later won/lost dispute finds the key released), over the account instead of one purchase. One
 * `account:dispute_freeze_released` audit row per released freeze, FIRST (it reads the open set the release then
 * closes): actor = the operator, item_id = the purchase (no account id to erase), detail = the purchase and credits.
 * The dispute itself is untouched: a later LOST close still reverses and bans.
 */
export function releaseDisputeFreezeStmts(db: DbLike, accountId: string, operator: string, ts: string): DbStatement[] {
  if (!ACCOUNT_RE.test(accountId)) throw new RefundError("bad account id", 500);
  if (typeof operator !== "string" || operator.length === 0 || operator.length > 200) throw new RefundError("bad operator", 500);
  if (typeof ts !== "string" || Number.isNaN(Date.parse(ts))) throw new RefundError("bad timestamp", 500);
  return [
    db
      .prepare(
        "INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) " +
          `SELECT ?, 'account:dispute_freeze_released', json_object('purchaseId', f.ref, 'credits', -f.credits), ?, f.ref, 'operator' FROM credit_ledger f WHERE ${OPEN_FREEZE_ON_ACCOUNT_SQL}`,
      )
      .bind(operator, ts, accountId),
    db
      .prepare(
        "INSERT INTO credit_ledger (entry_id, account_id, kind, origin, credits, ref, idem, reason, at) " +
          "SELECT 'cle_' || lower(hex(randomblob(10))), f.account_id, 'reversal', 'refund_reversal', -f.credits, f.ref, 'dispute_unfreeze' || substr(f.idem, 15), 'dispute_unfreeze', ? " +
          `FROM credit_ledger f WHERE ${OPEN_FREEZE_ON_ACCOUNT_SQL}`,
      )
      .bind(ts, accountId),
  ];
}

/**
 * D-1038(1) (I29-FX3): a fingerprint PROMOTED at erasure because a dispute was open (account_data.ts deleteAccount, the
 * D-1037(10)-extended INSERT) is deleted again when that dispute is WON, and kept only if it is lost. Rides the won's
 * batch AFTER its release, on the erased buyer's money tombstone. It deletes nothing while the tombstone shows a loss or
 * a possible one: any reversal (a lost close writes a 'dispute' one; a refund then a lost dispute promotes with no
 * 'dispute' row, pending-fingerprints.ts promotePendingFingerprintsOnLostDisputeStmt, so a refund row counts too -- and
 * `status = 'refunded'` is only ever written beside a reversal: credits.ts:567 and reverseOneGrant's status UPDATE), or another dispute still
 * open on the tombstone (an open freeze, OPEN_FREEZE_ON_ACCOUNT_FOR).
 */
function dropWonDisputeFingerprintsStmt(db: DbLike, purchase: PurchaseRow): DbStatement {
  return db
    .prepare(
      "DELETE FROM erased_dispute_fingerprints WHERE tombstone_id = ? " +
        "AND NOT EXISTS (SELECT 1 FROM credit_reversals r WHERE r.account_id = ?) " +
        `AND NOT EXISTS (SELECT 1 FROM credit_ledger f WHERE ${OPEN_FREEZE_ON_ACCOUNT_SQL})`,
    )
    .bind(purchase.account_id, purchase.account_id, purchase.account_id);
}

/**
 * Open -> freeze; won -> unfreeze, both keyed by the dispute. A won delivered
 * BEFORE its open writes that dispute's freeze and release together (net
 * zero), so the late open finds its key taken and freezes nothing. A won with
 * no dispute id releases whichever freeze is open. Nothing here bans or claws.
 */
async function freezeOrRelease(db: DbLike, purchase: PurchaseRow, input: ReverseInput & { kind: "dispute_open" | "dispute_won" }, mintEntryId: () => string, ts: string): Promise<ReverseResult> {
  const kind = input.kind;
  const disputeId = typeof input.disputeId === "string" && input.disputeId.length > 0 && input.disputeId.length <= 200 ? input.disputeId : null;
  const key = disputeId ?? input.processorEventId;
  const a = mintEntryId();
  const b = kind === "dispute_won" ? mintEntryId() : a;
  if (!CLE_RE.test(a) || !CLE_RE.test(b)) throw new RefundError("bad id shape", 500);
  const stmts: DbStatement[] = [];
  // RG-AM-01: a crypto link's open also asks whether its won already came on another link of the charge (the charge id is
  // the chain key up to its first "+", crypto-rail.ts cryptoGrantKey).
  const plus = purchase.external_txn_id.indexOf("+");
  const charge = purchase.rail === "crypto" ? (plus < 0 ? purchase.external_txn_id : purchase.external_txn_id.slice(0, plus)) : null;
  if (kind === "dispute_open") stmts.push(freezeStmt(db, purchase.purchase_id, key, a, ts, charge));
  else if (disputeId) stmts.push(freezeStmt(db, purchase.purchase_id, key, a, ts), releaseStmt(db, purchase.purchase_id, key, b, ts));
  else stmts.push(releaseOpenStmt(db, purchase.purchase_id, b, ts));
  if (kind === "dispute_won" && purchase.account_id.startsWith("del_")) stmts.push(dropWonDisputeFingerprintsStmt(db, purchase));
  await db.batch(stmts);
  const landed = await db.prepare("SELECT credits FROM credit_ledger WHERE entry_id = ?").bind(b).first<{ credits: number }>();
  if (!landed) return { reversed: false, reason: "duplicate" };
  return kind === "dispute_open"
    ? { reversed: false, reason: "frozen", credits_frozen: -landed.credits }
    : { reversed: false, reason: "unfrozen", credits_frozen: 0 };
}

/**
 * #640 (D-0758): the ban's audit row, naming the purchase whose dispute caused
 * it. Written in the ban's own batch and ONLY when this ban landed (`banned_at`
 * is now this `ts`), so the latest such row is the ban in force — the one
 * `liftChargebackBan` records. item_id = the purchase, as
 * `refund:dispute_after_ban_lifted` keys it: no account id to erase.
 */
function banRowStmt(db: DbLike, purchaseId: string, accountId: string, rail: string, ts: string): DbStatement {
  return db
    .prepare(
      "INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) " +
        "SELECT ?, 'account:chargeback_banned', ?, ?, ?, 'system' WHERE EXISTS (SELECT 1 FROM accounts WHERE account_id = ? AND banned_at = ?)",
    )
    .bind(`rail:${rail}`, JSON.stringify({ purchaseId }), ts, purchaseId, accountId, ts);
}

/**
 * D-0743 (RGA-598-01/02): the ban for a dispute on a purchase ALREADY reversed
 * (a redelivered lost close, a losing concurrent batch; an open never bans —
 * D-0753). If an
 * operator lifted the chargeback ban (`liftChargebackBan`) at or after this
 * purchase's reversal, AND that lift lifted a ban from THIS purchase (#640,
 * D-0756(1)/D-0758: the lift row's `detail.purchaseId`; a lift with no purchase
 * — every lift written before #640 — keeps the account-level meaning), the
 * appeal stands: no re-ban, and one
 * `refund:dispute_after_ban_lifted` row per purchase (item_id = purchase_id) for operator review
 * (NOT EXISTS: a redelivery adds none). Otherwise the idempotent ban, with its
 * ban row. Returns whether the account was (or stays) banned by this call.
 */
async function disputeBanOrReview(db: DbLike, purchase: PurchaseRow, input: ReverseInput, ts: string): Promise<boolean> {
  const lifted = await db
    .prepare(
      "SELECT 1 AS x FROM audit_log WHERE action = 'account:chargeback_ban_lifted' AND item_id = ? " +
        "AND created_at >= (SELECT MIN(at) FROM credit_reversals WHERE purchase_id = ?) " +
        "AND CASE WHEN NOT json_valid(detail) THEN 1 WHEN json_extract(detail, '$.purchaseId') IS NULL AND json_extract(detail, '$.paymentKey') IS NULL THEN 1 ELSE json_extract(detail, '$.purchaseId') = ? END LIMIT 1",
    )
    .bind(purchase.account_id, purchase.purchase_id, purchase.purchase_id)
    .first<{ x: number }>();
  if (!lifted) {
    // #640 (D-0810 (15), D-0814(4)): once an operator lift (any purchase)
    // followed this purchase's DISPUTE reversal, no later lost dispute on THIS
    // purchase bans again — the redelivered event or a distinct one. That
    // reversal's batch already banned or found the ban in force, and the lift
    // is the only production writer that clears it. The lift is purchase-scoped
    // (D-0758): a purchase with no dispute reversal before the lift still bans.
    const recorded = await db
      .prepare(
        "SELECT 1 AS x FROM credit_reversals r WHERE r.purchase_id = ? AND r.kind = 'dispute' " +
          "AND EXISTS (SELECT 1 FROM audit_log l WHERE l.action = 'account:chargeback_ban_lifted' AND l.item_id = ? AND l.created_at >= r.at)",
      )
      .bind(purchase.purchase_id, purchase.account_id)
      .first<{ x: number }>();
    if (recorded) return accountBanned(db, purchase.account_id);
    await db.batch([
      db.prepare("UPDATE accounts SET banned_at = ? WHERE account_id = ? AND banned_at IS NULL").bind(ts, purchase.account_id),
      banRowStmt(db, purchase.purchase_id, purchase.account_id, input.rail, ts),
    ]);
    return true;
  }
  // item_id = the purchase, as `refund:partial_credit_manual_review` keys it
  // (index.ts), so the row carries no account id past an erasure; NOT EXISTS
  // on (action, item_id) makes it one row per purchase.
  await db
    .prepare(
      "INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) " +
        "SELECT ?, 'refund:dispute_after_ban_lifted', ?, ?, ?, 'system' " +
        "WHERE NOT EXISTS (SELECT 1 FROM audit_log WHERE action = 'refund:dispute_after_ban_lifted' AND item_id = ?)",
    )
    .bind(
      `rail:${input.rail}`,
      JSON.stringify({ note: "dispute lost after ban lifted", purchaseId: purchase.purchase_id, processorEventId: input.processorEventId, paymentRef: input.paymentRef ?? null }),
      ts,
      purchase.purchase_id,
      purchase.purchase_id,
    )
    .run();
  return false;
}

/**
 * I17-FXA RGM-06/17 (D-0942(1), D-0938(1d) option A carried through): a LOST dispute (`kind: "dispute"`) for a payment
 * whose grant `credits.ts grantCredits` REFUSED (its `credit:grant_refused_dispute` row, which names the account)
 * applies the chargeback ban to that account. ONCE, by the rule `disputeBanOrReview` and the grant path follow: an
 * operator lift (`account:chargeback_ban_lifted`, item_id = the account) at or after this payment's FIRST lost-dispute
 * marker stands, so no redelivery re-bans; the ban row (item_id = the payment key, like the grant path's) is written
 * only when THIS ban landed. A refusal row naming no account (written before I17-FXA) bans nobody. A WON close is
 * untouched (D-0942(3), Jason's).
 */
async function banRefusedGrantAccount(db: DbLike, input: ReverseInput, ts: string): Promise<void> {
  const actor = `rail:${input.rail}`;
  const ref = input.paymentRef ?? null;
  const txn = input.externalTxnId ?? null;
  const refused = await db
    .prepare(
      "SELECT item_id, CASE WHEN json_valid(detail) THEN json_extract(detail, '$.accountId') END AS account_id FROM audit_log " +
        "WHERE action = ? AND actor_id = ? AND item_id IN (?, ?) ORDER BY audit_id LIMIT 1",
    )
    .bind(GRANT_REFUSED_DISPUTE_ACTION, actor, ref, txn)
    .first<{ item_id: string; account_id: unknown }>();
  if (!refused || typeof refused.account_id !== "string" || !ACCOUNT_RE.test(refused.account_id)) return;
  const accountId = refused.account_id;
  const liftStands =
    "EXISTS (SELECT 1 FROM audit_log l WHERE l.action = 'account:chargeback_ban_lifted' AND l.item_id = ? AND l.created_at >= " +
    "(SELECT MIN(m.created_at) FROM audit_log m WHERE m.action = ? AND m.actor_id = ? AND m.item_id IN (?, ?) " +
    "AND CASE WHEN json_valid(m.detail) THEN json_extract(m.detail, '$.kind') END = 'dispute'))";
  await db.batch([
    db.prepare(`UPDATE accounts SET banned_at = ? WHERE account_id = ? AND banned_at IS NULL AND NOT ${liftStands}`).bind(ts, accountId, accountId, UNRECONCILED_DISPUTE_ACTION, actor, ref, txn),
    db
      .prepare(
        "INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) " +
          "SELECT ?, 'account:chargeback_banned', ?, ?, ?, 'system' WHERE EXISTS (SELECT 1 FROM accounts WHERE account_id = ? AND banned_at = ?)",
      )
      .bind(actor, JSON.stringify({ purchaseId: null, paymentKey: refused.item_id, refusedGrant: true }), ts, refused.item_id, accountId, ts),
  ]);
}

/**
 * Reverse the credit grant for a refunded/disputed purchase. Fail-closed:
 * an unknown txn or an already-reversed purchase is a no-op (not an error), so a
 * redelivery or a foreign event is safely ignored.
 *
 * I26-W1-MONEY (JA-06): a CRYPTO charge settled in parts is granted as a key chain (crypto-rail.ts `cryptoGrantKey`:
 * the bare charge id, then `<charge>+<prior>`), and its refund/dispute arrives keyed by the charge id. Every grant of
 * that chain is reversed, each through the one-purchase path below (its own reversal row, claw and ban), and the
 * results are summed. A partial dispute's `disputedCents` is shared out by each link's gross (at least 1 cent, at most
 * the link's gross), so a link never reads a share of 0 as "the whole grant" (`clawFor`). Both passes (below) spend a
 * running cap of the disputed amount, so the chain never claws more than it (D-1027(1), D-1028(1)); a link whose capped
 * share is 0 is reversed with `clawNothing`.
 */
export async function reverseGrant(
  db: DbLike,
  input: ReverseInput,
  mintReversalId: () => string,
  mintEntryId: () => string,
  now: () => string = () => new Date().toISOString(),
  pubEnv?: AnalyticsTickEnv,
  pubCtx?: { waitUntil(p: Promise<unknown>): void },
): Promise<ReverseResult> {
  if (input.kind !== "refund" && input.kind !== "dispute" && input.kind !== "dispute_open" && input.kind !== "dispute_won") throw new RefundError("bad kind", 500);
  if (typeof input.processorEventId !== "string" || input.processorEventId.length === 0 || input.processorEventId.length > 200) throw new RefundError("bad event id", 500);
  // I21-Q70 / I21-Q86: a cents figure is a non-negative safe integer or absent — a caller bug is not read as "the whole grant".
  if (input.refundedCents !== undefined && !(Number.isSafeInteger(input.refundedCents) && input.refundedCents >= 0)) throw new RefundError("bad refunded cents", 500);
  if (input.cashRefundedCents !== undefined && input.cashRefundedCents !== null && !(Number.isSafeInteger(input.cashRefundedCents) && input.cashRefundedCents >= 0)) throw new RefundError("bad cash refunded cents", 500);
  const byPaymentRef = typeof input.paymentRef === "string" && input.paymentRef.length > 0;
  if (input.rail === "crypto" && !byPaymentRef && typeof input.externalTxnId === "string" && input.externalTxnId.length > 0) {
    const chain = await cryptoChainKeys(db, input.externalTxnId);
    // I28 JA-03 (#970) and I28 R1 (D-1025(8)): every kind is re-read after it is applied. A link that committed between
    // the read above and the per-link writes was treated by none of them (an open did not freeze it; a won saw it inherit
    // the still-open freeze; a refund or lost dispute did not reverse it). Only links the first pass did not target are
    // treated, so a dispute opened meanwhile on a link already treated is left alone. A link committing after the re-read
    // sees the result in its own batch: the reversal row (credits.ts CRYPTO_CHAIN_REVERSED_SQL: D-1030, I26-Q5, GRANTS it
    // and leaves one GRANTED_AFTER_REVERSAL_ACTION row) or the freeze state (CRYPTO_CHAIN_FREEZE_SQL).
    const results: ReverseResult[] = [];
    const shares = new Map<string, number | undefined>();
    // I21-Q70: a refund's refunded amount is shared out over the chain exactly as a lost dispute's disputed amount.
    const shared = input.kind === "dispute" || input.kind === "refund";
    const d = amountOf(input);
    const partialLost = shared && typeof d === "number" && Number.isInteger(d) && d > 0;
    const pass = async (links: { key: string; grossCents: number }[], budget: number | undefined, cap?: { left: number }) => {
      const gross = links.reduce((a, l) => a + l.grossCents, 0);
      const partial = shared && typeof budget === "number" && Number.isInteger(budget) && budget > 0 && budget < gross;
      for (const link of links) {
        let share = partial && link.grossCents > 0 ? Math.min(link.grossCents, Math.max(1, Math.round((budget * link.grossCents) / gross))) : budget;
        // D-1027(1): a capped (late) pass spends a running remainder; a link whose share it leaves at 0 takes nothing.
        if (cap && share !== undefined) {
          share = Math.min(share, cap.left);
          cap.left -= Math.min(share, link.grossCents);
        }
        shares.set(link.key, share);
        results.push(await reverseOneGrant(db, forLink(input, link.key, share), mintReversalId, mintEntryId, now, pubEnv, pubCtx, cap !== undefined && share === 0));
      }
    };
    if (chain.length > 1) await pass(chain, d, partialLost ? { left: d } : undefined); // D-1028(1): capped like the late pass
    else {
      shares.set(input.externalTxnId, d);
      results.push(await reverseOneGrant(db, input, mintReversalId, mintEntryId, now, pubEnv, pubCtx));
    }
    const reread = await cryptoChainKeys(db, input.externalTxnId);
    const late = reread.filter((l) => !shares.has(l.key));
    if (late.length > 0) {
      // A partial LOST dispute: the late links share what the links already reversed left of the disputed amount, as a
      // running cap, so the chain's take-back never exceeds the disputed amount (D-0745). D-1027(1) (I28-RJ-01 Q2): JA-06's
      // 1-cent floor applies only while something is left; a late link the cap leaves at 0 is still reversed (refunded,
      // reversal row, ban, freeze released) but claws nothing. Every other case passes the amount through unchanged (a
      // refund, no amount, or an amount covering the whole grant claws the link whole).
      const taken = partialLost ? reread.reduce((a, l) => (shares.has(l.key) ? a + Math.min(l.grossCents, shares.get(l.key) ?? l.grossCents) : a), 0) : 0;
      const left = partialLost ? Math.max(0, d - taken) : 0;
      await pass(late, partialLost ? left : d, partialLost ? { left } : undefined);
    }
    return sumReverseResults(results);
  }
  return reverseOneGrant(db, input, mintReversalId, mintEntryId, now, pubEnv, pubCtx);
}

/** JA-06: one result for a chain's per-link results — reversed if any link was, every amount summed, banned if any link banned. */
function sumReverseResults(rs: ReverseResult[]): ReverseResult {
  const reversed = rs.some((r) => r.reversed);
  const base = (reversed ? rs.find((r) => r.reversed) : rs.find((r) => r.reason !== "unknown")) ?? rs[0]!;
  const out: ReverseResult = { ...base, reversed };
  for (const k of ["credits_clawed", "credits_frozen", "sales_reversed_cents", "reserve_loss_cents", "dispute_debit_cents"] as const) {
    if (rs.some((r) => r[k] !== undefined)) out[k] = rs.reduce((a, r) => a + (r[k] ?? 0), 0);
  }
  if (rs.some((r) => r.banned !== undefined)) out.banned = rs.some((r) => r.banned === true);
  return out;
}

/** The one-purchase reversal (`reverseGrant` above validated `input`). */
async function reverseOneGrant(
  db: DbLike,
  input: ReverseInput,
  mintReversalId: () => string,
  mintEntryId: () => string,
  now: () => string,
  pubEnv?: AnalyticsTickEnv,
  pubCtx?: { waitUntil(p: Promise<unknown>): void },
  /** D-1027(1): a late chain link the disputed amount no longer covers — reversed, but nothing clawed. */
  clawNothing = false,
): Promise<ReverseResult> {

  // 1. Map the reversal to its purchase. Prefer the PaymentIntent (card rails);
  //    fall back to the external txn id (store/crypto rails).
  const findPurchase = async (): Promise<PurchaseRow | null> => {
    let found: PurchaseRow | null = null;
    if (typeof input.paymentRef === "string" && input.paymentRef.length > 0) {
      found = await db
        .prepare("SELECT purchase_id, account_id, credits_granted, gross_cents, external_txn_id, status, rail FROM credit_purchases WHERE payment_ref = ?")
        .bind(input.paymentRef)
        .first<PurchaseRow>();
    }
    if (!found && typeof input.externalTxnId === "string" && input.externalTxnId.length > 0) {
      found = await db
        .prepare("SELECT purchase_id, account_id, credits_granted, gross_cents, external_txn_id, status, rail FROM credit_purchases WHERE external_txn_id = ? AND rail = ?")
        .bind(input.externalTxnId, input.rail)
        .first<PurchaseRow>();
    }
    return found;
  };
  let purchase = await findPurchase();
  if (!purchase) {
    // I17-Q1 (D-1030 (a) "grant the credits"; I29-CASHLOTS H2): a WON dispute on a payment whose grant was REFUSED for
    // that dispute grants the refused credits now, once (credits.ts grantRefusedOnDisputeWin: the ordinary grant batch
    // with the dispute gate replaced by "the refusal row exists", and one GRANT_REFUSED_DISPUTE_RESOLVED_ACTION row; a
    // redelivered win finds the purchase and takes the freeze path, which grants nothing). With no refusal row a won
    // dispute on a payment we cannot find has nothing to unfreeze and nothing to reconcile: no trail row, as before.
    if (input.kind === "dispute_won") {
      const key = input.paymentRef || input.externalTxnId;
      if (!key) return { reversed: false, reason: "unknown" };
      // I29-LOOSE L4 (I29-REVERSALS won-before-grant): a won dispute on a payment whose OPEN marker is still unanswered
      // and whose grant has not arrived answers that marker with one DISPUTE_WON_BEFORE_GRANT_ACTION row (same actor_id
      // and item_id), which credits.ts DISPUTE_BEFORE_GRANT_SQL reads, so the grant that lands later is granted once.
      // Written BEFORE the refusal check below: a grant committing before this row was refused and is granted by
      // grantRefusedOnDisputeWin; one committing after it sees the row. NOT CAUGHT, as the marker it answers.
      const itemId = input.paymentRef ?? input.externalTxnId ?? null;
      // D-1037(2) (RGM-02): the row names the dispute id, and is ALSO written when no open marker exists yet but the won
      // names its dispute (a won delivered before its open; once per id, never for a subscription's PaymentIntent):
      // DISPUTE_BEFORE_GRANT_SQL answers a later open marker for the same id with it, so won, open, grant grants once.
      const did = disputeIdOf(input);
      await db
        .prepare(
          "INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, ?, ?, ?, ?, 'system' " +
            `WHERE ${DISPUTE_BEFORE_GRANT_SQL} ` +
            "OR (? IS NOT NULL AND NOT EXISTS (SELECT 1 FROM audit_log w WHERE w.action = ? AND w.actor_id = ? AND w.item_id IS ? " +
            "AND CASE WHEN json_valid(w.detail) THEN json_extract(w.detail, '$.disputeId') END = ?) " +
            "AND NOT EXISTS (SELECT 1 FROM billing_payments WHERE stripe_payment_intent_id = ?))",
        )
        .bind(
          `rail:${input.rail}`,
          DISPUTE_WON_BEFORE_GRANT_ACTION,
          JSON.stringify({ rail: input.rail, processorEventId: input.processorEventId, paymentRef: input.paymentRef ?? null, externalTxnId: input.externalTxnId ?? null, ...(did !== null ? { disputeId: did } : {}) }),
          now(),
          itemId,
          `rail:${input.rail}`,
          itemId,
          itemId,
          did,
          DISPUTE_WON_BEFORE_GRANT_ACTION,
          `rail:${input.rail}`,
          itemId,
          did,
          input.paymentRef ?? null,
        )
        .run();
      const g = await grantRefusedOnDisputeWin(db, input.rail, key, undefined, mintEntryId, now);
      if (g.reason === "no_refusal") return { reversed: false, reason: "unknown" };
      return g.granted ? { reversed: false, reason: "granted" } : { reversed: false, reason: "grant_refused", grant_refusal: g.reason ?? "refused" };
    }
    // S-040 AF-10: Stripe (and IAP/crypto) delivery order is not guaranteed —
    // a dispute/refund CAN be delivered before the checkout-completed webhook
    // that will insert this purchase's `credit_purchases` row. Answering
    // `unknown` used to be silent: the grant that lands afterward is never
    // clawed nor banned, and nothing durable said the race happened at all.
    // A `dispute` mapping to nothing is the highest-risk case (a REFUND with
    // no matching purchase is routinely just a subscription charge — see the
    // call sites), so it gets an operator-visible, queryable trail here.
    // D-0938(1d) (I17 J5b): `credits.ts grantCredits` now consults this row
    // inside its committing batch and refuses the grant (chargeback treatment).
    // I3-0355 — THE REFUND ARM GETS THE SAME TRAIL, UNDER ITS OWN ACTION.
    //
    // The paragraph above is still the reason the two are not equally urgent,
    // and the action word keeps them apart so a dispute alert cannot be
    // drowned by refund noise. What it does NOT justify is the refund arm
    // having no durable record at all: `reason: 'unknown'` is returned to a
    // webhook handler that acks and forgets, so before this the ONLY statement
    // that a refund arrived for a purchase we cannot find was the absence of
    // one. FAMILY-003: the early exit names the row a reader can find
    // afterwards, the STATE (the money moved at the processor and no local
    // grant was clawed) and the NEXT ACTOR (an operator reconciling this
    // processor event id by hand; for the DISPUTE arm, `credits.ts`'s grant
    // path refuses the later grant on this row — D-0938(1d)).
    {
      // BOUND, not interpolated. The value is a two-way compile-time literal
      // and could not be injected, but `action` is a column value like any
      // other and a built SQL string here would be the shape the fidelity
      // suite exists to stop appearing in this file at all.
      const isDispute = input.kind === "dispute" || input.kind === "dispute_open";
      const unreconciled = isDispute ? UNRECONCILED_DISPUTE_ACTION : "refund:unreconciled_refund";
      // NOT CAUGHT. D-0938(1d): the DISPUTE row is no longer a trail but the input of grantCredits' refusal — losing it
      // would let the later grant land, so a failure here fails loud (the webhook answers 500 and the processor
      // redelivers). I17-Q2 (D-1030; I29-CASHLOTS H5): the REFUND row is now the input of the same refusal (credits.ts
      // REFUND_BEFORE_GRANT_SQL), so it fails loud too; the refund arm's old catch-and-ack is gone.
      {
        // RGA-598-03 (D-0743): a Stripe dispute on a PaymentIntent the
        // subscription rail recorded (`billing_payments`, 0039) is that rail's
        // event, not an unreconciled credit purchase — no credit-rail alarm.
        const subscriptionOwned =
          isDispute && input.rail === "stripe" && typeof input.paymentRef === "string" && input.paymentRef.length > 0
            ? await db.prepare("SELECT 1 AS x FROM billing_payments WHERE stripe_payment_intent_id = ? LIMIT 1").bind(input.paymentRef).first<{ x: number }>()
            : null;
        if (!subscriptionOwned) {
          // D-0938(1d): `kind` is read by credits.ts grantCredits — a LOST dispute (`dispute`) bans the refused grant's
          // account, an open one (`dispute_open`) only refuses it (an open never bans, D-0753).
          // D-1037(1)/(2): a dispute marker names its dispute id (DISPUTE_BEFORE_GRANT_SQL matches a won row by it); a refund
          // marker carries the cumulative refunded and cash cents (credits.ts REFUND_BEFORE_GRANT_SQL refuses only a FULL one).
          const did = isDispute ? disputeIdOf(input) : null;
          const detail = JSON.stringify({
            rail: input.rail,
            kind: input.kind,
            processorEventId: input.processorEventId,
            paymentRef: input.paymentRef ?? null,
            externalTxnId: input.externalTxnId ?? null,
            ...(did !== null ? { disputeId: did } : {}),
            ...(!isDispute && typeof input.refundedCents === "number" ? { refundedCents: input.refundedCents } : {}),
            ...(!isDispute && typeof input.cashRefundedCents === "number" ? { cashRefundedCents: input.cashRefundedCents } : {}),
            // I30-CONSOLE REFUND-CURRENCY: the refund's currency, for credits.ts grantCredits partialRefundStmts to record.
            ...(refundCurrencyOf(input) !== null ? { currency: refundCurrencyOf(input) } : {}),
          });
          const itemId = input.paymentRef ?? input.externalTxnId ?? null;
          // NOT EXISTS on (action, item_id, detail): a redelivery of the same
          // processor event adds no second row.
          await db
            .prepare(
              "INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, ?, ?, ?, ?, 'system' " +
                "WHERE NOT EXISTS (SELECT 1 FROM audit_log WHERE action = ? AND item_id IS ? AND detail = ?)",
            )
            .bind(`rail:${input.rail}`, unreconciled, detail, now(), itemId, unreconciled, itemId, detail)
            .run();
        }
      }
      // D-0938(1d): CLOSE THE READ-THEN-MARK WINDOW. A grant that committed between the lookup above and the marker
      // has already passed its in-batch gate; re-read, and if it is there now, this dispute (or refund, I29-CASHLOTS
      // H5) takes the ordinary path. A grant committing after the marker is refused by that gate, so one of the two
      // always sees the other.
      purchase = await findPurchase();
      // I17-FXA RGM-06 (D-0942(1)): a LOST close for a payment whose grant was refused on an earlier dispute bans the
      // account the refusal row names — the ban this arm applies to a granted purchase.
      if (!purchase && input.kind === "dispute") await banRefusedGrantAccount(db, input, now());
    }
    if (!purchase) return { reversed: false, reason: "unknown" };
  }
  if (!ACCOUNT_RE.test(purchase.account_id)) throw new RefundError("bad account id on purchase", 500);
  if (input.kind === "dispute_open" || input.kind === "dispute_won") return freezeOrRelease(db, purchase, { ...input, kind: input.kind }, mintEntryId, now());
  // I21-Q70 (D-1030 (a)): a purchase a refund took back only IN PART is taken back further by a later refund or a lost
  // dispute, never past its grant — the same batch below, as a TOP-UP of the purchase's reversal row (`partialTopUp`).
  const top = purchase.status !== "granted" && (input.kind === "refund" || input.kind === "dispute") && !clawNothing ? await partialTopUp(db, purchase, input) : null;
  if (purchase.status !== "granted" && !top) {
    // Review F1: a DISPUTE that lands after a REFUND already clawed the same
    // purchase must STILL ban — a refund never bans, and the ban is the whole
    // fraud deterrent. The claw is a no-op (already reversed), but the ban is
    // applied idempotently (banned_at IS NULL). Order-independent: refund-then-
    // dispute and dispute-then-refund both end banned.
    // I21-Q70: unless a refund took it back only IN PART (`partialTopUp`, the batch below).
    if (input.kind === "dispute") {
      // D-1037(10) (I20-Q4, JN-01 REV-I20Q4): refund then chargeback on an erased buyer's tombstone keeps the fingerprints.
      if (purchase.account_id.startsWith("del_")) await promotePendingFingerprintsOnLostDisputeStmt(db, purchase.account_id).run();
      const banned = await disputeBanOrReview(db, purchase, input, now());
      return { reversed: false, reason: "duplicate", banned };
    }
    return { reversed: false, reason: "duplicate" };
  }

  const reversalId = mintReversalId();
  const entryId = mintEntryId();
  if (!CRV_RE.test(reversalId) || !CLE_RE.test(entryId)) throw new RefundError("bad id shape", 500);

  // A top-up's reversalId is the TEMPORARY id its sale walk marks; `topUpStmts` renames it to the row's own at the end.
  const clawCredits = top ? top.delta : clawNothing ? 0 : clawFor(input.kind, purchase.credits_granted, purchase.gross_cents, input.disputedCents, input.refundedCents);
  const releaseId = mintEntryId();
  if (!CLE_RE.test(releaseId)) throw new RefundError("bad id shape", 500);
  const ts = now();
  const banning = input.kind === "dispute";
  const cash = cashColumn(input);
  const cur = currencyColumn(input);

  // 2. The atomic reversal batch — EVERYTHING inside it, including which sales
  //    to reverse (F2 review, Concurrency: a select-then-batch let a tip land,
  //    or a concurrent reversal mark the same sale, between the two).
  //
  //    WHICH SALES. Within-hold, un-reversed, oldest first, while their
  //    CUMULATIVE gross backing fits the reversal BUDGET:
  //
  //        budget = max(0, this claw − wallet balance still held) x rate
  //               + Σ reserve_loss_cents of earlier reversals on this account
  //
  //    I3-0322/I3-1079 (2026-09-19): this used to be
  //    `thisClawBackingCents + Σ credits_clawed(priors) x rate − Σ reversed`,
  //    the RAW clawed amount, uncorrected for money the claw never actually
  //    took from a creator because it was still sitting unspent in the fan's
  //    own wallet. A claw fully covered by the wallet balance now nets to a
  //    zero contribution — the wallet itself already "recovers" it, no sale
  //    needs to be touched. `reserve_loss_cents` (below) already IS this
  //    per-event shortfall net of that same event's own recovered sales, so
  //    summing it over priors carries the multi-bucket property forward
  //    without re-litigating history: an unrecovered shortfall from an
  //    earlier claw is still owed and stays in the running budget until some
  //    later batch's sale pool actually recovers it (a stale reserve_loss row
  //    that already got swept up by a later batch inflates a THIRD claim's
  //    budget harmlessly — the sale it would have reversed is already
  //    reversed=1 and out of the pool, so nothing double-claws).
  //
  //    Never past the budget (a sale larger than what is left is not reversed —
  //    credits are fungible, the oldest within-hold sale may belong to an
  //    innocent creator the disputed money never funded, and over-reversing
  //    would penalise them beyond the chargeback). The budget still CARRIES: a
  //    $50 tip funded by two $25 purchases that are BOTH disputed, wallet
  //    empty throughout, is not reversed on the first dispute (5000 > 2500)
  //    but is on the second (5000 ≤ 2500 + 2500), so multi-bucket fraud no
  //    longer leaves the creator paid out of Dhwaja's reserve (F2 review,
  //    Economic Abuse). What still cannot be recovered is reserve loss,
  //    recorded on the row.
  //
  //    The UNIQUE(purchase_id) on credit_reversals is the idempotency marker:
  //    a redelivery's insert throws, the batch rolls back — the sale UPDATE
  //    with it — and nothing double-claws.
  // BF-12 — RESOLVE THE FINANCIAL IDENTITY IN THE BATCH, THROUGH THE IMMUTABLE
  // PURCHASE ID.
  //
  // Every statement below used to bind `purchase.account_id` — a string read
  // BEFORE this transaction opened. `deleteAccount` moves `credit_purchases`,
  // `credit_reversals` and `sales` of an erased fan onto ONE shared money
  // tombstone, and that tombstone exists precisely so the purchase -> sales walk
  // this function performs still works afterwards (`account_data.ts`: per-row
  // tombstones "would cut that walk and let 'tip, then erase, then charge back'
  // pay the creator out of Dhwaja's reserve").
  //
  // An erasure committing between that read and this batch therefore did the
  // walk under an identifier nothing carries any more: no sale matched, eligible
  // creator earnings stayed unreversed — while `credit_purchases.status` WAS set
  // to 'refunded' by the stable purchase id, so a redelivery finds the work
  // already done and the recovery can never be repaired. The reversal record and
  // the claw were also INSERTED under the erased id, putting back into the money
  // tables exactly the identifier the erasure had removed.
  //
  // `CURRENT_OWNER` reads that identity from the purchase row itself, inside the
  // same transaction as the writes, so both orderings — erase-then-reverse and
  // reverse-during-erase — produce the same recovered sales and one consistently
  // anonymised financial trail. The purchase id is immutable and is the PK, so
  // this is always exactly one row; the `status = 'refunded'` update is the LAST
  // statement, so every earlier statement still reads the row it names.
  // (`CURRENT_OWNER` is module-scope, after `CLE_RE`, so `topUpStmts` reads the same identity.)
  // RG-634-01: "wallet balance still held" EXCLUDES dispute freeze rows. A
  // freeze on another purchase is money still in the wallet, not money gone to
  // a creator; counting it inflated this budget and reversed an innocent
  // creator's in-hold sale. Spend gates, by contrast, SHOULD see the freeze.
  // (`WALLET_HELD_SQL` is module-scope beside `CURRENT_OWNER`.)
  const owner = purchase.purchase_id;

  // D-0641 U1 — MOST RECENT FIRST. The walk below is newest-first in BOTH
  // places or in neither: the candidate-pool `ORDER BY at DESC, sale_id DESC
  // LIMIT ?` decides which sales the cap admits, and the `SUM(...) OVER (ORDER
  // BY at DESC, sale_id DESC ...)` decides the order `cum` spends the budget.
  // Flipping one and not the other re-makes the I3-1604 disagreement recorded
  // under this statement. One test per site, each reddening alone:
  // test/b1504-newest-first.harness.test.ts NEWEST_FIRST_WINDOW / POOL_CAP_NEWEST.
  //
  // D-0641 U2 — `clears_at > ?` IS STILL THE BOUNDARY OF THE SALE UPDATE: a
  // cleared sale is never marked reversed. What happens to the shortfall past it
  // depends on the kind (D-0960(1), Jason 2026-10-11, option B):
  //   - a REFUND (our choice to refund): the remainder is written down as
  //     `credit_reversals.reserve_loss_cents` and Dhwaja absorbs it, as before.
  //   - a LOST DISPUTE: the remainder is attributed to the account's CLEARED sales
  //     by the same newest-first walk (D-0616), bounded to sales spent at or after
  //     the disputed purchase and to THIS dispute's own shortfall, and recorded one
  //     row per sale in `creator_dispute_debits` IN THIS BATCH — the record that
  //     names each creator and the reason before a cent moves (D-0616). No balance
  //     is touched here: `creator-payout.ts` nets the debits at the creator's next
  //     payout and CARRIES what the payout cannot cover (`owed_cents`), so an
  //     over-claw is visible, not floored away (QUEUE #490). Only what no cleared
  //     sale covers stays `reserve_loss_cents`.
  // Defended by CLEARED_MONEY_UNREACHABLE / CLEARED_REFUND_ABSORBED in
  // test/b1504-newest-first.harness.test.ts and test/i20-w1a-dispute-netting.test.ts.
  // ONE ANALYTICS ROW PER REVERSED SALE (ADR-0006 §2 multi-row statements),
  // read from the marks the UPDATE above just made — the same
  // `reversed_by = ? AND reversed = 1` the reversal record's totals use below, so
  // the events and the totals are the same set of sales by construction.
  // The count is bounded by `MAX_REVERSAL_SALES`. Creator-scoped and
  // amount-bearing; `clears_day` is the day of the sale's OWN `clears_at`,
  // which is the cell the scheduled-clear inverse decrements (ADR §3).
  //
  // THE INVERSE CARRIES THE SAME THREE FIGURES THE SALE DID (S-12): the gross
  // as the amount, net and cut as dimensions, read off the very row being
  // reversed. Each inverse then lands on the series it inverts —
  // `reversals_gross_cents` against `gross_credits`, `net_cents` against
  // `scheduled_clear_cents` — instead of one number standing in for both. The
  // dims are built by concatenation rather than `json_object()` for the
  // reason the sale's own `clears_day` already was: no JSON1 dependency in a
  // statement that runs inside the money batch. Keys are in canonical
  // (sorted) order, so these bytes match `canonicalDims()`'s.
  const saleEmit = db
    .prepare(
      outboxSelectSql("sale_reversed", {
        table: "sales",
        where: "reversed_by = ?",
        // The landed flip (INV-051 U-1): a row carrying this id that the UPDATE did not reverse is not an event.
        extraWhere: "reversed = 1",
        amountExpr: "gross_backing_cents",
        dimsExpr:
          `'{"clears_day":"' || substr(clears_at, 1, 10) || '","net_cents":' || creator_net_cents || ',"platform_cut_cents":' || platform_cut_cents || '}'`,
      }),
    )
    .bind(ts, reversalId);
  const stmts: DbStatement[] = top ? topUpStmts(db, purchase, input, top, reversalId, entryId, ts, saleEmit, releaseId) : [
    // D-0744: release the purchase's open freeze (whichever dispute holds it)
    // in the same batch, so the claw is not stacked on the freeze. The budget
    // below ignores freeze rows either way (WALLET_HELD_SQL). D-1037(3): NOT by a
    // refund that leaves part of the grant — the dispute still holds the rest
    // (the refunded share stays held too until the dispute closes, fail-closed).
    ...(banning || clawCredits >= purchase.credits_granted ? [releaseOpenStmt(db, purchase.purchase_id, releaseId, ts)] : []),
    saleWalkStmt(db, reversalId, owner, ts, clawCredits, input.kind === "dispute"),
    saleEmit,
    // D-0960(1): a LOST dispute's past-hold remainder -> `creator_dispute_debits`
    // (see the D-0641 U2 note above). Placed after the sale UPDATE (its budget
    // subtracts what that UPDATE reversed) and before the claw (WALLET_HELD_SQL
    // reads the balance before it) and before the reversal record (whose
    // reserve_loss_cents subtracts what this statement debited).
    ...(banning ? disputeDebitStmts(db, reversalId, owner, purchase.purchase_id, clawCredits, ts, CURRENT_OWNER, WALLET_HELD_SQL, input.rail) : []),
    // The reversal record, its two totals DERIVED from what the UPDATE marked:
    // `reversed_by = ? AND reversed = 1`, the emit's own landed flip (S-090 #313a).
    // Reserve loss (telemetry, not load-bearing): the charged-back money that
    // was already SPENT beyond the fan's other credits (so the claw pushes the
    // balance negative) and could NOT be recovered by a sale reversal. Unspent
    // clawed credits are not a loss (the claw recovers them); past-hold spends
    // that funded already-cleared creator earnings are. Reads the balance
    // BEFORE the claw below lands.
    db
      .prepare(
        `INSERT INTO credit_reversals (reversal_id, account_id, purchase_id, rail, external_txn_id, kind, credits_clawed, sales_reversed_cents, reserve_loss_cents, processor_event_id, at${cash.col}${cur.col})
         SELECT ?, ${CURRENT_OWNER}, ?, ?, ?, ?, ?,
                (SELECT COALESCE(SUM(creator_net_cents),0) FROM sales WHERE reversed_by = ? AND reversed = 1),
                MAX(0, MAX(0, ? - MAX(0, ${WALLET_HELD_SQL})) * ?
                       - (SELECT COALESCE(SUM(gross_backing_cents),0) FROM sales WHERE reversed_by = ? AND reversed = 1)
                       - (SELECT COALESCE(SUM(gross_backing_cents),0) FROM creator_dispute_debits WHERE reversal_id = ?)),
                ?, ?${cash.val}${cur.val}`,
      )
      .bind(reversalId, owner, purchase.purchase_id, input.rail, purchase.external_txn_id, input.kind, clawCredits, reversalId, clawCredits, owner, CREDIT_BACKING_CENTS, reversalId, reversalId, input.processorEventId, ts, ...cash.bind, ...cur.bind),
    // I20-Q4 (D-1030 (b); I29-FANERASE H1): a LOST dispute on an erased buyer's money tombstone keeps the buyer's pending
    // fingerprints (account_data.ts promotePendingFingerprintsStmt, self-guarded on this 'dispute' row, so it sits AFTER
    // the INSERT above). From the leaf pending-fingerprints.ts (I29-REG3 H3: account_data.ts -> closures.ts -> this file).
    ...(banning && purchase.account_id.startsWith("del_") ? [promotePendingFingerprintsStmt(db, purchase.account_id)] : []),
    // Claw `clawCredits` (the whole grant, or D-0745's share of it). Balance
    // may go negative → future spends refuse. idem = the purchase, so the
    // journal refuses a second claw of one grant on its own, beside
    // `credit_reversals.purchase_id UNIQUE`. A share of 0 writes no row
    // (CHECK credits <> 0); the reversal record and the ban still land.
    ...(clawCredits > 0
      ? [
          db
            .prepare(`INSERT INTO credit_ledger (entry_id, account_id, kind, origin, credits, ref, idem, reason, at) SELECT ?, ${CURRENT_OWNER}, 'reversal', 'refund_reversal', ?, ?, ?, ?, ?`)
            .bind(entryId, owner, -clawCredits, purchase.purchase_id, `reversal:${purchase.purchase_id}`, input.kind, ts),
        ]
      : []),
    // Mark the purchase refunded (also blocks a second reversal that somehow
    // slipped the UNIQUE — belt and suspenders).
    db.prepare("UPDATE credit_purchases SET status = 'refunded' WHERE purchase_id = ? AND status = 'granted'").bind(purchase.purchase_id),
  ];
  if (banning && !top) {
    // Dispute → ban (a top-up's ban is `disputeBanOrReview`'s, after the batch). First dispute wins the timestamp (idempotent under a later
    // refund/dispute on another purchase). After erasure there is no accounts
    // row to ban — the erased account cannot spend anyway.
    stmts.push(
      db.prepare("UPDATE accounts SET banned_at = ? WHERE account_id = ? AND banned_at IS NULL").bind(ts, purchase.account_id),
      banRowStmt(db, purchase.purchase_id, purchase.account_id, input.rail, ts),
    );
  }

  let seqs: number[] = [];
  try {
    const span = outboxSpan(db);
    const spanRes = await db.batch([span.open, ...stmts, span.close]);
    seqs = outboxSpanSeqs(spanRes[0], spanRes[spanRes.length - 1]);
  } catch (e) {
    // UNIQUE(purchase_id): a concurrent/earlier delivery already reversed
    // this purchase and won the race; this batch (INSERT INTO credit_reversals
    // among it) rolled back whole. I3-0344: that must NOT swallow a dispute's
    // separate ban obligation — the sequential case (`purchase.status !==
    // 'granted'` above) already bans here even though its own claw is a
    // no-op, and a losing concurrent batch is the same situation, just
    // discovered later (inside the transaction instead of before it). Apply
    // the same idempotent ban a dispute always gets.
    const msg = e instanceof Error ? e.message : String(e);
    if (/unique/i.test(msg)) {
      if (banning) {
        const banned = await disputeBanOrReview(db, purchase, input, ts);
        return { reversed: false, reason: "duplicate", banned };
      }
      return { reversed: false, reason: "duplicate" };
    }
    throw e;
  }

  if (pubEnv && pubCtx) publishAfterCommit(pubEnv, pubCtx, seqs);
  if (top) {
    const after = await db
      .prepare("SELECT sales_reversed_cents, reserve_loss_cents, (SELECT COALESCE(SUM(amount_cents),0) FROM creator_dispute_debits WHERE reversal_id = ? AND at = ?) AS debit FROM credit_reversals WHERE reversal_id = ?")
      .bind(top.reversalId, ts, top.reversalId)
      .first<{ sales_reversed_cents: number; reserve_loss_cents: number; debit: number }>();
    const up: ReverseResult = {
      reversed: true,
      credits_clawed: clawCredits,
      sales_reversed_cents: (after?.sales_reversed_cents ?? top.salesReversedCents) - top.salesReversedCents,
      reserve_loss_cents: (after?.reserve_loss_cents ?? top.reserveLossCents) - top.reserveLossCents,
      dispute_debit_cents: banning ? (after?.debit ?? 0) : 0,
    };
    return banning ? { ...up, banned: await disputeBanOrReview(db, purchase, input, ts) } : { ...up, banned: false };
  }
  const row = await db
    .prepare("SELECT sales_reversed_cents, reserve_loss_cents, (SELECT COALESCE(SUM(amount_cents),0) FROM creator_dispute_debits WHERE reversal_id = ?) AS debit FROM credit_reversals WHERE reversal_id = ?")
    .bind(reversalId, reversalId)
    .first<{ sales_reversed_cents: number; reserve_loss_cents: number; debit: number }>();
  return {
    reversed: true,
    credits_clawed: clawCredits,
    sales_reversed_cents: row?.sales_reversed_cents ?? 0,
    reserve_loss_cents: row?.reserve_loss_cents ?? 0,
    dispute_debit_cents: row?.debit ?? 0,
    banned: banning,
  };
}

/** I21-Q70: a top-up of a purchase's partial-refund reversal row — the row read and what this event takes on top. */
interface TopUp {
  reversalId: string;
  clawed: number;
  delta: number;
  salesReversedCents: number;
  reserveLossCents: number;
}

/**
 * I21-Q70 (D-1030 (a)): CUMULATIVE PARTIAL REFUNDS. A purchase whose reversal so far is a REFUND that took back only
 * part of the grant (`credit_reversals.kind = 'refund'`, credits_clawed < credits_granted) is taken back further, and
 * never past its grant:
 *   - a later `refund` carries the charge's cumulative refunded cents, so it takes clawFor(refund, cumulative) less what
 *     is already taken; a redelivery, or a figure no larger than the last, takes nothing;
 *   - a LOST `dispute` takes its own share (clawFor(dispute)) of what is left of the grant, once per purchase (a
 *     purchase whose journal already carries a dispute claw takes nothing more on a later dispute).
 * Null when there is nothing to take. The batch is `topUpStmts`; the ban stays `disputeBanOrReview`'s.
 */
async function partialTopUp(db: DbLike, purchase: PurchaseRow, input: ReverseInput): Promise<TopUp | null> {
  const prior = await db
    .prepare("SELECT reversal_id, credits_clawed, sales_reversed_cents, reserve_loss_cents FROM credit_reversals WHERE purchase_id = ? AND kind = 'refund'")
    .bind(purchase.purchase_id)
    .first<{ reversal_id: string; credits_clawed: number; sales_reversed_cents: number; reserve_loss_cents: number }>();
  if (!prior) return null;
  const isDispute = input.kind === "dispute";
  if (isDispute && (await db.prepare("SELECT 1 AS x FROM credit_ledger WHERE ref = ? AND kind = 'reversal' AND reason = 'dispute' LIMIT 1").bind(purchase.purchase_id).first()) !== null) return null;
  const delta = isDispute
    ? Math.min(purchase.credits_granted - prior.credits_clawed, clawFor("dispute", purchase.credits_granted, purchase.gross_cents, input.disputedCents))
    : clawFor("refund", purchase.credits_granted, purchase.gross_cents, undefined, input.refundedCents) - prior.credits_clawed;
  if (delta <= 0) return null;
  return { reversalId: prior.reversal_id, clawed: prior.credits_clawed, delta, salesReversedCents: prior.sales_reversed_cents, reserveLossCents: prior.reserve_loss_cents };
}

/**
 * I21-Q70: a top-up's batch — the reversal batch's statements under the purchase's EXISTING reversal row: the in-hold
 * sale walk and its `sale_reversed` rows (`saleEmit`, built by the caller) under the TEMPORARY id `walkId`, so only
 * THIS top-up's sales are emitted and summed; a lost dispute's cleared-sale debits under the existing id; the row's
 * figures (credits_clawed, + recovered sales, + reserve loss, the cash figure when given); the claw; and `walkId`
 * renamed to the existing id in `sales`. The claw's idem is `reversal:<purchase>:<credits_clawed read>`, so of two
 * top-ups that read the same row only one commits: the other batch rolls back whole on the UNIQUE idem (`duplicate`).
 */
function topUpStmts(db: DbLike, purchase: PurchaseRow, input: ReverseInput, top: TopUp, walkId: string, entryId: string, ts: string, saleEmit: DbStatement, releaseId: string): DbStatement[] {
  const owner = purchase.purchase_id;
  const cash = cashColumn(input);
  const cur = currencyColumn(input);
  const lost = input.kind === "dispute";
  return [
    // D-1037(3): the freeze a dispute opened on the rest of a partially refunded purchase is released by its LOST close
    // (or by a refund that takes the rest), in this batch, so the top-up claw is not stacked on it.
    ...(lost || top.clawed + top.delta >= purchase.credits_granted ? [releaseOpenStmt(db, purchase.purchase_id, releaseId, ts)] : []),
    // D-1037(10) (I20-Q4): a lost dispute on a purchase a refund reversed in part keeps an erased buyer's fingerprints.
    ...(lost && purchase.account_id.startsWith("del_") ? [promotePendingFingerprintsOnLostDisputeStmt(db, purchase.account_id)] : []),
    saleWalkStmt(db, walkId, owner, ts, top.delta, input.kind === "dispute"),
    saleEmit,
    ...(input.kind === "dispute" ? disputeDebitStmts(db, top.reversalId, owner, purchase.purchase_id, top.delta, ts, CURRENT_OWNER, WALLET_HELD_SQL, input.rail, walkId) : []),
    db
      .prepare(
        `UPDATE credit_reversals SET credits_clawed = ?,
                sales_reversed_cents = sales_reversed_cents + (SELECT COALESCE(SUM(creator_net_cents),0) FROM sales WHERE reversed_by = ? AND reversed = 1),
                reserve_loss_cents = reserve_loss_cents + MAX(0, MAX(0, ? - MAX(0, ${WALLET_HELD_SQL})) * ?
                       - (SELECT COALESCE(SUM(gross_backing_cents),0) FROM sales WHERE reversed_by = ? AND reversed = 1)
                       - (SELECT COALESCE(SUM(gross_backing_cents),0) FROM creator_dispute_debits WHERE reversal_id = ? AND at = ?))${cash.set}${cur.set}
          WHERE reversal_id = ?`,
      )
      .bind(top.clawed + top.delta, walkId, top.delta, owner, CREDIT_BACKING_CENTS, walkId, top.reversalId, ts, ...cash.bind, ...cur.bind, top.reversalId),
    db
      .prepare(`INSERT INTO credit_ledger (entry_id, account_id, kind, origin, credits, ref, idem, reason, at) SELECT ?, ${CURRENT_OWNER}, 'reversal', 'refund_reversal', ?, ?, ?, ?, ?`)
      .bind(entryId, owner, -top.delta, purchase.purchase_id, `reversal:${purchase.purchase_id}:${top.clawed}`, input.kind, ts),
    db.prepare("UPDATE sales SET reversed_by = ? WHERE reversed_by = ?").bind(top.reversalId, walkId),
  ];
}

/** The reversal batch's in-hold sale walk (D-0641 U1/U2, I3-1604; the comments at its use in `reverseOneGrant`), marking
 * the sales it takes `reversed_by = reversedBy`. `owner` is the purchase id `CURRENT_OWNER` reads the account through.
 * I20-Q1 (D-1030 (a) "only from creators the disputed purchase's credits went to"; I29-LOOSE L5): with `lotScoped` (a
 * LOST dispute) a sale is walked only when its lot edge names THIS purchase or it has no edge at all (spent before 0221:
 * today's walk) — `lotSalesForPurchaseSql`, the predicate `disputeDebitStmt` asks. A refund walks as before. */
function saleWalkStmt(db: DbLike, reversedBy: string, owner: string, ts: string, clawCredits: number, lotScoped: boolean): DbStatement {
  const lot = lotScoped ? ` AND ${lotSalesForPurchaseSql("sales")}` : "";
  return db
    .prepare(
      `UPDATE sales SET reversed = 1, reversed_by = ?
        WHERE reversed = 0 AND account_id = ${CURRENT_OWNER} AND sale_id IN (
          SELECT sale_id FROM (
            SELECT sale_id, SUM(gross_backing_cents) OVER (ORDER BY at DESC, sale_id DESC ROWS UNBOUNDED PRECEDING) AS cum
              FROM (SELECT sale_id, at, gross_backing_cents FROM sales
                     WHERE account_id = ${CURRENT_OWNER} AND reversed = 0 AND clears_at > ?${lot}
                       AND gross_backing_cents <= MAX(0, ? - MAX(0, ${WALLET_HELD_SQL})) * ?
                          + (SELECT COALESCE(SUM(reserve_loss_cents),0) FROM credit_reversals WHERE account_id = ${CURRENT_OWNER})
                     ORDER BY at DESC, sale_id DESC LIMIT ?))
           WHERE cum <= MAX(0, ? - MAX(0, ${WALLET_HELD_SQL})) * ?
                          + (SELECT COALESCE(SUM(reserve_loss_cents),0) FROM credit_reversals WHERE account_id = ${CURRENT_OWNER}))`,
    )
    // I3-1604: a sale whose OWN gross backing already exceeds the budget is
    // excluded from the candidate pool BEFORE the running cumulative sum is
    // taken — the window function sums ALL preceding rows regardless of
    // whether they individually qualified, so one such row used to poison
    // `cum` for every smaller, in-budget sale behind it and starve them
    // permanently. Excluding it up front (same budget expression, applied
    // once as a row filter and once as the running-total bound) removes the
    // poison without changing which sales are selected when no such row
    // exists.
    .bind(reversedBy, owner, owner, ts, ...(lotScoped ? [owner] : []), clawCredits, owner, CREDIT_BACKING_CENTS, owner, MAX_REVERSAL_SALES, clawCredits, owner, CREDIT_BACKING_CENTS, owner);
}

/**
 * QUEUE #490/c: a reversal's reconciliation is BROKEN when its purchase row cannot be read, or a sale of its past-hold
 * population (`account`'s sales spent at or after the purchase and cleared by `at`) was reversed by something other
 * than this reversal — the two limbs of admin.ts `reversalAttribution`'s `reconciliation`. Binds: `brokenBinds`, where
 * `account` is `CURRENT_OWNER` (one `?`, the purchase id it reads the account through).
 */
export function reconciliationBrokenSql(account: string): string {
  return (
    "((SELECT at FROM credit_purchases WHERE purchase_id = ?) IS NULL OR EXISTS (SELECT 1 FROM sales m WHERE m.account_id = " +
    account +
    " AND m.at >= (SELECT at FROM credit_purchases WHERE purchase_id = ?) AND m.clears_at <= ? AND m.reversed = 1 AND (m.reversed_by IS NULL OR m.reversed_by <> ?)))"
  );
}

/** The binds of `reconciliationBrokenSql(CURRENT_OWNER)`. `reversalId` is the reversal ROW's id (a top-up's existing
 * row): sales this batch's in-hold walk marks all clear after `at`, so they are outside the population whatever id
 * marked them. */
function brokenBinds(owner: string, purchaseId: string, at: string, reversalId: string): string[] {
  return [purchaseId, owner, purchaseId, at, reversalId];
}

/** QUEUE #490/c: the audit action a lost dispute writes INSTEAD of its cleared-sale debits when its reconciliation is broken. */
export const DISPUTE_DEBIT_WITHHELD_ACTION = "refund:dispute_debit_withheld_unreconciled";

/**
 * QUEUE #490/c: the lost dispute's cleared-sale debit walk (`disputeDebitStmt`), refused when the reversal's
 * reconciliation is broken (`reconciliationBrokenSql`): no `creator_dispute_debits` row, and one
 * `DISPUTE_DEBIT_WITHHELD_ACTION` audit row (item_id = the purchase, no account id) an operator reads instead; the
 * remainder stays `reserve_loss_cents`. The alarm is written first, so both read the state the sale walk left.
 */
function disputeDebitStmts(
  db: DbLike,
  reversalId: string,
  owner: string,
  purchaseId: string,
  clawCredits: number,
  ts: string,
  currentOwner: string,
  walletHeld: string,
  rail: string,
  reversedBy: string = reversalId,
): DbStatement[] {
  const alarm = db
    .prepare(
      "INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) " +
        "SELECT ?, ?, '{\"purchaseId\":\"' || ? || '\",\"reason\":\"' || CASE WHEN (SELECT at FROM credit_purchases WHERE purchase_id = ?) IS NULL THEN 'purchase_unreadable' ELSE 'credits_moved' END || '\",\"reversalId\":\"' || ? || '\"}', ?, ?, 'system' " +
        `WHERE ${reconciliationBrokenSql(currentOwner)}`,
    )
    .bind(`rail:${rail}`, DISPUTE_DEBIT_WITHHELD_ACTION, purchaseId, purchaseId, reversalId, ts, purchaseId, ...brokenBinds(owner, purchaseId, ts, reversalId));
  return [alarm, disputeDebitStmt(db, reversalId, owner, purchaseId, clawCredits, ts, currentOwner, walletHeld, reversedBy)];
}

/**
 * D-0960(1) — a LOST dispute's past-hold remainder, attributed and recorded.
 * One `creator_dispute_debits` row per CLEARED (`clears_at <= ts`, the exact
 * complement of the sale UPDATE's `clears_at > ?`), un-reversed, not-yet-debited
 * sale of the purchase's current owner spent at or after the purchase, walked
 * NEWEST-FIRST in both the pool and the window (D-0616 / D-0641 U1, the same two
 * sites as the sale UPDATE), while the cumulative gross fits the budget:
 *
 *     budget = max(0, claw - wallet still held) x rate - gross this reversal reversed in-hold
 *
 * THIS dispute's own shortfall only: priors' `reserve_loss_cents` is left out on
 * purpose, because a prior row stays written after a later batch covers it and
 * would debit a second, unrelated sale (the in-hold walk tolerates that only
 * because a reversed sale leaves its pool). `sale_id` is the table's key, so a
 * sale is netted at most once whatever later dispute walks it.
 *
 * I20-Q1 (D-1030 (a) "only from creators the disputed purchase's credits went to, most recent first"; I29-CASHLOTS
 * H4): a sale is walked only when its lot edge (`credit_spend_lots`, 0221) names THIS purchase, or when it has no edge
 * at all (spent before 0221 — today's newest-first walk, kept by `lotSalesForPurchaseSql`'s second limb).
 */
function disputeDebitStmt(
  db: DbLike,
  reversalId: string,
  owner: string,
  purchaseId: string,
  clawCredits: number,
  ts: string,
  currentOwner: string,
  walletHeld: string,
  /** The id the in-hold walk marked THIS take-back's sales with (`topUpStmts` walks under a temporary one). */
  reversedBy: string = reversalId,
): DbStatement {
  const budget = `(MAX(0, ? - MAX(0, ${walletHeld})) * ? - (SELECT COALESCE(SUM(gross_backing_cents),0) FROM sales WHERE reversed_by = ? AND reversed = 1))`;
  const budgetBinds = [clawCredits, owner, CREDIT_BACKING_CENTS, reversedBy];
  return db
    .prepare(
      `INSERT INTO creator_dispute_debits (sale_id, reversal_id, passport_id, amount_cents, gross_backing_cents, at)
       SELECT sale_id, ?, passport_id, creator_net_cents, gross_backing_cents, ? FROM (
         SELECT sale_id, passport_id, creator_net_cents, gross_backing_cents,
                SUM(gross_backing_cents) OVER (ORDER BY at DESC, sale_id DESC ROWS UNBOUNDED PRECEDING) AS cum
           FROM (SELECT s.sale_id, s.passport_id, s.creator_net_cents, s.gross_backing_cents, s.at FROM sales s
                  WHERE s.account_id = ${currentOwner} AND s.reversed = 0 AND s.clears_at <= ?
                    AND s.at >= (SELECT at FROM credit_purchases WHERE purchase_id = ?)
                    AND ${lotSalesForPurchaseSql("s")}
                    AND NOT EXISTS (SELECT 1 FROM creator_dispute_debits d WHERE d.sale_id = s.sale_id)
                    AND s.gross_backing_cents <= ${budget}
                    AND NOT ${reconciliationBrokenSql(currentOwner)}
                  ORDER BY s.at DESC, s.sale_id DESC LIMIT ?))
        WHERE cum <= ${budget}`,
    )
    .bind(reversalId, ts, owner, ts, purchaseId, purchaseId, ...budgetBinds, ...brokenBinds(owner, purchaseId, ts, reversalId), MAX_REVERSAL_SALES, ...budgetBinds);
}

/** Is this account chargeback-banned? Read at the economic surfaces (credit
 * checkout, buy-card, pack open) to refuse a banned fan. */
export async function accountBanned(db: DbLike, accountId: string): Promise<boolean> {
  if (!ACCOUNT_RE.test(accountId)) return false;
  const row = await db.prepare("SELECT banned_at FROM accounts WHERE account_id = ?").bind(accountId).first<{ banned_at: string | null }>();
  return !!row && row.banned_at !== null;
}

/**
 * LIFT a chargeback ban — S-040 C-13. `reverseGrant`'s dispute ban
 * (`accounts.banned_at`) is NOT `closures.ts`'s two-stage suspension: it is
 * set on every dispute, uniformly, and until this export existed nothing in
 * the codebase ever cleared it short of erasing the account (admin.ts's own
 * docblock says so). `docs/legal/APPEALS.md` promises a person reviews a
 * chargeback_fraud suspension and, if overturned, "the suspension is lifted
 * in the same act, and your account works again" — a promise this file could
 * not keep for its OWN ban: the only lever was erasure, which forfeits
 * promotional credits and ends the account, the opposite of "works again".
 *
 * This does not change the ban's trigger or its fraud-deterrent purpose
 * (still set on every dispute, still uniform) — it only gives an operator
 * honoring an appeal the same lever `closures.ts` already has for its own
 * suspensions (`liftClosure`). Idempotent: lifting an account that is not
 * currently banned is a no-op.
 *
 * #640 (D-0758), I17-L04 (D-0943(1) FXA:8): the lift row's `detail` names the
 * payment of the account's LATEST `account:chargeback_banned` row of EITHER
 * kind — a purchase ban (`banRowStmt`, item_id = this account's purchase) as
 * `{"purchaseId":…}`, a refused-grant ban (`banRefusedGrantAccount`, credits.ts
 * grantCredits; item_id = the payment key, the account named by the
 * `credit:grant_refused_dispute` row with that item_id) as
 * `{"purchaseId":null,"paymentKey":…}` — and `disputeBanOrReview` honours the
 * lift for that purchase only (a paymentKey lift shields no purchase). A ban
 * with no ban row (set before #640) gives `detail` "" — no purchase, the
 * account-level meaning every earlier lift keeps. item_id stays the account id, which
 * `deleteAccount` pseudonymises (account_data.ts, the audit_log IN() list).
 */
export async function liftChargebackBan(
  db: DbLike,
  accountId: string,
  operator: string,
  now: () => string = () => new Date().toISOString(),
  /** I39-OPS (D-1158(6b)): the caller's rows (dual_control.ts coSignerAuditStmts), joined into this function's batch. */
  extra: readonly DbStatement[] = [],
): Promise<{ lifted: boolean }> {
  if (!ACCOUNT_RE.test(accountId)) throw new RefundError("bad account id", 500);
  if (!(await accountBanned(db, accountId))) return { lifted: false };
  const ts = now();
  await db.batch([
    db.prepare("UPDATE accounts SET banned_at = NULL WHERE account_id = ? AND banned_at IS NOT NULL").bind(accountId),
    db
      .prepare(
        "INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) " +
          "SELECT ?, 'account:chargeback_ban_lifted', COALESCE((SELECT CASE WHEN p.purchase_id IS NOT NULL THEN json_object('purchaseId', b.item_id) " +
          "ELSE json_object('purchaseId', NULL, 'paymentKey', b.item_id) END FROM audit_log b LEFT JOIN credit_purchases p ON p.purchase_id = b.item_id AND p.account_id = ? " +
          "WHERE b.action = 'account:chargeback_banned' AND (p.purchase_id IS NOT NULL OR EXISTS (SELECT 1 FROM audit_log r WHERE r.action = ? " +
          "AND r.actor_id = b.actor_id AND r.item_id = b.item_id AND CASE WHEN json_valid(r.detail) THEN json_extract(r.detail, '$.accountId') END = ?)) " +
          "ORDER BY b.audit_id DESC LIMIT 1), ''), ?, ?, 'operator'",
      )
      .bind(operator, accountId, GRANT_REFUSED_DISPUTE_ACTION, accountId, ts, accountId),
    ...extra,
  ]);
  return { lifted: true };
}
