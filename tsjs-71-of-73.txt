/**
 * Tips — a fan sends credits to a verified creator (F2, 2026-09-03).
 *
 * THE DECISION. "Tips stay display-only" was the carve-out kept when the
 * Creator War store reversed the no-money stance (plan-creator-war.md). On
 * 2026-09-03 it was reversed too: a tip routes through Credits at 100% creator
 * share, 0 Dhwaja fee, on the SAME rails a card sale uses — the 14-day
 * chargeback hold (`sales.clears_at`), the refund engine (a disputed grant
 * reverses within-hold sales, tips included), the weekly Connect payout. The
 * display-only payout tags (`payout.ts`) still exist and still say "pay the
 * creator directly"; a tip is the on-platform alternative, not a replacement.
 *
 * WHAT A TIP IS IN THE JOURNAL. One guarded debit (`kind 'spend'`, `origin
 * 'tip'`) and one `sales` row with `product 'tip'`, `platform_cut_cents 0`,
 * `creator_net_cents = gross`, `instance_ids '[]'`. Same NULL-PK force-rollback
 * as a card buy: balance short → nothing moves, one generic refusal.
 *
 * WHO CAN BE TIPPED. A CURRENTLY VERIFIED, not retracted, not delisted
 * passport — the same predicate that gates whether payout tags are served.
 * Money to an unverified or delisted creator is exactly the laundering /
 * impersonation surface the trust layer exists to refuse. An operator opt-in
 * (like the war `war` grant) is NOT required: a verified creator may not want
 * the game; nobody refuses a tip. A creator with no Connect binding simply
 * accrues (creator-payout.ts: held indefinitely, never paid — DECIDED there).
 *
 * BOUNDS. `TIP_MIN..TIP_MAX` credits per tip ($0.50–$50 at the pinned
 * backing); `TIPS_PER_FAN_PER_DAY` across all creators (a spend cap, not a
 * trust fact — bounded by the fan's own balance anyway). Amounts are fixed
 * buckets the client offers; a free amount inside the bounds is accepted.
 *
 * WHAT FUNDS A TIP. Only CASH-BACKED credits (credits.ts CASH_ORIGINS): a tip
 * is 100% creator money, so a referral or promotional grant can never become
 * one — the guard checks the cash-backed sub-balance beside the total.
 *
 * IDEMPOTENT ON THE CLIENT'S KEY. A tip is the one spend a fan can send twice
 * by accident (a lost response, a double tap) and the one where "twice" is
 * money out of their pocket to someone else. The client sends `idem` (an
 * opaque key it mints per attempt and REUSES on retry); the journal row carries
 * `tip:<account>:<key>` in its UNIQUE `idem` column, and a replay returns the
 * ORIGINAL sale with `replayed: true` instead of debiting again. A tip with no
 * key is still accepted (the mobile clients predate the field) — it simply has
 * no replay protection, which the SPA copy is written around.
 */
export const TIP_IDEM_RE = /^[A-Za-z0-9_-]{8,64}$/;

import { occurredDay, outboxStatements } from "./analytics-events.js";

export const TIP_MIN_CREDITS = 50;
export const TIP_MAX_CREDITS = 5000;
export const TIPS_PER_FAN_PER_DAY = 20;
export const TIP_BUCKETS = [100, 250, 500, 1000] as const;

const ACCOUNT_RE = /^acc_[0-9a-f]{20}$/;
const PSP_RE = /^psp_[0-9a-f]{16}$/;
const CLE_RE = /^cle_[0-9a-f]{20}$/;
const SAL_RE = /^sal_[0-9a-f]{20}$/;

export class TipError extends Error {

/** Currently verified, not retracted, not delisted — `payout.ts`'s predicate,
 * spelled for a bare passport id. Binds (passportId, the tip's instant): I3-1768 (D-1082(1), I32-FX1-VERIFIED) — a
 * grant whose deadline has passed at the tip's own instant is not tippable even before the expiry sweep folds
 * `claim_state` back (standing.ts `grantLiveAtSql`, the rule `VERIFIED_AT_SUBQUERY` is built from). */
// D-0960(2)/TERMS 7.4 (I20-L02): a closed creator (creator_closures) cannot be tipped.
const TIPPABLE = `EXISTS (
  SELECT 1 FROM passports p
   WHERE p.passport_id = ?
     AND p.claim_state = 'verified'
     AND p.retracted = 0
     AND NOT ${CREATOR_CLOSED_SQL("p.passport_id")}
     AND NOT EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = p.passport_id)
     AND ${grantLiveAtSql("p.passport_id")})`;

/** The guarded debit's refusal (its NULL primary key) and the idempotency race, each by its own column. */
const TIP_GUARD_REFUSAL_RE = /NOT NULL constraint failed: credit_ledger\.entry_id\b/i;
const TIP_IDEM_RACE_RE = /UNIQUE constraint failed: credit_ledger\.idem\b/i;

export async function tipCreator(
  db: DbLike,
  accountId: string,
  passportId: string,
  credits: number,
  ids: { entryId: string; saleId: string },
  nowMs: number,
  clientIdem: string | null = null,
  pubEnv?: AnalyticsTickEnv,
  pubCtx?: { waitUntil(p: Promise<unknown>): void },
): Promise<{ sale_id: string; balance: number; credits: number; replayed: boolean }> {
  if (!ACCOUNT_RE.test(accountId)) throw new WarInstanceError("bad account id", 500);
  if (typeof passportId !== "string" || !PSP_RE.test(passportId)) throw new TipError("no such creator", 404);
  if (!Number.isInteger(credits) || credits < TIP_MIN_CREDITS || credits > TIP_MAX_CREDITS) throw new TipError(`tip must be between ${TIP_MIN_CREDITS} and ${TIP_MAX_CREDITS} credits`, 400);
  if (!CLE_RE.test(ids.entryId) || !SAL_RE.test(ids.saleId)) throw new WarInstanceError("bad id shape", 500);
  if (clientIdem !== null && !TIP_IDEM_RE.test(clientIdem)) throw new TipError("bad idempotency key", 400);
  // BF-20/U-07 (S-051-batch TIPS_TRU): bound to passportId AND credits, not
  // just accountId — a bare `tip:<account>:<key>` let a caller reuse one key
  // across a different creator or amount and get the ORIGINAL sale's
  // replayed:true for a tip that never happened to the second creator.
  const idem = clientIdem === null ? null : `tip:${accountId}:${passportId}:${credits}:${clientIdem}`;

  // A replay answers with the ORIGINAL tip before any guard can refuse it —
  // the first attempt may have drawn the balance below the amount, and a 409
  // there would read as "not sent" to a client that already sent it.
  const replay = async (): Promise<{ sale_id: string; balance: number; credits: number; replayed: boolean } | null> => {
    if (idem === null) return null;
    const prior = await db.prepare("SELECT ref, credits FROM credit_ledger WHERE idem = ? AND account_id = ?").bind(idem, accountId).first<{ ref: string; credits: number }>();
    if (!prior) return null;
    return { sale_id: prior.ref, balance: await creditBalance(db, accountId), credits: -prior.credits, replayed: true };
  };
  const replayed = await replay();
  if (replayed) return replayed;

  // ONE generic refusal for "cannot be tipped": unknown, unverified, retracted
  // and delisted all read the same, so a tip attempt is not a listing oracle
  // (the same discipline `buyCard` follows).
  const ts = new Date(nowMs).toISOString();
  const ok = await db.prepare(`SELECT ${TIPPABLE} AS ok`).bind(passportId, ts).first<{ ok: number }>();
  if (!ok || ok.ok !== 1) throw new TipError("this creator cannot be tipped", 409);
  // The pre-batch read above is a fast path only (fast generic refusal); it is
  // not what makes the bound true. BF-20 (S-051-batch TIPS_TRU): re-asked
  // INSIDE the guarded debit's CASE below, so a retraction/delisting that
  // commits between this read and the batch cannot still buy its way through
  // — the same discipline `buyCard`'s WAR_ELIGIBLE_SQL follows (credits.ts).

  const dayStart = new Date(nowMs - 24 * 60 * 60 * 1000).toISOString();
  const clearsAt = new Date(nowMs + PAYOUT_HOLD_MS).toISOString();
  const gross = credits * CREDIT_BACKING_CENTS;
  let raced: { sale_id: string; balance: number; credits: number; replayed: boolean } | null = null;
  let seqs: number[] = [];
  try {
    const span = outboxSpan(db);
    const spanRes = await db.batch([
      span.open,
      // 1. Guarded debit: the balance (total AND cash-backed) covers the tip AND
      //    the fan is under the daily tip count. NULL PK → the whole batch rolls
      //    back. `idem` is the client's key (UNIQUE) or NULL.
      db
        .prepare(
          `INSERT INTO credit_ledger (entry_id, account_id, kind, origin, credits, ref, idem, reason, at)
           SELECT CASE
                    WHEN ${CASH_GUARD_SQL}
                     AND (SELECT COUNT(*) FROM sales WHERE account_id = ? AND product = 'tip' AND at >= ?) < ?
                     AND ${TIPPABLE}
                    THEN ? ELSE NULL END,
                  ?, 'spend', 'tip', ?, ?, ?, 'tip', ?`,
        )
        .bind(...cashGuardBinds(accountId, credits, ts), accountId, dayStart, TIPS_PER_FAN_PER_DAY, passportId, ts, ids.entryId, accountId, -credits, ids.saleId, idem, ts),
      // 2. The sale — predicated on the debit having landed. 100% to the creator.
      db
        .prepare(
          `INSERT INTO sales (sale_id, account_id, passport_id, product, credits_spent, gross_backing_cents, platform_cut_cents, creator_net_cents, instance_ids, clears_at, reversed, at)
           SELECT ?, ?, ?, 'tip', ?, ?, 0, ?, '[]', ?, 0, ?
            WHERE EXISTS (SELECT 1 FROM credit_ledger WHERE entry_id = ? AND ref = ?)`,
        )
        .bind(ids.saleId, accountId, passportId, credits, gross, gross, clearsAt, ts, ids.entryId, ids.saleId),
      // I20-Q1: which purchase(s) paid for this tip, FIFO, predicated on the sale row (credit-lots.ts).
      lotEdgeStmt(db, ids.saleId, accountId, credits, ts),
      // 3. The analytics row, in THIS batch (ADR-0006 §2, S-12). A tip IS a
      //    `sales` row — same table, same hold, same reversal engine — so it is
      //    the same `market_sale` event a card buy emits, and leaving it out
      //    would make `creator_revenue_daily` disagree with `sales` by exactly
      //    the creator's tip income. 100% creator share, so `net_cents` is the
      //    gross and `platform_cut_cents` is 0 — declared rather than omitted,
      //    because "no cut was taken" and "the cut is unknown" must not read
      //    alike at the consumer.
      //
      //    It needs no predicate of its own: statement 1's CASE yields a NULL
      //    primary key when the balance or the daily cap refuses the tip, and a
      //    NULL into a NOT NULL primary key rolls the WHOLE batch back — so this
      //    row exists iff the tip does.
      ...outboxStatements(db, {
        kind: "market_sale",
        passportId,
        occurredDay: occurredDay(ts),
        amountCents: gross,
        dims: { clears_day: clearsAt.slice(0, 10), net_cents: gross, platform_cut_cents: 0 },
      }),
      span.close,
    ]);
    seqs = outboxSpanSeqs(spanRes[0], spanRes[spanRes.length - 1]);
  } catch (e) {
    const msg = e instanceof Error ? e.message : String(e);
    // Two identical attempts racing: whichever lost the race (UNIQUE on the
    // key, or NOT NULL because the winner drew the balance down first) is
    // answered with the winner's tip. FX-I16-RG-MONEY:10 (I16-W13S H6): matched on
    // the guarded debit's OWN column and the tip's OWN idempotency key, as `buyCard`
    // does (credits.ts BUY_GUARD_REFUSAL_RE) — any other constraint is a fault, not a refusal.
    const refused = TIP_GUARD_REFUSAL_RE.test(msg);
    if (refused || TIP_IDEM_RACE_RE.test(msg)) raced = await replay();
    if (!raced) {
      if (refused) throw new TipError("insufficient credits or daily tip limit reached", 409);
      throw e;
    }
  }
  if (!raced) {
    if (pubEnv && pubCtx) publishAfterCommit(pubEnv, pubCtx, seqs);
  }
  if (raced) return raced;
  // Did it land? (A batch carries no portable `changes`.)
  const landed = await db.prepare("SELECT 1 AS x FROM sales WHERE sale_id = ?").bind(ids.saleId).first<{ x: number }>();
  if (!landed) throw new TipError("insufficient credits or daily tip limit reached", 409);
  return { sale_id: ids.saleId, balance: await creditBalance(db, accountId), credits, replayed: false };
}

import type { DbLike } from "./db.js";
import { translator } from "@dhwaja/i18n";

/**
 * RFC 8058 ONE-CLICK UNSUBSCRIBE — the one capability in this service that
 * travels in a URL, and the argument for why.
 *
 * ============================================================================
 * WHAT WAS THERE BEFORE, AND WHY IT WAS NOT AN UNSUBSCRIBE
 * ============================================================================
 *
 * Both mail streams shipped `List-Unsubscribe` in the RFC 2369 form only —
 * `<https://app.dhwaja.example/#/me>` for fan alerts,
 * `<https://console.dhwaja.example/#/digests>` for creator digests. Neither
 * works as an unsubscribe for a MACHINE:
 *
 *   * both are SPA fragments on Cloudflare Pages origins, not this Worker. The
 *     fragment never leaves the client, a GET returns the static shell and a
 *     POST to a Pages asset is a 405;
 *   * both destinations are behind a session a mailbox provider does not have.
 *
 * It is a link for a human, and only that. Gmail's and Yahoo's bulk-sender
 * rules want RFC 8058: `List-Unsubscribe-Post: List-Unsubscribe=One-Click` and
 * a URL that accepts a POST and unsubscribes with no further interaction.
 *
 * ============================================================================
 * THE COLLISION AT THE CENTRE OF THIS, STATED RATHER THAN DESIGNED AROUND
 * ============================================================================
 *
 * Every other capability in this repository travels in a request BODY, and
 * `POST /removal/status` and `POST /recovery/status` both give the reason in
 * their own comments: `observability.enabled` is on in all three wrangler
 * configs, so Workers Logs records invocation URLs, and a path-borne capability
 * would sit in that log for the retention window.
 *
 * RFC 8058 makes that impossible. The provider POSTs a FIXED form body
 * (`List-Unsubscribe=One-Click`) to the URL taken from the header, so the token
 * MUST be in the URL or there is no one-click at all. This module cannot
 * satisfy both rules. It takes the trade deliberately and bounds it:
 *
 *   * SINGLE-USE — the token is spent by the same batch that performs the
 *     write, so a log-reader replaying the URL changes nothing;
 *   * SHORT-LIVED — `UNSUBSCRIBE_TTL_MS`, the life of one mail rather than one
 *     recipient;
 *   * A KEY FOR NOTHING ELSE — it resolves to one (stream, subject) row, grants
 *     no read, and cannot be traded for an address, a passport or any state.
 *     The route's answer is the same 200 in every case, so it is not even an
 *     existence oracle for itself;
 *   * THE ACT IS BENIGN AND REVERSIBLE BY ITS OWNER — it turns a mail stream
 *     off, and the owner turns it back on in their own console.
 *
 * ============================================================================
 * ONLY THE POST ACTS. A GET NEVER CHANGES ANYTHING.
 * ============================================================================
 *
 * There is deliberately no GET that unsubscribes, and this is a security
 * property rather than an omission: corporate mail scanners, link checkers and
 * client prefetchers fetch every URL in a message. A GET that unsubscribed
 * would hand every such scanner the power to silently stop a creator's
 * protection mail — including the payout-change notice. RFC 8058 is POST-only
 * for exactly this reason, and the human-facing "manage these emails" link in
 * the body still points at the app, where a session decides.
 *
 * I3-0163 (D-0882(2)) AMENDS THIS AND KEEPS ITS REASON: the same one-click URL
 * now also sits in the mail BODY, so a human click is a GET to it. The only GET
 * this path answers is `unsubscribeConfirmPage` below — static, the same bytes
 * for every token, and it changes NOTHING; the opt-out is still only the POST its
 * button sends. A GET that unsubscribed remains the defect described above. The
 * route is `app.get("/unsubscribe/:token")` in index.ts (served since I12-FX3).
 *
 * ============================================================================
 * NO ADDRESS, IN ANY FORM
 * ============================================================================
 *
 * Migration 0049's rule, for its reason. A row here names WHO by internal id
 * (`acc_` / `psp_`); it never stores the address, nor a digest of one — an
 * unsalted digest of a lowercased email is reversible by enumeration, so
 * storing one would reproduce the roster the design exists to avoid.
 */

export class UnsubscribeError extends Error {
  override name = "UnsubscribeError";
  constructor(
    message: string,
    public readonly status: 400 | 429 = 400,
  ) {
    super(message);
  }
}

/**
 * The mail streams a one-click token may turn off. Stored on the row rather
 * than derived, so a token minted for one stream can never be spent on another
 * even if the two id spaces ever overlapped.
 *
 * `fan_alerts` writes `accounts.email_alerts`, which gates the Save-Creator
 * alert's non-transactional kinds only (`WHERE_EMAILABLE` in alerts.ts). It no
 * longer gates `reports.ts`'s report-outcome mail: that rail dropped the prefs
 * gate in W3-03 (see its docblock), so a one-click unsubscribe does not stop a
 * fan's own report outcomes, and transactional alert kinds are mailed regardless.
 *
 * `fan_report_outcomes` (D-0996(3), I3-2072) writes `accounts.email_report_outcomes`, the report-outcome mail's own
 * switch (reports.ts emailReportersOnWarning), and nothing else: the one-click on that mail stops that mail only.
 *
 * `creator_digest_email` writes `creator_contacts.digest_email_enabled` and
 * NOT `digests_enabled`. The latter gates the digest email *and* the digest
 * push, and a button labelled "unsubscribe from these emails" that silently
 * stops a creator's phone notifications is not what the button says.
 *
 * `fan_newsletter` (QUEUE #237, D-0384; I41-NEWSLETTER) DELETES the one `newsletter_subscriptions` row whose
 * `unsubscribe_token` is this token (newsletter.ts mints it for the confirmation mail), and nothing else.
 */
export const UNSUBSCRIBE_STREAMS = ["fan_alerts", "creator_digest_email", "fan_report_outcomes", "fan_newsletter"] as const;
/** D-0996(3): the streams keyed on a FAN's `acc_` id. Every statement that retires a fan's tokens (erasure, an
 * address swap or removal) names this whole list, so a token for one fan stream cannot outlive the others. */
export const FAN_UNSUBSCRIBE_STREAMS_SQL = "('fan_alerts', 'fan_report_outcomes', 'fan_newsletter')";
export type UnsubscribeStream = (typeof UNSUBSCRIBE_STREAMS)[number];

/** `uns_` + 32 hex = 128 CSPRNG bits, matching `rmv_` and `trk_`. It is a
 * bearer token, not a row id, so it takes the bearer-token width. */
export const UNSUBSCRIBE_TOKEN_RE = /^uns_[0-9a-f]{32}$/;

/**
 * How long a token stays spendable. Thirty days is the life of ONE MAIL, not of
 * one recipient: the trade above is only bounded if the URL sitting in an
 * invocation log stops working, and a digest arrives weekly at worst so a month
 * covers a slow reader without leaving a capability alive for a year. The
 * `pruneUnsubscribeTokens` sweep deletes past twice this, so an expired row
 * cannot be resurrected by a clock skew and the table cannot grow forever.
 */
export const UNSUBSCRIBE_TTL_MS = 30 * 24 * 60 * 60 * 1000;

/**
 * How many tokens one `mintUnsubscribeTokens` call may write.
 *
 * DERIVED FROM D1'S BIND-PARAMETER CEILING, not chosen. The multi-row INSERT
 * below binds **four** parameters per row, and D1 rejects a statement past 100
 * bound parameters outright (`db.ts` pins the shared `IN_CHUNK` at 90 for the
 * one-parameter-per-row case; `test-fidelity/param-cap.d1.test.ts` proves real
 * workerd throws `too many SQL variables` at 101). Four per row makes the true
 * ceiling 25, and this sits under it with headroom.
 *
 * This number was 64 in the first draft, reasoned by analogy with `IN_CHUNK` —
 * which is one-parameter-per-row arithmetic that does not transfer. 64 rows is
 * 256 parameters, so the "guard" was 2.5× past the limit and every wave wider
 * than 25 would have thrown. The better-sqlite3 double's ceiling is ~32k, so no
 * test in this repository could have seen it; only real D1 would, in production,
 * silently (the callers catch, so the mail ships without a one-click header and
 * the feature is simply off).
 *
 * A wave wider than this mints for the first `MINT_BATCH_MAX` subjects and the
 * rest fall back to the RFC 2369 header — the same outcome as an absent public
 * base, which is a degradation rather than a failure. The alert leg's own width
 * is `DELIVER_CONCURRENCY` (16) today, so this does not bite at the shipped
 * numbers; it bounds a caller that passes a wider `concurrency`.
 */
export const MINT_BATCH_MAX = 22;

/**
 * Mint one token per subject, in ONE statement.
 *
 * The alert leg sends a wave of emails per pass and its statement budget is a
 * hard bound (`EMAIL_STATEMENT_BUDGET`), so a per-message INSERT would spend
 * the budget that exists to keep the cron inside D1's limits. A multi-row
 * INSERT costs the wave one statement whatever its width.
 *
 * Subjects are DEDUPED: two notifications for the same account in one wave get
 * the same token, which is right — the unsubscribe they offer is the same act,
 * and spending it once is the whole intent.
 *
 * Returns a map of subject id → token. A subject missing from the map (an empty
 * input, or a write that failed) simply gets no one-click header on that mail:
 * the RFC 2369 link and the body footer are unchanged, so the mail is never
 * blocked on this.
 */
export async function mintUnsubscribeTokens(db: DbLike, stream: UnsubscribeStream, subjectIds: readonly string[], nowIso: string): Promise<Map<string, string>> {
  const out = new Map<string, string>();
  const unique = [...new Set(subjectIds.filter((s) => typeof s === "string" && s.length > 0))].slice(0, MINT_BATCH_MAX);
  if (unique.length === 0) return out;
  const rows: unknown[] = [];
  for (const subjectId of unique) {
    const token = mintId("uns", 32);
    out.set(subjectId, token);
    rows.push(token, stream, subjectId, nowIso);
  }
  await db
    .prepare(`INSERT INTO unsubscribe_tokens (token, stream, subject_id, created_at) VALUES ${unique.map(() => "(?, ?, ?, ?)").join(", ")}`)
    .bind(...rows)
    .run();
  return out;
}

/** The absolute URL a mailbox provider POSTs. `publicBase` is THIS Worker —
 * never the app or console origin, which are Pages assets that answer a POST
 * with a 405. A caller with no `PUBLIC_BASE_URL` gets null and sends no
 * one-click header at all (fail closed: a header pointing nowhere is worse than
 * no header, because a provider that gets a non-2xx marks the sender down). */
export function unsubscribeUrl(publicBase: string | null | undefined, token: string): string | null {
  if (!publicBase || !UNSUBSCRIBE_TOKEN_RE.test(token)) return null;

/**
 * Spend a token and turn its stream off. Returns nothing, and that is the
 * anti-oracle property: the route answers the same 200 for a live token, an
 * unknown one, an expired one, one already spent, and one whose subject row has
 * since been deleted. Nothing a caller can observe distinguishes them.
 *
 * ORDER MATTERS INSIDE THE BATCH. The preference write comes FIRST and is
 * guarded on liveness ONLY — `used_at IS NULL` on the token row (`liveForWrite`
 * below). The claim that follows carries `used_at IS NULL AND created_at >=
 * cutoff`. If the claim ran first it would set `used_at` and the preference
 * write's guard would then find nothing, so a valid one-click would spend the
 * token and change no setting. The guards are NOT the same (P-03): an
 * unspent token past its TTL still turns the stream off, and is simply not
 * marked spent — the opt-out lands whenever the 200 says it did.
 *
 * A guarded pair rather than a read-then-write: two POSTs of the same token in
 * flight together both see `used_at IS NULL` at SELECT time, and it is the
 * claim's WHERE that makes exactly one of them win.
 */
export async function unsubscribeOneClick(db: DbLike, token: string, nowMs: number): Promise<void> {
  // Shape first, and it costs nothing to decide: the regex is public and
  // client-computable, so refusing to spend a database round-trip on a
  // malformed token discloses nothing that the token's own format does not.
  if (!UNSUBSCRIBE_TOKEN_RE.test(token)) return;
  const nowIso = new Date(nowMs).toISOString();
  const cutoff = new Date(nowMs - UNSUBSCRIBE_TTL_MS).toISOString();
  const row = await db.prepare("SELECT stream, subject_id FROM unsubscribe_tokens WHERE token = ?").bind(token).first<{ stream: string; subject_id: string }>();
  if (!row) return;
  // An unknown stream on a stored row is a migration or a tampered database,
  // never a caller's doing. Refuse rather than guess which table to write.
  if (!(UNSUBSCRIBE_STREAMS as readonly string[]).includes(row.stream)) return;
  const stream = row.stream as UnsubscribeStream;
  // THE GUARD IS EXACTLY WHAT DECIDES, AND NOTHING ELSE. It re-asserts the two
  // mutable facts — not yet spent, not yet expired — on the row the SELECT above
  // just read by primary key. `stream` and `subject_id` are deliberately NOT
  // repeated here: nothing in this codebase ever UPDATEs either column, so a
  // clause naming them could not fail, and a guard that cannot fail reads as
  // protection while providing none. The write TARGET still comes from those two
  // columns; it is their immutability, not a repeated WHERE, that makes reading
  // them a moment earlier safe. `used_at` is the only field that races, and it
  // is the one both statements below guard.
  // P-03 fix: the PREFERENCE write is guarded on liveness only (not yet spent),
  // never on the TTL. Answering `{ok:true}` for an expired token while writing
  // nothing is the RFC 8058 lie this fix closes: the response promised the
  // opt-out landed, so the effect must land too. The TOKEN-CLAIM statement
  // below keeps the cutoff — capability lifetime is unchanged, only the effect
  // it gates is no longer coupled to that lifetime.
  // CEN-psa (I16-FXC): the sub-select is held as the SQL it is (leading verb), so the SSR copy lint does not read it as prose.
  const liveTokenSelect = "SELECT 1 FROM unsubscribe_tokens t WHERE t.token = ? AND t.used_at IS NULL";
  const liveForWrite = `EXISTS (${liveTokenSelect})`;
  const write =
    stream === "fan_alerts"
      ? db.prepare(`UPDATE accounts SET email_alerts = 0 WHERE account_id = ? AND ${liveForWrite}`).bind(row.subject_id, token)
      : stream === "fan_report_outcomes"
        ? db.prepare(`UPDATE accounts SET email_report_outcomes = 0 WHERE account_id = ? AND ${liveForWrite}`).bind(row.subject_id, token)
        : stream === "fan_newsletter"
          ? db.prepare(`DELETE FROM newsletter_subscriptions WHERE account_id = ? AND unsubscribe_token = ? AND ${liveForWrite}`).bind(row.subject_id, token, token)
          : // A creator with no `creator_contacts` row cannot have been mailed (the
        // digest email leg requires a verified address, which lives on that
        // row), so a token for this stream always has a row to write. A missing
        // one changes nothing and is answered like every other case.
        db.prepare(`UPDATE creator_contacts SET digest_email_enabled = 0, updated_at = ? WHERE passport_id = ? AND ${liveForWrite}`).bind(nowIso, row.subject_id, token);
  await db.batch([
    write,
    db.prepare("UPDATE unsubscribe_tokens SET used_at = ? WHERE token = ? AND used_at IS NULL AND created_at >= ?").bind(nowIso, token, cutoff),
  ]);
}

/**
 * Drop dead tokens. Past TWICE the TTL, so a row this deletes is already
 * unspendable by `unsubscribeOneClick`'s own cutoff for any plausible clock
 * skew — the sweep can never be what makes a live unsubscribe stop working.
 *
 * The table is written only by our own send path, so it is not outsider-growable
 * the way `creator_otp` is; it is swept because a row that decides nothing is a
 * row that should not be kept, and because a spent token is a record of one
 * recipient having been mailed.
 */
export async function pruneUnsubscribeTokens(db: DbLike, nowMs: number): Promise<void> {
  await db
    .prepare("DELETE FROM unsubscribe_tokens WHERE created_at < ?")
    .bind(new Date(nowMs - 2 * UNSUBSCRIBE_TTL_MS).toISOString())
    .run();
}

/**
 * UUIDv7: time-ordered, CSPRNG-filled. Used for event_ids so the id order
 * roughly matches chain order and ids are unguessable.
 */
export function uuid7(now: number = Date.now()): string {
  const bytes = new Uint8Array(16);
  crypto.getRandomValues(bytes);
  // 48-bit big-endian unix ms timestamp
  const ms = BigInt(now);
  for (let i = 0; i < 6; i++) {
    bytes[i] = Number((ms >> BigInt(8 * (5 - i))) & 0xffn);
  }
  bytes[6] = (bytes[6]! & 0x0f) | 0x70; // version 7
  bytes[8] = (bytes[8]! & 0x3f) | 0x80; // variant 10
  const hex = [...bytes].map((b) => b.toString(16).padStart(2, "0")).join("");
  return `${hex.slice(0, 8)}-${hex.slice(8, 12)}-${hex.slice(12, 16)}-${hex.slice(16, 20)}-${hex.slice(20)}`;
}

import type { DraftEvent } from "@dhwaja/schema";
import { fanOutVerificationExpiring } from "./alerts.js";
import type { DbLike } from "./db.js";
import { noticeVerificationExpiring } from "./digest.js";
import { tryAppendEvents } from "./events.js";

/**
 * THE SWEEP THAT MAKES `expires_at` MEAN SOMETHING.
 *
 * `verification_granted` has REQUIRED an `expires_at` since the first grant, and
 * for the whole of that time nothing anywhere read it: a grant six years past
 * its stated deadline still badged a creator **Verified** on the registry, still
 * served the badge SVG and the share card, still served their payout tags, and
 * still passed `requireVerifiedCreator` on all four mint paths. That was
 * docs/VERIFIED-MINT.md known gap §1, measured rather than suspected.
 *
 * WHY A SWEEP AND NOT A CHECK AT THE READ SITES. There are ~180 places that read
 * `claim_state`, five of them standing readers with three different predicates
 * (see `standing.ts`). An expiry check bolted onto the mint chokepoint alone
 * would have produced the exact asymmetry that document exists to reject — a
 * creator publicly badged Verified while silently unable to mint, two different
 * answers to "are you verified?". So the LEDGER expresses the lapse instead: this
 * sweep appends `verification_expired`, the existing fold moves the passport
 * verified → claimed (events.ts `foldEffects`, mirrored in fold.ts), and every
 * standing reader converges on the same instant, with no per-read-site change.
 *
 * IT APPENDS. It never UPDATEs `trust_events` and never rewrites the passport
 * row itself — the state change is the fold's, downstream of the append, exactly
 * as it is for `correction`/`downgrade_verification`.
 */

/**
 * How long before a deadline the CREATOR gets told. A verification lapse costs
 * a creator their badge, their payout-tag presentation and their ability to
 * mint anything — finding that out from a fan is not acceptable, so the notice
 * goes out ahead of the event, not with it, and it goes to them (see the warn
 * leg below, which used to send it to their savers instead).
 */
export const EXPIRY_WARNING_MS = 14 * 24 * 60 * 60 * 1000;

/**
 * Verified passports examined per run. This is a read bound, not a write bound:
 * a candidate the sweep decides NOT to expire costs one `JSON.parse`, so the
 * scan is deliberately much wider than the append budget below.
 *
 * STALE CLAIM CORRECTED (S-072 SWEEPS, QUEUE #141/#142). This paragraph used to
 * end "above this many live verified creators the tail of `updated_seq ASC`
 * goes unexamined for a tick … and the fix when the population outgrows it is a
 * cursor, not a bigger number." That fix SHIPPED — see D8 below — so the
 * sentence described a behaviour this file no longer has, and it was the
 * sentence a reader of the sweep-cursor family reached for. The tail is not
 * skipped: a page shorter than this limit wraps, a full page resumes strictly
 * past its last COMPLETED row, and the whole verified population is therefore
 * covered across ticks. What this number still bounds is how much of that
 * population one tick sees, and the ordering is still chosen so the page starts
 * at the least-recently-touched end (a renewal bumps `updated_seq`, so freshly
 * renewed creators sort to the back).
 */
export const SWEEP_SCAN_LIMIT = 2000;

/** Ledger appends per run — the expensive leg, and the one the cron budget cares
 * about. Same order of magnitude as `sweepDuePayoutTags`/`sweepFounding`. */
export const SWEEP_APPEND_LIMIT = 200;

export interface VerificationSweepResult {
  /** Live, currently-verified passports examined. */
  scanned: number;
  /** `verification_expired` events appended. */
  expired: number;
  /** Creators warned that their grant lapses inside `EXPIRY_WARNING_MS`. One
   * per creator per deadline — this counts PEOPLE TOLD, not cron ticks. */
  warned: number;
  /**
   * Verified passports whose most recent grant/renewal carries an expiry that
   * CANNOT BE READ — absent, empty, or unparseable. Emphatically **not**
   * expired; counted so an operator can see how many there are and go and fix
   * them. See the landmine note on `parseVerificationExpiry`.
   */
  unreadable_expiry: number;
  /** Verified passports with no grant/renewal event at all (seeded/imported
   * state that predates or bypasses the review queue). Also never expired. */
  no_grant: number;
  /** Candidates whose append was refused — almost always a benign race with a
   * concurrent downgrade. Logged, never fatal. */
  refused: number;
  /** C-48/PS-001: candidates that WERE past their deadline but whose append was
   * skipped because `appendLimit` was already spent this run. The cursor is
   * held back to just before the first of these (see below), so they are the
   * FIRST candidates examined next tick — never "ride the next wrap". Counted
   * so an operator can see the sweep is chronically short of append budget
   * rather than inferring it from a gap between `scanned` and `expired`. */
  deferred: number;
}

interface CandidateRow {
  passport_id: string;
  updated_seq: number;
  grant_payload: string | null;
}

/**
 * Expire every live passport whose current verification is past its stated
 * deadline, and warn the ones about to be.
 *
 * ─────────────────────────────────────────────────────────────────────────────
 * THE EMPTY-STRING EXPIRY — READ THIS BEFORE CHANGING THE SKIP BELOW.
 * ─────────────────────────────────────────────────────────────────────────────
 * `review.ts` wrote `String(item.payload.expires_at ?? "")` into every grant it
 * ever approved, so `""` is a VALID, ALREADY-COMMITTED expiry sitting in real
 * append-only ledgers, and `Date.parse("")` is NaN. A sweep written the obvious
 * way — "if I can't read a deadline, the deadline has passed" — would append
 * `verification_expired` for every one of those creators in a single cron tick
 * and de-badge the entire legacy verified population at once, irreversibly as
 * far as the log is concerned. Absent, empty and unparseable therefore all mean
 * UNKNOWN here, never "long past": the creator is LEFT ALONE and counted into
 * `unreadable_expiry` for the operator. The source is fixed at the writer
 * (`review.ts` now refuses to grant without a parseable `expires_at`), which is
 * where it belongs; this counter is how anyone learns the backlog is shrinking.
 */
export async function sweepExpiredVerifications(
  db: DbLike,
  nowMs: number,
  scanLimit = SWEEP_SCAN_LIMIT,
  appendLimit = SWEEP_APPEND_LIMIT,
): Promise<VerificationSweepResult> {
  const out: VerificationSweepResult = { scanned: 0, expired: 0, warned: 0, unreadable_expiry: 0, no_grant: 0, refused: 0, deferred: 0 };

  // D8: RESUME CURSOR. Without it, the two never-appending classes (`no_grant`,
  // `unreadable_expiry`) squat the head of the `updated_seq ASC` scan forever and
  // the tail — where genuinely-lapsed creators sit — is never re-examined, so a
  // lapsed creator keeps minting. The cursor makes the scan resumable: each run
  // starts strictly past where the last stopped and wraps to the start on
  // reaching the end, so the whole verified population is covered across ticks.
  const cur = (await db.prepare("SELECT cursor_seq, cursor_id FROM sweep_cursor WHERE name = 'verification'").first<{ cursor_seq: number; cursor_id: string }>()) ?? { cursor_seq: 0, cursor_id: "" };

  // One statement, one correlated read of the LATEST grant-or-renewal payload.
  // The `IN (…)` is a two-element LITERAL, not an interpolated placeholder list,
  // so there is nothing for `inChunks` to chunk (see the IN-builder census in
  // tests/invariants/fidelity-suite.test.ts). Payloads are folded in JS rather
  // than with `json_extract`, per the codebase convention: `json_extract` raises
  // on malformed JSON and would take the whole sweep down with one bad row.
  // The tuple comparison `(updated_seq, passport_id) > (cursor)` resumes the
  // scan; it uses the same `(updated_seq, passport_id)` order as before. The
  // leading `updated_seq >= ?` is implied by the OR and changes no row; it is
  // there because SQLite cannot seek on an OR, so without it the page is a walk
  // of idx_passports_verified_live from its first entry (I4-17-R3-30).
  const { results } = await db
    .prepare(
      `SELECT p.passport_id AS passport_id, p.updated_seq AS updated_seq,
              (SELECT e.payload FROM trust_events e
                WHERE e.subject = p.passport_id
                  AND e.event_type IN ('verification_granted', 'verification_renewed')
                ${grantInForceOrderSql("e", instantSqlLiteral(nowMs))} LIMIT 1) AS grant_payload
         FROM passports p
        WHERE p.claim_state = 'verified' AND p.retracted = 0
          AND p.updated_seq >= ?
          AND (p.updated_seq > ? OR (p.updated_seq = ? AND p.passport_id > ?))
        ORDER BY p.updated_seq ASC, p.passport_id ASC
        LIMIT ?`,
    )
    .bind(cur.cursor_seq, cur.cursor_seq, cur.cursor_seq, cur.cursor_id, scanLimit)
    .all<CandidateRow>();

  // C-48/PS-001 FIX. The cursor used to be written HERE, unconditionally, to the
  // PAGE'S LAST ROW before any row was processed — so a row that turned out to
  // be past its deadline but was skipped for want of append budget (`:continue`
  // below, once `out.expired >= appendLimit`) was ALREADY behind the cursor by
  // the time its own iteration ran. The comment there used to say "the rest ride
  // the next tick"; they did not — they rode the next WRAP, because the query
  // above excludes anything at or behind the cursor and a wrap only happens on a
  // page shorter than `scanLimit`. `lastCompleted` tracks the last row this run
  // actually finished (appended, refused, unreadable, no-grant, or confirmed
  // still-good) — never a row whose expiry check the append budget cut off — and
  // the cursor write below uses THAT, not the raw page boundary. A budget-cut
  // run therefore leaves the cursor sitting just before the first deferred row,
  // so that row is the very first thing the next tick looks at.
  let lastCompleted: CandidateRow | undefined;
  let budgetSpent = false;
  // C-48/PS-001 GUARD — ONE DOOR FOR THE CURSOR (I3-0326).
  // The paragraph above states the property; this is the line that enforces it.
  // Once a row has been deferred at :250 for want of append budget, no later row
  // may move `lastCompleted` past it, or the cursor write at :299 strands the
  // deferred row — a creator already past their deadline — behind the cursor for
  // a whole wrap. Every "this row finished" site calls this instead of assigning,
  // so the predicate exists once rather than at five separate doors.
  const complete = (r: CandidateRow): void => {
    if (!budgetSpent) lastCompleted = r;
  };

  for (const row of results) {
    out.scanned += 1;
    if (row.grant_payload === null) {
      out.no_grant += 1;
      complete(row);
      continue;
    }
    let raw: unknown;
    try {
      const p = JSON.parse(row.grant_payload) as Record<string, unknown>;
      raw = p.expires_at;
    } catch {
      raw = undefined; // a payload that will not parse states no deadline
    }
    // F13 / I34-RJ-02 (I34-W2-EXPIRY-B, D-1112(8)): the deadline as EVERY door reads it (standing.ts grantDeadlineMs,
    // grantLiveAtSql's parse), so a stored form one reader takes and another does not is swept exactly when the doors
    // call it lapsed, and never otherwise.
    const expiry = grantDeadlineMs(raw);
    if (expiry === null) {
      out.unreadable_expiry += 1;
      complete(row);
      continue;
    }

    if (expiry > nowMs) {
      // Still good. Warn ONCE PER DEADLINE if it lapses soon — the dedup key is
      // the deadline itself, so a renewal moves the key and a re-warn is
      // impossible until there is genuinely a new deadline.
      //
      // THIS ONLY EVER REACHED THE CREATOR'S FANS. `fanOutVerificationExpiring`
      // inserts one `notifications` row per `saves` row, and its docstring said
      // that reached the creator too "by the SAME route `fanOutPayoutChange`
      // already relies on — a self-save". Part 4b (#24) deleted the founder
      // auto-save and dropped the table behind it, so the only remaining writer
      // of `saves` is the fan-scoped `saveCreator`, which needs an `accounts`
      // row a creator does not have. Two halves of one remediation cancelled,
      // and the outcome was the exact thing the paragraph above calls
      // unacceptable: the fans were told and the creator was not.
      //
      // The CREATOR leg is first now and it is the one `warned` counts, because
      // the counter is meant to answer "how many people did we warn about their
      // own deadline". It goes to `digest.ts`, the creator-directed channel —
      // verified email, push tokens and the in-app list — which this file's own
      // header named as the better long-term home. The console additionally
      // derives an app-wide banner from `GET /account/verification`, so a
      // creator with no contact point set up still cannot miss it.
      //
      // The saver leg is KEPT and is no longer load-bearing for the creator. It
      // is genuinely useful on its own terms: a lapse also removes that
      // creator's payout destinations from every fan-facing surface, and a fan
      // who saw a destination yesterday is owed the reason it is gone.
      const expiryIso = new Date(expiry).toISOString();
      if (expiry - nowMs <= EXPIRY_WARNING_MS) {
        out.warned += await noticeVerificationExpiring(db, row.passport_id, expiryIso, nowMs, () => new Date(nowMs).toISOString());
        await fanOutVerificationExpiring(db, row.passport_id, expiryIso, () => new Date(nowMs).toISOString());
      }
      complete(row);
      continue;
    }

    if (out.expired >= appendLimit) {
      // Budget spent. This row (and, in principle, any later one in this page)
      // is deferred, NOT completed — `lastCompleted` must not move past it, or
      // the cursor write below would strand it behind the cursor exactly as it
      // did before this fix.
      out.deferred += 1;
      budgetSpent = true;
      continue;
    }

    // `actor_kind: 'system'` is the whole reason ACTOR_POLICY names `system` for
    // this type, and the type is NOT in APPEND_ALLOWED — it is passed as a
    // governed type here, so this module is its one and only door.
    const draft: DraftEvent = {
      event_type: "verification_expired",
      subject: row.passport_id,
      actor_kind: "system",
      actor_id: "system:verification-sweep",
      // The deadline that was enforced, copied verbatim off the grant/renewal,
      // so the append-only row carries its own evidence.
      payload: { expired_at: new Date(expiry).toISOString() },
    } as DraftEvent;
    const r = await tryAppendEvents(db, [draft], () => new Date(nowMs).toISOString(), ["verification_expired"], [], ["system"]);
    if (r.ok) {
      out.expired += 1;
      complete(row);
      continue;
    }
    // A refusal here is a RACE, not a bug: an operator downgrade or a retraction
    // landed between the scan and the append, so the passport is no longer
    // verified and the fold rightly refuses. Never fatal — one racing creator
    // must not stop the sweep for everyone else.
    out.refused += 1;
    complete(row);
    // THE SUBJECT DOES NOT GO TO THE LOG (W3-10 item 1). `row.passport_id` was
    // interpolated into this line and `refusal.message` is free text composed by
    // the fold, which may name the creator; `refusal.kind` is the closed
    // two-value enum ("state" | "conflict") the allowlist is written for, and
    // `refused` is the running count this sweep already reports.
    logError({ cid: "unknown", hop: "cron", msg: "verification_expiry_sweep_refused", refused: out.refused, kind: r.refusal.kind });
  }

  // Advance (or wrap, or hold) the cursor.
  //   - Nothing deferred, short page: we reached the end of the table this pass
  //     → reset to the start so the next tick wraps around.
  //   - Nothing deferred, full page: resume strictly past the last row we saw.
  //   - Something deferred: hold the cursor at the last row actually completed
  //     (or, if the very FIRST row was deferred, do not move it at all) so the
  //     first deferred row is the first thing examined next tick, regardless of
  //     whether this page was short — a short page with a deferred tail is not
  //     the end of the table, it is exactly the row we must not skip past.
  const nowIsoForCursor = new Date(nowMs).toISOString();
  if (!budgetSpent && results.length < scanLimit) {
    await db.prepare("INSERT INTO sweep_cursor (name, cursor_seq, cursor_id, updated_at) VALUES ('verification', 0, '', ?) ON CONFLICT(name) DO UPDATE SET cursor_seq = 0, cursor_id = '', updated_at = excluded.updated_at").bind(nowIsoForCursor).run();
  } else if (lastCompleted) {
    await db.prepare("INSERT INTO sweep_cursor (name, cursor_seq, cursor_id, updated_at) VALUES ('verification', ?, ?, ?) ON CONFLICT(name) DO UPDATE SET cursor_seq = excluded.cursor_seq, cursor_id = excluded.cursor_id, updated_at = excluded.updated_at").bind(lastCompleted.updated_seq, lastCompleted.passport_id, nowIsoForCursor).run();
  }
  // else: the very first candidate was deferred — leave `sweep_cursor` untouched
  // so the next tick starts at exactly the same place and hits it first.

  return out;
}

/**
 * THE OTHER HALF OF THE EXPIRY: how a verification gets GRANTED AGAIN.
 *
 * `verification-sweep.ts` made `expires_at` load-bearing — it appends

 * that grants verification could only be filled by hand-editing D1.
 *
 * Every currently-verified creator was therefore on a fuse with no reset.
 *
 * WHAT THIS MODULE IS, AND WHAT IT DELIBERATELY IS NOT. It is the creator's end
 * of the renewal: a READ that tells them where they stand and how long they
 * have, and a REQUEST that puts a `verification` item in the ordinary review
 * queue. It grants nothing. The decision stays exactly where the original grant
 * lives — an operator approving a queue item (`review.ts approveItem`), which is
 * also why nothing here duplicates that path: the renewal reuses it, and
 * `approveItem` chooses `verification_renewed` over `verification_granted` by
 * looking at the passport's current state rather than at anything the requester
 * said.
 */

/**
 * The default term a creator's renewal request asks for: one year, the same
 * length the product has always issued. The operator can type a different date
 * in the queue before approving — this is the number the *request* carries so a
 * queue item is never created without a readable deadline (which is the exact
 * defect `review.ts` now refuses at approval time, and the reason the legacy
 * `unreadable_expiry` backlog exists).
 */
export const RENEWAL_TERM_MS = 365 * 24 * 60 * 60 * 1000;

/**
 * I3-1200 (DECIDED, I4-04-R4): after a REJECTED verification item the creator
 * may refile once this long has passed since the rejection. The refile door had
 * no cap; the cooldown rides the enqueue INSERT's own WHERE, never a read.
 */
export const REFILE_COOLDOWN_MS = 24 * 60 * 60 * 1000;

  override name = "VerificationError";

    message: string,
    public readonly status: 400 | 403 | 404 | 409 | 429 = 400,

export interface VerificationStatus {
  /** The folded passport state — what every other reader sees. */
  claim_state: "unclaimed" | "claimed" | "verified";
  /**
   * The deadline on the most recent `verification_granted`/`verification_renewed`,
   * when one exists AND parses. `null` covers three different worlds the
   * creator-facing copy has to keep apart: never verified, verified by a grant
   * that predates the queue, and a legacy grant carrying the empty-string
   * expiry the sweep refuses to act on (see `unreadable_expiry`).
   */
  expires_at: string | null;
  /** Whole days from `nowMs` to `expires_at`, rounded UP so "0 days" means the
   * deadline is today rather than "some time in the last 24 hours". Negative
   * after the deadline. `null` when `expires_at` is. */
  days_remaining: number | null;
  /** The `expired_at` of the latest `verification_expired`, if the ledger
   * records a lapse for this passport. This is what lets the console say
   * "your verification lapsed on <date>" instead of the present-tense
   * "unlocks once you're verified" — which is what a de-verified creator was
   * being told on seven screens. */
  expired_at: string | null;
  /** Live grant, inside `EXPIRY_WARNING_MS` of its deadline. The 14-day
   * heads-up the sweep was already computing and sending to the creator's
   * FANS. */
  expiring_soon: boolean;
  /** An open `verification` item already exists for this passport, so the
   * request button must say "already with an operator" rather than offering to
   * file a second one. */
  renewal_pending: boolean;
  /** I3-1453: the Founding Creator number, via `founding.ts foundingRank()` — null
   * unless the passport is live AND verified (the number stays burned, withheld). */
  founding_rank: number | null;
  /** I3-1200: when this passport's LATEST verification item was rejected, the
   * instant it was rejected; else null (a later request, open or approved,
   * supersedes it). The operator's reason text is internal and NOT served. */
  rejected_at: string | null;
  /** I3-1200: while `rejected_at` is inside `REFILE_COOLDOWN_MS`, the instant a
   * new request will be accepted; else null. */
  refile_after: string | null;
}

interface StatusRow {
  claim_state: string;
  grant_payload: string | null;
  expired_payload: string | null;
  open_request: number;
  rejected_at: string | null;
}

/** `expires_at` off a grant/renewal payload, in ms, or null when unreadable.
 * Same rule as the sweep and `standing.ts`: absent, empty and unparseable all
 * mean UNKNOWN, never "long past". */
function payloadInstant(payload: string | null, field: string): { iso: string; ms: number } | null {
  if (payload === null) return null;
  let raw: unknown;
  try {
    raw = (JSON.parse(payload) as Record<string, unknown>)[field];
  } catch {
    return null;
  }
  const ms = parseVerificationExpiry(raw);
  if (ms === null) return null;
  return { iso: String(raw), ms };
}

/** D-1112(8) (I34-FX1-CORE, RGC-N1): the grant's `expires_at`, read by standing.ts `grantDeadlineMs` — the one parse
 * every door applies (grantLapsed, grantLiveAtSql) — and served as that instant's canonical UTC ISO, so a deadline
 * stored as `2031-01-11` or with a `+05:30`/`+0530` offset is shown as the instant the doors enforce. */
function grantInstant(payload: string | null): { iso: string; ms: number } | null {
  if (payload === null) return null;
  let raw: unknown;
  try {
    raw = (JSON.parse(payload) as Record<string, unknown>).expires_at;
  } catch {
    return null;
  }
  const ms = grantDeadlineMs(raw);
  return ms === null ? null : { iso: new Date(ms).toISOString(), ms };
}

/**
 * Where this creator stands with their verification, in one statement.
 *
 * Keyed by the caller's OWN passport (the route is creator-session gated and
 * takes no subject), so there is no suppressed-vs-never-listed pair to leak —
 * see tests/invariants/suppression-oracles.test.ts.
 */
export async function verificationStatus(db: DbLike, passportId: string, nowMs: number): Promise<VerificationStatus | null> {
  const row = await db
    .prepare(
      `SELECT p.claim_state AS claim_state,
              (SELECT e.payload FROM trust_events e
                WHERE e.subject = p.passport_id
                  AND e.event_type IN ('verification_granted', 'verification_renewed')
                ${grantInForceOrderSql("e", instantSqlLiteral(nowMs))} LIMIT 1) AS grant_payload,
              (SELECT e2.payload FROM trust_events e2
                WHERE e2.subject = p.passport_id AND e2.event_type = 'verification_expired'
                ORDER BY e2.seq DESC LIMIT 1) AS expired_payload,
              EXISTS (SELECT 1 FROM review_queue q
                       WHERE q.subject = p.passport_id AND q.item_type = 'verification' AND q.state = 'open') AS open_request,
              (SELECT CASE WHEN q2.state = 'rejected' THEN q2.updated_at END FROM review_queue q2
                WHERE q2.subject = p.passport_id AND q2.item_type = 'verification'
                ORDER BY q2.created_at DESC, q2.item_id DESC LIMIT 1) AS rejected_at
         FROM passports p
        WHERE p.passport_id = ? AND p.retracted = 0`,
    )
    .bind(passportId)
    .first<StatusRow>();
  if (!row) return null;

  const grant = grantInstant(row.grant_payload);
  const lapse = payloadInstant(row.expired_payload, "expired_at");
  // I3-1768 (D-1085(1), I34-EXPIRY-A): a `verified` row whose latest grant or renewal has lapsed at `nowMs` (standing.ts
  // `grantLapsed`) reads as the expiry sweep's fold will leave it: `claimed`, lapsed at the grant's own deadline (the
  // `expired_at` verification-sweep.ts writes).
  const lapsed = row.claim_state === "verified" && grantLapsed(row.grant_payload, nowMs);
  const verified = row.claim_state === "verified";
  // A lapse is only news while it is the LATEST word. Once a renewal lands the
  // passport is verified again, and reporting the old `expired_at` would have
  // the console apologise for something that has been fixed.
  const expiredAt = lapsed && grant ? grant.iso : !verified && lapse ? lapse.iso : null;
  const days = grant === null ? null : Math.ceil((grant.ms - nowMs) / (24 * 60 * 60 * 1000));

  return {
    claim_state: (lapsed ? "claimed" : row.claim_state === "verified" || row.claim_state === "claimed" ? row.claim_state : "unclaimed") as VerificationStatus["claim_state"],
    expires_at: grant?.iso ?? null,
    days_remaining: days,
    expired_at: expiredAt,
    expiring_soon: verified && grant !== null && grant.ms > nowMs && grant.ms - nowMs <= EXPIRY_WARNING_MS,
    renewal_pending: Number(row.open_request) === 1,
    rejected_at: row.rejected_at ?? null,
    refile_after: refileAfter(row.rejected_at, nowMs),
    founding_rank: await foundingRank(db, passportId, new Date(nowMs).toISOString()),
  };
}

function refileAfter(rejectedAt: string | null | undefined, nowMs: number): string | null {
  const ms = rejectedAt ? Date.parse(rejectedAt) : Number.NaN;
  if (!Number.isFinite(ms) || ms + REFILE_COOLDOWN_MS <= nowMs) return null;
  return new Date(ms + REFILE_COOLDOWN_MS).toISOString();
}

/**
 * File a verification / re-verification request. The creator's own door onto
 * the review queue — the only thing in the product that has ever enqueued a
 * `verification` item.
 *
 * IT CHANGES NOTHING ABOUT THEIR BADGE. A request is a queue row and an audit
 * row; the passport is untouched until an operator approves, which is what the
 * copy on the console panel promises and what makes it safe to offer to a
 * still-verified creator (the "renew before the downgrade window" case) as well
 * as to a lapsed one.
 *
 * REFUSALS, and why each is a refusal rather than a silent no-op:
 *  - unclaimed → 403. Verification is granted on top of a proven claim
 *    (`verification_granted` requires `claimed`), so an unclaimed passport
 *    asking for one is a creator who needs the CLAIM flow, and saying so is
 *    more useful than queueing an item an operator must reject.
 *  - already open → 409, carrying the existing item id. Two open items for one
 *    subject is how an operator ends up granting twice.
 *  - rejected within `REFILE_COOLDOWN_MS` → 429 naming when to refile (I3-1200).
 *
 * I3-1936: both refusals ride the INSERT's own WHERE inside one batch (D1
 * serializes batches), so two concurrent requests cannot both queue an item;
 * the refusal message is re-derived AFTER the write from the rows that won.
 */
export async function requestVerification(
  db: DbLike,
  passportId: string,
  nowMs: number,
  now: () => string = () => new Date().toISOString(),
): Promise<{ item_id: string; expires_at: string; renewal: boolean }> {
  const row = await db
    .prepare("SELECT claim_state, retracted FROM passports WHERE passport_id = ?")
    .bind(passportId)
    .first<{ claim_state: string; retracted: number }>();
  if (!row) throw new VerificationError("no such passport", 404);
  if (Number(row.retracted) !== 0) throw new VerificationError("this passport is retracted — verification is not available", 403);
  if (row.claim_state === "unclaimed") throw new VerificationError("claim this passport before requesting verification", 403);

  // The requested deadline. Written by the SERVER from its own clock — the
  // creator does not get to name the term of their own grant — and always
  // parseable, so an approval can never be refused for the reason review.ts
  // refuses one.
  const expiresAt = new Date(nowMs + RENEWAL_TERM_MS).toISOString();
  const item_id = mintId("itm");
  const ts = now();
  const actor = `creator:${passportId}`;
  const payload = JSON.stringify({ expires_at: expiresAt, requested_by: "creator", requested_at: new Date(nowMs).toISOString() });
  const cooldownStart = new Date(nowMs - REFILE_COOLDOWN_MS).toISOString();
  await db.batch([
    db
      .prepare(
        `INSERT INTO review_queue (item_id, item_type, state, subject, payload, created_at, updated_at, import_batch_id)
         SELECT ?, 'verification', 'open', ?, ?, ?, ?, NULL
          WHERE NOT EXISTS (SELECT 1 FROM review_queue q WHERE q.subject = ? AND q.item_type = 'verification' AND q.state = 'open')
            AND NOT EXISTS (SELECT 1 FROM review_queue q WHERE q.subject = ? AND q.item_type = 'verification' AND q.state = 'rejected' AND q.updated_at >= ?)`,
      )
      .bind(item_id, passportId, payload, ts, ts, passportId, passportId, cooldownStart),
    // The row `review.ts auditStmt` writes for an enqueue, keyed on THIS item's
    // existence so a refused request leaves no line saying it was queued.
    db
      .prepare(
        `INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind)
         SELECT ?, 'enqueue:verification', ?, ?, ?, ? WHERE EXISTS (SELECT 1 FROM review_queue q WHERE q.item_id = ?)`,
      )
      .bind(actor, item_id, ts, item_id, auditActorKind(actor), item_id), // I34-W3-SEAM-P (D-1119(8), D-1120(6)): the review-queue audit copy keeps only the item id
  ]);
  const mine = await db.prepare("SELECT 1 AS x FROM review_queue WHERE item_id = ?").bind(item_id).first();
  if (!mine) {
    const open = await db
      .prepare("SELECT item_id FROM review_queue WHERE subject = ? AND item_type = 'verification' AND state = 'open' LIMIT 1")
      .bind(passportId)
      .first<{ item_id: string }>();
    if (open) throw new VerificationError(`a verification request is already with an operator (${open.item_id})`, 409);
    const rej = await db
      .prepare("SELECT MAX(updated_at) AS at FROM review_queue WHERE subject = ? AND item_type = 'verification' AND state = 'rejected'")
      .bind(passportId)
      .first<{ at: string | null }>();
    const after = refileAfter(rej?.at ?? null, nowMs);
    if (after) throw new VerificationError(`your last verification request was declined — you can file a new one after ${after}`, 429);
    throw new VerificationError("a verification request is already with an operator", 409);
  }
  return { item_id, expires_at: expiresAt, renewal: row.claim_state === "verified" };
}

import type { DbLike } from "./db.js";

/**
 * THE VERIFIED-MINT CHOKEPOINT (docs/VERIFIED-MINT.md).
 *
 * One rule, one read, one witness, repo-wide: **everything that creates
 * distributable value — a reverse-code pool, codes in one, Dhwaja vouchers,
 * one-time claim credentials, a release made publicly claimable — requires a
 * passport that is verified AT THE MOMENT OF THE ACTION.** Never inherited from
 * when the pool was made, the release registered, or the grant issued.
 *
 * WHY THIS FILE EXISTS AND IS NOT A HELPER. The rule was first written as an
 * `if` in a route handler, and the review that followed found the same hole
 * three times over: `POST /account/reverse-codes/pool` called
 * `isVerifiedPassport` and nothing else on that surface did; then
 * `issueCredentials` minted one-time codes with no verification read anywhere in
 * `credentials.ts`, on BOTH its doors; then `publishRelease` (since deleted,
 * D-0920(a)) made a release public with nothing but an ownership check. Each time, the gate was something
 * a handler had to REMEMBER, and the module it guarded knew nothing about it.
 *
 * So the gate is no longer remembered. `VerifiedCreator`'s brand is a
 * module-private `unique symbol`: no file outside this one can NAME it, so no
 * object literal outside this one can satisfy the type, and
 * `requireVerifiedCreator` is its only producer. Every function that mints takes
 * the witness WHERE IT USED TO TAKE A `passportId: string`. A mint path written
 * next year does not have to know this rule exists — it does not compile without
 * it.
 */

/** The brand. Module-private on purpose: exported, it could be spread into a
 * literal anywhere and the type would stop meaning anything. */
const VERIFIED = Symbol("dhwaja/verified-creator");

/**
 * Proof that a LIVE read of `passports` found this creator verified at the
 * moment of the action. Obtainable only from `requireVerifiedCreator`.
 */
export interface VerifiedCreator {
  readonly passport_id: string;
  readonly [VERIFIED]: true;
}

/**
 * The refusal a mint path owes a creator Dhwaja has not (or no longer has)
 * vetted. Its own class rather than each surface's, because the surfaces that
 * demand the witness — reverse codes, forward credentials, the release
 * storefront, drops — each carry their OWN 403s ("not your pool", "not your
 * release"), and a client has to be able to tell a STANDING answer from an
 * OWNERSHIP one. `apps/creator-console/src/codes.ts` mirrors this exact
 * sentence for that reason and pins it against this file.
 */
export class VerifiedCreatorError extends Error {
  override name = "VerifiedCreatorError";
  readonly status = 403 as const;
  constructor(message = "verified creators only") {
    super(message);

/**
 * THE ONE PLACE this repo asks whether a creator may mint. Reads `claim_state`,
 * `retracted` AND `request_suppressions` live, in ONE statement, and decides in
 * JS (deliberately: an `AND claim_state = 'verified'` in the WHERE returns the
 * same empty result for "not verified" and for "no such passport", and makes the
 * check invisible to a guard that drives a route through a DB double).
 *
 * ABSENT THE CONDITION: a missing row, a retracted passport, a SUPPRESSED
 * passport, or any `claim_state` other than `verified` is a 403 and nothing is
 * written. Fail closed — a passport this cannot read is not a passport it may
 * mint for.
 *
 * THE THIRD DOOR (review HIGH 1). This read used to ask two questions and there
 * are three. `request_suppressions` is the row `removal.ts` calls "the row that
 * already means 'gone' everywhere in this codebase": a delisted person 404s on
 * every public surface and is dropped from the registry. It was invisible here,
 * so a delisted creator's pool kept dispensing codes indefinitely — measured:
 * `poolOffer` returned null while `claimCode` handed out `EXTERNAL-CODE-1`, and
 * `issueCredentials` / `publishRelease` / `createPool` all still minted (a
 * measurement of that time; `publishRelease` is since deleted, D-0920(a)). Hiding
 * the offer PAGE stops nobody who holds the URL, and the URL is exactly how
 * these offers are distributed (the creator posts it themselves). "Gone" has to
 * mean the value stops being created, not that one page stops rendering.
 *
 * The refusal WORDING is deliberately the same "verified creators only" a
 * de-verified creator gets. It is only ever shown to the subject's own session
 * or to an agency that has already passed a grant check on that subject, so it
 * tells neither of them anything they did not already know — and a distinct
 * "you are delisted" sentence would be a new oracle on the surface this repo
 * works hardest to keep silent.
 *
 * De-verification is the population that matters and the one the old checks
 * missed: `correction`/`downgrade_verification` leaves `retracted = 0`, so a
 * `retracted`-only check waves through a creator whose badge has been withdrawn.
 * Verification is revocable, and that revocability is the entire reason the
 * badge means anything to a fan.
 *
 * NOT gated by this, deliberately (docs/VERIFIED-MINT.md): winding down
 * (`closePool`, `withdrawRelease`), own reads, honouring a credential a fan
 * ALREADY holds (`verifyCode`/`markRedeemed`, `redeemCredential`), and the whole
 * fan side.
 *
 * EXPIRY IS ENFORCED — through the LEDGER, not through this read (the "known
 * gap" this docstring used to describe is closed). `verification-sweep.ts`
 * appends `verification_expired` once a grant's `expires_at` has passed and the
 * fold walks `claim_state` back to `'claimed'`, so a lapsed creator fails the
 * `claim_state !== "verified"` test above with no per-call-site change — which is
 * why this function still asks its three questions and no fourth. Enforcing at
 * the fold rather than here is what makes every standing reader in the product
 * converge on the same instant instead of each re-deriving a deadline.
 *
 * The residual is a WINDOW, not a gap: between the moment a grant lapses and the
 * sweep tick that records it (bounded by the cron interval and the sweep's
 * per-run budget), the column still says `verified` and this read still passes.
 * `passportStanding` (standing.ts) covers that window for callers that need it,
 * by reading the latest grant-or-renewal's `expires_at` on the SAME statement and
 * answering `reverification_required`; its threshold is the sweep's exactly, so
 * the two can never disagree about a creator. Moving this chokepoint onto it is a
 * separate change — see docs/VERIFIED-MINT.md.
 */
export async function requireVerifiedCreator(db: DbLike, passportId: string): Promise<VerifiedCreator> {
  const p = await db
    .prepare(
      "SELECT p.claim_state, p.retracted, EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = p.passport_id) AS suppressed FROM passports p WHERE p.passport_id = ?",
    )
    .bind(passportId)
    .first<{ claim_state: string; retracted: number; suppressed: number }>();
  if (!p || p.retracted !== 0 || p.claim_state !== "verified" || Number(p.suppressed) !== 0) throw new VerifiedCreatorError("verified creators only");
  return { passport_id: passportId, [VERIFIED]: true };
}

import { PLATFORMS, isFetchableWebHost, isLinkBioVerifiableHost, profileUrl } from "@dhwaja/canonical";

/**
 * Bio-code verification fetch — SSRF-hardened by CONSTRUCTION.
 *
 * The target is never a user-supplied URL. It is built by profileUrl() from
 * the platform table (an allowlisted host) + the handle we already hold in the
 * account record. So the entire SSRF surface is "fetch one of a fixed set of
 * known public-profile hosts over HTTPS". On top of that, belt and suspenders:
 *   - scheme must be https
 *   - host must equal the platform's declared public host (re-checked here)
 *   - redirects are NOT followed (redirect: "manual") — any 3xx fails closed,
 *     so a platform open-redirect can't bounce us to an internal target
 *   - response is byte-capped and time-bounded
 *   - we only ever look for the exact code; nothing else from the page is
 *     trusted or parsed
 *
 * Workers has no raw sockets and no cloud metadata endpoint reachable via
 * fetch the way a VM does, but the host allowlist + no-redirect posture is the
 * correct defense regardless of runtime.
 */

export const MAX_FETCH_BYTES = 512 * 1024;
export const FETCH_TIMEOUT_MS = 8000;

export type VerifyOutcome =
  | { status: "verified" }
  | { status: "code_absent" }
  | { status: "manual" } // platform not auto-verifiable
  | { status: "unreachable"; reason: string };

export interface FetchTarget {
  url: string;
  host: string;
}

/**
 * The synthetic platform key a generic M3 `link` account carries (`platform:
 * "link"`, `id_kind: "url"`, `id`: the case-preserved norm). No curated platform
 * is keyed "link", so this value alone distinguishes a link account from a
 * Tier-1 handle/pid one everywhere the fetch path branches.
 */
export const LINK_PLATFORM = "link";

/**
 * Resolve the fetch target for an account, or null if it is not
 * bio-code-fetchable (manual-only platform, a pid, or an unfetchable link).
 *
 * A Tier-1 account (`handle`/`pid`) resolves through `profileUrl` — an
 * ALLOWLISTED platform host, the target never user-derived, and only for one of
 * the four platforms a human reviewed as owner-authored (`publicProfile`).
 *
 * A generic M3 `link` (`idKind: "url"`) resolves through `linkTarget` —
 * reconstructed from the account's stored norm, with its own SSRF hardening —
 * AND ONLY IF ITS HOST IS ON `LINK_BIO_VERIFIABLE_HOSTS`.
 *
 * THAT SECOND CONDITION IS THE FIX FOR A PASSPORT TAKEOVER, NOT A TIDY-UP. A
 * link used to be authorized for the fetch by `isFetchableWebHost` alone — a
 * predicate about SSRF (no IP literal, no internal name), which says NOTHING
 * about whether the page is owner-authored. Since the proof is `body.includes
 * (code)` anywhere in the first 512 KB, any page that renders third-party text
 * (a visitor comment, a wall post, a guestbook signature) let a STRANGER satisfy
 * the victim's own control proof: measured as `200 {"ok":true,"outcome":
 * "verified"}` plus a live creator session over somebody else's passport. The
 * curated list is the same remedy `publicProfile` already is for Tier-1, and it
 * ships EMPTY on purpose — see its header in `platforms.ts`.
 *
 * A link that does not clear the list gets `null`, which every caller already
 * reads as "this account cannot be auto-verified": `startClaim` /
 * `startAccountProof` refuse with the manual-verification error (and
 * `manual-claims.ts` now accepts `id_kind: "url"`, so that refusal points at a
 * door that exists), and `removal.ts` falls to its decoy path exactly as it does
 * for a login-walled platform.
 */
export function verificationTarget(platform: string, idKind: "handle" | "pid" | "url" | "domain", id: string): FetchTarget | null {
  if (idKind === "url") {
    const target = linkTarget(id);
    return target && isLinkBioVerifiableHost(target.host) ? target : null;
  }
  if (idKind === "domain") return null; // a domain proves control via DNS, never a fetched page
  return profileUrl(platform, idKind, id);
}

/**
 * Build the fetch target for a generic `link` account from its stored norm
 * (`<host>[:port]<path>[?query]`, host already lowercased, path/query
 * case-preserved). SSRF-hardened by CONSTRUCTION, the link analogue of
 * `profileUrl`:
 *   - forced to https (the norm carries no scheme; we never fetch http);
 *   - re-parsed so the check runs against the parser's OWN normalized host, not
 *     the raw string — no octal/hex/integer-IPv4 or case trick survives;
 *   - credentials and an explicit port are refused (a proof target is a plain
 *     public page; a port is the port-scan surface);
 *   - the host must pass `isFetchableWebHost` (no IP literal, no single-label or
 *     internal name) — the same predicate `canonicalizeLink` already applied at
 *     seed time, re-applied here because this function must not trust that its
 *     input came through the canonicalizer.
 * `verifyCodePresent` re-validates all of this a third time on the constructed
 * URL; the belt and both suspenders are deliberate on a fetch whose host is not
 * on a fixed allowlist.
 */
export function linkTarget(norm: string): FetchTarget | null {
  let url: URL;
  try {
    url = new URL(`https://${norm}`);
  } catch {
    return null;
  }
  if (url.protocol !== "https:") return null;
  if (url.username || url.password || url.port) return null;
  if (!isFetchableWebHost(url.hostname)) return null;
  return { url: url.toString(), host: url.hostname };
}

function hostAllowed(platformKey: string, host: string): boolean {

/**
 * Fetch the constructed profile URL and report whether `code` appears in it.
 * `fetchImpl` is injectable for tests. Fails closed on anything unexpected.
 */
export async function verifyCodePresent(
  platformKey: string,
  target: FetchTarget,
  code: string,
  fetchImpl: typeof fetch = fetch,
): Promise<VerifyOutcome> {
  let url: URL;
  try {
    url = new URL(target.url);
  } catch {
    return { status: "unreachable", reason: "bad target url" };
  }
  // Re-verify the target independently of how it was constructed. The two host
  // rules are deliberately SEPARATE spellings, not one merged check: a Tier-1
  // platform fetch is authorized by an exact match against the platform's
  // reviewed `publicProfile` host (the target was never user-derived), while a
  // generic `link` is authorized by its own reviewed host list. Everything else
  // — https, no credentials/port, no redirect, the byte/time cap below — is
  // shared and applies to both.
  //
  // THE LINK BRANCH CARRIES BOTH OF ITS PREDICATES, and they answer two
  // different questions: `isFetchableWebHost` is about SSRF (may we send a
  // request there at all — no IP literal, no internal name), and
  // `isLinkBioVerifiableHost` is about PROOF SEMANTICS (does a substring on that
  // page mean anything). Conflating them is what let a stranger satisfy a
  // victim's control proof from a comment box; see `verificationTarget`. This is
  // the THIRD place the link host is checked (canonicalizer, `linkTarget`, here)
  // — deliberate belt and both suspenders on a fetch whose host is not fixed by
  // the platform table, and the only one of the three that stands between a
  // forged `FetchTarget` and an outbound request.
  if (url.protocol !== "https:") return { status: "unreachable", reason: "non-https" };
  if (url.username || url.password || url.port) return { status: "unreachable", reason: "url has credentials or port" };
  if (platformKey === LINK_PLATFORM) {
    if (!isFetchableWebHost(url.hostname)) return { status: "unreachable", reason: "link host not fetchable" };
    if (!isLinkBioVerifiableHost(url.hostname)) return { status: "unreachable", reason: "link host not bio-code verifiable" };
  } else if (!hostAllowed(platformKey, url.hostname)) {
    return { status: "unreachable", reason: "host not allowlisted" };
  }

  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), FETCH_TIMEOUT_MS);
  let res: Response;
  try {
    res = await fetchImpl(url.toString(), {
      method: "GET",
      redirect: "manual", // any 3xx → fail closed, never chase into an internal host
      signal: controller.signal,
      headers: { "User-Agent": "DhwajaVerify/1.0 (+https://dhwaja)" },
    });
  } catch (e) {
    clearTimeout(timer);
    return { status: "unreachable", reason: e instanceof Error && e.name === "AbortError" ? "timeout" : "fetch failed" };
  }

  // Keep the abort timer ARMED across the body read (cleared in finally). The
  // FETCH_TIMEOUT bounds the WHOLE exchange, not just the headers — otherwise a
  // slow-drip body that stays under MAX_FETCH_BYTES holds the subrequest open
  // indefinitely (a read-side slowloris). If the timer fires mid-read, the
  // reader/`res.text()` reject and we return unreachable/timeout.
  try {
    if (res.status >= 300 && res.status < 400) return { status: "unreachable", reason: "redirect refused" };
    if (!res.ok) return { status: "unreachable", reason: `http ${res.status}` };

    // Bounded read: stop after MAX_FETCH_BYTES; search incrementally so we don't
    // hold an unbounded page in memory.
    const body = res.body;
    if (!body) {
      const text = (await res.text()).slice(0, MAX_FETCH_BYTES);
      return text.includes(code) ? { status: "verified" } : { status: "code_absent" };
    }
    const reader = body.getReader();
    const decoder = new TextDecoder();
    let acc = "";
    let total = 0;
    try {
      for (;;) {
        const { done, value } = await reader.read();
        if (done) break;
        const chunk = value ?? new Uint8Array();
        // Bound EACH chunk to the REMAINING budget before decoding/searching it —
        // never let bytes past the permitted MAX_FETCH_BYTES prefix influence the
        // decision, and never let it matter how the network happened to slice the
        // response into chunks. `remaining` is computed BEFORE this chunk is
        // counted, so the boundary always falls in the same place in the
        // underlying byte stream regardless of chunk sizes.
        const remaining = MAX_FETCH_BYTES - total;
        if (remaining <= 0) {
          await reader.cancel().catch(() => {});
          return { status: "code_absent" }; // budget already exhausted by a prior chunk
        }
        const bounded = chunk.byteLength > remaining ? chunk.subarray(0, remaining) : chunk;
        total += bounded.byteLength;
        acc = (acc.slice(-code.length) + decoder.decode(bounded, { stream: true })); // keep a small overlap so a code split across chunks still matches
        if (acc.includes(code)) {
          await reader.cancel().catch(() => {});
          return { status: "verified" };
        }
        if (total >= MAX_FETCH_BYTES) {
          await reader.cancel().catch(() => {});
          return { status: "code_absent" }; // searched exactly the permitted prefix, not found
        }
      }
    } catch (e) {
      return { status: "unreachable", reason: e instanceof Error && e.name === "AbortError" ? "timeout" : "read failed" };
    }
    return acc.includes(code) ? { status: "verified" } : { status: "code_absent" };
  } catch (e) {
    // res.text() abort/read failure on the no-body path.
    return { status: "unreachable", reason: e instanceof Error && e.name === "AbortError" ? "timeout" : "read failed" };
  } finally {
    clearTimeout(timer);
  }
}

/**
 * S-CW3: BOND & FUSION — collecting deeply, converted into game power.
 *
 * A fan's Bond with a creator (0..+10) IS the in-match loyalty bonus. It is
 * raised by COMMITTING surplus duplicate card instances of that same creator
 * (fusion — the economy's drain), decays on a pure-time monthly clock, and
 * earns a permanent floor: +1 each time the Bond reaches +10, capped at +3.
 * Decay standing between max-outs is what makes a +3 floor mean "proven,
 * multi-season devotion" — money can rent +10; only time-plus-loyalty makes
 * it permanent.
 *
 * STATE IS DERIVED, NEVER STORED. `bond_commits` is the append-only source;
 * `computeBond` replays it through the cost curve + decay + floor at read
 * time (the codebase's lazy idiom — no cron, no mass monthly write, nothing
 * to drift). The replay is a pure function of (commits, now).
 *
 * THE HARD LINE (design doc §4): fusion touches ONLY the fan's Bond — never
 * the creator's seven Power Ratings. Nothing in this module reads or writes
 * ratings state.
 *
 * SUPPRESSION DISCIPLINE: nothing here checks whether a passport EXISTS or
 * its standing. Every predicate is scoped to the caller's own instances, so
 * a suppressed creator, a never-listed id, and a creator the fan simply owns
 * no cards of all fail identically ("no such instances") — no oracle.
 */

// ── Dials (tunable constants; changing the curve is a balance decision) ──
export const BOND_MAX = 10;
export const FLOOR_MAX = 3;
/** Copies to climb FROM level i TO i+1. Sum = 73 editions to max — deep but
 * reachable for a true superfan of ONE creator; brutal across many (the
 * anti-whale shape the design doc wants). */
export const BOND_COST_CURVE: readonly number[] = Object.freeze([1, 1, 2, 3, 4, 6, 8, 12, 16, 20]);
/** Pure-time decay: one level per elapsed period, floor-clamped. 30 days —
 * "monthly" as a fixed period so the replay is timezone/calendar-proof. */
export const DECAY_PERIOD_MS = 30 * 24 * 60 * 60 * 1000;
export const DECAY_LEVELS_PER_PERIOD = 1;
/** Commit ops per fan per day (each op can carry many copies). */
export const BOND_COMMITS_PER_DAY = 60;
/** I21-Q83 (D-1030 (c)): copies one account may commit to ONE creator in one UTC day, enforced inside commitToBond's
 * atomic validator. Jason tunes it from the simulator later. */
export const BOND_COPIES_PER_CREATOR_PER_DAY = 10;
/** Instances one commit op may burn. D1-BOUNDED: the atomic validator binds
 * N ids + 19 fixed params in ONE statement and D1 rejects >100 bound
 * parameters, so this cap is a correctness bound, not taste (80 + 19 = 99).
 * The fidelity-suite IN-builder guard pins this reasoning. */
export const MAX_COPIES_PER_COMMIT = 80;
/** AA-05: the docblock above asserts 80 + 19 = 99 <= 100 — enforce it in code
 * so raising the constant without re-deriving the arithmetic fails LOUDLY at
 * import time, in every environment, rather than staying green until a real
 * D1 commit of 90+ instances 500s with nothing burned. */
if (19 + MAX_COPIES_PER_COMMIT > 100) throw new Error("MAX_COPIES_PER_COMMIT raised past D1's 100-bound-parameter ceiling (19 fixed params + N ids)");

/** D-1059(2) (RGM-N8): copies that still raise `s` toward `max` (a match uses Bond clamped to its dials' bondMax,
 * war-match.ts matchBond) — 0 at or above it. Pure. */
export function copiesToBondMax(s: Pick<BondState, "level" | "progress">, max: number): number {
  let need = 0;
  for (let l = s.level; l < Math.min(max, BOND_MAX); l++) need += BOND_COST_CURVE[l]!;
  return need - (s.level < max ? s.progress : 0);
}

export interface BondState {
  passport_id: string;
  /** The number added in a match when fielding this creator (floor ≤ level ≤ 10). */
  level: number;
  /** Earned permanent floor (0..3). */
  floor: number;
  /** Copies committed toward the NEXT level (cleared by decay). */
  progress: number;
  /** Copies still needed for the next level; null at BOND_MAX. */
  next_cost: number | null;
  /** Lifetime copies committed past max — the prestige counter. */
  overflow: number;
  /** Lifetime copies committed. */
  lifetime_copies: number;
  last_commit_at: string | null;
}

interface CommitRow {
  copies: number;
  at: string;
}

/**
 * Replay a fan's commits for one creator into the current Bond. Pure and
 * deterministic: same rows + same `nowMs` ⇒ same state, forever (the same
 * property the match resolver has, so a replayed match can re-derive the
 * exact Bond it used).
 *
 * Rules, in order, at each commit and finally at `now`:
 *  1. DECAY the whole periods elapsed since the previous event: level drops
 *     `DECAY_LEVELS_PER_PERIOD` per period, never below the floor; any decay
 *     also clears partial progress (a cooled Bond restarts its current step).
 *  2. ADD the commit's copies to progress; climb while progress covers the
 *     next step's cost.
 *  3. On each TRANSITION into BOND_MAX from below, the floor rises by 1 (cap
 *     FLOOR_MAX). Sitting at max never re-earns; you must cool and re-climb.
 *  4. Copies landing while at BOND_MAX are overflow (prestige), not progress.
 */
export function computeBond(passportId: string, rows: CommitRow[], nowMs: number): BondState {
  let level = 0;
  let floor = 0;
  let progress = 0;
  let overflow = 0;
  let lifetime = 0;
  let prevMs: number | null = null;
  let lastAt: string | null = null;

  const decayTo = (tMs: number) => {
    if (prevMs === null) return;
    const periods = Math.floor((tMs - prevMs) / DECAY_PERIOD_MS);
    if (periods > 0) {
      const next = Math.max(floor, level - periods * DECAY_LEVELS_PER_PERIOD);
      // I3-1500: only a level that actually FELL clears the partial step; a floor-clamped period changes nothing.
      if (next !== level) progress = 0;
      level = next;
      // The decay reference advances by the WHOLE periods consumed, so the
      // remainder keeps counting toward the next period (no free reset).
      prevMs = prevMs + periods * DECAY_PERIOD_MS;
    }
  };

  for (const r of rows) {
    const tMs = Date.parse(r.at);
    decayTo(tMs);
    // PURE-TIME decay (review F2): a commit must never advance the decay
    // anchor — only decayTo's whole consumed periods do, so the sub-period
    // remainder is NEVER forgiven by activity. (The old `prevMs = tMs` here
    // let an overflow commit right after a boundary delay the next decay by
    // a full period — a held level is match power, so that was competitive
    // fraud, not a cosmetic drift.) The first commit initializes the anchor.
    if (prevMs === null) prevMs = tMs;
    lastAt = r.at;
    lifetime += r.copies;
    let copies = r.copies;
    while (copies > 0) {
      if (level >= BOND_MAX) {
        overflow += copies;
        copies = 0;
        break;
      }
      const cost = BOND_COST_CURVE[level]!;
      const need = cost - progress;
      if (copies >= need) {
        copies -= need;
        progress = 0;
        level++;
        if (level === BOND_MAX) floor = Math.min(FLOOR_MAX, floor + 1);
      } else {
        progress += copies;
        copies = 0;
      }
    }
  }
  decayTo(nowMs);

  return {
    passport_id: passportId,
    level,
    floor,
    progress,
    next_cost: level >= BOND_MAX ? null : BOND_COST_CURVE[level]! - progress,
    overflow,
    lifetime_copies: lifetime,
    last_commit_at: lastAt,
  };
}

const PSP_RE = /^psp_[0-9a-f]{16}$/;
const CI_RE = /^ci_[0-9a-f]{20}$/;
const BC_RE = /^bc_[0-9a-f]{20}$/;
const CIL_RE = /^cil_[0-9a-f]{20}$/;

/**
 * Commit (burn) specific surplus instances of one creator into Bond — ONE
 * atomic batch, all-or-nothing by construction:
 *
 *  - each instance is consumed by a guarded UPDATE (own, that creator,
 *    unbound, living, and never gifted or transferred TO this account —
 *    I21-Q83: only cards minted to the owner fuel Bond) — a failed guard
 *    makes that UPDATE a no-op;
 *  - per-instance 'consumed' provenance rows are predicated on the consume
 *    having landed at THIS op's timestamp;
 *  - the final `bond_commits` INSERT selects its PRIMARY KEY through a CASE
 *    that yields NULL unless (a) every one of the N instances was consumed
 *    at this op's timestamp AND (b) at least one living copy of the creator
 *    REMAINS (the sacred first copy, checked post-burn, atomically) AND (c)
 *    this op keeps the account's copies to this creator in this UTC day within
 *    BOND_COPIES_PER_CREATOR_PER_DAY (I21-Q83). A NULL
 *    PK violates NOT NULL → the WHOLE batch rolls back — so a partial burn
 *    can never survive and the sacred copy can never be burned, even by a
 *    racing pair of commits (D1 serializes batches; the loser's post-state
 *    check fails and its batch rolls back).
 *
 * Returns the new BondState on success. Throws WarInstanceError("commit
 * refused") when the batch rolled back — the caller can re-read and retry.
 */
export async function commitToBond(
  db: DbLike,
  accountId: string,
  passportId: string,
  instanceIds: string[],
  mintCommitId: () => string,
  mintLogId: () => string,
  now: () => string = () => new Date().toISOString(),
): Promise<BondState> {
  if (!PSP_RE.test(passportId)) throw new WarInstanceError("bad passport id");
  const n = instanceIds.length;
  if (n < 1 || n > MAX_COPIES_PER_COMMIT) throw new WarInstanceError(`commit 1..${MAX_COPIES_PER_COMMIT} instances`);
  const uniq = new Set(instanceIds);
  if (uniq.size !== n) throw new WarInstanceError("duplicate instance in commit");
  for (const id of instanceIds) if (!CI_RE.test(id)) throw new WarInstanceError("bad instance id shape");
  const commitId = mintCommitId();
  if (!BC_RE.test(commitId)) throw new WarInstanceError("bad commit id shape", 500);
  const ts = now();
  if (!Number.isFinite(Date.parse(ts))) throw new WarInstanceError("bad timestamp", 500);
  const utcDay = new Date(Date.parse(ts)).toISOString().slice(0, 10);

  // D-1059(2) (RGM-N8): copies past the CURRENT dial's bondMax change no match (matchBond clamps), so they are refused
  // with the reason, never burned. The Bond is read here and the batch's validator re-binds the copies it was read at
  // plus the room left (INV-058): a commit landing in between that would take the two past the maximum rolls this one
  // back; two that fit together both land.
  const { results: prior } = await db
    .prepare("SELECT copies, at FROM bond_commits WHERE account_id = ? AND passport_id = ? ORDER BY at, commit_id")
    .bind(accountId, passportId)
    .all<{ copies: number; at: string }>();
  const room = copiesToBondMax(computeBond(passportId, prior, Date.parse(ts)), DEFAULT_DIALS.bondMax);
  const priorCopies = prior.reduce((t, r) => t + r.copies, 0);
  if (n > room) {
    // I30-FX-WAR: the Bond-maximum refusal (approved, sheet-2 W63, D-1365).
    throw new WarInstanceError(
      room === 0
        ? `commit refused: your Bond with this creator is already at the match maximum of ${DEFAULT_DIALS.bondMax}, so more copies would not count`
        : `commit refused: only ${room} more ${room === 1 ? "copy counts" : "copies count"} before your Bond with this creator reaches the match maximum of ${DEFAULT_DIALS.bondMax}`,
      409,
    );
  }

  const stmts = [];
  for (const id of instanceIds) {
    // Review F1: the burn is attributed to THIS op's unique commit id, never
    // the wall clock — two same-millisecond commits share a ts, and a
    // ts-counting validator let the later op count the earlier op's burns as
    // its own (bond inflation with zero fresh property burned). commitId is
    // unique + unguessable, so consumed_by cannot collide.
    stmts.push(
      db
        .prepare(
          // escrowed_by IS NULL (S-CW5; since QUEUE #236 only a LEGACY pre-removal
          // stake or a gift-offer hold can be escrowed): a card staked as a match ante is off
          // the table — burning it as Bond fuel while the match settles would
          // double-spend the same property. Same refusal as every other guard.
          // F4b review: a card under an open RECOVERY HOLD cannot be burned
          // either — a hold exists so the disputed item survives until it is
          // returned, and Bond fuel is the one destruction path erasure does
          // not cover. Same generic refusal.
          `UPDATE card_instances SET consumed_at = ?, consumed_by = ?
            WHERE instance_id = ? AND owner_id = ? AND passport_id = ? AND bound = 0 AND consumed_at IS NULL AND escrowed_by IS NULL
              AND NOT EXISTS (SELECT 1 FROM collectible_holds h WHERE h.kind = 'card' AND h.item_id = card_instances.instance_id AND h.state = 'open' AND h.expires_at > ?)
              AND NOT EXISTS (SELECT 1 FROM card_instance_log l WHERE l.instance_id = card_instances.instance_id AND l.event IN ('gifted','transferred') AND l.to_owner = ?)`,
        )
        .bind(ts, commitId, id, accountId, passportId, ts, accountId),
    );
    const logId = mintLogId();
    if (!CIL_RE.test(logId)) throw new WarInstanceError("bad log id shape", 500);
    stmts.push(
      db
        .prepare(
          `INSERT INTO card_instance_log (log_id, instance_id, event, from_owner, to_owner, detail, at)
           SELECT ?, ?, 'consumed', ?, NULL, 'bond', ?
            WHERE EXISTS (SELECT 1 FROM card_instances WHERE instance_id = ? AND owner_id = ? AND consumed_by = ?)`,
        )
        .bind(logId, id, accountId, ts, id, accountId, commitId),
    );
    // QUEUE #172 P4: a card burned as Bond fuel is the same "item leaves its
    // owner" shape as offerGift/recoverCollectible/war-match settlement -- an
    // open gift offer on this card must void, same predicate every door uses.
    stmts.push(voidOrphanedOffersStatement(db, "card", id, ts));
  }
  // The validator/committer. CASE → NULL PK → NOT NULL violation → rollback.
  const placeholders = instanceIds.map(() => "?").join(",");
  stmts.push(
    db
      .prepare(
        `INSERT INTO bond_commits (commit_id, account_id, passport_id, copies, at)
         SELECT CASE
                  WHEN (SELECT COUNT(*) FROM card_instances
                         WHERE owner_id = ? AND passport_id = ? AND consumed_by = ?
                           AND instance_id IN (${placeholders})) = ?
                   AND (SELECT COUNT(*) FROM card_instances
                         WHERE owner_id = ? AND passport_id = ? AND consumed_at IS NULL AND escrowed_by IS NULL) >= 1
                   AND (SELECT COALESCE(SUM(copies), 0) FROM bond_commits
                         WHERE account_id = ? AND passport_id = ? AND substr(at, 1, 10) = ?) + ? <= ${BOND_COPIES_PER_CREATOR_PER_DAY}
                   AND (SELECT COALESCE(SUM(copies), 0) FROM bond_commits WHERE account_id = ? AND passport_id = ?) + ? <= ?
                  THEN ? ELSE NULL END,
                ?, ?, ?, ?`,
      )
      .bind(accountId, passportId, commitId, ...instanceIds, n, accountId, passportId, accountId, passportId, utcDay, n, accountId, passportId, n, priorCopies + room, commitId, accountId, passportId, n, ts),
  );

  try {
    await db.batch(stmts);
  } catch {
    // I3-2170: an exception is not a verdict — a transport failure AFTER the
    // commit throws too. This op's commit_id is unique, so its row existing is
    // the proof the batch committed; a read-back failure propagates (unknown,
    // never a false 409).
    const landed = await db.prepare("SELECT 1 AS x FROM bond_commits WHERE commit_id = ? AND account_id = ?").bind(commitId, accountId).first();
    if (landed) return getBond(db, accountId, passportId, Date.parse(ts));
    // NULL-PK rollback: an instance failed its guards (not yours / bound /
    // already consumed / wrong creator) or the burn would leave no living
    // copy. Deliberately ONE indistinguishable refusal — the caller learns
    // nothing about ids they don't own.
    // I21-Q83: the sentence names the two new refusals too (approved, sheet-2 W63, D-1365) without saying which one fired.
    throw new WarInstanceError(`commit refused: instances must be your own living unbound copies of this creator that you bought or opened (gifted or traded cards do not count), one copy must remain, and at most ${BOND_COPIES_PER_CREATOR_PER_DAY} copies of one creator can be committed a day`, 409);
  }
  return getBond(db, accountId, passportId, Date.parse(ts));
}

/** Current Bond with one creator (lazy replay; no writes). */
export async function getBond(db: DbLike, accountId: string, passportId: string, nowMs: number): Promise<BondState> {
  if (!PSP_RE.test(passportId)) throw new WarInstanceError("bad passport id");
  const { results } = await db
    .prepare("SELECT copies, at FROM bond_commits WHERE account_id = ? AND passport_id = ? ORDER BY at, commit_id")
    .bind(accountId, passportId)
    .all<{ copies: number; at: string }>();
  return computeBond(passportId, results, nowMs);
}

/**
 * S-CW5: Bond LEVELS for one fan across a roster's creators, in ONE query —
 * the match resolver snapshots up to 50 bonds per side and per-creator
 * `getBond` reads would cost 50 round trips. Same replay (`computeBond`) fed
 * from one grouped read; `getBond` remains the authoritative single read and
 * this must agree with it by construction (same rows, same fold).
 * D1-BOUNDED: passportIds.length ≤ 50 keeps binds ≤ 51.
 */
export async function bondLevels(db: DbLike, accountId: string, passportIds: readonly string[], nowMs: number): Promise<Map<string, number>> {
  const out = new Map<string, number>();
  if (passportIds.length === 0) return out;
  if (passportIds.length > 50) throw new WarInstanceError("too many passports for one bond read", 500);
  for (const p of passportIds) {
    if (!PSP_RE.test(p)) throw new WarInstanceError("bad passport id");
    out.set(p, 0);
  }
  const placeholders = passportIds.map(() => "?").join(",");
  const { results } = await db
    .prepare(`SELECT passport_id, copies, at FROM bond_commits WHERE account_id = ? AND passport_id IN (${placeholders}) ORDER BY at, commit_id`)
    .bind(accountId, ...passportIds)
    .all<{ passport_id: string; copies: number; at: string }>();
  const byPassport = new Map<string, { copies: number; at: string }[]>();
  for (const r of results) {
    const rows = byPassport.get(r.passport_id) ?? [];
    rows.push({ copies: r.copies, at: r.at });
    byPassport.set(r.passport_id, rows);
  }
  for (const [psp, rows] of byPassport) out.set(psp, computeBond(psp, rows, nowMs).level);
  return out;
}

/** Distinct creators a fan may hold Bonds with, shown at once. */
export const MAX_BONDS_LISTED = 100;

/**
 * Every Bond the fan holds. Two-step so the replay is EXACT (review F4): a
 * flat row LIMIT truncated mid-passport and silently understated the last
 * creators alphabetically. Instead: a bounded set of distinct creators, then
 * the full, unbounded commit history per creator — `getBond` semantics for
 * every row shown. `getBond` remains the authoritative single-creator read.
 */
export async function listBonds(db: DbLike, accountId: string, nowMs: number): Promise<BondState[]> {
  const { results: passports } = await db
    .prepare(`SELECT DISTINCT passport_id FROM bond_commits WHERE account_id = ? ORDER BY passport_id LIMIT ${MAX_BONDS_LISTED}`)
    .bind(accountId)
    .all<{ passport_id: string }>();
  const out: BondState[] = [];
  for (const p of passports) out.push(await getBond(db, accountId, p.passport_id, nowMs));
  out.sort((a, b) => b.level - a.level || b.lifetime_copies - a.lifetime_copies || (a.passport_id < b.passport_id ? -1 : 1));
  return out;

import { inChunks, type DbLike } from "./db.js";
import { CLAIM_STATE_AT_SQL } from "./standing.js";

/**
 * S-CW4: CREATOR WAR consent — the derive-from-log read for `war_opt_in` /
 * `war_opt_out`, and the ONE place their precedence rule lives.
 *
 * The rule (plan S-CW4, signed off): a creator's OWN opt-out always wins over
 * an agency opt-in. Concretely, replaying the subject's war events in seq
 * order:
 *
 *   - `war_opt_out` always takes effect, whoever appended it (an agency
 *     opting a creator OUT is the safe direction and is never blocked);
 *     when the actor was the creator's passkey it also sets the CREATOR LOCK.
 *   - `war_opt_in` by `creator_passkey` takes effect and CLEARS the lock —
 *     the creator's own signature is the only key.
 *   - `war_opt_in` by anyone else (an agency) takes effect ONLY while the
 *     lock is off. While locked it is ignored at read time — and the agency
 *     door (agencies.ts appendAgencyAction) also refuses to append it, so the
 *     ledger isn't littered with rows that read as consent but aren't.
 *
 * Read-time enforcement is the authoritative half: even if a locked agency
 * opt-in row somehow reached the log (a future door forgetting the check),
 * the derived state still says opted-out. Fail closed: no rows → never opted
 * in → not in the game.
 *
 * SUPPRESSION DISCIPLINE: `warOptState` answers for any psp id without
 * checking the passport table — a never-listed id and a real non-opted
 * creator return the identical zero state, so callers can safely expose the
 * BOOLEAN without opening an existence oracle. (Appending the events at all
 * requires a live claimed+ passport — events.ts enforces that.)
 */

export interface WarOptState {
  /** In the game right now? */
  opted_in: boolean;
  /** Creator's own opt-out stands; only the creator's passkey can re-enter. */
  creator_locked: boolean;
  /** Provenance of the event that produced the current opted_in value:
   * "creator" | "agency" | "operator" (recovery's opt-out sweep) | null
   * (never any war event). */
  decided_by: "creator" | "agency" | "operator" | null;
  /** created_at of that deciding event, or null. */
  decided_at: string | null;
}

const ZERO: WarOptState = { opted_in: false, creator_locked: false, decided_by: null, decided_at: null };

/** Replay one passport's war consent state from the ledger. */
export async function warOptState(db: DbLike, passportId: string): Promise<WarOptState> {
  if (!/^psp_[0-9a-z]{8,64}$/.test(passportId)) return { ...ZERO };
  const { results } = await db
    .prepare(
      "SELECT event_type, actor_kind, created_at FROM trust_events WHERE subject = ? AND event_type IN ('war_opt_in','war_opt_out') ORDER BY seq ASC",
    )
    .bind(passportId)
    .all<{ event_type: string; actor_kind: string; created_at: string }>();
  return replayRows(results);
}

/** The ONE replay both reads share — `warOptState` and the bulk gate must
 * never be able to disagree, so there is exactly one implementation. Rows
 * must arrive in seq order. */
function replayRows(results: readonly { event_type: string; actor_kind: string; created_at: string }[]): WarOptState {
  const state: WarOptState = { ...ZERO };
  for (const r of results) {
    const byCreator = r.actor_kind === "creator_passkey";
    const who = byCreator ? "creator" : r.actor_kind === "operator" ? "operator" : "agency";
    if (r.event_type === "war_opt_out") {
      state.opted_in = false;
      // The lock records that the CREATOR said no. An agency or operator
      // opt-out doesn't lock (the creator never spoke — and after a recovery
      // sweep the creator must be able to re-enter with a fresh passkey), but
      // neither unlocks — a lock only clears on the creator's own opt-in.
      if (byCreator) state.creator_locked = true;
      state.decided_by = who;
      state.decided_at = r.created_at;
    } else {
      // war_opt_in
      if (byCreator) {
        state.opted_in = true;
        state.creator_locked = false;
        state.decided_by = "creator";
        state.decided_at = r.created_at;
      } else if (!state.creator_locked) {
        state.opted_in = true;
        state.decided_by = "agency";
        state.decided_at = r.created_at;
      }
      // agency opt-in while creator-locked: ignored (see docblock).
    }
  }
  return state;
}

/**
 * THE GATE every war mint and matchmaking path must check — fail closed
 * (false for anything unknown).
 *
 * Consent alone is not enough (review S-CW4/F1): a passport can leave
 * claimed+ AFTER opting in (adjudicated downgrade, retraction/takedown), and
 * in those states the person the cards depict is no longer a proven,
 * standing participant — so the game freezes for them the INSTANT standing
 * lapses, with no one needing to append anything. The consent record itself
 * survives (re-claiming restores the game without re-consent — the creator's
 * recorded intent still stands); only the gate goes cold. warOptState stays a
 * pure replay so the state read never becomes an existence oracle.
 *
 * I3-1768 (D-1085(1), I34-EXPIRY-A): the claim state is read at `nowIso` (standing.ts CLAIM_STATE_AT_SQL) in all
 * three gates here, so a grant lapsed by then is not `verified` before the expiry sweep folds it back.
 */
export async function warOptedIn(db: DbLike, passportId: string, nowIso: string = new Date().toISOString()): Promise<boolean> {
  if (!(await warOptState(db, passportId)).opted_in) return false;
  const p = await db
    .prepare(
      `SELECT ${CLAIM_STATE_AT_SQL} AS claim_state, p.retracted AS retracted FROM passports p
        WHERE p.passport_id = ?
          AND NOT EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = p.passport_id)`,
    )
    .bind(nowIso, passportId)
    .first<{ claim_state: string; retracted: number }>();
  return !!p && p.retracted !== 1 && p.claim_state === "verified";
}

/**
 * S-CW5: the same gate across a whole roster in TWO queries instead of 2N —
 * a 50-card roster check must not cost 100 round trips per queue join. Must
 * agree with `warOptedIn` by construction: the same replay (shared with
 * `warOptState`'s loop) over one grouped event read, and the same standing
 * predicate over one grouped passport read. Returns true only if EVERY id
 * passes. D1-BOUNDED: ids.length ≤ 50 keeps binds ≤ 51 (the ids, plus the standing read's instant).
 */
export async function allWarOptedIn(db: DbLike, passportIds: readonly string[], nowIso: string = new Date().toISOString()): Promise<boolean> {
  if (passportIds.length === 0) return false;
  if (passportIds.length > 50) return false;
  for (const p of passportIds) if (!/^psp_[0-9a-z]{8,64}$/.test(p)) return false;
  const placeholders = passportIds.map(() => "?").join(",");
  const { results: events } = await db
    .prepare(
      `SELECT subject, event_type, actor_kind, created_at FROM trust_events
        WHERE subject IN (${placeholders}) AND event_type IN ('war_opt_in','war_opt_out') ORDER BY seq ASC`,
    )
    .bind(...passportIds)
    .all<{ subject: string; event_type: string; actor_kind: string; created_at: string }>();
  const bySubject = new Map<string, { event_type: string; actor_kind: string; created_at: string }[]>();
  for (const e of events) {
    const rows = bySubject.get(e.subject) ?? [];
    rows.push(e);
    bySubject.set(e.subject, rows);
  }
  for (const p of passportIds) {
    if (!replayRows(bySubject.get(p) ?? []).opted_in) return false;
  }
  const { results: passports } = await db
    .prepare(
      `SELECT p.passport_id AS passport_id, ${CLAIM_STATE_AT_SQL} AS claim_state, p.retracted AS retracted FROM passports p
        WHERE p.passport_id IN (${placeholders})
          AND NOT EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = p.passport_id)`,
    )
    .bind(nowIso, ...passportIds)
    .all<{ passport_id: string; claim_state: string; retracted: number }>();
  const standing = new Set(
    passports.filter((p) => p.retracted !== 1 && p.claim_state === "verified").map((p) => p.passport_id),
  );
  return passportIds.every((p) => standing.has(p));
}

/**
 * S-1 (I23-RG-MONEY, I23-FX-TC): `warOptedIn` for MANY passports — the subset that passes — in TWO statements per
 * `inChunks` chunk (one grouped opt-in/opt-out event read, one grouped standing read), never 2N. The same replay
 * (`replayRows`, shared with `warOptState`) and the same standing predicate as `warOptedIn`, so the two agree by
 * construction (test/i23-fx-tc-pack-odds.test.ts pins the per-id agreement). Ids that are not passport-shaped never
 * pass.
 */
export async function warOptedInIds(db: DbLike, passportIds: readonly string[], nowIso: string = new Date().toISOString()): Promise<Set<string>> {
  const passing = new Set<string>();
  const ids = passportIds.filter((p) => /^psp_[0-9a-z]{8,64}$/.test(p));
  for (const chunk of inChunks(ids, 1)) {
    const placeholders = chunk.map(() => "?").join(",");
    const { results: events } = await db
      .prepare(
        `SELECT subject, event_type, actor_kind, created_at FROM trust_events
          WHERE subject IN (${placeholders}) AND event_type IN ('war_opt_in','war_opt_out') ORDER BY seq ASC`,
      )
      .bind(...chunk)
      .all<{ subject: string; event_type: string; actor_kind: string; created_at: string }>();
    const bySubject = new Map<string, { event_type: string; actor_kind: string; created_at: string }[]>();
    for (const e of events) {
      const rows = bySubject.get(e.subject) ?? [];
      rows.push(e);
      bySubject.set(e.subject, rows);
    }
    const opted = chunk.filter((p) => replayRows(bySubject.get(p) ?? []).opted_in);
    if (opted.length === 0) continue;
    const { results: passports } = await db
      .prepare(
        `SELECT p.passport_id AS passport_id, ${CLAIM_STATE_AT_SQL} AS claim_state, p.retracted AS retracted FROM passports p
          WHERE p.passport_id IN (${opted.map(() => "?").join(",")})
            AND NOT EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = p.passport_id)`,
      )
      .bind(nowIso, ...opted)
      .all<{ passport_id: string; claim_state: string; retracted: number }>();
    for (const p of passports) if (p.retracted !== 1 && p.claim_state === "verified") passing.add(p.passport_id);
  }
  return passing;
}

    instanceId: string;
    logId: string;
    ownerId: string;
    passportId: string;
    source: InstanceSource;

export const STARTER_ROSTER_SIZE = 10;
/** Bound on the candidates checked against the consent gate (each check is
 * two reads). At most STARTER_SAVES_CAP of them are saves (oldest first), so a

const STARTER_CANDIDATE_CAP = 60;

/** Deterministic per-account order key for the pool top-up — FNV-1a over
 * `account_id:passport_id`, the non-crypto hash `cards.ts` `paletteSeed` uses.
 * Not a secret and not a security property: it only spreads different fans
 * across different creators of the same weekly pool. */
function starterOrderKey(accountId: string, passportId: string): number {
  const s = `${accountId}:${passportId}`;
  let h = 0x811c9dc5;
  for (let i = 0; i < s.length; i++) {
    h ^= s.charCodeAt(i);
    h = Math.imul(h, 0x01000193) >>> 0;
  }
  return h;
}

  accountId: string,
  poolPassportIds: readonly string[],

  }
  const pool = [...new Set(poolPassportIds)].filter((p) => PSP_RE.test(p) && !seen.has(p));
  pool.sort((a, b) => starterOrderKey(accountId, a) - starterOrderKey(accountId, b) || (a < b ? -1 : a > b ? 1 : 0));
  candidates.push(...pool);
  // EXPB siblings (I34-FX2-EXPIRY): one instant for the opt-in reads and the batch, not the wall clock.
  const ts = now();
  const chosen: string[] = [];
  for (const psp of candidates.slice(0, STARTER_CANDIDATE_CAP)) {
    if (chosen.length >= STARTER_ROSTER_SIZE) break;
    // D-1123(1): a closed or suspended creator is not chosen.
    if ((await warOptedIn(db, psp, ts)) && !(await creatorClosedNow(db, psp))) chosen.push(psp);
  }
  if (chosen.length === 0) return { minted: [], already: false };
  const ids: string[] = [];
  const stmts: DbStatement[] = [db.prepare("INSERT INTO starter_grants (account_id, created_at) VALUES (?, ?)").bind(accountId, ts)];
  for (const psp of chosen) {
    const id = mintInstanceId();
    ids.push(id);
    stmts.push(...mintInstanceStatements(db, { instanceId: id, logId: mintLogId(), ownerId: accountId, passportId: psp, source: "starter", bound: true, ts }));
  }
  try {
    await db.batch(stmts);
  } catch (e) {
    // The claim (or a concurrent winner's starter card) already exists: the whole batch rolled back.
    const msg = e instanceof Error ? e.message : String(e);
    if (/unique/i.test(msg)) return { minted: [], already: true };
    throw e;
  }
  const minted: string[] = [];
  for (const id of ids) {
    if (await db.prepare("SELECT 1 AS x FROM card_instances WHERE instance_id = ?").bind(id).first()) minted.push(id);
  }
  return { minted, already: false };
}

/**
 * Statements that transfer one instance — built to ride the S-CW5 match-settle
 * batch (result + ante transfer commit atomically or not at all).
 *
 * Every guard lives INSIDE the statements' WHERE (current owner matches, not
 * bound, not consumed, real destination — and, in "surplus-only" mode, at
 * least two living copies so the KEPT copy can never leave), so a lost race,
 * a bound card, or a stale claim flips the whole thing into a zero-row no-op.
 * The log row is written FIRST, predicated on the PRE-state (the exact guards
 * the UPDATE carries): inside one atomic batch both statements see the same
 * world in sequence, so either both fire or both no-op. (Predicating the log
 * on the POST-state was a bug our tests caught — it also held when the
 * destination ALREADY owned the card, minting phantom provenance.)
 *
 * `mode` is REQUIRED, not defaulted — the caller must decide:
 *  - "surplus-only": the ante rule. The transfer no-ops unless the owner holds
 *    ≥ 2 living copies of this creator, enforced ATOMICALLY in the same
 *    statements (review F3 — a read-then-act surplus check has a race window;
 *    this one rides the batch).
 *  - "any": moves the last copy too. For flows where that is the point.
 *
 * ── CALLER CONTRACT (review F4/F9 — write-downs, enforced by convention) ──
 *  - These statements are ONLY meaningful as one atomic batch; running them
 *    separately can forge provenance (a 'transferred' row with no move).
 *  - Transfers require no recipient consent AT THIS LAYER and bypass the mint
 *    cap by design. Every HTTP-reachable transfer flow MUST be one the
 *    recipient opted into (a match IS consent; a bare gift endpoint is not) —
 *    otherwise shelf-flooding a victim is trivial.
 */
export function transferInstanceStatements(
  db: DbLike,
  args: {
    instanceId: string;
    fromOwner: string;
    toOwner: string;
    reason: string;
    logId: string;
    ts: string;
    mode: "surplus-only" | "any";
    /** S-CW5: by default a transfer requires the card NOT be escrowed (a
     * staked card is off the table for everything but its own settlement).
     * The match-settle path passes the entry that holds the escrow — the
     * transfer then REQUIRES that exact hold and clears it in the same
     * UPDATE, so only the settlement the stake was made for can move it. */
    escrow?: { heldBy: string };
    /** F3: the provenance event. `transferred` (default — the ante), `gifted`
     * (a consented gift, detail = offer id) or `recovered` (operator recovery,
     * detail = audit ref). The two F3 events also require the destination's
     * SHELF to have room — a gift is opted into, a flood is not. */
    event?: "transferred" | "gifted" | "recovered";
    /** F3: an EXTRA predicate both statements carry (SQL fragment beginning
     * with AND, plus its binds) — the accept path passes "the offer is still
     * open", so a cancel/decline/expiry that lands first makes the whole move
     * a no-op instead of a move under a resolved offer. */
    extraGuard?: { sql: string; binds: unknown[] };
  },
): DbStatement[] {
  checkIds(args.instanceId, args.logId);
  checkTs(args.ts);
  if (args.fromOwner === args.toOwner) throw new WarInstanceError("transfer to self");
  const event = args.event ?? "transferred";
  if (!["transferred", "gifted", "recovered"].includes(event)) throw new WarInstanceError("bad transfer event", 500);
  if (args.reason.length === 0 || args.reason.length > 40) throw new WarInstanceError("bad transfer reason");
  if (args.escrow && !/^wq_[0-9a-f]{20}$/.test(args.escrow.heldBy)) throw new WarInstanceError("bad escrow holder shape", 500);
  // In surplus-only mode the guard counts LIVING copies of the same creator
  // held by the CURRENT owner — evaluated inside the batch, atomically.
  const surplusGuard =
    args.mode === "surplus-only"
      ? ` AND (SELECT COUNT(*) FROM card_instances o
               WHERE o.owner_id = ? AND o.consumed_at IS NULL AND (o.escrowed_by IS NULL OR o.instance_id = ?)
                 AND o.passport_id = (SELECT passport_id FROM card_instances WHERE instance_id = ?)) >= 2`
      : "";
  // I3-0944: a copy staked elsewhere is not a kept copy; the moving card still counts itself.
  const surplusBinds = args.mode === "surplus-only" ? [args.fromOwner, args.instanceId, args.instanceId] : [];
  const escrowGuard = args.escrow ? " AND escrowed_by = ?" : " AND escrowed_by IS NULL";
  const escrowBinds = args.escrow ? [args.escrow.heldBy] : [];
  // F3: a gift/recovery lands on a shelf that must have room (the ante bypasses
  // this by design — a match is a 1-for-1 swap).
  const shelfGuard = event === "transferred" ? "" : ` AND (SELECT COUNT(*) FROM card_instances d WHERE d.owner_id = ? AND d.consumed_at IS NULL) < ?`;
  const shelfBinds = event === "transferred" ? [] : [args.toOwner, MAX_INSTANCES_PER_ACCOUNT];
  if (args.extraGuard && !/^ AND /.test(args.extraGuard.sql)) throw new WarInstanceError("extraGuard must begin with ' AND '", 500);
  const extraSql = args.extraGuard?.sql ?? "";
  const extraBinds = args.extraGuard?.binds ?? [];
  return [
    db
      .prepare(
        `INSERT INTO card_instance_log (log_id, instance_id, event, from_owner, to_owner, detail, at)
         SELECT ?, ?, ?, ?, ?, ?, ?
          WHERE EXISTS (SELECT 1 FROM card_instances
                         WHERE instance_id = ? AND owner_id = ? AND bound = 0 AND consumed_at IS NULL${escrowGuard})
            AND EXISTS (SELECT 1 FROM accounts WHERE account_id = ?)${surplusGuard}${shelfGuard}${extraSql}`,
      )
      .bind(args.logId, args.instanceId, event, args.fromOwner, args.toOwner, args.reason, args.ts, args.instanceId, args.fromOwner, ...escrowBinds, args.toOwner, ...surplusBinds, ...shelfBinds, ...extraBinds),
    db
      .prepare(
        `UPDATE card_instances SET owner_id = ?, escrowed_by = NULL
          WHERE instance_id = ? AND owner_id = ? AND bound = 0 AND consumed_at IS NULL${escrowGuard}
            AND EXISTS (SELECT 1 FROM accounts WHERE account_id = ?)${surplusGuard}${shelfGuard}${extraSql}`,
      )
      .bind(args.toOwner, args.instanceId, args.fromOwner, ...escrowBinds, args.toOwner, ...surplusBinds, ...shelfBinds, ...extraBinds),
  ];
}

/** QUEUE #236 (D-0434, D-0870(20)): PRODUCTION CALLERS = 0. Staking is removed

export function escrowInstanceStatement(db: DbLike, args: { instanceId: string; ownerId: string; entryId: string; ts?: string }): DbStatement {

  if (!/^wq_[0-9a-f]{20}$/.test(args.entryId)) throw new WarInstanceError("bad entry id shape", 500);

      // concurrent joins can no longer stake both of the last two copies.

/**
 * Duplicate summary for one (account, creator): total living copies and how
 * many are surplus fuel/stake material (unbound copies beyond the first —
 * the first copy is sacred: display, provenance, never consumed or staked).
 * S-CW3's fusion and S-CW5's ante both size themselves off this — as ADVICE
 * for UI; the enforced version is the atomic surplus guard in
 * `transferInstanceStatements` ("surplus-only") and its S-CW3 consume twin.
 *
 * Review F10 (pin before S-CW3): WHICH copy is "the kept one" is a count
 * argument here, not a chosen row. If kept-copy selection ever becomes
 * row-level (e.g. for display), it must be "oldest MINT-TO-THIS-OWNER per the
 * provenance log" or an explicit user choice — never bare `acquired_at`,
 * which transfers carry over from the previous owner and would let a
 * transferred-in older card displace the owner's true first copy.
 */
export async function duplicateSummary(
  db: DbLike,
  accountId: string,
  passportId: string,
): Promise<{ total: number; unbound: number; surplus: number }> {
  const row = await db
    .prepare(
      // AS-057: escrowed (staked) copies are committed elsewhere — same class of
      // bug as C-59's post-burn count and AS-112's gift_offers gap. `escrowInstanceStatement`
      // only ever stakes a SURPLUS copy (its own guard requires >=2 living
      // copies), so an escrowed copy has already been spent as stake material —
      // it must count in `total` (still owned, still live) but drop out of the
      // pool this advises as further surplus/fuel, and it must NOT serve as a
      // substitute "kept" copy the way a bound copy does (that would let a lone
      // remaining free duplicate be mis-advised as surplus too).
      `SELECT COUNT(*) AS total,
              SUM(CASE WHEN bound = 0 AND escrowed_by IS NULL THEN 1 ELSE 0 END) AS unbound,
              SUM(CASE WHEN escrowed_by IS NOT NULL THEN 1 ELSE 0 END) AS escrowed
         FROM card_instances
        WHERE owner_id = ? AND passport_id = ? AND consumed_at IS NULL`,
    )
    .bind(accountId, passportId)
    .first<{ total: number; unbound: number | null; escrowed: number | null }>();
  const total = row?.total ?? 0;
  const unbound = row?.unbound ?? 0;
  const escrowed = row?.escrowed ?? 0;
  // `available` excludes already-staked copies from the kept-copy accounting
  // (they can't stand in for the kept copy — see above). Surplus = available
  // unbound copies beyond the KEPT one. If every available copy is unbound,
  // one of them is the kept copy (surplus = unbound - 1); if any bound copy
  // exists it can serve as the kept one, so every available unbound copy is surplus.
  const available = total - escrowed;
  const surplus = available === 0 ? 0 : unbound === available ? Math.max(0, unbound - 1) : unbound;
  return { total, unbound, surplus };
}

 * ranked wins, shown on the profile's public war record. No money and no credits: points are a number

const DAY_MS = 24 * 60 * 60 * 1000;
/** Season 1 opens at this instant (a Monday, 00:00 UTC). Matches before it belong to no season. */
export const SEASON_EPOCH_MS = Date.parse("2026-01-05T00:00:00.000Z");
/** Fixed season length. */
export const SEASON_LENGTH_MS = 28 * DAY_MS;
/** Rating points for each DISTINCT qualifying opponent beaten in a season. */
export const WIN_POINTS = 10;
/** social.ts WAR_RECORD_OPPONENT_AGE_MS (the 7-day Sybil age). */
export const LADDER_OPPONENT_AGE_MS = 7 * DAY_MS;
/** How many PAST seasons the standing carries (the ones with points). */
export const LADDER_HISTORY_SEASONS = 6;

export interface SeasonWindow {
  season: number;
  starts_at: string;
  ends_at: string;
}

export interface LadderStanding extends SeasonWindow {
  points: number;
  wins: number;
  /** Past seasons with points, newest first, at most LADDER_HISTORY_SEASONS. */
  history: Array<{ season: number; points: number }>;
}

/** The season `ms` falls in (1-based), or null before season 1. */
export function seasonOf(ms: number): number | null {
  if (!Number.isFinite(ms) || ms < SEASON_EPOCH_MS) return null;
  return Math.floor((ms - SEASON_EPOCH_MS) / SEASON_LENGTH_MS) + 1;
}

export function seasonWindow(season: number): SeasonWindow {
  const start = SEASON_EPOCH_MS + (season - 1) * SEASON_LENGTH_MS;
  return { season, starts_at: new Date(start).toISOString(), ends_at: new Date(start + SEASON_LENGTH_MS).toISOString() };
}

/**
 * One account's ladder standing at `nowMs`: the current season's points and the past seasons'.
 * One bounded statement over the account's own ranked matches since the oldest season it reports.
 */
export async function warLadderStanding(db: DbLike, accountId: string, nowMs: number): Promise<LadderStanding | null> {
  const current = seasonOf(nowMs);
  if (current === null) return null;
  const oldest = Math.max(1, current - LADDER_HISTORY_SEASONS);
  const fromMs = SEASON_EPOCH_MS + (oldest - 1) * SEASON_LENGTH_MS;
  // The two CASTs make the season an INTEGER division: a JS number may bind as REAL, and a REAL quotient
  // would put every match in its own fractional "season".
  const { results } = await db
    .prepare(
      `SELECT (m.resolved_ms - CAST(? AS INTEGER)) / CAST(? AS INTEGER) + 1 AS season, COUNT(DISTINCT CASE WHEN m.won = 1 THEN m.opp END) AS wins
         FROM (SELECT CASE WHEN a_account = ? THEN b_account ELSE a_account END AS opp,
                      CASE WHEN (a_account = ? AND winner = 'a') OR (b_account = ? AND winner = 'b') THEN 1 ELSE 0 END AS won,
                      resolved_ms
                 FROM war_matches
                WHERE practice = 0 AND ghost = 0 AND (a_account = ? OR b_account = ?)
                  AND resolved_ms >= ? AND resolved_ms <= ?) m
         JOIN accounts o ON o.account_id = m.opp
        WHERE m.opp <> ? AND CAST(strftime('%s', o.created_at) AS INTEGER) * 1000 <= m.resolved_ms - ?
        GROUP BY season`,
    )
    .bind(SEASON_EPOCH_MS, SEASON_LENGTH_MS, accountId, accountId, accountId, accountId, accountId, fromMs, nowMs, accountId, LADDER_OPPONENT_AGE_MS)
    .all<{ season: number; wins: number | null }>();
  const winsBySeason = new Map<number, number>();
  for (const r of results) winsBySeason.set(Number(r.season), Number(r.wins ?? 0));
  const wins = winsBySeason.get(current) ?? 0;
  const history: Array<{ season: number; points: number }> = [];
  for (let s = current - 1; s >= oldest; s -= 1) {
    const w = winsBySeason.get(s) ?? 0;
    if (w > 0) history.push({ season: s, points: w * WIN_POINTS });
  }
  return { ...seasonWindow(current), points: wins * WIN_POINTS, wins, history };
}

/**
 * S-CW5: MATCHES — queue, resolution, replay. QUEUE #236 (D-0434, D-0870(20)):
 * STAKING IS REMOVED — no join escrows a card and no match moves one; `practice`
 * stays the mode flag (ranked = practice 0 = the public win/loss record). The
 * `ante_*` columns stay in the schema and are never written. The first playable
 * slice: everything under it (engine, instances, Bond, consent) already
 * shipped; this module is where they meet (no stakes since QUEUE #236).
 *
 * ── RECON (#241, D-0920): with `joinQueue(..., { recon: true })` a LIVE pairing
 * does not resolve in the pairing request. `openRecon` claims both entries and
 * writes one `war_recon` row per side (migration 0190) in ONE batch; each side is
 * shown HALF of the opponent's roster — `reconRevealIndices`, a pure function of
 * the waiting entry's COMMITTED server seed — and may commit its Champion and
 * doctrine once (`commitRecon`) before `deadline` = pairing + WAR_RECON_WINDOW_MS.
 * `resolveRecon` refuses to resolve until both sides committed or the deadline
 * passed; a silent side fights with its standing Champion and the single
 * doctrine — never voided on timeout. `sweepWarRecon` resolves expired windows
 * nobody came back to. Ghost matches never recon (there is no second player).
 *
 * ── PAIRING WITHOUT A DURABLE OBJECT (deliberate plan deviation, written
 * down): the approved plan sketched a matchmaking DO per bracket. This repo
 * has NO Durable Object infrastructure, and the authoritative state a match
 * settles against (instances, bonds, queue rows) lives in D1 regardless — a
 * DO in front of it would serialize pairing that D1's single-writer batches
 * already serialize, while adding an untestable deployment surface. So
 * pairing is an ATOMIC CLAIM: the settle batch flips the chosen waiting entry
 * `waiting → matched` with a guarded UPDATE, and the force-rollback validator
 * refuses the whole batch unless exactly that claim landed — two racing
 * joiners get one winner, the loser retries the next candidate. The module
 * boundary is DO-shaped on purpose: if scale ever demands it, a DO adopts
 * this data model unchanged.
 *
 * ── PROVABLE FAIRNESS — LIVE MATCHES ONLY (the engine's six server
 * preconditions, honored for every match whose side A is a live waiting
 * entry; a GHOST match does NOT meet 1 or 2 — see GHOSTS below):
 *  1. The stream's server seed belongs to the WAITING entry, minted and
 *     committed (hash returned to that fan) when the entry was created —
 *     BEFORE the opponent, their roster, or their client seed existed.
 *  2. Both client seeds are fixed before the server seed is revealed: the
 *     waiting side's at their join, the joiner's in the resolving request;
 *     the seed is revealed only in the resolved match record.
 *  3. Single SUCCESSFUL use: an entry pairs at most once (the atomic claim),
 *     and a cancelled entry's seed is discarded, never recycled. A settle that
 *     ROLLS BACK (a lost pairing race) leaves the candidate waiting with its
 *     seed intact for the next joiner — which is NOT a grinding vector (review
 *     F4): the seed is never revealed to any client before a committed match
 *     (queueStatus returns only the hash), and each joiner fixes their own
 *     client seed in-request before the server reads the candidate's seed, so
 *     no offline prediction against a fixed opponent seed is possible.
 *  4. Rosters/stats/bonds are the server's own authenticated snapshots.
 *  5. Dials are server config (engine defaults) — no player input reaches them.
 *  6. The seed stays in the row (server-side) until resolution.
 *  No route exposes ANOTHER account's waiting entries or commitments, so a
 *  joiner cannot aim a client seed at a known opponent (they can't see one).
 *
 * ── GHOSTS: practice-tier only, and only when no live practice opponent
 * waits — side A is replayed from a past match's stored snapshot (its entry
 * label, stats, bonds, client seed), and NO property (no card) moves. A ghost
 * match is NOT PROVABLY FAIR (I3-0992, D-0913(3)): its server seed is minted
 * in `tryGhost` AFTER the joiner's side and client seed are fixed, and no
 * commitment to it was ever returned to anyone, so preconditions 1-2 above do
 * not hold for it. That is accepted only because a ghost is practice with no
 * stakes; the match row, the result and the replay all carry `ghost: true`.
 * Ranked matches are always live-vs-live: a ghost cannot hold
 * stakes because its player isn't there to have consented to THIS match.
 * Ghost sources are still gated on the creators' CURRENT opt-in (the freeze
 * follows the creator, not the snapshot).
 *
 * ── ERASURE SHAPE: engine player labels and everything inside `inputs` /
 * replays are ENTRY ids and instance/passport ids — never account ids.
 * Account ids live only in war_queue/war_matches columns, which
 * account_data.ts erases/pseudonymizes; the immutable R2 replay never needs
 * to change.
 */

/**
 * W308-A-18 (Jason's ruling D-0549: opponents are randomly matched TO PREVENT
 * COLLUSION). The pool the opponent is drawn from, and it is deliberately much
 * wider than the number of candidates actually attempted.
 *
 * THE DEFECT THIS CLOSES. Selection used to shuffle the 5 OLDEST waiters and
 * take the first that settled. Shuffling a page a colluder controls buys
 * nothing: a group holding 5 entries in a bracket owns the entire page, so
 * every reachable candidate is a colluder and the pairing lands inside the
 * group with probability 1 — regardless of how good the PRNG is. The old
 * docblock's claim that pairing was "unarrangeable without per-match
 * telemetry" was true only of the narrow two-honest-waiters case it actually
 * described. Fixing the PRNG (W308-A-17, below) does not fix this, and
 * widening the pool does not fix the PRNG; they are independent limbs and both
 * are closed here.
 *
 * WHAT THE WIDTH BUYS, AND WHAT IT DOES NOT. Width alone buys only PRICE: it
 * moves the attack from owning 5 entries to owning PAIR_WINDOW of them. The W3-08
 * judge overturned FIX6's claim that this CLOSED the hole, and was right — while
 * the pool was an age-ordered prefix its membership stayed attacker-orderable at
 * any width, because arrival order is precisely what a colluding group controls.
 * The pool is therefore drawn with `ORDER BY RANDOM()` over the whole waiting
 * set (see the query), and the width is now just a work bound on the sample.
 *
 * THE TRADE, stated because it is real: strict FIFO pairing is given up. That
 * was never a promise the product made — the queue promises A match, not the
 * oldest available opponent — and it is the exact property a colluder was
 * exploiting. A long-waiting fan does not starve: every waiter in the bracket is
 * equally likely to be sampled on every join, so the chance of going unmatched
 * shrinks with each arrival rather than depending on a queue position.
 */
const PAIR_WINDOW = 100;
/** I3-0561 (D-0834): ranked matches pair only accounts this old — the alt/main self-match window. */
const RANKED_MIN_ACCOUNT_AGE_MS = 7 * 24 * 60 * 60 * 1000;
/**
 * W308-Q1, found by the W3-08 JUDGE-AUDIT overturning the judge.
 *
 * THE DEFECT. The pairing loop spent ONE budget on two different jobs. A
 * candidate that turns out to be STALE — its owner banned or closed, its roster
 * naming cards nobody owns — is VOIDED in passing and the loop moves on, but
 * that void consumed one of the five attempts. So a bracket salted with stale
 * entries DENIED matches to honest joiners while valid opponents sat waiting:
 * measured at 4 denials in 12 joins with 95 stale and 5 valid waiters, against
 * 0 in 12 with the salt removed.
 *
 * WHY THE JUDGE MISSED IT, and it is worth writing down: its `P(opponent ∈
 * group) = k/n` analysis is correct and answers WHO YOU FIGHT. It never asks
 * WHETHER YOU FIGHT. A random draw from a set that is mostly rubble is still a
 * fair draw, and still returns rubble. The auditor also measured that
 * `voidEntry` REFUNDS the salted entry's own escrow — so the channel the judge
 * priced in antes is, on that axis, not priced at all. What does price it is
 * `idx_wq_one_waiting`: one waiting entry per account, so 95 stale entries
 * costs 95 accounts that once held cards. Real, but Sybil-priced, not ante-priced.
 *
 * THE SPLIT. Voiding a stale entry is REAPING, not a match attempt, and it is
 * permanent — each entry is voided once, ever, so the cost is paid once across
 * the system rather than per join. It therefore gets its own budget and does
 * not consume the attempt budget.
 *
 * WHY THE REAP BUDGET IS THE WHOLE WINDOW, and the first attempt at this fix
 * got it wrong. A budget of 25 still denied roughly one joiner in four: with 95
 * rubble entries among 100, the chance that all five valid opponents sit past
 * the 25th position of a random permutation is about 23%, so the budget itself
 * became the denial channel the split was supposed to close — a smaller version
 * of the same defect, which is not a fix. Set to PAIR_WINDOW, the loop can walk
 * the whole sample, so a joiner reaches every valid opponent the sample
 * contains. The work stays bounded (the pool is at most PAIR_WINDOW rows, which
 * is the query we already ran) and the worst case is SELF-LIQUIDATING, because
 * every candidate it walks past is voided permanently.
 *
 * THE RESIDUAL, stated rather than hidden: the pool is a SAMPLE. In a bracket
 * with far more than PAIR_WINDOW waiters and only a few valid ones, the sample
 * may contain none of them, and the joiner queues. That is inherent to sampling
 * and is not a denial channel — it is the same odds every honest waiter faces —
 * but it is the reason this is bounded by "every valid opponent IN THE SAMPLE"
 * rather than "in the bracket".
 */
const PAIR_REAP_BUDGET = PAIR_WINDOW;

/**
 * W308-A-17. A Fisher-Yates shuffle whose randomness is a CSPRNG, because this
 * permutation decides WHO A FAN IS MATCHED AGAINST and money rides on it.
 *
 * THE DEFECT THIS CLOSES. Opponent selection drew from `Math.random()` — in
 * the same function that mints the commit-reveal seed through
 * `WarMinters.serverSeed`, which this file's own interface documents and
 * REQUIRES as CSPRNG. It was also the only randomness in the module that was
 * neither committed, injected, nor stubbable, so nothing could observe it. A
 * participant who can predict the permutation can arrange the pairing, which is
 * the whole thing D-0549 forbids.
 *
 * UNBIASED, by rejection sampling rather than `% (i + 1)`. A modulo of a
 * uniform 32-bit draw is biased toward the low indices whenever the range does
 * not divide 2^32 — small, but it is a bias in WHO YOU FIGHT, and the loop to
 * remove it is three lines. Values at or above the largest multiple of the
 * range are redrawn.
 *
 * EXPORTED FOR ITS GUARD, and only for that. W3-08's judge showed that every
 * SOURCE-LEVEL guard on this function falls to one token — keep the
 * `crypto.getRandomValues(buf)` call, discard its result, and swap in a counter:
 * a substring scan cannot see dataflow. The only guard that survives that is a
 * BEHAVIOURAL one that drives this function with a controlled byte stream and
 * checks the permutation is a FUNCTION of those bytes, which needs the symbol.
 */
export function shuffleCsprng<T>(xs: T[]): void {
  for (let i = xs.length - 1; i > 0; i--) {
    const range = i + 1;
    const limit = Math.floor(0x1_0000_0000 / range) * range; // largest multiple of range within 2^32
    const buf = new Uint32Array(1);
    let r: number;
    do {
      crypto.getRandomValues(buf);
      r = buf[0]!;
    } while (r >= limit);
    const j = r % range;
    [xs[i], xs[j]] = [xs[j]!, xs[i]!];
  }
}

/**
 * #241 (D-0920): WHICH HALF of side `shown`'s roster its opponent is shown
 * during recon — floor(size/2) distinct indices, ascending.
 *
 * A PURE FUNCTION of the waiting entry's COMMITTED server seed, the side and
 * the roster size, and of nothing else: the seed was committed (its hash
 * returned to the waiting fan) before either roster was paired, so the server
 * cannot choose the half after seeing the rosters, and anyone can re-derive it
 * once the match reveals the seed.
 *
 * DOMAIN-SEPARATED from the match stream: every block here hashes
 * `recon:<seed>:<side>:<i>`, and `r` is not a hex digit, so no block can equal
 * a match-stream block (`<hex>:<hex>:<hex>:<i>`, war-engine rng.ts) — the
 * reveal discloses nothing about the draws the match will make. A partial
 * Fisher-Yates over one byte per draw (size <= 255), unbiased by rejection
 * sampling exactly as rng.ts draws.
 */
export async function reconRevealIndices(serverSeed: string, shown: "a" | "b", size: number): Promise<number[]> {
  if (!/^[0-9a-f]{8,128}$/.test(serverSeed)) throw new Error("reconRevealIndices: bad seed");
  if (shown !== "a" && shown !== "b") throw new Error("reconRevealIndices: bad side");
  if (!Number.isInteger(size) || size < 2 || size > 255) throw new Error("reconRevealIndices: bad size");
  let block: Uint8Array = new Uint8Array(0);
  let off = 0;
  let blockIndex = 0;
  const nextByte = async (): Promise<number> => {
    if (off >= block.length) {
      block = await sha256Bytes(new TextEncoder().encode(`recon:${serverSeed}:${shown}:${blockIndex++}`));
      off = 0;
    }
    return block[off++]!;
  };
  const idx = Array.from({ length: size }, (_, i) => i);
  const k = Math.floor(size / 2);
  for (let i = 0; i < k; i++) {
    const range = size - i;
    const limit = 256 - (256 % range);
    let r: number;
    do {
      r = await nextByte();
    } while (r >= limit);
    const j = i + (r % range);
    [idx[i], idx[j]] = [idx[j]!, idx[i]!];
  }
  return idx.slice(0, k).sort((x, y) => x - y);
}

function refuse(msg: string, status = 400): never {

/** ONE indistinguishable refusal for every roster-side failure — a card you
 * don't own, a consumed card, a non-opted / suppressed / unrated creator all
 * answer identically (no oracle over other people's property or standing). */
const ROSTER_REFUSAL = "roster not currently playable: every card must be your own living copy of a distinct creator who is in the game";

function validateJoinShape(input: QueueJoinInput): void {
  if (!Array.isArray(input.roster) || !(BRACKETS as readonly number[]).includes(input.roster.length)) {
    refuse(`roster size must be one of ${BRACKETS.join("/")}`);
  }
  if (new Set(input.roster).size !== input.roster.length) refuse("duplicate instance in roster");
  for (const id of input.roster) if (typeof id !== "string" || !CI_RE.test(id)) refuse("bad roster instance id");
  if (!Number.isInteger(input.champion_index) || input.champion_index < 0 || input.champion_index >= input.roster.length) {
    refuse("champion index out of range");
  }
  if (typeof input.client_seed !== "string" || !SEED_RE.test(input.client_seed)) {
    refuse("client_seed must be 8..64 lowercase hex chars");
  }
  if (typeof input.practice !== "boolean") refuse("practice must be a boolean");
  // QUEUE #236 (D-0434, D-0870(20)): no join, ranked or practice, can put a
  // card at risk. A body naming one is REFUSED, never ignored, so no client
  // can believe a card was staked.
  if (input.ante_instance !== undefined) refuse("staking has been removed from Creator Wars: a join takes no ante");
}

/** Load + verify a roster for `accountId`: all owned, living, unescrowed, one
 * card per creator, every creator opted in. Returns instance→passport in
 * ROSTER ORDER, or null (one generic refusal — see ROSTER_REFUSAL). `nowIso` is the door's instant for the opt-in gate's
 * lapsed-grant term (I34-W3-SEAM-Z, I34-W3-SEAM-A h/o 4): joinQueue's `doorAt`, tryPairWithWaiter's `passAt`. */
async function loadRoster(db: DbLike, accountId: string, roster: readonly string[], nowIso: string): Promise<{ instance_id: string; passport_id: string }[] | null> {
  const placeholders = roster.map(() => "?").join(",");
  const { results } = await db
    .prepare(
      `SELECT instance_id, passport_id FROM card_instances
        WHERE owner_id = ? AND consumed_at IS NULL AND escrowed_by IS NULL AND instance_id IN (${placeholders})`,
    )
    .bind(accountId, ...roster)
    .all<{ instance_id: string; passport_id: string }>();
  if (results.length !== roster.length) return null;
  const byId = new Map(results.map((r) => [r.instance_id, r.passport_id]));
  const passports = roster.map((id) => byId.get(id)!);
  if (new Set(passports).size !== passports.length) return null; // one card per creator
  if (!(await allWarOptedIn(db, passports, nowIso))) return null;
  return roster.map((id) => ({ instance_id: id, passport_id: byId.get(id)! }));
}

/** Snapshot the seven Power Rating bars for each passport from the live
 * ratings artifacts. null = at least one creator is unrated or suppressed
 * (indistinguishable by design — serveRatings already folds suppression in). */
async function snapshotStats(
  store: ArtifactStore,
  db: DbLike,
  passports: readonly string[],
  cache: Map<string, PowerRatings | null>,
  /** The door's own instant for serveRatings' lapsed-grant term (I34-W3-EXPIRY h/o 5): joinQueue's now(), buildSideSnapshot's nowMs. */
  nowIso: string,
): Promise<Map<string, number[]> | null> {
  const out = new Map<string, number[]>();
  for (const psp of passports) {
    let ratings = cache.get(psp);
    if (ratings === undefined) {
      ratings = await serveRatings(store, db, psp, nowIso);
      cache.set(psp, ratings);
    }
    if (!ratings) return null;
    // RATING_DIMENSIONS order == engine STAT_NAMES order (both are the seven
    // Power Ratings in card display order); map index-wise.
    const bars = RATING_DIMENSIONS.map((dim) => ratings!.ratings[dim].bar);
    if (bars.length !== STAT_NAMES.length || bars.some((b) => !Number.isInteger(b) || b < 1 || b > 7)) return null;
    out.set(psp, bars);
  }
  return out;
}

async function buildSideSnapshot(
  db: DbLike,
  store: ArtifactStore,
  accountId: string | null,
  entryId: string,
  roster: { instance_id: string; passport_id: string }[],
  championIndex: number,
  clientSeed: string,
  nowMs: number,
  statsCache: Map<string, PowerRatings | null>,
): Promise<SideSnapshot | null> {
  const passports = roster.map((r) => r.passport_id);
  const stats = await snapshotStats(store, db, passports, statsCache, new Date(nowMs).toISOString());
  if (!stats) return null;
  const bonds = accountId ? await bondLevels(db, accountId, passports, nowMs) : new Map<string, number>();
  return {
    entry: entryId,
    roster: roster.map((r) => ({ instance_id: r.instance_id, passport_id: r.passport_id, stats: stats.get(r.passport_id)!, bond: bonds.get(r.passport_id) ?? 0 })),
    champion_index: championIndex,
    client_seed: clientSeed,
    doctrine: "guardian",
  };
}

/**
 * THE CREATOR A WAR IS ATTRIBUTED TO: the side's CHAMPION, and only the
 * champion (ADR-0006 §4 `war_joins` / `wars_completed`).
 *
 * A roster is 10..50 cards of 10..50 DISTINCT creators (`validateJoinShape`,
 * `loadRoster`), so "every creator whose card was played" would put 10..50
 * fan-keyed rows — each needing its own HMAC — into a request batch the ADR
 * budgets at "+1–2 statements" (§8), and would make one join look like fifty
 * joins across fifty dashboards. The champion is the one creator the fan
 * NAMED, it is one row, and it is bounded.
 *
 * Returns null when a stored side cannot name one. Only a GHOST side can reach
 * that — its snapshot is replayed out of `war_matches.inputs` and its
 * `champion_index` is not re-validated on the way back in — and a replay that
 * cannot say whose card led is not a fact to attribute to anybody.
 */
function championPassport(s: SideSnapshot): string | null {
  const c = s.roster[s.champion_index];
  return c && typeof c.passport_id === "string" ? c.passport_id : null;
}

function toEngineSide(s: SideSnapshot): WarSide {
  return {
    player: s.entry,
    roster: s.roster.map((c) => ({ passportId: c.passport_id, stats: c.stats })),
    championIndex: s.champion_index,
    bonds: s.roster.map((c) => c.bond),
    ...(s.roster.some((c) => c.flag !== undefined) ? { flags: s.roster.map((c) => c.flag ?? 0) } : {}),
    doctrine: s.doctrine,
    clientSeed: s.client_seed,
  };
}

/** I21-Q30: stored Bond (war-bond.ts computeBond, 0..BOND_MAX) as a match on `dials` uses it — clamped, never rewritten. */
const matchBond = (bond: number, dials: Dials = DEFAULT_DIALS): number => Math.min(bond, dials.bondMax);

// ── Q642 (D-1030): the creator flag and the fan flag ─────────────────────────
/** Starting value of each flag on its capture day; each falls WAR_FLAG_DECAY_PER_DAY a day, unbounded below
 * (docs/WAR-BALANCE-SIM.md "the rulings"). Jason's D-1030 numbers. */
export const WAR_FLAG_START = Object.freeze({ creator: 7, fan: 3 });
export const WAR_FLAG_DECAY_PER_DAY = 1;
export type WarFlagKind = keyof typeof WAR_FLAG_START;

/** A flag's addend on the `dayHeld`-th UTC day it is held (day 1 = the capture day). Pure. */
export function flagValue(start: number, decayPerDay: number, dayHeld: number): number {
  if (!Number.isInteger(dayHeld) || dayHeld < 1) throw new RangeError("flagValue: dayHeld must be an integer >= 1");
  return start - decayPerDay * (dayHeld - 1);
}

/** The UTC day `nowMs` falls on, counted from `captureDay` (YYYY-MM-DD) as day 1. */
export function flagDayHeld(captureDay: string, nowMs: number): number {
  const from = Date.parse(`${captureDay}T00:00:00.000Z`);
  if (!/^\d{4}-\d{2}-\d{2}$/.test(captureDay) || !Number.isFinite(from)) throw new RangeError("flagDayHeld: bad capture day");
  const d = new Date(nowMs);
  return Math.floor((Date.UTC(d.getUTCFullYear(), d.getUTCMonth(), d.getUTCDate()) - from) / 86_400_000) + 1;
}

export interface WarFlagHolders {
  /** The flagged creator and the flag's value today; null when nobody holds it. */
  creator: { passport_id: string; value: number } | null;
  /** The flagged fan account and the flag's value today. */
  fan: { account_id: string; value: number } | null;
}

/** The holder of each flag at `nowMs`: the latest captured week (war_flags, 0253) that has begun, valued for today. */
export async function currentWarFlags(db: DbLike, nowMs: number): Promise<WarFlagHolders> {
  const today = new Date(nowMs).toISOString().slice(0, 10);
  const { results } = await db
    .prepare(
      `SELECT f.kind AS kind, f.passport_id AS passport_id, f.account_id AS account_id, f.capture_day AS capture_day FROM war_flags f
        WHERE f.week_start = (SELECT MAX(g.week_start) FROM war_flags g WHERE g.kind = f.kind AND g.week_start <= ?)`,
    )
    .bind(today)
    .all<{ kind: string; passport_id: string | null; account_id: string | null; capture_day: string }>();
  const out: WarFlagHolders = { creator: null, fan: null };
  for (const r of results) {
    const held = flagDayHeld(r.capture_day, nowMs);
    if (held < 1) continue;
    if (r.kind === "creator" && r.passport_id) out.creator = { passport_id: r.passport_id, value: flagValue(WAR_FLAG_START.creator, WAR_FLAG_DECAY_PER_DAY, held) };
    if (r.kind === "fan" && r.account_id) out.fan = { account_id: r.account_id, value: flagValue(WAR_FLAG_START.fan, WAR_FLAG_DECAY_PER_DAY, held) };
  }
  return out;
}

/**
 * Q642 — each card's flag addend for a side played by `account` (null for a ghost): the fan flag on every card of the
 * holder's side, plus the creator flag on EVERY card of the flagged creator, in any roster slot (D-0388: "the flag
 * rides any roster slot, but wins and losses are only counted by champion"; D-1059(2)). The one predicate both doors
 * use: matchTimeSide (the match) and reconViewFor (scouting). Pure.
 */
function cardFlags(s: SideSnapshot, account: string | null, flags: WarFlagHolders): number[] {
  const fan = flags.fan !== null && account !== null && flags.fan.account_id === account ? flags.fan.value : 0;
  return s.roster.map((c) => fan + (flags.creator !== null && c.passport_id === flags.creator.passport_id ? flags.creator.value : 0));
}

/**
 * The side a NEW match fights with (resolveAndSettle, the one door every pairing, recon and ghost match passes):
 * Bond clamped to the match's dials (I21-Q30), and each card's flag addend stamped (Q642, `cardFlags`). A stale
 * `flag` on a replayed ghost snapshot is dropped.
 */
function matchTimeSide(s: SideSnapshot, account: string | null, flags: WarFlagHolders): SideSnapshot {
  const perCard = cardFlags(s, account, flags);
  return {
    ...s,
    roster: s.roster.map((c, i) => {
      const { flag: _stale, ...card } = c;
      const flag = perCard[i]!;
      return flag === 0 ? { ...card, bond: matchBond(c.bond) } : { ...card, bond: matchBond(c.bond), flag };
    }),
  };
}

/** The complete, self-contained engine input for one match — everything
 * resolveMatch needs, so a replay re-derives from THIS ALONE (review F3: the

interface MatchInputs {
  server_seed: string;
  dials: Dials;
  sides: [SideSnapshot, SideSnapshot];
}

/** The canonical replay JSON — what replay_hash commits to and R2 stores. */
function replayJson(inputs: MatchInputs, log: MatchLog): string {
  return stableStringify({ engine: log.engine, inputs, log });
}

/**
 * Join the queue. Either pairs against the oldest valid waiting entry (the
 * match resolves and settles in THIS request) or becomes the waiting entry.
 * No card is escrowed on either path (QUEUE #236).
 */
export async function joinQueue(
  db: DbLike,
  store: ArtifactStore,
  accountId: string,
  input: QueueJoinInput,
  m: WarMinters,
  now?: () => string,
  fanSecret?: string,
  pubEnv?: AnalyticsTickEnv,
  pubCtx?: { waitUntil(p: Promise<unknown>): void },
  opts?: { recon?: false },
): Promise<JoinResult>;
export async function joinQueue(
  db: DbLike,
  store: ArtifactStore,
  accountId: string,
  input: QueueJoinInput,
  m: WarMinters,
  now: (() => string) | undefined,
  fanSecret: string | undefined,
  pubEnv: AnalyticsTickEnv | undefined,
  pubCtx: { waitUntil(p: Promise<unknown>): void } | undefined,
  opts: { recon: boolean },
): Promise<JoinResult | ReconJoinResult>;
export async function joinQueue(
  db: DbLike,
  store: ArtifactStore,
  accountId: string,
  input: QueueJoinInput,
  m: WarMinters,
  now: () => string = () => new Date().toISOString(),
  /** `ANALYTICS_FAN_SECRET`. Undefined is refused by `fanKey` (ADR-0006 §2). */
  fanSecret?: string,
  pubEnv?: AnalyticsTickEnv,
  pubCtx?: { waitUntil(p: Promise<unknown>): void },
  /** #241 (D-0920): `recon: true` opens a recon window on a LIVE pairing instead
   * of resolving it in this request. Off by default until the commit route and
   * the `sweepWarRecon` cron leg are wired (war-routes.ts, index.ts). */
  opts: { recon?: boolean } = {},
): Promise<JoinResult | ReconJoinResult> {
  validateJoinShape(input);
  // Chargeback ban (review A-014): the same accountBanned() gate every other
  // economic surface reads (refunds.ts docblock; war-routes.ts buy-card/pack;
  // index.ts checkout/gifts/rewards). Written when joinQueue moved property
  // through the ranked ante (removed, QUEUE #236); kept because a ranked
  // result is still a public record a suspended account must not write.
  if (await accountBanned(db, accountId)) refuse("account_suspended", 403);
  // I3-0289: the closure half of the same gate, in the primitive. The session
  // fold (`currentSessionEpoch`) already refuses a frozen caller at the route;
  // this is the door that does not depend on which route called it.
  if (await accountFrozen(db, accountId)) refuse("account_suspended", 403);
  const bracket = input.roster.length;
  const doorAt = now();
  const roster = await loadRoster(db, accountId, input.roster, doorAt);
  if (!roster) refuse(ROSTER_REFUSAL, 409);
  const statsCache = new Map<string, PowerRatings | null>();
  // Own-side stats gate BEFORE anything with side effects — an unrated /
  // suppressed roster refuses now, and priming the cache means a later
  // buildSideSnapshot for THIS fan can no longer fail (it was written so
  // nothing after the pre-#236 ante escrow could throw and orphan a stake).
  if (!(await snapshotStats(store, db, roster.map((r) => r.passport_id), statsCache, doorAt))) refuse(ROSTER_REFUSAL, 409);

  // One waiting entry per account, checked up front (the partial UNIQUE index
  // is the real enforcement) so a fan cannot stage a second waiting entry.
  const already = await db.prepare("SELECT 1 AS x FROM war_queue WHERE account_id = ? AND state = 'waiting'").bind(accountId).first();
  if (already) refuse("you already have a waiting entry — leave the queue first", 409);
  // RG-WAROPS:2 (I16-FXC): an entry inside an OPEN recon window is `matched` with no match yet; the same account joining
  // again would field the same cards into a second window. One open war per account.
  const inWindow = await db
    .prepare("SELECT 1 AS x FROM war_queue q WHERE q.account_id = ? AND q.state = 'matched' AND q.match_id IS NULL AND EXISTS (SELECT 1 FROM war_recon r WHERE r.entry_id = q.entry_id AND r.resolved_at IS NULL)")
    .bind(accountId)
    .first();
  if (inWindow) refuse("you are already in a war that has not finished — finish it first", 409);
  // I3-0126: the global admission cap, before the pairing / ghost / waiting writes.
  if (!(await consumeRateLimit(db, WAR_QUEUE_GLOBAL_KEY, Date.now(), WAR_QUEUE_GLOBAL_JOINS_PER_MINUTE, 60_000))) refuse("queue_busy", 429);

  // THE JOINER'S IDENTITY, minted ONCE (review F1). It labels this fan's side
  // in every pairing attempt and, if none pairs, becomes their waiting entry —
  // so the one id is valid on whichever path wins.
  const myEntry = m.entryId();
  if (!WQ_RE.test(myEntry)) refuse("bad entry id shape", 500);

  // ── Try to pair with a live waiting entry (oldest first) ──
  // QUEUE #236: nothing is escrowed to `myEntry`, so a throw here strands
  // nothing (the I3-0310 orphan-release that guarded this region is gone with
  // the stake it released).
  const paired = await tryPairWithWaiter(db, store, m, { accountId, bracket, roster, input, statsCache, myEntry, bEntryExisting: false, now, fanSecret, pubEnv, pubCtx, recon: opts.recon === true });
  if (paired) return { queued: false, ...paired };

  // ── Practice with an empty queue: play a ghost snapshot ──
  if (input.practice) {
    const ghost = await tryGhost(db, store, m, accountId, bracket, roster, input, statsCache, now, doorAt, fanSecret, pubEnv, pubCtx);
    if (ghost) return { queued: false, match: ghost };
  }

  // ── Become the waiting entry ──
  const entryId = myEntry;
  const serverSeed = m.serverSeed();
  if (!/^[0-9a-f]{8,128}$/.test(serverSeed)) refuse("bad server seed shape", 500);
  const seedHash = await sha256Hex(serverSeed);
  const ts = now();
  // The key is awaited BEFORE the batch is built (ADR-0006 §2: no statement is
  // ever built from a promise). The index is in range by `validateJoinShape`
  // and the entry by `loadRoster`, so this side always names a champion — see
  // `championPassport` for why the champion, and only the champion.
  const champion = roster[input.champion_index]!.passport_id;
  const joinKey = await fanKey(fanSecret, accountId, champion);
  const stmts: DbStatement[] = [];
  stmts.push(
    db
      .prepare(
        // QUEUE #236: `ante_instance` is never written (NULL) — no stake.
        `INSERT INTO war_queue (entry_id, account_id, bracket, practice, roster, champion_index, ante_instance, client_seed, server_seed, server_seed_hash, state, created_at, updated_at)
         VALUES (?, ?, ?, ?, ?, ?, NULL, ?, ?, ?, 'waiting', ?, ?)`,
      )
      .bind(
        entryId,
        accountId,
        bracket,
        input.practice ? 1 : 0,
        JSON.stringify(input.roster),
        input.champion_index,
        input.client_seed,
        serverSeed,
        seedHash,
        ts,
        ts,
      ),
  );
  // In the same batch as the entry, gated only on the joiner's qualification
  // (D-1080(2), JOINER_QUALIFIED_SQL above): if the insert fails the
  // batch fails and this row goes with it (ADR-0006 §2 — the event exists iff
  // its fact does). The scope is the WAR EVENT GROUP (I3-1241, D-0913(3):
  // edition week + bracket, `warEventId`), the unit many fans share — a
  // per-entry scope held one fan forever and its cell could never reach the
  // floor. `war_completed` scopes the MATCH instead — see resolveAndSettle.
  stmts.push(...outboxStatements(db, { kind: "war_joined", passportId: champion, occurredDay: occurredDay(ts), fanKey: joinKey, scopeId: warEventId(ts, bracket), dims: { mode: input.practice ? "practice" : "ranked" } }, { accountId, extraWhere: { sql: JOINER_QUALIFIED_SQL, params: [accountId] } })); // D-1080(2): counted only for a qualified fan
  try {
    const span = outboxSpan(db);
    const spanRes = await db.batch([span.open, ...stmts, span.close]);
    const seqs = outboxSpanSeqs(spanRes[0], spanRes[spanRes.length - 1]);
    if (pubEnv && pubCtx) publishAfterCommit(pubEnv, pubCtx, seqs);
  } catch (e) {
    const msg = e instanceof Error ? e.message : String(e);
    // The partial UNIQUE index: one waiting entry per account.
    if (/unique/i.test(msg) && /idx_wq_one_waiting|war_queue/i.test(msg)) refuse("you already have a waiting entry — leave the queue first", 409);
    // QUEUE #236: the NULL-PK escrow validator is gone, so any other failure
    // is a real fault, not a refusal — it propagates unchanged.
    throw e;
  }
  // ── CW-01: THE GUARANTEED CONTINUATION ──
  // The waiting row now exists, so anything that arrived (or was passed over)
  // while this request was working is visible to a second bounded pass — and
  // this pass CLAIMS the row just inserted instead of inserting another. This
  // is what makes two overlapping empty-queue joins converge on one match with
  // no third arrival and no cancel/rejoin, and what lets a waiter that sat
  // beyond an exhausted candidate page be reached once the stale entries ahead
  // of it have been voided by the first pass.
  //
  // BEST-EFFORT, DELIBERATELY. The entry is already durably
  // committed above; this is an opportunistic attempt to settle them now. A
  // failure here must leave the fan queued (which is a correct, recoverable
  // state they can leave), never turn a landed queue join into an error — so
  // the throw is swallowed and the ordinary `queued` answer is returned.
  try {
    const late = await tryPairWithWaiter(db, store, m, { accountId, bracket, roster, input, statsCache, myEntry: entryId, bEntryExisting: true, now, fanSecret, pubEnv, pubCtx, recon: opts.recon === true });
    if (late) return { queued: false, ...late };
  } catch {
    /* stay queued — the entry is committed and cancellable,
       UNLESS an independent joiner claimed this exact entry as THEIR
       candidate in the meantime — checked below (I3-1501). */
  }
  // I3-1501: this pass's own five-candidate page can miss a live waiter and
  // still return null/throw while, in a fully separate transaction, a
  // DIFFERENT joiner picked THIS entry as their own candidate and committed
  // that match first — this entry's row is then genuinely 'matched', not
  // 'waiting', even though every branch above concluded "stay queued".
  // Re-read this entry's own state before answering, so a caller is never
  // told `queued: true` for an entry the database already calls settled.
  const afterLate = await db.prepare("SELECT state, match_id FROM war_queue WHERE entry_id = ?").bind(entryId).first<{ state: string; match_id: string | null }>();
  if (afterLate?.state === "matched" && afterLate.match_id) {
    const alreadyMatched = await getMatch(db, accountId, afterLate.match_id);
    if (alreadyMatched) return { queued: false, match: alreadyMatched };
  }
  // #241: claimed into a recon window by another joiner — matched, no match row yet.
  if (afterLate?.state === "matched" && !afterLate.match_id) {
    const r = await db.prepare("SELECT match_id FROM war_recon WHERE entry_id = ?").bind(entryId).first<{ match_id: string }>();
    const view = r ? await reconViewFor(db, r.match_id, accountId, Date.parse(now())) : null;
    if (view) return { queued: false, recon: view };
  }
  return { queued: true, entry_id: entryId, server_seed_hash: seedHash };
}

/**
 * CW-01 — ONE BOUNDED PAIRING PASS, callable BEFORE the caller becomes a
 * waiting entry and again AFTER they have become one.
 *
 * It was inline in `joinQueue` and ran exactly once, before the insert. That
 * ordering is what left compatible players stranded: two fans arriving at an
 * empty bracket can both finish this scan while it is empty, both insert a
 * valid waiting row (the UNIQUE index is per ACCOUNT, so it correctly permits
 * both), and then nothing in the system ever tries to pair them. Polling only
 * reads status and leaving only cancels, so the pair could sit there until
 * some third fan happened to arrive. The same shape
 * strands a pair when this scan's bounded page is filled by stale candidates
 * that get voided in passing: the valid waiter sat one row beyond the page.
 *
 * `bEntryExisting` is the whole difference between the two calls: on the second
 * one the caller's row already exists and is claimed rather than inserted, so
 * BOTH sides go through the same guarded atomic claim and two continuations
 * racing each other still produce exactly one match — the loser's batch fails
 * its validator, returns null, and the loser simply stays queued.
 */
async function tryPairWithWaiter(
  db: DbLike,
  store: ArtifactStore,
  m: WarMinters,
  p: {
    accountId: string;
    bracket: number;
    roster: { instance_id: string; passport_id: string }[];
    input: QueueJoinInput;
    statsCache: Map<string, PowerRatings | null>;
    myEntry: string;
    bEntryExisting: boolean;
    now: () => string;
    fanSecret?: string | undefined;
    pubEnv?: AnalyticsTickEnv | undefined;
    pubCtx?: { waitUntil(p: Promise<unknown>): void } | undefined;
    /** #241: open a recon window instead of resolving now. */
    recon?: boolean;
  },
): Promise<{ match: MatchView } | { recon: ReconView } | null> {
  const { accountId, bracket, roster, input, statsCache, myEntry, now, fanSecret, pubEnv, pubCtx } = p;
  // ── Try to pair with a live waiting entry ──
  // The bounded PAGE stays oldest-first (created_at ASC): that ordering is
  // what CW-01's liveness fix depends on (a fixed, deterministic page is what
  // lets a second pass reach a waiter the first pass's page missed; s070-war
  // pins the exact page composition on it). D-0532 / D-0549 (Jason): opponent
  // SELECTION off that page is RANDOM, not "try oldest waiter first", and the
  // stated purpose is PREVENTING COLLUSION. SELECTION ONLY: the CLAIM path
  // below (the guarded UPDATE, the claimToken stamped into `updated_at`, the
  // validator) is untouched, and nothing here reads `updated_at` as a time
  // (QUEUE #411).
  //
  // Two independent things make the draw unarrangeable, and W3-08's attacker
  // measured that the old code had NEITHER:
  //   1. the POOL is PAIR_WINDOW wide, not the 5 oldest (W308-A-18);
  //   2. the permutation comes from a CSPRNG, not Math.random() (W308-A-17).
  // W3-08 JUDGE, overturning FIX6's claim that widening the window CLOSED
  // A-18. It did not, and the judge's demonstration is the reason this is now
  // `ORDER BY RANDOM()`: while the pool was an AGE-ORDERED PREFIX
  // (`ORDER BY created_at ASC LIMIT n`) its membership stayed ATTACKER-ORDERABLE
  // at ANY width. Arrival order is the one thing a colluding group fully
  // controls, so a group holding the oldest `n` entries of a deeper bracket
  // owned the draw with probability 1 exactly as before — the fix moved the
  // price of the attack from 5 entries to `n`, and called that closure.
  //
  // THE POOL IS NOW A RANDOM SAMPLE OF THE WHOLE WAITING SET, not a prefix of
  // it. There is no ordering a joiner or a colluder can influence, because
  // there is no ordering: every waiter in the bracket has the same chance of
  // being in the sample regardless of when it arrived. That is the property
  // D-0549 asks for, and a prefix of any length cannot have it.
  //
  // TWO SOURCES ON PURPOSE. SQLite's `RANDOM()` decides MEMBERSHIP (it is a
  // sampler, and is not claimed as a cryptographic guarantee); `shuffleCsprng`
  // below decides ORDER within the sample, from `crypto.getRandomValues`.
  // Neither alone is enough: a CSPRNG shuffle of an attacker-chosen set still
  // lands inside the set, and a random sample handed to the loop in a
  // predictable order is still predictable.
  //
  // COST, stated because R1's live item this same batch was an unbounded
  // aggregate. `ORDER BY RANDOM()` sorts the matching rows rather than walking
  // an index prefix, and the matching set is one bracket's WAITING entries —
  // the population `idx_wq_pairing (bracket, practice, state, created_at)`
  // already narrows to. That set is self-limiting: every join consumes a
  // waiter, so a bracket holding enough simultaneous waiters for this sort to
  // matter is a pathology in its own right. Paying a bracket-scoped sort per
  // join is the price of a draw nobody can arrange, and it is the right trade.
  // I3-0561 (D-0834, Jason: full delegation) — ranked matchmaking pairs only
  // accounts at least RANKED_MIN_ACCOUNT_AGE_MS old, so an alt in a quiet
  // bracket cannot pair with its own main and farm a ranked record between
  // them (it was written for the ante, removed by QUEUE #236; the ranked
  // record it also protects is unchanged). Practice is exempt (the `? = 1` arm).
  const passAt = now();
  const rankedAgeCutoff = new Date(Date.parse(passAt) - RANKED_MIN_ACCOUNT_AGE_MS).toISOString();
  // I4RGWAR-F2 (I3-0561, D-0843(4)) — the age gate above only checked the
  // WAITING side. A fresh alt JOINING an aged main WAITER self-paired because
  // `accountId` (the joiner) was never age-checked. Same cutoff, same
  // practice escape hatch, applied to the joiner's own account.
  const { results: pool } = await db
    .prepare(
      `SELECT entry_id, account_id, bracket, practice, roster, champion_index, ante_instance, client_seed, server_seed, server_seed_hash, state, created_at
         FROM war_queue
        WHERE bracket = ? AND practice = ? AND state = 'waiting' AND account_id != ?
          AND (? = 1 OR EXISTS (SELECT 1 FROM accounts o WHERE o.account_id = war_queue.account_id AND o.created_at <= ?))
        ORDER BY RANDOM() LIMIT ${PAIR_WINDOW}`,
    )
    .bind(bracket, input.practice ? 1 : 0, accountId, input.practice ? 1 : 0, rankedAgeCutoff)
    .all<QueueRow>();
  if (!input.practice) {
    const joinerOld = await db.prepare("SELECT 1 AS x FROM accounts WHERE account_id = ? AND created_at <= ?").bind(accountId, rankedAgeCutoff).first();
    if (!joinerOld) pool.length = 0;
  }
  shuffleCsprng(pool);
  // W308-Q1: the pool is NO LONGER truncated here. The two budgets below do the
  // bounding, because truncating to PAIR_CANDIDATES up front is what let a
  // salted bracket spend a joiner's whole allowance on rubble. The shuffle
  // still runs first, so what the budgets consume is a random permutation of
  // the whole sample and never an arrival-ordered prefix of it.
  let attemptsLeft = PAIR_CANDIDATES;
  let reapsLeft = PAIR_REAP_BUDGET;

  for (const cand of pool) {
    // A genuine match attempt is owed a slot; a reap is not. Both budgets must
    // be exhausted before the joiner is queued, and either one hitting zero
    // stops the loop — attempts because that is the work bound, reaps because
    // no single join should be made to walk the whole window for someone else.
    if (attemptsLeft <= 0 || reapsLeft <= 0) break;
    // Same gate on the WAITING side: a fan banned OR closed (suspended /
    // terminated, I3-0343) after they queued must not write a ranked result
    // either -- the CALLER's own account gets this for free from the session
    // fold (fan-auth.ts refuses a frozen caller before any route runs), but
    // the waiting candidate is not the caller and was never re-checked by
    // that fold, so both halves of the required set are asked explicitly
    // here. Treat a since-banned-or-closed waiting entry as stale, exactly
    // like an invalid roster (void it, own escrow released to its own owner,
    // never to the joiner) and try the next candidate.
    if ((await accountBanned(db, cand.account_id)) || (await accountFrozen(db, cand.account_id))) {
      await voidEntry(db, cand.entry_id, now);
      reapsLeft -= 1; // W308-Q1: reaping rubble is not a match attempt
      continue;
    }
    const candRosterIds = ((): string[] | null => {
      try {
        const v = JSON.parse(cand.roster) as unknown;
        return Array.isArray(v) && v.every((x): x is string => typeof x === "string") ? v : null;
      } catch {
        return null;
      }
    })();
    // Re-verify the WAITING side's world at resolve time: still owns every
    // roster card, creators still in the game, stats still served. A stale
    // entry is VOIDED in passing (its escrow released) and the next candidate
    // tried — never matched against a roster its owner no longer has.
    const candRoster = candRosterIds ? await candRosterValid(db, cand, candRosterIds, passAt) : null;
    // I3-0011 / I3-0272 (D-0567). engine.ts's SERVER PRECONDITION 1 requires the
    // revealed seed to be re-checked against the commitment published at join,
    // and nothing did: the hash the match VIEW serves is RECOMPUTED from the
    // revealed seed (`resolveAndSettle`, `sha256Hex(a.serverSeed)`), so it
    // agrees with whatever seed was used — including one substituted after the
    // waiter was shown its commitment. Comparison, not recomputation.
    //
    // A candidate whose commitment does not open is REFUSED AT RESOLVE: voided
    // in passing with its escrow released, exactly as a stale roster is, and
    // the next candidate tried. It is not refused at the JOIN seam, because the
    // corrupted row belongs to the waiting side and denying the arriving
    // joiner would punish the wrong account.
    const candCommitted =
      cand.server_seed !== null && cand.server_seed_hash !== null && (await sha256Hex(cand.server_seed)) === cand.server_seed_hash;
    if (!candRoster || cand.server_seed === null || !candCommitted) {
      await voidEntry(db, cand.entry_id, now);
      reapsLeft -= 1; // W308-Q1
      continue;
    }
    const ts = now();
    const nowMs = Date.parse(ts);
    // I3-1882 (DECIDED, I4-02-R2): no RANKED rematch between the same two
    // accounts inside RANKED_REMATCH_MS. `account_id != ?` above only stops one
    // account playing itself; this bounds how often any pair of accounts can
    // write a ranked result together. The candidate is SKIPPED, never voided
    // (it is a valid entry, just not for this joiner) and charges no budget:
    // the loop is already bounded by the PAIR_WINDOW-row pool.
    if (!input.practice && (await rankedRecently(db, accountId, cand.account_id, nowMs))) continue;
    const sideA = await buildSideSnapshot(db, store, cand.account_id, cand.entry_id, candRoster, cand.champion_index, cand.client_seed, nowMs, statsCache);
    if (!sideA) {
      await voidEntry(db, cand.entry_id, now);
      reapsLeft -= 1; // W308-Q1: an unservable snapshot is staleness too
      continue;
    }
    // PAST EVERY STALENESS CHECK: this candidate is a REAL opponent, and costs
    // an attempt whatever happens next (a settle failure, a lost claim race).
    // Decremented HERE and not earlier, because a candidate voided above was
    // never an opponent — charging it to the attempt budget is the whole
    // W308-Q1 defect.
    attemptsLeft -= 1;
    // buildSideSnapshot for the joiner cannot return null here — the own-side
    // stats gate at the top primed the cache — but keep the guard for safety.
    const sideB = await buildSideSnapshot(db, store, accountId, myEntry, roster, input.champion_index, input.client_seed, nowMs, statsCache);
    if (!sideB) refuse(ROSTER_REFUSAL, 409);

    // QUEUE #236: nothing is escrowed to `myEntry`, so a throw from the
    // settle has nothing to release (C-33 / PS-071 / I3-0107 released the
    // up-front ante here) and simply propagates.
    const settleArgs: SettleArgs = {
      ts,
      nowMs,
      bracket,
      practice: input.practice,
      ghost: false,
      serverSeed: cand.server_seed,
      sideA,
      sideB,
      aAccount: cand.account_id,
      bAccount: accountId,
      aEntryExisting: cand.entry_id,
      ...(p.bEntryExisting ? { bEntryExisting: true } : {}),
      joinerInput: input,
      fanSecret,
      pubEnv,
      pubCtx,
    };
    // #241: under recon the pairing CLAIMS both entries and opens the window;
    // the engine runs later, in `resolveRecon`, never in this request.
    if (p.recon) {
      const opened = await openRecon(db, m, { ...settleArgs, pubEnv, pubCtx });
      if (opened) return { recon: opened };
      continue;
    }
    const settled = await resolveAndSettle(db, store, m, { ...settleArgs, pubEnv, pubCtx });
    if (settled) return { match: settled };
    // Settle refused (the candidate was claimed by a racing joiner, or a
    // guard failed) — try the next candidate.
  }
  return null;
}



/** I3-1882: one ranked result per pair of accounts per rolling window. */
const RANKED_REMATCH_MS = 86_400_000;

async function rankedRecently(db: DbLike, x: string, y: string, nowMs: number): Promise<boolean> {
  const row = await db
    .prepare(
      `SELECT 1 AS n FROM war_matches
        WHERE practice = 0 AND ghost = 0 AND resolved_ms > ?
          AND ((a_account = ? AND b_account = ?) OR (a_account = ? AND b_account = ?)) LIMIT 1`,
    )
    .bind(nowMs - RANKED_REMATCH_MS, x, y, y, x)
    .first<{ n: number }>();
  return row !== null && row !== undefined;
}

/** The waiting entry's roster re-verified at resolve: still owned by them,
 * living, escrow-free (a pre-#236 entry's ANTE may still be escrowed; roster cards never are), one per
 * creator, creators still in the game. */
async function candRosterValid(db: DbLike, cand: QueueRow, rosterIds: string[], nowIso: string): Promise<{ instance_id: string; passport_id: string }[] | null> {
  if (!(BRACKETS as readonly number[]).includes(rosterIds.length) || rosterIds.length !== cand.bracket) return null;
  if (!Number.isInteger(cand.champion_index) || cand.champion_index < 0 || cand.champion_index >= rosterIds.length) return null;
  return loadRoster(db, cand.account_id, rosterIds, nowIso);
}

/** Void a stale waiting entry and release its escrow — the release is
 * predicated on the void having landed, so a racing settle keeps the stake. */
async function voidEntry(db: DbLike, entryId: string, now: () => string): Promise<void> {
  await db.batch([
    db.prepare("UPDATE war_queue SET state = 'voided', updated_at = ? WHERE entry_id = ? AND state = 'waiting'").bind(now(), entryId),
    db
      .prepare(
        "UPDATE card_instances SET escrowed_by = NULL WHERE escrowed_by = ? AND EXISTS (SELECT 1 FROM war_queue WHERE entry_id = ? AND state = 'voided')",
      )
      .bind(entryId, entryId),
  ]);
}

interface SettleArgs {
  ts: string;
  nowMs: number;
  bracket: number;
  practice: boolean;
  ghost: boolean;
  serverSeed: string;
  sideA: SideSnapshot;
  sideB: SideSnapshot;
  aAccount: string | null; // null = ghost side
  bAccount: string;
  /** Set when side A is a live waiting entry to claim; undefined for ghosts. */
  aEntryExisting?: string;
  /** CW-01: set when side B's queue row ALREADY EXISTS as `waiting` (the
   * post-insertion pairing continuation), so it is CLAIMED like side A's rather
   * than inserted. Undefined on the ordinary path, where the joiner's row is
   * born `matched`. */
  bEntryExisting?: boolean;
  joinerInput: QueueJoinInput;
  fanSecret: string | undefined;
  pubEnv?: AnalyticsTickEnv | undefined;
  pubCtx?: { waitUntil(p: Promise<unknown>): void } | undefined;
  /** #241: resolving an OPEN recon window. Both entries were already claimed
   * (`matched`, match_id NULL) by `openRecon`; the match id was minted there; this
   * attempt's claim is `token` stamped into both war_recon rows' `resolver`. */
  recon?: { matchId: string; token: string };
}

/** Statement 1 of a pairing batch (resolveAndSettle, openRecon): the joiner's
 * entry row, born `matched` — or, on the CW-01 continuation, CLAIMED. */
function joinerRowStatement(db: DbLike, a: SettleArgs, claimToken: string): DbStatement {
  return a.bEntryExisting
      ? db
          .prepare("UPDATE war_queue SET state = 'matched', updated_at = ? WHERE entry_id = ? AND account_id = ? AND state = 'waiting'")
          .bind(claimToken, a.sideB.entry, a.bAccount)
      : db
          .prepare(
            `INSERT INTO war_queue (entry_id, account_id, bracket, practice, roster, champion_index, ante_instance, client_seed, server_seed, server_seed_hash, state, match_id, created_at, updated_at)
             VALUES (?, ?, ?, ?, ?, ?, NULL, ?, NULL, NULL, 'matched', NULL, ?, ?)`,
          )
          .bind(
            a.sideB.entry,
            a.bAccount,
            a.bracket,
            a.practice ? 1 : 0,
            JSON.stringify(a.joinerInput.roster),
            a.joinerInput.champion_index,
            a.joinerInput.client_seed,
            a.ts,
            a.ts,
          );
}

/** Statement 2 of a pairing batch: claim the waiting entry (the atomic pairing). */
function claimWaitingStatement(db: DbLike, entryId: string, claimToken: string): DbStatement {
  return db.prepare("UPDATE war_queue SET state = 'matched', updated_at = ? WHERE entry_id = ? AND state = 'waiting'").bind(claimToken, entryId);
}

/** The INV-058 settle guards (I3-1306 / I3-0289 — see resolveAndSettle): roster
 * ownership, creator standing + consent bound by value, account standing, as
 * one SQL conjunction suffix and its binds. Shared VERBATIM by `openRecon`
 * (#241), so a window opens under exactly the predicate a match settles under.
 * null = a side's creators are not all opted in right now. */
async function settleGuards(
  db: DbLike,
  a: Pick<SettleArgs, "ts" | "aAccount" | "bAccount" | "sideA" | "sideB" | "recon">,
): Promise<{ sql: string; binds: unknown[] } | null> {
  const liveSides: { account: string; ids: string[] }[] = [];
  if (a.aAccount !== null) liveSides.push({ account: a.aAccount, ids: a.sideA.roster.map((c) => c.instance_id) });
  liveSides.push({ account: a.bAccount, ids: a.sideB.roster.map((c) => c.instance_id) });
  const creators = [...new Set([...a.sideA.roster, ...a.sideB.roster].map((c) => c.passport_id))];
  const consentSeq =
    (
      await db
        .prepare("SELECT COALESCE(MAX(seq), 0) AS s FROM trust_events WHERE subject IN (SELECT value FROM json_each(?)) AND event_type IN ('war_opt_in','war_opt_out')")
        .bind(JSON.stringify(creators))
        .first<{ s: number }>()
    )?.s ?? 0;
  // Per side: `allWarOptedIn` refuses more than 50 passports, one roster's maximum. At the settle's own instant `a.ts`
  // (I34-FX3-INSTANT, I34-JN-23 SEAMA-WARMATCH-INSTANT): the lapsed-grant term is the door's, not the wall clock's.
  for (const side of [a.sideA, a.sideB]) {
    if (!(await allWarOptedIn(db, side.roster.map((c) => c.passport_id), a.ts))) return null;
  }
  const rosterCheck = liveSides
    .map(() => " AND (SELECT COUNT(*) FROM card_instances WHERE instance_id IN (SELECT value FROM json_each(?)) AND owner_id = ? AND consumed_at IS NULL AND escrowed_by IS NULL) = ?")
    .join("");
  const rosterBinds = liveSides.flatMap((side) => [JSON.stringify(side.ids), side.account, side.ids.length]);
  const standingCheck = ` AND (SELECT COUNT(*) FROM passports p WHERE p.passport_id IN (SELECT value FROM json_each(?)) AND COALESCE(p.retracted, 0) <> 1
            AND p.claim_state IN ('claimed','verified') AND NOT EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = p.passport_id)) = ?
        AND (SELECT COALESCE(MAX(seq), 0) FROM trust_events WHERE subject IN (SELECT value FROM json_each(?)) AND event_type IN ('war_opt_in','war_opt_out')) = ?
        AND NOT EXISTS (SELECT 1 FROM account_closures WHERE account_id IN (?, ?) AND state IN ${FROZEN_CLOSURE_STATES_SQL})
        AND NOT EXISTS (SELECT 1 FROM accounts WHERE account_id IN (?, ?) AND banned_at IS NOT NULL)`;
  const accountA = a.aAccount ?? a.bAccount;
  const standingBinds = [JSON.stringify(creators), creators.length, JSON.stringify(creators), consentSeq, accountA, a.bAccount, accountA, a.bAccount];
  // RG-WAROPS:1 (I16-FXC, D-0925(2)): a recon window is voided when a roster card or a creator's consent CHANGED inside
  // it, not only when the state at resolution differs — an opt-out and back in, or a card gifted away and back, leaves
  // the value checks above equal. Inside = at or after the window's war_recon.created_at (the open batch's instant).
  if (a.recon) {
    const opened = "(SELECT MIN(created_at) FROM war_recon WHERE match_id = ?)";
    const ids = [...a.sideA.roster, ...a.sideB.roster].map((c) => c.instance_id);
    const windowCheck = ` AND NOT EXISTS (SELECT 1 FROM trust_events WHERE subject IN (SELECT value FROM json_each(?)) AND event_type IN ('war_opt_in','war_opt_out') AND created_at >= ${opened})
        AND NOT EXISTS (SELECT 1 FROM card_instance_log WHERE instance_id IN (SELECT value FROM json_each(?)) AND at >= ${opened})`;
    return { sql: rosterCheck + standingCheck + windowCheck, binds: [...rosterBinds, ...standingBinds, JSON.stringify(creators), a.recon.matchId, JSON.stringify(ids), a.recon.matchId] };
  }
  return { sql: rosterCheck + standingCheck, binds: [...rosterBinds, ...standingBinds] };
}

/** Run the engine, then commit the WHOLE outcome — joiner entry row, waiting
 * entry claim, legacy-escrow release, match record — in ONE atomic batch with a
 * NULL-PK force-rollback validator. Returns the view, or null when the batch
 * refused (caller retries the next candidate). */
async function resolveAndSettle(db: DbLike, store: ArtifactStore, m: WarMinters, a: SettleArgs): Promise<MatchView | null> {
  // The parameter is rebound (not shadowed) so `a.pubEnv` stays the caller's slot (s003-publish-wiring slotOf).
  const flags = await currentWarFlags(db, a.nowMs);
  a = { ...a, sideA: matchTimeSide(a.sideA, a.aAccount, flags), sideB: matchTimeSide(a.sideB, a.bAccount, flags) };
  const log = await resolveMatch({ serverSeed: a.serverSeed, sides: [toEngineSide(a.sideA), toEngineSide(a.sideB)], dials: DEFAULT_DIALS });
  const matchId = a.recon ? a.recon.matchId : m.matchId();
  if (!WM_RE.test(matchId)) refuse("bad match id shape", 500);
  // CW-02: a transaction that LOSES the race for a waiting entry must not be
  // able to read the room's committed `state = 'matched'` as proof it did
  // the claiming -- that state is equally true after the WINNER's guarded
  // UPDATE commits. Stamping THIS attempt's own globally-unique matchId into
  // the claimed row's `updated_at` at claim time, then requiring the exact
  // match in the settle validator below, turns the check from "is this row
  // claimed by *someone*" into "did *my* guarded UPDATE land" -- the loser's
  // own no-op UPDATE (blocked by `state = 'waiting'`) leaves the winner's
  // token in place, so the loser's own check fails and it never reaches the
  // war_matches INSERT.
  const claimToken = matchId;
  const inputs: MatchInputs = { server_seed: a.serverSeed, dials: DEFAULT_DIALS, sides: [a.sideA, a.sideB] };
  const replay = replayJson(inputs, log);
  const replayHash = await sha256Hex(replay);
  const winner: "a" | "b" = log.result.winner === 0 ? "a" : "b";
  const summary = {
    winner,
    by: log.result.by,
    rounds: log.result.rounds,
    hp_remaining: log.result.hpRemaining,
    cards_remaining: log.result.cardsRemaining,
    rerolls_used: log.result.rerollsUsed,
    champion_survived: log.result.championSurvived,
  };


  // Both awaited before a single statement is built (ADR-0006 §2). The JOINER
  // is the only side that joins in THIS request — side A either queued earlier
  // (and emitted its own `war_joined` then) or is a ghost replay that nobody
  // joined — so exactly one `war_joined` is emitted here.
  const joinerChampion = championPassport(a.sideB)!;
  // #241: a recon resolution emits no `war_joined` (openRecon did), so it mints no key.
  const joinKey = a.recon ? null : await fanKey(a.fanSecret, a.bAccount, joinerChampion);
  // I3-2164: ONE `war_completed` per creator per match — a creator whose card
  // led BOTH sides fought one war, not two. I3-0456 (DECIDED, I4-02-R2): a
  // GHOST match emits no war analytics at all — nobody played the ghost's
  // champion and the joiner had no opponent — so both lists are empty for it.
  const completedFor = a.ghost ? [] : [...new Set([championPassport(a.sideA), joinerChampion].filter((p): p is string => p !== null))];

  // I3-1306 / I3-0289 (INV-058 CLAIM-THEN-ACT): every fact the pairing pre-read
  // (roster ownership, creator standing and consent, account standing) is
  // re-asserted INSIDE the settlement batch by the validator at 4, so a write
  // landing between the pre-read and the batch fails the settle instead of
  // being settled over. Consent is an event replay, so it is bound by VALUE:
  // the last consent seq is read first, consent is re-checked after that read,
  // and the validator requires the seq unchanged. A ghost side is nobody's
  // cards, so only its creators are bound, never an owner.
  const g = await settleGuards(db, a);
  if (!g) return null;

  const stmts: DbStatement[] = [];
  // 1. The joiner's entry row (state 'matched' from birth — never waits).
  //    CW-01: on the post-insertion continuation the row is already there and
  //    `waiting`, so it is CLAIMED with the same guarded UPDATE side A gets.
  //    Both sides then contend through the identical atomic claim, and the
  //    validator at 4 (which already required this entry to read `matched` with
  //    THIS match id) needs no change: a continuation that loses the race
  //    changes nothing and its whole batch rolls back.
  // #241: under recon both entries were claimed by `openRecon`; this attempt's
  //    claim is its `resolver` token on BOTH war_recon rows, landing only while
  //    neither row is resolved (a racing resolver's token makes this a no-op
  //    and the validator at 4 refuses the batch).
  if (a.recon) {
    stmts.push(db.prepare("UPDATE war_recon SET resolver = ?, resolved_at = ? WHERE match_id = ? AND resolved_at IS NULL").bind(a.recon.token, a.ts, a.recon.matchId));
  } else {
    stmts.push(joinerRowStatement(db, a, claimToken));
  }
  // 2. Claim the waiting entry (live matches): the atomic pairing.
  if (a.aEntryExisting && !a.recon) stmts.push(claimWaitingStatement(db, a.aEntryExisting, claimToken));
  // 3. QUEUE #236 (D-0870(20)): NO CARD MOVES. A waiting entry that escrowed
  //    an ante BEFORE staking was removed still holds it; settling releases it
  //    to its own owner (`escrowed_by` is on the owner's own row, so a release
  //    can only ever hand a card back). Same atomic batch as the validator at
  //    4, so a settle that rolls back releases nothing and the entry keeps
  //    it until it settles, leaves (`leaveQueue`) or is voided (`voidEntry`) —
  //    the three paths that end a waiting entry, all of which release. A stake
  //    whose entry ROW is gone is released by the cron leg `sweepOrphanEscrow`
  //    (war-instances.ts, D-0871(1)).
  if (a.aEntryExisting) stmts.push(releaseEscrowStatement(db, a.aEntryExisting));
  // 4. The validator/committer: the match row exists ONLY if the pairing
  //    claim landed (live) and the joiner row landed.
  //    #241: a recon resolution's claim is BOTH war_recon rows carrying THIS
  //    attempt's token, and both entries still `matched` with no match row.
  const claimCheck = a.recon
    ? "(SELECT COUNT(*) FROM war_recon WHERE match_id = ? AND resolver = ?) = 2 AND (SELECT COUNT(*) FROM war_queue WHERE entry_id IN (?, ?) AND state = 'matched' AND match_id IS NULL) = 2"
    : a.aEntryExisting
      ? "(SELECT COUNT(*) FROM war_queue WHERE entry_id = ? AND state = 'matched' AND updated_at = ?) = 1"
      : "1 = 1";
  const claimBinds = a.recon ? [a.recon.matchId, a.recon.token, a.sideA.entry, a.sideB.entry] : a.aEntryExisting ? [a.aEntryExisting, claimToken] : [];
  // CW-02, side B: only the post-insertion continuation (`a.bEntryExisting`)
  // claims an EXISTING row through the same racy guarded UPDATE as side A
  // (statement 1 above) and needs the identical own-claim proof; the ordinary
  // path's row is a fresh INSERT under a freshly minted entry_id (born
  // 'matched'), which nobody else can claim away, so it keeps the plain
  // state check.
  const bClaimCheck = a.recon
    ? "1 = 1"
    : a.bEntryExisting
      ? "(SELECT COUNT(*) FROM war_queue WHERE entry_id = ? AND state = 'matched' AND updated_at = ?) = 1"
      : "(SELECT COUNT(*) FROM war_queue WHERE entry_id = ? AND state = 'matched') = 1";
  const bClaimBinds = a.recon ? [] : a.bEntryExisting ? [a.sideB.entry, claimToken] : [a.sideB.entry];
  stmts.push(
    db
      .prepare(
        `INSERT INTO war_matches (match_id, bracket, practice, ghost, a_entry, b_entry, a_account, b_account, winner, decided_by, rounds, inputs, summary, replay_hash, ante_a, ante_b, ante_transferred, resolved_at, resolved_ms)
         SELECT CASE WHEN ${claimCheck}
                      AND ${bClaimCheck}${g.sql}
                THEN ? ELSE NULL END,
                ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, NULL, NULL, NULL, ?, ?`,
      )
      .bind(
        ...claimBinds,
        ...bClaimBinds,
        ...g.binds,
        matchId,
        a.bracket,
        a.practice ? 1 : 0,
        a.ghost ? 1 : 0,
        a.sideA.entry,
        a.sideB.entry,
        a.aAccount,
        a.bAccount,
        winner,
        log.result.by,
        log.result.rounds,
        stableStringify(inputs),
        stableStringify(summary),
        replayHash,
        a.ts,
        a.nowMs,
      ),
  );
  // 4b. Backfill war_queue.match_id NOW that the war_matches row (whose
  //     match_id THIS references, migration 0100) exists — setting it any
  //     earlier (in step 1/2, alongside the state='matched' claim) trips
  //     `war_queue.match_id REFERENCES war_matches(match_id)` under SQLite's
  //     immediate (non-deferred) FK checking: the referencing UPDATE/INSERT
  //     ran before the referenced row existed, on every single pairing
  //     attempt, so the whole batch threw and every candidate was rejected
  //     (`resolveAndSettle`'s catch returns null; `tryPairWithWaiter` moved on
  //     to the next candidate, found none, and every joiner queued instead of
  //     pairing). Unconditional like the outbox rows below: if statement 4
  //     didn't land, the whole batch already rolled back and this never runs.
  stmts.push(db.prepare("UPDATE war_queue SET match_id = ? WHERE entry_id = ? AND state = 'matched'").bind(matchId, a.sideB.entry));
  if (a.aEntryExisting) {
    stmts.push(db.prepare("UPDATE war_queue SET match_id = ? WHERE entry_id = ? AND state = 'matched'").bind(matchId, a.aEntryExisting));
  }
  // 5. THE ANALYTICS ROWS, unconditional because statement 4 above is a NULL-PK
  //    force-rollback validator: a match that was claimed away takes these with
  //    it (ADR-0006 §2 — the event exists iff its fact does; the `catch` below
  //    is the caller's retry, not a swallowed write).
  //
  //    `war_joined` scopes the WAR EVENT GROUP (`warEventId`: edition week +
  //    bracket, I3-1241) — the same id the waiting side's join carried;
  //    `war_completed` scopes the MATCH and is emitted once per side, so a
  //    creator whose champion fought sees the war on their own dashboard
  //    whether their fan queued or joined. It carries no fan at all: which of
  //    the two humans led with your card is not a fact a completion discloses.
  // I3-0946: on the ordinary / ghost paths this entry is born HERE, so this
  // is its only `war_joined`. On the CW-01 continuation (`a.bEntryExisting`)
  // the entry was already inserted (and its own `war_joined` emitted,
  // unconditionally, at that INSERT in joinQueue) before this pass claimed
  // it -- emitting a second one here double-counts one join in `war_joins`
  // against one row in `participants` (analytics-cells.ts's relation).
  //    U-1 (INV-051): each emit is ALSO bound to the fact by `extraWhere` — the
  //    war_matches row statement 4 just wrote — because the recon claim above is
  //    a single-use `resolved_at IS NULL` flip; the force-rollback validator
  //    already makes this true, the extraWhere states it at the emit.
  const matchLanded = { sql: MATCH_LANDED_SQL, params: [matchId] };
  if (!a.bEntryExisting && !a.ghost && !a.recon) {
    stmts.push(...outboxStatements(db, { kind: "war_joined", passportId: joinerChampion, occurredDay: occurredDay(a.ts), fanKey: joinKey!, scopeId: warEventId(a.ts, a.bracket), dims: { mode: a.practice ? "practice" : "ranked" } }, { accountId: a.bAccount, extraWhere: { sql: `${MATCH_LANDED_SQL} AND ${JOINER_QUALIFIED_SQL}`, params: [matchId, a.bAccount] } })); // D-1080(2)
  }
  for (const psp of completedFor) {
    stmts.push(...outboxStatements(db, { kind: "war_completed", passportId: psp, occurredDay: occurredDay(a.ts), scopeId: matchId, dims: { mode: a.practice ? "practice" : "ranked" } }, { extraWhere: matchLanded }));
  }

  let seqs: number[] = [];
  try {
    const span = outboxSpan(db);
    const spanRes = await db.batch([span.open, ...stmts, span.close]);
    seqs = outboxSpanSeqs(spanRes[0], spanRes[spanRes.length - 1]);
  } catch {
    return null; // claimed away / a guard failed — the caller tries the next candidate
  }
  if (a.pubEnv && a.pubCtx) publishAfterCommit(a.pubEnv, a.pubCtx, seqs);
  // Replay to the immutable store AFTER the authoritative commit; a lost R2
  // write self-heals on first read (getReplay regenerates from `inputs` and
  // backfills). putIfAbsent: a replay, like an edition, is written once.
  try {
    await store.putIfAbsent(replayKey(matchId), replay);
  } catch {
    /* self-heals on read */
  }
  return {
    match_id: matchId,
    bracket: a.bracket,
    practice: a.practice,
    ghost: a.ghost,
    you: "b",
    you_won: winner === "b",
    winner,
    decided_by: log.result.by,
    rounds: log.result.rounds,
    summary,
    ante_transferred: null, // QUEUE #236: permanently NULL for every match resolved after the removal
    server_seed: a.serverSeed,
    server_seed_hash: await sha256Hex(a.serverSeed),
    resolved_at: a.ts,
  };
}

// ── #241 RECON (D-0920) ─────────────────────────────────────────────────────
// A recon window is two `war_recon` rows (migration 0190), one per side, keyed
// (match_id, side). Account ids are NOT copied there: a side's account is read
// through its `war_queue` entry, the one place account_data.ts erases.

interface ReconRow {
  side: "a" | "b";
  entry_id: string;
  champion_index: number;
  doctrine: string;
  deadline: string;
  committed_at: string | null;
  snapshot: string;
  resolver: string | null;
  resolved_at: string | null;
  account_id: string;
  bracket: number;
  practice: number;
  server_seed: string | null;
  server_seed_hash: string | null;
}

/** Both sides of one recon window as [a, b], or null when either side's entry
 * row is missing (unknown id, or an entry erased with its account). */
async function loadRecon(db: DbLike, matchId: string): Promise<[ReconRow, ReconRow] | null> {
  const { results } = await db
    .prepare(
      `SELECT r.side, r.entry_id, r.champion_index, r.doctrine, r.deadline, r.committed_at, r.snapshot, r.resolver, r.resolved_at,
              q.account_id, q.bracket, q.practice, q.server_seed, q.server_seed_hash
         FROM war_recon r JOIN war_queue q ON q.entry_id = r.entry_id
        WHERE r.match_id = ?`,
    )
    .bind(matchId)
    .all<ReconRow>();
  const a = results.find((r) => r.side === "a");
  const b = results.find((r) => r.side === "b");
  return results.length === 2 && a && b ? [a, b] : null;
}

function parseSnapshot(raw: string): SideSnapshot | null {
  try {
    const v = JSON.parse(raw) as SideSnapshot;
    return v && typeof v.entry === "string" && Array.isArray(v.roster) && v.roster.length >= 2 ? v : null;
  } catch {
    return null;
  }
}

/** The lead a side fights with: its war_recon row's champion (the standing one
 * until committed) and the single doctrine; null when the row cannot name one. */
function reconLead(row: ReconRow, s: SideSnapshot): SideSnapshot | null {
  if (!Number.isInteger(row.champion_index) || row.champion_index < 0 || row.champion_index >= s.roster.length) return null;
  if (row.doctrine !== "guardian") return null;
  return { ...s, champion_index: row.champion_index, doctrine: "guardian" };
}

/** One account's view of a recon window, or null when it plays neither side. */
async function reconViewFor(db: DbLike, matchId: string, accountId: string, nowMs: number): Promise<ReconView | null> {
  const rows = await loadRecon(db, matchId);
  if (!rows) return null;
  const [ra, rb] = rows;
  const mine = ra.account_id === accountId ? ra : rb.account_id === accountId ? rb : null;
  if (!mine) return null;
  const theirs = mine === ra ? rb : ra;
  const own = parseSnapshot(mine.snapshot);
  const opp = parseSnapshot(theirs.snapshot);
  if (!own || !opp || ra.server_seed === null || ra.server_seed_hash === null) return null;
  const shown = await reconRevealIndices(ra.server_seed, theirs.side, opp.roster.length);
  const oppFlags = cardFlags(opp, theirs.account_id, await currentWarFlags(db, nowMs));
  const revealed = new Set(shown);
  return {
    match_id: matchId,
    bracket: mine.bracket,
    practice: mine.practice === 1,
    you: mine.side,
    deadline: mine.deadline,
    you_committed: mine.committed_at !== null,
    opponent_committed: theirs.committed_at !== null,
    champion_index: mine.champion_index,
    doctrine: "guardian",
    your_roster: own.roster.map((c) => c.passport_id),
    opponent_size: opp.roster.length,
    opponent_reveal: shown.map((i) => ({ index: i, passport_id: opp.roster[i]!.passport_id, stats: [...opp.roster[i]!.stats] })),
    your_bonds: own.roster.map((c) => matchBond(c.bond)),
    opponent_bonds: opp.roster.map((c) => matchBond(c.bond)),
    opponent_flags: oppFlags.map((f, i) => (revealed.has(i) ? f : null)),
    server_seed_hash: ra.server_seed_hash,
  };
}

/**
 * #241: the PAIRING half of a recon match. ONE batch — the joiner's row, the
 * waiting entry's claim, the legacy-escrow release, and BOTH war_recon rows,
 * the first behind the same NULL-PK force-rollback validator `resolveAndSettle`
 * uses (the claim tokens + `settleGuards`), so a lost race or a stale roster
 * opens nothing and the caller tries the next candidate. Both entries stay
 * `matched` with match_id NULL until `resolveRecon` writes the war_matches row.
 * The joiner's `war_joined` is emitted HERE, once; `war_completed` waits for
 * the result. No engine run and no seed reveal happen in this request.
 */
async function openRecon(db: DbLike, m: WarMinters, a: SettleArgs): Promise<ReconView | null> {
  if (!a.aEntryExisting) return null;
  const matchId = m.matchId();
  if (!WM_RE.test(matchId)) refuse("bad match id shape", 500);
  const claimToken = matchId;
  const joinerChampion = championPassport(a.sideB)!;
  const joinKey = a.bEntryExisting ? null : await fanKey(a.fanSecret, a.bAccount, joinerChampion);
  const g = await settleGuards(db, a);
  if (!g) return null;
  const deadline = new Date(a.nowMs + WAR_RECON_WINDOW_MS).toISOString();
  const bCheck = a.bEntryExisting
    ? "(SELECT COUNT(*) FROM war_queue WHERE entry_id = ? AND state = 'matched' AND updated_at = ?) = 1"
    : "(SELECT COUNT(*) FROM war_queue WHERE entry_id = ? AND state = 'matched') = 1";
  const bBinds = a.bEntryExisting ? [a.sideB.entry, claimToken] : [a.sideB.entry];
  const stmts: DbStatement[] = [
    joinerRowStatement(db, a, claimToken),
    claimWaitingStatement(db, a.aEntryExisting, claimToken),
    releaseEscrowStatement(db, a.aEntryExisting),
    db
      .prepare(
        `INSERT INTO war_recon (match_id, side, entry_id, champion_index, doctrine, deadline, committed_at, snapshot, resolver, resolved_at, created_at)
         SELECT CASE WHEN (SELECT COUNT(*) FROM war_queue WHERE entry_id = ? AND state = 'matched' AND updated_at = ?) = 1
                      AND ${bCheck}${g.sql}
                THEN ? ELSE NULL END, 'a', ?, ?, 'guardian', ?, NULL, ?, NULL, NULL, ?`,
      )
      .bind(a.aEntryExisting, claimToken, ...bBinds, ...g.binds, matchId, a.sideA.entry, a.sideA.champion_index, deadline, stableStringify(a.sideA), a.ts),
    db
      .prepare(
        `INSERT INTO war_recon (match_id, side, entry_id, champion_index, doctrine, deadline, committed_at, snapshot, resolver, resolved_at, created_at)
         VALUES (?, 'b', ?, ?, 'guardian', ?, NULL, ?, NULL, NULL, ?)`,
      )
      .bind(matchId, a.sideB.entry, a.sideB.champion_index, deadline, stableStringify(a.sideB), a.ts),
  ];
  if (!a.bEntryExisting) {
    stmts.push(...outboxStatements(db, { kind: "war_joined", passportId: joinerChampion, occurredDay: occurredDay(a.ts), fanKey: joinKey!, scopeId: warEventId(a.ts, a.bracket), dims: { mode: a.practice ? "practice" : "ranked" } }, { accountId: a.bAccount, extraWhere: { sql: `EXISTS (SELECT 1 FROM war_recon WHERE match_id = ? AND side = 'a') AND ${JOINER_QUALIFIED_SQL}`, params: [matchId, a.bAccount] } })); // D-1080(2)
  }
  let seqs: number[] = [];
  try {
    const span = outboxSpan(db);
    const res = await db.batch([span.open, ...stmts, span.close]);
    seqs = outboxSpanSeqs(res[0], res[res.length - 1]);
  } catch {
    return null; // claimed away / a guard failed — the caller tries the next candidate
  }
  if (a.pubEnv && a.pubCtx) publishAfterCommit(a.pubEnv, a.pubCtx, seqs);
  return reconViewFor(db, matchId, a.bAccount, a.nowMs);
}

/** A recon resolution mints nothing: its match id was minted by `openRecon`. */
const noMint = (): string => {
  throw new Error("recon resolution mints no id");
};
const RECON_NO_MINT: WarMinters = { entryId: noMint, matchId: noMint, logId: noMint, serverSeed: noMint };

/** One resolution attempt's claim token (stored in war_recon.resolver, compared by equality only). */
function reconToken(): string {
  return mintId("rr");
}

export type ReconOutcome = { state: "resolved" | "pending" | "voided" | "absent" };

/**
 * #241: resolve a recon window — ONLY when both sides committed or the
 * deadline passed (otherwise `pending`, and nothing is written). A side that
 * never committed fights with its standing Champion and the single doctrine:
 * a timeout NEVER voids (D-0920). The engine then runs on the committed
 * leads under `resolveAndSettle`'s recon mode (the same settle guards).
 * `voided` happens only when the settle guards refuse (a roster card, a
 * creator's standing or consent, or an account's standing changed under the
 * window — what voids a stale waiting entry), or a side's entry row is gone
 * (account erasure deletes war_queue rows).
 */
export async function resolveRecon(
  db: DbLike,
  store: ArtifactStore,
  matchId: string,
  now: () => string = () => new Date().toISOString(),
  pubEnv?: AnalyticsTickEnv,
  pubCtx?: { waitUntil(p: Promise<unknown>): void },
): Promise<ReconOutcome> {
  const matchRow = () => db.prepare("SELECT 1 AS x FROM war_matches WHERE match_id = ?").bind(matchId).first();
  const rows = await loadRecon(db, matchId);
  if (!rows) {
    const ts = now();
    const closed = await db.prepare("UPDATE war_recon SET resolver = 'orphan', resolved_at = ? WHERE match_id = ? AND resolved_at IS NULL").bind(ts, matchId).run();
    if (closed.changes === 0) return (await matchRow()) ? { state: "resolved" } : { state: "absent" };
    await db
      .prepare("UPDATE war_queue SET state = 'voided', updated_at = ? WHERE entry_id IN (SELECT entry_id FROM war_recon WHERE match_id = ?) AND state = 'matched' AND match_id IS NULL")
      .bind(ts, matchId)
      .run();
    return { state: "voided" };
  }
  const [ra, rb] = rows;
  if (ra.resolved_at !== null || rb.resolved_at !== null) return (await matchRow()) ? { state: "resolved" } : { state: "voided" };
  const ts = now();
  const nowMs = Date.parse(ts);
  // THE BLOCK: no resolution while a side may still commit.
  if ((ra.committed_at === null || rb.committed_at === null) && nowMs < Date.parse(ra.deadline)) return { state: "pending" };
  const snapA = parseSnapshot(ra.snapshot);
  const snapB = parseSnapshot(rb.snapshot);
  const sideA = snapA ? reconLead(ra, snapA) : null;
  const sideB = snapB ? reconLead(rb, snapB) : null;
  const seed = ra.server_seed;
  const seedOk = seed !== null && ra.server_seed_hash !== null && (await sha256Hex(seed)) === ra.server_seed_hash;
  if (sideA && sideB && seed !== null && seedOk) {
    const settled = await resolveAndSettle(db, store, RECON_NO_MINT, {
      ts,
      nowMs,
      bracket: ra.bracket,
      practice: ra.practice === 1,
      ghost: false,
      serverSeed: seed,
      sideA,
      sideB,
      aAccount: ra.account_id,
      bAccount: rb.account_id,
      aEntryExisting: ra.entry_id,
      joinerInput: { roster: sideB.roster.map((c) => c.instance_id), champion_index: sideB.champion_index, client_seed: sideB.client_seed, practice: ra.practice === 1 },
      fanSecret: undefined,
      pubEnv,
      pubCtx,
      recon: { matchId, token: reconToken() },
    });
    if (settled) return { state: "resolved" };
  }
  if (await matchRow()) return { state: "resolved" }; // a racing resolver won
  await db.batch([
    db.prepare("UPDATE war_recon SET resolver = 'void', resolved_at = ? WHERE match_id = ? AND resolved_at IS NULL").bind(ts, matchId),
    db
      .prepare(
        "UPDATE war_queue SET state = 'voided', updated_at = ? WHERE entry_id IN (?, ?) AND state = 'matched' AND match_id IS NULL AND EXISTS (SELECT 1 FROM war_recon WHERE match_id = ? AND resolver = 'void')",
      )
      .bind(ts, ra.entry_id, rb.entry_id, matchId),
  ]);
  return (await matchRow()) ? { state: "resolved" } : { state: "voided" };
}

/**
 * #241: commit the caller's lead for one recon window — ONCE, before the
 * deadline. `champion_index` indexes the caller's own roster; `doctrine`, when
 * sent, must be the single doctrine. Then tries to resolve (it resolves only
 * if the other side has committed too). Answers the match once resolved, the
 * caller's recon view while the window is still open.
 */
export async function commitRecon(
  db: DbLike,
  store: ArtifactStore,
  accountId: string,
  matchId: string,
  body: { champion_index?: unknown; doctrine?: unknown },
  now: () => string = () => new Date().toISOString(),
  pubEnv?: AnalyticsTickEnv,
  pubCtx?: { waitUntil(p: Promise<unknown>): void },
): Promise<{ match: MatchView } | { recon: ReconView }> {
  if (!WM_RE.test(matchId)) refuse("not found", 404);
  const rows = await loadRecon(db, matchId);
  const mine = rows ? rows.find((r) => r.account_id === accountId) : undefined;
  if (!rows || !mine) refuse("not found", 404);
  if (mine.resolved_at !== null) {
    const done = await getMatch(db, accountId, matchId);
    if (done) return { match: done };
    refuse("this match was voided", 409);
  }
  const own = parseSnapshot(mine.snapshot);
  const ci = body.champion_index;
  if (!own || typeof ci !== "number" || !Number.isInteger(ci) || ci < 0 || ci >= own.roster.length) refuse("champion index out of range");
  if (body.doctrine !== undefined && body.doctrine !== "guardian") refuse("unknown doctrine");
  const ts = now();
  if (mine.committed_at !== null) refuse("you already committed your lead", 409);
  if (Date.parse(ts) >= Date.parse(mine.deadline)) refuse("the recon window has closed", 409);
  const res = await db
    .prepare("UPDATE war_recon SET champion_index = ?, doctrine = 'guardian', committed_at = ? WHERE match_id = ? AND side = ? AND committed_at IS NULL AND resolved_at IS NULL")
    .bind(ci, ts, matchId, mine.side)
    .run();
  if (res.changes !== 1) refuse("the recon window has closed", 409);
  const out = await resolveRecon(db, store, matchId, now, pubEnv, pubCtx);
  if (out.state === "resolved") {
    const done = await getMatch(db, accountId, matchId);
    if (done) return { match: done };
  }
  if (out.state === "voided") refuse("this match was voided", 409);
  const view = await reconViewFor(db, matchId, accountId, Date.parse(now()));
  if (!view) refuse("not found", 404);
  return { recon: view };
}

/** #241: the caller's OPEN recon windows (own-scoped), soonest deadline first. */
export async function reconStatus(db: DbLike, accountId: string, nowMs: number = Date.now()): Promise<ReconView[]> {
  const { results } = await db
    .prepare(
      `SELECT r.match_id FROM war_recon r JOIN war_queue q ON q.entry_id = r.entry_id
        WHERE q.account_id = ? AND r.resolved_at IS NULL ORDER BY r.deadline LIMIT 10`,
    )
    .bind(accountId)
    .all<{ match_id: string }>();
  const views = await Promise.all(results.map((r) => reconViewFor(db, r.match_id, accountId, nowMs)));
  return views.filter((v): v is ReconView => v !== null);
}

/** Windows `sweepWarRecon` resolves per tick (oldest deadline first). D-0931(7)/D-0933(1): 30. */
export const WAR_RECON_SWEEP_BATCH = 30;

/** D-0939(6): a RESOLVED war_recon row is deleted 90 days after `resolved_at`; an unresolved row never is. */
export const WAR_RECON_RETENTION_DAYS = 90;
/** Matches (both sides together) `sweepWarRecon` prunes per tick, oldest `resolved_at` first; the rest is next tick's (500 = ORPHAN_SWEEP_LIMIT's size; the leg runs on the six-hourly sweep). */
export const WAR_RECON_PRUNE_BATCH = 500;

/**
 * #241 cron leg: resolve every recon window whose deadline has passed and that
 * nobody came back to — a silent pair still gets its match (standing
 * Champions), never a void. Bounded per tick; the remainder is next tick's.
 */
export async function sweepWarRecon(
  db: DbLike,
  store: ArtifactStore,
  now: () => string = () => new Date().toISOString(),
  pubEnv?: AnalyticsTickEnv,
  pubCtx?: { waitUntil(p: Promise<unknown>): void },
): Promise<{ resolved: number; voided: number; pending: number }> {
  const { results } = await db
    .prepare(`SELECT match_id FROM war_recon WHERE resolved_at IS NULL AND deadline <= ? GROUP BY match_id ORDER BY MIN(deadline) LIMIT ${WAR_RECON_SWEEP_BATCH}`)
    .bind(now())
    .all<{ match_id: string }>();
  const out = { resolved: 0, voided: 0, pending: 0 };
  for (const r of results) {
    const o = await resolveRecon(db, store, r.match_id, now, pubEnv, pubCtx);
    if (o.state === "resolved") out.resolved += 1;
    else if (o.state === "voided") out.voided += 1;
    else if (o.state === "pending") out.pending += 1;
  }
  // D-0939(6) retention, on this cron leg: the row predicate is on the DELETE itself, so an unresolved side is never deleted.
  const cutoff = new Date(Date.parse(now()) - WAR_RECON_RETENTION_DAYS * 86_400_000).toISOString();
  await db
    .prepare(
      `DELETE FROM war_recon WHERE resolved_at IS NOT NULL AND resolved_at < ? AND match_id IN
         (SELECT match_id FROM war_recon WHERE resolved_at IS NOT NULL AND resolved_at < ? GROUP BY match_id ORDER BY MIN(resolved_at) LIMIT ${WAR_RECON_PRUNE_BATCH})`,
    )
    .bind(cutoff, cutoff)
    .run();
  return out;
}

/** Practice ghost: replay side A from a recent match's stored snapshot. */
async function tryGhost(
  db: DbLike,
  store: ArtifactStore,
  m: WarMinters,
  accountId: string,
  bracket: number,
  roster: { instance_id: string; passport_id: string }[],
  input: QueueJoinInput,
  statsCache: Map<string, PowerRatings | null>,
  now: () => string,
  /** joinQueue's `doorAt`: the ghost pick asks each source creator's opt-in gate at the join's instant (I34-FX3-INSTANT). */
  doorAt: string,
  fanSecret?: string,
  pubEnv?: AnalyticsTickEnv,
  pubCtx?: { waitUntil(p: Promise<unknown>): void },
): Promise<MatchView | null> {
  // ghost = 0 (C-41): a ghost source must be a REAL past match. Sourcing from a
  // prior ghost replay would nest the label ("ghost:ghost:...") without bound
  // every time a bracket has exactly one lone practicer.
  const { results } = await db
    .prepare(`SELECT inputs, a_entry FROM war_matches WHERE bracket = ? AND ghost = 0 ORDER BY resolved_at DESC, match_id DESC LIMIT ${GHOST_CANDIDATES}`)
    .bind(bracket)
    .all<{ inputs: string; a_entry: string }>();
  for (const row of results) {
    let src: { sides?: unknown };
    try {
      src = JSON.parse(row.inputs) as { sides?: unknown };
    } catch {
      continue;
    }
    const side = Array.isArray(src.sides) ? (src.sides[0] as SideSnapshot | undefined) : undefined;
    if (!side || !Array.isArray(side.roster) || side.roster.length !== bracket) continue;
    // The freeze follows the CREATOR, not the snapshot: a ghost of an
    // opted-out creator's cards is still a new match depicting them.
    if (!(await allWarOptedIn(db, side.roster.map((c) => c.passport_id), doorAt))) continue;
    const ts = now();
    const nowMs = Date.parse(ts);
    const joinerEntry = m.entryId();
    if (!WQ_RE.test(joinerEntry)) refuse("bad entry id shape", 500);
    const sideB = await buildSideSnapshot(db, store, accountId, joinerEntry, roster, input.champion_index, input.client_seed, nowMs, statsCache);
    if (!sideB) refuse(ROSTER_REFUSAL, 409);
    const ghostSide: SideSnapshot = {
      entry: `ghost:${side.entry}`,
      roster: side.roster,
      champion_index: side.champion_index,
      client_seed: side.client_seed,
      doctrine: "guardian",
    };
    // I3-0992: minted AFTER the joiner's side and client seed are fixed, and
    // never committed — a ghost is practice-only and NOT provably fair.
    const serverSeed = m.serverSeed();
    if (!/^[0-9a-f]{8,128}$/.test(serverSeed)) refuse("bad server seed shape", 500);
    const settled = await resolveAndSettle(db, store, m, {
      ts,
      nowMs,
      bracket,
      practice: true,
      ghost: true,
      serverSeed,
      sideA: ghostSide,
      sideB,
      aAccount: null,
      bAccount: accountId,
      joinerInput: input,
      fanSecret,
      pubEnv,
      pubCtx,
    });
    if (settled) return settled;
  }
  return null;
}

/** Cancel the caller's waiting entry and free any pre-#236 stake — atomically, with
 * the release predicated on the cancel having landed (a racing settle wins). */
export async function leaveQueue(db: DbLike, accountId: string, now: () => string = () => new Date().toISOString()): Promise<boolean> {
  const row = await db
    .prepare("SELECT entry_id FROM war_queue WHERE account_id = ? AND state = 'waiting'")
    .bind(accountId)
    .first<{ entry_id: string }>();
  if (!row) return false;
  await db.batch([
    db.prepare("UPDATE war_queue SET state = 'cancelled', updated_at = ? WHERE entry_id = ? AND account_id = ? AND state = 'waiting'").bind(now(), row.entry_id, accountId),
    db
      .prepare(
        "UPDATE card_instances SET escrowed_by = NULL WHERE escrowed_by = ? AND EXISTS (SELECT 1 FROM war_queue WHERE entry_id = ? AND state = 'cancelled')",
      )
      .bind(row.entry_id, row.entry_id),
  ]);
  const after = await db.prepare("SELECT state FROM war_queue WHERE entry_id = ?").bind(row.entry_id).first<{ state: string }>();
  return after?.state === "cancelled";
}

/** The caller's waiting entry, if any (own-scoped — never anyone else's). */
export async function queueStatus(db: DbLike, accountId: string): Promise<{ entry_id: string; bracket: number; practice: boolean; server_seed_hash: string | null; created_at: string } | null> {
  const row = await db
    .prepare("SELECT entry_id, bracket, practice, server_seed_hash, created_at FROM war_queue WHERE account_id = ? AND state = 'waiting'")
    .bind(accountId)
    .first<{ entry_id: string; bracket: number; practice: number; server_seed_hash: string | null; created_at: string }>();
  if (!row) return null;
  return { entry_id: row.entry_id, bracket: row.bracket, practice: row.practice === 1, server_seed_hash: row.server_seed_hash, created_at: row.created_at };
}

/** RJb-T2-AS-017 (D-0920): how far back the war screen's "recent result" looks. */
export const WAR_RECENT_RESULT_MS = 7 * 24 * 60 * 60 * 1000;

/** RJb-T2-AS-017: the caller's most recent RESOLVED entry (`matched` or `voided`)
 * joined in the last WAR_RECENT_RESULT_MS, so a fan matched (or expired) while
 * away is told so instead of "nothing queued". Own-scoped, like queueStatus. */
export async function recentWarResult(
  db: DbLike,
  accountId: string,
  nowMs: number,
): Promise<{ entry_id: string; state: "matched" | "voided" | "pending"; match_id: string | null; bracket: number; practice: boolean; created_at: string } | null> {
  const row = await db
    .prepare(
      "SELECT entry_id, state, match_id, bracket, practice, created_at FROM war_queue WHERE account_id = ? AND state IN ('matched', 'voided') AND created_at >= ? ORDER BY created_at DESC, entry_id DESC LIMIT 1",
    )
    .bind(accountId, new Date(nowMs - WAR_RECENT_RESULT_MS).toISOString())
    .first<{ entry_id: string; state: "matched" | "voided"; match_id: string | null; bracket: number; practice: number; created_at: string }>();
  if (!row) return null;
  // RG-WAROPS:14 (I16-FXC): `matched` with no match row yet is a recon window still open (or awaiting its sweep), not a result.
  const state = row.state === "matched" && row.match_id === null ? "pending" : row.state;
  return { entry_id: row.entry_id, state, match_id: row.match_id, bracket: row.bracket, practice: row.practice === 1, created_at: row.created_at };
}

/** RJb-T-AS-016 / I3-0320 (D-0920): a waiting entry older than this is voided by `sweepStaleWarQueue`. */
export const WAR_QUEUE_STALE_MS = 24 * 60 * 60 * 1000;
/** Entries `sweepStaleWarQueue` voids per scheduled run; the remainder is next run's. */
export const WAR_QUEUE_STALE_BATCH = 50;

/** RJb-T-AS-016 / I3-0320 cron leg: void every `waiting` war_queue entry (ranked or
 * practice) created more than WAR_QUEUE_STALE_MS before `nowMs`, oldest first, at most
 * WAR_QUEUE_STALE_BATCH per run, through `voidEntry` — the void is guarded on
 * `state = 'waiting'` (a racing pairing wins) and any legacy escrowed ante is
 * released only once the void landed. */
export async function sweepStaleWarQueue(db: DbLike, nowMs: number, now: () => string = () => new Date(nowMs).toISOString()): Promise<{ voided: number }> {
  const { results } = await db
    .prepare(`SELECT entry_id FROM war_queue WHERE state = 'waiting' AND created_at < ? ORDER BY created_at, entry_id LIMIT ${WAR_QUEUE_STALE_BATCH}`)
    .bind(new Date(nowMs - WAR_QUEUE_STALE_MS).toISOString())
    .all<{ entry_id: string }>();
  let voided = 0;
  for (const r of results) {
    await voidEntry(db, r.entry_id, now);
    const after = await db.prepare("SELECT state FROM war_queue WHERE entry_id = ?").bind(r.entry_id).first<{ state: string }>();
    if (after?.state === "voided") voided += 1;
  }
  return { voided };
}

/** I3-0273 (D-0920 option A; D-0532 random matching stands): the operator read over
 * war_matches — unordered account pairs that met at least `minMatches` times since
 * `sinceMs`, and per-account win rate for accounts with at least `minMatches` matches.
 * Ghost matches (no live side A) are excluded. Read-only; at most 100 rows each, and a
 * `*_truncated` flag says when a list was cut there. */
export const WAR_PAIR_REPORT_LIMIT = 100;
export async function warPairReport(
  db: DbLike,
  opts: { minMatches: number; sinceMs: number },
): Promise<{
  pairs: { x: string; y: string; matches: number; ranked: number; x_wins: number; y_wins: number }[];
  accounts: { account_id: string; played: number; wins: number; win_rate: number }[];
  /** True when a list hit WAR_PAIR_REPORT_LIMIT: more rows qualify than are shown. */
  pairs_truncated: boolean;
  accounts_truncated: boolean;
}> {
  const { results: pairs } = await db
    .prepare(
      `SELECT MIN(a_account, b_account) AS x, MAX(a_account, b_account) AS y, COUNT(*) AS matches,
              SUM(CASE WHEN practice = 0 THEN 1 ELSE 0 END) AS ranked,
              SUM(CASE WHEN (winner = 'a' AND a_account < b_account) OR (winner = 'b' AND b_account < a_account) THEN 1 ELSE 0 END) AS x_wins
         FROM war_matches
        WHERE ghost = 0 AND a_account IS NOT NULL AND resolved_ms >= ?
        GROUP BY MIN(a_account, b_account), MAX(a_account, b_account)
       HAVING COUNT(*) >= ?
        ORDER BY matches DESC, x, y LIMIT ${WAR_PAIR_REPORT_LIMIT}`,
    )
    .bind(opts.sinceMs, opts.minMatches)
    .all<{ x: string; y: string; matches: number; ranked: number; x_wins: number }>();
  const { results: accounts } = await db
    .prepare(
      `SELECT account_id, COUNT(*) AS played, SUM(won) AS wins FROM (
          SELECT a_account AS account_id, CASE WHEN winner = 'a' THEN 1 ELSE 0 END AS won FROM war_matches WHERE ghost = 0 AND a_account IS NOT NULL AND resolved_ms >= ?
          UNION ALL
          SELECT b_account AS account_id, CASE WHEN winner = 'b' THEN 1 ELSE 0 END AS won FROM war_matches WHERE ghost = 0 AND a_account IS NOT NULL AND resolved_ms >= ?
        ) GROUP BY account_id HAVING COUNT(*) >= ? ORDER BY CAST(SUM(won) AS REAL) / COUNT(*) DESC, played DESC, account_id LIMIT ${WAR_PAIR_REPORT_LIMIT}`,
    )
    .bind(opts.sinceMs, opts.sinceMs, opts.minMatches)
    .all<{ account_id: string; played: number; wins: number }>();
  return {
    pairs: pairs.map((p) => ({ x: p.x, y: p.y, matches: p.matches, ranked: p.ranked, x_wins: p.x_wins, y_wins: p.matches - p.x_wins })),
    accounts: accounts.map((a) => ({ account_id: a.account_id, played: a.played, wins: a.wins, win_rate: a.played > 0 ? a.wins / a.played : 0 })),
    pairs_truncated: pairs.length >= WAR_PAIR_REPORT_LIMIT,
    accounts_truncated: accounts.length >= WAR_PAIR_REPORT_LIMIT,
  };
}

interface MatchRow {
  match_id: string;
  bracket: number;
  practice: number;
  ghost: number;
  a_entry: string;
  b_entry: string | null;
  a_account: string | null;
  b_account: string;
  winner: string;
  decided_by: string;
  rounds: number;
  inputs: string;
  summary: string;
  replay_hash: string;
  ante_transferred: string | null;
  resolved_at: string;
}

async function toView(r: MatchRow, accountId: string): Promise<MatchView> {
  const you: "a" | "b" = r.a_account === accountId ? "a" : "b";
  let serverSeed: string | null = null;
  try {
    serverSeed = (JSON.parse(r.inputs) as { server_seed?: string }).server_seed ?? null;
  } catch {
    /* summary view survives a malformed row */
  }
  // I3-0991: a resolved match's server seed is already revealed (`serverSeed`
  // above, straight out of `inputs`), so the commitment hash is a pure
  // function of a value this view already carries -- there is no reason for
  // the joiner-facing field to be a permanent null. Computed from the SAME
  // seed the view exposes, so the two can never disagree.
  const serverSeedHash = serverSeed !== null ? await sha256Hex(serverSeed) : null;
  return {
    match_id: r.match_id,
    bracket: r.bracket,
    practice: r.practice === 1,
    ghost: r.ghost === 1,
    you,
    you_won: r.winner === you,
    winner: r.winner as "a" | "b",
    decided_by: r.decided_by,
    rounds: r.rounds,
    summary: ((): unknown => {
      try {
        return JSON.parse(r.summary);
      } catch {
        return null;
      }
    })(),
    ante_transferred: r.ante_transferred,
    server_seed: serverSeed,
    server_seed_hash: serverSeedHash,
    resolved_at: r.resolved_at,
  };
}

/** The caller's matches, newest first (either side), bounded. */
export async function listMatches(db: DbLike, accountId: string): Promise<MatchView[]> {
  const { results } = await db
    .prepare(
      `SELECT match_id, bracket, practice, ghost, a_entry, b_entry, a_account, b_account, winner, decided_by, rounds, inputs, summary, replay_hash, ante_transferred, resolved_at
         FROM war_matches WHERE a_account = ? OR b_account = ?
        ORDER BY resolved_at DESC, match_id DESC LIMIT ${MATCHES_LISTED}`,
    )
    .bind(accountId, accountId)
    .all<MatchRow>();
  return Promise.all(results.map((r) => toView(r, accountId)));
}

/** One of the caller's matches (either side), or null — a match you weren't
 * in answers exactly like one that doesn't exist. */
export async function getMatch(db: DbLike, accountId: string, matchId: string): Promise<MatchView | null> {
  if (!WM_RE.test(matchId)) return null;
  const row = await db
    .prepare(
      `SELECT match_id, bracket, practice, ghost, a_entry, b_entry, a_account, b_account, winner, decided_by, rounds, inputs, summary, replay_hash, ante_transferred, resolved_at
         FROM war_matches WHERE match_id = ? AND (a_account = ? OR b_account = ?)`,
    )
    .bind(matchId, accountId, accountId)
    .first<MatchRow>();
  return row ? await toView(row, accountId) : null;
}

/**
 * The full provably-fair replay for one of the caller's matches. Serves the
 * R2 copy when its hash checks out; a missing or corrupt copy is REGENERATED
 * from the authoritative `inputs` (the match is a pure function of them),
 * re-verified against the pinned hash, and backfilled. Returns the raw
 * canonical JSON string.
 */
export async function getReplay(db: DbLike, store: ArtifactStore, accountId: string, matchId: string): Promise<string | null> {
  if (!WM_RE.test(matchId)) return null;
  const row = await db
    .prepare("SELECT inputs, replay_hash FROM war_matches WHERE match_id = ? AND (a_account = ? OR b_account = ?)")
    .bind(matchId, accountId, accountId)
    .first<{ inputs: string; replay_hash: string }>();
  if (!row) return null;
  // CW-03 (1) — A THROWING OBJECT READ IS A REASON TO RECONSTRUCT, NOT TO FAIL.
  // This `get` sat outside every `try` below it, so an R2 read (or body-read)
  // exception exited the function and the route answered an error — while the
  // authoritative D1 record, which holds the complete pinned inputs and is the
  // whole point of storing them, was perfectly healthy. The stored object is a
  // CACHE of something this function can always rebuild; a cache that cannot be
  // read is a cache miss. `null` here means exactly that, and falls into the
  // same reconstruction path a missing object already used.
  let stored: string | null;
  try {
    stored = await store.get(replayKey(matchId));
  } catch {
    stored = null;
  }
  if (stored !== null && (await sha256Hex(stored)) === row.replay_hash) return stored;
  // Reaching here with `stored !== null` means the bytes under the key exist
  // and FAILED the pin — a known-corrupt object, not a missing one.
  const corrupt = stored !== null;
  // Regenerate from inputs (deterministic), verify, backfill. The engine call
  // is wrapped (review F2): a malformed `inputs` row must yield a clean null
  // (→ 404), never a thrown 500 — and the dials come FROM the pinned inputs
  // (review F3). A legacy row that predates the pin is filled from
  // INITIAL_DIALS (D-0815(2)), never today's DEFAULT_DIALS: a re-default must
  // not un-verify the back-catalogue.
  let inputs: MatchInputs;
  try {
    const parsed = JSON.parse(row.inputs) as Partial<MatchInputs>;
    if (typeof parsed.server_seed !== "string" || !Array.isArray(parsed.sides) || parsed.sides.length !== 2) return null;
    inputs = { server_seed: parsed.server_seed, dials: parsed.dials ?? INITIAL_DIALS, sides: parsed.sides as [SideSnapshot, SideSnapshot] };
  } catch {
    return null;
  }
  let replay: string;
  try {
    // I26-INT (REGISTRY hand-off, D-1014(6)): re-resolve on the PINNED dials — a key the row lacks runs at its
    // INITIAL value, the stamp is the pinned set exactly — so the stored dials and the replay hash are unchanged.
    const log = await resolveMatchAsLogged({ serverSeed: inputs.server_seed, sides: [toEngineSide(inputs.sides[0]), toEngineSide(inputs.sides[1])] }, inputs.dials as unknown as Record<string, unknown>);
    replay = replayJson(inputs, log);
  } catch {
    return null; // a malformed snapshot cannot be re-resolved — 404, not 500
  }
  if ((await sha256Hex(replay)) !== row.replay_hash) return null; // never serve what the pin refuses
  // CW-03 (2) — REPAIR A KNOWN-CORRUPT KEY; CREATE A MISSING ONE.
  // `putIfAbsent` is the right primitive for a missing object and the WRONG one
  // for an occupied-but-corrupt key: the production adapter refuses to write an
  // existing key and returns false, which was ignored, so the corrupt bytes
  // stayed and every subsequent read re-ran the whole combat simulation to
  // produce the same answer. Storage never healed. Write-once creation is
  // preserved for the ordinary case; the only bytes ever REPLACED are ones this
  // function has just proved do not match the pin, with bytes it has just
  // proved do — so a valid pinned object can never be overwritten with
  // unchecked content.
  try {
    if (corrupt) await store.put(replayKey(matchId), replay);
    else await store.putIfAbsent(replayKey(matchId), replay);
  } catch {
    /* best-effort backfill */
  }
  return replay;
}

/** Monday 00:00 UTC of `nowMs`'s ISO week (the week `isoWeek(nowMs)` names). */
function isoWeekStartMs(nowMs: number): number {
  const d = new Date(nowMs);
  const sinceMonday = (d.getUTCDay() + 6) % 7;
  return Date.UTC(d.getUTCFullYear(), d.getUTCMonth(), d.getUTCDate()) - sinceMonday * 86_400_000;
}

/** One row per (ranked match, side): the side's champion creator, its account, the other side's account (`opp`),
 * whether it won (`won`, every ranked result: losses are counted raw), and `q`, whether the win COUNTS: D-1481(1)
 * (I60-FLAG), the season ladder's rule (war-ladder.ts warLadderStanding, social.ts warRecord) — the opponent's
 * `accounts` row exists, is not the side's own account, and was created (floored to the second) at least
 * LADDER_OPPONENT_AGE_MS before the match's resolved_ms. A range on bracket (BRACKETS is ascending) lets the
 * idx_wm_bracket index (bracket, practice, resolved_at) bound the scan. Binds: from, to, from, to. */
const STANDINGS_SIDES_SQL = `SELECT json_extract(inputs, '$.sides[0].roster[' || json_extract(inputs, '$.sides[0].champion_index') || '].passport_id') AS psp,
         a_account AS account, b_account AS opp, CASE WHEN winner = 'a' THEN 1 ELSE 0 END AS won,
         CASE WHEN winner = 'a' AND EXISTS (SELECT 1 FROM accounts o WHERE o.account_id = b_account AND b_account <> a_account AND CAST(strftime('%s', o.created_at) AS INTEGER) * 1000 <= resolved_ms - ${LADDER_OPPONENT_AGE_MS}) THEN 1 ELSE 0 END AS q
    FROM war_matches
   WHERE bracket BETWEEN ${BRACKETS[0]} AND ${BRACKETS[BRACKETS.length - 1]} AND practice = 0 AND ghost = 0 AND resolved_at >= ? AND resolved_at < ? AND a_account IS NOT NULL
  UNION ALL
  SELECT json_extract(inputs, '$.sides[1].roster[' || json_extract(inputs, '$.sides[1].champion_index') || '].passport_id'),
         b_account, a_account, CASE WHEN winner = 'b' THEN 1 ELSE 0 END,
         CASE WHEN winner = 'b' AND EXISTS (SELECT 1 FROM accounts o WHERE o.account_id = a_account AND a_account <> b_account AND CAST(strftime('%s', o.created_at) AS INTEGER) * 1000 <= resolved_ms - ${LADDER_OPPONENT_AGE_MS}) THEN 1 ELSE 0 END
    FROM war_matches
   WHERE bracket BETWEEN ${BRACKETS[0]} AND ${BRACKETS[BRACKETS.length - 1]} AND practice = 0 AND ghost = 0 AND resolved_at >= ? AND resolved_at < ?`;

/** D-1481(1) / W0 am.3-4: a tally's wins over STANDINGS_SIDES_SQL rows — distinct (winning account, opponent) pairs
 * whose win counts (`q`), within the window the statement binds. Per account that is distinct opponents; per champion
 * creator a pair recurring under two champions counts once for each. */
const WIN_PAIRS_SQL = "COUNT(DISTINCT CASE WHEN q = 1 THEN account || '|' || opp END)";

/** Build this week's standings artifact and write it. Four statements and one
 * object write; nothing here reads per creator or per fan. */
export async function buildWarStandings(
  db: DbLike,
  store: ArtifactStore & Partial<Pick<VersionedObjectStore, "getVersioned" | "putIfMatch">>,
  nowMs: number,
): Promise<{ week: string; creators: number; fans: number; fan_players: number }> {
  // D-0955(4) / #946(f) (I20-W1C H5, landed by I20-L02): the etag is read BEFORE the D1 reads and the write is
  // conditional on it, so an erasure's drop (account_data.ts dropAccountFromWarStandings) landing between this build's
  // D1 reads and its write fails the condition instead of being overwritten with the erased id. A lost condition
  // THROWS (the cron leg logs it) and the next tick rebuilds from D1. The cron's store is route-helpers.ts r2Store,
  // which carries both methods; a store without them (an ad-hoc test store) keeps the plain put.
  const versioned = typeof store.getVersioned === "function" && typeof store.putIfMatch === "function";
  const prior = versioned ? await store.getVersioned!(WAR_STANDINGS_KEY) : null;
  const start = isoWeekStartMs(nowMs);
  const from = new Date(start).toISOString();
  const to = new Date(start + 7 * 86_400_000).toISOString();
  const binds = [from, to, from, to];
  const { results: creators } = await db
    .prepare(
      `WITH sides AS (${STANDINGS_SIDES_SQL})
       SELECT psp AS passport_id, ${WIN_PAIRS_SQL} AS wins, COUNT(*) - SUM(won) AS losses
         FROM sides WHERE psp IS NOT NULL GROUP BY psp
        ORDER BY wins DESC, losses ASC, psp ASC LIMIT ${WAR_STANDINGS_CANDIDATES}`,
    )
    .bind(...binds)
    .all<{ passport_id: string; wins: number; losses: number }>();
  const { results: fans } = await db
    .prepare(
      `WITH sides AS (${STANDINGS_SIDES_SQL})
       SELECT s.account AS account_id, ${WIN_PAIRS_SQL} AS wins, COUNT(*) - SUM(s.won) AS losses
         FROM sides s JOIN user_profiles u ON u.account_id = s.account
        WHERE u.war_public = 1 AND u.handle IS NOT NULL
        GROUP BY s.account
        ORDER BY wins DESC, losses ASC, s.account ASC LIMIT ${WAR_STANDINGS_CANDIDATES}`,
    )
    .bind(...binds)
    .all<{ account_id: string; wins: number; losses: number }>();
  const players = await db
    .prepare(`WITH sides AS (${STANDINGS_SIDES_SQL}) SELECT COUNT(DISTINCT account) AS n FROM sides`)
    .bind(...binds)
    .first<{ n: number }>();
  const artifact: WarStandingsArtifact = {
    week: isoWeek(nowMs),
    generated_at: new Date(nowMs).toISOString(),
    creators: creators.map((r) => ({ passport_id: r.passport_id, wins: Number(r.wins), losses: Number(r.losses) })),
    fans: fans.map((r) => ({ account_id: r.account_id, wins: Number(r.wins), losses: Number(r.losses) })),
    fan_players: Number(players?.n ?? 0),
  };
  if (versioned) {
    if ((await store.putIfMatch!(WAR_STANDINGS_KEY, stableStringify(artifact), prior?.etag ?? null)) === null) {
      throw new Error("war standings: the file changed during the build (an erasure's drop); the next tick rebuilds");
    }
  } else await store.put(WAR_STANDINGS_KEY, stableStringify(artifact));
  return { week: artifact.week, creators: artifact.creators.length, fans: artifact.fans.length, fan_players: artifact.fan_players };
}

/** Q642: one candidate for a flag — last week's ranked wins and losses. */
export interface WarFlagTally {
  id: string;
  wins: number;
  losses: number;
}

/**
 * Q642 — the weekly hand-over as ruled (docs/WAR-BALANCE-SIM.md, D-0385..D-0390): the top scorer takes the flag; a
 * sole-top incumbent keeps it without a refresh; an incumbent never wins a tie; among the tied challengers the fewest
 * losses wins; no wins at all changes nothing. A full tie has NO ruling: the lowest id takes it (I30-WAR placeholder,
 * flagged to Jason). `top` is the tied-top tier, wins > 0. Pure.
 */
export function rotateWarFlag(incumbent: string | null, top: readonly WarFlagTally[]): string | null {
  if (top.length === 0) return incumbent;
  if (incumbent !== null && top.length === 1 && top[0]!.id === incumbent) return incumbent;
  const challengers = top.filter((t) => t.id !== incumbent);
  let pick = challengers[0]!;
  for (const t of challengers) if (t.losses < pick.losses || (t.losses === pick.losses && t.id < pick.id)) pick = t;
  return pick.id;
}

/**
 * Q642 — write THIS week's flag holders (war_flags, 0253) from LAST week's ranked results: the champion creators'
 * wins for the creator flag, the accounts' wins for the fan flag (STANDINGS_SIDES_SQL: ranked, not practice, not
 * ghost; wins are WIN_PAIRS_SQL, D-1481(1)). Idempotent per (kind, week): the first sweep of a week wins (ON CONFLICT
 * DO NOTHING). Scheduled weekly by index.ts scheduled() (I30-INT hand-off).
 * D-1481(2) (I60-FLAG): only an eligible id takes or keeps a flag — a fan not banned (accounts.banned_at, refunds.ts
 * accountBanned), a creator not closed or suspended (credits.ts CREATOR_CLOSED_SQL) and opted in (war-consent.ts
 * warOptState). The tally is the first wins tier with an eligible member, filtered to those members; an ineligible
 * incumbent is passed to rotateWarFlag as null. Nobody eligible: no row, and currentWarFlags keeps serving the latest
 * written week (JASON-QUESTIONS I60 9).
 */
export async function sweepWarFlags(db: DbLike, nowMs: number, now: () => string = () => new Date(nowMs).toISOString()): Promise<{ week_start: string; creator: string | null; fan: string | null }> {
  const start = isoWeekStartMs(nowMs);
  const weekStart = new Date(start).toISOString().slice(0, 10);
  const from = new Date(start - 7 * 86_400_000).toISOString();
  const to = new Date(start).toISOString();
  const optedIn = async (pid: string): Promise<boolean> => (await warOptState(db, pid)).opted_in;
  const top = async (kind: "creator" | "fan"): Promise<WarFlagTally[]> => {
    const idExpr = kind === "creator" ? "psp" : "account";
    const standing = kind === "creator" ? `NOT ${CREATOR_CLOSED_SQL("psp")}` : "NOT EXISTS (SELECT 1 FROM accounts b WHERE b.account_id = account AND b.banned_at IS NOT NULL)";
    const { results } = await db
      .prepare(
        `WITH sides AS (${STANDINGS_SIDES_SQL}),
              t AS (SELECT ${idExpr} AS id, ${WIN_PAIRS_SQL} AS wins, COUNT(*) - SUM(won) AS losses FROM sides WHERE ${idExpr} IS NOT NULL AND ${standing} GROUP BY ${idExpr})
         SELECT id, wins, losses FROM t WHERE wins > 0 ORDER BY wins DESC, losses ASC, id ASC`,
      )
      .bind(from, to, from, to)
      .all<{ id: string; wins: number; losses: number }>();
    const all = results.map((r) => ({ id: r.id, wins: Number(r.wins), losses: Number(r.losses) }));
    for (let i = 0; i < all.length; ) {
      let j = i;
      while (j < all.length && all[j]!.wins === all[i]!.wins) j++;
      const tier: WarFlagTally[] = [];
      for (const x of all.slice(i, j)) if (kind === "fan" || (await optedIn(x.id))) tier.push(x);
      if (tier.length > 0) return tier;
      i = j;
    }
    return [];
  };
  const eligible = async (kind: "creator" | "fan", id: string): Promise<boolean> =>
    kind === "fan"
      ? !(await accountBanned(db, id))
      : (await db.prepare(`SELECT ${CREATOR_CLOSED_SQL("?")} AS c`).bind(id).first<{ c: number }>())?.c !== 1 && (await optedIn(id));
  const { results: prior } = await db
    .prepare(
      `SELECT f.kind AS kind, f.passport_id AS passport_id, f.account_id AS account_id, f.capture_day AS capture_day FROM war_flags f
        WHERE f.week_start = (SELECT MAX(g.week_start) FROM war_flags g WHERE g.kind = f.kind AND g.week_start < ?)`,
    )
    .bind(weekStart)
    .all<{ kind: string; passport_id: string | null; account_id: string | null; capture_day: string }>();
  const ts = now();
  const stmts: DbStatement[] = [];
  const held: { creator: string | null; fan: string | null } = { creator: null, fan: null };
  for (const kind of ["creator", "fan"] as const) {
    const p = prior.find((r) => r.kind === kind);
    const prev = p ? (kind === "creator" ? p.passport_id : p.account_id) : null;
    const incumbent = prev !== null && (await eligible(kind, prev)) ? prev : null;
    const holder = rotateWarFlag(incumbent, await top(kind));
    held[kind] = holder;
    if (holder === null) continue;
    const captureDay = holder === incumbent && p ? p.capture_day : weekStart;
    stmts.push(
      kind === "creator"
        ? db
            .prepare("INSERT INTO war_flags (kind, week_start, passport_id, account_id, capture_day, created_at) VALUES (?, ?, ?, NULL, ?, ?) ON CONFLICT (kind, week_start) DO NOTHING")
            .bind(kind, weekStart, holder, captureDay, ts)
        : // D-1059(10): the holder was read in an earlier statement; an erasure committing in between already tombstoned
          // every war_flags row (account_data.ts deleteAccount), so the account is re-checked HERE, at write time, and
          // an erased holder's row gets the same per-row tombstone erasure writes (the flag still lifts nobody).
          db
            .prepare(
              `INSERT INTO war_flags (kind, week_start, passport_id, account_id, capture_day, created_at)
               SELECT ?, ?, NULL, CASE WHEN EXISTS (SELECT 1 FROM accounts a WHERE a.account_id = ?) THEN ? ELSE 'del_' || lower(hex(randomblob(16))) END, ?, ?
                WHERE 1 ON CONFLICT (kind, week_start) DO NOTHING`,
            )
            .bind(kind, weekStart, holder, holder, captureDay, ts),
    );
  }
  if (stmts.length > 0) await db.batch(stmts);
  return { week_start: weekStart, ...held };
}

const isCount = (x: unknown): x is number => typeof x === "number" && Number.isInteger(x) && x >= 0;

function parseStandings(raw: string | null): WarStandingsArtifact | null {
  if (!raw) return null;
  try {
    const a = JSON.parse(raw) as Partial<WarStandingsArtifact>;
    if (typeof a.week !== "string" || typeof a.generated_at !== "string" || !isCount(a.fan_players) || !Array.isArray(a.creators) || !Array.isArray(a.fans)) return null;
    const creators = a.creators.filter((c) => !!c && typeof c.passport_id === "string" && /^psp_[0-9a-z]{8,64}$/.test(c.passport_id) && isCount(c.wins) && isCount(c.losses));
    const fans = a.fans.filter((f) => !!f && typeof f.account_id === "string" && isCount(f.wins) && isCount(f.losses));
    return { week: a.week, generated_at: a.generated_at, creators, fans, fan_players: a.fan_players };
  } catch {
    return null;
  }
}

/** The served standings: the artifact with every serve-time drop applied, or
 * null when no artifact has been built (or it does not parse). */
export async function serveWarStandings(store: ArtifactReader, db: DbLike): Promise<WarStandingsServed | null> {
  const a = parseStandings(await store.get(WAR_STANDINGS_KEY));
  if (!a) return null;
  // Creators: the same live predicate the pack odds labels use (war-routes.ts
  // withCreatorLabels) — not retracted, claimed+, not suppressed — and not
  // closed or suspended (I23-Q12, D-1030 (a): CREATOR_CLOSED_SQL) — then the
  // war opt-in, which only a per-creator replay can answer (warOptedIn also
  // requires `verified`). Stops once the board is full.
  const live = new Map<string, string>();
  const ids = a.creators.map((c) => c.passport_id);
  for (const chunk of inChunks(ids)) {
    const { results } = await db
      .prepare(
        `SELECT p.passport_id AS passport_id, p.display_name AS display_name FROM passports p
          WHERE p.passport_id IN (${chunk.map(() => "?").join(",")})
            AND p.retracted = 0 AND p.claim_state IN ('claimed','verified')
            AND NOT EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = p.passport_id)
            AND NOT ${CREATOR_CLOSED_SQL("p.passport_id")}`,
      )
      .bind(...chunk)
      .all<{ passport_id: string; display_name: string }>();
    for (const r of results) live.set(r.passport_id, r.display_name);
  }
  const creators: WarStandingsServed["creators"] = [];
  for (const c of a.creators) {
    if (creators.length >= WAR_STANDINGS_SIZE) break;
    const name = live.get(c.passport_id);
    if (name === undefined || !(await warOptedIn(db, c.passport_id))) continue;
    creators.push({ passport_id: c.passport_id, display_name: name, wins: c.wins, losses: c.losses });
  }
  const fansWithheld = a.fan_players < WAR_FAN_BOARD_FLOOR;
  let fans: WarStandingsServed["fans"] = null;
  if (!fansWithheld) {
    const handles = new Map<string, string>();
    for (const chunk of inChunks(a.fans.map((f) => f.account_id))) {
      const { results } = await db
        .prepare(
          `SELECT u.account_id AS account_id, u.handle AS handle FROM user_profiles u
            WHERE u.account_id IN (${chunk.map(() => "?").join(",")}) AND u.war_public = 1 AND u.handle IS NOT NULL
              AND NOT EXISTS (SELECT 1 FROM account_closures c WHERE c.account_id = u.account_id AND c.state IN ${FROZEN_CLOSURE_STATES_SQL})`,
        )
        .bind(...chunk)
        .all<{ account_id: string; handle: string }>();
      for (const r of results) handles.set(r.account_id, r.handle);
    }
    fans = a.fans.flatMap((f) => (handles.has(f.account_id) ? [{ handle: handles.get(f.account_id) as string, wins: f.wins, losses: f.losses }] : [])).slice(0, WAR_STANDINGS_SIZE);
  }
  return { week: a.week, as_of: a.generated_at, creators, fans, fans_withheld: fansWithheld };
}
