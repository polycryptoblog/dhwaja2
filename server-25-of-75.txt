// findings-vocab.mjs — THE ONE HOME (D-0621) for the FINDINGS.jsonl `status`
// vocabulary. Before this file the seven words were hand-listed TWICE: as the
// keys of `VOCAB` in closure-ledger.mjs and as `CANON_STATUS` in
// validate-room-report.mjs. Two copies is how W3-10's two scouts ended up on
// two different vocabularies (SCOUT-A wrote the seven, SCOUT-B wrote `located`,
// same batch, same wave) — measured, not supposed. Anything that needs to know
// what a status word MEANS imports it from here; nothing re-lists it.
//
// The split below is the substance, not bookkeeping:
//   VOCAB       — the seven DISPOSITION words. They answer "what happened to
//                 this defect?". Only a room that DISPOSES of defects may use
//                 them, and _OUTPUT-CONTRACT.md fixes the set at seven.
//   RECON_VOCAB — a RECON word. It answers a different question — "what did
//                 the room FIND?" — and is deliberately NOT a disposition.
//                 A scout has no verdict to give (org/rooms/scout.md:18 mints
//                 `located | not_located | open` and forbids it `holds`), so
//                 folding its positive result into a disposition bucket makes
//                 it pretend to a verdict it is not allowed to make.
//
// The buckets are THREE, not two. CLOSED is unchanged and stays {fixed,
// defended}; UNRESOLVED is unchanged; RECON is neither, because "I established
// this anchor at a file:line I read" and "I ran out of scope before deciding"
// are different facts and a ledger that cannot tell them apart cannot report a
// batch's recon yield at all (#430).

/** The seven disposition words. Each answers ONE question about a register item. */
export const VOCAB = {
  fixed: "is the defect real, and did this room ship a change that makes it not reproduce (before/after evidence)?",
  defended: "is the defect real, but does an existing guard or check already stop it from biting?",
  not_a_defect: "after investigation, is this not a real defect at all (working as intended / stale claim)?",
  wrong_owner: "is the defect real, but the fix belongs in a file or scope this room does not own?",
  not_reproduced: "did the room try to reproduce the defect at the named file:line and fail to?",
  not_located: "could the room not find the file/line the defect claims exists?",
  open: "is the defect real, unresolved, and does someone still need to act on it (reason: needs_decision | blocked)?",
};

/** Recon words: what a read-only room FOUND. Not a disposition, not a verdict. */
export const RECON_VOCAB = {
  located: "did a read-only room establish this anchor at a file:line it actually read (recon yield, NOT a verdict on the defect)?",
};

export const CLOSED_SET = new Set(["fixed", "defended"]);
export const UNRESOLVED_SET = new Set(["open", "wrong_owner", "not_reproduced", "not_located"]);
export const RECON_SET = new Set(Object.keys(RECON_VOCAB));

/** The seven, in _OUTPUT-CONTRACT.md order. The validator's allow-list floor. */
export const CANON_STATUS = Object.keys(VOCAB);
/** Every word the ledger can bucket: the seven plus the recon word(s). */
export const ALL_STATUS = [...CANON_STATUS, ...RECON_SET];
export const KNOWN_STATUSES = new Set(ALL_STATUS);
/** _OUTPUT-CONTRACT.md: an `open` row owes a `reason` FIELD, one of these two. */
export const OPEN_REASONS = ["needs_decision", "blocked"];

// A room contract may declare its OWN status vocabulary; the extension is
// DERIVED from the contract file, never hand-listed, so a contract that mints a
// word tomorrow is honoured without a code change. The `(?!VERDICT)` guard is
// load-bearing: note-auditor.md and teachback.md both say "VERDICT vocabulary
// is `A | B`" and neither is a findings status, so the window between `status`
// and "vocabulary is" may not contain that word.
export const ROOM_VOCAB_RE = /`status`(?:(?!VERDICT)[\s\S]){0,600}?vocabulary is `([^`\n]+)`/;

/**
 * Pull a room's own status vocabulary out of its contract's TEXT.
 * @param {string} text contents of org/rooms/<room>.md
 * @returns {string[]} the declared words, or [] when the contract declares none
 */
export function readRoomVocabulary(text) {
  const m = ROOM_VOCAB_RE.exec(String(text ?? ""));
  if (!m) return [];
  return m[1].split("|").map((s) => s.trim()).filter((s) => /^[a-z_-]+$/.test(s));
}

/** Which of the three ledger buckets a mapped status falls in. */
export function bucketOf(status) {
  if (CLOSED_SET.has(status)) return "closed";
  if (RECON_SET.has(status)) return "recon";
  if (UNRESOLVED_SET.has(status)) return "unresolved";
  return null; // not_a_defect, unmappable_only — counted on their own lines
}

// THE ADDRESSEE (QUEUE #507, D-0555/D-0556). A `wrong_owner` row says "the defect
// is real and the fix belongs in a file this room does not own". That sentence
// names a RECIPIENT and, until now, the row shape
// `{id, sev, file, line, mechanism, status, evidence}` had nowhere to put one.
// MEASURED, not supposed: the judge sampled 113 rows in one batch, 32 were
// `wrong_owner`, 30 of the 32 carried an actionable mechanism, and 0 of 32
// carried an addressee; the teachback found `reason` absent on 41 of 41
// `wrong_owner` rows and present on the `open` ones. A field omitted 41 of 41
// times is not carelessness -- it is a field that does not exist. So 223 register
// defects are located, described, and undispatchable.
//
// AN ADDRESSEE IS A FILE PATH, NOT A ROOM ID, and the reason is mechanical
// rather than stylistic:
//   - The room cut is RE-DERIVED every wave (ORG-01-batch/R08-register-recut:
//     306 of 2,214 rows changed room in a single re-cut, and the <=12 part cap
//     minted +36 new room-parts). A room id written into a row in W3-11 names a
//     room that does not exist by W3-14. A path does not move when the cut moves.
//   - The ownership table is ALREADY KEYED BY PATH: `ownerOf` in
//     `scripts/register-ownership-audit.mjs:51-56` is built from BATCHES.json's
//     `rooms[].owned[]`, i.e. path -> [{batch, room}]. A path is the key that
//     table takes; a room id is the value it returns. Storing the value means
//     storing an answer that expires; storing the key means the answer is
//     recomputed, by R08's rule, at the moment it is needed.
//   - The room writing the row KNOWS the path -- it is the `file:line` it just
//     read -- and does NOT know the receiving room, because a room is never told
//     the whole cut. The field asks for what the writer has.
//
// Hence: the field carries a repo-relative POSIX path; the ROOM is resolved from
// it by `recutRoom()`, R08's own exported function, and never stored.

/** The field name. Nothing spells it as a string literal twice. */
export const ADDRESSEE_KEY = "addressee";

/** Exactly the statuses that owe an addressee. */
export const ADDRESSEE_REQUIRED_STATUSES = new Set(["wrong_owner"]);

/**
 * Does this status owe an addressee?
 * A predicate rather than a set literal at each call site, so the validator and
 * the router cannot drift apart the way `VOCAB` and `CANON_STATUS` once did.
 * @param {unknown} status
 */
export function requiresAddressee(status) {
  return ADDRESSEE_REQUIRED_STATUSES.has(String(status ?? ""));
}

/** Stable machine codes. The validator renders prose; the router renders a reason. */
export const ADDRESSEE_FAULTS = {
  missing: "no `addressee` field at all",
  not_a_string: "`addressee` is not a string",
  empty: "`addressee` is blank",
  absolute: "`addressee` is an absolute path; it must be repo-relative",
  escapes_root: "`addressee` escapes the repo root with `..`",
  not_path_shaped: "`addressee` is not a path — it has no `/` and no file extension, so no ownership table can be keyed by it",
};

/**
 * The TOTAL parse. Returns the bare repo-relative POSIX path, or null when the
 * value cannot be one.
 *
 * A trailing `:NN` or `:NN-NN` is STRIPPED rather than rejected: a room that
 * writes `services/trust-core/src/x.ts:40` has named the right file in a longer
 * form, and the ownership table is keyed by the bare path. Rejecting it would
 * cost a re-run to delete four characters.
 *
 * This is the SAME function the validator uses to decide legality and the router
 * uses to decide what to look up, on purpose: two parsers is how a row passes
 * validation and then fails to route.
 * @param {unknown} v
 * @returns {string|null}
 */
export function normalizeAddressee(v) {
  if (typeof v !== "string") return null;
  let s = v.trim();
  if (!s) return null;
  s = s.replace(/:\d+(?:[-:]\d+)?$/, "");
  if (!s) return null;
  if (s.startsWith("/") || /^[A-Za-z]:[\\/]/.test(s)) return null;
  if (s.includes("\\")) return null;
  if (s.split("/").some((seg) => seg === ".." )) return null;
  // Path-shaped: it has a directory separator, or it is a bare filename with an
  // extension (`DEPLOY.md` is a real fix site -- R08 measured two such rows), or it
  // is a dotfile (`.gitignore` -- MEASURED: one real wrong_owner row addresses it, and
  // an earlier `{1,8}` extension bound rejected it for being nine letters long. The
  // corpus found that, not review).
  if (!s.includes("/") && !/^\.[A-Za-z0-9_-]+$/.test(s) && !/\.[A-Za-z0-9]{1,8}$/.test(s)) return null;
  return s;
}

/**
 * Why this value cannot be an addressee, as a machine code; null when it can.
 * @param {unknown} v the RAW field value (pass `undefined` when the key is absent)
 * @returns {keyof typeof ADDRESSEE_FAULTS | null}
 */
export function addresseeFault(v) {
  if (v === undefined || v === null) return "missing";
  if (typeof v !== "string") return "not_a_string";
  const s = v.trim();
  if (!s) return "empty";
  const stripped = s.replace(/:\d+(?:[-:]\d+)?$/, "");
  if (!stripped) return "empty";
  if (stripped.startsWith("/") || /^[A-Za-z]:[\\/]/.test(stripped)) return "absolute";
  if (stripped.includes("\\")) return "not_path_shaped";
  if (stripped.split("/").some((seg) => seg === "..")) return "escapes_root";
  if (normalizeAddressee(v) === null) return "not_path_shaped";
  return null;
}

// THE REGISTER LINK (QUEUE #548). A FINDINGS row is a DISPOSITION of a register
// item, and until now it named that item only in PROSE, inside `mechanism`. The
// link was therefore recovered by string-scanning, and two defensible extraction
// methods over the same corpus returned 282 and 356 closed -- 26% apart. A number
// nobody can reproduce is not a measurement, and closure is the number this org
// steers by.
//
// MEASURED over the committed corpus (4,380 rows), not supposed:
//   - 2,433 rows (55.5%) are linked to NO register item at all.
//   - 37 rows already carry `iid` and 25 already carry `item`. Rooms invented the
//     field because the shape had nowhere to put one -- the same way `addressee`
//     was omitted 41 of 41 times above. 60 of those 62 rows are INVISIBLE to
//     `closure-ledger.mjs`s linker, so a room that did the right thing was
//     counted with the rooms that said nothing.
//
// THE NAME IS `iid`, AND IT IS REUSED, NOT MINTED (ADR-0002 §11):
//   - REGISTER.jsonl's own key is `iid` (`closure-ledger.mjs` loads `row.iid`),
//     so the findings row and the register row spell the join column the same.
//   - 37 rows in the corpus already write it. Minting a third spelling would make
//     the divergence this field exists to close.
// `item`, `register`, `register_id` and `registerId` are read as legacy aliases
// and never written: the corpus carries them and a reader that ignores them
// throws away work already done.
//
// THE DECLARATION BEATS THE PROSE, and that is the substance rather than a
// preference. A prose scan cannot tell a row's SUBJECT from a row's MENTION: a
// mechanism reading "this is NOT the same defect as I3-0963" links the row to
// I3-0963, and a row that names a sibling, a duplicate or a counter-example is
// counted as a disposition of it. That is a false CLOSURE, not a missing one.

/** The field name. Nothing spells it as a string literal twice. */
export const REGISTER_KEY = "iid";

/** Legacy spellings read from the committed corpus. READ-ONLY -- never written. */
export const REGISTER_ALIAS_KEYS = ["item", "register", "register_id", "registerId"];

// ONE literal for the token, because there were two and they disagreed:
// `closure-ledger.mjs` scanned with `/\bI3-\d{3,6}\b/g` and anchored with
// `/^I3-\d+$/i`, so a 7-digit id was legal to one limb and invisible to the
// other. The register's own ids are 4 digits; the bound is generous and SHARED.
export const REGISTER_ID_RE = /\bI3-\d{3,8}\b/g;
const REGISTER_ID_ANCHORED = /^I3-\d{3,8}$/i;

/** A FINDINGS `iid` naming a QUEUE row (QUEUE #992): exactly `QUEUE #<n>`, n from 1 (no leading zero), beside `I3-NNNN` and
 *  `none: <what>`. I34-TOOLS-B (I33 B21, FAMILY-033): moved here from validate-room-report.mjs, which re-exports it, so the
 *  validator, registerIdFault and the closure ledger read one predicate. */
export const QUEUE_IID_RE = /^QUEUE #[1-9]\d{0,5}$/;
export const isQueueIid = (v) => typeof v === "string" && QUEUE_IID_RE.test(v.trim());

/** The members of a DECLARED link value: an array's elements and each string's `,`/`;` pieces, trimmed; a piece that is a
 *  whole QUEUE id is one member, any other piece splits further at whitespace (so `I3-0001 I3-0002` is two members). */
function linkMembers(v) {
  const out = [];
  const push = (x) => {
    if (Array.isArray(x)) { for (const y of x) push(y); return; }
    if (typeof x !== "string") return;
    for (const piece of x.split(/[,;]+/)) {
      const p = piece.trim();
      if (!p) continue;
      if (isQueueIid(p)) out.push({ queue: true, v: p });
      else for (const tok of p.split(/\s+/)) if (tok) out.push({ queue: false, v: tok });
    }
  };
  push(v);
  return out;
}

/** Is this exactly one register id? */
export function isRegisterId(v) {
  return typeof v === "string" && REGISTER_ID_ANCHORED.test(v.trim());
}

/**
 * The TOTAL parse of a DECLARED register-link value. Accepts a single id, a
 * comma/space separated list, or an array of either; returns canonical
 * upper-case ids, deduped and in first-seen order, or `[]` when the value
 * declares none.
 *
 * This is the SAME function the validator uses to decide legality and the ledger
 * uses to decide what to link, on purpose: two parsers is how a row passes
 * validation and then fails to link -- the defect `normalizeAddressee` exists to
 * prevent, in the other field.
 * @param {unknown} v
 * @returns {string[]}
 */
export function normalizeRegisterIds(v) {
  const out = [];
  const seen = new Set();
  // I34-TOOLS-B (I33 B21): a whole `QUEUE #<n>` member is a declared link to a QUEUE row, not a register id: it is skipped
  // here (normalizeQueueIds reads it) and never poisons the register ids beside it.
  for (const { queue, v: p } of linkMembers(v)) {
    if (queue) continue;
    if (!REGISTER_ID_ANCHORED.test(p)) return [];       // a malformed member poisons the whole
    const u = p.toUpperCase();
    if (!seen.has(u)) { seen.add(u); out.push(u); }
  }
  return out;
}

/** The QUEUE ids a DECLARED link value names (`QUEUE #992`, or an array holding one), canonical and deduped; `[]` when any
 *  member is neither a QUEUE id nor a register id (the poison rule of normalizeRegisterIds). */
export function normalizeQueueIds(v) {
  const out = [];
  for (const { queue, v: p } of linkMembers(v)) {
    if (!queue && !REGISTER_ID_ANCHORED.test(p)) return [];
    const c = p.replace(/\s+/g, " ");
    if (queue && !out.includes(c)) out.push(c);
  }
  return out;
}

/** Stable machine codes, mirroring ADDRESSEE_FAULTS. */
export const REGISTER_FAULTS = {
  not_a_string: "`iid` is neither a string nor an array of strings",
  empty: "`iid` is blank",
  malformed: "`iid` is not a register id — the shape is `I3-` followed by 3 to 8 digits (or `QUEUE #` and a QUEUE row number from 1)",
};

/**
 * Why this DECLARED value cannot be a register link, as a machine code; null when
 * it can. `undefined`/absent is NOT a fault: the field is optional, because a
 * tooling or process finding is legitimately about no register item. What is a
 * fault is declaring a link the ledger cannot parse -- that is worse than
 * declaring none, since the row looks linked and is not.
 * @param {unknown} v
 * @returns {keyof typeof REGISTER_FAULTS | null}
 */
export function registerIdFault(v) {
  if (v === undefined || v === null) return null;
  if (typeof v !== "string" && !Array.isArray(v)) return "not_a_string";
  if (Array.isArray(v) && v.some((x) => typeof x !== "string")) return "not_a_string";
  const flat = Array.isArray(v) ? v.join(" ").trim() : String(v).trim();
  if (!flat) return "empty";
  // I34-TOOLS-B (I33 B21): a QUEUE id is a declared link too, alone or in an array (`QUEUE #0` is not one).
  return normalizeRegisterIds(v).length || normalizeQueueIds(v).length ? null : "malformed";
}

/** The QUEUE ids a row DECLARES in its `iid` (QUEUE ids post-date the legacy aliases, which never carry one). */
export function declaredQueueIds(row) {
  if (!row || typeof row !== "object" || row[REGISTER_KEY] === undefined || row[REGISTER_KEY] === null) return [];
  return normalizeQueueIds(row[REGISTER_KEY]);
}

/**
 * The register ids a row DECLARES, canonical key first and then the legacy
 * aliases. `[]` when the row declares none -- which is not the same fact as
 * "no id could be scraped out of its prose", and the ledger keeps them apart.
 * @param {Record<string, unknown>} row
 * @returns {string[]}
 */
export function declaredRegisterIds(row) {
  if (!row || typeof row !== "object") return [];
  for (const k of [REGISTER_KEY, ...REGISTER_ALIAS_KEYS]) {
    if (row[k] === undefined || row[k] === null) continue;
    const ids = normalizeRegisterIds(row[k]);
    if (ids.length) return ids;
  }
  return [];
}

/** sha256 hex of a file's bytes, or null (unreadable, not a file, no hasher). Shells out to the system hasher like
 * close-run.mjs's hashFile: tests/invariants/single-hasher.test.ts keeps node:crypto out of scripts/. */
export function fileSha256(file) {
  for (const [cmd, args] of [["sha256sum", []], ["shasum", ["-a", "256"]]]) {
    try {
      const h = execFileSync(cmd, [...args, "--", String(file)], { encoding: "utf8", stdio: ["ignore", "pipe", "ignore"] }).trim().split(/\s+/)[0];
      if (/^[0-9a-f]{64}$/.test(h)) return h;
    } catch { /* next hasher */ }
  }
  return null;
}

}

// ---- B28-TOOLING (QUEUE #655): NEGATION RECOGNISED BY POSITION, NOT LISTED BY PHRASE -------------
// ONE recogniser for "does this text ASSERT item N, or only mention it?", imported by close-run.mjs
// gate 0f (`#N`) and closure-ledger.mjs's prose limb (`I3-NNNN`). Before B28 each carried a copy of a
// whole-clause word test, which (J-FINAL-02) missed exclusion prepositions ('without #561', 'outside
// #561') and over-reached the other way ('fixes I3-0963 and no regressions' dropped the true link,
// because 'no' anywhere in the clause negated the id BEFORE it).
// The lexicon below is an OPEN LIST of negators and exclusion phrasings, grown ruling by ruling
// (D-0823(8), D-0825(1)) -- not a closed class: a negator that is not on it reads as an ASSERTION, so
// the mention it governs still routes at gate 0f and links in the ledger. What is structural is the
// scope each one takes:
//   WHOLE  -- a predicate negator negates its whole conjunct, subject included ('I3-0963 is not it',
//             'does not apply', "isn't", 'never', 'nor', 'unrelated'). `not` is always WHOLE: 'X not
//             affected' names X as the thing NOT affected, and ambiguity errs to the mention.
//   RIGHT  -- a determiner/preposition of exclusion negates from itself to the conjunct end ('no
//             regressions', 'without #561', 'outside #561', 'other than', 'distinct from').
// A conjunct ends at `but`, and at `and`/`or` when the next conjunct OPENS with a negator. A
// parenthetical is negated when its host conjunct is WHOLE-negated ('not the bug in foo.ts:12
// (I3-0963)'); a RIGHT operator's scope stops at it. Clause punctuation (`; | , :` + space, a newline, a
// dash, a sentence-ending `.`) splits; `foo.ts:12` and `a.b` do not.
// B28-FX-TOOL (D-0823(8) L1): apostrophe-less contractions ('isnt', 'doesnt') negate as their spelled forms do.
// D-0825(1) (J-TOOL-NEG-UNLISTED): 'has nothing to do with', 'none of ... applies', 'unconnected to',
// 'irrelevant to' negate their whole conjunct as 'unrelated' does; 'free of' excludes rightward as 'without'.
const NEG_WHOLE_WORDS = "not|never|nor|neither|cannot|unrelated|nothing\\s+to\\s+do\\s+with|none\\s+of|unconnected\\s+to|irrelevant\\s+to|no\\s+longer|\\w+n['\\u2019]t|(?:is|are|was|were|do|does|did|has|have|had|could|should|would|ca|wo|ai|need|must)nt";
const NEG_RIGHT_WORDS = "no|without|free\\s+of|outside|except|excepting|excluding|besides|beyond|barring|unlike|minus|other\\s+than|rather\\s+than|instead\\s+of|as\\s+opposed\\s+to|(?:distinct|different|differs|separate|apart|aside|away)\\s+from|different\\s+than|independent(?:ly)?\\s+of|regardless\\s+of|irrespective\\s+of|in\\s+lieu\\s+of|versus|vs|save\\s+for";
// B28-FX-TOOL (D-0823(8) L1): `no` AFTER a word that is not a preposition or conjunction is a predicate
// negator ('#561 has no bearing on this', 'plays no part', 'offers no protection') and negates its WHOLE
// conjunct, subject included, as base 3889e29's clause test did; conjunct-initial `no` and `with no`/`and no`
// keep the rightward scope ('fixes I3-0963 with no regressions' still links I3-0963).
const NO_PREDICATE = /(?<=[\w#)\]'"`\u2019]\s+)(?<!(?<![\w-])(?:with|and|or|nor|but|to|for|of|in|on|at|by|from|into|onto|under|over|via|as|than|if|so|yet|e\.g\.|i\.e\.)\s+)(?<![\w-])no(?![\w-]|\s+longer(?![\w-]))/i;
const NEG_WHOLE = new RegExp(`(?<![\\w-])(?:${NEG_WHOLE_WORDS})(?![\\w-])`, "i");
const NEG_RIGHT = new RegExp(`(?<![\\w-])(?:${NEG_RIGHT_WORDS})(?![\\w-])`, "i");
const NEG_OPENS = new RegExp(`^\\s*(?:${NEG_WHOLE_WORDS}|${NEG_RIGHT_WORDS})(?![\\w-])`, "i");
const CLAUSE_BREAK = /[;|\n—–]|[,:](?=\s|$)|(?<!(?<![\w-])(?:vs|cf))\.(?=\s|$)|\s-{1,2}\s/;
// B28-FX-TOOL (D-0823(8) L1): a clause that follows a NEGATED clause across a list comma and opens with an
// id is an item of the negated list ('NOT the same defect as #319, #561 or #655'; 'Unrelated to I3-0963, I3-0964').
const LIST_ITEM = /^\s*(?:(?:and|or|nor|&)\s+)?(?:(?:QUEUE|queue|docket|row|item|finding)\s+)?(?:#\d|[A-Z][A-Za-z0-9]*-\d)/;
// The SPLIT form of a RIGHT operator (D-0820(6), B28-TOOL2-3): an adjective of distinction may take its
// noun between itself and `from` ('separate mechanism from I3-0963', 'a distinct defect from #561').
// Scope starts at the adjective, as the adjacent form's does. Kept OUT of NEG_RIGHT_WORDS on purpose:
// that list also feeds statesNoneSubject's head cut, NEG_OPENS and CONJUNCT_BREAK, and the `none:`
// validator must not change with this recogniser (nothing binds B27 or earlier, and no row newly fails).
const NEG_RIGHT_SPLIT = /(?<![\w-])(?:distinct|different|separate)(?:\s+(?!(?:from|than)(?![\w-]))[^\s]+){1,4}\s+(?:from|than)(?![\w-])/i;
// B28-FX-TOOL: `so` opens a new conjunct ('fixes I3-0963 so the route no longer leaks' asserts I3-0963).
const CONJUNCT_BREAK = new RegExp(`\\s+but\\s+|\\s+so\\s+|\\s+(?:and|or)\\s+(?=(?:${NEG_WHOLE_WORDS}|${NEG_RIGHT_WORDS})(?![\\w-]))`, "i");

// I7-CLOSE-TOOLS (#685): three more exclusion phrasings and two predicate phrasings, kept OUT of
// NEG_WHOLE_WORDS/NEG_RIGHT_WORDS for the reason NEG_RIGHT_SPLIT is: those lists also feed
// statesNoneSubject, and the `none:` validator must not change with this recogniser.
//   RIGHT_EXTRA -- 'lacking #561', 'sans #561', 'unaffected by #561' exclude rightward as 'without' does.
//   WHOLE_EXTRA -- '#561 is out of scope here', '#561 aside' (the id PRECEDES the negator, so only a
//                  whole-conjunct negation reaches it); bare `aside` only -- 'aside from' is RIGHT already.
const NEG_RIGHT_EXTRA = /(?<![\w-])(?:lacking|sans|unaffected\s+by)(?![\w-])/i;
// I16-W02F (#831c): `out-of-scope` and `outside (of) (the) scope` are the same scope-out phrase.
const SCOPE_OUT = String.raw`out(?:\s+|-)of(?:\s+|-)(?:the\s+|this\s+)?scope|outside\s+(?:of\s+)?(?:the\s+|this\s+)?scope`;
// I16-W02F (#655b, D-0824(4) residual): '#561 with no connection', '#561 with zero relation' say the id is UNRELATED.
const NO_RELATION = String.raw`(?:with\s+)?(?:no|zero)\s+(?:connection|relation(?:ship)?|link|bearing|overlap)(?:\s+(?:to|with))?`;
const NEG_WHOLE_EXTRA = new RegExp(String.raw`(?<![\w-])(?:${SCOPE_OUT}|${NO_RELATION}|aside(?!\s+from(?![\w-]))(?![\w-]))`, "i");
// I7-CLOSE-TOOLS (#687): a negated STATUS predicate ('does not reproduce', 'is not reproduced', 'no longer
// reproduces') that follows a parenthetical id is the disposition of the defect the parenthetical names,
// not a statement that the id is unrelated: 'the defect (I3-1545) does not reproduce' still names I3-1545.
const STATUS_NEG_BODY = String.raw`(?:(?:does|do|did|is|are|was|were|has|have|had|can|could)\s+)?(?:not|never|cannot|no\s+longer|\w+n['\u2019]t|(?:is|are|was|were|do|does|did|has|have|had|ca|wo|ai)nt)\s+(?:be\s+|been\s+|being\s+)?(?:reproduc\w*|repro(?![\w-]))`;

// I16-W02F (#831c): the same negated status predicate INSIDE a conjunct ('I3-0001 not reproduced') is the row's
// disposition of the defect it names, not an exclusion of it: it is blanked before the WHOLE test.
const STATUS_NEG_ANY = new RegExp(String.raw`(?<![\w-])${STATUS_NEG_BODY}`, "gi");

/** The asserted (un-negated) prefix of one conjunct: "" when WHOLE-negated (the whole conjunct when
 *  `exclusionsOnly`, which still marks it negated so a following list inherits). */
function assertedPart(conj, exclusionsOnly = false) {
  // I16-W02F (#831c): a scope-out phrase negates only its OWN `and`/`or` conjunct: 'fixes X and the rest is out of
  // scope' still asserts X (the head is judged on its own; the tail is negated).
  const ex = NEG_WHOLE_EXTRA.exec(conj);
  const joins = ex ? [...conj.slice(0, ex.index).matchAll(/\s+(?:and|or)\s+/gi)] : [];
  if (joins.length) {
    const head = assertedPart(conj.slice(0, joins[joins.length - 1].index), exclusionsOnly);
    return { text: exclusionsOnly ? conj : head.text, open: true };
  }
  const probe = conj.replace(STATUS_NEG_ANY, (m) => " ".repeat(m.length));
  if (NEG_WHOLE.test(probe) || NEG_WHOLE_EXTRA.test(probe) || NO_PREDICATE.test(probe)) return { text: exclusionsOnly ? conj : "", open: true };
  const at = [NEG_RIGHT.exec(conj), NEG_RIGHT_SPLIT.exec(conj), NEG_RIGHT_EXTRA.exec(conj)].filter(Boolean).map((m) => m.index);
  return at.length ? { text: conj.slice(0, Math.min(...at)), open: true } : { text: conj, open: false };
}

/**
 * The fragments of `text` that ASSERT what they name: scan an id pattern over these, never over the
 * raw text. Deterministic; `[]` for a non-string. See the block comment above for the grammar.
 * `opts.exclusionsOnly` (B28-FX-TOOL, the sidecar's reading, D-0823(8) L4): only an EXCLUSION (the
 * rightward class: 'without', 'outside', 'independent of', determiner 'no' after a preposition) and the
 * list it governs cut; a negated PREDICATE ('I3-1411 premise does not reproduce', 'the door never
 * painted the I3-0002 notice' -- the disposition itself) does not, but still opens a list, so
 * 'NOT the same defect as #1, I3-0963' drops I3-0963. iid-backfill.mjs reads negated predicates through
 * base 3889e29's own cue window instead. The ledger and gate 0f use the default.
 * @param {unknown} text
 * @param {{ exclusionsOnly?: boolean }} [opts]
 * @returns {string[]}
 */
export function assertedClauses(text, opts = {}) {
  const exclusionsOnly = Boolean(opts && opts.exclusionsOnly);
  if (typeof text !== "string") return [];
  // Parentheticals are lifted out of the WHOLE text first, so a list inside one ("(I3-0687, I3-0725)")
  // is never cut apart by the clause split, and each is judged against the conjunct that hosts it.
  const parens = [];
  const host = text.replace(/\(([^()]*)\)/g, (all, inner, at) => { parens.push({ inner, at, len: all.length }); return " ".repeat(all.length); });
  const spans = (str, re) => {
    const out = [];
    let from = 0;
    for (const m of str.matchAll(new RegExp(re.source, "gi"))) { out.push([from, m.index]); from = m.index + m[0].length; }
    out.push([from, str.length]);
    return out;
  };
  const clauses = spans(host, CLAUSE_BREAK).map(([a, b]) => ({ a, conj: spans(host.slice(a, b), CONJUNCT_BREAK).map(([x, y]) => [a + x, a + y]) }));
  const out = [];
  let negatedBefore = false;
  for (const { a, conj } of clauses) {
    const listContinues = negatedBefore && a > 0 && host[a - 1] === "," && LIST_ITEM.test(host.slice(conj[0][0], conj[0][1]));
    conj.forEach(([x, y], i) => {
      if (i === 0 && listContinues) { negatedBefore = true; return; }
      const part = assertedPart(host.slice(x, y), exclusionsOnly);
      if (part.text.trim()) out.push(part.text);
      negatedBefore = part.open;
    });
  }
  for (const { inner, at, len } of parens) {
    // A parenthetical is part of its host conjunct: a WHOLE negation there negates it too ('not the
    // bug in foo.ts:12 (I3-0963)'). A RIGHT operator governs its own phrase; the parenthetical is an
    // apposition that may attach to the sentence ('no record distinguishing them (I3-0687, ...)').
    const hostConj = clauses.flatMap((c) => c.conj).find(([x, y]) => at >= x && at <= y);
    const hostText = hostConj ? host.slice(hostConj[0], hostConj[1]) : "";
    const before = hostConj ? host.slice(hostConj[0], at) : "";
    const statusAfter = hostConj && !NEG_WHOLE.test(before) && !NEG_WHOLE_EXTRA.test(before) && STATUS_NEG_AFTER.test(host.slice(at + len, hostConj[1]));
    if (!exclusionsOnly && hostConj && !statusAfter && (NEG_WHOLE.test(hostText) || NEG_WHOLE_EXTRA.test(hostText))) continue;
    out.push(...assertedClauses(inner, opts));
  }
  return out;
}

// A `none:` description must NAME what the row is about. Refused structurally: after unwrapping
// brackets and quotes it may not OPEN with a placeholder or a negator (D-0814(6)), and the head of its
// first clause -- the text before any negator -- must carry a content word, so a bare predicate
// ('does not apply', 'it is not applicable') names nothing. The placeholder class is words that
// stand for an absent description; the function-word class is what a head with no subject is made of.
const PLACEHOLDER_WORD = /^(?:n[\/-]?a|n\.a\.?|na|none|null|nil|nothing|nobody|tbd|tba|todo|unknown|inapplicable|irrelevant|various|misc|etc|same|ditto|see|above|below|pending|nope|nah|non-\w+|non(?:applicable|relevant|existent)|-+|\?+|\.+|…|\*+)$/i;
// D-1123(3) (I34-RG-W3T TB:14): x, ok, skip, whatever and placeholder are placeholders in a --no-families reason, which
// compile-room-prompt.mjs's door asks through statesNoneSubject. D-1126(5) (I34-RJ-16 SEAM-19): only there -- isIidNone
// keeps its i34-head-5 rule (iidNoneSubject) -- and each is matched against a whole word, so a hyphenated word is one
// word ("ok-path census", "X-ray scan of images" name a subject).
const NO_FAMILIES_WORD = /^(?:x|ok|skip|whatever|placeholder)$/i;
// I38-T3 (I34 B6, RG-W3T/TB-JA07; D-1123(3)): a one-word reason of filler or junk is no subject either. JUNK_WORD is read
// by statesNoneSubject (the --no-families door) and statesClause, never by isIidNone (D-1126(5)).
const JUNK_WORD = /^(?:yes|yeah|yep|foo|bar|baz|qux|asdf|blah|lorem|ipsum|dummy|filler|stub|tbc|wip|x{2,}|fixme|tk)$/i;
// I40-T-TOOLS (I40S-055): only these junk words refuse as a FIRST word; the rest of JUNK_WORD (bar, stub, dummy, filler, wip,
// yes ...) are English words that open real reasons (`bar chart labels only`) and refuse only as the reason's only content.
const JUNK_FIRST = /^(?:asdf|lorem|ipsum|tbc|x{2,}|fixme|tk)$/i;
const FUNCTION_WORD = /^(?:a|an|the|it|its|this|that|these|those|they|there|here|which|what|is|are|was|were|be|been|being|am|do|does|did|done|has|have|had|can|could|will|would|shall|should|may|might|must|just|simply|also|still|as|per|so|to|of|for|in|on|at|by|applies|apply|applicable|relevant|needed|required)$/i;
/**
 * @param {unknown} desc  the text after `none:`
 * @returns {boolean} true when the description states a subject, as a --no-families reason (D-1123(3), D-1126(5)):
 *   the `none:` rule (iidNoneSubject), no NO_FAMILIES_WORD or JUNK_FIRST word as its first word, and one word outside
 *   NO_FAMILIES_WORD and JUNK_WORD
 */
export function statesNoneSubject(desc) {
  return iidNoneSubject(desc) && namesSubject(desc, (w) => PLACEHOLDER_WORD.test(w) || NO_FAMILIES_WORD.test(w) || JUNK_WORD.test(w), (w) => PLACEHOLDER_WORD.test(w) || NO_FAMILIES_WORD.test(w) || JUNK_FIRST.test(w));
}
// D-1141(2) (RG-GATES-01, -05, I36REP-02): the ONE single-word placeholder list for a persona/method clause and gate 0i's
// no_persona_reason -- the `none:` placeholders, the --no-families words, and the line's own kind words standing in for a reason.
const CLAUSE_PLACEHOLDER = /^(?:skip|skipped|applied|executed|not[ _]?applicable|does[ _]?not[ _]?apply|to[ _]?do|ok|x|whatever|placeholder)$/i;
// I38-T3 (Q1000-JA01): words that carry no reason on their own -- negators, the line's own kind words, and `reason`/`given`
// and their kin -- so `no persona`, `no reason given` and `TODO fill in` name nothing. A clause needs one word outside
// FUNCTION_WORD, PLACEHOLDER_WORD, CLAUSE_PLACEHOLDER, JUNK_WORD and this class.
const CLAUSE_FILLER = /^(?:no|not|none|never|nothing|reason|reasons|given|provided|stated|supplied|needed|necessary|later|yet|fill|filled|persona|personas|lens|lenses|skill|skills|method|methods|step|steps|determined|decided|confirmed|announced|added|now|idk|dunno)$/i;
// I40-T-TOOLS (I40S-024, -081): the last eight make the spelled-out placeholders `to be determined`, `to be confirmed`,
// `will be added`, `n/a for now` and `idk` name nothing; a clause with one other word still states it.
// I38-T3 (Q1000-B3): zero-width characters and HTML tags are read as nothing. Strikethrough `~` and the low and angle quote
// marks need no rule of their own: the content-word test below trims every non-alphanumeric end of every word.
// I40-T-TOOLS (I40S-025, -054): batch-gen.mjs:44's class (zero-width, bidi, U+2060-2064) plus the soft hyphen, the combining
// grapheme joiner, U+180E and the Hangul fillers; HTML entities (ENTITY) are decoded first, so `n&#47;a` is `n/a`.
const INVISIBLE = /[\u00ad\u034f\u115f\u1160\u180e\u200b-\u200f\u202a-\u202e\u2060-\u2064\u3164\ufeff]/g;
const NAMED_ENTITY = { nbsp: " ", amp: "&", lt: "<", gt: ">", quot: '"', apos: "'", sol: "/", shy: "", zwsp: "", zwj: "", zwnj: "" };
const ENTITY = /&(?:#(\d{1,7})|#[xX]([0-9A-Fa-f]{1,6})|([A-Za-z]+));/g;
const decodeEntities = (s) => s.replace(ENTITY, (m, dec, hex, name) => {
  const cp = dec !== undefined ? Number(dec) : hex !== undefined ? parseInt(hex, 16) : null;
  if (cp !== null) return cp <= 0x10ffff ? String.fromCodePoint(cp) : m;
  return Object.hasOwn(NAMED_ENTITY, name.toLowerCase()) ? NAMED_ENTITY[name.toLowerCase()] : m;
});
const HTML_TAG = /<\/?[A-Za-z][^<>]*>/g;
/**
 * A persona or method line's clause, or a no_persona_reason, states something (D-1141(2)): false when it is empty, a whole
 * bracketed template (`<clause>`, `[clause]`, `{clause}`), or one placeholder word (PLACEHOLDER_WORD or CLAUSE_PLACEHOLDER,
 * after outer quotes, parens, emphasis and code marks (`*`, `_`, backticks) and punctuation are trimmed and joiners read as spaces); an ordinary sentence passes, including one
 * that opens with `no` or `not` (`no screen reader in the VM`). I38-T3 (Q1000-JA01, -B3): zero-width characters are dropped and
 * HTML tags read as spaces first, and the clause must hold one word (ends trimmed of every non-alphanumeric) outside
 * FUNCTION_WORD, PLACEHOLDER_WORD, CLAUSE_PLACEHOLDER, JUNK_WORD and CLAUSE_FILLER, so `no persona` and `no reason given` refuse.
 * I42-VALIDATE (I37-JA05): a word of one character repeated three or more times (`aaaa…`) is not that word either.
 * @param {unknown} clause
 * @returns {boolean}
 */
export function statesClause(clause) {
  if (typeof clause !== "string") return false;
  const c = decodeEntities(clause).replace(INVISIBLE, "").trim();
  if (/^(?:<[^<>]*>|\[[^[\]]*\]|\{[^{}]*\})[.!,;:]*$/.test(c)) return false;
  let d = c.replace(HTML_TAG, " ");
  // FXG-03: markdown emphasis and code marks (`*`, `_`, backticks) and typographic quotes are stripped from both ends too.
  for (let prev = ""; prev !== d;) { prev = d; d = d.replace(/^[\s("'`*_\u00ab\u201c\u2018]+|[\s)"'`*_\u00bb\u201d\u2019.!,;:]+$/g, ""); }
  if (!/[A-Za-z0-9]/.test(d)) return false;
  const one = d.replace(/\s*\/\s*/g, "/").replace(/(?<=[A-Za-z])[-_]+(?=[A-Za-z])/g, " ");
  if (PLACEHOLDER_WORD.test(d) || PLACEHOLDER_WORD.test(one) || CLAUSE_PLACEHOLDER.test(one)) return false;
  const bare = (w) => w.replace(/^[^A-Za-z0-9#]+|[^A-Za-z0-9]+$/g, "");
  const noReason = (w) => FUNCTION_WORD.test(w) || PLACEHOLDER_WORD.test(w) || CLAUSE_PLACEHOLDER.test(w) || JUNK_WORD.test(w) || CLAUSE_FILLER.test(w);
  // I42-VALIDATE (I37-JA05): a word that is one character repeated three or more times (`aaaa…`, `zzz`) states nothing either.
  return one.split(/\s+/).map(bare).some((w) => /[A-Za-z0-9]/.test(w) && !noReason(w) && !/^(.)\1{2,}$/i.test(w));
}
/** The `none: <what>` rule isIidNone applies (i34-head-5's statesNoneSubject, D-1126(5)). */
function iidNoneSubject(desc) {
  if (typeof desc !== "string") return false;
  // B28-FX-TOOL (D-0823(8) L2): a placeholder spelled with joiners ('not-applicable', 'does_not_apply')
  // or a spaced slash ('N / A') is the same placeholder: the normalised form must state a subject too.
  const joined = desc.replace(/(?<=[A-Za-z])[-_]+(?=[A-Za-z])/g, " ").replace(/\s*\/\s*/g, "/");
  return namesSubject(desc) && namesSubject(joined);
}
function namesSubject(desc, isPlaceholder = (w) => PLACEHOLDER_WORD.test(w), isFirstPlaceholder = isPlaceholder) {
  let d = desc.trim();
  // I38-T3 (Q1000-N2, FXG2-06): `_`, `*` and `~` are marks, not word characters, so `_tbd_` is the placeholder tbd.
  for (let prev = ""; prev !== d;) { prev = d; d = d.replace(/^[\s("'`[{<_*~]+|[\s)"'`\]}>.!,;:_*~]+$/g, ""); }
  if (!/[A-Za-z0-9]/.test(d)) return false;
  // I40-T-TOOLS (I40S-026, -082): the first word sheds its trailing `_`, `*` and `~` marks too (its leading ones went with the outer strip above), so `_tbd_ census` opens with tbd.
  const first = d.split(/\s+/)[0].replace(/[,:;.!?)\]_*~]+$/, "");
  if (isFirstPlaceholder(first) || NEG_OPENS.test(d)) return false;
  const clause = d.split(CLAUSE_BREAK)[0];
  const cut = [NEG_WHOLE.exec(clause), NEG_RIGHT.exec(clause)].filter(Boolean).map((m) => m.index);
  const head = cut.length ? clause.slice(0, Math.min(...cut)) : clause;
  return head.split(/\s+/).map((w) => w.replace(/^[^A-Za-z0-9#]+|[^A-Za-z0-9]+$/g, "")).some((w) => /[A-Za-z0-9]/.test(w) && !FUNCTION_WORD.test(w) && !isPlaceholder(w));
}

// ---- JUDGED-BY (I8-TOOLING #819; D-0734 rule, one parser) ------------------------------------------------
// A batch judged by rooms that live in ANOTHER run dir is credited only through `<batch>/JUDGED-BY.md`, one
// line per judge report: `JUDGED-BY: <repo-relative path to a REPORT.md> sha256=<64-hex>`. The file counts only
// if EVERY line (blank lines aside) names a REPORT.md inside `root` whose sha256 matches, whose sibling PROMPT.md
// CARRIES room type `judge` (no dir-name fallback) and whose SCOPE PACKET names the batch DIRECTORY name as a
// whole token. Any failing line refuses the file. Was two copies (close-run.mjs, check-orchestrator-compliance.mjs);
// both now import this. The hasher is INJECTED: the lib never hashes ([single-hasher]).
// I16-W02F (#622a, #845a-c): each listed report must RULE (carry a CONFIRMED|REJECTED|NEEDS_REPRODUCTION line); the path is
// judged by its REAL path (a symlink out of --root is refused), a
// directory named REPORT.md is refused, a report listed twice is refused, and ` prompt-sha256=<64-hex>` pins the sibling
// PROMPT.md -- REQUIRED for a batch that bindsAfterB27, optional (checked when present) for the history batches.
export const JUDGED_BY_LINE = /^JUDGED-BY: (\S+) sha256=([0-9a-f]{64})(?: prompt-sha256=([0-9a-f]{64}))?$/;
/** A judge's verdict word (close-run.mjs COVERAGE_VERDICT's shape): a line carrying one is a line that rules. */
const JUDGE_VERDICT_RE = /\b(CONFIRMED|REJECTED|NEEDS_REPRODUCTION)\b(?!-)/;
export function promptRoomType(text) {
  const tail = [...text.matchAll(/^- ROOM TYPE:\s*([A-Za-z0-9_-]+)/gm)].pop();
  if (tail) return tail[1].toLowerCase();
  const head = /^## ROOM CONTRACT \(org\/rooms\/([A-Za-z0-9_-]+)\.md\)/m.exec(text);
  return head ? head[1].toLowerCase() : null;
}
/**
 * D-1141(2) (RG-GATES-06): THE persona ids a compiled prompt delivered -- every `## PERSONA <id>` heading at line start
 * (trailing text after the id allowed), never `## PERSONA COMMON`, never inside the SCOPE PACKET body (scopePacketOf's
 * bounds) and never inside a ``` or ~~~ fenced block. validate-room-report.mjs and close-run.mjs gate 0i both read this.
 * @param {unknown} text
 * @returns {string[]} distinct ids, in order of first appearance
 */
/** The prompt with its SCOPE PACKET body (scopePacketOf's bounds) cut out, the heading kept. I40-T-TOOLS (I40S-056): ONE
 *  excision for promptPersonaIds and validate-room-report.mjs promptMethodSteps. */
export function withoutScopePacket(text) {
  const t = String(text ?? "");
  const at = /^## SCOPE PACKET[ \t]*$/m.exec(t);
  const packet = scopePacketOf(t);
  if (!at || packet === null) return t;
  const from = at.index + at[0].length;
  return `${t.slice(0, from)}\n${t.slice(from + packet.length)}`;
}
/** I40-T-TOOLS (I40S-057, -027): THE fence reader for promptPersonaIds and validate-room-report.mjs blankFences. A ``` or ~~~
 *  line (3+ characters, up to 3 spaces in) opens a fence, except a backtick line whose info string holds a backtick (inline
 *  code, CommonMark 4.5); inside a fence a marker line of the same character, at least as long, followed by spaces or tabs
 *  only, closes it (I43-ORGTOOLS FENCESTEP: `/^[ \t]*$/`, not trim(), which also took U+00A0, VT, FF and U+3000).
 *  \returns {{fence: string|null, fenceLine: boolean}} the state after `line`, and whether `line` is a fence line */
export function fenceStep(fence, line) {
  const f = /^ {0,3}(`{3,}|~{3,})(.*)$/.exec(String(line).replace(/\r$/, ""));
  if (!f) return { fence, fenceLine: false };
  if (fence === null) return f[1][0] === "`" && f[2].includes("`") ? { fence, fenceLine: false } : { fence: f[1], fenceLine: true };
  return { fence: f[1][0] === fence[0] && f[1].length >= fence.length && /^[ \t]*$/.test(f[2]) ? null : fence, fenceLine: true };
}
export function promptPersonaIds(text) {
  const t = withoutScopePacket(text);
  const ids = [];
  let fence = null;
  for (const line of t.split("\n")) {
    const s = fenceStep(fence, line);
    fence = s.fence;
    if (s.fenceLine || fence !== null) continue;
    const m = /^## PERSONA (\S+)(?:[ \t][^\n]*)?$/.exec(line.replace(/\r$/, ""));
    if (m && m[1] !== "COMMON" && !ids.includes(m[1])) ids.push(m[1]);
  }
  return ids;
}
export function scopePacketOf(text) {
  const at = /^## SCOPE PACKET[ \t]*$/m.exec(text);
  if (!at) return null;
  const rest = text.slice(at.index + at[0].length);
  const end = /^## (?:RULINGS ON IDS THIS PACKET NAMES|TAIL)\b/m.exec(rest);
  return end ? rest.slice(0, end.index) : rest;
}
const realOrNull = (p) => { try { return realpathSync(p); } catch { return null; } };
export function judgedByCredit({ batch, batchDir, root, hashFile }) {
  const f = join(batchDir, "JUDGED-BY.md");
  if (!existsSync(f)) return null;
  const lines = readFileSync(f, "utf8").split(/\r?\n/).filter((l) => l.trim() !== "");
  if (lines.length === 0) return { ok: false, why: "JUDGED-BY.md names no report" };
  const names = new RegExp("(?<![A-Za-z0-9_-])" + batch.replace(/[.*+?^${}()|[\]\\]/g, "\\$&") + "(?![A-Za-z0-9_-])");
  const reports = [];
  const seen = new Set();
  const realRoot = realOrNull(root);
  if (realRoot === null) return { ok: false, why: "--root does not resolve" };
  const inside = (r, p) => isInside(r, p);
  for (const l of lines) {
    const m = JUDGED_BY_LINE.exec(l);
    if (!m) return { ok: false, why: "malformed line: " + JSON.stringify(l.slice(0, 120)) };
    const abs = resolve(root, m[1]);
    const rel = relative(root, abs);
    if (isAbsolute(m[1]) || !relInside(rel) || basename(abs) !== "REPORT.md") return { ok: false, why: m[1] + " is not a REPORT.md inside --root" };
    if (!existsSync(abs)) return { ok: false, why: m[1] + " does not exist" };
    const real = realOrNull(abs);
    if (real === null || !inside(realRoot, real) || basename(real) !== "REPORT.md") return { ok: false, why: m[1] + " resolves outside --root or to another file (a link)" };
    if (!statSync(real).isFile()) return { ok: false, why: m[1] + " is not a file" };
    if (hashFile(real) !== m[2]) return { ok: false, why: m[1] + " sha256 does not match" };
    if (seen.has(real)) return { ok: false, why: m[1] + " is listed more than once" };
    seen.add(real);
    const pp = join(dirname(real), "PROMPT.md");
    if (!existsSync(pp)) return { ok: false, why: m[1] + " has no sibling PROMPT.md" };
    const realPrompt = realOrNull(pp);
    if (realPrompt === null || !inside(realRoot, realPrompt) || !statSync(realPrompt).isFile()) return { ok: false, why: m[1] + " sibling PROMPT.md is not a file inside --root" };
    if (m[3] === undefined && bindsAfterB27(batch)) return { ok: false, why: m[1] + " carries no prompt-sha256= pin (required for " + batch + ")" };
    if (m[3] !== undefined && hashFile(realPrompt) !== m[3]) return { ok: false, why: m[1] + " prompt-sha256 does not match its PROMPT.md" };
    const text = readFileSync(realPrompt, "utf8");
    const t = promptRoomType(text);
    if (t !== "judge") return { ok: false, why: m[1] + " room type is " + (t ?? "unstated") + ", not judge" };
    const scope = scopePacketOf(text);
    if (scope === null || !names.test(scope)) return { ok: false, why: m[1] + " scope does not name " + batch };
    if (!readFileSync(real, "utf8").split(/\r?\n/).some((x) => JUDGE_VERDICT_RE.test(x))) return { ok: false, why: m[1] + " rules nothing (no CONFIRMED, REJECTED or NEEDS_REPRODUCTION line)" };
    reports.push(rel);
  }
  return { ok: true, reports };
}

// ---- ROOM CLASSIFIER (I16-W03J, QUEUE #614a) -------------------------------------------------------------
// close-run.mjs (gates 0d/0e/0h), check-orchestrator-compliance.mjs (HR-4/HR-5) and validate-room-report.mjs
// each decided what kind of room a folder is; the ruling-name fallback was RE-TYPED in two of them. This is
// the one rule: the compiled PROMPT.md's type (promptRoomType: last TAIL `- ROOM TYPE:`, else the ROOM
// CONTRACT heading), else a legacy `=== ROOM: <x>` header, else -- only when the caller passes `dir` -- a
// case-insensitive ruling-room name (judge-audit names first, so `JUDGEAUDIT/` is never a judge).
// close-run.mjs's skeleton `role` (:1767) keeps its own parse by ruling D-0921(8).
export const RULING_TYPES = new Set(["judge", "judge-audit", "note-auditor", "invariant-auditor", "teachback"]);
export const RULING_NAME_FALLBACK = [
  [/^judge[-_]?audit/i, "judge-audit"],
  [/^judge/i, "judge"],
  [/^note[-_]?audit/i, "note-auditor"],
  [/^inv(ariant)?[-_]?audit/i, "invariant-auditor"],
  [/^teachback/i, "teachback"],
];
/**
 * @param {{ promptText?: string|null, dir?: string|null }} a
 * @returns {{ type: string|null, source: "prompt"|"prompt-room-header"|"name"|null }}
 */
export function classifyRoom({ promptText = null, dir = null } = {}) {
  if (typeof promptText === "string") {
    const t = promptRoomType(promptText);
    if (t) return { type: t, source: "prompt" };
    const h = /^=== ROOM: (\S+)/m.exec(promptText);
    if (h) return { type: h[1], source: "prompt-room-header" };
  }
  if (typeof dir === "string") for (const [re, t] of RULING_NAME_FALLBACK) if (re.test(dir)) return { type: t, source: "name" };
  return { type: null, source: null };
}

/** classifyRoom for a room directory `<batchDir>/<dir>`: its PROMPT.md when present, then the name. */
export function roomTypeOfDir(batchDir, dir) {
  const pp = join(batchDir, dir, "PROMPT.md");
  return classifyRoom({ promptText: existsSync(pp) ? readFileSync(pp, "utf8") : null, dir }).type;
}
/** D-1148(1) (IA37-01): whether a `_packets.jsonl` `dir` names a room of `batchDir` -- one path segment (not "", "." or
 *  "..", no "/", so never absolute) that lstat finds as a directory, not a symlink, directly inside it, or that is absent
 *  (ENOENT: a dispatched room that left no directory is still owed its USAGE row and still counts as compiled, not ran). */
export function packetRoomDir(batchDir, dir) {
  if (typeof dir !== "string" || dir === "" || dir === "." || dir === ".." || dir.includes("/")) return false;
  try { return lstatSync(join(batchDir, dir)).isDirectory(); } catch (e) { return /** @type {{ code?: string }} */ (e).code === "ENOENT"; }
}

// ---- THE RAN-RULE (D-1147(1), I37-FX1-STAGES) ---------------------------------------------------------------
// close-run.mjs gates 0i (population count) and 0j, and batch-close.mjs, credit a stage through roomRanStage only.
/**
 * A REPORT.md's STATUS verdict: in its LAST `## STATUS` section (the heading at line start, outside a ``` / ~~~ fence;
 * CRLF-blind; a room continued by message appends one), the first line that is neither blank nor a `BRIEF-DELTA:` line,
 * when its first token is DONE, NOT DONE or BLOCKED (an optional `STATUS:` prefix); else null. I42-FX-CLOSE (D-1227(2)): the
 * word may be wrapped in `**` or a backtick pair and NOT DONE may hold more than one space or tab, read as the bare word. The unfilled template
 * `DONE | NOT DONE | BLOCKED`, a fence or a heading before any such line, and an empty section are null.
 * @param {unknown} text
 * @returns {"DONE"|"NOT DONE"|"BLOCKED"|null}
 */
export function reportStatusVerdict(text) {
  let fence = null;
  let open = false; // inside a STATUS section whose verdict line is not yet decided
  /** @type {"DONE"|"NOT DONE"|"BLOCKED"|null} */
  let verdict = null;
  for (const line of String(text ?? "").split(/\r?\n/)) {
    // I40-T-TOOLS (FAMILY-013 census of I40S-057's class): the one fence reader, fenceStep.
    const st = fenceStep(fence, line);
    fence = st.fence;
    if (st.fenceLine) { open = false; continue; }
    if (fence !== null) continue;
    if (/^##\s/.test(line)) {
      open = /^##\s+STATUS\s*$/.test(line);
      if (open) verdict = null;
      continue;
    }
    if (!open || line.trim() === "" || /^\s*BRIEF-DELTA:/.test(line)) continue;
    open = false;
    const m = /^\s*(?:STATUS:\s*)?(?:(\*\*|`)(NOT[ \t]+DONE|DONE|BLOCKED)\1|(NOT[ \t]+DONE|DONE|BLOCKED))(?![\w-])(.*)$/.exec(line);
    if (m !== null && !/^\s*\|/.test(m[4])) verdict = /** @type {"DONE"|"NOT DONE"|"BLOCKED"} */ ((m[2] ?? m[3]).replace(/[ \t]+/, " "));
  }
  return verdict;
}
const isFileAt = (p) => { try { return statSync(p).isFile(); } catch { return false; } };
/**
 * Whether the room in `<batchDir>/<dir>` ran its stage: its PROMPT.md is a file whose promptRoomType (TAIL `- ROOM TYPE:`,
 * else `## ROOM CONTRACT (org/rooms/<x>.md)`) names a type -- never the directory name, never a _packets.jsonl row -- and
 * its REPORT.md is a file whose reportStatusVerdict is DONE. `why` names the first unmet half.
 * @param {string} batchDir
 * @param {string} dir
 * @returns {{ ran: boolean, type: string|null, personaIds: string[], verdict: string|null, why: string }}
 */
export function roomRanStage(batchDir, dir) {
  const pp = join(batchDir, dir, "PROMPT.md");
  const rp = join(batchDir, dir, "REPORT.md");
  const promptText = isFileAt(pp) ? readFileSync(pp, "utf8") : null;
  const type = promptText === null ? null : promptRoomType(promptText);
  const personaIds = promptText === null ? [] : promptPersonaIds(promptText);
  if (type === null) return { ran: false, type, personaIds, verdict: null, why: promptText === null ? "no compiled PROMPT.md" : "its PROMPT.md names no room type" };
  if (!isFileAt(rp)) return { ran: false, type, personaIds, verdict: null, why: "no REPORT.md" };
  const verdict = reportStatusVerdict(readFileSync(rp, "utf8"));
  if (verdict !== "DONE") return { ran: false, type, personaIds, verdict, why: verdict === null ? "its REPORT.md has no STATUS line" : `its REPORT.md says STATUS ${verdict}` };
  return { ran: true, type, personaIds, verdict, why: "ran" };
}

// ---- THE STAGE TABLE (D-1146(4), I37-STAGES) --------------------------------------------------------------
// batch-close.mjs's stage presence and close-run.mjs gate 0j both read this table and readStageWaivers, one each.
/** One row per batch stage: `types` = the room types that are it (lower case); `persona` = the `## PERSONA <id>` the room
 *  must also carry; `from37` = owed only where bindsFromI37 binds; `withRed` = owed only when a red room ran; `gate0j` =
 *  close-run gate 0j requires it; `from62` = gate 0j owes it only where bindsFromI62 binds (batch-close's stageRequired
 *  ignores it). Row order is the print order. */
export const STAGE_TABLE = Object.freeze([
  { stage: "scout", types: ["scout"], from62: true, gate0j: true },
  { stage: "builders", types: ["blue-builder"] },
  { stage: "test-engineer", types: ["test-engineer"], from37: true, gate0j: true },
  { stage: "regression", types: ["regression-attacker", "regression"], from62: true, gate0j: true },
  { stage: "red", types: ["red-attacker", "red"], from37: true, gate0j: true },
  { stage: "blue-response", types: ["blue-builder"], persona: "M07", from37: true, withRed: true, gate0j: true },
  { stage: "purple", types: ["judge"], persona: "M08", from37: true, withRed: true, gate0j: true },
  { stage: "invariant-auditor", types: ["invariant-auditor"], gate0j: true },
  { stage: "judge", types: ["judge"] },
  { stage: "judge-audit", types: ["judge-audit"] },
  { stage: "note-audit", types: ["note-auditor"] },
  { stage: "teachback", types: ["teachback"] },
].map((s) => Object.freeze(s)));
/** A waiver may name a stage by its pre-I37 name: batch-close's `attacker` stage is `regression` (`--waive attacker=<why>`). */
export const STAGE_ALIASES = Object.freeze({ attacker: "regression" });
/**
 * The stage a room is: a row whose `persona` the room carries wins over a row with none (blue-builder + M07 is
 * blue-response, judge + M08 is purple), else the row for its type alone, else null.
 * @param {unknown} type  room type (case-blind)
 * @param {unknown} personaIds  promptPersonaIds of its PROMPT.md
 * @returns {string|null}
 */
export function stageOfRoom(type, personaIds) {
  const t = String(type ?? "").toLowerCase();
  const ids = Array.isArray(personaIds) ? personaIds : [];
  const hit = STAGE_TABLE.find((s) => s.persona && s.types.includes(t) && ids.includes(s.persona))
    ?? STAGE_TABLE.find((s) => !s.persona && s.types.includes(t));
  return hit ? hit.stage : null;
}
/**
 * Whether close-run gate 0j (and the compliance re-derive) owes a STAGE_TABLE row: a `gate0j` row, a `withRed` one only when
 * `redRan`, a `from62` one only where bindsFromI62(name).
 * @param {{ gate0j?: boolean, withRed?: boolean, from62?: boolean }} s
 * @param {{ name: unknown, redRan: unknown }} o
 * @returns {boolean}
 */
export function gate0jOwed(s, { name, redRan }) {
  return s.gate0j === true && (!s.withRed || redRan === true) && (!s.from62 || bindsFromI62(name));
}
/** The TAIL's `- compiled <ISO> by` time of a compiled PROMPT.md (the last such line), as epoch ms, or null. */
export function promptCompiledAt(text) {
  const m = [...String(text ?? "").matchAll(/^- compiled (\d{4}-\d\d-\d\dT\d\d:\d\d:\d\d(?:\.\d+)?Z) by /gm)].pop();
  if (m === undefined) return null;
  const t = Date.parse(m[1]);
  return Number.isFinite(t) ? t : null;
}
/** Every tag on a compiled PROMPT.md's `scope tags: a, b` lines (line start), trimmed and lower-cased. */
export function promptScopeTags(text) {
  const tags = [];
  for (const m of String(text ?? "").matchAll(/^scope tags:[ \t]*(.*)$/gm)) for (const t of m[1].split(",")) if (t.trim() !== "") tags.push(t.trim().toLowerCase());
  return tags;
}
/**
 * The stage each named room dir is (stageOfRoom over roomRanStage's type and personas), with `ran` from roomRanStage.
 * D-1523: where bindsFromI62(name), a room of the `regression` stage is that stage only when its PROMPT.md was compiled
 * strictly after the batch's first `builders`-stage room (promptCompiledAt; a builder that never ran counts, a room with
 * no compiled line does not) and its scope tags carry no `brief-attack`; otherwise its stage is null, so a W0 alone
 * credits no regression stage.
 * @param {string} batchDir
 * @param {string[]} dirs
 * @param {unknown} name  the batch's run-dir basename
 * @returns {Map<string, { ran: boolean, stage: string|null, rr: ReturnType<typeof roomRanStage> }>}
 */
export function stagesOfRooms(batchDir, dirs, name) {
  const out = new Map();
  const at = new Map();
  for (const d of dirs) {
    const rr = roomRanStage(batchDir, d);
    const pp = join(batchDir, d, "PROMPT.md");
    const text = isFileAt(pp) ? readFileSync(pp, "utf8") : "";
    out.set(d, { ran: rr.ran, stage: stageOfRoom(rr.type, rr.personaIds), rr });
    at.set(d, { at: promptCompiledAt(text), briefAttack: promptScopeTags(text).includes("brief-attack") });
  }
  if (!bindsFromI62(name)) return out;
  let firstBuilder = null;
  for (const [d, r] of out) {
    const t = at.get(d).at;
    if (r.stage === "builders" && t !== null && (firstBuilder === null || t < firstBuilder)) firstBuilder = t;
  }
  for (const [d, r] of out) {
    if (r.stage !== "regression") continue;
    const { at: t, briefAttack } = at.get(d);
    if (briefAttack || t === null || firstBuilder === null || !(t > firstBuilder)) r.stage = null;
  }
  return out;
}

export function stageRequired(s, { binds, redRan }) {
  return (!s.from37 || binds === true) && (!s.withRed || redRan === true);
}

/** The batch-root file a stage waiver lives in: one `- <stage>: <reason>` line per waived stage. */
export const STAGE_WAIVERS_BASENAME = "STAGE-WAIVERS.md";
/**
 * One waiver, from STAGE-WAIVERS.md or batch-close's `--waive`: the canonical stage, or the fault that refuses it (a name
 * that is no STAGE_TABLE stage or alias, or a reason statesClause refuses).
 * @param {unknown} stage
 * @param {unknown} reason
 * @returns {{ stage: string } | { fault: string }}
 */
export function checkStageWaiver(stage, reason) {
  const n = String(stage ?? "").trim().toLowerCase();
  const canon = Object.hasOwn(STAGE_ALIASES, n) ? STAGE_ALIASES[n] : n;
  if (!STAGE_TABLE.some((s) => s.stage === canon)) return { fault: `waives ${JSON.stringify(n)}, which is no stage (${STAGE_TABLE.map((s) => s.stage).join(", ")})` };
  if (!statesClause(reason)) return { fault: `waives ${canon} with the reason ${JSON.stringify(String(reason ?? ""))}, which is empty, a template or a placeholder word, not a reason` };
  // I42-FX-CLOSE (D-1227(3), VAL-09, RG-05): a waiver reason, and only it, also needs >= 40 characters and >= 4 distinct
  // words of >= 3 letters outside the function, placeholder, junk and filler lists, and no word a 1-4 character unit x3+.
  const c = decodeEntities(String(reason)).replace(INVISIBLE, "").trim();
  const words = c.replace(HTML_TAG, " ").replace(/(?<=[A-Za-z])[-_]+(?=[A-Za-z])/g, " ").split(/\s+/).map((w) => w.replace(/^[^A-Za-z0-9]+|[^A-Za-z0-9]+$/g, "")).filter((w) => w !== "");
  const stated = new Set(words.filter((w) => (w.match(/[A-Za-z]/g) ?? []).length >= 3 && ![FUNCTION_WORD, PLACEHOLDER_WORD, CLAUSE_PLACEHOLDER, JUNK_WORD, CLAUSE_FILLER].some((re) => re.test(w))).map((w) => w.toLowerCase()));
  const repeated = words.find((w) => /^(.{1,4})\1{2,}$/i.test(w));
  if (c.length < 40 || stated.size < 4 || repeated !== undefined) return { fault: `waives ${canon} with the reason ${JSON.stringify(String(reason))}, which ${repeated !== undefined ? `repeats ${JSON.stringify(repeated)}` : `is ${c.length} characters with ${stated.size} distinct stated word(s)`}; a waiver reason needs at least 40 characters and 4 distinct words of three or more letters (D-1227(3))` };
  return { stage: canon };
}
const STAGE_WAIVER_LINE = /^-[ \t]+([A-Za-z][A-Za-z-]*)[ \t]*:[ \t]*(.*)$/;
/**
 * `<batchDir>/STAGE-WAIVERS.md` read: every `- <stage>: <reason>` line is a waiver or a fault (checkStageWaiver; a second
 * line for one stage is a fault too). Other lines are prose. Text inside `<!-- -->` comments and lines inside ``` / ~~~
 * fences are not read (D-1147(1)); LF or CRLF. No file: no waivers, no faults. A path that is not a file, or that
 * cannot be read, is a fault.
 * @param {string} batchDir
 * @returns {{ waivers: Map<string, string>, faults: string[] }}
 */
export function readStageWaivers(batchDir) {
  const waivers = new Map();
  const faults = [];
  const f = join(batchDir, STAGE_WAIVERS_BASENAME);
  if (!existsSync(f)) return { waivers, faults };
  let text;
  try {
    if (!statSync(f).isFile()) {
      faults.push(`${STAGE_WAIVERS_BASENAME} is not a file (a directory or other entry is at that path); it must be a file of \`- <stage>: <reason>\` lines`);
      return { waivers, faults };
    }
    text = readFileSync(f, "utf8");
  } catch (e) {
    faults.push(`${STAGE_WAIVERS_BASENAME} cannot be read (${/** @type {{ code?: string }} */ (e).code ?? String(e)})`);
    return { waivers, faults };
  }
  let fence = null;
  let comment = false;
  text.split(/\r?\n/).forEach((raw, i) => {
    if (!comment) {
      // I40-T-TOOLS (FAMILY-013 census of I40S-057's class): the one fence reader, fenceStep.
      const st = fenceStep(fence, raw);
      fence = st.fence;
      if (st.fenceLine || fence !== null) return;
    }
    let line = "";
    let rest = raw;
    while (rest !== "") {
      if (comment) {
        const end = rest.indexOf("-->");
        if (end < 0) break;
        comment = false;
        rest = rest.slice(end + 3);
      } else {
        const open = rest.indexOf("<!--");
        if (open < 0) { line += rest; break; }
        line += rest.slice(0, open);
        comment = true;
        rest = rest.slice(open + 4);
      }
    }
    const m = STAGE_WAIVER_LINE.exec(line);
    if (!m) return;
    const r = checkStageWaiver(m[1], m[2]);
    if ("fault" in r) faults.push(`${STAGE_WAIVERS_BASENAME}:${i + 1} ${r.fault}`);
    else if (waivers.has(r.stage)) faults.push(`${STAGE_WAIVERS_BASENAME}:${i + 1} waives ${r.stage} a second time; a stage is waived once`);
    else waivers.set(r.stage, m[2].trim());
  });
  return { waivers, faults };
}

// ---- THE RULING JOIN (I16-W03J, QUEUE #874 / #493a) ------------------------------------------------------
// close-run.mjs gate 0e counted a population row "ruled" when ANY judge line contained its id and a verdict
// word, so a line about another row ruled it (#493a), while scripts/judge-coverage.mjs joined `rules_on` keys
// -- two numbers for one batch (#874). This is the one join. A population row is {room, line, id}; its keys
// are `room:line`, `room/id` and, only when no other row shares the id, the bare id (case-blind). A judge
// record RULES a key only in these forms:
//  - FINDINGS.jsonl: a row whose OWN `verdict` field (else `status`) is a verdict word, keyed by its
//    `rules_on` (each value WHOLLY one documented key -- judge-coverage.mjs's forms, D-0872(2)) and its own id;
//  - REPORT.md / VERDICTS.md: a table row whose FIRST cell is the key and another cell opens with a verdict
//    word, or a line that OPENS (after a list marker) with the key followed directly by the verdict word
//    (`- X-1 — CONFIRMED`, `X-1: verdict REJECTED`). A line that merely mentions an id rules nothing.
export const VERDICT_WORDS = ["CONFIRMED", "REJECTED", "NEEDS_REPRODUCTION"];
const ONE_RULING_KEY = /^(?:[A-Za-z0-9][A-Za-z0-9_.-]*[:/][A-Za-z0-9_.#-]+|(?:[^\s()]+\/)?[^\s()/]+\/FINDINGS\.jsonl:\d+)$/i;
const normRulingKey = (t) => {
  const x = t.replace(/[.,:;]+$/, "");
  const path = /(?:^|\/)([^/]+)\/FINDINGS\.jsonl:(\d+)$/i.exec(x);
  return (path ? `${path[1]}:${path[2]}` : x).toLowerCase();
};
/** The key a documented-form `rules_on` value names (`<room>:<line>`, `<room>/<id>`, the FINDINGS.jsonl path
 *  form, or `<id> (<key>)`), lower-cased; null for anything else (prose rules on nothing). */
// I34-TOOLS-B (I32 C15, FAMILY-027): the batch-qualified form `<run>-batch/<room>/<id>` (or `<run>-batch/<room>:<line>`) names
// the same row as `<room>/<id>`; a leading run-dir segment ending `-batch` is stripped before the forms above are read.
const BATCH_QUALIFIER = /^[A-Za-z0-9][A-Za-z0-9_.-]*-batch\/(?=[^/\s]+[:/][^/\s]+$)/i;
export function rulingKeyOf(v) {
  if (typeof v !== "string" || /^\s*none:/i.test(v)) return null;
  const x = v.trim().replace(/[.,:;]+$/, "").replace(BATCH_QUALIFIER, "");
  if (ONE_RULING_KEY.test(x)) return normRulingKey(x);
  const c = /^[^\s()]+\s+\(([^\s()]+)\)$/.exec(x);
  // I34-FX3-SEAM (FX3T-F3): the parenthesised key may be batch-qualified too, `<label> (<run>-batch/<room>/<id>)`.
  const inner = c ? c[1].replace(/[.,:;]+$/, "").replace(BATCH_QUALIFIER, "") : "";
  return c && ONE_RULING_KEY.test(inner) ? normRulingKey(inner) : null;
}
const isVerdictWord = (v) => typeof v === "string" && VERDICT_WORDS.includes(v.trim().toUpperCase());
const stripMarks = (s) => s.trim().replace(/^[`*_[(]+|[`*_\]),.;:]+$/g, "");
const LINE_RULING = /^\s*(?:[-*+]\s+|\d+[.)]\s+)?[`*_]*([^\s`*_|]+?)[`*_]*\s*(?:[—–:=|]|-{1,2}|\s)\s*[[(]?\s*(?:verdict\s*[:=]?\s*)?[`*_]*(CONFIRMED|REJECTED|NEEDS_REPRODUCTION)\b(?!-)/;
/** Every key the judge dirs RULE, lower-cased. `judgeDirs` are absolute room directories. */
// I44-TOOLS (C6 / JA-20, D-1273): two judge key shapes the join could not read. Both look a key's room up as a sibling
// dir of the judge dir and read that room's FINDINGS.jsonl row ids (`<room>/<x>` ids also yield `<x>`).
//  - Re-verdict (shape B): a `<room>/<x>` key whose `<x>` is no row id of `<room>` but is one after ONE trailing
//    `-rv<digits>` is stripped (`I30-FX9/I30-FX9-01-RV1` -> `i30-fx9/i30-fx9-01`); a key that matches as written is kept.
//  - Outside record (shape A): a FINDINGS row whose id `<room>/<x>` names no row of `<room>` and whose `rules_on` names
//    `<room>:<file.ext>:<line>` (a file other than FINDINGS.jsonl) rules that record, not a population row: the key added
//    is `<room>:<file.ext>:<line>`, which matches no population key and is never a broken join.
//    I44-FX-VOCAB (NEW-3, D-1278): only when `<file.ext>` is a regular file of that room's dir, name compared exactly
//    (isRoomRecord); a cite of a source file or of a missing record leaves the id keyed as written, a broken join.
export function judgeRulingKeys(judgeDirs) {
  const keys = new Set();
  const roomIds = new Map();
  const hasRow = (d, room, x) => {
    const ck = `${join(d, "..")}\0${room}`;
    if (!roomIds.has(ck)) {
      let ids = null;
      try {
        const name = readdirSync(join(d, "..")).find((n) => n.toLowerCase() === room);
        const fp = name ? join(d, "..", name, "FINDINGS.jsonl") : "";
        if (fp && existsSync(fp)) {
          ids = new Set();
          for (const l of readFileSync(fp, "utf8").split("\n")) {
            let o = null;
            try { o = l.trim() ? JSON.parse(l) : null; } catch { o = null; }
            if (!o || typeof o !== "object" || o.id == null) continue;
            const id = String(o.id).trim().toLowerCase();
            ids.add(id);
            if (id.startsWith(`${room}/`)) ids.add(id.slice(room.length + 1));
          }
        }
      } catch { ids = null; }
      roomIds.set(ck, ids);
    }
    return roomIds.get(ck)?.has(x) ?? false;
  };
  const isRoomRecord = (d, room, file) => {
    try {
      // no dir for `room` -> `name` is undefined, join throws, and the catch answers false
      const name = readdirSync(join(d, "..")).find((n) => n.toLowerCase() === room);
      return readdirSync(join(d, "..", name), { withFileTypes: true }).some((e) => e.isFile() && e.name === file);
    } catch { return false; }
  };
  const RE_VERDICT = /-rv\d+$/;
  const ROOM_ID_KEY = /^([a-z0-9][a-z0-9_.-]*)\/(.+)$/;
  const add = (d, k) => {
    const m = ROOM_ID_KEY.exec(k);
    if (m && !hasRow(d, m[1], m[2]) && hasRow(d, m[1], m[2].replace(RE_VERDICT, ""))) keys.add(`${m[1]}/${m[2].replace(RE_VERDICT, "")}`);
    else keys.add(k);
  };

  for (const d of judgeDirs) {
    const fp = join(d, "FINDINGS.jsonl");
    if (existsSync(fp)) for (const l of readFileSync(fp, "utf8").split("\n")) {
      let o = null;
      try { o = l.trim() ? JSON.parse(l) : null; } catch { o = null; }
      if (!o || typeof o !== "object") continue;
      if (!isVerdictWord(o.verdict ?? (isVerdictWord(o.status) ? o.status : null))) continue;
      const on = typeof o.rules_on === "string" ? [o.rules_on] : Array.isArray(o.rules_on) ? o.rules_on : [];
      for (const v of on) { const k = rulingKeyOf(v); if (k) add(d, k); }
      if (o.id != null && String(o.id).trim()) {
        const id = String(o.id).trim().toLowerCase();
        const im = ROOM_ID_KEY.exec(id);
        const outside = im ? on.map((v) => (typeof v === "string" ? OUTSIDE_RECORD.exec(v.trim()) : null))
          .filter((m) => m && m[1].toLowerCase() === im[1] && m[2].toLowerCase() !== "findings.jsonl" && isRoomRecord(d, m[1].toLowerCase(), m[2])) : [];
        if (outside.length > 0 && !hasRow(d, im[1], im[2])) for (const m of outside) keys.add(`${im[1]}:${m[2].toLowerCase()}:${m[3]}`);
        else add(d, id);
      }
    }
    for (const b of ["REPORT.md", "VERDICTS.md"]) {
      const p = join(d, b);
      if (!existsSync(p)) continue;
      for (const l of readFileSync(p, "utf8").split("\n")) {

          continue;
        }
        const m = LINE_RULING.exec(l);
        if (m) add(d, stripMarks(m[1]).toLowerCase());
      }
    }
  }
  return keys;
}
/** The population rows the judge dirs rule. `pop` rows carry {room, line, id}. */

#!/usr/bin/env node
// scripts/lib/harness-placement.mjs — S-076 GATE D2 (QUEUE #257): shared placement engine for
// `gate.mjs --harnesses`, extracted verbatim from that mode so a second caller (a tooling test,
// or a future room) can run the SAME logic against a file that isn't in the pinned population
// without duplicating it. Nothing here reads the manifest, the retired list or the homes hint
// file — those stay in gate.mjs, which owns the pinned/retired population and its own `refuse`.
//
// FAMILY-007 (RUN-0015 F-4): `scanFrames` walks the five frames a JS file has — line comment,
// block comment, template, string, regex literal — in one ordered pass, so a quoted or
// templated "import" never counts as this file's own. `code` keeps "…"/'…' verbatim as a
// placeholder ("#<i>") with the real text in `strs[i]` (a quoted string is DATA, QUEUE #197);
// templates and regexes are blanked. An unterminated template leaves the file `unbalanced`.
import { existsSync, readFileSync } from "node:fs";
import { basename, dirname, resolve as resolvePath } from "node:path";
import ts from "typescript";
import { parse as parseTsFile } from "./ts-scan.mjs";

export const FRAME = /\/\/[^\n]*|\/\*[\s\S]*?\*\/|`(?:[^`\\]|\\[\s\S])*`|"(?:[^"\\\n]|\\.)*"|'(?:[^'\\\n]|\\.)*'|\/(?:\\[\s\S]|\[(?:\\[\s\S]|[^\]\n])*\]|[^\\/\n[])+\/[a-z]*/g;
const DIV_BEFORE = /[A-Za-z0-9_$)\]]$/; // a `/` after a value is division, not a regex literal

export function scanFrames(src) {
  let code = "";
  let bare = "";
  const strs = [];
  let last = 0;
  FRAME.lastIndex = 0;
  for (let m = FRAME.exec(src); m; m = FRAME.exec(src)) {
    const t = m[0][0];
    if (t === "/" && m[0][1] !== "/" && m[0][1] !== "*" && DIV_BEFORE.test(src.slice(0, m.index).replace(/\s+$/, ""))) {
      FRAME.lastIndex = m.index + 1; // division: not a frame, resume one char in
      continue;
    }
    const pre = src.slice(last, m.index);
    code += pre; bare += pre; last = m.index + m[0].length;
    if (t === '"' || t === "'") { code += `"#${strs.length}"`; strs.push(m[0].slice(1, -1)); bare += t + t; }
    else if (t === "`") { code += "``"; bare += ""; }        // a CLOSED template balances
    else if (t === "/" && m[0][1] !== "/" && m[0][1] !== "*") { code += "/ /"; bare += ""; }
    else { const nl = m[0].replace(/[^\n]/g, ""); code += nl; bare += nl; } // comments keep newlines
  }
  const rest = src.slice(last);
  return { code: code + rest, strs, unbalanced: (bare + rest).includes("`") };
}

// Every `import … from "<spec>"` statement of a file (any specifier), read over scanFrames' output.
export function importSpecsOf(src) {
  const { code, strs, unbalanced } = scanFrames(src);
  const stmts = [...code.matchAll(/(?<![\w$.])import\s+([^"';]*?)\s*from\s+"#(\d+)"/g)].map((m) => ({ clause: m[1], spec: strs[Number(m[2])] }));
  // S-075 J8 / #303 (I16-W03G hand-off): a side-effect `import "x"` and a dynamic `import("x")` name a module too; they bind
  // no names (clause ""), so placeOne checks only that the spec resolves at the candidate depth.
  const bare = [...code.matchAll(/(?<![\w$.])import\s*(?:\(\s*)?"#(\d+)"/g)].map((m) => ({ clause: "", spec: strs[Number(m[1])] }));
  return { unbalanced, stmts: [...stmts, ...bare] };
}

// Parses ONE import clause's text into the names an importer binds.
export function parseImportClause(clause) {
  const c = clause.replace(/^\s*type\s+/, "");
  const names = [];
  const brace = /\{([^}]*)\}/.exec(c);
  const head = brace ? c.slice(0, brace.index) : c;
  if (!/\*\s+as\s+\S+/.test(head) && head.replace(/,/g, "").trim()) names.push("default");
  if (brace) {
    for (const part of brace[1].split(",")) {
      const p = part.trim().replace(/^type\s+/, "");
      if (!p) continue;
      names.push(p.split(/\s+as\s+/)[0].trim());
    }
  }
  return names;
}

export function resolves(dir, spec, siblingDir) {
  const bases = [resolvePath(dir, spec)];
  if (spec.startsWith("./") && siblingDir) bases.push(resolvePath(siblingDir, spec));
  return bases.some((b) => [b, b.replace(/\.js$/, ".ts"), b.replace(/\.js$/, ".tsx"), `${b}.ts`].some(existsSync));
}

export function candidatePath(dir, spec, siblingDir) {
  const bases = [resolvePath(dir, spec)];
  if (spec.startsWith("./") && siblingDir) bases.push(resolvePath(siblingDir, spec));
  for (const b of bases) {
    for (const cand of [b, b.replace(/\.js$/, ".ts"), b.replace(/\.js$/, ".tsx"), `${b}.ts`]) {
      if (existsSync(cand)) return cand;
    }
  }
  return null;
}

const packageEntryCache = new Map();
function packageEntryPath(spec) {
  const m = /^@dhwaja\/([^/]+)$/.exec(spec);
  if (!m) return null;
  if (packageEntryCache.has(spec)) return packageEntryCache.get(spec);
  let result = null;
  const pkgDir = `packages/${m[1]}`;
  const pkgJsonPath = `${pkgDir}/package.json`;
  if (existsSync(pkgJsonPath)) {
    try {
      const main = JSON.parse(readFileSync(pkgJsonPath, "utf8")).main;
      if (main && existsSync(`${pkgDir}/${main}`)) result = `${pkgDir}/${main}`;
    } catch { /* unparseable package.json -- no entry, falls through to null */ }
  }
  packageEntryCache.set(spec, result);
  return result;
}

const exportedNamesCache = new Map();
export function exportedNames(path, seen = new Set()) {
  if (exportedNamesCache.has(path)) return exportedNamesCache.get(path);
  const names = new Set();
  if (!seen.has(path)) {
    seen.add(path);
    try {
      const sf = parseTsFile(path);
      for (const stmt of sf.statements) {
        const flags = ts.getCombinedModifierFlags(stmt);
        const isExported = (flags & ts.ModifierFlags.Export) !== 0;
        const isDefault = (flags & ts.ModifierFlags.Default) !== 0;
        if (isDefault) names.add("default");
        if (isExported) {
          if ((ts.isFunctionDeclaration(stmt) || ts.isClassDeclaration(stmt) || ts.isInterfaceDeclaration(stmt) ||
               ts.isTypeAliasDeclaration(stmt) || ts.isEnumDeclaration(stmt)) && stmt.name) {
            names.add(stmt.name.text);
          } else if (ts.isVariableStatement(stmt)) {
            for (const d of stmt.declarationList.declarations) if (ts.isIdentifier(d.name)) names.add(d.name.text);
          }
        }
        if (ts.isExportAssignment(stmt)) names.add("default");
        if (ts.isExportDeclaration(stmt)) {
          const spec = stmt.moduleSpecifier && ts.isStringLiteral(stmt.moduleSpecifier) ? stmt.moduleSpecifier.text : null;
          if (stmt.exportClause && ts.isNamedExports(stmt.exportClause)) {
            for (const el of stmt.exportClause.elements) names.add(el.name.text);
          } else if (!stmt.exportClause && spec) {
            const resolved = spec.startsWith(".") ? candidatePath(dirname(path), spec, null) : packageEntryPath(spec);
            if (resolved) for (const n of exportedNames(resolved, seen)) names.add(n);
          }
        }
      }
    } catch { /* not a parseable TS/TSX file — exports empty, every named check below fails it */ }
  }
  exportedNamesCache.set(path, names);
  return names;
}

// SELF-CHECK (QUEUE #197): the scanner's boundary, asserted by gate.mjs before any placement.

}

// D2 (QUEUE #257): a harness may declare its own home as the FIRST LINE of the file,
// `// harness-home: <dir>`. It is only a candidate when that directory exists on disk — a
// declared home naming a dir that was renamed or never existed falls through to the ordinary
// candidate order below, never a silent no-op. It does NOT skip the export-scan: `placeOne`
// tries it exactly like any other candidate and still requires every import to resolve there.
export function declaredHome(src) {
  const first = src.split("\n", 1)[0] ?? "";
  const m = /^\/\/\s*harness-home:\s*(\S+)\s*$/.exec(first);
  if (!m) return null;
  return existsSync(m[1]) ? m[1] : null;
}

// Places ONE harness file: `f` is its on-disk path (used only for `dirname(f)` as "home" and
// the org/** in-place exclusion below), `src` is its already-read text, `dests` is the caller's
// enumerated candidate list (gate.mjs's `DESTS`), `homesHint` is the optional S-021 HOMES file
// object ({[f]: {dest}}). Candidate order: declared-home (D2), home, hinted (S-021), then dests.
// REG-D2-HOME: `declared` goes FIRST — a `./` import (or no relative import at all) always
// resolves at `home` too (via the `siblingDir` fallback passed to every candidate below), so
// trying `home` first meant a declared `// harness-home:` was never actually reached whenever
// the harness's own relative imports were all still siblings in its org/** dir.
// Returns `{ unplaced: true, reason }` or `{ dest, siblings, declaredHomeUsed }`.
export function placeOne(f, src, { homesHint = {}, dests = [] } = {}) {
  const { unbalanced, stmts } = importSpecsOf(src);
  if (unbalanced) return { unplaced: true, reason: "FAMILY-007: import scan ended inside an unterminated template literal — placement not attempted" };
  const importStmts = stmts.filter((imp) => imp.spec.startsWith("."));
  const bySpec = new Map();
  for (const { clause, spec } of importStmts) {
    const names = bySpec.get(spec) || new Set();
    for (const n of parseImportClause(clause)) names.add(n);
    bySpec.set(spec, names);
  }
  const imports = [...bySpec.entries()].map(([spec, names]) => ({ spec, names: [...names] }));
  const specs = imports.map((i) => i.spec);
  const home = dirname(f);
  const hinted = homesHint[f]?.dest;
  const declared = declaredHome(src);
  const candidateOrder = [declared, home, hinted, ...dests].filter(Boolean);
  const nf = (r) => r.split("(not found)").length - 1;
  const reasonsByDest = [];
  const dest = candidateOrder.find((d) => {
    const missing = [];
    for (const imp of imports) {
      const cand = candidatePath(d, imp.spec, home);
      if (!cand) { missing.push(`${imp.spec} (not found)`); continue; }
      const have = exportedNames(cand);
      const miss = imp.names.filter((n) => !have.has(n));
      if (miss.length) missing.push(`${imp.spec}: missing ${miss.join(", ")}`);
    }
    if (missing.length) reasonsByDest.push(`${d} — ${missing.join("; ")}`);
    return !missing.length;
  });
  if (!dest) return { unplaced: true, reason: `no depth's imports resolve by export; closest: ${[...reasonsByDest].sort((a, b) => nf(a) - nf(b) || a.length - b.length)[0]}` };
  // N1 (S-075 FIX): a candidate whose closest resolving depth is its OWN directory is only a
  // real placement outside org/** — vitest.config.ts excludes org/** unconditionally.
  if (dest === home && f.startsWith("org/")) return { unplaced: true, reason: "in place under org/** — excluded by vitest.config.ts:27, never collected" };
  const siblings = specs.filter((i) => i.startsWith("./") && !resolves(dest, i, null) && resolves(dest, i, home)).map((i) => basename(i).replace(/\.js$/, ".ts"));
  return { dest, siblings, declaredHomeUsed: !!declared && dest === declared };
}

#!/usr/bin/env node
// scripts/lib/harness-registry.mjs — S-076 GATE D1 (QUEUE #162): a small, independently
// testable library of `gate.mjs --harnesses`/`--shard` helpers. Originally just the
// crash-recovery registry below (a copy this leg places is recorded here BEFORE the first real
// copy, so a run killed by the ~170s call budget — not a thrown JS error, the in-process
// `finally` block never runs for that — still leaves a trail the NEXT run reads and cleans
// before placing anything of its own). S-081 #275/#271/#151 add three more pure/small helpers
// here rather than to harness-placement.mjs, whose own docblock explicitly keeps the manifest,
// the retired list and the homes hint OUT of that file; this one carries no such restriction and
// already exists to make gate.mjs's own internals testable without a real vitest run.
import { execFileSync } from "node:child_process";
import { closeSync, existsSync, mkdirSync, openSync, readFileSync, rmSync, writeFileSync, writeSync } from "node:fs";
import { dirname, relative, sep } from "node:path";
import { realOrTail } from "./real-or-tail.mjs";

// REG-S-REG-DELETE: a registry entry names a PLANNED copy, not a made one — a run killed
// between writeRegistry and its first real copyFileSync leaves a path nobody ever wrote to, and
// a builder can commit a real, tracked file there before the next run. `git ls-files
// --error-unmatch` is the same population source check-invariants.mjs and check-claims.mjs use;
// any error (untracked path, or a path outside a git repo entirely, e.g. a test fixture) reads
// as "not tracked", which is the safe default for a leftover gate copy.
const isGitTracked = (p) => {
  try {
    execFileSync("git", ["ls-files", "--error-unmatch", p], { stdio: ["ignore", "pipe", "pipe"] });
    return true;
  } catch {
    return false;
  }
};

// Removes every path a PREVIOUS run's registry still lists — but ONLY when it is safe to: git
// does not track it, AND (when the registry recorded one) its current bytes still equal the
// SOURCE sha256 taken at planning time. A tracked file, or an untracked one whose bytes no
// longer match what this run's plan would have copied, was never this run's own gate copy to
// remove — REG-S-REG-DELETE (a plan-only path a builder later committed a real module to).
// Best-effort otherwise: a path already gone, or one that can't be removed, doesn't stop the
// rest. Returns the list the registry named (for callers/tests that want to see what a stale
// registry named — NOT only what was actually removed), `[]` when there's no registry or it's
// unreadable (an unreadable registry has nothing to remove BY NAME — the run after it still
// overwrites the file with its own list, same as an absent one). Registry entries are either a
// plain path string (pre-REG-S-REG-DELETE shape: no sha256 to check, skip that guard) or
// `{ path, sha256 }` (current shape, written by writeRegistry via gate.mjs).
export function cleanupPreviousRegistry(registryPath) {
  if (!existsSync(registryPath)) return [];
  let prev = [];
  try {
    prev = JSON.parse(readFileSync(registryPath, "utf8")).placed ?? [];
  } catch {
    return [];
  }
  for (const entry of prev) {
    const p = typeof entry === "string" ? entry : entry.path;
    const source = typeof entry === "string" ? null : entry.source;
    try {
      if (isGitTracked(p)) continue; // never delete a file git tracks
      if (source != null && existsSync(p) && !(existsSync(source) && readFileSync(p).equals(readFileSync(source)))) continue; // not this run's planned copy
      rmSync(p);
    } catch { /* already gone, or unremovable -- not this run's copy to guarantee */ }
  }
  return prev;
}

// ---------------------------------------------------------------- QUEUE #275 (S-081): the lock
// `gate.mjs --harnesses` takes before doing any placement/copy work, so two runs of that leg
// never race on the same shared destination directory (REG-GATE-REG-ORDER's copyIn). A run
// killed by the ~170s call budget between acquire and its own release leaves the lock file
// behind on disk; STALE_MS is how long the NEXT run waits before treating that as abandoned
// rather than live, same shape as the crash-recovery registry above. Pure w.r.t. wall clock
// (nowMs passed in) so a test can simulate staleness without sleeping.
export const HARNESS_LOCK_STALE_MS = 15 * 60 * 1000;

// P2-G2 (S-081 REG phase 2): the old acquire was check-(existsSync)-then-write
// (writeFileSync unconditionally), a real race between two processes; and release was
// unconditional, so a REFUSED run (one that found someone else's live lock) deleted that live
// holder's lock on its way out — a third run could then acquire concurrently with the second.
// Fixed by (1) an atomic O_CREAT|O_EXCL create (`openSync(..., "wx")`, EEXIST on contention, no
// window between the existence check and the write) and (2) release taking the exact
// {pid, startedAtMs} THIS call wrote and refusing to remove a lock file that doesn't match it
// byte-for-byte — a lock this run never actually owns (someone else's, or none) is never deleted.
export function acquireHarnessLock(lockPath, { pid, nowMs, staleMs = HARNESS_LOCK_STALE_MS } = {}) {
  mkdirSync(dirname(lockPath), { recursive: true });
  const owner = { pid, startedAtMs: nowMs };
  const payload = `${JSON.stringify(owner, null, 1)}\n`;
  for (let attempt = 0; attempt < 2; attempt++) {
    let fd;
    try {
      fd = openSync(lockPath, "wx");
    } catch (e) {
      if (e?.code !== "EEXIST") throw e;
      let holder = null;
      try {
        holder = JSON.parse(readFileSync(lockPath, "utf8"));
      } catch {
        holder = null; // unreadable/corrupt lock file — treated as stale below, same as a missing one
      }
      const age = holder?.startedAtMs != null ? nowMs - holder.startedAtMs : Infinity;
      if (holder && Number.isFinite(age) && age >= 0 && age < staleMs) {
        return { ok: false, reason: `held by pid ${holder.pid}, started ${new Date(holder.startedAtMs).toISOString()}, ${Math.round(age / 1000)}s ago`, holder };
      }
      // stale or unreadable — a previous run was killed or crashed without releasing it; reclaim
      // and retry the atomic create once. A racing reclaim by another process just fails EEXIST
      // again on the retry, reported below rather than looping forever.
      try { rmSync(lockPath); } catch { /* raced away already — fine, retry create */ }
      continue;
    }
    writeSync(fd, payload);
    closeSync(fd);
    return { ok: true, owner };
  }
  return { ok: false, reason: "lock file contended on every retry" };
}

export function releaseHarnessLock(lockPath, owner) {
  if (!owner) return; // this run never actually acquired the lock — nothing of ours to remove
  let current = null;
  try {
    current = JSON.parse(readFileSync(lockPath, "utf8"));
  } catch {
    return; // gone, or unreadable — either way not safely ours to remove
  }
  if (current?.pid === owner.pid && current?.startedAtMs === owner.startedAtMs) {
    try { rmSync(lockPath); } catch { /* already gone */ }
  }
}

// scripts/lib/real-or-tail.mjs — I28-REALPATH R2 (QUEUE #972): the ONE "realpath, or the realpath of the
// nearest existing ancestor plus the missing tail" helper. Callers: scripts/seed-plant.mjs, scripts/gate.mjs,
// scripts/lib/harness-registry.mjs (checkPinnedTestCounts), scripts/lib/audit-store.mjs (auditRootInFindings),
// scripts/compile-room-prompt.mjs (the --out refusal), scripts/intake/room-write-audit.mjs (REPO and --audit-dir;
// I28-FX-INT landed I28-FX-TOOLING's hand-off, so no copy is left — tests/tooling/real-or-tail.test.ts's census is []).
// Strict: it walks up only past components that do not exist, and a realpath failure on an EXISTING ancestor throws,
// as does a path with no resolvable ancestor — it never returns a lexical spelling that a containment check would
// then compare against a realpath. I28-FX-TOOLING F2: a component that is a DANGLING symlink (lstat sees it,
// existsSync does not) resolves to its link target plus the tail, because a write through it lands there; a link
// chain longer than MAX_HOPS throws ELOOP. `fsi` exists so tests can drive the refusals.
import { existsSync, lstatSync, readlinkSync, realpathSync } from "node:fs";
import { basename, dirname, join, resolve } from "node:path";

const MAX_HOPS = 40;
export function realOrTail(p, fsi = {}, hops = 0) {
  const { existsSync: exists = existsSync, realpathSync: real = realpathSync, lstatSync: lstat = lstatSync, readlinkSync: readlink = readlinkSync } = fsi;
  // ENOTDIR (a path through a regular file) or EACCES on lstat: not a link we can see — keep walking, as before.
  const isLink = (h) => { try { return lstat(h, { throwIfNoEntry: false })?.isSymbolicLink() === true; } catch { return false; } };
  let head = resolve(p);
  const tail = [];
  while (!exists(head) && !isLink(head) && dirname(head) !== head) {
    tail.unshift(basename(head));
    head = dirname(head);
  }
  if (!exists(head) && isLink(head)) {
    if (hops >= MAX_HOPS) throw Object.assign(new Error(`ELOOP: too many symbolic links resolving ${p}`), { code: "ELOOP" });
    return realOrTail(join(resolve(real(dirname(head)), readlink(head)), ...tail), fsi, hops + 1);
  }
  return join(real(head), ...tail);
}

// A grant token is tree-relative, has no glob, no escape and no whitespace. `.` / `..` / `/` are
// refused explicitly: a grant that normalises to the tree root would make `isAllowed` true for
// EVERY path, which is the "guard that cannot fail" shape (PROMPT-CORE, THE MUTATION MUST BE THE

export function isGrantToken(p) {
  const t = String(p ?? "").trim().replace(/\/+$/, "");
  if (!t || t === "." || t === "..") return false;
  if (/\s|\*|\?|\[|^\//.test(t)) return false;
  if (/(^|\/)\.\.(\/|$)/.test(t)) return false;
  if (!/^[A-Za-z0-9_.][A-Za-z0-9_./@-]*$/.test(t)) return false;
  return t.includes("/") || /\.[A-Za-z0-9]+$/.test(t);
}

function tokensIn(body) {

 * OWNS section starts and ends, used by parseOwns below and by the compiler's probe-prefix rule

 * line, then every (B) section, then every (C)/(D) label. Each is `{ form, lines, body }`: `form` is "files",

export function ownsSections(promptText) {
  const text = String(promptText || "");
  const out = [];
  const add = (form, m, body) => out.push({ form, lines: m[0].split("\n"), body });

  // so the two readers cannot drift on the two spellings they share. The section is the label line and the bullet
  // and blank lines after it, ending before the first other line; the grant is the label's inline tail when it has one.
  for (const m of text.matchAll(/^[ \t]*[-*>]*[ \t]*\*{0,2}Owns\*{0,2}(?:[ \t]*\([^)\n]*\))?[ \t]*:\*{0,2}[ \t]*([^\n]*)\n?([\s\S]*?)(?=\n(?![ \t]*(?:[-*+][ \t]|$))|(?![\s\S]))/gmi)) {
    add("label", m, m[1].trim() ? m[1] : m[2]);
  }
  return out;
}

export function parseOwns(promptText) {
  const text = String(promptText || "");
  // (A)-(D): the sections ownsSections finds, each read from its grant body.
  const bodies = ownsSections(text).map((s) => s.body);
  // (E) LAST RESORT -- THE COMPILER'S OWN RENDERING OF THE SUB-RUN'S FENCED `files:` KEY:
  // `**Files in this sub-run** ...:` plus its bullet run (compile-room-prompt.mjs:planFacts).
  // This is NOT "prose a scout happened to write": it is emitted mechanically from the one
  // machine-readable ownership list the PLAN carries, which is the only reason it is admitted.
  // It exists so prompts ALREADY ON DISK resolve without being recompiled -- ORG-02-R1/R2/R3
  // reached the write audit with no `FILES (OWNS` line at all, because `extractSubrun` dropped
  // everything below the yaml fence and those three sub-runs put their OWNS line there.
  // TWO GATES, and both are load-bearing:
  //   (1) LAST RESORT. Consulted only when (A)-(D) yielded no token, so every prompt that
  //       resolves today resolves BYTE-IDENTICALLY. MEASURED 2026-09-22 over the four prompts
  //       carrying both shapes (ORG-02-R4/R5/R6 and the W3-15-R1 control): the token sets are
  //       equal in all four, so the fallback is equivalent where it can be cross-checked and
  //       never wider.
  //   (2) WRITE-GRANT MARKER. Accepted only when the prompt carries the compiler's own write
  //       grant sentence, which `renderSubrunScope` emits ONLY for a persona whose contract
  //       frontmatter grants `Edit`. A read-only room's sub-run list is a READING list.
  // WHY NOT THE PLAN'S PROSE `FILES (OWNS): ...` LINE, which is the obvious repair and is wrong:
  // that line is prose and following it is unsafe in BOTH directions. `SCOUT/PLAN.md:116` reads
  // ``FILES (OWNS): scripts/gate.mjs (read `scripts/check-invariants.mjs` for context; do not
  // edit it).`` -- a parser widened to the continuation line would grant the one file the
  // sentence forbids. A fenced `files:` key has no such failure mode, and a grant recovered from
  // prose is the defect, not the parser that fails to recover it.
  if (!bodies.some((b) => tokensIn(b).length) && WRITE_GRANT_MARKER.test(text)) {
    for (const m of text.matchAll(/^[ \t]*\*\*Files in this sub-run\*\*[^\n]*:[ \t]*\n((?:[ \t]*[-*+][ \t]+[^\n]*\n?)+)/gm)) bodies.push(m[1]);
  }
  const out = [];
  for (const b of bodies) out.push(...tokensIn(b));
  return [...new Set(out)];
}

/** The compiled PROMPT.md for <run>/<room>, searched across every tree in play (repo, build tree,
 *  audited tree). Defect (1) above is exactly this search not happening. */
export function findPrompt(trees, runDir, roomDir) {
  for (const t of trees) {
    if (!t || !runDir || !roomDir) continue;
    const p = join(t, "org/07-findings", runDir, roomDir, "PROMPT.md");
    if (existsSync(p)) return p;
  }
  return null;
}

/** The packet log's own record of the grant -- `owns_paths`, written by compile-room-prompt.mjs
 *  :1511. Independent of the prompt text, so the two disagree loudly rather than silently.
 *  I9-FX4 (JN-04, sibling of RG615-01 in compile-room-prompt.mjs:~1491): the LATEST row for the room is the grant, even when
 *  its `owns_paths` is empty (an empty grant is a grant, never a cue to fall back to an older, stale one). A deferred row
 *  withholds `owns_paths`, so it is read from its `withheld_to` record (same run, dir and `at`, resolved against the tree
 *  that holds the log, then the others); a deferred row with no such record is an empty grant. Across trees the row with the
 *  greatest `at` wins (ties: first tree), so a stale repo log cannot shadow a newer build-tree compile. */
export function ownsFromPackets(trees, runDir, roomDir) {
  let best = null;
  for (const t of trees) {
    if (!t || !runDir || !roomDir) continue;
    const f = join(t, "org/07-findings", runDir, "_packets.jsonl");
    if (!existsSync(f)) continue;
    let latest = null;
    for (const line of readFileSync(f, "utf8").split("\n")) {
      if (!line.trim()) continue;
      try {
        const row = JSON.parse(line);
        // a {kind:"resume"} row (compile-room-prompt.mjs:207, QUEUE #554) carries no grant: it is not the room's latest grant row
        if (row && row.dir === roomDir && row.kind !== "resume") latest = row;
      } catch { /* a malformed line is not a grant */ }
    }
    if (!latest) continue;
    const at = typeof latest.at === "string" ? latest.at : "";
    if (!best || at > best.at) best = { row: latest, at, tree: t };
  }
  if (!best) return [];
  const { row } = best;
  if (Array.isArray(row.owns_paths)) return row.owns_paths.filter(isGrantToken);
  if (!(row.deferred && Array.isArray(row.withheld) && row.withheld.includes("owns_paths") && typeof row.withheld_to === "string")) return [];
  for (const t of [best.tree, ...[...trees].filter((x) => x && x !== best.tree)]) {
    const stash = join(t, row.withheld_to);
    if (!existsSync(stash)) continue;
    for (const line of readFileSync(stash, "utf8").split("\n")) {
      let full;
      try { full = line.trim() ? JSON.parse(line) : null; } catch { full = null; }
      if (full && full.run === row.run && full.dir === row.dir && full.at === row.at && Array.isArray(full.owns_paths)) return full.owns_paths.filter(isGrantToken);
    }
  }
  return [];
}

/** The BASENAMES of the room's own harness files, from its own report directory's `harness/`.
 *  Rule 33 orders the room to copy one of these INTO the tree (and the tree path is chosen by the
 *  suite layout, not by the room), so the grant cannot be expressed as a path -- it is expressed
 *  as "this exact file, wherever rule 33 put it". Narrower than a directory grant: it admits one
 *  named file the room itself wrote into its own report dir, and nothing else. */
export function harnessNames(trees, runDir, roomDir) {
  const out = new Set();
  for (const t of trees) {
    if (!t || !runDir || !roomDir) continue;
    const d = join(t, "org/07-findings", runDir, roomDir, "harness");
    if (!existsSync(d)) continue;
    try {
      for (const e of readdirSync(d, { withFileTypes: true })) if (e.isFile()) out.add(e.name);
    } catch { /* unreadable harness dir yields no names */ }
  }
  return [...out];
}

/**
 * W3-15 INSTRUMENT-A, DEFECT 2: THE RULE-33 HARNESS THAT NO GRANT COVERED.
 *
 * Constitution rule 33 orders a room to SHIP its regression harness into the suite. The suite
 * layout -- not the room -- chooses where it lands, so the grant cannot be written as a path in
 * the room's packet, and a room whose OWNS list names only a `src/` file therefore shipped a file
 * NOBODY held a grant for. `harnessNames` above covers the case where the room also left a copy in
 * its own report dir's `harness/`; when it did not, the shipped file had no owner at all and
 * reddened the write audit for the NEXT room to validate. MEASURED (2026-09-22): W3-15-R8 has no
 * `harness/` directory, and its shipped
 * `tests/tooling/w315-regression-w3-15-r8-halt-write-integrity.test.ts` was charged as a violation
 * against W3-15-R1 -- a room that did not write it and could not have.
 *
 * THE PATTERN IS CENSUSED, NOT ASSUMED (the lesson this file's own header already records). Census
 * 2026-09-22 over all 48 `*regression*.test.ts` outside org/07-findings in this tree: the dominant
 * shape is `<batch>-regression-<room-id>-<slug>.test.ts`, e.g.
 *   w315-regression-w3-15-r8-halt-write-integrity.test.ts   (W3-15-batch / W3-15-R8)
 *   w314-regression-w3-14-r5-enrol-scrutiny-origins.test.ts (W3-14-batch / W3-14-R5)
 *   w311-regression-w3-11-r10-frozen-sweep.test.ts          (W3-11-batch / W3-11-R10)
 * so the room id is matched as a DELIMITED SEGMENT, never as a substring: `w3-11-r1` must not
 * claim `...-w3-11-r10-...`, which a substring test would hand it. A leading batch-ish token and a
 * non-empty slug are both required, so an arbitrarily-named test file cannot become a grant.
 *
 * THIS GRANTS ONE ROOM'S OWN ID AND NOTHING ELSE. A harness naming a DIFFERENT room never matches
 * here, so it is never this room's `own` -- it stays whatever the audit's other tiers make of it.
 */
export function harnessPatternFor(roomDir) {
  const room = String(roomDir ?? "").trim().toLowerCase();
  if (!room || !/^[a-z0-9][a-z0-9.-]*$/.test(room)) return null;
  const esc = room.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
  // THE BATCH TOKEN ADMITS HYPHENS (ORG-02-R5, 2026-09-22). It was `[a-z0-9]+`, which cannot match
  // the name THIS ORG'S OWN COMPILER PRESCRIBES: `compile-room-prompt.mjs:326` builds a room's
  // harness path as `${findingPrefix.toLowerCase()}-regression-${room}-<topic>.test.ts`, and an
  // ORG-02 prefix is `FINDING-0000` -- a hyphen inside the token. MEASURED: the pattern for
  // `ORG-02-R5` rejected `finding-0000-regression-ORG-02-R5-owns-resolution.test.ts`, the exact
  // file the packet told this room to write. So a room that obeyed its own packet and shipped its
  // rule-33 harness held NO grant for it and was charged -- W3-15 INSTRUMENT-A DEFECT 2, the
  // defect this function exists to close, reopened through the compiler's prescription.
  // THE DELIMITED-SEGMENT PROPERTY IS UNCHANGED AND IS WHAT THE WIDENING MUST NOT COST: the room
  // id is still bracketed by literal `-regression-` and `-`, so `w3-11-r1` still cannot claim
  // `...-regression-w3-11-r10-...` (the char after the id must be `-`, and there it is `0`).
  // Asserted in tests/tooling/finding-0000-regression-ORG-02-R5-owns-resolution.test.ts.
  return new RegExp(`^[a-z0-9][a-z0-9-]*-regression-${esc}-[a-z0-9._-]+\\.test\\.tsx?$`, "i");
}

// The directory names this corpus uses for test suites (censused alongside the pattern above):
// apps/*/test, apps/mobile-check/native-checks, packages/*/test, services/*/test,
// services/trust-core/test-fidelity, tests/tooling, tests/invariants.
const TEST_DIR_SEGMENT = /^(?:tests?|__tests__|test-fidelity|native-checks)$/i;
// Pruned because they are large and never hold a shipped harness. `org/` is pruned too: a harness
// sitting in a room's own report dir is `harnessNames`' job, and org/ is the biggest tree here.
const PRUNE_DIR = /^(?:node_modules|\.git|\.scratch|org|dist|build|coverage|\.next|\.turbo|vendor)$/;

/**
 * Every shipped regression harness encoding THIS room's id, as tree-relative paths, from the files
 * actually present under a test directory. Returns paths (not bare names) so the grant admits the
 * exact file where it really sits and nothing else -- a name grant would match that basename
 * anywhere in the tree, which is wider than rule 33 needs.
 */
// `_runDir` is unused: the pattern derives from the ROOM id alone (the batch prefix is matched
// generically, so a room dispatched into a differently-named run still resolves). The parameter is
// kept so this helper has the SAME (trees, runDir, roomDir) shape as `harnessNames` and
// `ownsFromPackets` beside it, which is how `grantFor` calls all three.
export function regressionHarnessPaths(trees, _runDir, roomDir) {
  const re = harnessPatternFor(roomDir);
  if (!re) return [];
  const out = new Set();
  for (const t of trees) {
    if (!t || !existsSync(t)) continue;
    // Walk down to a test directory (bounded), then walk that subtree for matching files.
    const walk = (abs, rel, depth, inTestDir) => {
      let entries;
      try { entries = readdirSync(abs, { withFileTypes: true }); } catch { return; }
      for (const e of entries) {
        const childRel = rel ? `${rel}/${e.name}` : e.name;
        if (e.isDirectory()) {
          if (PRUNE_DIR.test(e.name)) continue;
          const nowInTest = inTestDir || TEST_DIR_SEGMENT.test(e.name);
          // Outside a test dir the walk is depth-capped (every test dir in this corpus is at
          // depth <= 3); inside one it runs to the leaves, which are small.
          if (!nowInTest && depth >= 3) continue;
          walk(join(abs, e.name), childRel, depth + 1, nowInTest);
        } else if (e.isFile() && inTestDir && re.test(e.name)) {
          out.add(childRel);
        }
      }
    };
    walk(t, "", 0, false);
  }
  return [...out];
}

/** Rooms sharing the audited tree RIGHT NOW. Derived, never guessed: two snapshots belong to the
 *  same live wave when they were taken against the same `repo` AND the same `head`. That is what
 *  "concurrent" means for this instrument -- a snapshot at another HEAD is a different base and
 *  its room's writes are not in this tree's window. 2026-09-21: 18 of 355 snapshots in the build
 *  tree share head d13736f7, and those 18 are exactly W3-12-batch + W3-13-batch. */
export function concurrentLabels(auditRoots, label) {
  const snaps = [];
  for (const root of auditRoots) {
    if (!root || !existsSync(root)) continue;
    let names;
    try { names = readdirSync(root); } catch { continue; }
    for (const n of names) {
      if (!n.endsWith(".json")) continue;
      const f = join(root, n);
      try {
        if (statSync(f).size === 0) continue;
        const s = JSON.parse(readFileSync(f, "utf8"));
        if (s && typeof s.head === "string") snaps.push({ label: n.slice(0, -5), head: s.head, repo: s.repo });
      } catch { /* an unreadable snapshot names no room */ }
    }
  }
  const me = snaps.find((s) => s.label === label);
  if (!me) return [];
  return snaps.filter((s) => s.head === me.head && s.repo === me.repo && s.label !== label).map((s) => s.label);
}

function splitLabel(l) {
  const i = String(l).indexOf("__");
  return i < 0 ? null : { run: l.slice(0, i), room: l.slice(i + 2) };
}

/** Every room dir under `run` that carries a compiled PROMPT.md, in any tree. Catches a room of
 *  the live wave whose snapshot is absent (a hand-dispatched room), which `concurrentLabels`
 *  alone cannot see. A dir with NO PROMPT.md is NOT a room here: it has no grant to read, so
 *  including it would claim an ownership nobody compiled. */
export function compiledRoomsUnder(trees, runDir) {
  const out = new Set();
  for (const t of trees) {
    if (!t || !runDir) continue;
    const runAbs = join(t, "org/07-findings", runDir);
    if (!existsSync(runAbs)) continue;
    try {
      for (const e of readdirSync(runAbs, { withFileTypes: true })) {
        if (!e.isDirectory() || e.name.startsWith(".")) continue;
        if (existsSync(join(runAbs, e.name, "PROMPT.md"))) out.add(e.name);
      }
    } catch { /* unreadable run dir contributes nothing */ }
  }
  return [...out];
}

/**
 * The grant for ONE room: `paths` (exact-or-directory-prefix) and `names` (bare basenames).
 * `packets` and `prompt` are both reported so a disagreement between the compiler's two records
 * is visible instead of being silently unioned away.
 */
export function grantFor({ trees, runDir, roomDir }) {
  const promptFile = findPrompt(trees, runDir, roomDir);
  const fromPrompt = promptFile ? parseOwns(readFileSync(promptFile, "utf8")) : [];
  const fromPackets = ownsFromPackets(trees, runDir, roomDir);
  // DEFECT 2: rule 33's shipped harness is an IMPLICIT grant -- derived from this room's own id,
  // never declared, because the suite layout chooses the path after the packet is written.
  const fromHarness = regressionHarnessPaths(trees, runDir, roomDir);
  return {
    runDir,
    roomDir,
    promptFile,
    fromPrompt,
    fromPackets,
    fromHarness,
    paths: [...new Set([...fromPrompt, ...fromPackets, ...fromHarness])],
    names: harnessNames(trees, runDir, roomDir),
  };
}

/**
 * THIS room's grant, plus every concurrently-live room's grant keyed by room dir.
 *
 * WHAT THIS DELIBERATELY DOES NOT DO: it does not widen `own`. A sibling's grant NEVER exculpates
 * the room being checked for its own writes -- it only lets the audit say "this path belongs to
 * W3-12-R6" instead of charging it here. A room cannot grant itself anything by declaring it; the
 * only thing that grants is a PROMPT.md the compiler wrote.
 */
// I34-TOOLS-B (I33 B22, FAMILY-027): it also RETURNS the room list it enumerated -- `rooms`, this room first, each
// `{ run, room, own, promptFile }` once -- and, given a `globsOf(promptText)` reader (the validator passes
// globGrantsInPrompt), `globs`: every glob grant those rooms' compiled prompts record, `{ run, room, own, glob }`, with
// this room's read from `ownPrompt` when one is given. isGrantToken refuses a glob token, so `own`/`attributed` never
// carry one; the write audit reads glob grants from `globs` instead of enumerating the rooms a second time.
export function resolveGrants({ trees, auditRoots, runDir, roomDir, label, globsOf = null, ownPrompt = null }) {
  const own = grantFor({ trees, runDir, roomDir });
  const runs = new Set([runDir]);
  const rooms = [{ run: runDir, room: roomDir }];
  for (const l of concurrentLabels(auditRoots, label)) {
    const s = splitLabel(l);
    if (!s) continue;
    runs.add(s.run);
    rooms.push(s);
  }
  for (const run of runs) for (const room of compiledRoomsUnder(trees, run)) rooms.push({ run, room });
  const attributed = [];
  const roomList = [];
  const seen = new Set();
  for (const { run, room } of rooms) {
    const key = `${run}__${room}`;
    if (seen.has(key)) continue;
    seen.add(key);
    const isOwn = run === runDir && room === roomDir;
    roomList.push({ run, room, own: isOwn, promptFile: isOwn && ownPrompt ? ownPrompt : findPrompt(trees, run, room) });
    if (isOwn) continue;
    const g = grantFor({ trees, runDir: run, roomDir: room });
    for (const p of g.paths) attributed.push({ room, path: p });
    for (const n of g.names) attributed.push({ room, path: n });
  }
  const globs = [];
  if (typeof globsOf === "function") {
    for (const r of roomList) {
      if (!r.promptFile || !existsSync(r.promptFile)) continue;
      for (const glob of globsOf(readFileSync(r.promptFile, "utf8"))) globs.push({ run: r.run, room: r.room, own: r.own, glob });
    }
  }
  return { own, attributed, liveRuns: [...runs], rooms: roomList, globs };
}

// scripts/lib/root.mjs — W3-11a-R4. THE ROOT DEFECT: `--root` defaulted to
// `process.cwd()` across scripts/, and a ROOM's cwd is the build tree, whose
// `org/` is a stale checkout of the organization's records. One default
// produced QUEUE #434 (a `0 finding(s) from 0 room file(s)` digest written over
// the inter-room channel while twelve rooms' FINDINGS.jsonl sat in the repo),
// QUEUE #422 (`fatal: bad object` and ~1,700 false accusations, worked around
// by eleven rooms across two batches) and QUEUE #432 (an unrecognised flag
// silently DROPPED, so a typo became a destructive write).
//
// Both halves are the same family: a tool that answers confidently about the
// WRONG INPUT. So this module refuses rather than guesses, on both axes.
//
// ANCHOR. The tree is located by walking up from the CALLER'S OWN FILE, not
// from cwd. `node <tree>/scripts/x.mjs` run from anywhere then resolves to
// <tree>, which is the property the guard measures (same root from a
// subdirectory and from a sibling tree).
import { existsSync, realpathSync, statSync } from "node:fs";
import { dirname, isAbsolute, parse, resolve } from "node:path";
import { fileURLToPath } from "node:url";
import { parseArgs as nodeParseArgs } from "node:util";

// MARKER. `org/00-constitution` and `.git` are NOT equivalent and the choice is
// measured, not argued: inside this build tree, `.git` also occurs at
// `.scratch/r7/{g1,f2,f3,f4,o2}/.git`, and none of those directories carries
// `org/00-constitution`. `.git` answers "am I in a git repository"; a scratch
// repo, a room worktree and an unrelated checkout all answer yes. The
// constitution directory answers "am I in THIS ORGANIZATION'S tree", which is
// the question every caller here is actually asking. A worktree also spells
// `.git` as a FILE, which a naive `existsSync` test reads as a hit.
export const ROOT_MARKER = "org/00-constitution";

export class RootError extends Error {
  constructor(message) { super(message); this.name = "RootError"; }
}
export class UsageError extends Error {
  constructor(message) { super(message); this.name = "UsageError"; }
}

/** Absolute directory for a path, a file:// URL, or an import.meta.url. */
function dirOf(from) {
  if (!from) throw new RootError("root.mjs: no anchor given");
  const p = typeof from === "string" && from.startsWith("file:") ? fileURLToPath(from)
    : from instanceof URL ? fileURLToPath(from)
    : String(from);
  const abs = isAbsolute(p) ? p : resolve(p);
  // An anchor is normally a FILE (import.meta.url). Treating a file as a
  // directory would start the walk one level too deep, which is invisible
  // until a marker sits exactly at that level.
  return existsSync(abs) && !isDir(abs) ? dirname(abs) : abs;
}
function isDir(p) {
  try { return statSync(p).isDirectory(); } catch { return false; }
}

/**
 * Walk up from `from` for `marker`. Returns the absolute directory that CONTAINS
 * the marker, or null. The marker must be a DIRECTORY: a worktree's `.git` file
 * would otherwise answer a `.git` marker test.
 */
export function findRoot(from, { marker = ROOT_MARKER } = {}) {
  let dir = dirOf(from);
  const stop = parse(dir).root;
  for (;;) {
    const candidate = resolve(dir, marker);
    if (existsSync(candidate) && isDir(candidate)) return dir;
    if (dir === stop) return null;
    const up = dirname(dir);
    if (up === dir) return null;
    dir = up;
  }
}

/**
 * The tree this instrument belongs to.
 *   explicit (--root/--repo/--tree)  ->  wins, verbatim, NOT marker-checked
 *                                        (temp-fixture trees are a real caller)
 *   else                             ->  located by walking up from `from`
 *   else                             ->  THROW. Never process.cwd().
 * `cwd` is accepted only as a DECLARED anchor (`{from: process.cwd()}`) by a
 * tool whose subject genuinely is the tree you are standing in; it is never a
 * fallback.
 */
export function resolveRoot({ from, explicit = null, marker = ROOT_MARKER, what = "--root" } = {}) {
  if (explicit !== null && explicit !== undefined && explicit !== "") return resolve(explicit);
  const found = findRoot(from, { marker });
  if (found) return found;
  throw new RootError(
    `could not locate the repository root: no \`${marker}/\` directory above ${dirOf(from)}. ` +
    `Pass ${what} <repo> explicitly. Refusing to fall back to the current directory — ` +
    `a room's cwd is the build tree, whose org/ is a stale checkout (QUEUE #434, #422).`
  );
}

export const REPO_HOME_REL = "mnt/dhwaja";
export const BUILD_TREE_HOME_REL = "mnt/dhwaja-linux-build";
const realOr = (p) => { try { return realpathSync(p); } catch { return resolve(p); } };
export function repoCensusRoot({ from, explicit = null, env = process.env, marker = ROOT_MARKER, what = "--repo" } = {}) {

/**
 * Strict argv parser (QUEUE #432). Wraps node:util.parseArgs — platform before
 * dependency — and adds the two shapes this tree's CLIs already use and
 * node:util cannot express:
 *   variadic: true   `--allow a b c`   (collects until the next --flag)
 *   rest             everything after a literal `--`, kept separate from
 *                    positionals (node:util folds them together, which loses a
 *                    git-style pathspec boundary)
 * spec: { options: { name: {type:"string"|"boolean", short?, multiple?, variadic?} },
 *         allowPositionals?: boolean }
 * THROWS UsageError on an unrecognised flag instead of dropping it.
 */
export function parseArgs(argv, spec = {}) {
  const options = spec.options ?? {};
  const allowPositionals = spec.allowPositionals ?? true;

  const dash = argv.indexOf("--");
  const head = dash === -1 ? argv.slice() : argv.slice(0, dash);
  const rest = dash === -1 ? [] : argv.slice(dash + 1);

  // Variadic options are lifted out before node:util sees them.
  const variadicNames = Object.keys(options).filter((k) => options[k].variadic);
  const variadicValues = {};
  const kept = [];
  for (let i = 0; i < head.length; i++) {
    const tok = head[i];
    const name = tok.startsWith("--") ? tok.slice(2).split("=")[0] : null;
    if (name && variadicNames.includes(name)) {
      const inline = tok.includes("=") ? [tok.slice(tok.indexOf("=") + 1)] : [];
      const collected = [...inline];
      while (i + 1 < head.length && !head[i + 1].startsWith("--")) collected.push(head[++i]);
      variadicValues[name] = [...(variadicValues[name] ?? []), ...collected];
      continue;
    }
    kept.push(tok);
  }

  const nodeOptions = {};
  for (const [k, v] of Object.entries(options)) {
    if (v.variadic) continue;
    nodeOptions[k] = { type: v.type ?? "boolean", ...(v.short ? { short: v.short } : {}), ...(v.multiple ? { multiple: true } : {}) };
  }

  let parsed;
  try {
    parsed = nodeParseArgs({ args: kept, options: nodeOptions, strict: true, allowPositionals });
  } catch (e) {
    const known = Object.keys(options).map((k) => `--${k}`).join(" ");
    throw new UsageError(`${e.message.split("\n")[0]}\n  recognised flags: ${known || "(none)"}`);
  }
  return { values: { ...parsed.values, ...variadicValues }, positionals: parsed.positionals, rest };
}

/**
 * One-call adopter entry. `from` is the CALLER'S `import.meta.url`.
 *   const { values, positionals, rest, root } = cli({
 *     from: import.meta.url,
 *     usage: "node scripts/x.mjs <run> [--root <repo>] [--json]",
 *     options: { root: { type: "string" }, json: { type: "boolean" } },
 *     rootFlag: "root",              // which option carries the explicit root
 *     argv: process.argv.slice(2),   // optional
 *   });
 * A usage error or an unlocatable root prints and exits 2 — the code this tree's
 * CLIs already use for "usage/config error" — unless `throwOnError` is set, which
 * is what the tests use.
 */
export function cli(spec) {
  const argv = spec.argv ?? process.argv.slice(2);
  const rootFlag = spec.rootFlag ?? "root";
  try {
    const parsed = parseArgs(argv, spec);
    const explicit = parsed.values[rootFlag] ?? null;
    const root = spec.noRoot
      ? null
      : resolveRoot({ from: spec.from, explicit, marker: spec.marker, what: `--${rootFlag}` });
    return { ...parsed, root };
  } catch (e) {
    if (spec.throwOnError) throw e;
    if (e instanceof UsageError || e instanceof RootError) {
      console.error(`${spec.name ?? "error"}: ${e.message}`);
      if (spec.usage) console.error(`usage: ${spec.usage}`);
      process.exit(2);
    }
    throw e;
  }
}

/** The tokens behind extractIds, each with the span of source it consumed (D-0720 needs to know which `#N` are bare). */
function scanIds(text) {
  const src = String(text ?? "");
  const out = [];
  let span = 0;
  const push = (id, end) => { out.push({ id, start: span, end }); };
  HEAD_RE.lastIndex = 0;
  for (let m = HEAD_RE.exec(src); m; m = HEAD_RE.exec(src)) {
    let cont, fmt;
    span = m.index;
    const e0 = HEAD_RE.lastIndex;
    if (m[1]) { cont = CONT_I3_RE; fmt = (n) => `I3-${n}`; push(fmt(m[2]), e0); }
    else if (m[3]) { cont = CONT_DOCKET_RE; fmt = (n) => `DOCKET#${Number(n)}`; push(fmt(m[4]), e0); }
    else if (m[5]) { cont = CONT_NUM_RE; fmt = (n) => `QUEUE#${Number(n)}`; push(fmt(m[6]), e0); }
    else {
      // canonical ids are upper-case in the register. I10-FX2 (#589): keep the leading upper-case segments
      // (`REV:PR-10/ps-186` is REV:PR-10, not nothing) and rescan after them.
      if (m[7] !== m[7].toUpperCase()) continue;
      const kept = [];
      for (const seg of m[8].split("/")) { if (seg !== seg.toUpperCase()) break; kept.push(seg); }
      if (kept.length === 0) continue;
      const tok = `${m[7]}:${kept.join("/")}`;
      push(tok, m.index + tok.length);
      HEAD_RE.lastIndex = m.index + tok.length;
      continue;
    }
    let at = HEAD_RE.lastIndex;
    for (;;) {
      cont.lastIndex = at;
      const c = cont.exec(src);
      if (!c) break;
      const n = c[1] ?? c[2];
      if (c[2] !== undefined) {
        const prev = Number(out[out.length - 1].id.slice(3));
        if (!(Number(n) > prev && Number(n) - prev <= COMMA_DASH_SPAN)) break;
      }
      span = at;
      push(fmt(n), cont.lastIndex);
      at = cont.lastIndex;
    }
    HEAD_RE.lastIndex = at;
  }
  return out;
}

// D-0720 (d) / D-0721 (6): `rules_on` must be an ARRAY of namespaced ids, each entry exactly one whole
// id (I3-NNNN, QUEUE#N, DOCKET#N, or a canonical id the register knows). `[]` rules on no id. Anything
// malformed — not an array, a non-string, a multi-id or partial entry, an unknown canonical id — skips
// the row's `rules_on` with a warning and falls back to topic extraction.
function wholeId(x, knownCanon) {
  if (typeof x !== "string") return null;
  const t = x.trim();
  const toks = scanIds(t);
  if (toks.length !== 1 || toks[0].start !== 0 || toks[0].end !== t.length) return null;
  const id = toks[0].id;

  return id;
}

    });
  }
  // D-0719: register aliases, both directions, keyed to the iid.
  const toIid = new Map();
  const toCanon = new Map();
  const finderSeen = new Map();
  if (present.register) {
    readFileSync(paths.register, "utf8").split("\n").forEach((l, i) => {
      const s = l.trim();
      if (!s) return;
      try {
        const r = JSON.parse(s);
        if (r.iid && r.canonical_id) { toIid.set(String(r.canonical_id).toUpperCase(), r.iid); toCanon.set(r.iid, String(r.canonical_id)); }
        if (r.iid && Array.isArray(r.members)) {
          for (const mem of r.members) {
            const f = String(mem).startsWith(`${FINDER_NS}:`) ? String(mem).slice(FINDER_NS.length + 1) : null;
            if (f) finderSeen.set(f, (finderSeen.get(f) ?? new Set()).add(r.iid));
          }
        }
      } catch (e) { errors.push(`${REGISTER_REL}:${i + 1} is not JSON (${e.message}) — skipped`); }
    });
  }
  // QUEUE #582: unique finder-id resolutions only; an ambiguous finder id is counted, never guessed.
  const finderToIid = new Map([...finderSeen].filter(([, set]) => set.size === 1).map(([f, set]) => [f, [...set][0]]));
  const finderAmbiguous = [...finderSeen].filter(([, set]) => set.size > 1).map(([f]) => f).sort();
  // QUEUE #589: a `/`-joined canonical id the register does not know is read as its namespaced segments, never as its first.
  const canonIds = (id) => {
    if (toIid.has(id) || !/^[A-Z]{2,5}:[^/]+\//.test(id)) return [toIid.get(id) ?? id];
    const ns = id.slice(0, id.indexOf(":"));
    return id.slice(ns.length + 1).split("/").map((s) => (/^[A-Z]{2,5}:/.test(s) ? s : `${ns}:${s}`)).map((s) => toIid.get(s) ?? s);
  };
  const knownCanon = present.register ? new Set(toIid.keys()) : null;
  const jason = rows.filter((r) => {
    if (!isJason(r.row)) return false;
    if (ID_SHAPE.test(String(r.row.id))) return true;
    errors.push(`${LEDGER_REL}:${r.line}: Jason row id ${safeId(r.row.id)} is not exactly D-NNNN — row skipped (D-0721)`);
    return false;
  });
  const byDid = new Map();
  for (const r of jason) byDid.set(r.row.id, [...(byDid.get(r.row.id) ?? []), r]);
  const map = new Map();
  const add = (id, r, via) => {
    const list = map.get(id) ?? [];
    if (!list.some((x) => x.line === r.line)) list.push({ ...r, via });
    map.set(id, list);
  };
  const yieldless = [];
  for (const r of jason) {
    const ids = idsOf(r.row, { errors, knownCanon });
    if (ids.length === 0) yieldless.push(`${r.row.id}@${LEDGER_REL}:${r.line}`);
    for (const id of ids) for (const c of canonIds(id)) add(c, r, "topic");
  }
  let queueRows = 0;
  const queueCrossRowSkipped = [];
  const queueLinks = [];
  if (present.queue) {
    readFileSync(paths.queue, "utf8").split("\n").forEach((l, i) => {
      const m = /^\|\s*#?(\d+)\s*\|/.exec(l);
      if (!m) return;
      queueRows++;

      const marked = markerDids(cell);
      for (const did of new Set(cell.match(DID_RE) ?? [])) {
        const qid = `QUEUE#${Number(m[1])}`;

          if (!marked.has(did) && namesOtherRow(r.row, qid, { knownCanon })) { queueCrossRowSkipped.push(`${qid}->${r.row.id}`); continue; }
          add(qid, r, `${QUEUE_REL}:${i + 1}`);
          queueLinks.push(`${qid}->${r.row.id}`);
        }
      }
    });
  }
  for (const list of map.values()) list.sort(newestFirst);
  const index = {
    root: key, paths, present, errors,
    /** `{ rulings, cut }` for `id` (any spelling extractIds accepts), or null when unruled. */
    lookup(id) {
      // I10-FX2 (#589): text naming MORE than one id is refused (null), never read as its first id; a `/`-joined canonical id
      // the register does not know gets only the rulings filed under EVERY one of its segments, never its first segment's.
      const named = extractIds(id);
      if (named.length > 1) return null;
      const finder = named.length === 0 ? finderToIid.get(String(id ?? "").trim()) : undefined;
      const segs = finder ? [finder] : canonIds(named[0] ?? String(id));
      const norm = segs.length === 1 ? segs[0] : named[0];
      const list = segs.length === 1 ? map.get(norm)
        : (map.get(segs[0]) ?? []).filter((r) => segs.every((s) => (map.get(s) ?? []).some((x) => x.line === r.line)));
      if (!list || list.length === 0) return null;
      return { id: norm, alias: toCanon.get(norm) ?? null, rulings: list.slice(0, MAX_RULINGS_PER_ID), cut: Math.max(0, list.length - MAX_RULINGS_PER_ID) };
    },
    census() {
      const exact = rows.filter((r) => { const w = r.row.decider ?? r.row.actor; return w === "jason" || w === "Jason"; }).length;
      return {
        ledger_rows: rows.length, parse_errors: errors.length, queue_rows: queueRows,
        jason_rows_first_word: jason.length, jason_rows_exact_match: exact,
        jason_rows_yielding_ids: jason.length - yieldless.length, jason_rows_yielding_none: yieldless,
        ids_ruled: map.size, register_aliases: toIid.size,
        finder_ids_mapped: finderToIid.size, finder_ids_ambiguous: finderAmbiguous,
        links_total: [...map.values()].reduce((n, l) => n + l.length, 0),
        links_via_queue: [...map.values()].reduce((n, l) => n + l.filter((x) => x.via !== "topic").length, 0),
        queue_cross_row_skipped: queueCrossRowSkipped,
        queue_links: [...new Set(queueLinks)].sort(),
      };
    },
  };
  CACHE.set(key, index);
  return index;
}

/** The block for one lookup result. mode "row" for ROWS.md, "packet" for the compiler's section. */

export function unruledRowBlocks(rowsText, index) {

/**
 * D-0721 (REG-579-10): every ROWS.md reference in a packet, resolved in order against the packet dir,
 * the batch dir(s), `org/07-findings/` and the repo root (`$HOME`/`${HOME}`/`~` expanded first; an
 * absolute path is taken as is). REG-579-17: NO reference is discarded. The token boundary is
 * Unicode-aware (letters/digits of any script, `_ . / -`, and template brackets `< > { } *`), so a
 * curly quote, NBSP or em dash before a path is a boundary, never a reason to skip it, and a template
 * such as `<batch>/<room>/ROWS.md` is kept whole and shown as unresolved. Returns the resolved files
 * (to be checked) and the unresolved refs, which a caller must SHOW — never refuse on, never drop.
 */
export function resolveRowRefs(packetText, { packetDir, batchDirs = [], root, home }) {
  const refs = new Set();
  for (const m of String(packetText).matchAll(/(?:\$\{HOME\}|\$HOME|~)?[\p{L}\p{N}_./<>{}*-]*ROWS\.md(?![\p{L}\p{N}_])/gu)) refs.add(m[0]);
  const bases = [...new Set([packetDir, ...batchDirs, join(root, "org/07-findings"), root].filter(Boolean))];
  const resolved = [];
  const unresolved = [];
  for (const ref of refs) {
    const expanded = ref.replace(/^(?:\$\{HOME\}|\$HOME|~)(?=\/)/, home);
    const candidates = expanded.startsWith("/") ? [expanded] : bases.map((b) => resolve(b, expanded));
    const abs = candidates.find((c) => existsSync(c) && statSync(c).isFile());
    if (abs) resolved.push({ ref, abs });
    else unresolved.push(ref);
  }
  return { resolved, unresolved };
}

// scripts/lib/seed-recall.mjs — QUEUE #277 (S-081 SCOUT): the ONE place that picks which
// instance's `seeds` is authoritative for a seeded-recall reading. Before this extraction,
// `halt.mjs`'s `recallTripwire` and `control-signals.mjs`'s `seed_recall.last` each picked their
// own instance independently and disagreed: control-signals.mjs took the array's LAST instance
// verbatim (not necessarily a SWEEP, let alone a SCORED one); halt.mjs did the right selection
// but only for itself. Both now call this.
//
// Selection (halt.mjs's recallTripwire logic, unchanged by the extraction — includes the
// D-0424.3 / #278 item 2 ruling: the scored instance must ALSO be SWEEP-tagged, not just any
// instance with seeds.declared=true, so a scored-but-non-SWEEP run never satisfies this):
//   1. the LATEST instance (by the caller's own ordering) whose composition.modes includes
//      "SWEEP".
//   2. within [0..that index], the LATEST instance that is ALSO SWEEP-tagged AND has
//      seeds.declared === true.
//   Neither found -> not resolvable; the caller decides how to report that (halt.mjs trips a
//   named tripwire with an evidence string; control-signals.mjs records `null` and a `source`
//   explaining why).
//
// `instances` must already be in the caller's own chronological order (halt.mjs: instance
// filename sort; control-signals.mjs: run_id sort) — this helper never re-sorts.

export function isSweepInstance(inst) {
  return Array.isArray(inst?.composition?.modes) && inst.composition.modes.includes("SWEEP");
}

/**
 * @param {ReadonlyArray<{composition?: {modes?: string[]}, seeds?: {declared?: boolean}}>} instances
 * @returns {{ok: true, sweepIndex: number, sweep: unknown, scoredIndex: number, scored: unknown, seeds: unknown}
 *         | {ok: false, reason: string, sweepIndex?: number, sweep?: unknown}}
 */
export function pickLatestScoredSweepSeeds(instances) {
  const sIdx = instances.map(isSweepInstance).lastIndexOf(true);
  if (sIdx < 0) return { ok: false, reason: "no SWEEP instance (composition.modes)" };
  const sweep = instances[sIdx];
  const tIdx = instances.slice(0, sIdx + 1).map((i) => isSweepInstance(i) && i?.seeds?.declared === true).lastIndexOf(true);
  if (tIdx < 0) return { ok: false, reason: "no instance up to the latest sweep declares seeds", sweepIndex: sIdx, sweep };
  const scored = instances[tIdx];
  return { ok: true, sweepIndex: sIdx, sweep, scoredIndex: tIdx, scored, seeds: scored.seeds };
}

const FM_RE = /^---\r?\n([\s\S]*?)\r?\n---\r?\n/;
/** I38-T4 (Q1000-B7): the ONE frontmatter reader for skills and personas — a leading `---` block with LF or CRLF line
 *  ends, after one optional leading BOM. `fm` and `body` come back with CRLF turned to LF; null when the text opens with no such block. */
export function readFrontmatter(text) {
  // I40-T-TOOLS (I40S-079): a Windows editor's leading byte-order mark (U+FEFF) is not part of the text.
  const t = String(text).replace(/^\ufeff/, "");
  const m = FM_RE.exec(t);
  return m ? { fm: m[1].replace(/\r\n/g, "\n"), body: t.slice(m[0].length).replace(/\r\n/g, "\n") } : null;
}
const fmValue = (fm, key) => {
  const line = fm.split("\n").find((l) => new RegExp(`^${key}:\\s*`).test(l));
  return line === undefined ? null : line.replace(new RegExp(`^${key}:\\s*`), "").trim().replace(/^["']|["']$/g, "");
};

/** The `skill_ids:` frontmatter list: null when the line is absent; throws when present but not a `[...]` list. */
export function parseSkillIdsLine(fm) {
  const line = fm.split("\n").find((l) => /^skill_ids:/.test(l));
  if (line === undefined) return null;
  if (!/\[.*\]/.test(line)) throw new SkillMethodError(`the persona's \`${line.trim()}\` is not a [..] list`);
  return parseListField(line);
}

/** The org/skills/<slug>/SKILL.md whose frontmatter `id:` equals `id`, or null. */
export function findSkillById(root, id) {
  const dir = join(root, "org/skills");
  if (!existsSync(dir)) return null;
  for (const e of readdirSync(dir, { withFileTypes: true })) {
    if (!e.isDirectory()) continue;
    const file = join(dir, e.name, "SKILL.md");

}

/** heading -> every body it heads, in order. A `## ` line inside a ``` or ~~~ fence (closed by a line of the same
 *  character, at least as long) is body text, not a heading (RG-COMPILE/2). Text before the first heading is dropped. */
function splitSections(body) {
  const sections = new Map();
  let cur = null;
  let fence = null;
  for (const line of body.split("\n")) {
    // I40-T-TOOLS (FAMILY-013 census of I40S-057's class): findings-vocab.mjs fenceStep, so an inline-code line opens no fence.
    const st = fenceStep(fence, line);
    fence = st.fence;
    if (!st.fenceLine && fence === null && line.startsWith("## ")) {
      const key = line.slice(3).trim();
      cur = [];
      sections.set(key, [...(sections.get(key) ?? []), cur]);
      continue;
    }
    cur?.push(line);
  }
  return new Map([...sections].map(([k, v]) => [k, v.map((lines) => lines.join("\n"))]));
}

/** Resolve one id to its METHOD block; throws SkillMethodError naming the refusal. */
export function resolveSkillMethod(root, id) {
  const found = findSkillById(root, id);
  if (!found) throw new SkillMethodError(`no org/skills/*/SKILL.md carries \`id: ${id}\``);
  const { slug, text, status } = found;
  const rel = `org/skills/${slug}/SKILL.md`;
  const mirror = join(root, ".claude/skills", slug, "SKILL.md");
  if (!existsSync(mirror) || !readFileSync(mirror).equals(readFileSync(found.file))) {
    throw new SkillMethodError(`.claude/skills/${slug}/SKILL.md is not byte-identical to ${rel}`);
  }
  const sections = splitSections(readFrontmatter(text)?.body ?? text.replace(/\r\n/g, "\n"));
  const missing = METHOD_SECTIONS.filter((s) => !sections.has(s));
  if (missing.length) throw new SkillMethodError(`${rel} (${id}) lacks ${missing.map((s) => `\`## ${s}\``).join(", ")}`);
  for (const s of METHOD_SECTIONS) {
    const bodies = sections.get(s) ?? [];
    if (bodies.length > 1) throw new SkillMethodError(`${rel} (${id}) carries \`## ${s}\` ${bodies.length} times; a required section appears once`);
    if (!bodies[0]?.trim()) throw new SkillMethodError(`${rel} (${id}) has an empty \`## ${s}\``);
  }
  const heading = `## METHOD ${id} ${slug} (${rel})`;
  const block = [heading, ...METHOD_SECTIONS.map((s) => `### ${s}\n${(sections.get(s) ?? [""])[0].replace(/^\n+|\s+$/g, "")}\n`)].join("\n");
  return { id, slug, status, heading, block };
}

#!/usr/bin/env node
// scripts/lib/ts-scan.mjs — ADR-0009 D3 item S5a-3: shared TypeScript compiler-API
// helpers for the four AST guards (never regex over source: call expressions at any
// nesting, string/template literal contents, the enclosing function of a node, imports).
import ts from "typescript";
import { readFileSync } from "node:fs";

/** Parse a file on disk into a ts.SourceFile with parent pointers set (needed by
 *  enclosingFunction/line lookups below). */
export function parse(path) {
  const text = readFileSync(path, "utf8");
  const kind = path.endsWith(".tsx") ? ts.ScriptKind.TSX : ts.ScriptKind.TS;
  return ts.createSourceFile(path, text, ts.ScriptTarget.Latest, /* setParentNodes */ true, kind);
}

/** Pre-order walk of every descendant of `node` (node itself included first).
 *  `visit` may return `false` to skip that node's children (used to treat a
 *  `"a" + "b"` string-concatenation chain as ONE unit rather than walking into
 *  its already-consumed operands). */
export function walk(node, visit) {
  const r = visit(node);
  if (r !== false) ts.forEachChild(node, (child) => walk(child, visit));
}

function isPlusChain(node) {
  return ts.isBinaryExpression(node) && node.operatorToken.kind === ts.SyntaxKind.PlusToken;
}

/** True if a `+`-chain has a string/template literal anywhere in it — the
 *  house style this repo wraps a long SQL statement in (`"DELETE FROM " +
 *  "t WHERE …"`), as opposed to ordinary numeric/other addition. */
function chainHasLiteral(node) {
  if (ts.isStringLiteral(node) || ts.isNoSubstitutionTemplateLiteral(node) || ts.isTemplateExpression(node)) return true;
  if (isPlusChain(node)) return chainHasLiteral(node.left) || chainHasLiteral(node.right);
  return false;
}

/** The concatenated text of a literal-bearing `+`-chain; a non-literal operand
 *  (a variable, a call) collapses to `?`, the same convention as a template
 *  hole — the seam between two literal halves must not itself look like a
 *  token boundary the SQL splitter could hide a verb or a table name across. */
function concatText(node) {
  if (ts.isStringLiteral(node) || ts.isNoSubstitutionTemplateLiteral(node)) return node.text;
  if (ts.isTemplateExpression(node)) {
    let s = node.head.text;
    for (const span of node.templateSpans) s += "?" + span.literal.text;
    return s;
  }
  if (isPlusChain(node)) return concatText(node.left) + concatText(node.right);
  return "?";
}

/** The nearest FunctionDeclaration / MethodDeclaration / ArrowFunction /
 *  FunctionExpression ancestor of `node`, or undefined at module scope. */
export function enclosingFunction(node) {
  let n = node.parent;
  while (n) {
    if (ts.isFunctionDeclaration(n) || ts.isMethodDeclaration(n) || ts.isArrowFunction(n) || ts.isFunctionExpression(n)) return n;
    n = n.parent;
  }
  return undefined;
}

/** Best-effort name for a function-like node: its own name (declaration/method),
 *  or the identifier/property it is assigned to (`const NAME = () => {}`). */
export function functionName(fn) {
  if (!fn) return undefined;
  if ((ts.isFunctionDeclaration(fn) || ts.isMethodDeclaration(fn)) && fn.name) return fn.name.getText();
  const p = fn.parent;
  if (p && ts.isVariableDeclaration(p) && p.name) return p.name.getText();
  if (p && ts.isPropertyAssignment(p) && p.name) return p.name.getText();
  return undefined;
}

/** The nearest NAMED function ancestor of `node` -- a FunctionDeclaration or
 *  MethodDeclaration, or the nearest ArrowFunction/FunctionExpression bound to a
 *  name (`const NAME = () => {}`, `NAME: () => {}`) per `functionName`'s own
 *  notion of "named". An anonymous arrow/closure in between (a `.flatMap(cb)`
 *  callback, an inline `.then(() => {})`) is TRANSPARENT: this answers "same
 *  unit of authorship", not "same execution frame" -- `enclosingFunction` above
 *  answers that one (QUEUE #97, FAMILY-006: the two questions are different and
 *  a guard that wants one must not be handed the other's answer). */
export function enclosingNamedFunction(node) {
  let n = node.parent;
  while (n) {
    if (ts.isFunctionDeclaration(n) || ts.isMethodDeclaration(n)) return n;
    if ((ts.isArrowFunction(n) || ts.isFunctionExpression(n)) && functionName(n) !== undefined) return n;
    n = n.parent;
  }
  return undefined;
}

function lineOf(sf, node) {
  return sf.getLineAndCharacterOfPosition(node.getStart(sf)).line + 1;
}

/** Every CallExpression anywhere under `sf` (any nesting) whose callee's own
 *  source text is exactly `name` (`"outboxStatements"`, `"vi.useFakeTimers"`, …) —
 *  the DECLARATION of a same-named function is never a call and is excluded. */
export function callsNamed(root, name) {
  const out = [];
  const sf = root.getSourceFile ? root.getSourceFile() ?? root : root;
  walk(root, (node) => {
    if (!ts.isCallExpression(node)) return undefined;
    const calleeText = node.expression.getText(sf).trim();
    if (calleeText !== name) return undefined;
    const fn = enclosingFunction(node);
    const namedFn = enclosingNamedFunction(node);
    out.push({
      node,
      text: node.getText(sf),
      line: lineOf(sf, node),
      arguments: node.arguments,
      enclosingFunction: fn,
      enclosingFunctionName: functionName(fn),
      enclosingNamedFunction: namedFn,
      enclosingNamedFunctionName: functionName(namedFn),
    });
    return undefined;
  });
  return out;
}

/** Every string / template literal's TEXT under `node`, in source order. A
 *  template's `${…}` holes are each collapsed to a single `?` — the literal's
 *  own words survive, an interpolated expression's do not (S5a-3: the guard
 *  reads what is literally spelled, not what an identifier might resolve to). */
export function literalsIn(node) {
  const out = [];
  walk(node, (n) => {
    if (ts.isStringLiteral(n) || ts.isNoSubstitutionTemplateLiteral(n)) {
      out.push(n.text);
      return false;
    }
    if (ts.isTemplateExpression(n)) {
      let s = n.head.text;
      for (const span of n.templateSpans) s += "?" + span.literal.text;
      out.push(s);
      return false;
    }
    if (isPlusChain(n) && chainHasLiteral(n)) {
      out.push(concatText(n));
      return false;
    }
    return undefined;
  });
  return out;
}

/** Every literal under `node`, split into `;`-delimited statements (SQL text is
 *  built out of literals only — a spread param or a bound `?` is not a statement
 *  boundary, so this never looks past the literal's own characters). Empty
 *  fragments (a trailing `;`, a literal with none) are dropped. */
export function sqlStatementsIn(node) {
  const out = [];
  for (const lit of literalsIn(node)) {
    for (const part of lit.split(";")) {
      const t = part.trim();
      if (t) out.push(t);
    }
  }
  return out;
}

/** Every import declaration in `sf`: `{ moduleSpecifier, names }`, `names` holding
 *  the default import (if any) and every named/namespace binding. */
export function importsOf(sf) {
  const out = [];
  for (const stmt of sf.statements) {
    if (!ts.isImportDeclaration(stmt) || !ts.isStringLiteral(stmt.moduleSpecifier)) continue;
    const names = [];
    const clause = stmt.importClause;
    if (clause) {
      if (clause.name) names.push(clause.name.text);
      const nb = clause.namedBindings;
      if (nb && ts.isNamedImports(nb)) for (const el of nb.elements) names.push(el.name.text);
      else if (nb && ts.isNamespaceImport(nb)) names.push(nb.name.text);
    }
    out.push({ moduleSpecifier: stmt.moduleSpecifier.text, names, node: stmt });
  }
  return out;
}

/** Every top-level CallExpression's callee identifier text, deduped (any nesting) — used to ask
 *  "does this file CALL an imported name", the caller side of the one-call-hop attribution
 *  (S-014 #86: a file that calls an imported `auditStmt(`/`withDualControl(` writes whatever
 *  THAT function writes, one hop out — the file's own SQL-literal scan cannot see it). */
export function calledIdentifiers(sf) {
  const out = new Set();
  walk(sf, (n) => {
    if (ts.isCallExpression(n) && ts.isIdentifier(n.expression)) out.add(n.expression.text);
    return undefined;
  });
  return out;
}

/** The literal-bearing SQL statements of the EXPORTED function declaration, or EXPORTED
 *  const-arrow/function-expression declaration, named `name` among `statements` — the callee
 *  side of the one-call-hop (S-014 #86). Mirrors the "declared among statements, exported only"
 *  shape `hoistedInto` uses for a conditional producer's import hop
 *  (services/trust-core/test/conditional-emit-scan.test.ts:97-135, read for the shape) but reads
 *  a CALLEE's body instead of an initialiser's text: an import can bind only an exported name,
 *  so a private helper of the target module is never the caller's write. One hop only — the
 *  callee's own further calls into a THIRD module are not followed. */
export function exportedFunctionStatements(statements, name) {
  const exported = (d) => (ts.getCombinedModifierFlags(d) & ts.ModifierFlags.Export) !== 0;
  for (const stmt of statements) {
    if (ts.isFunctionDeclaration(stmt) && stmt.name && stmt.name.text === name && stmt.body && exported(stmt)) {
      return sqlStatementsIn(stmt.body);
    }
    if (ts.isVariableStatement(stmt) && exported(stmt)) {
      for (const d of stmt.declarationList.declarations) {
        if (
          ts.isIdentifier(d.name) &&
          d.name.text === name &&
          d.initializer &&
          (ts.isArrowFunction(d.initializer) || ts.isFunctionExpression(d.initializer))
        ) {
          return sqlStatementsIn(d.initializer.body ?? d.initializer);
        }
      }
    }
  }
  return [];
}

/** An ObjectLiteralExpression's own property names (PropertyAssignment,
 *  ShorthandPropertyAssignment, method, or spread's target skipped) — used to ask
 *  "does this call's options argument NAME a property", never "does its printed
 *  text contain a substring". */
export function objectLiteralKeys(node) {
  if (!node || !ts.isObjectLiteralExpression(node)) return [];
  const keys = [];
  for (const prop of node.properties) {
    if (ts.isPropertyAssignment(prop) || ts.isShorthandPropertyAssignment(prop) || ts.isMethodDeclaration(prop)) {
      keys.push(prop.name.getText().replace(/^["'`]|["'`]$/g, ""));
    }
  }
  return keys;
}

// scripts/lib/usage-figures.mjs — D-1092(2): the ONE check on a usage row's counter figures, imported by both doors that
// append to org/08-control/USAGE.jsonl: scripts/record-room-usage.mjs (the orchestrator's recorder) and
// scripts/token-report.mjs --record (the ledger's append path, which the recorder calls and which can be called directly).
// A figure — tokens, tool uses, duration in milliseconds — is a positive whole number. A missing or zero figure is
// refused unless the row carries a non-blank allow-zero reason; a fraction, a negative or a non-number is refused with
// or without one (I33-RG-ORG:7/:17: `0.4` tokens and `-500` tokens were stored as `measured`).

/** Faults an allow-zero reason may waive: the figure was not reported, or was reported as zero. */
export const WAIVABLE_FAULTS = Object.freeze(["missing", "zero"]);

/** null for a positive whole number no larger than Number.MAX_SAFE_INTEGER; else the fault: missing | not a number | zero |
 *  negative | fractional | too large. */
export function figureFault(v) {
  if (v === undefined || v === null) return "missing";
  if (typeof v !== "number" || !Number.isFinite(v)) return "not a number";
  if (v === 0) return "zero";
  if (v < 0) return "negative";
  if (!Number.isInteger(v)) return "fractional";
  // I34-TOOLS-C (I33 JA-05 limb 1): 1e21 and 9007199254740993 are integers to Number.isInteger but not exact counts.
  if (!Number.isSafeInteger(v)) return "too large";
  return null;
}

/** An allow-zero reason is a string with a non-space character. */
export function allowZeroReasonOk(reason) {
  return typeof reason === "string" && reason.trim() !== "";
}

/** The faults that block a row: `figures` is [[name, value], ...]; `allowZero` is the row's reason (undefined when the
 *  row claims none). A declared but blank reason is itself a fault (name `allow-zero`). Returns [{ name, value, fault }]. */
export function usageFigureRefusals(figures, { allowZero } = {}) {
  const out = [];
  if (allowZero !== undefined && !allowZeroReasonOk(allowZero)) out.push({ name: "allow-zero", value: allowZero, fault: "blank reason" });
  const waive = allowZeroReasonOk(allowZero);
  for (const [name, value] of figures) {
    const fault = figureFault(value);
    if (fault && !(waive && WAIVABLE_FAULTS.includes(fault))) out.push({ name, value, fault });
  }
  return out;
}

/** The ledger stores duration as wall_seconds (ms / 1000); this recovers the millisecond figure, rounding away only
 *  float noise (relative, 1e-9), so 1534.174 s -> 1534174 ms while 0.0005 s stays 0.5 ms and 1e-12 s stays 1e-9 ms
 *  (an absolute tolerance rounded that one to zero, which an allow-zero reason then waived). */
export function durationMsFromWallSeconds(ws) {
  if (typeof ws !== "number" || !Number.isFinite(ws)) return ws;
  const ms = ws * 1000;
  const r = Math.round(ms);
  return Math.abs(ms - r) <= Math.abs(ms) * 1e-9 ? r : ms;
}
