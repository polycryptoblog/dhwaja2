import { sha256Bytes } from "@dhwaja/canonical";
import type { DbLike, DbStatement } from "./db.js";
import { logError } from "./log.js";

export const RL_DAY_MS = 24 * 60 * 60 * 1000;

/**
 * Atomically consume one unit from a windowed per-key counter in `rate_limits`.
 * Returns true if allowed, false if the cap is reached within the window.
 *
 * The ENTIRE decision — window reset AND increment — is one statement, so there
 * is no check-then-act race. The naive "read; if expired reset to 1; else
 * guarded-increment" pattern has a hole: N concurrent requests arriving at the
 * window boundary all read the expired window, all take the reset branch, and
 * all get served while leaving count = 1 — overshooting the cap by the burst
 * width every window. Folding the reset into the upsert closes it.
 *
 * window_start is stored ISO-8601 (lexicographically ordered), so "older than
 * the window" is `window_start < cutoffIso`. A fresh row inserts at count 1
 * (one used). On the row already existing: if expired, reset to count 1; else
 * if under cap, increment; else the guarded upsert-WHERE matches no row →
 * changes 0 → denied.
 */
export async function consumeRateLimit(db: DbLike, key: string, nowMs: number, cap: number, windowMs = RL_DAY_MS): Promise<boolean> {
  const nowIso = new Date(nowMs).toISOString();
  const cutoffIso = new Date(nowMs - windowMs).toISOString();
  const res = await db
    .prepare(
      `INSERT INTO rate_limits (rl_key, window_start, count) VALUES (?, ?, 1)
       ON CONFLICT(rl_key) DO UPDATE SET
         window_start = CASE WHEN rate_limits.window_start < ? THEN ? ELSE rate_limits.window_start END,
         count        = CASE WHEN rate_limits.window_start < ? THEN 1  ELSE rate_limits.count + 1 END
       WHERE rate_limits.window_start < ? OR rate_limits.count < ?`,
    )
    .bind(key, nowIso, cutoffIso, nowIso, cutoffIso, cutoffIso, cap)
    .run();
  if (res.changes > 0) return true;
  await recordRefusal(db, key, nowMs, cap, windowMs);
  return false;
}

/** The `audit_log.action` a refused `consumeRateLimit` writes (QUEUE #483).
 * Operators read it at `GET /ops/audit?action=ratelimit:refused`. */
export const RL_REFUSED_ACTION = "ratelimit:refused";

/**
 * The "this window's refusal is already recorded" TAG: one character appended
 * to the refused bucket's own `window_start`. `~` sorts above every character
 * an ISO-8601 instant contains, so for ANY cutoff `c` of the same format,
 * `W~ < c` iff `W < c` and `W~ >= c` iff `W >= c` -- every reader of the column
 * (the upsert above, `refundRateLimit`, `rateLimitCount`, `pruneRateLimits`)
 * decides exactly as it did untagged. The upsert's reset branch writes a fresh
 * untagged instant, so a new window starts untagged.
 */
const RL_REFUSED_TAG = "~";

/**
 * D-0810(2): the key as it may be STORED in `audit_log.detail` — every fan
 * account id (`acc_<hex>`) in it replaced by `acc#<hex>`, a one-way digest from
 * canonical's single hasher under its own label. Erasure pseudonymises
 * `audit_log` by `item_id`; a raw id in `detail` would outlive it.
 */
export async function refusalKeyForAudit(key: string): Promise<string> {
  const ids = [...new Set(key.match(/acc_[0-9a-f]+/g) ?? [])];
  let out = key;
  for (const id of ids) {
    const d = await sha256Bytes(new TextEncoder().encode(`dhwaja-rl-refusal-v1\n${id}`));
    out = out.split(id).join("acc#" + [...d.slice(0, 16)].map((b) => b.toString(16).padStart(2, "0")).join(""));
  }
  return out;
}

/**
 * A CEILING THAT FIRES IS SEEN (QUEUE #483). Called on every refusal, from the
 * one place every ceiling in the tree refuses, so no caller can forget it.
 *
 * AT MOST ONE ROW PER BUCKET PER WINDOW, NEVER ONE PER REFUSED REQUEST, AND NO
 * NEW `rate_limits` ROW EVER: the first refusal of a window tags the bucket's
 * own `window_start` in one guarded UPDATE (changes 1) and writes the audit
 * row; every later refusal of that window matches no row and writes nothing.
 * Concurrent first refusals race on that one UPDATE and exactly one wins.
 *
 * NEVER CHANGES THE ANSWER: the caller gets `false` whatever happens here. An
 * audit write that fails un-tags the bucket so the next refusal retries it,
 * and is logged.
 *
 * WHAT AN OPERATOR READS: `audit_log` row, action `ratelimit:refused`, actor
 * `system:ratelimit` (actor_kind `system`), detail
 * `{key, cap, window_ms, window_start, count}` -- whose ceiling (the key), which
 * window, and that it fired. `GET /ops/audit?action=ratelimit:refused`.
 */
async function recordRefusal(db: DbLike, key: string, nowMs: number, cap: number, windowMs: number): Promise<void> {
  try {
    const stored = await refusalKeyForAudit(key);
    const claim = await db
      .prepare("UPDATE rate_limits SET window_start = window_start || ? WHERE rl_key = ? AND substr(window_start, -1) <> ?")
      .bind(RL_REFUSED_TAG, key, RL_REFUSED_TAG)
      .run();
    if (!(claim.changes > 0)) return;
    try {
      await db
        .prepare(
          `INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind)
           SELECT 'system:ratelimit', ?, json_object('key', ?, 'cap', ?, 'window_ms', ?, 'window_start', rtrim(window_start, ?), 'count', count), ?, NULL, 'system'
           FROM rate_limits WHERE rl_key = ?`,
        )
        .bind(RL_REFUSED_ACTION, stored, cap, windowMs, RL_REFUSED_TAG, new Date(nowMs).toISOString(), key)
        .run();
    } catch (e) {
      await db
        .prepare("UPDATE rate_limits SET window_start = rtrim(window_start, ?) WHERE rl_key = ? AND substr(window_start, -1) = ?")
        .bind(RL_REFUSED_TAG, key, RL_REFUSED_TAG)
        .run();
      throw e;
    }
  } catch (e) {
    // Through log.ts's allowlist: the key names a subject (an account, a
    // passport, a masked IP) and is never logged (INV-004); it is in audit_log.
    logError({ cid: "unknown", hop: "http", msg: "ratelimit_refusal_not_recorded", cap, err: e instanceof Error ? e.name : typeof e });
  }
}

/**
 * A STOCK CEILING AS A BATCH STATEMENT (QUEUE #482). A rate limit bounds flow;
 * this bounds a quantity that accumulates. Put it in a `db.batch` AFTER the
 * insert it bounds: `countSql` (a scalar subquery, its `?`s bound by
 * `countBinds`) is evaluated inside the same transaction, and when it exceeds
 * `cap` the statement supplies NULL to the NOT NULL `rate_limits.window_start`,
 * so the WHOLE batch rolls back -- a no-match would not (standing.ts
 * `verifiedMintGate` states the mechanism). The row it leaves on success is a
 * `count 0` scratch row under `key`, swept by `pruneRateLimits`.
 * `isStockCeilingRefusal` recognises exactly this failure.
 */
export function stockCeilingGate(db: DbLike, key: string, countSql: string, countBinds: unknown[], cap: number, nowMs: number): DbStatement {
  return db
    .prepare(
      `INSERT INTO rate_limits (rl_key, window_start, count) VALUES (?, CASE WHEN (${countSql}) <= ? THEN ? END, 0)
       ON CONFLICT(rl_key) DO UPDATE SET window_start = excluded.window_start, count = 0`,
    )
    .bind(key, ...countBinds, cap, new Date(nowMs).toISOString());
}

const STOCK_REFUSED_RE = /NOT NULL constraint failed:\s*rate_limits\.window_start/i;

/** Did this error come from `stockCeilingGate` refusing its batch? Matched on
 * the exact column, never on /constraint/ (standing.ts GATE_REFUSED_RE). */
export function isStockCeilingRefusal(e: unknown): boolean {
  return e instanceof Error && STOCK_REFUSED_RE.test(e.message);
}

/**
 * GIVE ONE BACK. Decrements the bucket `consumeRateLimit` above debited, never
 * below zero, for the one caller shape that debits a slot and then finds it
 * bought nothing — `intelligence-routes.ts`'s refused explanation, on the
 * handler's own stated rule that a refusal is not a service rendered. It lives
 * HERE, beside the debit, because the bucket's shape is this module's: a caller
 * reaching into `rate_limits` with its own UPDATE is how one ceiling ends up
 * with two spellings and one of them stops matching the window.
 *
 * WINDOW-SCOPED, and that is the control: `window_start >= cutoff` means a
 * refund only ever reaches the window the debit was taken in, so a refund
 * arriving after the window rolled cannot hand the next day a free slot; and
 * `count > 0` means it cannot mint one either. The caller decides WHETHER to
 * refund and is responsible for bounding how often — this function bounds only
 * the arithmetic.
 *
 * I3-2166: `nowMs` is the DEBIT's instant (every caller passes the `nowMs` it
 * consumed with), and the window is bounded on BOTH sides: a window that was
 * reset after the debit starts after `nowMs`, never funded it, and is not
 * credited. `|| '~'` keeps a refusal-tagged window (RL_REFUSED_TAG) inside the
 * bound it would have had untagged.
 */
export async function refundRateLimit(db: DbLike, key: string, nowMs: number, windowMs = RL_DAY_MS): Promise<boolean> {
  const cutoffIso = new Date(nowMs - windowMs).toISOString();
  const debitIso = new Date(nowMs).toISOString();
  const res = await db
    .prepare("UPDATE rate_limits SET count = count - 1 WHERE rl_key = ? AND window_start >= ? AND window_start <= ? || '~' AND count > 0")
    .bind(key, cutoffIso, debitIso)
    .run();
  return res.changes > 0;
}

/**
 * READ-ONLY peek at a bucket's live count (0 when the row is absent or its
 * window has rolled). It decides nothing on its own — `consumeRateLimit` above
 * remains the single atomic gate — but it lets a route hand a module the REAL
 * bucket count where that module's pinned policy expects one (otp-policy.ts's
 * per-IP send cap), instead of the hardcoded 0 that quietly made the clause
 * inert on the fan-email path.
 */
export async function rateLimitCount(db: DbLike, key: string, nowMs: number, windowMs = RL_DAY_MS): Promise<number> {
  const cutoffIso = new Date(nowMs - windowMs).toISOString();
  const row = await db.prepare("SELECT count FROM rate_limits WHERE rl_key = ? AND window_start >= ?").bind(key, cutoffIso).first<{ count: number }>();
  return row?.count ?? 0;
}

/**
 * The LONGEST window any caller passes to `consumeRateLimit`. It is the free
 * creator's broadcast window (`broadcast.ts` BROADCAST_WINDOW_MS.free, one
 * week); every other key uses a day or an hour. Written here as its own
 * constant rather than imported, because the safety of the sweep below depends
 * on it being an UPPER BOUND, not on it being any particular caller's number:
 * raising a caller's window past a week without raising this one would make the
 * sweep delete rows that are still deciding something.
 */
export const RL_LONGEST_WINDOW_MS = 7 * 24 * 60 * 60 * 1000;

/**
 * Cron hygiene for `rate_limits`.
 *
 * THIS TABLE IS GROWTH-ONLY, NOT A SECRET STORE, and it is the one swept table
 * whose retention is a size argument rather than a deletion-past-expiry one.
 * It holds no identity: a key is a masked IP prefix, an account id or a
 * passport id under a fixed prefix, and a count. But it takes ONE ROW PER KEY
 * and never gives one back — the only DELETE in the tree is
 * `releaseBroadcastSlot`, which undoes a single creator's own claim — so every
 * anonymous caller that ever hit a throttled route leaves a row behind
 * permanently. `fanreg:<masked-ip>` alone means the table's steady-state size
 * is "every /24 that ever touched the fan sign-up route", forever.
 *
 * WHY DELETING IS SAFE, precisely: `consumeRateLimit` RESETS any row whose
 * `window_start` is older than the caller's window (`count = 1`, fresh
 * `window_start`) and `rateLimitCount` reports 0 for one. A row past the
 * LONGEST window in use is therefore already indistinguishable from an absent
 * row for every caller — deleting it cannot change a single allow/deny
 * decision. The margin is doubled on top of that, so a sweep that runs late,
 * or a clock that skews, still cannot reach a live row.
 *
 * The `bcast:<passport>` broadcast SLOT lives in this table too and is the one
 * key where a wrongly-dropped row would cost a real guarantee (a second
 * broadcast inside one window). It is protected by the same bound: its window
 * IS `RL_LONGEST_WINDOW_MS`, so a `bcast:` row this sweep can reach is one
 * `assertBroadcastAllowed` already counts as zero.
 */
export async function pruneRateLimits(db: DbLike, nowMs: number): Promise<number> {
  const cutoff = new Date(nowMs - 2 * RL_LONGEST_WINDOW_MS).toISOString();
  const res = await db.prepare("DELETE FROM rate_limits WHERE window_start < ?").bind(cutoff).run();
  return res.changes ?? 0;
}

/**

 * in WHOLE DAYS from the fan's qualification DAY to the day boundary T.

 * qualification INSTANT to the millisecond. Whole days carry only the
 * qualification day, and only while it is under 30 days old.
 *
 * ONE expression for the live arm, the community legs and the tombstone
 * writer (FAMILY-004(c)): the tombstone contributes exactly what the live row
 * counted. CONTRIBUTES ONE `?` — the UTC day T (`YYYY-MM-DD`).
 */
export function fanDayWeightExpr(alias = "qf"): string {
  if (!/^[A-Za-z_][A-Za-z0-9_]*$/.test(alias)) throw new Error(`ratings: not a SQL alias: ${JSON.stringify(alias)}`);
  return `MAX(0.0, MIN(1.0, (julianday(?) - julianday(substr(${alias}.qualified_at, 1, 10))) / ${FAN_WEIGHT_RAMP_DAYS}.0))`;
}

/** ISO-8601 week (UTC), e.g. "2026-W32". */
export function isoWeek(nowMs: number): string {
  const d = new Date(nowMs);
  const target = new Date(Date.UTC(d.getUTCFullYear(), d.getUTCMonth(), d.getUTCDate()));
  const dayNum = (target.getUTCDay() + 6) % 7; // Mon=0
  target.setUTCDate(target.getUTCDate() - dayNum + 3); // nearest Thursday
  const firstThursday = new Date(Date.UTC(target.getUTCFullYear(), 0, 4));
  const firstDayNum = (firstThursday.getUTCDay() + 6) % 7;
  firstThursday.setUTCDate(firstThursday.getUTCDate() - firstDayNum + 3);
  const week = 1 + Math.round((target.getTime() - firstThursday.getTime()) / (7 * DAY_MS));
  return `${target.getUTCFullYear()}-W${String(week).padStart(2, "0")}`;
}

/** #936(a) I3-1103: the instant an ISO week (as `isoWeek` spells it) closes — the next Monday 00:00 UTC. */
export function isoWeekEndMs(week: string): number {
  const jan4 = Date.UTC(Number(week.slice(0, 4)), 0, 4);
  const monday1 = jan4 - ((new Date(jan4).getUTCDay() + 6) % 7) * DAY_MS;
  return monday1 + Number(week.slice(6)) * 7 * DAY_MS;
}

function hex(b: Uint8Array): string {
  let s = "";
  for (const x of b) s += x.toString(16).padStart(2, "0");
  return s;
}

/** sha256 of a UTF-8 string, lowercase hex. The module's ONLY hashing call
 * site, routed through `@dhwaja/canonical` (single-hasher invariant). */
async function sha256Utf8(s: string): Promise<string> {
  return hex(await sha256Bytes(new TextEncoder().encode(s)));
}

/** Percentile → retro 1–7 bar. */
export function toBar(percentile: number): number {
  return Math.max(1, Math.min(7, 1 + Math.floor(percentile * 7)));
}

/** Gather every eligible creator's raw dimension values (cron only — this is
 * the full-population scan the public routes must never run).
 *
 * Eligibility is VERIFIED creators only (review finding): a merely-claimed
 * creator is a free sockpuppet who could self-redeem free credentials to pump
 * momentum/collector_base and pollute everyone's percentiles. Verified =
 * operator-vetted — gaming a rating now costs a revocable vetted identity
 * (same posture as the S24 referral economy). Claimed creators simply have no
 * Power Ratings until verified. */
async function populationValues(db: DbLike, nowMs: number): Promise<{ values: Map<string, Record<RatingDimension, number>>; scanned: number }> {
  // A SUPPRESSED creator is not eligible: a delisted person must not be ranked,
  // must not have a live ratings artifact refreshed, and must not be frozen into
  // a new weekly edition. This is the WRITE half of the suppression gate — the
  // READ half (`isSuppressed` below) is what makes an artifact written BEFORE
  // the suppression stop serving, which is the case that actually leaked.
  const { results: eligible } = await db
    .prepare(
      `SELECT passport_id FROM passports
        WHERE claim_state = 'verified' AND retracted = 0
          AND ${grantLiveAtSql("passports.passport_id")}
          AND NOT EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = passports.passport_id)`,
    )
    // I3-1768 (D-1085(1), I34-EXPIRY-A): a grant lapsed at the build instant is not eligible (standing.ts grantLiveAtSql).
    .bind(new Date(nowMs).toISOString())
    .all<{ passport_id: string }>();
  const out = new Map<string, Record<RatingDimension, number>>();
  for (const e of eligible) {
    out.set(e.passport_id, { reach: 0, momentum: 0, demand: 0, collector_base: 0, archive_depth: 0, community: 0, longevity: 0 });
  }
  const add = (psp: string, dim: RatingDimension, v: number): void => {
    const row = out.get(psp);
    if (row) row[dim] += v;
  };

  // I16 #125a: the momentum window is DAY-bounded (the first day 90 days back, fixed for the whole
  // UTC day), so its live arm and its tombstone arm partition one population (FAMILY-002 P1).
  const cutoffDay = dayOf(nowMs - 90 * DAY_MS);
  const today = dayOf(nowMs);
  // I3-0280 (D-0920 Option A): every grouped leg statement also returns `n`, the source rows it
  // aggregated; their sum is a FLOOR on the rows this scan read (DbLike drops D1's
  // meta.rows_read), alarmed at RATINGS_ROWS_READ_ALARM below.
  let scanned = 0;
  const q = async (sql: string, dim: RatingDimension, ...binds: unknown[]): Promise<void> => {
    const { results } = await db.prepare(sql).bind(...binds).all<{ psp: string; v: number; n?: number }>();
    for (const r of results) {
      add(r.psp, dim, r.v);
      scanned += Number(r.n ?? 0);
    }
  };

  // P1: A SHIELDED ACCOUNT DOES NOT COUNT TOWARD A PUBLISHED NUMBER.
  //
  // `reach` is "active accounts on the passport", and it is published as a
  // percentile ALONGSIDE the population it was computed over — and, on
  // `/p/:id/presentation`, in the same JSON response as the FILTERED roster. A
  // creator who shields one of five accounts would therefore publish a reach
  // consistent with five while showing four, and the published methodology says
  // what the dimension measures. That is a per-passport hidden-account COUNT,
  // on an anonymous edge-cached route, and it is exactly the kind of "nothing
  // Dhwaja publishes tells you there is more" the shield promises.
  //
  // Subtracted rather than filtered in SQL: the aggregate stays one grouped scan
  // (this runs over every account in the product), the shield set is one more
  // read, and the intersection is computed over the shielded keys only — which
  // is a set bounded by how many accounts creators have chosen to compartment,
  // not by the table.
  await q("SELECT passport_id AS psp, COUNT(*) AS v, COUNT(*) AS n FROM passport_accounts WHERE status = 'active' GROUP BY passport_id", "reach");
  // I3-1265: ONE exclusion set keyed by (passport, account). The two reasons
  // below overlap — an account can be shielded AND asserted-unproven — and each
  // used to subtract on its own, so the overlap cost reach 2. Both passes now
  // only COLLECT; the subtraction runs once per pair after both.
  const excluded = new Map<string, string>(); // proofPairKey -> passport_id
  const shields = await allShieldedKeys(db);
  if (shields.size > 0) {
    for (const chunk of inChunks([...shields.keys()])) {
      const { results } = await db
        .prepare(
          `SELECT passport_id AS psp, account_key AS k FROM passport_accounts
            WHERE status = 'active' AND passport_id IN (${chunk.map(() => "?").join(",")})`,
        )
        .bind(...chunk)
        .all<{ psp: string; k: string }>();
      for (const r of results) if (shields.get(r.psp)?.has(r.k)) excluded.set(proofPairKey(r.psp, r.k), r.psp);
    }
  }
  // S-036 C-52: an ASSERTED account with no CURRENT proof is exactly what the
  // public roster refuses to list (passport-view.ts:420 — `asserted_unproven`
  // and `transferred` both drop; only `proven_stale`, a proof THIS passport
  // once had, is kept). `reach`'s own methodology string ("active verified
  // accounts") already claims the narrower set; the query above counted every
  // active row instead, so a passport could publish a reach percentile
  // consistent with N accounts while the roster — the only surface that says
  // these accounts are the same person — shows fewer, the identical shape P1
  // above already fixed for shields. Same subtract-not-filter shape, same
  // reason: the aggregate stays two bulk reads (`assertedAccountKeys` is the
  // ledger-wide S101 registry-build form, `freshProofPairs` its two-bulk-read
  // proof-freshness twin) bounded by the asserted set, never a per-creator
  // `accountProofStanding` call over the whole population.
  const assertedKeys = await assertedAccountKeys(db);
  if (assertedKeys.size > 0) {
    const proven = await freshProofPairs(db, nowMs);
    for (const chunk of inChunks([...assertedKeys])) {
      const { results } = await db
        .prepare(
          `SELECT passport_id AS psp, account_key AS k FROM passport_accounts
            WHERE status = 'active' AND account_key IN (${chunk.map(() => "?").join(",")})`,
        )
        .bind(...chunk)
        .all<{ psp: string; k: string }>();
      for (const r of results) if (!proven.has(proofPairKey(r.psp, r.k))) excluded.set(proofPairKey(r.psp, r.k), r.psp);
    }
  }
  for (const psp of excluded.values()) add(psp, "reach", -1);
  // A dimension can never go negative: every excluded pair is an ACTIVE row the
  // aggregate above counted, once, so the floor is structural — the clamp stays
  // as the thing that fails loudly if that ever changes.
  for (const [, row] of out) if (row.reach < 0) row.reach = 0;
  // I16 #125a (INV-004 ERASE DIRECTION, FAMILY-002 P1): erasure DELETEs the fan's own
  // acquisitions (account_data.ts deleteAccount), so the live arm alone dropped the creator's
  // momentum the instant a holder erased. The tombstone arm (0164, kind 'acquisition') counts
  // exactly those rows back for the rest of the erasure day (`erased_day >= T`), under the SAME
  // day bound as the live arm, so an erasure moves nothing until the UTC day ends.
  await q(
    "SELECT psp, SUM(v) AS v, SUM(n) AS n FROM (" +
      "SELECT r.passport_id AS psp, COUNT(*) AS v, COUNT(*) AS n FROM acquisitions a JOIN releases r ON r.release_id = a.release_id WHERE a.acquired_at >= ? GROUP BY r.passport_id " +
      "UNION ALL " +
      "SELECT passport_id AS psp, SUM(n) AS v, COUNT(*) AS n FROM acquisition_tombstones WHERE kind = 'acquisition' AND created_day >= ? AND erased_day >= ? GROUP BY passport_id" +
      ") GROUP BY psp",
    "momentum",
    cutoffDay,
    cutoffDay,
    today,
  );
  // Docket #23 C-1 (Jason, 2026-09-07): the PUBLIC S83 demand signal day-lags
  // like the private one — INV-004 ERASE DIRECTION, one count, one partition.
  // QUEUE #134 / D-0797: only QUALIFIED requesters vote, qualified AS OF the
  // same boundary T (`analytics-read.ts readLegacyFacts` `req` is the identical
  // live predicate). The tombstone half is gated at the WRITE instead:
  // `account_data.ts deleteAccount` tombstones only qualified requesters, with
  // `created_day` = the later of the request day and the qualification day, so
  // `created_day < T` is this live arm's membership rule and erasing any

  // QUEUE #771 / D-0858(7): each requester is a VOTE WEIGHT
  // (`fanDayWeightExpr`, the boards' 30-day ramp in whole days), measured at

  await q(

      "UNION ALL " +
      "SELECT passport_id AS psp, fan_weight AS w FROM creator_request_tombstones WHERE created_day < ? AND erased_day >= ?" +
      ") GROUP BY psp",
    "demand",
    dayOf(nowMs),
    dayOf(nowMs),
    dayOf(nowMs),
    dayOf(nowMs),
    dayOf(nowMs),
  );
  // `archive_requests` is CREATOR-PRIVATE and, unlike `creator_requests` above,
  // is one row per fan, which made it
  // FINDING-0009: a creator who can induce one named fan to ask, and who can
  // read the population either side of that act, has learned that person's
  // membership. `requests.ts archiveDemand` closed that on the served count by
  // counting WHOLE UTC DAYS ONLY — but this composite was the last reader of
  // the table still counting today's rows, and a percentile is a served number
  // too: a rating artifact rebuilt after one induced request would move, on a
  // sweep the creator can wait out, at a granularity finer than a day.
  //
  // So the same predicate shape as `requests.ts:849` (the count inside
  // `archiveDemand`), bound with the BUILD's own clock — `nowMs`, the parameter this function already receives, never a
  // second `Date.now()` inside the query, because a second clock is a second
  // boundary and the two would drift apart under a slow sweep. A request made
  // today moves tomorrow's artifact and nothing built today.
  //
  // AND THE CUTOFF IS THE OBSERVED COMMIT DAY, NOT `created_at` (migration 0103,
  // QUEUE #515, Jason D-0653). `created_at` is a clock read `requestArchive`
  // took on the NEAR side of the INSERT's own `await`, so a row whose insert
  // COMMITTED after midnight carried a day that had already ended: it entered
  // this composite the moment it landed, mid-day, under a completed day. That is
  // worse HERE than at `requests.ts archiveDemand`, because this answer is BAKED
  // — `populationValues` feeds the percentile that is written into the ratings
  // artifact, so a boundary-straddled row does not merely get served early, it
  // outlives the case that produced it. `settled_day` is a day observed AFTER
  // that commit; NULL is a pre-0103 row and COALESCEs to the identical

  //
  // QUEUE #771 / D-0859(2): ONLY QUALIFIED requesters vote here too, each at
  // its day-pure age weight (`fanDayWeightExpr`) bound to the same T, like the
  // `creator_requests` arm above. The tombstone arm contributes `fan_weight`,

  // day. ONE statement and ONE ROUND over both arms (a UNION ALL, as above):
  // two separately rounded sums can differ from the rounded whole by 0.01, and
  // that difference would move at the erasure. The tombstone row still exists
  // for every requester, because `requests.ts archiveDemand` counts PEOPLE.
  await q(
    // FX-I16-W01G:4 (I16-FXC): this leg returns `n` like every other grouped leg, so its rows reach `scanned`.
    "SELECT psp, ROUND(COALESCE(SUM(w), 0), 2) AS v, COUNT(*) AS n FROM (" +
      `SELECT a.passport_id AS psp, ${fanDayWeightExpr("qf")} AS w FROM archive_requests a ${qualifiedFanJoin("a.account_id")} WHERE COALESCE(a.settled_day, substr(a.created_at, 1, 10)) < ? AND ${fanQualifiedAsOfExpr("qf")} ` +
      "UNION ALL " +
      "SELECT passport_id AS psp, fan_weight AS w FROM archive_request_tombstones WHERE created_day < ? AND erased_day >= ?" +
      ") GROUP BY psp",
    "demand",
    dayOf(nowMs),
    dayOf(nowMs),
    dayOf(nowMs),
    dayOf(nowMs),
    dayOf(nowMs),
  );
  // CO-04: JOIN accounts — the dimension is "distinct ACCOUNTS holding at least
  // one verified acquisition" (the methodology string above), and after an
  // erasure a retained row's `account_id` is not an account at all. Erasure
  // DELETEs the holder's own acquisitions but TOMBSTONES the ones somebody else
  // redeemed (deleteAccount's `UPDATE acquisitions SET account_id = 'del_' …`), giving EACH retained row its own
  // `del_<random>` holder id so the redeemer's evidence survives unlinkably.
  // A `COUNT(DISTINCT account_id)` therefore turned one erased holder of N such
  // rows into N phantom collectors, and erasing a fan RAISED the creator's
  // collector base. The dedup key is the exact column erasure randomises, so
  // the count has to be taken over LIVE owners, not over ids: a tombstone joins
  // nothing. Same "require a live owner" shape the intelligence/ROI collector
  // stock already gets from its erasure-decremented day counters
  // (`analytics-read.ts:351`) — this was the one holder metric still reading
  // the property table directly.
  // I3-0920 (D-0920 Option A) + I16 #125b: only QUALIFIED fans are collectors — a burst of
  // throwaway accounts that never qualified adds 0 — qualified AS OF the day boundary T and
  // holding an acquisition made before T, so the live arm is day-pure like demand's. Erasure
  // removes every acquisition the holder had from this arm at once; the tombstone arm (0164, kind
  // 'holder', one row per passport, created_day = the later of the first acquisition day and the
  // qualification day) holds them for the rest of the erasure day. A holder is in exactly one arm.
  await q(
    "SELECT psp, SUM(v) AS v, SUM(n) AS n FROM (" +
      // I3-1089 (I34-W2-EXPIRY-B): a gifted row keeps its original `acquired_at` (gifting.ts moves `account_id` only), so
      // a moved row is dated by its holder's receipt — the latest `acquisition_log` entry to that owner, the shape
      // social.ts's featured read uses — or the recipient is counted before they held it.
      `SELECT r.passport_id AS psp, COUNT(DISTINCT a.account_id) AS v, COUNT(*) AS n FROM acquisitions a JOIN releases r ON r.release_id = a.release_id JOIN accounts ac ON ac.account_id = a.account_id ${qualifiedFanJoin("a.account_id")} WHERE COALESCE((SELECT MAX(l.at) FROM acquisition_log l WHERE l.acquisition_id = a.acquisition_id AND l.to_owner = a.account_id), a.acquired_at) < ? AND ${fanQualifiedAsOfExpr("qf")} GROUP BY r.passport_id ` +
      "UNION ALL " +
      "SELECT passport_id AS psp, SUM(n) AS v, COUNT(*) AS n FROM acquisition_tombstones WHERE kind = 'holder' AND created_day < ? AND erased_day >= ? GROUP BY passport_id" +
      ") GROUP BY psp",
    "collector_base",
    today,
    today,
    today,
    today,
  );
  await q("SELECT passport_id AS psp, COUNT(*) AS v, COUNT(*) AS n FROM releases WHERE status = 'published' GROUP BY passport_id", "archive_depth");
  // I3-0321: qualified fans only — every other public board (rankings.ts,
  // requests.ts) gates its fan-count boards through `qualified_fan_accounts`;
  // this was the one still reading `saves`/`card_collections` raw, so a burst
  // of zero-day accounts could pump the published `community` percentile with
  // no ramp and no identity cost.
  // QUEUE #771 / D-0858(7): age-weighted like the boards, weight measured at
  // the UTC-day boundary T (the demand arm's instant; one instant per card).
  // QUEUE #793 (D-0797 partition, migration 0157): the same two-arm shape as
  // the `creator_requests` demand arm above. The live arm is DAY-PURE — row
  // made before T, fan qualified before T — so `created_day < T` is its

  // `community_tombstones` row per counted row with `created_day` = the later

  for (const [table, alias, kind] of [
    ["saves", "s", "save"],
    ["card_collections", "cc", "collect"],

      "SELECT psp, ROUND(COALESCE(SUM(w), 0), 2) AS v, COUNT(*) AS n FROM (" +

AND ${fanQualifiedAsOfExpr("qf")} ` +
        "UNION ALL " +
        `SELECT passport_id AS psp, fan_weight AS w FROM community_tombstones WHERE kind = '${kind}' AND created_day < ? AND erased_day >= ?` +
        ") GROUP BY psp",
      "community",
      dayOf(nowMs),
      dayOf(nowMs),
      dayOf(nowMs),
      dayOf(nowMs),
      dayOf(nowMs),
    );
  }
  // Longevity: days since first ledger event for the subject.
  const { results: births } = await db
    .prepare("SELECT subject AS psp, MIN(created_at) AS born FROM trust_events WHERE event_type = 'passport_created' GROUP BY subject")
    .all<{ psp: string; born: string }>();
  for (const b of births) {
    const row = out.get(b.psp);
    if (row) row.longevity = Math.max(0, Math.floor((nowMs - Date.parse(b.born)) / DAY_MS));
  }
  // I3-1060: the narrow demand opt-out (`demand_optouts`, migration 0017) is
  // "this subject stays listed and checkable; the public counter about them
  // comes off" (LISTING-POLICY.md §4.4, passport-view.ts `livePassportWithDemand`
  // docblock) — `requestCard`/`creatorDemand` already gate their per-passport
  // demand counters on it. This aggregate is the same public demand signal,
  // over the same table, and never read the opt-out at all: one bulk read
  // (the whole opt-out list is expected to be far smaller than the eligible
  // population, same shape as the shield/asserted bulk reads above) zeroes the
  // one dimension the lever is meant to silence, and only that one — reach,
  // collector_base etc. must keep counting this creator normally.
  const { results: optOuts } = await db.prepare("SELECT passport_id FROM demand_optouts").all<{ passport_id: string }>();
  for (const o of optOuts) {
    const row = out.get(o.passport_id);
    if (row) row.demand = 0;
  }
  // I3-0280 (D-0920 Option A): full-table reads stay for launch; past the threshold a person
  // decides on incremental counters (QUEUE #885/#912 residual).
  if (scanned >= RATINGS_ROWS_READ_ALARM) logError({ cid: "unknown", hop: "cron", msg: "ratings_rows_read_over_threshold", count: scanned, cap: RATINGS_ROWS_READ_ALARM });
  return { values: out, scanned };
}

/** Rank one dimension: percentile = fraction of the population STRICTLY below. */
function percentiles(values: Array<[string, number]>): Map<string, number> {
  const sorted = [...values].sort((a, b) => a[1] - b[1]);
  const n = sorted.length;
  const out = new Map<string, number>();
  let i = 0;
  while (i < n) {
    let j = i;
    while (j < n && sorted[j]![1] === sorted[i]![1]) j++;
    const pct = n <= 1 ? 0 : i / n; // ties share the lower bound (no rank inflation)
    for (let k = i; k < j; k++) out.set(sorted[k]![0], pct);
    i = j;
  }
  return out;
}

const LIST_PAGE = 1000;

const SHARD_PREFIX = "ratings/_live/";
/** A passport id is `psp_` + 16 hex; its first hex nibble picks the shard. */
export function ratingsShardKey(passportId: string): string {
  return `${SHARD_PREFIX}${passportId.charAt(4)}.json`;
}
interface RatingsShard {
  generated_at: string;
  population: number;
  methodology_version: string;
  cards: Record<string, PowerRatings>;
}

export interface RatingsBuildResult {
  computed: Map<string, PowerRatings>; // full population (for tests/inspection)
  batch: Map<string, PowerRatings>; // the slice actually written this sweep
  /** FX-I16-W01G:4: the source-row floor the scan summed (`scanned`, compared with RATINGS_ROWS_READ_ALARM). */
  rows_read: number;
}

/**
 * Cron: compute percentiles over the full eligible population, publish ALL of
 * it as the RATINGS_SHARDS live shards, then write the per-creator artifacts
 * for the next RATINGS_BATCH creators in cursor order.
 * Returns both the full computed map and the written batch. The six-hourly
 * sweep freezes this batch's editions at once; `freezeWeeklyEditionsStep`
 * freezes the rest of the population from the live shards (QUEUE #774).
 */
export async function buildPowerRatings(
  db: DbLike,
  store: ArtifactStore,
  nowMs: number,
  now: () => string = () => new Date().toISOString(),
  batchSize: number = RATINGS_BATCH,
  /** I14-CAPS (QUEUE #912): the leg's cap the batch is sized from (RATINGS_LEG_CAP). */
  legCap: number = RATINGS_LEG_CAP,
): Promise<RatingsBuildResult> {
  // I14-CAPS: populationValues executes each statement it prepares exactly once.
  let spent = 0;
  const { values, scanned } = await populationValues({ prepare: (sql: string) => (spent++, db.prepare(sql)), batch: (sts) => db.batch(sts) }, nowMs);
  const population = values.size;
  const perDim = new Map<RatingDimension, Map<string, number>>();
  for (const dim of RATING_DIMENSIONS) {
    perDim.set(dim, percentiles([...values.entries()].map(([psp, v]) => [psp, v[dim]])));
  }
  const ts = now();
  const computed = new Map<string, PowerRatings>();
  for (const psp of values.keys()) {
    const ratings = {} as Record<RatingDimension, DimensionRating>;
    for (const dim of RATING_DIMENSIONS) {
      const pct = perDim.get(dim)!.get(psp) ?? 0;
      ratings[dim] = { percentile: Math.round(pct * 1000) / 1000, bar: toBar(pct) };
    }
    computed.set(psp, { passport_id: psp, generated_at: ts, population, methodology_version: METHODOLOGY_VERSION, ratings });
  }

  // ALWAYS CURRENT: every shard is rewritten every sweep — an empty one too, so
  // a creator this sweep excluded (suppressed, retracted, unverified) is gone
  // from rest rather than lingering in last sweep's shard.
  const shards = new Map<string, RatingsShard>();
  for (let n = 0; n < RATINGS_SHARDS; n++) {
    shards.set(`${SHARD_PREFIX}${n.toString(16)}.json`, { generated_at: ts, population, methodology_version: METHODOLOGY_VERSION, cards: {} });
  }
  for (const [psp, card] of computed) {
    const shard = shards.get(ratingsShardKey(psp)); // an id outside `psp_<hex>` is unservable (serveRatings' regex) — no shard
    if (shard) shard.cards[psp] = card;
  }
  for (const [key, shard] of shards) await store.put(key, JSON.stringify(shard));
  spent += shards.size;
  // I25-FX-DATA (I3-1103 b): the population these shards now publish, recorded per ISO week of `nowMs` — AFTER the
  // shard puts, overwritten by every sweep that week, so the LAST sweep before the close leaves the population as of
  // the close. The anchor compares the week's complete freeze cursor with it (`weekFreezeState`), never with a
  // population a post-close sweep published. Same spelling as `liveShardPopulation` (`populationHash`).
  const published: string[] = [];
  for (const shard of shards.values()) published.push(...Object.keys(shard.cards));
  await store.put(weekPopulationKey(isoWeek(nowMs)), JSON.stringify({ population: await populationHash(published.sort()), generated_at: ts }));
  spent += 1;

  // I3-1154: MISSING-ARTIFACT PRIORITY. Plain lexicographic rotation makes a
  // newly-verified creator wait up to ceil(population/batchSize) sweeps for
  // their FIRST artifact, same as everyone else's refresh — but "never rated
  // yet" and "due for a refresh" are different urgencies (a creator with no
  // `ratings/<psp>.json` shows `not_rated_yet` on every public surface in the
  // meantime). One `list()` call (bounded — the same one-listing shape
  // `readWeeklySet` already uses, not a per-creator `get`, so it costs
  // nothing against the sweep's write-only subrequest budget) partitions the
  // population into never-written and already-written, and never-written
  // creators rotate first. Once written, a creator falls into the ordinary
  // lexicographic rotation for subsequent refreshes — this changes WHO GOES
  // FIRST, never the steady-state rotation shape.
  const writtenIds = new Set(
    (await store.list("ratings/"))
      .filter((k) => k !== CURSOR_KEY && k.endsWith(".json"))
      .map((k) => k.slice("ratings/".length, -".json".length)),
  );
  const allSorted = [...computed.keys()].sort();
  const ordered = [...allSorted.filter((p) => !writtenIds.has(p)), ...allSorted.filter((p) => writtenIds.has(p))];
  // Rotate the write window: pick up after the last-written passport.
  let startIdx = 0;
  const cursorRaw = await store.get(CURSOR_KEY);
  if (cursorRaw) {
    try {
      const cursor = (JSON.parse(cursorRaw) as { after?: string }).after ?? "";
      const i = ordered.findIndex((p) => p > cursor);
      startIdx = i === -1 ? 0 : i;
    } catch {
      startIdx = 0;
    }
  }
  // I14-CAPS: the list's pages and the cursor get are spent; the cursor put and the anchor's allowance are held.
  spent += Math.max(1, Math.ceil((writtenIds.size + 1) / LIST_PAGE)) + 1;
  let fit = Math.max(0, Math.floor((legCap - spent - 1 - RATINGS_ANCHOR_ALLOWANCE) / RATINGS_CREATOR_MAX));
  // I14-FX1 (RG-CAPS-02, D-0905(1)): THE LEG ALWAYS MAKES PROGRESS OR ALERTS. The population reads
  // above grow with the ledger (one statement per IN_CHUNK asserted keys), so past ~15k keys they
  // alone leave no room under the cap and the batch was 0 forever: nobody new rated, nobody
  // refreshed. One creator is still written (the rotation covers the population in `population`
  // runs) and the overrun is the alarm, so the cap is raised or the reads bounded by a person.
  if (fit < 1 && ordered.length > 0 && batchSize > 0) {
    logError({ cid: "unknown", hop: "cron", msg: "ratings_leg_over_cap", count: population, detail: `spent ${spent} of cap ${legCap} before the batch` });
    fit = 1;
  }
  const batch = new Map<string, PowerRatings>();
  let lastWritten: string | null = null;
  for (let i = 0; i < Math.min(batchSize, fit, ordered.length); i++) {
    const psp = ordered[(startIdx + i) % ordered.length]!;
    if (batch.has(psp)) break; // wrapped the whole population
    batch.set(psp, computed.get(psp)!);
    await store.put(`ratings/${psp}.json`, JSON.stringify(computed.get(psp)!));
    lastWritten = psp;
  }
  // The cursor is the last passport written in ROTATION order; the next sweep
  // resumes after it (findIndex(> cursor) wraps to 0 past the end).
  // I14-CAPS: a run the cap left no room for writes nobody and keeps the rotation where it was.
  if (lastWritten !== null || fit > 0 || ordered.length === 0) await store.put(CURSOR_KEY, JSON.stringify({ after: lastWritten ?? "", updated_at: ts }));
  return { computed, batch, rows_read: scanned };
}

/**
 * THE SUPPRESSION GATE FOR EVERY SUBJECT-SCOPED RATINGS ARTIFACT.
 *
 * `removal.ts` commits in writing that "SUPPRESSION IS INDISTINGUISHABLE FROM
 * NEVER-LISTED". It was NOT true for this module (measured): the three artifact
 * routes — `GET /p/:id/card/ratings`, `GET /p/:id/card/editions` and
 * `GET /p/:id/card/editions/:week` — answered 200 for a suppressed creator whose
 * artifacts had been written before the delisting, while a never-listed id got
 * 404. That 200-vs-404 split is itself the "did this person ask to be delisted?"
 * probe, and the frozen weekly edition carries the subject's DISPLAY NAME in its
 * bytes, so the worst of the three leaked the name too.
 *
 * WHY THE THREE `serve*` FUNCTIONS TAKE A `db` NOW, AND WHY IT IS REQUIRED.
 * The object store cannot answer "is this subject suppressed" — the suppression
 * lives in `request_suppressions`, and an artifact is immutable and may predate
 * the row by weeks, so a write-time filter alone can never close this (it does
 * not un-write what is already in the bucket). The gate therefore has to be a
 * LIVE read at serve, which means these functions need a database handle. It is
 * a REQUIRED POSITIONAL parameter, in the `(store, db, passportId)` order
 * `rankings.ts serveTopFans` already uses, precisely so it cannot be bypassed:
 * a future caller that forgets it does not silently serve an ungated artifact,
 * it fails `pnpm typecheck`. An optional `db?` would have made the leak the
 * DEFAULT for every new call site.
 *
 * WHY HERE AND NOT IN THE ROUTES. These three functions are the only readers of
 * `ratings/<psp>.json`, `editions/<psp>/index.json` and
 * `editions/<psp>/<week>.json` anywhere in the service. Gating the three routes
 * would have left `share-card.ts` (the live card SVG's ratings bars) and
 * `digest.ts` (the creator's weekly email) reading them ungated, and every
 * future consumer one forgotten line away from re-opening the hole.
 */
async function isSuppressed(db: DbLike, passportId: string, nowIso: string): Promise<boolean> {
  // S-026/C-152: `populationValues` (the WRITE half, above) already excludes
  // `retracted = 1` from who gets a NEW live artifact refreshed or frozen into
  // a new weekly edition. This READ half used to check only
  // `request_suppressions`, so a LIVE ratings artifact or a weekly edition
  // written before a retraction — immutable, same as the suppression case this
  // function's docblock describes — kept serving the erased name forever: the
  // write-time gate stopped new leaks, nothing stopped the old ones. Retracted
  // and suppressed are deliberately independent controls (passport-view.ts),
  // so a passport can be retracted without ever being suppressed and this
  // clause is not redundant with the one beside it.
  //
  // I3-1268: `claim_state <> 'verified'` is the READ half of the eligibility
  // `populationValues` writes with (verified only). A creator downgraded after
  // an artifact was written (`verification_expired`, `downgrade_verification`
  // -> events.ts statePassport) kept serving — and, through `serveRatings`, kept
  // counting as rated in war-match, packs, credits, share-card and digest. The
  // stored artifact is untouched; re-verification serves it again.
  // I3-1768 (D-1085(1), I34-EXPIRY-A): so is a grant lapsed at the serve instant, before the expiry sweep folds it back.
  const row = await db
    .prepare(
      `SELECT 1 AS x FROM passports p WHERE p.passport_id = ? AND (p.retracted = 1 OR p.claim_state <> 'verified' OR NOT (${grantLiveAtSql("p.passport_id")}))
       UNION SELECT 1 FROM request_suppressions r WHERE r.passport_id = ?`,
    )
    .bind(passportId, nowIso, passportId)
    .first<{ x: number }>();
  return row !== null;
}

/**
 * CONSTANT SHAPE — the gate and the object read ALWAYS BOTH RUN, in this fixed
 * order, and only then does anything branch.
 *
 * Written the obvious way (`if (await isSuppressed(...)) return null;` before
 * the read) these three functions cost a suppressed subject ZERO object reads
 * where a never-listed one cost a miss: same D1 sequence, same status, same
 * bytes, one whole network round trip apart, with SUPPRESSED being the FASTER
 * answer. Nothing about the responses could show it and the statement-sequence
 * guard recorded D1 only, so it measured as identical (review MEDIUM 3, this
 * pass). `rankings.ts serveTopFans` had already been given this shape for the
 * same reason; these three had not.
 *
 * RESIDUAL, STATED (the same one `serveTopFans` records): the object read is a
 * HIT for a subject that was rated before it was suppressed and a MISS for one
 * that was never rated, and no ordering fixes that — only serving a decoy
 * artifact for every id would, which is the registry's problem and not this
 * route's. What is removed here is the difference an attacker can measure
 * without ever having to calibrate a hit against a miss.
 */
async function gatedArtifact(store: ArtifactReader, db: DbLike, passportId: string, key: string): Promise<string | null> {
  const suppressed = await isSuppressed(db, passportId, new Date().toISOString());
  const raw = await store.get(key);
  return suppressed ? null : raw;
}

/** Serve a creator's live ratings artifact (O(1) read; absent or SUPPRESSED →
 * null, which every caller already renders as the never-listed answer: 404 on
 * the route, no ratings bars on the card, no ratings block in the digest). */
export async function serveRatings(
  store: ArtifactReader,
  db: DbLike,
  passportId: string,
  /** JN-02 (I34-W2-EXPIRY-B): the door's own instant for the lapsed-grant term; the wall clock when a caller has none. */
  nowIso: string = new Date().toISOString(),
): Promise<PowerRatings | null> {
  if (!/^psp_[0-9a-f]{16}$/.test(passportId)) return null;
  // Same fold as `gatedArtifact`, same constant shape: the gate and the shard
  // read both run before anything branches on the gate. The per-creator
  // fallback depends only on the shard's content, never on suppression.
  const suppressed = await isSuppressed(db, passportId, nowIso);
  let raw: string | null = null;
  const shardRaw = await store.get(ratingsShardKey(passportId));
  if (shardRaw) {
    try {
      const card = (JSON.parse(shardRaw) as RatingsShard).cards?.[passportId];
      if (card) raw = JSON.stringify(card);
    } catch {
      raw = null;
    }
  }
  // QUEUE #778 / D-0858(14): the per-creator object is read UNCONDITIONALLY.
  // Reading it only on a shard miss made a creator hidden since the last sweep
  // (still in its shard) cost one read and a never-listed id two — a latency
  // oracle over the sweep window. Gate, shard, object: one shape for every
  // well-formed id; the shard card still wins. Residual as `gatedArtifact`.
  const objectRaw = await store.get(`ratings/${passportId}.json`);
  if (raw === null) raw = objectRaw;
  if (suppressed || !raw) return null;
  try {
    return { ...(JSON.parse(raw) as PowerRatings), methodology: METHODOLOGY };
  } catch {
    return null;
  }
}

export interface EditionIndexEntry {
  week: string;
  content_hash: string;
}

/**
 * Cron (weekly): freeze this week's card for every rated creator. An edition
 * is written ONCE — if the key exists it is never touched again (old editions
 * never change, even if stats or methodology later do). The content_hash is
 * sha256 over the canonical (sorted-key) JSON of the edition minus the hash
 * itself, so any holder can independently re-verify a saved edition.
 */
export async function buildWeeklyEditions(
  db: DbLike,
  store: ArtifactStore,
  ratingsByPsp: Map<string, PowerRatings>,
  nowMs: number,
  now: () => string = () => new Date().toISOString(),
): Promise<number> {
  const week = isoWeek(nowMs);
  let written = 0;
  for (const [psp, ratings] of ratingsByPsp) {
    const key = `editions/${psp}/${week}.json`;
    const existing = await store.get(key);
    if (existing !== null) {
      // Immutable — never rewrite. But REPAIR the index if a prior sweep
      // crashed between the edition put and the index put (review finding:
      // an orphaned edition would otherwise stay missing from the index
      // forever, because this branch used to skip straight past it).
      try {
        const ed = JSON.parse(existing) as CardEdition;
        await ensureIndexEntry(store, psp, week, ed.content_hash);
      } catch {
        // corrupt stored edition — leave for operator forensics, never overwrite
      }
      continue;
    }
    // I3-1768 (D-1084(1), I32-FX2-SERVER): the class is read AT THE EDITION'S OWN INSTANT (`nowMs`): a `verified` row
    // whose latest grant or renewal has lapsed by then (standing.ts `grantLiveAtSql`) is frozen as the expiry sweep's
    // fold will draw it — `claimed`, with no Founding rank — the rule cards.ts applies to the live card.
    const p = await db
      .prepare(
        `SELECT display_name, ${shownClassSql("passports")} AS claim_state,
                claim_state AS stored_state, retracted FROM passports WHERE passport_id = ?`,
      )
      .bind(new Date(nowMs).toISOString(), psp)
      .first<{ display_name: string; claim_state: string; stored_state: string; retracted: number }>();
    if (!p || p.retracted === 1) continue;
    // I3-0910: read through `foundingRank`, the gate every live surface uses — a
    // rank withheld while the passport is not live+verified is not frozen here.
    const lapsed = p.stored_state === "verified" && p.claim_state !== "verified";
    const foundingRankNow = lapsed ? null : await foundingRank(db, psp, new Date(nowMs).toISOString()); // the edition's instant (I34-EXPIRY-A)
    // M56: a FOUNDING creator's frozen edition must wear the founding badge, not
    // the plain claim_state — same derivation as the live card (cards.ts): a
    // served founding rank promotes the claim_state to "founding-*". Storing
    // the bare claim_state dropped the founding distinction from every edition.
    const editionBadge = foundingRankNow !== null ? `founding-${p.claim_state}` : p.claim_state;
    const base = {
      edition_id: `${psp}@${week}`,
      week,
      passport_id: psp,
      display_name: p.display_name,
      badge: editionBadge,
      founding_rank: foundingRankNow,
      // ONE seed function with the live card (cards.ts) — same renderer, same
      // art. A slice of the passport id also embedded an ID FRAGMENT in the
      // frozen, permanently-published edition body; the hash does not.
      palette_seed: paletteSeed(psp),
      generated_at: now(),
      population: ratings.population,
      methodology_version: ratings.methodology_version,
      ratings: ratings.ratings,
    };
    const contentHash = await sha256Utf8(stableStringify(base));
    const edition: CardEdition = { ...base, content_hash: contentHash } as CardEdition;
    // AN EDITION IS IMMUTABLE, and the WRITE is what enforces it — the `get`
    // above is a fast path, not a guard. `generated_at` is wall-clock and it is
    // inside the hashed body, so two overlapping sweeps used to mint two
    // DIFFERENT editions for the same creator and week and the survivor was
    // decided by put ordering (measured: both sweeps returned written=1 and the
    // key was put twice). That is not an ordinary lost update: the anchored leaf
    // commits to a content_hash, and the arbitrary survivor can be the one the
    // anchor does not name — an edition that fails end-to-end verification
    // forever, because the anchor is on the ledger and cannot be corrected.
    // I3-1883: INTENT FIRST. The week's set member is written BEFORE the edition,
    // so no fault can leave a served edition that the week's anchor never names.
    // A member whose edition then never lands is a phantom, and a member left
    // holding a racing loser's hash is stale: `readWeeklySet` takes every
    // member's hash from the stored edition itself, so neither reaches an anchor.
    await store.put(weeklySetEntryKey(week, psp), JSON.stringify({ content_hash: contentHash }));
    if (!(await store.putIfAbsent(key, JSON.stringify(edition)))) {
      // Somebody else got there first. THEIR bytes are the edition; index from
      // those, never from ours — same repair the existing-object branch does.
      const stored = await store.get(key);
      if (stored !== null) {
        try {
          await ensureIndexEntry(store, psp, week, (JSON.parse(stored) as CardEdition).content_hash);
        } catch {
          // corrupt stored edition — leave for operator forensics, never overwrite
        }
      }
      continue;
    }
    await ensureIndexEntry(store, psp, week, contentHash);
    written++;
  }
  return written;
}

/** QUEUE #774 (D-0858(10)): creators one analytics tick freezes. The step shares
 * that invocation's ~1000-subrequest budget with the outbox and erasure drains;
 * a fresh edition costs ~6 subrequests, so 25 is ~170 with the shard reads, and
 * 2,016 ticks a week cover ~50k creators (the six-hourly batches alone: 2,800). */
export const EDITIONS_STEP_BUDGET = 25;
/** Per-week cursor. Under `editions/`, never `ratings/`: `buildPowerRatings`
 * reads every `ratings/*.json` key as a written creator. */
function editionsCursorKey(week: string): string {
  return `editions/_cursor/${week}.json`;
}
interface EditionsCursor {
  after: string;
  pass_written: number;
  complete: boolean;
  /** sha256 of the sorted shard population the current pass started on (D-0861(9)). */
  population: string;
}

/** I25-FX-DATA (I3-1103 b): the population `buildPowerRatings` last published in ISO week `week` (see there). */
function weekPopulationKey(week: string): string {
  return `editions/_population/${week}.json`;
}
/** sha256 of sorted passport ids joined by newlines — the ONE population spelling (cursor, live pass, week record). */
async function populationHash(sorted: readonly string[]): Promise<string> {
  return sha256Utf8(sorted.join("\n"));
}

/** The live shard population: every card in the RATINGS_SHARDS live shards, its sorted ids, and `population` = sha256 of
 * those ids joined by newlines. ONE spelling for the freeze (which stores it in the cursor) and the anchor (I3-1103 b,
 * which compares it), so the two compare the identical value. Spends RATINGS_SHARDS store gets. */
async function liveShardPopulation(store: ArtifactStore): Promise<{ cards: Map<string, PowerRatings>; sorted: string[]; population: string }> {
  const cards = new Map<string, PowerRatings>();
  for (let n = 0; n < RATINGS_SHARDS; n++) {
    const shardRaw = await store.get(`${SHARD_PREFIX}${n.toString(16)}.json`);
    if (shardRaw === null) continue;
    try {
      for (const [psp, card] of Object.entries((JSON.parse(shardRaw) as RatingsShard).cards ?? {})) cards.set(psp, card);
    } catch {
      // a corrupt shard contributes nobody; the next sweep rewrites it whole
    }
  }
  const sorted = [...cards.keys()].sort();
  return { cards, sorted, population: await populationHash(sorted) };
}

/**
 * QUEUE #774: freeze this week's edition for EVERY creator the last sweep
 * published, `budget` creators per call. The population is the RATINGS_SHARDS
 * live shards (already suppression/retraction/verification filtered by the
 * sweep); each call hands the next `budget` ids after the cursor, in sorted
 * order, to `buildWeeklyEditions` (which skips an already-frozen creator). A
 * pass that reaches the end of the population and froze nothing, over the same
 * population it started on, marks the week complete; a later call that week
 * stops only while the shard population is unchanged, and re-opens the week when
 * it changed (a creator verified mid-week — D-0861(9), RGOPS-774).
 */
export async function freezeWeeklyEditionsStep(
  db: DbLike,
  rawStore: ArtifactStore,
  nowMs: number,
  budget: number = EDITIONS_STEP_BUDGET,
  now: () => string = () => new Date().toISOString(),
  // I38-TICK-5: the withCronLease guard of the caller (index.ts "ratings-editions"); every store write below — each edition,
  // its set entry and index, and the cursor — reads it first (`leaseGuardedStore`), and a lost lease ends the step there.
  lease?: CronLeaseGuard,
): Promise<{ frozen: number; complete: boolean }> {
  const store = leaseGuardedStore(rawStore, lease);
  try {
    return await freezeWeeklyEditionsStepBody(db, store, nowMs, budget, now);
  } catch (e) {
    if (e instanceof StoreLeaseLost) return { frozen: 0, complete: false };
    throw e;
  }
}

/** I38-TICK-5: thrown by a `leaseGuardedStore` write once the lease is lost; the step that wrapped the store catches it. */
class StoreLeaseLost extends Error {}
/** I38-TICK-5: `store` with the lease guard read before every write (put, putIfAbsent, delete); reads pass through. */
function leaseGuardedStore(store: ArtifactStore, lease: CronLeaseGuard | undefined): ArtifactStore & { delete?: (k: string) => Promise<void> } {
  const refuse = (): void => {
    if (lease?.lost === true) throw new StoreLeaseLost("cron lease lost: this run stops before its next write");
  };
  const del = (store as ArtifactStore & { delete?: (k: string) => Promise<void> }).delete;
  return {
    get: (k) => store.get(k),
    list: (p) => store.list(p),
    put: async (k, v) => {
      refuse();
      return store.put(k, v);
    },
    putIfAbsent: async (k, v) => {
      refuse();
      return store.putIfAbsent(k, v);
    },
    ...(typeof del === "function"
      ? {
          delete: async (k: string) => {
            refuse();
            return del.call(store, k);
          },
        }
      : {}),
  };
}

async function freezeWeeklyEditionsStepBody(
  db: DbLike,
  store: ArtifactStore,
  nowMs: number,
  budget: number,
  now: () => string,
): Promise<{ frozen: number; complete: boolean }> {
  const cursorKey = editionsCursorKey(isoWeek(nowMs));
  let cursor: EditionsCursor = { after: "", pass_written: 0, complete: false, population: "" };
  const raw = await store.get(cursorKey);
  if (raw !== null) {
    try {
      const c = JSON.parse(raw) as Partial<EditionsCursor>;
      cursor = {
        after: typeof c.after === "string" ? c.after : "",
        pass_written: Number(c.pass_written) || 0,
        complete: c.complete === true,
        population: typeof c.population === "string" ? c.population : "",
      };
    } catch {
      // unreadable cursor: restart the pass; frozen editions are skipped, never rewritten
    }
  }
  const { cards, sorted, population } = await liveShardPopulation(store);
  if (sorted.length === 0) return { frozen: 0, complete: false };
  if (cursor.complete) {
    if (cursor.population === population) return { frozen: 0, complete: true };
    cursor = { after: "", pass_written: 0, complete: false, population: "" }; // population changed: re-open the week
  }
  if (cursor.after === "") cursor = { after: "", pass_written: 0, complete: false, population };
  const slice = sorted.filter((p) => p > cursor.after).slice(0, budget);
  // The shards can be up to one sweep old: re-apply the sweep's own eligibility
  // (populationValues) LIVE, so a creator suppressed, retracted or unverified
  // since is not frozen into a new edition.
  const eligible = new Set<string>();
  // I3-1768 (D-1084(1)): and a grant that has lapsed at the edition's instant (`grantLiveAtSql`) is not frozen either.
  for (const chunk of inChunks(slice, 1)) {
    const { results } = await db
      .prepare(
        `SELECT passport_id FROM passports
          WHERE passport_id IN (${chunk.map(() => "?").join(", ")}) AND claim_state = 'verified' AND retracted = 0
            AND ${grantLiveAtSql("passports.passport_id")}
            AND NOT EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = passports.passport_id)`,
      )
      .bind(...chunk, new Date(nowMs).toISOString())
      .all<{ passport_id: string }>();
    for (const r of results) eligible.add(r.passport_id);
  }
  const toFreeze = new Map(slice.filter((p) => eligible.has(p)).map((p) => [p, cards.get(p)!] as const));
  const frozen = toFreeze.size === 0 ? 0 : await buildWeeklyEditions(db, store, toFreeze, nowMs, now);
  const passWritten = cursor.pass_written + frozen;
  const endOfPass = slice.length === 0 || slice[slice.length - 1] === sorted[sorted.length - 1];
  const next: EditionsCursor = endOfPass
    ? { after: "", pass_written: 0, complete: passWritten === 0 && cursor.population === population, population }
    : { after: slice[slice.length - 1]!, pass_written: passWritten, complete: false, population: cursor.population };
  await store.put(cursorKey, JSON.stringify(next));
  return { frozen, complete: next.complete };
}

/** The prefix every member of one week's edition set lives under. */
function weeklySetPrefix(week: string): string {
  return `editions/weekly-sets/${week}/`;
}
/** ONE OBJECT PER MEMBER. The week's set used to be a single passport-keyed
 * map that every writer read, mutated and put back — a read-modify-write
 * justified in-comment by "cron is the only writer", which Cloudflare does not
 * guarantee: overlapping scheduled invocations are possible and a RETRY of a
 * failed invocation is routine. Reproduced with a store that holds every
 * reader on that get: two sweeps each freezing one creator produced an
 * anchored set of ONE. That is worse than an ordinary lost update, because the
 * ledger anchor then asserts that the incomplete set is the whole week.
 *
 * A key per member removes the shared cell the update was lost in: two writers
 * touch two keys, a retry rewrites its own key with the same bytes (editions
 * are immutable, so the hash cannot change), and the set is DERIVED at anchor
 * time by listing the prefix instead of accumulated. No lease, no
 * single-writer assumption, nothing the platform has to promise. */
function weeklySetEntryKey(week: string, psp: string): string {
  return `${weeklySetPrefix(week)}${psp}.json`;
}

/** Record (week, psp) -> content_hash for the anchor, and append (week, hash)
 * to the per-creator edition index. The repair path above re-runs this for
 * orphaned editions after a crashed sweep. */
async function ensureIndexEntry(store: ArtifactStore, psp: string, week: string, contentHash: string): Promise<void> {
  // Write the SET MEMBER first — it is the anchor's preimage, and a crash
  // after only the per-creator index was written would exclude a real edition
  // from the anchored set forever (past weeks are never re-swept; review
  // finding). A member-then-crash gap in the INDEX, by contrast, self-repairs
  // on the next current-week sweep. The put is unconditional because it is
  // idempotent: the same (week, psp) always carries the same immutable hash.
  await store.put(weeklySetEntryKey(week, psp), JSON.stringify({ content_hash: contentHash }));
  const idxKey = `editions/${psp}/index.json`;
  const idxRaw = await store.get(idxKey);
  let idx: EditionIndexEntry[] = [];
  if (idxRaw) {
    try {
      idx = JSON.parse(idxRaw) as EditionIndexEntry[];
    } catch {
      idx = [];
    }
  }
  if (!idx.some((e) => e.week === week)) {
    // I3-2010: the index is still one read-modify-write cell, so a writer that
    // read it before another week landed can put back a copy without that week.
    // Every GROWTH therefore rebuilds from the source of truth — the union of
    // the stored index and the edition objects listed under `editions/<psp>/`
    // (one list; a get only for a week the index is missing) — so a lost entry
    // is restored by the creator's next write instead of being lost for good.
    const have = new Set(idx.map((e) => e.week));
    const prefix = `editions/${psp}/`;
    for (const k of await store.list(prefix)) {
      const w = k.slice(prefix.length).replace(/\.json$/, "");
      if (!WEEK_RE.test(w) || w === week || have.has(w)) continue;
      const raw = await store.get(k);
      if (raw === null) continue;
      try {
        const h = (JSON.parse(raw) as CardEdition).content_hash;
        if (typeof h === "string" && HASH_RE.test(h)) {
          idx.push({ week: w, content_hash: h });
          have.add(w);
        }
      } catch {
        // corrupt stored edition — leave for operator forensics, never index it
      }
    }
    idx.push({ week, content_hash: contentHash });
    idx.sort((a, b) => (a.week < b.week ? -1 : a.week > b.week ? 1 : 0));
    await store.put(idxKey, JSON.stringify(idx));
  }
}

// --- Ledger anchoring of weekly edition sets (S27) ---
//
// Once a week has ENDED (so its set can no longer grow), the sweep hashes that
// week's edition set and appends one `edition_set_anchored` event to the
// hash-chained trust ledger. The ledger is checkpoint-anchored (S4), so this
// upgrades edition immutability from "object-store write discipline" to
// TAMPER-EVIDENT: silently replacing any stored edition breaks the anchored
// set hash.
//
// WHAT THE ANCHORED SET IS — AND WHY IT IS NOT A MAP.
// The anchored set used to be the literal {passport_id -> content_hash} map,
// and `GET /editions/anchors/:week` served that map verbatim. It named every
// rated creator. That broke the commitment removal.ts makes in writing
// (removal.ts:32-33, "SUPPRESSION IS INDISTINGUISHABLE FROM NEVER-LISTED"):
// a subject rated before they were suppressed stayed in the served map
// forever, and redacting them would have broken `set_hash`, which is committed
// to the ledger. It was also a standing ENUMERATION surface for every rated
// passport — the thing invariant 3 denies the registry, granted here by a
// public GET.
//
// So the set is no longer a map. It is a SORTED LIST OF OPAQUE LEAF HASHES,
// one per edition. A passport id is never in it, so nothing ever has to be
// redacted out of it, and no redaction can ever break the hash:
//
//   leaf     = sha256_hex( EDITION_LEAF_VERSION "\n" passport_id "\n" content_hash )
//   leaves   = every leaf of the week, sorted ascending (lexicographic hex)
//   set_hash = sha256_hex( stableStringify({ leaves, version: EDITION_SET_VERSION, week }) )
//
// Strictly as strong for the only two things the route exists to serve:
//  - A HOLDER of an edition knows their own passport_id and content_hash, so
//    they recompute their leaf (editionLeaf below) and check membership in the
//    served list.
//  - ANYONE recomputes set_hash from the served list (editionSetHash below) and
//    checks it against the `edition_set_anchored` payload, which is
//    checkpoint-anchored. Replacing any stored edition still breaks it.
//  - NOBODY goes from the list back to a passport id: sha256 preimage
//    resistance, and there is nothing to grind offline — testing a guessed
//    passport id requires that subject's content_hash, obtainable only from the
//    edition or the per-creator index, and BOTH of those reads are
//    suppression-gated (serveEdition / serveEditionIndex).
//
// PREIMAGE INJECTIVITY — the whole argument, because a separator that can occur
// INSIDE either field lets two different pairs produce one preimage:
//  1. Both operands are FIXED WIDTH and VALIDATED here, not assumed:
//     passport_id matches PSP_RE /^psp_[0-9a-f]{16}$/ (exactly 20 chars) and
//     content_hash matches HASH_RE /^[0-9a-f]{64}$/ (exactly 64). At fixed
//     widths the concatenation is already injective with no separator at all.
//  2. The separator is "\n" — the same newline join the event preimage uses
//     (events.ts eventPreimage, "every field is newline-free by validation, so
//     the newline join is unambiguous") and a byte NEITHER validated charset
//     can contain. Injectivity therefore survives even if a field later became
//     variable-width.
//  3. EDITION_LEAF_VERSION is the first line, so a leaf preimage can never be
//     confused with an event preimage (different tag) or with a content_hash /
//     set_hash preimage (canonical JSON, first byte "{").
//  4. A pair failing (1) is never hashed AT ALL: the week is skipped
//     unanchored, the same fail-closed branch a corrupt set takes. The fixed
//     widths are enforced, not hoped for.
//
// CANONICAL SERIALIZATION. `stableStringify` — this module's existing canonical
// form, already what the edition content_hash commits to: recursively
// key-sorted, JSON-escaped, no whitespace. Leaves are lowercase hex, so JSON
// escaping is a no-op on them and sorted-string order is byte order. The set
// preimage is an OBJECT and not the bare array so the hash is BOUND to its week
// and to the format version: one week's set can never be replayed as another
// week's anchor. Everything a verifier needs (version, week, leaves) is served.
//
// `editions_count` STAYS, deliberately. It is a population count and nothing
// else — how many creators were rated that week. It is already implied by
// leaves.length in the same response, already public in the
// `edition_set_anchored` payload, and already weaker than the `population`
// field every ratings artifact publishes. Removing it would be theatre. It is
// written as leaves.length so it can never drift into a redaction detector.
//
// THIS FORMAT IS NOW PINNED. Same rule the event preimage lives under, in the
// same voice: THE LEAF PREIMAGE AND THE SET SERIALIZATION ARE PINNED FOREVER.
// Changing either is a format change — bump EDITION_LEAF_VERSION /
// EDITION_SET_VERSION and re-anchor forward; never edit v1 in place. This was
// free to change today only because nothing has launched and there is no
// committed anchor history to preserve. That window is now closed.
//
// WHAT THIS DOES NOT DO. A suppressed subject's leaf STAYS in the sets of weeks
// they were rated in — it must, or set_hash breaks, which is precisely the trap
// the map shape had set. That is not an oracle: the leaf is opaque, and the
// only party who can test it is one already holding that subject's edition,
// i.e. one who already knows they were rated. Going FORWARD the subject leaves
// the sets entirely — buildPowerRatings does not rank a suppressed creator, so
// no new edition and no new leaf is ever minted for them (proved in
// tests/invariants/suppression-oracles.test.ts, "the cron sweep neither ranks nor freezes a
// suppressed creator").
//
// The week's membership still exists as the sweep's own working state — one
// object per member under `editions/weekly-sets/<week>/`, so overlapping or
// retried sweeps cannot drop each other's entries (see weeklySetEntryKey) —
// but it is INTERNAL: nothing serves it, and once the week's anchor object is
// stored it is pruned (I3-2030 — `pruneWeek`).

/** PINNED. First line of every edition leaf preimage. Never edit — bump. */
export const EDITION_LEAF_VERSION = "dhwaja-edition-leaf-v1";
/** PINNED. `version` field of every set_hash preimage. Never edit — bump. */
export const EDITION_SET_VERSION = "dhwaja-edition-set-v1";

const PSP_RE = /^psp_[0-9a-f]{16}$/;
const HASH_RE = /^[0-9a-f]{64}$/;

/**
 * The opaque membership leaf for one edition. PINNED (see above).
 * Throws on a malformed operand rather than hashing an ambiguous preimage —
 * the caller's only correct response is to not anchor.
 */
export async function editionLeaf(passportId: string, contentHash: string): Promise<string> {
  if (!PSP_RE.test(passportId)) throw new Error("edition leaf: malformed passport_id");
  if (!HASH_RE.test(contentHash)) throw new Error("edition leaf: malformed content_hash");
  return sha256Utf8(`${EDITION_LEAF_VERSION}\n${passportId}\n${contentHash}`);
}

/**
 * The anchored set hash over a week's sorted leaf list. PINNED (see above).
 * This is the exact computation a third party runs on the bytes served by
 * `GET /editions/anchors/:week`.
 */
export async function editionSetHash(week: string, leaves: readonly string[]): Promise<string> {
  return sha256Utf8(stableStringify({ leaves, version: EDITION_SET_VERSION, week }));
}

export interface EditionAnchor {
  week: string;
  set_version: string;
  set_hash: string;
  editions_count: number;
  /** Sorted opaque leaves — the anchored set. Contains no passport id. */
  leaves: string[];
  anchored_at: string;
  event_id: string;
}

/**
 * The ledger-append seam. Resolves to the minted event, or to `null` when the
 * append was REFUSED because this week is already anchored — the appender rides
 * a `UNIQUE(week)` row into the same atomic batch as the event, so the second
 * writer of a week loses the constraint and the event rolls back with it (see
 * edition-anchor.ts). `null` means "somebody else anchored it": skip the week.
 */
type AppendAnchorEvent = (week: string, setHash: string, editionsCount: number) => Promise<{ event_id: string } | null>;

/**
 * I3-2030 cost limb (QUEUE #737, D-0858(1)): at most this many member-edition
 * reads per anchor SWEEP, across every week it visits. A week's set is one read
 * per member (I3-1883: the hash is the stored edition's own), so a week with
 * more members than one Worker invocation can fetch (~1000 subrequests) used to
 * throw on every sweep and never anchor. Now a week is read over as many sweeps
 * as it needs, and anchored only once every member has been read.
 */
export const ANCHOR_MEMBER_READ_BUDGET = 400;

/** The week's resume point: `{ v: {psp: content_hash}, p: [psp] }` — pairs
 * already verified against the stored edition, and members whose edition was
 * absent when read. A `.checkpoint.json` sibling of the legacy accumulator, so
 * it is never under the member prefix and the member key shape is unchanged;
 * pruned with the week. Written only by this sweep, from edition bytes, and
 * editions are immutable (putIfAbsent), so a pair can never go stale. */
function weeklySetCheckpointKey(week: string, slot = ""): string {
  return `editions/weekly-sets/${week}.checkpoint${slot === "" ? "" : `-${slot}`}.json`;
}

/**
 * QUEUE #774-C (D-0859(1)): ONE CHECKPOINT OBJECT PER WRITER. The six-hourly
 * sweep writes slot `""` (the original key) and the analytics tick writes slot
 * `"tick"`; each is serialised by its own cron lease, so no object has two
 * writers at once and neither invoker's put can drop pairs the other verified
 * (a shared object was a read-modify-write: a slow reader's put erased the
 * other's progress). Every reader UNIONS all slots — a pair is read from an
 * immutable edition and filtered to current members, so any slot's pair is as
 * good as a fresh read.
 */
export const ANCHOR_CHECKPOINT_SLOTS = ["", "tick"] as const;
export type AnchorCheckpointSlot = (typeof ANCHOR_CHECKPOINT_SLOTS)[number];

/**
 * One week's set, gathered from its member objects, within `budget.left` edition
 * reads (decremented). `complete: false` means the week is not fully read yet:
 * the checkpoint was saved and the caller must NOT anchor it this sweep.
 *
 * A member whose edition is absent is a phantom (I3-1883) and is dropped — but it
 * is not dropped for good on an early sweep: when every other member has been
 * read, the phantoms are read once more (deferring one sweep if the remaining
 * budget cannot cover them), so an edition that landed meanwhile still joins.
 * Phantoms that outnumber a whole budget stay dropped, so progress is bounded.
 *
 * Also reads the LEGACY single-object accumulator when one is present, on the
 * completing sweep only. Nothing writes it any more; it is here so a week that
 * was half-accumulated in the old shape when this shipped still anchors
 * completely, and so a corrupt legacy object still fails the week closed
 * instead of anchoring a subset. THROWS on anything unparseable — the caller's
 * only correct response is to leave the week unanchored.
 */
async function readWeeklySet(
  store: ArtifactStore,
  week: string,
  memberKeys: readonly string[],
  budget: { left: number; full: number },
  slot: AnchorCheckpointSlot = "",
): Promise<{ set: Record<string, string>; phantoms: number; complete: boolean; pending: number }> {
  const prefix = weeklySetPrefix(week);
  const members: string[] = [];
  for (const key of memberKeys) {
    const psp = key.slice(prefix.length).replace(/\.json$/, "");
    if (!PSP_RE.test(psp)) throw new Error("edition set: malformed member key");
    members.push(psp);
  }
  const memberSet = new Set(members);
  const ckKey = weeklySetCheckpointKey(week, slot);
  const verified: Record<string, string> = {};
  const knownPhantom = new Set<string>();
  let ckRaw: string | null = null;
  // The checkpoint is a cache of reads, never a source of truth: an
  // unparseable one is ignored and rebuilt, and only current members count.
  // Every slot's pairs first, then every slot's phantoms (a pair one writer
  // verified beats another's stale phantom).
  const cks: { v?: unknown; p?: unknown }[] = [];
  for (const s of ANCHOR_CHECKPOINT_SLOTS) {
    const raw = await store.get(weeklySetCheckpointKey(week, s));
    if (s === slot) ckRaw = raw;
    if (raw === null) continue;
    try {
      const ck = JSON.parse(raw) as { v?: unknown; p?: unknown } | null;
      if (ck && typeof ck === "object") cks.push(ck);
    } catch {
      // ignored and rebuilt
    }
  }
  for (const ck of cks) {
    if (typeof ck.v === "object" && ck.v !== null && !Array.isArray(ck.v)) {
      for (const [psp, h] of Object.entries(ck.v as Record<string, unknown>)) if (memberSet.has(psp) && typeof h === "string") verified[psp] = h;
    }
  }
  for (const ck of cks) if (Array.isArray(ck.p)) for (const psp of ck.p) if (typeof psp === "string" && memberSet.has(psp) && !(psp in verified)) knownPhantom.add(psp);
  const phantom = new Set<string>();
  const readOne = async (psp: string): Promise<void> => {
    budget.left--;
    const raw = await store.get(`editions/${psp}/${week}.json`);
    if (raw === null) {
      phantom.add(psp);
      return;
    }
    const { content_hash } = JSON.parse(raw) as { content_hash?: unknown };
    if (typeof content_hash !== "string") throw new Error("edition set: malformed edition");
    verified[psp] = content_hash;
  };
  const fresh = members.filter((psp) => !(psp in verified) && !knownPhantom.has(psp)).sort();
  const old = [...knownPhantom].sort();
  let complete = true;
  for (const psp of fresh) {
    if (budget.left <= 0) {
      complete = false;
      break;
    }
    await readOne(psp);
  }
  if (complete && old.length > 0) {
    if (old.length <= budget.left) for (const psp of old) await readOne(psp);
    else if (old.length <= budget.full) complete = false; // one more sweep re-reads them
    else for (const psp of old) phantom.add(psp); // more phantoms than a whole budget: dropped
  }
  if (!complete) {
    const next = JSON.stringify({ v: Object.fromEntries(Object.entries(verified).sort()), p: [...new Set([...old, ...phantom])].filter((x) => !(x in verified)).sort() });
    if (next !== ckRaw) await store.put(ckKey, next);
    return { set: verified, phantoms: 0, complete: false, pending: members.filter((x) => !(x in verified)).length };
  }
  const set: Record<string, string> = {};
  const legacyRaw = await store.get(`editions/weekly-sets/${week}.json`);
  if (legacyRaw !== null) Object.assign(set, JSON.parse(legacyRaw) as Record<string, string>);
  Object.assign(set, verified);
  return { set, phantoms: phantom.size, complete: true, pending: 0 };
}

/** I14-FX4 (QUEUE #912, I14-FX2 HANDOFF 2): deletes ONE anchorCompletedWeeks call may spend pruning, shared by its three
 * prune sites. A number, not unbounded: the artifacts ratings leg leaves the anchor step RATINGS_ANCHOR_ALLOWANCE (40),
 * which its listing, the per-week anchor reads and RATINGS_ANCHOR_MEMBER_BUDGET (16, index.ts) member reads also spend.
 * test/i14-fx4-prune-bound.test.ts holds it per site.
 * #927e — THE MEMBER BUDGET IS SHARED, AND THIS IS ITS SPLIT: while an anchored week is still being pruned, its late-member
 * check reads each member it is about to delete OUT OF the same member budget (`budget.left`), so it spends up to
 * ANCHOR_PRUNE_DELETE_BUDGET (8) reads and the next unanchored week's `readWeeklySet` gets the rest: 16 - 8 = 8 member reads
 * per ratings-leg call instead of 16, until the old week is gone (it resumes from its checkpoint, never re-reads). Kept
 * shared on purpose: a separate allowance would add up to 8 reads to a call RATINGS_ANCHOR_ALLOWANCE already sizes.
 * test/i16-w01g-anchor-budget.test.ts pins both figures. */
export const ANCHOR_PRUNE_DELETE_BUDGET = 8;
/** I19-FXB (JN-B02 W1D:5): D1 round trips reserved for one ledger append (`editionSetAnchorAppender` ->
 * `tryAppendEvents`): measured 2 by test/i19-fxb-anchor-allowance.test.ts, which fails if one spends more than this. */
export const ANCHOR_APPEND_COST = 4;
/** I19-FXB (JN-B02 W1D:5): the most one owed week spends besides member reads and deletes: the anchor get, the freeze
 * cursor get, one get per checkpoint slot, the checkpoint put or the legacy get, the append, the two read-back reads,
 * the forced-anchor marker put and the anchor put. */
export const ANCHOR_WEEK_STEP_MAX = 1 + 1 + ANCHOR_CHECKPOINT_SLOTS.length + 1 + ANCHOR_APPEND_COST + 2 + 1 + 1;
/** I3-2030: the week's working state is removed once its anchor object is
 * stored — it is on no read path after that (see above). Keys come from the
 * sweep's own listing; a store that cannot delete keeps them. I14-FX4: at most
 * `prune.left` deletes; the keys left over keep the week listed, so the next
 * call takes the anchored-week branch and resumes, and `deferred` records the count. */
async function pruneWeek(store: ArtifactStore, keys: readonly string[], prune: { left: number }, deferred: (n: number) => void): Promise<void> {
  const del = (store as ArtifactStore & { delete?: (k: string) => Promise<void> }).delete;
  if (typeof del !== "function") return;
  for (const [i, k] of keys.entries()) {
    if (prune.left <= 0) return deferred(keys.length - i);
    prune.left--;
    await del.call(store, k);
  }
}

/** #936(a) I3-1103 (D-0954(4)): how long after a week closes its anchor waits for the week's edition freeze. The freeze
 * (`freezeWeeklyEditionsStep`) writes only the CURRENT week, so after the close its cursor moves only for a call that
 * straddled the boundary; a day covers that many times over and keeps an unfrozen week's anchor within a day. */
export const ANCHOR_FREEZE_DEADLINE_MS = DAY_MS;
/** I3-1103 b: store gets the anchor's population check spends — one pass over the live shards, at most once per
 * `anchorCompletedWeeks` call (RATINGS_ANCHOR_ALLOWANCE carries it on top of its earlier 40). */
export const ANCHOR_POPULATION_READS = RATINGS_SHARDS;

/** #936(a): the week's freeze cursor says `complete` (absent or unreadable: not complete) — AND (I3-1103 b) the
 * population AS OF THE WEEK'S CLOSE still equals the population that pass completed over (`cursor.population`). A
 * creator published after the week's last freeze pass but before the close changes the hash, so the week reads as not
 * complete: it waits (`freeze_open`) until the deadline, then anchors with the forced marker; one published after the
 * close does not (I25-FX-DATA). `atClose` is read only when the cursor says complete; it returns null when the call's
 * meter cannot pay it (`unpaid`: the caller stops and the week resumes next call). */
async function weekFreezeState(store: ArtifactStore, week: string, atClose: (week: string) => Promise<string | null>): Promise<"complete" | "open" | "unpaid"> {
  const raw = await store.get(editionsCursorKey(week));
  if (raw === null) return "open";
  let cursor: Partial<EditionsCursor>;
  try {
    cursor = JSON.parse(raw) as Partial<EditionsCursor>;
  } catch {
    return "open";
  }
  if (cursor.complete !== true) return "open";
  const population = await atClose(week);
  if (population === null) return "unpaid";
  return typeof cursor.population === "string" && cursor.population === population ? "complete" : "open";
}

/** #936(a): the durable record that a week was anchored past its freeze deadline with its freeze incomplete — beside
 * the cursor, under `editions/_` (never a creator prefix), create-if-absent like the anchor — and its log line.
 * Written BEFORE the anchor object: a crash between the two leaves the record and the ledger read-back finishes the anchor. */
export function anchorForcedKey(week: string): string {
  return `editions/_anchor_forced/${week}.json`;
}
async function anchorForcedMarker(store: ArtifactStore, anchor: EditionAnchor, cid: string): Promise<void> {
  logError({ cid, hop: "cron", msg: "edition_anchor_freeze_incomplete", kind: "freeze_deadline", detail: anchor.week, count: anchor.editions_count });
  await store.putIfAbsent(
    anchorForcedKey(anchor.week),
    JSON.stringify({ week: anchor.week, reason: "freeze_incomplete_at_deadline", anchored_at: anchor.anchored_at, event_id: anchor.event_id, editions_count: anchor.editions_count }),
  );
}

/**
 * Anchor every COMPLETED week that has a set artifact but no anchor yet.
 * `appendAnchor` is the ledger-append seam (the HTTP layer wires appendEvents
 * with actor system) — kept injected so this module stays out of the ledger
 * writer path and the events module stays the only one.
 */
export async function anchorCompletedWeeks(
  rawStore: ArtifactStore,
  nowMs: number,
  rawAppendAnchor: AppendAnchorEvent,
  now: () => string = () => new Date().toISOString(),
  weeksBack = 8,
  // S-036 C-34/FX-15: OPTIONAL, and added last so every existing positional
  // call (index.ts included) keeps compiling untouched. Without it the
  // conflict branch below just `continue`s, which is correct for the
  // concurrent-sweep case but wrong for the crash case: `appendAnchor` rides
  // the ledger event and the `edition_set_anchors` UNIQUE(week) row into ONE
  // atomic batch (edition-anchor.ts), so a crash between that commit and the
  // `store.put(anchorKey, ...)` below leaves the week permanently anchored in
  // the ledger with no object ever reachable — `appendAnchor` conflicts on
  // every future sweep (same UNIQUE(week) row) and returns null every time,
  // so line "if (appended === null) continue" was a permanent skip, not a
  // retry. Recovery needs the ledger (the event_id is minted there, never in
  // the store), which is why this needs a `db` this function did not
  // previously take: `ratings.ts` is deliberately kept out of the ledger
  // WRITE path (see edition-anchor.ts's docblock), but a READ of the already-
  // committed row is not a write. Wiring index.ts's call site to pass its
  // `db` through is a one-line follow-up outside this room's ownership
  // (services/trust-core/src/ratings.ts only) — see OPEN QUESTIONS.
  db: DbLike | null = null,
  // I3-0949: correlation id for the skip log (log.ts `cid`); index.ts may pass
  // its lease holder. Optional and last, like `db`, so every call compiles.
  cid = "cron:editions-anchor",
  // I3-2030: member-edition reads this sweep may spend (see readWeeklySet).
  memberBudget = ANCHOR_MEMBER_READ_BUDGET,
  // QUEUE #774-C: which checkpoint object this invoker writes (see
  // ANCHOR_CHECKPOINT_SLOTS). The sweep keeps `""`; the tick passes "tick".
  checkpointSlot: AnchorCheckpointSlot = "",
  // I19-FXB (JN-B02 W1D:5): subrequests this call may spend — every store call, the ledger append and the read-back.
  // The ratings leg passes RATINGS_ANCHOR_ALLOWANCE (index.ts); unbounded by default (the tick and direct callers).
  allowance = Infinity,
): Promise<EditionAnchor[]> {
  // I19-FXB (JN-B02 W1D:5): THE METER. Every store call below goes through `store`, which spends one unit per call
  // (a listing one per LIST_PAGE keys); the append spends ANCHOR_APPEND_COST; each ledger read-back read one. Before a
  // week starts, the meter must cover ANCHOR_WEEK_STEP_MAX, and the shared member-read and delete budgets are cut to
  // what is left after it — so a call never spends past `allowance`. The oldest owed week goes first and gets the
  // full budgets (55 - ANCHOR_WEEK_STEP_MAX - ANCHOR_POPULATION_READS >= 16 + 8 at 56), so a stop leaves later weeks for the next call, never
  // the oldest: no week is stranded.
  const meter = { left: allowance };
  const rawDelete = (rawStore as ArtifactStore & { delete?: (k: string) => Promise<void> }).delete;
  const spend = (n: number): void => {
    meter.left -= n;
  };
  const store: ArtifactStore & { delete?: (k: string) => Promise<void> } = {
    get: (k) => {
      spend(1);
      return rawStore.get(k);
    },
    list: async (p) => {
      const keys = await rawStore.list(p);
      spend(Math.max(1, Math.ceil(keys.length / LIST_PAGE)));
      return keys;
    },
    put: (k, v) => {
      spend(1);
      return rawStore.put(k, v);
    },
    putIfAbsent: (k, v) => {
      spend(1);
      return rawStore.putIfAbsent(k, v);
    },
    ...(typeof rawDelete === "function"
      ? {
          delete: (k: string) => {
            spend(1);
            return rawDelete.call(rawStore, k);
          },
        }
      : {}),
  };
  const appendAnchor: AppendAnchorEvent = (week, setHash, n) => {
    spend(ANCHOR_APPEND_COST);
    return rawAppendAnchor(week, setHash, n);
  };
  const currentWeek = isoWeek(nowMs);
  const out: EditionAnchor[] = [];
  // I3-0949: EVERY fail-closed branch below says so. They used to `continue`
  // in silence, so a week could stay unanchored — its editions unverifiable —
  // with no record anywhere that it happened. `detail` is the ISO week (not an
  // identifier); `kind` is the branch.
  const skipped = (kind: string, week: string, count?: number): void =>
    logError({ cid, hop: "cron", msg: "edition_anchor_skipped", kind, detail: week, ...(count === undefined ? {} : { count }) });
  // I3-0949: THE CANDIDATE WEEKS ARE DERIVED, NOT A WINDOW. This loop used to
  // visit the last `weeksBack` (8) weeks only, so a week that failed closed
  // for longer than that could never be anchored. Now it visits every closed
  // week that still has working state under the set prefix — which, because a
  // week's members are pruned once its anchor is stored (I3-2030), is exactly
  // the weeks still owed an anchor. Oldest first; `weeksBack` bounds how many
  // one sweep attempts (subrequest budget) and the rest are logged as deferred.
  const root = "editions/weekly-sets/";
  const byWeek = new Map<string, { members: string[]; all: string[] }>();
  for (const k of await store.list(root)) {
    const rest = k.slice(root.length);
    const slash = rest.indexOf("/");
    const week = slash === -1 ? rest.replace(/(\.checkpoint(-[a-z]+)?)?\.json$/, "") : rest.slice(0, slash);
    if (!WEEK_RE.test(week) || week >= currentWeek) continue; // still open (or not yet begun) — never anchor a growing set
    const entry = byWeek.get(week) ?? { members: [], all: [] };
    entry.all.push(k);
    if (slash !== -1) entry.members.push(k);
    byWeek.set(week, entry);
  }
  const weeks = [...byWeek.keys()].sort();
  // `full` is the phantom-drop bar (readWeeklySet): never below the SWEEP's
  // budget, so a small-budget invoker (the tick) defers phantoms the sweep
  // would still re-read instead of dropping them — who completes a week must
  // not change what it anchors (QUEUE #774-C).
  const budget = { left: memberBudget, full: Math.max(memberBudget, ANCHOR_MEMBER_READ_BUDGET) };
  const prune = { left: ANCHOR_PRUNE_DELETE_BUDGET }; // I14-FX4: every pruneWeek below shares it
  if (weeks.length > weeksBack) skipped("deferred", weeks[weeksBack]!, weeks.length - weeksBack);
  let livePopulation: string | null = null; // I3-1103 b: read at most once per call, only when a cursor says complete
  // I25-FX-DATA: the population as of the week's close is the week's record (`weekPopulationKey`, one get); a week with
  // no readable record (swept before the record existed) falls back to the live shard pass, read ONCE per call. The
  // meter must cover each on top of the rest of this week's step (its anchor and cursor gets are spent).
  const atClose = async (week: string): Promise<string | null> => {
    if (meter.left < 1 + ANCHOR_WEEK_STEP_MAX - 2) return null;
    const rec = await store.get(weekPopulationKey(week));
    try {
      const p = rec === null ? undefined : (JSON.parse(rec) as { population?: unknown }).population;
      if (typeof p === "string" && p.length > 0) return p;
    } catch {
      // unreadable record: the live pass below
    }
    if (livePopulation === null) {
      if (meter.left < ANCHOR_POPULATION_READS + ANCHOR_WEEK_STEP_MAX - 3) return null;
      livePopulation = (await liveShardPopulation(store)).population;
    }
    return livePopulation;
  };
  for (const [i, week] of weeks.slice(0, weeksBack).entries()) {
    if (meter.left < ANCHOR_WEEK_STEP_MAX) {
      skipped("allowance", week, Math.min(weeks.length, weeksBack) - i); // this week and the rest resume next call
      break;
    }
    budget.left = Math.min(budget.left, meter.left - ANCHOR_WEEK_STEP_MAX);
    prune.left = Math.min(prune.left, meter.left - ANCHOR_WEEK_STEP_MAX - budget.left);
    const keys = byWeek.get(week)!;
    const anchorKey = `editions/anchors/${week}.json`;
    const anchorRaw = await store.get(anchorKey);
    if (anchorRaw !== null) {
      // Already anchored — immutable. A member still under its prefix either
      // waits for a bounded prune (I14-FX4) or arrived after the anchor; only
      // the second is LATE (I14-FX5, FX4:4): its leaf is absent from the stored
      // anchor. Each member is read once, out of the member-read budget, on the
      // call that deletes it; one the budget cannot read stays listed for the
      // next call, so a late member is never deleted unlogged. An anchor with no
      // readable leaves counts every member late, as before.
      let anchored: Set<string> | null = null;
      try {
        const a = JSON.parse(anchorRaw) as { leaves?: unknown };
        if (Array.isArray(a.leaves)) anchored = new Set(a.leaves.filter((l): l is string => typeof l === "string"));
      } catch {
        anchored = null;
      }
      const memberKeys = new Set(keys.members);
      const prefix = weeklySetPrefix(week);
      const checked: string[] = [];
      let late = 0;
      for (const k of keys.all) {
        if (checked.length >= prune.left) break;
        if (memberKeys.has(k) && anchored !== null) {
          if (budget.left <= 0) break;
          budget.left--;
          const raw = await store.get(k);
          let inAnchor = false;
          try {
            const h = raw === null ? null : (JSON.parse(raw) as { content_hash?: unknown }).content_hash;
            inAnchor = raw === null || (typeof h === "string" && anchored.has(await editionLeaf(k.slice(prefix.length).replace(/\.json$/, ""), h)));
          } catch {
            inAnchor = false;
          }
          if (!inAnchor) late++;
        } else if (memberKeys.has(k)) late++;
        checked.push(k);
      }
      if (late > 0) skipped("late_member", week, late);
      await pruneWeek(store, checked, prune, (n) => skipped("prune_deferred", week, n));
      if (keys.all.length > checked.length) skipped("prune_deferred", week, keys.all.length - checked.length);
      continue;
    }
    // #936(a) I3-1103 (D-0954(4)): THE ANCHOR WAITS FOR THE WEEK'S FREEZE. `freezeWeeklyEditionsStep` marks the week's
    // cursor `complete` once a pass over an unchanged population froze nobody; until then the set may still be missing a
    // creator's edition. A week whose cursor is not complete is skipped (`freeze_open`) until ANCHOR_FREEZE_DEADLINE_MS
    // after it closed, then anchored ANYWAY — no week is stranded — and the anchor says so (`anchorForcedMarker`).
    // Read before the member reads, so a waiting week spends one read, not the member budget.
    // I3-1103 b: a complete cursor is trusted only over an unchanged live population; the shard pass is read ONCE per
    // call (`live`), and a call whose meter cannot pay it stops here and the week resumes next call.
    const freeze = await weekFreezeState(store, week, atClose);
    if (freeze === "unpaid") {
      skipped("allowance", week, Math.min(weeks.length, weeksBack) - i);
      break;
    }
    const freezeComplete = freeze === "complete";
    budget.left = Math.min(budget.left, meter.left - ANCHOR_WEEK_STEP_MAX);
    prune.left = Math.min(prune.left, Math.max(0, meter.left - ANCHOR_WEEK_STEP_MAX - budget.left));
    if (!freezeComplete && nowMs < isoWeekEndMs(week) + ANCHOR_FREEZE_DEADLINE_MS) {
      skipped("freeze_open", week);
      continue;
    }
    // THE SET IS DERIVED, NOT READ FROM ONE CELL. Every member is its own
    // object (see weeklySetEntryKey), so a concurrent or retried sweep can add
    // to the week without any writer being able to drop another's entry. The
    // week is closed by the time we get here, so the listing is stable.
    let set: Record<string, string>;
    try {
      const r = await readWeeklySet(store, week, keys.members, budget, checkpointSlot);
      if (!r.complete) {
        skipped("partial_set", week, r.pending); // checkpointed; a later sweep resumes it
        continue;
      }
      set = r.set;
      if (r.phantoms > 0) skipped("phantom_member", week, r.phantoms);
    } catch {
      skipped("unreadable_set", week);
      continue; // corrupt member — leave for forensics, never anchor garbage
    }
    const entries = Object.entries(set);
    if (entries.length === 0) {
      skipped("empty_set", week);
      continue; // no editions that week
    }
    // Malformed pair → skip the WHOLE week unanchored (fail closed). Anchoring
    // a partial set would silently commit to a set missing a real edition.
    let leaves: string[];
    try {
      leaves = await Promise.all(entries.map(([psp, h]) => editionLeaf(psp, h)));
    } catch {
      skipped("malformed_pair", week);
      continue;
    }
    leaves.sort();
    // Impossible from a map keyed by passport id, but a duplicate would make
    // editions_count and the list disagree — fail closed rather than commit it.
    if (new Set(leaves).size !== leaves.length) {
      skipped("duplicate_leaf", week);
      continue;
    }
    const setHash = await editionSetHash(week, leaves);
    // ANCHOR ONCE. The `store.get(anchorKey)` above is a cheap short-circuit,
    // not the guard — the anchor object is written AFTER the append, so two
    // overlapping sweeps both read null there and both used to append an
    // `edition_set_anchored` event for the same week (measured: two events, one
    // set_hash). The guard is inside the append: a `UNIQUE(week)` row rides the
    // same atomic batch, so the loser's event rolls back with the constraint and
    // surfaces here as a null. Skip the week — the winner owns it, artifact
    // included.
    const appended = await appendAnchor(week, setHash, leaves.length);
    if (appended === null) {
      // Either a concurrent sweep just won the SAME row (setHash/leaves are a
      // pure function of the closed week's store state, so it computed the
      // identical values) or an earlier sweep committed the ledger row and
      // crashed before the object write. Both cases are recoverable the same
      // way: read back the row this exact append lost to, and — ONLY if it
      // is byte-identical to what we just (re)computed — finish the object
      // write ourselves.
      if (db === null) {
        skipped("conflict_no_db", week);
        continue; // no ledger handle wired — leave for a sweep that has one
      }
      spend(2); // this read and the event read below (ANCHOR_WEEK_STEP_MAX counts both)
      const row = await db
        .prepare("SELECT set_hash, editions_count, anchored_at FROM edition_set_anchors WHERE week = ?")
        .bind(week)
        .first<{ set_hash: string; editions_count: number; anchored_at: string }>();
      // Anomaly, not the crash case: the committed row disagrees with the set
      // this sweep just derived from the store. Never anchor a guess — fail
      // closed and leave it for forensics, same posture as every other
      // fail-closed branch in this function.
      if (!row || row.set_hash !== setHash || row.editions_count !== leaves.length) {
        skipped("ledger_mismatch", week);
        continue;
      }
      const existing = await db
        .prepare("SELECT event_id FROM trust_events WHERE event_type = 'edition_set_anchored' AND subject = ? ORDER BY seq DESC LIMIT 1")
        .bind(`sha256:${setHash}`)
        .first<{ event_id: string }>();
      if (!existing) {
        skipped("ledger_event_missing", week);
        continue; // ledger row committed but the event itself is unreadable — never fabricate one
      }
      const recovered: EditionAnchor = {
        week,
        set_version: EDITION_SET_VERSION,
        set_hash: setHash,
        editions_count: leaves.length,
        leaves,
        anchored_at: row.anchored_at,
        event_id: existing.event_id,
      };
      // Create-if-absent: two sweeps racing to recover never replace each
      // other's object (the anchor is immutable once written).
      if (!freezeComplete) await anchorForcedMarker(store, recovered, cid);
      await store.putIfAbsent(anchorKey, JSON.stringify(recovered));
      await pruneWeek(store, keys.all, prune, (n) => skipped("prune_deferred", week, n));
      out.push(recovered);
      continue;
    }
    const { event_id } = appended;
    const anchor: EditionAnchor = {
      week,
      set_version: EDITION_SET_VERSION,
      set_hash: setHash,
      editions_count: leaves.length,
      leaves,
      anchored_at: now(),
      event_id,
    };
    if (!freezeComplete) await anchorForcedMarker(store, anchor, cid);
    await store.putIfAbsent(anchorKey, JSON.stringify(anchor));
    await pruneWeek(store, keys.all, prune, (n) => skipped("prune_deferred", week, n));
    out.push(anchor);
  }
  return out;
}

export async function anchorEditionsTickStep(

/**
 * Public: a completed week's anchor record — the sorted opaque leaf list, its
 * hash, and the ledger event id that committed it. Verify a saved edition end
 * to end: edition -> leaf -> membership -> set_hash -> checkpoint-anchored
 * ledger event.
 *
 * This function reads ONE object, the anchor itself, and that object contains
 * no passport id. The passport-keyed weekly-set accumulator is deliberately NOT
 * on this path: a future bug here cannot reach it.
 */
export async function serveEditionAnchor(store: ArtifactReader, week: string): Promise<EditionAnchor | null> {
  if (!WEEK_RE.test(week)) return null;
  const anchorRaw = await store.get(`editions/anchors/${week}.json`);
  if (!anchorRaw) return null;
  let anchor: EditionAnchor;
  try {
    anchor = JSON.parse(anchorRaw) as EditionAnchor;
  } catch {
    return null;
  }
  // Fail closed on anything that is not the pinned shape: a stored anchor whose
  // leaves are absent or not all opaque hashes is never served.
  if (!Array.isArray(anchor.leaves) || !anchor.leaves.every((l) => typeof l === "string" && HASH_RE.test(l))) return null;
  return anchor;
}

  /** The bases the ledger would accept right now (empty = not reactivatable). */
  bases_available: ReactivationBasis[];
  /** The one sentence the ledger would refuse with, when nothing is available. */
  refusal: string | null;
}

export function keyShapeOk(key: unknown): key is string {
  return typeof key === "string" && key.length > 0 && key.length <= ACCOUNT_KEY_MAX && key.startsWith("v1|");
}

export function basisOk(basis: unknown): basis is ReactivationBasis {
  return typeof basis === "string" && (REACTIVATION_BASES as readonly string[]).includes(basis);
}

/** Everything an operator reads before signing — and the refusal the ledger would give. */
export async function reactivationPreview(db: DbLike, key: string, nowMs: number): Promise<ReactivationPreview> {
  if (!keyShapeOk(key)) throw new ReactivationError("bad account key");
  const acct = await db
    .prepare(
      `SELECT a.passport_id AS passport_id, a.status AS status, p.retracted AS retracted,
              EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = a.passport_id) AS subj_gone,
              EXISTS (SELECT 1 FROM suppressed_account_keys k WHERE k.account_key = a.account_key) AS key_frozen
         FROM passport_accounts a JOIN passports p ON p.passport_id = a.passport_id
        WHERE a.account_key = ?`,
    )
    .bind(key)
    .first<{ passport_id: string; status: string; retracted: number; subj_gone: number; key_frozen: number }>();
  if (!acct) throw new ReactivationError("no such account", 404);
  const revoking = await latestRevokingEvent(db, key);
  const delisted = acct.subj_gone === 1 || acct.key_frozen === 1;
  // "Re-proven" = proved again AFTER the revocation, not lapsed, inside the window.
  const proofFresh = revoking ? await freshControlProof(db, acct.passport_id, key, nowMs, revoking.created_at) : false;
  const warning = await publishedWarningFor(db, key);
  const blocking = await activeBlockingWarning(db, key);
  const bases: ReactivationBasis[] = [];
  let refusal: string | null = null;
  if (acct.retracted === 1) refusal = "the passport is retracted";
  else if (acct.status !== "revoked" && acct.status !== "historical") refusal = `the account is ${acct.status}, not revoked or historical`;
  else if (delisted) refusal = "delisted at the subject's request and never re-listed — lift the removal (unsuppress) first, so the re-listing is a declared act";
  else if (blocking.warned) refusal = "the account carries an active public warning and cannot be reactivated while the warning is active — retract the warning first";
  else if (!revoking) refusal = "no revoking event on record for this key";
  else {
    if (revoking.event_type === "account_revoked") bases.push("takedown_reversed");
    if (revoking.event_type === "correction") bases.push("correction_reversed");
    if (proofFresh) bases.push("control_reproven");
  }
  return {
    account_key: key,
    passport_id: acct.passport_id,
    status: acct.status,
    revoking,
    delisted,
    proof_fresh: proofFresh,
    proof_window_days: REACTIVATION_PROOF_WINDOW_MS / 86_400_000,
    warning,
    bases_available: bases,
    refusal,
  };
}

/** The latest `warning_published` / `warning_retracted` for the key's subject hash decides. */
async function publishedWarningFor(db: DbLike, key: string): Promise<{ published: boolean; event_id: string | null }> {
  const subject = `sha256:${await subjectHash(key)}`;
  const row = await db
    .prepare("SELECT event_id, event_type FROM trust_events WHERE subject = ? AND event_type IN ('warning_published', 'warning_retracted') ORDER BY seq DESC LIMIT 1")
    .bind(subject)
    .first<{ event_id: string; event_type: string }>();
  return row && row.event_type === "warning_published" ? { published: true, event_id: row.event_id } : { published: false, event_id: null };
}

/**
 * Append `account_reactivated` for `key` on the passport that owns it, with an
 * audit line. `reverses` must be the event the caller previewed — two operators
 * sign the same reversal, and a revoking event that landed between preview and
 * execution is a refusal, never a silent re-target. `expect_passport` binds the
 * call to the passport the caller decided about (the console's preview, the
 * recovery case's target): a key that has moved since is a refusal, never a
 * re-listing on a passport nobody signed for (F5 review).
 */
export async function reactivateAccount(
  db: DbLike,
  operator: string,
  input: { account_key: string; basis: ReactivationBasis; reverses: string; reason?: string | null; expect_passport?: string },
  now: () => string = () => new Date().toISOString(),
  /** QUEUE #942(a) (D-0946(1)): the caller's audit row, after this function's own, in the same append batch. */
  extraStatements: DbStatement[] = [],
): Promise<{ event_id: string; passport_id: string }> {
  if (!OP_RE.test(operator)) throw new ReactivationError("bad operator", 400);
  if (!keyShapeOk(input.account_key)) throw new ReactivationError("bad account key");
  if (!basisOk(input.basis)) throw new ReactivationError(`basis must be ${REACTIVATION_BASES.join("|")}`);
  if (input.reason !== undefined && input.reason !== null && (typeof input.reason !== "string" || input.reason.length > 512)) throw new ReactivationError("reason must be ≤ 512 chars");
  if (input.expect_passport !== undefined && !PSP_RE.test(input.expect_passport)) throw new ReactivationError("bad expect_passport");
  const owner = await db.prepare("SELECT passport_id FROM passport_accounts WHERE account_key = ?").bind(input.account_key).first<{ passport_id: string }>();
  if (!owner || !PSP_RE.test(owner.passport_id)) throw new ReactivationError("no such account", 404);
  if (input.expect_passport !== undefined && owner.passport_id !== input.expect_passport) {
    throw new ReactivationError(`${input.account_key} is on ${owner.passport_id}, not ${input.expect_passport} — preview again`, 409);
  }
  const payload: Record<string, unknown> = { account_key: input.account_key, basis: input.basis, reverses: input.reverses };
  if (typeof input.reason === "string" && input.reason.length > 0) payload.reason = input.reason;
  const draft = { event_type: "account_reactivated", subject: owner.passport_id, actor_kind: "operator", actor_id: operator, payload } as DraftEvent;
  const ts = now();
  let ev: TrustEventEnvelope | undefined;
  try {
    [ev] = await appendEvents(db, [draft], now, [], [auditStmt(db, operator, "account:reactivate", input.account_key.slice(0, 200), `${input.basis} on ${owner.passport_id} reverses ${input.reverses}`, ts), ...extraStatements]);
  } catch (e) {
    if (e instanceof LedgerError) throw new ReactivationError(e.message, 409);
    throw e;
  }
  return { event_id: ev!.event_id, passport_id: owner.passport_id };
}

export async function publicRecognitionsFor(db: DbLike, accountId: string, limit = 50): Promise<PublicRecognition[]> {

    .prepare(

         FROM recognitions r JOIN passports p ON p.passport_id = r.passport_id
        WHERE r.account_id = ? AND r.retracted_at IS NULL AND p.retracted = 0
          AND r.accepted_at IS NOT NULL AND r.declined_at IS NULL
          AND NOT EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = r.passport_id)
        ORDER BY r.granted_at DESC LIMIT ?`,
    )
    .bind(accountId, capped)
    .all<{ kind: string; message: string | null; granted_at: string; by_passport_id: string; by_name: string | null }>();

  });

  // THE FAN'S HALF OF THE CONSENT (audit item #11). A creator may WRITE a
  // recognition about a fan; only the named FAN may PUBLISH it. Both doors are
  // FAN-session gated — `fanId`, not `creatorId` — because the whole point is that

    } catch (e) {
      if (e instanceof RecognitionError) return c.json(errorBody(e), e.status);
      return serverFault(c, "/account/recognition/:id/accept", e);
    }
  });

  app.post("/account/recognition/:id/decline", async (c) => {
    const id = await fanId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    try {
      return c.json({ ok: true, ...(await decideRecognition(d1(c.env.TRUST_DB), id, c.req.param("id"), "decline")) });
    } catch (e) {
      if (e instanceof RecognitionError) return c.json(errorBody(e), e.status);
      return serverFault(c, "/account/recognition/:id/decline", e);
    }
  });

  // S90b — the creator's own weekly recognition recap (verified-gated).
  app.get("/account/recognition/recap", async (c) => {
    const id = await creatorId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const db = d1(c.env.TRUST_DB);
    if (!(await isVerifiedPassport(db, id))) return c.json({ error: "recognition unlocks at verified status" }, 403);
    return c.json(await recognitionRecap(db, id, Date.now()));
  });

  return app;
}

import { sha256Hex } from "@dhwaja/canonical";
import { RECOGNITION_ACCOUNT_RE, RECOGNITION_ID_RE, RECOGNITION_KINDS, RECOGNITION_MESSAGE_KINDS, stripUnsafeDisplayText } from "@dhwaja/schema";
import type { DbLike, DbStatement } from "./db.js";
import { effectiveCreatorTier } from "./entitlements-store.js";
import { recognitionCaps } from "./entitlements.js";

/**
 * S90 — creator→fan recognition: the scarce, signed, attributable ladder (Phase 3
 * recognition core). Each recognition is a passkey-signed ledger event
 * (recognition_granted / recognition_retracted, derive-from-log — the source of
 * truth); this module owns the surrounding controls and the read projection.
 *
 * The ladder, and why it stays PREMIUM:
 *   - `follow` is the standing relationship. A creator may follow only a fan in
 *     their OWN linked audience — someone who saved them or claimed one of their
 *     codes — so recognition can never name an arbitrary account. Follows are
 *     capped per tier (ACTIVE_FOLLOW_CAP).
 *   - `like` / `thanks` / `shout_out` / `spotlight` are expressive recognitions and
 *     can only target a fan the creator ALREADY follows.
 *   - Every grant (follow or expressive) counts against a small per-day cap
 *     (RECOGNITION_DAILY_CAP, tiered) — the scarcity knob that keeps a shout-out
 *     meaningful. A retraction never refunds a day's slot.
 *
 * Privacy: the queue/read surfaces reveal fan account_ids ONLY to that fan's own
 * creator, and ONLY for fans already in the creator's audience (savers/claimers) —
 * the minimum needed to target a recognition; no handles/emails, aggregates + own
 * rows only. The recognized fan is carried in the event PAYLOAD, never as the
 * subject (the subject is always the granting creator's own passport).
 */

export class RecognitionError extends Error {
  override name = "RecognitionError";
  constructor(
    message: string,
    public readonly status: 400 | 401 | 403 | 404 | 409 | 429 = 400,
  ) {
    super(message);
  }
}

/** Tiered caps (locked S90; amended S90c). Two SEPARATE daily caps by weight:
 *  - RECOGNITION_DAILY_CAP gates the SCARCE named recognitions — follow + thanks +
 *    shout_out + spotlight — the premium signals.
 *  - LIKE_DAILY_CAP is the lightweight `like` rung: no follow required, its own more
 *    generous daily budget, so a creator can freely ack their audience.
 * ACTIVE_FOLLOW_CAP bounds how many fans a creator actively follows at once.
 *
 * M7: the NUMBERS now come from `recognitionCaps` (entitlements.ts — the one
 * pinned cap table) and the TIER from `effectiveCreatorTier` (the single read
 * that honours the 7-day billing grace). These exports keep their names for the
 * call sites and clients, but can no longer drift from what money actually
 * buys. */
export const RECOGNITION_DAILY_CAP = { free: recognitionCaps("free").namedDaily, pro: recognitionCaps("pro").namedDaily } as const;
export const LIKE_DAILY_CAP = { free: recognitionCaps("free").likeDaily, pro: recognitionCaps("pro").likeDaily } as const;
export const ACTIVE_FOLLOW_CAP = { free: recognitionCaps("free").activeFollows, pro: recognitionCaps("pro").activeFollows } as const;
export const RECOGNITION_MAX_LEN = 280;
const DAY_MS = 86_400_000;
const PSP_RE = /^psp_[0-9a-f]{16}$/;

/** A CSPRNG recognition id. Earlier this was `uuid7(nowMs).slice(0,16)` — 48
 * timestamp bits + ~12 random bits — so recognitions minted in the same
 * millisecond (across creators, under load) could birthday-collide the
 * `recognitions.recognition_id` PK. The id is opaque and never ordered by (reads
 * order by `granted_at`), so full 64 random bits is strictly better: collision-safe
 * and less timing-revealing in the signed payload. */
export function newRecognitionId(): string {
  return `rec_${crypto.randomUUID().replace(/-/g, "").slice(0, 16)}`;
}

/** I3-1229 (D-0833): the ledger commits to a recognition's message, never carries it.
 * The text lives only in the erasable `recognitions.message` projection column
 * (deleted with the fan's account, account_data.ts). The preimage binds the
 * recognition id, so one hash cannot be replayed onto another recognition. */
export const RECOGNITION_MESSAGE_HASH_VERSION = "recognition-message-v1";
export async function recognitionMessageHash(recognitionId: string, message: string): Promise<string> {
  return sha256Hex(`${RECOGNITION_MESSAGE_HASH_VERSION}\n${recognitionId}\n${message}`);
}

/** The signed `recognition_granted` payload: `message_hash` in place of `message`. */
export async function recognitionGrantPayload(recognitionId: string, kind: string, message: string | null): Promise<{ recognition_id: string; kind: string; message_hash?: string }> {
  const payload: { recognition_id: string; kind: string; message_hash?: string } = { recognition_id: recognitionId, kind };
  if (message !== null && message !== "") payload.message_hash = await recognitionMessageHash(recognitionId, message);
  return payload;
}

/** Sanitize an expressive recognition's message BEFORE it enters the signed draft
 * (idempotent, mirrors broadcast sanitize): strip angle brackets + control/bidi/
 * zero-width chars, collapse whitespace, clamp. Result still passes the schema.
 *
 * AND SCREEN IT FOR PLATFORM VOICE (audit item #11). Everything below this line
 * is INJECTION hygiene — it stops a message carrying markup or a bidi override
 * into a careless renderer. It says nothing whatever about what the message
 * CLAIMS. A creator's 280 characters land on somebody else's public profile and
 * on their own passport page beside a fan's handle, so a shout-out reading
 * "✓ Verified — Dhwaja Support" was, until now, published verbatim: a creator
 * could write a fake platform verification into a third party's page, while that
 * fan's own 40-character display name was refused for the identical string
 * (`social.ts` `displayNameImpersonates`, v14 P1). The same fold now runs here,
 * on the same shape of field, out of `platform-voice.ts`.
 *
 * Screened at SANITIZE time, which is the ceremony's FIRST leg, so a refused
 * message never becomes a signed draft the creator has to redo — and applied to
 * the CLEANED, CLAMPED string, because folding what will actually be stored is
 * what keeps the two ceremony legs in agreement. Retryable copy, not a 500. */
export function sanitizeRecognitionMessage(raw: unknown): string {
  if (typeof raw !== "string") throw new RecognitionError("message required");
  const cleaned = stripUnsafeDisplayText(raw.replace(/[<>]/g, ""))
    .replace(/\s+/g, " ")
    .trim();
  if (cleaned.length === 0) throw new RecognitionError("message is empty after sanitizing");
  const clamped = cleaned.slice(0, RECOGNITION_MAX_LEN).trim();
  if (claimsPlatformVoice(clamped)) throw new RecognitionError("recognition messages can't claim verification or platform-staff status", 409);
  return clamped;
}

/** Is this fan in the creator's OWN linked audience — a saver OR a code-claimer of
 * this creator? Only such fans may be followed; recognition can never name an
 * arbitrary account. */
async function isAudience(db: DbLike, passportId: string, accountId: string): Promise<boolean> {
  const saved = await db.prepare("SELECT 1 AS x FROM saves WHERE passport_id = ? AND account_id = ? LIMIT 1").bind(passportId, accountId).first();
  if (saved) return true;
  const claimed = await db.prepare("SELECT 1 AS x FROM reverse_code_claims k JOIN reverse_code_pools p ON p.pool_id = k.pool_id WHERE p.passport_id = ? AND k.account_id = ? LIMIT 1").bind(passportId, accountId).first();
  return claimed !== null;
}

/** I3-0909: has this fan blocked recognitions from this creator? */
async function isRecognitionBlocked(db: DbLike, passportId: string, accountId: string): Promise<boolean> {
  const r = await db.prepare("SELECT 1 AS x FROM recognition_blocks WHERE account_id = ? AND passport_id = ? LIMIT 1").bind(accountId, passportId).first();
  return r !== null;
}

/** I3-0909: the fan's own switch. Idempotent both ways; projection-only (the fan
 * has no passport to sign with — same stance as `decideRecognition`). */
export async function setRecognitionBlock(db: DbLike, accountId: string, passportId: string, blocked: boolean, now: () => string = () => new Date().toISOString()): Promise<{ blocked: boolean }> {
  if (typeof accountId !== "string" || !RECOGNITION_ACCOUNT_RE.test(accountId)) throw new RecognitionError("bad account id");
  if (typeof passportId !== "string" || !PSP_RE.test(passportId)) throw new RecognitionError("no such creator", 404);
  if (blocked) {
    // I4-RG-FAN-01: bound to the account still existing, so a write landing after
    // an erasure commit leaves no row keyed by the erased fan's raw account id.
    await db
      .prepare("INSERT OR IGNORE INTO recognition_blocks (account_id, passport_id, created_at) SELECT ?, ?, ? WHERE EXISTS (SELECT 1 FROM accounts WHERE account_id = ?)")
      .bind(accountId, passportId, now(), accountId)
      .run();
    const live = await db.prepare("SELECT 1 AS x FROM accounts WHERE account_id = ?").bind(accountId).first();
    if (!live) throw new RecognitionError("no such account", 404);
  } else await db.prepare("DELETE FROM recognition_blocks WHERE account_id = ? AND passport_id = ?").bind(accountId, passportId).run();
  return { blocked };
}

/** Does the creator ACTIVELY follow this fan? (the gate for expressive kinds) */
async function isFollowing(db: DbLike, passportId: string, accountId: string): Promise<boolean> {
  const r = await db.prepare("SELECT 1 AS x FROM recognitions WHERE passport_id = ? AND account_id = ? AND kind = 'follow' AND retracted_at IS NULL LIMIT 1").bind(passportId, accountId).first();
  return r !== null;
}

/** Grants in the trailing 24h, split by weight:
 *  - "like"       — only `like` grants (the lightweight budget);
 *  - "named"      — the NAMED recognitions (everything except `like`:
 *                   follow/thanks/shout_out/spotlight).
 *
 * A third mode, "expressive" (named MINUS follow), existed only to fund the
 * founder's exemption `recognition_follow_daily_budget` and was DELETED with it
 * (audit item #24). The founder no longer follows everyone, so there is no
 * account whose follow budget has to be uncoupled from its named budget.
 *
 * Counts regardless of later retraction, so grant→retract can't refund a slot. */
type GrantBudget = "like" | "named";
async function grantsToday(db: DbLike, passportId: string, nowMs: number, budget: GrantBudget): Promise<number> {
  const since = new Date(nowMs - DAY_MS).toISOString();
  const sql =
    budget === "like"
      ? "SELECT COUNT(*) AS n FROM recognitions WHERE passport_id = ? AND kind = 'like' AND granted_at >= ?"
      : "SELECT COUNT(*) AS n FROM recognitions WHERE passport_id = ? AND kind <> 'like' AND granted_at >= ?";
  const r = await db.prepare(sql).bind(passportId, since).first<{ n: number }>();
  return r?.n ?? 0;
}

async function activeFollows(db: DbLike, passportId: string): Promise<number> {
  const r = await db.prepare("SELECT COUNT(*) AS n FROM recognitions WHERE passport_id = ? AND kind = 'follow' AND retracted_at IS NULL").bind(passportId).first<{ n: number }>();
  return r?.n ?? 0;
}

/** Is there already an ACTIVE recognition of this exact (fan, kind)? The
 * duplicate-active guard — you can't stack ten identical shout-outs on one fan. */
async function hasActiveOfKind(db: DbLike, passportId: string, accountId: string, kind: string): Promise<boolean> {
  const r = await db.prepare("SELECT 1 AS x FROM recognitions WHERE passport_id = ? AND account_id = ? AND kind = ? AND retracted_at IS NULL LIMIT 1").bind(passportId, accountId, kind).first();
  return r !== null;
}

/**
 * Validate a grant fully — used as a fast fail at /start (before the passkey
 * ceremony) and again right before the append, so the creator gets a clean,
 * specific refusal instead of a constraint error. Returns the tier. Throws
 * RecognitionError; mints/writes nothing.
 *
 * THIS FUNCTION IS NO LONGER THE ENFORCEMENT. It is the UX and the fast path.
 * Every cap it names is ALSO carried by the write itself — see
 * `buildRecognitionGrantStatements` — so a racer that overtakes one of these
 * reads is refused by the INSERT, not admitted by it.
 *
 * Concurrency, all three answers.
 *
 * DUPLICATE-ACTIVE — a partial UNIQUE index (one active (creator,fan,kind)).
 * A racing duplicate grant fails the projection INSERT and rolls the whole
 * signed batch back, so a fan can never end up double-followed or
 * double-shouted no matter how the legs interleave. That is the guard that
 * faces the FAN, and it has always been a constraint rather than a read.
 *
 * DAILY BUDGET and ACTIVE-FOLLOW CEILING — now `INSERT … SELECT … WHERE
 * (SELECT COUNT(*) …) < cap` on the projection row itself, plus an admission
 * gate that makes a refused INSERT fail the batch. Both are in the write.
 *
 * WHAT THIS USED TO SAY, kept because the measurement is the record. Until
 * audit item #28 these two caps were read-checked HERE and nowhere else, and
 * that was written down as a known, accepted residual: *a creator who pre-opens
 * several passkey challenges and fires the /finish legs together lands one
 * grant per challenge — MEASURED at 4 follows against the free named cap of 1,
 * with the racers held on the COUNT read. The overshoot is the width of the
 * burst, not one.* It was accepted on the grounds that the overshoot reached
 * only the creator's own tiered budget and their own follow ceiling — a
 * Pro-differentiator leak and a table-growth bound — never another user's feed,
 * the way an over-sent broadcast would.
 *
 * THAT IS CLOSED. `tests/invariants/concurrency-guards.test.ts` now fires N
 * simultaneous signed /finish legs against a cap of 1 and asserts exactly one
 * lands; it fails against the old code with the measured 4.
 *
 * WHAT WAS ACTUALLY IN THE WAY, since the old text named it and got it half
 * right. The objection was real: *"a conditional insert that matches no row
 * does NOT fail a D1 batch, so the signed `recognition_granted` event would
 * land with no projection row behind it — worse than the overshoot."* True, and
 * it is why the plain idiom could not simply be copied. The missing piece was
 * not a different cap shape but a way to turn "matched nothing" into "the batch
 * failed": `recognitionQuotaGate` (standing.ts) reads the projection row back
 * by id inside the same transaction and feeds it to a NOT NULL column, so an
 * absent row aborts the batch and the event never lands. The second objection —
 * that `consumeRateLimit`'s TUMBLING window cannot express a cap of N over a
 * TRAILING 24h — is also still true, and is why the cap lives in the INSERT's
 * own subquery (which can carry any predicate, `granted_at >= ?` included)
 * rather than in the rate limiter.
 */
export async function assertRecognitionAllowed(db: DbLike, passportId: string, input: RecognitionInput, nowMs: number): Promise<{ tier: "free" | "pro" }> {
  const kind = input.kind;
  if (typeof kind !== "string" || !(RECOGNITION_KINDS as readonly string[]).includes(kind)) throw new RecognitionError("bad kind");
  if (typeof input.account_id !== "string" || !RECOGNITION_ACCOUNT_RE.test(input.account_id)) throw new RecognitionError("bad account id");
  if (input.message != null && input.message !== "" && !RECOGNITION_MESSAGE_KINDS.has(kind)) throw new RecognitionError("a message is only allowed on thanks/shout_out/spotlight");

  const tier = await effectiveCreatorTier(db, passportId, nowMs);
  // I3-0909: a fan who blocked this creator receives nothing from them. The
  // refusal is MASKED as the kind's ordinary eligibility refusal (below), for the
  // reason `decideRecognition` tells the creator nothing: "this fan blocked you"
  // has a retaliation use and no product use.
  if (await isRecognitionBlocked(db, passportId, input.account_id)) {
    throw new RecognitionError(kind === "follow" ? "you can only follow fans who saved you or claimed one of your codes" : kind === "like" ? "you can only like fans who saved you or claimed one of your codes" : "follow this fan before recognizing them", 403);
  }
  // NO FOUNDER BRANCH — three exemptions were deleted from here (audit item
  // #24): `recognition_active_follow_cap`, `recognition_follow_daily_budget`
  // and `recognition_audience_gate`. All three rested on "the founder follows
  // everyone", which was true only because of an auto-created save edge that no
  // longer exists. The audience gate in particular has to bind the founder
  // hardest of anyone: it asks "is this fan already yours?", and the whole
  // reason the founder could answer yes about every fan on the platform was
  // that the platform had answered for them. Every creator, founder included,
  // now passes the same caps and the same gates.
  //
  // Two separate daily budgets: a `like` draws from the generous LIKE cap; every
  // other kind draws from the scarce named-recognition cap.
  if (kind === "like") {
    if ((await grantsToday(db, passportId, nowMs, "like")) >= LIKE_DAILY_CAP[tier]) {
      throw new RecognitionError(tier === "pro" ? `you've hit today's like limit (${LIKE_DAILY_CAP.pro})` : `free creators can like ${LIKE_DAILY_CAP.free} fans a day — Creator Pro raises it to ${LIKE_DAILY_CAP.pro}`, 429);
    }
  } else if ((await grantsToday(db, passportId, nowMs, "named")) >= RECOGNITION_DAILY_CAP[tier]) {
    throw new RecognitionError(tier === "pro" ? `you've hit today's recognition limit — Pro allows ${RECOGNITION_DAILY_CAP.pro} a day` : `free creators can give ${RECOGNITION_DAILY_CAP.free} named recognition a day — Creator Pro raises it to ${RECOGNITION_DAILY_CAP.pro}`, 429);
  }

  if (kind === "follow") {
    if (!(await isAudience(db, passportId, input.account_id))) throw new RecognitionError("you can only follow fans who saved you or claimed one of your codes", 403);
    if (await hasActiveOfKind(db, passportId, input.account_id, "follow")) throw new RecognitionError("you already follow this fan", 409);
    if ((await activeFollows(db, passportId)) >= ACTIVE_FOLLOW_CAP[tier]) throw new RecognitionError(`you're at your active-follow limit (${ACTIVE_FOLLOW_CAP[tier]})`, 409);
  } else if (kind === "like") {
    // A like is the lightweight ack: any fan in the creator's OWN audience, no
    // follow required. Still one active like per fan (the duplicate-active guard).
    if (!(await isAudience(db, passportId, input.account_id))) throw new RecognitionError("you can only like fans who saved you or claimed one of your codes", 403);
    if (await hasActiveOfKind(db, passportId, input.account_id, "like")) throw new RecognitionError("you already liked this fan", 409);
  } else {
    // thanks / shout_out / spotlight — the named recognitions require an active follow.
    if (!(await isFollowing(db, passportId, input.account_id))) throw new RecognitionError("follow this fan before recognizing them", 403);
    if (await hasActiveOfKind(db, passportId, input.account_id, kind)) throw new RecognitionError("you've already given this recognition to this fan", 409);
  }
  return { tier };
}

/**
 * THE CAP IS IN THE INSERT (audit item #28).
 *
 * Three literal statements, never one with a column name or an operator
 * interpolated — the same reason `decideRecognition` below spells its two
 * spellings out in full. Which one a grant takes is decided by its KIND, and
 * every one of them is the canonical `INSERT … SELECT … WHERE (SELECT COUNT(*)
 * …) < cap` shape (accounts.ts `saveCreator`, eight sites in the tree):
 *
 *   LIKE   — the lightweight budget: `like` grants in the trailing 24h < cap.
 *   NAMED  — the scarce budget: everything that is not a `like`, same window.
 *   FOLLOW — the named budget AND the ACTIVE-follow ceiling, which counts live
 *            rows (`retracted_at IS NULL`) rather than a window, because
 *            retracting a follow DOES give the slot back while retracting a
 *            grant does NOT give the day's budget back.
 *
 * THE WINDOW IS TRAILING, and that is why this is a subquery rather than a
 * `consumeRateLimit` bucket: `granted_at >= ?` is exact at every instant, where
 * a tumbling window lets a burst straddling the boundary spend two budgets.
 * The counts read the SAME rows `grantsToday` / `activeFollows` read, so the
 * headroom the queue shows a creator and the cap the write enforces cannot
 * drift.
 *
 * Both statements are handed to `appendCreatorSigned` as extraStatements, so
 * they ride the SAME atomic batch as the signed `recognition_granted` event —
 * never a separate write that could diverge from it.
 *
 * WHY THERE ARE TWO STATEMENTS AND NOT ONE. A conditional INSERT that matches
 * no row does not fail a D1 batch; it just does nothing, and the batch commits
 * around it. On this path that would leave the signed event with no projection
 * row behind it, which is worse than the cap overshoot it replaced. The second
 * statement — `recognitionQuotaGate` — reads the row back by id in the same
 * transaction and supplies it to a NOT NULL column, so "the cap refused" and
 * "the batch fails" become the same event. They are returned TOGETHER, from one
 * function, for exactly the reason `requireVerifiedCreator` is a witness rather
 * than an `if`: a caller cannot take half of it.
 *
 * The partial-unique index still makes a concurrent duplicate-active grant fail
 * here too, rolling back the same batch — that guard is unchanged.
 */
const GRANT_INSERT_HEAD =
  "INSERT INTO recognitions (recognition_id, passport_id, account_id, kind, message, granted_at) SELECT ?, ?, ?, ?, ?, ?";

/** `like` grants in the trailing window < the like cap. */
const LIKE_BUDGET_CLAUSE =
  " WHERE (SELECT COUNT(*) FROM recognitions rb WHERE rb.passport_id = ? AND rb.kind = 'like' AND rb.granted_at >= ?) < ?";

/** NAMED grants (everything except `like`) in the trailing window < the named cap. */
const NAMED_BUDGET_CLAUSE =
  " WHERE (SELECT COUNT(*) FROM recognitions rb WHERE rb.passport_id = ? AND rb.kind <> 'like' AND rb.granted_at >= ?) < ?";

/** I3-0909: …and never toward a fan who blocked this creator. Appended to EVERY
 * grant INSERT, so a block landing between the two ceremony legs makes the INSERT
 * match nothing and `recognitionQuotaGate` fail the whole signed batch. */
const NOT_BLOCKED_CLAUSE = " AND NOT EXISTS (SELECT 1 FROM recognition_blocks bl WHERE bl.account_id = ? AND bl.passport_id = ?)";

/** …and, for a follow, the live active-follow ceiling on top of it. */
const FOLLOW_CEILING_CLAUSE =
  " AND (SELECT COUNT(*) FROM recognitions rf WHERE rf.passport_id = ? AND rf.kind = 'follow' AND rf.retracted_at IS NULL) < ?";

export function buildRecognitionGrantStatements(
  db: DbLike,
  passportId: string,
  input: RecognitionInput,
  recognitionId: string,
  tier: "free" | "pro",
  nowMs: number,
  now: () => string = () => new Date().toISOString(),
): DbStatement[] {
  const message = input.message != null && input.message !== "" && RECOGNITION_MESSAGE_KINDS.has(input.kind) ? input.message : null;
  const ts = now();
  const since = new Date(nowMs - DAY_MS).toISOString();
  const row = [recognitionId, passportId, input.account_id, input.kind, message, ts];
  let insert: DbStatement;
  if (input.kind === "like") {
    insert = db.prepare(GRANT_INSERT_HEAD + LIKE_BUDGET_CLAUSE + NOT_BLOCKED_CLAUSE).bind(...row, passportId, since, LIKE_DAILY_CAP[tier], input.account_id, passportId);
  } else if (input.kind === "follow") {
    insert = db.prepare(GRANT_INSERT_HEAD + NAMED_BUDGET_CLAUSE + FOLLOW_CEILING_CLAUSE + NOT_BLOCKED_CLAUSE).bind(...row, passportId, since, RECOGNITION_DAILY_CAP[tier], passportId, ACTIVE_FOLLOW_CAP[tier], input.account_id, passportId);
  } else {
    insert = db.prepare(GRANT_INSERT_HEAD + NAMED_BUDGET_CLAUSE + NOT_BLOCKED_CLAUSE).bind(...row, passportId, since, RECOGNITION_DAILY_CAP[tier], input.account_id, passportId);
  }
  return [
    insert,
    recognitionQuotaGate(db, recognitionId, () => ts),
    // THE STANDING GATE, WHICH THE QUOTA GATE IS NOT (audit v11, D5).
    //
    // `recognitionQuotaGate` proves "a `recognitions` row with this id exists at
    // commit time" — that the CAPS admitted the grant. It says nothing whatever
    // about the creator's standing, and the only standing check on this route was
    // `isVerifiedPassport` in the pre-read (recognition-routes.ts:87), which is a
    // read on the far side of the same window every mint path had to close. So a
    // creator downgraded or suppressed between the ceremony's two legs still
    // landed a signed `recognition_granted` event and wrote their passport id
    // into `recognitions` — a public, fan-visible affiliation minted under
    // standing they no longer had.
    //
    // Two gates, two propositions, one batch: `write_admissions` is keyed
    // (gate, subject) so 'recognition_quota' and 'verified_mint' are separate
    // rows and neither can be mistaken for the other. Either NOT NULL fires and
    // the whole signed batch rolls back.
    verifiedMintGate(db, passportId, recognitionId, () => ts),
  ];
}

/** Validate a retract: the caller owns an ACTIVE recognition with this id. Returns
 * its kind + fan (so the retract event can be built and the response is useful). */
export async function assertRetractAllowed(db: DbLike, passportId: string, recognitionId: string): Promise<{ kind: string; account_id: string }> {
  if (typeof recognitionId !== "string" || !RECOGNITION_ID_RE.test(recognitionId)) throw new RecognitionError("no such recognition", 404);
  const r = await db.prepare("SELECT passport_id, account_id, kind, retracted_at FROM recognitions WHERE recognition_id = ?").bind(recognitionId).first<{ passport_id: string; account_id: string; kind: string; retracted_at: string | null }>();
  if (!r) throw new RecognitionError("no such recognition", 404);
  if (r.passport_id !== passportId) throw new RecognitionError("not your recognition", 403);
  if (r.retracted_at !== null) throw new RecognitionError("already retracted", 409);
  return { kind: r.kind, account_id: r.account_id };
}

/** Build the projection UPDATE for a retract — passed as an extraStatement so it
 * rides the same atomic batch as the signed recognition_retracted event. The
 * guarded WHERE (own row + still-active) makes a double-submit a no-op. */
export function buildRecognitionRetractStatement(db: DbLike, passportId: string, recognitionId: string, now: () => string = () => new Date().toISOString()): DbStatement {
  return db.prepare("UPDATE recognitions SET retracted_at = ? WHERE recognition_id = ? AND passport_id = ? AND retracted_at IS NULL").bind(now(), recognitionId, passportId);
}

export interface ReceivedRecognition {

  passport_id: string; // the creator who recognized them
  kind: string;
  message: string | null;

  /** 0017 consent (item #11). Both NULL = PENDING: the fan can see it, nobody
   * else can. Carried on the fan's OWN read only — the creator's `granted` view
   * has no consent column, deliberately (see `decideRecognition`). */
  accepted_at: string | null;
  declined_at: string | null;
  /** I3-0373: the recognizing creator's display name, so the fan's list can say
   * WHO rather than "by this creator". NULL for a retracted passport — the name
   * is exactly what a retraction withdraws from view (the id row stays, above). */
  creator_display_name: string | null;

/** A fan's ACTIVE received recognitions — who recognized them, and how.
 *
 * A SUPPRESSED creator drops out, on the same terms as every other owner-scoped
 * fan read: `listSaves` and `listNotifications` (accounts.ts) and the codex
 * already drop one, and this row carries the delisted person's `passport_id`
 * AND the free text they wrote, so it was the one owner-scoped surface still
 * serving both. Owner-scoped is not a licence — the fan is not a third party,
 * but "gone" that holds everywhere except one screen is a promise with an
 * exception nobody documented, and this surface disagreed with the PUBLIC
 * projection of the SAME recognitions (`recognition-public.ts`) about the same
 * fact. The `recognitions` row is left alone: lifting the suppression restores
 * the entry exactly, and deleting a fan's own history to hide a creator would
 * be the wrong trade.
 *
 * NOT filtered here, and deliberately: `p.retracted`. The public projection
 * drops a retracted creator because it prints their display NAME to strangers;
 * this read is the fan's own record and shows a passport id, so it follows
 * `listNotifications` — suppression (the person asked to be gone) removes,
 * claim-state churn does not. `GET /account/export` (account_data.ts) still
 * carries every recognition unfiltered: a data-subject export is the one place
 * completeness outranks this, and that is its own documented stance. */
export interface ReceivedRecognitionPage {
  recognitions: ReceivedRecognition[];
  next: { granted_at: string; recognition_id: string } | null;
}

/** Q-16 fix: `granted_at` alone is not a total order (same-instant grants
 * collide), so `(granted_at, recognition_id) < (?, ?)` is the keyset — the
 * same row-value-comparison shape `listCollection` (collections.ts) already
 * uses for the identical reason. `recognition_id` is globally unique
 * (ids.ts), so the pair is a genuine total order over one account's rows. */
export async function recognitionsReceived(
  db: DbLike,
  accountId: string,
  before?: { granted_at: string; recognition_id: string },
  limit = 100,
): Promise<ReceivedRecognitionPage> {
  const capped = Math.max(1, Math.min(200, limit));
  const q = before
    ? db
        .prepare(
          `SELECT r.recognition_id, r.passport_id, r.kind, r.message, r.granted_at, r.accepted_at, r.declined_at,
                  CASE WHEN p.retracted = 0 THEN p.display_name END AS creator_display_name
             FROM recognitions r LEFT JOIN passports p ON p.passport_id = r.passport_id
            WHERE r.account_id = ? AND r.retracted_at IS NULL AND (r.granted_at, r.recognition_id) < (?, ?)
              AND NOT EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = r.passport_id)
            ORDER BY r.granted_at DESC, r.recognition_id DESC LIMIT ?`,
        )
        .bind(accountId, before.granted_at, before.recognition_id, capped)
    : db
        .prepare(
          `SELECT r.recognition_id, r.passport_id, r.kind, r.message, r.granted_at, r.accepted_at, r.declined_at,
                  CASE WHEN p.retracted = 0 THEN p.display_name END AS creator_display_name
             FROM recognitions r LEFT JOIN passports p ON p.passport_id = r.passport_id
            WHERE r.account_id = ? AND r.retracted_at IS NULL
              AND NOT EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = r.passport_id)
            ORDER BY r.granted_at DESC, r.recognition_id DESC LIMIT ?`,
        )
        .bind(accountId, capped);
  const { results } = await q.all<ReceivedRecognition>();
  const last = results[results.length - 1];
  const next = results.length === capped && last ? { granted_at: last.granted_at, recognition_id: last.recognition_id } : null;
  return { recognitions: results, next };
}

// ---------- fan-side consent (audit item #11) ----------

/**
 * THE CONSENT STATE MACHINE, and the only two doors into it.
 *
 *   pending  (accepted_at IS NULL, declined_at IS NULL)  — the state every new
 *            grant starts in. Visible to the named fan, published NOWHERE.
 *   accepted (accepted_at set)   — published on /u/:handle and, for a spotlight,
 *            in Fan of the Week.
 *   declined (declined_at set)   — published nowhere, TERMINALLY. Nothing the
 *            creator can do moves a row out of this state, and `acceptRecognition`
 *            refuses it too, so "no" does not decay into "yes" through a UI
 *            mis-tap, a replayed request, or a retract-and-regrant loop.
 *
 * WHY DECLINE IS TERMINAL RATHER THAN A TOGGLE. A creator who wanted a declined
 * shout-out live again could otherwise retract and re-grant until the fan
 * mis-taps; making the row terminal means the creator must retract (which the
 * fan sees) and the fan must affirmatively accept the NEW row. A fan who
 * declined by accident is not stuck with a wrong record either — the row simply
 * never publishes, and the creator can retract and offer again.
 *
 * WHAT THE CREATOR IS TOLD: nothing. `recognitionsGranted` is unchanged and
 * carries no consent column. "This fan refused you" is a fact with an obvious
 * retaliation use and no product use — the creator's own view already shows the
 * recognition they gave, and a fan's refusal to be quoted is the fan's business.
 * That is also why the public reads treat pending, declined and non-existent
 * identically: a withheld recognition must not be legible AS withheld.
 *
 * NEITHER DOOR TOUCHES THE LEDGER. `recognition_granted` / `recognition_retracted`
 * are passkey-signed CREATOR events; consent is the fan's answer to one, not a
 * creator act, and the fan has no passport to sign with. It is a projection-only
 * guarded UPDATE, exactly like `setBroadcastsMuted` — the other fan-side consent
 * valve — so the fold is untouched and a replay still reproduces state.
 */
export type ConsentDecision = "accept" | "decline";

/**
 * Record the named fan's decision on ONE received recognition. Returns the new
 * state. ONE guarded UPDATE, then one read-back only on the no-op path: the
 * WHERE carries every precondition (`account_id = ?` — the fan owns it; still
 * active; not already resolved the other way), so a double-submit and a
 * cross-account attempt are both `changes === 0` rather than a check-then-act
 * window two concurrent requests could both pass.
 *
 * The read-back exists to turn that 0 into an HONEST status: "no such
 * recognition" (404) and "you already answered this one" (409) are different
 * facts, and answering 404 for both would tell a fan their own accepted
 * recognition had vanished. It runs ONLY when the update matched nothing, so
 * the success path is a single statement.
 */
export async function decideRecognition(
  db: DbLike,
  accountId: string,
  recognitionId: string,
  decision: ConsentDecision,
  now: () => string = () => new Date().toISOString(),
): Promise<{ recognition_id: string; state: "accepted" | "declined" }> {
  if (typeof recognitionId !== "string" || !RECOGNITION_ID_RE.test(recognitionId)) throw new RecognitionError("no such recognition", 404);
  if (typeof accountId !== "string" || !RECOGNITION_ACCOUNT_RE.test(accountId)) throw new RecognitionError("no such recognition", 404);
  // Two whole literal statements, never one with the column name interpolated:
  // a SQL identifier assembled by concatenation is the shape the paywall guard
  // refuses outright elsewhere in this tree, and there is no reason to introduce
  // it here for two spellings that fit on two lines.
  //
  // A SUPPRESSED CREATOR'S RECOGNITION IS UNANSWERABLE, not merely unpublished.
  // `recognitionsReceived` already drops a delisted creator's rows from the
  // fan's own list, so without the same clause here a fan holding an old id
  // would get 200 for a delisted creator and 404 for an id that never existed —
  // the "did this person ask to be delisted?" oracle, re-opened on a brand new
  // route by the one caller who still remembers the id. The clause rides both
  // statements, so the answer and the D1 shape both match a nonexistent id
  // (tests/invariants/suppression-oracles.test.ts probes exactly this).
  const suppressedClause = "AND NOT EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = recognitions.passport_id)";
  const sql =
    decision === "accept"
      ? `UPDATE recognitions SET accepted_at = ?
          WHERE recognition_id = ? AND account_id = ? AND retracted_at IS NULL
            AND accepted_at IS NULL AND declined_at IS NULL ${suppressedClause}`
      : `UPDATE recognitions SET declined_at = ?
          WHERE recognition_id = ? AND account_id = ? AND retracted_at IS NULL
            AND accepted_at IS NULL AND declined_at IS NULL ${suppressedClause}`;
  const res = await db.prepare(sql).bind(now(), recognitionId, accountId).run();
  if (res.changes === 0) {
    // Owner-scoped read-back: `account_id = ?` is in the WHERE, so a stranger
    // probing another fan's recognition id gets the same 404 an unknown id gets
    // — this path must not become a "does this recognition exist" oracle.
    const row = await db
      .prepare(`SELECT accepted_at, declined_at, retracted_at FROM recognitions WHERE recognition_id = ? AND account_id = ? ${suppressedClause}`)
      .bind(recognitionId, accountId)
      .first<{ accepted_at: string | null; declined_at: string | null; retracted_at: string | null }>();
    if (!row || row.retracted_at !== null) throw new RecognitionError("no such recognition", 404);
    throw new RecognitionError(row.declined_at !== null ? "you already declined this recognition" : "you already accepted this recognition", 409);
  }
  return { recognition_id: recognitionId, state: decision === "accept" ? "accepted" : "declined" };
}

export interface QueueFan {
  account_id: string;
  source: "saver" | "claimer" | "both";
  following: boolean;
  given: string[]; // active recognition kinds already granted to this fan
}

export interface RecognitionQueue {
  passport_id: string;
  generated_at: string;
  tier: "free" | "pro";
  grants_today: number; // NAMED recognitions given today (follow/thanks/shout_out/spotlight)
  daily_cap: number; // the named-recognition daily cap
  likes_today: number;
  like_cap: number;
  active_follows: number;
  follow_cap: number;
  fans: QueueFan[];
}

/**
 * The Recognition Queue: the creator's own audience (savers ∪ code-claimers) with
 * their current follow/recognition state, plus the creator's live cap headroom.
 * Bounded (each source LIMIT-ed); reveals account_ids ONLY to that fan's own
 * creator. CALLER-GATING: the HTTP layer passes the authenticated creator's own
 * verified passport (like the ROI dashboard).
 */
export async function recognitionQueue(db: DbLike, passportId: string, nowMs: number = Date.now(), now: () => string = () => new Date().toISOString(), limit = 100): Promise<RecognitionQueue> {
  const capped = Math.max(1, Math.min(200, limit));
  const tier = await effectiveCreatorTier(db, passportId, nowMs);

  // I3-1894: PRIVACY.md:29 "Creators see a count, never a name" — a fan who
  // ONLY saved is listed by id to the creator only when that fan opted in with
  // `user_profiles.affiliation_visibility = 'public'` (the Top Fans consent,
  // leaderboard.ts). Claimers are unchanged; aggregates still count every saver.
  const savers = (
    await db
      .prepare(
        "SELECT sv.account_id AS account_id FROM saves sv JOIN user_profiles up ON up.account_id = sv.account_id AND up.affiliation_visibility = 'public' WHERE sv.passport_id = ? ORDER BY sv.created_at DESC LIMIT 500",
      )
      .bind(passportId)
      .all<{ account_id: string }>()
  ).results;
  // I3-1876: newest claim first, same direction as the savers arm — without an
  // ORDER BY, WHICH 500 claimers survive the LIMIT was the planner's choice.
  // I15-FX6 (FX2-SIB-RECOG): JOIN accounts — an erased or undone claim is a `del_`
  // tombstone with no accounts row, and is never listed to the creator as a claimer.
  const claimers = (
    await db
      .prepare("SELECT k.account_id AS account_id FROM reverse_code_claims k JOIN reverse_code_pools p ON p.pool_id = k.pool_id JOIN accounts a ON a.account_id = k.account_id WHERE p.passport_id = ? GROUP BY k.account_id ORDER BY MAX(k.claimed_at) DESC, k.account_id LIMIT 500")
      .bind(passportId)
      .all<{ account_id: string }>()
  ).results;

  const source = new Map<string, "saver" | "claimer" | "both">();
  for (const r of savers) source.set(r.account_id, "saver");
  for (const r of claimers) source.set(r.account_id, source.has(r.account_id) ? "both" : "claimer");

  // Active recognitions this creator holds, grouped by fan — ONLY for the fans
  // the two capped arms above returned (I3-2102). Bound: <= 1000 fans x one
  // active row per kind (idx_recognitions_active_unique). No LIMIT: a LIMIT here
  // would silently drop `given` kinds for fans that ARE listed.
  const active = (
    await db
      .prepare("SELECT account_id, kind FROM recognitions WHERE passport_id = ? AND retracted_at IS NULL AND account_id IN (SELECT value FROM json_each(?))")
      .bind(passportId, JSON.stringify([...source.keys()]))
      .all<{ account_id: string; kind: string }>()
  ).results;
  const givenByFan = new Map<string, Set<string>>();
  for (const r of active) {
    let s = givenByFan.get(r.account_id);
    if (!s) {
      s = new Set();
      givenByFan.set(r.account_id, s);
    }
    s.add(r.kind);
  }

  const fans: QueueFan[] = [...source.entries()].map(([account_id, src]) => {
    const given = givenByFan.get(account_id) ?? new Set<string>();
    return { account_id, source: src, following: given.has("follow"), given: [...given] };
  });
  // Fans the creator already follows float to the top (they're recognizable now).
  fans.sort((a, b) => (a.following === b.following ? (a.account_id < b.account_id ? -1 : 1) : a.following ? -1 : 1));

  return {
    passport_id: passportId,
    generated_at: now(),
    tier,
    grants_today: await grantsToday(db, passportId, nowMs, "named"),
    daily_cap: RECOGNITION_DAILY_CAP[tier],
    likes_today: await grantsToday(db, passportId, nowMs, "like"),
    like_cap: LIKE_DAILY_CAP[tier],
    active_follows: await activeFollows(db, passportId),
    follow_cap: ACTIVE_FOLLOW_CAP[tier],
    fans: fans.slice(0, capped),
  };
}

export interface RecognitionVolumeRow {
  passport_id: string;
  granted: number; // total recognitions ever granted by this creator
  active: number; // currently active
  retracted: number; // retracted
  spotlights_active: number; // active spotlights — the public Fan-of-the-Week surface, the likeliest gaming vector
}

/**
 * Per-creator recognition volume — OPERATOR-ONLY (gated at the /ops route). An
 * aggregate for spotting abuse (e.g. a creator mass-spotlighting to game the public
 * widget). No fan ids or handles — creator-level counts only. Bounded top-N by volume.
 */
export async function recognitionVolume(db: DbLike, limit = 50): Promise<RecognitionVolumeRow[]> {
  const capped = Math.max(1, Math.min(200, limit));
  const { results } = await db
    .prepare(
      `SELECT passport_id AS passport_id,
              COUNT(*) AS granted,
              SUM(CASE WHEN retracted_at IS NULL THEN 1 ELSE 0 END) AS active,
              SUM(CASE WHEN retracted_at IS NOT NULL THEN 1 ELSE 0 END) AS retracted,
              SUM(CASE WHEN kind = 'spotlight' AND retracted_at IS NULL THEN 1 ELSE 0 END) AS spotlights_active
         FROM recognitions
        GROUP BY passport_id
        ORDER BY granted DESC
        LIMIT ?`,
    )
    .bind(capped)
    .all<{ passport_id: string; granted: number; active: number; retracted: number; spotlights_active: number }>();
  return results.map((r) => ({
    passport_id: r.passport_id,
    granted: r.granted ?? 0,
    active: r.active ?? 0,
    retracted: r.retracted ?? 0,
    spotlights_active: r.spotlights_active ?? 0,
  }));
}

const RECAP_WINDOW_DAYS = 7;

export interface RecognitionRecap {
  passport_id: string;
  generated_at: string;
  window_days: number;
  granted_total: number; // grants in the window (all kinds)
  granted_by_kind: Record<string, number>; // window, per kind
  new_follows: number; // follows granted in the window
  active_follows: number; // current
  active_spotlights: number; // current
}

/** A creator's own weekly recognition recap (feature 79) — what they did in the
 * last 7 days + where they stand now. Creator-private, verified-gated at the route. */
export async function recognitionRecap(db: DbLike, passportId: string, nowMs: number = Date.now(), now: () => string = () => new Date().toISOString()): Promise<RecognitionRecap> {
  if (!PSP_RE.test(passportId)) throw new Error("bad passport id");
  const since = new Date(nowMs - RECAP_WINDOW_DAYS * DAY_MS).toISOString();
  const { results } = await db.prepare("SELECT kind, COUNT(*) AS n FROM recognitions WHERE passport_id = ? AND granted_at >= ? GROUP BY kind").bind(passportId, since).all<{ kind: string; n: number }>();
  const byKind: Record<string, number> = {};
  let total = 0;
  let newFollows = 0;
  for (const r of results) {
    const n = typeof r.n === "number" ? r.n : 0;
    byKind[r.kind] = n;
    total += n;
    if (r.kind === "follow") newFollows = n;
  }
  const activeSpot = await db.prepare("SELECT COUNT(*) AS n FROM recognitions WHERE passport_id = ? AND kind = 'spotlight' AND retracted_at IS NULL").bind(passportId).first<{ n: number }>();
  return {
    passport_id: passportId,
    generated_at: now(),
    window_days: RECAP_WINDOW_DAYS,
    granted_total: total,
    granted_by_kind: byKind,
    new_follows: newFollows,
    active_follows: await activeFollows(db, passportId),
    active_spotlights: activeSpot?.n ?? 0,
  };
}

export interface FanOfWeekEntry {
  handle: string; // the spotlighted fan's PUBLIC handle (their opt-in identity)
  display_name: string;
  message: string | null;
  granted_at: string;

/**
 * The creator's currently-spotlighted fans — "Fan of the Week" (feature 78), safe
 * to serve on the creator's PUBLIC passport page. Only fans with a PUBLIC profile
 * appear (their handle is their opt-in public identity); a private/handleless
 * spotlighted fan is omitted — the raw account_id is never exposed. A RETRACTED
 * creator serves nothing (retraction is the kill-switch for every circulating
 * share surface — the impostor must not keep a named social-proof widget live).
 * Bounded.
 *
 * A SUPPRESSED creator likewise serves nothing (measured gap; `removal.ts`:
 * "SUPPRESSION IS INDISTINGUISHABLE FROM NEVER-LISTED"). `GET
 * /p/:id/fan-of-the-week` answers `{fans: []}` for an id that never existed, so
 * the empty list IS the never-listed answer — a 404 for a suppressed subject
 * would MAKE the "did this person ask to be delisted?" oracle instead of closing
 * it. This route was also the worst of the set for third parties: it served the
 * FANS' handles, display names and free-text messages, i.e. other people's data,
 * off a delisted creator's passport id.
 *
 * The gate rides the SAME row-level join that already enforces retraction
 * (`NOT EXISTS` on `request_suppressions`), so it is one query, no extra
 * statement, and it sits in the only function that reads spotlight recognitions
 * for a public creator surface — a second caller of this function inherits it,
 * and there is no other path from `recognitions` to this shape.
 *
 * TWO CONSENT GATES WERE ADDED HERE, BOTH ON THAT SAME STATEMENT (0017):
 *
 *  - `r.accepted_at IS NOT NULL AND r.declined_at IS NULL` (audit item #11). A
 *    spotlight publishes the FAN's handle, display name and the CREATOR's free
 *    text on the creator's own page. `u.visibility = 'public'` was doing the
 *    consent work, and it cannot: it is the fan's answer to "may people see my
 *    profile", asked long before any creator wrote anything about them.
 *  - `u.affiliation_visibility = 'public'` (audit item #12). The separable
 *    lever: a fan may keep a public profile and still refuse to appear on other
 *    people's pages. It is a predicate on a column of the row this join ALREADY
 *    fetches, so it costs nothing and cannot be forgotten by a caller.
 *
 * Both are extra `AND`s rather than an early return, for the reason the
 * suppression gate is: `tests/invariants/suppression-oracles.test.ts` pins this
 * route for statement-SEQUENCE identity, and it pins the empty-200 semantics
 * explicitly "so a later hardening pass cannot invert it". A withheld spotlight
 * therefore reads exactly like a creator who has spotlighted nobody — which is
 * also the right answer for the FAN, whose refusal must not be legible to the
 * public as a refusal.
 */
export async function fanOfTheWeek(db: DbLike, passportId: string, limit = 10): Promise<FanOfWeekEntry[]> {
  if (!PSP_RE.test(passportId)) return [];
  const capped = Math.max(1, Math.min(50, limit));
  const { results } = await db
    .prepare(
      `SELECT r.message AS message, r.granted_at AS granted_at, u.handle AS handle, u.display_name AS display_name
         FROM recognitions r
         JOIN user_profiles u ON u.account_id = r.account_id AND u.visibility = 'public' AND u.affiliation_visibility = 'public'
         JOIN passports p ON p.passport_id = r.passport_id AND p.retracted = 0
        WHERE r.passport_id = ? AND r.kind = 'spotlight' AND r.retracted_at IS NULL
          AND r.accepted_at IS NOT NULL AND r.declined_at IS NULL
          AND NOT EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = r.passport_id)
        ORDER BY r.granted_at DESC LIMIT ?`,
    )
    .bind(passportId, capped)
    .all<{ message: string | null; granted_at: string; handle: string; display_name: string | null }>();
  return results.map((r) => ({ handle: r.handle, display_name: r.display_name ?? "", message: r.message ?? null, granted_at: r.granted_at }));
}
