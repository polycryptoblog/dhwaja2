-- 0047_subject_kind.sql — a passport gains a KIND, and a character gains an owner.
--
-- Dhwaja lists creators. Some of those creators are not people: an AI character
-- or a brand mascot has fans, gets impersonated, and needs the same protection —
-- but it has no identity of its own to verify and nobody it can consent on
-- behalf of. These two columns are what make that representable.
--
-- KIND IS AN AXIS, NOT A VERDICT. `subject_kind` is orthogonal to `claim_state`:
-- a character walks unclaimed → claimed → verified on exactly the proofs a human
-- uses. Nothing about the registry bucket entry changes, so the k-anonymous
-- artifact, its format version, the four client verdict tables and the
-- cross-client parity guard are all untouched. Adding a fifth VERDICT word would
-- have meant editing Swift and Kotlin; adding an axis does not.
--
-- DEFAULT 'person' AND SQLITE BACKFILLS IT, so every existing passport is a
-- person and no meaning is migrated. There is no row whose kind is unknown.
ALTER TABLE passports ADD COLUMN subject_kind TEXT NOT NULL DEFAULT 'person'
  CHECK (subject_kind IN ('person','ai_character','brand'));

-- WHO ANSWERS FOR A CHARACTER — an active agency (`agc_<16 hex>`) or a currently
-- verified PERSON (`psp_<16 hex>`). NULL for a person, by construction.
--
-- Why it is not an `agency_authorized` grant: that event requires a
-- `creator_passkey`, i.e. the creator consenting to be represented. A character
-- has no creator to consent, and cannot hold a passkey until it is claimed —
-- which is the very thing the binding is supposed to gate. Using the existing
-- grant would have deadlocked, so the declaration and the binding are one
-- operator act, because for a character they are one decision.
--
-- Deliberately NOT a foreign key: it points at either `agencies` or `passports`
-- depending on its prefix, and SQLite cannot express that. The claim gate
-- resolves it against both tables and refuses anything it cannot resolve — see
-- `claims.ts claimConsumedGateStmt`, where the check rides INSIDE the batch that
-- appends `claim_proven` rather than beside it.
--
-- NOTE ON INVARIANT [append-only]: `passports` is a FOLD of the ledger. Both
-- columns are written by the fold from `subject_kind_declared`, never by a SQL
-- backfill, and `fold.ts` mirrors them so the second reducer still agrees
-- byte-for-byte.
ALTER TABLE passports ADD COLUMN operated_by TEXT;

-- 0048_character_operator_watch.sql — the sweep's own memory of when a
-- character's operator went quiet.
--
-- WHY THIS TABLE EXISTS AT ALL. Slice 14 gated the claim and the verification on
-- a live accountable operator, and evaluated that gate AT THE WRITE — so a
-- character verified while its agency was active keeps its badge forever after
-- the agency is suspended. The sweep that closes that hole must not de-badge on
-- first sight: an agency suspended for twenty minutes during an ops action would
-- otherwise permanently strip every character it runs, each one needing its own
-- operator re-verification to come back. So there is a grace window.
--
-- A grace window needs a start, and there is nowhere to read one. `agencies` has
-- `status` and NO `suspended_at`, and the other three ways an operator dies
-- (the bound person unverified, retracted, or delisted at their own request)
-- each date themselves somewhere different. The sweep therefore records its own
-- first observation, which is also the only timestamp that means the right
-- thing: how long WE have known, not how long it has been true.
--
-- NOTE ON INVARIANT [append-only]: this is NOT ledger state and does not belong
-- in `trust_events`. "The sweep noticed something" is not a fact Dhwaja asserts
-- about a subject — nothing public reads this table, no verdict moves because of
-- it, and deleting every row costs nothing but a restarted clock. It is the same
-- class as `sweep_cursor` and `review_queue`. The CONSEQUENTIAL act is the
-- `operator_lapsed` event appended at the end of the window, and that is a
-- ledger row like everything else.
--
-- A ROW EXISTS ONLY WHILE THE LAPSE DOES. The sweep deletes it the moment the
-- operator is live again, so a lapse repaired inside the window leaves no trace
-- and sends no second notice — recovery is automatic and silent, which is what
-- makes a short suspension a non-event rather than an incident.
CREATE TABLE character_operator_watch (
  -- The CHARACTER, not the operator. One agency may run many characters and each
  -- one's window runs from when the sweep first saw THAT character unaccountable,
  -- so a character declared after the suspension gets its own full window.
  passport_id   TEXT PRIMARY KEY REFERENCES passports(passport_id),
  -- ISO. The window's start, and the `lapsed_at` copied verbatim into the
  -- `operator_lapsed` payload so the append-only row carries its own evidence.
  first_seen_at TEXT NOT NULL,
  -- THE GRANT THIS WINDOW WAS OPENED AGAINST — `trust_events.seq` of the most
  -- recent `verification_granted` for this passport when the window opened, or 0.
  --
  -- Without it a stale row silently truncates the NEXT window to nothing. A
  -- character can leave `verified` by a route this sweep never sees (the expiry
  -- sweep on the same tick, an operator downgrade, a retraction) and then be
  -- re-verified; the old row survives, and the first tick after the
  -- re-verification measures the window from the ORIGINAL observation — so a
  -- creator who has just been re-verified gets minutes of grace instead of seven
  -- days, and no notice, because the dedup key is still the old window's.
  -- Measured in adversarial review, on a sequence of ordinary operator actions.
  --
  -- A different grant seq means an operator has AFFIRMED this character since we
  -- opened the window, so the window is void and a fresh one starts.
  grant_seq     INTEGER NOT NULL DEFAULT 0,
  -- ISO, or '' when no notice has gone out yet. The digest is deduped on
  -- `first_seen_at` besides (period_key), so this is the cheap guard that stops
  -- the sweep re-deriving a notice it already sent on every six-hourly tick.
  notified_at   TEXT NOT NULL DEFAULT ''
);

-- The sweep's own read: "who is in grace, oldest first" for the ops queue, and
-- the window check on every tick.
CREATE INDEX idx_character_operator_watch_seen ON character_operator_watch (first_seen_at);

===== BEGIN FILE: services/trust-core/migrations/0049_recovery_tracking.sql =====
-- Slice 19 — M27: the recovery a creator files and never hears about again.
--
-- `POST /creator/recovery/file` proves an inbox, files a case for every live
-- passport behind it, and answers `{ok:true, received:true}`. That is the last
-- thing the product ever says. Every other recovery route is a write and the
-- queue read is Access-gated, so the person who filed has no surface at all —
-- while the creator console's own copy tells them "an operator checks it".
--
-- WHY A FILING TABLE AND NOT A COLUMN ON THE CASE. One inbox can carry several
-- passports, so a per-CASE ticket would hand back a list whose LENGTH counts
-- passports behind an address — precisely the number `fileCreatorRecovery`
-- already refuses to disclose (it drops the `deduped` flag its session-gated
-- sibling returns, for that reason, in those words). One ticket per FILING
-- carries no cardinality.
--
-- NO ADDRESS COLUMN, IN ANY FORM. The first version of this table carried
-- `address_hash TEXT NOT NULL` with a note saying the hash existed "only so the
-- prune can find a filing's rows". Two things were wrong with that. Nothing ever
-- read the column — the prune keys on `created_at`, which is indexed below — and
-- an UNSALTED sha256 of a lowercased email is not meaningfully private: the
-- input space is enumerable from any address list, so a rainbow pass turns this
-- straight back into the "plaintext roster of creator contact addresses joinable
-- from a capability" the column was introduced to avoid. A liability with no
-- reader is not a privacy measure; it is just the liability.
CREATE TABLE IF NOT EXISTS creator_recovery_filings (
  tracking_id  TEXT PRIMARY KEY,
  created_at   TEXT NOT NULL
);

CREATE INDEX IF NOT EXISTS idx_recovery_filings_created ON creator_recovery_filings (created_at);

-- A LINK TABLE, not a column on the case, because the case is shared.
--
-- `submitRecoveryRequest` DEDUPES: a second filing for a passport that already
-- has an open request-origin case folds into that case rather than creating one
-- (it returns the existing `item_id` with `deduped: true`). A `tracking_id`
-- column on the queue row would therefore hold the FIRST filer's ticket forever,
-- and every later filer's ticket would resolve to nothing — the silence this
-- migration exists to end, wearing a ticket. Many filings to one case is the
-- real shape, so it gets a real table.
--
-- No FK to `review_queue`: a resolved case can be pruned on its own schedule and
-- an orphan link must read as "with an operator" (see `recoveryFilingStatus`),
-- never as an error.
CREATE TABLE IF NOT EXISTS recovery_filing_cases (
  tracking_id TEXT NOT NULL,
  item_id     TEXT NOT NULL,
  created_at  TEXT NOT NULL,
  PRIMARY KEY (tracking_id, item_id)
);

CREATE INDEX IF NOT EXISTS idx_recovery_filing_cases_item ON recovery_filing_cases (item_id);

===== BEGIN FILE: services/trust-core/migrations/0050_caution_dispute_dedup.sql =====
-- Slice 19 — M78: two contests where there should be one.
--
-- `contestCaution` is check-then-act: SELECT for an open `caution_dispute` on
-- this subject, then `enqueue` with a fresh `mintId("itm")`. Two concurrent
-- contests both pass the SELECT and both enqueue, so the operator queue carries
-- the same complaint twice and the dispute screen shows a duplicate.
--
-- Severity is low and was lower still after the per-IP cap landed (20/hr, plus
-- 5 per subject), so this is duplicate operator work rather than a hole. It is
-- fixed because the fix is an index and because a half-closed item sitting
-- behind a comment that says "still OPEN" is exactly how a known gap becomes a
-- permanent one.
--
-- PARTIAL, AND DELIBERATELY NOT A TABLE-WIDE UNIQUE. `review_queue` carries
-- eight item types through eight doors — reports, warnings, disputes, removals,
-- nominations, imports, caution disputes, recovery cases — and several of them
-- legitimately hold more than one open row per subject (a recovery case and a
-- fan's request on the same passport are two different cases, on purpose, and
-- `submitRecoveryRequest` guards them apart by ORIGIN rather than by subject).
-- A constraint broad enough to fix this one would be a new failure mode for the
-- other seven. The index says only what is actually true: one open caution
-- dispute per subject.
--
-- SQLite applies a partial index only to rows matching the WHERE, so a resolved
-- or rejected dispute leaves the slot free and a subject may be contested again
-- after a decision — which is the intended behaviour, not a side effect.
--
-- THE DUPLICATES ALREADY IN THE TABLE ARE CLOSED FIRST, and without this the
-- migration cannot be applied at all. `CREATE UNIQUE INDEX` on a populated table
-- fails on a data violation, and `IF NOT EXISTS` does not help — it suppresses
-- "already exists", not "your rows break this". `contestCaution` has been
-- check-then-act since it shipped, on an anonymous route, so two open disputes
-- on one subject is not hypothetical: it is the defect below the index. Every
-- other UNIQUE index in this tree was created at table-creation time on an empty
-- table, so nothing here had exercised the retro-fit case, and the test double
-- migrates a FRESH database where the table is empty — which is exactly why this
-- would have failed first in production and nowhere else.
--
-- The SURVIVOR is the lowest `item_id`, which is stable and arbitrary in the
-- only way that matters: the losers carry the same subject and the same public
-- accusation, so an operator reading the survivor is reading the same case. They
-- are `rejected` rather than deleted — this table is an operator's decision
-- record, and a row vanishing is worse than a row marked. `disposition` is set
-- to match because the two are read together on the console.
UPDATE review_queue
   SET state = 'rejected',
       disposition = 'rejected'
 WHERE item_type = 'caution_dispute'
   AND state = 'open'
   AND item_id NOT IN (
     SELECT MIN(item_id) FROM review_queue
      WHERE item_type = 'caution_dispute' AND state = 'open'
      GROUP BY subject
   );

CREATE UNIQUE INDEX IF NOT EXISTS idx_open_caution_dispute
  ON review_queue (subject)
  WHERE item_type = 'caution_dispute' AND state = 'open';

===== BEGIN FILE: services/trust-core/migrations/0051_unsubscribe_tokens.sql =====
-- Slice 21 — RFC 8058 one-click unsubscribe.
--
-- Gmail and Yahoo's bulk-sender rules want `List-Unsubscribe-Post:
-- List-Unsubscribe=One-Click` and a URL that accepts a POST and unsubscribes
-- with no further interaction. What shipped before this was the RFC 2369 form
-- pointing at an SPA fragment on a Pages origin — a link for a human, inert to a
-- mailbox provider's machinery, and behind a session no mail bot has.
--
-- ============================================================================
-- WHY THE CAPABILITY IS IN THE URL, AGAINST THIS REPO'S OWN RULE
-- ============================================================================
--
-- Every other capability here travels in a request BODY, because
-- `observability.enabled` is on in all three wrangler configs and Workers Logs
-- records invocation URLs — so a path-borne capability sits in that log for the
-- retention window. `POST /removal/status` and `POST /recovery/status` both say
-- so in their own comments.
--
-- RFC 8058 makes that impossible: the provider POSTs a FIXED form body
-- (`List-Unsubscribe=One-Click`) to the URL from the header, so the token must
-- be in the URL or there is no one-click at all. This slice cannot satisfy both
-- rules. It picks the trade deliberately and bounds it:
--
--   * SINGLE-USE — the row is read by primary key, then the preference write and
--     the `used_at` claim run as ONE atomic D1 batch, both carrying the same
--     `used_at IS NULL AND created_at >= cutoff` guard. So a log-reader replaying
--     the URL matches zero rows and changes nothing;
--   * SHORT-LIVED — 30 days, the life of one mail rather than one recipient;
--   * A KEY FOR NOTHING ELSE — it resolves to one (stream, subject) row and
--     grants no read. It cannot be traded for an address, a passport, or state;
--   * THE ACT IS BENIGN AND REVERSIBLE BY ITS OWNER — turning a mail stream off,
--     which the owner turns back on in their own console.
--
-- ============================================================================
-- NO ADDRESS, IN ANY FORM
-- ============================================================================
--
-- Migration 0049's rule, for its reason: this table is keyed by a value handed
-- to a mail provider, and an unsalted digest of a lowercased email is reversible
-- by enumeration — so storing one would reproduce the roster it was meant to
-- prevent. The row names WHO by their internal id (an `acc_` account or a `psp_`
-- passport), which is what the write needs and nothing more.
CREATE TABLE IF NOT EXISTS unsubscribe_tokens (
  token       TEXT PRIMARY KEY,
  -- 'fan_alerts' | 'creator_digest_email' — which stream this token turns off.
  -- Stored rather than derived so a token minted for one stream can never be
  -- spent on another, even if the two id spaces ever overlap.
  stream      TEXT NOT NULL,
  -- The account_id or passport_id whose preference this token writes.
  subject_id  TEXT NOT NULL,
  created_at  TEXT NOT NULL,
  -- Set on first use, by the same atomic batch that writes the preference. Both
  -- statements guard on this column, so a second POST matches zero rows in both
  -- and is answered identically to an unknown token.
  used_at     TEXT
);

CREATE INDEX IF NOT EXISTS idx_unsub_created ON unsubscribe_tokens (created_at);

-- THE ERASURE PATH'S INDEX, and it is here because the erasure path is the one
-- that cannot afford a scan. `deleteAccount` runs
-- `DELETE FROM unsubscribe_tokens WHERE subject_id = ? AND stream = 'fan_alerts'`
-- inside its single large batch; this table is sized at (mail volume × the
-- retention window), which at the alert leg's own stated ceiling (~86,400 mails
-- a day) is millions of rows. A full scan there would be the one unindexed
-- statement in the GDPR path. Leading column is `stream` so the composite also
-- serves a stream-scoped sweep, and `subject_id` alone is never queried without
-- one.
CREATE INDEX IF NOT EXISTS idx_unsub_subject ON unsubscribe_tokens (stream, subject_id);

-- EMAIL-ONLY, SEPARATE FROM `digests_enabled` — the decision this slice was
-- signed off on.
--
-- `creator_contacts.digests_enabled = 0` turns off the digest EMAIL *and* the
-- digest PUSH (digest.ts gates both on it). A button labelled "unsubscribe from
-- these emails" that silently stops a creator's push notifications is not what
-- the button says, and a creator who clicked it in a mail client would have no
-- idea their phone had gone quiet.
--
-- So one-click writes THIS column and leaves `digests_enabled` alone. The email
-- leg reads `digests_enabled = 1 AND digest_email_enabled = 1`; push keeps
-- reading `digests_enabled` alone. DEFAULT 1 so every existing creator is
-- unchanged.
ALTER TABLE creator_contacts ADD COLUMN digest_email_enabled INTEGER NOT NULL DEFAULT 1 CHECK (digest_email_enabled IN (0,1));

-- Slice 23 — THE SAFETY LANE, indexed.
--
-- ============================================================================
-- WHAT WAS MEASURED, BECAUSE THE GAP WAS RECORDED IN THE WRONG PLACE
-- ============================================================================
--
-- `DEPLOY.md` §8 and `CARRYOVER.md` both carried "alert delivery is bounded
-- pull, not a queue — past the ceiling the answer is Cloudflare Queues", and
-- pointed at the FAN legs. Measured against the better-sqlite3 double, on the
-- two rails a payout-destination change actually travels:
--
--   FAN push, a payout alert buried under a full batch of OLDER rows and more
--   than PRIORITY_SCAN newer ones          → delivered on pass 2 = 10 minutes
--     (the positional window costs ⌈older/batch⌉ passes, and at the 5-minute
--      DELIVERY_CRON it takes a ~43,000-row undrained backlog to reach 24 h)
--
--   CREATOR digest, one payout-change alert behind a weekly herd (roundups
--   only; a herd of longer-clocked ALERTS reburies it inside the fast lane
--   unless that lane is money-first, which cost a further correction):
--        100 creators  → sweep  3 =  18 hours   (inside the window)
--        500 creators  → sweep 11 =  66 hours   (MISSED)
--      1,000 creators  → sweep 21 = 126 hours   (MISSED — five days)
--
-- So the rail that breaks the 24-hour payout cooling-off window is the CREATOR
-- one, at a few hundred creators, and it breaks because `deliverDigests` reads
-- `ORDER BY created_at ASC LIMIT 50` with **no priority ordering of any kind** —
-- the fan legs' `PRIORITY_SCAN`/`SAFETY_KINDS` answer was never given to it.
-- The queue was never the problem; the ordering was.
--
-- ============================================================================
-- WHY INDEXES, AND WHY THESE SHAPES
-- ============================================================================
--
-- `alert-fanout-budget.test.ts` records the reason the fan legs use a positional
-- window rather than a `WHERE kind IN (…)` arm: "the latter walks the entire
-- `delivered = 0` set to answer 'no safety rows here'". That was correct, and it
-- is correct only WITHOUT an index. Measured over the safety-kind arm:
--
--   rows undelivered      no index        (kind, notif_id) WHERE delivered = 0
--   2,000                  0.21 ms         0.045 ms
--   50,000                 5.13 ms         0.026 ms
--   200,000               20.49 ms         0.029 ms      ← linear vs FLAT
--
-- So the arm becomes O(matching rows) instead of a walk, and the window it
-- replaces stops being a window at all: a safety alert is selected by WHAT IT IS
-- rather than by WHERE IT SITS, so backlog depth cannot bury it.
--
-- PARTIAL, on purpose. `notifications` is retained 180 days and
-- `creator_digests` is not swept at all, but a row leaves these indexes the
-- moment its channel settles — so each index is sized by the WORKING SET, not by
-- the table. It is the shape migration 0050 already uses.
--
-- A SHAPE THAT DOES NOT WORK, recorded so it is not retried: an index whose
-- predicate names the kinds
-- (`(notif_id) WHERE delivered = 0 AND kind IN (…)`) is never chosen by the
-- planner — measured at 19.7 ms over 200,000 rows, i.e. the unindexed cost. The
-- kinds have to be in the index COLUMNS for a multi-value seek.
CREATE INDEX IF NOT EXISTS idx_notif_safety_push ON notifications (kind, notif_id) WHERE delivered = 0;
CREATE INDEX IF NOT EXISTS idx_notif_safety_email ON notifications (kind, notif_id) WHERE emailed = 0;

-- THE CREATOR RAIL. `kind` already carries the split this needs and no new
-- column is required: 0023's own comment says it — "'daily'/'weekly' are the
-- periodic roundups. 'alert' is an event-driven creator notice" — which is
-- exactly "has a clock on it" versus "is news". `payout_account_change`,
-- `payout_change`, `proof_expiry` and `operator_lapse` all ride `alert`.
--
-- The predicate repeats the read's OWN `(emailed = 0 OR pushed = 0)` so the
-- index holds only unsettled alerts. That OR is also why the existing
-- `idx_creator_digests_undelivered (emailed, pushed)` is unusable and the FIFO
-- head read is a full scan today (17 ms at 200,000 rows, measured) — an
-- existing cost this migration does not change and the manifest names. It works
-- HERE because the query's own WHERE matches this predicate exactly:
--
--   depth 200,000, alert arm      no index 7.25 ms   →   this index 0.012 ms
CREATE INDEX IF NOT EXISTS idx_creator_digests_alert ON creator_digests (created_at) WHERE kind = 'alert' AND (emailed = 0 OR pushed = 0);

-- Slice 24 — THE ACQUISITIONS PAGE KEY.
--
-- ============================================================================
-- WHAT THIS FIXES, AND WHY IT IS AN INDEX RATHER THAN JUST A ROUTE CHANGE
-- ============================================================================
--
-- `listAcquisitions` has always taken a `limit` (clamped to 500) that no caller
-- passed, so `GET /account/acquisitions` served a fixed 100 rows and three
-- clients printed a footnote saying older redemptions are not listed. That is a
-- fan's own purchase record, truncated with no way to reach the rest.
--
-- Paging it needs a key, and `acquired_at` alone is NOT one: it is not unique
-- (two codes redeemed inside the same millisecond share a stamp) and it is not
-- indexed for this account's rows at all. A `WHERE acquired_at < ?` cursor over
-- a non-unique column SILENTLY SKIPS every row that shares the boundary stamp
-- with the last row of the previous page — a page break that eats purchases.
-- So the cursor is the tuple `(acquired_at, acquisition_id)`, and this index is
-- the tuple's home:
--
--   WHERE account_id = ?
--     AND (acquired_at < ?1 OR (acquired_at = ?1 AND acquisition_id < ?2))
--   ORDER BY acquired_at DESC, acquisition_id DESC
--
-- …spelled as a ROW VALUE, `(acquired_at, acquisition_id) < (?, ?)`. The
-- expanded `a < ? OR (a = ? AND b < ?)` form is logically identical and was the
-- first draft; measured, it plans as a `MULTI-INDEX OR` with
-- `USE TEMP B-TREE FOR ORDER BY` under collected statistics — a SORT, which is
-- what this index exists to remove — and with no range constraint at all on
-- another data shape. The row value plans as one seek on this index.
--
-- SQLite serves it by seeking `account_id` and scanning the rest of the index
-- BACKWARDS, so the ASC declaration below is correct and a DESC index would buy
-- nothing.
--
-- `acquisition_id` is the primary key, so the tuple is TOTAL — which is the whole
-- requirement. The ids are RANDOM, not time-ordered: `ids.ts` exists to enforce
-- that an identifier must not encode when it was created, and `event_id` is its
-- one exemption. Rows sharing a millisecond therefore come back in an
-- arbitrary-but-stable order, which is fine: a keyset cursor needs a strict
-- order, not a meaningful one.
--
-- ============================================================================
-- WHY THE OLD INDEX GOES
-- ============================================================================
--
-- `idx_acq_account (account_id)` from 0007 is a STRICT PREFIX of the new one.
-- Every query it can serve, the new one serves — checked with EXPLAIN QUERY PLAN
-- rather than assumed, against all fourteen per-account readers of this table
-- (`ownsRelease`, `reverse-codes.ts`, `fan-auth.ts`, `account_data.ts`,
-- `admin.ts`'s per-account counts, `social.ts`'s taste graph, and the rest).
--
-- It is NOT dropped as unused: before this migration it was the ONLY index on
-- `account_id`, so every one of those readers chose it. The argument is
-- prefix-superset. Keeping it would be the dead weight slice 23's review found
-- in `idx_notif_safety_email` — a second index every redemption pays to write
-- and no read can prefer.
--
-- Two of those readers get FASTER rather than merely unchanged, because they
-- already order by `acquired_at` within one account and had no index for it:
-- `account_data.ts` (the data export) and `social.ts` (the taste graph's seed
-- set). Line numbers are deliberately not cited: the first draft cited three and
-- one of them was already wrong.
--
-- NOT touched: `idx_acq_release (release_id)`. It is the creator-side and
-- attribution-side reader's index and shares no prefix with this one.

CREATE INDEX IF NOT EXISTS idx_acq_account_page ON acquisitions (account_id, acquired_at, acquisition_id);
DROP INDEX IF EXISTS idx_acq_account;

-- Slice 29 — an `op_` id is not a person, and the two-person rule needs one.
--
-- THE DEFECT. `operatorActorId` maps a normalized email string to a random
-- `op_` id via a lookup table, so `j.brown@gmail.com`, `jbrown+ops@gmail.com`
-- and `jbrown@gmail.com` are three different operators. Each can be the "second
-- person" for the others, and every gate that says "two distinct operators"

-- instead of becoming a new one.

-- separate column rather than as "the first op_id we saw" because the other
-- half of the aliasing — a work address and a personal one, which share no
-- canonical form — can only be linked by somebody who KNOWS they are the same
-- human. `POST /ops/operators/link` is that door.
--
-- NOTHING IS REWRITTEN. Every existing `op_` id keeps its ledger attribution:
-- `person_id` defaults to the row's own `op_id`, so a deployment that never
-- links anything behaves exactly as it did, and an `op_` id that already
-- appears in `trust_events` is never touched. This table is ops state, not a
-- fold, so no reducer sees any of it.

ALTER TABLE operator_identities ADD COLUMN canonical_email TEXT;

-- already exists — `startDomainClaim` joins `passport_accounts` and 404s
-- ("no such account on that passport"), `startDomainProof` refuses an unclaimed
-- one, and `startDomainAdd` requires `subjectMayAddAccounts`. A BRAND WITH NO
-- PASSPORT has no id to put in this column, and that NOT NULL is the exact wall
-- that made "self-serve brand onboarding" impossible. `startDomainProof`'s own
-- refusal has been pointing at a door nobody had built since it was written:
-- "claim the passport first, then prove additional domains here".

-- The alternative was a second challenge table on `brand_enrolments`, with its
-- own TTL, its own attempt counter and its own live-slot rule. This repository
-- has one loud lesson about that shape — `issueDomainChallenge`'s docblock:

-- column gives, and a CHECK makes the exception nameable rather than general:
--

-- 0064: SCOUT — a budgeted, numbered, snapshot-carrying act of early support.
-- F6 (master list #44 Scout, #6 proof of early fandom). Additive: one table,
-- two indexes, nothing rewritten.
--
-- ---------------------------------------------------------------------------
-- WHY A FOURTH SCARCE FAN ORDINAL, AND WHY IT IS SHAPED LIKE THE OTHER THREE
-- ---------------------------------------------------------------------------
-- `founding.ts` holds two (the creator Founding badge, the registry join
-- number) and `founding_fan_ranks` (0020) holds the third, the fan-side
-- Founding Fan slot. All three obey one rule, written at the top of
-- `founding.ts` and proven by `0020`'s deletion-drift docblock:
--
--     A NUMBER IS A ROW, ALLOCATED ONCE, NEVER RECOMPUTED, AND NEVER
--     RE-MINTED INTO A SLOT SOMEBODY VACATED.
--
-- Scout is the fourth, and it copies every clause of that:
--
--   * PRIMARY KEY (passport_id, account_id) — a fan is numbered AT MOST ONCE
--     per creator, and their number never changes. Re-scouting is a no-op, not
--     a second ordinal and not a second budget spend.
--   * UNIQUE (passport_id, ordinal) — two fans can never hold Scout #7 on the
--     same creator, whatever a future writer believes. Concurrent allocations
--     race on this constraint; the loser retries the recomputed next number,
--     exactly as `awardFoundingIfEligible` does.
--   * NO FOREIGN KEY on `account_id`, deliberately, and it is the third time
--     this repository makes that call for the same reason (`founding_fan_ranks`
--     and `reverse_code_claims` are the other two): an FK would either block
--     erasure outright or cascade a spent slot away, and a slot that vanishes
--     is a slot that can be re-minted. `account_data.ts` deletes the row on
--     erasure and the allocator reads `MAX(ordinal) + 1` rather than
--     `COUNT(*) + 1`, so an erased fan's number stays UNFILLED forever instead
--     of being handed to the next arrival. That is the identical sentence
--     `account_data.ts` already carries for `founding_fan_ranks`.
--
-- ---------------------------------------------------------------------------
-- THE BUDGET IS NOT IN THIS TABLE, AND THAT IS THE POINT
-- ---------------------------------------------------------------------------
-- There is no `scout_budgets` table, no per-account counter and no grant row.
-- A fan's remaining Scouts for a period is COUNT(*) over their own rows in
-- that `cohort`, subtracted from one constant — F2's rule applied to a
-- non-money resource: A BALANCE IS A FOLD, NEVER A STORED TRUTH. Nothing to
-- decrement, nothing to drift, no second writer, and no repair path needed
-- because there is no derived state to repair.
--
-- `cohort` therefore does double duty: it is the display label ("Sept 2026")
-- AND the period key the budget folds over. It is written as the UTC calendar
-- month of `scouted_at` and frozen — a stored `YYYY-MM` rather than a
-- `strftime` over `scouted_at` on every read, so the budget window can never
-- be moved under an already-spent Scout by a later change of period rule.
--
-- ---------------------------------------------------------------------------
-- THE SNAPSHOT (#6) — WHAT THE CREATOR LOOKED LIKE WHEN YOU FOUND THEM
-- ---------------------------------------------------------------------------
-- "Proof of early fandom" is unprovable without a frozen number, and every
-- signal here is one this product ALREADY PUBLISHES about a creator, captured
-- at the instant of the scout:
--
--   demand_at_scout       the QUALIFIED requester count (`fan-quality.ts` —
--                         the same number /p/:id/requests serves, so the
--                         snapshot cannot claim more than the public counter
--                         ever did)
--   claim_state_at_scout  'unclaimed' | 'claimed'  (a VERIFIED passport cannot
--                         be scouted at all — see the CHECK)
--   join_no_at_scout      the registry join number, NULL if unnumbered
--   founding_rank_at_scout the creator Founding badge rank, NULL if none
--
-- DELIBERATELY ABSENT: a save count and a check count. `creatorDemand` in
-- `requests.ts` documents both exclusions and they hold here for the same
-- reasons — S22 treats the save list as private, and the k-anonymous check
-- lookup means the server never learns who checked whom, so any "checks at the
-- time" figure would be invented rather than measured.
--
-- The snapshot columns are NOT NULL where the value always exists and NULL-able
-- where the underlying fact legitimately does not (an unclaimed creator has no
-- join number and no Founding rank). A NULL here means "they had none", which
-- is itself the early-fandom evidence.

CREATE TABLE scout_ordinals (
  passport_id            TEXT    NOT NULL REFERENCES passports(passport_id),
  account_id             TEXT    NOT NULL,        -- no FK: erasure must be able to remove the account
  ordinal                INTEGER NOT NULL CHECK (ordinal >= 1),
  cohort                 TEXT    NOT NULL CHECK (length(cohort) = 7 AND substr(cohort, 5, 1) = '-'),
  scouted_at             TEXT    NOT NULL,
  -- The #6 snapshot, frozen in the same INSERT as the ordinal.
  demand_at_scout        INTEGER NOT NULL CHECK (demand_at_scout >= 0),
  claim_state_at_scout   TEXT    NOT NULL CHECK (claim_state_at_scout IN ('unclaimed','claimed')),
  join_no_at_scout       INTEGER,
  founding_rank_at_scout INTEGER,
  PRIMARY KEY (passport_id, account_id)
);

-- Two fans never hold the same number on the same creator. This is the
-- structural half of the ordinal promise; the allocator's retry loop is the
-- other half.
CREATE UNIQUE INDEX idx_scout_slot ON scout_ordinals(passport_id, ordinal);

-- The budget fold ("how many Scouts have I spent this month") and the fan's own
-- scout list, both keyed on the account. `cohort` leads the second column so
-- the period COUNT is an index-only scan.
CREATE INDEX idx_scout_account_cohort ON scout_ordinals(account_id, cohort);

--

-- re-attests, freshly and under the creator's own passkey, something the fan
-- can already verify — or it produces nothing at all.
--
-- That is what lets a FAN-INITIATED channel exist inside every lock the product
-- has (`docs/DIRECTION.md` "Decisions locked", clarified 2026-09-03):
--   * it is not a second source of truth about account ownership — the proof
--     model (bio-code / OAuth / operator) stays the only one, and `handshake.ts`
--     REFUSES to build an `identity` draft unless the account key is already an
--     active, unshielded row on that passport;
--   * it cannot name a payout destination that is not already a rendering
--     payout tag, so it can never point a fan at a new place to send money;
--   * it never names a fake — a negative answer is private to the asking fan
--     and non-accusatory, and impersonation goes to the report path, which has
--     operator review and the PSA rules;
--   * it changes no registry answer and no verdict: `handshake_answered` is
--     derive-from-log, exactly like `recognition_granted`.
--
-- ---------------------------------------------------------------------------
-- WHY THE NOTE IS HERE AND NOT IN THE LEDGER
-- ---------------------------------------------------------------------------
-- `note` is the ONE place fan-authored text reaches a creator through
-- Handshake: ≤120 characters, on the `offer` and `call` scopes only, escaped
-- and capped exactly like an archive-request note. It lives in THIS table,
-- which erasure can reach, and it NEVER enters the signed event — the S90
-- `recognition_id` precedent (window item 1), for the same reason: an
-- append-only log that `deleteAccount` cannot reach is the wrong home for
-- anything a person typed. The signed payload is
-- `{handshake_id, scope, disposition}` and the row is what says what was asked.
--
-- `identity` and `payment` carry NO note at all. The structured fields say
-- everything there is to say, and every character of free text into a
-- creator's inbox has to earn its place.
--
-- ---------------------------------------------------------------------------
-- THE STATE MACHINE, AND WHY `expired` IS NOT A VERDICT
-- ---------------------------------------------------------------------------
--   open ──answer(confirmed)──► confirmed        (signed, publishable)
--        ──answer(declined)───► declined         (signed, PRIVATE to the asker)
--        ──ttl───────────────► expired          (nothing happened, and that is
--                                                 not a signal about anybody)
--        ──creator withdraws──► expired
--
-- NON-RESPONSE IS THE DEFAULT AND MEANS NOTHING. Most creators will never
-- answer most handshakes. `expired` is never counted, never ranked, never
-- surfaced on a creator's public page, and never reaches `intelligence.ts` —
-- a product that renders silence as amber is a product that manufactures
-- suspicion, which is the exact harm this one exists to reduce.
--
-- ---------------------------------------------------------------------------
-- THE CAPS ARE INDEXES, BECAUSE THEY ARE ENFORCED IN THE INSERT
-- ---------------------------------------------------------------------------
-- F6's review lesson, applied on day one: a SELECT above an INSERT is not a
-- ceiling. Every cap `handshake.ts` claims rides inside the INSERT's own
-- WHERE, and these indexes are what make those counts cheap.
--
-- UNIQUE(passport_id, account_id) WHERE state = 'open' is the one-open-per-pair
-- rule, structurally: a fan cannot queue five questions at one creator, and a
-- second attempt is a 409 rather than a second row. It is PARTIAL so a fan may
-- ask again after the first is answered or expires.

CREATE TABLE handshakes (
  handshake_id TEXT PRIMARY KEY,                   -- hsk_<16 hex>
  passport_id  TEXT NOT NULL REFERENCES passports(passport_id),
  account_id   TEXT NOT NULL,                      -- the asking fan; no FK (erasure)
  scope        TEXT NOT NULL CHECK (scope IN ('identity','payment','offer','call')),
  -- The structured subject of the question, per scope. `identity`: the account
  -- key the fan encountered. `payment`: the destination string. `offer`/`call`:
  -- the structured offer/window. One column, because the server validates it
  -- per scope before it is ever written and no reader interprets it blind.
  subject      TEXT NOT NULL,
  note         TEXT,                               -- offer/call only, <= 120 printable chars
  state        TEXT NOT NULL CHECK (state IN ('open','confirmed','declined','expired')),
  created_at   TEXT NOT NULL,
  expires_at   TEXT NOT NULL,
  answered_at  TEXT,
  -- The signed event that answered it. NULL while open; set in the SAME atomic
  -- batch as the `handshake_answered` append, so a state of 'confirmed' with no
  -- event behind it is not a reachable row.
  answer_event TEXT
);

-- One OPEN handshake per fan per creator. Partial, so asking again after an
-- answer or an expiry is allowed — the cap is on the queue, not on the
-- relationship.
CREATE UNIQUE INDEX idx_hsk_open_pair ON handshakes(passport_id, account_id) WHERE state = 'open';
-- The creator's inbox, and the per-creator open ceiling.
CREATE INDEX idx_hsk_creator_state ON handshakes(passport_id, state, created_at);
-- The fan's own list, their global open cap, and the per-fan-per-creator window.
CREATE INDEX idx_hsk_fan ON handshakes(account_id, created_at);
-- The expiry sweep: only open rows, ordered by when they lapse.
CREATE INDEX idx_hsk_expiry ON handshakes(expires_at) WHERE state = 'open';

-- ---------------------------------------------------------------------------
-- THE CREATOR'S TWO SWITCHES
-- ---------------------------------------------------------------------------
-- `handshake_settings.accepting` is the whole-channel off switch, and
-- `handshake_blocks` is the per-fan one. THEY ANSWER IDENTICALLY to a creator
-- who simply has none, and `handshake.ts` spends the same statements in every
-- case: "you are blocked" is information a harasser can act on, so this door
-- does not give it. The DEFAULT is derived, not stored — a passport with no
-- settings row accepts handshakes iff it is verified — so the table only ever
-- holds a deliberate change.
CREATE TABLE handshake_settings (
  passport_id TEXT PRIMARY KEY REFERENCES passports(passport_id),
  accepting   INTEGER NOT NULL DEFAULT 1 CHECK (accepting IN (0,1)),
  updated_at  TEXT NOT NULL
);

CREATE TABLE handshake_blocks (
  passport_id TEXT NOT NULL REFERENCES passports(passport_id),
  account_id  TEXT NOT NULL,                       -- no FK: erasure must be able to remove the account
  created_at  TEXT NOT NULL,
  PRIMARY KEY (passport_id, account_id)
);

-- ---------------------------------------------------------------------------
-- THE ADMISSION GATE — `write_admissions` (0022) in a table of its own
-- ---------------------------------------------------------------------------
-- An answer is TWO writes that must be one: the signed `handshake_answered`
-- append, and the row moving `open` → `confirmed`/`declined`. They ride one
-- atomic batch, and the state guard lives inside the UPDATE — but an UPDATE
-- that matches nothing is not an error in SQLite, so a second answer that
-- overtook the first would append a signed event over a closed row.
--
-- This table is what turns "the UPDATE matched nothing" into "the batch fails".
-- `handshake.ts buildAnswerStatements` inserts a value that is a SUBQUERY
-- returning NULL exactly when the row does not now read the disposition just
-- written; NULL violates the NOT NULL and the whole batch rolls back. The same
-- mechanism 0022 chose over a trigger, and for the same reasons: a plain
-- NOT NULL needs no exemption from the append-only guard, shows up in
-- schema-shape.expected.txt, and behaves identically on better-sqlite3 and D1.
--
-- NOT NULL IS DECLARED EXPLICITLY. SQLite's legacy quirk lets a TEXT PRIMARY
-- KEY hold NULL in an ordinary rowid table, which would make the whole gate a
-- no-op. `write_admissions` declares it too, for the same reason.
--
-- IT DOES NOT GROW: one row per handshake that was ever answered, upserted.
-- Nothing reads it to decide anything — deleting every row changes no decision
-- the product makes and only loses the audit trail.
CREATE TABLE handshake_admissions (
  handshake_id TEXT NOT NULL PRIMARY KEY,
  admitted_at  TEXT NOT NULL
);

-- 0066: CONNECTED FAN PASSPORT / CONSTELLATIONS (F8, master list #45).
-- Additive: two tables, three indexes.
--
-- ---------------------------------------------------------------------------
-- WHAT IS STORED, AND WHAT DELIBERATELY HAS NO COLUMN
-- ---------------------------------------------------------------------------
-- A fan proves by OAuth that they control an account on a platform. Dhwaja
-- records THAT THEY DID and nothing else:
--
--   STORED     provider, sub
--   NOT STORED handle, display name, email, avatar, follower count, granted
--              scopes, access token, refresh token, profile URL — none of them
--              has a column, so a future writer cannot quietly begin keeping
--              one. That is the same structural move `fan-quality.ts` makes
--              about billing: a promise the schema enforces beats a promise a
--              docblock makes.
--
-- The frozen list's words are "store provider+sub only; expose yes/no claims",
-- and this table is the literal reading of them.
--
-- WHY `sub` IS STORED AT ALL. Only for `UNIQUE (provider, sub)`. That
-- constraint IS the feature: one external account backs at most one Dhwaja
-- account, which is the first thing in this product with a real per-account
-- price. Every other fan-quality signal (`fan-quality.ts`) can be minted for
-- free — a mail domain, a software passkey — and the F6 adversarial review said
-- so in as many words: *the floor is time, not identity*. This is the identity
-- half, and it is worth exactly as much as that UNIQUE.
--
-- RECORDED, NOT BUILT: hashing `sub` under a per-install pepper would harden a
-- database leak against bulk correlation with a provider's public ids. It is a
-- follow-up rather than part of this slice because it adds an operator secret
-- the feature cannot start without, and `oauth_identities` (0004) already
-- stores google/apple `sub` in the clear — one decision, taken once, for both.
--
-- ---------------------------------------------------------------------------
-- WHAT IS PUBLISHED
-- ---------------------------------------------------------------------------
-- A COUNT. `GET /u/:handle` carries `constellation: 3` — three independent
-- platforms vouch for this being a person — and never which three. A fan may
-- opt IN per link to naming a provider on their own profile (`provider_public`,
-- DEFAULT 0). There is no endpoint that maps a fan to a provider account and
-- none that maps a provider account to a fan, in either direction, at any
-- privilege level below an operator with database access.

CREATE TABLE fan_connections (
  account_id      TEXT    NOT NULL,        -- no FK: erasure must be able to remove the account
  provider        TEXT    NOT NULL,        -- 'twitch' | 'youtube' | 'discord' | 'instagram'
  sub             TEXT    NOT NULL,        -- the provider's STABLE subject id. The ONLY thing kept.
  connected_at    TEXT    NOT NULL,
  -- The per-link opt-in. 0 = the fan's profile publishes the COUNT only, which
  -- is the default and the posture: naming a platform somebody is on is a fact
  -- about them that they should have to volunteer.
  provider_public INTEGER NOT NULL DEFAULT 0 CHECK (provider_public IN (0,1)),
  PRIMARY KEY (account_id, provider)
);

-- THE SYBIL CONTROL, and the only reason `sub` is on the table.
CREATE UNIQUE INDEX idx_fanconn_sub ON fan_connections(provider, sub);

-- ---------------------------------------------------------------------------
-- THE RELINK COOLDOWN
-- ---------------------------------------------------------------------------
-- Unlinking deletes the row, which would otherwise free the external account
-- immediately — and one Twitch account walking a fleet of Dhwaja accounts up to
-- "connected", one after another, is exactly the farm this feature exists to
-- price. So an unlink reserves `(provider, sub)` to the SAME fan for
-- `RELINK_COOLDOWN_DAYS`; anyone else is refused until it lapses.
--
-- IT DENIES NOBODY A RESOURCE THEY COULD HAVE HAD. Only somebody who CONTROLS
-- the external account can ever reach this row — reaching it requires
-- completing the provider's own OAuth — so a stranger cannot reserve a slot on
-- another person's account. `tests/invariants/per-victim-resources.test.ts`
-- carries that verdict.
--
-- Erasure drops the hold with the account: the person is gone, and a
-- reservation held for nobody protects nobody.
CREATE TABLE fan_connection_holds (
  provider   TEXT NOT NULL,
  sub        TEXT NOT NULL,
  account_id TEXT NOT NULL,                -- who may relink before `until`
  until      TEXT NOT NULL,
  PRIMARY KEY (provider, sub)
);
CREATE INDEX idx_fanconn_hold_account ON fan_connection_holds(account_id);
-- The sweep: only rows past their lapse instant.
CREATE INDEX idx_fanconn_hold_until ON fan_connection_holds(until);

-- 0067: ASK THE CREATOR / Q&A (F9, master list #54). Additive: two tables,
-- four indexes.
--
-- ---------------------------------------------------------------------------
-- WHY THIS IS NOT THE DM CHANNEL `DIRECTION.md:548` LOCKS OUT
-- ---------------------------------------------------------------------------
-- The lock is "NO open text feed / comments / DMs", and its stated reason is
-- scams, grooming and moderation load. Four properties keep Q&A outside it, and
-- each is a property of this schema rather than a policy somebody follows:
--
--  1. ONE QUESTION, NO THREAD. There is no reply table, no parent column and no
--     second turn. `idx_q_open_pair` allows one OPEN question per fan per
--     creator, and answering it CLOSES the exchange rather than opening a
--     channel.
--  2. THE ONLY REPLY PATH IS PUBLIC. There is no column for a private answer
--     and no route that could carry one. A creator publishes a signed answer
--     everybody sees, or says nothing. A private creator→fan message is exactly
--     what turns a question box into a DM inbox, and there is nowhere to put
--     one.
--  3. STANDING GATES THE INBOUND. A stranger cannot type at a creator at all
--     (`questions.ts HANDSHAKE-style standing`, plus the creator's own raised
--     bar in `question_settings.rule`), which is the grooming and scam surface
--     the lock is about.
--  4. SILENCE COSTS THE CREATOR NOTHING. Unanswered questions are private to
--     them, lapse on a TTL, and are never counted, ranked or published — F7's
--     rule, for F7's reason.
--
-- ---------------------------------------------------------------------------
-- THE HARD PART: ANSWERING PUBLISHES A STRANGER'S SENTENCE ON A TRUST PAGE
-- ---------------------------------------------------------------------------
-- An answer without its question is meaningless, so publishing an answer
-- publishes the question. What makes that safe is that THE CREATOR READS IT
-- BEFORE THEY SIGN, and signing IS the publication: there is no auto-publish,
-- no queue that drains by default, and no way to answer without seeing.
--
-- `body_hash` is what makes that promise hold against the database rather than
-- against good intentions. The signed `question_answered` payload carries
-- `{question_id, question_hash, answer}`, so a published answer is bound to the
-- EXACT text the creator was shown; the row cannot be swapped underneath it,
-- and a creator can never be displayed answering a question they did not read.
--
-- WHERE EACH PIECE OF TEXT LIVES, and the split is the S90 `recognition_id`
-- rule applied to a harder case:
--   `body`   the FAN's words → this table, deletable, gone with the fan
--   `body_hash` → the ledger, so the binding survives without the words
--   `answer` the CREATOR's own words → the ledger, beside `creator_broadcast`'s
--            message, because erasing a fan must not erase a creator's
--            published statement
--
-- The consequence is deliberate and is stated in the console copy: an erased
-- fan's question disappears from under a published answer, and the answer
-- remains as a signed statement that has to stand on its own.

CREATE TABLE questions (
  question_id  TEXT NOT NULL PRIMARY KEY,          -- qst_<16 hex>
  passport_id  TEXT NOT NULL REFERENCES passports(passport_id),
  account_id   TEXT NOT NULL,                      -- the asking fan; no FK (erasure)
  body         TEXT NOT NULL,                      -- <= QUESTION_MAX printable chars
  body_hash    TEXT NOT NULL,                      -- sha256 of the canonical body
  -- 'retracted' is an ANSWERED question the creator later took down. It is a
  -- separate state rather than a flag because `publicAnswers` filters on
  -- `state = 'answered'`, so the row leaves the public page by the same
  -- predicate that put it there — no second condition to forget.
  state        TEXT NOT NULL CHECK (state IN ('open','answered','declined','expired','retracted')),
  created_at   TEXT NOT NULL,
  expires_at   TEXT NOT NULL,
  answered_at  TEXT,
  answer       TEXT,                               -- the creator's own words, public
  -- Set in the SAME atomic batch as the signed append (the F7 admission-gate
  -- mechanism), so an 'answered' row with no event behind it is not reachable.
  answer_event TEXT,
  -- The same mechanism for the way back. `retract_event` is the per-attempt
  -- nonce the retraction's admission row selects on, and `retracted_at` is when
  -- the answer left the page. `answer` is deliberately NOT cleared: the creator
  -- signed those words, the `question_answered` event still carries them, and
  -- blanking the column here would make the row disagree with the log.
  retract_event TEXT,
  retracted_at  TEXT
);
-- ---------------------------------------------------------------------------
-- WHY `body` IS BLANKED RATHER THAN THE ROW DELETED, FOR ANSWERED QUESTIONS
-- ---------------------------------------------------------------------------
-- `questions.ts PUBLISH_WARNING` tells a creator, before they sign, that an
-- erased fan's question disappears and their own answer stays. The first draft
-- made that sentence FALSE: erasure deleted the row outright, `publicAnswers`
-- reads this table, and the whole answer vanished with it — so a creator signed
-- on the strength of a warning that misdescribed the outcome, which is a
-- consent defect and not a cosmetic one.
--
-- `account_data.ts` therefore DELETES a question that was never answered (it is
-- the fan's words and nothing depends on it) and BLANKS one that was: `body` to
-- the empty string and `account_id` to a per-row tombstone. `body` is NOT NULL
-- so it cannot be nulled; `publicAnswers` maps the empty string back to a null
-- `question`, which is the shape the clients already render as "the person who
-- asked has since deleted their account".

-- One OPEN question per fan per creator. Partial, so a fan may ask again once
-- the first is answered or lapses: the cap is on the queue, not the
-- relationship.
CREATE UNIQUE INDEX idx_q_open_pair ON questions(passport_id, account_id) WHERE state = 'open';
-- The creator's inbox and the per-creator open ceiling.
CREATE INDEX idx_q_creator_state ON questions(passport_id, state, created_at);
-- The fan's own list, their global open cap, and the per-pair window.
CREATE INDEX idx_q_fan ON questions(account_id, created_at);
-- The expiry sweep: open rows only.
CREATE INDEX idx_q_expiry ON questions(expires_at) WHERE state = 'open';

-- ---------------------------------------------------------------------------
-- THE CREATOR'S SWITCHES — AND THIS ONE IS OPT-IN
-- ---------------------------------------------------------------------------
-- Handshake defaults ON for a verified creator (an answer there re-attests
-- something already on their page, and costs them nothing to ignore). Q&A
-- defaults OFF, because it is the only surface in the product that can put a
-- STRANGER'S SENTENCE on a creator's own trust page. A creator turns that on.
--
-- `rule` is an optional `eligibility.ts` rule that RAISES the default standing
-- bar — the frozen list's "creator can raise the bar". It is stored as the same
-- JSON the F4 predicate parses, so there is one rule language in the product
-- and no second evaluator to keep in step. NULL means the default disjunction.
CREATE TABLE question_settings (
  passport_id TEXT PRIMARY KEY REFERENCES passports(passport_id),
  accepting   INTEGER NOT NULL DEFAULT 0 CHECK (accepting IN (0,1)),
  rule        TEXT,
  updated_at  TEXT NOT NULL
);

-- ---------------------------------------------------------------------------
-- THE PER-FAN BLOCK — and why it exists here even though F7 got it first
-- ---------------------------------------------------------------------------
-- Handshake carries a ≤120-char note that is never published; this carries 280
-- characters of pure fan text and is the ONLY surface that can put a stranger's
-- sentence on a creator's trust page. It is therefore the one that most needs a
-- per-fan remedy, and the first draft shipped with LESS recourse than its
-- sibling: a creator harassed by one identified person could only close the
-- whole channel or raise the bar for everybody, which punishes every honest fan
-- to stop one.
--
-- The block is named BY QUESTION ID at the route (`blockQuestionAsker`), the
-- F7 shape: the creator holds that id because the question was addressed to
-- them, and the fan's `acc_` id never leaves the server.
CREATE TABLE question_blocks (
  passport_id TEXT NOT NULL REFERENCES passports(passport_id),
  account_id  TEXT NOT NULL,                       -- no FK: erasure must be able to remove the account
  created_at  TEXT NOT NULL,
  PRIMARY KEY (passport_id, account_id)
);

-- The admission gate, the `write_admissions` (0022) NOT NULL mechanism in a
-- table of its own — see `handshake_admissions` (0065) for the full argument.
-- NOT NULL is declared explicitly because SQLite lets a TEXT PRIMARY KEY hold
-- NULL in an ordinary rowid table, which would make the whole gate a no-op.
--
-- `kind` is in the PRIMARY KEY because TWO signed ceremonies now pass through
-- this gate for the same question — publishing an answer and retracting it —
-- and they must not share a slot.
--
-- AN EARLIER VERSION OF THIS COMMENT GAVE A REASON THAT IS FALSE, and the
-- correction is worth keeping because the false version is the intuitive one.
-- It claimed that without `kind`, a question_id already admitted for 'answer'
-- would let a retraction's INSERT succeed on the ON CONFLICT path even when its
-- own subquery found nothing. It would not: the subquery yields NULL, NULL
-- never equals an existing key, so the upsert branch is never taken and NOT
-- NULL aborts first — with a composite key or a single-column one, identically.
-- (Both shapes were probed against real SQLite rather than reasoned about.)
--
-- The two REAL reasons: each ceremony keeps its own `admitted_at` instead of
-- one silently overwriting the other's, so the table is an audit trail rather
-- than a last-writer-wins cell; and the key is the SECOND guard behind each
-- INSERT's state predicate — if `state = 'answered'` on the retract UPDATE ever
-- slipped, the key is what still refuses a duplicate admission and rolls the
-- signed append back with it. Neither INSERT carries an ON CONFLICT clause any
-- more, precisely so that key can do that job.
CREATE TABLE question_admissions (
  question_id TEXT NOT NULL,
  kind        TEXT NOT NULL CHECK (kind IN ('answer','retract')),
  admitted_at TEXT NOT NULL,
  PRIMARY KEY (question_id, kind)
);

-- 0068: DEMOGRAPHICS, AGGREGATE-ONLY (F10, master list #52). One table, and
-- what it does NOT have is the feature.
--
-- ---------------------------------------------------------------------------
-- THERE IS NO account_id COLUMN, AND THAT IS THE ENFORCEMENT
-- ---------------------------------------------------------------------------
-- `docs/BUILD-PLAN-2026-09-03.md:45` settled the open question this way:
-- country from `CF-IPCountry`, counted into K-floored buckets when a fan acts,
-- NEVER STORED PER ACCOUNT, no age, no language. So there is no per-fan country
-- column to protect here — the increment lands directly on an aggregate row,
-- and a fan's country exists only for the few microseconds between reading a
-- request header and adding one to a counter.
--
-- The absence is load-bearing rather than incidental. A future writer who
-- wanted to keep "which fan came from where" cannot do it quietly: they need a
-- migration, which means a `schema-shape.expected.txt` diff a reviewer sees.
-- The same is true of age and language, which have no column for the same
-- reason and are not coming.
--
-- ---------------------------------------------------------------------------
-- FAN-DAYS, NOT PEOPLE AND NOT ACTIONS
-- ---------------------------------------------------------------------------
-- The obvious metric is DISTINCT FANS per country. It is refused, and the
-- reason belongs next to the table rather than in a review thread:
--
-- Counting distinct fans means remembering WHICH fans have been counted. That
-- is either a row saying (passport, account, country, period) — the exact row
-- this feature exists not to write — or a PAIR of rows, `seen(psp, account,
-- period)` beside `counts(psp, country, period)`, which reconstruct it by
-- joining on the period whenever a bucket is small.
--
-- The FIRST DRAFT therefore counted raw actions, and that was worse in a way
-- the review had to execute to show: `GET /p/:id` writes one on every request
-- with no ceiling, so a creator's only Icelandic fan opening their share page
-- five times published `IS` on the dashboard, and anybody with a session and a
-- VPN could salt a country to floor-1 and read the exact real count off the
-- published bucket.
--
-- So the unit is a FAN-DAY: one fan, one creator, one UTC day, counted once
-- however many times they act. The dedupe rides the `creator_encounters` row
-- (0000) that ALREADY exists for that pair — `route-helpers.ts logEncounter`
-- counts only when that row's `last_seen_at` crosses into a new UTC day — so
-- no new per-person row is created to support it, and this table still holds
-- nothing but a creator, a day, a country and a number.
--
-- ---------------------------------------------------------------------------
-- WHY `day` AND NOT `period`
-- ---------------------------------------------------------------------------
-- `regions.ts audienceRegions` reports a MONTH but sums only days STRICTLY
-- BEFORE today, and storing days is what makes that possible.
--
-- The first draft stored months and leaned on `roi.ts dayFrozen` to stop a
-- creator differencing two reads. That control is weaker than it looks: it
-- computes on the FIRST READ of a UTC day, so a creator who simply does not
-- read chooses both endpoints — read at 23:59:50, have one known fan act, read
-- at 00:00:10, and the delta is that person. (That is a property of the shared
-- control, not of this feature; it is recorded against `dayFrozen` itself.)
--
-- Excluding the running day removes the primitive instead of timing it: for a
-- given creator, month and today, the answer is the same at every hour, so
-- there is no pair of reads to difference and no snapshot to sweep out of R2.
--
-- ---------------------------------------------------------------------------
-- RETENTION
-- ---------------------------------------------------------------------------
-- `regions.ts sweepOldRegions` drops days older than RETENTION_MONTHS (13) on
-- the cron. Thirteen so a creator can hold a month against the same month last
-- year, and no longer, because an aggregate kept forever is a permanent history
-- of where somebody's audience was that nobody asked anyone to keep.

CREATE TABLE audience_regions (
  passport_id TEXT NOT NULL REFERENCES passports(passport_id),
  day         TEXT NOT NULL,                       -- 'YYYY-MM-DD', UTC
  -- ISO-3166-1 alpha-2, uppercase. 'ZZ' is the real bucket for unknown:
  -- Cloudflare sends 'XX' when it cannot tell and 'T1' for Tor, and both land
  -- here along with anything off-shape. ZZ is SHOWN rather than dropped —
  -- hiding it would make every other number on the screen a lie.
  --
  -- The CHECK is defence in depth: `regions.ts normalizeCountry` is the only
  -- writer and already guarantees this, but a feature whose argument is "the
  -- schema is the enforcement" should not have a column the schema does not
  -- constrain. It also keeps arbitrary text out of the #53 prompt by
  -- construction rather than by the write path staying disciplined.
  country     TEXT NOT NULL CHECK (country GLOB '[A-Z][A-Z]'),
  fan_days    INTEGER NOT NULL DEFAULT 0,
  PRIMARY KEY (passport_id, day, country)
);

-- The creator's own read: one month of days for one passport.
CREATE INDEX idx_regions_read ON audience_regions(passport_id, day);
-- The retention sweep, which walks by day across every passport.
CREATE INDEX idx_regions_sweep ON audience_regions(day);

-- 0069: SMALL-CREATOR DISCOVERY (F11, master list #39). One table, and it is a
-- METER rather than a score.
--
-- ---------------------------------------------------------------------------
-- WHY A COUNTER AND NOT A RANKING
-- ---------------------------------------------------------------------------
-- Every other discovery rail (`rankings.ts`) orders `score DESC`. That is a
-- rich-get-richer loop by construction: the creators with the most collectors
-- hold the boards, being on a board wins collectors, and tomorrow the same
-- names are there. `breakout` looks like the exception and is not — it filters
-- to recently-verified and then sorts by score, so a large creator who verified
-- last week takes the slot a small one needed.
--
-- A newcomer rail sorted by score would be the same loop one size down: a
-- leaderboard of the biggest small creators, with the same names every day.
--
-- So a slot on this rail is a LOAN. `days` counts the days a passport has been
-- shown, the creator leaves the pool at `MAX_EXPOSURE_DAYS`, and selection
-- among the pool is `(days ASC, hash(passport_id || day) ASC)` — least-exposed
-- first, ties broken by a per-day hash nobody can steer. There is no score to
-- climb here, so there is nothing to farm: a creator cannot buy, request or
-- earn a better position, only wait for their turn and then lose it.
--
-- ---------------------------------------------------------------------------
-- WHAT IS DELIBERATELY NOT IN THIS TABLE, AND NOT ON THE WIRE
-- ---------------------------------------------------------------------------
-- No score, no collector count, no rank. The rail's membership rule is "under
-- the size ceiling", so publishing ANY of the ordering inputs would turn a
-- discovery surface into a bound on a creator's audience that they never chose
-- to publish — "this person has fewer than N collectors", addressable by name.
-- The counter, the ordering key and the pool size are all server-side only; a
-- viewer sees a list of creators and cannot tell who is at three collectors and
-- who is at three hundred.
--
-- A creator who OUTGROWS the ceiling leaves the pool holding whatever exposure
-- they had. They are not owed the remainder — the rail exists for people the
-- other boards cannot reach, and one of them reaching those boards is the rail
-- having worked.
--
-- No FK to `passports`: the sweep and the pool query both tolerate a stale row,
-- and an FK here would make a passport retraction depend on a discovery table.

CREATE TABLE discovery_exposure (
  passport_id    TEXT NOT NULL PRIMARY KEY,
  days           INTEGER NOT NULL DEFAULT 0,   -- days shown, never reset
  first_shown_at TEXT NOT NULL,
  last_shown_at  TEXT NOT NULL
);

-- The pool query's ordering key: least-exposed first.
CREATE INDEX idx_exposure_days ON discovery_exposure(days, passport_id);

-- creator who simply does not read chooses both endpoints, reads at 23:59:50,

-- about which dashboards are worth building, and a dashboard belongs to a
-- creator. `last_read_at` is a coarse marker rather than an access log: it is

-- 0071: THE PRECOMPUTED BRAND INDEX (F13, HANDOFF §1.1).
--
-- ---------------------------------------------------------------------------
-- WHAT IT IS, AND THE ONE THING IT IS NOT
-- ---------------------------------------------------------------------------
-- `domain-gate.ts` walks EVERY listed domain and EVERY listed name on every
-- verify. The cap that makes it fail closed at ~30,000 rows per arm was chosen
-- against a measurement rather than a feeling: "a full scan costs ~10
-- microseconds of SKELETON WORK per row". The ceiling is CPU spent folding the
-- same unchanged rows over and over.
--
-- This table caches that fold. It is A CACHE OF A PURE FUNCTION, KEYED ON ITS
-- OWN INPUT, and that is the whole safety argument.
--
-- IT IS NOT A SHORTLIST. The tempting design is to key on SKELETON and use it
-- to pick which rows to compare — look up the rows sharing a token, compare
-- only those. That would be FAIL-OPEN, because `gateMatch` is deliberately
-- fuzzy: containment, edit distance, seam segmentation and confusable slack all
-- match things that share no exact token. An index that narrows the candidate
-- set therefore MISSES matches the scan finds, and a lookalike gate that misses
-- is a lookalike gate that lets the lookalike through. "We keep the scan as a
-- fallback" does not rescue it, because the fallback is exactly what does not
-- run when the index answers.
--
-- So the scan still walks every row and `gateMatch` still decides. Not one
-- verdict moves. What changes is that the per-row cost drops from a skeleton
-- fold to a JSON parse.
--
-- ---------------------------------------------------------------------------
-- WHY A STALE ENTRY CANNOT HURT
-- ---------------------------------------------------------------------------
-- Every entry stores the INPUT it was computed from, and a reader uses it only
-- when that input equals the row's current value. So an entry that is stale,
-- truncated, corrupted or orphaned is IGNORED and the row is folded live. There
-- is no state in which consulting this table gives a different answer from not
-- having it — which is the property that makes it safe to put in front of a
-- fail-closed security gate.
--
-- No FK: the index tolerates a row that has gone (the sweep drops it), and an FK
-- would make deleting a passport account depend on a cache.

CREATE TABLE brand_index (
  -- 'domain' or 'name' — the two arms of the gate, with different inputs and
  -- different skeleton functions, kept in one table because they have identical
  -- lifecycle and identical staleness rules.
  kind       TEXT NOT NULL CHECK (kind IN ('domain','name')),
  -- The row's own identity: `account_key` for a domain, `passport_id|name` for
  -- a name. Exactly the cursor the scan already orders by.
  key        TEXT NOT NULL,
  -- The value the skeletons were computed FROM. A reader compares this against
  -- the live row and ignores the entry when they differ.
  input      TEXT NOT NULL,
  -- JSON: the brand list for a domain, the single skeleton for a name.
  skeletons  TEXT NOT NULL,
  built_at   TEXT NOT NULL,
  PRIMARY KEY (kind, key)
);

-- The builder's cursor, and the sweep.
CREATE INDEX idx_brand_index_built ON brand_index(kind, built_at);

-- 0073: ACCOUNT CLOSURE, CONFISCATION AND APPEAL (F18, master-list #32).
--
-- ---------------------------------------------------------------------------
-- WHAT THIS EXISTS FOR
-- ---------------------------------------------------------------------------
-- Before this, an operator who had verified theft, market manipulation or
-- economic abuse had exactly two options: erase the account
-- (`POST /ops/accounts/:id/delete`, irreversible, destroys the evidence with
-- it) or do nothing. The one automated lever, `accounts.banned_at`, is set ONLY
-- by a chargeback dispute in `refunds.ts` and gates only money.
--
-- So the shape here is a REVERSIBLE suspension with a real appeal window, and a
-- separate second act that makes it permanent. The two hardest grounds —
-- verified theft and market manipulation — are precisely the two an operator is
-- least likely to be certain about, and a wrong call has to be recoverable.
--
-- ---------------------------------------------------------------------------
-- THE POLICY THIS TABLE IS, decided 2026-09-04
-- ---------------------------------------------------------------------------
--  1. Two stages. `close` suspends and starts the clock; `terminate` is a
--     separate dual-control act.
--  2. Freeze everything. On TERMINATION, forfeit only value Dhwaja invented
--     (promotional, referral, game, admin grants). Cash-backed credits and owed
--     creator payouts REMAIN A DEBT WE OWE. "We took their money" is not a
--     position this product takes.
--  3. Thirty days to appeal, by email, reviewed by a person. An appeal does not
--     lift the suspension — but termination CANNOT EXECUTE while one is open and
--     undecided, and cannot execute before the deadline. That is enforced by the
--     write below and in `closures.ts`, not by an operator's memory.
--  4. The person is told WHICH GROUND was invoked and how to appeal. Not the
--     detection signal, not the reporter. `category` is NOT NULL for that
--     reason: a closure that cannot name its ground cannot be written.

CREATE TABLE account_closures (
  -- NO FOREIGN KEY, and that is not an oversight. `deleteAccount` erases the
  -- `accounts` row, and a `REFERENCES accounts(account_id)` here made erasing
  -- ANY closed account — including a LIFTED one, cleared of everything — fail
  -- with `FOREIGN KEY constraint failed`. So the operator erasure door was a 500
  -- for exactly the accounts most likely to need it, and a terminated account
  -- had no route out at all: not self-service (frozen), not appeal (the state is
  -- absorbing), not operator (this constraint). A review found it.
  --
  -- The link is TOMBSTONED on erasure instead (`account_data.ts`, the
  -- `scout_ordinals` pattern): the account id becomes a `del_` random, so no
  -- live account can ever match the row, while the fact that a closure happened
  -- — the evidence this table exists to keep — survives the person leaving.
  account_id   TEXT PRIMARY KEY,
  -- `lifted` is kept as a ROW rather than a delete: an account that was
  -- suspended and cleared has a history, and the appeal that cleared it is the
  -- most important part of it. Nothing reads a lifted row as a restriction.
  state        TEXT NOT NULL CHECK (state IN ('suspended','terminated','lifted')),
  -- THE ENUMERATED GROUNDS, and the vocabulary the notice is drawn from. It is
  -- a CHECK rather than a convention because decision 4 makes the category the
  -- thing the person is actually told, so an unnamed or invented ground is a
  -- notice nobody can act on.
  category     TEXT NOT NULL CHECK (category IN (
                 'chargeback_fraud','account_compromise','verified_theft',
                 'market_manipulation','economic_abuse','legal_compliance')),
  suspended_at TEXT NOT NULL,
  suspended_by TEXT NOT NULL,
  -- suspended_at + 30 days, stored rather than computed so the window a given
  -- account was actually given survives a later change to the constant.
  appeal_deadline   TEXT NOT NULL,
  appeal_opened_at  TEXT,
  appeal_decided_at TEXT,
  appeal_outcome    TEXT CHECK (appeal_outcome IN ('upheld','overturned')),
  terminated_at TEXT,
  terminated_by TEXT,
  lifted_at     TEXT,
  lifted_by     TEXT,
  -- OPERATOR-ONLY, and the reason decision 4 is implementable: the evidence
  -- summary lives here and the notice draws only on `category`. Nothing serves
  -- this column to the account.
  note         TEXT NOT NULL DEFAULT ''
);

-- The freeze is read on EVERY session resolution (`fanCaller` →
-- `currentSessionEpoch`'s fold), so it is a primary-key point lookup by
-- construction. This index is for the operator's own list and the termination
-- sweep: closures whose appeal window has run out, oldest first.
CREATE INDEX idx_closure_state ON account_closures(state, appeal_deadline);

-- 0074: WHEN AN ARCHIVE-REQUEST NOTE WAS LAST EDITED, AND WHAT IT SAID BEFORE
-- (RUN-0002, FINDING-0010 and FINDING-0012).
--
-- ---------------------------------------------------------------------------
-- WHAT THIS EXISTS FOR
-- ---------------------------------------------------------------------------
-- `archive_requests` is PRIMARY KEY (account_id, passport_id) — one row is one
-- fan asking one creator to open their back catalogue — so a count over it is a
-- DISTINCT-FAN POPULATION. RUN-0002 removed the differencing primitive from that
-- count at the data model: `requests.ts archiveDemand` counts only rows created
-- strictly before the start of the current UTC day, so the answer for a given
-- (creator, day) is the same at every hour and there is no pair of reads to
-- difference.
--
-- THE NOTES WERE STILL LIVE, and the row that carries them is not append-only.
-- `requestArchive`'s repeat branch turns a second request from the SAME fan into
-- `UPDATE archive_requests SET note = ?` and deliberately leaves `created_at`
-- alone — correctly, because membership was established the day the row was
-- inserted. But the notes read was bounded by `created_at` too, so text WRITTEN
-- TODAY by a fan whose row is from a completed day was served TODAY, stamped
-- with the older day. A creator could ask two named fans to send the same
-- string minutes apart and learn from the two screens which of them had already
-- requested the archive before — a per-fan membership oracle at second
-- granularity, rebuilt out of the one field the day-lag could not see.
-- (Executed; regression chain (f).)
--
-- AND LAGGING THE EDIT WAS ONLY HALF OF IT (FINDING-0012). `note_updated_at`
-- alone stops the NEW text appearing today, but it stops it by dropping the
-- whole row out of today's answer — and the OLD text was on the creator's three
-- live screens a moment earlier. So a creator could ask one NAMED fan to
-- re-send, watch exactly one note vanish within the second, and learn both that
-- this fan is in the counted population and WHICH of the messages on their
-- screen that fan wrote. Enumerable (n probes, n names), passive (a fan editing
-- unprompted shrinks the list at the instant the write lands), and with a
-- trace-free confirming branch (probe with the exact current text and the no-op
-- fires: the note survives and nothing is written). Clearing the note is the
-- same primitive with a permanent tail. A DAY-LAG OVER A MUTABLE FIELD IS ONLY
-- HALF A CONTROL; THE OTHER HALF IS RETAINING WHAT IT REPLACED, which is what
-- the two columns below are for. (Executed; regression chain (f'').)
--
-- ---------------------------------------------------------------------------
-- THE POLICY THESE THREE COLUMNS ARE, decided 2026-09-05
-- ---------------------------------------------------------------------------
--  1. THE READ SERVES THE TEXT THAT WAS CURRENT AT THE START OF TODAY, UTC. Per
--     row: if COALESCE(note_updated_at, created_at) is strictly before the start
--     of today, the row has not been touched today and `note` IS that text, dated
--     by that same expression. Otherwise the row WAS touched today and the text
--     it held this morning is in `prev_note`, dated `prev_note_day`. Either way
--     the served set for (creator, day) is a function of COMPLETED DAYS ONLY, so
--     nothing a fan does today can make a note appear or disappear today. An edit
--     or a clear takes effect tomorrow.
--  2. NULLABLE, AND NOT BACKFILLED. NULL `note_updated_at` means "never edited";
--     NULL `prev_note` means "nothing has been replaced". Both are the truth for
--     every row that exists today, so there is nothing to backfill, and a
--     backfill would have to invent an edit that never happened.
--  3. THE COUNT DOES NOT MOVE. These columns bound the NOTES only. Membership in
--     the population is the insert, so editing a note must not add a fan to the
--     count or remove one from it — `archiveDemand` keeps counting `created_at`.
--  4. ONLY A REAL TEXT CHANGE WRITES ANYTHING. `requestArchive` touches these
--     columns only when the submitted note differs from the stored one; an
--     identical, whitespace-only or identically-empty repeat performs no write at
--     all. Otherwise a fan who re-sent the same string would move the timestamp,
--     and the timestamp itself would become the second, finer clock this column
--     exists to remove.
--  5. AND A SECOND EDIT ON THE SAME DAY MUST NOT OVERWRITE THE RETAINED TEXT.
--     `prev_note`/`prev_note_day` are written only when the outgoing text has
--     already been published under a COMPLETED day. Copying today's first edit
--     into them would put text written today into today's answer — the original
--     defect, one field along.
--  6. A CLEAR IS A WITHDRAWAL, AND IT LAGS LIKE ANY OTHER EDIT (orchestrator,
--     2026-09-05). A fan who blanks their note removes it from TOMORROW's answer,
--     not today's; today still serves what stood there this morning. The fan
--     keeps the ability to correct or retract their own words, which is why this
--     shape was chosen over "notes are immutable once written".
--  7. NO INDEX. The notes read is already scoped to one passport and capped at
--     100 rows; an index here would buy nothing and would be one more artifact
--     carrying a per-fan instant.
--
-- Additive: three nullable columns on an existing table. No data is read,
-- rewritten or destroyed by this migration.

ALTER TABLE archive_requests ADD COLUMN note_updated_at TEXT NULL;
ALTER TABLE archive_requests ADD COLUMN prev_note TEXT NULL;
ALTER TABLE archive_requests ADD COLUMN prev_note_day TEXT NULL;

-- 0075: WHAT A COUNTED FAN LEAVES BEHIND WHEN THEY ERASE (RUN-0006,
-- FINDING-0013 and FINDING-0017 erase direction).
--
-- ---------------------------------------------------------------------------
-- WHAT THIS EXISTS FOR
-- ---------------------------------------------------------------------------
-- RUN-0002 made the archive-demand count a function of COMPLETED UTC DAYS:
-- `requests.ts archiveDemand` counts rows with `created_at < start of today`,
-- so the answer for a (creator, day) is the same at every hour and there is no
-- pair of reads to difference. But that predicate is evaluated over the LIVE
-- `archive_requests` table, and a row can leave the live table inside a day.
-- `account_data.ts deleteAccount` runs `DELETE FROM archive_requests WHERE
-- account_id = ?`, so a fan who was counted at 09:00 was not counted at 15:00 —
-- a delta of exactly one person, at an instant a creator can watch, on a
-- population a creator can grow by asking one named fan. The day-lag never saw
-- it, because the day-lag is a predicate over WHAT SURVIVES rather than a
-- property of WHAT WAS COUNTED. (Judge, RUN-0006.)
--
-- THIS TABLE MAKES THE DAY-KEY A PROPERTY OF THE COUNT. One row is inserted per
-- live `archive_requests` row in the SAME ATOMIC BATCH as the DELETE, carrying
-- only the two days the count needs: the day the request was made (which is
-- what decides whether a completed day counted it) and the day it was erased
-- (which is what decides how long it keeps being counted). Every day-lagged
-- read then counts `live + tombstones WHERE created_day < today AND erased_day
-- >= today`, so the erased fan's row is counted through the whole UTC day of
-- the erasure and drops at the next boundary — like every other change.
--
-- ---------------------------------------------------------------------------
-- WHAT IS NOT HERE, AND WHY EACH ABSENCE IS THE CONTROL
-- ---------------------------------------------------------------------------
-- NO account_id, and no tombstone id either. A pseudonymised link would make
-- this a record of "this erased person had asked these creators", which is the
-- fan data the erasure exists to destroy. The row names a CREATOR and two DAYS,
-- and nothing else can be recovered from it.
--
-- NO note, and no hash of one. The DELETE purges the fan's words (J1 (i),
-- Jason, 2026-09-06): the notes list may shrink within the day, which is a
-- stated property of this design, not an oversight — the note carries no
-- identity on the wire and the COUNT does not move. A retained note here would
-- be text a person asked us to destroy, kept for a reporting figure.
--
-- NO INSTANT. `substr(created_at, 1, 10)` and `dayOf(nowMs)` are ten-character
-- UTC days. An erasure timestamp would be a second clock at second granularity
-- attached to an act one person performs, which is the primitive RUN-0002 and
-- migration 0074 exist to remove.
--
-- ---------------------------------------------------------------------------
-- NO READER MAY EVER BE ADDED. THIS IS A HARD RULE, NOT A PREFERENCE.
-- ---------------------------------------------------------------------------
-- One fan who erases having asked five creators leaves FIVE rows that all carry
-- the same `(created_day, erased_day)` pair. Those rows are MUTUALLY LINKABLE:
-- anyone permitted to read them side by side can group by that pair and
-- recover "one person asked these five creators and erased on this day", which
-- is precisely the cross-creator profile the per-row tombstones elsewhere in
-- `deleteAccount` are shaped to prevent (see the `moneyTombstone` comment for
-- the one deliberate exception and what buys it). Unlinkability is bought here
-- by the ABSENCE OF A READER, not by the row shape: the only two statements
-- permitted to read this table are the COUNTs in `requests.ts archiveDemand`
-- (scoped to ONE `passport_id`) and `ratings.ts populationValues` (one COUNT
-- per `passport_id`, GROUP BY passport_id — grouping on the passport is the
-- only grouping allowed); each returns NUMBERS keyed by passport and nothing
-- else. Any query that groups, joins or filters on `(created_day, erased_day)`
-- across passports, any join, any export, any admin surface, any row-level
-- read reconstitutes the link and is refused. The prune step
-- (`requests.ts pruneArchiveTombstones`, in the `scheduled()` hygiene chain)
-- bounds the window a mistake could span to a single day.
--
-- ---------------------------------------------------------------------------
-- THE INDEX
-- ---------------------------------------------------------------------------
-- `(passport_id, erased_day)` serves both permitted readers (both filter one
-- passport and `erased_day >= today`) and the prune (`erased_day < today`).
-- `created_day` is deliberately NOT in it: it is a filter on the few rows the
-- index has already narrowed to, and putting it in the key would make the
-- index itself a `(creator, request day, erasure day)` ordering — a structure
-- that answers "which requests, made on which day, ended on this day" without
-- reading the table.
--
-- Additive: one new table and one new index. No existing table is read,
-- rewritten or destroyed by this migration; it ships with 0074.

CREATE TABLE archive_request_tombstones (
  passport_id TEXT NOT NULL REFERENCES passports(passport_id),
  created_day TEXT NOT NULL,
  erased_day  TEXT NOT NULL
);
CREATE INDEX idx_arch_tomb_read ON archive_request_tombstones(passport_id, erased_day);

-- 0076: THE TRANSACTIONAL OUTBOX (ADR-0006 §2). The row that makes an analytics
-- event exist IF AND ONLY IF its fact exists.
--
-- ---------------------------------------------------------------------------
-- WHY A TABLE AND NOT A QUEUE SEND
-- ---------------------------------------------------------------------------
-- A `ctx.waitUntil(queue.send(...))` beside a `db.batch([...])` is two writes
-- with no relationship: the batch can commit and the send can fail, and the
-- number a creator is shown is then quietly short with nothing to detect it
-- from. The outbox row is APPENDED INSIDE THE AUTHORITATIVE BATCH, so it
-- commits with the fact or not at all, and the publisher afterwards is free to
-- be at-least-once — `processed_events` in the analytics store absorbs the
-- duplicates.
--
-- ---------------------------------------------------------------------------
-- AN AMOUNT OR A PSEUDONYM, NEVER BOTH — AND THE CHECK IS WHY
-- ---------------------------------------------------------------------------
-- `CHECK (fan_key IS NULL OR amount_cents IS NULL)` is P2 written into the
-- schema. A row carrying both would be "this pseudonymous person paid this much
-- to this creator on this day" — a per-person spending record, assembled in the
-- one place that is copied out of production and into a second database. The
-- money events are creator-scoped and fan-key-free; the fan-keyed events carry
-- no amount. `buyers_new` is therefore written from the fan-keyed membership,
-- never from a money event (P3, A3).
--
-- `event_key` is a PER-ROW NONCE (`mintId("evt")`, CSPRNG), stored ON the row
-- so every re-delivery of that row carries the same key. It is derived from
-- NOTHING: an identity-derived key would make the idempotency ledger in the
-- second store a set of stable per-person identifiers, and would collapse two
-- genuine same-day events for one (fan, creator) into one (A3's same-day
-- re-follow). `occurred_day` is fixed by the producer at commit and is ten
-- characters: no sub-day timestamp crosses the queue (INV-051).
--
-- `dims` holds CLOSED ENUMS only — a JSON object whose keys and values are
-- module constants. No free text, no handle, no email, no IP, no
-- unbounded-cardinality value.
--
-- ---------------------------------------------------------------------------
-- NO FOREIGN KEYS. This is the deliberate opposite of `audience_regions`.
-- ---------------------------------------------------------------------------
-- The INSERT rides inside the AUTHORITATIVE batch. An FK to `passports` or to
-- `accounts` would therefore let a missing or concurrently-removed parent row
-- roll back a product write — a follow, a sale, a payout — because ANALYTICS
-- could not be recorded. Analytics never gets to veto the fact it describes.
-- The post-erasure guard the design does rely on is written as an explicit
-- `WHERE EXISTS (SELECT 1 FROM accounts WHERE account_id = ?)` in
-- `outboxStatements()`, which fails the ROW rather than the batch.
--
-- ---------------------------------------------------------------------------
-- `created_at`, AND WHY AN INSTANT IS ALLOWED ON THIS ROW
-- ---------------------------------------------------------------------------
-- The drain gives the request path a head start (`age > 60 s`) so the cron is a
-- REPAIR rather than the normal route. That age has to be measured from
-- something, and the something is the commit instant of the row itself.
--
-- IT IS A DEFAULT AND NOT A BIND. The producers' INSERT column lists
-- (`outboxStatements()`, `sqlOutboxSelect()`) are untouched, so no producer
-- learns a new column and no `INSERT … SELECT` fan-out has to carry a clock;
-- `strftime` fixes the value at COMMIT, which is exactly the instant the age is
-- wanted from.
--
-- IT IS NOT INV-051's SUB-DAY TIMESTAMP, and the reason is structural rather
-- than a promise: `analytics-outbox.ts MESSAGE_COLUMNS` is the queue contract
-- and does not name it, so this instant never crosses the queue and never
-- reaches the analytics store. It sits beside `attempts` and `published_at` —
-- three columns that are this publisher's own bookkeeping, all three stripped
-- from the body. `occurred_day` remains the only time the CONSUMER can see, and
-- it is ten characters.
--
-- ---------------------------------------------------------------------------
-- THE INDEXES
-- ---------------------------------------------------------------------------
-- The drain is `WHERE published_at IS NULL AND attempts < 5 AND created_at <=
-- (now - 60 s) ORDER BY seq LIMIT 100`, so the index it needs is a PARTIAL one over the
-- unpublished rows only — the published backlog (up to 7 days of every event on
-- the platform) is not in it, and the drain's cost is a function of what is
-- owed rather than of what has been done. `seq` is the rowid, so the partial
-- index is already in `seq` order.
--
-- `idx_outbox_repair` serves the republish rule: rows marked published that sit
-- above the consumer watermark's first gap and are older than 30 minutes
-- (pass-3 BREAK 7). The prune (`published_at IS NOT NULL AND seq < watermark`)
-- walks the rowid directly and needs no index of its own.

CREATE TABLE analytics_outbox (
  seq          INTEGER PRIMARY KEY AUTOINCREMENT,
  -- Per-row nonce. UNIQUE so a re-emit of the same logical row is refused here
  -- rather than becoming a second event downstream.
  event_key    TEXT NOT NULL UNIQUE,
  passport_id  TEXT NOT NULL,
  event_type   TEXT NOT NULL,
  occurred_day TEXT NOT NULL,                    -- 'YYYY-MM-DD', UTC, fixed at commit
  fan_key      TEXT,                             -- per-creator HMAC pseudonym, or NULL
  fan_key_v    INTEGER,                          -- the secret version behind fan_key
  scope_kind   TEXT,                             -- creator|event|buyer|release_attempt
  scope_id     TEXT,
  dims         TEXT,                             -- closed-enum JSON, or NULL
  amount_cents INTEGER,
  attempts     INTEGER NOT NULL DEFAULT 0,
  published_at TEXT,                             -- NULL => owed
  -- The commit instant, for the drain's age gate. Never sent (see the header).
  created_at   TEXT NOT NULL DEFAULT (strftime('%Y-%m-%dT%H:%M:%fZ', 'now')),
  CHECK (fan_key IS NULL OR amount_cents IS NULL)
);

-- The drain. PARTIAL: only what is owed.
CREATE INDEX idx_outbox_unpublished ON analytics_outbox(seq) WHERE published_at IS NULL;
-- The republish-above-the-watermark repair.
CREATE INDEX idx_outbox_repair ON analytics_outbox(published_at) WHERE published_at IS NOT NULL;

-- ---------------------------------------------------------------------------
-- THE DEAD LETTER TABLE. A message is never lost without a record (O-2).
-- ---------------------------------------------------------------------------
-- A row reaching 5 attempts, or a message the queue's DLQ hands back, is moved
-- here with the reason and the day it died, and the ops alert is raised. The
-- consumer watermark is ADVANCED PAST THAT SEQ (pass-3 BREAK 4): the dead row
-- is now the record of that event, so the prune below it is safe and the
-- repair rule does not chase a row that has already been accounted for.
--
-- Same columns as the outbox, and `seq` KEEPS ITS VALUE (no AUTOINCREMENT):
-- the seq is the dead row's identity, and an operator replay re-enqueues that
-- exact row. Kept 30 days — shorter than `analytics_erased`'s 90, so a replay
-- can never outlive the sentinel that would refuse it for an erased fan.
CREATE TABLE analytics_outbox_dead (
  seq          INTEGER PRIMARY KEY,
  event_key    TEXT NOT NULL UNIQUE,
  passport_id  TEXT NOT NULL,
  event_type   TEXT NOT NULL,
  occurred_day TEXT NOT NULL,
  fan_key      TEXT,
  fan_key_v    INTEGER,
  scope_kind   TEXT,
  scope_id     TEXT,
  dims         TEXT,
  amount_cents INTEGER,
  attempts     INTEGER NOT NULL DEFAULT 0,
  published_at TEXT,
  -- Carried over by the publisher's INSERT … SELECT so a dead row keeps the age
  -- it died at. The DLQ consumer rebuilds its row from a MESSAGE, which carries
  -- no instant (INV-051), so there the default stands and records the death.
  created_at   TEXT NOT NULL DEFAULT (strftime('%Y-%m-%dT%H:%M:%fZ', 'now')),
  dead_at      TEXT NOT NULL,
  reason       TEXT NOT NULL,                    -- closed enum: attempts|delivered_above_gap|dlq
  CHECK (fan_key IS NULL OR amount_cents IS NULL)
);

-- The 30-day prune.
CREATE INDEX idx_outbox_dead_prune ON analytics_outbox_dead(dead_at);

-- 0077: WHICH CREATORS A FAN'S PSEUDONYMS EXIST FOR (ADR-0006 §2/§5).
--
-- ---------------------------------------------------------------------------
-- THIS IS PRODUCTION DATA OF THE `creator_encounters` CLASS
-- ---------------------------------------------------------------------------
-- The row is `(this account, this creator)` in plain text: it says that this
-- person's activity has been counted into that creator's analytics. That is the
-- same fact `creator_encounters` (0005) already holds, held for the same kind
-- of reason, and it is governed the same way — it lives in PRODUCTION D1, is
-- covered by the account's erasure, and NEVER crosses the queue. Nothing here
-- is copied into the analytics store; what crosses is the HMAC.

-- creators the account has pseudonyms under, computes each `fan_key` from the

-- deletes these rows and the ledger row. After that no raw account id related
-- to analytics exists anywhere (P2).
--
-- ---------------------------------------------------------------------------
-- WHY NO KEYS ARE STORED HERE
-- ---------------------------------------------------------------------------
-- Storing `fan_key` beside `account_id` would make this table the join that the
-- per-creator pseudonym exists to prevent: one row would translate a pseudonym
-- back to a person, and a set of rows would translate one person across every
-- creator. The drain RECOMPUTES each key from `ANALYTICS_FAN_SECRET`, which the
-- table does not hold (A3, O-5).

-- ---------------------------------------------------------------------------
-- WHY THERE IS NO FOREIGN KEY TO `accounts`, AND WHY THE ROWS OUTLIVE THE BATCH
-- ---------------------------------------------------------------------------
-- Both halves are load-bearing and neither is an oversight.
--

-- rows must still be there for the drain to read. An FK would either fail that
-- batch or (ON DELETE CASCADE) destroy the only enumeration of the scopes the
-- erasure still has to reach — the fan would be deleted from production and
-- left, forever, inside five creators' analytics.
--

-- named: a producer whose `WHERE EXISTS (SELECT 1 FROM accounts ...)` guard was
-- evaluated just before the erasure committed writes its scope row too, and a
-- delete inside the batch would miss it. The scopes outlive the batch and the
-- drain re-reads them once (§5c) before deleting.
--
-- ---------------------------------------------------------------------------
-- WHAT IS NOT HERE
-- ---------------------------------------------------------------------------
-- No timestamp: the row is a set membership, not a history of when this person
-- first touched that creator — `creator_encounters` already answers that, under
-- its own governance, and a second clock on the same fact is a second thing to
-- erase. No count, no scope, no source: the drain needs the pair and nothing
-- else, and every extra column here is a column that has to be erased twice.

CREATE TABLE analytics_fan_scopes (
  account_id  TEXT NOT NULL,
  passport_id TEXT NOT NULL,
  PRIMARY KEY (account_id, passport_id)

-- 0078: THE ERASURE WORK LEDGER (ADR-0006 §5). One row, written INSIDE
-- `deleteAccount`'s batch, before its DELETEs.
--
-- ---------------------------------------------------------------------------
-- WHY A LEDGER AND NOT A QUEUE EVENT
-- ---------------------------------------------------------------------------
-- The erasure must reach a SECOND DATABASE, and two D1 bindings never share a
-- batch. Sending it as an analytics event would put the erasure on the same
-- at-least-once path as everything it is supposed to undo — behind the same
-- backlog, subject to the same dead-letter table, with the fan's departure
-- sitting in a queue. So the durable record is a row that commits WITH the
-- deletion, and a drain does the cross-database work afterwards; every step of
-- it is replay-safe, so a Worker death costs a retry and never a missed
-- erasure.
--
-- ---------------------------------------------------------------------------
-- NO KEYS. THE DRAIN COMPUTES THEM.
-- ---------------------------------------------------------------------------
-- An earlier shape stored the fan's `fan_key`s on this row so the drain would
-- not need the secret. That makes the erasure ledger a durable list of "these
-- pseudonyms are this person" — the exact translation table the per-creator key
-- exists to prevent, written down at the moment a person asked to be forgotten
-- (A3). This row carries the ACCOUNT and the DAY, and the drain derives
-- everything else from `analytics_fan_scopes` (0077) and the secret.
--
-- `requested_day` is ten characters and is what the tombstones' `erased_day`
-- carries, so an erased fan is counted through the whole UTC day of the erasure
-- and drops at the next boundary — the 0075 direction, and the reason no served
-- number moves inside a day.
--
-- `created_at` is the drain's ORDERING ONLY (oldest first, LIMIT 20 per tick).
-- It is an instant on a row that is deleted minutes later, in production, and
-- it never crosses into the analytics store — where the same fact is the
-- ten-character `erased_day` and nothing finer.
--
-- No FK to `accounts`, for the reason 0077 spells out: `DELETE FROM accounts`
-- is the last statement of the very batch that writes this row.

CREATE TABLE analytics_erasures (
  erasure_id    TEXT NOT NULL PRIMARY KEY,
  account_id    TEXT NOT NULL,
  requested_day TEXT NOT NULL,                   -- 'YYYY-MM-DD', UTC
  created_at    TEXT NOT NULL
);

-- The drain takes the oldest 20 per tick.
CREATE INDEX idx_analytics_erasures_drain ON analytics_erasures(created_at);

-- 0079: THE FAN-DAY CLAIM (ADR-0006 §2, O-12/O-23, pass-3 BREAK 1/2/11/12).
-- Two columns on `creator_encounters`, and they turn a read-then-write into an
-- atomic claim.
--
-- ---------------------------------------------------------------------------
-- WHAT WAS WRONG WITH THE SHIPPED PATH
-- ---------------------------------------------------------------------------
-- `route-helpers.ts logEncounter` decided "is this a new UTC day for this
-- (fan, creator)?" by POINT-READING `creator_encounters.last_seen_at`
-- (`codex.ts:70-77`) and then writing the geo increment. Two concurrent
-- requests both read the old day and both increment, so `fan_days` — which is
-- defined as DISTINCT FANS — could count one fan twice; and the day the
-- increment landed on came from a timestamp minted before the read rather than
-- from the row that won.
--
-- ---------------------------------------------------------------------------
-- THE CLAIM
-- ---------------------------------------------------------------------------
-- `claimed_day` is the UTC day this (fan, creator) pair has already been
-- counted into `audience_regions`. `claim_token` is a CSPRNG token minted by
-- the Worker for one request. The whole encounter path becomes ONE atomic D1
-- batch: bump the sighting counters; then an upsert that writes
-- `(claimed_day, claim_token)` only `WHERE creator_encounters.claimed_day <
-- excluded.claimed_day`; then the geo increment and the `fan_active` outbox and
-- scopes rows, EACH GATED on `claim_token = ?token` — so only the request whose
-- token won the claim writes them. Two requests in the same millisecond cannot
-- both win, and the batch is atomic, so a Worker death leaves a full claim or
-- none.
--
-- The day comes from `substr(ts, 1, 10)` where `ts` is minted after every await
-- and committed in that one round-trip, which is what bounds the request path's
-- window (the build grace rests on this).
--
-- ---------------------------------------------------------------------------
-- BOTH COLUMNS ARE NULLABLE, AND MUST BE
-- ---------------------------------------------------------------------------
-- SQLite cannot add a NOT NULL column without a default, and a default here
-- would be a lie about every existing row — a fabricated claim day would
-- SUPPRESS the first real count for every pair already in the table. NULL means
-- "never claimed", and `claimed_day < excluded.claimed_day` is written so the
-- first claim wins on a NULL through the upsert's IS-NULL handling in
-- `codex.ts`, not by inventing a day here.
--
-- NO INDEX. Every read of these columns is by the table's own PRIMARY KEY
-- `(account_id, passport_id)`; an index on `claimed_day` would order the whole
-- encounter table by "which fans were active on which day", across creators,
-- which is a structure this product does not have and does not want.
--
-- Additive: two columns, no rewrite of any existing row.

ALTER TABLE creator_encounters ADD COLUMN claimed_day TEXT;
ALTER TABLE creator_encounters ADD COLUMN claim_token TEXT;

-- 0080: WHAT A COUNTED FAN LEAVES BEHIND WHEN THEY ERASE, FOR THE TWO
-- POPULATIONS THE ROI DASHBOARD COUNTS (S-002 S-17, ADR-0006 §5; B7 OPEN
-- QUESTION 2). Migration 0075's shape, applied twice more.
--
-- ---------------------------------------------------------------------------
-- WHAT THIS EXISTS FOR
-- ---------------------------------------------------------------------------
-- S-1 moved every fan-derived count on the four legacy dashboards onto the
-- analytics store or onto a DAY-PURE production read. Two counts had neither
-- and were served WITHHELD: `CreatorRoi.discovery.requests` (`creator_requests`)
-- and `CreatorRoi.conversion.supporters_total` (`reverse_code_claims`). Both
-- are per-fan populations, and a day-lagged predicate over the LIVE table is
-- not day-pure — which is FINDING-0013 exactly: `deleteAccount` removes the
-- fan's `creator_requests` row and pseudonymises their `reverse_code_claims`
-- row (so the Sybil-gate JOIN to `accounts` stops matching), and a count a
-- COMPLETED day had already made moves by exactly one person at an instant a
-- creator can watch, on a population a creator can grow by asking one named
-- fan.
--
-- These two tables make the day-key a property of the count, as 0075 did for
-- archive demand. One row is written in the SAME ATOMIC BATCH as the statement
-- that removes the fan from the population, carrying only the two days the
-- count needs: the day the fan BECAME COUNTABLE and the day they erased. The
-- reads then count `live + tombstones WHERE created_day < T AND erased_day >=
-- T`, so the erased fan keeps being counted for the whole UTC day of the
-- erasure and drops at the next boundary, like every other change.
--
-- `created_day` IS NOT ALWAYS THE ROW'S OWN DAY, and that is the point: it is
-- the first day the live predicate would have counted the row.
--   * `creator_request_tombstones`: the request's day. The live count is
--     `created_at < T`, so `created_day < T` is the same membership rule.
--   * `reverse_code_claim_tombstones`: `max(claim day, account creation day +
--     7)`, because `supporter-status.ts` counts a claim only once the claiming
--     account is seven days old (the Sybil gate) — the day-pure form of which
--     is `accounts.created_at < T − 7 days`. Writing the LATER of the two days
--     makes the tombstone's rule character-for-character the live one, so an
--     erasure inside a fan's first week cannot make a count move UP.
--
-- ---------------------------------------------------------------------------
-- ONE ROW PER (CREATOR, ERASED FAN) — NOT ONE PER CLAIM
-- ---------------------------------------------------------------------------
-- `supporters_total` is DISTINCT FANS, and `reverse_code_claims` is keyed
-- (pool_id, account_id): one fan who claimed from three of a creator's pools is
-- one supporter and three rows. A per-claim tombstone would add three where the
-- live count lost one, and the total would jump by two at the erasure instant —
-- the same watchable mid-day delta this table exists to remove, pointing the
-- other way. So the INSERT is `GROUP BY passport_id` with `MIN(countable day)`,
-- and no `pool_id` column exists to tempt a future per-pool reader: per-pool
-- claim and redemption counts are creator-scoped STORE cells (S-1), never a
-- live count over this table.
--
-- ---------------------------------------------------------------------------
-- WHAT IS NOT HERE, AND WHY EACH ABSENCE IS THE CONTROL (0075's rules, verbatim
-- in force)
-- ---------------------------------------------------------------------------
-- NO account_id and no tombstone id. A pseudonymised link would make these a
-- record of "this erased person had asked / claimed from these creators", which
-- is the fan data the erasure exists to destroy. A row names a CREATOR and two
-- DAYS, and nothing else can be recovered from it.
--
-- NO INSTANT. Both columns are ten-character UTC days. An erasure timestamp
-- would be a second clock at second granularity attached to an act one person
-- performs.
--
-- NO RECENCY DAY, which is why `supporters_active` and `supporters_core` stay
-- WITHHELD rather than joining `supporters_total` here. Those two are windowed
-- populations ("claimed in the last 90 days", "had a code marked redeemed in
-- the last 90 days"), so counting an erased fan through their erasure day would
-- need a THIRD day on a per-person row — a finer fingerprint on a row whose
-- unlinkability is bought by having almost no shape — and `core` additionally
-- turns on `reverse_codes.redeemed_at`, an act the CREATOR performs, on the
-- creator's own clock, after the fan is gone. Withheld is the honest answer and
-- `analytics-read.ts readLegacyFacts` says so at the call site.
--
-- ---------------------------------------------------------------------------
-- NO READER MAY EVER BE ADDED. THIS IS A HARD RULE, NOT A PREFERENCE.
-- ---------------------------------------------------------------------------
-- One fan who erases having asked five creators leaves FIVE rows that all carry
-- the same `(created_day, erased_day)` pair, and those rows are MUTUALLY
-- LINKABLE: anyone permitted to read them side by side can group by that pair
-- and recover "one person asked these five creators and erased on this day".
-- Unlinkability is bought by the ABSENCE OF A READER, not by the row shape. The
-- only statements permitted to read these tables are the two COUNTs in
-- `analytics-read.ts readLegacyFacts`, each scoped to ONE `passport_id` and
-- each returning a NUMBER. Any query that groups, joins or filters on
-- `(created_day, erased_day)` across passports, any join, any export, any admin
-- surface, any row-level read reconstitutes the link and is refused. The prune
-- step (`analytics-read.ts pruneRoiTombstones`, in the `scheduled()` hygiene

-- to a single day.
--
-- ---------------------------------------------------------------------------
-- THE INDEXES
-- ---------------------------------------------------------------------------
-- `(passport_id, erased_day)` on each, serving the one reader (one passport,
-- `erased_day >= T`) and the prune (`erased_day < today`). `created_day` is
-- deliberately NOT in either: it is a filter on the few rows the index has
-- already narrowed to, and putting it in the key would make the index itself a
-- `(creator, membership day, erasure day)` ordering — a structure that answers
-- "which of them, counted from which day, ended on this day" without reading
-- the table.
--

-- rewritten or destroyed by this migration.

CREATE INDEX idx_creq_tomb_read ON creator_request_tombstones(passport_id, erased_day);

-- (creator, country, day) and N `fan_active` events under ONE fan_key.
--
-- `active_fans(day) := Σ_country audience_regions(day).fan_days` (ADR §4) is
-- the day's whole active population, and both the geo day gate
-- (GEO_DAY_MIN_ACTIVE) and the K floor are computed over it — so a fan-day one
-- account can mint at will is a k-anonymity bypass, not only a metrics error.
--
-- ---------------------------------------------------------------------------
-- THE FAN-DAY LEDGER IS NOT CODEX CONTENT
-- ---------------------------------------------------------------------------
-- The Codex is the fan's private sighting history, theirs to clear. The claim
-- is the ledger of which fan-days have already been counted into the creator's
-- geo cell — it is not something the fan is shown, exports, or owns, and a fan
-- clearing what they see must not re-arm what the creator is billed for. Two
-- different lifetimes, so two different tables. `clearCodex` keeps its
-- `DELETE FROM creator_encounters` and never touches this one.
--
-- ---------------------------------------------------------------------------
-- NO READER OUTSIDE TWO PLACES, AND THAT IS THE PRIVACY ARGUMENT
-- ---------------------------------------------------------------------------
-- This table holds `(account_id, passport_id, day)` — a raw fan id beside a
-- creator and a day, which is exactly the shape `fan_key` exists to keep out of
-- the analytics store. It stays out of it by having NO READER: the only
-- statements permitted to touch this table are
--   (a) `codex.ts recordEncounter`'s own batch — one upsert and the three
--       `claim_token = ?` gates that read it back inside that same batch; and
--   (b) the erasure `DELETE FROM creator_day_claims WHERE account_id = ?` in
--       `account_data.ts deleteAccount`'s batch.
-- No export, no admin surface, no join, no row-level read, ever. Same rule the
-- 0080 tombstones carry, for the same reason.
--
-- NO INDEX. Every read is by this table's own PRIMARY KEY
-- `(account_id, passport_id)`. An index on `claimed_day` would order the whole
-- table by "which fans were active on which day" across creators, which is a
-- structure this product does not have and does not want (0079's rule, kept).
--
-- BOTH VALUE COLUMNS ARE NOT NULL. 0079's had to be nullable because an ALTER
-- TABLE cannot add NOT NULL to a table that already has rows; this table has no
-- pre-existing rows and a row is only ever born from the claim upsert, which
-- always supplies both. The upsert's predicate keeps 0079's
-- `claimed_day IS NULL OR claimed_day < excluded.claimed_day` spelling verbatim
-- so the two do not diverge — on this table the IS-NULL limb is unreachable.
--
-- ---------------------------------------------------------------------------
-- 0079's COLUMNS STAY, AND ARE NO LONGER WRITTEN
-- ---------------------------------------------------------------------------
-- `creator_encounters.claimed_day` and `.claim_token` are NOT dropped. SQLite's
-- DROP COLUMN would rewrite the table, and the append-only migration rule means
-- a shipped column is not un-shipped. From this migration on, nothing writes or
-- reads them: `recordEncounter`'s INSERT no longer names them and its claim,
-- geo and outbox statements all address `creator_day_claims`. They are dead
-- columns carrying whatever 0079 last wrote, and no served value depends on
-- them.
--
-- ---------------------------------------------------------------------------
-- NO FOREIGN KEYS, BECAUSE `creator_encounters` HAS NONE
-- ---------------------------------------------------------------------------
-- This table is a SIBLING of `creator_encounters` split off by lifetime, not a
-- new relational entity, and `0005_social.sql:68-79` gives that table no FK on
-- either column. Declaring them here would change two behaviours that are not
-- this migration's to change: `recordEncounter`'s batch would THROW for a fan
-- whose `accounts` row is already gone (today the `accounts` EXISTS guard in
-- `outboxStatements` silently fails the row and the encounter still records —
-- `test/s002-producers-engagement.test.ts` pins exactly that), and an erasure
-- missing this table's DELETE would fail the whole batch rather than the one
-- assertion that names it.
--
-- Erasure completeness is enforced where it is already enforced: the
-- `FAN_COLUMNS` audit in `test/account-data.test.ts` declares this column
-- `deleted`, seeds a row, and asserts zero rows after `deleteAccount` — a

  claim_token TEXT NOT NULL, -- the request that won it

-- 0082: WHICH EPOCH BUMPS WERE THE CREATOR'S OWN (docket #37 (b), as amended by
-- the S-019 W0 brief-attack; closes FINDING-0014-C-1).
--
-- ---------------------------------------------------------------------------
-- WHAT THIS EXISTS FOR
-- ---------------------------------------------------------------------------
-- `consumeManualClaimRef` (src/manual-claims.ts) decides whether an approved
-- manual claim ref is still consumable by asking a TIMESTAMP question: was
-- there an epoch bump at or after the instant the operator approved? It asked
-- it of `updated_at`, which every bump stamps — the operator's force-logout,
-- the hijack recovery's, and the creator's own "log out everywhere" alike. So a
-- creator who tapped logout-all between her approval and her first poll burned
-- her own approved claim, permanently, and the support path was a ticket.
--
-- WHY A WATERMARK AND NOT A REASON. This table is `passport_id PRIMARY KEY` —
-- ONE row per creator, overwritten by every bump. A `reason` column records only
-- the LAST bump, so a recovery bump followed later by an ordinary self-service
-- logout would read `self_service` and re-arm a ref the recovery legitimately
-- killed. That is the INV-020 case the inertness rule exists for. A watermark is
-- ORDER-INDEPENDENT: it answers "was there ANY involuntary bump at or after T",
-- which is the question the code actually asks, and a later voluntary bump
-- cannot erase an earlier involuntary one.
--
-- WHO WRITES IT. The operator/recovery doors only — `console.ts` POST
-- /ops/creators/:id/force-logout and `recovery.ts` action `force_logout`, both
-- through `bumpCreatorEpoch(..., involuntary=true)`. The creator's own
-- logout-all (`creator-contact-routes.ts`) and her own passkey revocation
-- (`passkeys.ts`, which carries its own upsert so the bump can ride the revoke's
-- batch) do not write it, and `passkeys.ts` naming no column here is what leaves
-- an earlier watermark intact rather than clearing it.
--
-- ---------------------------------------------------------------------------
-- THE BACKFILL, AND WHY IT IS THE CONSERVATIVE DIRECTION
-- ---------------------------------------------------------------------------
-- Left NULL, every row that exists today would read "no involuntary bump on
-- record" and every ref burned in the past would silently become consumable
-- again — a retroactive verdict change on a population nobody has looked at,
-- including refs burned by a real recovery. Backfilling the watermark to the
-- row's own `updated_at` reads every historical bump as involuntary, which
-- preserves TODAY'S verdict for every existing row exactly and confines the
-- behaviour change to bumps taken after this migration runs. The cost is the
-- direction that fails CLOSED: a creator whose ref was burned by her own logout
-- before today still has to ask, which is the situation she is in now.
ALTER TABLE creator_sessions ADD COLUMN last_involuntary_bump_at TEXT;
UPDATE creator_sessions SET last_involuntary_bump_at = updated_at;

-- 0092: ONE PERSON, ONE SEAT PER AGENCY+PROVIDER — ENFORCED BY THE DATABASE.
--
-- AG-04. `addAgencyMember` guarded duplicate enrolment with a SELECT followed by
-- an INSERT, and `idx_agency_member_email` was a plain (non-unique) index. Two
-- administrators authorizing the same person at the same moment therefore both
-- read "no such member" and both inserted. The consequence is not a cosmetic
-- duplicate row: first login binds ONE of the two invitations, self-service
-- removal deletes only the member_id it is handed, and the residual unbound
-- invitation later re-authorizes the very identity the agency believed it had
-- removed — possibly at the other invitation's DIFFERENT role.
--
-- The boundary is deliberately (agency_id, provider, email) and not email
-- alone: the same person legitimately holds seats at several agencies, and the
-- same address under google and apple is two distinct provider identities.
-- Both of those remain allowed; this constraint only forbids the thing that was
-- never intended.
--
-- Deduplicate first, or the index cannot be created on a database that already
-- raced. The survivor is chosen by the same rule the application would: a BOUND
-- row (one somebody has actually signed in with) beats an unbound one, and
-- among equals the earliest enrolment wins — so nobody's live session is
-- deleted by a bookkeeping migration.
DELETE FROM agency_members
 WHERE rowid NOT IN (
   SELECT rowid FROM (
     SELECT rowid,
            ROW_NUMBER() OVER (
              PARTITION BY agency_id, provider, email
              ORDER BY (CASE WHEN sub IS NULL THEN 1 ELSE 0 END) ASC, added_at ASC, rowid ASC
            ) AS rn
       FROM agency_members
   ) WHERE rn = 1
 );

DROP INDEX idx_agency_member_email;
CREATE UNIQUE INDEX idx_agency_member_email ON agency_members(agency_id, provider, email);

-- ---------------------------------------------------------------------------
-- 0093 — THE FOUNDING-FAN FREEZE DECISION, RECORDED INDEPENDENTLY OF ITS AWARDS
-- (S-070 JOBS2, SJ-13)
-- ---------------------------------------------------------------------------
--
-- `freezeFoundingFanRanks` decided, once and forever, which fans hold a
-- passport's Founding Fan ordinals. It recorded that decision ONLY as the award
-- rows themselves — `founding_fan_ranks` — and used "does any row exist for this
-- passport?" as the proof that the decision had been made. A surrogate for
-- completion is not completion, and this one was wrong in two directions at
-- once:
--
--  1. AN EMPTY DECISION WRITES NOTHING. A passport whose requesters were all
--     unqualified (or too young) at the immutable claim instant has an empty
--     slate, so the freeze writes no row and leaves no trace that it ran. The
--     cron sweep selects the first 200 claimed passports that have requests and
--     no rank rows, ORDER BY passport_id ASC — so those passports sit at the
--     head of that query FOREVER. A full batch of them is a permanent wall: the
--     sweep re-decides the same 200 empty slates on every tick and never reaches
--     an eligible passport behind them. Nothing reports this; the sweep returns
--     `frozen = 0` and looks idle.
--
--  2. ERASURE OF EVERY HOLDER REOPENS SPENT ORDINALS. `deleteAccount` removes
--     that fan's request and their `founding_fan_ranks` row. If every original
--     rank holder erases while other historically eligible requests remain, the
--     "any row exists" check becomes FALSE again and a later sweep allocates
--     ordinals that were already spent — to different people, under the same
--     public numbers. `0020_fan_quality.sql` and `account_data.ts` both promise
--     in writing that a vacated slot stays empty and a number is never reused.
--
-- One row per passport, written in the SAME atomic batch as the award rows (or
-- alone, when the decision was empty), closes both. It is PASSPORT-KEYED and
-- carries no account identity, which is what makes it survive fan erasure:
-- `deleteAccount`'s deletions are all `WHERE account_id = ?`, so nothing in the
-- erasure path can reach this table, and it owes no erasure/export disposition
-- (`account-data.test.ts` derives that list from the schema by column name — a
-- table with no fan identifier in it arrives decided).
--
-- `slots` is the SIZE of the decision, including zero. It is the operator's
-- answer to "did this passport get an empty slate, or has it not been decided?",
-- which is precisely the question nothing could answer before.
--
-- BACKFILL. Every passport that already holds rank rows was already decided, and
-- its decision must not be re-opened by this migration's own arrival: the
-- SELECT below writes their freeze rows from the awards themselves, with the
-- `frozen_at` those rows already carry (all rows of one freeze share the claim
-- instant). A passport with an EMPTY historical decision cannot be recovered
-- here — there is nothing in the database that records it — so it is simply
-- decided again on the next sweep, which is correct and is the one-off cost of
-- having had no completion record.
--
-- OPERATOR NOTE (DEPLOY.md): after this migration, a passport present in
-- `founding_fan_ranks` but ABSENT from `founding_freezes` means the backfill
-- below did not run — the freeze check would then still be reading the award
-- rows and consequence (2) above is live. The reconciliation query is
--   SELECT DISTINCT r.passport_id FROM founding_fan_ranks r
--    WHERE NOT EXISTS (SELECT 1 FROM founding_freezes f WHERE f.passport_id = r.passport_id);
-- It must return zero rows. A row in `founding_freezes` with `slots = 0` is a
-- real, recorded, EMPTY decision and is not a fault.

CREATE TABLE founding_freezes (
  passport_id TEXT PRIMARY KEY,
  -- The claim instant the slate was derived from — the same value every
  -- `founding_fan_ranks.frozen_at` of this passport carries, so a re-derivation
  -- on another machine a week later is byte-identical.
  frozen_at   TEXT NOT NULL,
  -- How many ordinals the decision allocated. ZERO IS A DECISION.
  slots       INTEGER NOT NULL DEFAULT 0 CHECK (slots >= 0)
);

INSERT OR IGNORE INTO founding_freezes (passport_id, frozen_at, slots)
  SELECT passport_id, MIN(frozen_at), COUNT(*) FROM founding_fan_ranks GROUP BY passport_id;

-- ---------------------------------------------------------------------------
-- 0094 — AN INDEX SO THE PAYOUT FAN-OUT CAN RESUME ON IDENTITY (S-070 JOBS2,
-- SJ-16). Purely additive: one index, no table touched, no column added, no row
-- rewritten.
-- ---------------------------------------------------------------------------
--
-- `sweepPayoutChangeFanout` resumed across invocations by seeking the
-- `covered`-th saver in the CURRENT ordered set. 0029 chose a COUNT over an
-- account id on purpose — the marker must hold no fan identifier — and wrote
-- down the cost: "if the saver set changed between invocations the resume point
-- can drift by the number of savers added or removed… the insert is ON CONFLICT
-- DO NOTHING, so re-covering ground is free."
--
-- Re-covering is free. SKIPPING is not, and removals cause skipping. An
-- addition below the boundary pushes the numbered position BACKWARD (re-cover,
-- free); a removal pushes it FORWARD, so the next range begins AFTER savers who
-- were never notified, and `done = 1` makes the omission permanent. Unsaving and
-- account erasure both delete `saves` rows. The notice is the detection half of
-- the payout-change safety — it is what tells a fan the destination they were
-- about to send money to has moved — so an omitted recipient is the failure the
-- whole mechanism exists to prevent.
--
-- The repair keeps 0029's privacy property exactly: the resume point is now the
-- greatest `account_id` this change has ALREADY NOTIFIED, read from
-- `notifications` — a table that already holds the fan identifier, already has
-- an erasure and export disposition, and already carries the `payout:<change>`
-- event id. NOTHING new is stored anywhere, and `payout_change_fanout` stays
-- free of fan identity.
--
-- This index is what makes that read affordable. `notifications`' existing
-- indexes are keyed on `(account_id, notif_id)`, on `delivered`, on `emailed`
-- and (0052) on the safety-kind partial — none can seek `event_id`, so the
-- MAX() would scan the whole table once per open marker per invocation.
--
-- OPERATOR NOTE: this changes only the COST of the payout fan-out's resume, not
-- its result — but the fan-out's behaviour does change with it (fans are no
-- longer skipped), so if you are diffing notification volumes across this
-- deploy, a payout change that was mid-fan-out at the cutover may deliver to
-- more recipients than the old resume would have reached. That is the fix, not a
-- regression.

CREATE INDEX idx_notif_event_account ON notifications(event_id, account_id);

-- 0095 — S-070 BILLING2 (BF-13): durable per-payout retry state.
--
-- WHAT WAS WRONG. `runWeeklyPayouts` STEP 0 re-attempts pending payouts with
--   SELECT ... FROM creator_payouts WHERE status = 'pending' <window>
--   ORDER BY passport_id ASC, created_at ASC LIMIT PAYOUT_RETRY_BUDGET
-- and `creator_payouts` (0036) carried NO record of an attempt ever having been
-- made. The `failed` status in its CHECK is never written — a refused transfer
-- goes onto an in-memory `out.failed` and the row stays `pending` — so a
-- destination that fails permanently keeps its place at the head of that
-- ordering on every tick, for ever. Once PAYOUT_RETRY_BUDGET such rows exist,
-- every later pending row in the same window is never attempted again. Their
-- amounts stay RESERVED (`creatorEarnings` subtracts pending + paid), so those
-- creators are not merely unpaid: their cleared balance is also withheld from
-- new payouts. One creator's broken Stripe account denied service to others'.
--
-- Deployments with fewer connected creators than PAYOUT_PAGE are the common
-- case and the worst one: the cursor wraps every tick, so it is the same window
-- every time.
--
-- WHAT THIS ADDS. Two columns, both additive, no table rewritten, no index
-- dropped, no existing column touched:
--   `attempts`        how many transfer attempts this payout has been refused.
--   `next_attempt_at` the earliest RFC3339 instant STEP 0 may try it again.
--
-- OPERATOR NOTE — WHAT A PRE-MIGRATION ROW MEANS.
-- Every row that existed before this migration has `attempts = 0` and
-- `next_attempt_at IS NULL`. NULL means **"eligible now, never attempted"** —
-- it does NOT mean "never retry", and STEP 0 treats it as immediately payable,
-- which is exactly the behaviour those rows had before this migration. A
-- pending row that has been failing for months therefore gets one more
-- immediate attempt after deploy and only then starts backing off; that is
-- deliberate, because the database does not know how many times it already
-- failed and inventing a count would be worse than admitting there isn't one.
-- `attempts = 0` on a pending row is consequently ambiguous ONLY across this
-- deploy boundary: after the first post-deploy tick it means what it says.
--
-- A row with `next_attempt_at` in the future is NOT stuck and is NOT an error;
-- it is a row whose destination refused and which is waiting out its backoff.
-- To see the backlog:
--   SELECT passport_id, attempts, next_attempt_at FROM creator_payouts
--    WHERE status = 'pending' AND attempts > 0 ORDER BY attempts DESC;
-- Rows with a high `attempts` are the ones an operator should look at: the
-- money is still reserved and still owed, and the backoff means the run has
-- stopped spending its ceiling on them.
ALTER TABLE creator_payouts ADD COLUMN attempts INTEGER NOT NULL DEFAULT 0;
ALTER TABLE creator_payouts ADD COLUMN next_attempt_at TEXT;

-- The selection STEP 0 makes every tick: pending, in a passport window, due.
CREATE INDEX idx_payouts_pending_due ON creator_payouts(status, next_attempt_at, passport_id);

--
-- WHAT WAS WRONG. `buildRegistry` rebuilt the whole public registry inside one
-- Worker invocation, under a 10-minute D1 lease it never re-checked, and wrote
-- buckets and proofs over the live keys the public worker serves. A slow build
-- could outlive its lease and interleave with a second one (I3-0246), a fault

--  * registry_builds — one row per generation. `epoch` is the fence: every

--    could not admit is `pending` here; one still pending after a committed build
--    triggers a rung step at the next plan. Nothing is ever evicted.

--    `acct_vetted` / `acct_asserted` columns (migration 0097).

  seq        INTEGER PRIMARY KEY,

  state      TEXT NOT NULL CHECK (state IN ('open', 'committed', 'abandoned')),
  phase      TEXT NOT NULL,
  cursor     INTEGER NOT NULL DEFAULT 0,
  plan_json  TEXT NOT NULL,
  agg_json   TEXT NOT NULL DEFAULT '{}',
  failures   INTEGER NOT NULL DEFAULT 0,
  reason     TEXT,
  gc_state   TEXT NOT NULL DEFAULT 'pending' CHECK (gc_state IN ('pending', 'stage_done', 'done')),
  started_at TEXT NOT NULL,
  updated_at TEXT NOT NULL,
  ended_at   TEXT
);
CREATE INDEX idx_registry_builds_state ON registry_builds (state, seq);

CREATE TABLE registry_safety_admission (
  subject_hash TEXT PRIMARY KEY,
  state        TEXT NOT NULL CHECK (state IN ('pending', 'admitted', 'refused')),
  since        TEXT NOT NULL,
  since_seq    INTEGER NOT NULL,
  updated_seq  INTEGER NOT NULL,
  updated_at   TEXT NOT NULL
);
CREATE INDEX idx_registry_safety_pending ON registry_safety_admission (state, since_seq);

  name       TEXT PRIMARY KEY,
  watermark  INTEGER NOT NULL,
  updated_at TEXT NOT NULL
);

-- (constraint violation on CREATE UNIQUE INDEX) rather than silently letting

-- 0103: THE DAY A ROW WAS OBSERVED COMMITTED, WHICH IS THE ONLY DAY A CALENDAR
-- CUTOFF MAY SAFELY USE (QUEUE #515, Jason D-0653 — "stamp it the moment it
-- commits"; W3-11-R9 item I3-2180).
--
-- ---------------------------------------------------------------------------
-- WHAT THIS EXISTS FOR, DEMONSTRATED BY EXECUTION
-- ---------------------------------------------------------------------------
-- `requests.ts archiveDemand` serves each row AS IT STOOD AT THE START OF TODAY,
-- UTC (0074, policy 1). It decides that from a timestamp the WRITER took, and
-- until this migration the writer took it on the NEAR side of an `await`:
-- `const stamp = now()` (requests.ts:697) bound into an UPDATE that is not
-- awaited until :726, and `now()` evaluated inside `.bind(...)` on the INSERT
-- (:732) awaited at :733.
--
-- Hold that UPDATE at the database seam across midnight and the reader answers
-- twice inside one UTC day, differently:
--     00:00:00.000  ->  { "the 2019 convention set please", 2026-09-08 }
--     00:00:00.100  ->  { "actually the 2020 acoustic session", 2026-09-10 }
-- Same UTC day (2026-09-11), 100 ms apart, and the second answer is text whose
-- row version DID NOT EXIST when that day began — published under a day that is
-- STRICTLY BEFORE the day it became visible.
--
-- A clock read taken BEFORE an `await` is a LOWER bound on the commit. A
-- calendar cutoff needs an UPPER one. That is the whole defect, and it is not a
-- property of any single token — it is a property of WHERE the clock is read.
--
-- ---------------------------------------------------------------------------
-- THE TWO REMEDIES JASON REFUSED, AND WHY THE COLUMNS ARE SHAPED AS THEY ARE
-- ---------------------------------------------------------------------------
-- A DB-SIDE STAMP (`strftime('%Y-%m-%dT%H:%M:%fZ','now')` in the statement) is
-- the obvious answer and it is refused: it removes the injectable `now()` that
-- every test in this module drives and that the day-lag correctness proof rests
-- on. So NOTHING BELOW IS A DB-SIDE CLOCK. Every value in every column here is
-- bound by the caller from the injected `now()`, and a test can still drive the
-- whole boundary by injecting instants — which is what
-- `test/w312-b9-archive-commit-stamp.test.ts` does.
--
-- A STATED GRACE WINDOW on the reader's cutoff was also refused: it leaves the
-- behaviour in place and makes the product explain an oddity forever. So there
-- is no constant here, no slack, no "plus one hour". The reader's predicate
-- stays "strictly before the start of today, UTC" — only the COLUMN it reads
-- changes.
--
-- ---------------------------------------------------------------------------
-- THE PROTOCOL THESE FOUR COLUMNS ARE
-- ---------------------------------------------------------------------------
-- The only upper bound a client can obtain on a commit instant is a clock read
-- taken AFTER that write's `await` has resolved. So a change to a SERVED value
-- happens in two statements:
--
--   PHASE 1 — INVISIBLE. The committing statement writes the fan's new words to
--     `stage_note`/`stage_at` and touches nothing any reader reads. It may
--     commit at any instant, before or after any midnight, and NO served answer
--     moves. This is what makes the two reads above equal.
--   PHASE 2 — PUBLISHED, AND CLAIMED. Once phase 1's `await` has RESOLVED, the
--     caller reads the clock again; that instant is at-or-after phase 1's
--     commit. Phase 2 installs the staged words, rotates the retained pair and
--     stamps `note_settled_day` with that observed day, and its gate —
--     `AND stage_at = ?` — is INSIDE the committing statement. It claims the
--     exact version this call staged: a concurrent second edit that staged a
--     different instant matches zero rows and its phase 2 is a no-op, so the
--     two cannot both publish and neither can publish the other's text.
--     (A re-read before a write narrows a race; only a claim closes it.)
--
-- The INSERT path needs no staging because UNSETTLED is already invisible to
-- every calendar-cutoff reader: the row is inserted with `settled_day` =
-- `'9999-12-31'`, counted by nothing, and claimed by the same observed-after
-- statement.
--
-- ---------------------------------------------------------------------------
-- THE SENTINEL, AND WHY NULL COULD NOT CARRY BOTH MEANINGS
-- ---------------------------------------------------------------------------
-- `'9999-12-31'` means UNSETTLED. It is a ten-character UTC day like every other
-- value in these columns, and it sorts after every real one, so EVERY existing
-- `< <today>` predicate excludes it without being rewritten to know about it.
--
-- NULL means something different and irreplaceable: THIS ROW PREDATES 0103.
-- Every reader COALESCEs NULL back to the exact expression it used before this
-- migration — `substr(created_at,1,10)` for membership,
-- `substr(COALESCE(note_updated_at, created_at),1,10)` for the note — so a row
-- that existed before this migration causes EXACTLY the answer it caused
-- yesterday. No creator-facing figure moves retrospectively, which is why there
-- is no backfill: a backfill would have to invent an observation nobody made,
-- and the only honest value for a row nobody observed committing is "I do not
-- know, use what you used before". This is 0074's policy 2 in a second place.
--
-- NO DEFAULT CLAUSE, deliberately, and this is the load-bearing detail of the
-- four ALTERs below: SQLite's `ADD COLUMN ... DEFAULT x` fills every EXISTING
-- row with `x`, which would destroy the NULL that carries "predates 0103" and
-- would silently re-date every archive request in the product.
--
-- ---------------------------------------------------------------------------
-- WHAT IS NOT HERE
-- ---------------------------------------------------------------------------
-- NO INDEX. The membership count and the notes read are both already scoped to
-- one `passport_id` through `idx_areq_passport` and the notes read is capped at
-- 100 rows; the cron settle below is a single unfiltered UPDATE that visits the
-- table once either way. An index on a settle DAY would additionally be a
-- structure ordering fans by when they acted, which is the primitive 0074 and
-- 0075 exist to remove.
--
-- NO INSTANT ON THE SETTLE COLUMNS. `settled_day` and `note_settled_day` are
-- ten-character UTC days, not timestamps, for 0075's reason: a second clock at
-- second granularity attached to an act one named person performs is the
-- differencing primitive itself. `stage_at` IS an instant, but it is a CLAIM KEY
-- that exists only between phase 1 and phase 2 and is set back to NULL by
-- phase 2 — it is never served, never read by any cutoff, and never persists.
--
-- NO TABLE REBUILD. Four `ALTER TABLE ... ADD COLUMN`s with no DEFAULT are
-- O(1) catalogue edits in SQLite: no row is read, rewritten or destroyed. The
-- 12-step rebuild (create/copy/drop/rename) is what a migration in this tree
-- must not do without an enclosing transaction, and this migration does not
-- come near it.
--
-- ---------------------------------------------------------------------------
-- THE SUBREQUEST BUDGET (~1,000 D1 subrequests per cron tick, ~40 legs)
-- ---------------------------------------------------------------------------
-- This migration is DDL applied by the deploy, not by the cron, so it spends
-- none of that budget. The liveness path it enables — `requests.ts
-- settleArchivePublications`, which rescues rows stranded by a crash between
-- phase 1 and phase 2 — is exactly TWO set-based UPDATEs with no per-row loop:
-- 2 subrequests per tick whatever the row count, and it cannot grow with the
-- table. There is no backfill leg, because there is nothing to backfill.

ALTER TABLE archive_requests ADD COLUMN settled_day TEXT NULL;
ALTER TABLE archive_requests ADD COLUMN note_settled_day TEXT NULL;
ALTER TABLE archive_requests ADD COLUMN stage_note TEXT NULL;
ALTER TABLE archive_requests ADD COLUMN stage_at TEXT NULL;

SELECT account_id, at, sig FROM (
  SELECT account_id, at, sig, ROW_NUMBER() OVER (PARTITION BY account_id ORDER BY at ASC, sig ASC) AS rn FROM (
    SELECT account_id, created_at  AS at, 'oauth_email' AS sig FROM oauth_identities WHERE email_verified = 1

     WHERE EXISTS (SELECT 1 FROM accounts x WHERE x.account_id = COALESCE(a.redeemed_by, a.account_id))))

-- I3-2167 (I4-16-R9, D-0840(5); landed by I4-17-R9). ADDITIVE: one table + an idempotent backfill.
-- The live agency grant per (creator passport, agency), projected from the ledger so the analytics
-- sweep (analytics-build.ts agencyGrantedCreators) reads a bounded table instead of folding every
-- agency_authorized / agency_revoked event on every build. scope is NULL once revoked. updated_seq is
-- the trust_events.seq of the event the row reflects; a writer only moves a row FORWARD
-- (ON CONFLICT ... WHERE excluded.updated_seq > agency_report_grants.updated_seq).
CREATE TABLE agency_report_grants (
  passport_id TEXT NOT NULL,
  agency_id   TEXT NOT NULL,
  scope       TEXT,
  updated_seq INTEGER NOT NULL,
  PRIMARY KEY (passport_id, agency_id)
);
-- Backfill: the latest grant event per (subject, payload.agency_id), by seq — the same "later event
-- wins" fold analytics-build.ts and agencies.ts run. A payload that is not valid JSON, or names no
-- agency_id, identifies no pair and is skipped, as the fold skips it. Safe to re-run on its own (the
-- DEPLOY.md row says when): it only ever moves a row forward.
INSERT INTO agency_report_grants (passport_id, agency_id, scope, updated_seq)
SELECT e.subject,
       json_extract(e.payload, '$.agency_id'),
       CASE WHEN e.event_type = 'agency_authorized' THEN json_extract(e.payload, '$.scope') END,
       e.seq
  FROM trust_events e
 WHERE e.event_type IN ('agency_authorized', 'agency_revoked')
   AND json_valid(e.payload)
   AND json_extract(e.payload, '$.agency_id') IS NOT NULL
   AND e.seq = (SELECT MAX(l.seq) FROM trust_events l
                 WHERE l.event_type IN ('agency_authorized', 'agency_revoked')
                   AND l.subject = e.subject
                   AND json_valid(l.payload)
                   AND json_extract(l.payload, '$.agency_id') = json_extract(e.payload, '$.agency_id'))
ON CONFLICT (passport_id, agency_id) DO UPDATE SET scope = excluded.scope, updated_seq = excluded.updated_seq
 WHERE excluded.updated_seq > agency_report_grants.updated_seq;

-- 0135 created release_status_history with no backfill, so a release published or withdrawn before the
-- writers shipped has no history row and the as-of reader (back-catalog.ts, COALESCE(..., 'registered')) reads
-- it as 'registered'. One row per such release, carrying its CURRENT status at its created_at: exactly what
