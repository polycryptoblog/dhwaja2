// close-run.mjs — ADR-0009 D4 / QUEUE #31(3): assemble a run's closure skeleton
// mechanically from the judge's FINDINGS.md, without reading anything the judge
// did not write down. It parses exactly two sections of the judge report:
//   - "## 1." the ADJUDICATION table (id | verdict | sev | reach | detect | conf)
//   - "## 5. DISPATCH" the FIX-IN-SWEEP / QUEUE / NEEDS_REPRODUCTION / STATED
//     PROPERTY bullet buckets (matched back to table ids by first-letter +
//     trailing-number, since DISPATCH drops the room-name middle segment).
// It does NOT read "## 2. MERGES" (so it never merges two ids into one row the
// way a human registry does), does not read the STATUS section's "New this
// room:" line (so judge-authored extra findings like JUDGE-1 are invisible to
// it), and cannot know about a later room's findings (like an auditor's
// AUDIT-1) at all. Those are exactly the differences the SCOPE.md fixture
// task asks this script to report against the 22 hand-written REGISTRY.md
// rows for RUN-0013 — see SCRIPTS-B/REPORT.md.
//
// Usage:
//   node scripts/close-run.mjs <run-dir> --judge <FINDINGS.md> --out <dir> [--template <instance.json>]
//
// <run-dir> is the run's org/07-findings/<run> directory (used only for its
// basename, to derive the run id and run number — e.g. RUN-0013-sweep-s004 ->
// run number 0013). --template defaults to
// org/06-runs/instances/RUN-0013-sweep-s004.json relative to --root (or the
// nearest ancestor of <run-dir> two levels up if --root is not given), which
// is the RUN.schema 0.6.0 shape this script clones.
//
// Emits, under --out:
//   instance.skeleton.json   — RUN.schema 0.6.0 shape, findings[]/waves[]/
//                              discrepancies[] rebuilt, closure fields reset

// ---- 0c. DELIVERY-GAP gate (S-094-R3; org/rooms/teachback.md rule 4) ----------
// The teachback ALREADY computes this. W3-09a's measured FAMILY-014 delivered to 1 of
// 25 rooms while three rooms that never received it instantiated it, at 113 room-turns
// (W3-09a-batch/TEACHBACK.md:52-56) -- and then wrote it in prose, where nothing read
// it. Every matcher anyone writes will be wrong somehow; the question this gate answers
// is whether being wrong is VISIBLE WITHIN ONE BATCH, with a count, at close.
//
// OUTCOME-SHAPED, deliberately. A FLAG-shaped check is satisfied by a teachback that
// reports zero; a COUNT-shaped check is satisfied by a teachback that files nothing.
// So this reads the ATTRIBUTED OCCURRENCES and the PACKET ROWS and computes the
// difference itself, and an absent record on a batch that could carry one BLOCKS.
//
// What blocks is not a gap -- a gap is the expected steady state -- but a gap the
// teachback did not ACCOUNT FOR. Naming why a family did not reach a room is already
// rule 4's obligation in prose; this makes it discharge-able and counts what is left.
//
// VINTAGE, and it is derived from the batch's own artifacts rather than from anything
// the teachback declares about itself: a batch is MODERN when at least one packet row
// whose `room` is not `teachback` carries a `families` key. Measured: the only row
// with `families` in W3-08-batch/_packets.jsonl is its own TEACHBACK dir (the
// teachback ran later, under the newer compiler, on an older batch), so keying on "any
// row" would block a historical batch -- and a gate that blocks every historical batch
// is a gate someone passes `--force` to, which is how a gate dies.
// RJ550b-09b-TB-S094-16 (I16-W03J): this was `join(ROOT, ...)`, and ROOT is the SCRIPT'S tree (root.mjs), so a
// close run from the build tree appended to the build tree's copy and one run from the repo to the repo's: two
// ledgers that drifted apart (repo 9925 B, build 23702 B at i16-base). The ledger is the BATCH'S record, so it
// lives in the tree that holds the batch (queueOfBatch, the USAGE.jsonl rule at gate 0g), unless --gap-ledger names it.
const GAP_LEDGER_REL = "org/08-control/DELIVERY-GAP.jsonl";
function gapLedgerFor(batchDir) {
  const explicit = opt("gap-ledger");
  if (explicit) return resolve(explicit);
  const q = queueOfBatch(batchDir);
  // I43-ORGTOOLS (I42 INCIDENTS 5): a QUEUE.md tree found ABOVE an explicit --root is the tree the fixture root sits in, not
  // the batch's: close-run-teachback-gate.test.ts run with TMPDIR inside the build tree appended 116 fixture rows to its ledger.
  const r = opt("root") ? realPathOf(resolve(opt("root"))) : null;
  if (q && r?.startsWith(`${realPathOf(q.tree)}/`)) return join(r, GAP_LEDGER_REL);
  return join(q ? q.tree : ROOT, GAP_LEDGER_REL);
}

/** QUEUE #475(a). The full record of a deferred packet row lives at its `withheld_to` (resolved in
 *  the batch's own tree), matched on `dir` + `at` because compile-room-prompt.mjs writes both lines
 *  from one record. If it does not resolve, the room's PROMPT.md `### FAMILY-NNN` bodies -- what it
 *  actually read -- stand in, and the caller records which source it used. */
function withheldFamilies(batchDir, p) {
  const q = queueOfBatch(batchDir);
  const stash = typeof p.withheld_to === "string" ? join(q ? q.tree : ROOT, p.withheld_to) : null;
  if (stash && existsSync(stash)) {
    for (const l of readFileSync(stash, "utf8").split("\n")) {
      let s = null;
      try { s = l.trim() ? JSON.parse(l) : null; } catch { s = null; }
      if (s && s.dir === p.dir && s.at === p.at && (p.run === undefined || s.run === p.run) && Array.isArray(s.families)) return { families: s.families, from: p.withheld_to };
    }
  }
  const pp = join(batchDir, p.dir, "PROMPT.md");
  // I34-TOOLS-A (I33 B4 limb 2): the shared delivery predicate, so a NOT-delivered shadow heading is not a delivery.
  if (existsSync(pp)) return { families: deliveredFamilyIds(readFileSync(pp, "utf8")), from: "PROMPT.md bodies (withheld_to did not resolve)" };
  return { families: null, why: `withheld_to ${p.withheld_to ?? "(none)"} holds no record for this dir/at and no PROMPT.md is on disk` };
}

      const gap = row.instantiated_not_delivered + row.uncompiled_gap;

      if (row.undeterminable > 0) msgs.push(`     ${row.undeterminable} pair(s) undeterminable: the room has a PROMPT.md but its packet row predates the \`families\` field, so what it received is unknown. Not a gap; not decided either.`);
    }
  }

  // S-094-FX2 defeat 3, the printing half. These three were RECORDED to the ledger and never
  // printed, so the one artifact that names the true cause of a claimed_not_emitted block was

  }
  if (row.prompts_stale.length) {
    msgs.push(`     prompts NEWER than _packets.jsonl (recompiled after the row was claimed; the intersection is reading a later artifact than the room did): ${row.prompts_stale.join(", ")}`);
  }
  if (row.prompts_absent > 0) {
    msgs.push(`     PROMPT.md absent for ${row.prompts_absent} packet row(s): those claims stand unverified (a pruned dir does not invent a gap — and deleting a PROMPT.md therefore clears a claimed_not_emitted block; see S-094-FX2 QUEUED)`);
  }

  if (row.packet_probe_rows > 0) msgs.push(`NOTE: ${row.packet_probe_rows} probe packet row(s) (dir starting "_") excluded — QUEUE #474b`);

  // The CLOSE is the ledger's event; the preflight door (--delivery-gap-only) computes
  // the same figure from the same code and does not append a second row for it.
  const ledger = gapLedgerFor(batchDir);
  if (recordToLedger) {
    try {
      mkdirSync(dirname(ledger), { recursive: true });
      appendFileSync(ledger, JSON.stringify(row) + "\n");
      msgs.push(`     figure recorded: ${ledger}`);
    } catch (e) {
      msgs.push(`WARNING: could not append the delivery-gap figure to ${ledger}: ${e.message}`);
    }
  } else msgs.push(`     figure not recorded (preflight door); the close records it to: ${ledger}`);
  return { blocked, msgs, row };
}

// At batch (or run) close, the note is the org's own account of what

// wrote the note — reads the note plus every VERDICTS.md/REPORT.md under

// anything else until that file exists, names the exact note/commit-message
// file it audited with a matching sha256 (so a note edited after the audit
// is caught as stale, not silently accepted), and carries
// VERDICT: SUPPORTED — VERDICT: UNSUPPORTED blocks the close outright.
// Runs unconditionally, before --judge/--out are required, so it also

// [single-hasher] (docs/INVARIANTS.md, tests/invariants/single-hasher.test.ts): SHA-256 may only
// be computed inside packages/canonical; a repo script hashing a FILE for integrity (never a
// subject/URL) shells out to a system sha256 hasher instead of node:crypto, same shape as
// scripts/intake/w2-sync.mjs's own exemption reasoning, without needing one — no JS hasher is
// called at all.
// [REG-04] (S-080 FIX5): GNU `sha256sum` is absent on stock macOS. Try it first, fall back to
// `shasum -a 256` (BSD/macOS default) on ENOENT, and fail naming both if neither is on PATH —
// still no node:crypto import (single-hasher).
function hashFile(file) {
  try {
    return execFileSync("sha256sum", [file], { encoding: "utf8" }).trim().split(/\s+/)[0];
  } catch (e) {
    if (e.code !== "ENOENT") throw e;
    try {
      return execFileSync("shasum", ["-a", "256", file], { encoding: "utf8" }).trim().split(/\s+/)[0];
    } catch (e2) {
      if (e2.code !== "ENOENT") throw e2;
      console.error(`refused: close blocked — no sha256 hasher on PATH (tried "sha256sum" and "shasum -a 256")`);
      process.exit(1);
    }
  }
}

// ---- 0a. THE HR-1 DISCHARGE (W3-11a-R3; ORCHESTRATOR.md HR-1, Jason 2026-09-20) ----
// HR-1: an auditor gets ONE re-verdict; the orchestrator fixes everything it found, sends
// it back once, and ANYTHING STILL STANDING BECOMES A QUEUE ROW AND THE BATCH CLOSES. The
// rule was written because a note-auditor was sent back FOUR times in one batch, over a
// million tokens on one room re-issuing a verdict, three of the four round trips buying
// nothing. Until this block existed, `UNSUPPORTED` here exited 1 unconditionally, so the
// discharge HR-1 grants did not exist for the one auditor this gate binds on — and the
// only way to close was to make the auditor change its word, which is the pressure HR-1
// was written to remove.
//
// WHAT IS AND IS NOT BEING RELAXED. An UNSUPPORTED that NOBODY HAS ANSWERED still blocks,
// exactly as before. What closes is an UNSUPPORTED the orchestrator has ANSWERED: one
// re-verdict round, and one queue row per surviving claim, each row resolving on disk.
// This is the character the rest of this gate already has — A GAP IS DISCHARGED BY BEING
// NAMED, NEVER BY BEING ZERO (see the TEACHBACK-GAP-CAUSE branches above) — applied to the
// one verdict that had no discharge at all.
//
// WHY IT CANNOT BE SATISFIED BY AN EMPTY GESTURE. Four limbs, every one of which costs the
// writer something it could not produce without reading the audit:
//   L1 DISCHARGES-SHA256 must equal the sha256 of THIS NOTE-AUDIT.md. A discharge is bound
//      to the exact bytes it answers; re-issue the verdict and the hash moves and the
//      discharge stops covering it. (Same staleness discipline as AUDITED-SHA256 above,
//      and the same hasher, so no second implementation can drift from the first.)
//   L2 RE-VERDICT-OF-SHA256 must be present and DIFFERENT from L1. HR-1 grants the
//      discharge after the orchestrator "sends it back ONCE" — so a FIRST-PASS UNSUPPORTED,
//      which nobody has yet answered, has no distinct earlier audit to cite and CANNOT be
//      discharged. This limb is what bounds the round trip instead of abolishing it: it
//      buys the one re-verdict HR-1 promises the auditor and refuses the zero-verdict
//      bypass that would otherwise be the cheapest path to a green close.
//   L3 One QUEUE-ROW line per surviving claim, in bijection with the `Unsupported claims:`
//      bullets DERIVED FROM THE AUDIT ITSELF (never a hand list — meta-guard §1), each
//      carrying a >=16-character VERBATIM substring of the claim it covers, matching
//      exactly one bullet, and each bullet matched exactly once. Boilerplate repeated N
//      times matches one bullet N times and is refused; a discharge that covers 3 of 4
//      claims names the fourth and is refused.
//   L4 Every queue id must RESOLVE to a real row in org/01-current/QUEUE.md, matched with
//      scripts/check-queue.mjs:127's own row shape. "Becomes a QUEUE row" is a fact on
//      disk or it is a sentence. If QUEUE.md cannot be read, L4 cannot be evaluated and
//      the close is REFUSED — an unverifiable queue claim must never discharge.
// And the door underneath all four: an UNSUPPORTED whose audit carries ZERO parsable
// claim bullets BLOCKS, discharge or no discharge. Without that, L3's population is empty
// and its bijection passes over nothing — `for (const x of delivered)` with nothing
// delivered, the shape this org has already shipped twice and found by execution both
// times. The empty case is the one a rubber stamp would reach for, so it is the one case
// no discharge can reach.
//
// W3-11a-FX2A, AFTER THE REGRESSION ATTACKER GOT PAST FOUR OF THE FIVE. Every limb above
// asserted a SHAPE where it meant a FACT. Each is now resolved, and none is looser:
//   L2 RE-VERDICT-OF-SHA256 was checked for 64 hex digits and inequality with L1 and
//      NOTHING ELSE, so `0000...0001` bought "re-verdict round confirmed". It now needs a
//      `RE-VERDICT-OF-FILE:` line, and that file must exist inside the BATCH'S OWN RUN
//      DIRECTORY (W3-A/JW2H-01: this was bounded against `qloc.tree`, which queueOfBatch

/** The surviving claims, DERIVED FROM THE AUDIT DOCUMENT (W3-11a-FX2A; BREAK-3, BREAK-7).
 *
 *  The old implementation matched the literal line `^Unsupported claims:$` and stopped at
 *  the first line after it that was not a bullet, a blank, or an indented continuation.
 *  Both halves were measured wrong against the six real NOTE-AUDIT.md in this tree: W3-09a
 *  carries NINE unsupported bullets under TWO headings and parsed as SIX, and W3-03/W3-06
 *  carry no such literal line at all and parsed as ZERO while all six read UNSUPPORTED. An
 *  under-derived population is worse than no population, because the bijection below is
 *  EXACT OVER IT and therefore reads as rigour.
 *
 *  Two forms, and which applies is READ OFF THE DOCUMENT, never assumed:
 *    CLAIMS   the audit is a room report carrying a `## CLAIMS` fenced json array (the
 *             shared output contract). The population is the rows the auditor marked
 *             `"state": "broken"`. A parsable array with ZERO broken rows under a verdict
 *             of UNSUPPORTED contradicts itself and REFUSES.
 *    bullets  org/rooms/note-auditor.md's list. EVERY top-level bullet in the document is
 *             attached to the last non-blank non-bullet line above it, and that heading is
 *             CLASSIFIED. A heading this gate cannot classify REFUSES, naming it.
 *  So every bullet in the document is either COUNTED or REFUSED; there is no third path
 *  along which one silently disappears. */
const CLAIM_BULLET = /^[ \t]{0,3}[-*][ \t]+\S/;
const CLAIM_CONT = /^[ \t]+\S/;

/** surviving | supporting | unknown, from the heading's own words. The order is
 *  load-bearing: "Unsupported claims:" contains "supported". */
function classifyClaimHeading(h) {
  const t = String(h == null ? "" : h).replace(/^[#>\s*_]+/, "").toLowerCase();
  if (!t) return "unknown";
  if (/unsupported|not true|not supported|underivable|not derivable|contradict/.test(t)) return "surviving";
  if (/\bsupported\b|\bmatches\b|\bmatching\b|do(es)? derive|re-derived/.test(t)) return "supporting";
  return "unknown";
}

/** Every bullet run in the document, with the heading it sits under. */
function bulletSections(text) {
  const lines = text.split("\n");
  const sections = [];
  let heading = null, headingLine = 0, cur = null, inBullet = false, fenced = false;
  for (let i = 0; i < lines.length; i++) {
    const line = lines[i];
    if (/^\s*```/.test(line)) { fenced = !fenced; cur = null; inBullet = false; continue; }
    if (fenced) continue;
    if (CLAIM_BULLET.test(line)) {
      if (!cur) { cur = { heading, headingLine, claims: [] }; sections.push(cur); }
      cur.claims.push(line.trim().replace(/^[-*][ \t]+/, ""));
      inBullet = true;
      continue;
    }
    if (line.trim() === "") continue;
    if (inBullet && CLAIM_CONT.test(line) && cur && cur.claims.length) {
      cur.claims[cur.claims.length - 1] += " " + line.trim();
      continue;
    }
    heading = line.trim(); headingLine = i + 1; cur = null; inBullet = false;
  }
  return sections;
}

function survivingClaims(text) {
  const cm = /^##[ \t]+CLAIMS[ \t]*$/m.exec(text);
  if (cm) {
    const fence = /```(?:json)?[ \t]*\n([\s\S]*?)\n[ \t]*```/.exec(text.slice(cm.index + cm[0].length));
    let parsed = null;
    if (fence) { try { parsed = JSON.parse(fence[1]); } catch { parsed = null; } }
    if (Array.isArray(parsed) && parsed.length && parsed.every((r) => r && typeof r === "object" && "state" in r)) {
      const broken = parsed.filter((r) => String(r.state).toLowerCase() === "broken");
      if (broken.length === 0) {
        return { form: "CLAIMS", claims: [], sections: [], problem:
          `${notePath} says VERDICT: UNSUPPORTED and its "## CLAIMS" array parses (${parsed.length} row(s)), but NOT ONE row carries "state": "broken". The document contradicts itself — the verdict says something did not survive the audit and the claims table says everything holds — so there is no derivable claim population here. That is an audit to re-issue, not a population to discharge.` };
      }
      return { form: "CLAIMS", claims: broken.map((r) => `${r.id == null ? "(no id)" : r.id} — ${r.mechanism == null ? (r.expected == null ? "" : r.expected) : r.mechanism}`.trim()),
        sections: [`## CLAIMS, ${broken.length} of ${parsed.length} row(s) with "state": "broken"`], problem: null };
    }
  }
  const all = bulletSections(text).filter((s) => s.claims.length);
  const unknown = all.filter((s) => classifyClaimHeading(s.heading) === "unknown");
  if (unknown.length) {
    return { form: "bullets", claims: [], sections: [], problem:
      `${notePath} carries ${unknown.reduce((n, s) => n + s.claims.length, 0)} bullet(s) under ${unknown.length} heading(s) this gate cannot classify as surviving or as supported, so the surviving-claim population is NOT DERIVABLE:\n` +
      unknown.map((s) => `    line ${s.headingLine}: ${s.heading === null ? "(no heading above the bullets)" : String(s.heading).slice(0, 120)} — ${s.claims.length} bullet(s)`).join("\n") +
      `\n  Discharging only what the gate CAN read would be a bijection over the WRONG POPULATION, which is worse than no bijection because it reads as rigour. Head each bullet run with a line that says what the bullets are ("Unsupported claims:", "... NOT true of the code ...", "Supported (re-derived and matching):"). THE FIX IS THE AUDITOR'S, NOT THE ORCHESTRATOR'S: send it back for HR-1's one re-verdict. Nobody but the auditor writes ${basename(notePath)}.` };
  }
  const surviving = all.filter((s) => classifyClaimHeading(s.heading) === "surviving");
  return { form: "bullets", claims: surviving.flatMap((s) => s.claims),
    sections: surviving.map((s) => `"${String(s.heading).slice(0, 60)}" (line ${s.headingLine}, ${s.claims.length} bullet(s))`), problem: null };
}

/** The QUEUE the BATCH'S OWN TREE carries, located by walking up from the RUN DIRECTORY
 *  for the very artifact this limb is about (W3-11a-FX2A, BREAK-4). ROOT is anchored on
 *  `import.meta.url` (scripts/lib/root.mjs) while <run-dir> is a cwd-relative positional,
 *  and NOTHING asserted the two were one tree — so "became a QUEUE row" was answered
 *  against whichever org/ the SCRIPT'S tree carried. For a room that is the stale build
 *  checkout: a row genuinely present in the batch's own queue was REFUSED, and a row
 *  present only in the script's tree DISCHARGED. The old root defect FOLLOWED the run
 *  dir; this one IGNORED it. */
function queueOfBatch(runDirAbs) {
  let dir = resolve(runDirAbs);
  for (;;) {
    const qp = join(dir, "org/01-current/QUEUE.md");
    if (existsSync(qp)) return { tree: dir, path: qp };
    const up = dirname(dir);
    if (up === dir) return null;
    dir = up;
  }
}

/** id -> the ROW ITSELF, not just the id. Same row predicate as
 *  scripts/check-queue.mjs:127 — one predicate, not two spellings of one. `count` is how
 *  many row lines carry that id: "the QUEUE row" for an id is not always one row, and a
 *  limb that resolves an id says so rather than silently picking the first.
 *
 *  THE FIGURE THAT WAS HERE WAS A MOVING ONE, AND IT MOVED (W3-A, 2026-09-21). This docblock
 *  read "429 row lines and 426 distinct ids", measured when #410/#411/#412 were each written
 *  twice. QUEUE #448 was repaired the same day (R06 renumbered them to #450/#451/#452) and
 *  org/01-current/QUEUE.md now measures 452 row lines and 452 distinct ids -- no duplicates
 *  at all. The count is therefore no longer stated here: a board figure pinned in a comment
 *  is false by the next batch, and the caller below must not depend on the two numbers being
 *  equal TODAY. They are equal today and the multiplicity limb still has to hold. */
function queueRows(qp) {
  const rows = new Map();
  const lines = readFileSync(qp, "utf8").split("\n");
  for (let i = 0; i < lines.length; i++) {
    const m = /^\|\s*(\d+)\s*\|/.exec(lines[i]);
    if (!m) continue;
    const prev = rows.get(m[1]);
    if (prev) { prev.count += 1; prev.at.push(i + 1); continue; }
    const cells = lines[i].split("|");
    rows.set(m[1], { id: m[1], line: lines[i], status: (cells[2] === undefined ? "" : cells[2]).trim(), count: 1, at: [i + 1] });
  }
  return rows;
}

/** A queue row is LIVE when something still stands on it. Measured over this tree's

/** Two paths are the same tree when they are the same INODE path; /tmp is a symlink on
 *  macOS and a plain directory on Linux, and this gate runs on both. */
function realPathOf(p) { try { return realpathSync(p); } catch { return resolve(p); } }

// ---- 0d. REGRESSION RECONCILE gate (ORG-02-R4; QUEUE #501) ----------------
/**
 * THE PROPERTY: a row a batch leaves `open`, whose `file` a SIBLING ROOM IN THE SAME
 * BATCH reports itself having `fixed`, must carry a reconciliation record naming that
 * sibling and a disposition with a citation — or the batch does not close.
 *
 * WHY A CLOSURE STEP AND NOT A DUTY ON THE FIX ROOM. The attacker's verdicts live in the
 * ATTACKER's FINDINGS.jsonl, which is outside every fix room's ownership list; a fix room
 * writing it is the cross-room write B2 forbids and S-026/DISCREPANCY-0049 measured. The
 * fix room also usually cannot answer the question — it was briefed on a FILE, not on the
 * rows that file's defect produced. Closure is the only point where both populations are
 * readable at once, so the duty goes here.
 *
 * WHAT IT DOES NOT DO: it never writes a verdict and never touches a FINDINGS.jsonl. A
 * regression attacker does not fix, so `open` is the CORRECT verdict for it and it stays.
 * This gate refuses a CLOSE until a human-authored record exists. It moves no authority;
 * it removes the option of closing silently with nobody having exercised it.
 *
 * Both populations are DERIVED by walking the batch directory (FAMILY-004(a) COVERAGE:
 * from the source, never a hand list). No finding id is named in this file.
 */
const RECONCILE_BASENAME = "REGRESSION-RECONCILE.md";

/** The seven words org/rooms/_OUTPUT-CONTRACT.md fixes for a FINDINGS.jsonl `status`. A
 *  disposition here is one of THOSE: a status outside them is a contract violation against
 *  that rule, not a new word, so this gate mints no eighth. */
const FINDINGS_STATUS = ["fixed", "defended", "not_a_defect", "wrong_owner", "not_reproduced", "not_located", "open"];

/** A citation is a backticked command or a `path:line`. DELIBERATELY NOT A LENGTH FLOOR:
 *  a `why.length > N` floor is satisfied by one sentence stamped across every entry, and
 *  that is the measured defect of one of the very rows this gate exists to reconcile. The
 *  crediting-room check below is the other half — stamped prose cannot name the room the
 *  join computed. */

/** Every FINDINGS.jsonl row in the batch, with the room directory it came from. An
 *  unparsable line is COUNTED AND NAMED, never skipped — a population that silently drops
 *  what it could not read is the under-count this instrument has already paid for once. */
function batchRoomFindings(runDirAbs) {
  const rows = [], bad = [];
  let dirs = [];
  try { dirs = readdirSync(runDirAbs, { withFileTypes: true }).filter((e) => e.isDirectory()).map((e) => e.name).sort(); }
  catch { return { rows, bad, rooms: [] }; }
  for (const room of dirs) {
    const fp = join(runDirAbs, room, "FINDINGS.jsonl");
    if (!existsSync(fp)) continue;
    const lines = readFileSync(fp, "utf8").split("\n");
    for (let i = 0; i < lines.length; i++) {
      if (!lines[i].trim()) continue;
      let o = null;
      try { o = JSON.parse(lines[i]); } catch { o = null; }
      if (!o || typeof o !== "object" || o.id == null) { bad.push(`${room}/FINDINGS.jsonl:${i + 1}`); continue; }
      rows.push({ room, line: i + 1, id: String(o.id),
        file: o.file == null ? "" : String(o.file).trim(),
        status: String(o.status == null ? "" : o.status).trim().toLowerCase(),
        addressee: o.addressee, mechanism: o.mechanism, iid: o.iid });
    }
  }
  return { rows, bad, rooms: dirs };
}

/** The join. A row reconciles against a SIBLING's fix on the same file; a room's own fix
 *  of its own row is not a reconciliation, because nobody independent looked. */
function reconcileOverlap(rows, rulingRooms = []) {
  const open = rows.filter((r) => r.status === "open" && r.file);
  // I7-CLOSE-TOOLS (#638): a RULING room (judge, judge-audit, note-auditor, invariant-auditor, teachback) rules on
  // the batch's rows and fixes nothing; a `fixed` row it writes is a verdict, not a fix, so it never credits a
  // reconcile record. Same predicate as gate 0e's population cut (RULING_TYPES via coverageRoomType).
  const fixed = rows.filter((r) => r.status === "fixed" && r.file && !rulingRooms.includes(r.room));
  const overlap = [];
  for (const o of open) {
    const by = [...new Set(fixed.filter((f) => f.file === o.file && f.room !== o.room).map((f) => f.room))].sort();
    if (by.length) overlap.push({ room: o.room, id: o.id, file: o.file, by });
  }
  return { open, fixed, overlap };
}

/** null when the file is absent; otherwise its table rows. A cell containing a literal
 *  `|` splits early and the row fails the checks below rather than passing silently.
 *  The row cell is a bare `ID` or `ROOM/ID` (#632): split at the FIRST `/`, so `room` is
 *  null for a bare id. `ROOM/ID` exists because two rooms can file one id (a judge repeating
 *  a builder's), each occurrence has its own computed fixer set, and a bare line matches both. */
function reconcileRecords(p) {
  if (!existsSync(p)) return null;
  const recs = [];
  for (const line of readFileSync(p, "utf8").split("\n")) {
    if (!/^\s*\|/.test(line)) continue;
    const cells = line.split("|").slice(1, -1).map((c) => c.trim());
    if (cells.length < 4) continue;
    if (/^:?-+:?$/.test(cells[0]) || /^row$/i.test(cells[0])) continue;
    const key = cells[0].replace(/`/g, "");
    const slash = key.indexOf("/");
    recs.push({ key, room: slash < 0 ? null : key.slice(0, slash), id: slash < 0 ? key : key.slice(slash + 1), by: cells[1].replace(/`/g, ""), disposition: cells[2].toLowerCase(), evidence: cells[3] });
  }
  return recs;
}

/** The record for one open occurrence: an exact `ROOM/ID` line first, else a bare-id line. */
function matchRecord(recs, o) {
  return recs.find((x) => x.room === o.room && x.id === o.id) ?? recs.find((x) => x.room === null && x.id === o.id);
}

{
  const runDirAbs = resolve(runDirArg);
  const rf = batchRoomFindings(runDirAbs);
  const rulingRooms0d = rf.rooms.filter((d) => RULING_TYPES.has(coverageRoomType(runDirAbs, d)));
  const { open, fixed, overlap } = reconcileOverlap(rf.rows, rulingRooms0d);
  const reconPath = join(runDirAbs, RECONCILE_BASENAME);
  const msgs = rf.bad.length
    ? [`NOTICE: ${rf.bad.length} unparsable FINDINGS.jsonl line(s) in ${basename(runDirAbs)}, counted in NEITHER population: ${rf.bad.join(", ")}`]

  if (!overlap.length) {
    // NON-VACUITY. A pass that prints nothing is indistinguishable from a gate that never

    for (const m of msgs) console.log(m);
    console.log(`OK: regression reconcile — ${open.length} open row(s) and ${fixed.length} fixed row(s) across ${rf.rooms.length} room dir(s) in ${basename(runDirAbs)}; no open row shares a file with a sibling's fix, so there is nothing to reconcile.`);
  } else {

    const problems = [];
    if (recs === null) { problems.push(`${reconPath} does not exist.`); recs = []; }

    for (const rec of recs) {
      if (rec.room === null) {

      } else if (!rf.rooms.includes(rec.room)) {
        problems.push(`the record for "${rec.key}" names room "${rec.room}", which is NOT a room directory of this batch — a ROOM/ID cell names the room that FILED the open row.`);
      } else if (!open.some((r) => r.room === rec.room && r.id === rec.id)) {
        problems.push(`the record for "${rec.key}" names an id that is NOT an open row of room ${rec.room} in this batch, so it describes a row that no longer exists in the form it was written for — re-derive it or delete it.`);
      }
    }
    for (const o of overlap) {

The crediting room is COMPUTED from the batch, never accepted from the record${rec.room === null ? `; if another room filed the same id with a different fixer, give each occurrence its own ROOM/ID line` : ""}.`);
      if (!FINDINGS_STATUS.includes(rec.disposition)) problems.push(`${o.room}/${o.id} — the record "${rec.key}" has disposition "${rec.disposition}" is not one of the seven FINDINGS status words (${FINDINGS_STATUS.join(", ")}).`);
      if (!RECON_CITATION.test(rec.evidence)) problems.push(`${o.room}/${o.id} — the evidence cell of the record "${rec.key}" carries no citation: it needs a \`command\` in backticks or a path:line.     }
    if (problems.length) {
      for (const m of msgs) console.error(m);
      console.error(`refused: close blocked — ${basename(runDirAbs)} leaves ${overlap.length} row(s) OPEN whose file a SIBLING ROOM IN THE SAME BATCH reports fixed. A regression attacker does not fix, so its "open" is the CORRECT verdict and MUST NOT be edited to clear this. What is missing is the reconciliation, and it belongs in ${reconPath}:`);
      for (const p of problems) console.error(`  - ${p}`);
      console.error(`  NEXT ACTOR: whoever closes this batch writes ${RECONCILE_BASENAME} — one "| row | fixed by | disposition | evidence |" line per row above, disposition one of ${FINDINGS_STATUS.join("|")}, evidence a backticked command or a path:line that was actually RUN.       process.exit(1);
    }
    for (const m of msgs) console.log(m);
    console.log(`RECONCILED: ${overlap.length} of ${open.length} open row(s) in ${basename(runDirAbs)} name a file a sibling fixed in the same batch, and each carries a record in ${RECONCILE_BASENAME}:`);
    for (const o of overlap) {
      const rec = matchRecord(recs, o);
      console.log(`  ${o.room}/${o.id} (${o.file}) — fixed by ${rec.by} — ${rec.disposition} — ${rec.evidence.slice(0, 110)}`);
    }
  }
}

  }
  return out;
}

      process.exit(1);
    }
  }
}

// ---- 0h. SIBLING-SEARCH gate (QUEUE #570, #574; rule 20(b), org/rooms/judge.md 5a) ------------
/**
 * THE PROPERTY: every CONFIRMED judge row whose mechanism names a family (`family` not
 * `unclassified`) carries its searched scope and its outcome, or the batch does not
 * close. The record is the judge's own FINDINGS.jsonl row (judge.md 5a), which the teachback's
 * recurrence column reads (teachback.md rule 3). NON-VACUITY: a judge row with no verdict word,
 * a CONFIRMED row with no `family`, and a judge with no FINDINGS.jsonl each refuse -- otherwise
 * omitting the field would be the cheapest way past the gate.
 */
/** The names are org/06-runs/schemas/RUN.schema.json's own (ADR-0008 D4/D6: `outcome`, `sibling_searches[].searched_scope`,
 *  `candidates[].site|disposition`), reused so a run instance and a judge row spell one record one way. `not_yet_swept` is
 *  that enum's fifth word and is refused here: a judge that ran the search has swept. */
const SIBLING_DISPOSITION = new Set(["recurrence_in_scope", "none_in_scope", "novel_variant", "historical_sibling"]);
const CANDIDATE_DISPOSITION = new Set(["confirmed", "rejected", "unresolved", "out_of_scope"]);
const FAMILY_VALUE = /^(FAMILY-\d{3}(\/v\d+)?|new: \S.*|unclassified)$/;
/** B27-RG-ORG RG-570-PLACEHOLDER-SCOPE: `RECON_CITATION` accepted ANY backticked token, so '`n/a`'
 *  passed as a searched scope. A scope cites a `path:line`, or a backticked token that is a
 *  command or a path: it has a letter and a space or a '/', and is not a placeholder word. */
const SCOPE_PLACEHOLDER = /^\s*(n\/?a|none|null|nil|tbd|todo|unknown|nothing|same|see above|-+|\?+|\.+)\s*$/i;
/** B27-FX4B J-ORG-02 (QUEUE #570, D-0814(6)): the denylist passed '`not searched`' (a letter and a space).
 *  A token now has to LOOK like a search: its first word is a command, or it carries a path shape
 *  (`a/b`, `name.ext`); and its first word is not a negation or placeholder. */
const SCOPE_PATH = /[A-Za-z0-9_.*-]\/[A-Za-z0-9_.*-]|[A-Za-z0-9_*-]\.[A-Za-z]{1,6}\b/;
const SCOPE_NEGATED = /^(?:not|no|never|none|nothing|skipped|unsearched|pending|did|didn't|n\/?a|tbd|todo|same|see)$/i;
// I34-FX1-TOOLS (D-1110(5), I33 B3): D-1089(4)'s ordinary words, the only negation words exempt as an UNQUOTED (or escaped)
// pattern; a quoted pattern is pattern text whatever it holds (D-1089(4)(a)).
const PATTERN_ORDINARY = new Set(["did", "same", "see"]);
/** I7-CLOSE-TOOLS (#686): the searched scope is an ALLOWLIST, not a first-word denylist. A backticked token counts
 *  only when its first word is a known executable (since #831a: a SEARCH, isSearchCommand below) AND it carries a SCOPE ARGUMENT -- a path
 *  that is RUNNABLE (since I19 #831a: pathOperand -- the whole tree, a path shape, or a name that exists under ROOT;
 *  a bare word is no longer a scope). A bare command ('`grep`', '`ls -la`') names no scope; a bare path with no command ('`scripts/a.mjs`')
 *  names no search. The other arm, a `path:line` (COVERAGE_POINTER), is unchanged. */
// I19-W1F (#831a): scopeArgument() (its "symbol" arm: any 3+ letter identifier) is gone -- that arm is what let
// `grep the record` stand as a search; the path half is now pathOperand() below.
// #831a (I16-W03J): the first word being ANY known executable let `git status`, `git log`, `test true` and
// `node whatever` stand as proof of a search (a 3-letter word passed as a "symbol" scope argument). A searched
// scope is now a SEARCH: a search tool with a pattern AND a path (`grep -rn guard scripts`, `find scripts -name x`),
// `git grep <pattern> <path>`, `git ls-files <pathspec>`, or a script (a path, run directly or by an interpreter)
// with at least one argument. Reading a file (`sed`, `cat`) is not a search.
const SEARCH_TOOL = /^(?:grep|egrep|fgrep|rg|ag|ack|find|fd)$/;
const SCRIPT_RUNNER = /^(?:node|python3?|bash|sh)$/;
const SCRIPT_PATH = /^(?:\.{0,2}\/)?[A-Za-z0-9_.-]+(?:\/[A-Za-z0-9_.-]+)+\.(?:mjs|cjs|js|ts|py|sh)$/;
const unquote = (a) => a.replace(/^["'(]+|["'),;:]+$/g, "");
const WHOLE_TREE = /^\.{1,2}\/?$/;
const searchArg = (a) => { const w = unquote(a); return w !== "" && !w.startsWith("-") && !SCOPE_PLACEHOLDER.test(w) && !SCOPE_NEGATED.test(w); };
// I19-W1F (#831a): the PATH a search ran over must be runnable: the whole tree, a path shape (SCOPE_PATH), or a name that
// EXISTS in the tree -- under ROOT or under this script's own checkout (a closure gate run with a synthetic --root still
// reads a repo path the judge searched). A bare word (`grep the record`) was read as a "symbol" scope and passed.
const SCRIPT_TREE = fileURLToPath(new URL("..", import.meta.url));
// I19-FXD (RG-TOOLS L2a, D-0956(1)): a path SHAPE is no longer enough -- every path operand must EXIST, inside ROOT or
// inside this script's checkout (`nowhere/does-not-exist`, `notes.txt`, `/dev/null` were accepted). A glob is checked
// at its literal directory prefix (`scripts/*.mjs` -> `scripts/`); the whole-tree operands (`.`, `..`, `./`) are kept.
const globBase = (w) => { const i = w.search(/[*?[{]/); return i < 0 ? w : w.slice(0, i).replace(/[^/]*$/, ""); };
const within = (base, p) => isInside(base, p, { allowSelf: true });
// I34-FX1-TOOLS (RG-TOOLS-N2, D-1112(2)): a glob in the FIRST segment leaves an empty base, which checked ROOT itself
// (`{nowhere,fake}/x.mjs`, `?owhere/x`); such a word counts only when the glob matches an existing path inside the base.
const globHits = (b, w) => globSync(w, { cwd: b }).some((p) => within(b, resolve(b, p)));
const inTree = (w) => [ROOT, SCRIPT_TREE].some((b) => { const p = resolve(b, globBase(w)); return within(b, p) && existsSync(p) && (globBase(w) !== "" || globHits(b, w)); });
// I33-FX3-GUARDS (I33-RJ-02:2, D-1095(2)): a path operand is checked as written (shellWords has already removed its
// quotes): no trailing `,`, `;` or `:` comes off, so the `scripts,` of `grep callers: scripts, tests` is not `scripts`.
// I33-FX4-CLASSES (D-1097, I33-RJ-03:2): a CLASS, not a list of separators. A word with a `,` or `;`, or ending in `:`, is
// not a path operand, glob or not (`scripts/*,` was checked at its glob base `scripts/` and passed). A glob word is its
// literal base (checked by inTree) and a tail of path and glob characters only: nothing else may ride after it. Inside a
// backticked COMMAND a shell brace list (`src/{a,b}.ts`) is a glob, so its commas are not separators there; in an
// UNBACKTICKED field (barePathOperand, isBareSearchLine) a `,` or `;` anywhere refuses, braces included.
const LISTED_WORD = /[,;]|:$/;
const GLOB_WORD = /[*?[{]/;
const GLOB_TAIL = /^[A-Za-z0-9_.\-/*?[\]{}!]*$/;
const BRACE_LIST = /\{[A-Za-z0-9_.\-/*?,]*\}/g;
const unbraced = (w) => w.replace(BRACE_LIST, "{}");
const globClean = (w) => !GLOB_WORD.test(w) || GLOB_TAIL.test(unbraced(w.slice(globBase(w).length)));
const pathOperand = (w) => WHOLE_TREE.test(w) || (!LISTED_WORD.test(unbraced(w)) && globClean(w) && (SCOPE_PATH.test(w) || /^[A-Za-z0-9_.-]+\/?$/.test(w)) && inTree(w));
const barePathOperand = (w) => !LISTED_WORD.test(w) && pathOperand(w);
const scriptExists = (p) => inTree(unquote(p));
/** I33-GUARDS (D-1089(4)): split a command into words as a shell does. Unquoted whitespace separates words; a '...' or
 *  "..." span stays inside one word (quotes removed), which is marked `quoted`; `flag` marks a word whose first character
 *  is an unquoted '-'. `control` is true when an unquoted `;`, `|`, `&`, `<`, `>`, `$(` or a newline occurs:
 *  then the text is more than one plain command. I34-TOOLS-A (JA-01): POSIX backslash escapes -- unquoted, `\` takes
 *  the next character literally; inside "..." only `\"`, `\\`, `\$` and `\`` escape (any other `\` stays); inside '...' none. */
function shellWords(s) {
  const words = [];
  let cur = null, q = null, control = false;
  for (let i = 0; i < s.length; i++) {
    const c = s[i];
    if (q === '"' && c === "\\" && '"\\$`'.includes(s[i + 1])) { cur.text += s[++i]; continue; }
    if (q) { if (c === q) q = null; else cur.text += c; continue; }
    if (c === "\\" && i + 1 < s.length) { cur ??= { text: "", quoted: false, flag: false }; cur.text += s[++i]; continue; }
    if (c === "'" || c === '"') { cur ??= { text: "", quoted: false, flag: false }; cur.quoted = true; q = c; continue; }
    if (/\s/.test(c)) { if (c === "\n") control = true; if (cur) { words.push(cur); cur = null; } continue; }
    if (";|&<>".includes(c) || (c === "$" && s[i + 1] === "(")) control = true;
    cur ??= { text: "", quoted: false, flag: c === "-" };

const GREP_VALUE_FLAGS = new Set(["-e", "--regexp", "-f", "--file", "-A", "-B", "-C", "-m", "--max-count", "--context", "--after-context", "--before-context", "-d", "-D", "--include", "--exclude", "--exclude-dir"]);

const PATTERN_FLAGS = new Set(["-e", "--regexp"]);
const FIND_PATTERN_TESTS = new Set(["-name", "-iname", "-path", "-ipath", "-regex", "-iregex", "-wholename", "-iwholename"]);
const FIND_NO_VALUE = new Set(["-print", "-print0", "-o", "-or", "-a", "-and", "-not", "-delete", "-ls", "-prune", "-empty", "-follow", "-depth", "-xdev", "-mount", "-true", "-false", "-quit", "-readable", "-writable", "-executable", "-nouser", "-nogroup"]);
const FIND_OPERATORS = new Set(["(", ")", "\\(", "\\)", "!", ","]);
/** grep-family words from `start`: the pattern's index (-1 when none) and the indexes of the words after it. */
function grepParse(words, start, valueFlags) {
  let pattern = -1;
  const after = [];
  for (let i = start; i < words.length; i++) {
    const w = words[i];
    if (w.flag) {
      if (valueFlags.has(w.text)) { if (PATTERN_FLAGS.has(w.text) && pattern < 0 && i + 1 < words.length) pattern = i + 1; i++; }
      continue;
    }
    if (pattern < 0) pattern = i;
    else after.push(i);
  }
  return { pattern, after };
}
/** `find [-L|-H|-P] <path>... <expression>`: the leading path words, the first name-test value (the pattern), how many
 *  tests took a value, and `stray` -- expression words that are neither a test, its value nor an operator. */
function findParse(words) {
  let i = 1;
  while (i < words.length && ["-L", "-H", "-P"].includes(words[i].text)) i++;
  const paths = [];
  while (i < words.length && !words[i].flag && !FIND_OPERATORS.has(words[i].text)) paths.push(i++);
  let pattern = -1, valued = 0;
  const stray = [];
  for (; i < words.length; i++) {
    const w = words[i];
    if (w.flag) {
      if (FIND_NO_VALUE.has(w.text)) continue;
      if (i + 1 >= words.length) { stray.push(i); continue; }
      valued++;
      if (pattern < 0 && FIND_PATTERN_TESTS.has(w.text)) pattern = i + 1;
      i++;
      continue;
    }
    if (!FIND_OPERATORS.has(w.text)) stray.push(i);
  }
  return { paths, pattern, valued, stray };
}
/** The index of the PATTERN argument of a grep/rg/... or find command, or -1 (a script, `git ls-files`). */
function patternIndex(words) {
  const first = words[0]?.text ?? "";

  if (first === "git" && words[1]?.text === "grep") return grepParse(words, 2, GREP_VALUE_FLAGS).pattern;
  if (SEARCH_TOOL.test(first)) return grepParse(words, 1, first === "rg" ? RG_VALUE_FLAGS : GREP_VALUE_FLAGS).pattern;
  return -1;
}

// ---- 1. parse "## 1." ADJUDICATION table -----------------------------------
function section(text, headingRe) {
  const lines = text.split("\n");
  let start = -1, end = lines.length;
  for (let i = 0; i < lines.length; i++) {
    if (start === -1 && headingRe.test(lines[i])) { start = i + 1; continue; }
    if (start !== -1 && /^##\s/.test(lines[i]) && i > start) { end = i; break; }
  }
  if (start === -1) return "";
  return lines.slice(start, end).join("\n");
}
const adjudicationSection = section(judgeText, /^##\s*1\./);
const dispatchSection = section(judgeText, /^##\s*5\.\s*DISPATCH/);

// QUEUE #586 (I9-TOOLING): a finding id is letters/digits in at least two dash-joined segments (`--` allowed). The old
// `<x>-<digits>` test dropped ids ending in a word (I3-0304-b, TB-W309A-STATUS-VOCAB, I8-JN-PRIV). Census over every
// repo org/07-findings/**/FINDINGS.jsonl id during I9: 17,771 ids, 12,549 accepted before, 17,520 now, none accepted before and
// refused now; the rest have no dash (F01, q358), a shape a table's room-name rows (R2, SCOUT) share, so they stay out.
// I9-FX1 (QUEUE #586, RG586-02) / I10-CLOSER (QUEUE #855, D-0870(5)): the judge ADJUDICATION-table row filter is
// judgeTableFindingId in scripts/lib/finding-id.mjs, ONE function shared by close-run.mjs and audit-precheck.mjs (it
// was a byte-identical copy in each). It also accepts dotted (`I3-0304.b`) and lower-case-led (`i8-jn-priv`) ids.
const tableRows = [];
for (const line of adjudicationSection.split("\n")) {
  const m = line.match(/^\|(.+)\|$/);
  if (!m) continue;
  const cells = m[1].split("|").map((c) => c.trim());
  if (cells.length < 3) continue;
  if (/^-+$/.test(cells[0].replace(/\s/g, ""))) continue; // separator row
  const idCell = cells[0];
  const idTok = judgeTableFindingId(cells);
  if (idTok === null) continue; // header, aggregate or prose row: not a finding
  const [, verdictCell = "", sevCell = "", reachCell = "", detectCell = "", confCell = ""] = cells;
  const verdict = (verdictCell.match(/^[A-Z_]+/) || [""])[0];
  tableRows.push({

// ---- 2. parse "## 5. DISPATCH" buckets -------------------------------------
// Buckets are introduced by a bold heading line ("**FIX-IN-SWEEP —...**",
// "**QUEUE (docket).**", "**NEEDS_REPRODUCTION.**", "**STATED PROPERTY.**");
// each bullet under one reads "- **<bold lead, ids here>.** <body prose>".
// We extract ids ONLY from the bold lead (text inside the first `**...**` of
// the bullet) to avoid false-positive matches like "R-5" or "T2" inside body
// prose that also happens to look like `[A-Z]-\d+`.
const bucketHeadingRe = /^\*\*(FIX-IN-SWEEP|QUEUE|NEEDS_REPRODUCTION|STATED PROPERTY)\b/;
let bucket = null;
const idToBucket = {}; // shortId -> bucket name
const unmatchedMentions = []; // short mentions ("A-1") that matched no table id
const crossRefBucket = {}; // shortId -> the bucket of its first parenthesised mention (used only when no primary mention exists)
for (const line of dispatchSection.split("\n")) {
  const h = line.match(bucketHeadingRe);
  if (h) { bucket = h[1]; continue; }
  if (!bucket) continue;
  const bullet = line.match(/^-\s*\*\*(.+?)\*\*/);
  if (!bullet) continue;
  const lead = bullet[1];
  // QUEUE #586 (I9-TOOLING): every id-shaped token in the lead, whole (ORG-03-R1-04, REG-579-09, I8-JN-PRIV); the old
  // `\b[A-Z]-\d+\b` saw none of them, so their CONFIRMED rows fell to `open`. A token that IS a table id maps exactly.
  // Only a legacy short token (A-1, DISPATCH dropping the room segment) falls back to first letter + trailing number,
  // and only onto exactly ONE table row: two candidates is unmatched, never first-wins.
  // I9-FX1 (RG586-01): a token inside parentheses is a CROSS-REFERENCE (`B-probe-1 (sibling of A-sentinel-1)`), any other
  // is PRIMARY. An id takes the bucket of its FIRST primary mention; an id with no primary mention anywhere takes its first
  // cross-reference's bucket. No later bullet overwrites an earlier bucket (the old scan: last mention won).
  let depth = 0;
  const primaryAt = new Set();
  for (let i = 0; i < lead.length; i++) {
    if (lead[i] === "(") depth++;
    else if (lead[i] === ")") depth = Math.max(0, depth - 1);
    else if (depth === 0) primaryAt.add(i);
  }
  // QUEUE #855 (D-0870(5)): tokens and matching are finding-id.mjs's (dotted ids whole; a mixed/lower-case lead maps only
  // when exactly one table id equals it ignoring case; the legacy short form stays upper-case, onto exactly one row).
  const tableIds = tableRows.map((r) => r.shortId);
  for (const mm of findingIdTokens(lead)) {
    const short = mm.id;
    const hit = matchTableId(short, tableIds);
    const match = hit === null ? undefined : tableRows.find((r) => r.shortId === hit);
    if (match) {
      if (primaryAt.has(mm.index)) { if (!(match.shortId in idToBucket)) idToBucket[match.shortId] = bucket; }
      else if (!(match.shortId in crossRefBucket)) crossRefBucket[match.shortId] = bucket;
    } else unmatchedMentions.push({ mention: short, bucket, bulletLead: lead }); // I9-FX1 (RG586-04): digit-free too, never dropped
  }
}
for (const [id, b] of Object.entries(crossRefBucket)) if (!(id in idToBucket)) idToBucket[id] = b;

// ---- 4. small RUN.schema 0.6.0 validator (inlines $defs by $ref) ----------
function resolveRef(root, ref) {
  const path = ref.replace(/^#\//, "").split("/");
  let node = root;
  for (const p of path) node = node[p];
  return node;
}
function typeOf(v) {
  if (v === null) return "null";
  if (Array.isArray(v)) return "array";
  return typeof v;
}
function validate(schema, data, path, root, errors) {
  if (!schema) return;
  if (schema.$ref) return validate(resolveRef(root, schema.$ref), data, path, root, errors);
  if (schema.oneOf) {
    const subErrs = schema.oneOf.map((s) => { const e = []; validate(s, data, path, root, e); return e; });
    if (!subErrs.some((e) => e.length === 0)) errors.push(`${path}: matches none of oneOf (${subErrs.map((e) => e.length).join(",")} errors each)`);
    return;
  }
  if (schema.anyOf) {
    const subErrs = schema.anyOf.map((s) => { const e = []; validate(s, data, path, root, e); return e; });
    if (!subErrs.some((e) => e.length === 0)) errors.push(`${path}: matches none of anyOf`);
    return;
  }
  if (schema.allOf) for (const s of schema.allOf) validate(s, data, path, root, errors);
  if (schema.const !== undefined && data !== schema.const) errors.push(`${path}: expected const ${JSON.stringify(schema.const)}, got ${JSON.stringify(data)}`);
  if (schema.enum && !schema.enum.includes(data)) errors.push(`${path}: ${JSON.stringify(data)} not in enum ${JSON.stringify(schema.enum)}`);
  if (schema.type) {
    const types = Array.isArray(schema.type) ? schema.type : [schema.type];
    const t = typeOf(data);
    const ok = types.includes(t) || (types.includes("integer") && t === "number" && Number.isInteger(data));
    if (!ok) { errors.push(`${path}: expected type ${types.join("|")}, got ${t}`); return; }
  }
  if (typeof data === "string") {
    if (schema.minLength !== undefined && data.length < schema.minLength) errors.push(`${path}: length ${data.length} < minLength ${schema.minLength}`);
    if (schema.maxLength !== undefined && data.length > schema.maxLength) errors.push(`${path}: length ${data.length} > maxLength ${schema.maxLength}`);
    if (schema.pattern && !new RegExp(schema.pattern).test(data)) errors.push(`${path}: ${JSON.stringify(data)} fails pattern ${schema.pattern}`);
  }
  if (Array.isArray(data)) {
    if (schema.minItems !== undefined && data.length < schema.minItems) errors.push(`${path}: ${data.length} items < minItems ${schema.minItems}`);
    if (schema.maxItems !== undefined && data.length > schema.maxItems) errors.push(`${path}: ${data.length} items > maxItems ${schema.maxItems}`);
    if (schema.uniqueItems) { const seen = new Set(); for (const it of data) { const k = JSON.stringify(it); if (seen.has(k)) { errors.push(`${path}: duplicate item ${k}`); break; } seen.add(k); } }
    if (schema.items) data.forEach((it, i) => { validate(schema.items, it, `${path}[${i}]`, root, errors); });
  }
  if (data && typeof data === "object" && !Array.isArray(data)) {
    if (schema.required) for (const k of schema.required) if (!(k in data)) errors.push(`${path}: missing required "${k}"`);
    if (schema.properties) for (const k of Object.keys(schema.properties)) if (k in data) validate(schema.properties[k], data[k], `${path}.${k}`, root, errors);
    if (schema.additionalProperties === false && schema.properties) {
      for (const k of Object.keys(data)) if (!(k in schema.properties)) errors.push(`${path}: unexpected property "${k}" (additionalProperties: false)`);
    }
  }
  // schema.allOf conditionals (if/then) are intentionally NOT evaluated here —
  // a "small node validator" per the brief, not a full draft-2020-12 engine.
  // Reported as a known gap in REPORT.md rather than silently claimed correct.
}

function readJsonl(path) {
  const rows = [];
  const bad = [];
  const text = readFileSync(path, "utf8");

function declaredNoId(row) {

  if (v === undefined || v === null) return null;

export function extractRegisterIds(row, sidecar = null) {
  if (!row || typeof row !== "object") return [];
  const declared = declaredRegisterIds(row);
  if (declared.length) return declared;
  if (declaredNoId(row)) return [];
  const ownId = String(row.id ?? "").trim();
  if (isRegisterId(ownId)) return [ownId.toUpperCase()];
  const sc = sidecarVerdict(row, sidecar);
  if (sc) return sc.iid;
  // D-0815(5), FX4B-SIB-LEDGER-NEG: an id the text does not ASSERT ("NOT the same defect as I3-0963")
  // is not a citation. B28-TOOLING (QUEUE #655): the ONE recogniser close-run.mjs gate 0f also uses,
  // `assertedClauses()` in findings-vocab.mjs -- the copy that lived here is gone.
  const hay = [row.id, row.mechanism, row.file].filter(Boolean).map(String)
    .flatMap((f) => assertedClauses(f)).join(" ");
  const found = new Set();
  for (const m of hay.matchAll(REGISTER_ID_RE)) found.add(m[0].toUpperCase());
  return [...found];
}

/**

export function linkSource(row, sidecar = null) {

  if (declaredRegisterIds(row).length) return "declared";

/**
 * QUEUE #548 (B27-RG-ORG RG-548-SIDECAR-UNREAD): `_IID-BACKFILL.jsonl` (scripts/iid-backfill.mjs)
 * is READ here, keyed `<batch>/<room>:<line>` and checked against the row's own id so a stale
 * sidecar is ignored rather than trusted. For a row the prose scan would link, the sidecar's
 * attribution replaces the scrape: `attributed` -> its ids, a `prose-mention`/`prose-multi`
 * verdict -> no link. Absent file -> null (the ledger says so in its provenance line).
 */
export function loadSidecar(path) {
  if (!existsSync(path)) return null;
  const byKey = new Map();
  for (const l of readFileSync(path, "utf8").split("\n")) {
    if (!l.trim()) continue;
    let o = null;
    try { o = JSON.parse(l); } catch { continue; }
    const seg = String(o?.file ?? "").split("/");
    if (seg.length < 3 || typeof o.line !== "number") continue;
    byKey.set(`${seg[seg.length - 3]}/${seg[seg.length - 2]}:${o.line}`, o);
  }
  return byKey;
}

    let o = null;
    try { o = JSON.parse(l); } catch { continue; }
    if (o && typeof o === "object") out.push({ row: o.row, landed_by: o.landed_by, batch: o.batch });
  }
  return out;
}
function sidecarVerdict(row, sidecar) {
  if (!sidecar || !row.__at) return null;
  const o = sidecar.get(row.__at);
  if (!o || String(o.id ?? "") !== String(row.id ?? "")) return null;
  if (o.attribution === "attributed" && Array.isArray(o.iid)) return { iid: o.iid.map((x) => String(x).toUpperCase()) };
  if (o.attribution === "unattributable" && /^prose-/.test(String(o.reason ?? ""))) return { iid: [] };
  return null;
}

// ---------------------------------------------------------------------------
// LIMBS (QUEUE #884, I13-TOOLS). `PRIORITY` ranks `fixed` first among the claims on an item, so one row that fixed ONE
// limb of a multi-limb item made the whole item read fixed: I3-0027 (an interim guard shipped, the threshold is not
// calibrated) and I3-0028 (partly done) were counted CLOSED from a builder's row while a judge's row for another limb
// said `open`. A row now names the limb it disposes of — `limb`, or for a judge's ruling row its `rules_on` ("ROOM:N",
// which already names the limb it rules on) — and an item with an UNRESOLVED limb never resolves to a closed word:
// it resolves to that limb's word and is counted PARTIAL. Rows that carry neither field behave exactly as before.
// ---------------------------------------------------------------------------
/** One spelling for a limb key: `R:4`, `R-4` and `r_4` are the same limb (I13-RG-STRUCT-06). */
export function normLimb(v) {
  return String(v ?? "").toLowerCase().replace(/[^a-z0-9]+/g, "-").replace(/^-+|-+$/g, "");
}
/** Tokens of a limb/ruling string that can name a limb or a row: a letter AND a digit (`L1`, `R:4`, `I4-J-FAN/J-FAN-NEW-05`). */
function limbTokens(v) {
  return (String(v ?? "").match(/[A-Za-z0-9][A-Za-z0-9._]*(?:[-:/][A-Za-z0-9._]+)*/g) ?? []).filter((t) => /[A-Za-z]/.test(t) && /\d/.test(t));
}
/**
 * The limb a row disposes of by an EXPLICIT field (`limb`, else a judge's `rules_on`), or null. Free text that names
 * no letter+digit key ("none (own finding)", "(a) never-spilled claims") is NOT a limb: it used to be one distinct
 * limb per spelling, which no fixing row could ever clear (I13-RG-STRUCT-09, I3-1288 read PARTIAL when fully fixed).
 */
export function limbOf(row) {
  const v = row?.limb ?? row?.rules_on;
  return typeof v === "string" && v.trim() && limbTokens(v).length ? v.trim() : null;
}
/** Every name a row answers to: its normalised limb string, the key tokens inside it (and the tail after a `/`), and its own id. */
export function limbAliases(row) {
  const out = new Set();
  const lb = limbOf(row);
  if (lb) {
    out.add(normLimb(lb));
    for (const t of limbTokens(lb)) {
      const n = normLimb(t);
      if (n.length >= 4) out.add(n);
      if (t.includes("/")) { const tail = normLimb(t.slice(t.lastIndexOf("/") + 1)); if (tail.length >= 4) out.add(tail); }
    }
  }
  const id = normLimb(row?.id);
  if (id.length >= 5 && /\d/.test(id)) out.add(id);
  return [...out];
}

/**
 * Resolve ONE item's entries to a winner. `entries` carry `{status, limb, aliases?}`; `unmappable` ones are ignored.
 * Without limb rows this is the old rule (`PRIORITY.find`). With them, rows are grouped into limbs by SHARED NAME
 * (`aliases`, else the normalised `limb`): a fixing row that names a limb by any of its spellings, or by the id of the
 * row that opened it, clears it. A limb whose own winner is UNRESOLVED stops the item resolving to a closed word: the
 * result is `{ winner: <that limb's word>, partial: true }`. A group is a limb only if one of its rows carries a
 * `limb`. Pure; exported for tests.
 */
export function resolveItem(entries) {
  const known = entries.filter((e) => e.status !== "unmappable");
  if (known.length === 0) return { winner: "unmappable_only", distinct: [], partial: false, limbs: [] };
  const distinct = new Set(known.map((e) => e.status));
  const base = PRIORITY.find((p) => distinct.has(p)) ?? "unmappable_only";
  const names = known.map((e) => new Set([...(e.aliases ?? []), ...(e.limb ? [normLimb(e.limb)] : [])].filter(Boolean)));
  const parent = known.map((_, i) => i);
  const find = (i) => { while (parent[i] !== i) i = parent[i] = parent[parent[i]]; return i; };
  const owner = new Map();
  names.forEach((ns, i) => { for (const n of ns) { if (owner.has(n)) parent[find(i)] = find(owner.get(n)); else owner.set(n, i); } });
  const groups = new Map();
  known.forEach((e, i) => { const g = find(i); if (!groups.has(g)) groups.set(g, { limb: null, sts: new Set() }); const o = groups.get(g); if (!e.superseded) o.sts.add(e.status); if (e.limb && !o.limb) o.limb = e.limb; });
  const limbs = [...groups.values()].filter((g) => g.limb).map((g) => ({ limb: g.limb, winner: PRIORITY.find((p) => g.sts.has(p)) ?? "unmappable_only" }));
  const openWords = new Set(limbs.map((l) => l.winner).filter((w) => UNRESOLVED_SET.has(w)));
  if (CLOSED_SET.has(base) && openWords.size > 0) return { winner: PRIORITY.find((p) => openWords.has(p)), distinct: [...distinct], partial: true, limbs };
  return { winner: base, distinct: [...distinct], partial: false, limbs };
}

// Reconciliation
// ---------------------------------------------------------------------------
export function reconcile({ register, findingsFiles, sidecar = null, landed = null }) {
  const perItem = new Map(); // iid -> [{status, batch, room, sourceId, raw}]
  const byBatch = new Map(); // batch -> {rows, mapped:{}, unmappable, noId, unknownId}
  const unmappableCounts = new Map(); // raw string -> count
  let totalRows = 0, badRows = 0, noIdRows = 0, unknownIdRows = 0, multiIdRows = 0;
  // QUEUE #548: the link's PROVENANCE, counted. A closure figure carried by
  // `prose` is the figure two extraction methods disagreed about by 26%.
  const linkSources = { declared: 0, "declared-none": 0, "declared-queue": 0, "declared-bad": 0, "own-id": 0, sidecar: 0, prose: 0, none: 0 };

  // I13-FX8 (a): a judge names the row it rules on as `ROOM:LINE` (the physical line of ROOM's FINDINGS.jsonl), the
  // builder keys it by its own id (`RG-AUTH-07`, `I12-MAIL-09`): the two spellings never meet by name.
  const parsed = findingsFiles.map((f) => ({ f, ...readJsonl(f.path) }));
  const roomLine = new Map(); // normLimb("ROOM:N") -> ["batch/room:N", ...]
  const addresses = new Set(); // D-1077(3): every row's exact "batch/room:N"
  for (const { f, rows } of parsed) for (const row of rows) {
    if (!row || typeof row !== "object" || !row.__line) continue;
    const k = normLimb(`${f.room}:${row.__line}`);
    if (!roomLine.has(k)) roomLine.set(k, []);
    roomLine.get(k).push(`${f.batch}/${f.room}:${row.__line}`);
    addresses.add(`${f.batch}/${f.room}:${row.__line}`);
  }
  // I46-TOOLS (D-1304(1); I46SA-057): A LANDED HAND-OFF. A wrong_owner row is landed when a CLOSED row (the fixed row in
  // the batch that took the hand-off) carries `lands: "<room>/FINDINGS.jsonl:<n>"` naming it, or when a `_LANDED.jsonl` row
  // names it with a CLOSED `landed_by` row. Its item entry then carries the landing row's status; the row's own file is
  // never edited. `<room>/FINDINGS.jsonl:<n>` resolves in the given batch first, else in the one batch holding that room
  // and line; `<batch>/<room>/FINDINGS.jsonl:<n>` is exact.
  const rowAt = new Map();
  for (const { f, rows } of parsed) for (const row of rows) if (row && typeof row === "object" && row.__line) rowAt.set(`${f.batch}/${f.room}:${row.__line}`, row);
  const resolveAddr = (ref, batch) => {
    const m = /^(?:([^/\s]+)\/)?([^/\s]+)\/FINDINGS\.jsonl:(\d+)$/.exec(String(ref ?? "").trim());
    if (!m) return null;
    if (m[1]) return `${m[1]}/${m[2]}:${Number(m[3])}`;
    const c = roomLine.get(normLimb(`${m[2]}:${Number(m[3])}`)) ?? [];
    const here = c.filter((a) => a.startsWith(`${batch}/`));
    return here.length === 1 ? here[0] : c.length === 1 ? c[0] : null;
  };
  const landedBy = new Map(); // wrong_owner "batch/room:line" -> { by, status, via }
  const land = (target, by, via) => {
    const t = target ? rowAt.get(target) : undefined;
    const b = by ? rowAt.get(by) : undefined;
    if (!t || !b || mapLegacyStatus(t.status) !== "wrong_owner" || !CLOSED_SET.has(mapLegacyStatus(b.status))) return;
    // I46-FXR-TOOLS (D-1307 F9; I46-RG-TOOLS NEW-02): the landing row disposes of the SAME register item: every item the
    // hand-off links is among the landing row's items, and a hand-off linking none lands only on a row linking none.
    const ti = extractRegisterIds({ ...t, __at: target }, sidecar);
    const bi = extractRegisterIds({ ...b, __at: by }, sidecar);
    if (ti.length ? !ti.every((x) => bi.includes(x)) : bi.length > 0) return;
    landedBy.set(target, { by, status: mapLegacyStatus(b.status), via });
  };
  for (const { f, rows } of parsed) for (const row of rows) if (row && typeof row === "object" && row.__line) land(resolveAddr(row.lands, f.batch), `${f.batch}/${f.room}:${row.__line}`, "lands");
  for (const s of landed ?? []) land(resolveAddr(s.row, s.batch), resolveAddr(s.landed_by, s.batch), "sidecar");
  /**
   * The `batch/room:line` keys a row's limb string rules on: `ROOM:N` resolved in the row's own batch first, else the one
   * batch that has that room. D-1077(3): `<batch>/<room>:<line>` is an EXACT address — it names a row in a room whose name
   * repeats across batches (SCOUT, JUDGE, ...), which `ROOM:N` cannot — and resolves only to that row, never by ROOM:N.
   */
  const ruledAts = (row, batch, own) => {
    const out = [];
    for (const t of limbTokens(limbOf(row))) {
      const m = /^(.+):(\d+)$/.exec(t);
      if (!m) continue;
      let pick;
      if (m[1].includes("/")) {
        const a = `${m[1]}:${Number(m[2])}`;
        pick = addresses.has(a) && a !== own ? [a] : [];
      } else {
        const c = (roomLine.get(normLimb(`${m[1]}:${Number(m[2])}`)) ?? []).filter((a) => a !== own);
        const here = c.filter((a) => a.startsWith(`${batch}/`));
        pick = here.length ? here : c.length === 1 ? c : [];
      }
      for (const a of pick) if (!out.includes(a)) out.push(a);
    }
    return out;
  };

  for (const { f, rows, bad } of parsed) {
    badRows += bad.length;
    if (!byBatch.has(f.batch)) byBatch.set(f.batch, { rows: 0, mapped: {}, unmappable: 0, noId: 0, unknownId: 0, itemsTouched: new Set() });
    const bstat = byBatch.get(f.batch);
    for (const row of rows) {
      if (row && typeof row === "object" && row.__line) Object.defineProperty(row, "__at", { value: `${f.batch}/${f.room}:${row.__line}` });
      totalRows++;
      bstat.rows++;
      const mapped = mapLegacyStatus(row.status);
      bstat.mapped[mapped] = (bstat.mapped[mapped] ?? 0) + 1;
      if (mapped === "unmappable") {
        bstat.unmappable++;
        const raw = String(row.status ?? "(missing)");
        unmappableCounts.set(raw, (unmappableCounts.get(raw) ?? 0) + 1);
      }
      const landing = row.__at ? landedBy.get(row.__at) : undefined;
      if (landing) bstat.landed = (bstat.landed ?? 0) + 1;
      const ids = extractRegisterIds(row, sidecar);
      linkSources[linkSource(row, sidecar)]++;
      if (ids.length === 0) { noIdRows++; bstat.noId++; continue; }
      if (ids.length > 1) multiIdRows++;
      for (const iid of ids) {
        if (!register.byId.has(iid)) { unknownIdRows++; bstat.unknownId++; continue; }
        bstat.itemsTouched.add(iid);
        if (!perItem.has(iid)) perItem.set(iid, []);
        perItem.get(iid).push({ status: landing ? landing.status : mapped, landedBy: landing ? landing.by : null, batch: f.batch, room: f.room, sourceId: row.id, raw: row.status, limb: limbOf(row), aliases: limbAliases(row), at: row.__at, ruled: row.__at ? ruledAts(row, f.batch, row.__at) : [] });
      }
    }
  }

  // Per-item resolution + conflict detection.
  const itemDisposition = new Map(); // iid -> {winner, distinctStatuses, entries}
  let conflicts = 0;
  // TEACHBACK.md:57 (2) (I22-W1-TOOLS): CLOSED-ruled vs CLOSED-claimed. A closed item is RULED when a closing row of it is the
  // row some judge/attacker row's `rules_on` names, or is itself a ruling row; otherwise the close rests on its author's say-so.
  const ruledTargets = new Set();
  for (const es of perItem.values()) for (const e of es) for (const at of e.ruled ?? []) ruledTargets.add(at);
  for (const [iid, entries] of perItem) {
    // I13-RG-STRUCT-02: a builder names its limbs in its row ids. Where ONE room wrote two or more rows on an item
    // and a row names no limb field, that row's own id is its limb: a room's `fixed` row and its own `open` row are two
    // limbs, not one item that reads fixed. A lone row stays item-level, so a later whole-item fix is unchanged.
    const perRoom = new Map();
    for (const e of entries) perRoom.set(`${e.batch}/${e.room}`, (perRoom.get(`${e.batch}/${e.room}`) ?? 0) + 1);
    for (const e of entries) if (!e.limb && e.sourceId != null && String(e.sourceId).trim() && perRoom.get(`${e.batch}/${e.room}`) > 1) e.limb = `id:${String(e.sourceId).trim()}`;
    // I13-FX8 (a): a ruling row and the row on the line it names are ONE limb (`at:` alias, collision-free), and the
    // judge's word stands for it: the ruled row's own status stays in the item-level base but not in the limb's winner.
    // A whole-item row that names no limb and is not that row still clears nothing (D-0891(5)).
    for (const e of entries) for (const at of e.ruled ?? []) {
      const tag = `at:${at}`;
      e.aliases = [...(e.aliases ?? []), tag];
      const t = entries.find((x) => x.at === at);
      if (!t) continue;
      t.aliases = [...(t.aliases ?? []), tag];
      if (CLOSED_SET.has(e.status) || UNRESOLVED_SET.has(e.status)) t.superseded = true;
    }
    const knownEntries = entries.filter((e) => e.status !== "unmappable");
    const distinct = new Set(knownEntries.map((e) => e.status));
    if (distinct.size > 1) conflicts++;
    const r = resolveItem(entries);
    const ruledClosed = CLOSED_SET.has(r.winner) && entries.some((e) => CLOSED_SET.has(e.status) && ((e.ruled?.length ?? 0) > 0 || ruledTargets.has(e.at)));
    itemDisposition.set(iid, { winner: r.winner, distinctStatuses: [...distinct], entries, partial: r.partial, limbs: r.limbs, ruledClosed });
  }

  const landedRows = [...landedBy].map(([row, v]) => ({ row, ...v }));
  return { perItem, itemDisposition, byBatch, unmappableCounts, totalRows, badRows, noIdRows, unknownIdRows, multiIdRows, conflicts, linkSources, sidecarRead: sidecar !== null, landed: landedRows };
}

// ---------------------------------------------------------------------------
// Report assembly
// ---------------------------------------------------------------------------
export function buildReport({ register, recon }) {
  const defectIds = [...register.byId.values()].filter((r) => r.kind === "defect").map((r) => r.iid);
  const totalDefects = defectIds.length;
  const touched = defectIds.filter((iid) => recon.itemDisposition.has(iid));
  const untouched = totalDefects - touched.length;

  const vocabCounts = Object.fromEntries([...Object.keys(VOCAB), ...RECON_SET, "unmappable_only"].map((k) => [k, 0]));
  for (const iid of touched) vocabCounts[recon.itemDisposition.get(iid).winner]++;

  const closed = Object.entries(vocabCounts).filter(([k]) => CLOSED_SET.has(k)).reduce((a, [, v]) => a + v, 0);
  const unresolved = Object.entries(vocabCounts).filter(([k]) => UNRESOLVED_SET.has(k)).reduce((a, [, v]) => a + v, 0);
  // RECON is reported on its own line and is NOT added into closed or unresolved.
  const reconCount = Object.entries(vocabCounts).filter(([k]) => RECON_SET.has(k)).reduce((a, [, v]) => a + v, 0);
  const partialIids = touched.filter((iid) => recon.itemDisposition.get(iid).partial);
  const closedRuled = touched.filter((iid) => CLOSED_SET.has(recon.itemDisposition.get(iid).winner) && recon.itemDisposition.get(iid).ruledClosed).length;
  const notADefect = vocabCounts.not_a_defect;
  const unmappableOnly = vocabCounts.unmappable_only;

  return {
    register: { totalRows: register.totalRows, kindCounts: register.kindCounts },
    defects: { total: totalDefects, touched: touched.length, untouched, vocabCounts, closed, closedRuled, closedClaimed: closed - closedRuled, unresolved, recon: reconCount, notADefect, unmappableOnly, partial: partialIids.length, partialIids },
    conflicts: recon.conflicts,
    rows: { total: recon.totalRows, bad: recon.badRows, noId: recon.noIdRows, unknownId: recon.unknownIdRows, multiId: recon.multiIdRows },
    linkSources: recon.linkSources ?? { declared: 0, "declared-none": 0, "declared-bad": 0, "own-id": 0, sidecar: 0, prose: 0, none: 0 },
    sidecarRead: Boolean(recon.sidecarRead),
    landed: recon.landed ?? [],
    byBatch: recon.byBatch,
    unmappableCounts: recon.unmappableCounts,
  };
}

export function formatReport(r, { showUnmappable = 15 } = {}) {
  const L = [];
  L.push("=== CLOSURE LEDGER ===");
  L.push(`register: ${r.register.totalRows} rows — ` + Object.entries(r.register.kindCounts).map(([k, v]) => `${k}:${v}`).join(", "));
  L.push(`findings rows scanned: ${r.rows.total} (${r.rows.bad} unparseable, ${r.rows.noId} name no I3- id, ${r.rows.unknownId} name an id not in the register, ${r.rows.multiId} name >1 id)`);
  // QUEUE #548. Printed UNCONDITIONALLY and with its population, so it cannot
  // pass by being empty: `prose` is the share of this report that rests on
  // scraping `mechanism`, which is the 282-vs-356 gap expressed as a number
  // instead of an argument.
  const ls = r.linkSources;
  const linked = ls.declared + ls["own-id"] + (ls.sidecar ?? 0) + ls.prose;
  L.push(`register link provenance (#548): declared ${ls.declared}, declared-none ${ls["declared-none"] ?? 0}, declared-queue ${ls["declared-queue"] ?? 0}, declared-bad ${ls["declared-bad"] ?? 0}, own-id ${ls["own-id"]}, sidecar ${ls.sidecar ?? 0} (_IID-BACKFILL.jsonl ${r.sidecarRead ? "read" : "ABSENT"}), PROSE ${ls.prose}, none ${ls.none} — ${linked ? (100 * ls.prose / linked).toFixed(1) : "0.0"}% of linked rows are linked by scraping \`mechanism\`, which is the only part of this figure two extraction methods can disagree about`);
  L.push("");
  L.push(`defects in register: ${r.defects.total}`);
  L.push(`  touched (>=1 linked finding): ${r.defects.touched}`);
  L.push(`  untouched: ${r.defects.untouched}`);
  L.push(`  disposition-unmappable-only (touched, no readable status): ${r.defects.unmappableOnly}`);
  L.push("");
  L.push(`CLOSED (fixed + defended): ${r.defects.closed} / ${r.defects.total} (${(100 * r.defects.closed / r.defects.total).toFixed(1)}%)`);
  L.push(`  (#884: ${r.defects.partial ?? 0} PARTIAL item(s) are NOT in CLOSED — a limb row is unresolved; the pre-#884 rule would have read CLOSED as ${r.defects.closed + (r.defects.partial ?? 0)})`);
  L.push(`  CLOSED-ruled (a judge/attacker row rules on a closing row): ${r.defects.closedRuled ?? 0}; CLOSED-claimed (no ruling row names it): ${r.defects.closedClaimed ?? r.defects.closed}`);
  L.push(`  fixed:    ${r.defects.vocabCounts.fixed}`);
  L.push(`  defended: ${r.defects.vocabCounts.defended}`);
  L.push(`not_a_defect: ${r.defects.notADefect}`);
  L.push(`UNRESOLVED: ${r.defects.unresolved}`);
  L.push(`  of which PARTIAL (#884: one limb is fixed, another is still unresolved — counted here, never in CLOSED): ${r.defects.partial ?? 0}`);
  L.push(`  open:           ${r.defects.vocabCounts.open}`);
  L.push(`  wrong_owner:    ${r.defects.vocabCounts.wrong_owner}`);
  const ld = r.landed ?? [];
  L.push(`  landed hand-offs (D-1304(1): a wrong_owner row counted at its CLOSED landing row's status): ${ld.length} (lands field ${ld.filter((x) => x.via === "lands").length}, _LANDED.jsonl ${ld.filter((x) => x.via === "sidecar").length})`);
  L.push(`  not_reproduced: ${r.defects.vocabCounts.not_reproduced}`);
  L.push(`  not_located:    ${r.defects.vocabCounts.not_located}`);
  L.push(`RECON (read-only rooms' anchors; neither closed nor unresolved): ${r.defects.recon}`);
  L.push(`  located:        ${r.defects.vocabCounts.located}`);
  L.push("");
  L.push(`CONFLICTS (>1 distinct mapped status on the same register item): ${r.conflicts}`);
  L.push("");
  L.push("--- by batch ---");
  for (const [batch, b] of [...r.byBatch.entries()].sort()) {
    const mappedStr = Object.entries(b.mapped).filter(([, v]) => v > 0).map(([k, v]) => `${k}:${v}`).join(" ");
    L.push(`${batch}: rows=${b.rows} items_touched=${b.itemsTouched.size} no_id=${b.noId} unknown_id=${b.unknownId} unmappable=${b.unmappable}${b.landed ? ` landed=${b.landed}` : ""} | ${mappedStr}`);
  }
  L.push("");
  L.push(`--- top ${showUnmappable} unmappable raw status strings (word, count) ---`);
  const top = [...r.unmappableCounts.entries()].sort((a, b) => b[1] - a[1]).slice(0, showUnmappable);
  for (const [word, count] of top) L.push(`  ${count}\t${word}`);
  return L.join("\n");
}
