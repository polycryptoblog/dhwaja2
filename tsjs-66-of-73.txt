/** One guarded audit INSERT (the auditStmt columns), landing only when `where` holds. */
function guardedAuditStmt(db: DbLike, actor: string, action: string, passportId: string, detail: string, ts: string, where: string, binds: readonly unknown[]): DbStatement {
  return db
    .prepare(`INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, ?, ?, ?, ?, ? WHERE ${where}`)
    .bind(actor, action, detail, ts, passportId, auditActorKind(actor), ...binds);
}

/**
 * I54-FX-PRIV REQ-02: erase the names a subject's own delisting owed under a hold, once no hold names the passport.
 * A passport is taken when an OWED marker was written at or after its standing suppression row (a lift and an
 * operator re-delisting leave an older marker behind), no DONE marker follows that marker, no open hold names it and
 * names remain. Each passport is one batch: the DONE marker and the erase, both guarded on the suppression row and
 * on no open hold, so a hold opened or a lift landed since the read changes nothing. Returns the passports taken.
 */
export async function sweepSubjectEraseOwed(db: DbLike, nowIso: string): Promise<number> {
  const { results } = await db
    .prepare(
      `SELECT DISTINCT o.item_id AS passport_id, (SELECT m.created_at FROM audit_log m WHERE m.item_id = o.item_id AND m.action = o.action ORDER BY m.created_at DESC LIMIT 1) AS owed_at FROM audit_log o JOIN request_suppressions s ON s.passport_id = o.item_id
        WHERE o.action = ? AND o.created_at >= s.created_at
          AND NOT EXISTS (SELECT 1 FROM audit_log d WHERE d.item_id = o.item_id AND d.action = ? AND d.created_at >= o.created_at)
          AND NOT ${legalHoldOpenSql("passport", "o.item_id")}
          AND (EXISTS (SELECT 1 FROM passport_names n WHERE n.passport_id = o.item_id) OR EXISTS (SELECT 1 FROM passports p WHERE p.passport_id = o.item_id AND p.display_name <> ''))
        ORDER BY o.item_id LIMIT ?`,
    )
    .bind(SUBJECT_ERASE_OWED_AUDIT_ACTION, SUBJECT_ERASE_DONE_AUDIT_ACTION, SUBJECT_ERASE_SWEEP_LIMIT)
    .all<{ passport_id: string; owed_at: string }>();
  for (const { passport_id: id, owed_at: owedAt } of results) {
    await db.batch([
      guardedAuditStmt(db, "system:subject-erase-sweep", SUBJECT_ERASE_DONE_AUDIT_ACTION, id, "subject_request: hold released; owed names erased", doneAt(nowIso, owedAt), SUBJECT_ERASE_GUARD, [id, id]), // I65-HOLD F2
      ...erasePassportNamesStatements(db, id, SUBJECT_ERASE_GUARD, [id, id]),
    ]);
  }
  const taken = results.length + (await sweepRetractionNamesOwed(db, nowIso)) + (await sweepRenameNamesOwed(db, nowIso)) + (await sweepRemovalNamesOwed(db, nowIso)); await sweepFaceRevokeOwed(db, nowIso); return taken; // I64-HOLD H2 (W0 am.4): one arm per door; I65-HOLD F1: the face-data arm runs last and is not in the names count
}

/** I47-CODE B39 (I47-SCOUT-A/FINDINGS.jsonl I47-B39; the creator-close.ts JA-13 shape): journal the passport's

/**
 * Ops: suppress/unsuppress a passport's request surface (idempotent).
 *
 * THE FREEZE RIDES THE SUPPRESSION, AT EVERY WRITER, INCLUDING THIS ONE.
 * `request_suppressions` is keyed on the passport; the account keys it protects
 * are frozen into `suppressed_account_keys` (migration 0028) so the "never
 * re-seeded" promise stops hanging off `passport_accounts.passport_id`, a
 * MUTABLE pointer that `account_transferred` moves. `removal.ts` has written
 * both halves in one batch since 0028 landed. This writer did not: it ran a bare
 * `.run()` and left the promise, for the ops toggle alone, resting on the
 * union's weaker leg.
 *
 * That was correctness-neutral — `isSuppressedAccountKey` is a UNION of the
 * snapshot and the live join, and the live leg still covered a suppression with
 * no snapshot (which is exactly what `tests/invariants/removal.test.ts` pins,
 * and that leg and that test both stay). It was not ATOMICITY-neutral: two
 * writes with no transaction between them means a suppression can exist without
 * its freeze, and "the guarantee currently holds through the other leg" is not
 * the same claim as "the guarantee is written down". Both halves now land in one
 * batch, through the SAME shared statements every other door uses — no second
 * spelling of the INSERT to drift.
 *
 * THE LIFT DROPS THE FREEZE, AND DROPS IT FIRST. Without that, adding the
 * snapshot here would have made this toggle ONE-WAY: the `DELETE` below removes
 * the suppression while the frozen keys keep refusing forever, so an operator
 * lifting a removal would find the subject still un-relistable with nothing on
 * screen to say why. The order is the declared unsuppress ceremony's
 * (`unsnapshotSuppressedKeysStmt`, console.ts): snapshot first, suppression
 * second, so a half-applied lift leaves the subject STILL SUPPRESSED rather than
 * listed-but-unprotected — the only fail-closed direction there is here.
 */
export async function setRequestSuppression(
  db: DbLike,
  passportId: string,
  operator: string,
  suppress: boolean,
  now: () => string = () => new Date().toISOString(),
  // I18 #942(a) D12/D13 (D-0946(1)): the caller's rows (the console's audit row) ride the SAME batch as the
  // suppression and its snapshot, so a failing one rolls the delisting (or the lift) back.
  extraStatements: DbStatement[] = [],
  // D-0960(12): true only when this suppression serves the SUBJECT'S OWN request; then the image service's
  // biometric revoke is owed in the suppression's batch (likeness.ts imageConsentOwedOnRemovalStatement, drained by
  // drainImageConsentRevokeOutbox), guarded on the suppression row. An operator suppression for any other reason, and
  // every lift, owes nothing.
  subjectRequest = false,
): Promise<void> {
  if (!PSP_RE.test(passportId)) throw new RequestError("bad passport id");
  const p = await passport(db, passportId);
  if (!p) throw new RequestError("no such passport", 404);
  const ts = now();
  // I18-FXA D12/D13 (RGA-D13-NOOP, D-0949(1)): with joined rows, `changeGateStmt` sits DIRECTLY after the
  // suppression write, so a repeat suppress or a lift of a passport that was not suppressed (0 rows) commits
  // none of them; the door still answers as before.
  const gate = extraStatements.length > 0 ? [changeGateStmt(db)] : [];
  if (suppress) {
    // Guarded on the suppression row existing (it sees the INSERT above inside
    // this batch), exactly as removal.ts's is — an unguarded freeze would
    // refuse a key forever on a suppression that never landed.
    const freeze = () => snapshotSuppressedKeysStmt(db, passportId, operator, ts);
    // Upsert (one row per passport, `gen` bumped on a repeat), so a repeated subject-request suppress owes once.
    const owed = () => (subjectRequest ? [imageConsentOwedOnRemovalStatement(db, passportId, ts, "EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = ?)", [passportId]), faceRevokeOwedStatement(db, passportId, ts, "subject delisting", "EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = ?)", [passportId])] : []); // I65-HOLD F1: the marker under a hold
    // I54-REQ P28 a (INV-069): the subject's own delisting erases what a granted removal erases (passport_names and the
    // display_name fold), in this batch, guarded on the suppression row; an open legal hold on the passport keeps the
    // names (the erase statements change nothing) and writes the OWED marker, placed before the erase, so
    // sweepSubjectEraseOwed erases them after the release (I54-FX-PRIV REQ-02).
    const names = () => (subjectRequest ? erasePassportNamesStatements(db, passportId, SUBJECT_ERASE_GUARD, [passportId, passportId]) : []);
    const owedMarker = () =>
      subjectRequest
        ? [guardedAuditStmt(db, operator, SUBJECT_ERASE_OWED_AUDIT_ACTION, passportId, "subject_request: names kept under an open legal hold; erasure owed on release", ts, `EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = ?) AND ${PASSPORT_LEGAL_HOLD_SQL} AND ${NAMES_REMAIN_SQL}`, [passportId, passportId, passportId, passportId])]
        : [];
    try {
      await db.batch([
        db.prepare("INSERT INTO request_suppressions (passport_id, operator, created_at) VALUES (?, ?, ?) ON CONFLICT(passport_id) DO NOTHING").bind(passportId, operator, ts),
        ...gate,
        freeze(),
        ...owed(),
        ...owedMarker(),
        ...names(),
        ...extraStatements,
      ]);
    } catch (e) {
      if (!isChangeGateRefusal(e)) throw e;
      // Already suppressed: the freeze still runs (it did before the gate); it is guarded on the suppression row,
      // so it is safe whatever landed in between, and it is not the audited change. A subject's request on an
      // already-suppressed passport still owes the revoke (same guard). I20-FXC (I20-RG-SEC-09): the caller's
      // rows (its suppress audit) are dropped here, so the subject's request is recorded by its OWN audit row in
      // this batch, beside the revoke it queues; an operator suppress for any other reason still writes nothing.
      const subjectAudit = subjectRequest ? [auditStmt(db, operator, SUBJECT_REVOKE_AUDIT_ACTION, passportId, "subject_request: already delisted; face erasure owed", ts)] : [];
      await db.batch([freeze(), ...owed(), ...owedMarker(), ...names(), ...subjectAudit]);
    }
    // I25-Q8, I47-CODE B39 (JA-13 shape): reached on a committed batch AND on the change-gate refusal, so a repeat
    // suppress after a journal write that failed both ways still journals it, under the row's created_at (one key).
    await journalSuppressionRow(db, passportId);
  } else {
    try {
      await db.batch([
        unsnapshotSuppressedKeysStmt(db, passportId),
        db.prepare("DELETE FROM request_suppressions WHERE passport_id = ?").bind(passportId),
        ...gate,
        ...extraStatements,
      ]);
      // D-1037(9) RECORDS-08 (I29-FX-ERASE H1): the lift committed; journal it so a restore's replay does not
      // re-apply the suppression it lifted (restore-journal.ts replayOne "suppression_lift"). Never throws.
      await journalRestoreAct("suppression_lift", passportId, ts);
    } catch (e) {
      if (!isChangeGateRefusal(e)) throw e;
    }
  }
}

/**
 * Turn the public demand counter off (or back on) for one passport. Idempotent
 * both ways. `actor` is the passport's own id when a claimed creator does it
 * themselves and an operator email when an operator does it on a pre-claim
 * subject's request; it is recorded so "who switched this off" is answerable
 * later, which is the whole difference between a policy and a claim about one.
 *
 * Turning it back ON carries NONE of `UNSUPPRESS_WARNING`'s hazard: a demand
 * opt-out never removes a registry hash, so lifting one republishes nothing that
 * an archiver could correlate. It is an ordinary reversible setting, and saying
 * so is worth a sentence because the adjacent lever is not.
 */
export async function setDemandOptOut(
  db: DbLike,
  passportId: string,
  actor: string,
  optOut: boolean,
  now: () => string = () => new Date().toISOString(),
  // I18 #942(a) D14 (D-0946(1)): the caller's rows join the toggle's batch, gated on the toggle changing a row.
  extraStatements: DbStatement[] = [],
): Promise<void> {
  if (!PSP_RE.test(passportId)) throw new RequestError("bad passport id");
  const p = await passport(db, passportId);
  if (!p) throw new RequestError("no such passport", 404);
  const write = optOut
    ? db.prepare("INSERT INTO demand_optouts (passport_id, actor, created_at) VALUES (?, ?, ?) ON CONFLICT(passport_id) DO NOTHING").bind(passportId, actor, now())
    : db.prepare("DELETE FROM demand_optouts WHERE passport_id = ?").bind(passportId);
  // I18-FXA D14 (RGA-D13-NOOP, D-0949(1)): a repeat opt-out (ON CONFLICT DO NOTHING) or an opt-in with no
  // opt-out stored changes 0 rows, and `changeGateStmt` keeps the joined rows from committing for it.
  try {
    await db.batch(extraStatements.length > 0 ? [write, changeGateStmt(db), ...extraStatements] : [write]);
  } catch (e) {
    if (!isChangeGateRefusal(e)) throw e;
  }
}

/**
 * Request an unclaimed creator (idempotent; the PK makes the counter count
 * distinct accounts). Claimed/verified passports can't be requested — the ask
 * has already been answered — and retracted passports don't exist publicly.
 */
export async function requestCreator(
  db: DbLike,
  accountId: string,
  passportId: string,
  now: () => string = () => new Date().toISOString(),
  dailyCap: number = REQUEST_CAP_PER_DAY,
): Promise<void> {
  await requireAccount(db, accountId);
  // I62 P6 (D-1522): the daily allowance is spent BEFORE any passport read, so it answers the same whoever is asked
  // for (no oracle) and an idempotent repeat spends it too, as the nominate door's does (nominations.ts fileNomination).
  if (!(await consumeRateLimit(db, `reqday:${accountId}`, Date.parse(now()), dailyCap))) throw new RequestError(REQUEST_DAILY_LIMIT_MESSAGE, 429);
  // Suppressed reads exactly like nonexistent on this endpoint (no oracle: a
  // harasser can't confirm from the API that their target engaged and opted
  // out — review finding). It is resolved by `livePassport`, ONE statement, and
  // it comes FIRST: the suppression check used to sit BELOW the claim-state
  // branch, so a suppressed CLAIMED subject answered 409 "already claimed"
  // where a never-listed one answered 404 — the same delisting oracle one
  // status code further along (oracle sweep, this pass).
  const p = await livePassport(db, passportId);
  if (!p) throw new RequestError("no such creator", 404);
  if (p.claim_state !== "unclaimed") throw new RequestError("already claimed", 409);
  const already = await db.prepare("SELECT 1 AS x FROM creator_requests WHERE account_id = ? AND passport_id = ?").bind(accountId, passportId).first();
  if (already) return; // idempotent
  // Cap AND claim-state guards live INSIDE the insert (no check-then-act
  // window): a claim landing between the read above and this insert can't
  // produce an at-claim-time request row (which would fake "prior" support).
  const res = await db
    .prepare(
      `INSERT INTO creator_requests (account_id, passport_id, created_at)
       SELECT ?, ?, ?
        WHERE (SELECT COUNT(*) FROM creator_requests WHERE account_id = ?) < ?
          AND EXISTS (SELECT 1 FROM passports WHERE passport_id = ? AND claim_state = 'unclaimed' AND retracted = 0)
       ON CONFLICT(account_id, passport_id) DO NOTHING`,
    )
    .bind(accountId, passportId, now(), accountId, MAX_REQUESTS_PER_ACCOUNT, passportId)
    .run();
  if (res.changes === 0) {
    const raced = await db.prepare("SELECT 1 AS x FROM creator_requests WHERE account_id = ? AND passport_id = ?").bind(accountId, passportId).first();
    if (!raced) {
      // AE-08. A zero-change INSERT means one of THREE predicates failed —
      // the per-account cap, `claim_state = 'unclaimed'`, or `retracted = 0` —
      // and only the first is "request limit reached". Re-read with the SAME
      // gate the top of this function already used, so a claim or retraction
      // that landed in the window between that read and this INSERT gets the
      // SAME answer it would have gotten arriving a moment earlier, instead of
      // the fan being told a false thing about their own account's cap.
      const p2 = await livePassport(db, passportId);
      if (!p2) throw new RequestError("no such creator", 404);
      if (p2.claim_state !== "unclaimed") throw new RequestError("already claimed", 409);
      throw new RequestError("request limit reached");
    }
  }
}

/**
 * Withdraw a request (idempotent) — but only while the ask is still OPEN.
 * Once the passport is claimed, the request set is FROZEN: founding-fan rank
 * is computed live over the rows, so post-claim deletions would shift ranks
 * and mint never-revoked founding_fan badges to unbounded accounts (a
 * confirmed exploit in review). Freezing at claim keeps "first N at claim
 * time" true forever. Requests on retracted passports may always be removed.
 */
export async function unrequestCreator(db: DbLike, accountId: string, passportId: string): Promise<void> {
  await requireAccount(db, accountId);
  if (!PSP_RE.test(passportId)) throw new RequestError("bad passport id");
  // TWO PROPERTIES AT ONCE, AND THE TRAP BETWEEN THEM.
  //
  // (1) NO ORACLE. Read through `passport()` this answered 409 "frozen once the
  //     creator claims" for a suppressed CLAIMED subject and 200 for an id that
  //     never existed — the freeze rule leaked the claim state of someone who had
  //     asked to be delisted. So the RESPONSE is decided by `livePassport`: a
  //     suppressed subject takes the never-listed branch and answers 200.
  //
  // (2) THE FREEZE STILL HOLDS. The first repair of (1) also let the DELETE
  //     through, on the reasoning that "the ranks the freeze protects are
  //     computed over a passport that no longer serves". That reasoning is
  //     FALSE: suppression is REVERSIBLE (`POST /ops/requests/:id/unsuppress`),
  //     so a fan could withdraw an early request during a suppression window and
  //     shift the frozen rank set the moment it lifted — reopening the confirmed
  //     founding-fan exploit the freeze was added to close (review MEDIUM 4).
  //
  // THE SHAPE THAT SATISFIES BOTH: match the never-listed RESPONSE without
  // performing the side effect. The freeze moves INSIDE the delete as a
  // `NOT EXISTS` guard — the same "guards live inside the statement, no
  // check-then-act window" idiom `requestCreator` above uses for its insert — so
  // a claimed, unretracted passport is frozen whether or not it is suppressed,
  // whether or not anyone remembered to check, and whether or not a claim lands
  // between the read and the write. A suppressed CLAIMED subject therefore
  // spends the SAME two statements a never-listed id spends, returns the same
  // 200 bytes, and deletes exactly nothing.
  //
  // Retracted passports stay withdrawable (`p.retracted = 0` in the guard) —
  // there is no live listing whose rank set could ever be read.
  const p = await livePassport(db, passportId);
  if (p && p.claim_state !== "unclaimed") {
    throw new RequestError("request history is frozen once the creator claims", 409);
  }
  await db
    .prepare(
      `DELETE FROM creator_requests
        WHERE account_id = ? AND passport_id = ?
          AND NOT EXISTS (SELECT 1 FROM passports p
                           WHERE p.passport_id = ? AND p.retracted = 0 AND p.claim_state <> 'unclaimed')`,
    )
    .bind(accountId, passportId, passportId)
    .run();
}

export interface RequestCard {
  passport_id: string;
  display_name: string;
  claim_state: string;
  /** I54-REQ P40 a: null (withheld) below ROI_K_FLOOR, zero included; milestone and next_milestone leave with it. */
  request_count: number | null;
  milestone: number | null; // highest milestone reached
  next_milestone: number | null; // null once past the last one
}

/**
 * I54-REQ P40 a (INV-081): a PUBLIC count of fans who asked is served at ROI_K_FLOOR or above, else withheld (null),
 * zero included, so every population from nobody to K-1 serves the same bytes. MILESTONES[0] (10) is above the floor,
 * so a reached milestone never names a withheld count.
 */
export function publicDemandCount(n: number): number | null {
  return n >= ROI_K_FLOOR ? n : null;
}

/** The milestone pair for a SERVED count; both null when the count is withheld. */
function milestonesOf(shown: number | null): { milestone: number | null; next_milestone: number | null } {
  if (shown === null) return { milestone: null, next_milestone: null };
  let milestone: number | null = null;
  let next: number | null = null;
  for (const m of MILESTONES) {
    if (shown >= m) milestone = m;
    else {
      next = m;
      break;
    }
  }
  return { milestone, next_milestone: next };
}

/**
 * Public share-card data for a passport's request counter — the thing a fan
 * posts to tag the creator ("N of us want you on Dhwaja"). Also serves the
 * on-page counter. Counts only — never who.
 */
export async function requestCard(db: DbLike, passportId: string, nowMs: number = Date.now()): Promise<RequestCard | null> {
  // Live + unsuppressed on ONE statement (`livePassport`). The gate was correct
  // here already but cost a second round trip for a suppressed subject and one
  // for a never-listed one, behind byte-identical 404s — the same measurable
  // difference `/p/:id` had (review MEDIUM 5).
  //
  // `livePassportWithDemand` adds the narrow opt-out (item #13) to that SAME
  // statement. This function backs /p/:id/requests, the passport page's request
  // bar and both demand-card surfaces (`demand-card.ts` calls it), so one
  // predicate here takes the counter off every one of them at once and none of
  // them can drift.
  const p = await livePassportWithDemand(db, passportId, nowMs);
  if (!p) return null;
  // QUALIFIED requesters, on the SAME single statement the raw count used
  // (item #18). One JOIN, no extra round trip — which matters here because this
  // route is measured for statement-sequence parity between a suppressed
  // subject and a never-listed one (suppression-oracles.test.ts).
  // D-0845(7) DAY-FROZEN: a requester erased TODAY still counts until the UTC
  // day ends — the erasure batch's `creator_request_tombstones` row (written
  // only for a qualified requester, account_data.ts) is added back on the SAME
  // statement, so an erasure cannot move the public count mid-day.
  const c = await db
    .prepare(
      `SELECT COUNT(*) AS n, (SELECT COUNT(*) FROM creator_request_tombstones t WHERE t.passport_id = ? AND t.erased_day >= ?) AS tomb
         FROM creator_requests r ${qualifiedFanJoin("r.account_id")} WHERE r.passport_id = ?`,
    )
    .bind(passportId, dayOf(nowMs), passportId)
    .first<{ n: number; tomb: number }>();
  const shown = publicDemandCount((c?.n ?? 0) + (c?.tomb ?? 0));
  return { passport_id: p.passport_id, display_name: p.display_name, claim_state: p.claim_state, request_count: shown, ...milestonesOf(shown) };
}

export interface CreatorDemand {
  passport_id: string;
  display_name: string;
  claim_state: string;
  claimable: boolean; // unclaimed + live → the claim CTA applies
  /** I54-REQ P40 a: null below ROI_K_FLOOR, zero included; milestones, first_requested_at and recent_requests leave with it. */
  request_count: number | null;
  milestone: number | null;
  next_milestone: number | null;
  first_requested_at: string | null; // earliest request — "wanted since"
  /** requests in the last DEMAND_RECENT_DAYS (momentum); null when the count is withheld, or when it or the older
   * remainder (request_count - recent_requests) is below ROI_K_FLOOR (INV-004 spatial clause 2). */
  recent_requests: number | null;
}

const DEMAND_RECENT_DAYS = 30;

/**
 * S83 — the public creator DEMAND aggregate that backs the claim-CTA page. It
 * uses ONLY privacy-safe signals:
 *  - REQUESTS, the public-by-design demand signal (one row per distinct fan; the
 *    PK dedups, so COUNT(*) is already distinct-account).
 *  - NEVER saves: S22 treats the save list ("who you follow") as private, so a
 *    save COUNT must not surface publicly (rewards.ts F4).
 *  - NEVER a check count: the k-anonymous lookup means the server never learns
 *    who checked whom, so a "how many checked you" number cannot be shown
 *    truthfully — the page says so plainly instead of inventing one.
 * Suppressed passports return null (the harassment opt-out). Every query is a
 * bounded per-passport aggregate on the (passport_id) index — the same shape and
 * cost as the existing requestCard route, so it is safe to serve live.
 */
export async function creatorDemand(db: DbLike, passportId: string, nowMs: number): Promise<CreatorDemand | null> {
  const p = await livePassportWithDemand(db, passportId, nowMs); // live + unsuppressed + not opted out, one statement
  if (!p) return null;
  // Qualified requesters only, on the same two statements (item #18). "Wanted
  // since" moves with it: the earliest QUALIFIED ask, so a burner that arrived
  // first cannot backdate a creator's demand story.
  // D-0845(7) DAY-FROZEN: the count and the "wanted since" DATE hold for the
  // whole UTC day of a requester's erasure — today's tombstones (account_data.ts
  // erasure batch) are added back on the SAME statement. A tombstone carries a
  // day, never an instant, so "first" is the earliest COUNTED DAY over live rows
  // and today's tombstones, served as that day's midnight for BOTH (I5-FX2). A
  // live row's counted day is the later of the ask and the fan's qualification —
  // the same key the tombstone's `created_day` carries (I5-FX1).
  const today = dayOf(nowMs);
  const agg = await db
    .prepare(
      `SELECT COUNT(*) AS n, MIN(MAX(r.created_at, qf.qualified_at)) AS first,
              (SELECT COUNT(*) FROM creator_request_tombstones t WHERE t.passport_id = ? AND t.erased_day >= ?) AS tomb,
              (SELECT MIN(t.created_day) FROM creator_request_tombstones t WHERE t.passport_id = ? AND t.erased_day >= ?) AS tomb_first
         FROM creator_requests r ${qualifiedFanJoin("r.account_id")} WHERE r.passport_id = ?`,
    )
    .bind(passportId, today, passportId, today, passportId)
    .first<{ n: number; first: string | null; tomb: number; tomb_first: string | null }>();
  const n = (agg?.n ?? 0) + (agg?.tomb ?? 0);
  const liveFirst = agg?.first ?? null;
  const tombFirst = agg?.tomb_first ?? null;
  // I5-FX2 (I5RG-PRIV-01/-05): served at DAY precision, live or tombstone alike —
  // the erased requester's instant no longer exists, so an instant here either
  // moves within the erasure's day or marks it with a midnight no live row yields.
  const liveDay = liveFirst === null ? null : liveFirst.slice(0, 10);
  const firstDay = tombFirst !== null && (liveDay === null || tombFirst < liveDay) ? tombFirst : liveDay;
  const first = firstDay === null ? null : `${firstDay}T00:00:00.000Z`;
  // I5-J-K:7 (D-0853(3), D-0845(7)): `recent_requests` is day-frozen like the
  // count. The window is whole UTC days, keyed on the day a request is COUNTED —
  // the later of its own day and its fan's qualification day, the exact key the
  // erasure batch writes as the tombstone's `created_day` (account_data.ts) — and
  // today's in-window tombstones are added back, so an erasure moves it only at
  // the next boundary.
  const cutoffDay = dayOf(nowMs - DEMAND_RECENT_DAYS * 86_400_000);
  const rec = await db
    .prepare(
      `SELECT COUNT(*) AS n,
              (SELECT COUNT(*) FROM creator_request_tombstones t WHERE t.passport_id = ? AND t.erased_day >= ? AND t.created_day >= ?) AS tomb
         FROM creator_requests r ${qualifiedFanJoin("r.account_id")}
        WHERE r.passport_id = ? AND MAX(substr(r.created_at, 1, 10), substr(qf.qualified_at, 1, 10)) >= ?`,
    )
    .bind(passportId, today, cutoffDay, passportId, cutoffDay)
    .first<{ n: number; tomb: number }>();
  const shown = publicDemandCount(n);
  const recent = (rec?.n ?? 0) + (rec?.tomb ?? 0);
  return {
    passport_id: p.passport_id,
    display_name: p.display_name,
    claim_state: p.claim_state,
    claimable: p.claim_state === "unclaimed",
    request_count: shown,
    ...milestonesOf(shown),
    first_requested_at: shown === null ? null : first,
    // recent <= n (the same rows under one more filter), so a withheld count always withholds this too.
    recent_requests: publicDemandCount(recent) !== null && publicDemandCount(n - recent) !== null ? recent : null,
  };
}

export interface TopRequested {
  passport_id: string;
  display_name: string;
  request_count: number;
}

const TOP_REQUESTED_KEY = "rankings/top-requested.json";

/**
 * One qualified requester's vote on the board: THE SAME expression as
 * `ratings.ts fanDayWeightExpr` — the 30-day ramp (FAN_WEIGHT_RAMP_DAYS) in
 * whole UTC days from the qualification day — which is also the weight
 * `account_data.ts` stores on the erasure tombstone. It is spelled here, not
 * imported, because `requests.ts -> ratings.ts` closes a dependency cycle
 * (ratings -> regions -> analytics-cells -> intelligence -> roi -> back-catalog
 * -> requests; tests/invariants/i13-arch-cycles.test.ts). The two spellings are
 * held byte-identical by i16-w03d-top-requested.test.ts; the single home is a
 * hand-off (move it to fan-quality.ts). CONTRIBUTES ONE `?` — the UTC day.
 */
export function boardDayWeightExpr(alias = "qf"): string {
  if (!/^[A-Za-z_][A-Za-z0-9_]*$/.test(alias)) throw new Error(`ratings: not a SQL alias: ${JSON.stringify(alias)}`);
  return `MAX(0.0, MIN(1.0, (julianday(?) - julianday(substr(${alias}.qualified_at, 1, 10))) / ${FAN_WEIGHT_RAMP_DAYS}.0))`;
}

/** I21-Q125 (D-1030 round 2; default pending Jason): the public Most Requested board is built only while this
 * ops_settings key reads "1". Absent or any other value = OFF, and `computeTopRequested` returns [] (so the next
 * `buildTopRequested` writes an empty board). The private per-creator demand count (`creatorDemand`) does not read it. */
export const REQUESTS_TOP_BOARD_KEY = "requests_top_board";

/**
 * Compute "Most Requested Creators" — top unclaimed passports by
 * distinct-account request count. This is the AGGREGATE query, so it runs only
 * from the cron sweep (buildTopRequested), never from a public request: a
 * bare Cache-Control header does not actually edge-cache Worker responses, so
 * serving this query directly would be an unauthenticated DB-load amplifier.
 */
export async function computeTopRequested(db: DbLike, nowIso: string, limit = TOP_REQUESTED_LIMIT): Promise<TopRequested[]> {
  const capped = Math.max(1, Math.min(TOP_REQUESTED_LIMIT, limit));
  // TWO NUMBERS, ON PURPOSE (audit item #18). `score` is the age-weighted sum
  // over QUALIFIED requesters and it decides the ORDER and the ADMISSION — a
  // fresh cohort of accounts is worth a rounding error however many of them
  // there are. `request_count` is the whole number of qualified people and it is
  // the only one PUBLISHED, because "N of us want you on Dhwaja" has to be a
  // count of people; `score` never leaves this function.
  //
  // I16 #147 (D-0920): ONE vote is `boardDayWeightExpr` (= `ratings.ts
  // fanDayWeightExpr`) — the boards' 30-day ramp in whole UTC days, the
  // expression the ratings boards and the erasure tombstone writer
  // (account_data.ts) already share — and the floor is
  // applied to the WEIGHTED sum, so ten just-qualified accounts no longer put an
  // unclaimed subject on the public board. Every weight is <= 1, so the weighted
  // floor implies the headcount floor PS-149 set.
  //
  // I16 #26f (D-0845(7), FAMILY-002 P1): a requester erased TODAY still counts
  // until the UTC day ends, as on `requestCard`/`creatorDemand` — today's
  // `creator_request_tombstones` rows are added back in the SAME statement, each
  // with the `fan_weight` the erasure stored (the live row's weight on that day),
  // so an erasure moves neither the count, the order nor the admission mid-day.
  const today = nowIso.slice(0, 10);
  const { results } = await db
    .prepare(
      `SELECT p.passport_id AS passport_id, p.display_name AS display_name, COUNT(*) AS request_count, ROUND(SUM(u.w), 2) AS score
         FROM (SELECT r.passport_id AS psp, ${boardDayWeightExpr("qf")} AS w
                 FROM creator_requests r ${qualifiedFanJoin("r.account_id")}
               UNION ALL
               SELECT t.passport_id AS psp, t.fan_weight AS w
                 FROM creator_request_tombstones t WHERE t.erased_day >= ?) u
         JOIN rankable_passports p ON p.passport_id = u.psp
        WHERE p.claim_state = 'unclaimed' AND p.retracted = 0
          AND NOT EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = p.passport_id)
          AND NOT EXISTS (SELECT 1 FROM demand_optouts d WHERE d.passport_id = p.passport_id)
          AND EXISTS (SELECT 1 FROM ops_settings o WHERE o.key = '${REQUESTS_TOP_BOARD_KEY}' AND o.value = '1')
        GROUP BY u.psp
        HAVING SUM(u.w) >= ?
        ORDER BY score DESC, request_count DESC, p.passport_id ASC
        LIMIT ?`,
    )
    .bind(today, today, TOP_REQUESTED_FLOOR, capped)
    .all<TopRequested & { score: number }>();
  return results.map((r) => ({ passport_id: r.passport_id, display_name: r.display_name, request_count: r.request_count }));
}

interface RankingStore {
  put(key: string, value: string): Promise<void>;
  get(key: string): Promise<string | null>;
}

/** Cron: precompute the ranking into the object store (like the registry). */
export async function buildTopRequested(db: DbLike, store: RankingStore, now: () => string = () => new Date().toISOString(), nowMs: number = Date.now()): Promise<void> {
  const top = await computeTopRequested(db, new Date(nowMs).toISOString());
  await store.put(TOP_REQUESTED_KEY, JSON.stringify({ generated_at: now(), top }));
}

/** Serve the precomputed ranking (O(1) read + one bounded filter query).
 * Absent → empty, never an error and never a live aggregate. Suppressed
 * passports AND narrow demand opt-outs (0017) are filtered LIVE — a harassment
 * opt-out takes effect on the discovery surface immediately, not at the next
 * cron sweep (review finding), and the same has to be true of the opt-out that
 * exists so a subject does not HAVE to use the harassment one; the query is
 * bounded by the artifact's ≤50 ids and reads both tables on one statement.
 *
 * `db` IS REQUIRED, and that is the fix rather than a style choice (review
 * HIGH 3, applied to the whole class). It was `db?: DbLike` with `if (db && …)`
 * under it, which is the identical shape `serveDiscovery` shipped: the delisting
 * guarantee for a public board resting on one call site remembering to pass an
 * OPTIONAL argument, where deleting it is a change tsc accepts and the board
 * then serves a delisted creator's id and display name. Absent config must
 * disable a feature, never silently open it. */
export async function serveTopRequested(store: RankingStore, db: DbLike): Promise<{ generated_at: string | null; top: TopRequested[] }> {
  // D-1059(11) (I21-Q125, I30-RG-PRIV): the switch is read at SERVE as well as at build, so off is off at once — a board
  // built while it was on is never served after it goes off. Off answers exactly what an absent board answers.
  const on = await db.prepare("SELECT 1 AS on_ FROM ops_settings WHERE key = ? AND value = '1'").bind(REQUESTS_TOP_BOARD_KEY).first<{ on_: number }>();
  if (!on) return { generated_at: null, top: [] };
  const raw = await store.get(TOP_REQUESTED_KEY);
  if (!raw) return { generated_at: null, top: [] };
  let parsed: { generated_at?: string; top?: TopRequested[] };
  try {
    parsed = JSON.parse(raw) as { generated_at?: string; top?: TopRequested[] };
  } catch {
    return { generated_at: null, top: [] };
  }
  let top = Array.isArray(parsed.top) ? parsed.top : [];
  if (top.length > 0) {
    // CHUNKED. `top` is parsed out of an R2 ARTIFACT, so TOP_REQUESTED_LIMIT is
    // a property of whatever wrote that object and is never revalidated here —
    // an older, hand-edited or future artifact carrying more than 100 entries
    // would cross D1's parameter cap and 500 this route (audit v10, L7). The
    // filter this feeds is a DELISTING filter, so failing it open is the worse
    // half: re-derive the bound at the read instead of trusting the artifact.
    //
    // ALL THREE withholding facts are read on ONE statement per chunk (a UNION,
    // not a second loop): the narrow demand opt-out (item #13) must take the
    // board off immediately for exactly the reason a suppression must, and an
    // opt-out that only bit at the next cron sweep would be a promise with a
    // latency.
    //
    // RETRACTION IS THE THIRD, and it was missing (I3-1310). INV-005 names
    // three live filters over a precomputed board — retraction, suppression,
    // opt-out — and `retracted = 0` was applied at BUILD only
    // (`computeTopRequested`), so between two cron sweeps a creator who
    // retracted was still served BY NAME on the public board. Same argument as
    // the opt-out's, and the same remedy: the withholding fact is read live at
    // the serve, from the base table, on the statement that is already running.
    // The `passports` row is the subject's own, so this reads no population it
    // was not already naming.
    //
    // The chunk is now bound THREE times, which is what `inChunks`'s
    // extraBindings budget is for: leaving a third of the IN_CHUNK ceiling per
    // chunk keeps `3 x size <= IN_CHUNK`, so the statement clears D1's
    // parameter cap with the same margin the two-table version had. The
    // divisor and the number of `IN (...)` terms below move together.
    const withheld = new Set<string>();
    for (const chunk of inChunks(top.map((t) => t.passport_id), IN_CHUNK - Math.floor(IN_CHUNK / 3))) {
      const placeholders = chunk.map(() => "?").join(",");
      const { results } = await db
        .prepare(
          `SELECT passport_id FROM request_suppressions WHERE passport_id IN (${placeholders})
            UNION SELECT passport_id FROM demand_optouts WHERE passport_id IN (${placeholders})
            UNION SELECT passport_id FROM passports WHERE passport_id IN (${placeholders}) AND retracted = 1`,
        )
        .bind(...chunk, ...chunk, ...chunk)
        .all<{ passport_id: string }>();
      for (const r of results) withheld.add(r.passport_id);
    }
    if (withheld.size > 0) top = top.filter((t) => !withheld.has(t.passport_id));
  }
  return { generated_at: parsed.generated_at ?? null, top };
}

// --- Archive requests (per-creator back-catalog demand) ---

/**
 * Ask a CLAIMED/VERIFIED creator to catalog their archive. The optional note
 * ("which old content first") is shown only to that creator. Idempotent per
 * (account, creator); a repeat replaces the note.
 *
 * THE REPEAT BRANCH IS A PRIVACY-LOAD-BEARING PATH (FINDING-0010 and
 * FINDING-0012, migration 0074). It must do exactly three things and no others:
 *
 *   - it must NEVER touch `created_at`. That instant is the fan's MEMBERSHIP in
 *     the population, `archiveDemand` counts on it, and moving it would let a
 *     re-request add or remove a fan from a whole-day count.
 *   - it must stamp `note_updated_at` when, and only when, the note TEXT
 *     actually changes. `archiveDemand` bounds the notes read by
 *     COALESCE(note_updated_at, created_at), so an unstamped edit is text
 *     written today served today under an older day's date — which is the
 *     per-fan membership oracle the day-lag exists to remove. And an identical
 *     repeat that DID stamp it would make the stamp itself a second, finer
 *     clock a creator could drive, so an identical repeat writes nothing at all.
 *   - it must CARRY THE OUTGOING TEXT OUT OF THE WAY FIRST, into `prev_note` and
 *     the day that text was current under. Stamping alone lags the arrival of
 *     the new words and simultaneously makes the old ones VANISH from an answer
 *     the creator has already read today — the same identity-to-instant binding
 *     obtained in the opposite direction, and enumerable one named fan at a time
 *     (FINDING-0012). With the outgoing text retained, `archiveDemand` can still
 *     answer "what did this row say at 00:00 today" after the row has moved on,
 *     which is the only way the served note SET can be a function of completed
 *     days rather than of "what nobody has touched since midnight".
 *   - it must not act on a note the caller never sent. `note` is
 *     `string | undefined` and the two are different requests, not two spellings
 *     of one (FINDING-0015): `undefined` is "I am re-asking; I said nothing
 *     about the note", and on this path it must leave the stored note exactly
 *     where it is — no comparison, no UPDATE, no `note_updated_at`. Only an
 *     explicit `""` is a WITHDRAWAL, and it is lagged like every other edit.
 *     Collapsing the two destroyed a fan's words on a request that contained no
 *     instruction to destroy them, which is a consent defect before it is a
 *     privacy one. On the INSERT path below the two DO coincide — a first
 *     request with nothing to say about the note stores `''` — because there is
 *     no existing text for "leave it alone" to preserve.
 */
export async function requestArchive(
  db: DbLike,
  accountId: string,
  passportId: string,
  note: string | undefined,
  now: () => string = () => new Date().toISOString(),
): Promise<void> {
  await requireAccount(db, accountId);
  // Same gate, same order, same reason as `requestCreator`: suppressed reads as
  // never-listed and it is decided BEFORE the claim-state branch. Through
  // `passport()` this route accepted an archive request for a suppressed subject
  // (200) and refused one for an id that never existed (404) — `saveCreator`'s
  // bug in a second route (oracle sweep, this pass).
  const p = await livePassport(db, passportId);
  if (!p) throw new RequestError("no such creator", 404);
  if (p.claim_state === "unclaimed") throw new RequestError("creator has not claimed yet — request the creator instead", 409);
  const clean = typeof note === "string" ? note.trim() : "";
  if (clean.length > NOTE_MAX || hasBadNote(clean)) throw new RequestError(`note must be at most ${NOTE_MAX} printable chars without < or > or invisible formatting characters such as direction marks`);
  // THIS PROBE'S SQL IS LOAD-BEARING AS A STRING, and must stay exactly as it
  // is. `tests/invariants/concurrency-guards.test.ts` rendezvouses five racing
  // requests on this literal (`db.gateOn("SELECT 1 AS x FROM archive_requests
  // WHERE account_id = ? AND passport_id = ?", 5)`) to prove the per-account cap
  // holds under concurrency. Folding the note read into it — `SELECT note` —
  // moved the seam out from under that guard, and the guard did not fail: it
  // HUNG, because the barrier never reached five arrivals. So the note is read
  // by a SECOND statement below, on the repeat path only.
  const already = await db.prepare("SELECT 1 AS x FROM archive_requests WHERE account_id = ? AND passport_id = ?").bind(accountId, passportId).first();
  if (already) {
    // NOTHING WAS SAID ABOUT THE NOTE, SO NOTHING HAPPENS TO IT (FINDING-0015).
    // This returns BEFORE the note is even read: an absent key must not reach
    // the comparison below, where `clean` is `''` and `''` means withdrawal.
    // The probe above is untouched by this guard (it is the concurrency-guards
    // rendezvous and the cap check, neither of which is about the note), so the
    // per-account cap and the idempotency of the request itself are unchanged.
    if (note === undefined) return;
    const current = await db
      .prepare(
        `SELECT note AS note, COALESCE(note_updated_at, created_at) AS note_at, prev_note AS prev_note, prev_note_day AS prev_note_day,
                ${NOTE_PUBLISHED_DAY} AS note_pub_day
           FROM archive_requests WHERE account_id = ? AND passport_id = ?`,
      )
      .bind(accountId, passportId)
      .first<{ note: string | null; note_at: string; prev_note: string | null; prev_note_day: string | null; note_pub_day: string }>();
    // NO-OP ON AN IDENTICAL REPEAT. Not an optimisation: a write here would move
    // `note_updated_at` on demand, and a timestamp a creator can ask a named fan
    // to move is exactly the primitive migration 0074 removes. An empty note
    // repeated over an empty note is the same case.
    if ((current?.note ?? "") !== clean) {
      const stamp = now();
      // PHASE 1 — THE COMMITTING STATEMENT, AND IT IS INVISIBLE (migration 0103,
      // QUEUE #515, Jason D-0653 "stamp it the moment it commits").
      //
      // This used to be the ONE statement, and it published the fan's new words
      // stamped with a clock read taken on the NEAR side of its own `await` —
      // a LOWER bound on the commit, where a calendar cutoff needs an UPPER one.
      // Held at the database seam across midnight, it made `archiveDemand`
      // answer twice inside one UTC day, differently, for a row version that did
      // not exist when that day began (W3-11-R9, I3-2180; reproduced at two
      // instants in `test/w312-b9-archive-commit-stamp.test.ts`).
      //
      // So this statement now touches NO column any reader reads. `stage_note`
      // and `stage_at` are read by nothing except phase 2 and the cron rescue.
      // It may commit at any instant, before or after any boundary, and no
      // served answer moves — which is the whole fix: there is no longer an
      // instant at which a reader can see half of an edit.
      await db
        .prepare("UPDATE archive_requests SET stage_note = ?, stage_at = ? WHERE account_id = ? AND passport_id = ?")
        .bind(clean, stamp, accountId, passportId)
        .run();
      // PHASE 2 — PUBLISHED, STAMPED WITH AN OBSERVED DAY, AND CLAIMED.
      //
      // `now()` here is read AFTER phase 1's `await` has RESOLVED, so it is
      // at-or-after phase 1's commit: the only upper bound on a commit instant a
      // client clock can produce. It stays the INJECTED clock — nothing here is
      // a DB-side `strftime('now')`, which is the remedy Jason refused because
      // it removes the clock this module's day-lag proof is driven by.
      const settledDay = now().slice(0, 10);
      // HAS THE OUTGOING TEXT ALREADY BEEN PUBLISHED UNDER A COMPLETED DAY?
      // If yes, it is what the answer is serving and it must be retained before
      // `note` is overwritten. If no, the row has ALREADY been touched today —
      // `prev_note` already holds this morning's text, and copying today's first
      // edit over it would put text written today into today's answer, which is
      // the original defect one field along (0074 policy 5).
      //
      // THE COMPARISON IS AGAINST `settledDay`, NOT AGAINST THE STAMP'S DAY, and
      // that is the second half of the fix. The retention decision and the day
      // this version is published under are now the SAME day, decided together,
      // after the commit was observed — so there is no instant at which the row
      // is rotated for one day and stamped for another.
      const publishedBefore = (current?.note_pub_day ?? "") < settledDay;
      // `created_at` is deliberately NOT in this SET list. See the docblock.
      //
      // `AND stage_at = ? AND stage_note = ?` IS THE CLAIM, and it is INSIDE the
      // committing statement. Re-reading the row before this write would narrow
      // the race with a concurrent second edit and would not close it; claiming
      // the exact version this call staged does. A concurrent edit that staged
      // different words matches zero rows here and its phase 2 is a no-op, so
      // the two can never both publish and neither can publish the other's text.
      await db
        .prepare(
          `UPDATE archive_requests SET prev_note = ?, prev_note_day = ?, note = ?, note_updated_at = ?, note_settled_day = ?, stage_note = NULL, stage_at = NULL
            WHERE account_id = ? AND passport_id = ? AND stage_at = ? AND stage_note = ?`,
        )
        .bind(
          publishedBefore ? (current?.note ?? "") : (current?.prev_note ?? null),
          publishedBefore ? (current?.note_pub_day ?? "") : (current?.prev_note_day ?? null),
          clean,
          stamp,
          settledDay,
          accountId,
          passportId,
          stamp,
          clean,
        )
        .run();
      // PHASE 3 — THE CORRECTION, AND THE RESIDUAL IT BOUNDS RATHER THAN CLOSES.
      //
      // Phase 2 is the statement that makes something visible, and no client
      // clock can bound its OWN commit before it happens: read the clock again
      // once it has resolved and, if a boundary passed during it, the day it
      // stamped is a day that has already ended. Re-stamp forward, claimed on
      // the exact version and the exact day phase 2 wrote, so this cannot
      // re-date a row a later edit has already moved on from, and so it is a
      // no-op (and costs no subrequest at all) on every write that did not
      // straddle — which is all of them but the ones landing within one D1
      // round trip of 00:00:00Z.
      //
      // WHAT THIS LEAVES: between phase 2's commit and this correction — one
      // round trip — a reader can see the new text under the ended day. That
      // window is irreducible with an INJECTED clock, because the only thing
      // that can stamp a row at the instant it becomes visible is the database
      // itself, and a DB-side stamp is the remedy that was refused: it takes
      // away the injectable `now()` this module's day-lag proof is driven by.
      // The trade is stated, not hidden. Before this change the same window was
      // the whole REMAINDER of the day and did not heal.
      //
      // At most ONE correction is ever needed: it is issued a round trip after
      // phase 2, and two UTC midnights cannot fall inside one round trip.
      const observed = now().slice(0, 10);
      if (observed !== settledDay) {
        const publishedBeforeObserved = (current?.note_pub_day ?? "") < observed;
        await db
          .prepare(
            `UPDATE archive_requests SET prev_note = ?, prev_note_day = ?, note_settled_day = ?
              WHERE account_id = ? AND passport_id = ? AND note_updated_at = ? AND note_settled_day = ?`,
          )
          .bind(
            publishedBeforeObserved ? (current?.note ?? "") : (current?.prev_note ?? null),
            publishedBeforeObserved ? (current?.note_pub_day ?? "") : (current?.prev_note_day ?? null),
            observed,
            accountId,
            passportId,
            stamp,
            settledDay,
          )
          .run();
      }
    }
    return;
  }
  //
  // THE INSERT NEEDS NO STAGING COLUMN, because UNSETTLED is already invisible
  // to every calendar cutoff: `ROW_PUBLISHED_DAY` resolves to '9999-12-31',
  // which no `< <today>` predicate admits. The row is created, the per-account
  // cap still counts it (that cap is a property of the account, not of a served
  // day), and NO served population contains it until the claim below stamps a
  // day that was observed AFTER this statement committed. `now()` inside
  // `.bind(...)` was the membership half of I3-2180 — same seam, same lower
  // bound — and `created_at` keeps that instant because membership is dated by
  // the insert (0074 policy 3); it is simply no longer what decides visibility.
  const stamp = now();
  const res = await db
    .prepare(
      `INSERT INTO archive_requests (account_id, passport_id, note, created_at, settled_day, note_settled_day)
       SELECT ?, ?, ?, ?, ?, ? WHERE (SELECT COUNT(*) FROM archive_requests WHERE account_id = ?) < ?
       ON CONFLICT(account_id, passport_id) DO NOTHING`,
    )
    .bind(accountId, passportId, clean, stamp, ARCHIVE_UNSETTLED, ARCHIVE_UNSETTLED, accountId, MAX_REQUESTS_PER_ACCOUNT)
    .run();
  if (res.changes === 0) {
    const raced = await db.prepare("SELECT 1 AS x FROM archive_requests WHERE account_id = ? AND passport_id = ?").bind(accountId, passportId).first();
    if (!raced) throw new RequestError("request limit reached");
    return;
  }
  // THE CLAIM. `AND settled_day = ?` (UNSETTLED) and `AND created_at = ?` are in
  // the committing statement: they claim the exact row this call inserted, so a
  // retry, a racing duplicate or the cron rescue below cannot re-date a row that
  // is already settled. The day is read after the INSERT's `await` resolved.
  const settledDay = now().slice(0, 10);
  await db
    .prepare("UPDATE archive_requests SET settled_day = ?, note_settled_day = ? WHERE account_id = ? AND passport_id = ? AND created_at = ? AND settled_day = ?")
    .bind(settledDay, settledDay, accountId, passportId, stamp, ARCHIVE_UNSETTLED)
    .run();
  // The same one-round-trip correction the edit path carries, and the same
  // residual. Claimed on `settled_day = ?` (the day the claim above wrote), so
  // it can only ever move a row this call settled, and only forward.
  const observed = now().slice(0, 10);
  if (observed !== settledDay) {
    await db
      .prepare("UPDATE archive_requests SET settled_day = ?, note_settled_day = ? WHERE account_id = ? AND passport_id = ? AND created_at = ? AND settled_day = ?")
      .bind(observed, observed, accountId, passportId, stamp, settledDay)
      .run();
  }
}

/**
 * THE ARCHIVE-DEMAND FLOOR, AND NOTHING ELSE (Jason, 2026-09-05).
 *
 * A population below this is withheld entirely — zero included — on every
 * surface that carries back-catalog demand.
 *
 * IT IS DELIBERATELY NOT `roi.ts ROI_K_FLOOR`, and the two must never be
 * aliased again. They floor different populations with different disclosure:
 * `ROI_K_FLOOR` (5) covers saves, card collectors and collectors, which a fan
 * takes silently and which no creator can induce by asking; this one covers a
 * population a creator can grow BY ASKING ONE NAMED PERSON, which is the whole
 * mechanism FINDING-0009 turned on. A single shared number meant that raising
 * the archive floor silently raised three unrelated ones (and lowering either
 * lowered both), so the choice could not be made for one population without
 * being made for four. Two constants, two decisions.
 */
export const ARCHIVE_DEMAND_FLOOR = 10;

export interface ArchiveDemand {
  /** Distinct fans who asked — counted over WHOLE UTC DAYS ONLY, and k-floored.
   * `null` whenever the population is below `floor`, INCLUDING zero: see the
   * function's docblock for why zero is withheld with the rest. */
  request_count: number | null;
  /** Explicit, so no consumer has to read a null as "nobody asked". */
  withheld: boolean;
  /** `ARCHIVE_DEMAND_FLOOR`. DIAGNOSTIC ONLY — it MUST NOT be rendered, and no
   * served human-readable string may quote it or any other quantity for a
   * withheld population (Jason, 2026-09-05). It used to exist so a client could
   * say "fewer than N"; that sentence is now forbidden, because "fewer than 10"
   * is still a statement about how many fans asked, and because a creator who
   * reads it beside a screen that once said something else learns the
   * transition. It is kept on the wire so a reader can tell a withheld count
   * from a missing field, and so this control's threshold is inspectable
   * without reading the source. */
  floor: number;
  /** The last WHOLE UTC day counted, `YYYY-MM-DD`. Today is never counted.
   * Derived from the CLOCK alone — it carries no fan data, so publishing it
   * cannot be differenced. */
  through: string;
  /** Creator-visible only; no requester identity, and nothing finer than a day —
   * neither in the value nor in the ORDER.
   *
   * `note` is the text this fan's row carried at the START of the current UTC
   * day and `day` is the day that text was written; the SET and its contents are
   * therefore a function of completed days only, so nothing any fan does today
   * changes it (see property 3 on the function below).
   *
   * THE `created_at` ALIAS IS GONE (FINDING-0014). It carried the same
   * `YYYY-MM-DD` string as `day`, which made it a field whose name said
   * "when this fan requested the archive" while its value said "when this text
   * was last written" — two different facts, and clients rendering it by name
   * mis-dated the request. There is one name for one fact now. */
  notes: Array<{ note: string; day: string }>;
  /** THE NOTES WERE WITHHELD WHILE THE COUNT WAS SERVED (RUN-0006).
   *
   * `request_count` is the day-lagged population INCLUDING the fans who erased
   * today (migration 0075), because a count that dropped when one person left
   * would be the delta the day-lag exists to remove. The NOTES cannot follow
   * that population: erasure destroys the words, so the only notes that exist
   * are the LIVE ones, and serving them under a summed count would mean
   * serving notes over a live population below the floor — a creator whose ten
   * fans became one would read that one fan's words beside a count of ten.
   *
   * So the notes are gated on the LIVE count and this flag says which gate
   * fired: `true` means notes exist or existed but the live population is below
   * `floor` and none is served. It is `false` whenever notes are served AND
   * whenever the whole response is withheld — a withheld response says nothing
   * about notes, and a flag that distinguished "withheld with notes" from
   * "withheld without notes" would be one bit about the population the withheld
   * response refused to describe. Renderers show the shared withheld sentence
   * and NEVER "no notes", which would be false for a creator whose fans wrote
   * some. */
  notes_withheld: boolean;
}

/**
 * The creator's view of demand for their archive — and the ONE place the
 * `archive_requests` population is computed. roi.ts, intelligence.ts and
 * back-catalog.ts all call this rather than issuing their own `COUNT(*)`, so the
 * four payloads that carry the number carry the SAME number by construction and
 * cannot be differenced against each other.
 *
 * CALLER-GATING: the HTTP layer must pass the AUTHENTICATED creator's own
 * passport id (or an agency's actively-granted one). Notes come back without
 * requester identity — demand, not a DM channel.
 *
 * ---------------------------------------------------------------------------
 * WHY THIS IS A PRIVACY CONTROL AND NOT A REPORT
 * ---------------------------------------------------------------------------
 * `archive_requests` is `PRIMARY KEY (account_id, passport_id)` (migration 0006)
 * and its insert is `ON CONFLICT … DO NOTHING`, so this count is a DISTINCT-FAN
 * POPULATION that only ever goes up by one person at a time. Served live and
 * unfloored it was two separate breaches of the creator-private half of INV-004:
 *
 *   (a) RESTING — a population of one was served exactly. No attack, no timing:
 *       `request_count: 1` is one identifiable person.
 *   (b) MOVING — a creator who asks one known fan to act and reads before and
 *       after gets a delta of exactly 1, at two instants THEY chose. A k-floor
 *       does nothing about this: it is the identity function above the floor, so
 *       `7 - 6 = 1` no matter how high the floor is (roi.ts frozenRead says this
 *       in its own words).
 *
 * Three properties close them, and all three are load-bearing:
 *
 * 1. DAY-LAG AT THE SOURCE. Only rows created strictly before the start of the
 *    current UTC day are counted. The answer for a given (creator, day) is then
 *    the same at every hour BY CONSTRUCTION, so there is no pair of reads to
 *    difference and the finest delta anyone can obtain is one whole day of fans.
 *    This is `regions.ts audienceRegions` (`AND day < ?`), and the `FrozenAnalytics`
 *    docblock at roi.ts:241-247 argues why removing the primitive at the data model beats timing a
 *    build: no snapshot to be partial, nothing to sweep, no 503. The cost is that
 *    a creator cannot see today's requests until tomorrow — the same cost
 *    `audienceRegions` already pays, and the correct one.
 *
 * 2. K-FLOOR, INCLUDING ZERO. `roi.ts kFloor` passes 0 through; this does not,
 *    and the difference matters. If 0 were served as 0 while 1..4 were withheld,
 *    the TRANSITION from "0" to "withheld" would itself prove that a fan acted —
 *    the oracle rebuilt out of the mask. Every population in 0..floor-1 therefore
 *    produces byte-identical output. (`regions.ts` suppresses a sub-floor total
 *    the same way, zero included.)
 *
 * 3. NOTES DAY-GRANULAR, LAGGED ON THE EDIT TIME, AND WITHHELD ENTIRELY BELOW
 *    THE FLOOR. A served millisecond timestamp re-supplies the instant directly,
 *    so a lagged count beside a live timestamp would close nothing; the value is
 *    truncated to the day HERE rather than clamped in the console (a UI
 *    restriction is not a control — INV-010). The ORDER is day-granular too:
 *    ordering on the raw instant would rank same-day notes by their millisecond,
 *    which is a granularity finer than a day recoverable from the list itself.
 *    And below the floor the array is EMPTY, because one note under a withheld
 *    count proves the population is at least one — the choice is the private
 *    one, at the cost that a creator with a handful of note-writing fans sees
 *    none of them.
 *
 *    THE PREDICATE IS `COALESCE(note_updated_at, created_at)`, NOT `created_at`
 *    (migration 0074, FINDING-0010). The row is not append-only: a repeat
 *    request from an already-counted fan rewrites the note in place. Bounded on
 *    the insert time, text written today was served today under an older day's
 *    date, and two named fans sending the same string minutes apart told the
 *    creator which of them had asked before — the exact per-fan oracle the
 *    day-lag removes from the count.
 *
 *    AND THE ROW IS SERVED AS IT STOOD AT THE START OF TODAY, NOT DROPPED
 *    (FINDING-0012). Lagging the edit by that predicate alone kept the new text
 *    out of today's answer by taking the whole row out of it — so the OLD text,
 *    which the creator had already read this morning, VANISHED within a second
 *    of a named fan re-sending. That is the same binding of a person to an
 *    instant read backwards, it is enumerable one fan at a time, it names which
 *    message belongs to whom, and it has a trace-free confirming probe (send the
 *    exact current text and the no-op above fires, changing nothing). Clearing a
 *    note was the same primitive with a permanent tail. So `requestArchive`
 *    retains the outgoing text in `prev_note` with the day it was current under,
 *    and the read below serves `note` for a row untouched today and `prev_note`
 *    for a row touched today. The served SET — membership as well as dates — is
 *    then a function of COMPLETED DAYS ONLY, which is the property the day-lag
 *    needed and did not have: not "what nobody has touched since midnight", but
 *    "what stood here at midnight". An edit or a clear made today is served from
 *    tomorrow; a fan keeps the ability to correct or withdraw their own words.
 *
 *    THE COUNT ABOVE STAYS ON `created_at`, deliberately. Membership in the
 *    population is the INSERT; if an edit moved a fan across the count's
 *    boundary, a creator could ask one known fan to re-send and watch the total
 *    move, which is FINDING-0009 rebuilt through the note field. The two
 *    predicates measure two different facts and must not be unified.
 */
export async function archiveDemand(db: DbLike, passportId: string, nowMs: number = Date.now()): Promise<ArchiveDemand> {
  // `created_at` is always an ISO-8601 UTC instant (see `requestArchive`), so a
  // lexicographic comparison against the bare day IS "strictly before the start
  // of today, UTC": '…-04T23:59:59.999Z' < '2026-09-05' and
  // '2026-09-05T00:00:00.000Z' is not.
  const startOfToday = dayOf(nowMs);
  const through = dayOf(nowMs - 86_400_000);
  const floor = ARCHIVE_DEMAND_FLOOR;
  // I3-1338 (I22): THE LIVE COUNT AND THE TOMBSTONE COUNT ARE ONE STATEMENT.
  // They were two reads, and an erasure (`deleteAccount`: tombstone INSERT +
  // live DELETE in one batch) committing between them counted the erased fan
  // twice or not at all — one either side of the floor. One statement reads
  // one snapshot, so the erased fan is counted exactly once.
  const c = await db
    // `created_at` -> `ROW_PUBLISHED_DAY` (migration 0103). `created_at` is the
    // writer's clock read taken on the near side of the INSERT's own `await`, so
    // a row whose insert COMMITTED after midnight was counted from the moment it
    // landed, under a day that had already begun — the count moved mid-day, by
    // exactly one, on an act one named fan performs. `settled_day` is a day
    // OBSERVED after that commit; NULL is a pre-0103 row and COALESCEs back to
    // the identical expression, so no previously served count changes.
    .prepare(
      `SELECT (SELECT COUNT(*) FROM archive_requests WHERE passport_id = ? AND ${ROW_PUBLISHED_DAY} < ?) AS n,
              (SELECT COUNT(*) FROM archive_request_tombstones WHERE passport_id = ? AND created_day < ? AND erased_day >= ?) AS t`,
    )
    .bind(passportId, startOfToday, passportId, startOfToday, startOfToday)
    .first<{ n: number; t: number }>();
  const live = c?.n ?? 0;
  // AND THE FANS WHO ERASED TODAY (RUN-0006, FINDING-0013/0017 erase direction,
  // migration 0075). The count above is a predicate over the LIVE table, so
  // `deleteAccount`'s `DELETE FROM archive_requests` took a row a COMPLETED day
  // had already counted straight out of today's answer: a distinct-fan
  // population moving by exactly one, mid-day, on an act one named person
  // performs. The day-lag could not see it because the day-lag asks what
  // SURVIVES rather than what was COUNTED.
  //
  // `archive_request_tombstones` carries the two days and nothing else, so the
  // erased row keeps being counted for the whole UTC day of the erasure and
  // drops at the next boundary, exactly as a request made today appears
  // tomorrow. `created_day < today` is the SAME membership rule as the live
  // count (a row created and erased on the same day was never counted and never
  // is); `erased_day >= today` is what expires it. Both binds are the same
  // `startOfToday`, so there is one boundary, not two.
  const n = live + (c?.t ?? 0);
  // Below the floor NOTHING varies with the population: same count, same empty
  // notes array, same bytes for 0 and for floor-1. `notes_withheld` is false
  // here for the same reason - a withheld response describes nothing.
  if (n < floor) return { request_count: null, withheld: true, floor, through, notes: [], notes_withheld: false };
  // THE NOTES GATE IS THE LIVE COUNT, NOT THE SUM (RUN-0006 brief-attack).
  // Erasure destroys the words, so the notes can only ever be the live fans' -
  // and gating them on the summed count would serve notes over a live
  // population the floor is there to protect: at k = 10 with one erasure the
  // summed count is 10 while nine fans' notes would go out where today's read
  // sends none, and nine erasures of ten would put the last fan's words on the
  // screen under a count of ten. No erasure may cause a note to be served that
  // the pre-erasure read would have withheld. The COUNT still uses the sum
  // (that is the whole fix); only the notes step down to `live`.
  if (live < floor) return { request_count: n, withheld: false, floor, through, notes: [], notes_withheld: true };
  // EACH ROW IS SERVED AS IT STOOD AT THE START OF TODAY, UTC (migration 0074,
  // FINDINGS 0010 and 0012). A note is a piece of text a fan wrote at an
  // instant; the row it lives on is not append-only, and `requestArchive`'s
  // repeat branch rewrites it in place.
  //
  //   untouched today — COALESCE(note_updated_at, created_at) < start of today:
  //     `note` IS the text that stood here at midnight, and the same expression
  //     dates it. (COALESCE falls back to `created_at` for every row that was
  //     never edited, which is every row that existed before 0074.)
  //   touched today — otherwise: `note` is text written TODAY and must not be
  //     served (FINDING-0010), and the row must not simply drop out either
  //     (FINDING-0012), so we serve the text it replaced, which `requestArchive`
  //     put in `prev_note` under the day it was current, `prev_note_day`.
  //
  // The empty string is the "nothing to serve" value in BOTH branches and is
  // filtered identically — a fan who has no note, a fan who cleared one
  // yesterday, and a fan whose row is too new to have said anything at the start
  // of today are one case to the reader, which is what makes the disappearance
  // unobservable rather than merely delayed.
  //
  // `created_at < startOfToday` states the rest of the property: only rows that
  // EXISTED at the start of today can contribute at all. It changes no served
  // value (a row created today falls to the second branch with `prev_note` NULL)
  // and is kept because the property, not the accident, is what must survive an
  // edit to this query.
  const { results } = await db
    .prepare(
      // The three `COALESCE(note_updated_at, created_at)` cutoffs and the
      // `created_at` membership cutoff are now `NOTE_PUBLISHED_DAY` and
      // `ROW_PUBLISHED_DAY` (migration 0103). Same predicate, same "strictly
      // before the start of today, UTC", same ten-character lexicographic
      // identity — over a day that was OBSERVED after the write committed
      // instead of one read before it. A staged edit (`stage_note`) is read by
      // nothing here, which is why holding the write across midnight can no
      // longer move this answer.
      `SELECT note, day FROM (
         SELECT CASE WHEN ${NOTE_PUBLISHED_DAY} < ? THEN note
                     ELSE COALESCE(prev_note, '') END AS note,
                CASE WHEN ${NOTE_PUBLISHED_DAY} < ? THEN ${NOTE_PUBLISHED_DAY}
                     ELSE COALESCE(prev_note_day, '') END AS day
           FROM archive_requests WHERE passport_id = ? AND ${ROW_PUBLISHED_DAY} < ?
       ) WHERE note <> '' AND day <> ''
        ORDER BY day DESC, note ASC LIMIT 100`,
    )
    .bind(startOfToday, startOfToday, passportId, startOfToday)
    .all<{ note: string; day: string }>();
  return { request_count: n, withheld: false, floor, through, notes: results.map((r) => ({ note: r.note, day: r.day })), notes_withheld: false };
}

/**
 * CRON PRUNE for migration 0075's tombstones, and it can never change a served
 * value.
 *
 * Every reader requires `erased_day >= <its day>`: `archiveDemand` today's, and
 * the sweep's day object for T = yesterday (`analytics-read.ts` readLegacyFacts ->
 * `backCatalog` -> `archiveDemand` at the start of T) yesterday's. D-1037(13)
 * (I29-JN-04 PR-02, Jason "keep erasure markers two days"): this deletes
 * `erased_day < <yesterday>`, so the prune is disjoint from both reads for every
 * read day >= yesterday — the hour this runs at is irrelevant, and it is safe in a
 * 6-hourly chain (`run0006-builder-erasure.test.ts` asserts it at four hours).
 *
 * It is not merely hygiene. A row that decides nothing is a row not to keep, and
 * these rows are mutually linkable by `(created_day, erased_day)` for one fan
 * across creators (0075's header): bounding the table to two days of
 * erasures bounds what any future mistake could reconstruct.
 */
export async function pruneArchiveTombstones(db: DbLike, nowMs: number): Promise<number> {
  const res = await db.prepare("DELETE FROM archive_request_tombstones WHERE erased_day < ?").bind(dayOf(nowMs - 86_400_000)).run();
  return res.changes;
}

/**
 * THE RESCUE FOR A ROW STRANDED BETWEEN THE TWO PHASES (migration 0103).
 *
 * `requestArchive` publishes in two statements: an INVISIBLE one that stages the
 * fan's words, then a CLAIMED one that installs them stamped with a day observed
 * after the first committed. That is what makes the served answer a function of
 * completed days. It also means a request that dies between the two — the worker
 * is evicted, the isolate is killed, the second statement throws — leaves the
 * fan's words staged and unpublished, and the creator keeps seeing the previous
 * text. NOT a correctness defect (the served answer is still exactly what it was
 * at the start of the day, and still moves only at a boundary) but a LIVENESS
 * one, and a fan's own words are not something to strand on a crash.
 *
 * TWO SET-BASED STATEMENTS, NO PER-ROW LOOP. This is 2 D1 subrequests per tick
 * whatever the table holds — the figure that matters, because one `scheduled()`
 * invocation has on the order of 1,000 subrequests for all ~40 of its legs
 * (W3-11-R9, I3-2188). A per-row settle would have been a second unbounded
 * sweep in the same file as the one that item was filed against.
 *
 * `stage_at < <today>` and `created_at < <today>` are what keep this off an
 * IN-FLIGHT request: it only ever rescues a write from a STRICTLY EARLIER day,
 * so it can never race a phase 2 that is about to run, and it can never publish
 * something under a day it did not observe. Rows it rescues publish from the day
 * AFTER the tick that rescues them, like every other change.
 *
 * It CAN change a served value — that is its whole purpose, and it is the one
 * way in which it differs from `pruneArchiveTombstones` above.
 */
export async function settleArchivePublications(db: DbLike, nowMs: number): Promise<number> {
  const today = dayOf(nowMs);
  const promoted = await db
    .prepare(
      `UPDATE archive_requests
          SET prev_note     = CASE WHEN ${NOTE_PUBLISHED_DAY} < ? THEN note ELSE prev_note END,
              prev_note_day = CASE WHEN ${NOTE_PUBLISHED_DAY} < ? THEN ${NOTE_PUBLISHED_DAY} ELSE prev_note_day END,
              note = stage_note,
              note_updated_at = stage_at,
              note_settled_day = ?,
              stage_note = NULL,
              stage_at = NULL
        WHERE stage_at IS NOT NULL AND stage_at < ?`,
    )
    .bind(today, today, today, today)
    .run();
  const born = await db
    .prepare("UPDATE archive_requests SET settled_day = ?, note_settled_day = ? WHERE settled_day = ? AND created_at < ?")
    .bind(today, today, ARCHIVE_UNSETTLED, today)
    .run();
  return promoted.changes + born.changes;
}

// --- Achievement sweep hooks (called from social.awardAchievements) ---

/**
 * Achievements earned by this fan's answered requests: for every requested
 * passport that is NOW claimed/verified, early_supporter:<psp>; and if this fan
 * holds one of the FROZEN `founding_fan_ranks` slots for it, founding_fan:<psp>.
 * Requests can only ever be placed on unclaimed passports (guarded inside the
 * insert), the request set FREEZES at claim (unrequest is blocked), and the
 * ranks themselves are materialised once at claim (item #19) — so the badge is
 * a row lookup rather than a live count that a later deletion could shift.
 *
 * `settled` = achievement ids the account already holds. Both awards for a
 * passport are decided in the same sweep, so a passport whose early_supporter
 * is already held is fully settled — skipped entirely (no rank query). That
 * bounds repeat-sweep cost to passports claimed since the last sweep.
 */
// --- Founding Fan ranks: MATERIALISED ONCE, AT CLAIM (audit item #19) --------
//
// WHAT WAS WRONG. Rank was pure request order with no qualification, recomputed
// live on every read as "COUNT(*) of requests at or before mine". Requesting is
// free, and (until this change) the request route had no per-IP throttle at all,
// so an attacker minting FOUNDING_FAN_RANKS accounts captured ALL TEN permanent
// slots on any unclaimed passport before a real fan arrived. The badge is
// rendered publicly WITH ITS NUMBER (`Founding Fan #3`, apps/web/src/views.ts)
// and is redeemable as reverse-code eligibility (`reverse-codes.ts`), so those
// slots have economic value. This is the fan-side mirror of the creator-side
// tightening in `founding.ts` (WINDOW ITEM 10: `claimed` used to qualify for the
// creator Founding badge, and being EARLY absorbed ranks that should have gone
// to being VETTED). Same model, same reasoning.
//
// AND A FREEZE BYPASS THE AUDIT DID NOT HAVE. `unrequestCreator` above puts the
// post-claim freeze INSIDE its DELETE as a `NOT EXISTS`, precisely so a
// withdrawal cannot shift a rank. But `account_data.ts` `deleteAccount` runs a
// bare `DELETE FROM creator_requests WHERE account_id = ?` with no guard at all:
// delete your account after a passport claims and, under the old live COUNT,
// every rank behind you shifted up one and account #11 was handed a
// never-revoked `founding_fan` badge. That is exactly the exploit the freeze
// exists to close, reached through a different door.
//
// THE FIX IS TO STOP COMPUTING IT. A rank is now a ROW in `founding_fan_ranks`,
// allocated ONCE, at claim, over QUALIFIED requesters evaluated AS OF THE CLAIM
// INSTANT — which is stable forever for the same reason the current code
// already relied on (the request set itself freezes at claim). Deletion drift,
// requalification drift and query drift all disappear together, and the shape
// matches the existing scarce-ordinal pattern (`founding.ts` join numbers /
// Founding badges): numbers are rows, allocated once, never reused, never
// re-minted into a slot somebody vacated.
//
// AND A SECOND FREEZE BYPASS, FOUND BY ADVERSARIAL REVIEW (D2) AFTER THE ABOVE
// WAS WRITTEN. "Evaluated as of the claim instant" was IMPLEMENTED as a query
// over live, mutable state: `0020`'s `qualified_at` was a `MIN()` over four
// sub-selects recomputed on every read, so the answer depended on when the
// sweep ran. `fan-auth.ts`'s pending-email swap moves `fan_email.verified_at`
// FORWARD, past the claim; `removeFanEmail` deletes the row; `revokeFanPasskey`
// takes a passkey out of 'active'; erasure deletes `acquisitions`. Any of them,
// between the claim and the sweep, silently forfeited a permanent slot to the
// next requester. Proven by execution: two freezes, identical ledger, identical
// request set, ranks (1, 2) in one and (null, 1) in the other.
//
// `0024_fan_qualification.sql` closes it at the source rather than here: the
// four signals are now RECORDED as they happen into a monotone per-account
// watermark (`fan_qualification`, maintained by triggers on the evidence
// tables, `min()`-upserted so it only ever moves EARLIER), and the view reads
// that instead of recomputing. Every predicate below is therefore a function of
// facts that can no longer change — which is what this block always claimed and
// now is. `foundingFanRankDrift` / `repairFoundingFanRanks` are the operator
// remedy for the rows the old behaviour already wrote wrong; the freeze itself
// stays once-ever.

/**
 * THE CLAIM INSTANT, FROM THE LEDGER — not from the wall clock of whichever
 * sweep happened to get here first. Returns null for a passport that has not
 * been claimed, which is the fail-closed answer (no claim, no slate).
 */
async function foundingClaimInstant(db: DbLike, passportId: string): Promise<string | null> {
  const ev = await db
    .prepare("SELECT MIN(created_at) AS at FROM trust_events WHERE subject = ? AND event_type IN ('claim_proven','verification_granted')")
    .bind(passportId)
    .first<{ at: string | null }>();
  return ev?.at ?? null;
}

/**
 * THE SLATE: who holds which Founding Fan slot on this passport, in rank order,
 * derived from the claim instant alone.
 *
 * ONE DERIVATION, used by the freeze, by the drift audit and by the repair —
 * "two call sites computed the rank with the same expression copied twice; one
 * edit made them disagree about who holds #7" is QUERY DRIFT, and
 * `0020_fan_quality.sql` names it as one of the three things materialising the
 * rank was supposed to end. Copying this query a second time would reintroduce
 * it at the repair path instead.
 *
 * EVERY INPUT IS IMMUTABLE. `creator_requests` freezes at the claim (both
 * writers guard on it); `qualified_fan_accounts.qualified_at` is the monotone
 * watermark `0024_fan_qualification.sql` records as the evidence happens; and
 * `accounts.created_at` is the account's birth. So this function is a pure
 * function of `(ledger, request set)` and returns the same slate however long
 * after the claim it is called, and whatever the fan has done to their email,
 * their passkeys or their acquisitions since. That is the property D2 proved
 * the previous implementation did not have.
 *
 * I3-0965 (D-0829, 0115): a slot also needs a COSTED qualification instant
 * (`fan_costed_qualification`, monotone like the plain watermark) — a passkey
 * alone is free to mint and no longer takes a slot. D-0844(2): the 7-day floor
 * runs from that costed instant (`fanCostedAgedEnoughExpr`).
 */
async function foundingFanSlate(db: DbLike, passportId: string, claimedAt: string): Promise<string[]> {
  const { results } = await db
    .prepare(
      `SELECT r.account_id AS account_id
         FROM creator_requests r
         ${qualifiedFanJoin("r.account_id")}
        WHERE r.passport_id = ?
          AND r.created_at <= ?
          AND ${fanQualifiedAsOfExpr()}
          AND ${fanAgedEnoughExpr()}
          AND ${fanCostedAgedEnoughExpr()}
        ORDER BY r.created_at ASC, r.account_id ASC
        LIMIT ?`,
    )
    .bind(passportId, claimedAt, claimedAt, claimedAt, claimedAt, FOUNDING_FAN_RANKS)
    .all<{ account_id: string }>();
  return results.map((r) => r.account_id);
}

/** Is this passport in a state where slots may be allocated at all? */
async function foundingFreezeEligible(db: DbLike, passportId: string): Promise<boolean> {
  const p = await db.prepare("SELECT claim_state, retracted FROM passports WHERE passport_id = ?").bind(passportId).first<{ claim_state: string; retracted: number }>();
  // Only a LIVE, CLAIMED passport freezes: an unclaimed one is still taking
  // requests (the set is not final), and a retracted one has no public surface
  // whose ranks anyone could read.
  return !!p && p.retracted !== 1 && p.claim_state !== "unclaimed";
}

/**
 * Freeze this passport's Founding Fan ranks. Idempotent and once-ever: if ANY
 * row exists for the passport the whole function is a no-op, so a re-run, a
 * concurrent sweep and a lazy call from an achievement read all converge.
 *
 * `frozen_at` records the CLAIM instant, so a re-run — on another machine, a
 * week later — writes byte-identical rows.
 *
 * Returns how many slots this call allocated (0 if already frozen, not
 * eligible, or nobody qualified).
 */
export async function freezeFoundingFanRanks(db: DbLike, passportId: string, now: () => string = () => new Date().toISOString()): Promise<number> {
  if (!PSP_RE.test(passportId)) return 0;
  // THE DECISION IS ITS OWN ROW (S-070 SJ-13, migration 0093). This used to ask
  // `founding_fan_ranks` — "does any AWARD exist for this passport?" — as the
  // proof that the decision had been taken, and an award is a surrogate for a
  // decision, not the decision. It was wrong in both directions: an EMPTY slate
  // wrote nothing at all, so the passport stayed in the cron sweep's head batch
  // for ever and a full batch of empty decisions walled off every eligible
  // passport behind it; and erasure of EVERY holder deleted the last award, so
  // the check went false again and a later sweep re-allocated ordinals that had
  // already been spent, under the same public numbers that
  // `0020_fan_quality.sql` and `account_data.ts` both promise are never reused.
  //
  // `founding_freezes` is passport-keyed and carries no account identity, which
  // is exactly why it survives fan erasure: every deletion in `deleteAccount` is
  // `WHERE account_id = ?`, so nothing in the erasure path can reach it.
  //
  // The award-row check is KEPT beside it, as an OR: on a deployment where 0093
  // has not yet been applied — or where its backfill did not run — an existing
  // slate must still read as decided rather than be re-allocated by this
  // function. It is belt-and-braces, not the primary; after the backfill it is
  // never the clause that fires.
  const already = await db
    .prepare(
      `SELECT 1 AS x FROM founding_freezes WHERE passport_id = ?
        UNION SELECT 1 FROM founding_fan_ranks WHERE passport_id = ? LIMIT 1`,
    )
    .bind(passportId, passportId)
    .first();
  if (already) return 0; // frozen means frozen — never recomputed, whatever changed since
  if (!(await foundingFreezeEligible(db, passportId))) return 0;

  const claimedAt = (await foundingClaimInstant(db, passportId)) ?? now();
  const slate = await foundingFanSlate(db, passportId, claimedAt);

  try {
    // ONE ATOMIC BATCH: the decision row and the awards it decided commit
    // together or not at all. The decision row is FIRST so a reader that sees
    // any award necessarily sees the decision too. An EMPTY slate is still a
    // decision and still writes its row — `slots = 0` — which is the whole
    // point: "decided, and nobody qualified" and "not yet decided" stop being
    // the same state.
    // I3-0460 / I3-1219 (I22): THE ELIGIBILITY AND THE ACCOUNT RIDE THE MINTING
    // STATEMENTS (INV-058). `foundingFreezeEligible` and the slate are reads taken
    // before this batch; a retraction, a suppression or a fan's erasure landing
    // in between used to mint a permanent public ordinal anyway (for a hidden
    // creator, or for an account that no longer exists). Every INSERT now
    // re-asks the live predicate itself, and `slots` is re-derived from the
    // ranks that actually landed, by the batch's last statement.
    const live = `EXISTS (SELECT 1 FROM passports p WHERE p.passport_id = ? AND p.retracted = 0 AND p.claim_state <> 'unclaimed'
                          AND NOT EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = p.passport_id))`;
    const res = await db.batch([
      db.prepare(`INSERT INTO founding_freezes (passport_id, frozen_at, slots) SELECT ?, ?, ? WHERE ${live}`).bind(passportId, claimedAt, slate.length, passportId),
      ...slate.map((accountId, i) =>
        db
          .prepare(
            `INSERT INTO founding_fan_ranks (passport_id, account_id, rank, frozen_at) SELECT ?, ?, ?, ?
              WHERE ${live} AND EXISTS (SELECT 1 FROM accounts a WHERE a.account_id = ?)
             ON CONFLICT(passport_id, account_id) DO NOTHING`,
          )
          .bind(passportId, accountId, i + 1, claimedAt, passportId, accountId),
      ),
      db
        .prepare("UPDATE founding_freezes SET slots = (SELECT COUNT(*) FROM founding_fan_ranks WHERE passport_id = ?) WHERE passport_id = ? AND frozen_at = ?")
        .bind(passportId, passportId, claimedAt),
    ]);
    // What landed, read off the batch's own per-statement results (`changes`,
    // which `d1().batch` lifts from D1's `meta` — I3-1987 a — or `meta.changes`
    // on a handle that hands D1's shape back raw) rather than by a sixth
    // statement, so `FOUNDING_FREEZE_D1_CALLS` still holds.
    return (res as { changes?: number; meta?: { changes?: number } }[])
      .slice(1, 1 + slate.length)
      .filter((r) => (r?.changes ?? r?.meta?.changes ?? 0) > 0).length;
  } catch (e) {
    // A concurrent freeze got there first — on `founding_freezes`' PRIMARY KEY
    // now as well as on UNIQUE(passport_id, rank); the batch is atomic, so
    // nothing half-landed. Their rows are computed from the same frozen inputs,
    // so they are ours.
    //
    // SJ-14: this catch is the one that must NOT eat a transient failure, and
    // its caller is why. `requestAchievements` calls this function and then
    // reads the rank; a swallowed batch error returns 0, the caller finds no
    // rank, persists `early_supporter` alone, and every later scan SKIPS the
    // passport on the strength of that award — so a fan whose rank cron creates
    // five minutes later never receives the badge they earned. A collision is a
    // decision somebody else completed (safe to report as 0); anything else is
    // unfinished work and is propagated so the caller does not settle it.
    // `db.batch` surfaces a constraint refusal as an ordinary Error, so the
    // narrowing is on the message, the same test `social.ts` and
    // `creator-payout.ts` use at their own UNIQUE doors.
    if (e instanceof Error && /UNIQUE|constraint|PRIMARY KEY/i.test(e.message)) return 0;
    throw e;
  }
}

/** One passport's frozen slate against the slate its inputs imply. */
export interface FoundingRankDrift {
  readonly passport_id: string;
  /** The claim instant the slate is derived from; null if the passport never claimed. */
  readonly claimed_at: string | null;
  /** Account ids in rank order, as `founding_fan_ranks` currently holds them. */
  readonly frozen: string[];
  /** Account ids in rank order, as the (now immutable) inputs imply. */
  readonly expected: string[];
  /** True when the two differ — a row written by the pre-`0024` freeze. */
  readonly drifted: boolean;
  /**
   * OP-06. True when `frozen`'s ranks are NOT the contiguous run `1..frozen.length`
   * — a rank once written is now missing while a HIGHER rank still survives.
   * That shape is what a winner's OWN erasure leaves (`account_data.ts
   * deleteAccount` deletes exactly that account's `founding_fan_ranks` row and
   * their `creator_requests` row; the spent slot is policy-unfilled, never
   * recycled — see the comment at its `DELETE FROM founding_fan_ranks` call).
   * The pre-`0024` freeze bug this repair exists for produces the OTHER shape:
   * a slate under-frozen from rank 1 with no internal gap (fewer people, same
   * packing). The two are indistinguishable from `expected` alone, because
   * `foundingFanSlate` recomputes live over the CURRENT `creator_requests` /
   * `qualified_fan_accounts` state — with the erased winner's request row gone,
   * it happily slides the requester the cap excluded at claim time into the
   * vacated slot. `repairFoundingFanRanks` refuses to write when this is true.
   */
  readonly erasureGap: boolean;
}

/**
 * THE REPAIR PATH, half one: READ-ONLY. Which passports' frozen slates disagree
 * with what their inputs imply?
 *
 * WHY THIS EXISTS AT ALL. The freeze is once-ever and has no repair path by
 * design — a permanent public ordinal that a sweep can rewrite is not
 * permanent. But D2 was in production behaviour, and the reviewer's run ended
 * with a fan who WAS qualified at claim time holding `rank = null` for ever and
 * rank #1 minted to the person behind them. "Once-ever" is the right shape for
 * the NORMAL path and the wrong answer for rows a defect already wrote: there
 * is no honest way for the affected fan to get their slot back, because nothing
 * ever recomputes. So the remedy is deliberately split in two — this half only
 * READS and only REPORTS, and is safe to run against production on a schedule;
 * `repairFoundingFanRanks` is the half that writes.
 *
 * THERE IS NO DOOR, and this sentence used to say there was one ("an operator
 * invokes it per passport after reading this"), which reads as a mechanism and
 * is not one. Census at HEAD —
 *   grep -rn 'sweepFoundingFanRankDrift\|repairFoundingFanRanks' --include=*.ts services apps packages | grep -v '\.test\.'
 * — returns DECLARATIONS AND COMMENTS ONLY, in this one file (this comment is
 * among them, so the line count moves; the invocation count does not). No
 * route, no `scheduled()` entry, no console action, no script under
 * `scripts/`, and no non-test caller anywhere: the invocation count outside the
 * tests is ZERO, and the de facto invocation is therefore hand SQL, which is
 * not what either this docblock or DEPLOY.md §2a describes. Wiring it is a
 * decision (it is an unaudited direct write over a permanent public ordinal),
 * so what is corrected here is the CLAIM, not the reachability.
 *
 * A drifted slate is not necessarily this defect (a passport frozen before its
 * requesters aged past `FOUNDING_FAN_MIN_AGE_DAYS` also reports drift), which
 * is exactly why the write is not automatic.
 */
export async function foundingFanRankDrift(db: DbLike, passportId: string): Promise<FoundingRankDrift | null> {
  if (!PSP_RE.test(passportId)) return null;
  const { results: frozenRows } = await db
    .prepare("SELECT account_id, rank FROM founding_fan_ranks WHERE passport_id = ? ORDER BY rank ASC")
    .bind(passportId)
    .all<{ account_id: string; rank: number }>();
  const claimedAt = await foundingClaimInstant(db, passportId);
  const frozen = frozenRows.map((r) => r.account_id);
  const expected = claimedAt === null ? [] : await foundingFanSlate(db, passportId, claimedAt);
  // OP-06: a gap inside the WRITTEN ranks (rank 5 missing while rank 6
  // survives) can only come from a row that once existed being deleted —
  // `freezeFoundingFanRanks` always packs from rank 1 with no internal holes.
  //
  // THAT LIMB SEES AN INTERNAL HOLE ONLY, and erasure of the HIGHEST occupied
  // rank leaves the contiguous run `1..n-1` behind it — indistinguishable, from
  // the award rows alone, from a slate that was never longer. The count the
  // decision allocated is recorded away from the award rows, in
  // `founding_freezes.slots` (written at the freeze's own batch, bound to
  // `slate.length`), on a passport-keyed table nothing in the erasure path can
  // reach — which is what makes it the only in-database witness to a row that
  // is gone. So `frozenRows.length < slots` is the SAME question the limb above
  // asks, without depending on WHERE the missing rank sat. The pre-0024
  // under-freeze this repair exists for does not trip it: that freeze wrote
  // `slots` from the under-frozen slate it actually inserted, so the two agree.
  //
  // LIMITATION, stated because the code cannot fix it: `0093_founding_freezes`
  // backfills `slots` as `COUNT(*)` over the rows that SURVIVED to migration
  // time, so an erasure from BEFORE that migration is not recorded anywhere and
  // only the internal-hole limb can still see it. A passport with no freeze row
  // at all (backfill not run) falls back to that limb for the same reason.
  const freeze = await db.prepare("SELECT slots FROM founding_freezes WHERE passport_id = ?").bind(passportId).first<{ slots: number }>();
  const allocated = freeze ? freeze.slots : null;
  const erasureGap = frozenRows.some((r, i) => r.rank !== i + 1) || (allocated !== null && frozenRows.length < allocated);
  return {
    passport_id: passportId,
    claimed_at: claimedAt,
    frozen,
    expected,
    drifted: frozen.length !== expected.length || frozen.some((a, i) => a !== expected[i]),
    erasureGap,
  };
}

/** Bounded audit over every passport that already holds frozen slots. A caller's
 * `limit` is clamped to `FOUNDING_DRIFT_SWEEP_MAX`. */
export async function sweepFoundingFanRankDrift(db: DbLike, limit = FOUNDING_DRIFT_SWEEP_MAX): Promise<FoundingRankDrift[]> {
  const n = Math.max(1, Math.min(FOUNDING_DRIFT_SWEEP_MAX, Math.floor(Number.isFinite(limit) ? limit : FOUNDING_DRIFT_SWEEP_MAX)));
  const { results } = await db
    .prepare("SELECT DISTINCT passport_id FROM founding_fan_ranks ORDER BY passport_id ASC LIMIT ?")
    .bind(n)
    .all<{ passport_id: string }>();
  const out: FoundingRankDrift[] = [];
  for (const r of results) {
    const d = await foundingFanRankDrift(db, r.passport_id);
    if (d?.drifted) out.push(d);
  }
  return out;
}

/**
 * THE REPAIR PATH, half two: OPERATOR-INVOKED, one passport at a time.
 *
 * Rewrites this passport's frozen slate to the one its immutable inputs imply,
 * in ONE atomic batch (delete-then-reinsert; `UNIQUE(passport_id, rank)` makes
 * an in-place shuffle impossible to express as an upsert). Returns the drift it
 * acted on, or null when there was nothing to do — so re-running it is a
 * no-op, and running it on a passport that was frozen correctly changes
 * nothing.
 *
 * NOT WIRED TO CRON, deliberately, and not reachable from any route: it moves a
 * permanent, publicly-numbered, economically-valuable integer between two real
 * people, and that is a decision an operator makes with the drift report in
 * front of them (DEPLOY.md §2a). `frozen_at` is re-derived from the ledger, so
 * the repaired rows are byte-identical to what the fixed freeze would have
 * written at the claim.
 */
export async function repairFoundingFanRanks(db: DbLike, passportId: string): Promise<FoundingRankDrift | null> {
  const drift = await foundingFanRankDrift(db, passportId);
  if (!drift || !drift.drifted || drift.claimed_at === null) return null;
  // OP-06: a gapped frozen slate is a legitimately-erased winner's OWN spent
  // slot, not the pre-0024 freeze bug this function exists to fix — see
  // `FoundingRankDrift.erasureGap`. Recomputing and reinserting `expected`
  // here would slide the requester the cap excluded at claim time into that
  // vacated slot and renumber every surviving winner behind it: a real,
  // permanent, publicly-numbered ordinal reassigned to the wrong person, with
  // no concurrent write or corruption needed. Refuse rather than guess; the
  // drift report (`foundingFanRankDrift`) still flags it for a human.
  if (drift.erasureGap) return null;
  if (!(await foundingFreezeEligible(db, passportId))) return null;
  const claimedAt = drift.claimed_at;
  await db.batch([
    // THE DECISION ROW FIRST, and it is rewritten with the slate it decides.
    // `founding_freezes.slots` is "how many ordinals the decision allocated";
    // a repair that changes the slate's LENGTH and leaves the old count behind
    // makes the completion record false, and — since `foundingFanRankDrift`
    // now reads that count to see a trailing erasure — makes every later audit
    // of this passport report a phantom one. Same ordering and same atomicity
    // as the freeze's own batch, for the same reason.
    db
      .prepare(
        `INSERT INTO founding_freezes (passport_id, frozen_at, slots) VALUES (?, ?, ?)
           ON CONFLICT(passport_id) DO UPDATE SET slots = excluded.slots, frozen_at = excluded.frozen_at`,
      )
      .bind(passportId, claimedAt, drift.expected.length),
    db.prepare("DELETE FROM founding_fan_ranks WHERE passport_id = ?").bind(passportId),
    ...drift.expected.map((accountId, i) =>
      db.prepare("INSERT INTO founding_fan_ranks (passport_id, account_id, rank, frozen_at) VALUES (?, ?, ?, ?)").bind(passportId, accountId, i + 1, claimedAt),
    ),
  ]);
  return drift;
}

/** D1 calls `freezeFoundingFanRanks` spends on ONE passport, worst case: the
 * freeze probe, the eligibility read, the claim instant, the slate and the
 * write batch. A batch is ONE subrequest however many statements it carries,
 * which is the unit `alerts.ts` and `creator-payout.ts` budget in. Measured,
 * not counted by eye: `w311-regression-w3-11-r9-founding-sweep-budget.test.ts`
 * divides a tick's real spend by the passports it decided and asserts this. */
export const FOUNDING_FREEZE_D1_CALLS = 5;

/** This leg's share of the ~1000 subrequests ONE `scheduled()` invocation has
 * for all of its legs together (index.ts:151). Same order as `ratings.ts`
 * RATINGS_BATCH, and deliberately far below the ceiling: the chain that calls
 * this runs it among roughly forty other legs drawing on the same budget. */
export const FOUNDING_FAN_SWEEP_BUDGET = 150;

/** D1 calls `foundingFanRankDrift` spends on ONE passport, worst case: the
 * frozen rows, the claim instant, the slate and the decision row's `slots`. */
export const FOUNDING_DRIFT_D1_CALLS = 4;
/** I3-2188 b (I22): passports one drift audit may read, DERIVED from the same
 * per-leg budget the freeze sweep spends (`FOUNDING_FAN_SWEEP_BUDGET`, above)
 * less the audit's own selection. It was a round 200 — 801 subrequests, past
 * the ~1000 one invocation has for every leg together. */
export const FOUNDING_DRIFT_SWEEP_MAX = Math.floor((FOUNDING_FAN_SWEEP_BUDGET - 1) / FOUNDING_DRIFT_D1_CALLS);


/** Passports one tick may decide: the budget, less the sweep's own selection,
 * over what one passport costs. DERIVED, so the bound cannot drift away from
 * the two figures that justify it — a round literal here is what produced
 * I3-2188. */
export const FOUNDING_FAN_SWEEP_BATCH = Math.floor((FOUNDING_FAN_SWEEP_BUDGET - 1) / FOUNDING_FREEZE_D1_CALLS);

/**
 * Cron: freeze the ranks for any claimed passport that has requests and no
 * frozen slots yet. Bounded, idempotent, and safe to re-run — every row it
 * writes is a pure function of the ledger and the frozen request set, so the
 * sweep's own timing changes nothing.
 *
 * It selects DISTINCT rather than grouping, because this is an award
 * allocation and not a board (the same distinction founding.ts makes), so it
 * reads the base passports table on purpose — a founder's fans are entitled to
 * their ranks. Keeping it out of grouped form also keeps it outside the
 * rankable-view guard in test/founder.test.ts, which is scoped to boards.
 *
 * THE BOUND IS A SPEND, NOT A ROW COUNT (I3-2188). One selected passport costs
 * FOUNDING_FREEZE_D1_CALLS D1 calls, so the old default of 200 ROWS authorised
 * 1 + 200 x 5 = 1001 subrequests for this leg alone — the entire budget one
 * invocation has for every leg in the chain. Measured over a 210-passport
 * population, one tick submitted exactly 1001. A row limit and a spend are a
 * constant apart, and only the spend is comparable to the ceiling, so the
 * default is now derived from the budget instead of being a round number.
 *
 * IT MOVES NO RANK, AND THAT IS THE ONLY REASON IT IS SAFE TO CHANGE HERE. A
 * founding rank is a permanent, publicly-numbered ordinal; renumbering one is a
 * product decision, not a bound. Nothing here renumbers: every row the freeze
 * writes is a pure function of the ledger's claim instant and the request set
 * frozen at it, and `frozen_at` is that claim instant rather than the sweep's
 * clock, so a passport decided on tick 2 receives rows byte-identical to the
 * ones tick 1 would have written. What changes is HOW MANY passports are
 * decided per tick. Nobody's existing number moves, and no new number is
 * derived differently. A fan who reads their achievements before cron reaches
 * their passport is unaffected either way: `requestAchievements` freezes
 * lazily on that read.
 */
export async function sweepFoundingFanRanks(db: DbLike, now: () => string = () => new Date().toISOString(), limit = FOUNDING_FAN_SWEEP_BATCH): Promise<number> {
  const { results } = await db
    .prepare(
      `SELECT DISTINCT r.passport_id AS passport_id
         FROM creator_requests r
         JOIN passports p ON p.passport_id = r.passport_id
        WHERE p.retracted = 0 AND p.claim_state IN ('claimed','verified')
          AND NOT EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = r.passport_id)
          AND NOT EXISTS (SELECT 1 FROM founding_freezes f WHERE f.passport_id = r.passport_id)
        ORDER BY r.passport_id ASC
        LIMIT ?`,
    )
    .bind(limit)
    .all<{ passport_id: string }>();
  let frozen = 0;
  for (const r of results) {
    // PER-PASSPORT ISOLATION, and it is new because the throw is (S-070 SJ-14).
    // `freezeFoundingFanRanks` no longer swallows a non-collision failure — its
    // achievement caller must not settle a decision that did not complete — so
    // the cron leg, which has no such caller and 200 unrelated passports to get
    // through, takes the isolation instead. An unfrozen passport is simply
    // selected again next tick; it is not decided, and now it says so.
    try {
      if ((await freezeFoundingFanRanks(db, r.passport_id, now)) > 0) frozen += 1;
    } catch {
      /* one passport's failure never stops the sweep; retried next run */
    }
  }
  return frozen;
}

/**
 * The fan's FROZEN founding-fan rank for a passport — now a ROW LOOKUP, not a
 * recomputation. Returns null when they hold no slot. The psp is used
 * server-side only; the client sees just the number.
 *
 * Deleting an account can no longer move this: a rank is allocated once and a
 * vacated slot is never re-minted (see the block above, and
 * `0020_fan_quality.sql`).
 */
export async function foundingRank(db: DbLike, accountId: string, passportId: string): Promise<number | null> {
  if (!PSP_RE.test(passportId)) return null;
  const row = await db.prepare("SELECT rank FROM founding_fan_ranks WHERE passport_id = ? AND account_id = ?").bind(passportId, accountId).first<{ rank: number }>();
  const rank = row?.rank ?? 0;
  return rank >= 1 && rank <= FOUNDING_FAN_RANKS ? rank : null;
}

/** Cap on how many founding-fan ranks one read enriches (bounds the per-read
 * COUNT queries for a fan who is a founding fan of many creators). */
export const FOUNDING_RANK_ENRICH_CAP = 50;

export async function requestAchievements(db: DbLike, accountId: string, settled: ReadonlySet<string> = new Set()): Promise<string[]> {
  const { results } = await db
    .prepare(
      // I3-1313: the LIMIT bounds the passports still OWED a decision, never a
      // fixed first 200 of the fan's (up to MAX_REQUESTS_PER_ACCOUNT) requests.
      // A passport is left out once both of its awards are settled for good:
      // early_supporter recorded, and founding_fan either recorded or
      // impossible — the founding decision exists (founding_freezes or any rank
      // row, the freeze's own "decided" test) and holds no slot for this fan (a
      // slot is allocated once and never re-minted). The settled set is read
      // from `achievements`, the table the caller's `settled` comes from
      // (social.ts awardAchievements). Ordered, so the window is deterministic.
      `SELECT r.passport_id AS psp, r.created_at AS at
         FROM creator_requests r JOIN passports p ON p.passport_id = r.passport_id
        WHERE r.account_id = ? AND p.retracted = 0 AND p.claim_state IN ('claimed','verified')
          AND (
            NOT EXISTS (SELECT 1 FROM achievements a WHERE a.account_id = r.account_id AND a.achievement_id = 'early_supporter:' || r.passport_id)
            OR (
              NOT EXISTS (SELECT 1 FROM achievements a WHERE a.account_id = r.account_id AND a.achievement_id = 'founding_fan:' || r.passport_id)
              AND (
                EXISTS (SELECT 1 FROM founding_fan_ranks f WHERE f.passport_id = r.passport_id AND f.account_id = r.account_id)
                OR NOT (EXISTS (SELECT 1 FROM founding_freezes z WHERE z.passport_id = r.passport_id)
                        OR EXISTS (SELECT 1 FROM founding_fan_ranks g WHERE g.passport_id = r.passport_id))
              )
            )
          )
        ORDER BY r.created_at ASC, r.passport_id ASC
        LIMIT 200`,
    )
    .bind(accountId)
    .all<{ psp: string; at: string }>();
  const out: string[] = [];
  for (const { psp } of results) {
    // TWO AWARDS, TWO COMPLETIONS (S-070 SJ-14). `early_supporter` used to
    // settle the whole passport: a `continue` here skipped the founding check
    // with it. Combined with a freeze that swallowed its own failure and
    // returned 0, that made a transient rank-write failure PERMANENT — this
    // scan returned `early_supporter` alone, the caller persisted it, and every
    // later scan skipped the passport on the strength of that award while cron
    // went on to create the rank the fan had earned. Badge and rank then
    // disagreed for ever, with nothing reporting it.
    //
    // The freeze no longer swallows (above), which is the half that stops the
    // partial completion being recorded at all. This is the half that stops an
    // ALREADY-recorded one being permanent: a passport whose early-supporter
    // award is settled is no longer skipped outright — the award is simply not
    // re-offered, and the founding question is asked again. It costs a single
    // indexed row read per settled passport, bounded by the same `LIMIT 200`
    // this query already carries, and the freeze itself short-circuits on one
    // read when the decision exists.
    const settledEarly = settled.has(`early_supporter:${psp}`);
    if (!settledEarly) out.push(`early_supporter:${psp}`);
    if (settled.has(`founding_fan:${psp}`)) continue; // both decided in a prior sweep
    // LAZY FREEZE, then READ. The cron sweep is the primary writer, but this is
    // the S22 lazy achievement sweep and a fan reading their achievements
    // between the claim and the next cron tick should not see a hole. The
    // freeze is once-ever and every input it reads is immutable — the ledger's
    // claim instant, the request set (frozen at that claim), the monotone
    // `fan_qualification` watermark and `accounts.created_at` — so calling it
    // here cannot produce a different answer from calling it in cron.
    //
    // THAT SENTENCE USED TO BE FALSE, and adversarial review D2 proved it by
    // execution. `qualified_at` was a live recomputation over four mutable
    // tables, so WHICH of the two callers got here first decided who held rank
    // #1: a requester who changed their email address in between was silently
    // dropped from the slate and their permanent slot minted to the next
    // person. `0024_fan_qualification.sql` records the four signals as they
    // happen instead of re-deriving them, which is what makes the claim above
    // true rather than aspirational. It is proved, not asserted:
    // `test/founding-freeze-determinism.test.ts` runs the two freezes over an
    // identical ledger and requires an identical slate.
    //
    // Bounded by `settled`: a passport whose FOUNDING award is already decided
    // never reaches this line (the `continue` above). One whose early-supporter
    // award alone is settled DOES, deliberately — see TWO AWARDS, TWO
    // COMPLETIONS. The freeze is idempotent and short-circuits on one read.
    await freezeFoundingFanRanks(db, psp);
    const slot = await db.prepare("SELECT rank FROM founding_fan_ranks WHERE passport_id = ? AND account_id = ?").bind(psp, accountId).first<{ rank: number }>();
    if (slot && slot.rank >= 1 && slot.rank <= FOUNDING_FAN_RANKS) out.push(`founding_fan:${psp}`);
  }
  return out;
}

// I64-HOLD H2 (I63-RG-W2-08; D-1549, D-1555 (4); W0 am.4-7): the owed erase of the names a held retraction, a held erasing
// rename and a held granted removal kept. Each arm is its own SELECT, guard and LIMIT, called from sweepSubjectEraseOwed
// (the cron step "sweepSubjectEraseOwed", index.ts), so a release by any route is caught on the next run. Each passport is
// one batch whose statements re-ask the hold and the arm's own DONE marker, so a hold opened since the read, or a second
// run, erases and writes nothing.
// ---------------------------------------------------------------------------

/* Each arm takes at most SUBJECT_ERASE_SWEEP_LIMIT passports per run (the subject arm's existing bound); the rest are
 * taken by the next run, whose predicate is re-read. */

/**
 * W0 am.5, state-driven: a retracted passport whose retraction (its first retract_passport correction) wrote
 * `retraction:names_erase_owed` at or after it (I64-FX-HOLD ruling 1) or happened while a passport hold was open (the arm
 * held retractions from before the marker rely on), with no hold open now and no `retraction:names_erase_done` at or after the retraction. Erases
 * the names (the shared erase; the fold name is already blank) and writes DONE even when they are already gone (a
 * manual rebuild's erase writes no marker). Returns the passports taken.
 */
async function sweepRetractionNamesOwed(db: DbLike, nowIso: string): Promise<number> {
  const { results } = await db
    .prepare(
      `SELECT p.passport_id AS passport_id, t.created_at AS retracted_at FROM passports p
         JOIN trust_events t ON t.seq = (SELECT r.seq FROM trust_events r WHERE r.subject = p.passport_id AND r.event_type = 'correction' AND CASE WHEN json_valid(r.payload) THEN json_extract(r.payload, '$.action') END = 'retract_passport' ORDER BY r.seq ASC LIMIT 1)
        WHERE p.retracted = 1
          AND NOT ${legalHoldOpenSql("passport", "p.passport_id")}
          AND (EXISTS (SELECT 1 FROM audit_log o WHERE o.item_id = p.passport_id AND o.action = 'retraction:names_erase_owed' AND o.created_at >= t.created_at) OR EXISTS (SELECT 1 FROM legal_holds lh WHERE lh.subject_kind = 'passport' AND lh.subject_id = p.passport_id AND lh.opened_at <= t.created_at AND (lh.released_at IS NULL OR lh.released_at > t.created_at)))
          AND NOT EXISTS (SELECT 1 FROM audit_log d WHERE d.item_id = p.passport_id AND d.action = 'retraction:names_erase_done' AND d.created_at >= t.created_at)
        ORDER BY p.passport_id LIMIT ?`,
    )
    .bind(SUBJECT_ERASE_SWEEP_LIMIT)
    .all<{ passport_id: string; retracted_at: string }>();
  for (const { passport_id: id, retracted_at: at } of results) {
    const guard = `NOT ${PASSPORT_LEGAL_HOLD_SQL} AND NOT EXISTS (SELECT 1 FROM audit_log d WHERE d.item_id = ? AND d.action = 'retraction:names_erase_done' AND d.created_at >= ?)`;
    await db.batch([
      ...erasePassportNamesStatements(db, id, guard, [id, id, at]),
      db
        .prepare(
          `INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT 'system:names-erase-sweep', 'retraction:names_erase_done', 'retraction: hold released; owed names erased', ?, ?, 'system' WHERE ${guard}`,
        )
        .bind(doneAt(nowIso, at), id, id, id, at), // I65-HOLD F2: never before the retraction it answers
    ]);
  }
  return results.length;
}

/** The kept refs of a passport's pending rename OWED rows (no `rename:names_erase_done` at or after them), as a derived
 * table of `value`s, matched by EXISTS (a subquery, not a bound IN list). Binds the passport id once. */
const RENAME_OWED_REFS_SQL = `SELECT j.value FROM audit_log o, json_each(CASE WHEN json_valid(o.detail) THEN o.detail ELSE '[]' END) j
  WHERE o.item_id = ? AND o.action = 'rename:names_erase_owed'
    AND NOT EXISTS (SELECT 1 FROM audit_log d WHERE d.item_id = o.item_id AND d.action = 'rename:names_erase_done' AND d.created_at >= o.created_at)`;

/**
 * W0 am.6: a passport with a pending `rename:names_erase_owed` and no hold open. Deletes only the refs those OWED rows list
 * (never the current name, which is a later ref) and their brand_index name entries, keeping the current name's entry and a
 * handle's entry, as the rename does; then DONE. Returns the passports taken.
 */
async function sweepRenameNamesOwed(db: DbLike, nowIso: string): Promise<number> {
  const { results } = await db
    .prepare(
      `SELECT DISTINCT o.item_id AS passport_id, (SELECT m.created_at FROM audit_log m WHERE m.item_id = o.item_id AND m.action = o.action ORDER BY m.created_at DESC LIMIT 1) AS owed_at FROM audit_log o
        WHERE o.action = 'rename:names_erase_owed'
          AND NOT EXISTS (SELECT 1 FROM audit_log d WHERE d.item_id = o.item_id AND d.action = 'rename:names_erase_done' AND d.created_at >= o.created_at)
          AND NOT ${legalHoldOpenSql("passport", "o.item_id")}
        ORDER BY o.item_id LIMIT ?`,
    )
    .bind(SUBJECT_ERASE_SWEEP_LIMIT)
    .all<{ passport_id: string; owed_at: string }>();
  for (const { passport_id: id, owed_at: owedAt } of results) {
    await db.batch([
      db
        .prepare(
          `DELETE FROM brand_index
            WHERE kind = 'name'
              AND key IN (SELECT ? || '|' || n.display_name FROM passport_names n WHERE n.passport_id = ? AND EXISTS (SELECT 1 FROM (${RENAME_OWED_REFS_SQL}) r WHERE r.value = n.name_ref))
              AND key NOT IN (SELECT ? || '|' || p.display_name FROM passports p WHERE p.passport_id = ?)
              AND NOT EXISTS (SELECT 1 FROM passport_accounts h WHERE h.passport_id = ? AND h.id_kind = 'handle' AND ? || '|' || h.id = brand_index.key)
              AND NOT ${PASSPORT_LEGAL_HOLD_SQL}`,
        )
        .bind(id, id, id, id, id, id, id, id),
      db
        .prepare(`DELETE FROM passport_names WHERE passport_id = ? AND EXISTS (SELECT 1 FROM (${RENAME_OWED_REFS_SQL}) r WHERE r.value = passport_names.name_ref) AND NOT ${PASSPORT_LEGAL_HOLD_SQL}`)
        .bind(id, id, id),
      db
        .prepare(
          `INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT 'system:names-erase-sweep', 'rename:names_erase_done', 'rename: hold released; owed previous names erased', ?, ?, 'system'
            WHERE NOT ${PASSPORT_LEGAL_HOLD_SQL} AND EXISTS (${RENAME_OWED_REFS_SQL})`,
        )
        .bind(doneAt(nowIso, owedAt), id, id, id), // I65-HOLD F2
    ]);
  }
  return results.length;
}

/** The removal arm's guard: no open hold, and a pending `removal:names_erase_owed` written at or after the standing
 * suppression row (no `removal:names_erase_done` at or after it). Binds the passport id three times (I65-HOLD: a hold of either kind). */
const REMOVAL_OWED_GUARD = `NOT ${FACE_DATA_HOLD_SQL} AND EXISTS (SELECT 1 FROM audit_log o JOIN request_suppressions s ON s.passport_id = o.item_id
  WHERE o.item_id = ? AND o.action = 'removal:names_erase_owed' AND o.created_at >= s.created_at
    AND NOT EXISTS (SELECT 1 FROM audit_log d WHERE d.item_id = o.item_id AND d.action = 'removal:names_erase_done' AND d.created_at >= o.created_at))`;

/**
 * W0 am.7: a granted removal that met a hold. Once no hold is open and the suppression still stands, one batch owes the
 * face-data revoke (the outbox upsert), erases the names and writes DONE, every statement under REMOVAL_OWED_GUARD, DONE
 * last so the statements before it still see the OWED pending: a second run finds DONE and writes no second outbox gen.
 * Returns the passports taken.
 */
async function sweepRemovalNamesOwed(db: DbLike, nowIso: string): Promise<number> {
  const { results } = await db

    | {
        put(k: string, v: string, o?: unknown): Promise<unknown>;
        get(k: string): Promise<{ text(): Promise<string> } | null>;
        list(o: { prefix: string; cursor?: string; startAfter?: string }): Promise<{ objects: { key: string }[]; truncated: boolean; cursor?: string }>;
        delete(k: string): Promise<void>;

    pending: tdb && typeof tdb.prepare === "function" ? d1(tdb) : null,
    put: async (key, body) => void (await b.put(key, body, { httpMetadata: { contentType: "application/json" } })),

export interface RestoreReplayResult { checked: number; applied: number; already: number; absent: number; failed: number; next_cursor: string | null }

/**
 * Re-apply every journaled act at or after `since` (T, the restore point) to the restored database. Each act goes
 * through its own door with the replaying operator as actor and a `restore_journal:replay` audit row: a suppression
 * through setRequestSuppression, a retraction as a `retract_passport` correction (the journaled reason code), an
 * operator revocation through revokeOperator; D-1037(9): a suppression lift through setRequestSuppression(false), and a
 * creator erasure / creator wipe through the replayer creator-erasure.ts registers (restore-journal-store.ts
 * registerRestoreReplayer: eraseCreator, and the wipe door's batch, each with the journaled trigger); D-1041(1): a creator close
 * through the replayer creator-close.ts registers (closeCreatorAccount at the journaled instant, before any same-day
 * erasure by key order); D-1042(1): a creator termination (a `creator_close` with reason CREATOR_TERMINATE_JOURNAL_REASON)
 * through the same replayer (terminateCreator at the journaled instant, the replaying operator as actor) — each door
 * keeps its own refusals (a legal hold or unsettled money is a counted failure, never a forced write), except that a
 * creator erasure and a creator wipe re-apply with money unsettled and pay nothing (D-1119(2), D-1120(2)); a legal hold
 * still refuses both. An act already
 * in force is `already`; so is an act SUPERSEDED by its same-day sibling (`supersededSameDay`); a target the restored
 * database does not hold is `absent`; a failure is counted (never the id) and the page is re-run with the same cursor.
 */
export async function replayRestoreJournal(
  db: DbLike,
  store: RestoreJournalStore,
  opts: { since: string; operator: string; cursor?: string | null; now?: () => string },
): Promise<RestoreReplayResult> {
  const now = opts.now ?? (() => new Date().toISOString());
  const page = await store.list({ prefix: RESTORE_JOURNAL_PREFIX, cursor: opts.cursor ?? null, startAfter: `${RESTORE_JOURNAL_PREFIX}${opts.since.slice(0, 10)}` });
  const out: RestoreReplayResult = { checked: 0, applied: 0, already: 0, absent: 0, failed: 0, next_cursor: page.cursor };
  for (const key of page.keys) {
    const k = parseKey(key);
    if (!k) continue;
    let body: { at?: unknown; reason?: unknown } = {};
    try {
      body = JSON.parse((await store.get(key)) ?? "{}") as typeof body;
    } catch {
      body = {};
    }
    const at = typeof body.at === "string" ? body.at : `${k.day}T00:00:00.000Z`;
    if (at < opts.since) continue;
    out.checked += 1;
    try {
      const r = (await supersededSameDay(store, k.kind, k.id, k.day, at, page.keys, page.cursor !== null, Boolean(opts.cursor))) ? "already" : await replayOne(db, k.kind, k.id, typeof body.reason === "string" ? body.reason : null, opts.operator, now, at);
      out[r] += 1;
    } catch (e) {
      out.failed += 1;
      logError({ cid: crypto.randomUUID(), hop: "cron", msg: "restore_journal_replay_failed", err: e instanceof Error ? e.name : typeof e });
    }
  }
  return out;
}

 * by time. The act whose same-day sibling carries a LATER `at` is superseded (the later act decides the state); across
 * days the walk's key order is the time order, so the later act is replayed after the earlier one. The sibling is read
 * only when it can exist: listed in this page, or sorting past this page's end (more pages follow) or before its start
 * (this page is not the first), so a page with no sibling reads no extra key.
 */
const SAME_DAY_SIBLING: Partial<Record<RestoreJournalKind, RestoreJournalKind>> = { suppression: "suppression_lift", suppression_lift: "suppression" };
async function supersededSameDay(
  store: RestoreJournalStore,
  kind: RestoreJournalKind,
  id: string,
  day: string,
  at: string,
  pageKeys: string[],
  morePages: boolean,
  earlierPages: boolean,
): Promise<boolean> {
  const sibling = SAME_DAY_SIBLING[kind];
  if (!sibling) return false;
  const key = restoreJournalKey(sibling, id, `${day}T00:00:00.000Z`);
  const first = pageKeys[0] ?? "";
  const last = pageKeys[pageKeys.length - 1] ?? "";
  if (!pageKeys.includes(key) && !(morePages && key > last) && !(earlierPages && key < first)) return false;
  let body: { at?: unknown } | null = null;
  try {
    body = JSON.parse((await store.get(key)) ?? "null") as { at?: unknown } | null;
  } catch {
    return false;
  }
  return typeof body?.at === "string" && body.at > at;
}

async function replayOne(db: DbLike, kind: RestoreJournalKind, id: string, reason: string | null, operator: string, now: () => string, at: string): Promise<"applied" | "already" | "absent"> {

    return "applied";
  }
  if (kind === "creator_erasure" || kind === "creator_wipe" || kind === "creator_close" || kind === "creator_case") {

      logError({ cid: crypto.randomUUID(), hop: "cron", msg: "restore_journal_replayer_missing", err: kind });
      throw new Error(`no restore replayer registered for ${kind}`);
    }

      kind === "creator_erasure" ? "creator erasure" : kind === "creator_wipe" ? "creator wipe" : kind === "creator_case" ? `creator ${(reason ?? "").split(":")[0]?.replace("_", " ")}` : reason === CREATOR_TERMINATE_JOURNAL_REASON ? "creator termination" : "creator close";
    return replay(db, id, reason, now, [audit(what)], at, operator);
  }
  if (kind === "suppression") {
    if (await db.prepare("SELECT 1 AS x FROM request_suppressions WHERE passport_id = ?").bind(id).first()) return "already";
    if (!(await db.prepare("SELECT 1 AS x FROM passports WHERE passport_id = ?").bind(id).first())) return "absent";
    await setRequestSuppression(db, id, operator, true, now, [audit("suppression")]);
    return "applied";
  }
  if (kind === "retraction") {
    const p = await db.prepare("SELECT retracted FROM passports WHERE passport_id = ?").bind(id).first<{ retracted: number }>();
    if (!p) return "absent";
    if (p.retracted === 1) return "already";

/**
 * D-1043(1) (I29-JUDGE-AUDIT F1): re-put up to `limit` unwritten restore_journal_pending rows, oldest act first. A row
 * whose key the store already holds with an `at` at or after the pending one is marked written without a put (a later
 * act under that key was journaled meanwhile). A put that throws bumps `attempts` and leaves the row for the next tick;
 * nothing is deleted. `more: true` = rows beyond the bound may remain. scheduled() step: I30-INT (HANDOFFS).
 */
export async function sweepRestoreJournalPending(
  db: DbLike,
  store: RestoreJournalStore,
  now: () => string = () => new Date().toISOString(),
  limit = RESTORE_JOURNAL_PENDING_LIMIT,
): Promise<{ written: number; failed: number; more: boolean }> {
  const lim = Math.max(1, Math.min(limit, RESTORE_JOURNAL_PENDING_LIMIT));
  const { results } = await db
    .prepare("SELECT journal_key, body, act_at FROM restore_journal_pending WHERE written_at IS NULL ORDER BY act_at, journal_key LIMIT ?")
    .bind(lim)
    .all<{ journal_key: string; body: string; act_at: string }>();
  let written = 0;
  let failed = 0;
  for (const r of results) {
    try {
      let held: { at?: unknown } | null = null;
      try {
        held = JSON.parse((await store.get(r.journal_key)) ?? "null") as { at?: unknown } | null;
      } catch {
        held = null; // an unreadable object is overwritten with the pending body
      }
      if (!(typeof held?.at === "string" && held.at >= r.act_at)) await store.put(r.journal_key, r.body);
      await db.prepare("UPDATE restore_journal_pending SET written_at = ? WHERE journal_key = ? AND act_at = ? AND written_at IS NULL").bind(now(), r.journal_key, r.act_at).run();
      written += 1;
    } catch (e) {
      failed += 1;
      await db.prepare("UPDATE restore_journal_pending SET attempts = attempts + 1 WHERE journal_key = ?").bind(r.journal_key).run().catch(() => undefined);
      logError({ cid: crypto.randomUUID(), hop: "cron", msg: "restore_journal_unwritten", err: e instanceof Error ? e.name : typeof e });
    }
  }
  return { written, failed, more: results.length === lim };
}

/** backup_window_days (1..3650, default 90) + RESTORE_JOURNAL_GRACE_DAYS. */
export async function restoreJournalRetentionDays(db: DbLike): Promise<number> {
  return (await numericSetting(db, BACKUP_WINDOW_DAYS_KEY, BACKUP_WINDOW_DAYS_DEFAULT, 1, 3650)) + RESTORE_JOURNAL_GRACE_DAYS;
}

/**
 * Delete journal entries whose day is older than restoreJournalRetentionDays. Keys sort by day, so the walk stops at the
 * first kept entry; `more: true` = call again. scheduled() step: I29-INT.
 */
export async function pruneRestoreJournal(db: DbLike, store: RestoreJournalStore, nowMs: number = Date.now()): Promise<{ deleted: number; more: boolean }> {
  const cutoffMs = nowMs - (await restoreJournalRetentionDays(db)) * DAY_MS;
  const page = await store.list({ prefix: RESTORE_JOURNAL_PREFIX });
  let deleted = 0;
  for (const key of page.keys) {
    const k = parseKey(key);
    if (!k) continue; // not a key this module wrote: never deleted on a guess
    if (Date.parse(`${k.day}T00:00:00.000Z`) >= cutoffMs) return { deleted, more: false };
    await store.delete(key);
    deleted += 1;
  }
  return { deleted, more: page.cursor !== null };
}

// A code value: printable, bounded, no markup/control/whitespace-edge. The
// creator's external code — kept to a safe charset so it renders as plain text

const CODE_RE = /^[A-Za-z0-9._%+\-]{1,64}$/;
const ACHIEVEMENT_REF_RE = /^[a-z0-9:_]{1,64}$/;

/**
 * AUDIT ITEM #23 — the achievement namespace is GLOBAL, its members are not.
 *
 * `achievements.achievement_id` is one flat namespace holding two very different
 * kinds of id. Some are CREATOR-SCOPED and name their creator in the string —
 * `early_supporter:<psp>`, `founding_fan:<psp>`, `completion:<psp>`,
 * `milestone:<psp>:<n>` (social.ts). The rest belong to nobody: they are earned
 * against the platform, not against a creator.
 *
 * `createPool` only ever FORMAT-checked an achievement ref — no ownership read
 * at all, while the `acquisition` kind one branch above was ownership-checked
 * against `releases`. So `createPool({ kind: "achievement", ref:
 * "founding_fan:psp_<someone else>" })` was accepted, and the resulting pool
 * dispensed the attacker's codes to the VICTIM's most committed fans while
 * `poolOffer` published both the attacker's `passport_id` and a ref naming the
 * victim to anyone holding the pool URL. attribution.ts already documented this
 * exact hole ("Pool creation only format-checks the achievement ref (never
 * ownership)") and defended only the analytics COUNT; the pool itself was still
 * allowed to exist and dispense. docs/REVERSE-CODES-PLAN.md's design intent was
 * that createPool "validates the release/achievement belongs to / is grantable
 * by the creator" — only the release half had shipped.
 *
 * THE RULE, and it is the same predicate on all four surfaces so they cannot
 * disagree: an achievement ref is grantable by a creator iff it is
 *   (a) SUFFIXED with that creator's own passport id — byte-identical to the
 *       test attribution.ts:eligiblePopulation already uses, so a pool that can
 *       exist is exactly a pool whose audience size its owner may read; or
 *   (b) a member of CREATOR_NEUTRAL_ACHIEVEMENTS below.
 *
 * CROSS-PROMOTION IS NOT BUILT BY RELAXING THIS. If two creators ever want to
 * offer each other's fans a code, that needs a two-sided grant row — a record
 * that the OTHER creator agreed — not a wider string check here.
 */
export const CREATOR_NEUTRAL_ACHIEVEMENTS: ReadonlySet<string> = new Set([
  // Collection tiers (social.ts COLLECTOR_TIERS) — earned across all creators.
  "first_verified_acquisition",
  "collector_10",
  "collector_50",
  "collector_100",
  // Codex tiers (social.ts CODEX_TIERS) — distinct creators encountered.
  "codex_10",
  "codex_50",
  "codex_100",
  "codex_500",
  // Scout tiers (social.ts SCOUT_TIERS) — found creators before they verified.
  "scout",
  "scout_10",
  "scout_50",
  // DELIBERATELY ABSENT: `reporter:1` / `reporter:3` / `reporter:10`. They are
  // creator-neutral in the namespace sense, so (b) would otherwise admit them,
  // and they are excluded anyway: a reporter badge records that this fan filed
  // impersonation reports that were later confirmed, and the whole reports
  // subsystem is built so that a report cannot be turned back on its reporter.
  // A code pool gated on one is a selector for "people who report", offered by
  // an account that may be the subject of a report. Being able to name a
  // population is the first half of targeting it, and the discount is not worth
  // the second half.
]);

/**
 * Is this achievement ref one the creator may gate a pool on? Pure, no DB read
 * — the whole point is that the answer is derivable from the id, which is why
 * the namespace was designed with the passport id IN the string.
 */
export function grantableAchievementRef(ref: string, passportId: string): boolean {
  if (!ACHIEVEMENT_REF_RE.test(ref)) return false;
  if (CREATOR_NEUTRAL_ACHIEVEMENTS.has(ref)) return true;
  if (passportId.length === 0) return false;
  // F4: a creator's CUSTOM badge (`custom:<psp>:<slug>`, minted by a reward
  // policy) carries its owner in the middle — prefix-scoped, exact separator,
  // and NOTHING else decides it (a slug ending in another passport's id must
  // not fall through to the suffix rule).
  if (ref.startsWith("custom:")) return ref.startsWith(`custom:${passportId}:`);
  // Suffix, not `includes`: `completion:psp_A` must not be grantable by psp_B
  // just because a future id shape happens to mention two passports.
  return ref.endsWith(passportId);
}

/** Validate an `achievement` eligibility ref, or throw. One call site per
 * surface (createPool, validateDropPoolInput, buildDropPool) so a fourth pool
 * writer cannot forget it and still compile against the same helper. */
function requireGrantableAchievement(ref: string, passportId: string): void {
  if (!ACHIEVEMENT_REF_RE.test(ref)) throw new ReverseCodeError("bad achievement id");
  if (!grantableAchievementRef(ref, passportId)) {
    // ONE message for both failure modes. "That achievement belongs to another
    // creator" would confirm, to any verified creator, that a guessed
    // `founding_fan:psp_…` is well-formed and whose it is; the refusal must not
    // become the read the refusal exists to prevent.
    throw new ReverseCodeError("you can only gate a pool on your own achievements or a platform-wide one", 403);
  }
}

/** A CSPRNG id for pool + code mints. An earlier `mint()` derived the pool id
 * from `uuid7(nowMs).slice(0,16)` (48 timestamp bits + ~12 random bits), so at a
 * fixed millisecond — many pools minted "at once" — only ~12 bits varied and the
 * `reverse_code_pools.pool_id` PK could birthday-collide (a UNIQUE error surfaced
 * to the creator; a test flake here). Pool ids are opaque capability handles
 * (public via `GET /reverse-codes/:id`) and are NEVER ordered by (listPools orders
 * by `created_at`), so nothing needs them time-sortable. Full 64 random bits is
 * collision-safe for a 1000-code batch AND far less enumerable than a mostly-
 * timestamp id. */
function randId(prefix: string): string {
  const a = new Uint8Array(8);
  crypto.getRandomValues(a);
  let s = "";
  for (const b of a) s += b.toString(16).padStart(2, "0");
  return `${prefix}_${s}`;
}

// Unambiguous base32 (no 0/1/I/O) so a supporter can read a voucher aloud or
// type it without confusion. Every symbol is within CODE_RE, so a generated
// voucher validates on the same path as an uploaded one. 32^12 ≈ 2^60 of
// entropy — collision-safe far past any creator's realistic voucher volume.
const VOUCHER_ALPHABET = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";
const VOUCHER_LEN = 12;

/** Mint one human-transcribable Dhwaja voucher code (CSPRNG, rejection-sampled
 * so every symbol is uniform over the 32-char alphabet). */
function mintVoucher(): string {
  const bytes = new Uint8Array(VOUCHER_LEN);
  crypto.getRandomValues(bytes);
  let out = "";
  for (let i = 0; i < VOUCHER_LEN; i++) {
    // 256 % 32 === 0, so a byte maps to a symbol with no modulo bias.
    out += VOUCHER_ALPHABET[bytes[i]! & 31];
  }
  return out;
}

/** Strip markup + control chars from a pool title (shown to fans in the offer);
 * throws if nothing survives. */
function cleanTitle(raw: unknown): string {
  const t = typeof raw === "string" ? stripUnsafeDisplayText(raw.replace(/[<>]/g, "")).replace(/\s+/g, " ").trim().slice(0, TITLE_MAX) : "";
  if (t.length === 0) throw new ReverseCodeError("title required");
  return t;
}

/** Normalize an optional supporter-tier gate: null/"" → null (ungated); a valid
 * tier passes; anything else throws. */
function normalizeTier(raw: unknown): SupporterTier | null {
  if (raw == null || raw === "") return null;
  if (typeof raw !== "string" || !SUPPORTER_TIERS.includes(raw as SupporterTier)) throw new ReverseCodeError("bad min_supporter_tier");
  return raw as SupporterTier;
}

/** Validate + de-dup a submitted batch of code values (within the batch). */
function cleanCodeValues(codes: unknown): string[] {
  if (!Array.isArray(codes) || codes.length === 0) throw new ReverseCodeError("codes array required");
  if (codes.length > MAX_CODES_PER_BATCH) throw new ReverseCodeError(`at most ${MAX_CODES_PER_BATCH} codes per batch`);
  const seen = new Set<string>();
  const clean: string[] = [];
  for (const c of codes) {
    if (typeof c !== "string" || c.length > MAX_CODE_LEN || !CODE_RE.test(c)) throw new ReverseCodeError("every code must be 1..64 chars of [A-Za-z0-9._%+-]");
    if (!seen.has(c)) {
      seen.add(c);
      clean.push(c);
    }
  }
  return clean;
}

/** Mint `count` distinct voucher codes not colliding with `existing`. Bounded
 * rejection loop (guards a pathologically bad RNG); throws on a bad count. */
function mintVouchers(count: unknown, mint: () => string, existing: Set<string> = new Set()): string[] {
  if (!Number.isInteger(count) || (count as number) < 1 || (count as number) > MAX_CODES_PER_BATCH) throw new ReverseCodeError(`count must be 1..${MAX_CODES_PER_BATCH}`);
  const n = count as number;
  const taken = new Set(existing);
  const fresh: string[] = [];
  let guard = 0;
  while (fresh.length < n) {
    if (guard++ > n * 50 + 100) throw new ReverseCodeError("could not mint unique codes, try again", 409);
    const v = mint();
    if (!CODE_RE.test(v) || taken.has(v)) continue;
    taken.add(v);
    fresh.push(v);
  }
  return fresh;
}

  const res = await db
    .prepare(
      "INSERT INTO reverse_code_pools (pool_id, passport_id, title, eligibility_kind, eligibility_ref, eligibility_rule, min_supporter_tier, status, created_at, updated_at) " +
        `SELECT ?, ?, ?, ?, ?, ?, ?, 'active', ?, ? WHERE ${VERIFIED_AT_SUBQUERY} IS NOT NULL`,
    )
    .bind(poolId, passportId, title, kind, ref, ruleJson, minTier, ts, ts, passportId, ts)
    .run();
  // The ONLY way this matches no row is the standing predicate: the pool id is a
  // fresh CSPRNG value, so there is nothing else in the statement that can fail
  // to match. Same sentence a de-verified creator gets from the witness.
  if (res.changes === 0) throw new VerifiedCreatorError("verified creators only");
  return { pool_id: poolId };
}

/** Upload code values into a pool the caller owns. Returns how many were added
 * (dedup within the pool drops repeats). Takes the verification witness: a pool
 * made while verified does NOT keep the right to be filled after that status is
 * taken away (the review finding — this path asked nothing at all). */
export async function uploadCodes(db: DbLike, creator: VerifiedCreator, poolId: string, codes: unknown, nowMs: number, now: () => string = () => new Date().toISOString()): Promise<{ added: number; skipped: number }> {
  const passportId = creator.passport_id;
  const pool = await requireOwnedPool(db, passportId, poolId);
  if (pool.status !== "active") throw new ReverseCodeError("pool is closed", 409);
  // Validate + de-dup within the submitted batch before touching the DB.
  const clean = cleanCodeValues(codes);
  const ts = now();
  // Explicit dedup against codes already in the pool, so `added` is exact
  // without depending on res.changes, and the INSERT can never mask anything.
  const { results: existingRows } = await db.prepare("SELECT code_value FROM reverse_codes WHERE pool_id = ?").bind(poolId).all<{ code_value: string }>();
  const existing = new Set(existingRows.map((r) => r.code_value));
  const fresh = clean.filter((v) => !existing.has(v));
  if (fresh.length > 0) {
    // One atomic batch; each code_id is CSPRNG (collision-safe). UNIQUE(pool,
    // value) is a backstop against a concurrent duplicate upload → the whole
    // batch rolls back and the creator retries (idempotent by value).
    //
    // THE STANDING RE-CHECK RIDES THE BATCH (audit item #29). This write is N
    // statements, not one, so the cheap in-WHERE form would refuse each insert
    // silently and report `added: N` for nothing — the batch does not fail on a
    // no-match. `verifiedMintGate` makes it fail: its NOT NULL subject is the
    // standing predicate, so a creator de-verified or suppressed inside the
    // window takes the whole batch down instead of filling a pool they may no
    // longer fill.
    try {
      await db.batch([verifiedMintGate(db, passportId, poolId, () => ts), ...fresh.map((value) => db.prepare("INSERT INTO reverse_codes (code_id, pool_id, code_value, created_at) VALUES (?, ?, ?, ?)").bind(randId("rc"), poolId, value, ts))]);
    } catch (e) {
      if (isWriteAdmissionRefusal(e)) throw new VerifiedCreatorError("verified creators only");
      throw e;
    }
  }
  return { added: fresh.length, skipped: clean.length - fresh.length };
}

/**
 * Generate `count` Dhwaja voucher codes into a pool the caller owns and return
 * them (so the creator can hand them to supporters). Unlike uploadCodes — where
 * the creator supplies their platform's own codes — these are Dhwaja-minted
 * vouchers the creator later VERIFIES and MARKS USED when a supporter presents
 * one (the conversion loop). Values are stored so the creator can re-view and
 * verify them, exactly like uploaded codes.
 */
export async function generateCodes(db: DbLike, creator: VerifiedCreator, poolId: string, count: unknown, now: () => string = () => new Date().toISOString(), mint: () => string = mintVoucher): Promise<{ added: number; codes: string[] }> {
  const passportId = creator.passport_id;
  const pool = await requireOwnedPool(db, passportId, poolId);
  if (pool.status !== "active") throw new ReverseCodeError("pool is closed", 409);
  // The standing `retracted` read that used to live here is gone because
  // `requireVerifiedCreator` — which the caller had to pass to obtain the
  // witness — reads `claim_state` AND `retracted` in one statement and is
  // strictly stronger. It was also strictly too weak on its own: de-verification
  // (`correction`/`downgrade_verification`) leaves `retracted = 0`, so this
  // check waved through the exact creator the review found.
  // Codes already in the pool — generated values must not collide with them
  // (UNIQUE(pool,value)) so every requested voucher actually lands.
  const { results: existingRows } = await db.prepare("SELECT code_value FROM reverse_codes WHERE pool_id = ?").bind(poolId).all<{ code_value: string }>();
  const fresh = mintVouchers(count, mint, new Set(existingRows.map((r) => r.code_value)));
  const ts = now();
  // The standing re-check rides the batch, exactly as in `uploadCodes` above and
  // for the same reason — see the comment there. Minting Dhwaja vouchers is the
  // most distributable thing on this surface, so it is the last one that should
  // survive a downgrade committed mid-flight.
  try {
    await db.batch([verifiedMintGate(db, passportId, poolId, () => ts), ...fresh.map((value) => db.prepare("INSERT INTO reverse_codes (code_id, pool_id, code_value, source, created_at) VALUES (?, ?, ?, 'generated', ?)").bind(randId("rc"), poolId, value, ts))]);
  } catch (e) {
    if (isWriteAdmissionRefusal(e)) throw new VerifiedCreatorError("verified creators only");
    throw e;
  }
  return { added: fresh.length, codes: fresh };
}

// ---------- drop unifier: build a pool + codes as part of a signed batch ----------

export interface DropPoolInput {
  title: string;
  eligibility_kind?: string; // default 'save'; 'acquisition' auto-ties to the drop's own release
  eligibility_ref?: string | null; // used only for 'achievement'
  min_supporter_tier?: string | null;
  values?: unknown; // uploaded external codes, OR…
  generate_count?: number; // …generate N Dhwaja vouchers (exactly one of the two)
}

/** Cheap validation of a drop's reverse-code block, WITHOUT minting or DB access
 * — so the /drop/start leg can fail fast before the passkey ceremony.
 *
 * TAKES THE PASSPORT ID (audit item #23). It used to take the input alone and
 * format-check the achievement ref, which is how the drop path became the
 * second door onto the cross-creator pool: `buildDropPool` is documented
 * "Pure/no-read" and delegates its validation here, so a check that could not
 * see whose pool it was could not refuse another creator's fanbase. The
 * ownership predicate needs no read at all — the passport id is IN the ref —
 * so the caller passing its own id is the whole fix, and it is a required
 * parameter so a caller cannot omit it and still compile. */
export function validateDropPoolInput(input: DropPoolInput, passportId: string): void {
  cleanTitle(input.title);
  const kind = input.eligibility_kind ?? "save";
  // A drop's code block is the legacy three-kind shape; a composite `rule`
  // pool is created on its own through createPool (it needs a DB read to
  // scope release refs, and this validator is deliberately read-free).
  if (!ELIGIBILITY_KINDS.includes(kind as EligibilityKind) || kind === "rule") throw new ReverseCodeError("bad eligibility_kind");
  if (kind === "achievement") requireGrantableAchievement(String(input.eligibility_ref ?? ""), passportId);
  normalizeTier(input.min_supporter_tier);
  const hasGen = input.generate_count != null;
  const hasVals = input.values != null;
  if (hasGen === hasVals) throw new ReverseCodeError("provide either generate_count or values (not both)");
  if (hasGen) {
    if (!Number.isInteger(input.generate_count) || (input.generate_count as number) < 1 || (input.generate_count as number) > MAX_CODES_PER_BATCH) throw new ReverseCodeError(`count must be 1..${MAX_CODES_PER_BATCH}`);
  } else {
    cleanCodeValues(input.values);
  }
}

/**
 * Build (WITHOUT executing) the pool + code INSERTs for a drop's reverse-code
 * offer, to ride the drop's ONE atomic signed batch. Pure/no-read: eligibility
 * 'acquisition' is auto-tied to the drop's own `releaseId` (created in the same
 * batch — so there's no cross-release ownership read to do here; the creator is
 * signing the whole drop). Returns the new pool id, the statements, and any
 * generated voucher values (to hand back to the creator).
 *
 * It takes the verification WITNESS rather than a passport id precisely because
 * it does no read of its own: this is the path the review reached directly, and
 * a function that mints a whole monetization surface with no DB access cannot
 * check anything for itself. Requiring the witness moves the check to the only
 * place that can make it — the caller — and makes omitting it a type error
 * instead of an oversight. The drop's OTHER verification gate
 * (`releases.requireVerified`, via the collectible) is not this gate: it fires
 * because a drop mints a release, and would stop firing the moment a drop could
 * carry a pool without one.
 */
export function buildDropPool(db: DbLike, creator: VerifiedCreator, releaseId: string, input: DropPoolInput, nowMs: number, now: () => string = () => new Date().toISOString(), mintCode: () => string = mintVoucher): { pool_id: string; statements: DbStatement[]; generated: string[] } {
  const passportId = creator.passport_id;
  validateDropPoolInput(input, passportId);
  const title = cleanTitle(input.title);
  const kind = (input.eligibility_kind ?? "save") as EligibilityKind;
  let ref: string | null = null;
  if (kind === "acquisition") {
    ref = releaseId; // the drop's own collectible — trusted, same batch, no read
  } else if (kind === "achievement") {
    ref = String(input.eligibility_ref ?? "");
    // Re-asserted HERE and not merely upstream: this function is the one that
    // actually builds the INSERT, and "the caller validated" is exactly the
    // assumption that let the drop path mint a cross-creator pool. Pure, so it
    // costs nothing to state twice.
    requireGrantableAchievement(ref, passportId);
  }
  const minTier = normalizeTier(input.min_supporter_tier);
  const generating = input.generate_count != null;
  const values = generating ? mintVouchers(input.generate_count, mintCode) : cleanCodeValues(input.values);
  const source = generating ? "generated" : "uploaded";
  const poolId = randId("rcp");
  const ts = now();
  const statements: DbStatement[] = [
    // THE STANDING RE-CHECK GOES OUT WITH THE STATEMENTS (audit item #29). This
    // function does no read and executes nothing — it hands a caller the writes
    // to fold into the drop's ONE atomic signed batch — so it is the one mint
    // path where the gate could not be "checked here". Shipping the gate AS one
    // of the statements is what makes that structural: the caller cannot execute
    // the pool without also executing the check, because they arrive in the same
    // array. If the creator is de-verified or suppressed by the time the batch
    // commits, the gate's NOT NULL fires and the whole drop — signed event,
    // release, pool, codes — rolls back together.
    verifiedMintGate(db, passportId, poolId, () => ts),
    db.prepare("INSERT INTO reverse_code_pools (pool_id, passport_id, title, eligibility_kind, eligibility_ref, min_supporter_tier, status, created_at, updated_at) VALUES (?, ?, ?, ?, ?, ?, 'active', ?, ?)").bind(poolId, passportId, title, kind, ref, minTier, ts, ts),
    ...values.map((v) => db.prepare("INSERT INTO reverse_codes (code_id, pool_id, code_value, source, created_at) VALUES (?, ?, ?, ?, ?)").bind(randId("rc"), poolId, v, source, ts)),
  ];
  return { pool_id: poolId, statements, generated: generating ? values : [] };
}

/**
 * Re-view the VALUES of the generated Dhwaja vouchers in a pool the caller
 * owns (never uploaded codes — the creator already holds those; this is
 * exclusively for the ones Dhwaja minted, which exist nowhere else).
 *
 * AS-077 / Q-21 (S-037-batch/REVERSECOD_TRU) — the SAME mechanism seen from
 * two angles. `generateCodes`' own docblock has always promised "Values are
 * stored so the creator can re-view and verify them" but no route ever
 * served a pool's generated values back after the initial `generateCodes`
 * response: if that one response is lost (a dropped connection, a client
 * crash between the write landing and the body arriving), the rows are
 * permanently write-only — nobody can ever read the voucher values a
 * creator is supposed to hand to supporters, though the rows keep existing
 * and (pre Q-08 fix) kept inflating `remaining`. `verifyCode` doesn't
 * substitute: it looks up ONE value the caller already knows, it doesn't
 * list what a pool holds. Ownership-scoped exactly like `uploadCodes`.
 */
export async function listGeneratedCodes(db: DbLike, passportId: string, poolId: string): Promise<{ code_value: string; redeemed_at: string | null; created_at: string }[]> {
  await requireOwnedPool(db, passportId, poolId);
  const { results } = await db
    .prepare("SELECT code_value, redeemed_at, created_at FROM reverse_codes WHERE pool_id = ? AND source = 'generated' ORDER BY created_at ASC")
    .bind(poolId)
    .all<{ code_value: string; redeemed_at: string | null; created_at: string }>();
  return results;
}

/**
 * Stop distribution. DELIBERATELY NOT verification-gated — it takes a plain
 * passport id, not the witness, and that is a decision rather than an omission
 * (docs/VERIFIED-MINT.md). Fail-closed means refusing the
 * PRIVILEGED act; closing is the opposite of privileged. Absent this exemption a
 * de-verified creator could neither fill their pool nor wind it down, and the
 * offer would stand open with no one able to retire it.
 */
export async function closePool(db: DbLike, passportId: string, poolId: string, now: () => string = () => new Date().toISOString()): Promise<void> {
  await requireOwnedPool(db, passportId, poolId);
  // I3-1770: bind the mutation to the SAME ownership the precheck read, not
  // just the pool id — a change between the precheck and this UPDATE (a
  // transfer, a race) must not close a pool this passport no longer owns.
  // Zero rows means the precheck's answer no longer holds; map to the same
  // not-found the precheck itself uses.
  const res = await db.prepare("UPDATE reverse_code_pools SET status = 'closed', updated_at = ? WHERE pool_id = ? AND passport_id = ?").bind(now(), poolId, passportId).run();
  if ((res.changes ?? 0) === 0) throw new ReverseCodeError("no such pool", 404);
}

/**
 * Resolve a pool the caller owns.
 *
 * A SUPPRESSED owner's pool is INVISIBLE to anyone who is not that owner: it
 * takes the same "no such pool" 404 an unknown id takes. Without that, any
 * creator session was a delisting oracle — a delisted creator's pool id
 * answered `403 not your pool` while an unknown one answered `404 no such
 * pool`, which is the "did this person ask to be delisted?" question
 * `removal.ts` exists to make unanswerable (review HIGH 1, the CREATOR_SCOPED
 * audit). The gate rides the SAME statement, so it costs no round trip.
 *
 * THE OWNER IS EXEMPT, deliberately: a delisted creator must keep being able to
 * CLOSE their own pool. Winding down is the safe direction.
 */
async function requireOwnedPool(db: DbLike, passportId: string, poolId: string): Promise<{ status: string }> {
  const pool = await db
    .prepare(
      "SELECT p.passport_id, p.status, EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = p.passport_id) AS suppressed FROM reverse_code_pools p WHERE p.pool_id = ?",
    )
    .bind(poolId)
    .first<{ passport_id: string; status: string; suppressed: number }>();
  if (!pool) throw new ReverseCodeError("no such pool", 404);
  if (pool.passport_id !== passportId) throw new ReverseCodeError(Number(pool.suppressed) !== 0 ? "no such pool" : "not your pool", Number(pool.suppressed) !== 0 ? 404 : 403);
  return { status: pool.status };
}

export interface PoolSummary {
  pool_id: string;
  title: string;
  eligibility_kind: string;
  eligibility_ref: string | null;
  min_supporter_tier: string | null; // the pool's first-access gate, if any (creator-facing config)
  status: string;
  total_codes: number;
  /** null = withheld: a tier-limited pool's claimers are a supporter population (S-4, D-0995(10)); see listPools. */
  claimed: number | null;
  /** null = withheld with `claimed`, from which it is derivable (total uploaded minus claimed minus redeemed-direct). */
  remaining: number | null;
  redeemed: number; // codes the creator marked USED — the real conversion signal (a supporter actually bought), distinct from a claim
  created_at: string;
  /** F4: the creator's own rule, readable back (a `rule` pool's JSON, plus the
   * one-line summary every kind gets). */
  eligibility_rule: EligibilityRule | null;
  eligibility_summary: string;
}

/** A creator's pools with counts: the claimer count (active-follower signal) and
 * the redeemed count (the conversion signal — a supporter bought and presented
 * the code, and the creator marked it used).
 *
 * I3-1242 a (D-0993(1), census pick: yes) — served only to the owning creator (`passport_id = ?`, bound from the
 * creator session at the route). `total_codes` and `redeemed` count the creator's own acts (codes uploaded or
 * generated, codes the creator marked used) and are exact even at 1. `claimed` is NOT the creator's act: it counts
 * FANS who claimed. On an ungated pool that is exact by the census pick (D-0993(1)). On a TIER-LIMITED pool
 * (`min_supporter_tier` set) every claimer is a supporter of that tier, so the count is a supporter population and
 * obeys the supporter list's minimum-group floor (S-4, D-0995(10); `SUPPORTER_STATS_FLOOR`, supporter-status.ts):
 * below it `claimed` is null, and `remaining` with it, because uploaded minus remaining minus redeemed-direct
 * re-derives the claim count. Pinned by test/i23-money-pool-counts.test.ts and test/i23-fx-tc-pool-tier-floor.test.ts. */
export async function listPools(db: DbLike, passportId: string): Promise<PoolSummary[]> {
  // Q-08 (S-037-batch/REVERSECOD_TRU): `remaining` used to be `total_codes -
  // claimed`, which is not "codes a fan can still be given" — it silently
  // counted (a) generated Dhwaja vouchers, which claimCode never dispenses
  // (source != 'uploaded'), and (b) an uploaded code the creator redeemed
  // directly without it ever being claimed (markRedeemed does not require a
  // prior claim). `remaining` is now the EXACT candidate set claimCode's own
  // dispensing query selects from — same three predicates, same shape, so the
  // two can't drift.
  const { results } = await db
    .prepare(
      `SELECT p.pool_id, p.title, p.eligibility_kind, p.eligibility_ref, p.eligibility_rule, p.min_supporter_tier, p.status, p.created_at,
              (SELECT COUNT(*) FROM reverse_codes c WHERE c.pool_id = p.pool_id) AS total_codes,
              (SELECT COUNT(*) FROM reverse_code_claims k WHERE k.pool_id = p.pool_id) AS claimed,
              (SELECT COUNT(*) FROM reverse_codes c WHERE c.pool_id = p.pool_id AND c.source = 'uploaded' AND c.redeemed_at IS NULL
                 AND NOT EXISTS (SELECT 1 FROM reverse_code_claims k WHERE k.code_id = c.code_id)) AS remaining,
              (SELECT COUNT(*) FROM reverse_codes c WHERE c.pool_id = p.pool_id AND c.redeemed_at IS NOT NULL) AS redeemed
       FROM reverse_code_pools p WHERE p.passport_id = ? ORDER BY p.created_at DESC`,
    )
    .bind(passportId)
    .all<{ pool_id: string; title: string; eligibility_kind: string; eligibility_ref: string | null; eligibility_rule: string | null; min_supporter_tier: string | null; status: string; created_at: string; total_codes: number; claimed: number; remaining: number; redeemed: number }>();
  return results.map((r) => {
    const rule = poolRule({ passport_id: passportId, eligibility_kind: r.eligibility_kind, eligibility_ref: r.eligibility_ref, eligibility_rule: r.eligibility_rule });
    const withheld = r.min_supporter_tier !== null && Number(r.claimed) < SUPPORTER_STATS_FLOOR;
    return {
      pool_id: r.pool_id,
      title: r.title,
      eligibility_kind: r.eligibility_kind,
      eligibility_ref: r.eligibility_ref,
      min_supporter_tier: r.min_supporter_tier,
      status: r.status,
      total_codes: r.total_codes,
      claimed: withheld ? null : r.claimed,
      remaining: withheld ? null : r.remaining,
      redeemed: r.redeemed,
      created_at: r.created_at,
      eligibility_rule: r.eligibility_kind === "rule" ? rule : null,
      eligibility_summary: rule ? describeRule(rule) : "nobody qualifies (the stored rule could not be read)",
    };
  });
}

// ---------- fan: public offer, claim, re-view ----------

export interface PoolOffer {
  pool_id: string;
  title: string;
  passport_id: string;
  eligibility_kind: string;
  eligibility_ref: string | null;
  /** F4: a fan-safe one-liner of the rule (kinds, never ids) — the offer page's
   * hint for composite pools. */
  eligibility_summary: string;
  available: boolean; // active AND at least one unclaimed code remains
}

/**
 * THE STANDING A POOL'S NON-OWNER SURFACES REQUIRE OF ITS CREATOR: not
 * retracted, not suppressed. ONE const, interpolated at every door that needs
 * it and never re-typed (FAMILY-004 P1) — a second spelling of one rule is how
 * the read door and the write door drifted apart in the first place. The pool
 * alias is `p` at every site. Costs no round trip: it rides the statement.
 */
/** I3-1260: the CLAIMANT half of the allocation's in-statement re-check (the
 * creator half is CREATOR_STANDING_SQL): the account still exists (not erased)
 * and is not suspended/terminated. Binds: accountId, accountId. */
const CLAIMANT_STANDING_SQL = `EXISTS (SELECT 1 FROM accounts a WHERE a.account_id = ?)
                AND NOT EXISTS (SELECT 1 FROM account_closures ac WHERE ac.account_id = ? AND ac.state IN ${FROZEN_CLOSURE_STATES_SQL})`;

const CREATOR_STANDING_SQL = `EXISTS (SELECT 1 FROM passports ps WHERE ps.passport_id = p.passport_id AND ps.retracted = 0)
          AND NOT EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = p.passport_id)`;

/**
 * Public pool offer — metadata only, NEVER any code value.
 *
 * A SUPPRESSED creator's pool is the same `null` an unknown pool id gets, which
 * `GET /reverse-codes/:id` renders as the identical 404. The pool id is an
 * unguessable capability, so this was never a probe — it was residual
 * PUBLICATION: the route handed back a delisted person's `passport_id`, and an
 * `achievement`-kind `eligibility_ref` names them outright (`founding_fan:psp_…`)
 * (review LOW 5, this pass). The gate rides the SAME statement (`NOT EXISTS`,
 * the `livePassport` / `getReleaseProvenance` idiom), so it costs no round trip.
 *
 * The offer comes back the moment a suppression is lifted — suppression is
 * reversible and this reads live, never a snapshot.
 */
export async function poolOffer(db: DbLike, poolId: string): Promise<PoolOffer | null> {
  const p = await db
    .prepare(
      `SELECT pool_id, passport_id, title, eligibility_kind, eligibility_ref, eligibility_rule, status
         FROM reverse_code_pools p
        WHERE p.pool_id = ?
          AND ${CREATOR_STANDING_SQL}`,
    )
    .bind(poolId)
    .first<{ pool_id: string; passport_id: string; title: string; eligibility_kind: string; eligibility_ref: string | null; eligibility_rule: string | null; status: string }>();
  if (!p) return null;
  const rule = poolRule(p);
  // Only fan-claimable ('uploaded'), never-used, unclaimed codes count toward
  // availability. Generated handout-vouchers are not self-serve claimable.
  const remain = await db.prepare("SELECT COUNT(*) AS n FROM reverse_codes c WHERE c.pool_id = ? AND c.source = 'uploaded' AND c.redeemed_at IS NULL AND NOT EXISTS (SELECT 1 FROM reverse_code_claims k WHERE k.code_id = c.code_id)").bind(poolId).first<{ n: number }>();
  return {
    pool_id: p.pool_id,
    title: p.title,
    passport_id: p.passport_id,
    eligibility_kind: p.eligibility_kind,
    // A composite rule is never serialised to the fan (a release id or an
    // achievement id inside it is the creator's business); the summary is.
    eligibility_ref: p.eligibility_kind === "rule" ? null : p.eligibility_ref,
    eligibility_summary: rule ? describeRule(rule) : "the creator's conditions",
    available: p.status === "active" && (remain?.n ?? 0) > 0,
  };
}

/** The pool's rule, whatever shape it was stored in — the legacy three kinds
 * and the F4 composite evaluate through ONE module. A stored rule that no
 * longer parses is nobody's eligibility (null → refuse). */
export function poolRule(pool: { passport_id: string; eligibility_kind: string; eligibility_ref: string | null; eligibility_rule?: string | null }): EligibilityRule | null {
  if (pool.eligibility_kind === "rule") return ruleFromStored(pool.eligibility_rule ?? null, pool.passport_id, CREATOR_NEUTRAL_ACHIEVEMENTS);
  if (pool.eligibility_kind === "achievement") {
    // DEFENCE IN DEPTH (audit item #23): pools written BEFORE the cross-creator
    // fix are still in the table; an ungrantable ref is nobody's eligibility.
    if (!pool.eligibility_ref || !grantableAchievementRef(pool.eligibility_ref, pool.passport_id)) return null;
  }
  return ruleFromLegacy(pool.eligibility_kind, pool.eligibility_ref);
}

/** Is this fan eligible for the pool's rule? Server-side, by construction —
 * and since F4 the answer carries its SNAPSHOT (what was true, when), which
 * the claim stores. */
async function eligibilitySnapshotFor(db: DbLike, accountId: string, pool: { passport_id: string; eligibility_kind: string; eligibility_ref: string | null; eligibility_rule?: string | null }, nowMs: number): Promise<EligibilitySnapshot | null> {
  const rule = poolRule(pool);
  if (!rule) return null;
  return eligibilitySnapshot(db, accountId, pool.passport_id, rule, nowMs);
}

/** I3-1261: the claim-time re-check of a pool's rule. Every leaf is read live
 * EXCEPT `supporter`, which is judged against `tierBefore` — the tier read
 * before the claim landed — because `supporterTier` counts `reverse_code_claims`
 * and the claim being checked would otherwise qualify the fan by itself. */
async function qualifiesWithTier(db: DbLike, accountId: string, passportId: string, rule: EligibilityRule, nowMs: number, tierBefore: SupporterStatus): Promise<boolean> {
  if (rule.kind === "all") {
    for (const r of rule.rules) if (!(await qualifiesWithTier(db, accountId, passportId, r, nowMs, tierBefore))) return false;
    return true;
  }
  if (rule.kind === "any") {
    for (const r of rule.rules) if (await qualifiesWithTier(db, accountId, passportId, r, nowMs, tierBefore)) return true;
    return false;
  }
  if (rule.kind === "supporter") return meetsTier(tierBefore, rule.tier);
  return (await eligibilitySnapshot(db, accountId, passportId, rule, nowMs)).satisfied;
}
const ruleHasSupporter = (rule: EligibilityRule): boolean => (rule.kind === "all" || rule.kind === "any" ? rule.rules.some(ruleHasSupporter) : rule.kind === "supporter");

async function isEligible(db: DbLike, accountId: string, pool: { passport_id: string; eligibility_kind: string; eligibility_ref: string | null; eligibility_rule?: string | null }, nowMs: number = Date.now()): Promise<boolean> {
  const snap = await eligibilitySnapshotFor(db, accountId, pool, nowMs);
  return snap !== null && snap.satisfied;
}

/** The pre-F4 evaluator, kept ONLY as the reference the module must agree
 * with — `test/eligibility.test.ts` runs both on the same fixtures. */
export async function legacyIsEligible(db: DbLike, accountId: string, pool: { passport_id: string; eligibility_kind: string; eligibility_ref: string | null }): Promise<boolean> {
  if (pool.eligibility_kind === "acquisition") {
    // EVIDENCE, NOT PROPERTY (F3 review, 0061): the fan must have REDEEMED a
    // credential of this release themselves. A gifted-in acquisition is theirs
    // to hold, feature and gift on — but one paid credential passed through N
    // accounts must not drain N of the creator's limited codes. F4b: the
    // redeemer keeps the standing after gifting the item away.
    const r = await db.prepare("SELECT 1 AS x FROM acquisitions WHERE release_id = ? AND redeemed_by = ? LIMIT 1").bind(pool.eligibility_ref, accountId).first();
    return r !== null;
  }
  if (pool.eligibility_kind === "achievement") {
    // DEFENCE IN DEPTH (audit item #23). The three write paths now refuse a
    // cross-creator ref, but pools written BEFORE that fix are still in the
    // table and would otherwise keep dispensing to a victim's fanbase forever.
    // Checking here stops them at the dispensing door with no migration and no
    // backfill, and it means a future fifth writer that forgets the gate leaks
    // a row rather than an audience. A pool that fails it is simply not
    // eligible for anyone — the same generic "you don't qualify" every other
    // eligibility miss returns, so it is never a probe.
    if (!pool.eligibility_ref || !grantableAchievementRef(pool.eligibility_ref, pool.passport_id)) return false;
    const r = await db.prepare("SELECT 1 AS x FROM achievements WHERE account_id = ? AND achievement_id = ? LIMIT 1").bind(accountId, pool.eligibility_ref).first();
    return r !== null;
  }
  if (pool.eligibility_kind === "save") {
    const r = await db.prepare("SELECT 1 AS x FROM saves WHERE account_id = ? AND passport_id = ? LIMIT 1").bind(accountId, pool.passport_id).first();
    return r !== null;
  }
  return false;
}

export interface ClaimResult {
  code_value: string;
  pool_id: string;
}

/**
 * Claim one code from a pool. Checks eligibility, then atomically assigns a
 * single unclaimed code to this fan. Race-proof via reverse_code_claims:
 * PK(pool,account) = one per fan; UNIQUE(code_id) = one fan per code. On a
 * code-race the loser retries another code; a repeat claim by the same fan
 * returns their EXISTING code (idempotent, so they can't be told "sold out"
 * after already claiming). Reveals the code value to the claimer.
 */
export async function claimCode(
  db: DbLike,
  accountId: string,
  poolId: string,
  now: () => string = () => new Date().toISOString(),
  pubEnv?: AnalyticsTickEnv,
  pubCtx?: { waitUntil(p: Promise<unknown>): void },
): Promise<ClaimResult> {
  // ORDER IS THE FIX, not just the filter (review HIGH 1). Already claimed →
  // return the existing code (idempotent re-view), and do it FIRST, before the
  // pool is resolved at all: a fan who already holds a code keeps holding it
  // whatever happens to the creator afterwards, which is the promise
  // docs/VERIFIED-MINT.md makes and `listClaimedCodes` already keeps. This
  // statement touches no pool row, so it answers the same way for an unknown id.
  const existing = await db.prepare("SELECT c.code_value AS code_value FROM reverse_code_claims k JOIN reverse_codes c ON c.code_id = k.code_id WHERE k.pool_id = ? AND k.account_id = ?").bind(poolId, accountId).first<{ code_value: string }>();
  if (existing) return { code_value: existing.code_value, pool_id: poolId };

  // A SUPPRESSED creator's pool IS an unknown pool id, in the same statement and
  // with the same answer — `poolOffer`'s gate, on the door that actually
  // dispenses. Two things were broken without it, and they are the same edit:
  //
  //  1. DISTRIBUTION. `poolOffer` returning null hides the offer PAGE. It stops
  //     nobody who holds the URL, and the URL is precisely how these offers are
  //     distributed — the creator posts it on their own platform. Measured on a
  //     delisted creator: `poolOffer -> null`, `claimCode -> EXTERNAL-CODE-1`.
  //     A delisted creator's pool dispensed codes indefinitely.
  //  2. THE ORACLE. `GET /reverse-codes/:id` answered 404 for both a suppressed
  //     pool and an unknown id, and `POST /account/reverse-codes/:id/claim`
  //     answered 200-with-a-code for the first and 404 for the second — a clean
  //     "did this person ask to be delisted?" probe on a fan route, which is the
  //     exact question `removal.ts` exists to make unanswerable.
  //
  // Both legs now spend the SAME two statements and return the SAME 404 body, so
  // there is nothing to compare — not the wording, not the statement count.
  const pool = await db
    .prepare(
      `SELECT pool_id, passport_id, eligibility_kind, eligibility_ref, eligibility_rule, min_supporter_tier, status
         FROM reverse_code_pools p
        WHERE p.pool_id = ?
          AND NOT EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = p.passport_id)`,
    )
    .bind(poolId)
    .first<{ pool_id: string; passport_id: string; eligibility_kind: string; eligibility_ref: string | null; eligibility_rule: string | null; min_supporter_tier: string | null; status: string }>();
  if (!pool) throw new ReverseCodeError("no such pool", 404);

  if (pool.status !== "active") throw new ReverseCodeError("this offer is closed", 409);
  // Standing check: a retracted creator (e.g. removed for impersonation) stops
  // distributing immediately, even though the pool was created while verified
  // (review finding — verification was gated only at creation). Fail closed.
  const creator = await db.prepare("SELECT retracted FROM passports WHERE passport_id = ?").bind(pool.passport_id).first<{ retracted: number }>();
  if (!creator || creator.retracted !== 0) throw new ReverseCodeError("this offer is closed", 409);
  // Derive the window "now" from now(); fall back to Date.now() if now() ever
  // yields an unparseable string, so nothing below can 500 on a bad clock.
  const parsed = Date.parse(now());
  const nowMs = Number.isFinite(parsed) ? parsed : Date.now();
  // F4: the eligibility SNAPSHOT — what was true, when — is what the claim
  // stores; the refusal is the same generic sentence for every miss.
  const snap = await eligibilitySnapshotFor(db, accountId, pool, nowMs);
  if (!snap || !snap.satisfied) throw new ReverseCodeError("you don't qualify for this offer yet", 403);
  // Optional first-access gate: the pool may require an existing supporter tier.
  // Invisible by design — a fan who doesn't meet it gets the same generic
  // "don't qualify" as any other eligibility miss (never a leak of their status).
  // (The tier itself is now Sybil-gated in supporterTier: a fresh account can't
  // reach active/core, so it can't unlock a tier-gated pool.)
  // I3-1261: the tier BEFORE this claim exists — the re-check after the write
  // judges `supporter` leaves against it (see qualifiesWithTier).
  const tierBefore: SupporterStatus = pool.min_supporter_tier || ruleHasSupporter(snap.rule) ? await supporterTier(db, accountId, pool.passport_id, nowMs) : "none";
  if (pool.min_supporter_tier && !meetsTier(tierBefore, pool.min_supporter_tier as SupporterTier)) throw new ReverseCodeError("you don't qualify for this offer yet", 403);
  const ts = now();
  // Sybil throttle: bound how many codes ONE account can seize per day. Consumed
  // exactly ONCE, and ONLY after a claimable candidate is confirmed — so a fan who
  // taps sold-out offers never burns a slot (review LOW), and the idempotent
  // re-view above (which returned earlier) never consumes one either. Over the cap
  // → the same generic error, so it leaks nothing. Fail-closed (failure denies).
  let throttled = false;
  // Bounded retry: pick an unclaimed code, try to seize it; UNIQUE(code_id)
  // rejects a concurrent winner → pick another. A handful of tries clears any
  // realistic contention; if none are seizable, the pool is exhausted.
  for (let attempt = 0; attempt < 8; attempt++) {
    // Fan self-serve dispenses ONLY 'uploaded', never-used, unclaimed codes.
    // Generated handout-vouchers (source='generated') are for the creator's
    // off-platform conversion loop and must not be claimable here (review
    // finding — otherwise a save would net a free supporter voucher).
    const candidate = await db
      .prepare("SELECT c.code_id, c.code_value FROM reverse_codes c WHERE c.pool_id = ? AND c.source = 'uploaded' AND c.redeemed_at IS NULL AND NOT EXISTS (SELECT 1 FROM reverse_code_claims k WHERE k.code_id = c.code_id) LIMIT 1")
      .bind(poolId)
      .first<{ code_id: string; code_value: string }>();
    if (!candidate) throw new ReverseCodeError("this offer is sold out", 409); // no slot burned
    if (!throttled) {
      if (!(await consumeRateLimit(db, `rcclaim:${accountId}`, nowMs, MAX_CLAIMS_PER_ACCOUNT_PER_DAY))) {
        throw new ReverseCodeError("you're claiming too fast — try again later", 429);
      }
      throttled = true;
    }
    try {
      // The claim and its analytics row in ONE batch (ADR-0006 §2).
      //
      // CO-07 — THE INSERT CARRIES ITS ADMISSION IN ITS OWN `WHERE`. It used to
      // carry none, on the reasoning that UNIQUE(code_id) makes a double claim
      // impossible. It does — but uniqueness is not the only thing that has to
      // still be true at insert time. Everything this function read BEFORE the
      // await chain above (the code was `redeemed_at IS NULL`; the pool was
      // `active`; the creator was not retracted and not suppressed) is a read,
      // and a creator's ordinary fulfillment marking that very code redeemed —
      // or closing the pool, or a retraction landing — between the SELECT at
      // the top of this loop and here does NOT violate claim uniqueness. So the
      // allocator handed the fan a voucher the creator had already burned AND
      // consumed their one-per-pool claim, after which every later request
      // returned that same dead code. Not a race in the exotic sense: one fan
      // claiming while the creator works their fulfillment queue.
      //
      // `INSERT ... SELECT ... WHERE` is the repo's guarded-write shape
      // (`mintInstanceStatements`, `forfeitStmt`, `acceptGift`): a losing
      // admission changes NOTHING — no voucher, no consumed pool claim — and
      // the read-back below turns that into another attempt rather than a
      // silent success. `pool_code_claimed` is CREATOR-SCOPED (the funnel cell
      // is the creator's, no fan pseudonym is minted for a code claim) and is
      // predicated on THIS INSERT having written its row: `changes() = 1` is the
      // row count of the claim INSERT, the statement directly above the outbox
      // row in this batch (a creator-scoped emit is ONE statement,
      // analytics-events.ts), asked inside the EXISTS that names the landed row
      // (tests/invariants/conditional-emit-scan.test.ts reads that row as the
      // fact). I3-0178 (D-1078(21)): the EXISTS alone was the row's shape, so a
      // same-account double tap whose second INSERT was refused (the code
      // already this account's) emitted a second event for one claim.
      const evKey = eventKey();
      const landed = { sql: "EXISTS (SELECT 1 FROM reverse_code_claims WHERE pool_id = ? AND account_id = ? AND code_id = ? AND changes() = 1)", params: [poolId, accountId, candidate.code_id] };
      const span = outboxSpan(db);
      const spanRes = await db.batch([
        span.open,
        db
          .prepare(
            `INSERT INTO reverse_code_claims (pool_id, account_id, code_id, claimed_at, eligibility_snapshot)
             SELECT ?, ?, ?, ?, ?
              WHERE EXISTS (SELECT 1 FROM reverse_codes c
                             WHERE c.code_id = ? AND c.pool_id = ? AND c.source = 'uploaded' AND c.redeemed_at IS NULL
                               AND NOT EXISTS (SELECT 1 FROM reverse_code_claims k WHERE k.code_id = c.code_id))
                AND EXISTS (SELECT 1 FROM reverse_code_pools p
                            WHERE p.pool_id = ? AND p.status = 'active'
                              AND ${CREATOR_STANDING_SQL})
                AND ${CLAIMANT_STANDING_SQL}`,
          )
          .bind(poolId, accountId, candidate.code_id, ts, JSON.stringify(snap), candidate.code_id, poolId, poolId, accountId, accountId),
        ...outboxStatements(db, { kind: "pool_code_claimed", passportId: pool.passport_id, occurredDay: occurredDay(ts), scopeId: poolId }, { extraWhere: landed, eventKey: evKey }),
        span.close,
      ]);
      const seqs = outboxSpanSeqs(spanRes[0], spanRes[spanRes.length - 1]);
      // A guarded INSERT that matched nothing does not throw, so "did it land"
      // is a read. Not landed → the admission changed under us: try the next
      // candidate (a redeemed code is excluded by the SELECT above, so the loop
      // makes progress) and let the bounded loop's 409 answer a closure.
      const mineNow = await db.prepare("SELECT 1 AS x FROM reverse_code_claims WHERE pool_id = ? AND account_id = ? AND code_id = ?").bind(poolId, accountId, candidate.code_id).first();
      if (!mineNow) {
        // I3-1260: the claimant lost standing inside the window — refuse, don't burn retries.
        if (!(await db.prepare(`SELECT 1 AS x WHERE ${CLAIMANT_STANDING_SQL}`).bind(accountId, accountId).first())) throw new ReverseCodeError("you don't qualify for this offer yet", 403);
        continue;
      }
      // I3-1261: eligibility is not one SQL predicate, so it is re-checked AFTER
      // the write, against the claim's own row. A fan who stopped qualifying
      // inside the window has the claim and its event undone in ONE batch, the
      // event only while unpublished and the claim only while its event is, so
      // the event exists iff its fact. (Minimum tier cannot fall inside the
      // window except by erasure, which CLAIMANT_STANDING_SQL re-checks.)
      // RGM-07 (I15-FX2): the undo PSEUDONYMISES the fan's claim row (erasure's
      // `del_` tombstone, account_data.ts) instead of deleting it, so the code
      // stays claimed and is never dispensed again — a same-fan call inside the
      // window (the existing-claim read above, the PK-catch read below, or
      // listClaimedCodes) may already have returned it.
      if (!(await qualifiesWithTier(db, accountId, pool.passport_id, snap.rule, nowMs, tierBefore))) {
        await db.batch([
          db
            .prepare("UPDATE reverse_code_claims SET account_id = 'del_' || lower(hex(randomblob(16))) WHERE pool_id = ? AND account_id = ? AND code_id = ? AND claimed_at = ? AND NOT EXISTS (SELECT 1 FROM analytics_outbox WHERE event_key = ? AND published_at IS NOT NULL)")
            .bind(poolId, accountId, candidate.code_id, ts, evKey),
          db
            .prepare("DELETE FROM analytics_outbox WHERE event_key = ? AND published_at IS NULL AND NOT EXISTS (SELECT 1 FROM reverse_code_claims WHERE pool_id = ? AND account_id = ? AND code_id = ?)")
            .bind(evKey, poolId, accountId, candidate.code_id),
        ]);
        const kept = await db.prepare("SELECT 1 AS x FROM reverse_code_claims WHERE pool_id = ? AND account_id = ? AND code_id = ?").bind(poolId, accountId, candidate.code_id).first();
        if (!kept) throw new ReverseCodeError("you don't qualify for this offer yet", 403);
      }
      if (pubEnv && pubCtx) publishAfterCommit(pubEnv, pubCtx, seqs);
      return { code_value: candidate.code_value, pool_id: poolId };
    } catch (e) {
      // PK(pool,account) conflict → this fan already claimed between our read
      // and here → return their code. UNIQUE(code_id) conflict → someone took
      // this code → retry another candidate.
      if (e instanceof Error && /UNIQUE|constraint|PRIMARY/i.test(e.message)) {
        const mine = await db.prepare("SELECT c.code_value AS code_value FROM reverse_code_claims k JOIN reverse_codes c ON c.code_id = k.code_id WHERE k.pool_id = ? AND k.account_id = ?").bind(poolId, accountId).first<{ code_value: string }>();
        if (mine) return { code_value: mine.code_value, pool_id: poolId };
        continue; // code-race: try the next candidate
      }
      throw e;
    }
  }
  throw new ReverseCodeError("could not claim right now, try again", 409);
}

export interface ClaimedCode {
  pool_id: string;
  title: string;
  code_value: string;
  claimed_at: string;
}

/** Opaque pagination cursor for `listClaimedCodes`: the (claimed_at, pool_id)
 * of the last row a page ended on. `claimed_at` alone is not a total order —
 * two claims can share a millisecond — so `pool_id` (the other half of
 * PK(pool,account): one claim per fan per pool) breaks ties without dropping
 * or repeating a row across pages. */
export interface ClaimedCodesCursor {
  claimed_at: string;
  pool_id: string;
}

/**
 * A fan's own claimed codes (re-viewable — they need them to redeem).
 *
 * Q-10 (S-037-batch/REVERSECOD_TRU): a fan with more than `limit` claims had
 * NO way to reach anything past the newest page — `limit` defaulted to 100,
 * was clamped to 200, and there was no cursor at all, so a code claimed
 * earlier than that window was unrecoverable through this function (the
 * account-data export deliberately omits `code_value` — it's the creator's
 * inventory, already shown to the fan at claim time — so this is the only
 * surface that can ever hand a fan a code value again). `before` pages
 * backward from a prior page's last row; omit it for the first page.
 */
export async function listClaimedCodes(db: DbLike, accountId: string, limit = 100, before?: ClaimedCodesCursor | null): Promise<ClaimedCode[]> {
  const capped = Math.max(1, Math.min(200, limit));
  const { results } = await db
    .prepare(
      before
        ? `SELECT k.pool_id, p.title, c.code_value, k.claimed_at
       FROM reverse_code_claims k
       JOIN reverse_codes c ON c.code_id = k.code_id
       JOIN reverse_code_pools p ON p.pool_id = k.pool_id
       WHERE k.account_id = ? AND (k.claimed_at < ? OR (k.claimed_at = ? AND k.pool_id < ?))
       ORDER BY k.claimed_at DESC, k.pool_id DESC LIMIT ?`
        : `SELECT k.pool_id, p.title, c.code_value, k.claimed_at
       FROM reverse_code_claims k
       JOIN reverse_codes c ON c.code_id = k.code_id
       JOIN reverse_code_pools p ON p.pool_id = k.pool_id
       WHERE k.account_id = ? ORDER BY k.claimed_at DESC, k.pool_id DESC LIMIT ?`,
    )
    .bind(...(before ? [accountId, before.claimed_at, before.claimed_at, before.pool_id, capped] : [accountId, capped]))
    .all<ClaimedCode>();
  return results;
}

// ---------- creator: verify-before-discount + mark-used (the conversion loop) ----------

export interface VerifyResult {
  found: boolean; // the code is a genuine voucher in ONE of this creator's pools
  pool_id?: string;
  title?: string;
  redeemed?: boolean; // already marked used → the creator must NOT give the discount again
  redeemed_at?: string | null;
  // was also dispensed to a fan via the claim endpoint (informational). null = WITHHELD: a tier-limited pool whose claim
  // count is below SUPPORTER_STATS_FLOOR (S-4, D-0995(10)) — per-code claimed state would re-derive the count listPools withholds.
  claimed?: boolean | null;
  matches?: number; // how many of the creator's codes carry this value (normally 1)
}

/**
 * A creator checks a presented code BEFORE giving a discount: is it real (one of
 * MY vouchers) and unused? Scoped to the caller's own pools by construction, so
 * a creator can never probe another creator's codes — an unknown or foreign
 * value simply returns { found: false } (no cross-creator enumeration, no
 * distinction between "never existed" and "belongs to someone else"). Read-only;
 * the authoritative single-use decision is markRedeemed.
 */
export async function verifyCode(db: DbLike, passportId: string, codeValue: unknown): Promise<VerifyResult> {
  if (typeof codeValue !== "string" || !CODE_RE.test(codeValue)) return { found: false };
  // M36: a GENERATED Dhwaja voucher is uppercase base32 by construction (the
  // alphabet promise: "a supporter can read it aloud or type it without
  // confusion"), so a lowercased entry must still match. UPLOADED codes are
  // opaque tokens for an external platform where case may be significant — two
  // codes that differ only in case can be two distinct codes — so those stay an
  // exact match. Hence the case-fold is scoped to source = 'generated' only.
  const upper = codeValue.toUpperCase();
  const { results } = await db
    .prepare(
      `SELECT c.pool_id, p.title, c.redeemed_at, p.min_supporter_tier,
              EXISTS (SELECT 1 FROM reverse_code_claims k WHERE k.code_id = c.code_id) AS claimed,
              (SELECT COUNT(*) FROM reverse_code_claims k WHERE k.pool_id = c.pool_id) AS pool_claimed
       FROM reverse_codes c JOIN reverse_code_pools p ON p.pool_id = c.pool_id
       WHERE p.passport_id = ? AND (c.code_value = ? OR (c.source = 'generated' AND c.code_value = ?))
       ORDER BY (c.redeemed_at IS NULL) DESC, c.created_at ASC`,
    )
    .bind(passportId, codeValue, upper)
    .all<{ pool_id: string; title: string; redeemed_at: string | null; min_supporter_tier: string | null; claimed: number; pool_claimed: number }>();
  if (results.length === 0) return { found: false };
  // Surface the most actionable match: an unredeemed one first (so the creator
  // can redeem it), else a redeemed one (so they see it's already used).
  const best = results[0]!;
  return {
    found: true,
    pool_id: best.pool_id,
    title: best.title,
    redeemed: best.redeemed_at !== null,
    redeemed_at: best.redeemed_at,
    // RJ-02 S-4: listPools' withholding rule, applied per code (one rule: tier pool AND claim count below the floor).
    claimed: best.min_supporter_tier !== null && Number(best.pool_claimed) < SUPPORTER_STATS_FLOOR ? null : best.claimed === 1,
    matches: results.length,
  };
}

export interface RedeemResult {
  pool_id: string;
  redeemed_at: string;
}

/**
 * Mark a presented voucher USED, exactly once. The creator calls this AFTER
 * applying the discount so the code can't be redeemed twice. Atomic single-use:
 * a guarded `UPDATE ... WHERE redeemed_at IS NULL` that also re-checks ownership,
 * so a double-submit (or a race) flips exactly one row — the loser gets 409
 * "already redeemed". Scoped to the caller's own pools; an unknown/foreign value
 * is a 404. Optionally disambiguated by poolId when a value spans pools.
 */
export async function markRedeemed(
  db: DbLike,
  passportId: string,
  codeValue: unknown,
  now: () => string = () => new Date().toISOString(),
  poolId?: string | null,
  pubEnv?: AnalyticsTickEnv,
  pubCtx?: { waitUntil(p: Promise<unknown>): void },
): Promise<RedeemResult> {
  if (typeof codeValue !== "string" || !CODE_RE.test(codeValue)) throw new ReverseCodeError("no such code", 404);
  const scope = typeof poolId === "string" && poolId.length > 0 ? poolId : null;
  // M36: case-fold scoped to generated vouchers, exactly as verifyCode — the
  // creator marks used whatever case the supporter presented, while an uploaded
  // external token stays an exact match.
  const upper = codeValue.toUpperCase();
  // Candidate codes of THIS value in THIS creator's pools; prefer an unredeemed
  // one (oldest first) so a re-presented value redeems a still-open copy.
  const cand = await db
    .prepare(
      `SELECT c.code_id, c.pool_id, c.redeemed_at
       FROM reverse_codes c JOIN reverse_code_pools p ON p.pool_id = c.pool_id
       WHERE p.passport_id = ? AND (c.code_value = ? OR (c.source = 'generated' AND c.code_value = ?)) AND (? IS NULL OR c.pool_id = ?)
       ORDER BY (c.redeemed_at IS NULL) DESC, c.created_at ASC LIMIT 1`,
    )
    .bind(passportId, codeValue, upper, scope, scope)
    .first<{ code_id: string; pool_id: string; redeemed_at: string | null }>();
  if (!cand) throw new ReverseCodeError("no such code", 404);
  if (cand.redeemed_at !== null) throw new ReverseCodeError("this code was already used", 409);

  const ts = now();
  // Atomic single-use gate. The WHERE re-checks redeemed_at IS NULL (a concurrent
  // winner already flipped it → 0 rows → we report already-used) AND re-checks
  // ownership via the pool subquery (defense in depth: the write can't touch a
  // code outside the caller's pools even if the id were wrong).
  //
  // ONE BATCH, EMIT FIRST, ON THE PRE-STATE (S-088 G291 REG-1). The flip used to
  // commit through `run()` and the event followed as a second statement, so a
  // failure between the two left the code redeemed with no event, and the retry
  // answered 409 forever. The emit now carries the flip's own WHERE as its
  // `extraWhere` and rides the same batch ahead of it: both read one state in one
  // transaction, so the event exists iff this batch flips the code, and a failed
  // batch rolls both back (a retry starts clean).
  //
  // WHO WON (C-60, S-037-batch/REVERSECOD_TRU): never a `redeemed_at = ts`
  // read-back — two same-millisecond callers produce the same `ts`. The flip's
  // `RETURNING code_id` row is this call's own proof: it is in this batch's
  // result only if this batch's UPDATE wrote the row (db.ts: enumerate, never
  // count), and a serialised loser's batch selects no event row and returns none.
  const span = outboxSpan(db);
  const spanRes = await db.batch([
    span.open,
    db
      .prepare(
        outboxSelectSql("pool_code_redeemed", {
          table: "reverse_code_pools",
          where: "pool_id = ? AND passport_id = ?",
          extraWhere: "EXISTS (SELECT 1 FROM reverse_codes WHERE code_id = ? AND pool_id = reverse_code_pools.pool_id AND redeemed_at IS NULL)",
          scopeIdExpr: "pool_id",
        }),
      )
      .bind(ts, cand.pool_id, passportId, cand.code_id),
    db
      .prepare("UPDATE reverse_codes SET redeemed_at = ? WHERE code_id = ? AND redeemed_at IS NULL AND pool_id IN (SELECT pool_id FROM reverse_code_pools WHERE passport_id = ?) RETURNING code_id")
      .bind(ts, cand.code_id, passportId),
    span.close,
  ]);
  const seqs = outboxSpanSeqs(spanRes[0], spanRes[spanRes.length - 1]);
  const res = spanRes.slice(1, -1);
  if (((res[1] as { results?: unknown[] } | undefined)?.results ?? []).length !== 1) throw new ReverseCodeError("this code was already used", 409);
  if (pubEnv && pubCtx) publishAfterCommit(pubEnv, pubCtx, seqs);
  return { pool_id: cand.pool_id, redeemed_at: ts };
}
