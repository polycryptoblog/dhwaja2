/** I3-0261: how many `operator:identity_first_seen` rows `GET /ops/integrity` returns, newest first. */
export const OPERATOR_FIRST_SEEN_LIMIT = 50;
export interface OperatorFirstSeen {
  rows: { op_id: string; at: string }[];
  limit: number;
  /** True when more first-seen rows exist than `rows` carries: the list is the newest `limit`, not every one. */
  more: boolean;
}
/**
 * I3-0261: the `operator:identity_first_seen` audit rows operators.ts writes when an OPERATOR_EMAILS address mints an
 * identity on its first request — read here so the integrity page shows them. The allowlist change itself is
 * `operatorAccessChanges` below (D-1082(4)). Rows the audit log no longer holds are not here. A failed read is `null`, never an
 * empty list, so the page says it could not read them rather than that there are none.
 */
export async function operatorFirstSeen(d: DbLike): Promise<OperatorFirstSeen | null> {
  try {
    const { results } = await d
      .prepare("SELECT item_id AS op_id, created_at AS at FROM audit_log WHERE action = 'operator:identity_first_seen' ORDER BY audit_id DESC LIMIT ?")
      .bind(OPERATOR_FIRST_SEEN_LIMIT + 1)
      .all<{ op_id: string; at: string }>();
    return { rows: results.slice(0, OPERATOR_FIRST_SEEN_LIMIT), limit: OPERATOR_FIRST_SEEN_LIMIT, more: results.length > OPERATOR_FIRST_SEEN_LIMIT };
  } catch {
    return null;
  }
}

/**
 * I3-0261 (D-1082(4)): the operator-access audit rows operators.ts writes (`OPERATOR_ACCESS_ACTIONS`: an identity's
 * role first seen or changed, the OPERATOR_EMAILS / group configuration a request ran under, addresses added,
 * re-added and removed, the full group or groups claim changed), newest first, for the integrity page. Rows the
 * audit log no longer holds are not here. A failed read is `null`, never an empty list.
 */
export async function operatorAccessChanges(d: DbLike): Promise<OperatorAccessChanges | null> {
  try {
    const { results } = await d
      .prepare(`SELECT action, item_id AS item, created_at AS at, detail FROM audit_log WHERE action IN (${OPERATOR_ACCESS_ACTIONS.map(() => "?").join(", ")}) ORDER BY audit_id DESC LIMIT ?`)
      .bind(...OPERATOR_ACCESS_ACTIONS, OPERATOR_ACCESS_CHANGES_LIMIT + 1)
      .all<{ action: string; item: string | null; at: string; detail: string }>();
    return { rows: results.slice(0, OPERATOR_ACCESS_CHANGES_LIMIT), limit: OPERATOR_ACCESS_CHANGES_LIMIT, more: results.length > OPERATOR_ACCESS_CHANGES_LIMIT };
  } catch {
    return null;
  }
}

/**
 * S7-4 / FINDING-0014-D-2 — a brand-enrolment decision mail that never sends
 * leaves a row. `mailer.send` returns FALSE for an unmapped template, a
 * malformed address and every provider rejection (mailer.ts:51-53, :205-213),
 * and the deferred promise REJECTS for a transport failure; both are
 * "undelivered", and the audited-send contract mailer.ts:84-94 states for
 * `report-outcome` applies here too — the decision itself is recorded, so the
 * notice that did not reach the enroller must be recorded beside it.
 * Counts and the enrolment id only, never the address (mailer.ts rule).
 * Best-effort like reports.ts:1809-1820: the audit line never fails a decision.
 */
async function auditUndeliveredDecisionMail(
  d: DbLike,
  op: string,
  template: string,
  enrolmentId: string,
  reason: string,
  ts: string,
): Promise<void> {
  try {
    await auditStmt(d, op, "brand_enrol:decision_mail_undelivered", enrolmentId, `${template}: ${reason}`, ts).run();
  } catch {
    /* the record of a lost mail must not itself lose the decision */
  }
}

/** Wraps one deferred decision send so BOTH undelivered shapes audit. The
 * returned promise never rejects, so the `defer` fallback's `catch(() => {})`
 * has an audited row behind it rather than a swallowed boolean. */
function sendDecisionMail(
  mailer: { send(template: string, to: string, vars: Record<string, string>): Promise<boolean> },
  d: DbLike,
  op: string,
  template: string,
  to: string,
  vars: Record<string, string>,
  enrolmentId: string,
  ts: string,
): Promise<void> {
  return mailer.send(template, to, vars).then(
    (ok) => (ok ? undefined : auditUndeliveredDecisionMail(d, op, template, enrolmentId, "the mailer did not accept it", ts)),
    (e) => auditUndeliveredDecisionMail(d, op, template, enrolmentId, `send failed: ${e instanceof Error ? e.name : "error"}`, ts),
  );
}

/**
 * S-022 (docket #36j, ruled by Jason 2026-09-11: "yes with INV-025 wording") —
 * the incumbent (`fromPassportId`) is told their account was transferred
 * away. INV-025's wording: point only to the REAL account, never name the
 * fake. INV-003's no-oracle shape is WHY this is careful, not merely terse:
 * naming the challenger here — their handle, their id, or the destination
 * passport — would turn a routine transfer into an intelligence channel about
 * who now holds the account. `accountKey` is `v1|platform|idKind|id`
 * (canonicalAccount, packages/canonical); ONLY the platform segment (index 1)
 * is ever read here. idKind and id — the parts that could actually identify
 * the account, and through it the challenger — never enter this function's
 * own locals, let alone the mail. Best-effort and meant to be called on
 * `waitUntil` by the caller: a lost notice never unwinds an already-committed
 * transfer, the same stance `sendDecisionMail` above takes for enrolment mail.
 */
/** The creator's verified contact address, or null — the one lookup the transfer notice sends to and (I40-OPS, D-1169(7))
 * the scheduling door checks before a one-signature transfer. */
export async function verifiedCreatorAddress(d: DbLike, passportId: string): Promise<string | null> {
  const contact = await d.prepare("SELECT email, email_verified FROM creator_contacts WHERE passport_id = ?").bind(passportId).first<{ email: string | null; email_verified: number }>();
  return contact?.email && contact.email_verified === 1 ? contact.email : null;
}

async function notifyIncumbentAccountTransferred(
  d: DbLike,
  mailer: { send(template: string, to: string, vars: Record<string, string>): Promise<boolean> },
  fromPassportId: string,
  accountKey: string,
  actorId: string,
  ts: string,
  /** I39-OPS (D-1158 answer 4): given, this is the WARNING of a scheduled transfer — template `account-transfer-scheduled`
   * (vars `platform`, `due_utc`), each audit detail tagged with the transfer id, and the outcome written to the row's
   * `notice` column in the audit row's batch. Absent, the notice of a completed transfer, unchanged. */
  scheduled: { transferId: string; dueUtc: string } | null = null,
): Promise<void> {
  try {
    const platform = accountKey.split("|")[1] ?? "";
    const tag = scheduled === null ? "" : ` (scheduled ${scheduled.transferId})`;
    const mark = (notice: "sent" | "undelivered" | "unsent"): DbStatement[] =>
      scheduled === null ? [] : [d.prepare("UPDATE scheduled_account_transfers SET notice = ? WHERE transfer_id = ?").bind(notice, scheduled.transferId)];
    const email = await verifiedCreatorAddress(d, fromPassportId);
    if (email === null) {
      await d.batch([auditStmt(d, actorId, "account:transfer_notice_unsent", fromPassportId, `no_verified_address${tag}`, ts), ...mark("unsent")]);
      return;
    }
    let sent = false;
    try {
      sent = await (scheduled === null
        ? mailer.send("account-transfer-notice", email, { platform })
        : mailer.send("account-transfer-scheduled", email, { platform, due_utc: scheduled.dueUtc }));
    } catch {
      sent = false;
    }
    await d.batch([
      auditStmt(
        d,
        actorId,
        sent ? "account:transfer_notice_sent" : "account:transfer_notice_undelivered",
        fromPassportId,
        `${sent ? "ok" : "the mailer did not accept it"}${tag}`,
        ts,
      ),
      ...mark(sent ? "sent" : "undelivered"),
    ]);
  } catch (e) {
    // The record of a lost notice must not itself lose the transfer — same
    // stance as auditUndeliveredDecisionMail above.
    logError({ cid: crypto.randomUUID(), hop: "http", msg: "account transfer notice audit failed", err: e instanceof Error ? e.name : typeof e });
  }
}

/** I40-OPS (D-1169(4), W0 amendments 3-4): a same-item door's one-operator path, unless a retraction countersign is pending
 * on `subject` (warnings.ts countersignPendingOnSubject), when the same `run` waits for two distinct persons under `action`
 * (dual_control.ts COUNTERSIGN_HOLD_ACTIONS, UNCLAMPED). An empty subject (no such item) matches no review item. */
async function underCountersignHold<T>(
  d: DbLike,
  env: Record<string, unknown>,
  op: string,
  subject: string,
  action: string,
  key: string,
  run: () => Promise<T>,
  now: () => string,
): Promise<{ executed: true; result: T } | { executed: false; approvals: number; required: number }> {

  return outcome.executed ? { executed: true, result: outcome.result as T } : { executed: false, approvals: outcome.approvals, required: outcome.required };
}

/** I39-OPS: schedule a one-signature transfer. Refuses (LedgerError, 409 at the route) an account that is not on

export async function scheduleSoloTransfer(
  d: DbLike,
  a: { accountKey: string; fromPassport: string; to: string; heldSeq: string; op: string; nowMs: number; noAddress?: { signers: readonly string[] } },
): Promise<{ transfer_id: string; due_at: string; delay_hours: number }> {
  await assertTransferSchedulable(d, a.accountKey, a.fromPassport, a.to);
  const hours = await soloTransferDelayHours(d);
  const ts = new Date(a.nowMs).toISOString();
  const due = new Date(a.nowMs + hours * 3_600_000).toISOString();
  const id = mintId("stx");
  // I41-CONSOLE (D-1195(7) "a"; migration 0282): a no-address row stores every signer withDualControl consumed, in the
  // row's own INSERT (so in the audit's batch); runDueScheduledTransfers and revokeOperator's cancel read it. A
  // one-signer row stores none: its INSERT is i41-base's.
  const insert = a.noAddress
    ? d
        .prepare("INSERT INTO scheduled_account_transfers (transfer_id, account_key, from_passport, to_passport, held_seq, requested_by, requested_at, due_at, signers) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)")
        .bind(id, a.accountKey, a.fromPassport, a.to, a.heldSeq, a.op, ts, due, JSON.stringify([...new Set(a.noAddress.signers)]))
    : d
        .prepare("INSERT INTO scheduled_account_transfers (transfer_id, account_key, from_passport, to_passport, held_seq, requested_by, requested_at, due_at) VALUES (?, ?, ?, ?, ?, ?, ?, ?)")
        .bind(id, a.accountKey, a.fromPassport, a.to, a.heldSeq, a.op, ts, due);
  try {
    await d.batch([
      insert,
      auditStmt(d, a.op, "account:transfer_scheduled", a.to, `${id} due ${due}: ${a.noAddress ? "notice cannot be sent (no verified address; two operators approved): " : ""}${a.accountKey} from ${a.fromPassport}`.slice(0, 150), ts),
    ]);
  } catch (e) {
    if (/UNIQUE constraint failed: scheduled_account_transfers\.account_key/i.test(e instanceof Error ? e.message : String(e))) {
      throw new LedgerError("a transfer of this account is already scheduled — withdraw it first");
    }
    throw e;
  }
  return { transfer_id: id, due_at: due, delay_hours: hours };
}

/**
 * I39-OPS: THE EXECUTOR of scheduled transfers (idle until the scheduled sweep calls it — index.ts, I39-LAND). Each due
 * `scheduled` row is re-checked — the account still on `from_passport` at the `updated_seq` recorded when it was
 * requested, and (with `requireNotice`) the incumbent warned — then moved with the immediate path's effects: the same
 * `account_transferred` event and `account:transfer` audit row, actor = the requesting operator, the row's flip to
 * `executed` change-gated in the same append batch (a withdraw that landed first aborts it), then today's notice. A
 * refused row is `refused` with `account:transfer_scheduled_refused`; a row whose batch throws anything else stays
 * `scheduled` for the next run (counted in `failed`). At most `limit` rows a run, oldest due first.
 */
export async function runDueScheduledTransfers(
  d: DbLike,
  mailer: { send(template: string, to: string, vars: Record<string, string>): Promise<boolean> },
  nowMs: number,
  opts: { limit?: number; requireNotice?: boolean; env?: Record<string, unknown> } = {},
): Promise<{ executed: number; refused: number; skipped: number; failed: number }> {
  const now = new Date(nowMs).toISOString();
  const requireNotice = opts.requireNotice ?? SCHEDULED_TRANSFER_REQUIRES_NOTICE;
  const allowlist = await scheduledTransferAllowlist(d, opts.env);
  const { results } = await d
    .prepare(
      "SELECT transfer_id, account_key, from_passport, to_passport, held_seq, requested_by, notice, signers FROM scheduled_account_transfers WHERE state = 'scheduled' AND due_at <= ? ORDER BY due_at, transfer_id LIMIT ?",
    )
    .bind(now, Math.max(1, Math.min(opts.limit ?? 50, 200)))
    .all<{ transfer_id: string; account_key: string; from_passport: string; to_passport: string; held_seq: string; requested_by: string; notice: string; signers: string | null }>();
  const out = { executed: 0, refused: 0, skipped: 0, failed: 0 };
  for (const r of results) {
    const refuse = (reason: string) =>
      d.batch([
        d
          .prepare("UPDATE scheduled_account_transfers SET state = 'refused', settled_at = ?, settled_by = ?, outcome = ? WHERE transfer_id = ? AND state = 'scheduled'")
          .bind(now, SCHEDULED_TRANSFER_ACTOR, reason, r.transfer_id),
        changeGateStmt(d),
        auditStmt(d, SCHEDULED_TRANSFER_ACTOR, "account:transfer_scheduled_refused", r.to_passport, `${r.transfer_id}: ${reason}`, now),
      ]);
    try {
      const cur = await d.prepare("SELECT passport_id, updated_seq FROM passport_accounts WHERE account_key = ?").bind(r.account_key).first<{ passport_id: string; updated_seq: number }>();
      const reason =
        (await signersRefusal(d, [r.requested_by, ...(r.signers === null ? [] : (JSON.parse(r.signers) as string[]))], allowlist)) ??
        (requireNotice && r.notice !== "sent" ? "creator_not_warned" : cur?.passport_id !== r.from_passport || String(cur.updated_seq) !== r.held_seq ? "source_changed" : null);
      if (reason !== null) {
        await refuse(reason);
        out.refused += 1;
        continue;
      }
      try {
        await appendEvents(
          d,
          [{ event_type: "account_transferred", subject: r.to_passport, actor_kind: "operator", actor_id: r.requested_by, payload: { account_key: r.account_key, from_passport: r.from_passport } }],
          () => now,
          [],
          [
            d
              .prepare("UPDATE scheduled_account_transfers SET state = 'executed', settled_at = ?, settled_by = ? WHERE transfer_id = ? AND state = 'scheduled'")
              .bind(now, SCHEDULED_TRANSFER_ACTOR, r.transfer_id),
            changeGateStmt(d),
            auditStmt(d, r.requested_by, "account:transfer", r.to_passport, `${r.account_key} from ${r.from_passport}`.slice(0, 150), now),
          ],
        );
      } catch (e) {
        if (!(e instanceof LedgerError)) throw e;
        await refuse("ledger_refused");
        out.refused += 1;
        continue;
      }
      out.executed += 1;
      await notifyIncumbentAccountTransferred(d, mailer, r.from_passport, r.account_key, r.requested_by, now);
    } catch (e) {
      if (isChangeGateRefusal(e)) {
        out.skipped += 1; // withdrawn (or settled) between the read and the write: that state stands
        continue;
      }
      out.failed += 1;
      logError({ cid: crypto.randomUUID(), hop: "cron", msg: "scheduled account transfer failed", err: e instanceof Error ? e.name : typeof e });
    }
  }
  return out;
}

/**
 * I39-FX-OPS (D-1158(4), INV-030): the executor's second stop after `revokeOperator`'s cancel (operators.ts). A
 * requester with an `operator_revocations` row is refused `requester_revoked`; with an allowlist in force, one whose
 * `operator_identities.email` is missing or not on it is refused `requester_not_operator` (a removal from
 * `OPERATOR_EMAILS` writes no revocation row).
 */
/** I41-CONSOLE (D-1195(7) "a"): a no-address row is refused when ANY stored signer would be (requester first), answered
 * as a revoked requester is; a row with no stored signers (`signers` NULL) checks its requester alone. */
async function signersRefusal(d: DbLike, signers: readonly string[], allowlist: readonly string[] | null): Promise<string | null> {
  for (const s of new Set(signers)) {
    const reason = await requesterRefusal(d, s, allowlist);
    if (reason !== null) return reason;
  }
  return null;
}
async function requesterRefusal(d: DbLike, requestedBy: string, allowlist: readonly string[] | null): Promise<string | null> {
  if (await isOperatorRevoked(d, requestedBy)) return REVOKED_REQUESTER_OUTCOME;
  if (allowlist === null) return null;
  const row = await d.prepare("SELECT email FROM operator_identities WHERE op_id = ?").bind(requestedBy).first<{ email: string | null }>();
  return row?.email && allowlist.includes(row.email) ? null : "requester_not_operator";
}

/** The allowlist the executor checks: `env.OPERATOR_EMAILS` when the caller hands it a string (parsed as
 * dual_control.ts parses it: trim, lowercase), else the newest `operator:access_config` row (`recordAccessConfig`,
 * the configuration the last /ops request ran under; an unreadable one is an empty list), else null: none on record. */
async function scheduledTransferAllowlist(d: DbLike, env: Record<string, unknown> | undefined): Promise<string[] | null> {
  if (typeof env?.OPERATOR_EMAILS === "string") return env.OPERATOR_EMAILS.split(",").map((e) => e.trim().toLowerCase()).filter((e) => e.length > 0);
  const row = await d.prepare("SELECT detail FROM audit_log WHERE action = 'operator:access_config' ORDER BY audit_id DESC LIMIT 1").bind().first<{ detail: string }>();
  if (!row) return null;
  try {
    return (JSON.parse(row.detail) as { operator_emails: unknown[] }).operator_emails.map(String);
  } catch {
    return [];
  }
}

/** I39-OPS: every remaining operator is told of a break-glass revocation — each `OPERATOR_EMAILS` address whose
 * identity (by address or canonical address) is not revoked, template `operator-break-glass-revoke` (vars `revoked`,
 * `when_utc`). Best-effort and off the response path; mails the mailer did not accept are recorded as one
 * `operator:break_glass_notice_undelivered` row (a count, no address). */
export async function notifyOperatorsOfBreakGlass(
  d: DbLike,
  env: Record<string, unknown>,
  mailer: { send(template: string, to: string, vars: Record<string, string>): Promise<boolean> },
  op: string,
  revoked: string,
  ts: string,
): Promise<{ sent: number; failed: number }> {
  const out = { sent: 0, failed: 0 };
  try {
    const raw = typeof env.OPERATOR_EMAILS === "string" ? env.OPERATOR_EMAILS : "";
    const addresses = [...new Set(raw.split(",").map((e) => e.trim().toLowerCase()).filter((e) => e.length > 0 && !/todo/i.test(e)))];
    for (const to of addresses) {
      const idRow = await d
        .prepare("SELECT op_id FROM operator_identities WHERE email = ? OR canonical_email = ? ORDER BY op_id LIMIT 1")
        .bind(to, canonicalizeEmail(to))
        .first<{ op_id: string }>();
      if (idRow && (await isOperatorRevoked(d, idRow.op_id))) continue;
      let ok = false;
      try {
        ok = await mailer.send("operator-break-glass-revoke", to, { revoked, when_utc: ts });
      } catch {
        ok = false;
      }
      if (ok) out.sent += 1;
      else out.failed += 1;
    }
    if (out.failed > 0) await auditStmt(d, op, "operator:break_glass_notice_undelivered", null, `${out.failed} of ${out.sent + out.failed} operator notices not accepted`, ts).run();
  } catch (e) {
    logError({ cid: crypto.randomUUID(), hop: "http", msg: "break-glass operator notice failed", err: e instanceof Error ? e.name : typeof e });
  }
  return out;
}

export type CreditCorrectionOutcome = { applied: true; entry_id: string; credits: number } | { applied: false; refusal: string; duplicate?: true };
/**
 * I4-Q5: the `credit:correct` executor. Re-reads the cap (a lowered cap refuses a correction signed under the old one),
 * batches credits.ts `creditCorrectionStmts` with `idem` bound to the dual-control CLAIM (dual_control.ts claimIdFor), so a
 * re-run of one claim is a UNIQUE refusal and never a second row, then reads the entry back: the statements write
 * nothing when their guards refuse (unknown or terminated account, a debit below zero).
 */
export async function applyCreditCorrection(db: DbLike, accountId: string, credits: number, operator: string, reason: string, claim: string, now: () => string): Promise<CreditCorrectionOutcome> {
  const cap = await creditCorrectionCap(db);
  if (Math.abs(credits) > cap) return { applied: false, refusal: creditCorrectionCapRefusal(cap) };
  const { stmts, entryId } = creditCorrectionStmts(db, accountId, credits, operator, reason, { idem: `credit_correct:${claim}`, now });
  try {
    await db.batch(stmts);
  } catch (e) {
    if (e instanceof Error && /UNIQUE constraint failed:\s*credit_ledger\.idem/i.test(e.message)) return { applied: false, refusal: CREDIT_CORRECTION_DUPLICATE, duplicate: true };
    throw e;
  }
  const landed = await db.prepare("SELECT 1 AS x FROM credit_ledger WHERE entry_id = ?").bind(entryId).first();
  return landed === null ? { applied: false, refusal: CREDIT_CORRECTION_REFUSED } : { applied: true, entry_id: entryId, credits };
}

/**
 * I29-FX4 (FX3-05 / FX-HOLDS H4, D-1039(2)): EVIDENCE KEPT UNDER A LEGAL HOLD. reports.ts notifyReportersOnWarning keeps a
 * held item's evidence on retraction and writes `evidence:purge_held`; nothing purges it later by itself. This lists each
 * item with such a row whose `report_evidence` rows still exist (its takedown is still owed), oldest hold first, with
 * whether a hold still keeps it — D-1042(2): reports.ts itemPurgeHeldSql, the check the purge doors and the retention sweep
 * ask (the item, an uploader's account, the item's passport). Served on GET /ops/fan-reports as `evidence_held`;
 * at most HELD_EVIDENCE_LIST_MAX items, and `truncated: true` says more are waiting beyond them.
 * I30-CONSOLE FX10-11 (D-1051(3)): `closed_as` is the closing action that kept it, read off the item's LATEST
 * `evidence:purge_held` row: 'retracted' when that row is the retraction leg's (reports.ts notifyReportersOnWarning,
 * actor HELD_RETRACTION_ACTOR), else the item's `fan_report_items.state` when 'dismissed' or 'suppressed' (evidence.ts
 * purgeItemEvidenceQuietly, reached from dismissReport / suppressReport), else null — the admin banner names no action.
 */
/** The actor reports.ts writes the retraction leg's `evidence:purge_held` row under (recordNoticeLoss). */
export const HELD_RETRACTION_ACTOR = "system:reports";
export const HELD_EVIDENCE_LIST_MAX = 50;
export interface HeldEvidenceItem {
  item_id: string;
  objects: number;
  held_at: string;
  hold_open: boolean;
  closed_as: "retracted" | "dismissed" | "suppressed" | null;
}
export async function heldEvidenceList(db: DbLike): Promise<{ items: HeldEvidenceItem[]; truncated: boolean }> {
  const { results } = await db
    .prepare(
      `SELECT a.item_id AS item_id, MAX(a.created_at) AS held_at, (SELECT COUNT(*) FROM report_evidence e WHERE e.item_id = a.item_id) AS objects, ${itemPurgeHeldSql("a.item_id")} AS hold_open,
              CASE WHEN (SELECT l.actor_id FROM audit_log l WHERE l.action = 'evidence:purge_held' AND l.item_id = a.item_id ORDER BY l.created_at DESC, l.audit_id DESC LIMIT 1) = ? THEN 'retracted'
                   ELSE (SELECT i.state FROM fan_report_items i WHERE i.item_id = a.item_id AND i.state IN ('dismissed', 'suppressed')) END AS closed_as
         FROM audit_log a WHERE a.action = 'evidence:purge_held' AND EXISTS (SELECT 1 FROM report_evidence e WHERE e.item_id = a.item_id)
        GROUP BY a.item_id ORDER BY held_at ASC, a.item_id ASC LIMIT ?`,
    )
    .bind(HELD_RETRACTION_ACTOR, HELD_EVIDENCE_LIST_MAX + 1)
    .all<{ item_id: string; held_at: string; objects: number; hold_open: number; closed_as: HeldEvidenceItem["closed_as"] }>();
  const rows = results ?? [];
  return {
    items: rows.slice(0, HELD_EVIDENCE_LIST_MAX).map((r) => ({ item_id: r.item_id, objects: Number(r.objects), held_at: r.held_at, hold_open: r.hold_open === 1, closed_as: r.closed_as ?? null })),
    truncated: rows.length > HELD_EVIDENCE_LIST_MAX,
  };
}

export function freezeReleaseIdems(raw: unknown): string[] | null {
  if (!Array.isArray(raw) || raw.length === 0 || raw.length > FREEZE_RELEASE_MAX) return null;
  const out = new Set<string>();
  for (const x of raw) {
    const p = x !== null && typeof x === "object" ? (x as { purchase_id?: unknown }).purchase_id : undefined;
    const k = x !== null && typeof x === "object" ? (x as { dispute_key?: unknown }).dispute_key : undefined;
    if (typeof p !== "string" || !FREEZE_PURCHASE.test(p) || typeof k !== "string" || k.length === 0 || k.length > 200 || hasUnsafeIdentifierText(k)) return null; // I29-FX4 (FX3-14): the shared strict class (#523 census)
    out.add(`dispute_freeze:${p}:${k}`);
  }
  return [...out].sort();
}

/** Parse a limits body `{ <key>: <whole number> , ... }`: every key known, every value a whole number in its range. */
export function parseLimitSettings(body: unknown): { entries: [string, string][] } | { error: string } {
  if (!body || typeof body !== "object" || Array.isArray(body)) return { error: "send an object of setting keys to whole numbers" };
  const entries: [string, string][] = [];
  for (const [k, v] of Object.entries(body as Record<string, unknown>)) {
    const spec = OPS_LIMIT_SETTINGS.find((s) => s.key === k);
    if (!spec) return { error: `unknown setting ${k.slice(0, 60)}` };
    const n = typeof v === "number" ? v : typeof v === "string" && /^[0-9]{1,9}$/.test(v.trim()) ? Number(v.trim()) : NaN;
    if (!Number.isInteger(n) || n < spec.min || n > spec.max) return { error: `${k} must be a whole number from ${spec.min} to ${spec.max}` };
    entries.push([k, String(n)]);
  }
  if (entries.length === 0) return { error: "send at least one setting" };
  return { entries: entries.sort(([a], [b]) => (a < b ? -1 : a > b ? 1 : 0)) };
}
/** GET /ops/settings/limits: each key's stored value, the value its reader uses, its range and default. */
export async function opsLimitSettings(db: DbLike): Promise<{ key: string; stored: string | null; effective: number; min: number; max: number; default: number; unit: string }[]> {
  const out = [];
  for (const s of OPS_LIMIT_SETTINGS) {
    const stored = await getSetting(db, s.key);
    const n = stored !== null && /^[0-9]{1,9}$/.test(stored.trim()) ? Number(stored.trim()) : NaN;
    out.push({ key: s.key, stored, effective: Number.isInteger(n) && n >= s.min && n <= s.max ? n : s.def, min: s.min, max: s.max, default: s.def, unit: s.unit });
  }
  return out;
}

    // UNDECIDED, and its docblock says why: the thirty days are "enforced HERE
    // rather than by an operator's memory". This door is strictly MORE
    // destructive — termination forfeits invented credits and keeps the
    // evidence, erasure destroys the account AND the evidence with it,
    // irreversibly — and until now it asked only for the id's SHAPE and a
    // matching `confirm`. An operator who could not terminate an account with
    // a live appeal could erase it instead, and the appeal, the closure row
    // and the person's case all went with it.
    //
    // ONE CLAUSE OF THE FIVE, and the choice is the point. The other four
    // (`no closure`, `already terminated`, `lifted`, `deadline not passed`)
    // are about the TWO-STAGE PATH, and importing them here would make an
    // ordinary erasure — a `legal_compliance` ground, a fan's own right
    // exercised through the fan door's twin — depend on a closure that need
    // not exist. This clause alone is about the PERSON'S OUTSTANDING RIGHT: a
    // case a human has not yet decided.
    //
    // BEFORE `withDualControl`, so an appeal opened after a first sign-off
    // still stops the second approver's request — the check runs on every
    // request that reaches this door, not once per action. A banked sign-off
    // that can no longer execute is the shape slice 21 already refused here.
    const closure = await closureFor(d, id);
    if (closure && closure.appeal_opened_at && !closure.appeal_decided_at) {
      return c.json({ error: "an appeal is open and undecided on this account" }, 409);
    }
    // I3-0567 (I26-W1-CONTENT, D-0522): THE ERASURE FULFILS THE REQUEST ALREADY OPEN. It used to record a fresh request
    // "arrived" at fulfilment, so an erasure answered late read as on time and the open one stayed open. Resolved here,
    // before the sign-off is banked: `dsr_id` names it (an open `erasure` row, else 409). The sign-off key carries it, so
    // both approvers approve the same pairing. None open for this account -> recorded fresh below as before.
    // I26-FX-AI (D-1015(2), JN-05 I26C-02 + I26-INT-09, RG): BOUND TO THE SUBJECT. A request is this account's when its
    // subject is the account id OR the account's email (`POST /ops/dsr` stores the subject as given — usually an email):
    // a named request whose subject is neither is refused (409). D-1024(5) widens "this account's" to every sign-in
    // identity's email and replaces "one open -> id REQUIRED" with "one open -> paired" (below); several open -> 409
    // `dsr_required`, listing no id and no subject. So no erasure closes another person's request or runs beside an open
    // one unpaired.
    const namedDsr = body?.dsr_id;
    if (namedDsr !== undefined && (typeof namedDsr !== "string" || !DSR_ID.test(namedDsr))) return c.json({ error: "no such request" }, 404);
    if (namedDsr !== undefined) {
      const named = await d
        .prepare("SELECT subject FROM data_subject_requests WHERE request_type = 'erasure' AND state IN ('received', 'in_progress') AND dsr_id = ?")
        .bind(namedDsr)
        .first<{ subject: string }>();
      if (!named) return c.json({ error: "dsr_id is not an open erasure request" }, 409);
      if (!dsrSubjectNamesAccount(named.subject, id, await accountSignInEmails(d, id))) return c.json({ error: "dsr_id is not an open erasure request about this account" }, 409);
    }
    // D-1024(5) (I28-ERASURE): a BLANK dsr_id pairs the single open request about this account (same predicate as a named
    // one); several are refused with a request for the id (listing none); none -> recorded fresh below as before.
    let openDsr: string | null = namedDsr ?? null;
    if (namedDsr === undefined) {
      const open = await openErasuresFor(d, id);
      if (open.length > 1) return c.json({ error: "more than one open erasure request names this account: give the dsr_id of the one this erasure fulfils", dsr_required: true }, 409);
      openDsr = open[0] ?? null;
    }
    // THE FAN DOOR'S DEFERRAL, ON THIS DOOR TOO (A-2). The post-commit fan-out
    // is an outbound Stripe call with no timeout; awaited here it held an
    // ALREADY-COMMITTED operator erasure open for as long as Stripe stayed
    // silent, and the operator's own `account:delete` row — written after the
    // fan-out — did not exist while it hung. Captured BEFORE the erasure by the
    // shared helper (account_data.ts `captureWaitUntil`), never re-typed here.
    // With no ExecutionContext (a test, a non-fetch caller) the fan-out is
    // collected instead of awaited inside the primitive, so the audit row still
    // precedes it, and the request then waits for it exactly as it did before.
    const ctxWaitUntil = captureWaitUntil(c);
    const deferred: Promise<unknown>[] = [];
    const waitUntil = (p: Promise<unknown>): void => {
      if (ctxWaitUntil) ctxWaitUntil(p);
      else deferred.push(p);
    };
    try {
      const outcome = await withDualControl(d, c.env as Record<string, unknown>, op, "account:delete", openDsr === null ? id : `${id}|${openDsr}`, async (signers) => {
        // THE SAME PRIMITIVE THE FAN DOOR CALLS (S-013 / AS-007). Before this,
        // the operator door erased and left the card being charged, so one
        // right had two financial outcomes depending on who exercised it. This
        // closure is ALSO the dual-control deferred execution — `withDualControl`
        // re-enters it inside the second approver's request — so there is no
        // third site to route.
        const env = c.env as unknown as Record<string, unknown>;
        // S-021/docket #52 half 2 (CONSOLEENV, docket #54): the fan door
        // (fan-account-routes.ts:331,339) threads pubEnv/pubCtx so its erasure
        // publishes to the outbox immediately; this door used to have no way
        // to typecheck the same call (ConsoleEnv carried none of TRUST_DB /
        // ANALYTICS_DB / ANALYTICS_QUEUE) and fell back to the cron drain only
        // (QUEUE #23). Same shape as the fan door, exactly.
        const [pubEnv, pubCtx] = analyticsPublishArgs(c);
        const r = await eraseAccountWithSubscriptions(d, id, {
          actorId: op,
          actorKind: "operator",
          env,
          waitUntil,
          evidence: evidence(c),
          nowMs: nowMs(),
          ...(pubEnv && pubCtx ? { pubEnv, pubCtx } : {}),
        });
        // BEFORE THE FAN-OUT, ALWAYS: this is the one row an operator
        // reconciles a delete from, and the erasure it names has committed.
        // QUEUE #378 (D-0543, built under D-0808): written after the sweep, so
        // `item_id` is pseudonymised at WRITE time; the operator actor is kept.
        //
        // QUEUE #126 L1 (OP-03/FX-10): the erasure above has COMMITTED, so this
        // receipt cannot decide the answer. `withDualControl` keeps no catch
        // around execute() (D-0646), so the guard is here: a failed receipt is
        // logged with the error class name only (never the message), and the operator is
        // told through `audit_recorded:false` instead of a 500.
        let auditRecorded = true;
        try {
          // I39-OPS (D-1158(6b)): the co-signer rows ride the receipt's batch (the erasure above is several batches).
          await d.batch([erasedSubjectAuditStmt(d, op, "operator", "account:delete", id, "operator erasure", new Date(nowMs()).toISOString()), ...coSignerAuditStmts(d, "account:delete", op, signers, new Date(nowMs()).toISOString())]);
        } catch (e) {
          auditRecorded = false;
          logError({ cid: crypto.randomUUID(), hop: "http", route: "/ops/accounts/:id/delete", msg: "operator erasure audit write failed", err: e instanceof Error ? e.name : typeof e });
        }
        // I3-0567 a (I22-FX-HUB): the erasure IS a fulfilled erasure request, so it is on the request clock's record.
        // Written after the erasure, like the receipt above, so the subject is a fresh `del_` pseudonym and never the
        // erased id (QUEUE #378's write-time rule); keeping the requester's id is Jason's retention call, not made here.
        // Best-effort for the same reason: the erasure has committed and this row cannot decide the answer.
        // I3-0567 (I26-W1-CONTENT): an open request resolved above is CLOSED as fulfilled, keeping its arrival; a close
        // that lands between the resolve and here (the request withdrawn meanwhile) changes no row and reports false.
        let dsrRecorded = true;
        try {
          const ts = new Date(nowMs()).toISOString();
          if (openDsr !== null) {
            const res = await d.prepare("UPDATE data_subject_requests SET state = 'fulfilled', closed_at = ? WHERE dsr_id = ? AND state IN ('received', 'in_progress') RETURNING dsr_id").bind(ts, openDsr).all();
            if (res.results.length !== 1) throw new Error("the open request was closed meanwhile");
            await auditStmt(d, op, "dsr:close", openDsr, "closed as fulfilled by operator erasure", ts).run();
          } else await d
            .prepare("INSERT INTO data_subject_requests (dsr_id, request_type, subject, state, arrival_at, closed_at) VALUES (?, 'erasure', 'del_' || lower(hex(randomblob(16))), 'fulfilled', ?, ?)")
            .bind(`dsr_${[...crypto.getRandomValues(new Uint8Array(16))].map((b) => b.toString(16).padStart(2, "0")).join("")}`, ts, ts)
            .run();
        } catch (e) {
          dsrRecorded = false;
          logError({ cid: crypto.randomUUID(), hop: "http", route: "/ops/accounts/:id/delete", msg: "operator erasure request record failed", err: e instanceof Error ? e.name : typeof e });
        }
        if (deferred.length > 0) await Promise.allSettled(deferred.splice(0));
        return { ...r, audit_recorded: auditRecorded, dsr_recorded: dsrRecorded };
      });
      if (!outcome.executed) return c.json({ ok: true, pending: true, approvals: outcome.approvals, required: outcome.required });
      return c.json({ ok: true, ...outcome.result });
    } catch (e) {
      return mapError(c, e);
    }
  });

        const warnWaitUntil = captureWaitUntil(c) ?? ((p: Promise<unknown>) => void Promise.resolve(p).catch(() => {}));
