function withBotTokens(init: RequestInit | undefined): RequestInit | undefined {
  if (!BOT_WIDGETS.turnstile && !BOT_WIDGETS.solvemedia) return init;
  const raw = init?.body;
  if (raw !== undefined && raw !== null && typeof raw !== "string") return init;
  let body: Record<string, unknown>;
  try {
    body = raw ? (JSON.parse(raw) as Record<string, unknown>) : {};
  } catch {
    return init;
  }
  if (BOT_WIDGETS.turnstile) body.turnstile_token = turnstileToken;

  return { ...init, body: JSON.stringify(body) };
}

export function navScopedFetch(input: RequestInfo | URL, init?: RequestInit): Promise<Response> {
  const method = (init?.method ?? "GET").toUpperCase();
  if (method === "POST" && botDoor(input)) return fetch(input, withBotTokens(init)).finally(resetTurnstile);
  if (method !== "GET" && method !== "HEAD") return fetch(input, init);
  const nav = navAbort.signal;
  const own = init?.signal ?? null;
  const link = new AbortController();
  const stop = (): void => link.abort();
  if (nav.aborted || own?.aborted) link.abort();
  nav.addEventListener("abort", stop, { once: true });
  own?.addEventListener("abort", stop, { once: true });
  const done = (): void => {
    nav.removeEventListener("abort", stop);
    own?.removeEventListener("abort", stop);
  };
  return fetch(input, { ...init, signal: link.signal }).finally(done);
}

/**
 * F8 — the OAuth RETURN leg.
 *
 * The provider redirects the browser back with `?code=&state=`. This exchanges
 * them over a POST and then STRIPS THEM FROM THE URL before anything renders,
 * so the single-use code does not sit in the address bar, in history, or in a
 * referer header on the next outbound link.
 *
 * The `state` is NOT checked here, and deliberately: the server binds the
 * challenge row to the caller's own account and reads it with that account in
 * the WHERE, so a foreign state is already indistinguishable from one that
 * never existed. A browser-side copy would have added nothing but another place
 * for a capability to live — and this app persists none (`m8-guards.test.ts`).
 */
/** Resolves `true` when it navigated (`go` renders by itself), `false` when there was no return to finish. */
async function finishPendingConnection(): Promise<boolean> {
  const q = new URLSearchParams(window.location.search);
  const code = q.get("code");
  const state = q.get("state");
  if (!code || !state) return false;
  // Strip FIRST, so a failure below still leaves no code in the URL.
  window.history.replaceState(null, "", window.location.pathname + window.location.hash);
  // I3-0740: a return that lands after the session expired used to be dropped
  // without a word — the fan thought the platform was connected. The code is
  // single-use and already stripped, so it is never retried: say so and sign in.
  if (!session.isSignedIn()) {
    goKeeping("#/signin", "You were signed out before that connection finished, so nothing was connected. Sign in and start the connection again.");
    return true;
  }
  const r = await deps.api.finishConnection(state, code);
  if (!r.ok) {
    const outcome = typeof r.data?.outcome === "string" ? r.data.outcome : "";
    // I3-0538a — the reason is shown ON the account page (a note, with a Try
    // again control for the one outcome a retry can change), not in a browser alert.
    noteThen(refusalText(r,
      outcome === "already_linked_elsewhere"
        ? "That account is already connected to a different Dhwaja account. Each platform account can back one Dhwaja account."
        : outcome === "held"
          ? "That account was disconnected from another Dhwaja account recently and is reserved to them for 30 days."
          : "We couldn't complete that connection. Nothing was saved — try again."),
      outcome !== "already_linked_elsewhere" && outcome !== "held",
    );
  }
  go("#/me");
  return true;
}

/** PS-101 / T-04 — the navigation epoch. `renderRoute` is async, so two
 * renders can be in flight (hashchange during a slow fan-out) and the SLOWER
 * one used to win the innerHTML write, painting a route the fan had already
 * left. Every render takes an epoch; a render whose epoch is stale when its
 * await returns paints nothing and wires nothing. The guard is the comparison
 * at `epoch !== renderEpoch`, on both the success and the backstop arm.
 *
 * I3-1212 — THE GUARD IS A COUNTER, SO IT SEES ONLY THE WRITERS THAT CLAIM.
 * It used to be claimed too late to cover render()'s OWN early return: the
 * account-closed arm below painted the notice and RETURNED above the claim, so
 * it made nothing stale, and a passport fan-out already parked on its await came
 * back with `epoch === renderEpoch` still true and wiped the one screen that
 * carries the fan's appeal address and its deadline. Every writer of `root`
 * claims through `claimEpoch()`; a paint that does not claim is invisible to the
 * comparison, whatever the comparison says. */
let renderEpoch = 0;

/** Claim the navigation epoch for a paint of `root`: any render still parked on
 * its await is stale from here on. Called by `render()` and by every out-of-band
 * painter — `paint()` and the search fallback — which write `root` without
 * going through the router at all. */
function claimEpoch(): number {
  navAbort.abort(); // I3-0801 — the previous route's reads are abandoned from here
  navAbort = new AbortController();
  return ++renderEpoch;
}

/** I3-2121 — the signed-in state the last render painted; the `storage`
 * listener at the bootstrap tail repaints only when another tab flips it. */
let seenSignedIn = session.isSignedIn();

/** I3-1461 (I4-10-R5): the route last painted. A REFRESH of that same route
 * (an action re-reading its page) keeps the fan's scroll position across the
 * innerHTML swap; a NAVIGATION (hashchange, or go() to the current hash) still
 * resets to the top (PS-109). */
let lastPaintedHash: string | null = null;
async function render(nav = false): Promise<void> {
  seenSignedIn = session.isSignedIn();
  const route = parseHash(window.location.hash);
  const keepY = !nav && buildHash(route) === lastPaintedHash ? window.scrollY : null;
  // Claimed HERE rather than at the fan-out below: every early return in this
  // function either paints or navigates, and each one must make an in-flight
  // render stale. The two `go(...)` arms claim too — a navigation HAS happened.
  const epoch = claimEpoch();
  // I3-1594: a route whose page lives on the SSR origin leaves the SPA here —
  // renderRoute (app.ts) would otherwise paint home under #/how-it-works.
  const external = externalHref(route, deps.ssrBase ?? null);
  if (external) {
    window.location.assign(external);
    return;
  }
  if (route.name === "signin" && session.isSignedIn()) return go("#/me"); // already in → skip the sign-in screen
  // The lost-passkey path is for someone who is OUT. Already in → send them to
  // the page that manages this, not to a form that would mail them a code they
  // do not need.
  if (route.name === "signin-email" && session.isSignedIn()) return go("#/security");
  // PS-323 teardown, on the ROUTE CHANGE rather than on the close button only:
  // the trap's release is what clears `inert`/`aria-hidden` on this very
  // element (a11y.ts), and the swap below does not. A Back press taken with
  // the modal open reaches render() and no close, so without this line the new
  // route paints behind a still-open dialog on a still-inert root, with no
  // reachable control to release it. Idempotent when no modal is open.
  closeRequestModal();
  // I3-0002 — A CLOSED ACCOUNT IS TOLD, AT THE DOOR, WHAT THE SERVER SAID.
  //
  // `closures.ts:110` states the scope itself: this notice is "what a sign-in
  // door tells somebody who has just PROVED they hold the account". So it is
  // painted on the sign-in routes and nowhere else — a closed account can still
  // read a public passport page, and nothing here takes that away. The notice
  // is recorded in `api.ts`'s two request primitives, so it is set by whichever
  // door answered 403: OAuth exchange, passkey assertion or the OTP verify.
  const closure = deps.api.lastClosure();
  if (closure && (route.name === "signin" || route.name === "signin-email")) {
    root.innerHTML = shell(accountClosedView(closure, { rights: deps.api.hasRightsSession() }), { signedIn: false, route: buildHash(route) });
    wireRightsDoors(); // QUEUE #417 — no-op unless the view rendered the two doors
    document.title = `Account closed · Dhwaja`;
    root.setAttribute("tabindex", "-1");
    root.focus();
    announce(document, "This account is closed");
    return;
  }
  root.innerHTML = `<div class="loading">Loading…</div>`;
  try {
    // I3-0602 (D-1112(3)): `#/odds` is app.ts renderRoute's own arm (I34-FX1-WEB X1, I34-W2-CLIENTS).
    const html = await renderRoute(route, deps);
    if (epoch !== renderEpoch) return; // a newer route is already painting
    root.innerHTML = html;
  } catch {
    if (epoch !== renderEpoch) return;
    // Last-resort backstop: a view must never leave the page stuck on
    // "Loading…". The pure views already fail soft; this catches anything else.
    // I3-1484: a one-off failure gets a retry next to the way out.
    root.innerHTML = `<div class="err"><p>Something went wrong loading this page.</p><button type="button" data-act="retry-route" class="ds-btn act primary">Try again</button> <a href="#/" data-nav="#/" class="ds-btn nav">← Home</a></div>`;
  }
  wire();
  paintSaveNotice();
  root.querySelector<HTMLButtonElement>('[data-act="retry-route"]')?.addEventListener("click", () => void render());
  // PS-109/PS-313 — a hashchange painted a new route with no title update, no
  // scroll reset and no focus/announcement: the fan's scroll position and
  // focus (usually the link/back-button that caused the change, or nothing)
  // both carried over from the OLD page onto the new one, and AT users got no
  // signal a navigation happened at all. `check` keeps its own richer,
  // content-derived focus + announcement (`announceVerdict`, CL:D-05) but was
  // missing the same title update and scroll reset as every other route.
  document.title = `${ROUTE_TITLE[route.name] ?? "Dhwaja"} · Dhwaja`;
  lastPaintedHash = buildHash(route);
  try {
    window.scrollTo(0, keepY ?? 0); // not implemented in every test/embedding realm — never block a render on it
  } catch {
    /* noop */
  }
  if (route.name === "check") {
    // A pending landing is deliberately LEFT SET here: `announceVerdict` is the
    // richer, content-derived sentence and must not be talked over, and the next
    // render delivers the landing instead. Deferred, never dropped.
    announceVerdict();
  } else {
    root.setAttribute("tabindex", "-1");
    root.focus(keepY !== null ? { preventScroll: true } : undefined); // I3-1461: a refresh keeps its place
    // I3-0537 — a landing that outlived the navigation that set it OUTRANKS the
    // routine "<Route> loaded": a fan whose account has just been irreversibly
    // deleted needs to be told THAT, not where they ended up. This flush lived in
    // `wire()` in the first version of this fix, and the announcement on the very
    // next line overwrote it every time — caught by execution, not by reading.
    const landed = pendingLanding;
    pendingLanding = null;
    announce(document, landed ?? `${ROUTE_TITLE[route.name] ?? "Dhwaja"} loaded`);
    const note = pendingNote; // I3-1163b / I3-0538a — see noteThen
    pendingNote = null;
    if (note) showLandingNote(note);
  }
}

function announceVerdict(): void {

function go(hash: string): void {

  // (e.g. a redeem deep-link) can resume after auth.

/**
 * FU-07 (S-068) — A FAILED LOAD IS NOT A CACHED SUCCESS.
 *
 * This resolved immediately whenever ANY element with that `src` existed,
 * without ever checking whether that element had loaded or ERRORED — and the
 * failed element was left in `<head>`. One transient network failure therefore
 * poisoned every later attempt in the document: the Apple button (a direct
 * mounted control) resolved off the dead element, found no `AppleID` global and
 * failed again having issued no fresh request, and the Google container — whose
 * whole UI is rendered BY the SDK — stayed permanently empty, including after
 * leaving Sign in and coming back, because `setupSignin` re-ran into the same
 * dead element. Only a full document reload cleared it. That is the way back
 * in, so it is treated as lockout-adjacent rather than cosmetic.
 *
 * Three states now, not one: an in-flight load is SHARED (concurrent callers no
 * longer resolve early off a half-appended element), a LOADED script is still
 * cached (one element, one request — a retry must not re-fetch a working SDK),
 * and a FAILED one is removed from both the map and the document so the next
 * DELIBERATE attempt goes to the network again. Nothing here retries on its own.
 */
const scriptLoads = new Map<string, Promise<void>>();

function loadScript(src: string): Promise<void> {
  const inflight = scriptLoads.get(src);
  if (inflight) return inflight;
  const p = new Promise<void>((resolve, reject) => {
    // `data-loaded` is set by our own onload handler, so this is OUR record of
    // a completed load — not the mere presence of a tag, which is what the old
    // check read and what a failed attempt also leaves behind.
    if (document.querySelector(`script[src="${src}"][data-loaded="1"]`)) return resolve();
    // Clear anything a previous FAILED attempt left behind before re-requesting.
    for (const dead of Array.from(document.querySelectorAll(`script[src="${src}"]`))) dead.remove();
    const s = document.createElement("script");
    s.src = src;
    s.async = true;
    s.onload = () => {
      s.dataset.loaded = "1";
      resolve();
    };
    s.onerror = () => {
      s.remove();
      reject(new Error("script load failed"));
    };
    document.head.appendChild(s);
  });
  scriptLoads.set(src, p);
  // The awaiting caller still sees the rejection (each has its own catch); this
  // only drops the poisoned entry. Attached BEFORE the promise is handed out so
  // a synchronous rejection cannot be seen as unhandled.
  void p.catch(() => {
    scriptLoads.delete(src);
  });
  return p;
}

async function onSigninToken(provider: "google" | "apple", idToken: string | undefined, nonce: string): Promise<void> {
  const msg = root.querySelector<HTMLElement>("[data-signin-msg]");
  if (!idToken) {
    if (msg) msg.textContent = "Sign-in didn't finish, so you're not signed in. Try again.";
    return;
  }
  const signin = await deps.api.oauthSigninResult(provider, idToken, nonce, pendingReferral());
  if (signin.ok) {
    // M13: the code has now been offered to the server (attributed only if this
    // was a brand-new account); either way it must not ride a second sign-in.
    clearReferral();
    // EVERY gated action gets its place back, not just a redeem deep-link.
    // `consumeReturn` parses the remembered hash through `safeReturnHash` and
    // re-emits the canonical form, so the return is validated rather than
    // trusted — and it CLEARS it, so no later sign-in replays it (FU-08).
    go(consumeReturn());
  } else if (deps.api.lastClosure()) {
    // "Please try again" is FALSE advice for a closed account — trying again
    // cannot work, and the two facts that can still help (the appeal address
    // and its deadline) were being thrown away. Repaint through render(), which
    // owns the closed-account branch, rather than writing a second copy here.
    await render();
  } else if (msg) {
    msg.textContent = refusalText(signin, "Sign-in failed. Please try again.");
  }
}

// Load + wire the provider SDKs on the sign-in screen. This is the only place

    btn.classList.remove("primary"); // I3-2078 sibling (I22-FX-WEB): a requested state is never the primary style
    // left disabled — a terminal, honest state, no retry loop.
  } else {
    btn.textContent = "Try again";
    btn.removeAttribute("disabled");
    if (reqMsg) reqMsg.textContent = refusalText(r, ""); // I24 (D-0996(2)): the server's sentence when it sent one
  }
}

/** WebAuthn feature detection — absent APIs degrade to a graceful message +
 * the #/signin fallback link, never a broken button. */
/** I3-0934 (I34-W2-CLIENTS): the card the three passkey sign-in doors show for a ceremony that did not sign in. */
function signInOutcomeCard(outcome: Exclude<PasskeyOutcome, "done" | "signin_only">): Outcome {
  return outcome === "expired" ? PASSKEY_SIGNIN_EXPIRED : passkeySignInOutcome(outcome === "signin_failed" ? "failed" : outcome);
}

function passkeySupported(): boolean {
  return typeof navigator !== "undefined" && !!navigator.credentials && typeof window.PublicKeyCredential !== "undefined";
}

/** I3-1429 — ONE PASSKEY CEREMONY AT A TIME, AND A STUCK ONE CAN BE REPLACED.
 * An OS interruption (a call, another system sheet) can leave a WebAuthn call
 * pending with nothing on screen; the browser then refuses every new ceremony
 * as "a request is already pending", so a retry read as a cancel and the modal
 * never recovered. Each ceremony takes this signal: starting a new one aborts
 * the old one, and closing the modal (or leaving the route) aborts it too. */
let ceremony: AbortController | null = null;
function beginCeremony(): AbortSignal {
  ceremony?.abort();
  ceremony = new AbortController();
  return ceremony.signal;
}
function abortCeremony(): void {
  ceremony?.abort();
  ceremony = null;
}

/** The ONLY navigator.credentials.get call in the app. Encodes exactly the
 * base64url fields finishFanAuth expects; null on cancel/failure. */
async function getAssertionAdapter(opts: FanAssertionOptionsDto): Promise<AssertionFieldsDto | null> {
  const cred = (await navigator.credentials.get({
    signal: beginCeremony(),
    publicKey: {
      challenge: b64urlToBytes(opts.challenge).buffer as ArrayBuffer,
      rpId: opts.rpId,
      allowCredentials: [], // usernameless — the resident key identifies the account
      userVerification: (opts.userVerification || "required") as UserVerificationRequirement,
    },
  })) as PublicKeyCredential | null;
  if (!cred) return null;
  const resp = cred.response as AuthenticatorAssertionResponse;
  return {
    credentialId: cred.id, // already base64url per the WebAuthn spec
    clientDataJSON: bytesToB64url(resp.clientDataJSON),
    authenticatorData: bytesToB64url(resp.authenticatorData),
    signature: bytesToB64url(resp.signature),
  };
}

/** The ONLY navigator.credentials.create call in the app (the JOIN path —
 * account birth happens server-side at finish). The user entry is a local,
 * random placeholder: the server binds the account by challenge, never by
 * this client-chosen id, and no email/name is collected here. */
async function createCredentialAdapter(opts: FanRegistrationOptionsDto): Promise<AttestationFieldsDto | null> {
  const cred = (await navigator.credentials.create({
    signal: beginCeremony(),
    publicKey: {
      challenge: b64urlToBytes(opts.challenge).buffer as ArrayBuffer,
      rp: { id: opts.rp.id, name: "Dhwaja" },
      user: { id: crypto.getRandomValues(new Uint8Array(16)).buffer as ArrayBuffer, name: "Dhwaja fan", displayName: "Dhwaja fan" },
      pubKeyCredParams: opts.pubKeyCredParams as PublicKeyCredentialParameters[],
      authenticatorSelection: {
        residentKey: (opts.authenticatorSelection.residentKey || "required") as ResidentKeyRequirement,
        requireResidentKey: true,
        userVerification: (opts.authenticatorSelection.userVerification || "required") as UserVerificationRequirement,
      },
      attestation: "none",
    },
  })) as PublicKeyCredential | null;
  if (!cred) return null;
  const resp = cred.response as AuthenticatorAttestationResponse;
  return { clientDataJSON: bytesToB64url(resp.clientDataJSON), attestationObject: bytesToB64url(resp.attestationObject) };
}

/**
 * THE JOIN DOOR'S adapter. Identical ceremony, one extra field: the inbound
 * referral code captured at `:177`, carried on the join body so the server can
 * attribute the new account at birth (`fan-account-routes.ts:541` links it on
 * `r.created`, and ignores a bad/foreign/self code — a referral never fails a
 * signup). The OAuth leg has carried it since M13 (`:451`); this door did not,
 * and the passkey door is the ONLY signup door on a deployment with no OAuth
 * provider configured, so a referred fan cost the referrer their credit with
 * nothing logged.
 *
 * SEPARATE from `createCredentialAdapter` on purpose: that one also serves the
 * two ADD doors (`:1897`, `:2125`), where a session already exists, `created`
 * is false and a referral code could never be spent.
 */
async function joinCredentialAdapter(opts: FanRegistrationOptionsDto): Promise<(AttestationFieldsDto & { referral_code?: string }) | null> {
  const fields = await createCredentialAdapter(opts);
  if (!fields) return null;
  const ref = pendingReferral();
  return ref ? { ...fields, referral_code: ref } : fields;
}

/** Release for the modal's focus trap; null when no modal is open. Held here
 * because open and close are two functions and the page must never be left
 * `inert` by a close that forgot (PS-323). */
let modalRelease: (() => void) | null = null;

function closeRequestModal(): void {
  // Release FIRST: it clears `inert` on the main region and puts focus back on
  // the control that opened the dialog. Removing the holder first would drop
  // focus to <body> with nothing to restore — the old behaviour.
  modalRelease?.();
  modalRelease = null;
  abortCeremony(); // I3-1429 — a dismissed modal's ceremony is over
  document.querySelector("[data-modal-holder]")?.remove();
  // I3-1178 — an in-app dialog shares this teardown, and a promise left pending
  // here would strand its flow for the life of the page. A dialog dismissed by
  // a route change answers exactly what Cancel answers.
  const stranded = dialogSettle;
  dialogSettle = null;
  stranded?.(null);
}

/**
 * I3-1178 / I3-1177 / I3-0537 — THE FAN'S CONSENT AND THE FAN'S WORDS LEAVE THE
 * DOCUMENT. `window.prompt` and `window.confirm` are browser chrome, not page.
 * Three consequences, and they are three of this room's rows rather than three
 * complaints about taste:
 *
 *   - they take no markup, so a disclosure written as paragraphs arrives as one
 *     unformattable wall of text (I3-0537);
 *   - they take no validation, so a ceiling the copy advertises cannot be
 *     enforced where the fan is actually typing (I3-1177);
 *   - they are outside the tree `trapFocus` and `announce` operate on, so this
 *     app's focus handling and its live region stop at their edge (I3-1178).
 *
 * This is the one replacement for all three. It is deliberately the same
 * promise-shaped contract the native calls have, so a call site changes by one
 * `await` and nothing else, and it reuses the modal convention already in this
 * file — `[data-modal-holder]`, `trapFocus`, `closeRequestModal` — rather than
 * starting a second one.
 *
 * The content is built as NODES with `textContent`, never as an HTML string:
 * there is no escaping question to get wrong, now or at a later call site.
 */
let dialogSettle: ((v: string | null) => void) | null = null;

interface AppDialogSpec {
  title: string;
  /** Rendered one `<p>` per entry. This is I3-0537's first limb in one line. */
  paragraphs: readonly string[];
  confirmLabel: string;
  cancelLabel?: string;
  /** Present ⇒ the dialog collects text. Absent ⇒ it is a confirm. */
  input?: { label: string; max: number };
}

/**
 * The server measures a question AFTER collapsing whitespace and trimming
 * (`services/trust-core/src/questions.ts:179-181`), so the client counts the
 * same form. Counting the RAW value — which is what a bare `maxlength` would
 * do — refuses submissions the server would have accepted, which is the same
 * class of defect as not counting at all, pointed the other way.
 */
function canonicalLength(raw: string): number {
  return raw.replace(/\s+/g, " ").trim().length;
}

/** Resolves to the fan's text for an input dialog, `""` for a plain confirm, or
 * `null` on Cancel, Escape, or a route change. */
function appDialog(spec: AppDialogSpec): Promise<string | null> {
  closeRequestModal(); // one modal at a time; settles anything already pending
  return new Promise<string | null>((resolve) => {
    dialogSettle = resolve;
    const holder = document.createElement("div");
    holder.setAttribute("data-modal-holder", "");
    holder.setAttribute("data-app-dialog", "");
    const box = document.createElement("section");
    box.className = "modal";
    box.setAttribute("role", "dialog");
    box.setAttribute("aria-modal", "true");
    box.setAttribute("aria-labelledby", "app-dialog-h");
    const h = document.createElement("h2");
    h.id = "app-dialog-h";
    h.textContent = spec.title;
    box.appendChild(h);
    for (const para of spec.paragraphs) {
      const p = document.createElement("p");
      p.textContent = para;
      box.appendChild(p);
    }
    let field: HTMLTextAreaElement | null = null;
    let count: HTMLElement | null = null;
    if (spec.input) {
      const lab = document.createElement("label");
      lab.setAttribute("for", "app-dialog-input");
      lab.textContent = spec.input.label;
      field = document.createElement("textarea");
      field.id = "app-dialog-input";
      field.setAttribute("data-app-dialog-input", "");
      count = document.createElement("p");
      count.className = "muted";
      count.setAttribute("data-app-dialog-count", "");
      count.setAttribute("aria-live", "polite");
      box.appendChild(lab);
      box.appendChild(field);
      box.appendChild(count);
    }
    const row = document.createElement("p");
    const confirm = document.createElement("button");
    confirm.className = "ds-btn act primary";
    confirm.setAttribute("data-app-dialog-confirm", "");
    confirm.textContent = spec.confirmLabel;
    const cancel = document.createElement("button");
    cancel.className = "ds-btn act";
    cancel.setAttribute("data-app-dialog-cancel", "");
    cancel.textContent = spec.cancelLabel ?? "Cancel";
    row.appendChild(confirm);
    row.appendChild(cancel);
    box.appendChild(row);
    holder.appendChild(box);
    document.body.appendChild(holder);

    const settle = (v: string | null): void => {
      const done = dialogSettle;
      dialogSettle = null;
      modalRelease?.();
      modalRelease = null;
      holder.remove();
      done?.(v);
    };
    // The ceiling is enforced on the CONTROL, not only in the copy: over it, the
    // dialog will not submit and says by how much. A `maxlength` alone would
    // silently truncate the fan's sentence, which is a worse answer than a
    // refusal they can see and correct.
    const refresh = (): void => {
      if (!spec.input || !field || !count) return;
      const n = canonicalLength(field.value);
      const over = n - spec.input.max;
      count.textContent =
        over > 0 ? `${over} character${over === 1 ? "" : "s"} too long — ${n} of ${spec.input.max}` : `${n} of ${spec.input.max} characters`;
      confirm.disabled = n === 0 || over > 0;
    };
    field?.addEventListener("input", refresh);
    refresh();
    confirm.addEventListener("click", () => settle(field ? field.value : ""));
    cancel.addEventListener("click", () => settle(null));
    const trigger = document.activeElement as HTMLElement | null;
    modalRelease = trapFocus(holder, { onEscape: () => settle(null), restoreTo: trigger });
    (field ?? confirm).focus();
    announce(document, spec.title);
  });
}

/** I3-0422a — THE IN-PAGE CONFIRM. Every `window.confirm` in this app went through
 * the browser's own box (no markup, no focus trap, no title, one wall of text);
 * this is the same question asked in the document, with the cost sentences as
 * their own paragraphs. `true` only on the confirm button -- Cancel, Escape and a
 * route change are all `false`, so nothing proceeds on a dismissal. */
async function confirmDialog(title: string, paragraphs: readonly string[], confirmLabel: string): Promise<boolean> {
  return (await appDialog({ title, paragraphs, confirmLabel })) !== null;
}

/**

 * Nine handlers said "Saved" and then ran `setTimeout(render, 700-1400)`, which

function noteThen(text: string, retry = false): void {

}

/** Open the interstitial for a validated passport id and wire its actions.
 * The markup comes from the pure, tested view fn (renderer-owned values only);

function showRequestModal(psp: string): void {
  closeRequestModal();

  if (!html) return; // hostile/malformed id → nothing renders

    if (browserOwnsClick(e)) return; // I3-2106
    e.preventDefault();
    closeRequestModal();
    go("#/signin"); // full sign-in screen fallback (Google/Apple)
  });
  let attempt = 0; // I3-1429 — only the latest press may speak
  const run = (mode: "auth" | "register") => async (): Promise<void> => {
    const mine = ++attempt;
    const supported = passkeySupported();
    if (msg) msg.textContent = "Waiting for your passkey…";
    let refusal: string | null = null;
    // I24 (D-0996(2)): the follow-up request's own refusal, shown on the reqbar when sign-in landed but it did not.

      onRefusal: (m) => {

      }

/**
 * UI-AUDIT F-W2 — A PAYMENT CONFIRMATION ANYONE COULD FABRICATE.
 *
 * Two call sites both read the return flag as a SUBSTRING test on the raw query
 * string — `search.includes("upgraded=1")`. Every one of these passes it:
 *
 *     ?not-upgraded=1        ?ref=upgraded=1        ?x=see-upgraded=1
 *     ?utm_campaign=upgraded=1                      ?q=upgraded=1%20scam
 *
 * …and what it turns on is not decoration: `planState` renders the post-payment
 * welcome, and `billingStateCopy` answers with a sentence that states the fan
 * WAS CHARGED. So any link — a post, a DM, a redirect chain, an ad tracker
 * appending its own params — could put a confirmation of a payment that never
 * happened in front of a signed-in fan, on the real domain, over their real
 * account page. That is a phishing primitive handed out by the app itself: the
 * next message ("we couldn't take your card, re-enter it") is the whole attack,
 * and this page did the convincing.
 *
 * `URLSearchParams` and an EXACT value. The flag is one we set on our own
 * Stripe success URL, so nothing is lost by reading it precisely.
 */
function returnFlag(name: string): boolean {
  if (typeof window === "undefined") return false;
  try {
    return new URLSearchParams(window.location.search).get(name) === "1";
  } catch {
    return false; // a search string URLSearchParams cannot parse is not a receipt
  }
}

/** I3-1384 — strip one return flag from the SEARCH string, keeping every other
 * param and the hash, so the next render of the same route does not re-read it. */
function dropReturnFlag(name: string): void {
  try {
    const q = new URLSearchParams(window.location.search);
    if (!q.has(name)) return;
    q.delete(name);
    const rest = q.toString();
    window.history.replaceState(window.history.state, "", `${window.location.pathname}${rest ? `?${rest}` : ""}${window.location.hash}`);
  } catch {
    /* an unparseable search or a refused replaceState leaves the URL as it was */
  }
}

/**

    // M10: while the payment is still confirming (Stripe webhook lag), re-check
    // the entitlement a few times so the card flips to "active" on its own — the
    // "this page updates itself" promise the processing copy makes. Bounded so a
    // never-arriving webhook doesn't poll forever; the fan can also just refresh.
    if (state.name === "processing" && polls < 6 && typeof window !== "undefined") {
      polls += 1;
      window.setTimeout(() => void deps.api.entitlementStatus().then(paint), 3000);
      return; // the processing card carries no button
    }
    // FU-14: re-paint on an interval change, so the price on the BUTTON can
    // never disagree with the option the fan has selected. Same wiring as the
    // creator console's `pro_interval` radios.
    for (const el of holder.querySelectorAll<HTMLInputElement>('input[name="tier_interval"]')) {
      el.addEventListener("change", () => paint({ status, data }));
    }
    const btn = holder.querySelector<HTMLButtonElement>('[data-act="plan-action"]');
    if (!btn) return; // the 503, signed-out and processing cards carry no button by design
    btn.addEventListener("click", async () => {
      btn.disabled = true;
      const kind = btn.dataset.planKind;
      const interval = btn.dataset.planInterval === "month" ? "month" : "year";
      // A PRE-M7 subscriber is paying through the previous processor and has no
      // Stripe customer: they get that processor's portal. Opening a Stripe
      // checkout for them would be a SECOND live subscription, so `planState`
      // never offers checkout on an active/grace state.
      const r = kind === "portal" ? await deps.api.stripePortal() : await deps.api.startTierCheckout(interval);
      const url = r.data?.url;
      // I3-1251 (BF-06): an in-flight checkout on the OTHER interval was replayed. Say so and let the fan choose; never a silent redirect.
      const switched = kind === "portal" ? null : checkoutIntervalSwitched(interval, (r.data ?? null) as Parameters<typeof checkoutIntervalSwitched>[1]);
      if (switched && typeof url === "string" && /^https:\/\//.test(url)) {
        btn.disabled = false;
        const note = holder.querySelector<HTMLElement>("[data-plan-msg]");
        if (note) note.textContent = `You already have a checkout open for the ${switched === "month" ? "monthly" : "yearly"} plan. Continuing will open that one, not the ${interval === "month" ? "monthly" : "yearly"} plan you picked.`;
        btn.textContent = `Continue to the ${switched === "month" ? "monthly" : "yearly"} checkout`;
        btn.onclick = () => { window.location.href = url; };
        return;
      }
      if (typeof url === "string" && /^https:\/\//.test(url)) {
        window.location.href = url; // hand off to the processor-hosted page
        return;
      }
      if (r.status === 503) {
        // Billing is switched off on this deployment. Swap the whole card for
        // the designed state — the control disappears rather than dies.
        holder.innerHTML = planStateView(planState({ status: 503, ent: data, signedIn }));
        return;
      }
      btn.disabled = false;
      const msg = holder.querySelector<HTMLElement>("[data-plan-msg]");
      if (msg) msg.textContent = refusalText(r, "Couldn't reach the payment page just now. Nothing protective is affected — checks, verdicts and warnings are unchanged.");
    });
  };
  void deps.api.entitlementStatus().then(paint);
  void readPromo().then(paintPromo);
}

/**
 * S-CW7a — the wallet page (`#/wallet`). Drops the live credit balance into
 * [data-wallet-state] and wires the buy-credits buttons: each carries its bucket
 * + rail declaratively, POSTs to start a hosted checkout, and hands off to the
 * processor page. A rail that is switched off (503) swaps in a calm note rather
 * than a dead button; a 403 means the account was suspended (chargeback) and the
 * whole card is repainted to the banned state. Nothing protective is read here.
 */
/** I3-2126 — the re-read schedule of the credit-checkout return leg (about 30 s in all). */
const CREDIT_RETURN_POLL_MS = [1500, 3000, 5000, 8000, 12000] as const;

function wireWalletPage(): void {
  const holder = root.querySelector<HTMLElement>("[data-wallet-state]");
  if (!holder) return;
  const signedIn = session.isSignedIn();
  let lastCredits = 0; // the most recent balance read, so a mid-session ban can be repainted with it
  let readCredits: number | null = null; // I3-2126 — the last balance the SERVER actually answered with
  // The message slot lives in the STATIC grid now (F-W3), so it is present in
  // every state the buy buttons are — including signed-out and banned, where a
  // click used to produce no visible reaction at all.
  const setMsg = (t: string): void => {
    const el = root.querySelector<HTMLElement>("[data-wallet-msg]");
    if (el) el.textContent = t;
  };
  // UI-AUDIT F-W3 — the buttons follow the state. A signed-out or suspended fan
  // sees why they cannot buy instead of a live control that answers nothing.
  const setBuyable = (on: boolean, why: string): void => {
    root.querySelectorAll<HTMLButtonElement>('[data-act="buy-credits"]').forEach((b) => {
      b.disabled = !on;
      if (on) b.removeAttribute("title");
      else b.title = why;
    });
    if (!on) setMsg(why);
  };
  // I30-SPEND H5 (I21-Q50, D-1030): the fan's own spending-limit card in [data-wallet-spend] (wallet.ts spendLimitsView),
  // read once signed in; each of its four controls sends one change (api.ts setSpendLimits) and the card is re-drawn
  // from the server's answer; a refusal is written into [data-spend-msg].
  const spendHolder = root.querySelector<HTMLElement>("[data-wallet-spend]");
  const paintSpend = (s: Parameters<typeof spendLimitsView>[0]): void => {
    if (!spendHolder?.isConnected) return;
    spendHolder.innerHTML = spendLimitsView(s);
    const on = (sel: string, change: () => Parameters<typeof deps.api.setSpendLimits>[0] | null): void => {
      spendHolder.querySelector<HTMLButtonElement>(sel)?.addEventListener("click", async (e) => {
        const btn = e.currentTarget as HTMLButtonElement;
        const say = (t: string): void => {
          const m = spendHolder.querySelector<HTMLElement>("[data-spend-msg]");
          if (m) m.textContent = t;
        };
        const c = change();
        if (!c) return say("Enter a whole number of dollars.");
        btn.disabled = true;
        const r = await deps.api.setSpendLimits(c);
        btn.disabled = false;
        if (r.status === 200 && r.data) return paintSpend(r.data);
        say(refusalText(r, "Couldn't change your spending limit just now. Nothing was changed."));
      });
    };
    on('[data-act="spend-cap-save"]', () => {
      const v = spendHolder.querySelector<HTMLInputElement>("[data-spend-cap-input]")?.value.trim() ?? "";
      return /^\d{1,7}$/.test(v) ? { monthly_cap_cents: Number(v) * 100 } : null;
    });
    on('[data-act="spend-cap-clear"]', () => ({ monthly_cap_cents: null }));
    on('[data-act="spend-pause"]', () => ({ pause_days: Number(spendHolder.querySelector<HTMLSelectElement>("[data-spend-pause-days]")?.value) }));
    on('[data-act="spend-pause-end"]', () => ({ end_pause: true }));
  };
  if (spendHolder && signedIn) void deps.api.spendLimits().then((r) => paintSpend(r.status === 200 ? r.data : null));
  const paintWallet = (r: Awaited<ReturnType<typeof deps.api.warCredits>>): void => {
    if (!holder.isConnected) return;
    const st = walletState({ status: r.status, data: r.data, signedIn });
    lastCredits = st.credits;
    if (r.status === 200 && typeof r.data?.credits === "number") readCredits = r.data.credits;
    holder.innerHTML = walletStateView(st);
    setBuyable(
      canBuy(st),
      st.name === "signed-out" ? "Sign in to buy credits — a balance belongs to an account." : BANNED_BUY_HINT,
    );
  };
  // Buy buttons live in the STATIC grid (always present), so wire them once here.
  root.querySelectorAll<HTMLButtonElement>('[data-act="buy-credits"]').forEach((btn) =>
    { btn.addEventListener("click", async () => {
      const credits = Number(btn.dataset.credits);
      // QUEUE #350 (D-0434): card is the only rail; wallet.ts renders no crypto
      // button and api.ts no longer carries a crypto checkout call.
      if (!(CREDIT_BUCKETS as readonly number[]).includes(credits)) return;
      btn.disabled = true;
      const r = await deps.api.startCreditCheckout(credits);
      const url = r.data?.url;
      if (typeof url === "string" && /^https:\/\//.test(url)) {
        window.location.href = url; // hand off to the processor-hosted page
        return;
      }
      btn.disabled = false;
      if (r.status === 403) {
        // The account was suspended (chargeback) between load and click — repaint
        // the whole card to the banned state, keeping the last balance we read.
        const banned = walletState({ status: 200, data: { credits: lastCredits, banned: true }, signedIn });
        holder.innerHTML = walletStateView(banned);
        setBuyable(false, BANNED_BUY_HINT);
        return;
      }
      setMsg(refusalText(r,
        r.status === 401
          ? "Sign in to buy credits — a balance belongs to an account."
          : r.status === 503
            ? "Card payments aren't set up on this deployment yet — nothing was charged. Check back later."
            : "Couldn't reach the payment page just now — nothing was charged. Give it another try in a moment.",
      ));
    }); },
  );
  // UI-AUDIT F-W1 — THE RETURN FROM A CREDIT PURCHASE, which had no landing at
  // all. Stripe's credit success URL used to be the SUBSCRIPTION one
  // (`?upgraded=1#/me`), so a fan who bought credits was returned to the account
  // page under the flag that means a Premium subscription completed, and the
  // billing copy answered "Confirming your Premium payment…" — a spinner
  // polling for a subscription webhook that was never coming. It never resolved.
  // The credits had been granted correctly, on a page the fan was not on.
  //
  // The rail now returns to `?credits=1#/wallet`. The balance read below is what
  // confirms the purchase — no separate claim is made, because the balance IS
  // the receipt. If the webhook has not landed yet the number simply has not
  // moved, and the note says so rather than asserting a charge we cannot see.
  if (!returnFlag("credits") || !signedIn) {
    if (returnFlag("credits")) dropReturnFlag("credits");
    void deps.api.warCredits().then(paintWallet);
    return;
  }
  setMsg("Thanks — your credits are being added. Your balance below updates as soon as the payment clears, usually within a few seconds.");
  dropReturnFlag("credits"); // I3-1384 — a receipt is read once, not on every later wallet visit
  // I3-2126 — ONE read was the whole return leg, so a webhook that landed a
  // second after it left the old number on screen under a sentence promising
  // it would update. Re-read on a bounded schedule until the balance moves off
  // the first good read; a failed re-read never repaints over a good balance.
  // Leaving the page stops it (`holder.isConnected`). No pre-checkout balance
  // is stored — this client keeps nothing in browser storage but the session
  // token (m8-guards) — so a balance that never moves is reported as "not
  // moved", never as "not cleared": the webhook may have beaten the first read.
  let before: number | null = null;
  const moved = (): boolean => readCredits !== null && before !== null && readCredits > before;
  void (async () => {
    paintWallet(await deps.api.warCredits());
    if (before === null) before = readCredits;
    for (const wait of CREDIT_RETURN_POLL_MS) {
      if (moved()) break;
      await new Promise<void>((res) => setTimeout(res, wait));
      if (!holder.isConnected) return;
      const r = await deps.api.warCredits();
      if (r.status === 200 && typeof r.data?.credits === "number") paintWallet(r);
      if (before === null) before = readCredits; // the first read failed: baseline on the first good one
    }
    if (!holder.isConnected) return;
    setMsg(
      moved()
        ? "Payment cleared — your balance below is up to date."
        : "Your balance below hasn't changed while this page was open. If it doesn't include this purchase yet, reload in a minute — payments usually clear within a few seconds.",
    );
  })();
}

 * control (or the reason there isn't one), and on open mints a client seed,
 * spends 500 credits server-side, and renders the CALM reveal (all six cards at

/**
 * F3 — GIFTING. Per-item controls (offer + history) and the offers panel.
 * Every outcome message is the server's own refusal or a fixed line; a lost
 * response says "couldn't confirm", never "nothing happened" — an offer is not
 * money, but an accept moves property.
 */
function wireGifts(): void {
  const showGiftOutcome = (): void => {
    if (!giftOutcome || giftOutcome.hash !== window.location.hash) return;
    const el = root.querySelector<HTMLElement>("[data-gifts-msg]");
    if (el) el.textContent = giftOutcome.text;
  };

  // INSIDE `[data-gifts-panel]` (gifts.ts:82), and the very next statement
  // destroyed that element: `render()` on accept, `repaintPanel()` on
  // decline/cancel and on every 4xx. So it is held here and re-applied after
  // whichever paint follows, and announced as well — the live region is the one
  // copy of the sentence no repaint can wipe (a11y.ts:56). Held per-hash so a
  // navigation cannot carry one page's outcome onto another.
  const panelMsg = (t: string): void => {
    giftOutcome = t ? { text: t, hash: window.location.hash } : null;
    announce(document, t);
    showGiftOutcome();
  };
  const repaintPanel = async (): Promise<void> => {
    const panel = root.querySelector<HTMLElement>("[data-gifts-panel]");
    if (!panel) return;
    const r = await deps.api.gifts();
    panel.outerHTML = giftsPanelView(r.data ?? null);
    wireGiftsPanel();
    showGiftOutcome();
  };
  const wireGiftsPanel = (): void => {
    const act = (name: "gift-accept" | "gift-decline" | "gift-cancel" | "gift-ignore" | "gift-block", call: (id: string) => Promise<{ ok: boolean; status: number; data: unknown; message?: string }>, done: string, confirmFirst?: string): void => {

        { btn.addEventListener("click", async () => {
          const id = btn.dataset.offer ?? "";
          if (!/^gft_[0-9a-f]{32}$/.test(id)) return;

          const r = await call(id);
          if (r.ok) {
            panelMsg(done);
            // An accepted gift changes the SHELF too — re-render the page so the
            // card appears where it now lives.
            if (name === "gift-accept") return void render();
            return void repaintPanel();
          }
          btn.closest(".giftacts")?.querySelectorAll<HTMLButtonElement>("button").forEach((b) => { b.disabled = false; });
          if (r.status === 0 || r.status >= 500) return panelMsg("We couldn't confirm that — reload to see where things stand before trying again.");
          const err = (r.data as { error?: unknown } | null)?.error;
          panelMsg(refusalText(r, typeof err === "string" ? err : "That didn't go through."));
          void repaintPanel();
        }); },
      );
    };
    act("gift-accept", (id) => deps.api.acceptGift(id), "It's yours — it's on your shelf now.");
    act("gift-decline", (id) => deps.api.declineGift(id), "Declined. Nothing moved.");
    act("gift-cancel", (id) => deps.api.cancelGift(id), "Withdrawn. The item never left your shelf.");
    // QUEUE #172 (B27-MONEY H4): the recipient's quiet exits. Block has no end
    // and no undo on this client, so it asks first; Ignore lapses in 30 days.
    // D-0812(1)/D-0813(7): under Ignore a re-offer is ACCEPTED and never delivered,
    // so the toast must not promise the sender is refused.
    // D-0811(2): neither exit tells the sender; to them the offer stays pending and
    // ends at its own expiry. A blocked sender's later offers land and never reach you.
    act("gift-ignore", (id) => deps.api.ignoreGift(id), "Ignored. They aren't told: to them the offer ends at its own expiry, and anything they offer you in the next 30 days is quietly dropped.");
    act("gift-block", (id) => deps.api.blockGiftSender(id), "Blocked. They aren't told: to them the offer ends at its own expiry, and nothing they offer you will reach you again.", "Block this sender? They won't be told, and nothing they offer you will reach you again. There's no way to undo this here.");
  };
  wireGiftsPanel();

  root.querySelectorAll<HTMLElement>(".giftctl").forEach((ctl) => {
    const kind = ctl.dataset.giftKind === "acquisition" ? "acquisition" : "card";
    const itemId = ctl.dataset.giftItem ?? "";

    const hist = ctl.querySelector<HTMLElement>("[data-gift-history]");

    };
    ctl.querySelector<HTMLButtonElement>('[data-act="gift-open"]')?.addEventListener("click", () => {
      if (form) form.hidden = false;
      form?.querySelector<HTMLInputElement>("input")?.focus();
    });
    ctl.querySelector<HTMLButtonElement>('[data-act="gift-close"]')?.addEventListener("click", () => {
      if (form) form.hidden = true;
      say("");
    });
    form?.addEventListener("submit", async (ev) => {
      ev.preventDefault();
      const input = form.querySelector<HTMLInputElement>("input[name=to_handle]");
      const handle = (input?.value ?? "").trim().toLowerCase().replace(/^@/, "");
      if (!handleOk(handle)) return say("A handle is 3–20 letters, digits or underscores.");
      form.querySelectorAll<HTMLButtonElement>("button").forEach((b) => { b.disabled = true; });
      const r = await deps.api.offerGift(kind, itemId, handle);
      form.querySelectorAll<HTMLButtonElement>("button").forEach((b) => { b.disabled = false; });
      if (r.ok && r.data?.offer) {
        form.hidden = true;
        say(`Offered to @${handle}. It stays on your shelf until they accept — you can withdraw it below.`);
        return void repaintPanel();
      }
      if (r.status === 0 || r.status >= 500) return say("We couldn't confirm that — check your gifts below before offering again.");
      if (r.status === 429) return say(refusalText(r, "Too many offers for now — try again in a bit."));
      const err = (r.data as { error?: unknown } | null)?.error;
      say(refusalText(r, typeof err === "string" ? err : "That didn't go through."));
    });
    ctl.querySelector<HTMLButtonElement>('[data-act="gift-history"]')?.addEventListener("click", async () => {
      if (!hist) return;
      if (!hist.hidden) {
        hist.hidden = true;
        return;
      }
      const r = await deps.api.collectibleHistory(kind, itemId);
      hist.innerHTML = historyView(r.data?.history ?? null);
      hist.hidden = false;
    });
  });
  showGiftOutcome(); // `wire()` runs this after every render(), including the accept repaint
}

  });

/**

/**

 * the server names in `subscriptions_unknown` the subscription reads it could
 * not make before erasing (account_data.ts `eraseAccountWithSubscriptions`);

function deletedLanding(data: unknown): string {

      : "")
  );
}

/** I3-1316 — THE BROWSER CHECKS THE SEED ITSELF. The reveal used to compare the
 * hash the server returned with the hash the server committed to -- two strings
 * the server supplied -- and then told the fan their browser had checked the seed.
 * The commitment is `sha256Hex(server_seed)` (trust-core packs.ts); it is
 * recomputed here from the revealed seed. A missing or non-matching seed fails
 * closed, which renders the broken-commitment reveal. */
async function seedAnswersCommitment(seed: unknown, returnedHash: string, heldHash: string): Promise<"match" | "mismatch" | "unchecked"> {
  if (returnedHash !== heldHash) return "mismatch";
  if (typeof seed !== "string" || seed === "") return "mismatch";
  try {
    return (await sha256Hex(seed)) === heldHash ? "match" : "mismatch";
  } catch {
    // This browser cannot hash (no WebCrypto digest): that is not evidence against the draw, so it is neither
    // called broken nor called "checked for you" -- the reveal renders without the committed-hash line.
    return "unchecked";
  }
}

function wireStorePage(): void {
  const stateHolder = root.querySelector<HTMLElement>("[data-store-state]");
  const resultHolder = root.querySelector<HTMLElement>("[data-pack-result]");
  if (!stateHolder || !resultHolder) return;
  const signedIn = session.isSignedIn();
  // Null until a balance actually reads (F-W5). A repaint that has never had a
  // number must not invent one — see storeStateView's unread branch.
  let lastCredits: number | null = null;
  let lastCashBacked: number | null | undefined; // I3-0059 — undefined = the server did not send it
  let lastBanned = false;
  /** I3-0438 — set when an open returns an outcome we cannot classify as
   * "refused before the debit". While set, no `open-pack` button is pressable:
   * a second press would spend a NEW pack price against the fresh commitment
   * while the first may already have landed. Only a reload clears it, which is
   * what the sentence on screen tells the fan to do. */
  let spendHeld = false;
  /** I5-645/RGC-115 (QUEUE #645 row 10) -- true from the moment `open()` is
   * called until its `openPack()` response resolves (or an earlier return
   * skips the request entirely). `paintState`'s repaint mints a fresh,
   * unbound button on every call (including the fire-and-forget trailing
   * `takeCommitment()` a completed open leaves running) and only this flag
   * -- not `spendHeld`/`cannotSpend` -- tells that repaint a purchase is
   * still unresolved. Cleared as soon as the response lands (before any of
   * open()'s own repaints), with `open()`'s `finally` as the backstop for
   * every path that returns before a request is even sent. */
  let openInFlight = false;
  /** I3-0438 second limb — `[data-store-msg]` lives inside `[data-store-state]`
   * (store.ts:181), which `paintState` replaces, and `open()` fires
   * `takeCommitment()` → `paintState()` right after writing the sentence. Held
   * and re-applied so the fan can actually read it. */
  let storeMsg = "";
  // `commitment.odds` is untyped JSON off the wire; `parsePackOdds` is its only
  // validator and nothing here reads the field directly (S-022 ODDS, docket #36 (d)).
  // I3-1432 (D-0829, I4-13-R7 P4): a signed-out visitor never gets a commitment,
  // so it reads the PUBLIC odds instead; a signed-in fan's sale gate stays on the
  // commitment's own disclosure (ruling (d)) and never on the public read.
  let publicOdds: PackOddsDisclosureDto | null = null;
  const currentOdds = (): PackOddsDisclosureDto | null => (commitment ? parsePackOdds(commitment.odds) : !signedIn ? publicOdds : null);
  const paintState = (credits: number | null, banned: boolean, cashBacked: number | null | undefined = lastCashBacked): void => {
    lastCredits = credits;
    lastBanned = banned;
    lastCashBacked = cashBacked;
    stateHolder.innerHTML = storeStateView({ signedIn, credits, banned, cashBacked, odds: currentOdds() });
    bindOpen();
    if (storeMsg) {
      const m = stateHolder.querySelector<HTMLElement>("[data-store-msg]");
      if (m) m.textContent = storeMsg;
    }
    // I3-1387 — the reveal's "Open another" lives outside `stateHolder` and was
    // never repainted: it stayed live on a suspended account and on a balance
    // that can no longer buy a pack. This only ever DISABLES.
    const cannotSpend = banned || credits === null || credits < PACK_PRICE_CREDITS || (typeof cashBacked === "number" && cashBacked < PACK_PRICE_CREDITS);
    if (spendHeld || cannotSpend || openInFlight) root.querySelectorAll<HTMLButtonElement>('[data-act="open-pack"]').forEach((b) => { b.disabled = true; });
  };
  // I3-0059 — a pack is debited from the cash-backed class (origin store_spend, credits.ts CASH_ORIGINS), so the sub-balance drops by the price.
  const afterSpendCashBacked = (balance: number | null): number | undefined => (typeof lastCashBacked === "number" ? Math.max(0, Math.min(balance ?? 0, lastCashBacked - PACK_PRICE_CREDITS)) : undefined);
  const sayStore = (t: string): void => {
    storeMsg = t;
    announce(document, t);
    const m = stateHolder.querySelector<HTMLElement>("[data-store-msg]");
    if (m) m.textContent = t;
  };
  /**
   * UI-AUDIT F-P4 — THE COMMITMENT, HELD BEFORE THE FAN'S SEED EXISTS.
   *
   * Taken when the store page loads and re-taken after each open, so pressing
   * Buy never waits on it. Order is the whole point and is why this is a
   * separate value rather than something fetched inside `open()`: the server's
   * hash must be in hand BEFORE `mintClientSeed()` runs, or the fan's seed
   * contributes nothing and we are back to a server that could pick its seed
   * after seeing theirs.
   */
  let commitment: PackCommitDto | null = null;
  /** I24 (D-0996(2)): the server's sentence for the last refused commitPack ("" when it sent none). */
  let commitRefusal = "";
  const takeCommitment = async (): Promise<void> => {
    const c = await deps.api.commitPack();
    commitRefusal = c.ok ? "" : refusalText(c, "");
    commitment = c.ok && c.data && typeof c.data.pack_id === "string" ? c.data : null;
    // The buy button's existence depends on this landing: repaint with the
    // credits/banned state already known, so the odds (or the "not available"
    // state) appear as soon as the commitment resolves rather than on the next
    // unrelated repaint.
    paintState(lastCredits, lastBanned);
  };

  const open = async (btn: HTMLButtonElement): Promise<void> => {
    // I5-645/RGC-115 -- true for the whole open() call, across every await,
    // so a rebind mid-flight (paintState's innerHTML swap, including the
    // fire-and-forget trailing takeCommitment() a PRIOR open leaves running)
    // cannot offer a live Buy button while this purchase is unresolved.
    openInFlight = true;
    try {
    btn.disabled = true;
    // QUEUE #115 — the held sentence answers ONE press. Without this reset a
    // refusal from an earlier press is re-applied by `paintState` over the
    // result of a later press that succeeded.
    storeMsg = "";
    // A missing commitment is a retry, not an open: opening without one is the
    // shape this flow replaced, and the server refuses it anyway.
    if (!commitment) await takeCommitment();
    if (!commitment) {
      btn.disabled = false; // no pack was ever requested, so nothing was spent
      sayStore(commitRefusal || "Couldn't set up a verifiable draw just now — nothing was spent. Try again in a moment.");
      return;
    }
    const held = commitment;
    // The seed is generated HERE, after the server's hash is already held.
    let clientSeed: string;
    try {
      clientSeed = mintClientSeed();
    } catch {
      // I3-0917: no entropy, no open. openPack was never sent, so nothing was spent and `commitment` stays held.
      btn.disabled = false;
      sayStore("This browser couldn't generate the random seed a checkable draw needs — nothing was spent. Try again, or use another browser.");
      return;
    }
    const r = await deps.api.openPack(held.pack_id, clientSeed);
    // The request settled (whatever the outcome) -- clear the guard here,
    // before any of the repaints below, so a resolved purchase's own
    // success/refusal paint is the one place the button legitimately
    // re-arms. The `finally` below is the backstop for every path that
    // returns before reaching this line (no commitment, no seed entropy).
    openInFlight = false;
    // Spent or refused, this commitment is not reused: a spent one is single-use
    // server-side, and a refused one may have been expired or superseded.
    commitment = null;
    void takeCommitment();
    let seedCheck: "match" | "mismatch" | "unchecked" = "unchecked";
    if (r.data && r.data.pack_id) {
      // The promise the fan was holding, checked against what came back. A
      // mismatch means the revealed seed does not answer the hash they were
      // shown, which is the one thing this whole flow exists to make checkable.
      seedCheck = await seedAnswersCommitment(r.data.server_seed, r.data.server_seed_hash, held.server_seed_hash);
      if (seedCheck === "mismatch") {
        resultHolder.innerHTML = packRevealView({ ...r.data, commitment_broken: true });
        // I3-0532 (I4-13-R7 P1): the Copy pack id control store.ts renders on a broken reveal.
        resultHolder.querySelector<HTMLButtonElement>('[data-act="copy-pack-id"]')?.addEventListener("click", async (e) => {
          const b = e.currentTarget as HTMLButtonElement;
          const id = b.dataset.packId ?? "";
          if (!/^[A-Za-z0-9_-]{1,64}$/.test(id)) return;
          try {
            await navigator.clipboard.writeText(id);
            b.textContent = "Copied ✓";
          } catch {
            b.textContent = "Copy blocked — select the pack id above";
          }
        });
        appendFirstPackNotice(resultHolder);
        paintState(typeof r.data.balance === "number" ? r.data.balance : lastCredits, false, afterSpendCashBacked(typeof r.data.balance === "number" ? r.data.balance : lastCredits));
        bindOpen();
        return;
      }
    }
    if (r.data && r.data.pack_id) {
      resultHolder.innerHTML = packRevealView({ ...r.data, ...(seedCheck === "match" ? { committed_hash: held.server_seed_hash } : {}) }); // I3-1316: the "checked" line only when the hash ran
      appendFirstPackNotice(resultHolder);
      paintState(typeof r.data.balance === "number" ? r.data.balance : lastCredits, false, afterSpendCashBacked(typeof r.data.balance === "number" ? r.data.balance : lastCredits)); // balance dropped by the spend
      bindOpen();
      return;
    }
    // I3-0438 — the re-enable is now per OUTCOME CLASS, not unconditional. 403,
    // 409, 503 and 429 are all refused BEFORE the debit (the shelf and balance
    // checks ride the same guarded INSERT; 503 is an absent odds table; 429 is
    // war-routes.ts's limiter, which runs before openPack), so those stay
    // retryable. Everything else — a 500 raised after `db.batch` committed, a
    // proxy timeout, a dropped connection — may have spent the credits, and the
    // sentence below says so; the button must not invite a second spend against
    // the fresh commitment `takeCommitment()` has already taken.
    if (r.status === 403) {
      // I3-1387 — suspended mid-session: the pressed control stays off.
      return paintState(lastCredits, true);
    }
    const refusedBeforeDebit = r.status === 409 || r.status === 503 || r.status === 429;
    if (refusedBeforeDebit) btn.disabled = false;
    else spendHeld = true;
    {
      // UI-AUDIT F-P3 — "NOTHING WAS SPENT", ON THE ONE BRANCH THAT CANNOT KNOW.
      //
      // 409 and 503 are refusals the server makes BEFORE the debit: the shelf
      // check and the balance check ride the same guarded INSERT, and 503 is the
      // odds table being absent. Those two may promise nothing was spent, and
      // they still do.
      //
      // The fallback caught everything else — a 500 raised after `db.batch`
      // committed, a proxy timeout on a request the worker completed, a dropped
      // connection after the bytes went out — and told the fan their credits
      // were untouched. In every one of those the pack may well have been
      // opened and paid for. Telling somebody their money is safe when you do
      // not know is the one direction this error must not fail in; the fan's
      // next move on reading it is to press the button again.
      // D-26/AS-091 — packs.ts throws FIVE distinct 409 reasons (no commitment,
      // expired commitment, odds rolled over, and two insufficient-credits/
      // shelf-full wordings); this used to collapse all of them into "short on
      // credits… Top up", which is actively wrong advice for the first three —
      // a fresh commitment is already in flight (`takeCommitment()` above) and
      // topping up fixes nothing. Distinguish on the server's own reason text
      // rather than re-deriving it. 429 (war-routes.ts's rate limiter) is
      // refused BEFORE `openPack` runs, same as 409/503: nothing was spent, so
      // it must not fall into the "can't tell" fallback below.
      const errField = (r.data as { error?: unknown } | null)?.error;
      const reason = typeof errField === "string" ? errField : "";
      sayStore(refusalText(r,
        r.status === 409
          ? /commitment|odds/i.test(reason)
            ? "That draw expired before you opened it — nothing was spent. We've started a new one; press Open again."
            : PACK_GUARD_REFUSAL
          : r.status === 503
            ? "No pack is available this week yet — the odds haven't been published. Nothing was spent."
            : r.status === 429
              ? "You're opening packs faster than we can keep up — nothing was spent. Wait a moment and try again."
              : "Something went wrong opening that pack, and we can't tell from here whether it went through. Reload before trying again — if your balance dropped, the cards are on your shelf.",
      ));
    }
    } finally {
      openInFlight = false;
    }
  };
  function bindOpen(): void {
    root.querySelectorAll<HTMLButtonElement>('[data-act="open-pack"]').forEach((b) => {
      if (b.dataset.bound === "1") return; // wire each button once
      b.dataset.bound = "1";
      // I5-645/RGC-115 -- a rebind (paintState's innerHTML swap) mints a
      // fresh, unbound button and re-arms it here on every repaint; guard
      // the re-arm itself so a click that reaches this listener while a
      // purchase is in flight is a no-op, not just a disabled attribute a
      // caller could route around.
      b.addEventListener("click", () => {
        if (openInFlight) return;
        // I3-0058 (D-0920 option A) — the reveal's "Open another" sits where the
        // fan's thumb already is, so ONE tap re-bought a pack. It is marked
        // `data-confirm-tap` (store.ts revealView) and spends only on a SECOND
        // tap inside OPEN_AGAIN_WINDOW_MS; the first tap arms it and says so.
        if (b.dataset.confirmTap === "1") {
          const armedAt = Number(b.dataset.armedAt ?? "");
          if (!(armedAt > 0 && Date.now() - armedAt <= OPEN_AGAIN_WINDOW_MS)) {
            b.dataset.armedAt = String(Date.now());
            const label = b.textContent ?? "";
            b.textContent = `Tap again to open — ${PACK_PRICE_TEXT}`;
            setTimeout(() => {
              if (!b.dataset.armedAt) return;
              delete b.dataset.armedAt;
              b.textContent = label;
            }, OPEN_AGAIN_WINDOW_MS);
            return;
          }
          delete b.dataset.armedAt;
        }
        void open(b);
      });
    });
  }
  // PS-238 (docket #36 (d), S-022 ODDS): the fan's own pack spend history. Only
  // for a signed-in fan — signed out there is nothing to read, and an empty
  // panel would read as "you have spent nothing", which is a different claim.
  const spendPanel = root.querySelector<HTMLElement>("[data-pack-spend]");
  const spendBody = root.querySelector<HTMLElement>("[data-pack-spend-body]");
  if (signedIn && spendPanel && spendBody) {
    void deps.api.packSpendHistory().then((r) => {
      spendBody.innerHTML = packSpendHistoryView(r.data?.history ?? null);
      spendPanel.hidden = false;
    });
  }
  void deps.api.warCredits().then((r) => paintState(typeof r.data?.credits === "number" ? r.data.credits : null, r.data?.banned === true, r.data?.cash_backed));
  // Commit on load, so the server is bound before the fan has a seed to send.
  void takeCommitment();
  if (!signedIn) {
    void deps.api.packOddsPublic().then((r) => {
      publicOdds = parsePackOdds(r.data?.odds);
      paintState(lastCredits, lastBanned);
    });
  }
}

function val(sel: string): string {
  return (root.querySelector<HTMLInputElement | HTMLTextAreaElement>(sel)?.value ?? "").trim();
}
function picked(name: string): string {
  return root.querySelector<HTMLInputElement>(`input[name="${name}"]:checked`)?.value ?? "";
}
function say(sel: string, text: string): void {
  const el = root.querySelector<HTMLElement>(sel);
  if (el) el.textContent = text;
}
/** Render a pure-module HTML string into a holder. The string is built entirely
 * by the escaping view helpers from module-owned copy — no server text and no
 * user text reaches it unescaped. */
function put(sel: string, html: string): void {
  const el = root.querySelector<HTMLElement>(sel);
  if (el) el.innerHTML = html;
  // CL:M-06: nine flows say "Sending…" into an `aria-live` span, BLANK it, then
  // land the answer here — in a bare div nobody is listening to. The
  // announcement is made at the ONE place all nine converge, so a new call site
  // of an existing flow cannot forget it; `announce` ignores a clear (`put(sel,
  // "")`), which is not an outcome. `el.textContent` is the card's own words —
  // no second sentence is invented for AT that the screen does not show.
  if (el && OUTCOME_HOLDERS.includes(sel)) announce(document, el.textContent ?? "");
}

/**
 * Replace the whole page with a view rendered OUTSIDE the router — a report that
 * just filed, a removal code, a removal status — WITH the shell around it.
 *
 * These three sites used to assign `root.innerHTML` directly, which threw the
 * shell away: no top bar, no search box, no footer safety links, no way back
 * into the product except the browser's Back button. On the report confirmation
 * that combined with a suppressed action row to produce a page with no link of
 * any kind (see `reportDoneView`) — a dead end at the end of the single most
 * important journey the product has, reached by the fan who has just been
 * impersonated. Every out-of-band paint goes through here now, so a view cannot
 * lose the navigation by being rendered from the wrong place again.
 *
 * `renderRoute` (app.ts) does the same wrap for every routed view, and
 * `renderCheck` for a check result; this is the third and last door.
 */
function paint(inner: string, route: string): void {
  claimEpoch(); // I3-1212 — an out-of-band paint outranks any render still in flight
  root.innerHTML = shell(inner, { signedIn: session.isSignedIn(), route });
  wire();
}

/**
 * Resolve the case ref for a report we JUST filed, by reading the fan's own
 * list and matching the canonical subject. The matching rule (and why it is an
 * exact key match rather than "the newest row") lives in report.ts
 * `findCaseRef`, which is pure and tested; this only does the fetch.
 *
 * Null on any doubt. A null ref renders the honest "screenshots aren't
 * available for this case" panel — never a control that would 404.
 */
async function resolveCaseRef(url: string): Promise<string | null> {
  return findCaseRef((await deps.api.ownReports()).data?.reports, url);
}

/** The 3-step report intake. */
function wireReport(): void {
  const form = root.querySelector<HTMLElement>("[data-report]");
  if (!form) return;
  const show = (n: number): void => {
    root.querySelectorAll<HTMLElement>("[data-panel-step]").forEach((p) => { p.classList.toggle("on", p.dataset.panelStep === String(n)); });
    root.querySelectorAll<HTMLElement>(".st").forEach((p) => { p.classList.toggle("on", Number(p.dataset.step) <= n); });
    if (n === 3) {
      const cat = CATEGORY_CHOICES.find((c) => c.key === picked("category"));
      // textContent, never innerHTML — this line echoes what the fan typed.
      say("[data-report-review]", `Reporting ${val("#rp-url")} — ${cat ? cat.label : "not chosen yet"}.`);
    }
  };
  root.querySelectorAll<HTMLElement>("[data-report-next]").forEach((b) =>
    { b.addEventListener("click", () => {
      const to = Number(b.dataset.reportNext);
      // The gate IS the pure, tested function — not a re-typed copy of it, which
      // is how a client-side rule drifts away from the one under test.
      const gate = stepReady({ step: (to - 1) as ReportStep, url: val("#rp-url"), category: picked("category") as ReportCategory, note: val("#rp-note") });
      if (!gate.ok) return say("[data-report-msg]", gate.msg);
      say("[data-report-msg]", "");
      show(to);
    }); },
  );
  root.querySelectorAll<HTMLElement>("[data-report-back]").forEach((b) => { b.addEventListener("click", () => show(Number(b.dataset.reportBack))); });

  root.querySelector<HTMLButtonElement>('[data-act="report-send"]')?.addEventListener("click", async (e) => {
    const btn = e.currentTarget as HTMLButtonElement;
    const url = val("#rp-url");
    const category = picked("category") as ReportCategory;
    const note = val("#rp-note");
    btn.disabled = true;
    say("[data-report-msg]", "Sending…");
    const r = await deps.api.fileReport({ url, category, ...(note ? { note } : {}) });
    const out = refusalOutcome(r, reportSubmitOutcome({ ok: r.ok, status: r.status, result: r.data?.result ?? null }));
    // The case ref is resolved ONLY for a report that actually landed. A
    // failure never reaches the evidence step, so nothing can offer to attach
    // a screenshot to a report that does not exist.
    const ref = out.filed ? await resolveCaseRef(url) : null;
    paint(reportDoneView(out, ref), "#/report");
  });
}

/** The one fan upload. The failure ladder lives in report.ts; this only reads
 * the file and reports what the server said. */
function wireEvidence(): void {
  const box = root.querySelector<HTMLElement>("[data-evidence]");
  const input = root.querySelector<HTMLInputElement>("[data-evfile]");
  if (!box || !input) return;
  const caseRef = box.dataset.evidence ?? "";
  const list = root.querySelector<HTMLElement>("[data-evlist]");
  let attached = 0;
  input.addEventListener("change", async () => {
    const file = input.files?.[0];
    input.value = ""; // let the same file be retried after a failure
    if (!file) return;
    const pre = evidencePrecheck({ size: file.size }, attached);
    const done = (o: EvidenceOutcome): void => {
      say("[data-ev-msg]", o.attached ? "" : `${o.headline} — ${o.body}`);
      if (!list) return;
      // The row's text (and its class) come from report.ts, so the tick can
      // never be rendered for something the server did not accept.
      const row = evidenceRowText(file.name, o);
      const li = document.createElement("li");
      li.className = row.cls;
      // textContent throughout: a file name is attacker-supplied text.
      li.textContent = row.text;
      list.appendChild(li);
    };
    if (pre) return done(pre);
    say("[data-ev-msg]", "Uploading…");
    let bytes: ArrayBuffer;
    try {
      bytes = await file.arrayBuffer();
    } catch {
      return done(evidenceOutcome({ ok: false, status: 400 }));
    }
    const r = await deps.api.uploadEvidence(caseRef, bytes);
    const o = evidenceOutcome(r);
    if (o.attached) attached += 1;
    if (evidenceSlotsSpent(attached)) input.disabled = true;
    done(o);
  });
}

/** Remove / correct / contest — the legal safety valve. */
function wireRemoval(): void {
  if (!root.querySelector("[data-removal]")) return;
  const intent = (): RemovalIntent => (picked("intent") || "remove") as RemovalIntent;

  root.querySelector<HTMLButtonElement>('[data-act="removal-start"]')?.addEventListener("click", async (e) => {
    const btn = e.currentTarget as HTMLButtonElement;

    const chosen = intent();

    const out = refusalOutcome(r, removalStartOutcome(r));
    say("[data-removal-msg]", "");

    paint(removalCodeView(r.data, chosen), "#/remove");
  });

  root.querySelector<HTMLElement>('[data-act="removal-manual-open"]')?.addEventListener("click", () => {
    const box = root.querySelector<HTMLElement>("[data-manual]");
    if (box) box.hidden = false;
  });

  root.querySelector<HTMLButtonElement>('[data-act="removal-manual-send"]')?.addEventListener("click", async (e) => {
    const btn = e.currentTarget as HTMLButtonElement;
    btn.disabled = true;
    say("[data-removal-msg]", "Sending…");
    const detail = val("#rm-detail");
    const email = val("#rm-email");
    const r = await deps.api.removalManual({
      intent: intent(),
      ...removalSubject(val("#rm-url")),
      ...(detail ? { detail } : {}),
      ...(email ? { contact_email: email } : {}),
    });
    const out = refusalOutcome(r, manualOutcome(r));
    say("[data-removal-msg]", "");
    // FU-05 (S-068) — THIS WAS `!out.queued`, AND BOTH LIMBS OF ONE NEGATION
    // WERE BACKWARDS.
    //
    // `manualOutcome` sets queued:false for a definite 400 (the mistyped
    // profile link), a 429 cap and a general failure (`removal.ts:281-287`), so
    // the old polarity DISABLED the control after exactly the outcomes that
    // need a retry — on the legal safety valve, the door someone being
    // impersonated uses precisely because they cannot get into the account.
    // Nothing re-enabled it: there is no input/change handler here, and
    // re-opening the panel only unhides it. The failure card says "it is worth
    // another go" directly above a button that cannot.
    //
    // And it RE-ENABLED after a real receipt, where a second press replaces the
    // sole `[data-removal-out]` node — a later refusal renders no
    // `referenceBlock`, destroying the reference `removal.ts:449-459` calls the
    // only way to check on the request, with no account behind it to recover it.
    //
    // So: a queued receipt closes the control (the request is in; the reference
    // on screen is the artifact to protect), and every refusal leaves it live so
    // a corrected request submits from this same view with its draft intact.
    btn.disabled = out.queued;
    put("[data-removal-out]", outcomeCard(out, out.queued && r.data?.request_id ? referenceBlock(r.data.request_id) : ""));
  });
}

/** The bio-code verify step. */
function wireRemovalVerify(): void {
  const btn = root.querySelector<HTMLButtonElement>('[data-act="removal-verify"]');
  if (!btn) return;
  btn.addEventListener("click", async () => {

 * 0043 — the creator's inbox proof, inside the recovery form.
 *
 * Two legs, both unauthenticated. The first mails a code and answers the same
 * way for every address, so the button must NOT report whether the address was
 * found — the copy says so and this handler must not contradict it. The second
 * redeems the code and files the SAME form the signed-in path files, with the
 * creator's identity attached instead of a session.
 */
/**
 * I3-0060 — THE DOOR FOR A CREATOR WITH NO SESSION MUST NOT ANSWER IN TOKENS.
 *
 * `/creator/recovery/file` answers with an `error` field that is a MACHINE
 * string on every reachable failure: `email_unavailable` (503,
 * services/trust-core/src/index.ts:1177), `rate limit exceeded` (429, :1180),
 * `internal` (500, :1201), and on 400 an enumeration of the server's own kind
 * vocabulary (`lost_passkey, hijack, agency_dispute or competing_claim`,
 * creator-recovery.ts:395). That field was rendered verbatim as the outcome
 * card's body, so the person with the least slack in the product read
 * "email_unavailable".
 *
 * Keyed on the STATUS, which is ours to read and cannot drift into a new
 * vocabulary behind our back. A server sentence is passed through ONLY when it
 * looks like one; anything token-shaped (no space, or carrying an
 * `identifier_like_this`) falls back to the status sentence rather than being
 * shown. Every branch says what happened to the FILING, because that is the
 * one thing the creator needs and none of these strings said.
 */
/** I3-0671 (I34-W2-CLIENTS): a server `error` is shown only when it is a sentence, never a snake_case code — the
 * predicate creatorRecoveryFailureText below applies to its 400. */
/** W2P-N1 (I34-FX2-CLIENTS; DRAFT wording): the recon card's line when the commit got no answer or a 5xx. */
export const RECON_OUTCOME_UNKNOWN = "We don't know yet whether your lead was locked in — the answer didn't reach us. Refresh to see where the match stands before you try again.";

function serverSentence(error: unknown): string | null {
  const raw = typeof error === "string" ? error.trim() : "";
  return raw.length > 0 && /\s/.test(raw) && !/[a-z0-9]_[a-z0-9]/.test(raw) && raw !== "not found" && raw !== "rate limit exceeded" ? raw : null;
}

function creatorRecoveryFailureText(status: number, error: unknown): string {
  const raw = typeof error === "string" ? error.trim() : "";
  const sentence = raw.length > 0 && /\s/.test(raw) && !/[a-z0-9]_[a-z0-9]/.test(raw) && raw !== "not found" && raw !== "rate limit exceeded";
  if (status === 503) return "We can't send or check recovery codes right now — our mail system is down. Nothing was filed, so nothing was lost; try again later.";
  if (status === 429) return "That's too many recovery attempts from this connection. Nothing was filed. Wait an hour and try again — and if you are being locked out by someone else, file from a different network.";
  if (status === 400) return sentence ? raw : "Something in the form wasn't accepted — check the address and the 6-digit code from that email, then file again. Nothing was filed.";
  if (status === 404) return "That code doesn't match an address we hold. Nothing was filed. Ask for a new code and use the newest email.";
  return "Something went wrong on our side and nothing was filed. Nothing on your passport changed. Try again in a few minutes.";
}

/**
 * M2 — `#/security` (Sign-in & recovery). Passkeys first, backup email second.
 *
 * Everything a fan is told here comes from a pure outcome function in
 * backup-email.ts, so a 503 can never render as "no email on file" and a
 * refusal to strand the account can never render as a generic failure. The
 * address is read from the field at send time and never stored, never logged,
 * and never placed in a hash.
 */
function wireSecurity(): void {
  const page = root.querySelector<HTMLElement>("[data-security]");
  if (!page) return;
  const show = (o: Outcome): void => {
    say("[data-email-msg]", "");
    put("[data-email-out]", emailOutcomeCard(o));
  };
  /**
   * FU-01 (S-068) — THE CANCEL ACT, ONCE, FOR BOTH CONTROLS THAT CAN REACH IT.
   *
   * `DELETE /account/email` is TWO ACTS ON ONE CALL: `removeFanEmail`
   * (`services/trust-core/src/fan-auth.ts:1188-1199`) cancels a pending
   * replacement and KEEPS the confirmed address, and only falls through to the
   * destructive delete when no change is in flight. The route answers
   * `{ok:true}` for either (`fan-account-routes.ts:621-626`), so the response
   * cannot tell the client which act it got — the account has to be READ BACK.
   * This is why the mid-change button never claimed a removal, and it is now
   * also the path the replacement-era Remove button takes.
   */
  const cancelChangeAct = async (btn: HTMLButtonElement): Promise<void> => {
    if (!(await confirmDialog("Cancel the change you started?", ["The address you already confirmed stays exactly as it is — only the new one is dropped."], "Cancel the change"))) return;
    btn.disabled = true;
    say("[data-email-msg]", "Cancelling…");
    const answer = await deps.api.removeBackupEmail();
    const after = authPanelState(await deps.api.authState(), true);
    // W-09: THREE VALUES, NOT TWO. `after.name === "verified"` collapsed
    // `unreadable` onto `false`, and `false` is the branch that STATES the
    // address is gone. `null` says "the read-back did not answer" and reaches
    // the row that claims nothing.
    const out = refusalOutcome(answer, cancelPendingOutcome(answer, after.name === "unreadable" ? null : after.name === "verified"));
    // I49-A050: a stale sign-in is refused (403 reauth_required); sign this browser out before the card, as the add does.
    if (out.kind === "reauth") await deps.api.signOut();
    show(out);
    renderKeepingFrom("[data-email-out]");
  };
  root.querySelector<HTMLButtonElement>('[data-act="add-passkey"]')?.addEventListener("click", async (e) => {
    const btn = e.currentTarget as HTMLButtonElement;
    btn.disabled = true;
    say("[data-passkey-msg]", "Waiting for your passkey…");
    let refusal: string | null = null;
    const result = await addPasskeyFlow({ api: deps.api, createCredential: passkeySupported() ? createCredentialAdapter : null, onRefusal: (m) => { refusal = m; }, onReauth: async () => { await deps.api.signOut(); } });
    say("[data-passkey-msg]", "");
    const added = addPasskeyOutcome(result);
    put("[data-passkey-out]", emailOutcomeCard(result === "added" ? added : { ...added, body: refusal ?? added.body })); // I24 (D-0996(2))
    // Re-render on success so the passkey COUNT (and therefore whether the
    // email may be removed) reflects what the server actually holds — the
    // client never guesses that number.
    //
    // "signed_out" deliberately does NOT re-enable the button. The session this
    // screen was loaded under is dead (the transport has already dropped the
    // token), so pressing it again can only fail the same way; the card says so
    // and points at the way back in. Re-rendering here would replace that card
    // with the gate before the fan had read it.
    if (result === "added") renderKeepingFrom("[data-passkey-out]");
    else if (result !== "signed_out" && result !== "reauth") btn.disabled = false; // RG NEW-2: reauth signed this browser out
  });
  // Send (and re-send: one button, because the client is never given the
  // address back — see ADD_FORM in backup-email.ts).
  root.querySelector<HTMLButtonElement>('[data-act="email-send"]')?.addEventListener("click", async (e) => {
    const btn = e.currentTarget as HTMLButtonElement;
    btn.disabled = true;
    const address = cleanAddress(val("#be-addr"));
    if (!looksLikeAddress(address)) {
      // Shape only, and shape is public: this spends no send-cap slot and says
      // nothing about any account.
      say("[data-email-msg]", "That doesn't look like an email address.");
      btn.disabled = false;
      return;
    }
    // I3-0215 (I16-FXE): changing a CONFIRMED address is two posts. The first answers `confirm_current`
    // (a code went to the address already on file); the second carries that code as `current_code`.
    // A malformed code is refused here, before any POST, so it spends no send-cap slot.
    const curInput = page.querySelector<HTMLInputElement>("#be-current-code");
    const currentCode = curInput ? cleanCode(curInput.value) : "";
    if (curInput && !looksLikeCode(currentCode)) {
      say("[data-email-msg]", `Enter the ${OTP_DIGITS}-digit code we sent to the address you already confirmed.`);
      btn.disabled = false;
      curInput.focus();
      return;
    }
    say("[data-email-msg]", "Sending…");
    const enrolStart = await deps.api.addBackupEmail(address, curInput ? currentCode : undefined);
    const out = refusalOutcome(enrolStart, enrolStartOutcome(enrolStart));
    // D-1016(1): the way back is a fresh sign-in — sign this browser out first, as the passkey buttons do, and leave
    // the button off: pressing it again on this session can only be refused the same way.
    if (out.kind === "reauth") await deps.api.signOut();
    show(out);
    btn.disabled = out.kind === "reauth";
    if (out.kind === "confirm_current" || out.kind === "bad_current_code") {
      // Reveal the current-address code field once, beside the address it confirms; never auto-retry.
      let field = curInput;
      if (!field) {
        const wrap = document.createElement("div");
        wrap.dataset.currentCodeStep = "";
        wrap.innerHTML = `<label class="fl" for="be-current-code">The ${OTP_DIGITS}-digit code we sent to the address you already confirmed</label><input id="be-current-code" name="current_code" type="text" inputmode="numeric" autocomplete="one-time-code" spellcheck="false" maxlength="${OTP_DIGITS}" placeholder="${"0".repeat(OTP_DIGITS)}">`;
        (btn.closest(".sacts") ?? btn).before(wrap);
        field = wrap.querySelector<HTMLInputElement>("#be-current-code");
      } else field.value = "";
      field?.focus();
    }
    // Reveal the code step in place rather than re-rendering: a re-render would
    // wipe the "check that inbox" card the fan is meant to act on. NO auto-retry
    // on any failure — a client retry loop would eat the per-IP send cap that
    // makes mailing strangers expensive.
    if (out.kind === "code_sent") {
      page.querySelector<HTMLElement>("[data-code-step]")?.classList.add("on");
      page.querySelector<HTMLInputElement>("#be-code")?.focus();
      // FU-01 — AND THE DESTRUCTIVE CONTROL MUST STOP ADVERTISING AN ACT THE
      // SERVER WILL NO LONGER PERFORM. This branch reveals the code step in
      // place instead of re-rendering (a re-render would wipe the "check that
      // inbox" card), which left "Remove backup email" mounted and wired from
      // the no-pending render. A change is now in flight, so the very same
      // DELETE cancels it and keeps the confirmed address — exactly the state
      // the mid-change render exists to describe. Retarget the button rather
      // than removing it: a fan who typed the wrong new address still needs a
      // way out of this state, and `securityView` offers none other here.
      // `data-enrol="pending"` (the never-confirmed enrolment's Cancel) is left
      // alone — that state has no confirmed address to preserve.
      const destructive = page.querySelector<HTMLButtonElement>('[data-act="email-remove"]');
      if (destructive && destructive.dataset.enrol !== "pending") {
        destructive.dataset.changePending = "1";
        destructive.textContent = "Cancel this change";
      }
    }
  });
  root.querySelector<HTMLButtonElement>('[data-act="email-verify"]')?.addEventListener("click", async (e) => {
    const btn = e.currentTarget as HTMLButtonElement;
    btn.disabled = true;
    const code = cleanCode(val("#be-code"));
    // A code dies after five wrong tries, so a blank or half-typed field must
    // not be posted — that would spend one of the five on a typo.
    if (!looksLikeCode(code)) {
      say("[data-email-msg]", `Enter the ${OTP_DIGITS} digits from the email.`);
      btn.disabled = false;
      return;
    }
    say("[data-email-msg]", "Checking…");
    const enrolVerify = await deps.api.verifyBackupEmail(code);
    const out = refusalOutcome(enrolVerify, enrolVerifyOutcome(enrolVerify));
    show(out);
    // On success the screen re-renders into the state the SERVER holds (the
    // confirmed pill and the masked address), which says the same thing the
    // card did — the client never leaves a claim on screen it did not re-read.
    if (out.kind === "verified") renderKeepingFrom("[data-email-out]");
    else btn.disabled = false;
  });
  /**
   * THE LOST-DEVICE ACT. One handler for every row, because the rows are a list
   * the server owns; the credential id rides on the button that was pressed.
   *
   * ALWAYS RE-RENDER AFTERWARDS, on every outcome including the failures. A
   * revoke ends every session, so the screen this button was pressed on is
   * either running on a brand-new token (success) or on one the server has
   * already stopped accepting — and either way the only honest next screen is
   * one read back from the server. Leaving a stale list up here would leave a
   * removed device on screen next to a "Remove" button that can only 404.
   */
  root.querySelectorAll<HTMLButtonElement>('[data-act="revoke-passkey"]').forEach((btn) => {
    btn.addEventListener("click", async () => {
      const cred = btn.dataset.cred;
      if (!cred) return;
      if (!(await confirmDialog("Remove this passkey?", [REVOKE_COST], "Remove passkey"))) return;
      root.querySelectorAll<HTMLButtonElement>('[data-act="revoke-passkey"]').forEach((b) => { b.disabled = true; });
      say("[data-passkey-msg]", "Removing…");
      const revokeAnswer = await deps.api.revokeFanPasskey(cred);
      const out = refusalOutcome(revokeAnswer, revokeOutcome(revokeAnswer));
      say("[data-passkey-msg]", "");
      put("[data-passkey-out]", emailOutcomeCard(out));
      renderKeepingFrom("[data-passkey-out]");
    });
  });
  /**
   * #814a — REMOVE A LINKED GOOGLE/APPLE SIGN-IN. Two steps in the page (no
   * browser pop-up, I3-0422): "Remove" reveals the cost and a "Yes, remove"
   * control; only that one calls `DELETE /account/oauth/:provider`. The server
   * re-mints this device's session (every other device is signed out), and the
   * screen is read back afterwards on every outcome, as the passkey revoke is.
   */
  root.querySelectorAll<HTMLButtonElement>('[data-act="oauth-unlink"]').forEach((btn) => {
    btn.addEventListener("click", () => {
      const p = btn.dataset.provider;
      const box = p ? root.querySelector<HTMLElement>(`[data-unlink-confirm="${p}"]`) : null;
      if (box) box.hidden = false;
    });
  });
  root.querySelectorAll<HTMLButtonElement>('[data-act="oauth-unlink-confirm"]').forEach((btn) => {
    btn.addEventListener("click", async () => {
      const p = btn.dataset.provider;
      if (p !== "google" && p !== "apple") return;
      root.querySelectorAll<HTMLButtonElement>('[data-act="oauth-unlink-confirm"]').forEach((b) => { b.disabled = true; });
      say("[data-unlink-msg]", "Removing…");
      const unlinkAnswer = await deps.api.unlinkOAuth(p);
      const out = refusalOutcome(unlinkAnswer, unlinkOutcome(unlinkAnswer, p));
      say("[data-unlink-msg]", "");
      put("[data-unlink-out]", emailOutcomeCard(out));
      renderKeepingFrom("[data-unlink-out]");

  root.querySelector<HTMLButtonElement>('[data-act="email-remove"]')?.addEventListener("click", async (e) => {

    // CONFIRMED address is the fan's way back in if the passkey goes, while a
    // never-confirmed enrolment can sign nobody in and costs nothing to drop.
    // One confirm each, and `removeOutcome` is told which act it is describing.
    // FU-01 — a replacement was started in THIS document since the render, so
    // this button's call is a cancel, not a removal. One shared act, so the
    // receipt comes from the read-back rather than from the flattened 200.
    if (btn.dataset.changePending === "1") return void cancelChangeAct(btn);
    const pendingOnly = btn.dataset.enrol === "pending";
    const question = pendingOnly
      ? "Cancel this backup email? It was never confirmed, so nothing about how you sign in changes — and you can add a different address later."
      : "Remove your backup email? If you then lose your passkey, nobody can let you back in.";

      remInput.focus();
      return;

    if (!remInput && !(await confirmDialog(pendingOnly ? "Cancel this backup email?" : "Remove your backup email?", [question], pendingOnly ? "Cancel it" : "Remove it"))) return;
    btn.disabled = true;
    say("[data-email-msg]", pendingOnly ? "Cancelling…" : "Removing…");

    let out = refusalOutcome(answer, removeOutcome(answer));
    // FU-01, the cross-document half: a replacement started in ANOTHER tab (or
    // on a phone) since this screen rendered turns the same 200 into a cancel,
    // and no field on the response says so. Read the account back and only
    // re-describe the act when the read-back DEFINITELY shows the confirmed
    // address survived. An unreadable read-back changes nothing — it is no
    // evidence of a pending change, and the 200's ordinary meaning stands.
    if (out.kind === "removed" && authPanelState(await deps.api.authState(), true).name === "verified") {
      out = cancelPendingOutcome(answer, true);
    }
    // I49-A050: as the add door (D-1016(1)): sign this browser out before the card, and leave the button off.
    if (out.kind === "reauth") await deps.api.signOut();

    if (out.kind === "remove_confirm_current" || (remInput && out.kind === "bad_current_code")) {
      // Reveal the code field once, beside the button it arms; never auto-retry.
      let field = remInput;
      if (!field) {
        const wrap = document.createElement("div");
        wrap.dataset.removeCodeStep = "";
        wrap.innerHTML = `<label class="fl" for="be-remove-code">The ${OTP_DIGITS}-digit code we sent to the address you're removing</label><input id="be-remove-code" name="current_code" type="text" inputmode="numeric" autocomplete="one-time-code" spellcheck="false" maxlength="${OTP_DIGITS}" placeholder="${"0".repeat(OTP_DIGITS)}">`;
        (btn.closest(".sacts") ?? btn).before(wrap);
        field = wrap.querySelector<HTMLInputElement>("#be-remove-code");
      } else field.value = "";
      field?.focus();
    }
    if (out.kind === "removed" || out.kind === "change_cancelled") renderKeepingFrom("[data-email-out]");
    else btn.disabled = out.kind === "reauth";
  });
  /**
   * CANCEL A CHANGE — the same `DELETE /account/email` call the removal makes.
   * The server decides which act it is from the row (a change in flight is
   * cancelled and the confirmed address is left alone), and this button is only
   * ever rendered in the state where the cancel is the act — see securityView.
   *
   * The success card is not claimed from the 200. The account is READ BACK and
   * "Change cancelled" is said only if it still holds a confirmed address,
   * because the one way this client can be wrong about which act happened is a
   * change that completed between the render and the press.
   */
  root.querySelector<HTMLButtonElement>('[data-act="email-cancel-change"]')?.addEventListener("click", async (e) => {
    await cancelChangeAct(e.currentTarget as HTMLButtonElement);
  });
}

/**
 * M2 — `#/signin/email`, the lost-passkey path. Three steps in ONE document:
 * the address the fan typed stays in its own field and is read back from there
 * at verify time, so it is never copied into storage, into a hash, or into a
 * log. The code is read, sent, and dropped.
 */
function wireOtpSignin(): void {
  const page = root.querySelector<HTMLElement>("[data-otpsignin]");
  if (!page) return;
  const step = (n: 1 | 2 | 3): void => {
    page.querySelectorAll<HTMLElement>("[data-otp-step]").forEach((p) => { p.classList.toggle("on", p.dataset.otpStep === String(n)); });
  };
  const addr = page.querySelector<HTMLInputElement>("#ot-addr");
  // I21-Q152: /fan/otp/start takes the bot check on every call; its widgets sit above the send button.
  insertBotSlots(page.querySelector('[data-act="otp-start"]'));
  void mountBotChecks(page);
  root.querySelector<HTMLButtonElement>('[data-act="otp-start"]')?.addEventListener("click", async (e) => {
    const btn = e.currentTarget as HTMLButtonElement;
    btn.disabled = true;
    const address = cleanAddress(val("#ot-addr"));
    if (!looksLikeAddress(address)) {
      say("[data-otp-msg]", "That doesn't look like an email address.");
      btn.disabled = false;
      return;
    }
    say("[data-otp-msg]", "Sending…");
    const otpStart = await deps.api.fanOtpStart(address);
    const out = refusalOutcome(otpStart, otpStartOutcome(otpStart));
    say("[data-otp-msg]", "");
    put("[data-otp-out]", emailOutcomeCard(out));
    btn.disabled = false;
    // ADVANCE ON EVERY 200, unconditionally. The step the fan lands on must not
    // depend on whether the address has an account — the server made those two
    // answers identical and a UI that branched here would hand the oracle back.
    if (out.kind === "maybe_sent") {
      if (addr) addr.readOnly = true;
      step(2);
    }
  });
  root.querySelector<HTMLButtonElement>('[data-act="otp-restart"]')?.addEventListener("click", () => {
    if (addr) {
      addr.readOnly = false;
      addr.focus();
    }
    put("[data-otp-out]", "");
    say("[data-otp-msg]", "");
    step(1);
  });
  root.querySelector<HTMLButtonElement>('[data-act="otp-verify"]')?.addEventListener("click", async (e) => {
    const btn = e.currentTarget as HTMLButtonElement;
    btn.disabled = true;
    const code = cleanCode(val("#ot-code"));
    if (!looksLikeCode(code)) {
      say("[data-otp-msg]", `Enter the ${OTP_DIGITS} digits from the email.`);
      btn.disabled = false;
      return;
    }
    say("[data-otp-msg]", "Checking…");
    const otpVerify = await deps.api.fanOtpVerify(cleanAddress(val("#ot-addr")), code);
    const out = refusalOutcome(otpVerify, otpVerifyOutcome(otpVerify));
    // QUEUE #417 — a closed account that just proved itself here is told so, and
    // is offered its data-rights doors, by render()'s closed-account branch — the
    // same repaint the OAuth door does (`oauthSignin` above). Without this the
    // OTP door answered a closed account with the generic failure card.
    if (out.kind !== "signed_in" && deps.api.lastClosure()) {
      await render();
      return;
    }
    say("[data-otp-msg]", "");
    put("[data-otp-out]", emailOutcomeCard(out));
    if (out.kind === "signed_in") {
      // The session is live. Do NOT navigate away: registering a passkey right
      // here is the act that actually recovers the account, and this is the one
      // moment the fan will do it.
      const code = page.querySelector<HTMLInputElement>("#ot-code");
      if (code) code.value = ""; // the code is spent; never leave it on screen
      step(3);
    } else {
      btn.disabled = false;
    }
  });
  // FU-08 — "Not now" is an EXIT, so it takes the same exit path as the others.
  // It was an `<a href="#/me">` in the view, which both dropped the destination
  // the fan was gated out of and left the remembered value uncleared for a later
  // unrelated sign-in to pick up.
  root.querySelector<HTMLButtonElement>('[data-act="otp-skip"]')?.addEventListener("click", () => {
    go(consumeReturn());
  });
  root.querySelector<HTMLButtonElement>('[data-act="otp-passkey"]')?.addEventListener("click", async (e) => {
    const btn = e.currentTarget as HTMLButtonElement;
    btn.disabled = true;
    say("[data-otp-msg]", "Waiting for your passkey…");
    let refusal: string | null = null;
    const result = await addPasskeyFlow({ api: deps.api, createCredential: passkeySupported() ? createCredentialAdapter : null, onRefusal: (m) => { refusal = m; }, onReauth: async () => { await deps.api.signOut(); } });
    say("[data-otp-msg]", "");
    const added = addPasskeyOutcome(result);
    put("[data-otp-out]", emailOutcomeCard(result === "added" ? added : { ...added, body: refusal ?? added.body })); // I24 (D-0996(2))
    if (result === "added") {
      // M2: honor the deep-link the fan arrived on (e.g. a redeem code) instead
      // of always dumping them on #/me. They came here to DO something; the
      // email backup + passkey step is a detour, not the destination.
      const dest = consumeReturn();
      goKeeping(dest, root.querySelector("[data-otp-out]")?.textContent ?? "");
    } else if (result !== "signed_out" && result !== "reauth") btn.disabled = false; // RG NEW-2: reauth signed this browser out
  });
}

/** I3-2106 — a modified or non-primary click on a real link (Ctrl/Cmd/Shift/
 * Alt, middle button) is the browser's: open-in-new-tab must keep working, so
 * the SPA only takes a plain primary click. A `data-nav` on a non-link has no
 * browser behaviour to defer to and is always taken. */
function browserOwnsClick(e: MouseEvent): boolean {
  const t = e.currentTarget as Element | null;
  if (!t || t.tagName !== "A" || !t.hasAttribute("href")) return false;
  return e.button !== 0 || e.metaKey || e.ctrlKey || e.shiftKey || e.altKey;
}

// Declarative event wiring (delegation): data-nav navigates, data-act runs an
// action, data-tab switches a panel, data-edition opens a frozen edition.
function wire(): void {
  // I3-0036 — THE SKIP LINK MUST NOT REACH THE ROUTER. `href="#main-content"`
  // is the correct markup for a bypass block and is what works with no
  // controller running, but this app is hash-routed: `parseHash` reads
  // `main-content` as a route head, matches nothing, and returns
  // `{ name: "home" }` — so pressing the skip link NAVIGATED AWAY from the page
  // it was supposed to skip into. Cancelling the default means the hash never
  // changes, `hashchange` never fires, and `render()` is never entered; the
  // focus move is what the link was for and is done here explicitly, because
  // moving focus is not something the browser does for a fragment target that
  // is not itself focusable.
  root.querySelectorAll<HTMLElement>("[data-skip]").forEach((el) => {
    el.addEventListener("click", (e) => {
      e.preventDefault();
      const target = document.getElementById(el.dataset.skip!);
      if (!target) return;
      target.setAttribute("tabindex", "-1");
      target.focus();
      try {
        target.scrollIntoView();
      } catch {
        /* not implemented in every test realm — never block on it */
      }
    });
  });
  root.querySelectorAll<HTMLElement>("[data-nav]").forEach((el) => {
    el.addEventListener("click", (e) => {
      if (browserOwnsClick(e)) return; // I3-2106 — new tab / new window stays the browser's
      e.preventDefault();
      go(el.dataset.nav!);
    });
  });
  // I3-1063 — THE TAB STRIP SAID ONE THING TO THE EYE AND ANOTHER TO A SCREEN
  // READER. The strip declares `role="tab"` and ships a correct INITIAL
  // `aria-selected` (views.ts:1263-1268: Card true, the rest false), and this
  // handler then moved only the `on` CLASS. So the markup's promise held for
  // exactly as long as nobody used it: after any tab press the page still
  // announced "Card, selected" on every tab, and `aria-selected` is the ONLY
  // thing that answers "which one am I on" for a `tablist`. The class and the
  // state are set from the same comparison now, so they cannot diverge again.
  // Roving tabindex rides with it — a tablist takes ONE tab stop, and the
  // selected tab is the one that takes it.
  root.querySelectorAll<HTMLElement>("[data-tab]").forEach((btn) => {
    btn.addEventListener("click", () => {
      const which = btn.dataset.tab!;
      root.querySelectorAll<HTMLElement>("[data-tab]").forEach((b) => {
        const on = b === btn;
        b.classList.toggle("on", on);
        b.setAttribute("aria-selected", on ? "true" : "false");
        b.setAttribute("tabindex", on ? "0" : "-1");
      });
      root.querySelectorAll<HTMLElement>("[data-panel]").forEach((p) => { p.classList.toggle("on", p.dataset.panel === which); });
    });
  });
  // I3-0436 (I16-W04F HO-8): the shelf controls re-render the route with the fan's choice.
  root.querySelector<HTMLSelectElement>("[data-shelf-sort]")?.addEventListener("change", (e) => {
    const v = (e.target as HTMLSelectElement).value;

    void render();
  });
  root.querySelector<HTMLSelectElement>("[data-shelf-creator]")?.addEventListener("change", (e) => {
    shelf = { ...shelf, creator: (e.target as HTMLSelectElement).value || null };
    void render();
  });
  root.querySelectorAll<HTMLElement>("[data-edition]").forEach((chip) => {
    chip.addEventListener("click", async () => {

      const psp = chip.dataset.psp!;

      if (!view) return;
      view.innerHTML = `<div class="loading">Loading edition…</div>`;
      const ed = await deps.api.edition(psp, week);
      // The subject kind travels on the chip, not in the edition body (see
      // `editionsGallery`). It is a dataset attribute in the viewer's OWN DOM,
      // so it is not a trust boundary — anyone who can rewrite it can rewrite
      // the rendered SVG next to it. What matters is that an arbitrary string
      // never reaches the renderer: `asCardSubjectKind` narrows it to one of the
      // three known kinds and drops anything else.

    });
  });
  // v14: the lookup screen's platform buttons — each runs the SAME local
  // canonicalize→hash→one-bucket check the URL box runs; nothing about the
  // handle leaves the device beyond the k-anonymous prefix.
  root.querySelectorAll<HTMLElement>('[data-act="lookup-check"]').forEach((btn) =>
    { btn.addEventListener("click", async () => {
      const platform = btn.dataset.platform ?? "";
      const handle = btn.dataset.handle ?? "";

      // same LOOKUP_HANDLE_RE the route admits. checkAccount canonicalizes it
      // locally for the k-anon check; the handle is xml-escaped wherever rendered.
      if (!/^[a-z0-9_]{1,32}$/.test(platform) || !LOOKUP_HANDLE_RE.test(handle)) return;

      // local state ONLY on a confirmed ok, so the button never claims a
      // withdrawal the server did not make.
      const on = btn.dataset.on === "1";
      const r = act === "save" ? await deps.api.save(psp) : on ? await deps.api.uncollect(psp) : await deps.api.collect(psp);

      if (r.ok) {
        btn.dataset.on = on ? "0" : "1";

  //

  // read is deliberately tri-state in `warView` for the same reason.
  // I3-0529 (D-0858(13)) — the PRACTICE join. `joinWarPractice` (app.ts) picks

  root.querySelectorAll<HTMLButtonElement>('[data-act="war-recon-commit"]').forEach((btn) => {

function openHandshakeForm(btn: HTMLElement, psp: string): void {

    radio.value = opt.value;

  }
  form.appendChild(scopeFieldset);

  // The numbered account list, as radios -- same subjects/labels the old menu

async function submitHandshake(btn: HTMLElement, form: HTMLFormElement, psp: string, scope: "identity" | "payment", subject: string): Promise<void> {
  form.remove();
  btn.setAttribute("disabled", "true");
  const r = await deps.api.askHandshake(psp, scope, subject);
  if (r.ok) {
    btn.textContent = "Asked — you'll see the answer in your account";
  } else if (r.status === 409) {
    // FU-11: the refusal now says what WOULD be accepted. The service's one
    // refusal for unknown/revoked/shielded stays one refusal -- this adds no
    // information about which, it only names the form.
    btn.textContent = "Nothing on this page matches that — pick one of the listed accounts, or paste its full link";
    btn.removeAttribute("disabled");
  } else if (r.status === 403) {
    btn.textContent = refusalText(r, "This creator isn't taking handshakes from you");
  } else if (r.status === 429) {
    btn.textContent = refusalText(r, "You've reached your handshake limit");
  } else {
    btn.textContent = "Try again";
    btn.removeAttribute("disabled");
  }
}

/** The confirmation token a `#/p/<psp>/newsletter/<token>` hash carries for THIS passport, else null. */
export function newsletterTokenFromHash(hash: string, psp: string): string | null {
  const m = /^#\/p\/(psp_[0-9a-f]{16})\/newsletter\/(nwc_[0-9a-f]{32})$/.exec(hash ?? "");
  return m && m[1] === psp ? m[2]! : null;
}

const err = (data: unknown): string => (data !== null && typeof data === "object" && typeof (data as { error?: unknown }).error === "string" ? (data as { error: string }).error : "");

export function subscribeSentence(r: { ok: boolean; status: number; data: unknown }): string {
  if (r.ok) {
    const d = r.data as { state?: unknown; mailed?: unknown } | null;
    if (d?.state === "confirmed") return "You're subscribed.";
    return d?.mailed === true ? "Check your email for the confirmation link." : "Your request is waiting for you to confirm it: check your email.";
  }
  if (r.status === 401) return "Sign in to subscribe.";
  if (r.status === 409 && err(r.data) === "no_verified_email") return "Add and verify an email address in your account first.";
  if (r.status === 404) return "This creator isn't taking newsletter subscriptions.";
  if (r.status === 429) return "Too many tries. Try again later.";
  if (r.status === 503) return "We couldn't send the confirmation email just now. Nothing was saved. Try again later.";
  return "We couldn't do that just now. Try again later.";
}

export function confirmSentence(r: { ok: boolean; status: number; data: unknown }, name: string): string {
  if (r.ok) return `Confirmed. ${name} can now export your email address.`;
  if (r.status === 401) return "Sign in, then open the link again.";
  if (r.status === 409) return "This link was already used. Your subscription is confirmed.";
  if (r.status === 410) return "This link has expired. Subscribe again to get a new one.";
  if (r.status === 429) return "Too many tries. Try again later.";
  if (r.status === 404 && err(r.data) === "creator_unavailable") return "This creator isn't taking newsletter subscriptions.";
  if (r.status === 404 || r.status === 400) return "This link isn't valid for the account you're signed in with.";
  return "We couldn't do that just now. Try again later.";
}

export function unsubscribeSentence(r: { ok: boolean; status: number; data: unknown }): string {

/**
 * THE TWO PASSKEY DECISION TABLES — the ONE copy of "what did that ceremony
 * actually mean", shared by three clients.
 *
 * It lives in its own module, with NO imports at all, for two reasons:
 *
 *  1. `apps/mobile-check/test/backup-email-cross-client.test.ts` imports these
 *     functions and RUNS them, then parses the Swift and Kotlin mirrors out of
 *     their source and evaluates those against the same scenarios. That guard
 *     compiles under a config with no DOM lib, so anything reachable from here
 *     must not name `fetch`, `Response` or `RequestInit`.
 *  2. Being importable from nothing makes it obvious that these are pure
 *     decisions over primitives — no transport, no storage, no DOM.
 *
 * WHY THEY EXIST. Both ceremonies used to collapse a status-carrying answer
 * back into a `boolean` at the transport, on all three clients, and every
 * honest distinction died in that collapse:
 *
 *  • `created` — the server's own "I made a NEW account" signal — was
 *    discarded outright. `/fan/passkey/register` serves two acts: a JOIN
 *    (anonymous, account birth) and an ADD (put another passkey on the account
 *    already in hand, which is the last step of lost-passkey recovery). A join
 *    answer to an add request means the fan's real account got nothing and a
 *    brand-new empty one got their passkey. Every client rendered that as
 *    "Passkey ready on this device."
 *  • On the sign-in leg, 503-passkeys-off, 401-unknown-credential, a user
 *    cancel and a request that never completed all arrived as one `false`
 *    rendered as "Sign-in failed. Please try again." Retrying can never fix the
 *    first two, and the last is not a refusal at all.
 */

/** Which of the two acts the shared `/fan/passkey/register*` routes is being
 * asked to perform. */
export type PasskeyIntent = "join" | "add";

/**
 * What the server said about a finished registration. `created` is ITS OWN
 * signal — true means it made a new account — and `null` means the answer did
 * not carry it (an unreadable body, or a server too old to say), which is
 * treated as "we do not know", never as "add".
 */
export interface PasskeyRegisterAnswer {
  ok: boolean;
  status: number;
  created: boolean | null;
  /** Set only on the SYNTHETIC 401 a client invents when it holds no token and
   * therefore made no request. See NoSessionEnvelope in api.ts. */
  noSession?: true;
  /**
   * I3-0934 — the server's own `error` string on a REAL 401 answer, when the
   * transport read one. Purely additive: `registerVerdict` below does not
   * read it (widening `RegisterVerdict` broke every exhaustive switch over it
   * downstream — apps/web/src/passkey.ts's `AddPasskeyOutcome` among them,
   * none of them this room's to edit). `registerFailureCause` reads it
   * instead, as a SEPARATE, optional finer-grained answer a caller may
   * consult without changing what `registerVerdict` returns or means.
   */
  serverError?: string;
  /** I3-0792 (D-0995(12)): the refusal's plain `message` (api.ts refusalMessage), when the server sent one —
   * shown verbatim by the sign-up screens instead of their own copy. */
  message?: string;
}

/** What the server said about a finished sign-in. `signedIn` is the token
 * landing in the store, read back — never merely a 200. */
export interface PasskeyAuthAnswer {
  ok: boolean;
  status: number;
  signedIn: boolean;
  /** I3-0934 (I34-W2-CLIENTS): the refusal's `error` string (fan-auth.ts finishFanAuth throws four different 401s). */
  serverError?: string;
}

/** What a finished `/fan/passkey/register` means for the act that was asked for. */
export type RegisterVerdict = "joined" | "added" | "signed_out" | "unavailable" | "failed";

/**
 * THE RULE THAT WOULD HAVE CAUGHT THE ORIGINAL DEFECT, stated once:
 *
 *   an ADD that comes back `created:true` is NOT a success.
 *
 * `created:true` means the server made a brand-new empty account. That is the
 * right answer to a join and it is never the answer to "put a passkey on my
 * account" — the fan asking for that already HAS an account, and the one they
 * were trying to recover got nothing.
 *
 * `created: null` (the answer did not carry the field) is fail-closed for the
 * same reason: an add may only report success on an answer that positively said
 * it did not create an account.
 */
export function registerVerdict(intent: PasskeyIntent, a: PasskeyRegisterAnswer): RegisterVerdict {
  // Fail closed on config before anything else: passkeys are off on this
  // deployment, which is a fact about the deployment and not about the fan.
  if (a.status === 503) return "unavailable";
  // 401 is "the credential you presented is not a live fan session" — on an
  // ADD, and the synthetic 401 (this client held no token, so it never asked)
  // says the same thing one step earlier.
  if (a.noSession === true || a.status === 401) return intent === "add" ? "signed_out" : "failed";
  if (!a.ok) return "failed";
  if (intent === "join") return a.created === true ? "joined" : "failed";
  return a.created === false ? "added" : "failed";
}

/**
 * I3-0934 — THE FINER-GRAINED CAUSE BEHIND A 401, when the server's answer
 * carries one. `registerVerdict` above stays exactly as it was (three
 * distinct causes still fold into "signed_out"/"failed" there, on purpose:
 * widening its own return type breaks every exhaustive switch a downstream
 * client already wrote over it). This is the SEPARATE, additive answer a
 * caller can consult to give distinct advice for the three server-side 401s
 * on `/fan/passkey/register` (services/trust-core/src/fan-auth.ts):
 * `"unauthorized"` (the session itself is dead), `"challenge invalid or
 * expired"` (a spent/expired ceremony) and `"challenge was not issued for
 * this session"` (bound to a different account). `"unknown"` covers an
 * unrecognized or absent `serverError` (an older server, or the synthetic
 * 401) — never a guess.
 */
export type RegisterFailureCause = "session_dead" | "challenge_expired" | "wrong_account" | "unknown";

export function registerFailureCause(a: PasskeyRegisterAnswer): RegisterFailureCause {
  if (a.noSession === true) return "session_dead";
  if (a.status !== 401) return "unknown";
  if (a.serverError === "challenge invalid or expired") return "challenge_expired";
  if (a.serverError === "challenge was not issued for this session") return "wrong_account";
  if (a.serverError === "unauthorized") return "session_dead";
  return "unknown";
}

/** What a finished `/fan/passkey/auth` means. Five answers where there was one. */
export type SignInVerdict = "signed_in" | "unavailable" | "no_credential" | "expired" | "no_answer" | "failed";

/**
 * The sign-in leg's table. Note what is NOT here: "cancelled". A cancel happens
 * in the authenticator, before any request, so the flow reports it directly —
 * there is no server answer to classify.
 *
 * Written as a flat, ordered list of `if <condition> return <value>` rows, the
 * same shape as the Swift and Kotlin mirrors, because the cross-client guard
 * evaluates those rows against the scenarios it runs this function on.
 */
export function signInVerdict(a: PasskeyAuthAnswer): SignInVerdict {
  if (a.status === 503) return "unavailable";
  if (a.status === 0) return "no_answer"; // the request never completed — we know nothing
  // I3-0934: a 401 is four causes at fan-auth.ts finishFanAuth; only "unknown credential" means the passkey is unknown.
  if (a.status === 401 && a.serverError === "challenge invalid or expired") return "expired";
  if (a.status === 401 && typeof a.serverError === "string" && a.serverError !== "unknown credential") return "failed";
  if (a.status === 401) return "no_credential"; // "unknown credential", or an answer with no cause (the native mirrors' row)
  if (a.ok && a.signedIn) return "signed_in";
  return "failed";
}

/**
 * The passkey-first sign-in + auto-complete flow. mode "auth" = existing
 * passkey (usernameless sign-in); "register" = brand-new fan (join — account
 * birth at finish). On successful sign-in it AUTO-COMPLETES the original
 * intent by re-firing `completeRequest` for the same passport, so the fan
 * lands in the requested state without a re-click.
 *
 * Outcomes: "unavailable" (no adapter / passkeys not configured server-side),
 * "no_credential" (the server does not know that passkey), "cancelled" (the
 * authenticator returned nothing — no request was sent), "no_answer" (the
 * request never completed, which is not a refusal), "signin_failed" (everything
 * else), "signin_only" (signed in but the request re-fire failed — the session
 * still stands), "done".
 */
export async function passkeyRequestFlow(deps: {
  api: Api;
  mode: "auth" | "register";
  getAssertion: GetAssertion | null;
  createCredential: CreateCredential | null;
  completeRequest: () => Promise<boolean>;
  /** I3-0792 (D-0995(12)): told the server's refusal `message`, verbatim, when a refused leg carried one. */
  onRefusal?: (message: string) => void;
}): Promise<PasskeyOutcome> {
  const refused = (m: string | undefined): void => {
    if (typeof m === "string" && m !== "") deps.onRefusal?.(m);
  };
  if (deps.mode === "auth") {
    if (!deps.getAssertion) return "unavailable";
    const start = await deps.api.fanPasskeyAuthStart();
    if (!start.ok) refused(start.message);
    if (!start.ok || !start.data) return start.status === 503 ? "unavailable" : start.status === 0 ? "no_answer" : "signin_failed";
    let fields: AssertionFieldsDto | null = null;
    try {
      fields = await deps.getAssertion(start.data);
    } catch {
      fields = null; // authenticator errors are a cancel, never a crash
    }
    // A cancel happens IN the authenticator, before any request — there is no
    // server answer to classify, and it is not a failed sign-in.
    if (!fields) return "cancelled";
    const verdict = signInVerdict(await deps.api.fanPasskeyAuth(fields));
    if (verdict !== "signed_in") return verdict === "failed" ? "signin_failed" : verdict;
  } else {
    if (!deps.createCredential) return "unavailable";
    // JOIN, and it says so. This is the signed-out interstitial: a fan with no
    // account taps once and gets one, so the bearer is deliberately not sent
    // and `created:true` is the only answer that counts as a sign-in.
    const start = await deps.api.fanPasskeyRegisterStart("join");
    if (!start.ok) refused(start.message);
    if (!start.ok || !start.data) return start.status === 503 ? "unavailable" : start.status === 0 ? "no_answer" : "signin_failed";
    let fields: AttestationFieldsDto | null = null;
    try {
      fields = await deps.createCredential(start.data);
    } catch {
      fields = null;
    }
    if (!fields) return "cancelled";
    const answer = await deps.api.fanPasskeyRegister(fields, "join");
    if (registerVerdict("join", answer) !== "joined") {
      if (!answer.ok) refused(answer.message);
      return "signin_failed";
    }
  }
  // Auto-complete the original request under the fresh session.
  let requested = false;
  try {
    requested = await deps.completeRequest();
  } catch {
    requested = false;
  }
  return requested ? "done" : "signin_only";
}

/**

 * Two callers, one act: the "Add a passkey" button on `#/security`, and the

 * actually recovers the fan (the mailed code only got them through the door).
 *
 * Distinct from `passkeyRequestFlow` above: there is no join leg and nothing to
 * auto-complete afterwards — the server binds the ceremony to the caller's
 * account, so this must only ever run with a live session. It says `"add"` at
 * BOTH legs, which is what sends the bearer; without that the server sees an
 * anonymous caller and answers with a join, and finishing a join creates a
 * brand-new empty account. That is not a failure mode of this flow any more:
 * the server refuses a stale credential outright (401 → `signed_out`), and
 * `registerVerdict` refuses to call a `created:true` answer an "added".
 *

 * device, and passkeys unconfigured server-side (503).

export async function addPasskeyFlow(deps: { api: Api; createCredential: CreateCredential | null; onRefusal?: (message: string) => void; /** RG NEW-2: REQUIRED, so every caller decides the sign-out (typecheck is the per-site guard). */ onReauth: () => Promise<void> }): Promise<AddPasskeyOutcome> {

    if (typeof m === "string" && m !== "") deps.onRefusal?.(m);
  };
  // RG NEW-2: the way back is a fresh sign-in; the caller signs this browser out (`onReauth`) before the card shows.
  const reauth = async (): Promise<"reauth"> => {
    await deps.onReauth();
    return "reauth";

  if (!deps.createCredential) return "unavailable";
  const start = await deps.api.fanPasskeyRegisterStart("add");
  if (!start.ok) refused(start.message);
  if (!start.ok || !start.data) {
    if (start.status === 503) return "unavailable";
    // The session died between opening this screen and pressing the button (a
    // 30-day TTL crossing, another device signing out everywhere, the account
    // deleted elsewhere) — or there was never a token to send. Either way the
    // ceremony must not start: there is no account here to bind it to.
    if (start.noSession === true || start.status === 401) return "signed_out";
    if (isReauthRequired(start.status, (start.data as { error?: unknown } | null)?.error)) return reauth();
    return "failed";
  }
  let fields: AttestationFieldsDto | null = null;
  try {
    fields = await deps.createCredential(start.data);
  } catch {
    fields = null; // authenticator errors are a cancel, never a crash
  }
  if (!fields) return "failed";
  const answer = await deps.api.fanPasskeyRegister(fields, "add");
  if (!answer.ok) refused(answer.message);
  if (isReauthRequired(answer.status, answer.serverError)) return reauth();
  const verdict = registerVerdict("add", answer);
  // I3-0934: three different 401s used to read as "signed out". Two of them
  // are about the CEREMONY, not the session: an expired challenge (start
  // again) and one issued to another account (reload, start again).
  if (verdict === "signed_out") {
    const cause = registerFailureCause(answer);
    if (cause === "challenge_expired") return "expired";
    if (cause === "wrong_account") return "wrong_account";

  // answer to "failed" rather than adopting the new account — but the narrowing
  // is written out so a future edit to the table cannot quietly widen this.
  return verdict === "joined" ? "failed" : verdict;
}

// --- base64url helpers for the main.ts adapters (pure, testable) ---

export function b64urlToBytes(s: string): Uint8Array {
  const b64 = s.replace(/-/g, "+").replace(/_/g, "/") + "=".repeat((4 - (s.length % 4)) % 4);
  const bin = atob(b64);
  const out = new Uint8Array(bin.length);
  for (let i = 0; i < bin.length; i++) out[i] = bin.charCodeAt(i);
  return out;
}

export function bytesToB64url(bytes: ArrayBuffer | Uint8Array): string {
  const u8 = bytes instanceof Uint8Array ? bytes : new Uint8Array(bytes);
  let bin = "";
  for (const b of u8) bin += String.fromCharCode(b);
  return btoa(bin).replace(/\+/g, "-").replace(/\//g, "_").replace(/=+$/, "");
}
