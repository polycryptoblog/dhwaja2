const FAN_REPORT_STATES = ["open", "escalated", "suppressed", "dismissed", "published"] as const;
let fanReportState: string = "open";
let fanReportItems: FanReportRow[] = [];
let openReportId: string | null = null;
let selectedEvidenceId: string | null = null;
/** The live blob URL for the object on screen. Exactly one at a time, revoked
 * on every change: an operator-eyes-only object must not accumulate reachable
 * URLs in the document for the rest of the session. */
const evidenceUrl = objectUrlSlot();
let evidenceObjects: EvidenceObject[] | null = null;

function releaseEvidenceUrl(): void {
  // I3-1315 — supersede anything still in flight BEFORE revoking. A navigation
  // or a queue reload must invalidate every fetch already out, not only the URL
  // that happens to be live now.
  evidenceUrl.begin();
  evidenceUrl.release();
}

/** I3-1228: the keyset cursor the queue's last page answered with (`next_cursor`),
 * or null when there is no further page — or the worker does not page. */
let fanReportNext: string | null = null;
/**
 * I32-REPORTS hand-off 4 (I3-0304 limb 2): SEARCH THE QUEUE BY HANDLE. The search is the server's (`GET
 * /ops/fan-reports?handle=`, HANDOFF-fan-report-handle-search.patch for reports.ts/console.ts), because the queue is
 * paged and capped at `read_ceiling` and only a server-side filter can find a report past either. The server echoes
 * the handle it filtered on; a server that does not (one without the patch) is not trusted to have filtered, so the
 * loaded items are filtered here and the screen says the search covered only what is loaded.
 */
let fanReportHandle = "";
/** The handle as searched: trimmed, one leading `@` dropped, lower-cased, at most 100 characters. */
function fanReportHandleOf(raw: string): string {
  return raw.trim().replace(/^@/, "").toLowerCase().slice(0, 100);
}
function fanReportSearchForm(): string {
  return `<form id="fr-search" class="row" role="search" style="margin-bottom:10px"><label>Search by handle <input name="handle" type="search" maxlength="100" autocomplete="off" spellcheck="false" value="${esc(fanReportHandle)}"></label> <button class="ds-btn" type="submit">Search</button>${fanReportHandle ? ` <span class="muted">Search with the box empty to see the whole queue again.</span>` : ""}</form>`;
}
async function viewFanReports(more = false): Promise<void> {
  const cursor = more ? fanReportNext : null;
  const handleQ = fanReportHandle ? `&handle=${encodeURIComponent(fanReportHandle)}` : "";
  // D-1064(1): child_safety profile reports ride the ops overview and are listed ABOVE this queue.
  const [d, ov] = await Promise.all([api<{ items: FanReportRow[]; next_cursor?: string | null; truncated?: boolean; read_ceiling?: number; evidence_held?: HeldEvidenceList; handle?: string }>(`/ops/fan-reports?state=${encodeURIComponent(fanReportState)}${handleQ}${cursor ? `&cursor=${encodeURIComponent(cursor)}` : ""}`, ["items"]), api<OverviewStats>("/ops/overview")]);
  if (!d) return fail("Couldn't load the fan-report queue.");
  fanReportItems = cursor ? [...fanReportItems, ...d.items] : d.items;
  fanReportNext = typeof d.next_cursor === "string" && d.next_cursor !== "" ? d.next_cursor : null;
  const serverSearched = fanReportHandle !== "" && d.handle === fanReportHandle;
  const shown = fanReportHandle !== "" && !serverSearched ? fanReportItems.filter((i) => fanReportHandleOf(String(i.handle ?? "")) === fanReportHandle) : fanReportItems;
  const searchNote = fanReportHandle === ""
    ? ""
    : serverSearched
      ? `<div class="ds-banner banner" role="status">Showing only reports on the handle <b>${esc(fanReportHandle)}</b>, searched across this whole state.</div>`
      : `<div class="ds-banner banner banner-warn" role="status">This server does not search the queue by handle, so this shows only the reports on <b>${esc(fanReportHandle)}</b> among the ${esc(String(fanReportItems.length))} already loaded. There may be more further down the queue.</div>`;
  releaseEvidenceUrl();
  // #132PS-036 H-4 (I16-W10B, landed I16-L11): the worker reads at most `read_ceiling` items per state and says so.
  const ceiling = d.truncated === true && typeof d.read_ceiling === "number"
    ? `<div class="ds-banner banner banner-warn" role="status">More than ${esc(String(d.read_ceiling))} reports are in this state; only the first ${esc(String(d.read_ceiling))} by priority are listed. Close some and the rest appear. The oldest open report is on the overview.</div>`
    : "";
  app.innerHTML = profileChildSafetyBlock(ov === null ? null : (ov.profile_child_safety ?? null)) + ceiling + fanReportSearchForm() + searchNote + evidenceReviewScreen(shown, fanReportState, FAN_REPORT_STATES, ROLE, d.evidence_held) // I29-FX4 (FX3-05)
    + (fanReportNext ? `<div class="row" style="margin-top:10px"><span class="muted">${esc(String(shown.length))} shown — the queue has more.</span><span style="flex:1"></span><button class="ds-btn" data-act="fr-more">More</button></div>` : "");
  app.querySelector<HTMLButtonElement>('button[data-act="fr-more"]')?.addEventListener("click", () => void viewFanReports(true));
  app.querySelector<HTMLFormElement>("#fr-search")?.addEventListener("submit", (e) => {
    e.preventDefault();
    fanReportHandle = fanReportHandleOf(String(new FormData(e.currentTarget as HTMLFormElement).get("handle") ?? ""));
    openReportId = null; // the open report may not be one the search shows
    void viewFanReports();
  });
  app.querySelectorAll<HTMLButtonElement>("button[data-s]").forEach((b) => { b.onclick = () => {
    fanReportState = b.dataset.s ?? "open";
    openReportId = null;
    selectedEvidenceId = null;
    void viewFanReports();
  }; });
  wireRowlinks(app, (id) => void openReport(id));
  if (openReportId) void openReport(openReportId);
}

async function openReport(id: string): Promise<void> {
  const box = document.getElementById("frdetail");
  if (!box) return;
  const item = fanReportItems.find((i) => i.item_id === id);
  if (!item) return;
  openReportId = id;
  releaseEvidenceUrl();
  const ev = await api<{ evidence: EvidenceObject[] }>(`/ops/fan-reports/${encodeURIComponent(id)}/evidence`, ["evidence"]);
  evidenceObjects = ev ? ev.evidence : null;
  box.innerHTML = reportDetailPane(item, evidenceObjects, selectedEvidenceId, ROLE);
  wireReportDetail(item);
}

/** Fetch ONE object through the Access-gated route and hand the viewer a blob
 * URL. The bytes never become a link to storage and never leave the document. */
async function loadEvidenceObject(itemId: string, evId: string): Promise<void> {
  const slot = document.getElementById("evviewer");
  const obj = (evidenceObjects ?? []).find((o) => o.evidence_id === evId);
  if (!slot || !obj) return;
  // I3-1315 — the slot guarantees AT MOST ONE live URL, not that the one live
  // URL is the one the operator asked for. Two selections in flight, the first
  // resolving second, and the slot holds evidence A while the screen says B.
  const token = evidenceUrl.begin();
  slot.innerHTML = evidenceViewer(itemId, obj, "loading", null, ROLE);
  let status: ViewerStatus = "error";
  let url: string | null = null;
  try {
    const r = await fetch(`${OPS}/ops/fan-reports/${encodeURIComponent(itemId)}/evidence/${encodeURIComponent(evId)}`, { credentials: "include" });
    if (r.ok) {
      const candidateUrl = URL.createObjectURL(await r.blob());
      // I3-2165: a successful fetch used to be marked "ok" (and rendered as
      // <img src>) immediately — before the browser had decoded a single
      // pixel. A truncated or corrupt evidence file fetched cleanly and was
      // shown as successfully reviewed with no decode-failure handling at
      // all. `HTMLImageElement.decode()` resolves only once the image is
      // decoded and safe to paint, and rejects on a corrupt/undecodable
      // image — feature-detected because this console's DOM test harness
      // (jsdom) does not implement it, and a browser that lacks it keeps the
      // prior (immediate) behaviour rather than getting stuck on "Loading…".
      const probe = new Image();
      probe.src = candidateUrl;
      if (typeof probe.decode === "function") {
        try {
          await probe.decode();
          url = candidateUrl;
          status = "ok";
        } catch {
          // Revoked through a slot like every other object URL in this console
          // (console-safety: "revokes only via a slot"): adopt, then release.
          const discard = objectUrlSlot();
          discard.adopt(candidateUrl);
          discard.adopt(null);
          status = "error";
        }
      } else {
        url = candidateUrl;
        status = "ok";
      }
    } else if (r.status === 503) status = "unavailable";
    else if (r.status === 404) status = "missing";
  } catch {
    status = "error";
  }
  // The `return` is load-bearing: it drops the STALE RENDER as well as the

function openPurge(itemId: string, evId: string): void {
  const html = purgeConfirm(itemId, evId, WHO, ROLE);
  if (!html) return; // support seat: no confirm exists to press
  showModal(html);
  const cancel = document.querySelector<HTMLButtonElement>('#m-purge button[data-act="purge-cancel"]');
  if (cancel) cancel.onclick = () => closeModal();
  const go = document.querySelector<HTMLButtonElement>('#m-purge button[data-act="purge-confirm"]');
  if (go) go.onclick = async () => {
    const r = await apiDelete(`/ops/fan-reports/${encodeURIComponent(itemId)}/evidence/${encodeURIComponent(evId)}`);
    closeModal();
    if (!r.ok) { toast(failText(r.status, r.data, "Purge failed — the object is still there")); return; }
    const held = heldText(r.ok, r.data);
    if (held) { toast(held); return; }
    // UI-AUDIT F-A5 — THE SERVER SAYS WHETHER IT WROTE THE AUDIT LINE, AND THIS
    // TOAST USED TO ASSERT IT UNCONDITIONALLY.
    //
    // `DELETE /ops/fan-reports/:id/evidence/:ev` answers `{ok:true, audited}`,
    // and `audited` is false on purpose: the route deliberately moved the audit
    // write OUT of its try so an audit fault can never invert the outcome of an
    // irreversible purge (which is right). But the flag it added to say so was
    // never read here, and the console answered "the audit line is written" over
    // a response that had just said it was not.
    //
    // Evidence purge is the one action where the audit line IS the remaining
    // record — the bytes are gone and cannot be re-examined — so an operator who
    // believes a line exists has no reason to write one down. That is the whole
    // cost of this being wrong.
    const purged = (r.data ?? {}) as { audited?: unknown };
    toast(
      purged.audited === false
        ? "Purged — but the audit line was NOT written. The bytes and the row are gone; record this purge by hand now, with the item and evidence ids."
        : "Purged. The bytes and the row are gone; the audit line is written.",
    );
    selectedEvidenceId = null;
    releaseEvidenceUrl();
    void openReport(itemId);
  };
}

function leverFreezes(raw: string): { purchase_id: string; dispute_key: string }[] | null {
  const items = raw.split("\n").map((s) => s.trim()).filter((s) => s !== "");
  const out: { purchase_id: string; dispute_key: string }[] = [];
  for (const it of items) {
    const i = it.indexOf(":");
    const purchase = i < 0 ? "" : it.slice(0, i);
    const key = i < 0 ? "" : it.slice(i + 1);
    if (!/^cpu_[0-9a-f]{20}$/.test(purchase) || key === "" || key.length > 200) return null;
    out.push({ purchase_id: purchase, dispute_key: key });
  }
  return out.length === 0 ? null : out;
}

function wireLever(s: LeverSpec): void {
  const form = document.getElementById(`${s.id}-form`) as HTMLFormElement | null;
  form?.addEventListener("submit", (ev) => {
    ev.preventDefault();
    void (async () => {
      const fd = new FormData(form);
      const v: Record<string, string> = {};
      for (const f of s.fields) v[f.name] = String(fd.get(f.name) ?? "").trim();
      const path = s.path(v);
      if (path === null) return leverMsg(`${s.id}-msg`, s.bad);
      const r = await apiPost<Record<string, unknown>>(path, s.body(v));
      if (!r.ok) return leverMsg(`${s.id}-msg`, failText(r.status, r.data, "That didn't go through — nothing changed"));
      if (r.data?.pending) return leverMsg(`${s.id}-msg`, `Recorded your sign-off (${String(r.data.approvals ?? 1)} of ${String(r.data.required ?? 2)}). Nothing has changed yet — a second, different operator must approve with the same values.`);
      leverMsg(`${s.id}-msg`, s.done(r.data));
    })();
  });
}

 * written under the operator's name). A list drops every row such a done row names, read from GET /ops/audit for
 * `<action>_done`. A done read that fails drops nothing (the rows stay listed, and the list says it could not tell);
 * a done row beyond its first AUDIT_PAGE_MAX is not read, so its row may still be listed (never hidden unread).

async function byHandRead(action: string, title: string, who: string, empty: string): Promise<string> {

  if (!Array.isArray(list)) return `<h3>${esc(title)}</h3><p class="err">${esc(failText(r.status, r.data, `Couldn't load ${title.toLowerCase()}`))} Treat that as unknown, not as none.</p>`;
  const d = await apiGet<Record<string, unknown>>(`/ops/audit?action=${encodeURIComponent(`${action}_done`)}${page}`, ["entries"]);
  const doneRead = d.ok && d.data !== null && Array.isArray(d.data.entries);
  const done = new Set((doneRead ? (d.data!.entries as Record<string, unknown>[]) : []).filter((x) => x.action === `${action}_done`).map((x) => String(x.detail ?? "")));

function openRemoval(kind: "resolve" | "decline", id: string, intent: string, hasPassport = true, filedBy: string | null = null): void {
  // F-A4: whether the request resolved to a passport decides whether resolving
  // suppresses anything, so the confirm has to know it BEFORE the operator
  // commits — the `suppressed` flag in the response arrives after the decision.
  // `filedBy` is the same argument for the `correct` intent: only a signed-in
  // creator's own filing detaches anything, and the operator must be told which
  // kind they are looking at before they press, not after.
  const html = kind === "resolve" ? removalResolveConfirm(id, intent, WHO, ROLE, hasPassport, filedBy) : removalDeclineConfirm(id, intent, WHO, ROLE);
  if (!html) return;
  showModal(html);
  const scope = kind === "resolve" ? "#m-removal-resolve" : "#m-removal-decline";
  const cancel = document.querySelector<HTMLButtonElement>(`${scope} button[data-act="removal-cancel"]`);
  if (cancel) cancel.onclick = () => closeModal();
  const go = document.querySelector<HTMLButtonElement>(`${scope} button[data-act="removal-${kind}-confirm"]`);
  if (go) go.onclick = async () => {
    closeModal();
    let body: { note?: string } = {};
    if (kind === "decline") {
      const note = ((await promptText("Why is this removal declined? A note is required.")) ?? "").trim();
      if (!note) { toast("Declining needs a note — nothing was sent."); return; }
      body = { note };
    }
    const r = await apiPost<{ suppressed?: boolean; detached?: boolean; state?: string }>(`/ops/removals/${encodeURIComponent(id)}/${kind}`, body);
    if (!r.ok) { toast(failText(r.status, r.data, `${kind} failed — the request is still open`)); return; }
    // THE TOAST READS `detached`, NOT ONLY `suppressed`. On a `correct`,
    // `suppressed` is ALWAYS false — so reporting only that flag said "NOTHING
    // was suppressed, the listing is still served" identically whether an
    // account had just been permanently revoked or nothing had happened at all.
    // The success case was reported as a no-op, and the refusal was
    // indistinguishable from it.
    toast(kind === "resolve"
      ? intent === "correct"
        ? (r.data?.detached
            ? "Resolved — the account is off that passport for good and stops being published at the next build. Nobody was suppressed or delisted."
            : `Nothing was detached — the request is ${r.data?.state === "open" ? "still OPEN, because the ledger refused the change" : "closed with the listing untouched"}. Read the case again: only a signed-in creator's own \u201cnot mine\u201d executes, and only against a row Dhwaja listed and nobody proved.`)
        : (r.data?.suppressed ? "Resolved — the subject is suppressed and leaves the registry on the next build." : "Resolved, and NOTHING was suppressed — the listing is still served. If this was a remove request, the person is still listed; suppress their passport directly or re-file against one.")
      : "Declined. The subject stays listed — reply to them yourself, this console emails nobody.");
    void viewRemovals();
  };
}

/**
 * OP-02 (QUEUE #149): THE PLAN THE OPERATOR IS COMPOSING IS WHAT THE CARD SHOWS.
 * A request-origin case arrives with `actions: []`, and `recoveryCaseCard` renders
 * the account/destination inputs only for actions ALREADY in the payload — so
 * ticking `transfer_account` (or `reactivate_account`) changed the plan map and
 * drew nothing, `openRecoveryResolve` found no input to read, and the case could
 * only ever be rejected. The card is now redrawn from a COPY of the case with the
 * composed plan overlaid (the stored case is never edited; an operator-authored
 * plan is fixed at open and is never overlaid).
 */
function withComposedPlan(c: RecoveryCase): RecoveryCase {
  const plan = recoveryPlans.get(c.item_id);
  if (!plan || c.payload?.origin === "operator") return c;
  return { ...c, payload: { ...c.payload, actions: ACTION_ORDER.filter((a) => plan.has(a)) } };
}

function wireRecoveryCards(root: ParentNode): void {
  root.querySelectorAll<HTMLInputElement>('input[data-act="recovery-action"]').forEach((cb) => { cb.onchange = () => {
    const id = cb.dataset.id ?? "";
    const act = (cb.dataset.action ?? "") as RecoveryAction;
    const plan = recoveryPlans.get(id) ?? new Set<RecoveryAction>();
    if (cb.checked) plan.add(act); else plan.delete(act);
    recoveryPlans.set(id, plan);
    redrawRecoveryCard(cb, id);
  }; });
  root.querySelectorAll<HTMLButtonElement>('button[data-act="recovery-resolve-open"]').forEach((b) => { b.onclick = () => openRecoveryResolve(b.dataset.id ?? ""); });
  root.querySelectorAll<HTMLButtonElement>('button[data-act="recovery-reject-open"]').forEach((b) => { b.onclick = () => openRecoveryReject(b.dataset.id ?? ""); });
}

/** Redraw ONE case card in place, keeping whatever the operator already typed
 * into the transfer / re-list inputs (they are index-keyed, `trKeyId(idx)`). */
function redrawRecoveryCard(from: Element, id: string): void {
  const idx = recoveryCases.findIndex((x) => x.item_id === id);
  const c = idx >= 0 ? recoveryCases[idx] : undefined;
  const card = from.closest(".card.detail");
  if (!c || !card || c.payload?.origin === "operator") return;
  const typedKey = (document.getElementById(trKeyId(idx)) as HTMLInputElement | null)?.value;
  const typedDest = (document.getElementById(trDestId(idx)) as HTMLInputElement | null)?.value;
  const tpl = document.createElement("template");
  tpl.innerHTML = recoveryCaseCard(withComposedPlan(c), ROLE, idx);
  const fresh = tpl.content.firstElementChild;
  if (!fresh) return;
  card.replaceWith(fresh);
  const keyEl = document.getElementById(trKeyId(idx)) as HTMLInputElement | null;
  const destEl = document.getElementById(trDestId(idx)) as HTMLInputElement | null;
  if (keyEl && typedKey !== undefined) keyEl.value = typedKey;
  if (destEl && typedDest !== undefined) destEl.value = typedDest;
  wireRecoveryCards(fresh);
}

/** F3: the collectible-return form → POST /ops/collectibles/recover (dual control). */
async function recoverCollectibleFromForm(): Promise<void> {
  const v = (id: string): string => (document.getElementById(id) as HTMLInputElement | HTMLSelectElement | null)?.value.trim() ?? "";
  const kind = v("cr-kind") === "acquisition" ? "acquisition" : "card";
  const item = v("cr-item");
  const to = v("cr-to");
  const ref = v("cr-ref");
  const breakEscrow = kind === "card" && (document.getElementById("cr-break") as HTMLInputElement | null)?.checked === true;
  if (!(kind === "card" ? /^ci_[0-9a-f]{20}$/ : /^acq_[0-9a-f]{20}$/).test(item)) { toast(`The item id must be ${kind === "card" ? "ci_" : "acq_"} followed by 20 hex characters. Nothing was run.`); return; }
  if (!/^acc_[0-9a-f]{20}$/.test(to)) { toast("The destination must be a fan account id (acc_ followed by 20 hex characters). Nothing was run."); return; }
  // OU-09: THE SERVICE HAS ONE REFERENCE CONTRACT AND THIS IS IT.
  // `RECOVERY_REF_RE` (gifting.ts) is `{1,40}` and console.ts refuses anything
  // longer with a 400 — BEFORE the dual-control sign-off is recorded. Accepting
  // 64 here meant the operator composed a reference, read a two-person
  // confirmation naming it, pressed through, and only then learned the action
  // could never have taken it. The hold form beside this one already checks 40.
  if (!/^[A-Za-z0-9_:.\-]{1,40}$/.test(ref)) { toast("A case reference (1–40 of letters, digits, _ : . -) is required — it is written to the item's history. Nothing was run."); return; }
  if (!(await confirmText(`Return ${item} to ${to}?${breakEscrow ? "\n\nIf the card is staked in a waiting war-queue entry, that entry is CANCELLED so the card can move." : ""}\n\n${await signoffClause("moves the item", `the same item, destination, reference (${ref})${breakEscrow ? " and escrow choice" : ""}`)}`, "Return item"))) return;
  const body: Record<string, unknown> = { kind, item_id: item, to_account: to, ref };
  if (breakEscrow) body.break_escrow = true;

  if (!r.ok) { toast(failText(r.status, r.data, "Return failed — nothing moved")); return; }

" Its war-queue entry was cancelled." : ""}` : "Nothing moved: the item is already there, is staked in a match (tick the escrow box for a waiting queue entry), or no longer exists.");
}

/** The plan actually sent: the boxes the operator ticked, or — for an
 * operator-authored case whose plan is fixed at open — the stored one. */
function plannedActions(c: RecoveryCase): RecoveryAction[] {
  const stored = Array.isArray(c.payload?.actions) ? c.payload.actions : [];
  // `withForced` mirrors the server's FORCED_ACTIONS: a hijack always carries
  // the payout retraction and the agency-grant revocation. Sending the same set
  // the confirm displayed matters for dual control — the server folds the forced
  // actions into the signed plan key either way, so a request that omitted them
  // would still execute them, and the operator would have approved a plan that
  // did not describe what ran.
  const chosen = c.payload?.origin === "operator" ? stored : [...(recoveryPlans.get(c.item_id) ?? [])];
  return withForced(c.payload?.kind, chosen);
}

function openRecoveryResolve(id: string): void {
  const idx = recoveryCases.findIndex((x) => x.item_id === id);
  const c = idx >= 0 ? recoveryCases[idx] : undefined;
  if (!c) return;
  const actions = plannedActions(c);
  // THE TRANSFER PLAN, TYPED. A request-origin case carries no transfer fields
  // — `submitRecoveryRequest` deliberately refuses to let a requester name a
  // destination — so both values come off the card's inputs, and both are
  // validated HERE before the confirm is even rendered: a confirm reading
  // "moves `—` to `—`" is not a thing anybody should be signing off on, and the
  // 400 it produced named nothing.
  const typed = {
    contested_account_key: (document.getElementById(trKeyId(idx)) as HTMLInputElement | null)?.value.trim() ?? "",
    to_passport: (document.getElementById(trDestId(idx)) as HTMLInputElement | null)?.value.trim() ?? "",
  };
  const key = typed.contested_account_key || String(c.payload?.contested_account_key ?? "");
  const dest = typed.to_passport || String(c.payload?.to_passport ?? "");
  if (actions.includes("transfer_account")) {
    if (!key) { toast("Name the account being moved — the canonical account key, from the subject's passport page. Nothing was run."); return; }
    if (!/^psp_[0-9a-f]{16}$/.test(dest)) { toast("The destination must be a passport id (psp_ followed by 16 hex characters). Nothing was run."); return; }
  } else if (actions.includes("reactivate_account") && !key) {
    toast("Name the account being re-listed — the canonical account key. Nothing was run.");
    return;
  }
  // QUEUE #146 (D-0807): the fan plan's clear-email option, only when ticked.
  const clearEmail = c.payload?.kind === "fan_takeover" && document.querySelector<HTMLInputElement>(`input[data-act="recovery-clear-email"][data-id="${CSS.escape(c.item_id)}"]`)?.checked === true;
  const html = recoveryResolveConfirm(c, actions, WHO, ROLE, { contested_account_key: key, to_passport: dest, clear_email: clearEmail });
  if (!html) return;
  showModal(html);
  const cancel = document.querySelector<HTMLButtonElement>('#m-recovery-resolve button[data-act="recovery-cancel"]');
  if (cancel) cancel.onclick = () => closeModal();
  const go = document.querySelector<HTMLButtonElement>('#m-recovery-resolve button[data-act="recovery-resolve-confirm"]');
  if (go) go.onclick = async () => {
    closeModal();
    const plan: Record<string, unknown> = { actions };
    if (key) plan.contested_account_key = key;
    if (dest) plan.to_passport = dest;
    if (clearEmail) plan.clear_email = true;
    const r = await apiPost<{ pending?: boolean; approvals?: number; required?: number; actions_executed?: string[] }>(`/ops/recovery/${encodeURIComponent(id)}/resolve`, plan);
    if (!r.ok) { toast(r.status === 409 && !clearEmail ? "The ledger refused a step (409) — a transfer that cannot land, or a re-listing without a fresh control proof. The lockdown steps that ran are idempotent; re-resolve once the cause is fixed." : failText(r.status, r.data, "Resolve failed — nothing was run")); return; }
    if (r.data?.pending) { toast(`Your sign-off is recorded — nothing has run. ${r.data.approvals ?? 0} of ${r.data.required ?? 2} required.`); return; }
    toast(`Ran: ${(r.data?.actions_executed ?? []).join(", ") || "no actions"}.`);
    recoveryPlans.delete(id);
    void viewRecovery();
  };
}

function openRecoveryReject(id: string): void {

  showModal(html);

    toast(r.ok ? "Rejected. Reply to them yourself — this console sends no email." : failText(r.status, r.data, "Reject failed — the case is still open"));
    if (r.ok) void viewRecovery();
  };
}

    const held = heldText(r.ok, r.data);
    if (held) { toast(held); return; }

    // suppression left the dispute row `open`, so the only lever this screen
    // offers for draining the queue drained nothing, and an operator working it

// --- Media takedown (the CSAM lever) ----------------------------------------
let takedownHash = "";
let takedownStatus: TakedownStatus | null = null;
let takedownLast = "";
/** The URL the operator must purge by hand when the edge purge failed after the
 * origin removal succeeded (F-A1). Carried in the 502 body. */
let takedownPurgeUrl = "";
/** D-1064(7): the last `?hash=` link taken, so a redraw on the same link keeps what the operator typed since. */
let takedownLinked = "";

function viewTakedown(): void {
  // D-1064(7): a row of the screen-failed list links here with its hash.
  const linked = takedownHashFromRoute(location.hash);
  if (linked !== null && linked !== takedownLinked) { takedownLinked = linked; takedownHash = linked; takedownStatus = null; takedownPurgeUrl = ""; }
  app.innerHTML = takedownScreen(ROLE, takedownHash, takedownStatus, takedownLast, takedownPurgeUrl);
  const input = app.querySelector<HTMLInputElement>("#tdhash");
  // OU-10 — UPDATE WHAT DEPENDS ON THE HASH, NOT THE FIELD HOLDING IT.
  //
  // This used to call `viewTakedown()` on every input event, and that function's
  // first act is `app.innerHTML = …`. So each typed character, each deletion and
  // each paste destroyed the focused input and built a new one: an operator
  // correcting two characters of a 64-character hash had to reacquire the field
  // between them. The REASON for re-rendering was real and is preserved exactly
  // — the lever must enable only on a well-formed hash — but it needs three
  // nodes updated, not a whole screen rebuilt.
  if (input) input.oninput = () => {
    takedownHash = input.value;
    takedownStatus = null;
    takedownPurgeUrl = "";
    const note = document.getElementById("tdnote");
    if (note) note.textContent = takedownNote(takedownHash);
    const btn = app.querySelector<HTMLButtonElement>('button[data-act="takedown-open"]');
    // The ATTRIBUTE, not a class: `writeControl` emits a real `disabled`, and a
    // control that only LOOKS dead is a live control. `openTakedown` re-checks
    // the hash itself, so the two agree in both directions.
    if (btn) btn.disabled = !looksLikeContentHash(takedownHash);
    const res = document.getElementById("tdresultbox");
    if (res) res.innerHTML = "";
  };
  const open = app.querySelector<HTMLButtonElement>('button[data-act="takedown-open"]');
  if (open) open.onclick = () => openTakedown(takedownHash.trim());
  const retry = app.querySelector<HTMLButtonElement>('button[data-act="takedown-retry"]');
  if (retry) retry.onclick = () => { takedownStatus = null; takedownPurgeUrl = ""; viewTakedown(); };
}

function openTakedown(hash: string): void {
  if (!looksLikeContentHash(hash)) return;
  const html = takedownConfirm(hash, WHO, ROLE);
  if (!html) return; // support seat: the confirm does not exist in the document
  showModal(html);
  const cancel = document.querySelector<HTMLButtonElement>('#m-takedown button[data-act="takedown-cancel"]');
  if (cancel) cancel.onclick = () => closeModal();
  const go = document.querySelector<HTMLButtonElement>('#m-takedown button[data-act="takedown-confirm"]');
  if (go) go.onclick = async () => {
    const typed = document.querySelector<HTMLInputElement>("#tdconfirm")?.value ?? "";
    const msg = document.querySelector<HTMLElement>("#tdmsg");
    if (!typedConfirmMatches(typed, hash)) { if (msg) msg.textContent = "That is not the same hash. Nothing has been taken down."; return; }
    closeModal();
    const r = await apiDelete(`/ops/media/${encodeURIComponent(hash)}`);
    takedownLast = hash;
    // F-A1: `removed: true` with `edge_purged: false` is the DESTRUCTION-DONE,
    // EDGE-STILL-SERVING state. It is read off the body rather than inferred
    // from the status, because 502 alone cannot distinguish it from a takedown
    // that never started — and reading it as the latter is what produced
    // "Nothing was taken down" over a permanent removal.
    const body = (r.data ?? {}) as { removed?: unknown; edge_purged?: unknown; purge_url?: unknown };
    const partial = body.removed === true && body.edge_purged === false;
    takedownPurgeUrl = partial && typeof body.purge_url === "string" ? body.purge_url : "";
    takedownStatus = r.ok ? "done" : partial ? "edge-unpurged" : r.status === 503 ? "unavailable" : r.status === 404 ? "missing" : "error";
    viewTakedown();
  };
}

/** The concentration note beside a row's counts. Never a verdict — the words

}

/**
 * THE ATTENTION STRIP. Rows appear only when there is something to say. A quiet
 * console is the normal console: on a healthy deployment this is one line.
 */
export function attentionStrip(input: AttentionInput): string {
  const rows: string[] = [integrityAttentionRow(input.integrity, input.checkedLabel)];

  if (input.coolingOffElapsed === null) {
    rows.push(`<div class="arow"><span class="ic amb">!</span><b>Warnings pipeline could not be read</b>
      <span class="sub">cooling-off state unknown — check the warnings screen directly</span>
      <span class="right"><a href="#/warnings">Warnings</a></span></div>`);
  } else if (input.coolingOffElapsed > 0) {
    rows.push(`<div class="arow"><span class="ic amb">!</span>
      <b>${esc(String(input.coolingOffElapsed))} drafted warning${input.coolingOffElapsed === 1 ? " is" : "s are"} past the cooling-off window</b>
      <span class="sub">past the clock is not the same as ready — second look and an approved wording still gate publication</span>
      <span class="right"><a href="#/warnings">Warnings</a></span></div>`);
  }

  // The registry rebuild. AMBER, not red: red on this screen means the hash
  // chain is broken and nothing else, and a strip that shouts twice is a strip
  // that is read once. The row still names the CONSEQUENCE rather than the
  // symptom, because "the manifest is old" is not what an operator has to act
  // on — "the delisting you actioned is not published" is.
  if (input.registry === null) {
    rows.push(`<div class="arow"><span class="ic amb">!</span><b>Registry build state could not be read</b>
      <span class="sub">freshness unknown — check the registry health tab directly</span>
      <span class="right"><a href="#/agencies?tab=registry">Registry health</a></span></div>`);
  } else if (input.registry.stale) {
    const age = input.registry.age_hours === null ? "age unknown" : `${input.registry.age_hours}h old`;
    rows.push(`<div class="arow"><span class="ic amb">!</span>
      <b>The registry has stopped rebuilding</b>
      <span class="sub">last published build: ${esc(age)} — removal requests are not reaching the served buckets; check <code>registry_decoy_seed</code> on Config health and the cron logs</span>
      <span class="right"><a href="#/agencies?tab=registry">Registry health</a></span></div>`);
  }

  if (input.pendingApprovals === null) {
    rows.push(`<div class="arow"><span class="ic amb">!</span><b>Dual-control queue could not be read</b>
      <span class="sub">open Warnings and check for actions awaiting a second signature</span>
      <span class="right"><a href="#/warnings">Warnings</a></span></div>`);
  } else if (input.pendingApprovals > 0) {
    rows.push(`<div class="arow"><span class="ic inf">◎</span>
      <b>${esc(String(input.pendingApprovals))} action${input.pendingApprovals === 1 ? " is" : "s are"} awaiting dual-control sign-off</b>
      <span class="sub">a second, different operator must sign the same action — re-signing yourself does nothing</span>
      <span class="right"><a href="#/warnings">Open</a></span></div>`);
  }

  // The strip goes loud on exactly the condition the row draws in red — read
  // from the SAME resolver, so a `broken` state can never arrive with a red row
  // inside a hushed frame.
  const loud = integrityState(input.integrity) === "broken";
  return `<div class="attn${loud ? " loud" : ""}">${rows.join("")}</div>`;
}

import { esc, fmtTs } from "./html.js";

/**
 * THE OPERATOR'S DISCOVERY-METADATA OVERRIDE — the correction path for a
 * miscategorised seeded listing.
 *
 * `POST /ops/creators/:id/profile-meta` has been shipped, audited and
 * Access-gated with no caller since S80. The creator's own form exists at
 * `#/profile` in the creator console; the operator's did not, so the only way to
 * fix a listing we categorised wrongly was a hand-built `curl` — for a listing
 * whose subject never asked to be listed in the first place.
 *
 * ============================================================================
 * THREE TRAPS, AND TWO OF THEM WOULD HAVE MADE THIS CONSOLE LIE
 * ============================================================================
 *
 * **1. The public read is not the stored row.** `getProfileMeta` MASKS a
 * consent-gated category on an unclaimed listing (a stored `adult` reads back as
 * `null`) and drops a suppressed passport entirely. Populating this form from
 * `GET /p/:id/profile-meta` would therefore show "no category" over a stored
 * one — and the save below would clear it. The dossier read carries
 * `profile_meta` from `storedProfileMeta` instead, which is the row as stored
 * plus `withheld_from_public`, the answer to "why is their public page blank?".
 *
 * **2. The write is an UPSERT where an omitted field CLEARS.** `setProfileMeta`
 * treats every field as optional and a null/empty one as "clear it". A form that
 * posted only the field the operator touched would silently wipe the other
 * three. `profileMetaPayload` below always emits all four, seeded from what was
 * rendered — pinned by test, because this is invisible until it has happened.
 *
 * **3. The consent gate refuses `adult` on an unclaimed listing.**
 * `CONSENT_REQUIRED_CATEGORIES` is `{adult}` and `setProfileMeta` throws for it
 * while `claim_state === "unclaimed"` — "the creator may set it themselves after
 * claiming". The screen says so BEFORE the operator picks it rather than
 * rendering a raw 400 afterwards. The server check is untouched and stays the
 * boundary; this is a label, not an enforcement.
 *
 * **…and the case the first draft got wrong.** A gated category can ALREADY be
 * stored on an unclaimed listing — `correction / downgrade_claim` returns a
 * claimed passport to `unclaimed` and leaves `passport_profile` alone, which is
 * the scenario `getProfileMeta`'s read-time gate exists for. Rendering that
 * stored value as a SELECTED-but-disabled option made the form unsaveable: the
 * select still reports `adult`, every save posts it, and the server throws the
 * exact 400 this module claims to prevent — blocking the correction path on
 * precisely the listings it exists to correct. So a gated stored value is shown
 * as the withheld line and the select falls back to "none", and the copy says
 * plainly that saving from this form clears it. That is a real consequence and
 * it is stated rather than discovered.
 *
 * ============================================================================
 * A FAILED READ IS NEVER A SAVEABLE FORM
 * ============================================================================
 *
 * The fourth way this could lie, and the one that survived the first draft: with
 * `GET /taxonomy` down, every select renders EMPTY — the stored value is not in
 * the option list, so nothing is selected and every category reads as "none". An
 * operator who came to fix the region types it, saves, and the upsert clears the
 * creator's own categories. The screen warned and left the button live.
 *
 * So the save control is DISABLED whenever the form cannot faithfully represent
 * what is stored: no vocabulary, or no `profile_meta` on the dossier at all (an
 * older worker). Every other surface in this slice refuses to render a failed
 * read as data; this one refuses to let you commit it.
 */

export interface StoredProfileMeta {
  primary_category: string | null;
  secondary: string[];
  language: string | null;
  region: string | null;
  updated_by: string | null;
  updated_at: string | null;
  /** Stored categories the PUBLIC read withholds right now (consent-by-claim). */
  withheld_from_public?: string[];
  /** The OTHER reason the public read is empty: this subject is delisted, so the
   * whole row is withheld whatever the claim state. Independent of the array. */
  public_read_suppressed?: boolean;
}

export interface TaxonomyVocabulary {
  version: string;
  categories: { key: string; label: string }[];
  max_secondary: number;
}

/** The categories the server refuses to publish for an UNCLAIMED listing. Named
 * here to LABEL the option, never to decide: `setProfileMeta` throws either way
 * and remains the boundary. Kept in one place so the label and the sentence
 * under the form cannot drift apart. */
const CONSENT_GATED = new Set<string>(["adult"]);

const LANG_RE = /^[a-z]{2}$/;
const REGION_RE = /^[A-Z]{2}$/;

/**
 * The payload, ALWAYS four fields.
 *
 * `secondary` is capped at the vocabulary's `max_secondary` client-side too —
 * not because the server needs it (it validates) but because a form that let an
 * operator pick five and then 400'd would be a form that wasted their time
 * without saying why. An empty string becomes `null`, which is how this API
 * spells "clear it".
 */
export function profileMetaPayload(v: { primary: string; secondary: string[]; language: string; region: string }, maxSecondary: number): {
  primary_category: string | null;
  secondary: string[];
  language: string | null;
  region: string | null;
} {
  const clean = (s: string): string | null => (s.trim() === "" ? null : s.trim());
  return {
    primary_category: clean(v.primary),
    // De-duped, primary removed, capped. NOT "the same rules the server
    // applies" — the server THROWS on a secondary repeating the primary and on
    // a duplicate (`validate()` in taxonomy.ts), so normalising here where it
    // refuses would silently drop a selection and show the operator a reload
    // with no explanation. The SELECTS are what prevent it: `profileMetaBlock`
    // omits the chosen primary from every secondary list, so the collision is
    // not expressible in the UI. This stays as the last line of defence for a
    // stale DOM (the primary changed after the secondaries were rendered), and
    // that is the only case it fires in.
    secondary: [...new Set(v.secondary.filter((c) => c !== "" && c !== v.primary))].slice(0, Math.max(0, maxSecondary)),
    language: clean(v.language),
    region: clean(v.region),
  };
}

/** Client-side shape check, so a typo is refused where it was typed rather than
 * as a server 400 with no field named. The server's own regexes are the
 * boundary; these are the same two, restated to say WHICH field is wrong. */
export function profileMetaProblem(language: string, region: string): string | null {
  if (language.trim() !== "" && !LANG_RE.test(language.trim())) return "Language must be a two-letter ISO 639-1 code in lower case — en, ja, pt. Nothing was saved.";
  if (region.trim() !== "" && !REGION_RE.test(region.trim())) return "Region must be a two-letter ISO 3166-1 code in UPPER case — US, JP, BR. Nothing was saved.";
  return null;
}

const sel = `padding:6px 10px;background:var(--panel2);color:var(--ink);border:1px solid var(--line);border-radius:7px`;

/**
 * One category select.
 *
 * A GATED option on an unclaimed listing is drawn disabled, with the reason in
 * its own label: hiding it would leave an operator hunting for a category they
 * know exists, and offering it live would produce a 400 the screen then has to
 * explain after the fact.
 *
 * IT IS NEVER SELECTED AND DISABLED AT THE SAME TIME. A browser keeps an
 * explicitly-selected option selected however it is drawn, so a stored `adult`
 * rendered that way still posts `adult` on every save and the server throws —
 * which made the whole form unusable for the one listing shape the withheld line
 * is about. When the stored value is gated, the select falls back to "none" and
 * the block's copy says what saving will do to it.
 *
 * `exclude` drops the chosen primary from the secondary lists, because
 * `validate()` THROWS on a secondary repeating the primary. Preventing the
 * collision in the control is the only way to keep the payload's normalisation
 * from silently discarding a selection the operator made.
 */
function categoryOptions(vocab: TaxonomyVocabulary | null, current: string | null, unclaimed: boolean, blankLabel: string, exclude = ""): string {
  const cats = (vocab?.categories ?? []).filter((c) => c.key !== exclude || c.key === "");
  const selectable = (key: string): boolean => !(unclaimed && CONSENT_GATED.has(key));
  const chosen = current !== null && current !== "" && selectable(current) ? current : "";
  const opts = cats.map((c) => {
    const gated = !selectable(c.key);
    return `<option value="${esc(c.key)}"${chosen === c.key ? " selected" : ""}${gated ? " disabled" : ""}>${esc(c.label)}${gated ? " — the creator must set this themselves after claiming" : ""}</option>`;
  }).join("");
  return `<option value=""${chosen === "" ? " selected" : ""}>${esc(blankLabel)}</option>${opts}`;
}

/**
 * The block on the creator dossier. Renders the STORED row, says what the public
 * currently sees when the two differ, and offers the override.
 */
export function profileMetaBlock(meta: StoredProfileMeta | undefined, vocab: TaxonomyVocabulary | null, claimState: string, role: OpsRole): string {
  // THE TWO STATES IN WHICH THIS FORM CANNOT TELL THE TRUTH, and therefore
  // cannot be saved from: the dossier did not carry the stored row (an older
  // worker), or the vocabulary read failed so every select is empty and reads as
  // "none". Either way a save would post blanks the operator never chose into an
  // upsert that treats a blank as "clear it".
  const unreadable = meta === undefined || vocab === null;
  const m: StoredProfileMeta = meta ?? { primary_category: null, secondary: [], language: null, region: null, updated_by: null, updated_at: null, withheld_from_public: [], public_read_suppressed: false };
  const unclaimed = claimState !== "claimed" && claimState !== "verified";
  const withheld = Array.isArray(m.withheld_from_public) ? m.withheld_from_public : [];
  // WHO SET THIS is the question an override is actually about. `creator` means
  // the subject chose it and an operator overriding it is overruling them; an
  // `op_` id means we did.
  // "NOTHING HAS BEEN SET" IS AN ASSERTION, and an absent field is not evidence
  // for it. An older worker sends no `profile_meta` at all, which is a different
  // fact from an empty row and must not be reported as one.
  const provenance = meta === undefined
    ? `<p class="err" style="margin:0 0 8px">This Dhwaja build did not return the stored metadata, so what is set for this listing is <strong>unknown</strong> — not empty. Saving is disabled rather than risk clearing something you were never shown.</p>`
    : m.updated_at === null
      ? `<p class="muted" style="margin:0 0 8px">Nothing has been set for this listing.</p>`
      : `<p class="muted" style="margin:0 0 8px">Last set by <code class="mono">${esc(m.updated_by ?? "—")}</code> on ${esc(fmtTs(m.updated_at))}${m.updated_by === "creator" ? " — <strong>the creator set this themselves</strong>. Overriding it overrules them; prefer asking." : ""}</p>`;
  const withheldLine = withheld.length > 0
    ? `<p class="err" style="margin:0 0 8px">Stored, but <strong>not published</strong>: ${esc(withheld.join(", "))}. An unclaimed listing publishes no consent-gated category however it got there, so their public page shows nothing for it — it becomes visible if they claim the listing. <strong>Saving from this form clears it</strong>: the server refuses to store a gated category for an unclaimed listing, so there is no way to leave it in place from here, and it goes from the row and not just from the public page.</p>`
    : "";
  // THE OTHER REASON THE PUBLIC PAGE IS BLANK, and it is not the one above. A
  // delisted subject's whole row is withheld from every served read whatever the
  // claim state, so an operator asking "why is their page empty?" about a
  // claimed, delisted creator would otherwise be told nothing is withheld.
  const suppressedLine = m.public_read_suppressed
    ? `<p class="err" style="margin:0 0 8px">This subject is <strong>delisted</strong>, so none of this is published at all — the public read withholds the whole row whatever it holds. That is separate from the consent gate, and it is undone by re-listing them, not by changing anything here.</p>`

/** Validation before the gate is touched, mirroring the recovery card. A confirm

/**
 * RECOVERY ADJUDICATION — the operator half of "I have lost my passport".
 *
 * A creator whose passkey is gone, whose passport was hijacked, or whose account
 * is being claimed by someone else files a self-service request; it becomes a
 * `dispute` case in the review queue. The intake was wired and the adjudication
 * was not, so those cases arrived somewhere no screen could reach. Nobody could
 * answer, and the person on the other end could not tell "we are considering it"
 * from "nobody read it".
 *
 * THE ONE THING THIS SCREEN MUST GET RIGHT is that a REQUEST carries no plan.
 * The API takes the remediation at resolve time, and the operator supplies it —
 * which means the operator is choosing, right here, whether to bump a session
 * epoch, wipe someone's passkeys, or MOVE AN ACCOUNT FROM ONE PASSPORT TO
 * ANOTHER. That last one writes a ledger event and is the only action on this
 * screen that changes what fans are told about who owns a handle. So:
 *
 *  - the three actions are separate, labelled, and OFF by default;
 *  - `transfer_account` names both passports on screen before you can pick it,
 *    because a transfer to the wrong destination is an impersonation Dhwaja
 *    performed itself;
 *  - the confirm restates the exact plan, not a summary of it.
 *
 * ORIGIN IS RENDERED, ALWAYS. An operator-authored case has its plan fixed at
 * open time; a request-origin case does not, and its dual-control key folds the
 * plan in for exactly that reason. Those are different objects and the screen
 * says which one you are looking at rather than letting them blur.
 */

export type RecoveryAction = "force_logout" | "reset_passkeys" | "retract_payout_tags" | "revoke_connect_binding" | "revoke_agency_grants" | "evict_contact_channels" | "transfer_account" | "reactivate_account" | "revoke_fan_sign_in";
/** Every action except the fan-account one: the server's EXEC_ORDER / ACTIONS set. */
export type PassportRecoveryAction = Exclude<RecoveryAction, "revoke_fan_sign_in">;

/** The order the SERVER runs them in (recovery.ts EXEC_ORDER), so the screen and
 * the confirm never imply an order the executor does not use. */
export const ACTION_ORDER: PassportRecoveryAction[] = ["force_logout", "reset_passkeys", "retract_payout_tags", "revoke_connect_binding", "revoke_agency_grants", "evict_contact_channels", "transfer_account", "reactivate_account"];

/**
 * ACTIONS THE SERVER FORCES, per case kind (recovery.ts FORCED_ACTIONS). These
 * are not the operator's to leave out: after a confirmed hijack, no authority
 * the attacker created may remain able to move money or act for the creator.
 * Rendered CHECKED and DISABLED with the reason, so the screen tells the truth
 * about what will run rather than showing a box that does nothing.
 */
export const FORCED_BY_KIND: Record<string, RecoveryAction[]> = {
  hijack: ["force_logout", "reset_passkeys", "retract_payout_tags", "revoke_connect_binding", "revoke_agency_grants", "evict_contact_channels"],
  // QUEUE #146 / D-0793: a fan takeover case has exactly one plan (recovery.ts
  // validateFanPlan), so it is shown as the one fixed box and sent every time.
  fan_takeover: ["revoke_fan_sign_in"],
};

/** The boxes a case offers. A fan takeover case offers ONLY the fan revocation;
 * a passport case never offers it (the server refuses it on either mismatch). */
export function actionsFor(kind: string | undefined): RecoveryAction[] {
  return kind === "fan_takeover" ? ["revoke_fan_sign_in"] : [...ACTION_ORDER];
}

/** The plan that will actually execute for this case: what was chosen, plus
 * whatever the kind forces. Exported because main.ts must send the same set the
 * confirm displays — a confirm that understates the plan is worse than none. */
export function withForced(kind: string | undefined, chosen: readonly RecoveryAction[]): RecoveryAction[] {
  const forced = FORCED_BY_KIND[kind ?? ""] ?? [];
  const all = new Set<RecoveryAction>([...chosen, ...forced]);
  return actionsFor(kind).filter((a) => all.has(a));
}

import { esc, escClamp, fmtTs } from "./html.js";
import { seatBanner, writeControl, type OpsRole } from "./role.js";

/**
 * THE REMOVAL / CORRECT / CONTEST QUEUE — the operator half of self-delisting.
 *
 * `PRIVACY.md` §7.3 and `LISTING-POLICY.md` §5.1 promise a person can get
 * themselves off Dhwaja. The intake half of that promise has been wired for
 * three milestones; the resolution half had no screen, so every `correct` and
 * `contest`, and every `remove` the fast path deliberately refuses to automate,
 * landed in a queue nobody could open. A promise whose fulfilment is a route
 * with no button is not a promise.
 *
 * The screen is built around three distinctions the server already makes and
 * which an operator must not have to re-derive:
 *
 * 1. THE DEFAULT VIEW INCLUDES `auto_suppressed`, and it says why. The fast path
 *    takes a listing down with no human in the loop; if those were invisible
 *    unless asked for, a self-delisting really would happen unseen. They are
 *    shown as ALREADY DONE, not as work — the row carries no disposition
 *    control, because there is nothing left to decide.
 *
 * 2. INTENT IS THE FIRST COLUMN. `remove` ends a listing; `correct` and
 *    `contest` do not. Resolving all three looks identical in the API and means
 *    three different things to the person who filed it, so intent is rendered
 *    before anything else and the disposition copy changes with it.
 *
 * 3. RESOLVE IS NOT "APPROVE". On a `remove` intent it writes the suppression in
 *    the same batch as the disposition — the subject stops being served. On the
 *    other two it records a decision and nothing public changes. The confirm
 *    says which of those you are about to do, in those words.
 *
 * DECLINE IS NOT DRAWN AS THE SAFE OPTION. It is the one that leaves a person
 * listed against their wishes, so it carries its own confirm and asks for the
 * reason to be in the audit trail, rather than being the quiet button next to
 * the loud one.
 */

export interface RemovalRow {
  request_id: string;
  passport_id: string | null;
  account_key: string;
  intent: string;
  path: string;
  /** 'creator' = a signed-in creator's own "not mine" on their own passport;
   * null = the public unauthenticated path (or a pre-0042 row). This is the
   * difference between evidence and an anonymous note, and it is what the server
   * gates the detach on — so the queue must show it. */
  filed_by?: string | null;
  state: string;
  contact_email: string | null;
  detail: string | null;
  created_at: string;
  updated_at: string;
  resolved_by: string | null;
}

/** The states the queue filter offers. `""` is the server's DEFAULT view —
 * open + auto_suppressed — and is deliberately first. */
export const REMOVAL_VIEWS: readonly string[] = ["", "open", "auto_suppressed", "resolved", "declined"];

/**
 * What resolving THIS request actually does, in the operator's words. The
 * server's behaviour differs by intent AND by whether the request resolved to a
 * passport, and the copy must differ with it.
 *
 * UI-AUDIT F-A4 — THE SECOND ARGUMENT, and the delisting this screen used to
 * promise that the server does not perform.
 *
 * `removal.ts resolveRemoval` decides:
 *
 *     const suppress = disposition === "resolved"
 *                   && req.intent === "remove"
 *                   && req.passport_id !== null;      // ← this clause
 *
 * `request_suppressions` is keyed by passport, so a `remove` request that never
 * resolved to one — the row this very screen renders as "no passport resolved —
 * an account key only" — is marked `resolved` and SUPPRESSES NOTHING. The
 * confirm said "the subject stops being served" and "the subject leaves the
 * published registry on the next build" over exactly that request.
 *
 * What follows is the part that makes it more than wrong copy: the request then
 * sits in the queue as `resolved`, which is the state that means handled. Nobody
 * opens it again. A person who asked to be delisted is recorded as having been
 * delisted, and is still listed — and the only trace of the difference is a
 * `suppressed: false` field the toast reads AFTER the decision is already made.
 *
 * So the meaning is computed from both facts, and the no-passport case says what
 * actually happens and what to do instead of pressing.
 */
export function resolveMeaning(intent: string, hasPassport = true, filedBy: string | null = null): string {
  if (intent === "remove" && !hasPassport) {
    return "the request is marked resolved and NOTHING IS SUPPRESSED — suppression is keyed by passport and this request never resolved to one, so the subject keeps being served exactly as it is. Resolving this closes it as handled while the person stays listed";
  }
  if (intent === "remove") return "the subject stops being served — the suppression is written in the same batch as the disposition, and this is the same terminal the automatic path reaches";
  // A `correct` NOW EXECUTES, and this sentence had to move with it. Until the
  // welcome-screen slice a `correct` was pure bookkeeping — the copy said so, and
  // said to go make the correction by hand somewhere else. `resolveRemoval` now
  // appends an operator `account_revoked` for the account named on the request,
  // in the same disposition, whenever that account is still active on the
  // resolved passport. Leaving the old sentence in place would have been the
  // F-A4 defect inverted: copy promising LESS than the button does, on the one
  // control that takes an account off a creator's passport.
  if (intent === "correct" && !hasPassport) {
    return "the request is marked accepted and NOTHING IS DETACHED — a correction executes against a passport and this request never resolved to one. Resolving closes it as handled while the listing stays exactly as it is";
  }
  if (intent === "correct" && filedBy !== "creator") {
    return "the request is marked accepted and NOTHING IS DETACHED — this one was filed on the PUBLIC path, by somebody who proved nothing and holds no session on that passport, so the server refuses to execute it. Only a signed-in creator's own \u201cnot mine\u201d detaches anything. If you believe the listing IS wrong, act on it yourself rather than through this button";
  }
  if (intent === "correct") {
    return "the account named on this request comes OFF that passport, permanently. An `account_revoked` is appended as you, the row stops being published at the next registry build, and the handle goes back to being reportable and nominatable. Nothing is suppressed — this delists nobody. There is no undo: `account_added` refuses a key that already exists at any status and `account_transferred` preserves status, so nothing in the vocabulary puts this handle back on any passport";
  }
  if (intent === "contest") return "the contest is recorded as upheld. Nothing is suppressed by this alone; if the right outcome is a delisting, the person needs a `remove` request or an operator suppression";
  return "the request is marked resolved. Read the detail before pressing — this intent is not one this console has copy for";
}

/** Mirrors the server's WIRED_TRIGGERS — the ones that actually fire. The

/** Mirrors the server's `PLATFORM_PER_CREATOR_SHARE` (reward-policy.ts): a
 * platform-wide policy grants at most this share of `per_day` per triggering
 * creator per day. Shown, never sent — the server derives it. */
export const PLATFORM_PER_CREATOR_SHARE = 10;
export function perCreatorPerDay(perDay: number): number {

/**
 * THE SUPPORT SEAT, rendered.
 *
 * The operator console has two privilege tiers. `full` is unrestricted;
 * `support` may perform exactly three low-risk writes — file a report, run an
 * import dry-run, and preview a reactivation (a read despite the verb; the key
 * rides in the body) — and is 403'd by the server on everything else. That
 * boundary is enforced in `console.ts` (`OPS_ROLE_POLICY`) — this module is
 * only about not lying to the person sitting in the seat. AS-099/D-50: this
 * comment and `SUPPORT_WRITE_EXCEPTIONS` below must list the same three
 * routes as that table, or the copy below drifts back into a blanket claim.
 *
 * The rule every screen follows: a support seat SEES everything (reading is not
 * the dangerous half) and every control it cannot use is drawn inert, labelled
 * with why, and never silently missing. A missing button teaches an operator
 * that the feature does not exist; an inert one teaches them who to hand the
 * work to.
 *
 * Write controls are marked `full-only` in the class list. That single marker is
 * what the click interceptor in main.ts keys on and what the role-gating tests
 * assert, so a new write control that forgets it fails a test rather than
 * shipping a live button to a seat the server will refuse.

export const SUPPORT_WRITE_EXCEPTIONS = "filing a report, an import dry run, and a reactivation preview";

 * so the query itself is scoped."*
 *
 * The console holds `seeded` and it holds `live` for both tiers, so that
 * scoreboard is one division away from existing again on the client — and a
 * blank column is exactly the thing a future reader "fixes". So the rule is
 * written here as code rather than as an intention: **this screen renders the
 * numbers the server sent and derives no rate the server withheld.** The
 * coverage tier gets a sentence saying the number is absent on purpose, which is
 * the only thing that stops somebody adding it back.
 *
 * A coverage seed is a person who never asked to be listed. That is also why
 * nothing on this screen is a leaderboard, and why the roster's only row action
 * is navigation to the dossier.
 */

export interface TierMetrics {
  seeded: number;
  live: number;
  requests: number;
  requests_per_seed: number;
  suppressed: number;
  /** CONVERSION TIER ONLY — absent on coverage, deliberately. Never derived. */
  claimed?: number;
  claims_per_seed?: number;
}

export interface SeedingMetrics {
  generated_at: string;
  total_seeded: number;

 * "the server's own SEED_LIST_MAX"; it is not, and a page size that the server

/**
 * THE FILTER VOCABULARIES ARE DERIVED FROM THE ANSWER, NOT RESTATED HERE.
 *
 * `seedingList` refuses an out-of-vocabulary tier or source by returning NOTHING
 * rather than ignoring the filter, so a client-side copy of `SEED_SOURCES` that
 * drifted from `@dhwaja/schema` would produce a filter that silently matches
 * zero rows — and this console has no build-time dependency on that package to
 * keep the two honest. (The first draft of this file guessed the list and got
 * four of five values wrong, which is the whole argument.)
 *
 * So the options come out of `GET /ops/seeding/metrics`, which reports the
 * sources actually present (`by_source`) and the tiers it computes. A source
 * with no rows is not offered, which costs nothing: filtering by it would return
 * an empty table anyway.
 */
export function seedFilterOptions(m: SeedingMetrics | null): { tiers: string[]; sources: string[] } {
  return {
    tiers: m?.tiers ? Object.keys(m.tiers) : [],
    sources: Array.isArray(m?.by_source) ? m.by_source.map((s) => s.source).filter((x) => typeof x === "string" && x.length > 0) : [],
  };
}

const n = (v: unknown): string => (typeof v === "number" && Number.isFinite(v) ? String(v) : "—");

/**

/** The metrics summary. A failed read says so — it never renders zeros, which
 * would read as "we have seeded nothing" over a connection problem. */
export function seedingMetricsCard(m: SeedingMetrics | null): string {
  if (!m) return `<p class="err">Couldn't load the seeding metrics — this is a read failure, not a count of zero. The register below is a separate read and may still work.</p>`;
  // A 200 WITH THE WRONG SHAPE IS NOT AN ANSWER. `api()` collapses every
  // non-2xx to null, so a malformed 200 is the one failure that reaches here
  // typed as success — and `.map` on a missing array throws before
  // `mbox.innerHTML` is assigned, leaving the whole Imports screen on "Loading…".
  const sources = (Array.isArray(m.by_source) ? m.by_source : [])
    .map((s) => `<tr><td><code>${esc(s.source)}</code></td><td class="num mono">${n(s.seeded)}</td></tr>`)
    .join("");
  return `<div class="row" style="align-items:flex-start;gap:12px;flex-wrap:wrap">
    ${tierCard("Coverage tier", m.tiers?.coverage, true)}
    ${tierCard("Conversion tier", m.tiers?.conversion, false)}
    <div class="card" style="flex:1;min-width:220px">
      <div class="row"><strong>By source</strong></div>
      <table class="ds-table" style="margin-top:8px"><tbody>${sources || `<tr><td class="muted">Nothing seeded yet.</td></tr>`}</tbody></table>
      <p class="muted" style="margin:8px 0 0">${n(m.total_seeded)} seeded in total · ${n(m.suppressed_total)} delisted · read ${esc(fmtTs(m.generated_at))}</p>
    </div>
  </div>`;
}

/**
 * One option list, with the current value selected.
 *
 * A CURRENT VALUE MISSING FROM THE LIST IS STILL RENDERED, and that is the whole
 * point of this function rather than an inline map. The options are derived from
 * the metrics answer, so a metrics read that fails leaves the list EMPTY — and
 * with nothing carrying `selected`, a browser shows the first option instead.
 * The control would then read "any tier" while `loadSeedRoster` kept sending
 * `tier=coverage` from its own state: a filtered register under a control that
 * says it is unfiltered, which is how an operator concludes a partial list is
 * the whole one.
 */
function options(values: readonly string[], current: string, anyLabel: string): string {
  const known = current === "" || values.includes(current);
  return `<option value=""${current === "" ? " selected" : ""}>${esc(anyLabel)}</option>` +
    (known ? "" : `<option value="${esc(current)}" selected>${esc(current)}</option>`) +
    values.map((v) => `<option value="${esc(v)}"${current === v ? " selected" : ""}>${esc(v)}</option>`).join("");
}

/** The filter row. `batch` is a free field rather than a select because batch

  const { tiers, sources } = seedFilterOptions(m);
  return `<div class="row" style="margin:0 0 10px;flex-wrap:wrap">
    <select id="stier" style="${sel}">${options(tiers, f.tier, "any tier")}</select>
    <select id="ssource" style="${sel}">${options(sources, f.source, "any source")}</select>
    <input id="sbatch" placeholder="batch id" value="${esc(f.batch)}" style="${sel};min-width:200px">
    <button class="ds-btn" data-act="seed-filter">Filter</button>
    <button class="ds-btn" data-act="seed-clear">Clear</button>
  </div>`;
}

/**
 * The roster. Rows are navigation only — the same discipline the abuse screen's
 * signal half states — because a seeded listing is a person who has not asked
 * for anything and there is no verdict to render about them here.
 */
export function seedingRoster(list: SeedList | null, f: SeedFilters): string {
  if (!list) return `<p class="err">Couldn't load the seeded-listing register.</p>`;
  const filtered = f.tier !== "" || f.source !== "" || f.batch !== "";
  const count = Array.isArray(list.rows) ? list.rows.length : 0;
  const offset = Number.isFinite(list.offset) ? list.offset : 0;
  const limit = Number.isFinite(list.limit) && list.limit > 0 ? list.limit : SEED_PAGE;
  const pastEnd = count === 0 && offset > 0;
  const rows = (Array.isArray(list.rows) ? list.rows : [])
    .map((r) => `<tr class="rowlink" data-id="${esc(r.passport_id)}">
      <td>${esc(r.display_name ?? "—")}</td>
      <td><code>${esc(r.claim_state)}</code>${r.retracted ? ' <span class="ds-chip chip">retracted</span>' : ""}${r.suppressed ? ' <span class="ds-chip chip">delisted</span>' : ""}</td>
      <td><code>${esc(r.tier)}</code></td>
      <td><code>${esc(r.source)}</code></td>
      <td class="mono">${esc(r.batch_id ?? "—")}</td>
      <td class="num mono">${n(r.requests)}</td>
      <td class="mono">${esc(fmtTs(r.created_at))}</td>
    </tr>`)
    .join("");
  // AN EMPTY TABLE UNDER A FILTER IS AMBIGUOUS, and one of its two meanings is a
  // typo: `seedingList` refuses an out-of-vocabulary tier or source by returning
  // NOTHING rather than ignoring the filter, so "no seeded listings" and "you
  // asked for a tier that does not exist" look identical. The empty line names
  // the filters so the operator can tell which one they are looking at.
  const truncated = list.truncated === true;
  const empty = truncated
    ? `Past the register's 100,000-row window — narrow by tier, source or batch.`
    : pastEnd
    ? `You have paged past the last seeded listing. Go back.`
    : filtered
      ? `No seeded listings match these filters. An unknown tier or source returns nothing rather than being ignored, so check the filter values before concluding the register is empty.`
      : `No seeded listings yet.`;
  // PAST THE END IS A STATE THIS CONTROL CAN REACH, and it used to be a dead
  // end: "next" stayed live on an exact multiple of the page size, the next page
  // came back empty, and the paging row — rendered only when there were rows —
  // vanished with the Previous button. The operator was left reading "No seeded
  // listings yet" under a metrics card saying 900 were seeded, with no way back.
  //
  // So the row renders whenever there is a page to leave OR rows to page, and an
  // empty page past the start says what it is instead of borrowing the
  // never-seeded sentence.
  const page = count > 0 || offset > 0
    ? `<div class="row" style="margin-top:10px"><span class="muted">${truncated ? "Past the register's 100,000-row window — narrow by tier, source or batch" : pastEnd ? "Past the end of the register" : `Showing ${esc(String(count))} from #${esc(String(offset + 1))}`}</span><span style="flex:1"></span>
        <button class="ds-btn" data-act="seed-prev"${offset === 0 ? " disabled" : ""}>Previous</button>
        <button class="ds-btn" data-act="seed-next"${count < limit ? " disabled" : ""}>Next</button></div>`
    : "";
  return `<table class="ds-table"><thead><tr><th scope="col">Name</th><th scope="col">State</th><th scope="col">Tier</th><th scope="col">Source</th><th scope="col">Batch</th><th scope="col" class="num">Requests</th><th scope="col">Seeded</th></tr></thead>
    <tbody>${rows || `<tr><td colspan="7" class="muted">${esc(empty)}</td></tr>`}</tbody></table>${page}`;
}

import { esc } from "./html.js";

/**
 * THE MEDIA TAKEDOWN LEVER — `DELETE /ops/media/:hash`.
 *
 * `docs/CSAM-SCANNING.md` names this endpoint twice: once as the durable
 * takedown the product relies on, and once as step 2 of the on-a-match runbook.
 * Until this screen existed it had **no button**. An operator following that
 * runbook at 3am reached a step that said "run `DELETE /ops/media/:hash`" and had
 * to find a terminal, an Access token and a curl invocation to do it. A runbook
 * step with no control is a runbook step that gets skipped, and this is the one
 * step that must not be.
 *
 * Four things about this screen are deliberate:
 *
 * 1. THERE IS NO LIST. Media is content-addressed and there is no enumeration
 *    route — by design, and this screen does not invent one. You arrive here
 *    holding a content hash from a Cloudflare match email or a report, and you
 *    paste it. A browsable gallery of creator art on the operator console would
 *    be a surveillance surface built to service a delete button.
 *
 * 2. THE ORDER IS PRINTED, AND REPORTING COMES FIRST. NCMEC reporting is a legal
 *    obligation Cloudflare does not discharge for you, and deletion can destroy
 *    evidence counsel needs preserved. The runbook's order is on screen above the
 *    control, not in a document the operator is not currently reading.
 *
 * 3. IT IS HONEST THAT IT CANNOT BE UNDONE — twice, and specifically. This is
 *    stronger than the evidence purge next door: the purge deletes bytes, this
 *    also writes a `media_takedowns` tombstone that bans the identical bytes
 *    from ever being stored again, by anyone, forever. That is the property that
 *    makes it right for illegal content and wrong for a takedown filed in error,
 *    so the confirm says which one you are doing.
 *
 * 4. A FAILURE NEVER READS AS "DONE". 503 is fail-closed configuration — the
 *    media binding OR the edge-purge credentials absent in this environment —
 *    and it means NOTHING WAS REMOVED, not that the object is gone. 404 means no
 *    such asset, which on this screen usually means a mistyped hash and NOT that
 *    the content is not there. `error` now carries a third, worse case: the row
 *    and the object went and the EDGE CACHE DID NOT. All are drawn amber and say
 *    what is still true.
 *
 * 5. THE COPY IS PINNED TO WHAT THE SERVER DOES, and this screen is why that
 *    rule exists. The done state used to read "The row and the object are gone"
 *    over a takedown that purged nothing from the edge: `GET /media/:hash` is
 *    served `public, max-age=31536000, immutable` (deliberately — Cloudflare's
 *    CSAM tool scans at cache-fill), so every PoP that had already filled kept
 *    serving the identical bytes for up to a YEAR while this console said they
 *    were gone, under a legal commitment reading "We remove or disable the
 *    material". `deleteMediaAsset` now purges the edge and refuses the takedown
 *    outright when it cannot — and the sentences below live in ONE place
 *    (`TAKEDOWN_DONE_FACTS`, `TAKEDOWN_NOT_REMOVED_NOTE`,
 *    `TAKEDOWN_EDGE_UNPURGED_NOTE`), pinned against the route by
 *    apps/admin/test/takedown.test.ts, the same way evidence-review.ts pins
 *    `SUPPRESS_FACTS` against `suppressReport`.
 */

/** A content hash is 64 lowercase hex. Checked here only to keep the operator
 * from firing a confirm at an obvious typo — the server validates it too, and
 * this is not a security boundary. */
const HASH_RE = /^[0-9a-f]{64}$/;

export function looksLikeContentHash(s: string): boolean {
  return HASH_RE.test(s.trim());
}

/** How the takedown went. `unavailable` is the fail-closed 503 — the media store
 * binding OR the edge-purge credentials absent, and in BOTH cases nothing was
 * removed; `missing` is a 404; `error` is anything else, INCLUDING the 502 that
 * says the storage removal happened and the edge purge did not. */
/**
 * UI-AUDIT F-A1 — `edge-unpurged` IS A FIFTH STATE, and its absence is what let
 * this screen tell an operator two false things in a row about a completed,
 * irreversible takedown.
 *
 * `deleteMediaAsset` throws `MediaEdgeNotPurged` AFTER the tombstone is written,
 * the row is dropped and the R2 object is deleted (media.ts: "The origin removal
 * is NOT rolled back"). The console collapsed that 502 into `"error"`, which
 * renders "Nothing was taken down · Do not record this as handled · Retry" —
 * and the retry answers 404, which this same screen explains as "nearly always a
 * mistyped or truncated hash". So: the bytes were banned forever and the object
 * destroyed, and the operator was told it had not happened and then that they
 * had probably typed the hash wrong.
 *
 * The four old states each describe ONE world — done, or nothing. This is the
 * third world: origin done, edge still serving. It gets its own copy because
 * neither of the others is true in it, and the action it calls for (purge one
 * URL by hand; do NOT retry) is the opposite of both.
 */
export const TAKEDOWN_STATUSES = ["done", "edge-unpurged", "unavailable", "missing", "error"] as const;

export type TakedownStatus = (typeof TAKEDOWN_STATUSES)[number];

      <p><b>Do not press Retry.</b> The stored object is already gone, so a retry answers <i>no such asset</i> — and on this screen that message means a mistyped hash, which it would not be. Retrying cannot finish this; only the purge can.</p>
      ${purgeUrl ? `<p><b>Purge this URL by hand, now:</b> <code>${esc(purgeUrl)}</code></p>` : `<p><b>Purge the media URL for this hash by hand, now</b>, from the CDN dashboard.</p>`}
      <p>Then fix the purge configuration so the next takedown completes in one step, and record this one as <b>done at origin, purged by hand</b> — it is written in the audit log that way. <b>If this was illegal content, your reporting duty is unchanged and unfinished.</b></p>
      <div class="row"><a class="btnlink" href="#/config">Check config health</a></div>
    </div>`;
  }
  if (status === "done") {
    return `<div class="tomb spaced">
      <div class="t"><span class="ds-chip chip">◌ taken down</span> <code>${esc(hash)}</code></div>
      <p><b>Storage, cache and re-upload are all closed.</b> ${TAKEDOWN_DONE_FACTS.map((f) => esc(f)).join(" ")} This line is the record of the takedown, not a retained copy.</p>
      <p><b>Your reporting duty does not end here.</b> If this was illegal content, finish the CyberTipline report and the escalation runbook now. Nothing on this screen has notified anyone.</p>
    </div>`;
  }
  const why = status === "unavailable"
    ? `<b>The route fail-closed with a 503.</b> That is configuration being off on purpose, not a fault. ${esc(TAKEDOWN_NOT_REMOVED_NOTE)} On production, this would be a live object still being served.`
    : status === "missing"
      ? "<b>No stored asset has that content hash.</b> On this screen that is nearly always a mistyped or truncated hash — check it character for character against the match email. It is <b>not</b> evidence that the content is not stored."
      : `<b>The takedown did not complete.</b> The likeliest cause is the one this console can name: ${esc(TAKEDOWN_EDGE_UNPURGED_NOTE)} If the response said something else, assume nothing about how far it got — check the object and the tombstone before you record anything.`;
  return `<div class="failbox spaced">
    <div class="hl2"><span class="badge">!</span>Nothing was taken down</div>
    <p>${why} <b>Do not record this as handled.</b> Retry, then check config health; if it keeps failing, escalate — a takedown that silently did not happen is the worst outcome on this screen.</p>
    <div class="row"><button class="ds-btn" data-act="takedown-retry">Retry</button><a class="btnlink" href="#/config">Check config health</a></div>
  </div>`;
}

/**
 * THE CONFIRM. Typed, like account erasure — the operator retypes the exact
 * hash, so a takedown fired at the wrong object requires getting 64 characters
 * wrong twice. Returns "" for a support seat: the button that opens it is
 * already inert and the server already 403s the DELETE, and this is the third
 * defence — the confirm button does not exist in the document at all.
 */
export function takedownConfirm(hash: string, who: string, role: OpsRole): string {
  if (role !== "full") return "";
  return `<div class="scrim open" id="m-takedown">
    <div class="confirm destructive" role="alertdialog" aria-modal="true" aria-labelledby="td-h">
      <h3 id="td-h">⚠ Take down <code>${esc(hash)}</code>?</h3>
      <p><b>This cannot be undone.</b> ${TAKEDOWN_DONE_FACTS.map((f) => esc(f)).join(" ")} There is no untombstone lever and no support path to one.</p>
      <p>That permanence is the point when the content is illegal, and it is the harm when the hash is wrong. <b>An honest mistake by a creator stays re-uploadable only if you never press this</b> — for a creator who uploaded the wrong art, tell them to delete it themselves.</p>
      <p><b>If this is a CSAM match: report to NCMEC first.</b> Deletion can destroy evidence you are required to preserve, and pressing this notifies nobody.</p>
      <label for="tdconfirm">Type the content hash exactly to confirm</label>
      <input id="tdconfirm" value="" placeholder="${esc(hash)}" spellcheck="false" autocomplete="off">
      <div class="auditline mono">audit · media:takedown · ${esc(hash)} · ${esc(who)}</div>
      <div class="row spaced">
        <button class="ds-btn" data-act="takedown-cancel">Cancel</button>
        ${writeControl({ label: "Take down permanently", role, tone: "destructive", attrs: { "data-act": "takedown-confirm", "data-hash": hash }, lock: false })}
      </div>
      <p id="tdmsg" class="muted"></p>
    </div>
  </div>`;
}
