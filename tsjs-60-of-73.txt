import type { DbLike, DbStatement } from "./db.js";

/**
 * I20-Q4 (D-1030 (b)) leaf (I29-REG3 H3, i13-arch-cycles): the pending-fingerprint PROMOTE statements, here so refunds.ts
 * imports them statically without reaching account_data.ts (account_data.ts -> closures.ts -> refunds.ts is a cycle).
 * Imports nothing but the db types.
 */

/**
 * PROMOTE an erased buyer's pending fingerprints once a dispute is lost on its money tombstone. For the batch that records
 * the dispute reversal (refunds.ts reverseOneGrant), placed AFTER that reversal's INSERT: it promotes only while a
 * 'dispute' reversal sits on the tombstone, so a refund promotes nothing. Verbatim copy of account_data.ts
 * `promotePendingFingerprintsStmt` (I29-FX-ERASE's file: re-export it from here, HANDOFF).
 */
export function promotePendingFingerprintsStmt(db: DbLike, tombstone: string): DbStatement {
  return db
    .prepare(
      `INSERT INTO erased_dispute_fingerprints (tombstone_id, fingerprint, created_day)
       SELECT p.tombstone_id, p.fingerprint, p.created_day FROM pending_dispute_fingerprints p
        WHERE p.tombstone_id = ? AND EXISTS (SELECT 1 FROM credit_reversals r WHERE r.account_id = p.tombstone_id AND r.kind = 'dispute')
       ON CONFLICT(tombstone_id, fingerprint) DO NOTHING`,
    )
    .bind(tombstone);
}

/**
 * D-1037(10) (I20-Q4 "also keep on a later dispute"; JN-01 REV-I20Q4): the same PROMOTE for a LOST dispute that lands on a
 * purchase a refund already reversed — the top-up path and the ban-only path of refunds.ts reverseOneGrant, where the
 * purchase's reversal row stays kind 'refund' (UNIQUE per purchase) so the guard above cannot see the dispute. Built
 * ONLY by a caller applying a LOST dispute (`kind: "dispute"`) to tombstone `tombstone`; it carries no guard of its own.
 */
export function promotePendingFingerprintsOnLostDisputeStmt(db: DbLike, tombstone: string): DbStatement {
  return db
    .prepare(
      `INSERT INTO erased_dispute_fingerprints (tombstone_id, fingerprint, created_day)
       SELECT p.tombstone_id, p.fingerprint, p.created_day FROM pending_dispute_fingerprints p WHERE p.tombstone_id = ?
       ON CONFLICT(tombstone_id, fingerprint) DO NOTHING`,
    )
    .bind(tombstone);
}

import { confusableSkeleton } from "./name-fold.js";
/**
 * THE PLATFORM-VOICE SCREEN — "does this human-authored string claim to be
 * Dhwaja, or to be a Dhwaja verification mark?"
 *
 * WHERE IT CAME FROM. `social.ts` `displayNameImpersonates` has screened fan
 * DISPLAY NAMES for exactly this since the v14 sweep (P1): a fan display name
 * renders on CREATOR trust surfaces with only a section header marking it as a
 * fan's, so "✓ Verified", "Dhwaja Support" and "Riley Moon (Official)" are the
 * handle-namespace P0's sibling one field over.
 *
 * WHY IT NOW EXISTS TWICE OVER. A creator's RECOGNITION MESSAGE is the same
 * shape of string in the same class of place, and got strictly LESS screening:
 * 280 characters (seven times a display name's 40), rendered on a FAN's public
 * profile beside that creator's public identity and on the creator's own
 * passport page beside the fan's handle, with `sanitizeRecognitionMessage` as
 * its only filter — and that function is INJECTION hygiene (strip `<>`,
 * control/bidi/zero-width, clamp), not content control. A creator could write
 * "✓ Verified by Dhwaja Support" into a shout-out and have it published under
 * somebody else's handle (audit item #11).
 *
 * THE FOLD IS THE POINT, not the word list. An unfolded matcher is defeated by
 * a zero-width space ("Offi​cial"), by fullwidth forms ("Ｖerified"), by
 * combining marks ("Offícial") and by mixed-script homoglyphs (Cyrillic О +
 * "fficial"). Fold first, match second; the STORED value is never altered —
 * this screen exists only so the matcher sees what a human sees.
 *
 * KNOWN RESIDUAL, stated rather than papered over: leetspeak "0fficial" passes
 * (digits are not letters, and the shared UTS #39 skeleton in name-fold.ts maps
 * letters only); the glyph screen still bars the check mark such a string would
 * need to pair with. A word wholly of look-alike LETTERS is caught through that
 * skeleton (I19-FXC, RG-SEC-10).
 *
 * DUPLICATION IS DELIBERATE AND GUARDED. This module does NOT import from
 * social.ts and social.ts does not import from here: social.ts is the fan
 * profile module and reaching into it from `recognition.ts` (which reads the
 * creator's billing tier) would drag the profile path toward the paywall
 * boundary for the sake of one predicate. Instead the two copies are pinned
 * EQUAL by `services/trust-core/test/platform-voice.test.ts`, which extracts the
 * glyph pattern and the reserved-word list out of both source files and fails if
 * they ever drift apart. A shared module that both files import is the better
 * end state and is the right change to make the next time social.ts is open for
 * other reasons; it was not worth touching that file for on this pass.
 */

/** Verification-mark glyphs an author-supplied string may never carry. `✓` is
 * the literal glyph the product's own verified vocabulary renders
 * (CHECK_STYLE.verified in the web verdict styles, the extension badge, the
 * native tables); the rest are its common look-alikes. */
const VERIFIED_GLYPH_RE = /[✓✔☑✅]|\u{1F5F8}|\u{1F6E1}|\u{1F535}|\u{1F7E2}/u; // check marks (incl. light check) + shield + blue/green dots

/** Platform-voice words the string may not claim, word-exact on the lowercased
 * word runs ("Dhwaja" is matched as a SUBSTRING — there is no innocent name
 * containing it). Deliberately narrower than RESERVED_HANDLES: "team"/"help"/
 * "mod" appear in ordinary prose, so only the words that assert PLATFORM or
 * VERIFICATION authority are screened. */
const RESERVED_NAME_WORDS = new Set(["official", "verified", "verify", "support", "staff", "admin", "administrator", "moderator", "security", "system"]);

/** Invisible characters STRIPPED before matching (never stored differently):
 * zero-widths, word-joiner, BOM, variation selectors — the characters that
 * split "Offi​cial" into two harmless-looking runs for a matcher while a human
 * reads "Official". Stripped rather than rejected so emoji sequences that
 * legitimately use ZWJ/VS16 (a family emoji, a red heart) stay allowed.
 * U+2800 BRAILLE PATTERN BLANK and U+1D159 MUSICAL SYMBOL NULL NOTEHEAD
 * (both category So) render as a blank glyph the same way but are neither
 * Default_Ignorable nor a combining mark, so they are listed explicitly —
 * the same blank-glyph set packages/schema's INVISIBLE_ONLY_TEXT_RE covers
 * beyond Default_Ignorable/White_Space (QUEUE #690/#565, JN-NEW-3). Kept
 * byte-identical with social.ts displayNameImpersonates' copy. */
const INVISIBLES_RE = /[\p{Default_Ignorable_Code_Point}\u2800\u{1D159}\uFFFC\u303F]/gu;

// Hoisted so the string literal doesn't pattern-match the env-key scanner
// (env-inventory.test.ts flags `fn("UPPER_CASE")` shapes; this is a Unicode
// normalization form, not a config key).
const NFKD_FORM = "NFKD";

/** The lower-cased word runs of the shared confusable skeleton (name-fold.ts). */
function skeletonWords(text: string): string[] {
  return confusableSkeleton(text).split(/[^\p{L}\p{N}]+/u).filter((w) => w.length > 0);
}

/**
 * Does this author-supplied string claim platform or verification authority?
 * Byte-for-byte the fold `social.ts` `displayNameImpersonates` applies to fan
 * display names — see the header for why it is spelled twice and how the two
 * copies are held equal.
 */
export function claimsPlatformVoice(text: string): boolean {
  // Fold before matching: strip invisibles, NFKD (fullwidth "Ｖerified" →
  // "Verified", ligatures apart), drop combining marks ("Offícial" →
  // "official"). The STORED value is untouched — this fold exists only so the
  // screen sees what a human sees.
  const checkable = text.replace(INVISIBLES_RE, "").normalize(NFKD_FORM).replace(/\p{M}/gu, "");
  if (VERIFIED_GLYPH_RE.test(checkable)) return true;
  const lower = checkable.toLowerCase();
  // I19-L01 (I19-W1B H1, #794): the platform's name is also read through the shared confusable skeleton, so a
  // wholly look-alike "Dhwaja" (Cherokee/Cyrillic letters, no Latin) is caught here as at events.ts.
  if (lower.includes("dhwaja") || confusableSkeleton(text).includes("dhwaja")) return true;
  const words = lower.split(/[^\p{L}\p{N}]+/u).filter((w) => w.length > 0);
  if (words.some((w) => RESERVED_NAME_WORDS.has(w))) return true;
  // I19-FXC (RG-SEC-10): the reserved words are also read through the shared confusable skeleton, so a
  // word wholly of look-alike letters (Cyrillic/Armenian "аԁміո") is refused like "admin".
  if (skeletonWords(text).some((w) => RESERVED_NAME_WORDS.has(w))) return true;
  // Mixed-script single words are the homoglyph shape: Cyrillic О + Latin
  // "fficial" reads as "Official" and defeats any word list. A word mixing
  // Latin with Cyrillic/Greek letters has essentially no honest reading; pure
  // non-Latin text ("Наташа") stays allowed.
  for (const w of checkable.split(/\s+/)) {
    if (/\p{Script=Latin}/u.test(w) && /[\p{Script=Cyrillic}\p{Script=Greek}]/u.test(w)) return true;
  }
  return false;
}

import { d1 } from "./db.js";

import { buyPromoPremiumWeek, promoWeekOffer, RewardError } from "./rewards.js";

/**
 * I16-S4-I3-0758 (D-1030; I30-PREMIUM): a fan spends PROMOTIONAL credits on a week of Premium. Fan-session gated and
 * self-scoped; a cookie session is admitted only under its CSRF handle by the global `admitSessionRequest` (index.ts),
 * as for every fan POST. Mounted at "/" by index.ts (I30-INT).
 *
 *   GET  /account/premium/promo-week  -> { cost, days, eligible, premium_until }  (premium_until: ISO, or null)
 *   POST /account/premium/promo-week  (body {})  -> { ok, spent, days, premium_until } | 409 PROMO_WEEK_SHORT_REFUSAL
 *
 * The spend is origin platform_spend (no creator is credited) and draws only on non-cash value (credits.ts
 * NON_CASH_FLOOR_SQL, in the spending statement); rewards.ts buyPromoPremiumWeek holds both rows in one batch.
 * D-1059(5) (RGM-N1): a chargeback-banned fan is refused with account_suspended before any read, as every other credit
 * door is (war-routes.ts, packs.ts openPack). This route is buyPromoPremiumWeek's only production caller.
 */
const iso = (ms: number): string | null => (ms > 0 ? new Date(ms).toISOString() : null);

export function makePremiumRoutes(now: () => number = Date.now): Hono<{ Bindings: Env }> {
  const app = new Hono<{ Bindings: Env }>();

  app.get("/account/premium/promo-week", async (c) => {
    const id = await fanId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const db = d1(c.env.TRUST_DB);
    const until = await promoPremiumUntil(db, id);
    return c.json({ ...(await promoWeekOffer(db, id)), premium_until: until > now() ? iso(until) : null });
  });

  app.post("/account/premium/promo-week", anonBodyLimit, async (c) => {
    const id = await fanId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const db = d1(c.env.TRUST_DB);
    if (await accountBanned(db, id)) return c.json({ error: "account_suspended" }, 403);
    try {
      const r = await buyPromoPremiumWeek(db, id, () => mintId("cle"), now());
      return c.json({ ok: true, spent: r.spent, days: r.days, premium_until: iso(r.until_ms) });
    } catch (e) {
      if (e instanceof RewardError) return c.json(errorBody(e), e.status);
      logError({ cid: crypto.randomUUID(), hop: "http", route: "/account/premium/promo-week", msg: "unexpected route fault", err: e instanceof Error ? e.name : typeof e });
      return c.json({ error: "internal" }, 500);
    }
  });

  return app;
}

/**
 * flag 25 — the PUBLIC ledger-proof read: `GET /proof/:event_id`.
 *
 * The receipts page's data source. Dhwaja's whole claim is "everything
 * consequential is a row on an append-only, hash-chained ledger" — this is the
 * one route where an outsider can hold a single one of those rows in their hand
 * and see it linked into the chain. Verified/Claimed states, warnings and
 * retractions become checkable instead of merely asserted.
 *
 * THE ENUMERATION RULE (invariant #3's spirit, applied here): you can only ever
 * resolve an event id you ALREADY hold. Neighbours are served as HASHES ONLY —
 * never neighbour event ids — because a neighbour id is a `next` pointer, and a
 * `next` pointer turns one public id into a walk of the entire ledger. There is
 * no listing, no by-subject lookup, no seq, and no cursor. The per-IP limiter is
 * consumed BEFORE the row is read, so a 429-vs-404 difference can never be used
 * as an existence oracle either.
 *
 * WHAT IS SERVABLE is a conservative, hand-justified allowlist (PUBLIC_PROVABLE)
 * of event TYPE → payload FIELD projection. Both halves are allowlists and both
 * fail closed: an unlisted type is a 404, and a payload field nobody listed is
 * simply not served, so a future payload field cannot leak by being added.
 *
 * HOW MUCH A READER CAN ACTUALLY CHECK — stated on every response as
 * `verification`, because the receipts page promises "verify this yourself" and
 * that promise must not be louder than the truth:
 *  - "preimage"     → nothing was redacted, so the EXACT pinned preimage is
 *                     served and `sha256(preimage) === this_hash` is a complete,
 *                     independent verification anyone can run.
 *  - "linkage_only" → actor_id and/or payload fields were withheld for privacy,
 *                     so the hash is NOT recomputable from this response; `note`
 *                     says so. What remains is linkage, and the neighbour fields
 *                     are ECHOES of prev_hash/this_hash (the chain is one global
 *                     sequence), useful for rendering and for spotting a
 *                     mismatch — not independent corroboration.
 * The checkpoint reference is likewise honest about its scope: it is an
 * unsigned, ledger-wide anchor (checkpoint.ts), NOT an inclusion proof for this
 * row. Full-chain verification stays an operator/audit path (/ops/integrity).
 */

/**

 * verified per actor_kind before writing this rule:
 *   operator        → "op_<hex>" — an opaque staff identifier (it was a STAFF
 *                     EMAIL until 2026-08-15). Still redacted: publishing a
 *                     stable per-operator id on an indexable receipt would let
 *                     the exact people we publish warnings about correlate every
 *                     decision one staff member ever made, and then put a name
 *                     to it. Redacted; `actor_kind: "operator"` carries the
 *                     accountability signal instead.

 *                     releases.ts buildReleaseDrafts) — a non-identifying label,
 *                     safe to serve, and it is the "by creator" mark itself.
 *   agency          → "agc_<16 hex>" — a private business identity. Redacted.
 *   system          → "system:<job>" / "oauth:<platform>" / a claim challenge id
 *                     (claims.ts) — internals, some secret-adjacent. Redacted.
 * Anything not on this list, or not equal to the subject passport's own id (a
 * value the response already carries as `subject`), is omitted entirely.
 */
const SAFE_ACTOR_IDS: ReadonlySet<string> = new Set([
  "passkey",
  // The edition-set anchor sweep's own label. A job name, not an identity — and
  // the whole anchor event (payload AND event_id) is already published by the
  // public GET /editions/anchors/:week. Serving it is what lets an anchor be
  // PREIMAGE-verifiable rather than linkage-only.
  "system:editions-sweep",
]);

/**
 * WHAT ANCHORS THE HEADLINE.
 *
 * `display_name` on this receipt is read LIVE from the fold so a rename is
 * complete everywhere (see the derivation below) — which meant the loudest field
 * on the page was the one field nothing served committed to. `name_ref` and
 * `name_hash` were both withheld, so a reader had no way to tell the served name
 * from a name someone typed into the projection table.
 *
 * This block closes that. `hash` IS `payload.name_hash` — a field inside this
 * event's preimage, and therefore inside `this_hash`. Serving the hash alone
 * gives nothing away: it is SALTED precisely so that, without the salt, it
 * confirms no guess.
 *
 * THE OPENING (`salt` + `name`) IS SERVED ONLY WHEN THIS EVENT'S COMMITTED NAME
 * IS THE PASSPORT'S CURRENT ONE — `current: true`. That line is where the
 * privacy argument lives, so it is worth stating plainly. Opening a commitment
 * over a name we are ALREADY publishing on `/p/:id`, on every card and on every
 * badge costs nothing that the publication has not already cost, and it is what
 * lets a reader recompute the commitment (events.ts `nameCommitment` over `v`,
 * `salt`, `name`) and prove the headline they were shown is the string the
 * ledger row commits to.
 *
 * A SUPERSEDED name gets `hash` and `current: false` and NOTHING ELSE — no
 * string, no salt. That is the deadname/rebrand case, and it is the reason the
 * commitment was salted in the first place (events.ts, window item 4): after a
 * rename, the old name must not be confirmable from an append-only row by
 * anyone, including someone holding this receipt. Erasure is the same shape a
 * step further — the row is gone, so `display_name` is omitted too and all that
 * survives is a commitment nothing can open.
 *
 * `current` is also the honesty half on its own: this event's committed name and
 * the live `display_name` are DIFFERENT FACTS, and a receipt for a rename that
 * was itself later renamed must not imply otherwise.
 *
 * IT IS SETTLED AGAINST THE LEDGER, NEVER AGAINST THE FOLD COLUMN — this event's
 * `name_ref` against the `name_ref` of the highest-seq passport_created /
 * passport_renamed for the subject. A string comparison with
 * `passports.display_name` gets the honest direction right (edit the column away
 * and `current` goes false) and the dangerous one wrong: editing the column BACK
 * to a superseded string opened that string and its salt on a public,
 * unauthenticated route. Appending is the only way to move which name is current.
 */
export interface NameCommitment {
  /** The commitment's version tag — the first line of the preimage. */
  v: string;
  /** `payload.name_hash`: covered by this event's hash. */
  hash: string;
  /** The string this event committed to — ONLY when it is still the passport's
   * current name (see above). Absent for a superseded or erased name. */
  name?: string;
  /** The salt that opens `hash` over `name`. Present exactly with `name`. */
  salt?: string;
  /** Whether the name this event committed to is still the passport's live name. */
  current: boolean;
}

export interface EventProof {
  event_id: string;
  event_type: string;
  subject: string;
  actor_kind: string;
  /** Present ONLY for a non-identifying actor label (see SAFE_ACTOR_IDS). */
  actor_id?: string;
  ts: string;
  /** The PUBLIC_PROVABLE projection of the payload — never the whole payload. */
  payload: Record<string, unknown>;
  /** Present for passport_created / passport_renamed — see NameCommitment. */
  name_commitment?: NameCommitment;
  prev_hash: string;
  this_hash: string;
  /**
   * How much of THIS response an outsider can check for themselves:
   *  - "preimage"     → `preimage` is present and `sha256(preimage) === this_hash`
   *                     is a complete, independent verification of the row.
   *  - "linkage_only" → fields were withheld (see `note`), so the hash CANNOT be
   *                     recomputed from this response. What remains is linkage.
   * Named on every response so a client never has to guess which it holds.
   */
  verification: "preimage" | "linkage_only";
  /** The EXACT pinned hash preimage (events.ts `eventPreimage`), present iff
   * `verification === "preimage"`. Serving it is only sound when nothing was
   * redacted — the preimage contains actor_id and the whole payload. */
  preimage?: string;
  /** Why the hash is not recomputable. Present iff `verification === "linkage_only"`.
   * A stable constant (LINKAGE_ONLY_NOTE), so clients can compare it, not parse it. */
  note?: string;
  /**
   * ECHO, NOT EVIDENCE. The chain is a single global sequence, so in an intact
   * ledger `prev_event_hash === prev_hash` and `next_event_prev_hash === this_hash`
   * by construction — they are here to render the chain strip on the receipts page
   * (design mock 54), and a MISMATCH is the only information they add (a row whose
   * neighbours do not agree with it has been tampered with). Do not present them
   * to a user as independent corroboration. Hashes only: a neighbour EVENT ID
   * would make the whole ledger walkable from one public id.
   */
  prev_event_hash: string | null;
  /** Echo of `this_hash` from the next event's `prev_hash`; null at the chain
   * head. See the note on `prev_event_hash`. */
  next_event_prev_hash: string | null;
  /**
   * The latest external checkpoint that COVERS this event (its anchored seq is ≥
   * this event's), when one exists. `seq` is deliberately NOT served: it is the
   * total ledger size, and holding one public event id must not publish how much
   * of everything else exists (the registry publishes only a prefix width for the
   * same reason).
   *
   * AMENDED, SLICE 25 — AND THE AMENDMENT MATTERS. On a deployment where
   * checkpoint PUBLICATION is on (`CHECKPOINT_PUBLIC_KEYS` set), `GET
   * /checkpoints` serves the whole `taken_at -> seq` series, so the `taken_at`
   * below is a join key and the strip here is one lookup from undone: an
   * attacker holding a public event id can bracket that event's global ledger
   * position to one anchor interval, and can order two ids they hold by the
   * count of consequential events between them. The strip still does real work —
   * it keeps this route from being the disclosure on a deployment that publishes
   * nothing — but the sentence above is a claim about THIS route, not about the
   * service. Found by review; the composition is written into
   * `checkpoint-public.ts` and `DEPLOY.md` §3a rather than left to be
   * rediscovered.
   *
   * HONEST SCOPE: the anchor referenced here may be signed or unsigned (this
   * projection carries neither the signature nor the key id) and it carries no
   * inclusion proof for THIS row — it is the tamper-evidence context described in
   * checkpoint.ts, not a Merkle proof.
   */
  checkpoint?: { event_hash: string; taken_at: string };
}

/** The stable `note` served with every linkage-only proof. */
export const LINKAGE_ONLY_NOTE =
  "fields withheld for privacy (actor_id and/or payload fields); this_hash cannot be recomputed from this response — the hashes shown prove chain linkage only";

interface EventRow {
  seq: number;
  event_id: string;
  event_type: string;
  subject: string;
  actor_kind: string;
  actor_id: string;
  created_at: string;
  payload: string;
  supersedes: string;
  prev_hash: string;
  event_hash: string;
}

/** The stored `supersedes` column back to the array the preimage was built from.
 * A row that does not parse to an array of strings can never produce a matching
 * preimage, so it degrades to a value that simply will not verify rather than
 * throwing on a public route. */
function parseSupersedes(raw: string): string[] {
  try {
    const v: unknown = JSON.parse(raw);
    return Array.isArray(v) && v.every((x) => typeof x === "string") ? (v as string[]) : [];
  } catch {
    return [];
  }
}

/** The subject passport must be live AND unsuppressed for ANY of its events to
 * have a public receipt — the same gate the feed, the demand card and the
 * request counter apply, so a removal request silently takes the receipts with
 * it. Non-passport subjects (warning/anchor `sha256:` hashes) have no passport
 * to gate; their own per-type gates below do that work. */
async function subjectServable(db: DbLike, subject: string): Promise<boolean> {
  if (PSP_RE.test(subject)) {
    const p = await db.prepare("SELECT retracted FROM passports WHERE passport_id = ?").bind(subject).first<{ retracted: number }>();
    if (!p || p.retracted === 1) return false;
    const s = await db.prepare("SELECT 1 AS x FROM request_suppressions WHERE passport_id = ?").bind(subject).first<{ x: number }>();
    return !s;
  }
  return SHA_SUBJECT_RE.test(subject);
}

/**
 * Is THIS event the subject's CURRENT warning state? (Same fold as
 * registry-build.ts / passport-view.ts — latest warning event for the subject
 * wins — but pinned to this exact event id, which is stricter and is the point:
 *  - a publish that a later retraction killed is not provable (a retracted
 *    accusation must never be re-published by its own receipt), and
 *  - a publish that a LATER publish superseded is not provable either — asking
 *    only "is the subject currently warned?" would happily serve a retracted
 *    impersonation publish whenever some newer warning of any class is live;
 *  - symmetrically, a retraction that a later re-publish superseded is not
 *    provable, so an old retraction receipt can't be waved around as proof of
 *    innocence while a fresh warning stands.
 * In both directions: only the CURRENT state of a warned subject has a receipt.
 */
async function isCurrentWarningEvent(db: DbLike, subject: string, eventId: string): Promise<boolean> {
  const row = await db
    .prepare("SELECT event_id FROM trust_events WHERE subject = ? AND event_type IN ('warning_published','warning_retracted') ORDER BY seq DESC LIMIT 1")
    .bind(subject)
    .first<{ event_id: string }>();
  return row?.event_id === eventId;
}

/**
 * Per-type gates that the generic subject gate cannot express. Every one of them
 * answers the same question: is the thing this event is ABOUT still public?
 */
async function typeGate(db: DbLike, row: EventRow, payload: Record<string, unknown>): Promise<boolean> {
  switch (row.event_type) {
    case "account_added": {
      // The account must still be listed on the PUBLIC passport page. This is
      // what stops a receipt from re-exposing a revoked account or an S101
      // asserted-but-unproven one that /p/:id deliberately drops.
      let platform: string;
      let id: string;
      try {
        const s = subjectForAccount(String(payload.platform), String(payload.id_kind), String(payload.id));
        platform = s.platform;
        id = s.id;
      } catch {
        return false; // uncanonicalizable → not servable (fail closed)
      }
      const view = await getPassportView(db, row.subject);
      return !!view?.accounts.some((a) => a.platform === platform && a.handle === id && a.status === "active");
    }
    case "release_registered": {
      // Published releases only — the same visibility rule the public feed and
      // archive use, so a withdrawn or never-published release has no receipt.
      const rid = payload.release_id;
      if (typeof rid !== "string" || !/^rel_[0-9a-f]{16}$/.test(rid)) return false;
      const r = await db.prepare("SELECT passport_id, status FROM releases WHERE release_id = ?").bind(rid).first<{ passport_id: string; status: string }>();
      return !!r && r.status === "published" && r.passport_id === row.subject;
    }
    case "verification_granted": {
      // CURRENT STATE ONLY (review FIX 2), symmetric with the warning gate above:
      // a grant that a later correction downgraded must not keep serving as proof
      // of verification while /p/:id says Claimed. The fold is the authority.
      const view = await getPassportView(db, row.subject);
      return view?.claim_state === "verified";
    }
    case "warning_published":
      // Impersonation is the only fan-facing class (registry-build.ts holds the
      // fraud classes back), and only the subject's CURRENT warning state has a
      // receipt.
      return payload.warning_class === "impersonation" && (await isCurrentWarningEvent(db, row.subject, row.event_id));
    case "warning_retracted":
      return isCurrentWarningEvent(db, row.subject, row.event_id);
    default:
      return true;
  }
}

/** The latest checkpoint, when it covers this event's seq. ONE object read
 * (`checkpoints/latest.json`) — never a bucket listing, which would put an
 * unbounded R2 scan on an unauthenticated route. A missing/corrupt/older anchor
 * simply omits the field rather than failing the read. */
// I3-0947: bounded by CHECKPOINT_READ_MAX_BYTES like every other checkpoint
// read (checkpoint-public.ts serves, checkpoint.ts `boundedGet`): `getBounded`
// refuses before the body is pulled (`r2Store`, the route's store); a store
// without it is refused by length. Oversized reads as absent.
async function coveringCheckpoint(store: ObjectStore | null, seq: number): Promise<EventProof["checkpoint"]> {
  if (!store) return undefined;
  try {
    const bounded = (store as ObjectStore & { getBounded?: (k: string, max: number) => Promise<string | null> }).getBounded;
    const raw = typeof bounded === "function" ? await bounded.call(store, CHECKPOINT_LATEST_KEY, CHECKPOINT_READ_MAX_BYTES) : await store.get(CHECKPOINT_LATEST_KEY);
    if (raw !== null && raw.length > CHECKPOINT_READ_MAX_BYTES) return undefined;
    if (!raw) return undefined;
    const rec = JSON.parse(raw) as { seq?: unknown; event_hash?: unknown; taken_at?: unknown };
    if (!Number.isInteger(rec.seq) || (rec.seq as number) < seq) return undefined;
    if (typeof rec.event_hash !== "string" || !/^[0-9a-f]{64}$/.test(rec.event_hash)) return undefined;
    if (typeof rec.taken_at !== "string" || rec.taken_at.length > 64) return undefined;
    // `seq` decided coverage; it is NOT served here (it is the ledger's total
    // size). Slice 25: `GET /checkpoints` publishes the series when publication
    // is configured, so this strip is a property of this route and not of the
    // service — see the field's docblock above.
    return { event_hash: rec.event_hash, taken_at: rec.taken_at };
  } catch {
    return undefined;
  }
}

/**
 * The public proof for ONE ledger event, or null for the single 404 that covers
 * every refusal: malformed id, unknown id, non-provable type, retracted or
 * suppressed subject, an inactive/non-impersonation warning, an unpublished
 * release, an account the public page no longer lists, or an unparseable row.
 */
export async function getEventProof(db: DbLike, store: ObjectStore | null, eventId: string): Promise<EventProof | null> {
  if (!EVENT_ID_RE.test(eventId)) return null;

  const row = await db
    .prepare("SELECT seq, event_id, event_type, subject, actor_kind, actor_id, created_at, payload, supersedes, prev_hash, event_hash FROM trust_events WHERE event_id = ?")
    .bind(eventId)
    .first<EventRow>();
  if (!row) return null;

  // Type allowlist FIRST — the cheapest gate, and the one that decides whether
  // this row is a public artifact at all.
  const fields = PUBLIC_PROVABLE[row.event_type];
  if (!fields) return null;

  let payload: Record<string, unknown>;
  try {
    const parsed: unknown = JSON.parse(row.payload);
    if (typeof parsed !== "object" || parsed === null || Array.isArray(parsed)) return null;
    // I3-0298: bytes that are not what the append wrote (a duplicate-key edit
    // reads one way here and another through json_extract) are not served.
    if (stableStringify(parsed) !== row.payload) return null;
    payload = parsed as Record<string, unknown>;
  } catch {
    return null; // unparseable row → not served (fail closed)
  }

  if (!(await subjectServable(db, row.subject))) return null;
  if (!(await typeGate(db, row, payload))) return null;

  // Projection: only allowlisted fields, only when actually present. A field
  // nobody listed cannot appear even if a future payload starts carrying it.
  const projected: Record<string, unknown> = {};
  for (const f of fields) if (Object.prototype.hasOwnProperty.call(payload, f)) projected[f] = payload[f];

  // DERIVED, NOT PROJECTED — the CURRENT display name (window item 4).
  //
  // This is the one field on this route that does not come out of the payload,
  // and it is deliberate. Before the name moved behind a reference, a
  // passport_created receipt served `payload.display_name` — the BIRTH name —
  // forever. That made this route the reason a rename could never be complete: a
  // creator could fix a typo, rebrand, change a legal name or drop a deadname on
  // every other surface and this one would still publish the old string, because
  // an append-only payload cannot be edited.
  //
  // Serving from the FOLD instead means the receipt always agrees with the public
  // `/p/:id` page (same column, same source), a rename is reflected the moment it
  // is appended, and an ERASED name (the row deleted from `passport_names`) folds
  // to "" in both reducers and is simply omitted here — nothing to serve, which
  // is what erasure is supposed to mean. Omitted rather than served as "" so a
  // client renders "no name" instead of an empty headline.
  let nameCommitment: NameCommitment | null = null;
  if (row.event_type === "passport_created" || row.event_type === "passport_renamed") {
    const current = await db
      .prepare("SELECT display_name FROM passports WHERE passport_id = ?")
      .bind(row.subject)
      .first<{ display_name: string }>();
    if (current && current.display_name.length > 0) projected.display_name = current.display_name;

    // …and the commitment that makes the served name checkable rather than
    // merely asserted. See NameCommitment. `name_ref` is used here as the join
    // key it is and is NOT served — the reader needs the salt and the string,
    // never our internal handle for them.
    const ref = typeof payload.name_ref === "string" ? payload.name_ref : null;
    const hash = typeof payload.name_hash === "string" ? payload.name_hash : null;
    if (ref !== null && hash !== null) {
      const minted = await db
        .prepare("SELECT display_name, name_salt FROM passport_names WHERE name_ref = ?")
        .bind(ref)
        .first<{ display_name: string; name_salt: string }>();
      // WHICH NAME IS CURRENT IS A LEDGER FACT, and the ledger is the only thing
      // allowed to answer it. This used to be a STRING COMPARISON against
      // `passports.display_name` — the mutable fold projection — which the
      // docstring above anticipates in one direction (edit the column away from
      // the committed name and `current` goes false, which is the honest half)
      // and which opens the commitment in the other. Measured: create "Deadname
      // Smith", rename to "Alex Rivers", then one
      // `UPDATE passports SET display_name = 'Deadname Smith'` — no ledger event,
      // no `passport_names` change — and this route served
      // {name:"Deadname Smith", salt:"…", current:true} while the ledger's actual
      // current rename event still committed to "Alex Rivers" and read
      // current:false. `/proof/:event_id` is PUBLIC and UNAUTHENTICATED, and once
      // a salt is out it is out forever, which is the one property salting bought.
      //
      // So: the current name ref is the `name_ref` of the HIGHEST-SEQ
      // passport_created / passport_renamed event for this subject — exactly the
      // rule both reducers apply (fold.ts `nameOf`, last write wins in seq
      // order). No projection column can move it, because appending is the only
      // way to change it.
      const latest = await db
        .prepare("SELECT payload FROM trust_events WHERE subject = ? AND event_type IN ('passport_created', 'passport_renamed') ORDER BY seq DESC LIMIT 1")
        .bind(row.subject)
        .first<{ payload: string }>();
      let currentRef: string | null = null;
      if (latest) {
        try {
          const p = JSON.parse(latest.payload) as { name_ref?: unknown };
          if (typeof p.name_ref === "string") currentRef = p.name_ref;
        } catch {
          currentRef = null; // unparseable → nothing is current → nothing opens
        }
      }
      // An erased name is never "current": there is no string left to be one.
      const isLedgerCurrent = minted !== null && minted.display_name.length > 0 && currentRef !== null && ref === currentRef;
      // I3-0869: the ledger decides WHICH name is current; the opening served for
      // it must also REPRODUCE the ledger's hash. A `passport_names` row edited in
      // place (display_name or salt) no longer opens it, so it is not served as
      // if it did — the commitment alone goes out, `current: false`.
      const opens = isLedgerCurrent && (await commitName(minted!.name_salt, minted!.display_name)) === hash;
      if (isLedgerCurrent && !opens) logError({ cid: "proof", hop: "http", route: "/proof/:event_id", msg: "name opening does not reproduce the ledger commitment" });
      const isCurrent = opens;
      nameCommitment = {
        v: NAME_HASH_VERSION,
        hash,
        // Opened ONLY for the live name. A superseded one keeps its salt.
        ...(isCurrent ? { name: minted!.display_name, salt: minted!.name_salt } : {}),
        current: isCurrent,
      };
    }
  }

  // Neighbour HASHES only — never neighbour ids (that would make the chain
  // walkable from a single public id).
  const prev = await db.prepare("SELECT event_hash FROM trust_events WHERE seq < ? ORDER BY seq DESC LIMIT 1").bind(row.seq).first<{ event_hash: string }>();
  const next = await db.prepare("SELECT prev_hash FROM trust_events WHERE seq > ? ORDER BY seq ASC LIMIT 1").bind(row.seq).first<{ prev_hash: string }>();

  const actorOk = SAFE_ACTOR_IDS.has(row.actor_id) || row.actor_id === row.subject;
  const checkpoint = await coveringCheckpoint(store, row.seq);

  // INDEPENDENT VERIFIABILITY (review FIX 3). The preimage contains actor_id and
  // the WHOLE payload, so it may be served only when this response redacted
  // NEITHER. That condition is COMPUTED, not hand-listed, so a type can never
  // drift into preimage-serving by someone editing a list: today it holds for
  // edition_set_anchored (job-label actor, wholly public payload — already served
  // by /editions/anchors/:week) and for release_registered (the "passkey" actor
  // label, payload wholly served by /release/:id). Every other allowlisted type
  // fails it — passport_created/verification_granted/warning_* all carry an
  // operator EMAIL as actor_id, and account_added/verification_granted/
  // warning_retracted withhold payload fields — so they are honestly labelled
  // linkage_only instead of implying a check the reader cannot run.
  const payloadComplete = Object.keys(payload).every((k) => fields.includes(k));
  const preimage =
    actorOk && payloadComplete
      ? eventPreimage({
          event_id: row.event_id,
          event_type: row.event_type as TrustEventEnvelope["event_type"],
          subject: row.subject,
          actor_kind: row.actor_kind as TrustEventEnvelope["actor_kind"],
          actor_id: row.actor_id,
          created_at: row.created_at,
          payload,
          supersedes: parseSupersedes(row.supersedes),
          prev_hash: row.prev_hash,
        })
      : null;

  return {
    event_id: row.event_id,
    event_type: row.event_type,
    subject: row.subject,
    actor_kind: row.actor_kind,
    ...(actorOk ? { actor_id: row.actor_id } : {}),
    ts: row.created_at,
    payload: projected,
    ...(nameCommitment !== null ? { name_commitment: nameCommitment } : {}),
    prev_hash: row.prev_hash,
    this_hash: row.event_hash,
    ...(preimage === null ? { verification: "linkage_only" as const, note: LINKAGE_ONLY_NOTE } : { verification: "preimage" as const, preimage }),
    prev_event_hash: prev?.event_hash ?? null,
    next_event_prev_hash: next?.prev_hash ?? null,
    ...(checkpoint ? { checkpoint } : {}),
  };
}

 * `escalated` IS NOT ONLY A REVIEWER'S ACT. No FILING can reach it, which is
 * what the paragraph was actually arguing — but `face-triage.ts` sets it from
 * the actor `system:face-triage` after an auto-draft, so an item can enter this
 * population with no human having opened it. That is why the console's headline
 * no longer says "a person at Dhwaja is looking at". The face sweep is
 * unprovisioned today; the sentence has to be right before that flips.
 *
 * THE CATEGORY HALF IS FILER-REACHABLE. `fileReport` inserts a reporter row with
 * no gate on the item's state, so a fan can add an `impersonation` reporter row
 * to an item a reviewer escalated for some other reason — a `dead_code` report
 * on a lookalike handle — and flip this EXISTS without any reviewer having seen
 * an impersonation allegation. It still takes an operator escalating that item
 * first, so it is a far narrower lever than the unfiltered count this replaced;
 * it is not the closed door the paragraph implied.
 *
 * ONE DEFINITION, TWO SURFACES. `digest.ts` counted the same population with
 * neither filter and mailed the raw number to every claimed creator weekly —
 * so the careful new surface would have sat beside a careless old one answering
 * the same credential. Both now read `escalatedImpersonationReports`.

/**
 * The shared predicate. `escalated` only, impersonation only, and scoped to the
 * passport being asked about — the three things that make the answer mean
 * "a person at Dhwaja is looking at this" rather than "somebody typed a URL".
 */
const ESCALATED_IMPERSONATION =
  `FROM fan_report_items i WHERE i.suspected_passport_id = ? AND i.state = 'escalated' ` +
  `AND EXISTS (SELECT 1 FROM fan_report_reporters r WHERE r.item_id = i.item_id AND r.category = 'impersonation')`;

/** Is at least one triaged impersonation report naming this passport with a
 * reviewer right now? Indexed by 0045; `LIMIT 1` because the answer is a
 * boolean and counting past one computes a number this product has decided not
 * to publish. */
export async function underReview(db: DbLike, passportId: string): Promise<boolean> {
  const row = await db.prepare(`SELECT 1 AS x ${ESCALATED_IMPERSONATION} LIMIT 1`).bind(passportId).first<{ x: number }>();
  return row !== null;
}

/** The same population, counted — for the digest, which has always reported a
 * number and whose defect was WHICH number rather than that it was one.
 * I3-1145 (D-0913(3)): only the items ESCALATED inside [startIso, endIso) — the escalation instant is the
 * `review_queue` row the escalation itself writes (`reports.ts` escalate: same batch, `created_at` = the
 * flip's `ts`, joined on `review_item_id`), so a standing report is counted on the day it was escalated and
 * not again. `underReview` above keeps the all-time predicate. */
export async function escalatedImpersonationReports(db: DbLike, passportId: string, startIso: string, endIso: string): Promise<number> {
  const row = await db
    .prepare(`SELECT COUNT(*) AS n ${ESCALATED_IMPERSONATION} AND EXISTS (SELECT 1 FROM review_queue q WHERE q.item_id = i.review_item_id AND q.created_at >= ? AND q.created_at < ?)`)
    .bind(passportId, startIso, endIso)
    .first<{ n: number }>();
  return row?.n ?? 0;
}

/**
 * WHY DHWAJA THINKS A REPORTED ACCOUNT IS ABOUT THIS CREATOR — the fact both
 * creator-facing surfaces asserted and never qualified.
 *
 * `under_review` here and `new_reports_naming_you` in `digest.ts` are scoped by
 * `suspected_passport_id`, and THREE code paths set that column (a fourth only
 * ever clears it), between them producing claims of very different weight:
 *
 *   - an operator's own identification, backfilled by `warnings.ts` from a
 *     warning draft's `protected_passport_id`;
 *   - the handle or domain matcher on a strong kind — equal after
 *     `lookalikeNormalize` collapses padding and punctuation, a bounded typo, a
 *     TLD swap, or a confusable rewrite — all of which `reports.ts` calls claims
 *     about IDENTITY;
 *   - the image check, whose pointer is stamped at BOTH of its tiers (the
 *     `COALESCE` runs before the `match_medium` return), so the tier has to be
 *     read off `face_state` rather than assumed from the pointer's presence;
 *   - a bounded CONTAINMENT either way round, which `reports.ts` describes as a
 *     case where "a fan page, a shop and an impostor are lexically identical".
 *
 * All of them produced the same sentence on the creator's screen and the same
 * clause in their email. A creator could not tell "a person here decided this is
 * you" from "a machine saw a resemblance it cannot interpret", which is the
 * difference between a finding and a guess.
 *
 * WHAT THIS IS NOT. It does not say which account, how many, or on what
 * platform — those decisions stand and are argued above. It answers one
 * question the surfaces were already implying an answer to: on what did the
 * naming rest.
 */
export type NamingBasis = "operator" | "identity" | "likeness" | "resemblance";

/**
 * Rank order, weakest first; the array index plus one is the rank the SQL below
 * emits. `MAX()` over the population takes the STRONGEST reason any one
 * escalated item names this creator, and the direction is the security
 * property: a filer who adds a weak item can never talk the answer DOWN, and
 * cannot reach `operator` at all, because only a reviewer writes that pointer.
 */
const BASIS_BY_RANK: readonly NamingBasis[] = ["resemblance", "likeness", "identity", "operator"];

/**
 * Classify ONE item, in SQL, so the whole population costs one query.
 *
 * `IS` / `IS NOT` rather than `=` / `<>` on purpose: `json_extract` returns NULL
 * for an absent path, and SQLite's null-safe comparison is what lets "the
 * evidence carries no pointer at all" fall into the OPERATOR branch instead of
 * evaluating to NULL on every arm and dropping through to the weakest one.
 * Getting that backwards would report an operator's finding as a machine's
 * guess.
 *
 * The operator branch is stated as an ABSENCE: no matcher in the evidence bundle
 * corroborates the column, so the pointer came from the one path that writes it
 * without evidence — `warnings.ts`'s backfill from a warning draft's
 * `protected_passport_id`. That inference is sound only while the enumeration of
 * writers holds, and it is the strongest claim this product makes to a creator,
 * so the enumeration is PINNED rather than asserted here:
 * `tests/invariants/suspected-pointer-writers.test.ts`.
 *
 * `caution.ts` runs a corroboration test of the same family before publishing,
 * but it is NOT the same test and the difference matters in both directions: it
 * checks the pointer on `handle_similarity` only, takes `domain_similarity` on
 * its kind without checking whose it is, never reads `$.face`, and its
 * `CAUTION_STRONG_KINDS` excludes `edit_distance` — which this classifier ranks
 * as identity. A creator can therefore be told a handle is a near-miss of theirs
 * on evidence the product refuses to publish a public caution from. That is the
 * intended asymmetry (telling the person named is not accusing anybody), and it
 * is written down here so the two are not mistaken for one rule.
 *
 * The final branch is deliberately a catch-all rather than a list of the weak
 * kinds. An evidence bundle carrying a kind this code has never heard of is a
 * bundle written by a future matcher, and the safe reading of an unknown claim
 * is the weakest one on offer — which is why the copy for that rank describes a
 * CONFIDENCE and never a mechanism. It has four members and only one of them is
 * the containment the first draft's wording asserted.
 *
 * A BUNDLE THAT IS NOT A JSON OBJECT CONTRIBUTES NOTHING, and the arm is first
 * because all three of its cases are traps. `json_extract` raises on a column
 * that is not JSON, and one such row used to take the WHOLE aggregate down with
 * it — a creator with nine good rows and one bad one lost the basis for all ten.
 * A bundle that is a valid JSON ARRAY answers NULL to every path below, which
 * falls into the operator arm and mints a human vouch out of a data-loss bug
 * (`face-triage.ts` writes `[]` back if it is ever handed an array, because
 * `JSON.stringify` drops the `.face` property it just set on one). And an
 * explicit SQL NULL would slip past a bare `json_valid(...) = 0` test, which
 * evaluates to NULL rather than true — unreachable while `0010`'s
 * `evidence TEXT NOT NULL DEFAULT '{}'` holds, and tested for rather than
 * leaning on a schema line three migrations away. `MAX` skips NULLs, so an
 * unclassifiable row abstains and its neighbours still speak.
 */
const BASIS_RANK_SQL = `CASE
    WHEN i.evidence IS NULL OR json_valid(i.evidence) = 0 OR json_type(i.evidence) <> 'object' THEN NULL
    WHEN json_extract(i.evidence, '$.handle_similarity.suspected_passport_id') IS NOT i.suspected_passport_id
     AND json_extract(i.evidence, '$.domain_similarity.suspected_passport_id') IS NOT i.suspected_passport_id
     AND json_extract(i.evidence, '$.face.matched_passport_id') IS NOT i.suspected_passport_id THEN 4
    WHEN json_extract(i.evidence, '$.handle_similarity.suspected_passport_id') IS i.suspected_passport_id
     AND json_extract(i.evidence, '$.handle_similarity.kind') IN ('exact_lookalike', 'edit_distance') THEN 3
    WHEN json_extract(i.evidence, '$.domain_similarity.suspected_passport_id') IS i.suspected_passport_id
     AND json_extract(i.evidence, '$.domain_similarity.kind') IN ('homoglyph', 'exact_lookalike', 'edit_distance') THEN 3
    WHEN json_extract(i.evidence, '$.face.matched_passport_id') IS i.suspected_passport_id
     AND i.face_state = 'match_high' THEN 2
    ELSE 1
  END`;

/**
 * The strongest basis on which any escalated impersonation item names this
 * passport, or null.
 *
 * NULL MEANS "NO CLAIM", and it means it for two different reasons that the
 * callers treat identically on purpose: there is nothing under review, or every
 * row in the population abstained. A read that cannot tell a creator why they
 * were named falls back to the unqualified wording that shipped before this —
 * never to a confident one, and never to an error on a screen whose job is to be
 * calm.
 *
 * The `try/catch` is a BACKSTOP, not the primary mechanism, and that changed
 * during review: the first cut let `json_extract` throw on a malformed column
 * and caught it here, which silenced the basis for every OTHER row the creator
 * had. The `json_type` arm now abstains per row, so this catch covers only what
 * is left — a query failure with no per-row reading.
 */
export async function namingBasis(db: DbLike, passportId: string): Promise<NamingBasis | null> {
  try {
    const row = await db
      .prepare(`SELECT MAX(${BASIS_RANK_SQL}) AS r ${ESCALATED_IMPERSONATION}`)
      .bind(passportId)
      .first<{ r: number | null }>();
    const rank = row?.r ?? null;
    if (rank === null) return null;
    return BASIS_BY_RANK[rank - 1] ?? null;
  } catch {
    return null;
  }
}

export interface ProtectionState {
  /**
   * A triaged impersonation report naming this passport is with a reviewer.

   * A BOOLEAN, not a count (decided 2026-08-30): a tally is a number that goes
   * up, and that is pressure to go hunting on platforms Dhwaja does not control.
   * `escalated`-only is what makes it a true statement as well as a coarse one —
   * see the header.

  under_review: boolean;
  /**

   * reviewed — see `NamingBasis`. Null when nothing is under review, and also
   * null when the basis could not be established, which the console renders as
   * the unqualified sentence it rendered before this field existed.
   *
   * It is deliberately ONE value for the whole population rather than one per
   * item: a list would be a count, and a count is the lever `under_review` is a
   * boolean to avoid.
   */
  review_basis: NamingBasis | null;
  /** Published warnings actively protecting this creator. Adjudicated, public on
   * the impostor's own registry entry, and already counted on their ROI screen —
   * a number here reveals nothing new. */
  warnings_active: number;
  /**

   * that is not verified, is retracted, or is suppressed, and `creatorId` checks
   * none of those — so the first cut handed a suppressed or merely-claimed
   * creator a link to a page that does not exist.

  psa_path: string | null;

  notifications_paused: boolean;
  /** I3-1997 — `claim_state = 'claimed'` at the read's instant (not retracted,
   * not suppressed), whether or not a warning protects it yet, so the creator
   * learns before an emergency that the PSA page (verified-only, public 404
   * unchanged) becomes available once the passport is verified. */
  psa_needs_verification: boolean;

/** W312-R2-8: THE passport-keyed "not suppressed" predicate for `passports p`,
 * one const at both `protectionState` doors (servable, claimedOnly). Per
 * passport, not per account key (`seeding.ts suppressedKeyStmt` is per key). */
const PASSPORT_UNSUPPRESSED = "NOT EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = p.passport_id)";

export async function protectionState(db: DbLike, passportId: string, nowIso: string = new Date().toISOString()): Promise<ProtectionState> {
  const reviewing = await underReview(db, passportId);
  // Asked only when there is something to explain. `namingBasis` answers null
  // for an empty population anyway, but the read is a second scan of the same
  // rows and this route is the one a merely-claimed creator may poll.
  const basis = reviewing ? await namingBasis(db, passportId) : null;
  const p = await protectionSummary(db, passportId);
  // The three conditions `/p/:id/psa` itself applies, asked here rather than
  // published as a link that 404s. I3-1768 (D-1085(1), I34-EXPIRY-A): the claim state is the one /psa serves through,
  // at `nowIso` (standing.ts CLAIM_STATE_AT_SQL), so a lapsed grant moves from servable to claimedOnly together.
  const servable = await db
    .prepare(
      `SELECT 1 AS x FROM passports p WHERE p.passport_id = ? AND ${CLAIM_STATE_AT_SQL} = 'verified' AND p.retracted = 0
        AND ${PASSPORT_UNSUPPRESSED}`,
    )
    .bind(passportId, nowIso)
    .first<{ x: number }>();
  const claimedOnly = await db
    .prepare(
      `SELECT 1 AS x FROM passports p WHERE p.passport_id = ? AND ${CLAIM_STATE_AT_SQL} = 'claimed' AND p.retracted = 0
        AND ${PASSPORT_UNSUPPRESSED}`,
    )
    .bind(passportId, nowIso)
    .first<{ x: number }>();
  const paused = await db
    .prepare("SELECT 1 AS x FROM creator_contacts WHERE passport_id = ? AND (digests_enabled = 0 OR digest_email_enabled = 0)")
    .bind(passportId)
    .first<{ x: number }>();
  return {
    under_review: reviewing,
    review_basis: basis,
    warnings_active: p.active_warnings,
    psa_path: p.protected && servable !== null ? `/p/${passportId}/psa` : null,
    notifications_paused: paused !== null && paused !== undefined,
    psa_needs_verification: claimedOnly !== null && claimedOnly !== undefined,
  };
}

/**

 * ledger `actor_kind` to the four provenance classes the product cares about.

 * This is a LEAF module (no local imports) so it can be shared by BOTH the live

 * invariant). The `added_by` column on `passport_accounts` stores this value.

export type TouchedBy = "creator" | "agency" | "operator" | "system";

/** Map a ledger `actor_kind` to its provenance class. A `creator_passkey` mark is
 * unforgeable (only `appendCreatorSigned` writes it); everything unrecognized
 * folds to `operator` (the conservative, least-privileged-looking class). */
export function touchedByFromActor(actorKind: string): TouchedBy {
  switch (actorKind) {
    case "creator_passkey":
      return "creator";
    case "agency":
      return "agency";
    case "system":
      return "system";

      return "operator";
  }
}

import { absHttps, accountDisplay, type PassportView, kindDisclosure } from "./passport-view.js";
import { translator, type MessageKey } from "@dhwaja/i18n";

/** The bound translator this page threads through its row renderer. */
type Translate = (key: MessageKey, vars?: Readonly<Record<string, string>>) => string;

/**
 * S67 — the impersonator PSA (#12). Decision locked 2026-08-10: the PSA
 * points to the REAL account only — it never names, links, or hints at the
 * fake. That makes it a pure redirect-to-safety artifact the creator can post
 * to their own audience ("someone is impersonating me — THIS is my real,
 * verified account"), with almost no defamation surface: the only accusatory
 * content is the generic, fixed sentence that a warning exists.
 *
 * Lifecycle is derived from the ledger on EVERY request (the derive-from-log
 * pattern): the PSA exists iff a currently-published (not retracted)
 * impersonation warning carries this passport as its protected creator AND
 * the passport is verified + live right now. Retract the warning (or downgrade
 * the creator) and the PSA link dies on the next request — shares circulating
 * outside Dhwaja go 404 rather than keep accusing.
 *
 * ALL COPY IS FIXED CODE — like the warnings schema, there is nowhere to put
 * free text, so a bad day cannot produce a bad public sentence.
 */

export interface PsaPageOpts {
  /** Inline foil card SVG (trusted renderer output), or null. */
  cardSvg?: string | null;
  /** Absolute canonical URL of the passport page (from configured env, never
   * the Host header) — both the og:url base and the link target. */
  passportUrl?: string | null;
  /** Absolute APP_ORIGIN deep link into the interactive SPA (route-derived and
   * already validated). Absent → the shell's hand-off link is omitted. */
  appHref?: string | null;
  /** W2 — the reader's language, negotiated from `Accept-Language` by the
   * route. Absent → `en`, i.e. exactly what this page rendered before. */
  locale?: string;
}

/** One proven account, as a row of the "only real accounts" table. Same
 * exposure as `/p/:id` — platform, handle, provenance — and nothing else.
 *
 * The status word and the inert row are NOT decoration (review HIGH 3): a
 * non-active row must never read as "this is them" on the page a fan pastes
 * under a live scam. `/p/:id` and `/is/:platform/:handle` render it the same
 * way; this row used to be the odd one out, emitting neither. */
function accountRow(a: PassportView["accounts"][number], m: Translate): string {
  const mark =
    a.touched_by === "creator"
      ? `<span class="mark creator">${esc(m("mark.by_creator"))}</span>`
      : a.touched_by === "agency"
        ? `<span class="mark agency">${esc(m("mark.by_agency"))}</span>`
        : `<span class="mark none">${esc(m("mark.listed_by_dhwaja"))}</span>`;
  const state = a.status !== "active" ? ` <span class="m">(${esc(a.status)})</span>` : "";
  // PU-01 (S-070): the CONFIRMATION axis, orthogonal to `touched_by`. `mark`
  // above says WHO asserted the row; it says nothing about whether anybody ever
  // demonstrated control of it. This page's own heading presents the table as
  // the real accounts, so an unconfirmed row rendered with provenance alone
  // received a STRONGER assurance here than on either surface that already
  // qualifies it — `/p/:id` (`accountMarks` in passport-view.ts) and the SPA
  // Links panel (apps/web/src/views.ts, C-523). Same catalogue strings as
  // `/p/:id`, so the three public views cannot invent three wordings for one
  // fact, and `tabindex="0"` for the same reason `accountMarks` carries it
  // (M-24): a `title` on a non-focusable span is unreachable by keyboard, and
  // this page ships no script to offer an alternative.
  const confirm = a.confirmed
    ? ""
    : ` <span class="mark unconfirmed" tabindex="0" title="${esc(m("pv.unconfirmed_title"))}">${esc(m("pv.unconfirmed"))}</span>`;
  // ONE RULE FOR PRINTING AN ACCOUNT (passport-view.ts:accountDisplay) — a
  // Tier-2 `link` row's handle is a full URL and must render as "web link" +
  // bare URL, never `@<url>`, which this row used to print unchecked.
  const disp = accountDisplay(a.platform, a.handle);
  return `<tr${a.status !== "active" ? ' class="inert"' : ""}><td>${esc(disp.platform)}</td><td><code>${esc(disp.ident)}</code>${state}</td><td>${mark}${confirm}</td></tr>`;
}

/**
 * The shareable PSA page. Screenshot-native like the S64 share landing:
 * warning banner + "this is the real X" + the foil card + the proven-accounts
 * link. NOTHING about the impersonator appears — no handle, no hash, no
 * platform. It is the most-shared page in the set (it gets pasted into replies
 * under a live scam), so it wears the same script-free, remote-free shell and
 * has to render intact on a stranger's locked-down browser.
 */
export function renderPsaHtml(v: PassportView, opts: PsaPageOpts = {}): string {
  // W2 — every fixed sentence on this page comes from the catalogue, and the
  // creator's NAME is a variable rather than a concatenation, so a translator
  // can put it where their language puts it. Defaulted to `en`, so a caller
  // that does not negotiate renders byte-for-byte what it always did.
  const m = translator(opts.locale ?? "en");
  // The scheme+charset gate `seo-page.ts` and `discovery-page.ts` already apply
  // (review LOW 7): `esc()` makes a value inert as markup but does nothing about
  // `javascript:` in an href, and this is the most-pasted page in the set.
  const passportUrl = absHttps(opts.passportUrl);
  const appHref = absHttps(opts.appHref, "#");
  const title = m("psa.title", { name: v.display_name });
  const description = m("psa.description", { name: v.display_name });
  const og = [
    `<meta property="og:title" content="${esc(title)}">`,
    `<meta property="og:description" content="${esc(description)}">`,
    `<meta property="og:type" content="website">`,
    // A noindex page must not also ask to be ranked/shared under a canonical
    // URL (ssr-shell.test.ts "noindex pages carry no canonical and no og:url").
    // This page is always noindex,nofollow (S67), so og:url is never emitted.
    `<meta name="twitter:card" content="summary">`,
    `<meta name="twitter:title" content="${esc(title)}">`,
    `<meta name="twitter:description" content="${esc(description)}">`,
  ]
    .filter(Boolean)
    .join("\n");
  // #152 D-59: the same scope caption `/p/:id` (passport-view.ts) and
  // `/is/:platform/:handle` (seo-page.ts) put under the card — a card that
  // travels in a PSA says what it is, like the other two doors.
  const card = opts.cardSvg ? `<figure class="cardcol">${opts.cardSvg}<figcaption class="cardcap">${esc(m("pv.card_caption"))}</figcaption></figure>` : "";
  const link = passportUrl ? `<a class="linkcta" href="${esc(passportUrl)}">${esc(m("psa.link_cta"))}</a>` : "";
  // "The accounts below are the ones Dhwaja verified as the real X" is only true
  // of the ACTIVE ones. `getPassportView` returns every status, and a revoked
  // handle is revoked precisely BECAUSE it was sold, recycled or taken over —
  // i.e. it is now the most likely thing an impersonator is contacting people
  // from. So it never appears in the "real accounts" table; it appears below it,
  // under its own fixed heading, as a thing that is no longer theirs.
  // ONE derivation of the disclosure, shared with /p/:id and /is/:platform/:handle
  // (`kindDisclosure` in passport-view.ts). The PSA is the page a creator sends
  // fans TO when an impostor is active, so it is the last surface that can
  // afford to be the one that forgot to say what the subject is.
  const kind = kindDisclosure(v.subject_kind, m);
  const live = v.accounts.filter((a) => a.status === "active");
  const past = v.accounts.filter((a) => a.status !== "active");
  const cols = `<tr><th scope="col">${esc(m("col.platform"))}</th><th scope="col">${esc(m("col.account"))}</th><th scope="col">${esc(m("col.asserted"))}</th></tr>`;
  const rows = live.map((a) => accountRow(a, m)).join("");
  const pastBlock = past.length
    ? `<h2>${esc(m("psa.past_heading", { name: v.display_name }))}</h2>
<p class="honest">${esc(m("psa.past_note"))}</p>
<table>${cols}${past.map((a) => accountRow(a, m)).join("")}</table>`
    : "";
  // THE EMPHASIS IS APPLIED HERE, NOT INSIDE THE STRING. A catalogue value is
  // plain text and is escaped — a translator supplies words, and a word that can
  // carry markup can carry a `<script>` onto the most-pasted page in the set. So
  // the two sentences that used to embed `<strong>` mid-clause now wear it whole.
  const body = `<p class="warnbar"><strong>${esc(m("psa.warnbar", { name: v.display_name }))}</strong></p>
<div class="top">
<div class="idcol">
<p class="eyebrow">${esc(m("psa.eyebrow"))}</p>
<h1>${esc(m("psa.heading", { name: v.display_name }))}</h1>
<div class="pillrow">${verdictPill("verified", m("psa.pill"))}${kind.chip}</div>
<p class="honest">${esc(m("psa.honest", { name: v.display_name }))}</p>
${kind.plain ? `<p class="kindnote">${esc(kind.plain)}</p>` : ""}
</div>
${card}
</div>
<h2>${esc(m("psa.accounts_heading"))}</h2>
<table>${cols}${rows}</table>

import type { DeviceTarget, PushOutcome, PushPayload, PushSender } from "./alerts.js";
import { settingValue } from "./env-config.js";
import { logError } from "./log.js";

/**
 * Real push delivery — APNs (Apple) + FCM (Google) adapters behind the stable
 * `PushSender` seam that `alerts.deliverPending` already drives. Delivery stays
 * best-effort and retryable: a non-2xx or a transport error returns false, and
 * the notification row stays undelivered for the next pass. No push provider is
 * ever the trust boundary — a notification can only ever restate a ledger event.
 *
 * Auth model (both providers mint a short-lived bearer token from an operator
 * secret, cached per-isolate so we sign at most once per ~50 min):
 *  - APNs: an ES256 provider JWT signed with the team's .p8 key (header kid =
 *    key id, claims iss = team id). Apple rejects tokens refreshed more than
 *    once per 20 min, so caching is mandatory, not just an optimization.
 *  - FCM: a service-account RS256 JWT exchanged at Google's OAuth2 token
 *    endpoint for an access token (scope firebase.messaging).
 *
 * Everything network-facing takes an injectable `fetchImpl`/`now` so the crypto
 * (real signatures over a generated keypair) and request shaping are tested
 * headlessly; only the live send needs real credentials + a device.
 */

// ---------------------------------------------------------------------------
// base64 / PEM helpers
// ---------------------------------------------------------------------------

function b64url(bytes: Uint8Array): string {
  let bin = "";
  for (const b of bytes) bin += String.fromCharCode(b);
  return btoa(bin).replace(/\+/g, "-").replace(/\//g, "_").replace(/=+$/, "");
}

function b64urlJson(o: unknown): string {
  return b64url(new TextEncoder().encode(JSON.stringify(o)));
}

/** Decode a PEM private key (PKCS#8) to DER bytes. Tolerates env-mangled keys
 * where real newlines were stored as the two-character sequence "\n". */
export function pemToDer(pem: string): Uint8Array {
  const body = pem
    .replace(/\\n/g, "\n")
    .replace(/-----BEGIN [^-]+-----/, "")
    .replace(/-----END [^-]+-----/, "")
    .replace(/\s+/g, "");
  const bin = atob(body);
  const out = new Uint8Array(bin.length);
  for (let i = 0; i < bin.length; i++) out[i] = bin.charCodeAt(i);
  return out;
}

// ---------------------------------------------------------------------------
// JWT signing (ES256 for APNs, RS256 for FCM)
// ---------------------------------------------------------------------------

type Alg = "ES256" | "RS256";
// `CryptoKey` isn't in this package's TS lib (workerd typings), so derive it.
type SubtleKey = Awaited<ReturnType<typeof crypto.subtle.importKey>>;

const keyCache = new Map<string, Promise<SubtleKey>>();

function importSigningKey(pem: string, alg: Alg): Promise<SubtleKey> {
  const cacheKey = `${alg}:${pem}`;
  let p = keyCache.get(cacheKey);
  if (!p) {
    const der = pemToDer(pem);
    p = crypto.subtle.importKey(
      "pkcs8",
      der,
      alg === "ES256" ? { name: "ECDSA", namedCurve: "P-256" } : { name: "RSASSA-PKCS1-v1_5", hash: "SHA-256" },
      false,
      ["sign"],
    );
    keyCache.set(cacheKey, p);
  }
  return p;
}

/** Sign a compact JWS (header.payload.signature). ES256 signatures come out of
 * WebCrypto in raw r||s form — exactly what JOSE requires — so no DER unwrap. */
export async function signJwt(pem: string, alg: Alg, header: Record<string, unknown>, claims: Record<string, unknown>): Promise<string> {
  const key = await importSigningKey(pem, alg);
  const signingInput = `${b64urlJson({ alg, typ: "JWT", ...header })}.${b64urlJson(claims)}`;
  const sig = new Uint8Array(
    await crypto.subtle.sign(
      alg === "ES256" ? { name: "ECDSA", hash: "SHA-256" } : { name: "RSASSA-PKCS1-v1_5" },
      key,
      new TextEncoder().encode(signingInput),
    ),
  );
  return `${signingInput}.${b64url(sig)}`;
}

// ---------------------------------------------------------------------------
// Token cache (provider bearer tokens; sign at most once per window)
// ---------------------------------------------------------------------------

const TOKEN_TTL_SEC = 3000; // 50 min — under APNs' 60-min cap and 20-min refresh floor
const tokenCache = new Map<string, { token: string; exp: number }>();

async function cachedToken(cacheKey: string, nowSec: number, mint: () => Promise<{ token: string; ttlSec: number }>): Promise<string> {
  const hit = tokenCache.get(cacheKey);
  if (hit && hit.exp > nowSec) return hit.token;
  const { token, ttlSec } = await mint();
  tokenCache.set(cacheKey, { token, exp: nowSec + Math.min(ttlSec, TOKEN_TTL_SEC) });
  return token;
}

const placeholder = (v: string | undefined): boolean => !v || String(v).trim() === "" || /TODO/i.test(v);

/** One PEM block: BEGIN/END lines with the same label around base64 lines (I47-GFX-CODE G1). */
const PEM_BLOCK_RE = /^-----BEGIN ([A-Z0-9 ]+)-----\n([A-Za-z0-9+/=\n]+)\n-----END \1-----$/;

/** `placeholder` for key material: a TODO inside the base64 body of a well-formed PEM block is key bytes, not a
 * marker, so that block is configured; every other value keeps `placeholder`'s answer. Normalised the way
 * `pemToDer` reads a key (escaped `\n`, CR, surrounding whitespace). */
export function pemPlaceholder(v: string | undefined): boolean {
  if (!placeholder(v)) return false;
  const m = PEM_BLOCK_RE.exec((v ?? "").replace(/\\n/g, "\n").replace(/\r/g, "").trim());
  return m === null || /TODO/i.test(m[1]!);
}

/**
 * Hard ceiling on ONE provider call — the same convention and the same number
 * `mailer.ts` (`MAIL_TIMEOUT_MS`), `verify-fetch.ts` and the face-triage fetch
 * in `index.ts` already use.
 *
 * WHY IT MATTERS MORE HERE THAN ANYWHERE ELSE. Every other outbound call in the
 * codebase happens once per request. These happen in a LOOP, serially, inside
 * the cron: `alerts.deliverPending` awaits one per device per undelivered
 * notification. With no timeout, the ceiling on that loop was whatever the
 * platform eventually enforced — and `cron-lease.ts` picked its 30-minute TTL on
 * the STATED ASSUMPTION that it "sits far above a real sweep", with nothing
 * enforcing that and no fencing token to make an overrun safe. At 3 s a request,
 * 1000 notifications is fifty minutes: the lease lapses at minute thirty, the
 * next tick acquires it and starts a SECOND sweep, and both hand the same rows
 * to APNs/FCM. That is precisely the duplicate push `cron-lease.ts`'s own header
 * says the lease exists to prevent, produced by the lease's own TTL.
 *
 * Three bounds now stand between a hung provider and that outcome: this
 * per-request deadline, `alerts.DELIVERY_BUDGET_MS` over the whole leg, and
 * `scheduled()` re-asserting the lease between legs so an overrun run STOPS
 * instead of racing its successor. A timeout is an ordinary transient failure —
 * it returns the existing "retry" outcome, so the notification stays pending and
 * the next pass tries again. There are deliberately no retries inside one call,
 * for the same reason mailer.ts gives: retrying a push is amplification.
 */
export const PUSH_TIMEOUT_MS = 10_000;

/**
 * `fetch` with a deadline. Aborting is what makes the deadline real — without
 * the AbortController the request keeps running and keeps holding the
 * invocation open after we have stopped waiting for it. Never throws for the
 * timeout itself; the caller's own catch treats every failure as "retry".
 */
function fetchWithDeadline<T>(fetchImpl: typeof fetch, url: string, init: RequestInit, read: (res: Response) => Promise<T>, timeoutMs?: number): Promise<T>;
// The implementation keeps the literal `async function fetchWithDeadline(` that
// tests/invariants/evidence-scan-gate.test.ts reads to confirm the wrapper forces
// `redirect: "manual"`; the generic is carried by the overload above.
async function fetchWithDeadline(fetchImpl: typeof fetch, url: string, init: RequestInit, read: (res: Response) => Promise<unknown>, timeoutMs = PUSH_TIMEOUT_MS): Promise<unknown> {
  // I3-1255 c (I22): THE DEADLINE COVERS THE BODY, NOT ONLY THE HEADERS. The
  // timer used to be cleared as soon as `fetch` resolved, and every caller then
  // read the body (`res.json()`, `res.clone().text()`) with no bound at all, so
  // a provider that sent headers and stalled held the delivery leg open. The
  // caller's read now runs INSIDE the deadline, and the race rejects even when
  // the stalled reader ignores the abort.
  const ctl = new AbortController();
  let timer: ReturnType<typeof setTimeout> | undefined;
  const expired = new Promise<never>((_, reject) => {
    timer = setTimeout(() => {
      ctl.abort();
      reject(new Error("push deadline exceeded"));
    }, timeoutMs);
  });
  try {
    return await Promise.race([fetchImpl(url, { ...init, redirect: "manual", signal: ctl.signal }).then(read), expired]);
  } finally {
    clearTimeout(timer);
  }
}

/** Build APNs config from env, or null (→ APNs disabled) if any secret is
 * missing/placeholder. Never fabricates a half-configured sender. */
export function apnsConfigFromEnv(env: Record<string, unknown>): ApnsConfig | null {
  // I63 C2 (I62-CC-05): each setting through settingValue (padding trimmed, blank or TODO unset); the key keeps pemPlaceholder.
  const keyId = settingValue(env.APNS_KEY_ID);
  const teamId = settingValue(env.APNS_TEAM_ID);
  const bundleId = settingValue(env.APNS_BUNDLE_ID);
  const p8Pem = env.APNS_P8 as string | undefined;
  if (keyId === null || teamId === null || bundleId === null || pemPlaceholder(p8Pem)) return null;
  // H-03: APNS_HOST was accepted verbatim with no shape check — a typo or a
  // stray sandbox host (the DEPLOY.md misconfig) would silently route every
  // device request. Fail closed (same as the other four vars) on anything
  // that is not one of the two hosts Apple actually publishes.
  const rawHost = env.APNS_HOST as string | undefined;

  return { keyId, teamId, bundleId, p8Pem: p8Pem!, host };
}

export function fcmConfigFromEnv(env: Record<string, unknown>): FcmConfig | null {

  const projectId = settingValue(env.FCM_PROJECT_ID);

/** Mint (or reuse) an FCM access token via the service-account JWT-bearer
 * grant. Cached per client email. Returns "" if the exchange fails (caller
 * treats a falsy token as "provider unavailable" and leaves the alert pending). */
export async function fcmAccessToken(cfg: FcmConfig, nowSec: number, fetchImpl: typeof fetch): Promise<string> {
  return cachedToken(`fcm:${cfg.clientEmail}`, nowSec, async () => {
    const assertion = await signJwt(cfg.privateKeyPem, "RS256", {}, {
      iss: cfg.clientEmail,
      scope: "https://www.googleapis.com/auth/firebase.messaging",
      aud: cfg.tokenUri,
      iat: nowSec,
      exp: nowSec + 3600,
    });
    // Deadlined like the sends themselves: the token exchange is the FIRST
    // outbound hop of an FCM delivery, so a hung token endpoint stalls the whole
    // leg exactly as a hung send would. A throw here is caught by the caller and
    // read as "provider unavailable" → retry next pass.
    const j = await fetchWithDeadline(fetchImpl, cfg.tokenUri, {
      method: "POST",
      headers: { "content-type": "application/x-www-form-urlencoded" },
      body: `grant_type=${encodeURIComponent("urn:ietf:params:oauth:grant-type:jwt-bearer")}&assertion=${encodeURIComponent(assertion)}`,
    }, async (res) => {
      if (!res.ok) throw new Error(`fcm token ${res.status}`);
      return (await res.json()) as { access_token?: string; expires_in?: number };
    });
    if (!j.access_token) throw new Error("fcm token missing");
    return { token: j.access_token, ttlSec: Math.max(60, (j.expires_in ?? 3600) - 60) };
  });
}

/** The FCM v1 message body. `data` values MUST be strings. */
export function buildFcmMessage(device: DeviceTarget, payload: PushPayload): Record<string, unknown> {
  return {
    message: {
      token: device.push_token,
      // Same rule as APNs: a fixed, name-free line and the kind, never the
      // stored summary or the passport id. See alerts.ts PushPayload.
      notification: { title: "Dhwaja", body: payload.notice },
      data: { kind: payload.kind },
    },
  };
}

/**
 * Build a PushSender routing by device platform, using whichever providers are
 * configured. An unconfigured provider (or a token/exchange failure) returns
 * false for its devices — the notification stays pending and retries once the
 * secret is supplied. Never throws to the caller (deliverPending also guards).
 */
export function pushSenderFromEnv(env: Record<string, unknown>, fetchImpl: typeof fetch = fetch, now: () => number = () => Date.now()): PushSender {
  const apns = apnsConfigFromEnv(env);
  const fcm = fcmConfigFromEnv(env);

  return async (device: DeviceTarget, payload: PushPayload): Promise<PushOutcome> => {
    const nowSec = Math.floor(now() / 1000);
    try {
      if (device.platform === "apns") {
        if (!apns) return "retry";
        const token = await apnsProviderToken(apns, nowSec);
        const req = buildApnsRequest(apns, device, payload, token);
        const res = await fetchWithDeadline(fetchImpl, req.url, { method: "POST", headers: req.headers, body: req.body }, async (r) => ({
          ok: r.ok,
          status: r.status,
          // Read only where it is consulted (a non-429 4xx), and inside the deadline.
          reason: !r.ok && r.status >= 400 && r.status < 500 && r.status !== 429 ? await r.clone().text().then((t) => t).catch(() => "") : "",
        }));
        if (res.ok) return "ok";
        // H6: APNs signals a permanently-dead token with 410 Gone, or a 4xx whose
        // JSON `reason` is BadDeviceToken / Unregistered / DeviceTokenNotForTopic.
        // Everything else (429, 5xx, network) is transient → retry.
        if (res.status === 410) return "dead";
        if (res.status >= 400 && res.status < 500 && res.status !== 429) {
          const reason = res.reason;
          if (/BadDeviceToken|Unregistered|DeviceTokenNotForTopic|BadDeviceTokenSize/.test(reason)) return "dead";
          // I3-0819: a misrotated/revoked provider key (not a device problem)
          // comes back as 403 InvalidProviderToken/InvalidProviderKey. That is
          // neither "dead" (the token is fine) nor a per-attempt transient like
          // 429/5xx: it will keep failing every retry until an operator rotates
          // the key, and PushOutcome (alerts.ts) has no room for a fourth,
          // config-error outcome, so the queued notification is left correctly
          // pending ("retry") but the stall must not be silent — surface it so
          // it pages instead of looking like ordinary backpressure.
          if (/InvalidProviderToken|InvalidProviderKey/.test(reason)) {
            logError({ cid: "unknown", hop: "cron", msg: "apns provider credential rejected — rotation needed, notifications stalled", status: res.status });
          }
        }
        return "retry";
      }
      if (device.platform === "fcm") {
        if (!fcm) return "retry";
        const token = await fcmAccessToken(fcm, nowSec, fetchImpl);
        if (!token) return "retry";
        const res = await fetchWithDeadline(fetchImpl, `https://fcm.googleapis.com/v1/projects/${fcm.projectId}/messages:send`, {
          method: "POST",
          headers: { authorization: `Bearer ${token}`, "content-type": "application/json" },
          body: JSON.stringify(buildFcmMessage(device, payload)),
        }, async (r) => ({ ok: r.ok, status: r.status, body: r.status === 400 ? await r.clone().text().then((t) => t).catch(() => "") : "" }));
        if (res.ok) return "ok";
        // H6: FCM signals a dead token with 404 NOT_FOUND, or a 400 whose error
        // status is UNREGISTERED / INVALID_ARGUMENT on the registration token.
        if (res.status === 404) return "dead";
        if (res.status === 400) {
          const body = res.body;
          // I3-1330: FCM v1 also returns 400 INVALID_ARGUMENT for a malformed
          // request that has nothing to do with the token (bad message field,
          // oversized payload, bad TTL, ...). A bare substring match on
          // INVALID_ARGUMENT deleted a healthy device registration on any such
          // error. Only treat it as a dead token when the error also names the
          // registration token, matching the message FCM actually sends for a
          // token-shaped 400 ("The registration token is not a valid FCM
          // registration token").
          if (/UNREGISTERED|registration-token-not-registered/.test(body)) return "dead";
          if (/INVALID_ARGUMENT/.test(body) && /registration token/i.test(body)) return "dead";
        }
        return "retry";
      }
    } catch {
      // Token mint / transport hiccup — leave undelivered, retry next pass.
      return "retry";
    }
    return "retry";
  };
}

/**
 * Send a SILENT (background/data) push carrying a device-attach challenge code
 * to one device, and report success. This is the proof-of-possession channel:
 * only whoever actually holds the token receives the code. Routes by platform;
 * self-disables (false) if the provider isn't configured. Never throws.
 */
export async function sendChallengePush(env: Record<string, unknown>, device: DeviceTarget, code: string, fetchImpl: typeof fetch = fetch, now: () => number = () => Date.now()): Promise<boolean> {
  const nowSec = Math.floor(now() / 1000);
  try {
    if (device.platform === "apns") {
      const cfg = apnsConfigFromEnv(env);
      if (!cfg) return false;
      if (!/^[0-9a-fA-F]{32,200}$/.test(device.push_token)) return false;
      const token = await apnsProviderToken(cfg, nowSec);
      const res = await fetchWithDeadline(fetchImpl, `https://${cfg.host}/3/device/${device.push_token}`, {
        method: "POST",
        headers: {
          authorization: `bearer ${token}`,
          "apns-topic": cfg.bundleId,
          "apns-push-type": "background", // silent — no alert shown
          "apns-priority": "5",
          "content-type": "application/json",
        },
        body: JSON.stringify({ aps: { "content-available": 1 }, dhwaja_challenge: code }),
      }, async (r) => r.ok);
      return res;
    }
    if (device.platform === "fcm") {
      const cfg = fcmConfigFromEnv(env);
      if (!cfg) return false;
      const token = await fcmAccessToken(cfg, nowSec, fetchImpl);
      if (!token) return false;
      const res = await fetchWithDeadline(fetchImpl, `https://fcm.googleapis.com/v1/projects/${cfg.projectId}/messages:send`, {
        method: "POST",
        headers: { authorization: `Bearer ${token}`, "content-type": "application/json" },
        // data-only message (no `notification`) → delivered silently to the app.
        body: JSON.stringify({ message: { token: device.push_token, data: { dhwaja_challenge: code }, android: { priority: "high" } } }),
      }, async (r) => r.ok);
      return res;
    }
  } catch {
    return false;
  }
  return false;
}

/**
 * I3-0764 (D-0833(4)) — the fallback for a throw no typed arm above it claimed:
 * a storage fault, a bug, an outage is ours — logged, answered 500, never a 400
 * that tells the caller their request was the problem.
 */
function serverFault(c: Context<{ Bindings: Env }>, route: string, e: unknown): Response {
  logError({ cid: crypto.randomUUID(), hop: "http", route, msg: "unexpected route fault", err: e instanceof Error ? e.name : typeof e });
  return c.json({ error: "internal" }, 500);
}


/**
 * ASK THE CREATOR ROUTES (F9, master list #54).
 *
 * THERE IS NO PRIVATE CREATOR→FAN REPLY ROUTE, and its absence is the feature.
 * The creator has exactly two ways to act on a question: publish a signed
 * answer everybody can read, or decline it — and a decline carries no words at
 * all, only a state. Nothing here can carry a message from a creator to one
 * fan, which is precisely what would make this a DM inbox.
 *
 * The answer is the `recognition-routes.ts` two-leg ceremony: `/answer/start`
 * builds the draft SERVER-SIDE from the stored question and binds the challenge
 * to it; `/answer` rebuilds it identically and finishes with the echoed values.
 * The creator supplies a question id and their own words; the question text and
 * its hash come out of the database on both legs, so a client cannot cause an
 * answer to be published against a question the creator never read.
 */
export function makeQuestionRoutes(): Hono<{ Bindings: Env }> {
  const app = new Hono<{ Bindings: Env }>();

  // --- the fan half ---------------------------------------------------------

  app.post("/account/questions", anonBodyLimit, async (c) => {
    const id = await fanId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    // I3-0065 / D-0885(1): this door writes a fan-keyed analytics fact (`fanKey`), the
    // same 503 gate as /account/handshakes (handshake-routes.ts), before any work.
    if (!usableFanSecret(c.env.ANALYTICS_FAN_SECRET)) return c.json({ error: "analytics_unavailable", detail: "asking is temporarily unavailable" }, 503);
    // The per-IP ceiling every fan write carries, and this one puts free text in
    // front of a human, which is the surface audit #18/#19 added it for.
    if (!(await fanIpAllowed(c, "qst", RANKING_WRITES_PER_IP_HOUR))) return c.json(refusal("rate limit exceeded", RATE_LIMIT_MESSAGE), 429);
    const raw = await c.req.text();
    if (raw.length > 4 * 1024) return c.json({ error: "too_large" }, 413);
    const body = ((): { passport_id?: unknown; body?: unknown } | null => {
      try {
        return JSON.parse(raw) as { passport_id?: unknown; body?: unknown };
      } catch {
        return null;
      }
    })();
    if (!body || typeof body.passport_id !== "string") return c.json({ error: "passport_id and body required" }, 400);
    try {
      const q = await askQuestion(d1(c.env.TRUST_DB), id, body.passport_id, body.body, Date.now(), undefined, c.env.ANALYTICS_FAN_SECRET, ...analyticsPublishArgs(c));
      // The meaning of silence travels WITH the receipt, so no client can render
      // the waiting state without the sentence that says it means nothing.
      return c.json({ ok: true, question: { question_id: q.question_id, passport_id: q.passport_id, body: q.body, state: q.state, expires_at: q.expires_at }, no_answer_meaning: NO_ANSWER_MEANING });
    } catch (e) {
      if (e instanceof QuestionError) return c.json(errorBody(e), e.status);
      return serverFault(c, "/account/questions", e);
    }
  });

  app.get("/account/questions", async (c) => {
    const id = await fanId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    return c.json({
      questions: await listOwnQuestions(d1(c.env.TRUST_DB), id),
      question_max: QUESTION_MAX,
      open_limit: MAX_OPEN_PER_FAN,
      no_answer_meaning: NO_ANSWER_MEANING,
      decline_meaning: DECLINE_MEANING,
      // Every outcome a fan can see ships with the sentence that says what it
      // means. This one had none, and it is the only one where their words were
      // published and then removed.
      retract_meaning: RETRACT_MEANING,
    });
  });

  // --- the creator half -----------------------------------------------------

  app.get("/account/question-inbox", async (c) => {
    const id = await creatorId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const db = d1(c.env.TRUST_DB);
    const s = await questionSettings(db, id);
    // The answers page cursor, opaque to the client and validated here: a
    // malformed one reads as the first page rather than an error, because a
    // creator who mangled a URL should land on their answers, not on a refusal.
    const after = c.req.query("after");
    const at = c.req.query("after_at");
    const cursor = typeof after === "string" && QUESTION_ID_RE.test(after) && typeof at === "string" && at.length <= 40 ? { question_id: after, answered_at: at } : null;
    const own = await ownAnswers(db, id, cursor);
    // Q-04: the inbox is paged the same way `own` above already is, with its
    // OWN cursor pair -- `question_id`/`question_at` are unambiguous next to
    // `after`/`after_at` because the client threads them back on the SAME
    // request as a distinct pair.
    const qAfter = c.req.query("question_id");
    const qAt = c.req.query("question_at");
    const inboxCursor = typeof qAfter === "string" && QUESTION_ID_RE.test(qAfter) && typeof qAt === "string" && qAt.length <= 40 ? { question_id: qAfter, created_at: qAt } : null;
    const inbox = await questionInbox(db, id, Date.now(), inboxCursor);
    return c.json({
      questions: inbox.questions,
      questions_next: inbox.next,
      // What they have already published, so the retraction lever is on the
      // same screen as the button that created the thing it undoes. A remedy
      // filed somewhere else is a remedy most people never find — and PAGED,
      // because a remedy that stops at the fiftieth answer is not a remedy for
      // the fifty-first.
      answers: own.answers,
      answers_next: own.next,
      accepting: s.accepting,
      rule: s.rule,
      answer_max: ANSWER_MAX,
      // The asymmetry a creator must know BEFORE they publish, not after.
      publish_warning: PUBLISH_WARNING,
    });

    const id = await creatorId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const raw = await c.req.text();
    if (raw.length > 8 * 1024) return c.json({ error: "too_large" }, 413);
    const body = ((): { accepting?: unknown; rule?: unknown } | null => {
      try {
        return JSON.parse(raw) as { accepting?: unknown; rule?: unknown };
      } catch {
        return null;
      }
    })();
    if (!body || typeof body.accepting !== "boolean") return c.json({ error: "accepting (boolean) required" }, 400);
    try {
      // Parsed HERE, so a creator learns immediately that a raised bar is
      // malformed rather than discovering it later as a silently closed inbox.
      // BF-19: an OMITTED `rule` is not an instruction to clear one. The
      // console's ordinary off/on toggle sends no rule at all, and `?? null`
      // turned that into "delete the creator's audience restriction". `"rule"
      // in body` is the only thing that distinguishes the two on the wire, and
      // an explicit `null` still clears, as it always did.
      await setQuestionSettings(d1(c.env.TRUST_DB), id, body.accepting, "rule" in body ? body.rule : undefined);
      return c.json({ ok: true, accepting: body.accepting });
    } catch (e) {
      if (e instanceof EligibilityError) return c.json({ error: e.message }, 400);
      if (e instanceof QuestionError) return c.json(errorBody(e), e.status);
      return serverFault(c, "/account/question-settings", e);
    }
  });

  // Decline. NO WORDS — only a state. A decline that could carry a message would
  // be the private reply path this feature does not have.
  app.post("/account/questions/:id/decline", async (c) => {
    const id = await creatorId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const qid = c.req.param("id");
    if (!QUESTION_ID_RE.test(qid)) return c.json({ error: "no such question" }, 404);
    try {
      return c.json({ ok: true, declined: await declineQuestion(d1(c.env.TRUST_DB), id, qid) });
    } catch (e) {
      if (e instanceof QuestionError) return c.json(errorBody(e), e.status);
      return serverFault(c, "/account/questions/:id/decline", e);
    }
  });

  // Block the fan behind ONE question, named BY THE QUESTION. Self-scoped
  // twice: `creatorId` is the caller's own passport, and `blockQuestionAsker`
  // refuses any question not addressed to it. The fan's account id is never in
  // the request and never in a response — the inbox does not even name them.
  app.post("/account/questions/:id/block", async (c) => {
    const id = await creatorId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const qid = c.req.param("id");
    if (!QUESTION_ID_RE.test(qid)) return c.json({ error: "no such question" }, 404);
    try {
      await blockQuestionAsker(d1(c.env.TRUST_DB), id, qid);
      return c.json({ ok: true });
    } catch (e) {
      if (e instanceof QuestionError) return c.json(errorBody(e), e.status);
      return serverFault(c, "/account/questions/:id/block", e);
    }
  });

  // --- the two-leg signed PUBLIC answer (the S100 lock) ---------------------

  app.post("/account/questions/:id/answer/start", anonBodyLimit, async (c) => {
    const id = await creatorId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const cfg = passkeyConfig(c.env);
    if (!cfg) return c.json({ error: "passkeys_unavailable" }, 503);
    const qid = c.req.param("id");
    if (!QUESTION_ID_RE.test(qid)) return c.json({ error: "no such question" }, 404);
    const raw = await c.req.text();
    if (raw.length > 8 * 1024) return c.json({ error: "too_large" }, 413);
    const body = ((): { answer?: unknown } | null => {
      try {
        return JSON.parse(raw) as { answer?: unknown };
      } catch {
        return null;
      }
    })();
    if (!body) return c.json({ error: "answer required" }, 400);
    const db = d1(c.env.TRUST_DB);
    try {
      // The question and its hash come from the STORED ROW; only the answer is
      // the caller's, and it is canonicalized before it enters the draft.
      const { answer, draft } = await buildAnswerDraft(db, id, qid, body.answer, Date.now());
      const opts = await startAssertionForAction(db, id, [draft], cfg, Date.now());
      // Echo the CANONICAL answer — the finish leg re-sends this, not the raw
      // form, so both legs hash identically.
      return c.json({ ...opts, question_id: qid, answer });
    } catch (e) {
      if (e instanceof QuestionError) return c.json(errorBody(e), e.status);
      if (e instanceof PasskeyError) return c.json(errorBody(e), e.status);
      return serverFault(c, "/account/questions/:id/answer/start", e);
    }
  });

  app.post("/account/questions/:id/answer", anonBodyLimit, async (c) => {
    const id = await creatorId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const cfg = passkeyConfig(c.env);
    if (!cfg) return c.json({ error: "passkeys_unavailable" }, 503);
    const qid = c.req.param("id");
    if (!QUESTION_ID_RE.test(qid)) return c.json({ error: "no such question" }, 404);
    const raw = await c.req.text();
    if (raw.length > 16 * 1024) return c.json({ error: "too_large" }, 413);
    const body = ((): { answer: string; assertion: AssertionResponse } | null => {
      try {
        const o = JSON.parse(raw) as Record<string, unknown>;
        const a = o.assertion as Record<string, unknown> | undefined;
        if (typeof o.answer !== "string" || !a || typeof a.credentialId !== "string" || typeof a.clientDataJSON !== "string" || typeof a.authenticatorData !== "string" || typeof a.signature !== "string") return null;
        return { answer: o.answer, assertion: { credentialId: a.credentialId, clientDataJSON: a.clientDataJSON, authenticatorData: a.authenticatorData, signature: a.signature } };
      } catch {
        return null;
      }
    })();
    if (!body) return c.json({ error: "answer and assertion required" }, 400);
    const db = d1(c.env.TRUST_DB);
    const nowMs = Date.now();
    try {
      // REBUILT SERVER-SIDE from the row, exactly as /start did — so the action
      // hash matches only if the question is still the one the creator read.
      const { answer, draft } = await buildAnswerDraft(db, id, qid, body.answer, nowMs);
      const nowFn = () => new Date().toISOString();
      const nowIso = nowFn();
      const stmts = buildAnswerStatements(db, qid, answer, nowIso, id);
      await appendCreatorSigned(db, id, [draft], body.assertion, cfg, nowMs, nowFn, stmts, undefined, publishHook(...analyticsPublishArgs(c)));
      // I3-1273 sibling (I22-FX-AUTH): the answer is ON THE LEDGER now, so the door answers `ok` whatever the
      // echo read does — a fault there is logged and the canonical answer this request signed is returned instead.
      let q: Awaited<ReturnType<typeof getQuestion>> = null;
      try {
        q = await getQuestion(db, qid);
      } catch (e) {
        logError({ cid: crypto.randomUUID(), hop: "http", route: "/account/questions/:id/answer", msg: "question echo read failed after a committed answer", err: e instanceof Error ? e.name : typeof e });
      }
      return c.json({ ok: true, question_id: qid, answer: q?.answer ?? answer });
    } catch (e) {
      if (e instanceof QuestionError) return c.json(errorBody(e), e.status);
      if (e instanceof PasskeyError) return c.json(errorBody(e), e.status);
      if (e instanceof LedgerContentionError) return contentionResponse(c);
      // The admission gate fired: somebody answered this between the two legs.
      // Same words as the pre-check, so the paths are indistinguishable.
      if (e instanceof Error && /NOT NULL constraint failed: question_admissions\.question_id/i.test(e.message)) {
        return c.json({ error: "this question has already been answered or has lapsed" }, 409);
      }
      // The STANDING gate fired: the creator stopped being verified, or was
      // suppressed, between the two legs. Answered with the refusal it is
      // rather than an opaque 400 — a creator whose page just stopped being
      // published needs to know that is what happened.
      if (isWriteAdmissionRefusal(e)) return c.json({ error: "publishing an answer needs a currently verified passport" }, 403);
      return serverFault(c, "/account/questions/:id/answer", e);
    }
  });

  // --- taking a published answer back down (two-leg, signed) ----------------
  //
  // Same shape as the answer above, and deliberately so: undoing a signed public
  // statement is itself a signed public statement. The draft is built from the
  // STORED ROW on both legs and carries only the question id — there is nothing
  // for a client to choose but which of its own answers to withdraw.

  app.post("/account/questions/:id/retract/start", async (c) => {
    const id = await creatorId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const cfg = passkeyConfig(c.env);
    if (!cfg) return c.json({ error: "passkeys_unavailable" }, 503);
    const qid = c.req.param("id");
    if (!QUESTION_ID_RE.test(qid)) return c.json({ error: "no such question" }, 404);
    const db = d1(c.env.TRUST_DB);
    try {
      const { draft } = await buildRetractionDraft(db, id, qid);
      const opts = await startAssertionForAction(db, id, [draft], cfg, Date.now());
      return c.json({ ...opts, question_id: qid });
    } catch (e) {
      if (e instanceof QuestionError) return c.json(errorBody(e), e.status);
      if (e instanceof PasskeyError) return c.json(errorBody(e), e.status);
      return serverFault(c, "/account/questions/:id/retract/start", e);
    }
  });

  app.post("/account/questions/:id/retract", anonBodyLimit, async (c) => {
    const id = await creatorId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const cfg = passkeyConfig(c.env);
    if (!cfg) return c.json({ error: "passkeys_unavailable" }, 503);
    const qid = c.req.param("id");
    if (!QUESTION_ID_RE.test(qid)) return c.json({ error: "no such question" }, 404);
    const raw = await c.req.text();
    if (raw.length > 16 * 1024) return c.json({ error: "too_large" }, 413);
    const body = ((): { assertion: AssertionResponse } | null => {
      try {
        const o = JSON.parse(raw) as Record<string, unknown>;
        const a = o.assertion as Record<string, unknown> | undefined;
        if (!a || typeof a.credentialId !== "string" || typeof a.clientDataJSON !== "string" || typeof a.authenticatorData !== "string" || typeof a.signature !== "string") return null;
        return { assertion: { credentialId: a.credentialId, clientDataJSON: a.clientDataJSON, authenticatorData: a.authenticatorData, signature: a.signature } };
      } catch {
        return null;
      }
    })();
    if (!body) return c.json({ error: "assertion required" }, 400);
    const db = d1(c.env.TRUST_DB);
    const nowMs = Date.now();
    try {
      const { draft } = await buildRetractionDraft(db, id, qid);
      const nowFn = () => new Date().toISOString();
      // I3-1770 (I4-17-R1 hand-off): the owner bind rides into the batch too.
      const stmts = buildRetractStatements(db, qid, nowFn(), id);
      await appendCreatorSigned(db, id, [draft], body.assertion, cfg, nowMs, nowFn, stmts);
      return c.json({ ok: true, question_id: qid, state: "retracted" });
    } catch (e) {
      // ALREADY DOWN IS SUCCESS, and this is what makes a dropped connection
      // safe to retry. The finish leg is the one place a creator loses the
      // reply rather than the effect, so the honest answer to "take this down"
      // on a row that is already down is 200 — not a 409 that reads as failure
      // and sends them looking for a lever that already worked.
      if (e instanceof QuestionError && e.status === 409 && /already been taken down/i.test(e.message)) {
        return c.json({ ok: true, question_id: qid, state: "retracted", already: true });
      }
      if (e instanceof QuestionError) return c.json(errorBody(e), e.status);
      if (e instanceof PasskeyError) return c.json(errorBody(e), e.status);
      if (e instanceof LedgerContentionError) return contentionResponse(c);
      // The admission gate fired: the row left `answered` between the two legs,
      // so a concurrent ceremony took it down first. Same outcome, same words.
      if (e instanceof Error && /NOT NULL constraint failed: question_admissions\.question_id/i.test(e.message)) {
        return c.json({ ok: true, question_id: qid, state: "retracted", already: true });
      }
      // The LEDGER refused it — the only reachable case is a passport retracted
      // between the two legs. Answered with what happened rather than a bare
      // "error": the one refusal this path can produce was the one it explained
      // worst.
      if (e instanceof LedgerError) return c.json({ error: e.message }, 409);
      return serverFault(c, "/account/questions/:id/retract", e);
    }
  });

  return app;
}

/**
 * ASK THE CREATOR / Q&A — F9, master list #54.
 *
 * A fan with standing asks ONE question. The creator either publishes a
 * passkey-signed PUBLIC answer, or says nothing at all.
 *
 * ---------------------------------------------------------------------------
 * WHY THIS IS NOT THE DM CHANNEL THE PRODUCT LOCKS OUT
 * ---------------------------------------------------------------------------
 * `DIRECTION.md:548` locks **NO open text feed / comments / DMs**, and names its
 * reason: scams, grooming and moderation load. Four properties keep Q&A outside
 * that, and each is structural rather than a policy somebody has to follow:
 *
 *  1. ONE QUESTION, NO THREAD. No replies, no follow-ups, no second turn. One
 *     open question per fan per creator, enforced by a partial UNIQUE, and
 *     answering it CLOSES the exchange rather than opening a channel.
 *  2. THE ONLY REPLY PATH IS PUBLIC. There is no private creator→fan message
 *     anywhere in this feature — no column for one and no route that could
 *     carry one. A private reply is exactly what turns a question box into a DM
 *     inbox.
 *  3. STANDING GATES THE INBOUND, so a stranger cannot type at a creator at
 *     all. That is the grooming and scam surface the lock is about.
 *  4. SILENCE COSTS THE CREATOR NOTHING. Unanswered questions are private to
 *     them, lapse on a TTL, and are never counted, ranked or published — F7's
 *     rule, for F7's reason, and `NO_ANSWER_MEANING` is the same sentence.
 *
 * ---------------------------------------------------------------------------
 * THE HARD PART, SAID PLAINLY
 * ---------------------------------------------------------------------------
 * Answering PUBLISHES A STRANGER'S SENTENCE on a creator's own trust page. An
 * answer without its question is meaningless, so there is no version of this
 * feature that avoids it.
 *
 * What makes it safe is that the creator READS the question before they sign,
 * and signing IS the publication: no auto-publish, no queue that drains by
 * default, no way to answer without seeing. And `question_hash` makes that hold
 * against the database rather than against good intentions — the signed payload
 * binds the exact text the creator was shown, so the row cannot be swapped
 * underneath a published answer.
 *
 * WHERE EACH PIECE OF TEXT LIVES:
 *   the FAN's words     → the `questions` row, deletable, gone with the fan
 *   their hash          → the ledger, so the binding survives without the words
 *   the CREATOR's words → the ledger, beside `creator_broadcast.message`
 *
 * The consequence is deliberate and the console says it out loud: an erased
 * fan's question disappears from under a published answer, and the answer has
 * to stand on its own.
 *
 * ---------------------------------------------------------------------------
 * Q&A IS OPT-IN. HANDSHAKE IS OPT-OUT.
 * ---------------------------------------------------------------------------
 * A Handshake answer re-attests something already on the creator's page and
 * costs nothing to ignore, so it defaults ON for a verified creator. This is the
 * only surface in the product that can put a stranger's sentence on that page,
 * so it defaults OFF and a creator turns it on.
 */

}

/**

 * private-note one this first shipped with.
 *
 * `BAD_TEXT_RE` above is copied from the surfaces whose text NOBODY ELSE READS
 * (an archive-request note, a handshake note). This text goes on a creator's
 * public trust page, and the two other surfaces that publish text there —
 * `broadcast.ts` and `recognition.ts` — both strip bidi controls, isolates and

 *
 * It matters beyond tidiness: JavaScript's `\s+` does not collapse U+200B or
 * U+202E and `xml()` escapes them rather than removing them, so a question
 * could render one way in the console the creator signed from and another way
 * on the page it was published to. Not XSS — angle brackets are refused and
 * every surface escapes — but a creator signing for text that reads differently
 * where it lands is the same consent problem as the warning above.
 */
/** Whitespace controls become a space (so `\s+` collapses them as before); C0 controls
 * that are NOT whitespace stay visible to BAD_TEXT_RE so they are still REFUSED;
 * everything else the shared display rule refuses becomes a space too, by the shared
 * helper's own default (#832, D-0920 A: the same answer broadcast/recognition/
 * reverse-codes give, so `x<U+200B>y` is 'x y' here as there; ZWNJ/ZWJ kept). */
const WS_CTRL_RE = /[\t\n\v\f\r\u2028\u2029]/g;
function cleanQuestionText(raw: unknown): { t: string; bad: boolean } {
  if (typeof raw !== "string") return { t: "", bad: false };
  const ws = raw.replace(WS_CTRL_RE, " ");
  const bad = /[\u0000-\u001f]/.test(ws); // DEL is stripped below, as before; C0 is refused
  return { t: stripUnsafeDisplayText(ws).replace(/\s+/g, " ").trim(), bad };
}

export type QuestionState = "open" | "answered" | "declined" | "expired" | "retracted";

export interface QuestionRow {
  question_id: string;
  passport_id: string;
  account_id: string;
  body: string;
  body_hash: string;
  /** C-28: per-question random value, mixed into `questionHash` so the
   * ledger's committed digest cannot be confirmed against a guessed body once
   * erasure blanks it — see `questionHash` and migration 0085. */
  salt: string;
  state: QuestionState;
  created_at: string;
  expires_at: string;
  answered_at: string | null;
  answer: string | null;
  retracted_at?: string | null;
}

export function newQuestionId(): string {
  return `qst_${crypto.randomUUID().replace(/-/g, "").slice(0, 16)}`;
}

/**
 * The canonical form of a question, and the ONE place it is computed.
 *
 * Trim, collapse internal whitespace, and refuse control characters and angle
 * brackets — the archive-request-note rules. The hash is taken over THIS form,
 * so a question that renders identically hashes identically and a creator's
 * signature cannot be invalidated by a stray trailing space.
 */
export function canonicalQuestion(raw: unknown): string {
  const { t, bad } = cleanQuestionText(raw);
  if (t.length === 0) throw new QuestionError("a question needs some words");
  if (t.length > QUESTION_MAX) throw new QuestionError(`a question is at most ${QUESTION_MAX} characters`);
  if (bad || BAD_TEXT_RE.test(t)) throw new QuestionError("a question must not contain angle brackets or control characters");
  return t;
}

/** The creator's own answer, canonicalized the same way and bounded higher. */
export function canonicalAnswer(raw: unknown): string {
  const { t, bad } = cleanQuestionText(raw);
  if (t.length === 0) throw new QuestionError("an answer needs some words");
  if (t.length > ANSWER_MAX) throw new QuestionError(`an answer is at most ${ANSWER_MAX} characters`);
  if (bad || BAD_TEXT_RE.test(t)) throw new QuestionError("an answer must not contain angle brackets or control characters");
  return t;
}

export async function questionHash(body: string, salt: string): Promise<string> {
  // C-28: salted like `proof.ts nameCommitment` — a bare digest of the fan's
  // exact sentence "confirms any guess" once the plaintext is gone; the salt
  // is erased in the same statement as `body`/`body_hash` (`account_data.ts`
  // F9 block), so the ledger's committed hash outlives the fan but stops
  // being checkable against a guess once they're gone.
  return sha256Hex(`${salt}:${body}`);
}

/**
 * THE DEFAULT STANDING BAR — the frozen list's disjunction, plus the seven-day
 * floor, expressed as an `eligibility.ts` rule so the product has ONE predicate
 * language and a creator's raised bar needs no second evaluator.
 */
export const DEFAULT_STANDING: EligibilityRule = {
  kind: "all",
  rules: [
    { kind: "account_age", min_days: FOUNDING_FAN_MIN_AGE_DAYS },
    {
      kind: "any",
      rules: [{ kind: "save" }, { kind: "bond", min_level: 1 }, { kind: "founding_fan" }, { kind: "early_scout" }, { kind: "supporter", tier: "active" }],
    },
  ],
};

export interface QuestionSettings {
  accepting: boolean;
  /** The creator's RAISED bar, or null for the default disjunction. */
  rule: EligibilityRule | null;
}

/**
 * A creator's Q&A settings.
 *
 * DEFAULT OFF (see the module docblock), and off is also what an unverified,
 * retracted or unknown passport gets — nobody there can sign, so accepting a
 * question would only ever manufacture the silence this feature promises means
 * nothing.
 */
export async function questionSettings(db: DbLike, passportId: string): Promise<QuestionSettings> {
  const row = await db
    .prepare(
      `SELECT p.claim_state AS claim_state,
              (SELECT accepting FROM question_settings q WHERE q.passport_id = p.passport_id) AS accepting,
              (SELECT rule FROM question_settings q WHERE q.passport_id = p.passport_id) AS rule
         FROM passports p WHERE p.passport_id = ? AND p.retracted = 0`,
    )
    .bind(passportId)
    .first<{ claim_state: string; accepting: number | null; rule: string | null }>();
  if (!row || row.claim_state !== "verified" || row.accepting !== 1) return { accepting: false, rule: null };
  let rule: EligibilityRule | null = null;
  if (row.rule) {
    try {
      rule = parseRule(JSON.parse(row.rule), passportId, new Set<string>());
    } catch {
      // FAIL CLOSED. An unparseable stored rule is not "no bar" — it is a bar
      // nobody can evaluate, and treating it as absent would silently open a
      // creator's inbox wider than they set it. `askQuestion` refuses on null
      // `accepting` below, so the whole channel closes until it is fixed.
      return { accepting: false, rule: null };
    }
  }
  return { accepting: true, rule };
}

/** Set the switch and, optionally, the raised bar. Parsing happens HERE so a
 * creator learns immediately that a rule is malformed, rather than discovering
 * it as a silently closed inbox. */
export async function setQuestionSettings(
  db: DbLike,
  passportId: string,
  accepting: boolean,
  /**
   * BF-19 — `undefined` and `null` ARE DIFFERENT INSTRUCTIONS, and collapsing
   * them destroyed creators' configuration.
   *
   * The console toggles acceptance without sending a rule at all
   * (`apps/creator-console/src/main.ts`, `api.ts`); the route turned that
   * omission into `null` and this upsert wrote `rule = excluded.rule`. So an
   * ordinary off/on cycle silently cleared a restriction the creator had set
   * through the supported settings API, and fans outside their selected
   * audience could submit from then on — with nothing said to anyone.
   *
   * `undefined` now means LEAVE IT ALONE. `null` still means CLEAR IT, because
   * explicitly clearing a rule is a real and valid instruction — it is just not
   * the one an omission gives.
   */
  rawRule: unknown,
  now: () => string = () => new Date().toISOString(),
): Promise<void> {
  const touchRule = rawRule !== undefined;
  let stored: string | null = null;
  if (rawRule !== null && rawRule !== undefined) {
    const parsed = parseRule(rawRule, passportId, new Set<string>());
    // I4-04-R1 P7: an acquisition leaf must name the creator's OWN release —
    // the same helper and the same 403/404 pools and reward policies use.
    try {
      await requireOwnedReleases(db, parsed, passportId);
    } catch (e) {
      if (e instanceof ReverseCodeError) throw new QuestionError(e.message, e.status === 404 ? 404 : 403);
      throw e;
    }
    stored = JSON.stringify(parsed);
  }
  const upsert = db
    .prepare(
      `INSERT INTO question_settings (passport_id, accepting, rule, updated_at) VALUES (?, ?, ?, ?)
       ON CONFLICT(passport_id) DO UPDATE SET accepting = excluded.accepting, rule = ${touchRule ? "excluded.rule" : "question_settings.rule"}, updated_at = excluded.updated_at`,
    )
    .bind(passportId, accepting ? 1 : 0, stored, now());
  if (accepting) {
    await upsert.run();
    return;
  }
  // I3-1264: switching OFF closes every open question in the SAME batch — the
  // off switch is a block on everyone, and a question admitted under the old
  // setting must not stay answerable after it (mirrors handshake.ts
  // setAcceptingHandshakes). `declined` is this feature's own terminal state
  // for an off-switch close (DECLINE_MEANING copy), not handshake.ts's
  // `expired`.
  await db.batch([upsert, db.prepare("UPDATE questions SET state = 'declined', answered_at = ? WHERE passport_id = ? AND state = 'open'").bind(now(), passportId)]);
}

/**
 * Block the fan behind ONE question, named BY THE QUESTION.
 *
 * The F7 shape and for the F7 reason: the creator holds that id because the
 * question was addressed to them, so it is the capability, and the fan's `acc_`
 * id never leaves the server. Blocking also CLOSES whatever is open from them —
 * a block that left the question sitting in the inbox would be a block in name
 * only.
 */
export async function blockQuestionAsker(db: DbLike, passportId: string, questionId: string, now: () => string = () => new Date().toISOString()): Promise<void> {
  const row = await getQuestion(db, questionId);
  if (!row || row.passport_id !== passportId) throw new QuestionError("no such question", 404);
  // I3-0892 (D-0920 A): ONE block across both channels — the console calls it a
  // block on the fan, so the handshake door (handshake_blocks) shuts in the same batch.
  await db.batch([
    db.prepare("INSERT INTO question_blocks (passport_id, account_id, created_at) VALUES (?, ?, ?) ON CONFLICT(passport_id, account_id) DO NOTHING").bind(passportId, row.account_id, now()),
    db.prepare("INSERT INTO handshake_blocks (passport_id, account_id, created_at) VALUES (?, ?, ?) ON CONFLICT(passport_id, account_id) DO NOTHING").bind(passportId, row.account_id, now()),
    db.prepare("UPDATE questions SET state = 'declined', answered_at = ? WHERE passport_id = ? AND account_id = ? AND state = 'open'").bind(now(), passportId, row.account_id),
  ]);
}

/**
 * Ask.
 *
 * ORDER: the passport gate first and through `livePassport` (suppressed /
 * retracted / never-listed are one 404 — F6's lesson); then the door, where
 * "not accepting", "blocked by the raised bar" and "not verified" answer
 * IDENTICALLY, because "you specifically are not welcome" is information a
 * harasser acts on; then the caps, ALL INSIDE THE INSERT, because a SELECT
 * above an INSERT is not a ceiling.
 */
export async function askQuestion(
  db: DbLike,
  accountId: string,
  passportId: string,
  rawBody: unknown,
  nowMs: number,
  now: () => string = () => new Date().toISOString(),
  /** `ANALYTICS_FAN_SECRET`. Undefined is refused by `fanKey` (ADR-0006 §2). */
  fanSecret?: string,
  pubEnv?: AnalyticsTickEnv,
  pubCtx?: { waitUntil(p: Promise<unknown>): void },
): Promise<QuestionRow> {
  if (!ACCOUNT_RE.test(accountId)) throw new QuestionError("no such account", 401);
  const acct = await db.prepare("SELECT 1 AS x FROM accounts WHERE account_id = ?").bind(accountId).first();
  if (!acct) throw new QuestionError("no such account", 401);
  if (!PSP_RE.test(passportId)) throw new QuestionError("no such creator", 404);

  const p = await livePassport(db, passportId);
  if (!p) throw new QuestionError("no such creator", 404);

  const settings = await questionSettings(db, passportId);
  // BF-18 — A RAISED BAR IS A CONJUNCTION, NOT A REPLACEMENT.
  //
  // `0067_questions.sql:116-121`: "`rule` is an optional `eligibility.ts` rule
  // that RAISES the default standing bar — the frozen list's 'creator can raise
  // the bar'." `plan-f9-ask.md` says the same. This was `settings.rule ??
  // DEFAULT_STANDING`, which is REPLACEMENT: a creator who set a relationship
  // condition silently dropped the seven-day account-age floor, and one who set
  // an age condition silently dropped the relationship requirement. A creator
  // asking for a HIGHER bar could therefore open their inbox WIDER than the
  // default — the opposite of what the feature says it does, and the opposite
  // of what they intended.
  //
  // Both, always. `evaluate` already composes `all`, so there is no second
  // evaluator and no new rule language — which is the whole reason the raised
  // bar is expressed as an `eligibility.ts` rule in the first place.
  const bar: EligibilityRule = settings.rule ? { kind: "all", rules: [DEFAULT_STANDING, settings.rule] } : DEFAULT_STANDING;
  // ONE refusal for "the channel is off", "you do not clear the bar" and "the
  // creator raised it past you". A fan learns that they cannot ask; they do not
  // learn which of the three it is, and they cannot binary-search a creator's
  // private bar by varying their own standing.
  if (!settings.accepting || !(await evaluate(db, accountId, passportId, bar, nowMs))) {
    throw new QuestionError("this creator is not taking questions from you", 403);
  }

  const body = canonicalQuestion(rawBody);
  // C-28: a fresh per-question salt, mixed into the hash so the value the
  // creator's answer commits to in the ledger cannot be confirmed against a
  // guessed body after erasure blanks this row's copy.
  const salt = crypto.randomUUID().replace(/-/g, "");
  const bodyHash = await questionHash(body, salt);
  const id = newQuestionId();
  const createdAt = now();
  const expiresAt = new Date(nowMs + QUESTION_TTL_DAYS * DAY_MS).toISOString();
  const windowStart = new Date(nowMs - PER_CREATOR_WINDOW_DAYS * DAY_MS).toISOString();

  // THE ANALYTICS ROW RIDES THIS BATCH (ADR-0006 §2) AND CARRIES THE INSERT'S
  // OWN VERDICT. Every cap above lives INSIDE the INSERT, so "the question was
  // asked" is decided by that statement and by nothing this function read; the
  // event is therefore predicated on the row existing (`extraWhere`), which the
  // fresh `question_id` makes an exact test — a refused ask emits nothing, and a
  // successful one cannot emit twice.
  const key = await fanKey(fanSecret, accountId, passportId);
  const span = outboxSpan(db);
  const spanRes = await db
    .batch([
      span.open,
      db
        .prepare(
          `INSERT INTO questions (question_id, passport_id, account_id, body, body_hash, salt, state, created_at, expires_at)
       SELECT ?, ?, ?, ?, ?, ?, 'open', ?, ?
        WHERE NOT EXISTS (SELECT 1 FROM question_blocks b WHERE b.passport_id = ? AND b.account_id = ?)
          AND (SELECT COUNT(*) FROM questions WHERE account_id = ? AND state = 'open' AND expires_at > ?) < ?
          AND (SELECT COUNT(*) FROM questions WHERE account_id = ? AND passport_id = ? AND created_at >= ?) < ?
          AND (SELECT COUNT(*) FROM questions WHERE passport_id = ? AND state = 'open' AND expires_at > ?) < ?
          AND (SELECT q.accepting FROM question_settings q WHERE q.passport_id = ?) = 1
          AND COALESCE((SELECT q.rule FROM question_settings q WHERE q.passport_id = ?), '') = COALESCE(?, '')`,
        )
        .bind(
          id, passportId, accountId, body, bodyHash, salt, createdAt, expiresAt,
          passportId, accountId, accountId, createdAt, MAX_OPEN_PER_FAN, accountId, passportId, windowStart, MAX_PER_FAN_PER_CREATOR, passportId, createdAt, MAX_OPEN_PER_CREATOR,
          passportId, passportId, settings.rule ? JSON.stringify(settings.rule) : null,
        ),
      ...outboxStatements(
        db,
        { kind: "question_submitted", passportId, occurredDay: occurredDay(createdAt), fanKey: key },
        { accountId, extraWhere: { sql: "EXISTS (SELECT 1 FROM questions WHERE question_id = ?)", params: [id] } },
      ),
      span.close,
    ])
    .catch((e: unknown) => {
      if (e instanceof Error && /UNIQUE constraint/i.test(e.message)) throw new QuestionError("you already have an open question with this creator", 409);
      throw e;
    });
  const seqs = outboxSpanSeqs(spanRes[0], spanRes[spanRes.length - 1]);
  if (pubEnv && pubCtx) publishAfterCommit(pubEnv, pubCtx, seqs);
  // `DbLike.batch` returns `unknown[]` — no `changes` (db.ts:20) — so the
  // refusal is read off the TABLE by the id this call alone minted.
  const landed = await db.prepare("SELECT 1 AS x FROM questions WHERE question_id = ?").bind(id).first();
  if (!landed) {
    // A BLOCK answers with the SAME 403 a closed channel and an unmet bar give.
    // "You specifically are not welcome" is information a harasser acts on.
    const blocked = await db.prepare("SELECT 1 AS x FROM question_blocks WHERE passport_id = ? AND account_id = ?").bind(passportId, accountId).first();
    if (blocked) throw new QuestionError("this creator is not taking questions from you", 403);
    // I3-1264: the INSERT's own settings predicate (accepting/rule, re-checked
    // above) is what actually refused this — a raised bar or an off switch
    // that landed between the read at the top of this function and the INSERT
    // answers with the SAME 403 the initial read gives, never the generic cap
    // message below.
    const settingsNow = await db.prepare("SELECT accepting, rule FROM question_settings WHERE passport_id = ?").bind(passportId).first<{ accepting: number; rule: string | null }>();
    const raisedRuleText = settings.rule ? JSON.stringify(settings.rule) : null;
    if (!settingsNow || settingsNow.accepting !== 1 || (settingsNow.rule ?? null) !== raisedRuleText) {
      throw new QuestionError("this creator is not taking questions from you", 403);
    }
    // I3-1046: a lapsed question no longer holds a slot, swept or not (:398).
    const mine = await db.prepare("SELECT COUNT(*) AS n FROM questions WHERE account_id = ? AND state = 'open' AND expires_at > ?").bind(accountId, createdAt).first<{ n: number }>();
    if ((mine?.n ?? 0) >= MAX_OPEN_PER_FAN) throw new QuestionError(`you already have ${MAX_OPEN_PER_FAN} questions waiting for an answer`, 429);
    const recent = await db
      .prepare("SELECT COUNT(*) AS n FROM questions WHERE account_id = ? AND passport_id = ? AND created_at >= ?")
      .bind(accountId, passportId, windowStart)
      .first<{ n: number }>();
    if ((recent?.n ?? 0) >= MAX_PER_FAN_PER_CREATOR) throw new QuestionError(`you have already asked this creator in the last ${PER_CREATOR_WINDOW_DAYS} days`, 429);
    throw new QuestionError("this creator's question inbox is full — try again later", 429);
  }
  return { question_id: id, passport_id: passportId, account_id: accountId, body, body_hash: bodyHash, salt, state: "open", created_at: createdAt, expires_at: expiresAt, answered_at: null, answer: null };
}

export async function getQuestion(db: DbLike, questionId: string): Promise<QuestionRow | null> {
  if (!QUESTION_ID_RE.test(questionId)) return null;
  const r = await db
    .prepare("SELECT question_id, passport_id, account_id, body, body_hash, salt, state, created_at, expires_at, answered_at, answer FROM questions WHERE question_id = ?")
    .bind(questionId)
    .first<QuestionRow>();
  return r ?? null;
}

/**
 * THE DRAFT, BUILT SERVER-SIDE FROM THE STORED ROW (the S100 lock).
 *
 * The caller supplies a question id and their answer text; the question and its
 * hash come out of the database. So the two legs cannot disagree, a hostile
 * client cannot substitute somebody else's question, and — because the hash is
 * recomputed from the STORED body rather than trusted from the column — a row
 * edited between the legs fails rather than publishing an answer against text
 * the creator never read.
 */
export async function buildAnswerDraft(
  db: DbLike,
  passportId: string,
  questionId: string,
  rawAnswer: unknown,
  nowMs: number,
): Promise<{ row: QuestionRow; answer: string; draft: DraftEvent }> {
  const row = await getQuestion(db, questionId);
  if (!row || row.passport_id !== passportId) throw new QuestionError("no such question", 404);
  if (row.state !== "open") throw new QuestionError("this question has already been answered or has lapsed", 409);
  if (Date.parse(row.expires_at) <= nowMs) throw new QuestionError("this question has already been answered or has lapsed", 409);
  const answer = canonicalAnswer(rawAnswer);
  // RECOMPUTED, never read from the column. The stored `body_hash` is a
  // convenience for readers; the thing that gets SIGNED is a hash of the body
  // this call actually loaded, so the signature is a statement about text that
  // demonstrably exists in the row right now.
  // I3-1510 (I4-01-R6 patch H, adapted): a pre-0085 row carries salt ''. Salt
  // it ONCE, in the row, before signing — both passkey legs rebuild this draft,
  // so a per-call salt would sign two different hashes. The UPDATE only ever
  // moves '' -> a salt; the stored value is re-read and is what gets hashed.
  let salt = row.salt;
  if (salt === "") {
    const fresh = crypto.randomUUID().replace(/-/g, "");
    await db.prepare("UPDATE questions SET salt = ?, body_hash = ? WHERE question_id = ? AND salt = '' AND state = 'open'").bind(fresh, await questionHash(row.body, fresh), questionId).run();
    const again = await getQuestion(db, questionId);
    if (!again || again.state !== "open" || again.salt === "") throw new QuestionError("this question has already been answered or has lapsed", 409);
    salt = again.salt;
  }
  const hash = await questionHash(row.body, salt);
  const draft: DraftEvent = {
    event_type: "question_answered",
    subject: passportId,
    actor_kind: "creator_passkey",
    actor_id: "passkey",
    payload: { question_id: row.question_id, question_hash: hash, answer },
  };
  return { row, answer, draft };
}

/**
 * The projection write that rides the SAME signed batch as the append.
 *
 * A per-attempt nonce in `answer_event` is the admission gate (F7's fix): the
 * gate reads back a row THIS batch alone created, so a second answer that
 * overtook the first cannot append a signed event over a closed question — not
 * even with identical text, and not through `appendEvents`' retry after a lost
 * ledger-head race.
 *
 * `verifiedMintGate` rides along too. `events.ts` refuses a `question_answered`
 * from an unverified creator, but that check runs BEFORE the batch and never
 * looks at suppression; this is the commit-time half, and it matters more here
 * than anywhere because the output is public.
 */
export function buildAnswerStatements(db: DbLike, questionId: string, answer: string, nowIso: string, passportId: string): DbStatement[] {
  const nonce = `ans_${crypto.randomUUID().replace(/-/g, "")}`;
  return [
    // I3-1770: bound to the SAME passportId every caller already has in hand
    // (question-routes.ts's ownership read), not just the question id.
    db.prepare("UPDATE questions SET state = 'answered', answered_at = ?, answer = ?, answer_event = ? WHERE question_id = ? AND state = 'open' AND passport_id = ?").bind(nowIso, answer, nonce, questionId, passportId),
    db
      .prepare(
        // NO `ON CONFLICT` CLAUSE, and its removal is a real tightening. The
        // upsert that used to sit here was unreachable — the `state = 'open'`
        // guard on the UPDATE above makes a second admission for the same
        // question impossible while the subquery still resolves — so it never
        // fired in normal operation and did nothing but convert the PRIMARY KEY
        // beside it from a second guard into a no-op: with the state guard
        // removed, the upsert let a duplicate COMMIT, and without it the key
        // refuses and the batch rolls back. Belt fitted, braces cut.
        `INSERT INTO question_admissions (question_id, kind, admitted_at)
         VALUES ((SELECT question_id FROM questions WHERE question_id = ? AND answer_event = ?), 'answer', ?)`,
      )
      .bind(questionId, nonce, nowIso),
    verifiedMintGate(db, passportId, questionId, () => nowIso),
    // UNCONDITIONAL, and correct BECAUSE the admission gate above is a NOT NULL
    // that fails the WHOLE batch when the UPDATE matched nothing (ADR-0006 §2:
    // the event exists iff its fact does). `question_answered` is
    // creator-scoped and carries no fan at all — who asked is not a fact a
    // creator's dashboard gets out of having answered.
    ...outboxStatements(db, { kind: "question_answered", passportId, occurredDay: occurredDay(nowIso) }),
  ];
}

/**
 * THE WAY BACK (F9 review finding: `answered` was terminal).
 *
 * Publishing an answer puts a STRANGER'S SENTENCE on the creator's own trust
 * page, signed by them. Every other publishing surface here has a lever that
 * undoes it, and this one — the one where the words are not even the creator's
 * — shipped without one: a creator who misread a question, or read it correctly
 * and was still manoeuvred into amplifying it, could do nothing at all.
 *
 * WHAT IT DOES NOT DO, and the omissions are the design:
 *   - it does not delete the `question_answered` event. The chain is
 *     append-only, and a retraction that erased the original would let a
 *     creator un-say something a fan already screenshotted. The public record
 *     is "said, then withdrew".
 *   - it does not delete the fan's words from the row. Erasure is the fan's
 *     lever, not the creator's.
 *   - it carries no message. `onlyFields` in the schema refuses one — a parting
 *     note aimed at the fan whose question was just removed would be the
 *     private creator→fan reply this whole feature is built not to have.
 */
export async function buildRetractionDraft(db: DbLike, passportId: string, questionId: string): Promise<{ row: QuestionRow; draft: DraftEvent }> {
  const row = await getQuestion(db, questionId);
  // Same 404 for "no such row" and "not yours": a creator must not be able to
  // probe another creator's inbox by retraction id.
  if (!row || row.passport_id !== passportId) throw new QuestionError("no such question", 404);
  // ALREADY DOWN gets its own sentence, and the split is not cosmetic. One
  // refusal covering both cases told a creator whose connection dropped
  // mid-ceremony "only a published answer can be retracted" — which reads as
  // *it did not work*, on the one lever whose entire purpose is getting words
  // off a page. The honest answer to "take this down" when it is already down
  // is that it is already down; `question-routes.ts` turns this one into a 200.
  if (row.state === "retracted") throw new QuestionError("this answer has already been taken down", 409);
  if (row.state !== "answered") throw new QuestionError("only a published answer can be retracted", 409);
  // A RETRACTED PASSPORT is refused here, before a challenge is minted. The
  // ledger rule refuses it too (`events.ts question_answer_retracted`), but by
  // then the creator has already signed: the assertion is consumed, the
  // credential's sign_count advances, and every retry repeats it. `ownAnswers`
  // skips `livePassport` on purpose — a suppressed or downgraded creator must
  // keep this lever — and this is the one case that over-reached, because a
  // retracted passport publishes nothing and so has nothing to take down.
  const live = await db.prepare("SELECT retracted FROM passports WHERE passport_id = ?").bind(passportId).first<{ retracted: number }>();
  if (!live || live.retracted === 1) throw new QuestionError("no such question", 404);
  const draft: DraftEvent = {
    event_type: "question_answer_retracted",
    subject: passportId,
    actor_kind: "creator_passkey",
    actor_id: "passkey",
    payload: { question_id: row.question_id },
  };
  return { row, draft };
}

export function buildRetractStatements(db: DbLike, questionId: string, nowIso: string, passportId: string): DbStatement[] {
  const nonce = `rtr_${crypto.randomUUID().replace(/-/g, "")}`;
  // `state = 'answered'` in the WHERE is most of the race control: two retract
  // ceremonies finishing at once, or one finishing after the row already
  // moved, change nothing here — and the admission row below then finds no
  // question_id, so the batch (signed append included) rolls back. I3-1770
  // adds the SAME ownership bind `closePool`/`buildAnswerStatements` take, so
  // an ownership change landing after `buildRetractionDraft`'s precheck and
  // before this batch is caught the same way. `passportId` is REQUIRED and the
  // UPDATE has no unbound form (I4-J-FAN J-FAN-NEW-03): the production caller
  // (question-routes.ts) passes the session's passport.
  const retractUpdate = db
    .prepare("UPDATE questions SET state = 'retracted', retracted_at = ?, retract_event = ? WHERE question_id = ? AND state = 'answered' AND passport_id = ?")
    .bind(nowIso, nonce, questionId, passportId);
  return [
    retractUpdate,
    db
      .prepare(
        // Bare, for the reason spelled out in `buildAnswerStatements`.
        `INSERT INTO question_admissions (question_id, kind, admitted_at)
         VALUES ((SELECT question_id FROM questions WHERE question_id = ? AND retract_event = ?), 'retract', ?)`,
      )
      .bind(questionId, nonce, nowIso),
    // NO `verifiedMintGate` HERE, and the omission is deliberate — see the
    // `question_answer_retracted` rule in `events.ts`. Publishing needs a
    // verified creator; taking a publication DOWN must stay available to a
    // creator who has just lost their verification, because that is precisely
    // when they most need the page to stop saying things.
  ];
}

/** Decline. PRIVATE to the asking fan, non-accusatory, and it needs no
 * signature: nothing is published, so there is nothing to prove. */
export async function declineQuestion(db: DbLike, passportId: string, questionId: string, now: () => string = () => new Date().toISOString()): Promise<boolean> {
  const row = await getQuestion(db, questionId);
  if (!row || row.passport_id !== passportId) throw new QuestionError("no such question", 404);
  // I3-1770: bound to the SAME passportId the precheck above just verified,
  // not just the question id — an ownership change landing between that read
  // and this UPDATE must not decline a question that is no longer this
  // creator's.
  const res = await db.prepare("UPDATE questions SET state = 'declined', answered_at = ? WHERE question_id = ? AND state = 'open' AND passport_id = ?").bind(now(), questionId, passportId).run();
  return (res.changes ?? 0) > 0;
}

export async function questionInbox(
  db: DbLike,
  passportId: string,
  nowMs: number,
  cursor?: { created_at: string; question_id: string } | null,
): Promise<{ questions: InboxQuestion[]; next: { created_at: string; question_id: string } | null }> {
  const page = INBOX_PAGE;

    ? await db
        .prepare(
          `SELECT question_id, body, created_at, expires_at
             FROM questions
            WHERE passport_id = ? AND state = 'open' AND expires_at > ?
              AND (created_at > ? OR (created_at = ? AND question_id > ?))
            ORDER BY created_at ASC, question_id ASC LIMIT ?`,
        )
        .bind(passportId, new Date(nowMs).toISOString(), cursor.created_at, cursor.created_at, cursor.question_id, page + 1)
        .all<InboxQuestion>()
    : await db
        .prepare(

             FROM questions
            WHERE passport_id = ? AND state = 'open' AND expires_at > ?
            ORDER BY created_at ASC, question_id ASC LIMIT ?`,
        )
        .bind(passportId, new Date(nowMs).toISOString(), page + 1)
        .all<InboxQuestion>();

    ? await db
        .prepare(
          `SELECT question_id, body AS question, answer, answered_at
             FROM questions
            WHERE passport_id = ? AND state = 'answered' AND answer IS NOT NULL
              AND (answered_at < ? OR (answered_at = ? AND question_id < ?))
            ORDER BY answered_at DESC, question_id DESC LIMIT ?`,
        )
        .bind(passportId, cursor.answered_at, cursor.answered_at, cursor.question_id, page + 1)

    : await db

             FROM questions WHERE passport_id = ? AND state = 'answered' AND answer IS NOT NULL
            ORDER BY answered_at DESC, question_id DESC LIMIT ?`,
        )
        .bind(passportId, page + 1)
        .all<{ question_id: string; question: string; answer: string; answered_at: string }>();
  // One extra row is fetched purely to decide whether there IS a next page, and
  // dropped before it is served — a "next" cursor minted from an empty tail
  // would give the client a page that renders nothing.
  const results = rows.results.slice(0, page);

/**

export async function sweepExpiredQuestions(db: DbLike, nowMs: number, limit = 500): Promise<number> {
  const res = await db
    .prepare("UPDATE questions SET state = 'expired' WHERE question_id IN (SELECT question_id FROM questions WHERE state = 'open' AND expires_at <= ? ORDER BY expires_at LIMIT ?)")
    .bind(new Date(nowMs).toISOString(), limit)
    .run();
  return res.changes ?? 0;
}

/* I3-1768 (D-1084(1), I32-FX2-SERVER): THE CLASS A FAN IS SHOWN for a passport row is standing.ts `shownClassSql`
 * (I32 B7: the one copy; standing.ts CLAIM_STATE_AT_SQL is the same text over `p`). ONE bound parameter, the instant.
 * Read in ONE place for the discovery boards — `ineligibleSet`, which `dropSuppressed` runs at build (the board's own

async function breakout(db: DbLike, nowMs: number, nowIso: string): Promise<RankedCreator[]> {
  const cutoff = iso(nowMs - BREAKOUT_WINDOW_DAYS * DAY_MS);
  // Recently-verified creators, ranked by collects gained in the same window.
  // EXISTS (not a JOIN) on verification_granted so a re-grant can't multiply the
  // collect count. A just-verified creator with zero collects still surfaces
  // (score 0) — "newly broken out" is itself the signal, which is why the
  // qualified-fan gate rides a LEFT JOIN here: an unqualified collector must
  // contribute nothing WITHOUT removing the creator from the board.
  const { results } = await db
    .prepare(
      `SELECT p.passport_id AS passport_id, p.display_name AS display_name, p.claim_state AS claim_state,
              ${fanWeightSum()} AS score
         FROM rankable_passports p
         LEFT JOIN card_collections cc ON cc.passport_id = p.passport_id AND cc.created_at >= ?
         ${qualifiedFanLeftJoin("cc.account_id")}
        WHERE p.retracted = 0 AND p.claim_state = 'verified' AND ${grantLiveAtSql("p.passport_id")} AND ${notSuppressed("p.passport_id")}
          AND EXISTS (SELECT 1 FROM trust_events te WHERE te.subject = p.passport_id
                        AND te.event_type = 'verification_granted' AND te.created_at >= ?)
        GROUP BY p.passport_id
        ORDER BY score DESC, p.passport_id ASC
        LIMIT ?`,
    )
    .bind(nowIso, cutoff, nowIso, cutoff, RANK_LIMIT)
    .all<Row>();
  return results.map(clean);
}

// --- F11 / #39 — the newcomer rail ------------------------------------------

/** How many creators the rail shows. Small on purpose: a rail of twenty-five
 * newcomers is a directory nobody reads, and the point is that a few people get
 * a real look rather than everyone getting a listing. */
export const NEWCOMER_LIMIT = 8;
/**
 * The size ceiling for the pool, in QUALIFIED-FAN-WEIGHTED collects — the same
 * unit every other board is scored in, so "small" here means what "big" means
 * there. Generous rather than tight: the ceiling is the one thing membership on
 * this rail reveals about a creator, so a low one would make the rail a precise
 * bound on somebody's audience. At this value it says only "not yet one of the
 * larger accounts", which most of the registry could be.
 */
export const NEWCOMER_CEILING = 50;
/** A passport must be this old before it can be shown. Nothing is served on the
 * day an account appears — the same floor `fan-quality.ts` puts on a founding
 * fan, for the same reason: a surface that shows brand-new things immediately
 * is one you can mint your way onto. */
export const NEWCOMER_MIN_AGE_DAYS = 3;
/** Days of exposure a creator gets, ever. When it is spent they leave the pool
 * and do not come back — the slot is a loan, and a loan that renewed itself
 * would be the leaderboard this rail exists not to be. */
export const MAX_EXPOSURE_DAYS = 14;
/**
 * The ceiling is measured over collects at least this old.
 *
 * A THRESHOLD IS NOT A LEADERBOARD, and that flips the sign on sybil collects.
 * On every other board a fake collector helps the creator they land on, so the
 * age ramp's "worth a rounding error however many of them there are" is enough.
 * Here, crossing 50 REMOVES a creator from the only surface built for them — so
 * a rival with a fleet of free accounts could evict them, silently and with no
 * appeal. Lagging the measurement means a burst has to be planned a month
 * ahead and sits visible in the data the whole time, and by then the creator's
 * real growth is the thing that graduates them.
 */
export const CEILING_LAG_DAYS = 30;

/** A per-day rotation offset, 0..15 — the amount the id is rotated by before it
 * is used as an ordering column. */
export function dayRotation(day: string): number {
  return rotationKey(day, "window") % 16;
}

/**
 * SQL for "the passport id, rotated by N characters".
 *
 * The candidate window has to move with the day, not just the final sort: a
 * window taken in `passport_id ASC` order and shuffled afterwards is still the
 * lowest ids, and everything above them starves. Rotating the id inside the
 * ORDER BY makes the WINDOW itself a per-day permutation, so every id reaches
 * the front of the queue on some day. `psp_` + 16 hex is fixed-width, which is
 * what makes a substring rotation a clean permutation.
 */
export function dayShuffleSql(col: string, offsetParam: string): string {
  return `substr(substr(${col}, 5) || substr(${col}, 5), ${offsetParam} + 1, 16)`;
}

/**
 * A per-day tiebreak nobody can steer.
 *
 * Ordering the pool by `days ASC` alone leaves every creator at zero exposure
 * tied, and SQLite would break that tie by whatever the index offers — in
 * practice `passport_id ASC`, which is stable across days and hands the front
 * of the queue permanently to whoever minted a low id. This mixes the day in,
 * so the order is deterministic (the same for every reader, auditable from the
 * artifact) and unpredictable more than a day ahead, and no property a creator
 * controls improves their position.
 *
 * Not a cryptographic hash and does not need to be: there is nothing to forge,
 * only a queue to shuffle, and the passport id is fixed long before the day is.
 */
export function rotationKey(passportId: string, day: string): number {
  let h = 2166136261;
  for (const ch of `${passportId}|${day}`) {
    h ^= ch.charCodeAt(0);
    h = Math.imul(h, 16777619) >>> 0;
  }
  return h;
}

/**
 * THE NEWCOMER RAIL — creators the other boards structurally cannot reach.
 *
 * Every other rail here orders `score DESC`, which is a rich-get-richer loop:
 * the biggest accounts hold the boards, being on a board wins collectors, and
 * tomorrow the same names are there. Sorting a newcomer pool by score would be
 * that same loop one size down — a leaderboard of the biggest small creators.
 *
 * So this one is METERED. Being shown spends a day, and a creator leaves the
 * pool when their days are gone. There is no score to climb and therefore
 * nothing to farm.
 *
 * ---------------------------------------------------------------------------
 * FOUR THINGS THE FIRST DRAFT GOT WRONG. Each was a claim the code did not
 * honour, and the review measured every one of them.
 * ---------------------------------------------------------------------------
 *
 * 1. THE CANDIDATE WINDOW RE-IMPOSED THE ORDER THE HASH WAS THERE TO REMOVE.
 *    It selected `ORDER BY days ASC, passport_id ASC LIMIT 160` and then
 *    shuffled in JS — but the shuffle only reorders the 160 rows SQLite already
 *    picked. Once the pool exceeds the window at `days = 0`, which it does
 *    permanently (a creator only leaves zero by being shown), the window is
 *    just the 160 lowest passport ids. Simulated over 90 days on a 5,000-strong
 *    pool: 42% of creators never appeared at all, and nobody reached their
 *    exposure cap. NOW the ordering column is `dayShuffle`, a per-day rotation
 *    of the id, so the WINDOW moves with the day and every id reaches the front
 *    eventually.
 *
 * 2. THE METER WAS SPENT PER SWEEP, NOT PER DAY. The artifact cron runs four
 *    times a day. Sweep one showed eight creators and debited them; sweep two
 *    found those eight now sorting behind everyone at zero, picked a DIFFERENT
 *    eight, and debited them too. Thirty-two creators a day each spent one of
 *    their fourteen for six hours of visibility, and the pool drained: a
 *    hundred-creator registry emptied the rail on day 43, after which the
 *    section silently vanished. NOW anyone already shown TODAY is excluded from
 *    the pool query (`last_shown_at` day <> today), and a re-run only TOPS UP
 *    today's cohort to a full slate — it never picks a fresh eight (I3-1157,
 *    I3-1503).
 *
 * 3. THE RAIL WAS A SIZE ORACLE, TWICE. Rendering `min(8, |pool|)` names
 *    published the exact pool size whenever it was under eight — the launch
 *    state, and the drained state. And because the published ORDER was the
 *    ordering key, and `rotationKey` is an unkeyed hash of public inputs,
 *    anyone could compute a target's key and read their absence as "over the
 *    size ceiling" — a named bound on a creator's audience, provable in about a
 *    day. NOW the rail is withheld entirely below a full slate, and the emitted
 *    array is shuffled by a key nobody outside can compute, so published order
 *    carries no ordering information.
 *
 * 4. `score` IS STILL ZEROED ON THE WAY OUT, which the review confirmed holds.
 *    Membership already implies "under the ceiling"; a score beside it would
 *    publish a second bound.
 */
async function newcomers(db: DbLike, nowMs: number, nowIso: string): Promise<RankedCreator[]> {
  const day = nowIso.slice(0, 10);
  // TODAY'S COHORT IS ALREADY CHOSEN, if a sweep ran today. The artifact cron
  // fires every six hours, and the first version let each sweep pick a fresh
  // eight — the ones it had just charged now sorted behind everyone at zero —
  // so thirty-two creators a day each spent one of their fourteen for six hours
  // of visibility, and a hundred-creator registry emptied the rail on day 43.
  // A re-run must re-render the same rail, not mint another one.
  // ORDERED before it is LIMITed (I3-1877) by the day's own rotation, and held
  // to the pool's eligibility, so a delisted or downgraded member is not re-shown.
  const todays = await db
    .prepare(
      `SELECT p.passport_id AS passport_id, p.display_name AS display_name, p.claim_state AS claim_state
         FROM discovery_exposure e JOIN rankable_passports p ON p.passport_id = e.passport_id
        WHERE substr(e.last_shown_at, 1, 10) = ? AND p.retracted = 0 AND p.claim_state IN ('claimed','verified')
          AND ${notSuppressed("p.passport_id")}
        ORDER BY ${dayShuffleSql("p.passport_id", "?")} ASC, p.passport_id ASC
        LIMIT ?`,
    )
    .bind(day, dayRotation(day), NEWCOMER_LIMIT)
    .all<Row>();
  const kept = todays.results.map((r) => ({ ...clean(r), score: 0 }));
  if (kept.length >= NEWCOMER_LIMIT) return shuffleForDisplay(kept, day);
  const bornBefore = iso(nowMs - NEWCOMER_MIN_AGE_DAYS * DAY_MS);
  // The ceiling is measured over collects that are ALREADY OLD (see
  // `CEILING_LAG_DAYS`), so a burst cannot push a creator over it this week.
  const ceilingAsOf = iso(nowMs - CEILING_LAG_DAYS * DAY_MS);
  const { results } = await db
    .prepare(
      `SELECT p.passport_id AS passport_id, p.display_name AS display_name, p.claim_state AS claim_state,
              ${fanWeightSum()} AS score,
              COALESCE(e.days, 0) AS days
         FROM rankable_passports p
         LEFT JOIN card_collections cc ON cc.passport_id = p.passport_id AND cc.created_at < ?
         ${qualifiedFanLeftJoin("cc.account_id")}
         LEFT JOIN discovery_exposure e ON e.passport_id = p.passport_id
        WHERE p.retracted = 0 AND p.claim_state IN ('claimed','verified')
          -- AGE FROM THE LEDGER, since the passports table carries a sequence
          -- rather than a timestamp; passport_created is the one event that
          -- exists for every passport exactly once, so it is the birthday
          AND EXISTS (SELECT 1 FROM trust_events te WHERE te.subject = p.passport_id
                        AND te.event_type = 'passport_created' AND te.created_at < ?)
          AND COALESCE(e.days, 0) < ?
          AND ${notSuppressed("p.passport_id")}
          -- EXCLUSION, not a count (I3-1157/I3-1503): nobody already shown today is re-picked
          AND (e.last_shown_at IS NULL OR substr(e.last_shown_at, 1, 10) <> ?)
        GROUP BY p.passport_id
       HAVING score < ?
        ORDER BY days ASC, ${dayShuffleSql("p.passport_id", "?")} ASC
        LIMIT ?`,
    )
    .bind(ceilingAsOf, ceilingAsOf, bornBefore, MAX_EXPOSURE_DAYS, day, NEWCOMER_CEILING, dayRotation(day), NEWCOMER_LIMIT - kept.length)
    .all<Row & { days: number }>();
  const slate = [...kept, ...results.map((r) => ({ ...clean(r), score: 0 }))];

  // A PARTIAL SLATE IS WITHHELD ENTIRELY. Rendering however many the pool has
  // publishes the pool size exactly, which is a live count of how many creators
  // are under the size ceiling. Better to show nothing than to show a number
  // nobody agreed to publish.
  // Re-applied after every later filter too (`withholdPartial` in dropSuppressed, I3-1158).
  if (slate.length < NEWCOMER_LIMIT) return [];

  return shuffleForDisplay(slate, day);
}

/**
 * SHUFFLED BEFORE IT IS SERVED, so the published order carries none of the
 * ordering.
 *
 * `rotationKey` is an unkeyed hash of public inputs, and the rail was served in
 * key order — so anyone could compute a target's key, see where it would have
 * fallen, and read their absence as "over the size ceiling". A named bound on a
 * creator's audience, provable in about a day. The order a viewer sees is now
 * unrelated to the queue that produced it.
 */
function shuffleForDisplay(items: RankedCreator[], day: string): RankedCreator[] {
  return [...items].sort((a, b) => rotationKey(`${a.passport_id}|shuffle`, day) - rotationKey(`${b.passport_id}|shuffle`, day));
}

/**
 * Hygiene: drop exposure rows for passports that are gone or delisted.
 *
 * The row itself says "this creator was on the SMALL-CREATOR rail between these
 * dates" — which is "they were under the size ceiling then", kept forever, for
 * somebody who has retracted or asked to be delisted. It is correctly invisible
 * to the fan-erasure audit (there is no fan identifier in it), which is exactly
 * why it needs its own line here rather than a reason written in a doc.
 */
export async function sweepDiscoveryExposure(db: DbLike): Promise<number> {
  const res = await db
    .prepare(
      `DELETE FROM discovery_exposure
        WHERE passport_id IN (SELECT passport_id FROM passports WHERE retracted = 1)
           OR passport_id IN (SELECT passport_id FROM request_suppressions)
           OR NOT EXISTS (SELECT 1 FROM passports p WHERE p.passport_id = discovery_exposure.passport_id)`,
    )
    .run();
  return res.changes ?? 0;
}

/**
 * Debit a day of exposure from everyone the rail is about to show.
 *
 * Runs AFTER `dropSuppressed`, so a creator filtered out of the artifact is not
 * charged for an appearance nobody saw. ONE BATCH rather than a loop: the
 * artifact is written before this runs, and a throw partway through a sequence
 * of eight statements would charge a prefix and leave the rest on the rail
 * unmetered, which is the static leaderboard this whole design exists to
 * prevent.
 *
 * Idempotent within a day twice over: `last_shown_at` carries the day and only
 * moves forward, and `newcomers` excludes anyone already shown today from its
 * pool query, so a re-run re-shows today's cohort and only tops it up.
 */
async function debitExposure(db: DbLike, ids: string[], nowIso: string): Promise<void> {
  const day = nowIso.slice(0, 10);
  const good = ids.filter((psp) => PSP_RE.test(psp));
  // A shown-but-unchargeable id is a BUG, not something to skip past: it would
  // sit on the rail forever at zero exposure. The artifact is already written,
  // so this cannot un-publish it — it makes the mismatch loud instead of silent.
  if (good.length !== ids.length) logError({ cid: "unknown", hop: "cron", msg: "newcomer_rail_id_not_meterable", count: ids.length - good.length });
  if (good.length === 0) return;
  await db.batch(
    good.map((psp) =>
      db
        .prepare(
          `INSERT INTO discovery_exposure (passport_id, days, first_shown_at, last_shown_at)
           VALUES (?, 1, ?, ?)
           ON CONFLICT(passport_id) DO UPDATE SET
             days = days + CASE WHEN substr(discovery_exposure.last_shown_at, 1, 10) < ? THEN 1 ELSE 0 END,
             last_shown_at = MAX(discovery_exposure.last_shown_at, excluded.last_shown_at)`,
        )
        .bind(psp, nowIso, nowIso, day),
    ),
  );
}

/**
 * Most Collected within each category — the per-niche discovery ranking that
 * powers category browse (S82). NOTE a deliberate deviation from the plan's
 * "Most Requested by category": creator_requests only ever target UNCLAIMED
 * passports (the demand loop), while a category is set by a claimed creator or
 * an operator — so a request-by-category board would be near-empty. Ranking the
 * claimed/verified creators IN a category by collection count is the signal
 * category browse actually needs. Only non-empty categories are emitted.
 */
async function byCategory(db: DbLike, nowIso: string): Promise<CategoryBoard[]> {
  const boards: CategoryBoard[] = [];
  // F-16 / #184 — CONSENT_REQUIRED_CATEGORIES GOVERNS PUBLICATION SURFACES, NOT
  // JUST STORAGE. `discovery-page.ts` already applies exactly this exclusion to
  // the sitemap, to the homepage/browse niche chips, and to `/browse/:category`'s
  // robots directive; it could not apply it HERE, which is why the artifact kept
  // carrying the board those three surfaces had stopped pointing at.
  //
  // The board is not the protective function. Taxonomy's stance is that we LIST
  // an adult creator so a fan can check an account is really them — and that
  // "needs no label". Every un-labelled board above still ranks them. What a
  // per-category board adds, and the only thing it adds, is the label: an
  // ordered, machine-readable, unauthenticated roster of up to 25 people under
  // the heading "Adult", with their language and region attached, served to
  // anyone who GETs /rankings/discovery.
  //
  // A CLAIM IS NOT CONSENT TO THAT ROSTER. Yes, `primary_category = 'adult'` can
  // only be stored on a claimed/verified passport (`setProfileMeta`'s chokepoint),
  // so nobody here was labelled against their will — that is precisely the
  // argument F-16 heard and rejected for the chip and the sitemap entry. The
  // consent recorded is consent to the label on one's OWN passport, at an
  // address a fan reaches by already knowing who they are looking for. It is not
  // consent to being enumerated into a directory that is reachable by knowing
  // nothing at all. Answering that question a second way here is how this defect
  // family reproduces.
  for (const category of CATEGORIES.filter((k) => !CONSENT_REQUIRED_CATEGORIES.has(k))) {
    const { results } = await db
      .prepare(
        `SELECT p.passport_id AS passport_id, p.display_name AS display_name, p.claim_state AS claim_state,
                pp.language AS language, pp.region AS region,
                ${fanWeightSum()} AS score
           FROM passport_profile pp
           JOIN rankable_passports p ON p.passport_id = pp.passport_id AND p.retracted = 0 AND p.claim_state IN ('claimed','verified')
           LEFT JOIN card_collections cc ON cc.passport_id = pp.passport_id
           ${qualifiedFanLeftJoin("cc.account_id")}
          WHERE pp.primary_category = ? AND ${notSuppressed("p.passport_id")}
          GROUP BY p.passport_id
          ORDER BY score DESC, p.passport_id ASC
          LIMIT ?`,
      )
      .bind(nowIso, category, RANK_LIMIT)
      .all<Row>();
    if (results.length > 0) {
      boards.push({ category, label: CATEGORY_LABELS[category], top: results.map(clean) });
    }
  }
  return boards;
}

// --- Impersonation intelligence (derive-from-log, mirrors registry-build) ---

/**
 * #20 — STOP COMPETITIVELY RANKING VICTIMS.
 *
 * This was `most_targeted`: the top 25 verified creators sorted DESCENDING by
 * the number of active impersonators pointed at them, served anonymously from
 * `GET /rankings/impersonation` and rendered as "Most protected — Verified
 * creators impersonators target most", unit "warnings". Whatever the intent, an
 * ordered public board whose score is "how often this person is attacked" is a
 * league table of victimisation, it is legible as a target list, and being at
 * the top of it is not an achievement anyone opted into.
 *
 * The stated fan-facing purpose — "every link goes to the real account" — was
 * ALREADY served without an ordered board, by the per-creator `impostor_count`
 * chip on the Passport (`passport-view.ts`) and by the PSA route. So the board
 * loses the two things that made it a competition and keeps the one thing that
 * helped a fan:
 *
 *   1. NO SCORE IN THE DTO. `ProtectedCreator` has no `score` field at all, so
 *      a client cannot re-derive the ranking we stopped publishing. The
 *      aggregate counters (creators_protected, warnings_active, heat_recent)
 *      stay: they carry no per-creator identity and they are the honest way to
 *      say how much impersonation is happening.
 *   2. NO ORDER. Sorted alphabetically by display name (passport_id as the
 *      deterministic tie-break), which is the same list every time and encodes
 *      nothing. `protection` is one flat bucket for everyone on the list —
 *      deliberately ONE bucket, because two buckets is a coarse ranking and a
 *      coarse ranking is still a ranking.
 *
 * The count itself is not deleted, only unpublished: it is still what decides
 * WHO is on the list (≥1 active impersonation warning), and the creator's own
 * Passport still shows their own protection stats.
 *
 * RESIDUAL, STATED: the only exit from this list is still `request_suppressions`,
 * the FULL delisting switch — the same hostage the `demand_optouts` lever was
 * added to fix for the request counter (item #13). A dedicated
 * "don't list me as protected" opt-out belongs here for the same reason and is
 * NOT built in this change; with the score and the order gone, what remains is
 * a fact the Passport already states publicly, which is what makes deferring it
 * defensible rather than merely convenient.
 */
async function impersonationIntel(db: DbLike, nowMs: number): Promise<Omit<ImpersonationIntel, "generated_at">> {
  const heatCutoff = iso(nowMs - HEAT_WINDOW_DAYS * DAY_MS);
  // The disposition breakdown behind `resolution_rate` — see the field's
  // docblock. Read here rather than derived from the ledger fold below,
  // because a retraction's REASON lives on the publication row (0019), not in
  // the `warning_retracted` payload.
  const disp = await publicationDispositions(db);
  const { results } = await db
    .prepare(
      `SELECT e.subject AS subject, e.event_type AS event_type, e.created_at AS created_at, e.payload AS payload, ${operatorSetPointerSql("e")} AS ptr_operator
         FROM trust_events e WHERE e.event_type IN ('warning_published','warning_retracted') ORDER BY e.seq ASC`,
    )
    .bind()
    .all<{ subject: string; event_type: string; created_at: string; payload: string; ptr_operator: number | null }>();

  const active = new Map<string, boolean>(); // subject → latest state (publish=true)
  const cls = new Map<string, string>(); // subject → latest warning_class
  const ptr = new Map<string, string>(); // subject → protected_passport_id (impersonation)
  const impostors = new Set<string>(); // distinct subjects ever published
  let publishedTotal = 0;
  let retractedTotal = 0;
  let heatRecent = 0;

  for (const r of results) {
    if (r.event_type === "warning_published") {
      publishedTotal++;
      impostors.add(r.subject);
      active.set(r.subject, true);
      let warningClass = "";
      let protectedPid: string | undefined;
      try {
        const p = JSON.parse(r.payload) as { warning_class?: unknown; protected_passport_id?: unknown };
        warningClass = String(p.warning_class);
        // The SHARED shape (PASSPORT_ID_RE), not a wider local one: this board
        // and the registry builder must agree on which pointers are real, or a
        // creator counted as "protected" here would carry an id the builder
        // refuses to serve and no client can route.
        // …and only a pointer the OPERATOR set (QUEUE #292 / D-0802).
        if (r.ptr_operator === 1 && typeof p.protected_passport_id === "string" && PASSPORT_ID_RE.test(p.protected_passport_id)) {
          protectedPid = p.protected_passport_id;
        }
      } catch {
        warningClass = "";
      }
      cls.set(r.subject, warningClass);
      if (protectedPid) ptr.set(r.subject, protectedPid);
      else ptr.delete(r.subject);
      if (warningClass === "impersonation" && r.created_at >= heatCutoff) heatRecent++;
    } else {
      retractedTotal++;
      active.set(r.subject, false); // a later retraction wins (seq ASC)
    }
  }

  // Group ACTIVE impersonation warnings by the protected creator; gate the
  // pointer to currently-verified, non-retracted creators (the S63 rule),
  // re-checked here at build. The impersonator account is never named.
  const perCreator = new Map<string, number>();
  for (const [subject, isActive] of active) {
    if (!isActive || cls.get(subject) !== "impersonation") continue;
    const pid = ptr.get(subject);
    if (!pid) continue;
    perCreator.set(pid, (perCreator.get(pid) ?? 0) + 1);
  }

  // Fetch every protected creator's status in ONE chunked IN (was an N+1: a
  // per-creator SELECT in the loop). The gate must still run over ALL protected
  // creators — `creators_protected` counts them, not just the top-N — so this
  // batches the lookup rather than slicing first. Chunked for D1's 100-param cap
  // (a widely-targeted moment can carry >100 protected creators).
  const meta = await passportMetaByIds(db, [...perCreator.keys()], iso(nowMs));
  // Delisted creators are dropped BEFORE the RANK_LIMIT slice (I3-1087's shape),
  // and still counted in `creators_protected` exactly as before.
  const delisted = await suppressedSet(db, [...perCreator.keys()]);
  const targeted: ProtectedCreator[] = [];
  let creatorsProtected = 0;
  for (const pid of perCreator.keys()) {
    const row = meta.get(pid);
    if (!row || row.retracted !== 0 || row.claim_state !== "verified") continue;
    const name = String(row.display_name ?? "").replace(/[<>]/g, "").slice(0, 64).trim();
    if (name.length === 0) continue; // an unrenderable pointer is worse than none
    creatorsProtected++;
    if (delisted.has(pid)) continue;
    // The COUNT decides membership and is then DISCARDED — it is never written
    // to the artifact and never served (#20).
    targeted.push({ passport_id: pid, display_name: name, claim_state: row.claim_state, protection: "active" });
  }
  targeted.sort((a, b) => a.display_name.localeCompare(b.display_name) || (a.passport_id < b.passport_id ? -1 : 1));

  let warningsActive = 0;
  for (const [subject, isActive] of active) if (isActive && cls.get(subject) === "impersonation") warningsActive++;

  return {
    most_targeted: targeted.slice(0, RANK_LIMIT),
    counters: {
      creators_protected: creatorsProtected,
      warnings_active: warningsActive,
      warnings_published_total: publishedTotal,
      warnings_retracted_total: retractedTotal,
      impostors_flagged: impostors.size,
      warnings_corrected_total: disp.corrected_false_positive,
      resolution_rate: publishedTotal > 0 ? Math.round((disp.resolved_threat / publishedTotal) * 1000) / 1000 : 0,
      heat_recent: heatRecent,
    },
  };
}

/**
 * I14-CAPS (QUEUE #912, D-0904(1)): THE RANKINGS LEG'S CAP on the artifacts trigger — what one
 * buildRankings run may spend (D1 statements, a batch per statement, and store calls), so the leg
 * ends on a creator boundary instead of being refused mid-loop by the trigger's meter (index.ts
 * tailMeter). The boards, the impersonation artifact and the rotation record are always written;
 * the per-creator top-fans phase gets what is left: empty boards for creators who left the boards
 * first, then creators new to the boards, then the rest in rotation. What a run does not reach is
 * carried in TOPFANS_ROTATION_KEY. test/i14-artifacts-leg-caps.test.ts asserts the artifacts caps
 * sum under the trigger budget.
 */
export const RANKINGS_LEG_CAP = 120;
/** One board creator's top-fans refresh: the ranking statement and the put (test/i14-rankings-leg-cap.test.ts). */
export const TOPFANS_CREATOR_COST = 2;
/** I14-CAPS: `{ after, pending, held, dirty_since }` — the last creator the rotation refreshed, the
 * creators who left the boards and still hold a non-empty top-fans board, (I14-FX7) the account ids
 * each top-fans file this leg wrote holds (the same ids the file stores; dropped when the file is
 * rewritten or emptied), and the first run each boarded creator still waiting for an erasure
 * rewrite was found. Not under TOPFANS_PREFIX, which index.ts purgeLegacyTopFans lists. */
export const TOPFANS_ROTATION_KEY = "rankings/topfans-rotation.json";
/** I14-Q1 (b): the rotation record's `ids_v` once `held` (and `handles`) record every top-fans file this leg writes.
 * A record without it is from before, and its boarded creators with no held list are rewritten (dirty). */
export const TOPFANS_IDS_V = 2;
/**
 * I14-FX7 (QUEUE #912, D-0908(1)): THE ERASURE SHARE — subrequests of the rotation reserve that
 * rewrite, ahead of the rotation, every boarded creator whose stored top-fans file holds a fan
 * erased since that file was written (half the reserve: 18 of a 75 phase, 9 creators a run). Taken
 * from the rotation's half only, so leavers' empty boards and newcomers keep their budget
 * (D-0908(2)); what it does not use flows back to the rotation. More than it fits: logged
 * `topfans_erasure_backlog`, carried, oldest first.
 */
export function topFansErasureShare(phase: number): number {
  return Math.floor(Math.floor(Math.max(0, phase) / 2) / 2);
}

/** I14-CAPS: counts what buildRankings sends — every D1 statement executed (a batch per statement) and every store call. */
function rankingsMeter(db: DbLike, store: RankingStore): { db: DbLike; store: RankingStore; spent: () => number } {
  let n = 0;
  const real = new WeakMap<object, unknown>();
  const stmt = (st: object): unknown => {
    const p = new Proxy(st, {
      get(t, k) {
        const v = Reflect.get(t, k) as unknown;
        if (typeof v !== "function") return v;
        if (k === "bind") return (...a: unknown[]) => stmt((v as (...x: unknown[]) => object).apply(t, a));
        if (k === "first" || k === "all" || k === "run" || k === "raw") return (...a: unknown[]) => (n++, (v as (...x: unknown[]) => unknown).apply(t, a));
        return (v as (...x: unknown[]) => unknown).bind(t);
      },
    });
    real.set(p, st);
    return p;
  };
  return {
    db: {
      prepare: (sql: string) => stmt(db.prepare(sql) as unknown as object) as ReturnType<DbLike["prepare"]>,
      batch: (sts) => {
        n += sts.length;
        return db.batch(sts.map((x) => (real.get(x as unknown as object) ?? x) as typeof x));
      },
    },
    store: { put: (k, v) => (n++, store.put(k, v)), get: (k) => (n++, store.get(k)) },
    spent: () => n,
  };
}

/** Cron: precompute both artifacts into the object store. */
export async function buildRankings(dbIn: DbLike, storeIn: RankingStore, nowMs: number, now: () => string = () => new Date().toISOString(), legCap: number = RANKINGS_LEG_CAP): Promise<void> {
  const meter = rankingsMeter(dbIn, storeIn);
  const db = meter.db;
  const store = meter.store;
  // ONE instant for every board in a sweep, taken from the sweep's own `nowMs`
  // rather than from the clock inside each query, so the age ramp cannot shift
  // between two boards of the same artifact.
  const nowIso = iso(nowMs);
  const built: DiscoveryRankings = {
    generated_at: now(),
    most_collected: await mostCollected(db, nowIso),
    most_discovered: await mostDiscovered(db, nowIso),
    fastest_rising: await fastestRising(db, nowMs, nowIso),
    breakout: await breakout(db, nowMs, nowIso),
    newcomers: await newcomers(db, nowMs, nowIso),
    by_category: await byCategory(db, nowIso),
  };
  // FILTERED BEFORE IT IS WRITTEN, not only before it is served (review LOW 7).
  // `ratings/`, `editions/`, `rankings/topfans/`, `rankings/top-requested.json`
  // and `rankings/leaderboards.json` all exclude a delisted subject at BUILD
  // time; discovery alone deferred to the serve filter, so a real removal left
  // `rankings/discovery.json` holding the person's passport id AND display name
  // at rest, in an object nobody would think to purge. The serve-time filter
  // stays — it is what makes an opt-out take effect immediately rather than at
  // the next sweep — but the artifact no longer carries the name in the first
  // place, which is the data-minimisation half.
  const discovery = await dropSuppressed(db, built, nowIso);
  // I3-0898: a top-fans artifact is written only for creators on a board, so a
  // creator who LEAVES the boards kept their last one forever, re-filtered live
  // by fan HANDLE — a handle a stranger can later take. Every creator the
  // previous artifact boarded and this sweep does not gets an empty board.
  // I14-CAPS: those creators are RECORDED (TOPFANS_ROTATION_KEY `pending`, with any a capped run
  // left) BEFORE the new discovery artifact replaces the previous one, so a sweep that fails or is
  // capped here still empties them — first, ahead of every refresh, in the top-fans phase below.
  const nextBoarded = new Set(discoveryIds(discovery));
  const previousIds = new Set(discoveryIds(parseDiscovery(await store.get(DISCOVERY_KEY))));
  const rotation = parseRotation(await store.get(TOPFANS_ROTATION_KEY));
  const pending = [...new Set([...rotation.pending, ...previousIds])].filter((psp) => PSP_RE.test(psp) && !nextBoarded.has(psp)).sort();
  await store.put(TOPFANS_ROTATION_KEY, JSON.stringify({ after: rotation.after, pending, held: rotation.held, handles: rotation.handles, dirty_since: rotation.dirtySince, ...(rotation.idsRecorded ? { ids_v: TOPFANS_IDS_V } : {}) }));
  await store.put(DISCOVERY_KEY, JSON.stringify(discovery));
  // Charged only for what actually shipped: `dropSuppressed` may have removed a
  // creator from the rail, and a slot nobody saw is not a slot spent.
  await debitExposure(db, discovery.newcomers.map((c) => c.passport_id), nowIso);

  const intel = await impersonationIntel(db, nowMs);
  // Most Targeted NAMES the protected creator, so the same rule applies to it.
  const targetedIds = intel.most_targeted.map((c) => c.passport_id);
  const targetedSup = await suppressedSet(db, targetedIds);
  const targetedInel = await ineligibleSet(db, targetedIds, iso(nowMs), { requireVerified: true });
  const targetedDrop = new Set([...targetedSup, ...targetedInel]);
  const impersonation: ImpersonationIntel = { generated_at: now(), ...intel, most_targeted: drop(intel.most_targeted, targetedDrop) };
  await store.put(IMPERSONATION_KEY, JSON.stringify(impersonation));

  // #19 — precompute the public TOP-FAN board for every creator with real
  // collector traction (i.e. who appears on a discovery board). This keeps the
  // public `/p/:id/top-fans` route an O(1) read like every other public board —
  // never a live GROUP-BY over a creator's whole audience (the amplifier the
  // "precompute, don't serve live" decision exists to prevent). Creators with no
  // board traction simply have no artifact and serve an empty board.
  // A creator who has opted out of discovery gets no board artifact written at
  // all — `serveTopFans` gates the READ (an artifact written before the opt-out
  // must stop serving immediately), and this stops the sweep from refreshing or
  // minting one afterwards. Bounded and chunked, the same helper the serve path
  // uses for the discovery boards.
  const candidates = [...new Set(discoveryIds(discovery))].filter((id) => PSP_RE.test(id));
  const suppressedBoards = await suppressedSet(db, candidates);
  const boardCreators = candidates.filter((id) => !suppressedBoards.has(id));
  // I14-FX7 (QUEUE #912, D-0908(1)): ERASED SINCE WRITTEN. Erasure deletes the fan's `accounts` row
  // (account_data.ts deleteAccount), so an id a boarded creator's file holds with no account row left
  // was erased after that file was written. I14-Q1 (c) (Jason, D-1030): the same ONE statement (none
  // when nothing is held) also finds a held fan who no longer carries BOTH public consents, or whose
  // handle is not the one the file was written with (`handles`, recorded beside `held`; a pair with no
  // recorded handle is checked on consent only). Those creators are rewritten first, oldest first,
  // from topFansErasureShare.
  const held = new Map(Object.entries(rotation.held));
  const handles = new Map(Object.entries(rotation.handles));
  const pairOf = (p: string, i: number): string => JSON.stringify([held.get(p)![i]!, handles.get(p)?.length === held.get(p)!.length ? handles.get(p)![i]! : null]);
  const pairs = [...new Set(boardCreators.flatMap((p) => (held.get(p) ?? []).map((_, i) => pairOf(p, i))))].sort();
  const stale = new Set(
    pairs.length === 0
      ? []
      : (
          await db
            .prepare(
              `SELECT j.key AS i FROM json_each(?) j
                WHERE NOT EXISTS (SELECT 1 FROM accounts a WHERE a.account_id = json_extract(j.value, '$[0]'))
                   OR NOT EXISTS (SELECT 1 FROM rankable_profiles u WHERE u.account_id = json_extract(j.value, '$[0]')
                        AND u.visibility = 'public' AND u.affiliation_visibility = 'public'
                        AND (json_type(j.value, '$[1]') = 'null' OR u.handle = json_extract(j.value, '$[1]')))`,
            )
            .bind(`[${pairs.join(",")}]`)
            .all<{ i: number }>()
        ).results.map((r) => pairs[Number(r.i)]!),
  );
  // I14-Q1 (b) / I4-Q12 (I3-0898): THE FIRST RUN AFTER DEPLOY. A rotation record without `ids_v` was
  // written before `held` recorded every file this leg wrote, so a boarded creator that had a board
  // (`previousIds`) and holds no recorded list has a file whose fans this leg cannot check: dirty.
  // A creator carried in `dirty_since` (a backlog) stays dirty until it is rewritten.
  const legacy = !rotation.idsRecorded;
  const dirtySince = new Map<string, string>();
  for (const p of boardCreators) {
    const hit = (held.get(p) ?? []).some((_, i) => stale.has(pairOf(p, i)));
    if (hit || (legacy && previousIds.has(p) && !held.has(p)) || rotation.dirtySince[p] !== undefined) dirtySince.set(p, rotation.dirtySince[p] ?? nowIso);
  }
  const dirty = [...dirtySince.keys()].sort((x, y) => (dirtySince.get(x)! === dirtySince.get(y)! ? (x < y ? -1 : 1) : dirtySince.get(x)! < dirtySince.get(y)! ? -1 : 1));
  // I14-CAPS: the top-fans phase spends what the cap leaves, less the closing rotation put (1).
  const fits = (cost: number): boolean => meter.spent() + cost + 1 <= legCap;
  // I14-FX1 (RG-CAPS-06, D-0905(1)): THE ROTATION RESERVE. Half of what the cap leaves this phase
  // is held for the STABLE boarded creators in rotation order; leavers' empty boards and creators
  // new to the boards spend the other half first, and whatever either side leaves flows to the
  // other. Under churn the rotation used to get nothing (leavers cost 1, newcomers 2) and `after`
  // never moved. BOUND: every stable boarded creator is refreshed within
  // topFansRefreshBound(stable, phase) runs (see it). Not shortened by policy (D-0905(5)).
  const reserve = Math.floor(Math.max(0, legCap - meter.spent() - 1) / 2);
  const erasureShare = Math.floor(reserve / 2);
  let dirtySpent = 0;
  // Erasure rewrites spend the rotation's half, never the half ahead of it.
  const fitsAhead = (cost: number): boolean => meter.spent() + cost + 1 + reserve - dirtySpent <= legCap;
  // The rotation after `after`, wrapping; new to the boards = not on the previous artifact (no board yet).
  const sorted = [...boardCreators].sort();
  const start = sorted.findIndex((p) => p > rotation.after);
  const rotated = start <= 0 ? sorted : [...sorted.slice(start), ...sorted.slice(0, start)];
  const fresh = rotated.filter((p) => !previousIds.has(p));
  const stable = rotated.filter((p) => previousIds.has(p));
  const refreshed = new Set<string>();
  const refresh = async (psp: string): Promise<void> => {
    // D-0844(1) / RG2R-01, D-0850(1) / QUEUE #748: every board row is KEYED TO
    // THE ACCOUNT it was ranked as, taken from the ranking statement itself — no
    // second handle -> account lookup, so an erase + handle re-claim between two
    // statements cannot key the row to a stranger. `serveTopFans` re-filters on
    // that account.
    const keyed = (await publicTopFansKeyed(db, psp)).map((f) => ({ handle: f.handle, display_name: f.display_name, collected: f.collected, account_id: f.account_id }));
    await store.put(topFansKey(psp), JSON.stringify({ passport_id: psp, generated_at: discovery.generated_at, fans: keyed }));
    refreshed.add(psp);
    held.set(psp, keyed.map((f) => f.account_id));
    handles.set(psp, keyed.map((f) => f.handle));
  };
  let emptied = 0;
  let fi = 0;
  let si = 0;
  const emptyNext = async (): Promise<void> => {
    const psp = pending[emptied++]!;
    await store.put(topFansKey(psp), JSON.stringify({ passport_id: psp, generated_at: discovery.generated_at, fans: [] }));
    held.delete(psp);
    handles.delete(psp);
  };
  const refreshOnce = async (psp: string): Promise<void> => {
    if (!refreshed.has(psp)) await refresh(psp);
  };
  let di = 0;
  while (di < dirty.length && dirtySpent + TOPFANS_CREATOR_COST <= erasureShare && fits(TOPFANS_CREATOR_COST)) {
    const before = meter.spent();
    await refresh(dirty[di++]!);
    dirtySpent += meter.spent() - before;
  }
  while (emptied < pending.length && fitsAhead(1)) await emptyNext();
  while (fi < fresh.length && fitsAhead(TOPFANS_CREATOR_COST)) await refreshOnce(fresh[fi++]!);
  while (si < stable.length && fits(TOPFANS_CREATOR_COST)) await refreshOnce(stable[si++]!);
  while (emptied < pending.length && fits(1)) await emptyNext();
  while (fi < fresh.length && fits(TOPFANS_CREATOR_COST)) await refreshOnce(fresh[fi++]!);
  while (di < dirty.length && fits(TOPFANS_CREATOR_COST)) await refreshOnce(dirty[di++]!);
  const backlog = dirty.filter((p) => !refreshed.has(p));
  if (backlog.length > 0) logError({ cid: "unknown", hop: "cron", msg: "topfans_erasure_backlog", count: backlog.length, oldest: dirtySince.get(backlog[0]!) });
  // The next run resumes after the longest run of the rotation this one refreshed with no STABLE
  // creator missed (I14-FX7, FX6-h2): a creator new to the boards that no share reached is stepped
  // over, not stopped at — stopping there stalled `after` for good under churn. It is stable from
  // its next run and the rotation reaches it within a lap (topFansNewcomerBound).
  let after = rotation.after;
  for (const psp of rotated) {
    if (refreshed.has(psp)) after = psp;
    else if (previousIds.has(psp)) break;
  }
  const left = pending.slice(emptied);
  const keep = new Set([...nextBoarded, ...left]);
  const heldOut: Record<string, string[]> = {};
  const handlesOut: Record<string, string[]> = {};
  for (const p of [...held.keys()].sort()) {
    if (!(keep.has(p) && held.get(p)!.length > 0)) continue;
    heldOut[p] = held.get(p)!;
    if (handles.get(p)?.length === heldOut[p].length) handlesOut[p] = handles.get(p)!;
  }
  const dirtyOut: Record<string, string> = {};
  for (const p of backlog) dirtyOut[p] = dirtySince.get(p)!;
  await store.put(TOPFANS_ROTATION_KEY, JSON.stringify({ after, pending: left, held: heldOut, handles: handlesOut, dirty_since: dirtyOut, ids_v: TOPFANS_IDS_V }));
}

/**
 * I14-FX1 (RG-CAPS-06): the most runs between two refreshes of one STABLE boarded creator's
 * top-fans board, given `stable` creators in the rotation (at most the boarded count: a churner
 * that returns re-enters it) and `phase` subrequests left to buildRankings' per-creator phase (RANKINGS_LEG_CAP less the boards, the impersonation artifact and the
 * rotation record). The reserve guarantees floor(floor(phase / 2) / TOPFANS_CREATOR_COST) stable
 * refreshes a run, whatever the churn; at 6-hourly artifacts runs the bound in hours is 6x this.
 * I14-FX7: `dirtyPerRun` erasure rewrites a run take up to topFansErasureShare(phase) of that reserve
 * first (0, the default, is a run with no erased fan held; the worst case is the whole share).
 */
export function topFansRefreshBound(stable: number, phase: number, dirtyPerRun = 0): number {
  const reserve = Math.floor(Math.max(0, phase) / 2);
  const taken = Math.min(Math.max(0, dirtyPerRun) * TOPFANS_CREATOR_COST, topFansErasureShare(phase));
  const perRun = Math.floor((reserve - taken) / TOPFANS_CREATOR_COST);
  return perRun === 0 ? Infinity : Math.ceil(Math.max(0, stable) / perRun);
}

/**
 * I14-FX7 (D-0908(1)): the most runs a creator NEW to the boards that stays boarded waits for its
 * first top-fans board: its first run it is `fresh` (reached only with what leavers leave), from its
 * second it is in the stable rotation, which reaches it within topFansRefreshBound runs. A creator
 * boarded for fewer runs than this may leave before it gets one (its board then serves empty).
 */
export function topFansNewcomerBound(boarded: number, phase: number, dirtyPerRun = 0): number {
  return 1 + topFansRefreshBound(boarded, phase, dirtyPerRun);
}

/** I14-CAPS: TOPFANS_ROTATION_KEY, read defensively (absent or malformed = from the top, nothing pending). */
function parseRotation(raw: string | null): { after: string; pending: string[]; held: Record<string, string[]>; handles: Record<string, string[]>; dirtySince: Record<string, string>; idsRecorded: boolean } {
  try {
    const r = JSON.parse(raw ?? "null") as { after?: unknown; pending?: unknown; held?: unknown; handles?: unknown; dirty_since?: unknown; ids_v?: unknown } | null;
    const obj = (v: unknown): [string, unknown][] => (v && typeof v === "object" && !Array.isArray(v) ? Object.entries(v).filter(([k]) => PSP_RE.test(k)) : []);
    const held: Record<string, string[]> = {};
    for (const [k, v] of obj(r?.held)) if (Array.isArray(v)) held[k] = v.filter((a): a is string => typeof a === "string");
    const dirtySince: Record<string, string> = {};
    for (const [k, v] of obj(r?.dirty_since)) if (typeof v === "string") dirtySince[k] = v;
    const handles: Record<string, string[]> = {};
    for (const [k, v] of obj(r?.handles)) if (Array.isArray(v) && v.every((h) => typeof h === "string")) handles[k] = v as string[];
    return { after: typeof r?.after === "string" ? r.after : "", pending: Array.isArray(r?.pending) ? r.pending.filter((p): p is string => typeof p === "string") : [], held, handles, dirtySince, idsRecorded: r?.ids_v === TOPFANS_IDS_V };
  } catch {
    return { after: "", pending: [], held: {}, handles: {}, dirtySince: {}, idsRecorded: false };
  }
}

// --- Serve (O(1) read + one bounded suppression filter) ---

// D1 caps bound parameters at 100 per query. The discovery artifact can carry
// far more than 100 distinct ids (up to ~25×4 boards + 17 category boards ×25),
// so the suppression lookup MUST be chunked — a single IN(...) over every id
// would be rejected by D1 and 500 the public route exactly once discovery has
// real data. (The SqliteDouble's ~999-var limit hides this in tests.)
//
// This file was RIGHT and alone: the knowledge lived in this comment while four
// other IN(...) builders repeated the bug (audit v10, B1-B3). It now uses the
// shared `inChunks` helper so there is ONE definition of the chunk size, pinned
// against real D1 by test-fidelity/param-cap.d1.test.ts.

interface PassportMeta {
  passport_id: string;
  display_name: string;
  claim_state: string;
  retracted: number;
}

/** Fetch passport status rows for many ids in one chunked IN — replaces a
 * per-id N+1 in Most-Targeted. Same 100-param chunking as `suppressedSet`; a
 * missing id simply isn't in the returned map (the caller treats absent/off-gate
 * identically). */
async function passportMetaByIds(db: DbLike, ids: string[], nowIso: string): Promise<Map<string, PassportMeta>> {
  const distinct = [...new Set(ids)];
  const out = new Map<string, PassportMeta>();
  for (const chunk of inChunks(distinct, 1)) {
    const placeholders = chunk.map(() => "?").join(",");
    const { results } = await db
      // I3-1768 (D-1084(1)): the class at the board's instant (`shownClassSql`), so a lapsed grant is not 'verified' here.
      .prepare(`SELECT rp.passport_id AS passport_id, rp.display_name AS display_name, ${shownClassSql("rp")} AS claim_state, rp.retracted AS retracted FROM rankable_passports rp WHERE rp.passport_id IN (${placeholders})`)
      .bind(nowIso, ...chunk)
      .all<PassportMeta>();
    for (const r of results) out.set(r.passport_id, r);
  }
  return out;
}

/** Collect the suppressed subset of the given passport ids (bounded, chunked). */
async function suppressedSet(db: DbLike, ids: string[]): Promise<Set<string>> {
  const distinct = [...new Set(ids)];
  const out = new Set<string>();
  for (const chunk of inChunks(distinct)) {
    const placeholders = chunk.map(() => "?").join(",");
    const { results } = await db
      .prepare(`SELECT passport_id FROM request_suppressions WHERE passport_id IN (${placeholders})`)
      .bind(...chunk)
      .all<{ passport_id: string }>();
    for (const r of results) out.add(r.passport_id);
  }
  return out;
}

/**
 * TP-04 — the retraction/downgrade counterpart to `suppressedSet`.
 *
 * Every board here is built with `p.retracted = 0 AND p.claim_state IN
 * ('claimed','verified')` (Most Targeted: `claim_state = 'verified'`), but
 * that WHERE clause only runs in the cron sweep. Before this, the only thing
 * re-checked LIVE at serve was the opt-out (`request_suppressions`) — a
 * retraction or a verification downgrade landing between sweeps left the old
 * display name and claim-state badge on the board, unchanged, until the next
 * successful rebuild. Same bounded/chunked shape as `suppressedSet` so this
 * costs the same one extra query, not a per-row lookup.
 */
async function ineligibleSet(db: DbLike, ids: string[], nowIso: string, opts: { requireVerified?: boolean; shown?: Map<string, string> } = {}): Promise<Set<string>> {
  const distinct = [...new Set(ids)];
  const out = new Set<string>();
  for (const chunk of inChunks(distinct, 1)) {
    const placeholders = chunk.map(() => "?").join(",");
    const { results } = await db
      // I3-1768 (D-1084(1)): the class a fan is shown at `nowIso` (`shownClassSql`), not the stored column: a lapsed
      // grant is not verified here, and `opts.shown` hands that class to the caller that relabels a board row.
      .prepare(`SELECT p.passport_id AS passport_id, p.retracted AS retracted, ${shownClassSql("p")} AS claim_state FROM passports p WHERE p.passport_id IN (${placeholders})`)
      .bind(nowIso, ...chunk)
      .all<{ passport_id: string; retracted: number; claim_state: string }>();
    for (const r of results) {
      opts.shown?.set(r.passport_id, r.claim_state);
      const stillEligible = r.retracted === 0 && (opts.requireVerified ? r.claim_state === "verified" : r.claim_state !== "unclaimed");
      if (!stillEligible) out.add(r.passport_id);
    }
  }
  return out;
}

/** Passport ids across every board of a discovery artifact — defensive against a
 * structurally-odd (non-array top / missing id) board so id-extraction never
 * throws past the empty-shape guarantee. */
function discoveryIds(d: DiscoveryRankings): string[] {
  const boards: RankedCreator[][] = [d.most_collected, d.most_discovered, d.fastest_rising, d.breakout, d.newcomers, ...d.by_category.map((b) => (Array.isArray(b?.top) ? b.top : []))];
  const ids: string[] = [];
  for (const board of boards) {
    if (!Array.isArray(board)) continue;
    for (const c of board) if (c && typeof c.passport_id === "string") ids.push(c.passport_id);
  }
  return ids;
}

const drop = <T extends { passport_id: string }>(board: T[], sup: Set<string>): T[] => board.filter((c) => !sup.has(c.passport_id));

/** I3-1158: "a partial slate is withheld entirely" holds AFTER every filter, at
 * build and at serve — a filter that shrinks a full rail withholds it. */
const withholdPartial = (rail: RankedCreator[]): RankedCreator[] => (rail.length < NEWCOMER_LIMIT ? [] : rail);

const EMPTY_DISCOVERY: DiscoveryRankings = {
  generated_at: null,
  most_collected: [],
  most_discovered: [],
  fastest_rising: [],
  breakout: [],
  newcomers: [],
  by_category: [],
};

/**
 * THE DELISTING FILTER IS NOT AN OPTIONAL ARGUMENT (review HIGH 3).
 *
 * `db` used to be `db?: DbLike` with `if (!db) return d;` under it — so the
 * whole delisting guarantee for `/discover`, `/browse/:category` and
 * `/rankings/discovery` rode on three call sites remembering to pass it, and
 * deleting it from one was a change tsc was happy with: the parameter was
 * optional, and the route then served a delisted creator's passport id and
 * display name to anonymous callers with the whole invariant suite green.
 *
 * That is invariant 6 in shape — the ABSENCE of the suppression source read as
 * "no suppression" instead of "refuse". A required parameter is the smallest
 * thing that fails closed here: there is no longer a way to call this function
 * without the source of truth, so the failure mode is a compile error rather
 * than a silent unfiltered board. `rankings.test.ts` no longer exercises a
 * no-db path either — that path does not exist.
 *
 * The ROUTES are measured on top of this, in `suppression-oracles.test.ts`: a
 * suppressed subject and a never-listed one get byte-identical answers from
 * every board route, so a future re-loosening is caught at the surface as well
 * as at the signature.
 */
/** Drop every suppressed subject from every board of a discovery artifact. ONE
 * implementation, shared by the cron builder and the serve path, so the two can
 * never disagree about what a delisting removes. */
async function dropSuppressed(db: DbLike, d: DiscoveryRankings, nowIso: string): Promise<DiscoveryRankings> {
  const ids = discoveryIds(d);
  const sup = await suppressedSet(db, ids);
  const shown = new Map<string, string>();
  const inel = await ineligibleSet(db, ids, nowIso, { shown });
  // I3-1768 (D-1084(1)): a row the artifact says is `verified` whose class at `nowIso` is not (a lapsed grant, or a
  // downgrade the fold landed since the build) is relabelled with that class; it stays on its board.
  const stale = (c: RankedCreator): boolean => !!c && c.claim_state === "verified" && shown.has(c.passport_id) && shown.get(c.passport_id) !== "verified";
  const boards = [d.most_collected, d.most_discovered, d.fastest_rising, d.breakout, d.newcomers, ...d.by_category.map((b) => (Array.isArray(b?.top) ? b.top : []))];
  if (sup.size === 0 && inel.size === 0 && !boards.some((b) => Array.isArray(b) && b.some(stale))) return d;
  const sup2 = inel.size === 0 ? sup : new Set([...sup, ...inel]);
  const fix = (b: RankedCreator[]): RankedCreator[] => drop(b, sup2).map((c) => (stale(c) ? { ...c, claim_state: shown.get(c.passport_id) as string } : c));
  return {
    generated_at: d.generated_at,
    most_collected: fix(d.most_collected),
    most_discovered: fix(d.most_discovered),
    fastest_rising: fix(d.fastest_rising),
    breakout: fix(d.breakout),
    newcomers: withholdPartial(fix(d.newcomers)),
    by_category: d.by_category.map((b) => ({ ...b, top: fix(b.top) })).filter((b) => b.top.length > 0),
  };
}

/**
 * I3-0401 (D-1082(12)) — THE ACCOUNT A VERIFIED DISCOVER ROW LINKS, read LIVE at serve, one statement per id chunk.
 * The row qualifies only when, NOW, the passport is verified, and the account is active, a handle, not shielded
 * (`shieldedSqlPredicate`, the same predicate `/is/` resolves under — seo-page.ts resolveIsPage) and proved by THIS passport (a `used` non-contest
 * claim_challenges row or a `proven` oauth_challenges row: the `mine_count > 0` arm of getPassportView, which its
 * S101 drop never removes). `isPagePath` (passport-view.ts) is then the route's own shape gate, so the link is a page
 * `/is/` serves. First qualifying account by platform, id. A delisted or retracted creator gets no link because
 * `dropSuppressed` removes the whole row first; this read runs over the PRE-filter ids, so its statement count does not
 * move with a delisting.
 */
async function answerAccounts(db: DbLike, ids: string[], nowIso: string): Promise<Map<string, { platform: string; handle: string }>> {
  const out = new Map<string, { platform: string; handle: string }>();
  for (const chunk of inChunks([...new Set(ids)], 1)) {
    const placeholders = chunk.map(() => "?").join(",");
    const { results } = await db
      .prepare(
        `SELECT a.passport_id AS passport_id, a.platform AS platform, a.id AS id
           FROM passport_accounts a
           JOIN passports p ON p.passport_id = a.passport_id AND p.claim_state = 'verified' AND ${grantLiveAtSql("p.passport_id")}
          WHERE a.passport_id IN (${placeholders}) AND a.status = 'active' AND a.id_kind = 'handle'
            AND NOT ${shieldedSqlPredicate()}
            AND (EXISTS (SELECT 1 FROM claim_challenges c WHERE c.account_key = a.account_key AND c.passport_id = a.passport_id AND c.status = 'used' AND c.purpose != 'contest')
              OR EXISTS (SELECT 1 FROM oauth_challenges o WHERE o.account_key = a.account_key AND o.passport_id = a.passport_id AND o.status = 'proven'))
          ORDER BY a.passport_id, a.platform, a.id`,
      )
      .bind(nowIso, ...chunk)
      .all<{ passport_id: string; platform: string; id: string }>();
    for (const r of results) {
      if (out.has(r.passport_id) || isPagePath(r.platform, "handle", r.id, "active") === null) continue;
      out.set(r.passport_id, { platform: r.platform, handle: r.id });
    }
  }
  return out;
}

/** Every row loses any stored platform/handle; a row whose creator `answerAccounts` qualified gains its own. */
function withAnswerPages(d: DiscoveryRankings, pages: Map<string, { platform: string; handle: string }>): DiscoveryRankings {
  const one = (c: RankedCreator): RankedCreator => {
    if (!c || typeof c !== "object") return c;
    const { platform: _p, handle: _h, ...rest } = c;
    const page = pages.get(c.passport_id);
    return page ? { ...rest, platform: page.platform, handle: page.handle } : rest;
  };
  const all = (b: RankedCreator[]): RankedCreator[] => b.map(one);
  return {
    generated_at: d.generated_at,
    most_collected: all(d.most_collected),
    most_discovered: all(d.most_discovered),
    fastest_rising: all(d.fastest_rising),
    breakout: all(d.breakout),
    newcomers: all(d.newcomers),
    by_category: d.by_category.map((b) => (b && Array.isArray(b.top) ? { ...b, top: all(b.top) } : b)),
  };
}

/** I3-1768 (D-1084(1)): `nowMs` is the serve instant every class and the `/is/` link are read at (default: the clock). */
export async function serveDiscovery(store: RankingStore, db: DbLike, nowMs: number = Date.now()): Promise<DiscoveryRankings> {
  const d = parseDiscovery(await store.get(DISCOVERY_KEY));
  if (d === EMPTY_DISCOVERY) return d;
  const nowIso = iso(nowMs);
  const pages = await answerAccounts(db, discoveryIds(d), nowIso);
  return withAnswerPages(await dropSuppressed(db, d, nowIso), pages);
}

/** The discovery artifact as built (shape-checked, consent-gated categories
 * removed), BEFORE the live suppression filter. */
function parseDiscovery(raw: string | null): DiscoveryRankings {
  if (!raw) return EMPTY_DISCOVERY;
  try {
    const parsed = JSON.parse(raw) as DiscoveryRankings;
    return {
      generated_at: parsed.generated_at ?? null,
      most_collected: Array.isArray(parsed.most_collected) ? parsed.most_collected : [],
      most_discovered: Array.isArray(parsed.most_discovered) ? parsed.most_discovered : [],
      fastest_rising: Array.isArray(parsed.fastest_rising) ? parsed.fastest_rising : [],
      breakout: Array.isArray(parsed.breakout) ? parsed.breakout : [],
      newcomers: Array.isArray(parsed.newcomers) ? parsed.newcomers : [],
      // #184, serve side. A precomputed board OUTLIVES the sweep that built it,
      // so the build-time exclusion in `byCategory` alone would keep publishing
      // a consent-gated roster out of any artifact written before this fix,
      // until the next sweep ran. This module already takes that exact stance
      // for opt-outs one function down — suppression is "honored LIVE at serve,
      // so an opt-out takes effect immediately, not at the next sweep" — and a
      // consent gate has no weaker claim on it. Cheap, and it fails closed for
      // an artifact from any source.
      by_category: Array.isArray(parsed.by_category) ? parsed.by_category.filter((b) => !(b && typeof b.category === "string" && CONSENT_REQUIRED_CATEGORIES.has(b.category as Category))) : [],
    };
  } catch {
    return EMPTY_DISCOVERY;
  }
}

const EMPTY_INTEL: ImpersonationIntel = {
  generated_at: null,
  most_targeted: [],
  counters: {
    creators_protected: 0,
    warnings_active: 0,
    warnings_published_total: 0,
    warnings_retracted_total: 0,
    impostors_flagged: 0,
    warnings_corrected_total: 0,
    resolution_rate: 0,
    heat_recent: 0,
  },
};

/** Rebuild an artifact's protected list into the served shape: no score, no
 * ranked order, one flat marker. Junk entries are dropped rather than guessed
 * at. */
function normaliseProtected(raw: unknown): ProtectedCreator[] {
  if (!Array.isArray(raw)) return [];
  const out: ProtectedCreator[] = [];
  for (const c of raw as Array<Partial<ProtectedCreator>>) {
    if (!c || typeof c.passport_id !== "string" || !PSP_RE.test(c.passport_id)) continue;
    const name = String(c.display_name ?? "").replace(/[<>]/g, "").slice(0, 64).trim();
    if (name.length === 0) continue;
    out.push({ passport_id: c.passport_id, display_name: name, claim_state: typeof c.claim_state === "string" ? c.claim_state : "verified", protection: "active" });
  }
  out.sort((a, b) => a.display_name.localeCompare(b.display_name) || (a.passport_id < b.passport_id ? -1 : 1));
  return out;
}

/** Same required-`db` rule as `serveDiscovery` above, and for the same reason:
 * the protected list NAMES the creator, so an optional filter is a filter
 * one deleted argument away from being absent. */
export async function serveImpersonation(store: RankingStore, db: DbLike, nowMs: number = Date.now()): Promise<ImpersonationIntel> {
  const raw = await store.get(IMPERSONATION_KEY);
  if (!raw) return EMPTY_INTEL;
  let intel: ImpersonationIntel;
  try {
    const parsed = JSON.parse(raw) as ImpersonationIntel;
    if (!parsed || !parsed.counters) return EMPTY_INTEL;
    // NORMALISED AT THE READ, not only at the write (#20). The artifact on disk
    // may have been written by an older sweep that still carried a numeric
    // `score`, and an artifact is not a contract — this function is. Every
    // entry is rebuilt into the scoreless shape and re-sorted alphabetically,
    // so no vintage of the object can put a victimisation count on the wire.
    intel = {
      generated_at: parsed.generated_at ?? null,
      most_targeted: normaliseProtected(parsed.most_targeted),
      counters: parsed.counters,
    };
  } catch {
    return EMPTY_INTEL;
  }
  if (intel.most_targeted.length === 0) return intel;
  // A creator can opt out of being surfaced as "targeted"; the aggregate
  // counters stay (no identity). Filter the named board live — retraction and
  // a verification downgrade are re-checked here too (TP-04): the artifact is
  // only as current as the last sweep otherwise.
  const targetIds = intel.most_targeted.map((c) => c.passport_id);
  const sup = await suppressedSet(db, targetIds);
  const inel = await ineligibleSet(db, targetIds, iso(nowMs), { requireVerified: true });
  if (sup.size === 0 && inel.size === 0) return intel;
  const drop2 = new Set([...sup, ...inel]);
  return { ...intel, most_targeted: drop(intel.most_targeted, drop2) };
}

/**
 * #19 — serve the precomputed public top-fan board for a creator (O(1) read).
 * Three LIVE guards keep the stale artifact honest between sweeps, all bounded:
 *  - a RETRACTED creator serves nothing (the share-surface kill-switch),
 *  - a SUPPRESSED creator serves nothing (see below), and
 *  - any fan who has since withdrawn EITHER consent this board needs — their
 *    profile visibility or their affiliation visibility — is dropped (their
 *    handle was their opt-in; opting out takes effect immediately, not at the
 *    next sweep).
 * The heavy GROUP-BY that ranks the board runs ONLY in the cron sweep.
 *
 * THE SUPPRESSION GUARD (measured gap, `removal.ts`: "SUPPRESSION IS
 * INDISTINGUISHABLE FROM NEVER-LISTED"). `GET /p/:id/top-fans` answers
 * `{fans: []}` for a passport id that never existed, so an EMPTY board is
 * indistinguishable — but a suppressed creator who had a board artifact written
 * before the suppression kept serving it, handing back other people's FAN
 * HANDLES plus the has-a-board-vs-not distinguisher to anyone holding the id.
 * The correct answer is the NEVER-LISTED one for THIS route, which is the empty
 * board — NOT a 404, which would re-create the very oracle it closed (a 404 only
 * for suppressed subjects IS the "did this person ask to be delisted?" probe).
 *
 * It is checked HERE, in the single reader of the top-fans artifact, and not in
 * the route: `serveTopFans` is the only path from `rankings/topfans/<psp>.json`
 * to any caller, so every future consumer inherits the gate. The cron builder
 * additionally stops WRITING boards for suppressed creators (see buildRankings),
 * so the artifact also stops being refreshed — read gate for correctness now,
 * write gate so a stale object cannot outlive the delisting.
 */
export async function serveTopFans(store: RankingStore, db: DbLike, passportId: string): Promise<PublicTopFan[]> {
  if (!PSP_RE.test(passportId)) return [];
  // CONSTANT SHAPE. The live kill-switch runs FIRST and the artifact is read
  // UNCONDITIONALLY, so every input — suppressed, retracted, listed-but-never-
  // ranked, and an id that never existed — costs exactly one D1 read and one
  // object read. Ordered the other way (artifact, then gate) the gate only ran
  // when there was something to gate, so a suppressed subject who HAD a board
  // spent a D1 round trip that a never-listed id did not: the
  // has-a-board-vs-not distinguisher, behind an identical empty body (oracle
  // sweep, this pass).
  //
  // Residual, stated: the object read is a HIT for a previously-ranked subject
  // and a MISS for one that was never ranked, and no ordering fixes that — only
  // serving a decoy artifact for every id would, which is the registry's
  // problem and not this route's. What is removed here is the extra D1 round
  // trip, which is the part an attacker can measure without calibration.
  // The suppression count rides the same row as the retracted flag (the
  // `removal.ts` subselect shape), so the gate is one statement, not two.
  const p = await db
    .prepare("SELECT retracted, (SELECT COUNT(*) FROM request_suppressions s WHERE s.passport_id = p.passport_id) AS suppressed FROM passports p WHERE p.passport_id = ?")
    .bind(passportId)
    .first<{ retracted: number; suppressed: number }>();
  const raw = await store.get(topFansKey(passportId));
  if (!p || p.retracted || p.suppressed) return [];
  if (!raw) return [];
  let fans: (PublicTopFan & { account_id?: unknown })[];
  try {
    const parsed = JSON.parse(raw) as { fans?: unknown };
    fans = Array.isArray(parsed.fans) ? (parsed.fans as PublicTopFan[]).filter((f) => f && typeof f.handle === "string") : [];
  } catch {
    return [];
  }
  if (fans.length === 0) return [];
  // Live opt-out: keep only handles that STILL carry BOTH consents.
  //
  // BOTH, and the second one was missing. `leaderboard.ts publicTopFans` BUILDS
  // this board requiring `visibility = 'public' AND affiliation_visibility =
  // 'public'`; the live re-filter here re-checked only the first. The second
  // flag exists precisely because Top Fans has no per-row entity to consent on
  // (migration 0017, item #12): it is how a fan declines to be named on somebody
  // else's page without taking down their own profile, their collection, their
  // achievements and their discovery presence. Re-checking one of two consents
  // meant flipping the affiliation switch left the fan named on the artifact for
  // up to a full sweep interval — while the docstring above promises the opt-out
  // takes effect immediately. A re-filter that does not re-check the consent the
  // board's own builder requires is not a re-filter.
  //
  // CHUNKED even though the writer bounds this to the board size: `fans` is
  // parsed out of an R2 ARTIFACT, so the bound is a property of whatever wrote
  // that object, not of anything checked here. An artifact from an older build,
  // a hand-edited object, or a future board-size change would push this over
  // D1's 100-parameter cap and 500 a public board — the same shape as B1
  // (audit v10, L7). Re-derive the guarantee at the read.
  const handles = [...new Set(fans.map((f) => f.handle).filter((h) => /^[a-z0-9_]{3,20}$/.test(h)))];
  if (handles.length === 0) return [];
  const stillPublic = new Map<string, { account_id: string; display_name: string }>();
  for (const chunk of inChunks(handles)) {
    const placeholders = chunk.map(() => "?").join(",");
    const { results } = await db
      .prepare(`SELECT account_id, handle, display_name FROM rankable_profiles WHERE visibility = 'public' AND affiliation_visibility = 'public' AND handle IN (${placeholders})`)
      .bind(...chunk)
      .all<{ account_id: string; handle: string; display_name: string }>();
    for (const r of results) stillPublic.set(r.handle, { account_id: r.account_id, display_name: r.display_name });
  }
  // D-0844(1) / RG2R-01, I3-1309: a row is served only to the ACCOUNT it was
  // built for — never re-resolved through a handle erasure freed. An UN-keyed
  // row (an artifact written before buildRankings keyed them, :926) is DROPPED
  // here, not matched by handle: it shows nothing until the sweep rewrites it.
  // D-1078(4), I3-1309: the NAME is the consenting holder's LIVE `display_name`, read in the consent statement
  // above, never the artifact's build-time copy — a fan who renamed is never shown under a name they no longer have.
  return fans
    .filter((f) => {
      const cur = stillPublic.get(f.handle);
      if (!cur) return false;
      return typeof f.account_id === "string" && f.account_id === cur.account_id;
    })
    .map((f) => ({ handle: f.handle, display_name: (stillPublic.get(f.handle) as { display_name: string }).display_name, collected: f.collected }));
}

export interface NicheRank {
  category: Category;
  label: string;
  rank: number; // 1-based position in the creator's by-category board
  of: number; // how many creators are ranked in that board (top-N cap)
}

/**
 * A creator's own "you rank #N in <niche>" line — creator-PRIVATE, read off the
 * S81 discovery artifact (no new aggregate; the by-category board is
 * precomputed and suppression-filtered at BUILD). Returns null when the
 * creator has no primary category set, when their niche board hasn't been built
 * yet, or when they're not in it (outside the top-N, suppressed, or unclaimed) —
 * so the caller shows the line only when there's a real, honest rank.
 */
export async function nicheRank(store: RankingStore, db: DbLike, passportId: string): Promise<NicheRank | null> {
  if (!PSP_RE.test(passportId)) return null;
  const prof = await db.prepare("SELECT primary_category FROM passport_profile WHERE passport_id = ?").bind(passportId).first<{ primary_category: string | null }>();
  const category = prof?.primary_category;
  if (!category) return null;
  // I3-1125: rank/of are read off the board AS BUILT, not the live-filtered one,
  // so a THIRD party's delisting between sweeps cannot move this creator's two
  // numbers. Only the creator's OWN eligibility is re-checked live.
  if ((await suppressedSet(db, [passportId])).size > 0 || (await ineligibleSet(db, [passportId], iso(Date.now()))).size > 0) return null;
  const discovery = parseDiscovery(await store.get(DISCOVERY_KEY));
  const board = discovery.by_category.find((b) => b.category === category);
  if (!board) return null;
  const idx = board.top.findIndex((c) => c.passport_id === passportId);
  if (idx < 0) return null; // not ranked in the top-N of their niche
  return { category: board.category, label: board.label, rank: idx + 1, of: board.top.length };
}
