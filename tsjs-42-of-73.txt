        await auditStmt(db(c), actor(c), "registry:rebuild", null, "skipped: build already in progress", ts).run().catch(() => {});
        return c.json({ ok: true, seq: open?.seq, state: "in_progress" }, 202);
      }
      if (e instanceof RegistryCapacityError) {
        // The declared envelope refused the build with zero writes; the served
        // generation is untouched and an operator has to decide what gives.
        await auditStmt(db(c), actor(c), "registry:rebuild", null, "refused: capacity", ts).run().catch(() => {});
        return c.json({ error: "registry_capacity_exhausted" }, 503);
      }
      // The same posture the cron takes: a failed rebuild leaves the last
      // published buckets serving, and says so rather than half-reporting.
      await auditStmt(db(c), actor(c), "registry:rebuild", null, "failed", ts).run().catch(() => {});
      return c.json({ error: "rebuild_failed" }, 503);
    }
  });

  // Seed-batch history + detail (read-only). The apply/retract actions are the
  // POST routes below.
  app.get("/ops/imports", async (c) => {
    return c.json({ batches: await importBatchList(db(c)) });
  });
  app.get("/ops/imports/:batch", async (c) => {
    const detail = await importBatchDetail(db(c), c.req.param("batch"));
    if (!detail) return c.json({ error: "not_found" }, 404);
    return c.json({ detail });
  });

  // Seed import. Dry-run unless ?apply=true. Body is raw JSONL text. An apply
  // answers `complete`/`next`; the client re-POSTs with `&batch=` until complete.
  // #932f (I16-W05A): the 32 MiB ceiling enforced WHILE the body streams, before
  // `c.req.text()` buffers it — the declared-length check below trusts a header a
  // chunked request does not send.
  app.post("/ops/import/:kind", bodyLimit({ maxSize: 32 * 1024 * 1024, onError: (c) => c.json({ error: "file_too_large" }, 413) }), async (c) => {
    const kind = c.req.param("kind");
    if (kind !== "profile" && kind !== "determination") return c.json({ error: "bad_kind" }, 400);
    const apply = c.req.query("apply") === "true";
    // QUEUE #775 (D-0858(11)): `?batch=` continues a chunked apply with the SAME file
    // (importer.ts claimApplyContinuation: 409 on a different file, a finished or
    // retracted batch, or a continuation already running). Apply only.
    const batch = c.req.query("batch");
    if (batch !== undefined && (!apply || !/^imp_[0-9a-f]{16,64}$/.test(batch))) return c.json({ error: "bad_batch" }, 400);
    // Reject oversized bodies by the declared length before buffering where
    // possible (Workers also enforces its own request-size ceiling).
    const declared = Number(c.req.header("Content-Length") ?? "0");
    if (Number.isFinite(declared) && declared > 32 * 1024 * 1024) {
      return c.json({ error: "file_too_large" }, 413);
    }
    const text = await c.req.text();
    if (new TextEncoder().encode(text).length > 32 * 1024 * 1024) {
      return c.json({ error: "file_too_large" }, 413);
    }
    try {
      const report = await importFile(db(c), actor(c), kind, text, { apply, ...(batch !== undefined ? { batch } : {}) });
      return c.json(report);
    } catch (e) {
      return mapError(c, e);
    }
  });

  app.post("/ops/import/:batch/retract", async (c) => {
    try {
      // I3-1269 W4: the retracted passports' media is purged from the edge.
      // I3-1207 (D-0850(3)): one call does at most RETRACT_PER_CALL records and
      // answers `complete`; the client re-POSTs while `complete === false`.
      const purge = deps.edgePurger ? deps.edgePurger(c.env) : edgePurgerFromEnv(c.env);
      const r = await retractBatch(db(c), actor(c), c.req.param("batch"), undefined, { purge });
      return c.json({ ok: true, ...r });
    } catch (e) {
      return mapError(c, e);
    }
  });

  // --- M5: seeding metrics + the seeded-listing register (flag 14) ---
  // OPERATOR-ONLY by construction: seed_metadata describes how WE sourced a
  // listing, not a fact about the creator, so these two routes are the ONLY
  // place it is ever readable. Nothing here leaks onto a public payload
  // (asserted in tests/invariants/seeding.test.ts, a protected guard, against
  // the passport view / feed / SEO / proof responses).
  app.get("/ops/seeding/metrics", async (c) => {
    return c.json(await seedingMetrics(db(c)));
  });
  app.get("/ops/seeding/list", async (c) => {
    const tier = c.req.query("tier");
    const source = c.req.query("source");
    const batch = c.req.query("batch");
    return c.json(
      await seedingList(db(c), {
        ...(tier ? { tier } : {}),
        ...(source ? { source } : {}),
        ...(batch ? { batch } : {}),
        limit: Number(c.req.query("limit") ?? "50"),
        offset: Number(c.req.query("offset") ?? "0"),
      }),
    );
  });

  // --- M5: the fan nomination queue (flag 4) ---
  // Ordered by distinct-nominator count DESC — that count is the whole
  // prioritization signal and exists nowhere public. Marking a nomination
  // `listed` records the DECISION ONLY: it creates no passport. The operator
  // then runs a normal profile import, so minting stays one human-checked path.
  app.get("/ops/nominations", async (c) => {
    const state = c.req.query("state");
    return c.json(
      await nominationQueue(db(c), {
        ...(state ? { state } : {}),
        limit: Number(c.req.query("limit") ?? "50"),
        offset: Number(c.req.query("offset") ?? "0"),
      }),
    );
  });
  app.post("/ops/nominations/:id/list", async (c) => {
    try {
      const r = await decideNomination(db(c), actor(c), c.req.param("id"), "listed", null);
      return r.ok ? c.json(r) : c.json({ error: r.error }, 409);
    } catch (e) {
      return mapError(c, e);
    }
  });
  app.post("/ops/nominations/:id/decline", async (c) => {
    // I7 #618: same gate as the removal decline, refused before any state is read or written.
    const note = declineNote(await safeJson(c.req.raw));
    if (typeof note !== "string") return c.json({ error: note.error }, 400);
    try {
      const r = await decideNomination(db(c), actor(c), c.req.param("id"), "declined", note);
      return r.ok ? c.json(r) : c.json({ error: r.error }, 409);
    } catch (e) {
      return mapError(c, e);
    }
  });

  // ---------------------------------------------------------------------------
  // BRAND ENROLMENTS (slice 35) — proofs waiting for a person.
  //
  // A row here is a domain whose control somebody PROVED, asking to be listed.
  // The proof makes the operator's job confirmation rather than investigation;
  // it does not make the decision. `brand-enrolment.ts` has the full argument.
  app.get("/ops/brand-enrolments", async (c) => {
    const state = c.req.query("state");
    const lane = c.req.query("lane");
    const d = db(c);
    const q = await brandEnrolQueue(d, {
      ...(state ? { state } : {}),
      ...(lane ? { lane } : {}),
      limit: Number(c.req.query("limit") ?? "50"),
      offset: Number(c.req.query("offset") ?? "0"),
    });
    // QUEUE #178 AS-056: THE BANKED SIGN-OFF IS SHOWN. The scrutiny approval is
    // keyed on `${id}|scrutiny|${name}`, so a second operator used to have to
    // guess the first one's exact spelling with nothing on screen saying anyone
    // was waiting. Each open scrutiny row now carries every LIVE sign-off on it —
    // who, when, and the name it was for — read from the gate's own table with
    // the gate's own cutoff and person count. The name is recovered from the key
    // only when re-deriving the key from it is byte-exact (`boundedKey`); a
    // hashed long key shows `display_name: null` rather than a guess.
    const nowIso = new Date(nowMs()).toISOString();
    const required = requiredApprovalsFor(c.env as Record<string, unknown>, "brand_enrol:approve");
    const rows = await Promise.all(
      q.rows.map(async (r) => {
        if (r.state !== "open" || r.lane !== "scrutiny") return r;
        const prefix = `${r.enrolment_id}|scrutiny|`;
        const banked = await liveSignoffsByPrefix(d, "brand_enrol:approve", prefix, nowIso);
        const labels = await operatorLabels(d, banked.flatMap((b) => b.signed.map((s) => s.op_id)));
        const signoffs = await Promise.all(
          banked.map(async (b) => {
            const rest = b.params_hash.slice(prefix.length);
            const exact = (await boundedKey(prefix + rest)) === b.params_hash;
            return { display_name: exact ? rest : null, approvals: b.approvals, required, signed: b.signed.map((s) => ({ by: labelFor(labels, s.op_id), at: s.at })) };
          }),
        );
        return { ...r, signoffs };
      }),
    );
    return c.json({ rows });
  });

  // APPROVE — this is the one route in the slice that MINTS, and the scrutiny
  // lane is why it wears dual control.
  //
  // A standard-lane approval is an ordinary listing decision and goes through on
  // one operator, like every other import. A SCRUTINY-lane approval is a domain
  // `domainAddGate` says resembles a name already listed under somebody else —
  // so approving it is the act of deciding that a lookalike is legitimate, and
  // being wrong publishes an impersonator as `confirmed` under the product's own
  // standing. That is the same shape as publishing a warning, and it gets the
  // same two-person rule, keyed on the enrolment AND the name being granted:
  // changing either is a different decision and needs its own pair of eyes.
  app.post("/ops/brand-enrolments/:id/approve", async (c) => {
    const body = await safeJson(c.req.raw);
    const displayName = typeof body?.display_name === "string" ? body.display_name : "";
    const d = db(c);
    const op = actor(c);
    const id = c.req.param("id");
    // S7-4: the notice mails from INSIDE the executed callback both lanes
    // share, never on a merely-`pending` sign-off — `defer` closes over the
    // request's executionCtx the way :996-1003 does, so a pending decision
    // (dual control's first sign-off) never reaches here at all.
    let defer: (p: Promise<unknown>) => void;
    try {
      const ctx = c.executionCtx;
      defer = (p) => ctx.waitUntil(p);
    } catch {
      defer = (p) => void Promise.resolve(p).catch(() => {});
    }
    const notifyApproved = (domain: string, result: { passport_id: string; contact_email: string | null }): void => {
      if (!result.contact_email) return;
      defer(
        sendDecisionMail(
          mailerFromEnv(c.env as Record<string, unknown>),
          d,
          op,
          "brand-enrol-approved",
          result.contact_email,
          { brand_domain: domain, passport_id: result.passport_id },
          id,
          new Date(nowMs()).toISOString(),
        ),
      );
    };
    try {
      const row = await d.prepare("SELECT lane, state, domain FROM brand_enrolments WHERE enrolment_id = ?").bind(id).first<{ lane: string; state: string; domain: string }>();
      if (!row) return c.json({ error: "no such enrolment" }, 404);
      // VALIDATE BEFORE BANKING A SIGN-OFF. Slice 21's rule — "a decision that
      // can never execute banks NOTHING" — and this route broke it three ways:
      // an empty name, an over-long name and an ALREADY-DECIDED enrolment each
      // returned "Recorded your approval (1 of 2)" and left a permanent
      // `pending_actions` row, because the `state` read here was never used
      // (adversarial review, MEDIUM).
      if (row.state !== "open") return c.json({ error: "that enrolment is already decided" }, 409);
      if (!validDisplayName(displayName.trim())) return c.json({ error: "a display name is required: up to 100 characters, no angle brackets, no verification mark, no platform name" }, 400);
      const lane = row.lane === "scrutiny" ? "scrutiny" : "standard";
      if (lane === "standard") {
        const result = await approveBrandEnrol(d, id, op, displayName, { lane, dualControlSatisfied: false });
        notifyApproved(row.domain, result);
        return c.json(result);
      }
      // The params key carries the LANE as well as the id and the name: two
      // operators must be agreeing about the same decision, and "approve this
      // scrutiny row as X" is not the same decision as "approve this standard
      // row as X". `approveBrandEnrol` re-reads the lane and refuses if it moved.
      const outcome = await withDualControl(d, c.env as Record<string, unknown>, op, "brand_enrol:approve", `${id}|${lane}|${displayName.trim()}`, async () => {
        const result = await approveBrandEnrol(d, id, op, displayName, { lane, dualControlSatisfied: true });
        notifyApproved(row.domain, result);
        return result;
      }, undefined, id);
      if (!outcome.executed) return c.json({ ok: true, pending: true, approvals: outcome.approvals, required: outcome.required });
      return c.json(outcome.result);
    } catch (e) {
      return mapError(c, e);
    }
  });

  app.post("/ops/brand-enrolments/:id/decline", async (c) => {
    const body = await safeJson(c.req.raw);
    // #177 AS-054: a decline closes that domain (until a dual-controlled reopen) and `declineBrandEnrol`
    // writes the note as the audit detail, so an empty note is an audit line
    // that is silent on why. Refused before any state is read or written.
    const note = brandDeclineNote(body);
    if (typeof note !== "string") return c.json({ error: note.error }, 400);
    try {
      const result = await declineBrandEnrol(db(c), c.req.param("id"), actor(c), note);
      if (result.contact_email) {
        let defer: (p: Promise<unknown>) => void;
        try {
          const ctx = c.executionCtx;
          defer = (p) => ctx.waitUntil(p);
        } catch {
          defer = (p) => void Promise.resolve(p).catch(() => {});
        }
        defer(
          sendDecisionMail(
            mailerFromEnv(c.env as Record<string, unknown>),
            db(c),
            actor(c),
            "brand-enrol-declined",
            result.contact_email,
            { brand_domain: result.domain },
            c.req.param("id"),
            new Date(nowMs()).toISOString(),
          ),
        );
      }
      return c.json({ ok: true });
    } catch (e) {
      return mapError(c, e);
    }
  });

  // REOPEN — QUEUE #163 / F-30. `reopenBrandEnrol` was built, tested and
  // dual-control gated in the primitive, with no door. Mounted beside
  // approve/decline under the SAME seam: the params key is the enrolment and
  // the state being undone, so two operators must agree to look again at the
  // same decline. Validated before banking (slice 21): a missing row, a row that
  // is not declined, or an empty note banks nothing.
  app.post("/ops/brand-enrolments/:id/reopen", async (c) => {
    const body = await safeJson(c.req.raw);
    const note = typeof body?.note === "string" ? body.note.trim() : "";
    const d = db(c);
    const op = actor(c);
    const id = c.req.param("id");
    try {
      const row = await d.prepare("SELECT state FROM brand_enrolments WHERE enrolment_id = ?").bind(id).first<{ state: string }>();
      if (!row) return c.json({ error: "no such enrolment" }, 404);
      if (row.state !== "declined") return c.json({ error: "only a declined enrolment can be reopened" }, 409);
      if (!note || !hasVisibleText(note)) return c.json({ error: "note required" }, 400);
      const noteDigest = await planTextDigest(note);
      const outcome = await withDualControl(d, c.env as Record<string, unknown>, op, "brand_enrol:reopen", `${id}|declined|${noteDigest}`, async () =>
        reopenBrandEnrol(d, id, op, { dualControlSatisfied: true }, note, () => new Date(nowMs()).toISOString()), undefined, id,
      );
      if (!outcome.executed) return c.json({ ok: true, pending: true, approvals: outcome.approvals, required: outcome.required });
      return c.json({ ok: true, reopened: true, domain: outcome.result?.domain ?? null });
    } catch (e) {
      return mapError(c, e);
    }
  });

  // ---------------------------------------------------------------------------
  // BRAND CONTESTS (D-0561 / J-02) — TWO DOMAIN-PROVEN CLAIMANTS AND ONE NAME.
  //
  // A row here is a filing by somebody who proved DNS control of a host the
  // lookalike gate refused, naming the listing that refused them and the
  // passport holding it. Both parties are domain-proven: the machine cannot
  // decide between them, which is the whole reason there is a queue.
  //
  // WHAT AN ADMIT DOES AND DOES NOT DO (D-0594) is in `admitBrandContest`'s
  // docblock: it attaches the CONTESTED host to the filer's passport and records
  // the conflict; it does not strip, suppress or re-fold the incumbent's brand,
  // and nothing reads the record to decide what to serve.
  app.get("/ops/brand-contests", async (c) => {
    const state = c.req.query("state");
    const lane = c.req.query("lane");
    return c.json(
      await brandContestQueue(db(c), {
        ...(state ? { state } : {}),
        ...(lane ? { lane } : {}),
        limit: Number(c.req.query("limit") ?? "50"),
        offset: Number(c.req.query("offset") ?? "0"),
      }),
    );
  });

  // ADMIT — dual control, always. Every contest row is born in the `scrutiny`
  // lane because a gate hit is what creates it, so unlike the enrolment approve
  // route there is no one-operator path here at all; `admitBrandContest` refuses
  // a lone admit IN THE PRIMITIVE as well, for the reasons
  // `approveBrandEnrol`'s docblock records about a route-only check.
  app.post("/ops/brand-contests/:id/admit", async (c) => {
    const d = db(c);
    const op = actor(c);
    const id = c.req.param("id");
    try {
      const row = await d.prepare("SELECT lane, state, incumbent_passport_id, filer_passport_id FROM brand_contests WHERE contest_id = ?").bind(id).first<{ lane: string; state: string; incumbent_passport_id: string; filer_passport_id: string }>();
      if (!row) return c.json({ error: "no such contest" }, 404);
      // VALIDATE BEFORE BANKING A SIGN-OFF — slice 21's rule, and the defect the
      // enrolment approve route shipped: a decision that can never execute must
      // bank nothing, or the first operator's signature sits in `pending_actions`
      // forever against an already-decided row.
      if (row.state !== "open") return c.json({ error: "that contest is already decided" }, 409);
      const lane = row.lane === "standard" ? "standard" : "scrutiny";
      // The params key carries the LANE and the INCUMBENT as well as the id: two
      // operators must be agreeing about the same decision, and "admit this host
      // against incumbent X" is not the same decision as "admit it against Y".
      // I16 #932d (I16-W01H hand-off H1): and the FILER — a refile by another
      // passport rewrites `filer_passport_id` on the open row, so sign-offs banked
      // for filer A must not execute for filer B; the primitive re-checks it.
      const outcome = await withDualControl(d, c.env as Record<string, unknown>, op, "brand_contest:admit", `${id}|${lane}|${row.incumbent_passport_id}|${row.filer_passport_id}`, async () =>
        admitBrandContest(d, id, op, { lane, dualControlSatisfied: true, filer: row.filer_passport_id }), undefined, id,
      );
      if (!outcome.executed) return c.json({ ok: true, pending: true, approvals: outcome.approvals, required: outcome.required });
      return c.json(outcome.result);
    } catch (e) {
      return mapError(c, e);
    }
  });

  app.post("/ops/brand-contests/:id/decline", async (c) => {
    const body = await safeJson(c.req.raw);
    // RG2-B1-CONTEST-NOTE — the enrolment decline's #177 gate, same shape: a
    // decline is a final one-shot decision and `declineBrandContest` writes the
    // note as the audit detail, so an empty note is an audit line silent on why.
    // Refused before any state is read or written.
    const note = brandDeclineNote(body);
    if (typeof note !== "string") return c.json({ error: note.error }, 400);
    try {
      await declineBrandContest(db(c), c.req.param("id"), actor(c), note);
      return c.json({ ok: true });
    } catch (e) {
      return mapError(c, e);
    }
  });

  return app;
}

// THE ROLE POLICY — one explicit line per route, reads included.
// ---------------------------------------------------------------------------
/**
 * S53's original guard exempted GET/HEAD wholesale, on the assumption that
 * "reading is not the dangerous half". That assumption shipped a hole:
 * `GET /ops/accounts/:id/export` assembles a stranger's ENTIRE personal-data
 * package (email, OAuth identities, saves, collection, acquisitions, profile,
 * follows, requests, notifications) and a support seat could pull it by typing
 * the URL into the address bar with the Access cookie attached — while three
 * places in this codebase said in writing that it was 403. A method-based
 * default also means the NEXT bulk-PII read is exposed the moment it is added,
 * silently.
 *
 * So the decision is per route and written down here, for every method:
 *  - a route absent from this table is FULL-OPERATOR-ONLY (fail closed);
 *  - `role: "support"` is a deliberate, reviewed grant;
 *  - `unless` narrows a grant by request shape (the import dry-run).
 * `trust-core-routes` + `console-roles.test.ts` assert the table and the routes
 * cannot drift apart in either direction.
 *
 * READ DECISIONS, and why the support seat keeps them: the seat exists to work
 * the triage queues and answer a named subject's question, so it sees queues,
 * per-subject dossiers, aggregates and health. What it does NOT get is the one
 * read that ACTS on a subject's behalf (the DSAR export — legal weight, needs a
 * named accountable operator on the audit line, exactly like erasure) and the
 * one read that replays the whole ledger (`/ops/fold/verify`).
 *
 * CLOSED (was a known gap): `GET /ops/subscriptions` with no `q` used to return
 * every subscriber's email + Stripe customer state to a support seat — a browse
 * of strangers, precisely what `/ops/accounts` refuses ("you cannot arrive at a
 * stranger by wandering"). The route now applies BOTH narrowings: no search
 * term ⇒ no rows for a support seat, and the unfiltered listing is full-only.
 * The grant below stays `support` because the SEARCH is the seat's job; the
 * narrowing lives in the handler (with the reasoning) because it depends on the
 * request's shape AND the caller's tier, which `unless` alone cannot express,
 * and because the answer must be a renderable empty state rather than a 403.
 */
export type OpsRole = "full" | "support";

export interface OpsRoutePolicy {
  /** HTTP method, uppercase. HEAD is matched by the GET entry. */
  method: "GET" | "POST" | "DELETE";
  /** Hono-shaped path with `:param` segments — matched segment-wise. */
  path: string;
  /** Minimum seat that may call it. */
  role: OpsRole;
  /** For a `support` grant only: narrows it away when this returns true. */
  unless?: (url: URL) => boolean;
  /** Why this seat may (or may not) call it — kept next to the decision. */
  why: string;
}

/** Segment-wise match of a concrete path against a `:param` pattern. */
function pathMatches(pattern: string, pathname: string): boolean {
  const a = pattern.split("/");
  const b = pathname.split("/");
  if (a.length !== b.length) return false;
  return a.every((seg, i) => (seg.startsWith(":") ? b[i] !== undefined && b[i] !== "" : seg === b[i]));
}

/**
 * The declaration for one request, or null when the route is undeclared (which
 * the guard treats as full-only). HEAD is answered by the GET declaration —
 * a HEAD is a GET whose body is dropped, never a way around the table.
 */
/** I38-OPSUNDO (DRAFT wording, REPORT.md ## JASON): the refunds-owed worklist label for a refused settlement that was also credited. */
export const SETTLEMENT_ALSO_CREDITED_LABEL = "Also credited: credits for this purchase were granted to the account.";

export function matchOpsPolicy(method: string, pathname: string): OpsRoutePolicy | null {
  const m = method.toUpperCase() === "HEAD" ? "GET" : method.toUpperCase();
  return OPS_ROLE_POLICY.find((r) => r.method === m && pathMatches(r.path, pathname)) ?? null;
}

/**
 * I7-FX1 #618: a nomination decision is ONE guarded batch. `setNominationState` has no
 * state guard, so it is not used here: the flip, the note (declines only) and the audit
 * line all bind `state = 'open'`, and the audit line is predicated on that guard, so a
 * row that is already listed/declined keeps its state and note and records nothing.
 */
async function decideNomination(d: DbLike, operator: string, id: string, to: "listed" | "declined", note: string | null): Promise<{ ok: true; state: "listed" | "declined" } | { ok: false; error: string }> {
  const cur = await d.prepare("SELECT state FROM nominations WHERE nomination_id = ?").bind(id).first<{ state: string }>();
  if (!cur) return { ok: false, error: "no such nomination" };
  if (cur.state !== "open") return { ok: false, error: "this nomination is already decided" };
  await d.batch([
    d
      .prepare(
        `INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind)
           SELECT ?, ?, '', ?, ?, ?
            WHERE EXISTS (SELECT 1 FROM nominations WHERE nomination_id = ? AND state = 'open')`,
      )
      .bind(operator, `nomination:${to}`, new Date().toISOString(), id, auditActorKind(operator), id),
    d.prepare("UPDATE nominations SET state = ?, decision_note = ? WHERE nomination_id = ? AND state = 'open'").bind(to, note, id),
  ]);
  const after = await d.prepare("SELECT state, decision_note FROM nominations WHERE nomination_id = ?").bind(id).first<{ state: string; decision_note: string | null }>();
  if (after?.state !== to || after.decision_note !== note) return { ok: false, error: "this nomination is already decided" };
  return { ok: true, state: to };
}

/** I7 #618: the most characters of an operator's decline note (migration 0156's CHECK repeats it). */
export const DECLINE_NOTE_MAX = 1000;
/** The operator's decline note, trimmed at the ends only (never rewritten), that has at least one character a reader can see (I8-FX4; zero-width, filler and blank-glyph notes are no reason), or the 400 error word. */
function declineNote(body: Record<string, unknown> | null): string | { error: string } {
  const note = typeof body?.note === "string" ? body.note.trim() : "";
  if (!note || !hasVisibleText(note)) return { error: "note required" };
  if (note.length > DECLINE_NOTE_MAX) return { error: "note too long" };
  return note;
}

/** I8-FX2: a brand decline note is `declineNote` (string, trimmed, 1..1000) AND has a character a reader can see; same words, same 400. */
function brandDeclineNote(body: Record<string, unknown> | null): string | { error: string } {
  const note = declineNote(body);
  if (typeof note !== "string") return note;
  return hasVisibleText(note) ? note : { error: "note required" };
}

async function safeJson(req: Request): Promise<Record<string, unknown> | null> {
  try {
    const v = (await req.json()) as unknown;
    return typeof v === "object" && v !== null && !Array.isArray(v) ? (v as Record<string, unknown>) : null;
  } catch {
    return null;
  }
}

 * one never implies the other, in either direction. That is deliberate: a
 * fallback is exactly how a single list silently over-grants, and an operator

 * influence — no request header, body, path or query reaches it — so this
 * cannot be used to write attacker-chosen text into the logs. The behaviour is
 * unchanged: the entry is still dropped, never repaired.
 */
export function corsAllowedOrigins(
  raw: string | undefined,
  envKey: CorsEnvKey = "CORS_ORIGINS",
): readonly string[] {
  if (typeof raw !== "string") return [];
  const key = `${envKey} ${raw}`;
  const hit = parseCache.get(key);
  if (hit) return hit;

  const kept: string[] = [];
  const dropped: string[] = [];
  for (const part of raw.split(",")) {
    const entry = part.trim().toLowerCase();
    if (entry === "") continue; // an empty slot is a formatting artifact, not a typo
    // I63 C2 (W0-10, I62-CC-16): the setting rule per ENTRY — an entry holding TODO in any case is dropped (and
    // announced below); the rest of the list is kept. Never settingValue on the whole value.
    if (settingValue(entry) !== null && ORIGIN_RE.test(entry)) {
      // FM-02: a browser's ASCII origin serialization omits the HTTPS default
      // port (WHATWG URL "port state" / HTML "ascii serialization of an
      // origin") — an operator who wrote the explicit ":443" the parser
      // happily accepts produced a configured entry no real https:// request
      // can ever equal, a silent CORS/passkey-origin outage with no warning.
      // Canonicalize it away here, once, so every caller of this list (and of
      // `matchCorsOrigin`, exact-equality only) compares like the browser
      // does. Nondefault ports are untouched — only a literal trailing
      // ":443" is stripped, never a substring of a different port.
      const canonical = entry.replace(/:443$/, "");
      kept.push(canonical);
    } else dropped.push(entry);
  }
  for (const entry of dropped) {
    logWarn({ cid: "config", hop: "http", msg: "cors: ignoring malformed origin entry (expected an https origin with no path, e.g. https://app.example.com)", env_key: envKey, entry });
  }

  const frozen = Object.freeze(kept);
  if (parseCache.size >= PARSE_CACHE_MAX) parseCache.clear();
  parseCache.set(key, frozen);
  return frozen;
}

/**
 * The allowlist entry matching this request's Origin, or null. Exact equality
 * only.
 *
 * IT RETURNS THE CONFIGURED ENTRY, NEVER THE ARGUMENT. That is the property the
 * caller depends on: whatever comes back from here is emitted verbatim in
 * `Access-Control-Allow-Origin` next to `Access-Control-Allow-Credentials:
 * true`, so it must be a string this deployment's operator wrote, not bytes the
 * caller sent. Today `find(o => o === origin)` makes the two identical, which
 * is precisely why it is easy to lose: relax the comparison even slightly (a
 * `.toLowerCase()` on the request header, say, "to be lenient about case") and
 * returning `origin` starts putting caller-controlled bytes in the header while
 * every equality-based test stays green. Return the ENTRY. cors.test.ts pins
 * this with a case-varied Origin, which is red for both mistakes.
 */
export function matchCorsOrigin(
  raw: string | undefined,
  origin: string | undefined,
  envKey: CorsEnvKey = "CORS_ORIGINS",
): string | null {
  if (!origin) return null;
  return corsAllowedOrigins(raw, envKey).find((o) => o === origin) ?? null;
}

/**
 * Registered as `app.use("*", cors())` so it covers the whole surface,
 * including a preflight to a path that does not exist (403, not 404 — a 404
 * reads as a routing bug to whoever is debugging it).
 */
/**
 * Union two `Vary` values, case-insensitively, preserving the order they were
 * added. `Vary` is a comma-separated LIST — a second writer must add to it, not
 * replace it, and a duplicated token makes a cache key twice for one dimension.
 */
export function mergeVary(existing: string | null | undefined, add: string): string {
  const seen = new Set<string>();
  const out: string[] = [];
  for (const raw of [...(existing ?? "").split(","), ...add.split(",")]) {
    const v = raw.trim();
    if (!v) continue;
    const k = v.toLowerCase();
    if (seen.has(k)) continue;
    seen.add(k);
    out.push(v);
  }
  return out.join(", ");
}

export function cors(): MiddlewareHandler<{ Bindings: CorsEnv }> {
  return async (c, next) => {
    // Which surface, and therefore which list. Decided per request from the
    // path, because one list spanning both surfaces is what over-granted.
    const surface: CorsSurface = isOpsPath(c.req.path) ? "ops" : "public";
    const envKey = corsEnvKey(surface);
    const configured = surface === "ops" ? c.env?.OPS_CORS_ORIGINS : c.env?.CORS_ORIGINS;
    const allowed = matchCorsOrigin(configured, c.req.header("origin"), envKey);

    if (c.req.method === "OPTIONS") {
      // Preflight carries no credentials, so it is answered here, ahead of the
      // Access gate on /ops/* — which would otherwise 401 it and break the
      // real request that follows.
      if (!allowed) return c.json({ error: "forbidden" }, 403, { Vary: "Origin" });
      return c.body(null, 204, {
        // `allowed` is the CONFIGURED entry, never c.req.header("origin").
        "Access-Control-Allow-Origin": allowed,
        "Access-Control-Allow-Credentials": "true",
        "Access-Control-Allow-Methods": ALLOW_METHODS,
        "Access-Control-Allow-Headers": ALLOW_HEADERS,
        "Access-Control-Max-Age": MAX_AGE,
        Vary: "Origin",
      });
    }

    await next();

    // AFTER next(), so this also lands on responses a handler built as a raw
    // `new Response(...)` and on the notFound/onError responses.
    //
    // MERGED, NOT SET, and that is a real bug fixed rather than tidiness.
    // `c.header` REPLACES, and the i18n seam's routes set
    // `Vary: Accept-Language` inside their handler — so this line silently
    // deleted it on every server-rendered page. Nothing noticed because only one
    // catalogue ships, so the branch that adds it is dead; the day a second
    // language lands, `/discover`, `/browse/:category`, `/is/:platform/:handle`
    // and `/p/:id` would have served locale-varying bodies under
    // `Cache-Control: public, max-age=300` advertising only `Vary: Origin`, and
    // a shared cache would hand the first reader's language to everybody else.
    // The seam's promise — "the day a second language lands the header appears
    // with it and nobody has to remember" — was false, in this line.
    c.header("Vary", mergeVary(c.res.headers.get("Vary"), "Origin"));
    if (allowed) {
      // Again: the configured entry. Nothing caller-controlled reaches a header.
      c.header("Access-Control-Allow-Origin", allowed);
      c.header("Access-Control-Allow-Credentials", "true");
    }
  };
}

    scheduled_clear_cents: number | null;
  } | null;
  /** Served event cells only. `id` is opaque; `kind` is a module constant. */
  events: Array<{ id: string; kind: string; participants: number | null; new_fans: number | null; acquired_d30: number | null }>;
  /** Served cohort cells only, per (cohort day, closed-enum source). */
  cohorts_by_source: Array<{ cohort_day: string; source: string; size: number | null; d1: number | null; d7: number | null; d30: number | null }>;
}

/**
 * The served cells of `analytics-read.ts DaySummary` this seam reads.
 *
 * A STRUCTURAL SUBSET, declared rather than imported (see the module docblock).
 * A `DaySummary` is assignable to it; the compiler checks that at the call site
 * and `test/s002-summary.test.ts` checks it against an object built by
 * `buildDayObject` rather than against a literal.
 */
export interface SummaryInput {
  readonly day: string;
  readonly served_day: string;
  readonly revenue: Readonly<Record<string, number | null>>;
  readonly geo: { readonly active_fans: number | null; readonly countries: ReadonlyArray<{ readonly country: string; readonly fan_days: number }>; readonly other: number | null };
  readonly cohorts: ReadonlyArray<{ readonly cohort_day: string; readonly source: string; readonly size: number | null; readonly d1: number | null; readonly d7: number | null; readonly d30: number | null }>;
  readonly events: ReadonlyArray<{ readonly kind: string; readonly id: string; readonly participants: number | null; readonly new_fans: number | null; readonly acquired_d30: number | null }>;
  readonly stocks: { readonly supporters: number | null };
}

/** Is anything in this row served at all? A row with nothing served is DROPPED
 * rather than listed as a row of withheld cells: the row's existence is itself a
 * fact about the creator's events, and the standing instruction in `SYSTEM`
 * already tells the model the lists are incomplete. */
const anyServed = (...vs: Array<number | null>): boolean => vs.some((v) => v !== null);

export function projectForBrief(intel: CreatorIntelligence, summary?: SummaryInput | null): BriefInput {
  const s = summary ?? null;
  return {
    collectors_total: intel.audience.collectors_total,
    suppressed: intel.audience_suppressed,
    repeat: intel.audience.repeat_collectors,
    one_time: intel.audience.one_time_collectors,
    new_90d: intel.audience.new_collectors_90d,
    dormant: intel.audience.dormant_collectors,
    // The bucket LABELS are module constants (`intelligence.ts DEPTH_BUCKETS`),
    // not anything a creator or a fan can set — the one string that crosses.
    depth: intel.depth_distribution,
    completion_eligible: intel.completion.eligible,
    completed: intel.completion.completed_collectors,
    near_complete: intel.completion.near_completion_collectors,
    releases: intel.catalog.releases_published,
    acquisitions_total: intel.catalog.acquisitions_total,
    acquisitions_90d: intel.catalog.acquisitions_90d,
    archive_requests: intel.demand.archive_requests,
    // `integrity` is deliberately absent: it is the only part of
    // CreatorIntelligence carrying free text, and an abuse signal is not
    // something a creator needs a model's prose about.
    day: s === null ? null : s.day,
    supporters: s === null ? null : s.stocks.supporters,
    active: s === null ? null : s.geo.active_fans,
    revenue:
      s === null
        ? null
        : {
            gross_credits: s.revenue.gross_credits ?? null,
            creator_net_cents: s.revenue.creator_net_cents ?? null,
            paid_cents: s.revenue.paid_cents ?? null,
            scheduled_clear_cents: s.revenue.scheduled_clear_cents ?? null,
          },
    // SERVED CELLS ONLY, and the id/kind pass a shape gate before they cross:
    // an unknown kind or a non-opaque id is dropped, so the failure mode of a
    // new event kind is a missing row and never a creator's own title in a
    // prompt.
    events:
      s === null
        ? []
        : s.events
            .filter((e) => EVENT_KIND_LABELS[e.kind] !== undefined && OPAQUE_ID_RE.test(e.id) && anyServed(e.participants, e.new_fans, e.acquired_d30))
            .map((e) => ({ id: e.id, kind: EVENT_KIND_LABELS[e.kind] as string, participants: e.participants, new_fans: e.new_fans, acquired_d30: e.acquired_d30 })),
    cohorts_by_source:
      s === null
        ? []
        : s.cohorts
            .filter((c) => BRIEF_COHORT_SOURCES.has(c.source) && anyServed(c.size, c.d1, c.d7, c.d30))
            .map((c) => ({ cohort_day: c.cohort_day, source: c.source, size: c.size, d1: c.d1, d7: c.d7, d30: c.d30 })),
  };
}

/**
 * THE GEO BLOCK IS A PROJECTION OF THE SEALED DAY, NOT A LIVE READ (ADR-0006
 * section 7: "geo via the `AudienceRegions` rule rewritten for per-day cells").
 *
 * `projectSummary` already put the served country cells in the object
 * (`analytics-read.ts geo`), floored once by `serveDay` with everything else.
 * `explain` used to call `regions.ts audienceRegions(db, id)` beside the frozen
 * reads — a second store on a second clock (`dayOf(now - 1 day)`), so one
 * response carried a sealed summary beside a live geo block about a different
 * day, and the response cache keyed on `served_day` alone could not tell the
 * two apart. This adapter is the whole of the difference: the same four
 * branches in `renderBrief` below, fed from the object.
 */
export function regionsFromSummary(s: SummaryInput | null | undefined): AudienceRegions | null {
  if (s === null || s === undefined) return null;
  const countries = s.geo.countries.map((c) => ({ country: c.country, fans: c.fan_days }));
  const accounted = countries.reduce((n, c) => n + c.fans, 0) + (s.geo.other ?? 0);
  // `withheld` = the served cells do not enumerate the day (`regions.ts GeoServed`).
  return { day: s.day, active_fans: s.geo.active_fans, countries, other: s.geo.other, withheld: s.geo.active_fans === null || accounted !== s.geo.active_fans };
}

/**
 * A RATIO IS SERVED ONLY WHEN BOTH OPERANDS ARE (ADR-0006 §6 row 3, INV-052).
 *
 * The same rule `analytics-cells.ts ratio` applies to a `ServedSet`, applied
 * here to two already-served values — this module cannot import that function
 * without widening its import graph, and `test/s002-summary.test.ts` asserts
 * the two agree case for case rather than leaving the duplication unchecked.
 */
export function servedRatio(numerator: number | null, denominator: number | null): number | null {
  if (numerator === null || denominator === null || denominator === 0) return null;
  return numerator / denominator;
}

/** A withheld cell leaves as a SENTENCE, never as a number, a bound or a
 * statement that the thing it counts did or did not happen. One spelling, so a

/** I26-W1-CONTENT (I3-0151 money limb, D-0535(b)): the money line's four labels, ONE constant read by the renderer

  { label: "SCHEDULED to clear this day, in cents — revisable by a later reversal, and never to be called cleared or paid", key: ["scheduled"] },

/** I26-FX3 (D-1017): the SERVED NON-money measures' distinctive label words (`labelWords` form), each with its fixed
 * synonyms — the words a reply clause names to be ABOUT that measure on a withheld day (`groundedClause`; I28-AI: any withheld day). Kept
 * beside `MONEY_LABELS` so the two lists are read together: no word of a money label is, or may become, one of these. */
export const GROUNDING_LABELS: ReadonlyMap<string, readonly string[]> = new Map<string, readonly string[]>([
  ["audience", []], ["repeat", ["returning"]], ["dormant", ["lapsed", "inactive"]], ["depth", []], ["completion", ["complete", "completed"]],

const moneyLabel = (i: number): string => (MONEY_LABELS[i] as { label: string }).label;
/** The whole money line's withheld label (`revenue === null`). */
const MONEY_LINE_LABEL = "Money for this day";
/** I26-FX-AI (D-1015(1)): the prompt half of the money-topic refusal (`claimsWithheldMoney`), pushed on any day a money cell is
 * withheld. No numeral and no `<label>: WITHHELD` shape, so it adds nothing to `numeralPhrasesOf` or `withheldLabelsOf`. */
export const MONEY_DAY_RULE =
  "Money rule for this packet: a money measure is withheld today, so do not discuss money in any form, in any words: no earnings, income, takings, spending, paying, buying, prices, worth, cash, currency or amounts, and no judgement of how the money went.";
const withheldSentence = (what: string): string =>
  `${what}: WITHHELD — no figure is available. Do not state, estimate, bound or imply one, and do not mention this measure in your reply at all — not its name, a comparison, or whether it happened.`;

/**
 * WHAT THE PACKET ACTUALLY SAID, split by what it said it AS — the allowlist the
 * output half compares a reply against (`explainIntelligence` `:607`, `restatesPacket` `:667`).
 *
 * `renderBrief` prints three kinds of text and only ONE of them is a served
 * quantity. `figures` are the values it interpolated for served cells.
 * `keys` are the identifiers it printed BESIDE those values — the sealed day,
 * a cohort day, an opaque event id, a country code, a depth bucket label: a
 * reply may restate a key AS ITSELF and never as a quantity. Everything else in
 * the brief is FIXED LABEL PROSE (`in the last 90 days`, `after 7 days`,
 * `30-day retention rate`) and contributes nothing at all, which is the whole
 * of the E-2 correction: the label a WITHHELD cell hangs on is not a figure.
 */
export interface BriefPacket {
  figures: Set<string>;
  keys: Set<string>;
  /** C-22 fix: an exact decimal STRING `fig()` actually printed (e.g. a
   * retention rate). A decimal is never admitted by splitting it into its two
   * digit runs and checking each independently against `figures` — that would
   * let a model mint an invented rate out of any two UNRELATED served
   * integers either side of a ".", which is not a restatement of anything the
   * packet said. Only the literal decimal `fig()` printed restates it. */
  decimals: Set<string>;
}

/** A served VALUE, recorded and returned as the packet prints it. A `null` is
 * printed by `numberOrWithheld` as the withheld sentence and never reaches here. */
const fig = (p: BriefPacket, v: number | string): string => {
  const printed = String(v);
  // C-22: a decimal is ONE served cell, not two. Splitting `12.5` into `12`
  // and `5` and admitting each into `figures` independently mints an
  // allowlist entry for a cell that was never served (the bare `5`), which a
  // model can later spend on an unrelated invented quantity. A decimal is
  // recorded as the exact string printed instead, and matched whole.
  if (/^\d+\.\d+$/.test(printed)) {
    p.decimals.add(printed);
    return printed;
  }
  for (const n of integersOf(printed)) p.figures.add(n);
  return printed;
};

/** A KEY the packet prints beside a value. A key with no digit can neither
 * admit nor refuse a quantity and is not recorded. A key that is NOTHING BUT
 * digits (only a depth bucket label can be — `intelligence.ts:193` makes the
 * whole block `null` in production) is indistinguishable from a quantity when a
 * model writes it back, so it is recorded as a FIGURE rather than as a key: a
 * bare number may not be restatable-as-itself, because that is a hole any
 * invented number fits through. */
const key = (p: BriefPacket, k: string): string => {
  if (/^\d+$/.test(k)) fig(p, k);
  else if (/\d/.test(k)) p.keys.add(k);
  return k;
};

const numberOrWithheld = (p: BriefPacket, what: string, v: number | null, tail = ""): string => (v === null ? withheldSentence(what) : `${what}: ${fig(p, v)}${tail}.`);

/** UTF-8 bytes, which is what the cap is stated in (ADR-0006 §8). */
const byteLength = (s: string): number => new TextEncoder().encode(s).length;

/**
 * THE REDUCTION LADDER (S-18). Deterministic in both senses: the same input
 * gives the same brief, and the ORDER things are dropped in is fixed and stated
 * above — it does not depend on which block happens to be biggest.
 */
export function buildBrief(intel: BriefInput, regions: AudienceRegions | null, out?: BriefPacket): string {
  // Step 1. Whole days, most recent, in one stable order; events likewise.
  const days = [...new Set(intel.cohorts_by_source.map((c) => c.cohort_day))].sort();
  const keptDays = days.slice(-MAX_COHORT_DAYS);
  const cohorts = intel.cohorts_by_source
    .filter((c) => keptDays.includes(c.cohort_day))
    .sort((a, b) => (a.cohort_day === b.cohort_day ? (a.source < b.source ? -1 : a.source > b.source ? 1 : 0) : a.cohort_day < b.cohort_day ? -1 : 1));
  const events = [...intel.events].sort((a, b) => (a.kind === b.kind ? (a.id < b.id ? -1 : a.id > b.id ? 1 : 0) : a.kind < b.kind ? -1 : 1)).slice(0, MAX_EVENT_ROWS);

  // Steps 2-4, in order. `days: n` keeps the n most recent of the kept days.
  const ladder: Array<{ geo: boolean; events: boolean; days: number }> = [
    { geo: true, events: true, days: keptDays.length },
    { geo: false, events: true, days: keptDays.length },
    { geo: false, events: false, days: keptDays.length },
  ];
  for (let n = keptDays.length - 1; n >= 0; n--) ladder.push({ geo: false, events: false, days: n });

  for (const step of ladder) {
    const window = new Set(keptDays.slice(keptDays.length - step.days));
    const windowedCohorts = cohorts.filter((c) => window.has(c.cohort_day));
    // A FRESH ACCUMULATOR PER RUNG, copied out only on the rung that is
    // RETURNED: a block dropped for length was not sent, so it may not admit a
    // digit either.
    const p: BriefPacket = { figures: new Set(), keys: new Set(), decimals: new Set() };
    const brief = renderBrief(
      { ...intel, events: step.events ? events : [], cohorts_by_source: windowedCohorts },
      step.geo ? regions : null,
      // A BLOCK DROPPED FOR LENGTH IS NOT A BLOCK THAT WAS NOT THERE. Rung 2
      // drops `geo` whole; without this flag the packet it produces is
      // byte-identical to one for a creator with no regions at all, so a
      // WITHHELD geo and a DROPPED geo were the same silence.
      !step.geo && regions !== null,
      // R-07: the same distinction, extended to events and cohorts. `events`
      // is dropped WHOLE at this rung (never partially), so "omitted" is
      // exactly "this rung dropped it and there was something to drop".
      !step.events && events.length > 0,
      // Cohorts are dropped one whole day at a time, so the block reads as
      // "omitted" only once the window empties a list that was not empty to
      // start with — a window that still has room shows every day it covers.
      windowedCohorts.length === 0 && cohorts.length > 0,
      p,
    );
    if (byteLength(brief) <= MAX_BRIEF_BYTES) {
      if (out) {
        for (const f of p.figures) out.figures.add(f);
        for (const k of p.keys) out.keys.add(k);
        for (const d of p.decimals) out.decimals.add(d);
      }
      return brief;
    }
  }
  // Unreachable while the fixed blocks fit: below the last rung the brief is
  // the audience, catalog and revenue paragraphs, which are bounded by their
  // own shape. Kept as the floor rather than returning something over the cap.
  throw new CreatorAiError("the explanation packet is too large to send", 502, AI_NO_ANSWER_MARKER); // LEAD-creator-ai-422 (I16-W02H): no inference ran, so the route refunds the debited slot (intelligence-routes.ts:310).
}

function renderBrief(intel: BriefInput, regions: AudienceRegions | null, geoOmitted: boolean, eventsOmitted: boolean, cohortsOmitted: boolean, p: BriefPacket): string {
  const L: string[] = [];
  // I3-1243: a withheld audience is a WITHHELD sentence, never "0 collectors".
  L.push(intel.collectors_total === null ? withheldSentence("Audience size") : `Audience size: ${fig(p, intel.collectors_total)} collectors.`);
  if (intel.suppressed) {
    // The suppression travels IN as an explicit absence. A model handed a bare
    // `null` narrates around it and invents a shape for the missing data; told
    // plainly that a floor withheld it, it leaves it out (I13-AI: the service appends the note).
    L.push("Behavioural breakdowns are WITHHELD: this audience is below the reporting floor, so repeat/dormant/depth/completion figures are not available. Do not mention them in your reply at all, and do not estimate them.");
  } else {
    // I3-0150: "One-time" and "New in 90 days" are FIXED TEMPLATE PROSE, not
    // served cells — a model explaining the sentence naturally echoes its own
    // wording back, and `90`/the word `one` in "One-time" were never admitted
    // anywhere, so a faithful quote of the brief's own label was refused. Fix:
    // register the exact fixed PHRASE as a KEY (restatable as itself, exactly
    // as `:441`'s day sentence already is), never the bare numeral as a
    // FIGURE — a figure would let the digit be reused to invent an unrelated
    // fan count (I3-0151's exact hazard; `test/s003-reply-check.test.ts`'s
    // "About 90 fans came back" pin must keep failing).
    p.keys.add("One-time");
    p.keys.add("New in 90 days");
    L.push(`Repeat collectors: ${fig(p, intel.repeat ?? "null")}. One-time: ${fig(p, intel.one_time ?? "null")}. New in 90 days: ${fig(p, intel.new_90d ?? "null")}. Dormant: ${fig(p, intel.dormant ?? "null")}.`);
    if (intel.depth) L.push(`Collection depth: ${intel.depth.map((d) => `${key(p, d.bucket)}→${fig(p, d.collectors)}`).join(", ")}.`);
    // I3-0150: "own all but one" spells the word "one" in fixed prose.
    if (intel.completion_eligible) {
      p.keys.add("own all but one");
      L.push(`Completion: ${fig(p, intel.completed ?? "null")} own everything, ${fig(p, intel.near_complete ?? "null")} own all but one.`);
    }
  }
  // I3-0150: "in the last 90 days" is fixed prose; register the exact phrase.
  p.keys.add("in the last 90 days");
  if (intel.acquisitions_total !== null && intel.acquisitions_90d !== null) {
    L.push(`Catalog: ${fig(p, intel.releases)} releases, ${fig(p, intel.acquisitions_total)} acquisitions total, ${fig(p, intel.acquisitions_90d)} in the last 90 days.`);
  } else {
    // I3-1243: a withheld acquisition cell is a WITHHELD sentence, never 0.
    L.push(`Catalog: ${fig(p, intel.releases)} releases.`);
    L.push(numberOrWithheld(p, "Acquisitions total", intel.acquisitions_total));
    L.push(numberOrWithheld(p, "Acquisitions in the last 90 days", intel.acquisitions_90d));
  }
  // The suppression travels IN as an explicit absence, exactly as the audience
  // floor does above: a model handed a bare null narrates around it and invents
  // a number, told plainly that a floor withheld it, it leaves it out (I13-AI).
  L.push(
    intel.archive_requests === null
      // NO QUANTITY, AND NO PRESENCE OR ABSENCE (Jason, 2026-09-05). The older
      // wording said the number "is below the reporting floor", which is itself
      // a statement about how many fans asked — an upper bound is a disclosure,
      // and one a model will happily turn into prose ("only a handful of fans").
      // This sentence names no number, no floor and no threshold, does not say
      // whether anybody asked, and tells the model not to supply either.
      ? "Back-catalog demand: WITHHELD — no back-catalog request figures are available for this creator. Do not mention back-catalog demand in your reply at all — not a number of requests, a bound, a comparison, or whether anyone has or has not asked."
      : `Back-catalog requests: ${fig(p, intel.archive_requests)}.`,
  );

  // ---- ADR-0006 §7: the day summary. One deterministic paragraph per block.
  if (intel.day !== null) {
    L.push(`The figures below are for the sealed UTC day ${key(p, intel.day)} and do not move.`);
    L.push(numberOrWithheld(p, "Supporters, as a running total of people who saved this creator", intel.supporters));
    L.push(numberOrWithheld(p, `Active fans on ${key(p, intel.day)}`, intel.active));

    // MONEY IS THE CREATOR'S OWN AND IS NEVER FLOORED (INV-033, O-10). The two
    // series are labelled and must stay labelled: `scheduled` is revisable by a
    // later reversal and `paid` is the payout ledger's, so calling either the
    // other is the one money error this seam can make (ADR-0006 §3, A5/O-11).
    const rev = intel.revenue;
    if (rev === null) {
      L.push(withheldSentence(MONEY_LINE_LABEL), MONEY_DAY_RULE);
    } else {
      L.push(
        [
          numberOrWithheld(p, moneyLabel(0), rev.gross_credits),
          numberOrWithheld(p, moneyLabel(1), rev.creator_net_cents),
          numberOrWithheld(p, moneyLabel(2), rev.paid_cents),
          numberOrWithheld(p, moneyLabel(3), rev.scheduled_clear_cents),
        ].join(" "),
      );
      if ([rev.gross_credits, rev.creator_net_cents, rev.paid_cents, rev.scheduled_clear_cents].some((v) => v === null)) L.push(MONEY_DAY_RULE);
    }

    if (intel.events.length === 0) {
      // R-07 fix: DISTINCT FROM THE EMPTY CASE. Reusing one sentence for
      // "this creator had no events" and "events existed and were dropped
      // for length" is the geo defect this file already fixed once (`:504`
      // `geoOmitted`) left standing for events and cohorts: a reader cannot
      // tell "nothing happened" from "the packet didn't have room," and a
      // WITHHELD-shaped silence in the second case reads as a fact about the
      // creator's audience that it is not.
      L.push(
        eventsOmitted
          ? "Events: OMITTED FROM THIS PACKET FOR LENGTH — event figures for this day were not included here. Do not name an event, do not say how many there were, do not estimate any of it, and do not read this as the creator having no events."
          : "Events: no event figures could be shown for this day. Do not name an event, do not say how many there were, and do not estimate any of it.",
      );
    } else {
      for (const e of intel.events) {
        // I3-0150, completed: this block's fixed label is registered as an
        // exact-phrase KEY exactly like its eight siblings above and below.
        // It was the ONE label the original fix missed, and the miss failed
        // CLOSED, not open: `restatesPacket` strips registered keys and then
        // checks leftover digits against served figures, so the label's own
        // "30" had to match a served figure and did not — a reply that
        // faithfully quoted this phrase was rejected `integer_not_in_packet`
        // and 502'd a paid creator-facing feature. Registered as a KEY and
        // never as a bare FIGURE, which would let the digit be reused for an
        // invented quantity (I3-0151). Found by the W3-06 second-pass
        // regression attacker, reproduced by probe before this line was added.
        p.keys.add("of those still acquiring 30 days later");
        L.push(
          `Event ${e.kind} ${key(p, e.id)} — ` +
            [
              numberOrWithheld(p, "participants", e.participants),
              numberOrWithheld(p, "fans new to this creator", e.new_fans),
              numberOrWithheld(p, "of those still acquiring 30 days later", e.acquired_d30),
            ].join(" "),
        );
      }
    }

    if (intel.cohorts_by_source.length === 0) {
      // R-07 fix: same distinction as events, above.
      L.push(
        cohortsOmitted
          ? "Cohorts: OMITTED FROM THIS PACKET FOR LENGTH — cohort figures for this day were not included here. Do not name a source, do not say how many there were, do not estimate any of it, and do not read this as the creator having no cohorts."
          : "Cohorts: no cohort figures could be shown for this day. Do not name a source, do not say how many there were, and do not estimate any of it.",
      );
    } else {
      for (const c of intel.cohorts_by_source) {
        // THE RATIO IS SERVED ONLY WHEN BOTH OPERANDS ARE. `size` withheld or
        // `d30` withheld means the sentence says so; it never means a rate
        // computed against a number the creator was not served.
        const r = servedRatio(c.d30, c.size);
        // I3-0150: "still there after 1 day" / "after 7 days" / "after 30
        // days" / "30-day retention rate" are this row's fixed labels — each
        // registered as an exact-phrase KEY (never a bare FIGURE, which
        // would let the digit be reused for an invented per-fan quantity;
        // I3-0151), tied to THIS row's own print, never to a rung that
        // dropped the row.
        p.keys.add("still there after 1 day");
        p.keys.add("after 7 days");
        p.keys.add("after 30 days");
        p.keys.add("30-day retention rate");
        L.push(
          `Cohort ${key(p, c.cohort_day)} from ${key(p, c.source)} — ` +
            [
              numberOrWithheld(p, "joined", c.size),
              numberOrWithheld(p, "still there after 1 day", c.d1),
              numberOrWithheld(p, "after 7 days", c.d7),
              numberOrWithheld(p, "after 30 days", c.d30),
              r === null
                ? "30-day retention rate: WITHHELD — one of its two operands is withheld, so there is no rate. Do not mention this rate in your reply at all, and do not compute or estimate one."
                : `30-day retention rate: ${fig(p, (r * 100).toFixed(1))} percent.`,
            ].join(" "),
        );
      }
    }
  }

  if (regions) {
    // B8's four-branch contract (`regions.ts geoServedSet`, ADR-0006 §6 geo
    // row). The unit is DISTINCT FANS for ONE SEALED UTC DAY — never fan-days,
    // never a month, never residence.
    const unit = `Regions for ${key(p, String(regions.day))} (DISTINCT FANS — one person, one sealed UTC day, counted once however many times they acted; a country is where somebody acted from, and never where anyone lives)`;
    if (regions.active_fans === null) {
      // NO NUMBER, NO FLOOR, NO THRESHOLD, NO PRESENCE OR ABSENCE — the
      // `archive_requests` wording above is the precedent and the reason: an
      // upper bound is a disclosure.
      L.push(`${unit}: WITHHELD — no country figures and no total are available for this day. Do not mention regions for this day in your reply at all — not a number of fans, a country, a bound, or whether anyone acted anywhere.`);
    } else if (regions.countries.length === 0) {
      // The breakdown was NOT COMPUTED (below `GEO_DAY_MIN_ACTIVE`), so the
      // sentence must not imply that a country was seen and withheld.
      L.push(`${unit}: ${fig(p, regions.active_fans)} distinct fans in total. There is no country breakdown for this day — it does not exist. Do not name a country, do not say any country was or was not seen, and do not estimate a split.`);
    } else {
      const shown = regions.countries.map((b) => `${key(p, b.country)}=${fig(p, b.fans)}`).join(", ");
      // THE TOTAL IS PRINTED ONLY WHEN THE ROWS ACCOUNT FOR IT, and that is
      // CHECKED rather than asserted: printing a total beside rows that do not
      // sum to it hands back the withheld residual by subtraction, which is the
      // v1 defect `regions.ts` (3) records.
      const accounted = regions.countries.reduce((s, b) => s + b.fans, 0) + (regions.other ?? 0);
      if (accounted !== regions.active_fans) {
        L.push(`${unit}: ${shown}. No total is available. Do not add these up, do not state or estimate a total, and do not compute percentages.`);
      } else if (regions.other === null) {
        L.push(`${unit}: ${shown}. Total ${fig(p, regions.active_fans)} distinct fans, and every country is shown, so the countries add up to the total exactly. Do not compute percentages against anything not shown here.`);
      } else {
        L.push(`${unit}: ${shown}, Other=${fig(p, regions.other ?? "null")}. Other is a POOL of countries too small to name and is not itself a country — do not name it as one. Total ${fig(p, regions.active_fans)} distinct fans, and the countries shown plus Other add up to the total exactly. Do not compute percentages against anything not shown here.`);
      }
    }
  } else if (geoOmitted) {
    // DISTINCT FROM THE WITHHELD SENTENCE ABOVE. That one says no figures are
    // available; this one says figures exist and were not sent, which is a fact
    // about the PACKET and not about the creator's audience. Neither states,
    // bounds or implies a number.
    L.push(
      "Regions: OMITTED FROM THIS PACKET FOR LENGTH — region figures for this day were not included here. Do not state, estimate, bound or imply any number of fans or any country, do not say whether anyone acted anywhere, and do not read this as the creator having no regions.",
    );
  }

  return L.join("\n");
}

/** R1: WORKERS AI ONLY. The model name is a `@cf/` id, which is the only thing
 * an `InferenceBinding` can resolve — there is no host, no URL and no `fetch`
 * anywhere in this module, and `test/s002-summary.test.ts` scans for all three. */
export const AI_MODEL = "@cf/meta/llama-3.1-8b-instruct";
const MAX_TOKENS = 300;

/** Longest a reply may be before it is refused rather than truncated: a brief
 * cut off mid-sentence reads as the platform having more to say. */
const MAX_REPLY = 1200;

export interface Explanation {
  brief: string;
  /** The model's reply, then — when the brief withheld anything — the service's own fixed notes, one per withheld
   * measure (`withheldNotesOf`, I13-AI #899). The notes carry no figure and are not the model's words. */
  explanation: string;
  model: string;
  /** GENERATED MARKING, EXPLICIT (INV-052 OUTPUT HALF, docket #17 E-2). The
   * `model` field is an id a reader has to recognise before it means anything;
   * this is the claim itself, in the served object, beside it. `true` whenever a
   * model reply is in `explanation`; `false` only on the money-withheld day's fixed
   * template (I26-Q1, `moneyDayTemplate`), where no model was asked. The appended
   * withheld notes are fixed service text (`explanation` docblock). */
  generated: boolean;
}

/**

 * single-hasher invariant, enforced at run time by `scripts/invariant-setup.mjs`)
 * — an import this module cannot take without widening the two-type-import
 * graph that is its whole structural claim. So the key and the store live in
 * `intelligence-routes.ts`, which already imports the platform, and this module
 * stays pure and deterministic: same aggregates, same bytes, no state.
 */
export interface ExplainOptions {
  /** The `summary` kind of the SAME day object the intelligence came from. */
  readonly summary?: SummaryInput | null;
}

/**
 * Explain a creator's own aggregates.
 *
 * 503 WHEN UNSET, and never a fallback. An operator who has not configured an
 * inference binding has not configured this feature, and a canned paragraph
 * would be indistinguishable to a creator from a real reading of their numbers.
 */
export async function explainIntelligence(
  ai: InferenceBinding | undefined,
  intel: CreatorIntelligence,
  regions: AudienceRegions | null,
  opts?: ExplainOptions,
): Promise<Explanation> {
  // LEAD-creator-ai-422 (I16-W02H): a binding with no `run` passes the route's presence check and its debit; no inference runs, so the slot is refunded.
  if (!ai || typeof ai.run !== "function") throw new CreatorAiError("this deployment has no inference binding configured", 503, AI_NO_ANSWER_MARKER);
  // The check's allowlist is built BY THE RENDERER, in the same call that built
  // the bytes: only the rung actually returned contributes (`buildBrief`).
  const packet: BriefPacket = { figures: new Set(), keys: new Set(), decimals: new Set() };
  const brief = buildBrief(projectForBrief(intel, opts?.summary ?? null), regions, packet);
  const measures = withheldMeasuresOf(brief);
  // I26-Q1 (D-1030, Jason (c) "a template summary"): on a day ANY money measure is withheld the model is not asked.
  if (moneyWithheldDay(measures)) return moneyDayTemplate(brief);
  let out: unknown;
  try {
    out = await ai.run(AI_MODEL, { messages: [{ role: "system", content: SYSTEM }, { role: "user", content: brief }], max_tokens: MAX_TOKENS });
  } catch {
    // The provider's own error text is not passed through: it is written for an
    // operator reading logs, and a creator does not need to learn which vendor
    // Dhwaja bought inference from in order to be told it did not work.
    throw new CreatorAiError("the explanation service did not answer", 502, AI_NO_ANSWER_MARKER);
  }
  const text = textOf(out);
  if (typeof text !== "string" || text.trim().length === 0) throw new CreatorAiError("the explanation service did not answer", 502, AI_NO_ANSWER_MARKER);
  if (text.length > MAX_REPLY) throw new CreatorAiError("the explanation service did not answer", 502, AI_NO_ANSWER_MARKER);
  // THE OUTPUT HALF (INV-052, docket #17 E-2 ruling), the mirror of the input
  // half `buildBrief` builds: the model READS the packet's numbers back and
  // introduces none of its own. The check is `restatesPacket` (`:667`) over the
  // `BriefPacket` (`:300`) the renderer filled, and it lives here rather than in
  // the route because the packet is one side of the comparison and this frame
  // is the only one holding it.
  const withheld = withheldLabelsOf(brief);
  // I13-FX3 (899c) + I13-FX6: every channel reads every NORMALISED reading of the reply (`replyReadings`); the served text is the model's own.
  const ids = briefIdsOf(brief);
  const attribution = servedAttributionOf(brief, packet, withheld, sharedNameWords(measures));
  const context = sentenceAttributionOf(brief, packet);
  // #26a (docket #26 (b)): the numeral phrases the model was handed are fixed prose, read off what was actually sent.
  const prose = numeralPhrasesOf(`${SYSTEM}\n${brief}`);
  const offenceOf = (t: string): PacketOffence | null => {
    for (const reading of replyReadings(t, ids)) {
      const o = restatesPacket(reading, packet, withheld, attribution, measures, ids, prose);
      if (o !== null) return o;
    }
    // I13-FX12 (D-0894(1), JNFX-N2): the pipe-fold readings face ONLY the note-form and withheld-name checks — a pipe
    // read as a letter fuses `Collectors|90`, so the number channels never see those readings.
    return replyReadings(t, ids, true).some((r) => namesOrImitates(r, measures, ids, attribution)) ? "reference" : null;
  };
  let offence = offenceOf(text);
  // I26-FX3 (D-1017): AFTER every whole-reply check, on a money-withheld day only the GROUNDED clauses are the reply
  // (`groundedReply`); none left is the refusal, and a cut reply faces every whole-reply check again as served.
  // I28-AI (D-0535(b) non-money half): the same cut on a day where only a NON-money measure is withheld (`withheldDay`).
  const reply = offence === null && withheldDay(measures) ? groundedReply(text, ids, packet, attribution, prose, moneyWithheldDay(measures), servedAttributionOf(brief, packet, [])) : text;
  if (offence === null && reply !== text) offence = reply.trim().length === 0 ? "reference" : offenceOf(reply);
  // I62-AI (I45-21): on every day, AFTER the cut, so a clause the cut drops keeps the cut's reason (`mislabelledFigure`).
  if (offence === null && replyReadings(reply, ids).some((r) => mislabelledFigure(r, packet, attribution, context, prose) !== null)) offence = "integer";
  if (offence !== null) {
    // THE ALLOWLIST IS THE FIGURES, NOT THE PROSE (E-2). A withheld cell
    // contributes NOTHING: `withheldSentence` carries no digit and the LABEL it
    // hangs on (`Active fans on 2026-09-30`) is a key, not a value, so neither
    // the label nor the sealed day can hand the model a number to state the
    // cell with. Fixed label text (`in the last 90 days`, `after 7 days`) is
    // not in the set at all. Bounding a withheld figure and inventing a fan
    // count are the same refusal, and now by the packet's own accounting of
    // what it served rather than by a property of the string.
    rejectedReply(offence);
    throw new CreatorAiError("the explanation service did not answer", 502, AI_REPLY_REJECTED_MARKER);
  }
  // I28-AI (D-1024(2), I26-FX3 open question): a CUT reply that is served is recorded like a refusal, with its own reason.
  if (reply !== text) cutReply();
  // The BRIEF is returned beside the explanation on purpose: a creator can see
  // exactly what was sent, which is the only way "aggregates only" is a claim
  // they can check rather than one they have to take.
  // I13-AI (#899, D-0888(2)): the service, never the model, says what was withheld — one fixed note per measure.
  const notes = withheldNotesOf(brief);
  return { brief, explanation: notes.length === 0 ? reply.trim() : `${reply.trim()}\n\n${notes.join(" ")}`, model: AI_MODEL, generated: true };
}

/** I26-Q1: the `model` id of the money-withheld day's fixed template — no model wrote it. */
export const MONEY_DAY_TEMPLATE_MODEL = "none: fixed template, a money measure is withheld";
/** I26-Q1 (D-1030, Jason (c)): the money-withheld day's reply, built by the service alone — the brief's own sentences that
 * state a SERVED NON-money figure (no WITHHELD / OMITTED cell, no instruction to the model, no money cell label), then the fixed
 * withheld notes (`withheldNotesOf`). Every word is service text the renderer already produced. `MONEY_LINE_LABEL` is not in
 * the list: `renderBrief` prints it only as a WITHHELD sentence. Neither part is ever empty on a money-withheld day (the
 * events and cohorts blocks always print a sentence; a withheld money cell always has a note). */
function moneyDayTemplate(brief: string): Explanation {
  const money = MONEY_LABELS.map((m) => m.label);
  const served = brief
    .split("\n")
    .flatMap((line) => line.split(/(?<=\.)\s+(?=[A-Z])/))
    .filter((s) => !/WITHHELD|OMITTED|\bdo not\b/i.test(s) && !money.some((l) => s.includes(l)));
  return { brief, explanation: `${served.join(" ")}\n\n${withheldNotesOf(brief).join(" ")}`, model: MONEY_DAY_TEMPLATE_MODEL, generated: false };
}

/**
 * EVERY QUANTITY THE STRING STATES, as integers — the digit runs first, then the
 * ones it SPELLS (`wordIntegersOf` `:664`, INV-052 output half, docket #23 E-3).
 * Leading zeros are dropped, so the `04` of a packet date and a `4` in prose are
 * ONE integer, and `12.5` contributes `12` and `5` on both sides of the
 * comparison. Comparing the raw runs instead would refuse a correct reply for
 * writing a packet month without its zero. Digit runs come FIRST because
 * `restatesPacket` `:764` reads `[0]` of a token that is digits by construction.
 */
function integersOf(s: string, prose: ReadonlySet<string> = NO_PROSE): string[] {
  return [...(s.match(/\d+/g) ?? []).map((d) => d.replace(/^0+(?=\d)/, "")), ...spelledQuantitiesOf(s, prose)];
}

const WORD_ONES = new Map<string, number>([
  ["zero", 0], ["one", 1], ["two", 2], ["three", 3], ["four", 4], ["five", 5], ["six", 6], ["seven", 7], ["eight", 8], ["nine", 9],
  ["ten", 10], ["eleven", 11], ["twelve", 12], ["thirteen", 13], ["fourteen", 14], ["fifteen", 15], ["sixteen", 16], ["seventeen", 17], ["eighteen", 18], ["nineteen", 19],
]);
const WORD_TENS = new Map<string, number>([["twenty", 20], ["thirty", 30], ["forty", 40], ["fifty", 50], ["sixty", 60], ["seventy", 70], ["eighty", 80], ["ninety", 90]]);
const WORD_SCALE = new Map<string, number>([["thousand", 1000], ["million", 1000000]]);
/** #26c (I16-W02H): ordinals read as their cardinal and END the run (`twenty-first` is 21). `first` is absent on purpose:
 * position 1 asserts no population beyond the thing named, so it states no quantity (REPORT, BRIEF-DELTA). */
const ORD_ONES = new Map<string, number>([
  ["second", 2], ["third", 3], ["fourth", 4], ["fifth", 5], ["sixth", 6], ["seventh", 7], ["eighth", 8], ["ninth", 9], ["tenth", 10],
  ["eleventh", 11], ["twelfth", 12], ["thirteenth", 13], ["fourteenth", 14], ["fifteenth", 15], ["sixteenth", 16], ["seventeenth", 17], ["eighteenth", 18], ["nineteenth", 19],
]);
const ORD_TENS = new Map<string, number>([["twentieth", 20], ["thirtieth", 30], ["fortieth", 40], ["fiftieth", 50], ["sixtieth", 60], ["seventieth", 70], ["eightieth", 80], ["ninetieth", 90]]);
/** #26d (I16-W02H): in-group multipliers (`hundred`, `dozen`) and their ordinal/plural forms; a PLURAL multiplies a count
 * before it (`three hundreds` is 300) and is no number when bare (`hundreds of fans` states none). */
const WORD_GROUP = new Map<string, number>([["hundred", 100], ["hundredth", 100], ["hundreds", 100], ["dozen", 12], ["dozens", 12]]);
const SCALE_FORMS = new Map<string, number>([["thousandth", 1000], ["thousands", 1000], ["millionth", 1000000], ["millions", 1000000]]);
const ORDINAL_FORMS = new Set(["hundredth", "thousandth", "millionth"]);
const PLURAL_FORMS = new Set(["hundreds", "dozens", "thousands", "millions"]);

/** One spelled numeral: its value, where it sits in the LOWERCASED string, and its word when it is one word long. */
interface SpelledNumeral {
  value: number;
  start: number;
  end: number;
  lone: string | null;
}

/**
 * A SPELLED QUANTITY IS A QUANTITY (E-3): the integers an English numeral in
 * `s` states, for `integersOf` `:634` to hand to the same packet comparison a
 * digit run faces. Three properties, each one a direction a looser reading gets
 * wrong:
 *
 * 1. THE ALPHABET IS WHOLE WORDS. The string is cut into `[a-z]+` tokens and a
 *    token is compared entire, so `often`, `someone`, `money` and `nothing`
 *    carry no number — a `\b`-anchored search over the raw text would have to
 *    get that right once per word instead of once.
 * 2. A RUN IS ONE NUMERAL, NOT A SUM (the E-4 rule that a separator may never
 *    join two unrelated numbers, in words). A word that cannot continue the
 *    numeral in hand ENDS it and starts the next: `twenty thirty` is 20 and 30,
 *    `thirty-two nine` is 32 and 9. A run also ends at anything but a single
 *    hyphen or space, so a sentence boundary can never fuse two figures. The one
 *    joiner is `and` directly after a scale word and before a ones/tens word
 *    (docket #27): `three hundred and forty` is 340, `twenty and thirty` is two.
 * 3. `a`/`an` ARE NOT IN THE ALPHABET, and they do not have to be: `hundred`
 *    with an empty accumulator is 100, which is what makes `a hundred` 100
 *    without letting an article count as a quantity anywhere else.
 * This is the PARSER: every numeral, in or out of context. Which of them a reply
 * ASSERTS is `spelledQuantitiesOf` (docket #26 (b)).
 */
export function wordIntegersOf(s: string): string[] {
  return spelledNumeralsOf(s.toLowerCase()).map((n) => String(n.value));
}

function spelledNumeralsOf(lower: string): SpelledNumeral[] {
  const out: SpelledNumeral[] = [];
  let total = 0;
  let cur = 0;
  let ones = false;
  let tens = false;
  let hundreds = false;
  let scale = Infinity;
  let open = false;
  let afterScale = false;
  let start = 0;
  let end = 0;
  let words = 0;
  let first = "";
  const flush = (): void => {
    if (open) out.push({ value: total + cur, start, end, lone: words === 1 ? first : null });
    total = cur = words = 0;
    ones = tens = hundreds = open = afterScale = false;
    scale = Infinity;
  };
  const take = (at: number, w: string): void => {
    if (words === 0) {
      start = at;
      first = w;
    }
    words++;
    end = at + w.length;
    open = true;
  };
  const toks: Array<{ w: string; at: number; joined: boolean }> = [];
  let prev = 0;
  for (const m of lower.matchAll(/[a-z]+/g)) {
    toks.push({ w: m[0], at: m.index, joined: /^[- ]?$/.test(lower.slice(prev, m.index)) });
    prev = m.index + m[0].length;
  }
  for (let k = 0; k < toks.length; k++) {
    const { w, at, joined } = toks[k] as { w: string; at: number; joined: boolean };
    if (!joined) flush();
    // `first` alone states no quantity (ORD_ONES); closing a run (`twenty-first`, `one hundred first`) it is its 1.
    const one = WORD_ONES.get(w) ?? ORD_ONES.get(w) ?? (w === "first" && open && !ones ? 1 : undefined);
    const ten = WORD_TENS.get(w) ?? ORD_TENS.get(w);
    const group = WORD_GROUP.get(w);
    const mult = WORD_SCALE.get(w) ?? SCALE_FORMS.get(w);
    const ordinal = ORD_ONES.has(w) || ORD_TENS.has(w) || ORDINAL_FORMS.has(w) || w === "first";
    if (w === "and") {
      const next = toks[k + 1];
      const joins = open && afterScale && next?.joined && (WORD_ONES.has(next.w) || WORD_TENS.has(next.w) || ORD_ONES.has(next.w) || ORD_TENS.has(next.w));
      if (!joins) flush();
      continue;
    }
    if (one !== undefined) {
      if (ones || (tens && one >= 10)) flush();
      cur += one;
      take(at, w);
      ones = true;
      if (one >= 10) tens = true;
      afterScale = false;
    } else if (ten !== undefined) {
      if (ones || tens) flush();
      cur += ten;
      take(at, w);
      tens = true;
      afterScale = false;
    } else if (group !== undefined) {
      if (PLURAL_FORMS.has(w) && cur === 0) {
        flush();
        continue;
      }
      if (hundreds) flush();
      cur = cur === 0 ? group : cur * group;
      take(at, w);
      hundreds = afterScale = true;
      ones = tens = false;
    } else if (mult !== undefined) {
      if (PLURAL_FORMS.has(w) && cur === 0 && total === 0) {
        flush();
        continue;
      }
      if (mult >= scale) flush();
      total += (cur === 0 ? 1 : cur) * mult;
      cur = 0;
      take(at, w);
      ones = tens = hundreds = false;
      scale = mult;
      afterScale = true;
    } else {
      flush();
      continue;
    }
    if (ordinal) flush();
  }
  flush();
  return out;
}

const NO_PROSE: ReadonlySet<string> = new Set();
/** Before a lone `one`, the words that make it a pronoun (`no one`, `each one`), not a count. (`all but one` is a
 * FIGURE since D-0922(6) — see `spelledQuantitiesOf`.) */
const ONE_PRONOUN_BEFORE = new Set(["no", "each", "every", "any", "the", "this", "that", "which", "another", "such"]);
/** Before a lone `one`, the words that make it a figure (`exactly one`, `gained one`): a figure label in prose. */
const ONE_LABEL_BEFORE = new Set(["exactly", "only", "just", "reached", "gained", "lost", "counted", "total", "totals", "totaled", "totalled"]);
/** After a lone `one`, the words that show it modifies no noun (`one of the best`, `one or`, `one is`). */
const NOT_A_NOUN = new Set([
  "of", "and", "or", "nor", "but", "the", "a", "an", "is", "are", "was", "were", "be", "been", "by", "to", "in", "on", "at", "for", "with", "from", "than", "as", "if",
  "so", "that", "this", "it", "its", "they", "them", "their", "there", "then", "we", "you", "your", "he", "she", "who", "which", "what", "when", "where", "while",
  "another", "else", "too", "also", "now", "still", "again", "here", "can", "could", "will", "would", "should", "may", "might", "must", "do", "does", "did", "has", "have", "had", "not",
]);
const phraseKey = (lower: string, n: SpelledNumeral, sep: string, next: string): string => `${lower.slice(n.start, n.end).replace(/\s+/g, " ")}${sep === "-" ? "-" : " "}${next}`;

/** FIXED PROSE (docket #26 (b): "`one` in fixed prose is exempt"): every spelled numeral the text prints with the word
 * after it. The brief prints every served figure in DIGITS (`fig`), so a spelled numeral in the brief or the SYSTEM prompt
 * is template prose by construction; `explainIntelligence` derives the set from exactly what the model was sent. */
function numeralPhrasesOf(text: string): Set<string> {
  const lower = text.toLowerCase();
  const out = new Set<string>();
  for (const n of spelledNumeralsOf(lower)) {
    const next = /^(\s+|-)([a-z]+)/.exec(lower.slice(n.end));
    if (next !== null) out.add(phraseKey(lower, n, next[1] as string, next[2] as string));
  }
  return out;
}

/**
 * WHICH SPELLED NUMERALS THE REPLY ASSERTS AS QUANTITIES (docket #26 (b), Jason 2026-09-07; QUEUE #26 (1), RUN-0013 D-1):
 * a numeral is a QUANTITY only when it modifies a countable noun or follows a figure label, and fixed prose is exempt.
 * Built as: (1) a numeral phrase the model was handed (`prose`) is exempt; (2) every numeral other than a lone `one`
 * stays a quantity — fail closed, the reading no reply of this seam has been shown to need relaxed (REPORT OPEN QUESTIONS);
 * (3) a lone `one` is a quantity after `:`/`=` or a label word, before a noun, or as half of a range whose other half is
 * one (`one or two fans`), and NOT as a pronoun (`no one`, `one of`), a compound modifier (`one-time`) or at a clause end.
 */
export function spelledQuantitiesOf(s: string, prose: ReadonlySet<string> = NO_PROSE): string[] {
  const lower = s.toLowerCase();
  const nums = spelledNumeralsOf(lower);
  const counts = (k: number): boolean => {
    const n = nums[k] as SpelledNumeral;
    // D-0922(6): `all but one` restates a quantity — a figure whatever follows (`of`) and even in handed prose.
    if (n.lone === "one" && /\bbut\s+$/.test(lower.slice(0, n.start))) return true;
    const after = lower.slice(n.end);
    const next = /^(\s+|-)([a-z]+)/.exec(after);
    if (next !== null && prose.has(phraseKey(lower, n, next[1] as string, next[2] as string))) return false;
    if (n.lone !== "one") return true;
    const before = lower.slice(0, n.start);
    if (/[:=]\s*$/.test(before)) return true;
    const prev = /([a-z]+)\s+$/.exec(before)?.[1];
    if (prev !== undefined && ONE_LABEL_BEFORE.has(prev)) return true;
    const following = nums[k + 1];
    if (following !== undefined && /^\s+(?:or|to|and)\s+$/.test(lower.slice(n.end, following.start))) return counts(k + 1);
    if (prev !== undefined && ONE_PRONOUN_BEFORE.has(prev)) return false;
    if (next === null || next[1] === "-") return false;
    return !NOT_A_NOUN.has(next[2] as string);
  };
  return nums.flatMap((n, k) => (counts(k) ? [String(n.value)] : []));
}

/** A packet KEY as the model may write it back: the same token, with each of
 * its digit runs allowed to lose or keep leading zeros, so the packet's
 * `2026-09-04` also matches a reply's `2026-9-4`. Anchored on both sides so a
 * key is matched WHOLE and never as a piece of a longer number. */
function keyPattern(k: string): RegExp {
  const body = k.replace(/[.*+?^${}()|[\]\\-]/g, "\\$&").replace(/\d+/g, (d) => `0*${d.replace(/^0+(?=\d)/, "")}`);
  return new RegExp(`(?<![0-9A-Za-z])${body}(?![0-9A-Za-z])`, "g");
}

/**
 * DOES THE REPLY ONLY RESTATE THE PACKET? (INV-052 OUTPUT HALF.)
 *
 * Four steps, and each one is a direction the raw `\d+` comparison got wrong:
 *
 * 1. KEYS COME OUT (E-2). A key may be written back AS ITSELF — the
 *    creator's own day sentence is the thing a model is most likely to quote —
 *    but its digits are not a quantity, so once the key is removed a leftover
 *    `7` from `2026-09-07` has to stand on its own against `figures` and fails.
 * 2. A THOUSANDS SEPARATOR IS NOT A BOUNDARY (E-4), and it is removed BEFORE
 *    the keys are. `1,234` is the packet's
 *    `1234`: same digits, same order, one integer written for a human. Only a
 *    comma between a digit and a group of exactly three is removed, so a list
 *    (`12, 345`) is untouched and no two unrelated numbers are ever joined.
 * 3. TWO DECIMAL PLACES MAY BE A CENTS CELL WRITTEN IN DOLLARS (E-4). The money
 *    labels say "in cents" (`renderBrief`), so `$50.00` for a served `5000` is
 *    a faithful restatement and refusing it counts an invention that did not
 *    happen. Accepted only for EXACTLY two decimals, and only as an ALTERNATIVE
 *    to the existing run-by-run rule, so `12.5` and `12.9` keep the behaviour
 *    `test/s003-reply-check.test.ts` pins. The equivalence is one-way safe: the
 *    point-free form is admitted only when the packet actually served that
 *    integer, so it can introduce no quantity the packet did not carry.
 * 4. A QUANTITY SPELLED IN WORDS IS STILL A QUANTITY (E-3, docket #23). The
 *    check is "no invented quantity", not "no invented digit": `wordIntegersOf`
 *    `:664` normalises the numerals and they face `figures` on the same terms,
 *    so a spelled restatement of a served figure is served and a spelled
 *    invention is refused. The step is LAST because the digit tokens carry
 *    their own alternatives (2, 3) and must be decided before they are blanked.
 */
/** WHICH CHANNEL REFUSED. `integer` is a digit run the reply wrote; `quantity`
 * is a numeral the reply SPELLED (step 4). They are different facts about a
 * reply and the record below has been calling both the first one. */
type PacketOffence = "integer" | "quantity" | "reference";

const INVISIBLE_RE = /[\p{Cf}\p{Default_Ignorable_Code_Point}]/gu;
/**
 * I13-FX6 (D-0890(2)): THE READINGS THE CHECK REFUSES ON. An invisible character is read both as nothing
 * (`Act\u3164ive` is `Active`) and as a space (`Active\u3164fans` is `Active fans`, not `Activefans`); a date
 * written in words reads as the brief's own ISO id (`foldDates`). Every channel runs on every reading; the
 * served text is the model's own.
 */
function replyReadings(text: string, ids: readonly string[], pipes = false): string[] {
  const folded = [...text.normalize(NFKD_FORM)].map((c) => CONFUSABLE.get(c) ?? c).join("").replace(/\p{M}/gu, "");
  // I13-FX12 (JNFX2-S2): invisible characters go first, so the stroke and I readings below see `withhe\u200bId` as `withheId`.
  const visible = [...new Set([folded.replace(INVISIBLE_RE, ""), folded.replace(INVISIBLE_RE, " ")])];
  // I13-FX11 (JNFX-N2): a pipe touching a letter is an upright stroke too (`withhe|d`); a pipe between spaced words is left alone.
  // I13-FX12 (D-0894(1)): only in the `pipes` readings, which the caller hands to the note and withheld-name checks alone.
  const piped = pipes ? visible.map((v) => v.replace(/(?<=\p{L})\||\|(?=\p{L})/gu, STROKE)) : visible;
  const strokes = piped.flatMap((v) => [v.replace(STROKE_RE, "l"), v.replace(STROKE_RE, "I")]);
  // I13-FX9 (D-0891(6), J-07): a capital I directly after a lowercase letter may be the l it is drawn as (`withheId`, `avaiIable`);
  // I13-FX11 (JNFX-N1): it may also be the i it stands for (`ActIve`), so BOTH readings are kept — the fold is an extra reading, never a replacement.
  const readings = [...new Set(strokes.flatMap((f) => [f.replace(/(?<=\p{Ll})I/gu, "l"), f]))];
  return readings.map((r) => foldDates(r, ids));
}
/** I13-FX12: the private-use placeholder an upright-stroke letter folds to; `replyReadings` reads it as l and as I. */
const STROKE = "\ue000";
const STROKE_RE = /\ue000/gu;
const MONTHS = ["jan", "feb", "mar", "apr", "may", "jun", "jul", "aug", "sep", "oct", "nov", "dec"];
const MONTH_RE = "(January|February|March|April|May|June|July|August|September|October|November|December|Jan|Feb|Mar|Apr|Jun|Jul|Aug|Sept|Sep|Oct|Nov|Dec)\\.?";
const ORDINAL_WORDS = ["first", "second", "third", "fourth", "fifth", "sixth", "seventh", "eighth", "ninth", "tenth", "eleventh", "twelfth", "thirteenth", "fourteenth", "fifteenth", "sixteenth", "seventeenth", "eighteenth", "nineteenth", "twentieth"];
const dayOf = (d: string): number => {
  const w = d.toLowerCase().replace(/[\s-]+/g, " ");
  if (/^\d/.test(w)) return Number.parseInt(w, 10);
  const i = ORDINAL_WORDS.indexOf(w);
  if (i >= 0) return i + 1;
  if (w === "thirtieth") return 30;
  const m = /^(twenty|thirty) (\S+)$/.exec(w);
  const j = m === null ? -1 : ORDINAL_WORDS.indexOf(m[2] as string);
  return m === null || j < 0 || j > 8 ? Number.NaN : (m[1] === "twenty" ? 20 : 30) + j + 1;
};
const DAY_RE = "(\\d{1,2}(?:st|nd|rd|th)?|(?:twenty|thirty)[\\s-]+(?:first|second|third|fourth|fifth|sixth|seventh|eighth|ninth)|thirtieth|twentieth|[a-z]+?(?:st|nd|rd|th))";
const DATE_WORD_RES = [
  new RegExp(`\\b${MONTH_RE}\\s+(?:the\\s+)?${DAY_RE}(?:,?\\s+((?:19|20)\\d{2}))?\\b`, "gi"),
  new RegExp(`\\b(?:the\\s+)?${DAY_RE}\\s+(?:of\\s+)?${MONTH_RE}(?:,?\\s+((?:19|20)\\d{2}))?\\b`, "gi"),
];
/**
 * I13-FX6 (D-0890(2)): a date written in words (`September 30, 2026`, `the 30th of September`, `Sept. thirtieth`)
 * is rewritten as the ISO id the brief printed — and ONLY when it resolves to one (with no year, to the one id with
 * that month and day), so a count beside a month is never eaten as a year. `May` counts only capitalised.
 */
function foldDates(s: string, ids: readonly string[]): string {
  const dates = ids.filter((id) => DATE_ID_RE.test(id));
  if (dates.length === 0) return s;
  const resolve = (whole: string, month: string, day: string, year: string | undefined): string => {
    if (month.toLowerCase().startsWith("may") && !month.startsWith("May")) return whole;
    const mm = MONTHS.indexOf(month.slice(0, 3).toLowerCase()) + 1;
    const dd = dayOf(day);
    if (mm < 1 || !(dd >= 1 && dd <= 31)) return whole;
    const tail = `-${String(mm).padStart(2, "0")}-${String(dd).padStart(2, "0")}`;
    const hits = dates.filter((d) => (year === undefined ? d.endsWith(tail) : d === `${year}${tail}`));
    return hits.length === 1 ? (hits[0] as string) : whole;
  };
  return s
    .replace(DATE_WORD_RES[0] as RegExp, (whole, month: string, day: string, year?: string) => resolve(whole, month, day, year))
    .replace(DATE_WORD_RES[1] as RegExp, (whole, day: string, month: string, year?: string) => resolve(whole, month, day, year));
}
/** I13-FX3 (D-0889(6), 899d): the shape of the service's own note (`withheldNotesOf`); a reply that writes it is refused. */
const NOTE_FORM_RE = /(?::|\s[—–-])\s*withheld\b|\bno figure (?:is )?available\b/i;
/** I12-FX9: a window ("after 7 days", "30-day") is a WORD of its label, so `after 7 days` (served) and
 * `after 30 days` (withheld) are different measures; digits are spelled as letters so the split keeps them. */
const windowWord = (_: string, n: string): string => ` window${[...n].map((d) => "abcdefghij"[Number(d)]).join("")} `;
const labelWords = (s: string): string[] =>
  s
    .toLowerCase()
    .replace(/\d{4}-\d{2}-\d{2}/g, " ")
    .replace(/(?<![\d.,])(\d+)(?=[ -]?days?\b)[ -]?days?\b/g, windowWord)
    .split(/[^a-z]+/)
    .filter((w) => w.length > 0 && !LABEL_STOPWORDS.has(w))
    .map((w) => w.replace(/s$/, ""));
/** A brief SENTENCE: the unit a label and the figure printed under it share (`Repeat collectors: 30. One-time: 60.` is two). */
const briefSentences = (brief: string): string[] => brief.split("\n").flatMap((line) => line.split(/(?<=\.)\s+/));
export function withheldLabelsOf(brief: string): string[][] {
  const out: string[][] = [];

  for (const seg of briefSentences(brief)) {

/** I12-FX9 (D-0884(7), I12S-0151-b): served figure -> the content words of the brief LABEL that printed it, minus
 * every word of every withheld label. A sentence carrying WITHHELD contributes nothing.
 * I3-0151 a (D-0981(7), D-0535(a)): the label is the figure's OWN, not its sentence's — the words from the end of the
 * previous figure's unit (or the sentence start) up to it, plus its unit up to the next comma or semicolon; a
 * figure with no words of its own takes the sentence head (the text before the first colon). So
 * `Catalog: 3 releases, 300 acquisitions total` binds 300 to `acquisitions` only, and a reply restating 300 beside
 * `releases` (or any word that is not its label's) names no measure the brief printed 300 under. */
export type ServedAttribution = ReadonlyMap<string, ReadonlySet<string>>;
export function servedAttributionOf(brief: string, packet: BriefPacket, withheld: readonly string[][], shared: ReadonlySet<string> = new Set()): ServedAttribution {
  // I13-AI: a word of a withheld name that ALSO labels a served cell (another row's `after 30 days`) stays that cell's word.
  const banned = new Set(withheld.flat().filter((w) => !shared.has(w)));
  const out = new Map<string, Set<string>>();
  const add = (n: string, words: readonly string[]): void => {
    const ws = out.get(n) ?? new Set<string>();
    for (const w of words) ws.add(w);
    out.set(n, ws);
  };
  const own = (t: string): string[] => labelWords(t).filter((w) => !banned.has(w));
  for (const seg of briefSentences(brief)) {
    if (/\bWITHHELD\b/.test(seg)) continue;
    const text = seg.replace(/(\d),(?=\d{3}(?!\d))/g, "$1");
    if (own(text).length === 0) continue;
    // Keys, dates and fixed window labels are not figures: masked to letters of the same length, so positions hold.
    let masked = text;
    for (const k of packet.keys) masked = masked.replace(keyPattern(k), (m) => "x".repeat(m.length));
    masked = masked.replace(/\d{4}-\d{1,2}-\d{1,2}/g, (m) => "x".repeat(m.length)).replace(WINDOW_LABEL_DIGITS, (m) => "x".repeat(m.length));
    const colon = text.indexOf(":");
    const head = own(colon > 0 ? text.slice(0, colon) : text);
    let from = 0;
    for (const m of masked.matchAll(/\d+(?:\.\d+)?/g)) {
      const at = m.index ?? 0;
      const after = at + m[0].length;
      const stop = masked.slice(after).search(/[,;]/);
      const unitEnd = stop < 0 ? masked.length : after + stop;
      const words = own(`${text.slice(from, at)} ${text.slice(after, unitEnd)}`);
      const label = words.length > 0 ? words : head;
      add(m[0].includes(".") ? m[0] : (integersOf(m[0])[0] as string), label);
      from = Math.max(after, unitEnd);
    }
    // Spelled quantities the brief's own prose states keep the sentence as their label (as before I22).
    for (const n of integersOf(masked.replace(/\d+(?:\.\d+)?/g, " "))) add(n, own(text));
  }
  return out;
}

const CLAUSE_SPLIT = /,(?=\s)|;|\s[-—–]\s|\b(?:but|however|while|whereas|although|though|yet|and|so)\b/i;

/**
 * I12-FX12 (D-0886(1)) — THE NAMES OF A WITHHELD MEASURE. A reply names one by its whole label, by the label's
 * HEAD (the text before its first comma or dash: `creator net` of `Creator net this day, in cents`), by its

const namesOfLabel = (label: string): { words: string[]; generic: boolean }[] => {
  const full = labelWords(label);
  if (full.length === 0) return [];
  const raw = [full];
  const head = labelWords(label.split(/,|\s[-—–]\s|\(/)[0] as string);
  if (head.length > 0 && head.length < full.length) raw.push(head);
  for (const w of full) if (w.startsWith("window") && full.length > 1) raw.push([w]);
  // I13-FX3 (D-0889(6)): a name REQUIRES only its distinctive words (899b: `active` of `Active fans`); a name with none

  const out: { words: string[]; generic: boolean }[] = [];

};

/**
 * I13-AI (#899, D-0888(2), D-0887) — THE STRUCTURAL RULE. The model is told never to mention a withheld measure
 * (`SYSTEM`, `withheldSentence`), the service appends its own note for each (`withheldNotesOf`), and a reply that
 * NAMES one — whole label, head, window alone, or a `LABEL_PARAPHRASE` word — is refused whatever it says about it
 * (`namesWithheldMeasure`): there is no declared form to get right, so no phrasing list to outrun.
 * A NAME SHARED WITH A SERVED CELL (cohort A withholds `after 30 days`, cohort B serves it) is not by itself a
 * reference to the withheld one (I12-JN-FX2 (i)): it names the withheld measure only beside an identifier of A's
 * row that no served carrier of that name has (`ctx`), in the same or the previous sentence.
 */
const WITHHELD_LABEL_RE = /^[^A-Za-z0-9]*(.+?)(?::| are| is) WITHHELD\b/;
const ROW_COHORT_RE = /^(Cohort (\S+) from (\S+)) — /;
const ROW_EVENT_RE = /^(Event .+? (\S+)) — /;
interface BriefSeg { text: string; label: string | null; row: string; ids: string[] }
const SEALED_DAY_RE = /sealed UTC day (\d{4}-\d{2}-\d{2})\b/;
const DATE_ID_RE = /^\d{4}-\d{2}-\d{2}$/;
function briefSegs(brief: string): BriefSeg[] {
  const out: BriefSeg[] = [];
  // I13-FX3 (899b): a row-less segment below the sealed-day line is identified by that day.
  let day: string | null = null;
  for (const line of brief.split("\n")) {
    const sealed = SEALED_DAY_RE.exec(line);
    const c = ROW_COHORT_RE.exec(line);
    const e = c === null ? ROW_EVENT_RE.exec(line) : null;
    const row = c !== null ? (c[1] as string) : e !== null ? (e[1] as string) : "";
    const ids = c !== null ? [c[2] as string, c[3] as string] : e !== null ? [e[2] as string] : day !== null && sealed === null ? [day] : [];
    if (sealed !== null) day = sealed[1] as string;
    for (const text of line.split(/(?<=\.)\s+/)) {
      const m = WITHHELD_LABEL_RE.exec(text);
      let label = m === null ? null : (m[1] as string);
      if (label !== null && row !== "" && label.startsWith(`${row} — `)) label = label.slice(row.length + 3);
      out.push({ text, label, row, ids });
    }
  }
  return out;
}

/** The index of the family a brief label (or a served segment's label part) belongs to; -1 when it has none. */
const synonymFamilyOf = (label: string): number => WITHHELD_SYNONYMS.findIndex((f) => f.label.test(label.trim()));
const FAMILY_WORDS: readonly (readonly string[])[] = WITHHELD_SYNONYMS.map((f) => [...new Set(f.words.flatMap((w) => labelWords(w)))]);
/** A served segment's own label: the text before its first `: `, after its row prefix (`Cohort … — `). */
const servedLabelOf = (text: string): string => (text.replace(/^.*? — /, "").split(/:\s/)[0] as string).trim();
export function withheldMeasuresOf(brief: string): WithheldMeasure[] {
  const segs = briefSegs(brief);
  const served = segs
    .filter((s) => s.label === null && !/\bWITHHELD\b/.test(s.text) && /\d/.test(s.text.replace(/\d{4}-\d{2}-\d{2}/g, " ")))
    .map((s) => ({ words: new Set(labelWords(s.text)), ids: s.ids, text: s.text, row: s.row, family: synonymFamilyOf(servedLabelOf(s.text)) }));
  const out: WithheldMeasure[] = [];
  for (const s of segs) {
    if (s.label === null) continue;
    const family = synonymFamilyOf(s.label);
    const synonyms = family < 0 ? [] : (FAMILY_WORDS[family] as readonly string[]).map((w) => ({ words: [w], generic: false, synonym: true }));
    const names = [...namesOfLabel(s.label).map((n) => ({ ...n, synonym: false })), ...synonyms].map(({ words, generic, synonym }): WithheldName => {
      const carriers = served.filter((v) => words.every((w) => v.words.has(w)) || (synonym && v.family === family && v.row !== s.row));
      const ctx = s.ids.filter((id) => !carriers.some((v) => v.ids.includes(id)));
      return generic ? { words, shared: true, ctx, generic: true } : { words, shared: carriers.length > 0, ctx };
    });
    if (names.length > 0) out.push({ label: s.label, row: s.row, names });
  }
  // I13-FX6 (D-0890(2)): a generic name's served neighbours — every served segment carrying one of its identifiers
  // (the cohort rows of the sealed day), with the words of every non-shared withheld name, generic words, the row
  // glue and the row's own identifiers removed: what is left names a MEASURE (`joined`, `still there`).
  const banned = new Set([...out.flatMap((m) => m.names.filter((n) => !n.shared).flatMap((n) => n.words)), ...GENERIC_LABEL_WORDS, ...ROW_GLUE_WORDS]);
  for (const m of out)
    for (const n of m.names) {
      if (n.generic !== true) continue;
      const words = new Set<string>();
      const figures = new Map<string, Set<string>>();
      for (const v of served) {
        if (!v.ids.some((id) => n.ctx.includes(id))) continue;
        // a row's own identifiers (`referral`) say which row, not which measure: `60 cents from referral` attributes nothing
        const own = new Set(v.ids.flatMap((id) => labelWords(id)));
        const ws = [...v.words].filter((w) => !banned.has(w) && !own.has(w));
        for (const w of ws) words.add(w);
        for (const f of figuresOf(v.text, v.ids)) figures.set(f, new Set([...(figures.get(f) ?? []), ...ws]));
      }
      if (words.size > 0) n.servedBy = { words, figures };
    }
  return out;
}
/** I13-FX6: words of a row's own prefix (`Cohort <day> from <source> — `) that name no measure. */
const ROW_GLUE_WORDS = ["cohort", "from", "event"];
/** I13-FX6: the quantities a text states once identifiers, dates and fixed window labels are out (`servedAttributionOf`'s cut). */
function figuresOf(text: string, ids: readonly string[]): string[] {
  let rest = text.replace(/(\d),(?=\d{3}(?!\d))/g, "$1");
  for (const id of ids) rest = rest.replace(keyPattern(id), " ");
  rest = rest.replace(/\d{4}-\d{1,2}-\d{1,2}/g, " ").replace(WINDOW_LABEL_DIGITS, " ");
  return [...(rest.match(/\d+\.\d+/g) ?? []), ...integersOf(rest.replace(/\d+\.\d+/g, " "))];
}
/**
 * I13-FX6 (D-0890(2)): a GENERIC word (fans, paid, cents) beside a withheld measure's identifier is NOT a reference
 * to it when every clause carrying that word attributes itself to a measure the brief SERVED under that identifier:
 * each figure in the clause was printed under a word the clause names (`On 2026-09-30, 60 fans joined via
 * referral.`), and a figure-less clause names such a word. `90 fans among your collectors` attributes 90 to no
 * served cell of the day, so the day's withheld `Active fans` stays named.
 */
function attributedToServed(sentence: string, n: WithheldName, ids: readonly string[]): boolean {
  const by = n.servedBy;
  if (by === undefined) return false;
  const clauses = sentence.split(CLAUSE_SPLIT).filter((c) => n.words.every((w) => namedWords(c).has(w)));
  if (clauses.length === 0) return false;
  return clauses.every((c) => {
    const named = namedWords(c);
    const figs = figuresOf(c, ids);
    if (figs.length === 0) return [...named].some((w) => by.words.has(w));
    return figs.every((f) => [...(by.figures.get(f) ?? [])].some((w) => named.has(w)));
  });
}
/** The words of every withheld name that also labels a served cell (`servedAttributionOf`'s exemption). */
export const sharedNameWords = (measures: readonly WithheldMeasure[]): Set<string> =>
  new Set(measures.flatMap((m) => m.names.filter((n) => n.shared && n.generic !== true).flatMap((n) => n.words)));
/** The service's own sentence for each withheld measure, in brief order, row-qualified, deduplicated. */
export function withheldNotesOf(brief: string): string[] {
  const out: string[] = [];
  for (const s of briefSegs(brief)) {
    if (s.label === null) continue;
    const label = (s.label.split(/\s[—–]\s|\s\(/)[0] as string).trim();
    const note = `${label.charAt(0).toUpperCase()}${label.slice(1)}${s.row === "" ? "" : ` (${s.row})`}: withheld, no figure is available.`;
    if (!out.includes(note)) out.push(note);
  }
  return out;
}
const mentions = (s: string, id: string): boolean => new RegExp(keyPattern(id).source, "i").test(s);
function namesWithheldMeasure(text: string, measures: readonly WithheldMeasure[], ids: readonly string[] = []): boolean {
  const sentences = text.replace(/(\d),(?=\d{3}(?!\d))/g, "$1").split(/[.!?](?=\s|$)|\n/);
  // I13-FX3 (899a): the referent of a shared name is also the LAST row (and the last date) the reply named, however far back.
  const classes = [ids.filter((id) => DATE_ID_RE.test(id)), ids.filter((id) => !DATE_ID_RE.test(id))];
  const last: string[][] = classes.map(() => []);
  for (let k = 0; k < sentences.length; k++) {
    const sentence = sentences[k] as string;
    const all = namedWords(sentence);
    const near = `${sentences[k - 1] ?? ""} ${sentence}`;
    classes.forEach((cls, i) => {
      const hit = cls.filter((id) => mentions(sentence, id));
      if (hit.length > 0) last[i] = hit;
    });
    const inContext = (id: string): boolean => mentions(near, id) || last.some((l) => l.includes(id));
    for (const m of measures)
      for (const n of m.names) {
        if (!n.words.every((w) => all.has(w))) continue;
        if (n.generic === true && attributedToServed(sentence, n, ids)) continue;
        if (!n.shared || n.ctx.some(inContext)) return true;
      }
  }
  return false;
}
/**
 * I12-FX9 (D-0884(7), I12S-0151-b) — ATTRIBUTION, NOT PHRASING. Once the brief withheld anything,
 * every quantity in the reply must sit in a clause that names a word of a SERVED, NON-withheld
 * measure the brief printed THAT figure under (`servedAttributionOf`). A figure bound to a withheld
 * label by any phrasing, any negation, the next sentence, or a paraphrase names no such measure.
 */
function unattributedFigure(text: string, packet: BriefPacket, attribution: ServedAttribution, prose: ReadonlySet<string> = NO_PROSE): PacketOffence | null {
  for (const sentence of text.replace(/(\d),(?=\d{3}(?!\d))/g, "$1").split(/[.!?](?=\s|$)|\n/)) {
    for (const clause of sentence.split(CLAUSE_SPLIT)) {
      const named = new Set(labelWords(clause));
      const ok = (n: string | undefined): boolean => {
        const ws = n === undefined ? undefined : attribution.get(n);
        return ws !== undefined && [...ws].some((w) => named.has(w));
      };
      let rest = clause;
      for (const k of packet.keys) rest = rest.replace(keyPattern(k), " ");
      rest = rest.replace(WINDOW_LABEL_DIGITS, " ");
      for (const token of rest.match(/\d+(?:\.\d+)?/g) ?? []) {
        const [whole, frac = ""] = token.split(".");
        if (frac.length === 2 && ok(integersOf(`${whole}${frac}`)[0])) continue;
        if (frac.length > 0 ? ok(token) : ok(integersOf(token)[0])) continue;
        return "integer";
      }
      for (const n of integersOf(rest.replace(/\d+(?:\.\d+)?/g, " "), prose)) if (!ok(n)) return "quantity";
    }
  }
  return null;
}

/**
 * I62-AI (I45-21, D-1523 W0-2) — A FIGURE UNDER ANOTHER FIGURE'S LABEL, ON EVERY DAY. In a reply clause a figure (digits,
 * or a spelled quantity `spelledQuantitiesOf` counts) has a UNIT, the word right after it (`60 cents`), and a HEAD: for a
 * colon before it, the text from the unit of the last figure before that colon up to the colon, shared by every figure
 * after it (`Creator net this day, in cents: 90`, `collectors|repeat: 90|30`); with no colon, the text from the previous
 * figure's unit up to it (`You were paid 60`). A figure its unit names is not read against a head. Only a MEASURE word is
 * read (`measureWordOf`: a `GROUNDING_LABELS` word or synonym, a money key, or the money unit `cent`) that the brief printed some served figure under
 * (`servedAttributionOf`). A figure's own words are its label's plus its brief sentence's (`sentenceAttributionOf`: the
 * brief prints `40 in the last 90 days` after `300 acquisitions`). Refused: a unit that is not the figure's own; a head
 * naming measures that none of the figures it heads owns; and (I62-FX-AI, I62-JN-WEB-AI I62-AI-1) a head naming measures
 * over a figure that owns no word of it and has none of its own words beside it (`in cents: 500 then 90`). THE RUN: a figure
 * with no colon before it in its clause, none of its own words beside it and no measure in its own head is read against
 * the words of the latest head earlier in its sentence that named a measure (a colon's, or one a figure owns), across the
 * clause splits `CLAUSE_SPLIT` makes (`Paid out to you this day: 400, then 90`). I63-AI: a measure UNIT after that head
 * starts a run of its own (`You were paid 400 cents and 90`); a line's last run carries into the next line's first sentence
 * (`400\n- 90`); in a run a unit starts or a line carries, a figure owning a word of its own tail is not read against it.
 * A WINDOW word (`in the last 90 days`) in a figure's head that the brief printed some served figure under must be among
 * the figure's own label words (`attribution`). A figure with no such head before it in its sentence (or line) is not read here.
 */
const MONEY_MEASURE_WORDS: ReadonlySet<string> = new Set([...MONEY_LABELS.flatMap((m) => m.key.flatMap((k) => labelWords(k))), "cent"]);
/** I63-AI: a window label word (`windowWord`: `90 days` -> `windowja`). */
const WINDOW_WORD_RE = /^window[a-j]+$/;
/** I65-AI (R4-PAIR, PLAN ruling 3): the digits-plus-unit span `labelWords` turns into a window word (`90 days`). */
const WINDOW_SPAN_RE = /(?<![\d.,])\d+[ -]?days?\b/gi;
/** I65-AI (W1): `CLAUSE_SPLIT` read at one position — the split that ends a clause. */
const CLAUSE_SEP = new RegExp(CLAUSE_SPLIT.source, "iy");
/** I65-AI (RG05): a list item line (`- 300 acquisitions`); I65-FX-AI (ruling 8 c): also `+`, an en or em dash, `1.` and `1)`;
 * a star only before a space, so a bold line (`**In the last 90 days**`) is not an item. */
const BULLET_LINE_RE = /^\s*(?:[-+\u2022\u2013\u2014]|\*(?=\s)|\d+[.)])/;
/** I65-FX-AI (ruling 8 b): words that qualify a window span (`90 days or so`), masked so they are not words said after it. */
const QUALIFIED_SPAN_RE = new RegExp(`(${WINDOW_SPAN_RE.source})((?:[\\s,]*(?:or so|alone|only|or more|ago))+)`, "gi");
/** I65-FX-AI (ruling 8 a, D-1575): an all-time or present-state marker in a figure's own clause frees it from the window. */
const WINDOW_FREE_RE = /\b(?:overall|in total|all-time|so far|now|currently)\b/i;
/** I65-FX-AI (ruling 8 d): a line that ends mid-sentence (its last character, emphasis and space aside, is not . ! ? or :). */
const OPEN_LINE_RE = /[^\s*_.!?:][\s*_]*$/;
/** I65-FX-AI (ruling 8 c): a line ending in a colon, emphasis aside (`**In the last 90 days:**`). */
const COLON_LINE_RE = /:[\s*_]*$/;
const measureWordOf = (w: string): string | undefined => GROUNDING_SYNONYM.get(w) ?? (MONEY_MEASURE_WORDS.has(w) ? w : undefined);
function sentenceAttributionOf(brief: string, packet: BriefPacket): ServedAttribution {
  const out = new Map<string, Set<string>>();
  for (const seg of briefSentences(brief)) {
    const text = seg.replace(/(\d),(?=\d{3}(?!\d))/g, "$1");
    // Every brief label that carries a digit (a window, a day) is a key (`key`), so the keys are the whole mask; a WITHHELD
    // sentence prints no figure, and a reply naming a withheld measure is refused before this is read (`namesOrImitates`).
    let masked = text;
    for (const k of packet.keys) masked = masked.replace(keyPattern(k), " ");
    const words = labelWords(text);
    for (const m of masked.match(/\d+(?:\.\d+)?/g) ?? []) {
      const n = m.includes(".") ? m : (integersOf(m)[0] as string);
      out.set(n, new Set([...(out.get(n) ?? []), ...words]));
    }
  }
  return out;
}
function mislabelledFigure(text: string, packet: BriefPacket, attribution: ServedAttribution, context: ServedAttribution, prose: ReadonlySet<string>): PacketOffence | null {
  const labels = new Set([...attribution.values()].flatMap((ws) => [...ws]));
  const measures = (words: readonly string[]): string[] => words.map(measureWordOf).filter((w): w is string => w !== undefined && labels.has(w));
  // Every word of a head the brief printed some served figure under, measure or not (`collector` in `repeat|collectors`).
  const saidOf = (words: readonly string[]): string[] => words.map((w) => measureWordOf(w) ?? w).filter((w) => labels.has(w));
  const ownOf = (n: string | undefined): string[] => (n === undefined ? [] : [...(attribution.get(n) ?? []), ...(context.get(n) ?? [])]);
  const winOf = (n: string | undefined): string[] => (n === undefined ? [] : [...(attribution.get(n) ?? [])]);
  // I63-AI (JA-01): read per LINE; the run a line ends on is carried into the next line's first sentence (`400\n- 90`).
  let carried: string[] = [];
  // I64-AI (A3): a colon's run (`in cents: 500`) is also carried over a sentence split inside a line (`500. 90`).
  let held: string[] = [];
  let colonRun = false;
  // I65-AI (RG05): the window a line ending in a colon left pending (`In the last 90 days:`) binds each list item line after it.
  let listed: string[] = [];
  // I65-FX-AI (ruling 8 d): the line before ended mid-sentence (`In the last 90 days, you had`), so a line that goes on with no
  // capital (`300 acquisitions.`) is bound by the window it left pending, as a list item is.
  let open = false;
  for (const raw of text.replace(/(\d),(?=\d{3}(?!\d))/g, "$1").split("\n")) {
    let lineStart = true;
    const bullet = BULLET_LINE_RE.test(raw);
    const line = raw.replace(QUALIFIED_SPAN_RE, (_, span: string, words: string) => span + " ".repeat(words.length));
    // I64-AI (A3): the served window a figure-less clause names (`In the last 90 days,`) binds the next clause that has figures.
    let pending: string[] = bullet || (open && !/^[^\p{L}\d]*\p{Lu}/u.test(raw)) ? listed : [];
    // I64-FX-AI (ruling 4): a figure-less clause after the window that says something of its own (`things were quiet`) leaves
    // the window bound only to a figure that opens its clause (`300 acquisitions`), not to one with words before it (`You have 300`).
    // I65-AI (W1): 0 nothing said since the window; 1 said, followed so far only by commas (`, honestly, you had 300` stays
    // bound); 2 said and then a sentence end or another clause split (`. You have 300`, `; it was calm`); in an aside, a conjunction
    // only as `aside` below says (I65-FX-AI).
    let spoken = 0;
    // I65-FX-AI (ruling 8 d, D-1575): the words said came in an aside after a bare window clause (`In the last 90 days, slowly`);
    // there a conjunction frees the window only straight after said words (`quiet but you have`, not `, and you sold`), and only
    // for the clause it opens: a figure-less one followed by a comma is still the aside (`slowly but surely, you had 300`).
    let aside = false;
    let lapse = false;
    for (const sentence of line.split(/[.!?](?=\s|$)/)) {
      if (sentence.trim().length === 0) continue;
      // I62-FX-AI: THE RUN — the words of the latest head in this sentence that named a measure (a colon's, or a figure's own);
      // a bare figure after it in a later clause owns one of them. I63-AI (FX-AI-3): a measure unit (`400 cents`) starts one.
      // A run a unit starts or a line carries is LOOSE: a figure owning a word of its own tail (`30 of them repeat`) leaves it.
      let run: string[] = lineStart ? carried : held;
      // Read only once `run` is non-empty, which at a sentence's start is only a carried run (a head or a unit sets it after).
      let loose = true;
      lineStart = false;
      let cursor = 0;
      let sepBefore = "";
      for (const clause of sentence.split(CLAUSE_SPLIT)) {
        let rest = clause;
        for (const k of packet.keys) rest = rest.replace(keyPattern(k), (m) => " ".repeat(m.length));
        rest = rest.replace(WINDOW_LABEL_DIGITS, (m) => " ".repeat(m.length));
        const figures: { at: number; end: number; own: Set<string>; win: Set<string> }[] = [];
        for (const m of rest.matchAll(/\d+(?:\.\d+)?/g)) {
          const [whole, frac = ""] = m[0].split(".");
          const own = new Set([...ownOf(frac.length > 0 ? m[0] : integersOf(m[0])[0]), ...(frac.length === 2 ? ownOf(integersOf(`${whole}${frac}`)[0]) : [])]);
          const win = new Set([...winOf(frac.length > 0 ? m[0] : integersOf(m[0])[0]), ...(frac.length === 2 ? winOf(integersOf(`${whole}${frac}`)[0]) : [])]);
          figures.push({ at: m.index ?? 0, end: (m.index ?? 0) + m[0].length, own, win });
        }
        const counted = new Set(spelledQuantitiesOf(rest, prose));
        for (const n of spelledNumeralsOf(rest.toLowerCase())) if (counted.has(String(n.value))) figures.push({ at: n.start, end: n.end, own: new Set(ownOf(String(n.value))), win: new Set(winOf(String(n.value))) });
        figures.sort((a, b) => a.at - b.at);
        // I65-FX-AI (ruling 8 a): a marker in the figure's own clause (with the split before it: `so far`) is read first (D-1575).
        const free = WINDOW_FREE_RE.test(sepBefore + clause);
        const bound = free || (spoken === 2 && /\p{L}/u.test(clause.slice(0, figures[0]?.at ?? 0))) ? [] : pending;
        const heads = new Map<number, { head: string; members: { own: Set<string>; self: boolean; unit: boolean }[] }>();
        let runAt = -1;
        // froms[i]: where figure i's head may start — the end of the previous figure's unit (0 for the first).
        const froms: number[] = [0];
        for (const [i, f] of figures.entries()) {
          const unit = /^[\s-]*([A-Za-z]*)/.exec(clause.slice(f.end)) as RegExpExecArray;
          froms[i + 1] = f.end + unit[0].length;
          const unitWords = labelWords(unit[1] as string);
          const named = measures(unitWords);
          if (named.some((w) => !f.own.has(w))) return "integer";
          // I63-AI (FX-AI-4 b): a WINDOW in the figure's head that some served figure was printed under (`in the last 90 days:
          // 300`) is a word of its own label only (`attribution`; the brief sentence's `context` gives it to every figure there).
          const colon = clause.lastIndexOf(":", f.at);
          // The first figure after the colon fixes where the colon's head starts.
          const first = colon < 0 ? i : figures.findIndex((g) => g.at > colon);
          const windows = labelWords(clause.slice(froms[first] as number, colon < 0 ? f.at : colon)).filter((w) => WINDOW_WORD_RE.test(w) && labels.has(w));
          if ([...windows, ...bound].some((w) => !f.win.has(w))) return "integer";
          // A figure its own unit names (`30 returning collectors`) is labelled; only an unnamed one is read against its head.
          if (named.length > 0) {
            run = named;
            colonRun = false;
            loose = true;
            runAt = f.at;
            continue;
          }
          const ownHead = colon < 0 ? clause.slice(froms[i] as number, f.at) : "";
          // `90 collectors`, `collectors 90`: the figure's own words, naming no measure, beside it; still read against its head (below).
          const unitSelf = unitWords.some((w) => f.own.has(w));
          const self = unitSelf || labelWords(ownHead).some((w) => f.own.has(w));
          const headed = measures(labelWords(ownHead)).length > 0;
          const tail = labelWords(clause.slice(froms[i + 1] as number, figures[i + 1]?.at ?? clause.length));
          if (colon < 0 && !self && !headed && run.length > 0 && !(loose && tail.some((w) => f.own.has(w)))) {
            if (!run.some((w) => f.own.has(w))) return "integer";
            // I63-AI: a window the run's head named binds every figure of the run (`in the last 90 days: 40, 300`).
            if (run.some((w) => WINDOW_WORD_RE.test(w) && !f.win.has(w))) return "integer";
            continue;
          }
          // A figure under a head of its own it does not own is refused below, so only an owned label starts or ends a run.
          if (self) {
            run = headed && !unitSelf ? saidOf(labelWords(ownHead)) : [];
            colonRun = false;
            loose = false;
            runAt = f.at;
          }
          // I64-FX-AI (ruling 3): a figure after a figure under a measure's head (`Repeat buyers among your 90 collectors number 30`, the
          // A2 widening below) is in the run that head starts in its plain form (`Repeat buyers number 30`), so `, then 60` is read
          // against it. Under a head of its own (`self`) it keeps its own run; a later colon's head still replaces it (`runAt`).
          const prev = self ? undefined : heads.get(-i);
          if (prev !== undefined && measures(labelWords(prev.head)).length > 0) run = saidOf(labelWords(prev.head));
          const key = colon < 0 ? -1 - i : colon;
          const group = heads.get(key) ?? { head: clause.slice(froms[first] as number, colon < 0 ? f.at : colon), members: [] };
          group.members.push({ own: f.own, self, unit: unitSelf });
          heads.set(key, group);
        }
        const opened = figures.length > 0 ? [] : labelWords(clause).filter((w) => WINDOW_WORD_RE.test(w) && labels.has(w));
        if (opened.length > 0) {
          // I65-AI (R4-PAIR): the end of the last window span is a clause boundary; words after it say something (`In the last
          // 90 days things were quiet`).
          const spans = [...clause.matchAll(WINDOW_SPAN_RE)];
          const last = spans[spans.length - 1] as RegExpMatchArray;
          spoken = /\p{L}/u.test(clause.slice((last.index ?? 0) + last[0].length)) ? 1 : 0;
          aside = false;
        } else if (spoken === 0 && /\p{L}/u.test(clause)) {
          spoken = 1;
          aside = true;
        }
        pending = figures.length > 0 ? [] : [...pending, ...opened];
        cursor += clause.length;
        CLAUSE_SEP.lastIndex = cursor;
        const sep = CLAUSE_SEP.exec(sentence)?.[0] ?? "";
        cursor += sep.length;
        sepBefore = sep;
        const conjunction = /^\p{L}/u.test(sep);
        // The clause a conjunction opened in an aside: a comma after it is still the aside, a further conjunction goes on.
        if (lapse) {
          if (sep.startsWith(",")) spoken = 1;
          lapse = conjunction;
        }
        if (spoken === 1 && !sep.startsWith(",")) {
          if (!(aside && conjunction)) spoken = 2;
          else if (/\p{L}/u.test(clause)) {
            spoken = 2;
            lapse = true;
          }
        }
        for (const [key, group] of heads) {
          const words = labelWords(group.head);
          const measured = measures(words);
          if (measured.length === 0) continue;
          // I64-AI (A2): a head whose every figure names itself by its unit (`Repeat buyers among your 90 collectors`) also heads
          // the clause's next figure when that one heads a colon-less group of its own (`number 30`, key - 1). A colon group's
          // key is the colon's index, and no group sits one below it, so `in cents: 90 collectors` keeps its one member.
          const members = group.members.every((m) => m.unit) ? [...group.members, ...(heads.get(key - 1)?.members ?? [])] : group.members;
          if (!members.some((m) => measured.some((w) => m.own.has(w)))) return "integer";
          // I62-FX-AI: and EVERY figure it heads owns a word of it, or names itself (`in cents: 500 then 90` is refused).
          const said = saidOf(words);
          if (members.some((m) => !m.self && !said.some((w) => m.own.has(w)))) return "integer";
          if (key > runAt) {
            run = said;
            // Only a colon's group sets the run here: a colon-less group's figures are `self`, so `runAt` is past its key.
            colonRun = true;
            loose = false;
            runAt = key;
          }
        }
      }
      carried = run;
      held = colonRun ? run : [];
    }
    listed = COLON_LINE_RE.test(line) ? pending : bullet || raw.trim().length === 0 ? listed : OPEN_LINE_RE.test(raw) ? pending : [];
    open = OPEN_LINE_RE.test(raw);
  }
  return null;
}

/**
 * I3-0151 (I22-FX-MONEY, D-0535(a)(b)) — A WITHHELD MONEY MEASURE BY ANY NAME. The label-word rule above catches
 * `creator net` but not `earnings`, `income`, `revenue` or `money`, so a served figure could be restated as the
 * withheld money (`Your 300 acquisitions equal your earnings.`) and a number-free claim about it went unchecked
 * (`Your revenue was strong.`). When the brief withholds ANY money measure, a reply that uses ANY money word is a
 * reference to it, whatever it says and whatever figure (or none) it carries.
 * I26-W1-CONTENT (I22 OPEN-ROWS: `pay`, `royalties`, `paid`, `tips` passed a whole-word list): STRUCTURAL on both sides.
 * The measure is money by IDENTITY with `MONEY_LABELS`; the reply is read per CLAUSE, by word FAMILY (a stem and every
 * suffix, a currency sign) — and it was still a word list: RG served `You made a killing`, `Collectors spent a lot on you`,
 * `Your take was healthy`, and a SERVED key cleared a clause (`net` cleared `Your net was paid to you in full` over a withheld
 * PAID).
 * I26-FX-AI (D-1015(1)): BY TOPIC. On a day where ANY money measure is withheld (a `MONEY_LABELS` cell, or the whole
 * `Money for this day`), a reply clause about money IN ANY FORM is refused: `MONEY_TOPIC_RE` (earning, making, taking,
 * spending, paying, income, cash, currency, worth, prices, amounts and their idioms) or `MONEY_AMOUNT_RE` (a figure with
 * k/bn/grand, `six figures`). NO served measure's label clears a clause. The one pass is the 899e purchase construction —
 * `paid for|bought|purchased <a served NON-money measure>`, no second person, no other money word in the clause. The brief
 * also tells the model not to discuss money on such a day (`MONEY_DAY_RULE`, `buildBrief`).
 * I26-FX3 (D-1017): the lexicon is now the SECOND check — I26-RJ-01 served 16 of 20 everyday idioms past it. The first is
 * GROUNDING (`groundedReply`, below): a clause that names no served non-money measure is dropped whatever its words; the
 * lexicon still refuses a grounded clause that carries a listed money word. RESIDUAL: below, at `saysOnlyServed`.
 */

  /[$€£¥₹¢₩₽₺₿]|\b(?:earn\w*|income\w*|revenue\w*|profit\w*|lucrativ\w*|proceeds|takings?|take-?home|took\s+home|(?:your|the|their|my|a|daily|day's|net|gross|total)\s+(?:take|cut)|killings?|(?:brought|bring|brings|bringing|rak(?:e|es|ed|ing))\s+in(?!\s+(?:\d|new\b|more\b|fresh\b|first\b|repeat\b))|hauls?|windfalls?|jackpots?|fortunes?|riches|wealth\w*|in\s+the\s+(?:black|red)|bank\w*|wallets?|purses?|pocket\w*|spen(?:d|ds|t|ding|der|ders)|splash\w*|splurg\w*|(?:shell|fork|cough)\w*\s+(?:out|up)|pay(?:outs?|ments?|days?|checks?|cheques?|roll\w*|out|able|ers?|ees?)|wages?|salar\w*|royalt\w*|tip(?:s|ped|ping|pers?|\s+jars?)?|gratuit\w*|compensat\w*|remunerat\w*|fees?|commissions?|stipends?|bonus\w*|refund\w*|generous\w*|handsome(?:ly)?|sales?|sell(?:s|ing|er|ers)?|sold|turnover|worth\w*|pric(?:e|es|ed|ey|ing|eless)|cost(?:s|ly|ing)?|expensive|cheap\w*|afford\w*|valuations?|budget\w*|invest\w*|returns?\s+on|roi|margins?|amounts?|sums?|gross\w*|net(?:s|ted|ting)?|credits?|financ\w*|fiscal|monetar\w*|economic\w*|money\w*|monies|monet\w*|cash\w*|funds?|funding|currenc\w*|dollars?|bucks?|(?<!per\s)cents?|penn(?:y|ies)|pence|dimes?|nickels?|quid|euros?|pounds?|sterling|yen|rupees?|pesos?|reais|dough|moolah|loot|coins?|usd|eur|gbp|brl|jpy|inr|cad|aud|mxn|chf|cny|usdc|usdt|btc|eth)\b/i;
const MONEY_AMOUNT_RE = /\b\d+(?:[.,]\d+)?\s?(?:k|mm|bn|grand)\b|\b(?:a|one|two|three|four|five|ten|several|a\s+few|a\s+couple(?:\s+of)?)\s+grand\b|\b(?:four|five|six|seven|eight)[- ]figures?\b/i;

const SECOND_PERSON_RE = /\byou(?:r|rs|rself)?\b/i;
/** The labels whose withholding makes the day a money-withheld day: the four money cells and the whole money line. */
const MONEY_DAY_LABELS: ReadonlySet<string> = new Set([...MONEY_LABELS.map((m) => m.label), MONEY_LINE_LABEL]);
const MONEY_LABEL_WORDS: ReadonlySet<string> = new Set(MONEY_LABELS.flatMap((m) => [...labelWords(m.label), ...m.key]));
const moneyWithheldDay = (measures: readonly WithheldMeasure[]): boolean => measures.some((m) => MONEY_DAY_LABELS.has(m.label));
/** I28-AI (D-0535(b)): any measure withheld, money or not — the days `groundedReply` cuts on. */
const withheldDay = (measures: readonly WithheldMeasure[]): boolean => measures.length > 0;
function claimsWithheldMoney(text: string, measures: readonly WithheldMeasure[], attribution: ServedAttribution): boolean {
  if (!moneyWithheldDay(measures)) return false;
  const servedWords = new Set([...attribution.values()].flatMap((ws) => [...ws]).filter((w) => !MONEY_LABEL_WORDS.has(w) && !MONEY_TOPIC_RE.test(w)));
  for (const sentence of text.split(/[.!?](?=\s|$)|\n/))
    for (const clause of sentence.split(CLAUSE_SPLIT)) {
      if (MONEY_TOPIC_RE.test(clause) || MONEY_AMOUNT_RE.test(clause)) return true;
      if (!PURCHASE_RE.test(clause)) continue;

      const object = at === null ? [] : labelWords(clause.slice(at.index + at[0].length));
      if (at === null || SECOND_PERSON_RE.test(clause) || !object.some((w) => servedWords.has(w))) return true;
    }
  return false;
}

/**
 * I26-FX3 (D-1017, QUEUE #955; I26-RJ-01 rejected the lexicon alone) — BY GROUNDING. On a money-withheld day
 * (`moneyWithheldDay`) the reply is cut to the clauses (`CLAUSE_SPLIT`, per sentence) that pass `groundedClause` on
 * EVERY reading (`replyReadings`): (a) the clause names a served NON-money measure — a `GROUNDING_LABELS` word or
 * synonym among that figure's own label words (`servedAttributionOf`), or the figure itself beside one of its non-money
 * label words; (b) every number it states, digit or spelled, keys and window labels masked, is the value of a measure
 * it is grounded in. The lexicon stays the SECOND check, on the whole reply (`claimsWithheldMoney`, run first). A failing clause is DROPPED
 * (`groundedReply`); a reply with nothing left is refused as `reference` (`explainIntelligence`). Nothing dropped =
 * the reply byte-identical. It runs AFTER the whole-reply checks (`restatesPacket`), so it only ever removes: every reply
 * those refuse is still refused whole, and a cut reply faces them again. Days with no money measure withheld never reach this.
 * A clause with no served measure in it (`You raked it in this day.`) is dropped whatever its words; since I28-FX-AI a
 * grounded clause must also SAY nothing else (`saysOnlyServed`), so `Your acquisitions made you a mint.` is dropped too.
 * I28-AI (D-0535(b), its NON-money half; I26-JN-05 H1): the same cut on a day where only a NON-money measure is withheld
 * (`withheldDay`), so a clause characterising it in any words (`Your backers multiplied.` over withheld supporters) is
 * dropped: its own words are banned from `servedAttributionOf`, so it names no served measure. With money served
 * (`money` false) a served money measure's key words (`MONEY_GROUNDING`) ground too and its label words are not stripped;
 * with money withheld nothing here changes.
 * I28-FX-AI (D-1025(1), RG-AI-01): a clause naming a served measure but characterising a withheld one (`Your audience of 90
 * brought in a flood of new backers.`) is dropped: `saysOnlyServed`. RESIDUAL (stated): its vocabulary is every SERVED label's,
 * not only the grounding measure's, plus `fan`/`people` — a claim built from those words alone is served.
 */
const MONEY_GROUNDING: ReadonlyMap<string, string> = new Map(MONEY_LABELS.flatMap((m) => m.key.map((k) => [k, k] as const)));
const GROUNDING_SYNONYM: ReadonlyMap<string, string> = new Map([...GROUNDING_LABELS].flatMap(([w, syn]) => [[w, w] as const, ...syn.map((s) => [s, w] as const)]));
/** Label words too weak to bind a stated figure (`7` of `people who saved`): never grounds with or without the value. */
const GROUNDING_WEAK: ReadonlySet<string> = new Set(["a", "an", "and", "or", "to", "be", "by", "who", "but", "never", "it"]);
const MONEY_DAY_WORDS: ReadonlySet<string> = new Set([...MONEY_LABEL_WORDS, ...[...MONEY_DAY_LABELS].flatMap((l) => labelWords(l))]);
/** A clause of only these (a date, `in the last 90 days`, `overall`) MODIFIES its sentence and claims nothing (`neutral`). */
const GROUNDING_NEUTRAL: ReadonlySet<string> = new Set(["last", "past", "since", "over", "during", "within", "at", "from", "overall", "far"]);
/** I28-FX-AI (D-1025(1), RG-AI-01): the closed word classes a grounded clause may carry besides the brief's own served label words
 * and its figures (`labelWords` form, so `was` is `wa`, `across` is `acros`): pronouns, articles, prepositions that set no bound,
 * the auxiliaries, the plain verbs that state a figure, and the count nouns of a people measure (`people`). No quantifier, comparative, modal, negation or particle (`up`, `out`):
 * those characterise. A word outside these classes is a claim the brief did not make, and drops its clause. */
const GROUNDING_GLUE: ReadonlySet<string> = new Set([
  "i", "me", "my", "we", "u", "our", "you", "it", "they", "them", "their", "he", "she", "him", "her", "hi", "these", "those", "there", "here", "who", "which", "what",
  "to", "at", "by", "with", "within", "from", "into", "via", "acros", "through", "per", "as", "after", "before", "during", "since", "until", "among", "between",
  "be", "been", "being", "am", "wa", "were", "ha", "have", "had", "kept", "do", "doe", "did", "now", "today", "currently", "later",
  "people", "person", "stand", "stood", "total", "totaled", "totalled", "number", "numbered", "count", "counted", "show", "shown", "showed",
]);
const NUMBER_WORDS: ReadonlySet<string> = new Set(["zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine", "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen", "seventeen", "eighteen", "nineteen", "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety", "hundred", "thousand", "million", "billion"]);
/** One spelling per inflection (`acquired` = `acquiring`, `joined` = `join`), so a served label word is not a claim in another tense. */
const wordStem = (w: string): string => w.replace(/(?:ing|ed|e)$/, "");
type Grounding = "grounded" | "neutral" | "ungrounded";
function groundedReading(c: string, packet: BriefPacket, attribution: ServedAttribution, prose: ReadonlySet<string>, money: boolean): Grounding {
  const named = new Set(labelWords(c));
  const groundWords = new Set([...named].map((w) => GROUNDING_SYNONYM.get(w) ?? (money ? undefined : MONEY_GROUNDING.get(w))).filter((w): w is string => w !== undefined));
  let rest = c.replace(/(\d),(?=\d{3}(?!\d))/g, "$1");
  for (const k of packet.keys) rest = rest.replace(keyPattern(k), " ");
  rest = rest.replace(WINDOW_LABEL_DIGITS, " ");
  const stated = [...(rest.match(/\d+(?:\.\d+)?/g) ?? []).map((t) => (t.includes(".") ? t : (integersOf(t)[0] as string))), ...integersOf(rest.replace(/\d+(?:\.\d+)?/g, " "), prose)];
  // I28-AI: with money served, a two-place decimal is also its cents (`$50.00` = the served 5000), as `unattributedFigure` reads it.
  const alts = (s: string): string[] => {
    const [w, f = ""] = s.split(".");
    return !money && f.length === 2 ? [s, integersOf(`${w}${f}`)[0] as string] : [s];
  };
  const grounds = new Set<string>();
  for (const [n, ws] of attribution) {
    const own = [...ws].filter((w) => !money || !MONEY_DAY_WORDS.has(w));
    if (own.some((w) => groundWords.has(w)) || (stated.some((s) => alts(s).includes(n)) && own.some((w) => !GROUNDING_WEAK.has(w) && named.has(w)))) grounds.add(n);
  }
  if (stated.length === 0 && [...named].every((w) => GROUNDING_NEUTRAL.has(w) || w.startsWith("window"))) return "neutral";
  return grounds.size > 0 && stated.every((s) => alts(s).some((a) => grounds.has(a))) ? "grounded" : "ungrounded";
}
/** I28-FX-AI (D-1025(1), RG-AI-01): a grounded clause says NOTHING ELSE — every word is the brief's own for a served figure
 * (`served`: label words with a withheld name's kept; money's dropped on a money-withheld day) in any inflection, a synonym of
 * one, a figure (a window is a served label's word), a 899e purchase verb, or glue. `Your audience was out in force.` names a served measure and characterises a
 * withheld one: its clause is dropped. */
function saysOnlyServed(c: string, money: boolean, served: ServedAttribution): boolean {
  const sayable = new Set([...served.values()].flatMap((ws) => [...ws]).filter((w) => !money || !MONEY_DAY_WORDS.has(w)).map(wordStem));
  const says = (w: string): boolean =>
    w.length <= 1 || sayable.has(wordStem(w)) || GROUNDING_SYNONYM.has(w) || GENERIC_LABEL_WORDS.has(w) || GROUNDING_GLUE.has(w) || GROUNDING_NEUTRAL.has(w) || NUMBER_WORDS.has(w);
  return labelWords(c.replace(new RegExp(PURCHASE_OBJECT_RE.source, "gi"), " ")).every(says);
}
/** The WEAKEST reading decides: one ungrounded reading fails the clause. */
function groundedClause(clause: string, ids: readonly string[], packet: BriefPacket, attribution: ServedAttribution, prose: ReadonlySet<string>, money: boolean, served: ServedAttribution): Grounding {
  const readings = replyReadings(clause, ids);
  const verdicts = readings.map((r) => groundedReading(r, packet, attribution, prose, money));
  if (verdicts.includes("ungrounded")) return "ungrounded";
  if (verdicts.includes("neutral")) return "neutral";
  // A word a confusable reading invents (`It` read as `lt`) is not a claim: one reading saying only served words suffices.
  return readings.some((r) => saysOnlyServed(r, money, served)) ? "grounded" : "ungrounded";
}
function groundedReply(text: string, ids: readonly string[], packet: BriefPacket, attribution: ServedAttribution, prose: ReadonlySet<string>, money: boolean, served: ServedAttribution): string {
  const kept: string[] = [];
  let dropped = false;
  // Dates in words are read as the brief's ids first (`foldDates`), so `On September 30, 2026` is one modifier, not three.
  for (const sentence of foldDates(text.trim(), ids).split(/(?<=[.!?])\s+|\n+/)) {
    const end = /[.!?]+$/.exec(sentence)?.[0] ?? "";
    const clauses = sentence.slice(0, sentence.length - end.length).split(CLAUSE_SPLIT).filter((c) => c.trim().length > 0);
    const verdicts = clauses.map((c) => groundedClause(c, ids, packet, attribution, prose, money, served));
    // A sentence stands on a GROUNDED clause: its neutral modifiers stay with it, and alone they are dropped with it.
    const ok = verdicts.includes("grounded") ? clauses.filter((_, i) => verdicts[i] !== "ungrounded") : [];
    if (ok.length === clauses.length) kept.push(sentence);
    else {
      dropped = true;
      if (ok.length > 0) kept.push(`${ok.map((c) => c.trim()).join(", ")}${end === "" ? "." : end}`);
    }
  }
  return dropped ? kept.join(" ") : text;
}

/**
 * I23 (I3-1966 a, D-0535(b), D-0993(1)) — A NUMBER-FREE QUALITATIVE CLAIM. A cause, a trend or a
 * comparison with other creators is a statement about a quantity, and with anything withheld the
 * model cannot know it does not rest on the withheld one (`Your growth is accelerating.` over a
 * withheld new-supporters cell). A closed vocabulary, as `MONEY_TOPIC_RE`: when the brief withholds
 * ANY measure, a reply using any of these words is refused. Trust claims are not in it yet (they need
 * a wording list — the census pick).
 */
export const QUALITATIVE_CLAIM_RE =
  /\b(?:because|caused by|causes?|causing|due to|thanks to|as a result|driven by|led to|leads? to|grow(?:s|ing|th)?|grew|declin(?:e|es|ed|ing)|accelerat(?:e|es|ed|ing)|slow(?:s|ed|ing)? down|slowing|rising|rose|falling|fell|dropp(?:ed|ing)|surg(?:e|ed|ing)|trend(?:s|ing)?|momentum|other creators?|average creator|typical creator|most creators|creators like you|your peers|peers?|compared (?:to|with) others)\b/i;
function claimsQualitative(text: string, measures: readonly WithheldMeasure[]): boolean {
  return measures.length > 0 && QUALITATIVE_CLAIM_RE.test(text);
}

/** I13-FX3 (899d): the service's note form is the service's alone; I13-AI: naming a withheld measure at all is the refusal. */
function namesOrImitates(text: string, measures: readonly WithheldMeasure[], ids: readonly string[], attribution: ServedAttribution = new Map()): boolean {
  return NOTE_FORM_RE.test(text) || namesWithheldMeasure(text, measures, ids) || claimsWithheldMoney(text, measures, attribution) || claimsQualitative(text, measures);
}

function restatesPacket(text: string, packet: BriefPacket, withheld: readonly string[][] = [], attribution: ServedAttribution = new Map(), measures: readonly WithheldMeasure[] = [], ids: readonly string[] = [], prose: ReadonlySet<string> = NO_PROSE): PacketOffence | null {
  if (namesOrImitates(text, measures, ids, attribution)) return "reference";
  if (withheld.length > 0) {
    const loose = unattributedFigure(text, packet, attribution, prose);
    if (loose !== null) return loose;
  }
  // Separators go BEFORE keys: `1,234` must be one number before anything is
  // matched against it, or a one-character key would split it again.
  let rest = text.replace(/(\d),(?=\d{3}(?!\d))/g, "$1");
  for (const k of packet.keys) rest = rest.replace(keyPattern(k), " ");
  // D-0376 (QUEUE #181): a fixed-window label's digits (90, 30, 7, 1) are
  // allowed IN THE LABEL POSITION ONLY: the digit run directly followed by
  // `day`/`days` (`90 days`, `30-day`, `past 7 days`). ACCEPTED HOLE, by that
  // ruling: one of those four written as an invented DURATION ("stayed 30
  // days") is not caught. `30 fans`, `7%` and a bare `90` are still refused;
  // a spelled window ("ninety days") is not allowed (it faces `figures`).
  rest = rest.replace(WINDOW_LABEL_DIGITS, " ");
  for (const token of rest.match(/\d+(?:\.\d+)?/g) ?? []) {
    const [whole, frac = ""] = token.split(".");
    if (frac.length === 2 && packet.figures.has(integersOf(`${whole}${frac}`)[0] as string)) continue;
    if (frac.length > 0) {
      // C-22: a decimal that is not the cents-written-as-dollars equivalence
      // above must match a decimal the packet actually PRINTED, exactly. It
      // is never accepted by checking its whole and fractional parts against
      // `figures` independently — that treats "." as a boundary a model may
      // exploit, pairing any two unrelated served integers into a rate the
      // packet never stated.
      if (packet.decimals.has(`${whole}.${frac}`)) continue;
      return "integer";
    }
    if (integersOf(token).every((n) => packet.figures.has(n))) continue;
    return "integer";
  }
  // 4. AND THE SAME GATE FOR A SPELLED ONE. The digit tokens are decided above,
  // so they are blanked and what is left is read for numerals (`integersOf`
  // `:634`): `thirty-two` meets `figures` exactly as `32` would.
  for (const n of integersOf(rest.replace(/\d+(?:\.\d+)?/g, " "), prose)) if (!packet.figures.has(n)) return "quantity";
  return null;
}

/** The rejection is counted PLATFORM-WIDE AND NOWHERE ELSE (INV-005): a record
 * saying which creator's explanation was refused is a per-creator fact about a
 * fan-derived surface, and the operator's question here is "is the model
 * inventing numbers", which is one number for the platform. The reply itself is
 * never written: it is the object suspected of carrying a figure nobody
 * floored, and a log is a place it would outlive the request. Marker constant
 * plus one line, as `analytics-outbox.ts OUTBOX_DEAD_MARKER` does. */
export const AI_REPLY_REJECTED_MARKER = "AI_REPLY_REJECTED";
/** I3-0597 (I15-W2J): the inference produced no usable reply — the provider threw, answered
 * empty, or answered over MAX_REPLY. Like a refused reply, not a service rendered: the route
 * refunds it through the same bounded `explain-refused:` bucket. Same 502 message. */
export const AI_NO_ANSWER_MARKER = "AI_NO_ANSWER";
function rejectedReply(offence: PacketOffence): void {
  // THE REASON NAMES THE CHANNEL THAT REFUSED (RUN-0013 D-1 (a)). It said
  // `integer_not_in_packet` for every refusal, including the ones where the
  // reply wrote no digit at all and the spelled channel (`:771`) is what
  // refused it — a record stating a fact that did not happen. The OFFENDING
  // TOKEN is deliberately NOT here: the refused quantity is the one object
  // suspected of carrying a number nobody floored, which is the same reason the
  // reply is not written, and `test/s003-reply-check.test.ts` pins that no
  // number from a reply reaches this line.
  // I13-AI (#899): `reference` = the reply named a withheld measure at all (`namesWithheldMeasure`).
  const reason = offence === "integer" ? "integer_not_in_packet" : offence === "quantity" ? "quantity_not_in_packet" : "withheld_reference";
  replyRecord(reason);
}

export const AI_REPLY_CUT_REASON = "ungrounded_clauses_cut";
function cutReply(): void {
  replyRecord(AI_REPLY_CUT_REASON);
}
function replyRecord(reason: string): void {
  logError({ cid: crypto.randomUUID(), hop: "http", msg: AI_REPLY_REJECTED_MARKER, marker: AI_REPLY_REJECTED_MARKER, reason });
}

function textOf(out: unknown): string | null {
  if (typeof out === "string") return out;
  if (out && typeof out === "object" && "response" in out) {
    const r = (out as { response: unknown }).response;
    if (typeof r === "string") return r;
  }
  return null;
}
