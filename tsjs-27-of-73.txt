#!/usr/bin/env node
// scripts/release-export.mjs — I21-Q76 (Jason, D-1030: "(a) one private repo plus a release script that ships product
// paths only"). Copies the product paths (PRODUCT_DIRS, and the ROOT_CONFIGS present) from --root (default: this tree)
// into --out, which must not exist. It SCANS FIRST and copies nothing when any path it would copy is an org record:
// a path segment in FORBIDDEN_DIRS or a file name starting with a FORBIDDEN_PREFIXES entry — exit 1, naming every
// such path. `node_modules` is never shipped. --check scans only (exit 0 clean, 1 refused, nothing written).
// D-1059(17) (I30-RG-REG NEW-3): the file list is the GIT INDEX under those paths, never a working-tree walk, so an
// untracked gitignored file (services/trust-core/.dev.vars, .DS_Store) never ships; a TRACKED file that the ignore
// rules match (force-added) is refused like an org record. A --root that is not a git work tree is refused (exit 2).
// I34-TOOLS-C (I30 B13): the BYTES are git's too, never the working tree's: each file is written from its index blob
// (`git ls-files -s`, read with `git cat-file --batch`), or with --rev <rev> the list and the bytes are that commit's
// (`git ls-tree -r <rev>`; the ignore rules are checked with `git check-ignore --no-index`). An indexed path that is
// deleted or modified in the working tree no longer throws: it ships its git bytes, and a deleted one is named on
// stdout. Mode 100755 keeps its executable bit, 120000 is written as a symlink, and a submodule entry (160000) is named
// and not exported.
//   node scripts/release-export.mjs --out <dir> [--root <dir>] [--rev <rev>] [--check]
import { execFileSync, spawnSync } from "node:child_process";
import { chmodSync, existsSync, lstatSync, mkdirSync, realpathSync, symlinkSync, writeFileSync } from "node:fs";
import { dirname, join, resolve } from "node:path";
import { fileURLToPath } from "node:url";

export const PRODUCT_DIRS = ["apps", "services", "packages", "docs", "ops"];
export const ROOT_CONFIGS = [
  "package.json",
  "pnpm-lock.yaml",
  "pnpm-workspace.yaml",
  "tsconfig.json",
  "tsconfig.base.json",
  "biome.json",
  "vitest.config.ts",

export const FORBIDDEN_DIRS = ["org", "reports", "outputs"];
export const FORBIDDEN_PREFIXES = ["MANIFEST-", "COMMIT-MSG-", "plan-"];
const SKIPPED_DIRS = ["node_modules"];

const git = (root, args) => execFileSync("git", ["--no-optional-locks", "--literal-pathspecs", ...args], { cwd: root, encoding: "utf8", stdio: ["ignore", "pipe", "pipe"] });
/** `git ls-files -z <args> -- <product dirs and root configs>` in `root`, minus SKIPPED_DIRS; throws unless `root` is the top of a work tree. */
function indexed(root, args) {
  const top = git(root, ["rev-parse", "--show-toplevel"]).trim();
  if (realpathSync(top) !== realpathSync(root)) throw new Error(`the work tree's top is ${top}`);
  const out = git(root, ["ls-files", "-z", ...args, "--", ...PRODUCT_DIRS, ...ROOT_CONFIGS]);
  return out.split("\0").filter((p) => p !== "" && !p.split("/").slice(0, -1).some((s) => SKIPPED_DIRS.includes(s)));
}

/** Every file (repo-relative) the export would copy: the git index's entries under the product directories and root configs. */
export function exportPaths(root) {
  return indexed(root, ["--cached"]);
}

/** The indexed paths the ignore rules match (force-added): a local file such as `.dev.vars` that must never ship. */
export function ignoredPaths(root) {
  return indexed(root, ["--cached", "--ignored", "--exclude-standard"]);
}

const skipped = (p) => p.split("/").slice(0, -1).some((s) => SKIPPED_DIRS.includes(s));
/** [{ mode, sha, path }] for every export path: the index's stage-0 entries, or with `rev` that commit's tree entries. */
export function exportEntries(root, rev) {
  const scope = ["--", ...PRODUCT_DIRS, ...ROOT_CONFIGS];
  const out = rev === undefined ? git(root, ["ls-files", "-s", "-z", ...scope]) : git(root, ["ls-tree", "-r", "-z", rev, ...scope]);
  const entries = [];
  for (const rec of out.split("\0")) {
    const m = rev === undefined ? /^(\d{6}) ([0-9a-f]{40,64}) (\d)\t(.+)$/s.exec(rec) : /^(\d{6}) \w+ ([0-9a-f]{40,64})\t(.+)$/s.exec(rec);
    if (!m) continue;
    const [mode, sha, stage, path] = rev === undefined ? [m[1], m[2], m[3], m[4]] : [m[1], m[2], "0", m[3]];
    if (stage === "0" && !skipped(path)) entries.push({ mode, sha, path });
  }
  return entries;
}

/** The bytes of each blob sha, from one `git cat-file --batch`. */
function blobBytes(root, shas) {
  if (shas.length === 0) return new Map();
  const buf = execFileSync("git", ["--no-optional-locks", "cat-file", "--batch"], { cwd: root, input: `${shas.join("\n")}\n`, maxBuffer: 2 ** 31 - 1, stdio: ["pipe", "pipe", "pipe"] });
  const bytes = new Map();
  let at = 0;
  for (const sha of shas) {
    const nl = buf.indexOf(10, at);
    const head = buf.subarray(at, nl).toString("utf8").split(" ");
    if (head[0] !== sha || head[1] !== "blob") throw new Error(`git cat-file --batch: expected blob ${sha}, got "${head.join(" ")}"`);
    const size = Number(head[2]);
    bytes.set(sha, buf.subarray(nl + 1, nl + 1 + size));
    at = nl + 1 + size + 1;
  }
  return bytes;
}

/** The paths among `paths` that are org records and may never ship. */
export function forbiddenPaths(paths) {
  return paths.filter((p) => {
    const parts = p.split("/");
    return parts.slice(0, -1).some((s) => FORBIDDEN_DIRS.includes(s)) || FORBIDDEN_PREFIXES.some((x) => parts[parts.length - 1].startsWith(x));
  });
}

  const root = resolve(arg("--root") ?? join(dirname(fileURLToPath(import.meta.url)), ".."));
  const out = arg("--out");
  const check = argv.includes("--check");
  const rev = arg("--rev");
  if (argv.includes("--rev") && (rev === undefined || rev.startsWith("--"))) {
    console.error("release-export: --rev needs a revision");
    return 2;
  }
  if (!check && (out === undefined || existsSync(out))) {
    console.error(`release-export: --out <dir> is required and must not exist${out === undefined ? "" : ` (${out} exists)`}`);
    return 2;
  }
  let paths;
  let ignored;
  let entries;
  try {
    paths = exportPaths(root);
    ignored = ignoredPaths(root);
  } catch (e) {
    console.error(`release-export: --root ${root} is not the top of a git work tree; the file list is the git index (${String(e.stderr || e.message).trim().split("\n")[0]})`);

  }

  } catch (e) {
    console.error(`release-export: cannot read ${rev === undefined ? "the index" : `revision ${rev}`} (${String(e.stderr || e.message).trim().split("\n")[0]})`);
    return 2;
  }
  const bad = forbiddenPaths(paths);
  if (bad.length > 0) {
    console.error(`release-export: REFUSED — ${bad.length} org record path(s) would ship:\n${bad.map((p) => `  ${p}`).join("\n")}`);
    return 1;
  }
  if (ignored.length > 0) {
    console.error(`release-export: REFUSED — ${ignored.length} tracked path(s) match the ignore rules (a local file force-added):\n${ignored.map((p) => `  ${p}`).join("\n")}`);
    return 1;
  }
  if (check) {
    console.log(`release-export: ${paths.length} product file(s), none forbidden`);
    return 0;
  }
  const files = entries.filter((e) => e.mode !== "160000");
  const bytes = blobBytes(root, [...new Set(files.map((e) => e.sha))]);
  for (const e of files) {
    const dest = join(out, e.path);
    mkdirSync(dirname(dest), { recursive: true });
    if (e.mode === "120000") symlinkSync(bytes.get(e.sha).toString("utf8"), dest);
    else {
      writeFileSync(dest, bytes.get(e.sha));
      if (e.mode === "100755") chmodSync(dest, 0o755);
    }
  }
  const from = rev === undefined ? "the index" : rev;
  console.log(`release-export: copied ${files.length} file(s) from ${from} to ${out}`);
  const gone = files.filter((e) => !onDisk(join(root, e.path))).map((e) => e.path);
  if (gone.length > 0) console.log(`release-export: ${gone.length} path(s) deleted in the working tree; their bytes from ${from} shipped:\n${gone.map((p) => `  ${p}`).join("\n")}`);
  const subs = entries.filter((e) => e.mode === "160000").map((e) => e.path);
  if (subs.length > 0) console.log(`release-export: ${subs.length} submodule path(s) not exported:\n${subs.map((p) => `  ${p}`).join("\n")}`);
  return 0;
}

const onDisk = (p) => { try { lstatSync(p); return true; } catch { return false; } };
/** The paths among `paths` that the ignore rules match, tracked or not (`git check-ignore --no-index`; exit 1 = none). */
function checkIgnored(root, paths) {
  if (paths.length === 0) return [];
  const r = spawnSync("git", ["--no-optional-locks", "check-ignore", "--no-index", "-z", "--stdin"], { cwd: root, input: `${paths.join("\0")}\0`, encoding: "utf8" });
  if (r.status === 1) return [];
  if (r.status !== 0) throw new Error(`git check-ignore: ${String(r.stderr).trim().split("\n")[0]}`);
  return r.stdout.split("\0").filter((p) => p !== "");
}

if (process.argv[1] !== undefined && realpathSync(process.argv[1]) === fileURLToPath(import.meta.url)) process.exit(main(process.argv.slice(2)));

/**
 * INV-094 (I56-EDGE, D-1433): the restore drill record and its cadence check.
 *
 * The record is `ops/restore-drills.jsonl`: line 1 is the header `{"format":"dhwaja-restore-drills","version":1}`;
 * every later line is one drill `{"at":"YYYY-MM-DD","target":"<scratch resource restored into>","source":"<backup or
 * point-in-time restored from>","result":"pass"|"fail","by":"<who ran it>"}`. A drill counts as REHEARSED only when its
 * result is "pass". DEPLOY.md's "Date of the last restore actually REHEARSED" row must name the same date (UNKNOWN when
 * none), which tests/invariants/inv094-restore-drill.test.ts checks.
 *
 * The cadence is a pure function of its arguments: it never reads the clock. N (the most days a pass may age after GA)
 * is DRAFT, Jason's: RESTORE_DRILL_MAX_AGE_DAYS stays null until he sets it, and "after GA" is a parameter because the
 * code holds no GA date or flag. After GA an age that is not a finite number of days at or above 0 (NaN, negative,
 * Infinity) is not a fresh drill: it answers "fail", as no drill does (I59 FX1-M5).
 */

export const RESTORE_DRILL_FORMAT = "dhwaja-restore-drills";
export const RESTORE_DRILL_VERSION = 1;
/** DRAFT (Jason): the most days the last passing drill may age after GA. No value is chosen here. */
export const RESTORE_DRILL_MAX_AGE_DAYS = null;

const DATE_RE = /^\d{4}-\d{2}-\d{2}$/;
const KEYS = ["at", "by", "result", "source", "target"];

/**
 * @param {string} text the record file's contents
 * @returns {{ at: string, target: string, source: string, result: "pass" | "fail", by: string }[]}
 */
export function parseRestoreDrills(text) {
  const lines = text.split("\n").filter((l) => l.trim() !== "");
  if (lines.length === 0) throw new Error("restore-drills: no header line");
  const head = JSON.parse(lines[0]);
  if (head?.format !== RESTORE_DRILL_FORMAT || head?.version !== RESTORE_DRILL_VERSION) throw new Error("restore-drills: line 1 is not the format header");
  return lines.slice(1).map((l, i) => {
    const r = JSON.parse(l);
    const keys = r && typeof r === "object" ? Object.keys(r).sort() : [];
    if (keys.join(",") !== KEYS.join(",")) throw new Error(`restore-drills: line ${i + 2} keys are ${keys.join(",")}, want ${KEYS.join(",")}`);
    if (!DATE_RE.test(r.at) || Number.isNaN(Date.parse(`${r.at}T00:00:00Z`)) || new Date(`${r.at}T00:00:00Z`).toISOString().slice(0, 10) !== r.at) throw new Error(`restore-drills: line ${i + 2} at is not a calendar date`);
    if (r.result !== "pass" && r.result !== "fail") throw new Error(`restore-drills: line ${i + 2} result is not pass or fail`);
    for (const k of ["target", "source", "by"]) if (typeof r[k] !== "string" || r[k].trim() === "") throw new Error(`restore-drills: line ${i + 2} ${k} is empty`);
    return r;
  });
}

/**
 * The date of the latest passing drill, or null when none passed.
 * @param {{ at: string, result: string }[]} records
 * @returns {string | null}
 */
export function lastRehearsedAt(records) {
  let last = null;
  for (const r of records) if (r.result === "pass" && (last === null || r.at > last)) last = r.at;
  return last;
}

/**
 * @param {number | null} ageDays days since the last passing drill, or null when none is recorded
 * @param {number} atMs the instant the age was measured at (reported, never read from the clock)
 * @param {{ gaReached: boolean, maxAgeDays: number | null }} opts
 * @returns {{ level: "note" | "fail", message: string }}
 */
export function restoreDrillCadenceStatus(ageDays, atMs, opts) {
  const when = new Date(atMs).toISOString().slice(0, 10);
  const seen = ageDays === null ? "no restore drill is recorded" : `the last passing restore drill is ${Math.floor(ageDays)} days old`;
  if (!opts.gaReached) return { level: "note", message: `${seen} (measured ${when}); the cadence is not checked before GA.` };
  if (opts.maxAgeDays === null) return { level: "fail", message: `${seen} (measured ${when}); after GA the cadence needs RESTORE_DRILL_MAX_AGE_DAYS, which is unset (DRAFT, Jason).` };
  if (ageDays === null || !Number.isFinite(ageDays) || ageDays < 0 || ageDays > opts.maxAgeDays) return { level: "fail", message: `${seen} (measured ${when}); after GA a passing drill is due every ${opts.maxAgeDays} days.` };
  return { level: "note", message: `${seen} (measured ${when}); within ${opts.maxAgeDays} days.` };
}

const slug = (s) => s.toLowerCase().replace(/[^a-z0-9]+/g, "-").replace(/^-|-$/g, "");

function fail(msg) {
  const e = new Error(msg);
  e.routing = true;
  throw e;
}

function walk(dir, out = []) {
  for (const n of readdirSync(dir).sort()) {
    const p = join(dir, n);
    if (statSync(p).isDirectory()) walk(p, out);
    else out.push(p);
  }
  return out;
}

// I48-TOOLS (A115's sibling, FAMILY-013 class census): the persona block is read by skill-method.mjs readFrontmatter, the ONE
// reader (LF or CRLF line ends, one leading BOM), as red-stage-gen.mjs findPersona and the compiler read it.
function frontmatter(text) {
  const r = readFrontmatter(text);
  if (r === null) return null;
  const fm = {};
  for (const line of r.fm.split("\n")) {
    const m = /^(id|name):\s*"?(.*?)"?\s*$/.exec(line);
    if (m) fm[m[1]] = m[2];
  }
  return fm;
}

/** Every persona file's frontmatter, keyed both ways. A duplicate id or name refuses (ambiguous lens). */
export function indexPersonas(root) {
  const byId = new Map();
  const byName = new Map();
  for (const p of walk(join(root, PERSONAS))) {
    if (!p.endsWith(".md")) continue;
    const fm = frontmatter(readFileSync(p, "utf8"));
    if (!fm?.id || !fm.name) continue;
    const rel = relative(root, p);
    if (byId.has(fm.id)) fail(`persona id ${fm.id} is declared twice: ${byId.get(fm.id).file}, ${rel}`);
    byId.set(fm.id, { id: fm.id, name: fm.name, file: rel });
    byName.set(fm.name, [...(byName.get(fm.name) ?? []), fm.id]);
  }
  return { byId, byName };
}

function resolveName(idx, name, where) {
  const ids = idx.byName.get(name) ?? [];
  if (ids.length !== 1) fail(`${where}: lens "${name}" resolves to ${ids.length} persona files (need exactly 1)`);
  return ids[0];
}

/** Templates and red persona for one campaign id, from the tree. */
export function campaignFiles(root, rid) {
  const tdir = join(root, TEMPLATES);
  const templates = existsSync(tdir) ? readdirSync(tdir).filter((f) => f.startsWith(`${rid}-`) && f.endsWith(".json")).sort().map((f) => `${TEMPLATES}/${f}`) : [];
  const rdir = join(root, RED);
  const red = existsSync(rdir) ? readdirSync(rdir).filter((f) => f.startsWith(`${rid}-`) && f.endsWith(".md")).sort().map((f) => `${RED}/${f}`) : [];
  return { templates, red };
}

function resolveCampaign(root, idx, rid, name, where) {
  const { templates, red } = campaignFiles(root, rid);
  if (templates.length === 0) fail(`${where}: campaign ${rid} has no template ${TEMPLATES}/${rid}-*.json`);
  if (red.length !== 1) fail(`${where}: campaign ${rid} has ${red.length} red persona files ${RED}/${rid}-*.md (need exactly 1)`);
  const p = idx.byId.get(rid);
  if (!p || p.file !== red[0]) fail(`${where}: red persona file ${red[0]} does not declare id ${rid}`);
  if (name !== undefined && p.name !== name) fail(`${where}: guide names ${rid} "${name}", ${red[0]} names it "${p.name}"`);
  return rid;
}

function parseWave(root, idx, step, text) {
  const lenses = [];
  const campaigns = [];
  const unresolved = [];
  for (const raw of text.split("→")) {
    const tok = raw.trim();
    let handled = false;
    const range = /\((\d+)–(\d+)\)/.exec(tok);
    if (range) {
      handled = true;
      for (let n = Number(range[1]); n <= Number(range[2]); n++) {
        if (!idx.byId.has(String(n))) fail(`wave ${step}: persona id ${n} in range ${range[0]} has no persona file`);
        lenses.push(String(n));
      }
    }
    for (const m of tok.matchAll(/\b(M\d\d) /g)) {
      handled = true;
      const p = idx.byId.get(m[1]);
      if (!p || !tok.includes(`${m[1]} ${p.name}`)) fail(`wave ${step}: "${tok}" does not name ${m[1]} by its persona name`);
      lenses.push(m[1]);
    }
    const rr = /\bR(\d\d)–R(\d\d)\b/.exec(tok);
    if (rr) {
      handled = true;
      for (let n = Number(rr[1]); n <= Number(rr[2]); n++) campaigns.push(resolveCampaign(root, idx, `R${String(n).padStart(2, "0")}`, undefined, `wave ${step}`));
    }
    if (handled) continue;
    const ids = idx.byName.get(tok) ?? [];
    if (ids.length === 1) lenses.push(ids[0]);
    else unresolved.push(tok);
  }
  return { step, text, lenses, campaigns, unresolved };
}

/** Parse the guide and resolve everything; returns the ROUTING.json object. Refuses on any unresolved input. */
export function build(root) {
  const idx = indexPersonas(root);
  const lines = readFileSync(join(root, GUIDE), "utf8").split("\n");
  const sections = [];
  let cur = null;
  for (const line of lines) {
    const h = /^## (.+)$/.exec(line);
    if (h) {
      cur = { title: h[1].trim(), body: [] };
      sections.push(cur);
    } else if (cur && line.trim()) cur.body.push(line.trim());
  }
  const subsystems = [];
  let prerelease_waves = null;
  for (const s of sections) {
    if (/^Pre-release/.test(s.title)) {
      prerelease_waves = s.body.map((l) => {
        const m = /^(\d+)\. (.+)$/.exec(l);
        if (!m) fail(`pre-release section: line is not a numbered step: "${l}"`);
        return parseWave(root, idx, Number(m[1]), m[2]);
      });
      continue;
    }
    const where = `section "${s.title}"`;
    if (s.body.length !== 1 || !s.body[0].includes(" Red: ")) fail(`${where}: expected exactly one body line carrying " Red: "`);
    const [lensPart, redPart] = s.body[0].split(" Red: ");
    const lenses = lensPart.split(";").map((x) => x.trim().replace(/\.$/, "")).filter(Boolean).map((n) => resolveName(idx, n, where));
    const campaigns = redPart.split(";").map((x) => x.trim().replace(/\.$/, "")).filter(Boolean).map((c) => {
      const m = /^(R\d\d) (.+)$/.exec(c);
      if (!m) fail(`${where}: campaign "${c}" is not "Rnn Name"`);
      return resolveCampaign(root, idx, m[1], m[2], where);
    });
    const tags = s.title.split("/").map((t) => slug(t)).filter(Boolean);
    const id = tags[0];
    const decl = PATHS[id];
    if (!decl) fail(`${where}: subsystem id "${id}" has no PATHS entry in scripts/route-personas.mjs`);
    for (const p of decl.paths) if (!prefixExists(root, p)) fail(`${where}: path prefix "${p}" matches nothing in the tree`);
    subsystems.push({ id, title: s.title, paths: [...decl.paths], tags, lenses, campaigns });
  }
  if (!prerelease_waves) fail("guide has no ## Pre-release section");
  const ids = subsystems.map((s) => s.id);
  if (new Set(ids).size !== ids.length) fail(`duplicate subsystem ids: ${ids.join(", ")}`);
  for (const k of Object.keys(PATHS)) if (!ids.includes(k)) fail(`PATHS entry "${k}" names no guide section`);
  const others = [PERSONAS, "docs/review-personas"].flatMap((d) => (existsSync(join(root, d)) ? walk(join(root, d)) : [])).map((p) => relative(root, p)).filter((p) => /routing/i.test(p) && p !== GUIDE && p !== OUT);
  if (others.length) fail(`absent claim is false: another routing guide exists: ${others.join(", ")}`);
  return { source: GUIDE, generated_from: "node scripts/route-personas.mjs --build", subsystems, prerelease_waves, absent: ["pack 1 routing guide: in neither tree"] };
}

function prefixExists(root, prefix) {
  const full = join(root, prefix);
  if (prefix.endsWith("/")) return existsSync(full);
  if (existsSync(full)) return true;
  const dir = dirname(full);
  const base = full.slice(dir.length + 1);
  return existsSync(dir) && readdirSync(dir).some((n) => n.startsWith(base));
}

export const serialize = (obj) => `${JSON.stringify(obj, null, 2)}\n`;

/** Every problem with the committed file; [] means it is current and every id resolves. */
export function check(root) {
  const problems = [];
  let want;
  try {
    want = serialize(build(root));
  } catch (e) {
    if (!e.routing) throw e;
    return [`build refuses: ${e.message}`];
  }
  const outPath = join(root, OUT);
  if (!existsSync(outPath)) return [`${OUT} is missing; run --build`];
  const have = readFileSync(outPath, "utf8");
  if (have !== want) problems.push(`${OUT} differs from what --build would write; run --build`);
  let doc;
  try {
    doc = JSON.parse(have);
  } catch {
    return [...problems, `${OUT} is not JSON`];
  }
  const idx = indexPersonas(root);
  for (const s of doc.subsystems ?? []) {
    for (const id of s.lenses ?? []) if (!idx.byId.has(id)) problems.push(`${s.id}: lens ${id} has no persona file`);
    for (const rid of s.campaigns ?? []) {
      const { templates, red } = campaignFiles(root, rid);
      if (templates.length === 0) problems.push(`${s.id}: campaign ${rid} has no template`);
      if (red.length !== 1) problems.push(`${s.id}: campaign ${rid} has no single red persona`);
    }
  }
  return problems;
}

/** Product units the routing must cover: every .ts file under services/<x>/src, and every top-level apps/<x>/ and packages/<x>/ dir. */
export function productUnits(root) {
  const units = [];
  const dirs = (d) => (existsSync(join(root, d)) ? readdirSync(join(root, d)).filter((n) => statSync(join(root, d, n)).isDirectory()).sort() : []);
  for (const svc of dirs("services")) {
    const src = join(root, "services", svc, "src");
    if (existsSync(src)) units.push(...walk(src).filter((p) => p.endsWith(".ts")).map((p) => relative(root, p).replace(/\\/g, "/")));
  }
  for (const top of ["apps", "packages"]) units.push(...dirs(top).map((n) => `${top}/${n}/`));
  return units;
}

/** The product units no subsystem in `doc` routes (lookup by path finds none). [] means coverage is complete. */
export const coverage = (doc, root) => productUnits(root).filter((u) => lookup(doc, u, root).subsystems.length === 0);

/** Subsystems for a query: exact id, else exact tag, else longest string-prefix path match (ties all returned). */
export function lookup(doc, query, root) {
  const subs = doc.subsystems;
  const byId = subs.filter((s) => s.id === query);
  if (byId.length) return { by: "subsystem", subsystems: byId };
  const byTag = subs.filter((s) => s.tags.includes(query));
  if (byTag.length) return { by: "tag", subsystems: byTag };
  let q = query.replace(/\\/g, "/");
  if (isAbsolute(q)) q = relative(root, q).replace(/\\/g, "/");
  q = q.replace(/^\.\//, "");
  let best = 0;
  let hits = [];
  for (const s of subs) {
    for (const p of s.paths) {
      if (!q.startsWith(p)) continue;
      if (p.length > best) {
        best = p.length;
        hits = [s];
      } else if (p.length === best && !hits.includes(s)) hits.push(s);
    }
  }
  return { by: "path", prefix_length: best, subsystems: hits };
}

  let root = resolve(dirname(fileURLToPath(import.meta.url)), "..");
  const ri = args.indexOf("--root");
  if (ri >= 0) {
    if (!args[ri + 1]) return usage("--root needs a value");
    root = resolve(args[ri + 1]);
    args.splice(ri, 2);

  const json = args.includes("--json");
  const rest = args.filter((a) => a !== "--json");
  try {
    if (rest[0] === "--build" && rest.length === 1) {
      writeFileSync(join(root, OUT), serialize(build(root)));
      console.log(`wrote ${OUT}`);
      return 0;
    }
    if (rest[0] === "--check" && rest.length === 1) {
      const problems = check(root);
      for (const p of problems) console.log(`FAIL ${p}`);
      if (problems.length === 0) console.log(`OK ${OUT} is current and every lens and campaign resolves`);
      return problems.length ? 1 : 0;
    }
    if (rest[0] === "--coverage" && rest.length === 1) {
      const doc = JSON.parse(readFileSync(join(root, OUT), "utf8"));

      if (un.length === 0) console.log(`OK ${productUnits(root).length} product units route to at least one subsystem`);
      return un.length ? 1 : 0;
    }
    if (rest.length !== 1 || rest[0].startsWith("--")) return usage("expected exactly one of --build, --check, --coverage or a query");
    const doc = JSON.parse(readFileSync(join(root, OUT), "utf8"));
    const res = lookup(doc, rest[0], root);
    if (json) console.log(JSON.stringify(res, null, 2));
    else if (res.subsystems.length === 0) console.log(`no subsystem matches "${rest[0]}"`);
    else {
      const idx = indexPersonas(root);
      for (const s of res.subsystems) {
        console.log(`${s.id} — ${s.title} (matched by ${res.by})`);
        console.log(`  lenses: ${s.lenses.map((id) => `${id} ${idx.byId.get(id)?.name ?? "?"}`).join("; ")}`);
        console.log(`  campaigns: ${s.campaigns.map((rid) => `${rid} ${campaignFiles(root, rid).templates.join(",")}`).join("; ")}`);

//   node scripts/s4-due.mjs [--json] [--root <repo>]

// due if: (a) pnpm-lock.yaml's last commit is later than S4-CALENDAR.scanned_at, or

// Exit 0 either way (this is intelligence, not a gate).

import { execSync } from "node:child_process";
import { existsSync, readFileSync } from "node:fs";
import { join, resolve } from "node:path";

const args = process.argv.slice(2);
const opt = (k, d) => (args.includes(k) ? args[args.indexOf(k) + 1] : d);
const JSON_OUT = args.includes("--json");
const ROOT = resolve(opt("--root", process.cwd()));
const p = (...s) => join(ROOT, ...s);

const calPath = p("org/08-control/S4-CALENDAR.json");
const THIRTY_DAYS_MS = 30 * 24 * 60 * 60 * 1000;

let due = false, reason = "", scannedAt = null, lockfileAt = null;

if (!existsSync(calPath)) {
  due = true;
  reason = "no calendar";
} else {
  const cal = JSON.parse(readFileSync(calPath, "utf8"));
  scannedAt = cal.scanned_at;
  const scannedMs = Date.parse(scannedAt);

  let lockfileMs = null;
  try {
    const out = execSync("git log -1 --format=%cI -- pnpm-lock.yaml", { cwd: ROOT, encoding: "utf8", stdio: ["ignore","pipe","ignore"] }).trim();
    if (out) { lockfileMs = Date.parse(out); lockfileAt = out; }

    // git itself failing (the build tree has had a .git since D-0409), and rule (a) cannot be skipped in silence.
    console.error(`s4-due: git log for pnpm-lock.yaml cannot be read at ${ROOT}: ${String(e.stderr ?? e.message).trim()} — rule (a) cannot be checked`);
    process.exit(1);
  }

  if (lockfileMs != null && lockfileMs > scannedMs) {
    due = true;
    reason = `lockfile changed ${lockfileAt} > scanned_at ${scannedAt}`;
  } else if (Date.now() - scannedMs > THIRTY_DAYS_MS) {
    due = true;
    reason = `scanned_at ${scannedAt} older than 30 days`;
  } else {
    due = false;
    const next = new Date(scannedMs + THIRTY_DAYS_MS).toISOString().slice(0, 10);
    reason = `next ${next}; lockfile unchanged since ${scannedAt}${lockfileMs == null ? " (lockfile git history unavailable at this ROOT)" : ""}`;
  }
}

if (JSON_OUT) {
  console.log(JSON.stringify({ due, reason, scanned_at: scannedAt, lockfile_commit_at: lockfileAt }));

#!/usr/bin/env node
// seed-score.mjs — QUEUE #32 (S-075; SEEDS2 R1; D-0424.3/#278 item e — S-075 R3): unseal a run's
// seeds and score seeded recall.
//
//   node scripts/seed-score.mjs --run <RUN-id> --unseal <file> --findings <judge FINDINGS>
//        [--root <org root>] [--git <build tree>] [--discard] [--allow-empty]
//
// Checks the unseal file (kept outside the repo) against the in-repo
// <root>/org/06-runs/instances/<run>/seeds.sealed.json (seeds.sealed/2: run, planted count, one
// salted commitment per seed over file + seeded blob + id + family + description; any mismatch
// exits 2); with --git, that each seeded blob is still at <seed_tip>:<file> and — while the
// worktree exists — in the worktree. A seed is HIT when the FINDINGS text cites its file AT A LINE
// INSIDE ITS EXACT WINDOW (no slack) AND names the seed's family/mechanism token within
// MECHANISM_PROXIMITY_CHARS of that citation (`path:line` or `path:a-b`) — see the hit-rule block
// below for the tightened rule (retires the old ±WINDOW_SLACK-only rule, S-075 R3, latent).
// Writes <root>/org/06-runs/instances/<run>/seeds.scored.json (per seed, per family) and sets
// the instance's seeds.scored / seeds.seeded_recall / seeds.scored_ref (read by halt.mjs D5).
// A FINDINGS file with no `path:line` citation at all exits 2 unless --allow-empty: an
// unreadable judge output is not recall 0.
// --discard (needs --git) then removes the worktree and deletes the sweep branch named in the
// unseal file.
import { execFileSync } from "node:child_process";
import { existsSync, readFileSync, renameSync, rmSync, writeFileSync } from "node:fs";
import { dirname, join, relative, resolve } from "node:path";
import { fileURLToPath } from "node:url";

const args = process.argv.slice(2);
const val = (f) => (args.includes(f) ? args[args.indexOf(f) + 1] : null);
const die = (m) => { console.error(`seed-score: ${m}`); process.exit(2); };
const ROOT = resolve(val("--root") ?? join(HERE, ".."));
const INST_DIR = join(ROOT, "org/06-runs/instances");
const WINDOW_SLACK = 3; // kept only as a recorded figure (seeds.scored.json) — no longer widens the hit window
const MECHANISM_PROXIMITY_CHARS = 400;

const run = val("--run"), unsealPath = val("--unseal"), findingsPath = val("--findings"), gitDir = val("--git");

if (inst.seeds?.declared !== true) die(`${instPath} does not declare seeds`);
const sealed = JSON.parse(readFileSync(sealedPath, "utf8"));
const unseal = JSON.parse(readFileSync(unsealPath, "utf8"));

  die("unseal and sealed seed counts differ");

// the old rule scored a hit on ANY finding that named the file near the window, seed-related or
// not; the first tightening then required the family SLUG verbatim, under-counting a judge who
// correctly describes the mechanism in prose). A hit now needs three conjuncts: (1) same file
// (suffix match), (2) the citation's line range overlaps the seed's EXACT window (no
// +/-WINDOW_SLACK widening — the citation must land inside the range the seed actually changed),
// and (3) EITHER the seed's family token (its mechanism shape — one of FAMILIES) OR one of the
// seed's recorded mechanism keywords (3-5 per family, written at plant time — seed-plant.mjs
// FAMILY_KEYWORDS) is named in the FINDINGS text within MECHANISM_PROXIMITY_CHARS characters of
// that citation, tying the mechanism mention to THAT citation rather than anywhere in the document.
const samePath = (p, f) => p === f || f.endsWith(`/${p}`) || p.endsWith(`/${f}`);
const mechanismNear = (seed, idx) => {
  const lo = Math.max(0, idx - MECHANISM_PROXIMITY_CHARS);
  const hi = Math.min(text.length, idx + MECHANISM_PROXIMITY_CHARS);
  const around = text.slice(lo, hi).toLowerCase();
  const family = seed.family ?? "";
  const variants = [family, family.replace(/-/g, " "), family.replace(/-/g, "_"), ...(Array.isArray(seed.keywords) ? seed.keywords : [])];
  return variants.some((v) => v && around.includes(String(v).toLowerCase()));
};

const results = unseal.seeds.map((s) => {
  const [lo, hi] = s.window;
  const inWindow = cites.filter((c) => samePath(c.path, s.file) && c.a <= hi && c.b >= lo);
  const withToken = inWindow.filter((c) => mechanismNear(s, c.index));
  return { n: s.n, id: s.id, family: s.family, file: s.file, window: s.window, hit: withToken.length > 0,
    by: [...new Set(withToken.map((c) => c.raw))],
    near_but_no_mechanism: withToken.length === 0 ? [...new Set(inWindow.map((c) => c.raw))] : [] };
});
const byFamily = {};

  byFamily[r.family] ??= { planted: 0, hit: 0 };
  byFamily[r.family].planted++; byFamily[r.family].hit += r.hit ? 1 : 0;
}

const recall = hits / results.length;

  const docket = readFileSync(docketPath, "utf8");

    const found = rulingCandidates.some((r) => brief.includes(r));
    console.log(`docket #${id}: ${found ? "QUOTED" : "NOT QUOTED"}`);
    if (!found) anyNotQuoted = true;
  }
  process.exitCode = anyNotQuoted ? 1 : 0;

#!/usr/bin/env node
/**
 * ADR-0009 D4 / QUEUE #31(4) — sweep coverage check.
 *
 * Before an attack wave, every claim the serial batch's SERIAL-NOTE.md makes
 * (one `=== ITEM` block per builder) and every product/test file that actually
 * changed in the swept commit range must map to a sub-run in the sweep's own
 * `scout/PLAN.md` — or the tool names the gap instead of the sweep silently
 * skipping it (this is what DISCREPANCY-0024 was: WATERMARK's `recomputeWatermark`
 * fix shares a file, `analytics-outbox.ts`, with SENTINEL's `pruneAnalyticsRetention`
 * fix, so file-level coverage alone hides the gap — only item-level name
 * matching against the plan catches it).
 *
 * Usage: node scripts/sweep-coverage.mjs <run-dir> <from> <to>
 *   <run-dir>  path (relative to repo root, i.e. cwd) to the run's findings dir,
 *              e.g. org/07-findings/RUN-0013-sweep-s004
 *   <from> <to> commit range swept (git --no-optional-locks diff --name-only)
 *
 * Reads, relative to repo root (cwd; this script assumes it is run from the
 * mounted repo root, same convention as scripts/check-invariants.mjs):
 *   - SERIAL-NOTE.md            (repo root; NOT under <run-dir>)
 *   - <run-dir>/scout/PLAN.md   ("### Sub-run N — TITLE" blocks; "FILES:" lines)
 *   - `git --no-optional-locks diff --name-only <from>..<to>`
 *
 * Exit 1 if any SERIAL-NOTE item, or any changed file under a `src/` directory,
 * is UNCOVERED. Changed test files (`test/`, `tests/`, or `*.test.ts`) are
 * printed for visibility but do NOT gate the exit code (the brief's exit
 * condition names "product file or note item", not test file).
 *
 * BRIEF-DELTA: the brief describes SERIAL-NOTE item headings as `### <ITEM>`.
 * The live SERIAL-NOTE.md at repo root uses `=== ITEM` (three equals signs) for
 * its per-builder blocks; `###` is used for structural headings with mixed-case
 * prose ("### Orchestrator W2 …") that must NOT be read as items. This script
 * accepts EITHER marker but requires the heading text to be all-caps (one or
 * more space-separated ALL-CAPS/digit/-/_ tokens) with nothing but a bracketed
 * or dashed comment after it — that is what distinguishes a real item heading
 * from a prose subheading in both slices seen so far (S-003, S-004).
 *
 * Refuses to run with a bare `git status` — this tool only ever calls
 * `git --no-optional-locks diff --name-only`, never `git status` in any form.
 */

  <run-dir>  path to the run's findings dir (relative to repo root), e.g.
             org/07-findings/RUN-0013-sweep-s004
  <from>     git ref/commit the sweep started from
  <to>       git ref/commit the sweep ended at

Run from the repo root. Reads SERIAL-NOTE.md (repo root), <run-dir>/scout/PLAN.md,
and \`git --no-optional-locks diff --name-only <from>..<to>\`. Prints a coverage
table and exits 1 if any note item or changed src/ file is UNCOVERED.`);
}

const args = process.argv.slice(2);
if (args.includes("--help") || args.includes("-h") || args.length === 0) {

const root = process.cwd();
const runDir = join(root, runDirArg);

// ---------------------------------------------------------------- read inputs
let noteText;
try {
  noteText = readFileSync(join(root, "SERIAL-NOTE.md"), "utf8");
} catch (e) {

// An item heading is `===` or `###` followed by one-or-more ALL-CAPS tokens
// (letters/digits/-/_), optionally followed by a bracket/dash/paren comment.
// Anything with a lowercase letter in the heading text is NOT an item (it is a
// structural/prose subheading, e.g. "### Orchestrator W2 (what the builders...").
const HEADING_RE = /^(?:={3,}|#{2,4})\s+([A-Z][A-Z0-9_-]*(?:\s+[A-Z][A-Z0-9_-]*)*)\s*(?:[-–—([].*)?$/;

  console.error(`error: no "### Sub-run N — TITLE" blocks found in ${runDirArg}/scout/PLAN.md`);

// Extract each block's FILES: paragraph (from a line starting "FILES:" up to the
// next ALL-CAPS-word ":" label, a "scope tags:" line, a blank line, or the next
// heading) and the backtick paths inside it. A path may carry a `::symbol`
// annotation (e.g. `src/analytics-events.ts::outboxSelectSql`) — we keep the
// symbol for display and match on the path half only.
const LABEL_RE = /^([A-Z][A-Z ]{1,30}:|scope tags:)/;
function extractFilesParagraph(blockText) {
  const lines = blockText.split("\n");
  let capture = false;
  const buf = [];
  for (const raw of lines) {
    const line = raw.trim();
    if (!capture) {
      if (/^FILES:/.test(line)) {
        capture = true;
        buf.push(line);
      }
      continue;
    }
    if (line === "" || (LABEL_RE.test(line) && !/^FILES:/.test(line))) break;
    buf.push(line);
  }
  return buf.join(" ");
}
function extractBacktickEntries(text) {
  const entries = [];
  for (const m of text.matchAll(/`([^`]+)`/g)) {
    const raw = m[1];
    const [path, symbol] = raw.split("::");
    entries.push({ path, symbol: symbol ?? null });
  }
  return entries;
}
for (const b of blocks) {
  b.filesParagraph = extractFilesParagraph(b.text);
  b.fileEntries = extractBacktickEntries(b.filesParagraph);
}

// --------------------------------------------------------- item-name coverage
function coveringSubruns(itemName) {
  const wordRe = new RegExp(`\\b${itemName.replace(/[.*+?^${}()|[\]\\]/g, "\\$&")}\\b`, "i");
  return blocks.filter((b) => wordRe.test(b.title) || wordRe.test(b.text)).map((b) => `Sub-run ${b.n}`);
}

// --------------------------------------------------------- changed-file coverage
function isTestPath(p) {
  return /(^|\/)(tests?)(\/|$)/.test(p) || /\.test\.ts$/.test(p);
}
function isSrcPath(p) {
  return /(^|\/)src\//.test(p);
}
const relevantFiles = changedFiles.filter((p) => isSrcPath(p) || isTestPath(p));

  return blocks
    .filter((b) => b.fileEntries.some((e) => basename(e.path) === base))
    .map((b) => `Sub-run ${b.n}`);
}

console.log("");
console.log("## SERIAL-NOTE items");

console.log("## changed src/test files");

  }
}
if (relevantFiles.length === 0) console.log("  (none found)");


// #310c (S-088 RECALL, D-0431): a scoped file this large is asking to be its own sub-run (or
// split by line range across rooms) -- RUN-0016's seeded recall (0.25) traced partly to rooms
// declaring large parts of their scope unread; a scope that names console.ts (3,028 lines) or
// admin.ts (1,479 lines) alongside eight other files all but guarantees that. WARNING only --
// this tool judges coverage, not scope design; splitting a large file is the scout's/

const LARGE_FILE_LINES = 1000;
// QUEUE #310 REG-8 (S-088 REG round 1): a PLAN.md FILES: backtick entry is `path:line-ranges`
// (extractBacktickEntries only splits on "::" for a symbol suffix) -- every real plan uses this
// shape (RUN-0015 scout/PLAN.md:21,80,137), so `git show <ref>:<path>` must run on the path
// ALONE, never on "path:1-400,900-1119" (git reads that whole string as one bad path and fails).
function stripRanges(path) {
  return path.replace(/:[\d,\-]+$/, "");
}
function lineCountAtRef(ref, path) {
  try {
    const text = execFileSync("git", ["--no-optional-locks", "show", `${ref}:${stripRanges(path)}`], { cwd: root, encoding: "utf8", stdio: ["ignore", "pipe", "pipe"] });

    // counts as 1595, not 1596.
    return text.endsWith("\n") ? text.split("\n").length - 1 : text.split("\n").length;
  } catch {
    return null; // not present at that ref (deleted/renamed), or git failed -- nothing to warn about, and its stderr is swallowed (stdio above), never leaked to this tool's stdout
  }
}

// this repo's whole-tree listing was 1,757,577 bytes at i49-head-1. 64 MiB is about 38 times that.
const GIT_MAX_BUFFER = 64 * 1024 * 1024;
const git = (...a) => execFileSync("git", ["--no-optional-locks", ...a], { encoding: "utf8", maxBuffer: GIT_MAX_BUFFER });
const showCache = new Map();
const show = (ref, f) => {
  const key = `${ref}:${f}`;
  if (showCache.has(key)) return showCache.get(key);
  let text;
  // QUEUE #230: "" means "the path is not in that ref" (a file added or deleted in the range) and
  // nothing else; any other git failure (a bad ref, no repository) cannot be read as an absent file.
  try { text = execFileSync("git", ["--no-optional-locks", "show", key], { encoding: "utf8", stdio: ["ignore", "pipe", "pipe"], maxBuffer: GIT_MAX_BUFFER }); } catch (e) {
    const err = String(e.stderr ?? "");
    if (!isPathNotInRef(err)) { console.error(`sweep-graph: git show ${key} cannot be read: ${err.trim() || e.message}`); process.exit(2); }
    text = "";
  }
  showCache.set(key, text);
  return text;

const CODE_RE = /^(services|apps|packages|tests)\//;

// the ones this scan can read; the rest get one line under "## (no SQL surface)" instead of
// vanishing from the report entirely.

/** S-014 #86(i): one call hop, same shape as attack-scope.mjs's — for every imported name this
 *  file actually CALLS, follow the relative specifier (resolveSpec, below) and read that
 *  exported function's own SQL literals. Needs resolveSpec, defined further down this file, so
 *  it is called lazily rather than at parse time above it. */
function oneHopStatements(sf) {
  const out = [];
  const called = calledIdentifiers(sf);
  for (const imp of importsOf(sf)) {
    const names = imp.names.filter((nm) => called.has(nm));
    if (!names.length) continue;
    const resolved = resolveSpec(sf.fileName, imp.moduleSpecifier);
    if (!resolved) continue;
    const targetSf = parsedModule(resolved);
    if (!targetSf) continue;
    for (const nm of names) out.push(...exportedFunctionStatements(targetSf.statements, nm));
  }
  return out;
}

function tablesOf(src, path) {
  const written = [], read = [];
  let dynamicWrite = false, dynamicRead = false;
  try {
    const kind = path.endsWith(".tsx") ? ts.ScriptKind.TSX : ts.ScriptKind.TS;
    const sf = ts.createSourceFile(path, src, ts.ScriptTarget.Latest, true, kind);
    for (const s of [...sqlStatementsIn(sf), ...oneHopStatements(sf)]) {
      const m = s.match(TABLE);
      if (m) {
        if (WRITE_VERB.test(s)) written.push(m[1]);
        else if (READ_VERB.test(s)) read.push(m[1]);
        continue;
      }
      if (DYNAMIC_TABLE.test(s)) {
        if (WRITE_VERB.test(s)) dynamicWrite = true;
        else if (READ_VERB.test(s)) dynamicRead = true;
      }
    }
  } catch (e) { console.error(`sweep-graph: ${path} cannot be scanned for tables: ${e.message}`); process.exit(2); }
  const w = [...new Set(written)];
  const r = [...new Set(read)];
  if (dynamicWrite) w.push("<dynamic>");
  if (dynamicRead) r.push("<dynamic>");
  return { written: w, read: r };
}

function importsOfFile(src, path) {
  try {
    const kind = path.endsWith(".tsx") ? ts.ScriptKind.TSX : ts.ScriptKind.TS;
    const sf = ts.createSourceFile(path, src, ts.ScriptTarget.Latest, true, kind);
    return importsOf(sf).map((i) => i.moduleSpecifier).filter((m) => m.startsWith("."));
  } catch (e) { console.error(`sweep-graph: ${path} imports cannot be read: ${e.message}`); process.exit(2); }
}

// resolve a relative specifier from `file` to a repo-relative .ts path among allSrcFiles, best-effort
function resolveSpec(file, spec) {
  const dir = dirname(file);
  const p = spec.replace(/\.js$/, "");
  const joined = p.startsWith(".") ? `${dir}/${p}`.replace(/\/\.\//g, "/") : p;
  const norm = joined.split("/").reduce((acc, seg) => {
    if (seg === "." || seg === "") return acc;
    if (seg === "..") { acc.pop(); return acc; }
    acc.push(seg);
    return acc;
  }, []).join("/");
  return allSrcFiles.find((f) => f === `${norm}.ts` || f === `${norm}.tsx` || f === norm) || null;
}

#!/usr/bin/env node
// scripts/sync-candidate.mjs — copy a candidate from the build tree to the repo,
// verify per file by hash, and REFUSE rather than overwrite when the destination
// holds a line the source does not.
//
// Why this exists (W3-10, W3-11a-R9): rooms build in a sandbox build tree and the
// repo copy was a hand copy nobody re-checked. W3-10 closed with NOT ONE of its 60
// code-level `fixed` rows in the repo — 0 identical, 49 differing, 11 absent, the 11
// being every guard the batch shipped — and the closure ledger booked all 60 `fixed`.
// A note-auditor found it by deriving the population itself. No instrument did.
//
// The refusal is the deliverable, not the copy. A bulk copy that silently wins is the
// same defect with a script wrapped around it. So: snapshot the destination FIRST,
// diff it, and refuse on any repo-only line. There is no --force. The only override
// is --ack <path>=<digest>, which names the exact lines being discarded and goes
// stale the moment the destination changes again.
//
// Usage:
//   node scripts/sync-candidate.mjs --from <build-tree> --to <repo> \
//        (--manifest <file> | --from-report <dir>) [--apply] [--snapshot <dir>]
//        [--ack <path>=<digest>]...
//
// Exit codes: 0 all identical or synced and verified
//             1 at least one file REFUSED (a repo-only line would be lost)
//             2 usage / config / missing source / snapshot failure
//             3 a copied file did not verify by hash afterwards

import { createHash } from "node:crypto";
import {
  readFileSync, writeFileSync, existsSync, mkdirSync, readdirSync, statSync, copyFileSync,
} from "node:fs";
import { join, dirname, resolve, relative, isAbsolute } from "node:path";

const md5 = (buf) => createHash("md5").update(buf).digest("hex");
const sha1short = (s) => createHash("sha1").update(s).digest("hex").slice(0, 12);

function usage(msg) {
  process.stderr.write(`sync-candidate: ${msg}\n\n` +
    "  --from <dir>          source tree (the build tree the rooms worked in)\n" +
    "  --to <dir>            destination tree (the repository)\n" +
    "  --manifest <file>     population: one destination-relative path per line\n" +
    "  --from-report <dir>   population: derived from '## FILES TOUCHED' in REPORT.md files under <dir>\n" +
    "  --apply               actually copy (default is a dry run that still snapshots)\n" +
    "  --snapshot <dir>      where to snapshot the destination before touching it\n" +
    "  --ack <path>=<digest> discard the named repo-only lines; digest must match sha1-12 of them\n");
  process.exit(2);
}

function parseArgs(argv) {
  const a = { acks: new Map(), apply: false };
  for (let i = 0; i < argv.length; i++) {
    const t = argv[i];
    const eat = () => { const v = argv[++i]; if (v === undefined) usage(`${t} needs a value`); return v; };
    if (t === "--from") a.from = eat();
    else if (t === "--to") a.to = eat();
    else if (t === "--manifest") a.manifest = eat();
    else if (t === "--from-report") a.fromReport = eat();
    else if (t === "--snapshot") a.snapshot = eat();
    else if (t === "--apply") a.apply = true;
    else if (t === "--ack") {
      const v = eat();
      const eq = v.lastIndexOf("=");
      if (eq <= 0) usage("--ack takes <path>=<digest>");
      a.acks.set(v.slice(0, eq), v.slice(eq + 1));
    } else usage(`unknown argument ${t}`);
  }
  return a;
}

// ---- population -------------------------------------------------------------
// Derived, never hand-listed inside this file (meta-guard section 1, FAMILY-020).

function fromManifest(file) {
  if (!existsSync(file)) usage(`manifest not found: ${file}`);
  return readFileSync(file, "utf8").split("\n")
    .map((l) => l.replace(/#.*$/, "").trim())
    .filter(Boolean);
}

function walk(dir, out = []) {
  for (const e of readdirSync(dir)) {
    const p = join(dir, e);
    if (statSync(p).isDirectory()) walk(p, out); else out.push(p);
  }
  return out;
}

// Parses the FILES TOUCHED manifest a room is contractually required to write
// (_OUTPUT-CONTRACT.md). Anything that is not a path line is skipped explicitly.
function fromReportDir(dir) {
  if (!existsSync(dir)) usage(`--from-report dir not found: ${dir}`);
  const paths = new Set();
  const skipped = [];
  for (const f of walk(dir)) {
    if (!/REPORT(-[A-Za-z0-9]+)?\.md$/.test(f)) continue;
    const lines = readFileSync(f, "utf8").split("\n");
    let inSection = false;
    for (const raw of lines) {
      if (/^##\s/.test(raw)) { inSection = /^##\s+FILES TOUCHED\s*$/.test(raw); continue; }
      if (!inSection) continue;
      const line = raw.trim();
      if (!line) continue;
      if (/^(base|lesson|lesson-file|candidate|SIBLINGS)\b/i.test(line)) continue;
      if (/^none\b/i.test(line)) continue;
      const m = line.match(/^[-*]?\s*`?([A-Za-z0-9._][A-Za-z0-9._/-]*\.[A-Za-z0-9]+)`?\s*(?:[—:-]|$)/);
      if (m) paths.add(m[1]); else skipped.push(`${relative(dir, f)}: ${line.slice(0, 80)}`);
    }
  }
  // INV-060 posture: a dropped input line is visible, never silent.
  for (const s of skipped) process.stderr.write(`  note: FILES TOUCHED line not parsed as a path — ${s}\n`);
  return [...paths];
}

// ---- the diff that decides ---------------------------------------------------
// A repo-only line is a line the DESTINATION holds that the SOURCE does not.
// Multiset, not set: three copies in the destination against one in the source is
// two repo-only lines, and a set comparison would call that identical.
function repoOnlyLines(srcText, dstText) {
  const have = new Map();
  for (const l of srcText.split("\n")) have.set(l, (have.get(l) ?? 0) + 1);
  const only = [];
  for (const l of dstText.split("\n")) {
    const n = have.get(l) ?? 0;
    if (n > 0) have.set(l, n - 1); else only.push(l);
  }
  return only;
}

const args = parseArgs(process.argv.slice(2));

if (!args.manifest && !args.fromReport) {
  // No default population. A default here would be the hand list the note-auditor
  // refused to take on trust, compiled in.
  usage("give a population: --manifest <file> or --from-report <dir>");
}
const FROM = resolve(args.from), TO = resolve(args.to);
for (const [label, d] of [["--from", FROM], ["--to", TO]]) {
  if (!existsSync(d) || !statSync(d).isDirectory()) usage(`${label} is not a directory: ${d}`);
}
if (FROM === TO) usage("--from and --to are the same tree");

const population = (args.manifest ? fromManifest(args.manifest) : fromReportDir(args.fromReport))
  .map((p) => (isAbsolute(p) ? relative(TO, p) : p))
  .filter((p, i, arr) => arr.indexOf(p) === i)
  .sort();
if (population.length === 0) usage("the population is empty — nothing was derived");

const stamp = new Date().toISOString().replace(/[:.]/g, "-");
const SNAP = resolve(args.snapshot ?? join(FROM, "org/08-control/sync-snapshots", stamp));

// ---- 1. classify, reading only -----------------------------------------------
const rows = [];

  const row = { path: rel, status: "", srcMd5: null, dstMd5: null, repoOnly: 0, repoOnlyNonBlank: 0, ackDigest: null };
  if (!existsSync(src)) { row.status = "missing_source"; rows.push(row); continue; }
  const srcBuf = readFileSync(src);
  row.srcMd5 = md5(srcBuf);
  if (!existsSync(dst)) { row.status = "new"; rows.push(row); continue; }
  const dstBuf = readFileSync(dst);
  row.dstMd5 = md5(dstBuf);
  if (row.srcMd5 === row.dstMd5) { row.status = "identical"; rows.push(row); continue; }
  const only = repoOnlyLines(srcBuf.toString("utf8"), dstBuf.toString("utf8"));
  row.repoOnly = only.length;
  row.repoOnlyNonBlank = only.filter((l) => l.trim() !== "").length;
  if (only.length === 0) { row.status = "safe"; rows.push(row); continue; }
  row.ackDigest = sha1short(only.join("\n"));
  row.sample = only.filter((l) => l.trim() !== "").slice(0, 3);
  row.status = args.acks.get(rel) === row.ackDigest ? "acked" : "refused";
  rows.push(row);
}

// ---- 2. SNAPSHOT THE DESTINATION BEFORE ANY WRITE -----------------------------
// The pre-sync state is the thing that did not exist: the auditor measured repo-only
// lines in 21 of 27 files and then could not certify afterwards that none was unique,
// because nothing had recorded what the destination held before the copy.
let snapshotted = 0;
try {
  mkdirSync(SNAP, { recursive: true });
  for (const r of rows) {
    if (r.dstMd5 === null) continue;
    const out = join(SNAP, "destination", r.path);
    mkdirSync(dirname(out), { recursive: true });
    copyFileSync(join(TO, r.path), out);
    const back = md5(readFileSync(out));
    if (back !== r.dstMd5) throw new Error(`snapshot of ${r.path} does not hash to the destination`);
    snapshotted++;
  }
  writeFileSync(join(SNAP, "SNAPSHOT.json"), JSON.stringify({
    at: new Date().toISOString(), from: FROM, to: TO, apply: args.apply,
    files: rows.map(({ path, status, srcMd5, dstMd5, repoOnly, repoOnlyNonBlank, ackDigest }) =>
      ({ path, status, srcMd5, dstMd5, repoOnly, repoOnlyNonBlank, ackDigest })),
  }, null, 2) + "\n");
} catch (e) {
  process.stderr.write(`sync-candidate: SNAPSHOT FAILED, nothing was copied — ${e.message}\n`);
  process.exit(2);
}

// ---- 3. copy, then VERIFY BY HASH ---------------------------------------------
if (args.apply) {
  for (const r of rows) {
    if (r.status !== "safe" && r.status !== "new" && r.status !== "acked") continue;
    const src = join(FROM, r.path), dst = join(TO, r.path);
    const wasAcked = r.status === "acked";

      mkdirSync(dirname(dst), { recursive: true });
      copyFileSync(src, dst);
      r.status = wasAcked ? "synced_acked" : "synced";
    } catch (e) {

      // need not leave the destination intact: copyFileSync onto a symlinked device

      r.status = "copy_failed";

#!/usr/bin/env node
// scripts/test-scope.mjs — the minimal set of SUITES a change to a file set can redden.
//
//   node scripts/test-scope.mjs --files <a> <b> …   [--root <dir>] [--print list|cmd|json|audit|why]
//   node scripts/test-scope.mjs --why <suite> --files <a>          (which edge put that suite in)
//   node scripts/test-scope.mjs --print opaque [--shard k/N]        (just the OPAQUE roster + causes)
//   node scripts/test-scope.mjs --print audit  [--shard k/N]        (--shard: every N-th suite from
//                                                                    the k-th, so one leg fits a
//                                                                    40 s cap; the N legs partition
//                                                                    the population exactly)
//
// WHY: startup does not cache. ~138 s of CPU before the first assertion, paid in full on every
// invocation, so a room that mutates one function and re-runs its package pays a hundred and
// forty seconds for ten seconds of work. The fix is not fewer runs — mutation testing NEEDS N
// runs — it is making each run name only the suites that could possibly move.
//
// THE ONE INSTRUMENT for "which tests do I run" (QUEUE #462, D-0870(9)). scripts/affected-tests.mjs
// is retired: it had no caller and answered the same question differently, so its body now refuses
// (exit 2) and prints this tool's command. Every scope this tool prints states how many OPAQUE
// suites it added, because those are the price of an answer the graph could not bound.
//
// THE BAR IS SUPERSET, NOT MINIMALITY. An under-inclusive scope silently stops defending
// something; a slow suite only costs money. Every channel below therefore fails toward MORE
// suites, and the one class whose read set cannot be bounded is returned in EVERY scope and
// counted out loud (see OPAQUE).
//
// FOUR EDGE CHANNELS, file -> input:
//   import        `import` / `import type` / `export … from` (barrels) / `import()` with a
//                 literal / `require()` with a literal / `vi.mock|vi.doMock|vi.importActual`.
//                 Resolved with TS extension inference (`./x.js` -> `x.ts`) and by workspace
//                 package name (`@dhwaja/schema`, `@dhwaja/mobile-check/registry-proof`).
//   read-file     a string/template literal in the file that resolves to an existing FILE.
//                 This is the channel that sees a SOURCE-READING test — one that reads a file's
//                 bytes and asserts on them without ever importing it. An import graph cannot
//                 see that edge at all; 254 of this tree's 853 collected suites take one.
//   read-dir      a literal that resolves to an existing DIRECTORY: the file may read ANYTHING
//                 under it (the `readdirSync(SRC)` + `readFileSync(`${SRC}${f}`)` idiom), so the
//                 edge is a prefix and every file beneath it is an input.
//   read-closure  a read-file edge that lands on a .ts/.tsx/.mjs/.js file also drags that file's
//                 OWN import closure in: a guard that shells out to `node scripts/close-run.mjs`
//                 reddens when a module close-run imports moves, and the literal alone misses it.
//
// OPAQUE — the class this tool CANNOT bound, and does not pretend to. A suite that performs a
// filesystem read or spawns a subprocess whose path strings can come from outside its own
// literals has an unbounded input set. It is returned in EVERY scope, unconditionally, and its
// roster is printed by `--print audit`. That is the honest answer, not a dropped edge.
//
// LOUD, NOT SILENT (FAMILY-012 / INV-060): an unresolvable relative import, an unparseable file
// and a --files argument that is not in the tree are each reported on stderr; --strict makes

/** Directories never walked. DERIVED, not narrated: the caller passes the set vitest itself
 *  excludes (see excludeDirsFromVitestConfig). The hardcoded floor is only what a walk needs to
 *  terminate at all. */
const HARD_SKIP = new Set(["node_modules", ".git"]);

/** Read `vitest.config.ts`'s own `exclude` array plus `configDefaults.exclude`, and reduce it to
 *  the set of directory NAMES a walk must skip. FAMILY-013: the literal is derived from the
 *  config that states the scope, never re-typed here, so the two cannot drift. Patterns this
 *  reducer does not understand are returned in `unreduced` and printed by --print audit. */
export function excludeDirsFromVitestConfig(root, defaults = []) {
  const dirs = new Set(HARD_SKIP);
  const unreduced = [];
  const pats = [...defaults];
  const cfg = join(root, "vitest.config.ts");
  if (existsSync(cfg)) {
    const sf = parse(cfg);
    walk(sf, (n) => {
      if (ts.isPropertyAssignment(n) && ts.isIdentifier(n.name) && n.name.text === "exclude" && ts.isArrayLiteralExpression(n.initializer)) {
        for (const el of n.initializer.elements) for (const l of literalsIn(el)) pats.push(l);
      }
      return undefined;
    });
  }
  for (const p of pats) {
    // `**/x/**`, `x/**`, `**/.{a,b}/**` -> the directory name(s); anything else is unreduced.
    const m = /^(?:\*\*\/)?([^*/]+)\/\*\*$/.exec(p);
    if (m) {
      const brace = /^\.\{([^}]+)\}$/.exec(m[1]);
      if (brace) for (const part of brace[1].split(",")) dirs.add("." + part.trim());
      else dirs.add(m[1]);
      continue;
    }
    if (/^\*\*\/[^*/]+\.[^*/]*\*$/.test(p) || /config\.\*$/.test(p)) continue; // file patterns, not dirs
    unreduced.push(p);
  }
  return { dirs, unreduced };
}

/** `setupFiles` from vitest.config.ts, resolved. THESE LOAD FOR EVERY SUITE, so they — and the
 *  config that names them — are inputs of every suite in the population. An import graph rooted
 *  at the test file cannot see them: nothing imports `test-setup.ts`, the runner injects it.
 *  Derived from the config's own array, never re-typed (FAMILY-013). */
export function setupFilesFromVitestConfig(root) {
  const cfg = join(root, "vitest.config.ts");
  const out = [];
  if (!existsSync(cfg)) return out;
  const sf = parse(cfg);
  walk(sf, (n) => {
    if (ts.isPropertyAssignment(n) && ts.isIdentifier(n.name) && n.name.text === "setupFiles" && ts.isArrayLiteralExpression(n.initializer)) {
      for (const el of n.initializer.elements) for (const l of literalsIn(el)) {
        const r = relative(root, resolve(root, l.replace(/^\.\//, ""))).split("\\").join("/");
        if (relInside(r) && existsSync(join(root, r))) out.push(r);
      }
    }
    return undefined;
  });
  return out;
}

/** Every file in the tree, repo-relative, POSIX separators. */
export function indexTree(root, skipDirs) {
  const files = [];
  const walkDir = (rel) => {
    let names;
    try { names = readdirSync(rel ? join(root, rel) : root); } catch { return; }
    for (const n of names) {
      if (skipDirs.has(n)) continue;
      const r = rel ? `${rel}/${n}` : n;
      let st;
      try { st = statSync(join(root, r)); } catch { continue; }
      if (st.isDirectory()) walkDir(r);
      else files.push(r);
    }
  };
  walkDir("");
  return files;
}

/** The suite population, taken from the RUNNER rather than narrated. `vitest list --filesOnly`
 *  is authoritative: it applies the real include/exclude, so a suite this tool names is a suite
 *  `vitest run` would collect. Falls back to a glob walk — loudly — where vitest cannot run
 *  (a fixture tree with no vitest installed), because a silent fallback is how a population
 *  goes wrong without anyone noticing. */
export function suitePopulation(root, treeFiles, warn) {
  const bin = join(root, "node_modules", ".bin", "vitest");
  if (existsSync(bin)) {
    try {
      const out = execFileSync(bin, ["list", "--filesOnly"], { cwd: root, encoding: "utf8", maxBuffer: 64 * 1024 * 1024, stdio: ["ignore", "pipe", "pipe"] });
      const list = out.split("\n").map((l) => l.trim()).filter((l) => /\.(test|spec)\.[cm]?[jt]sx?$/.test(l)).map((l) => l.replace(/^\.\//, ""));
      if (list.length) return { suites: new Set(list), source: "vitest list --filesOnly" };
      warn(`vitest list returned no suites; falling back to a glob walk`);
    } catch (e) {
      warn(`vitest list failed (${String(e.message).split("\n")[0]}); falling back to a glob walk`);
    }
  } else {
    warn(`no ${relative(root, bin)}; falling back to a glob walk for the suite population`);
  }
  return { suites: new Set(treeFiles.filter((f) => /\.(test|spec)\.[cm]?[jt]sx?$/.test(f))), source: "glob walk (FALLBACK)" };
}

/** Workspace packages by `name`, read from every package.json the walk found. Derived from the
 *  tree, never a hand list (FAMILY-020: a hand-listed population fails in the direction of the
 *  claim it supports). */
export function workspacePackages(root, treeFiles) {
  const map = new Map();
  for (const f of treeFiles) {
    if (!f.endsWith("package.json") || f === "package.json") continue;
    const dir = dirname(f);
    let pkg;
    try { pkg = JSON.parse(readFileSync(join(root, f), "utf8")); } catch { continue; }
    if (pkg && typeof pkg.name === "string") map.set(pkg.name, { dir, main: pkg.main, exports: pkg.exports });
  }
  return map;
}

const inTree = (root, abs) => {
  const r = relative(root, abs).split("\\").join("/");
  return relInside(r) ? r : null;
};

/** One stat for a path under a skipped directory; extension inference only for an extensionless name. */
function existsOutside(abs, r, filesOnly) {
  let st = null;
  try { st = statSync(abs); } catch { /* absent */ }
  if (st) return st.isFile() || (!filesOnly && st.isDirectory());
  return !/\.[^./]+$/.test(r) && SRC_EXT.some((x) => { try { return statSync(abs + x).isFile(); } catch { return false; } });
}
const statIsFile = (root) => (r) => { try { return statSync(join(root, r)).isFile(); } catch { return false; } };
/** TS/bundler extension inference. `./x` -> x.ts|x.tsx|…|x/index.ts; `./x.js` -> x.ts first
 *  (verbatimModuleSyntax writes the .js the emit would have, and the file on disk is .ts).
 *  `has(rel)` answers "is this a file". createGraph passes the in-memory tree index (I28-TESTSCOPE
 *  S1: statSync here was 96 of 108 s of a 2-file query on the mounted tree); a file outside the
 *  index has no edges and is never a --files target, so answering from it drops no scope. */
function tryFile(relPath, has) {
  const p = relPath.split("\\").join("/");
  if (has(p)) return p;
  const m = /^(.*)\.(js|jsx|mjs|cjs)$/.exec(p);
  if (m) for (const e of [".ts", ".tsx", ".mts", ".cts"]) if (has(m[1] + e)) return m[1] + e;
  for (const e of SRC_EXT) if (has(p + e)) return p + e;
  for (const e of SRC_EXT) if (has(`${p}/index${e}`)) return `${p}/index${e}`;
  return null;
}

/** A module specifier as written in `fromRel` -> a repo-relative path, or null for external. */
export function resolveSpecifier(root, fromRel, spec, pkgs, has = statIsFile(root)) {
  if (!spec || spec.includes("?")) return null;                       // a template hole: unresolvable
  if (spec.startsWith("node:") || spec.startsWith("data:") || spec.startsWith("http")) return null;
  if (spec.startsWith(".")) {
    const rel = inTree(root, resolve(root, dirname(fromRel), spec));
    return rel ? tryFile(rel, has) : null;
  }
  if (spec.startsWith("/")) return null;
  // workspace package, longest name first (`@a/b` before `@a`)
  for (const name of [...pkgs.keys()].sort((a, b) => b.length - a.length)) {
    if (spec !== name && !spec.startsWith(name + "/")) continue;
    const { dir, main, exports: exp } = pkgs.get(name);
    const sub = spec === name ? "." : "." + spec.slice(name.length);
    if (exp && typeof exp === "object") {
      const e = exp[sub];
      const target = typeof e === "string" ? e : e && typeof e === "object" ? (e.import ?? e.default ?? e.require) : null;
      if (typeof target === "string") return tryFile(join(dir, target).split("\\").join("/"), has);
    }
    if (sub === ".") {
      if (typeof exp === "string") return tryFile(join(dir, exp).split("\\").join("/"), has);
      if (typeof main === "string") return tryFile(join(dir, main).split("\\").join("/"), has);
      return tryFile(`${dir}/src/index`, has) ?? tryFile(`${dir}/index`, has);
    }
    return tryFile(`${dir}/${sub.slice(2)}`, has) ?? tryFile(`${dir}/src/${sub.slice(2)}`, has);
  }
  return null;                                                        // node_modules dependency
}

const literalText = (x) => (x && (ts.isStringLiteral(x) || ts.isNoSubstitutionTemplateLiteral(x)) ? x.text : null);
/** An identifier `process` in a REFERENCE position: not a declared name, a member name, an import binding or a type. */
const isProcessIdentifierRef = (x) => isNameRef(x, "process");
/** An identifier spelled `name` in a REFERENCE position (I64-TOOLS G1: shared by `process` and `eval`). */
function isNameRef(x, name) {
  if (!ts.isIdentifier(x) || x.text !== name) return false;
  const p = x.parent;
  if (!p) return false;
  if ((ts.isPropertyAccessExpression(p) || ts.isPropertyAssignment(p) || ts.isPropertyDeclaration(p) || ts.isPropertySignature(p) || ts.isMethodDeclaration(p)
    || ts.isMethodSignature(p) || ts.isGetAccessorDeclaration(p) || ts.isSetAccessorDeclaration(p) || ts.isEnumMember(p) || ts.isVariableDeclaration(p)
    || ts.isParameter(p) || ts.isFunctionDeclaration(p) || ts.isFunctionExpression(p) || ts.isClassDeclaration(p) || ts.isClassExpression(p)) && p.name === x) return false;
  if (ts.isBindingElement(p) && (p.name === x || p.propertyName === x)) return false;
  if (ts.isImportClause(p) || ts.isImportSpecifier(p) || ts.isNamespaceImport(p) || ts.isExportSpecifier(p) || ts.isImportEqualsDeclaration(p)) return false;
  if (ts.isQualifiedName(p) || ts.isTypeReferenceNode(p) || ts.isTypeQueryNode(p) || ts.isLabeledStatement(p) || ts.isBreakOrContinueStatement(p)) return false;
  return true;
}
/** `process`, or `globalThis.process` / `global.process` / `globalThis["process"]` — read the way `process` is (D-1528). */
const GLOBAL_OBJECTS = new Set(["globalThis", "global"]);
// I64-TOOLS G1 (D-1556 (b)): the WIDE view, set only while callShape's second pass walks a file (callShape below). It adds
// `self`, every alias of a global object (`const g = globalThis`), const-resolved keys and createRequire aliases. null = narrow view.
let WIDE = null;
const SYMBOL_KEY = Symbol("symbol-key");
/** A key's known value: its literal text, SYMBOL_KEY, or undefined (unresolved). Only the wide view resolves a const or `Symbol.x`
 *  (the narrow view staying narrow keeps every narrow reason as it was). */
const keyValue = (k) => {
  const y = skipWrap(k);
  const lit = literalText(y);
  if (lit !== null) return lit;
  if (ts.isNumericLiteral(y)) return y.text;
  if (WIDE === null) return undefined;
  if (ts.isIdentifier(y)) return constValue(y);
  if (ts.isPropertyAccessExpression(y) && ts.isIdentifier(y.expression) && y.expression.text === "Symbol") return SYMBOL_KEY;
  return undefined;
};
// A key value that may itself reach process (or the global object, Function or eval) never resolves a for-of key.
const KEY_UNSAFE = new Set(["process", "Function", "eval", "globalThis", "global", "self"]);
const bindsName = (b, t) => (ts.isIdentifier(b) ? b.text === t : b.elements.some((el) => !ts.isOmittedExpression(el) && bindsName(el.name, t)));
/** The value of one `const` declaration: a literal, SYMBOL_KEY (`Symbol(..)`/`Symbol.for(..)`), a for-of over string literals none of
 *  which is in KEY_UNSAFE (its first), or undefined. */
function declValue(d) {
  if (!ts.isIdentifier(d.name) || !(ts.getCombinedNodeFlags(d) & ts.NodeFlags.Const)) return undefined;
  const loop = d.parent?.parent;
  if (loop && ts.isForOfStatement(loop)) {
    const arr = skipWrap(loop.expression);
    if (!ts.isArrayLiteralExpression(arr)) return undefined;
    const vals = arr.elements.map((el) => literalText(el));
    return vals.some((v) => v === null || KEY_UNSAFE.has(v)) ? undefined : vals[0];
  }
  if (!d.initializer) return undefined;
  const init = skipWrap(d.initializer);
  const lit = literalText(init);
  if (lit !== null) return lit;
  if (ts.isNumericLiteral(init)) return init.text;
  if (ts.isCallExpression(init)) {
    const c = skipWrap(init.expression);
    if ((ts.isIdentifier(c) && c.text === "Symbol") || (ts.isPropertyAccessExpression(c) && c.name.text === "for" && ts.isIdentifier(c.expression) && c.expression.text === "Symbol")) return SYMBOL_KEY;
  }
  return undefined;
}
/** The nearest lexical declaration of identifier `id` (for-head, block, file, parameter), resolved by declValue; undefined otherwise. */
function constValue(id) {
  for (let at = id.parent; at; at = at.parent) {
    let decls = [];
    if ((ts.isForOfStatement(at) || ts.isForInStatement(at) || ts.isForStatement(at)) && at.initializer && ts.isVariableDeclarationList(at.initializer)) decls = at.initializer.declarations;
    else if (ts.isBlock(at) || ts.isSourceFile(at) || ts.isModuleBlock(at) || ts.isCaseClause(at) || ts.isDefaultClause(at)) decls = at.statements.filter(ts.isVariableStatement).flatMap((st) => st.declarationList.declarations);
    else if (ts.isFunctionLike(at) && at.parameters.some((q) => bindsName(q.name, id.text))) return undefined;
    else if (ts.isCatchClause(at) && at.variableDeclaration && bindsName(at.variableDeclaration.name, id.text)) return undefined;
    const hit = decls.filter((d) => bindsName(d.name, id.text));
    if (hit.length > 0) return hit.length === 1 ? declValue(hit[0]) : undefined;
  }
  return undefined;
}
const isGlobalName = (t) => GLOBAL_OBJECTS.has(t) || (WIDE !== null && t === "self");
// I65-TOOLS T1: arguments a call cannot be read from (a spread, a non-literal `.apply` array).
const UNKNOWN_ARGS = Symbol("unknown-args");
/** `Reflect.get(o, k)` -> [o, k], also `Reflect["get"]`, a destructured or aliased get and `.call`/`.apply` (I65-TOOLS T1);
 *  UNKNOWN_ARGS when its arguments cannot be read; null when x is not a Reflect.get call. */
function reflectGetArgs(x) {
  if (!ts.isCallExpression(x)) return null;
  const t = callTarget(x);
  if (!memberOf(t.fn).some(([b, k]) => k === "get" && b !== null && isReflectRef(b))) return null;
  if (t.args === null) return UNKNOWN_ARGS;
  return t.args.length >= 2 ? [t.args[0], t.args[1]] : null;
}
/** `Reflect`; in the wide view also a renamed Reflect (`const R = Reflect`, I64-FX-TOOLS) and `globalThis.Reflect`. */
const isReflectRef = (x) => {
  const y = skipWrap(x);
  if (ts.isIdentifier(y) && (y.text === "Reflect" || WIDE?.reflects.has(y.text) === true)) return true;
  return WIDE !== null && isGlobalMember(y, "Reflect");
};
/** I63-TOOLS U3: parens, a type wrapper (`as`, `!`, `<T>x`, `satisfies`) and a comma's right operand do not change the value. */
const skipWrap = (x) => {
  for (;;) {
    if (ts.isParenthesizedExpression(x) || ts.isAsExpression(x) || ts.isNonNullExpression(x) || ts.isTypeAssertionExpression(x) || ts.isSatisfiesExpression(x)) x = x.expression;
    else if (ts.isBinaryExpression(x) && x.operatorToken.kind === ts.SyntaxKind.CommaToken) x = x.right;
    else return x;
  }
};
/** I63-TOOLS U3: `globalThis`, `(globalThis)`, `(0, globalThis)`, `globalThis.globalThis`, `global["globalThis"]`, … */
const isGlobalObjectRef = (x) => {
  const y = skipWrap(x);
  if (ts.isIdentifier(y)) return GLOBAL_OBJECTS.has(y.text) || (WIDE !== null && (y.text === "self" || WIDE.aliases.has(y.text)));
  if (ts.isPropertyAccessExpression(y)) return isGlobalName(y.name.text) && isGlobalObjectRef(y.expression);
  if (ts.isElementAccessExpression(y)) return isGlobalName(keyValue(y.argumentExpression)) && isGlobalObjectRef(y.expression);
  const ra = WIDE === null ? null : reflectGetArgs(y);
  if (ra !== null && ra !== UNKNOWN_ARGS) return isGlobalName(keyValue(ra[1])) && isGlobalObjectRef(ra[0]);
  // I64-FX-TOOLS (D-1556 (b)): either branch of `c ? a : b`, either operand of `||`, `??`, `&&` may be the value.
  if (WIDE !== null && ts.isConditionalExpression(y)) return isGlobalObjectRef(y.whenTrue) || isGlobalObjectRef(y.whenFalse);
  if (WIDE !== null && ts.isBinaryExpression(y) && LOGICAL_OPS.has(y.operatorToken.kind)) return isGlobalObjectRef(y.left) || isGlobalObjectRef(y.right);
  return false;
};
const LOGICAL_OPS = new Set([ts.SyntaxKind.BarBarToken, ts.SyntaxKind.QuestionQuestionToken, ts.SyntaxKind.AmpersandAmpersandToken]);
/** I64-TOOLS G1 (wide view): a global object indexed by a key not resolved to a non-process value, or Reflect.get of one. */
const isWideProcessRef = (x) => {
  if (ts.isElementAccessExpression(x) && isGlobalObjectRef(x.expression)) { const v = keyValue(x.argumentExpression); return v === undefined || v === "process"; }
  const ra = reflectGetArgs(x);
  if (ra !== null && ra !== UNKNOWN_ARGS && isGlobalObjectRef(ra[0])) { const v = keyValue(ra[1]); return v === undefined || v === "process"; }
  return false;
};
export const isProcessRef = (x) => isProcessIdentifierRef(x)
  || (ts.isPropertyAccessExpression(x) && isGlobalObjectRef(x.expression) && x.name.text === "process")
  || (ts.isElementAccessExpression(x) && isGlobalObjectRef(x.expression) && literalText(x.argumentExpression) === "process")
  || (WIDE !== null && isWideProcessRef(x));
/** A key bound from process by destructuring or a named node:process import: a runner reason, an opaque reason, "env", or null (bounded). */
function processKeyClass(key, verb) {
  if (key === null) return `${verb} process by a computed key or rest element, which this scan cannot follow`;
  if (RUNNER_PROCESS_METHODS.has(key)) return `${verb} process.${key}, which runs another program`;
  if (key === "env") return "env";
  if (INERT_PROCESS_METHODS.has(key) || INERT_PROCESS_PROPERTIES.has(key)) return null;
  return `${verb} process.${key}, a name not on the known-inert lists`;
}

/** Every module specifier `fromRel` imports, at any nesting: static import, `export … from`
 *  (the barrel form — `importsOf` in lib/ts-scan.mjs sees only the first), dynamic `import()`,
 *  `require()`, and the vitest mock family, whose literal is a real module edge. */
export function specifiersOf(sf) {
  const out = [];
  walk(sf, (n) => {
    if ((ts.isImportDeclaration(n) || ts.isExportDeclaration(n)) && n.moduleSpecifier && ts.isStringLiteral(n.moduleSpecifier)) out.push(n.moduleSpecifier.text);
    else if (ts.isImportEqualsDeclaration(n) && ts.isExternalModuleReference(n.moduleReference) && ts.isStringLiteral(n.moduleReference.expression)) out.push(n.moduleReference.expression.text);
    else if (ts.isCallExpression(n)) {
      const callee = n.expression;
      const name = ts.isIdentifier(callee) ? callee.text : ts.isPropertyAccessExpression(callee) && ts.isIdentifier(callee.name) ? callee.name.text : callee.kind === ts.SyntaxKind.ImportKeyword ? "import" : null;
      if ((name === "import" || name === "require" || name === "mock" || name === "doMock" || name === "importActual" || name === "importMock") && n.arguments.length) {
        const a = n.arguments[0];
        if (ts.isStringLiteral(a) || ts.isNoSubstitutionTemplateLiteral(a)) out.push(a.text);
      }
    }
    return undefined;
  });
  return out;
}

/** The call-shape facts opacity turns on: does the file read the filesystem, enumerate a
 *  directory, spawn a subprocess, or take a NAMED process.env value (NON_TREE_ENV excepted).
 *
 *  GATED ON THE IMPORT, and measured: `exec` and `fork` are ordinary method names —
 *  `/re/.exec(text)` is everywhere in this tree — so matching the bare call name marked 715 of
 *  847 suites opaque on the first run, which is a tool that answers "run everything" in 44
 *  seconds. A file that never imports `node:child_process` cannot spawn one directly; a file
 *  that reaches the filesystem through a HELPER is still caught, because opacity propagates
 *  along the import closure (see inputsOf). */
/** `process.env` or `process["env"]` (QUEUE #461: the bracket spelling was not seen as env at all). */
const isProcessEnv = (x) => (ts.isPropertyAccessExpression(x) && isProcessRef(skipParens(x.expression)) && ts.isIdentifier(x.name) && x.name.text === "env")
  || (ts.isElementAccessExpression(x) && isProcessRef(skipParens(x.expression)) && ts.isStringLiteral(x.argumentExpression) && x.argumentExpression.text === "env");

const REFLECTIVE_CALLS = new Set(["call", "apply", "bind"]);
// I64-TOOLS G1 (D-1556 (b), the narrow reading of D-1538 ruling 4): reasons of the wide pass.
export const WIDE_REASONS = {
  globalComputed: "reads the global object by a computed key, which may reach process and this scan cannot follow",
  reflect: "reads process from the global object through Reflect.get, which this scan cannot follow",
  fn: "builds code from a string with Function(), which this scan cannot read",
  evalIndirect: "calls eval indirectly, which runs a string this scan cannot read",
  createRequire: "loads a module through createRequire by a computed specifier, which this scan cannot follow",
  evalDirect: "calls eval directly, which runs a string this scan cannot read",
  vmRun: "runs a string through node:vm, which this scan cannot read",
  dataImport: "imports a data: URL module, which runs a string this scan cannot read",
  globalReader: "reads the global object's values in bulk or by descriptor, which may reach process and this scan cannot follow",

// I64-FX-TOOLS (D-1556 (b), FIX-ROUND-1 ruling 5): node:vm calls that run a string (`vm.runInThisContext(s)`, `script.runInContext(c)`).
const VM_RUNS = new Set(["runInThisContext", "runInNewContext", "runInContext", "compileFunction"]);
// Object readers that hand back the global object's values (`Object.values(globalThis)`); getOwnPropertyDescriptor is keyed.
const OBJECT_READERS = new Set(["values", "entries", "getOwnPropertyDescriptors"]);
/** The key text of a property name or binding name: an identifier or literal; a computed name via keyValue; null for none. */
const propKey = (pn) => (pn === undefined ? null : ts.isIdentifier(pn) ? pn.text : ts.isComputedPropertyName(pn) ? keyValue(pn.expression) : keyValue(pn));
// I65-TOOLS T1 (D-1568, PLAN ruling 7): a member read, as [base, key] pairs: `b.k`, `b[k]` (keyValue), and in the wide view an
// identifier recorded in WIDE.members (`const r = vm.runInThisContext`, `const { get } = Reflect`, `import { createRequire as cr }`),
// followed through alias chains. A named import's base is null.
function memberOf(c, seen = new Set()) {
  const y = skipWrap(c);
  if (ts.isPropertyAccessExpression(y)) return [[y.expression, y.name.text]];
  if (ts.isElementAccessExpression(y)) return [[y.expression, keyValue(y.argumentExpression)]];
  if (WIDE === null || !ts.isIdentifier(y) || seen.has(y.text)) return [];
  seen.add(y.text);
  return (WIDE.members.get(y.text) ?? []).flatMap((m) => ("init" in m ? memberOf(m.init, seen) : [[m.base, typeof m.key === "string" ? m.key : propKey(m.key)]]));
}
/** I65-TOOLS T1: a call's real callee and arguments: `.call` (args after this), `.apply` (an array literal's elements),
 *  `f.bind(t, ..)(..)` and `Reflect.apply(f, t, [..])` are taken off. args null = unreadable (a spread, a non-literal array).
 *  I65-FX-TOOLS (ruling 9): `recv` is the receiver a step passed (`t`); a `.call`/`.apply`/`.bind` reached with one runs
 *  on it when useRecv (`Function.prototype.call.call(Function, ..)` runs Function), and a bare `f.bind(t, ..)` is f with args null. */
function callTarget(n, useRecv = false) {
  if (ts.isNewExpression(n)) return { fn: skipWrap(n.expression), args: n.arguments ? [...n.arguments] : [], recv: undefined };
  const listed = (a) => {
    if (a === undefined) return [];
    const y = skipWrap(a);
    return ts.isArrayLiteralExpression(y) && !y.elements.some(ts.isSpreadElement) ? [...y.elements] : null;
  };
  let fn = skipWrap(n.expression);
  // A direct call keeps its arguments as written (a spread stays an element, as before I65); only the forms below lose them.
  let args = [...n.arguments];
  let recv;
  const spread = (xs) => xs.some(ts.isSpreadElement);
  for (;;) {
    const k = callKey(fn);
    const self = recv;
    if (k === "apply" && isReflectRef(fn.expression)) {
      if (args === null || args[0] === undefined) return { fn, args: null, recv };
      fn = skipWrap(args[0]);
      recv = args[1];
      args = spread(args) ? null : listed(args[2]);
      continue;
    }
    if (k === "call" || k === "apply" || k === "bind") {
      fn = skipWrap((useRecv ? self : undefined) ?? fn.expression);
      recv = args?.[0];
      args = args === null || spread(args) || k === "bind" ? null : k === "call" ? args.slice(1) : listed(args[1]);
    } else if (ts.isCallExpression(fn) && callKey(fn.expression) === "bind") {
      const bound = fn;
      fn = skipWrap(skipWrap(bound.expression).expression);
      args = args === null || spread(bound.arguments) ? null : [...bound.arguments.slice(1), ...args];
    } else return { fn, args, recv };
  }
}
const callKey = (x) => { const y = skipWrap(x); return ts.isPropertyAccessExpression(y) ? y.name.text : ts.isElementAccessExpression(y) ? keyValue(y.argumentExpression) : null; };
/** A member `m` of the global object (`globalThis.m`, `global["m"]`, `const { m: x } = globalThis`, `Reflect.get(globalThis, "m")`). */
function isGlobalMember(c, m) {
  if (memberOf(c).some(([b, k]) => k === m && b !== null && isGlobalObjectRef(b))) return true;
  const ra = reflectGetArgs(skipWrap(c));
  return ra !== null && ra !== UNKNOWN_ARGS && keyValue(ra[1]) === m && isGlobalObjectRef(ra[0]);
}
/** Function: `Function`, a wide alias (`const F = Function`), the global's Function, and any `.constructor` (I65-TOOLS T1:
 *  `(() => 1).constructor`, `[].constructor.constructor`, `Function.prototype.constructor`, the async-function constructor). */
/** I65-TOOLS T1: a call or construction of a Function ref builds code, except a `.constructor` reached with no argument, which
 *  builds an empty function at most (`new (app.constructor)()` makes an instance of app's class). */
const buildsCode = (fn, args) => isFunctionRef(fn) && (args === null || args.length > 0 || !memberOf(fn).some(([, k]) => k === "constructor"));
function isFunctionRef(c) {
  const y = skipWrap(c);
  if (ts.isIdentifier(y) && (y.text === "Function" || WIDE?.functions.has(y.text) === true)) return true;
  return memberOf(y).some(([, k]) => k === "constructor") || isGlobalMember(y, "Function");
}
/** Object: `Object`, a wide alias (`const O = Object`) and the global's Object (`globalThis.Object`). */
function isObjectRef(c) {
  const y = skipWrap(c);
  if (ts.isIdentifier(y) && (y.text === "Object" || WIDE?.objects.has(y.text) === true)) return true;
  return isGlobalMember(y, "Object");
}
const isCreateRequireName = (c) => {
  const y = skipWrap(c);
  if (ts.isIdentifier(y) && y.text === "createRequire") return true;
  return memberOf(y).some(([, k]) => k === "createRequire");
};
// I65-TOOLS T1 (JA-02): a data: URL module is source text in the specifier; the URL parser drops leading C0 and space, and the
// scheme is case-insensitive.
const isDataUrl = (spec) => /^[\x00-\x20]*data:/i.test(String(spec));
/** The wide view of one file (keys resolve lexically, constValue): global aliases (to a fixpoint) and createRequire aliases (a `require` alias is the import edge's). */
function wideView(sf) {
  const view = { aliases: new Set(), requires: new Set(), reflects: new Set(), functions: new Set(), objects: new Set(), members: new Map() };
  const member = (name, m) => { if (!view.members.has(name)) view.members.set(name, []); view.members.get(name).push(m); };
  const pairs = [];
  // I64-FX-TOOLS: [name, key, source] for `const { globalThis: g } = <src>` and `({ globalThis: g } = <src>)`.
  const keyed = [];
  walk(sf, (n) => {
    // I64-FX-TOOLS: a default (`function f(g = globalThis)`, `const { g = globalThis } = o`) binds its initializer too.
    if ((ts.isParameter(n) || ts.isBindingElement(n)) && ts.isIdentifier(n.name) && n.initializer) pairs.push([n.name.text, n.initializer]);
    // I65-TOOLS T1 (R5-G1): the key NODE is kept and resolved in the fixpoint below, after `WIDE = view`, so a const key resolves.
    if (ts.isBindingElement(n) && ts.isIdentifier(n.name) && !n.dotDotDotToken && ts.isObjectBindingPattern(n.parent)
      && ts.isVariableDeclaration(n.parent.parent) && n.parent.parent.initializer) {
      keyed.push([n.name.text, n.propertyName, n.parent.parent.initializer]);
      member(n.name.text, { base: n.parent.parent.initializer, key: n.propertyName ?? n.name });
    }
    if (ts.isBinaryExpression(n) && n.operatorToken.kind === ts.SyntaxKind.EqualsToken && ts.isObjectLiteralExpression(skipWrap(n.left))) {
      for (const pr of skipWrap(n.left).properties) {
        if (ts.isPropertyAssignment(pr) && ts.isIdentifier(skipWrap(pr.initializer))) {
          keyed.push([skipWrap(pr.initializer).text, pr.name, n.right]);
          member(skipWrap(pr.initializer).text, { base: n.right, key: pr.name });
        }
        if (ts.isShorthandPropertyAssignment(pr)) member(pr.name.text, { base: n.right, key: pr.name });
      }
    }
    if (ts.isImportSpecifier(n)) member(n.name.text, { base: null, key: (n.propertyName ?? n.name).text });
    if (ts.isVariableDeclaration(n) && ts.isIdentifier(n.name)) {
      const name = n.name.text;
      const init = n.initializer ? skipWrap(n.initializer) : null;
      if (init) pairs.push([name, init]);
    }
    if (ts.isBinaryExpression(n) && n.operatorToken.kind === ts.SyntaxKind.EqualsToken && ts.isIdentifier(n.left)) pairs.push([n.left.text, n.right]);
    return undefined;
  });
  // I65-TOOLS T1: `const r = vm.runInThisContext`, `const g = r` are member aliases memberOf follows.
  for (const [name, init] of pairs) {
    const y = skipWrap(init);
    if (ts.isIdentifier(y) || ts.isPropertyAccessExpression(y) || ts.isElementAccessExpression(y)) member(name, { init: y });
  }
  WIDE = view;
  for (let grew = true; grew;) {
    grew = false;
    for (const [name, init] of pairs) if (!view.aliases.has(name) && isGlobalObjectRef(init)) { view.aliases.add(name); grew = true; }
    for (const [name, key, src] of keyed) if (!view.aliases.has(name) && isGlobalName(propKey(key)) && isGlobalObjectRef(src)) { view.aliases.add(name); grew = true; }
    for (const [name, init] of pairs) if (!view.reflects.has(name) && isReflectRef(init)) { view.reflects.add(name); grew = true; }
    // I65-TOOLS T1: aliases of Function and Object (a createRequire alias is a member alias); a createRequire result (by
    // `.call`/`.apply` too) is a require.
    for (const [name, init] of pairs) {
      if (!view.functions.has(name) && isFunctionRef(init)) { view.functions.add(name); grew = true; }
      if (!view.objects.has(name) && isObjectRef(init)) { view.objects.add(name); grew = true; }
      if (!view.requires.has(name) && name !== "require" && ts.isCallExpression(init) && isCreateRequireName(callTarget(init).fn)) { view.requires.add(name); grew = true; }
    }
  }
  return view;
}
/** I64-TOOLS G1: the wide pass's own arms, checked first on each node so their reason is the node's. null = none. */
function wideReason(n) {
  if (ts.isElementAccessExpression(n) && isGlobalObjectRef(n.expression) && keyValue(n.argumentExpression) === undefined) return WIDE_REASONS.globalComputed;
  const ra = reflectGetArgs(n);
  if (ra === UNKNOWN_ARGS) return WIDE_REASONS.reflect;
  if (ra !== null && isGlobalObjectRef(ra[0])) { const v = keyValue(ra[1]); if (v === undefined || v === "process") return WIDE_REASONS.reflect; }
  // I65-TOOLS T1 (JA-02): `import("data:..")`, `import x from "data:.."`, `export * from "data:.."`.
  if (ts.isCallExpression(n) && n.expression.kind === ts.SyntaxKind.ImportKeyword && isDataUrl(literalText(n.arguments[0]))) return WIDE_REASONS.dataImport;
  if ((ts.isImportDeclaration(n) || ts.isExportDeclaration(n)) && isDataUrl(literalText(n.moduleSpecifier))) return WIDE_REASONS.dataImport;
  // A destructuring of the global object (`const { [p]: q } = globalThis`, `({ ...r } = globalThis)`): a computed key that does not
  // resolve, or a rest element, may take process; the key process under another name aliases it.
  const pattern = ts.isVariableDeclaration(n) && n.initializer && ts.isObjectBindingPattern(n.name) && isGlobalObjectRef(n.initializer)
    ? n.name.elements.map((el) => [Boolean(el.dotDotDotToken), el.propertyName])
    : ts.isBinaryExpression(n) && n.operatorToken.kind === ts.SyntaxKind.EqualsToken && ts.isObjectLiteralExpression(skipWrap(n.left)) && isGlobalObjectRef(n.right)
      ? skipWrap(n.left).properties.map((pr) => [ts.isSpreadAssignment(pr), ts.isPropertyAssignment(pr) ? pr.name : undefined])
      : [];
  for (const [rest, pn] of pattern) {
    const key = rest ? undefined : pn === undefined ? null : ts.isIdentifier(pn) ? pn.text : ts.isComputedPropertyName(pn) ? keyValue(pn.expression) : keyValue(pn);
    if (key === undefined) return WIDE_REASONS.globalComputed;
    if (key === "process") return "uses process as a value (an alias, an argument or a return), which this scan cannot follow";
  }
  // I64-FX-TOOLS: the callee with `.call`/`.apply`/`.bind` taken off (`Function.call(null, s)`), and a tagged template's tag.
  const unreflect = (x) => {
    let c = skipWrap(x);
    while ((ts.isPropertyAccessExpression(c) && REFLECTIVE_CALLS.has(c.name.text)) || (ts.isElementAccessExpression(c) && REFLECTIVE_CALLS.has(literalText(c.argumentExpression)))) c = skipWrap(c.expression);
    return c;
  };
  if (ts.isTaggedTemplateExpression(n) && isFunctionRef(unreflect(n.tag))) return WIDE_REASONS.fn;
  if (ts.isCallExpression(n)) {
    // I65-TOOLS T1: the callee and arguments through `.call`, `.apply`, `.bind` and Reflect.apply (callTarget).
    // I65-FX-TOOLS (ruling 9): read once on the receiver a `.call`/`.apply`/`.bind` step passed and once on the member's own
    // base (the I65 reading), opaque if either is; `(Object.values.call as any).apply(null, xs)` stays opaque.
    for (const useRecv of [true, false]) {
      const { fn: c, args, recv } = callTarget(n, useRecv);
      if (buildsCode(c, args)) return WIDE_REASONS.fn;
      // I65-FX-TOOLS (RG-W1-04): `globalThis.__lookupGetter__(k)` (also by `.call` on Object.prototype's) returns k's getter,
      // process's among them, unless k resolves outside KEY_UNSAFE.
      for (const [b, k] of memberOf(c)) {
        const self = recv ?? b;
        if (k !== "__lookupGetter__" || self === null || !isGlobalObjectRef(self)) continue;
        if (args === null) return WIDE_REASONS.globalReader;
        if (args[0] !== undefined) { const v = keyValue(args[0]); if (v === undefined || KEY_UNSAFE.has(v)) return WIDE_REASONS.globalReader; }
      }
      if (isGlobalMember(c, "eval")) return WIDE_REASONS.evalIndirect;
      const m = ts.isIdentifier(c) ? c.text : null;
      // I65-TOOLS T1: a renamed vm import, `const r = vm.runInThisContext`, `const { runInThisContext: r } = vm` (memberOf).
      if (VM_RUNS.has(m) || memberOf(c).some(([, k]) => VM_RUNS.has(k))) return WIDE_REASONS.vmRun;
      const a0 = args?.[0];
      // `Reflect.construct(Function, ..)`, also `Reflect["construct"]` (I65-TOOLS T1); `Reflect.apply(Function, ..)` is callTarget's.
      if (memberOf(c).some(([b, k]) => k === "construct" && b !== null && isReflectRef(b)) && a0 && isFunctionRef(a0)) return WIDE_REASONS.fn;
      // Object readers; I65-TOOLS T1: on `Object["values"]`, an alias of Object, `globalThis.Object`, by `.call`/`.apply`, and
      // Reflect.getOwnPropertyDescriptor; a descriptor key that may reach process, the global object, Function or eval (KEY_UNSAFE).
      for (const [b, k] of memberOf(c)) {
        if (b === null || !(isObjectRef(b) || (k === "getOwnPropertyDescriptor" && isReflectRef(b)))) continue;
        if (args === null) { if (OBJECT_READERS.has(k) || k === "getOwnPropertyDescriptor" || k === "assign") return WIDE_REASONS.globalReader; continue; }
        if (OBJECT_READERS.has(k) && a0 && isGlobalObjectRef(a0)) return WIDE_REASONS.globalReader;
        if (k === "getOwnPropertyDescriptor" && a0 && isGlobalObjectRef(a0) && args[1] !== undefined) { const v = keyValue(args[1]); if (v === undefined || KEY_UNSAFE.has(v)) return WIDE_REASONS.globalReader; }
        if (k === "assign" && args.slice(1).some((a) => isGlobalObjectRef(a))) return WIDE_REASONS.globalReader;
      }
    }
  }
  // `{ ...globalThis }` copies every value of the global object, process among them.
  if (ts.isSpreadAssignment(n) && isGlobalObjectRef(n.expression)) return WIDE_REASONS.globalReader;
  if (ts.isCallExpression(n) || ts.isNewExpression(n)) {
    const c = skipWrap(n.expression);
    // I65-TOOLS T1: `createRequire.call(null, u)`, `.apply`, and `req.call(null, k)` through callTarget.
    const t = callTarget(n);
    if (buildsCode(c, t.args)) return WIDE_REASONS.fn;
    const viaRequire = (ts.isIdentifier(t.fn) && WIDE.requires.has(t.fn.text)) || (ts.isCallExpression(t.fn) && isCreateRequireName(callTarget(t.fn).fn));
    if (viaRequire && (t.args === null || t.args.length)) {
      const lit = t.args === null ? null : literalText(t.args[0]);
      if (lit === null) return WIDE_REASONS.createRequire;
      if (/^(node:)?process$/.test(lit)) return "loads node:process as a value, which this scan cannot follow";
    }
  }
  // `eval` anywhere but the callee of a direct call (`eval(s)`, `(eval)(s)`) is an indirect eval: `(0, eval)(s)`, `const e = eval`.
  if (isNameRef(n, "eval")) {
    let at = n;
    while (at.parent && ts.isParenthesizedExpression(at.parent)) at = at.parent;
    const direct = at.parent && ts.isCallExpression(at.parent) && at.parent.expression === at;
    // I64-FX-TOOLS (FIX-ROUND-1 ruling 5, settles I64-TOOLS:2): a direct eval runs a string too, so it fails closed.
    if (!(at.parent && ts.isTypeOfExpression(at.parent))) return direct ? WIDE_REASONS.evalDirect : WIDE_REASONS.evalIndirect;
  }
  return null;
}
/** The call-shape facts (see shapeWalk). Each file is walked again in the WIDE view (I64-TOOLS G1); that pass's runner reason is
 *  kept apart as `wide`, which inputsOf takes only where no narrow reason applies, so a suite opaque today keeps its reason. */
export function callShape(sf, specs = []) {
  const shape = shapeWalk(sf, specs);
  try {
    wideView(sf);
    const wide = shapeWalk(sf, specs);
    shape.wide = wide.runner;
    if (wide.argvOrEnv) shape.argvOrEnv = true;
  } finally {
    WIDE = null;
  }
  return shape;
}
function shapeWalk(sf, specs) {
  const shape = { reads: false, enumerates: false, probes: false, spawns: false, spawnCalls: [], spawnArgLiterals: [], argvOrEnv: false, computedImports: [], evaluates: false, runner: null };
  const has = (re) => specs.some((m) => re.test(m));
  const canSpawn = has(/^(node:)?child_process$/);
  const canRead = has(/^(node:)?fs(\/promises)?$/) || has(/^(node:)?(fs-extra|glob|fast-glob|tinyglobby)$/);
  walk(sf, (n) => {
    if (WIDE !== null) { const r = wideReason(n); if (r !== null) shape.runner = shape.runner ?? r; }
    // I62-TOOLS O2: `process.<m>(…)` / `process["m"](…)` / `process[x](…)` with m a runner or unclassified fails closed.
    // I62-FX-TOOLS (D-1528): the same by name through a comma call `(0, process.m)(…)` or `process.m.call|apply|bind(…)`.
    // I63-TOOLS U3: a tagged template (process.m`x`) is a call too, and `process.m["call"|"apply"|"bind"](…)` is reflective.
    if (ts.isCallExpression(n) || ts.isTaggedTemplateExpression(n)) {
      let target = skipWrap(ts.isCallExpression(n) ? n.expression : n.tag);
      for (;;) {
        if (ts.isPropertyAccessExpression(target) && REFLECTIVE_CALLS.has(target.name.text)) target = skipWrap(target.expression);
        else if (ts.isElementAccessExpression(target) && REFLECTIVE_CALLS.has(literalText(target.argumentExpression))) target = skipWrap(target.expression);
        else break;
      }
      if ((ts.isPropertyAccessExpression(target) || ts.isElementAccessExpression(target)) && isProcessRef(skipParens(target.expression))) {
        const m = ts.isPropertyAccessExpression(target) ? target.name.text : literalText(target.argumentExpression);
        if (m === null) shape.runner = shape.runner ?? "calls a process method by a computed name";
        else if (RUNNER_PROCESS_METHODS.has(m)) shape.runner = shape.runner ?? `calls process.${m}(), which runs another program`;
        else if (!INERT_PROCESS_METHODS.has(m)) shape.runner = shape.runner ?? `calls process.${m}(), a process method not on the known-inert list`;
      }
    }
    // I63-TOOLS U3 (W0-21): any READ of a runner member of a process ref that the call scan above did not classify as a
    // direct call (`const f = process.execve`, `Reflect.apply(process.execve, …)`) fails closed. The walk is pre-order, so a
    // direct call's CallExpression sets its own reason first and this one never overrides it.
    if ((ts.isPropertyAccessExpression(n) || ts.isElementAccessExpression(n)) && isProcessRef(skipParens(n.expression))) {
      const m = ts.isPropertyAccessExpression(n) ? n.name.text : literalText(n.argumentExpression);
      // I63-FX-TOOLS U3: a computed key (`process[k]`, m null) may name any member, so it fails closed as the call arm's does.
      if (m === null) shape.runner = shape.runner ?? "accesses a process member by a computed key, which this scan cannot follow";
      if (RUNNER_PROCESS_METHODS.has(m)) shape.runner = shape.runner ?? `reads process.${m} as a value, which runs another program wherever it is called`;
    }
    // I63-TOOLS U3: the name itself, on a receiver this scan cannot resolve (`const g = globalThis; g.process.execve()`,
    // `const { execve } = g.process`): an identifier named execve/execveSync anywhere, or that name as a member key.
    if (ts.isIdentifier(n) && RUNNER_PROCESS_METHODS.has(n.text)) shape.runner = shape.runner ?? `names ${n.text}, a process runner method, on a receiver this scan cannot resolve`;
    if ((ts.isStringLiteral(n) || ts.isNoSubstitutionTemplateLiteral(n)) && RUNNER_PROCESS_METHODS.has(n.text)
      && ((ts.isElementAccessExpression(n.parent) && n.parent.argumentExpression === n) || (ts.isBindingElement(n.parent) && n.parent.propertyName === n)
        || ts.isComputedPropertyName(n.parent) || (ts.isPropertyAssignment(n.parent) && n.parent.name === n))) {
      shape.runner = shape.runner ?? `names ${n.text}, a process runner method, on a receiver this scan cannot resolve`;
    }
    if (ts.isNewExpression(n)) {
      const c = n.expression;
      const ctor = ts.isIdentifier(c) ? c.text : ts.isPropertyAccessExpression(c) && ts.isIdentifier(c.name) ? c.name.text : null;
      if (ctor && EVALUATING_CTORS.has(ctor)) shape.evaluates = true;
    }
    if (ts.isCallExpression(n)) {
      const callee = n.expression;
      const name = ts.isIdentifier(callee) ? callee.text : ts.isPropertyAccessExpression(callee) && ts.isIdentifier(callee.name) ? callee.name.text : null;
      if (name && EVALUATING_CALLS.has(name)) shape.evaluates = true;
      // I10-FX11 (QUEUE #461, JN-TOOLS-B): `import(x)` / `require(x)` with a COMPUTED specifier is no edge in
      // specifiersOf; its literals are kept so edgesOf can bound it or fail it open.
      if ((callee.kind === ts.SyntaxKind.ImportKeyword || (ts.isIdentifier(callee) && callee.text === "require")) && n.arguments.length) {
        const a0 = n.arguments[0];
        if (!(ts.isStringLiteral(a0) || ts.isNoSubstitutionTemplateLiteral(a0))) shape.computedImports.push([...literalsIn(a0)]);
      }
      if (name && canRead && READERS.has(name)) shape.reads = true;
      if (name && canRead && METADATA.has(name)) shape.probes = true;
      if (name && canRead && ENUMERATORS.has(name)) shape.enumerates = true;
      if (name && canSpawn && SPAWNERS.has(name)) {
        const lits = [];
        for (const a of n.arguments) for (const l of literalsIn(a)) lits.push(l);
        const a0 = n.arguments[0];
        const program = a0 && (ts.isStringLiteral(a0) || ts.isNoSubstitutionTemplateLiteral(a0)) ? a0.text : null;
        if (program !== null && READER_PROGRAMS.has(program)) shape.reads = true;
        else {
          shape.spawns = true;
          shape.spawnCalls.push({ program, lits });
          for (const l of lits) shape.spawnArgLiterals.push(l);
        }
      }
    }
    // A NAMED environment VALUE is a path channel; `process.argv` and a bare `process.env` are not
    // (D-0870(10)). argv: in-process under vitest it is the runner's own argv, and a SPAWNED
    // script's argv is spelled at the spawn site, whose literals are already that suite's edges.
    // A bare `process.env` (`{ ...process.env, X: out }`) hands the environment to a child that
    // is modelled at its own spawn site. Measured: argv/env was the cause for 89 of 275 opaque
    // suites at 2add002, and every CLI script imported for its exports was one of them.
    if ((ts.isPropertyAccessExpression(n) || ts.isElementAccessExpression(n)) && isProcessEnv(n.expression)) {
      // An ASSIGNMENT target (`process.env.AUDIT_ROOT = ROOT`) hands a value OUT; it reads nothing.
      const p = n.parent;
      const written = p && ts.isBinaryExpression(p) && p.left === n && p.operatorToken.kind === ts.SyntaxKind.EqualsToken;
      if (!written) {
        const key = ts.isPropertyAccessExpression(n) ? n.name.text : ts.isStringLiteral(n.argumentExpression) ? n.argumentExpression.text : null;
        if (key === null || !NON_TREE_ENV.has(key)) shape.argvOrEnv = true;
      }
    }
    // I10-FX3 (QUEUE #461, D-0871(10)): `process.env` read with no key AT this node is the same named
    // channel when it is DESTRUCTURED (`const { X } = process.env` — each bound name checked against
    // NON_TREE_ENV, a rest element always counts) or ALIASED (`const e = process.env`, `e = process.env`,
    // `({ X } = process.env)`): the key is read later, where this scan cannot follow it. A spread or an
    // argument (`{ ...process.env }`, `{ env: process.env }`) still hands it to a child (D-0870(10)).
    // I62-TOOLS O2: `const { execve } = process` binds a runner method the call scan above cannot see by name.
    // I62-FX-TOOLS (D-1528): from globalThis.process too, and any key not on the inert lists (or a rest/computed key) fails closed.
    const destructured = (ts.isVariableDeclaration(n) && n.initializer && isProcessRef(skipParens(n.initializer)) && ts.isObjectBindingPattern(n.name)) ? n.name.elements : [];
    for (const el of destructured) {
      const pn = el.propertyName ?? el.name;
      const key = el.dotDotDotToken ? null : ts.isIdentifier(pn) || ts.isStringLiteral(pn) ? pn.text : null;
      const c = processKeyClass(key, "binds");
      if (c === "env") shape.argvOrEnv = true;
      else if (c !== null) shape.runner = shape.runner ?? c;
    }
    // I62-FX-TOOLS (D-1528): `process` used as a VALUE — aliased (`const p = process`, `p = process`), passed (`f(process)`,
    // `{ process }`), returned or destructured by assignment — is followed nowhere by this scan, so it fails closed. A member
    // access, a `typeof` operand and a `const { … } = process` (classified by key above) are not values.
    if (isProcessRef(n)) {
      let at = n;
      while (at.parent && ts.isParenthesizedExpression(at.parent)) at = at.parent;
      const p = at.parent;
      const member = p && (ts.isPropertyAccessExpression(p) || ts.isElementAccessExpression(p)) && p.expression === at;
      const typeOf = p && ts.isTypeOfExpression(p);
      const pattern = p && ts.isVariableDeclaration(p) && p.initializer === at && ts.isObjectBindingPattern(p.name);
      if (!member && !typeOf && !pattern) shape.runner = shape.runner ?? "uses process as a value (an alias, an argument or a return), which this scan cannot follow";
    }
    // I62-FX-TOOLS (D-1528): `node:process` is the same object: a named import is classified by key, any other binding is an alias.
    if (ts.isImportDeclaration(n) && /^(node:)?process$/.test(literalText(n.moduleSpecifier) ?? "") && n.importClause) {
      const ic = n.importClause;
      if (ic.name && ic.name.text !== "process") shape.runner = shape.runner ?? "imports node:process under another name, which this scan cannot follow";
      if (ic.namedBindings && ts.isNamespaceImport(ic.namedBindings)) shape.runner = shape.runner ?? "imports node:process under another name, which this scan cannot follow";
      if (ic.namedBindings && ts.isNamedImports(ic.namedBindings)) {
        for (const el of ic.namedBindings.elements) {
          const c = processKeyClass((el.propertyName ?? el.name).text, "imports");
          if (c === "env") shape.argvOrEnv = true;
          else if (c !== null) shape.runner = shape.runner ?? c;
        }
      }
    }
    if (ts.isCallExpression(n) && (n.expression.kind === ts.SyntaxKind.ImportKeyword || (ts.isIdentifier(n.expression) && n.expression.text === "require"))
      && /^(node:)?process$/.test(literalText(n.arguments[0]) ?? "")) shape.runner = shape.runner ?? "loads node:process as a value, which this scan cannot follow";
    if (isProcessEnv(n)) {
      const p = n.parent;
      if (p && ts.isVariableDeclaration(p) && p.initializer === n) {
        if (ts.isObjectBindingPattern(p.name)) {
          for (const el of p.name.elements) {
            const pn = el.propertyName;
            const key = el.dotDotDotToken ? null
              : pn ? (ts.isIdentifier(pn) || ts.isStringLiteral(pn) ? pn.text : null)
              : ts.isIdentifier(el.name) ? el.name.text : null;
            if (key === null || !NON_TREE_ENV.has(key)) shape.argvOrEnv = true;
          }
        } else shape.argvOrEnv = true;
      } else if (p && ts.isBinaryExpression(p) && p.right === n && p.operatorToken.kind === ts.SyntaxKind.EqualsToken) shape.argvOrEnv = true;
    }
    return undefined;
  });
  return shape;
}

/** Path-shaped literals only. A bare word (`"test"`, `"utf8"`) is not a path edge: it has no
 *  separator and no extension, and admitting it made the scope the whole tree on the first run.
 *  A template's `${…}` hole is a `?` (lib/ts-scan.mjs), and everything from the first hole on is
 *  discarded — the prefix before it is still a real directory. */
export function bareWordLiterals(sf) {
  const out = new Set();
  for (const raw of literalsIn(sf)) {
    const lit = raw.split("?")[0];
    if (lit && lit.length < 60 && /^[A-Za-z0-9._-]+$/.test(lit) && !/\.[A-Za-z0-9]{1,6}$/.test(lit)) out.add(lit);
  }
  return out;
}

const pathShaped = (lit) => Boolean(lit) && lit.length <= 300 && !/^[a-z]+:\/\//i.test(lit)
  && (lit.includes("/") || /\.[A-Za-z0-9]{1,6}$/.test(lit) || lit === ".." || lit === ".");

/** Path-shaped literals, split into the ones spelled WHOLE and the FRAGMENTS that sit either
 *  side of a template hole. The distinction is load-bearing and was measured: this tree writes
 *  an esbuild entry as `` `${ROOT}src/main.ts` ``, which `literalsIn` renders `?src/main.ts`.
 *  Taking the text before the first hole gives the empty string, so eighteen creator-console DOM
 *  suites and four web ones — every one of which BUNDLES the app at test time and reddens when a
 *  package two hops down moves — had no edge at all. A fragment is resolved against the
 *  directories the file's whole literals already named, which is what `ROOT` is. */
export function pathLiteralParts(sf) {
  const whole = new Set();
  const frags = new Set();
  for (const raw of literalsIn(sf)) {
    const parts = raw.split("?");
    // The head — a hole-free literal, or the text BEFORE the first hole — is anchored at the
    // file and may name a directory: `` `../migrations/${f}` `` is a read of that directory.
    // Dropping that distinction cost 437 suites off the migration scope in one edit, every one
    // of which the full suite then showed reddening. Only what sits AFTER a hole is
    // unanchored, and that is the part restricted to files below.
    if (pathShaped(parts[0])) whole.add(parts[0]);
    for (const part of parts.slice(1)) if (pathShaped(part)) frags.add(part.replace(/^\/+/, ""));
  }
  return { whole, frags };
}

export function pathLiterals(sf) {
  return pathLiteralParts(sf).whole;
}

// ── the graph ─────────────────────────────────────────────────────────────────────────────────
/** A lazily-parsed, memoised view of the tree. Only files actually reached from a suite are
 *  parsed, so a query costs the closure it touches and not the whole repo. */
export function createGraph(root, opts = {}) {
  const warnings = [];
  const warn = (m) => warnings.push(m);
  const { dirs: skipDirs, unreduced } = excludeDirsFromVitestConfig(root, opts.vitestDefaults ?? []);
  const treeFiles = indexTree(root, skipDirs);
  const treeSet = new Set(treeFiles);
  const has = (r) => treeSet.has(r);
  const pkgs = workspacePackages(root, treeFiles);
  const { suites, source: popSource } = opts.population
    ? { suites: new Set(opts.population), source: "caller-supplied" }
    : suitePopulation(root, treeFiles, warn);
  const dirSet = new Set();
  for (const f of treeFiles) { let d = dirname(f); while (d && d !== "." && !dirSet.has(d)) { dirSet.add(d); d = dirname(d); } }

  // UNIVERSAL INPUTS — loaded by the runner for every suite, invisible to an import graph
  // rooted at the test file: the config itself and everything in its `setupFiles`.
  const universal = [];
  if (existsSync(join(root, "vitest.config.ts"))) universal.push("vitest.config.ts");
  for (const f of setupFilesFromVitestConfig(root)) universal.push(f);

  const edgeCache = new Map();
  /** file -> { imports:Set, readFiles:Set, readDirs:Set, opaque:string|null } */
  function edgesOf(rel) {
    if (edgeCache.has(rel)) return edgeCache.get(rel);
    const empty = { imports: new Set(), readFiles: new Set(), readDirs: new Set(), readOutside: new Set(), opaque: null, wide: null, executes: false };
    edgeCache.set(rel, empty);
    const ext = /\.[^./]+$/.exec(rel)?.[0] ?? "";
    if (!PARSEABLE.has(ext) || !treeSet.has(rel)) return empty;
    let sf;
    try { sf = parse(join(root, rel)); } catch (e) { warn(`unparseable: ${rel} — ${String(e.message).split("\n")[0]}`); return empty; }
    const e = { imports: new Set(), readFiles: new Set(), readDirs: new Set(), readOutside: new Set(), opaque: null, wide: null, executes: false };
    const specs = specifiersOf(sf);
    for (const spec of specs) {
      const r = resolveSpecifier(root, rel, spec, pkgs, has);
      if (r) e.imports.add(r);
      else if (spec.startsWith(".")) warn(`unresolved relative import: ${rel} -> ${spec}`);
    }
    const shape = callShape(sf, specs);
    // WHICH FILES' LITERALS ARE PATHS AT ALL. A `services/trust-core/src` module that never
    // touches the filesystem spells route paths, ids and URLs, not reads: harvesting its
    // literals gave `auth.ts` a directory edge on its OWN directory and, because half the tree
    // imports auth.ts, put all 849 suites in the scope of every file under that directory.
    // But `node:fs` is not the only way a path is consumed, and keying on it alone MISSED a
    // whole class, measured against the full suite: a DOM suite that imports `esbuild` and
    // bundles `src/main.ts` in-process reads that entry's entire module graph without ever
    // naming `readFileSync`. So: a suite, a script, or any file that imports a bundler/loader
    // has path literals; an ordinary source module has them only if it reads or spawns.
    const PATH_CONSUMING = /^(esbuild|rollup|vite|webpack|jsdom|typescript|tsup|@swc\/core|playwright|puppeteer)$/;
    const isSuite = suites.has(rel) || /\.(test|spec)\.[cm]?[jt]sx?$/.test(rel);
    const bundles = specs.some((m) => PATH_CONSUMING.test(m));
    const canTouchFs = shape.reads || shape.probes || shape.enumerates || shape.spawns || bundles || isSuite || rel.startsWith("scripts/");
    let rootAnchored = false;
    const parts = canTouchFs ? pathLiteralParts(sf) : { whole: new Set(), frags: new Set() };
    const land = (lit, bases, filesOnly = false) => {
      for (const base of bases) {
        const abs = resolve(root, base, lit);
        const r = relative(root, abs).split("\\").join("/");
        // A literal landing ON the repository root is a whole-tree read (`new URL("../../..",
        // import.meta.url)` + a walk). There is no prefix that bounds it, so it is opaque —
        // NOT a dropped edge, which is what an `inTree` null would have made it.
        if (r === "") { if (!filesOnly) rootAnchored = true; return; }
        if (!relInside(r)) continue;
        if (treeSet.has(r)) { e.readFiles.add(r); return; }
        if (dirSet.has(r)) { if (filesOnly) continue; e.readDirs.add(r); return; }
        const inferred = tryFile(r, has);
        if (inferred) { e.readFiles.add(inferred); return; }
        // Under a directory the walk SKIPS (`org/`, `node_modules/`, …): a BOUNDED read of a path no
        // --files target can be. It bounds the file (no opacity) and adds no edge. Statted only here.
        if (r.split("/").some((s) => skipDirs.has(s)) && existsOutside(abs, r, filesOnly)) { e.readOutside.add(r); return; }
      }
    };
    for (const lit of parts.whole) land(lit, [dirname(rel), ""]);
    // Second pass: a fragment beside a template hole is resolved against the directories the
    // whole literals just named — `${ROOT}src/main.ts` with `ROOT` from `new URL("../", …)`.
    // FILES ONLY. A fragment may name an entry point (`${ROOT}src/main.ts`); letting it name a
    // DIRECTORY made `scripts/invariant-setup.mjs` — which vitest injects into every suite —
    // depend on all of `packages/canonical/`, and every one of the 853 suites inherited it, so
    // any change under that package scoped to the whole tree. The base directory a fragment is
    // measured against already came from a whole literal, so nothing is lost: the coarse edge
    // was the base, and the fragment's job is to sharpen it.
    const dirBases = [dirname(rel), "", ...e.readDirs];
    for (const lit of parts.frags) land(lit, dirBases, true);
    // PER SPAWN CALL, not per file. A test that runs `execFileSync("node", [CLOSE_RUN])` — where
    // CLOSE_RUN is a const built from the literal "scripts/close-run.mjs" higher up — has no
    // literal INSIDE the call, and judging the file by the call's own literals marked all four
    // close-run guards opaque while the read-file edge that actually names the script sat right
    // there. A spawn is bounded when the program is a tree file, or when the program is a script
    // RUNNER (`node`, `npx`, …) and some script-shaped file in this file's read set could be
    // what it runs. `git`, `sh`, `pnpm` with nothing resolvable stay opaque, because what they
    // read is the working tree, not a file this tool can name.
    const RUNNERS = new Set(["node", "npx", "npm", "pnpm", "tsx", "ts-node", "bun", "deno"]);
    const resolvesInTree = (l) => {
      const q = String(l).split("?")[0];
      for (const base of [dirname(rel), ""]) { const r = inTree(root, resolve(root, base, q)); if (r && treeSet.has(r)) return true; }
      return false;
    };
    const scriptish = [...e.readFiles].some((f) => /\.(mjs|cjs|js|ts)$/.test(f));
    const spawnResolves = shape.spawnCalls.length > 0 && shape.spawnCalls.every((c) => {
      if (c.program && resolvesInTree(c.program)) return true;
      if (c.lits.some(resolvesInTree)) return true;
      // `execFileSync(process.execPath, [CLOSE_RUN, …])` — the program is computed, not spelled.
      return Boolean((c.program === null || RUNNERS.has(c.program)) && scriptish);
    });
    // A ROOT-ANCHORED literal (`new URL("../..", import.meta.url)`) is the repo root itself.
    // It bounds nothing on its own, but it does not by itself make the file unbounded either:
    // a targeted `join(repo, "services", "trust-core", "src", "x.ts")` is still spelled out in
    // literals. So the root anchor is opaque only in the company of an ENUMERATOR (or argv/env,
    // or an unresolvable spawn) — measured: treating the anchor alone as opaque put 93 extra
    // suites into every scope, 90 of which never enumerate anything. What the anchor DOES buy
    // is the multi-segment join, whose segments are bare words `pathLiterals` rejects: with an
    // anchor present, a bare word naming a real top-level directory becomes a prefix edge.
    if (rootAnchored) {
      if (shape.enumerates) e.opaque = "reads a path that resolves to the repository root and enumerates a directory";
      else for (const w of bareWordLiterals(sf)) if (dirSet.has(w)) e.readDirs.add(w);
    }
    // I10-FX11 (QUEUE #461): a computed import is an EDGE when a literal inside its argument names a tree
    // file (`import(join(ROOT, "scripts/x.mjs"))`), and OPAQUE otherwise: what it loads, and that module's
    // own closure, cannot be named.
    for (const lits of shape.computedImports) {
      const hit = lits.map((l) => String(l).split("?")[0]).filter(pathShaped)
        .flatMap((q) => [dirname(rel), ""].map((base) => inTree(root, resolve(root, base, q)))).find((r) => r && treeSet.has(r));
      if (hit) e.imports.add(hit);
      else e.opaque = e.opaque ?? "imports a module by a computed specifier that names no file in the tree";
    }
    // OPAQUE — the read set cannot be bounded by this file's own literals.
    // May this file RUN a file it reads? Only then does a read-file edge carry that file's opacity (inputsOf).
    // Keyed on the IMPORT, not the call shape: `promisify(execFile)(…)` spawns with no call named execFile.
    // (A computed import is an import edge when it names a tree file, and self-opaque when it does not.)
    e.executes = specs.some((m) => RUNS_CODE.test(m)) || shape.evaluates || bundles;
    // I62-TOOLS O2: a runner channel is opaque first, whatever else the file's literals bound.
    const runnerSpec = specs.find((m) => RUNNER_SPECIFIERS.test(m));
    if (runnerSpec !== undefined) e.opaque = e.opaque ?? `runs code through ${runnerSpec}, a test or worker runner whose input set cannot be bounded`;
    if (shape.runner !== null) e.opaque = e.opaque ?? shape.runner;
    if (shape.spawns && !spawnResolves) e.opaque = e.opaque ?? "spawns a subprocess whose program/args name no file in the tree";
    else if ((shape.reads || shape.enumerates) && e.readFiles.size === 0 && e.readDirs.size === 0 && e.readOutside.size === 0) e.opaque = e.opaque ?? "reads the filesystem with no path literal that resolves into the tree";
    else if (shape.argvOrEnv && (shape.reads || shape.enumerates)) e.opaque = e.opaque ?? "reads the filesystem and takes a named process.env value";
    // I64-TOOLS G1 (D-1556 (b)): a process reach only the wide view sees (callShape) is kept apart; inputsOf takes it last.
    e.wide = shape.wide;
    edgeCache.set(rel, e);
    return e;
  }

  const inputCache = new Map();
  /** Every tree file `rel` can be affected by, plus the directory prefixes it may read under. */
  function inputsOf(rel) {
    if (inputCache.has(rel)) return inputCache.get(rel);
    const files = new Set();
    const dirs = new Set();
    let opaque = null;
    let wide = null;
    // Each file is visited as RUN (it executes in the suite's process or a child) or READ (only
    // its bytes are read). A READ file's closure is still an input, but its opacity is not: a
    // module that never runs performs none of its own reads (I28-TESTSCOPE S1).
    const ran = new Set();
    const read = new Set();
    const queue = [[rel, true]];
    if (suites.has(rel)) for (const u of universal) queue.push([u, true]);
    while (queue.length) {
      const [cur, runs] = queue.pop();
      if (ran.has(cur) || (!runs && read.has(cur))) continue;
      (runs ? ran : read).add(cur);
      files.add(cur);
      const e = edgesOf(cur);
      // Opacity PROPAGATES. A helper module the suite imports that reads the filesystem
      // unboundedly makes the SUITE's input set unbounded; scoping on the suite's own call
      // shape alone would drop exactly the indirection a helper exists to create.
      if (runs && e.opaque && !opaque) opaque = cur === rel ? e.opaque : `via ${cur}: ${e.opaque}`;
      // I64-TOOLS G1: a wide-view reach makes the suite opaque only where no narrow reason does, so every reason today is kept.
      if (runs && e.wide && !wide) wide = cur === rel ? e.wide : `via ${cur}: ${e.wide}`;
      for (const d of e.readDirs) dirs.add(d);
      // read-closure: a literal landing on a module drags that module's import closure too. It
      // is RUN only when the reader can run it (spawn, eval, Worker, bundler, computed import).
      for (const r of e.imports) queue.push([r, runs]);
      for (const r of e.readFiles) queue.push([r, runs && e.executes]);
    }
    const out = { files, dirs, opaque: opaque ?? wide };
    inputCache.set(rel, out);
    return out;
  }

  return { root, treeFiles, treeSet, dirSet, pkgs, suites, popSource, unreduced, warnings, universal, edgesOf, inputsOf };
}

const underAny = (p, dirs) => { for (const d of dirs) if (p === d || p.startsWith(d + "/")) return true; return false; };

/** scope(changed) — every suite that can redden, with the channel that put it there. */
export function scopeFor(graph, changed) {
  const hits = new Map();
  const add = (s, why) => { if (!hits.has(s)) hits.set(s, []); hits.get(s).push(why); };
  for (const s of graph.suites) {
    const { files, dirs, opaque } = graph.inputsOf(s);
    if (opaque) { add(s, `opaque — ${opaque}`); continue; }
    for (const c of changed) {
      if (c === s) add(s, "is the changed suite");
      else if (files.has(c)) add(s, `input: ${c}`);
      else if (underAny(c, dirs)) add(s, `reads a directory containing ${c}`);
    }
  }
  return hits;
}

/** Expand a --files argument: a directory means every file under it. A path that is not in the
 *  tree is an ERROR, never a silent no-op — the shape that lets a scope come back empty and
 *  read as "nothing to run". */
export function expandTargets(graph, args) {
  const out = new Set();
  const missing = [];
  for (const a of args) {
    const rel = a.replace(/^\.\//, "").replace(/\/+$/, "").split("\\").join("/");
    if (graph.treeSet.has(rel)) { out.add(rel); continue; }
    if (graph.dirSet.has(rel)) {
      let n = 0;
      for (const f of graph.treeFiles) if (f.startsWith(rel + "/")) { out.add(f); n++; }
      if (n === 0) missing.push(`${a} (directory, but empty)`);
      continue;
    }
    missing.push(a);
  }
  return { targets: [...out], missing };
}

// ── CLI ───────────────────────────────────────────────────────────────────────────────────────
async function main(argv) {
  const opt = (n, d = null) => { const i = argv.indexOf(`--${n}`); return i >= 0 ? argv[i + 1] : d; };
  const flag = (n) => argv.includes(`--${n}`);
  // STOPS at the next flag. `.filter(a => !a.startsWith("--"))` over the whole tail swallowed

    const withField = pk.filter((p) => "dispatch_model" in p);
    const missing = pk.length - withField.length;
    const known = withField.filter((p) => p.dispatch_model !== null);
    const disagree = known.filter((p) => p.model && p.dispatch_model !== p.model);
    console.log(`\nRECONCILE — #234: contract \`model\` vs actually-dispatched \`dispatch_model\``);
    console.log(`  packets carrying the field: ${withField.length}/${pk.length} (missing: ${missing}) · known (non-null): ${known.length}`);
    if (disagree.length) {
      console.log(`  DISAGREE (${disagree.length}) — contract model differs from what was actually dispatched:`);
      for (const p of disagree.slice(0, 20)) console.log(`    ${p._batch ?? "?"}/${p.dir ?? p.room ?? "?"} — model=${p.model} dispatch_model=${p.dispatch_model}`);
      if (disagree.length > 20) console.log(`    … and ${disagree.length - 20} more`);
    } else if (known.length) {
      console.log(`  no disagreement across ${known.length} packet(s) with a known dispatch_model`);
    } else {
      console.log(`  no packet carries a known dispatch_model yet — nothing to reconcile`);
    }
  }
  console.log("");
}

// ---------------------------------------------------------------- --check
if (has("check")) {
  const n = rows.length;
  const bad = [];
  if (n === 0) bad.push("USAGE.jsonl is empty — nothing is being recorded");
  else for (const [f, floor] of Object.entries(COVERAGE_FLOORS)) {
    // QUEUE #289 (D-0424.6/.7): the `turns:` header only became mandatory for reports under a
    // batch dir S-081+/RUN-0016+ (validate-room-report.mjs's own contract-0.8.1 gate) — applying
    // this floor to the WHOLE ledger would fail rows that were never required to carry it. Scope
    // the denominator to in-cutoff rows only; a ledger with none yet is not a failure.
    if (f === "turns") {
      const turnsRows = rows.filter((r) => isAtOrAfterCutoff(r.run, LEGACY_DIRS));
      if (turnsRows.length === 0) continue;
      const have = turnsRows.filter((r) => typeof r.turns === "number").length;
      if (have / turnsRows.length < floor) bad.push(`turns: ${have}/${turnsRows.length} (S-081+/RUN-0016+ rows only) below floor ${Math.round(floor * 100)}%`);
      continue;
    }
    const have = f === "model" ? rows.filter((r) => r.model && r.model !== "?").length
      : f === "any_tokens" ? rows.filter(anyTokens).length
      : rows.filter((r) => typeof r[f] === "number").length;
    if (have / n < floor) bad.push(`${f}: ${have}/${n} below floor ${Math.round(floor * 100)}%`);
  }
  if (bad.length) { console.error("token-report --check FAILED:\n  " + bad.join("\n  ")); process.exit(1); }
  console.log("token-report --check OK");
}
