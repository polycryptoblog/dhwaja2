/**
 * I3-0401 (D-1078(3)) — the landing and Discover link an `/is/` answer page. Those pages are served by
 * trust-core, so the link is absolute on the API base (the base `checkShareLink` builds its `/is/` link on).
 * This is the landing/Discover EXAMPLE chip (JASON-QUESTIONS 7), drawn ONLY when it is `<API base>/is/<platform>/<handle>`
 * in the shape the server route admits; any other value, or no base, draws nothing — never a dead or off-site link.
 * A verified Discover row links its own answer page through `discoverAnswerLink` below (I3-0401, D-1082(12)).
 */
export function answerPageExample(base: string = apiBase(), example: string = exampleVerifiedUrl()): string {
  const b = typeof base === "string" ? base.replace(/\/+$/, "") : "";
  if (!/^https:\/\/[^\s"'<>]+$/.test(b)) return "";
  if (typeof example !== "string" || !example.startsWith(`${b}/is/`)) return "";
  if (!/^\/is\/[a-z0-9_]{1,24}\/[a-z0-9][a-z0-9._-]{0,63}$/.test(example.slice(b.length))) return "";
  return `<a class="ds-btn nav" href="${xml(example)}" rel="noopener">See what a Verified answer looks like</a>`;
}

/** I3-0401 (D-1082(12)): a Discover row's link to its creator's `/is/` answer page — only for a row the server marked

/**
 * Pure view renderers: data → HTML-string fragment. Every dynamic value flows
 * through the card-render escape helpers (xml/textClamp/num); the only href/src
 * sinks are https-allowlisted. Tab switching, navigation, and fetching are the
 * DOM controller's job — these functions never touch the DOM or the network.
 *
 * Navigation is expressed as data-nav="#/..." attributes the controller wires
 * to hash changes, so a rendered link can never carry an untrusted URL into an
 * href (no open-redirect from rendered content).
 */

const SAFE_HTTPS = /^https:\/\/[^\s"'<>]+$/;
function safeHref(u: string | null | undefined): string | null {
  return typeof u === "string" && SAFE_HTTPS.test(u) ? u : null;
}

// Shape guards: the API is trusted-ish but a malformed/partial payload must
// fail SOFT (an empty/plain fallback), never throw and freeze the view on
// "Loading…" (review finding). hx() tolerates non-strings; arr() tolerates
// non-arrays.
function hx(s: unknown, n: number): string {
  return typeof s === "string" ? s.slice(0, n) : "";
}
function arr<T>(x: unknown): T[] {
  return Array.isArray(x) ? (x as T[]) : [];
}
/** Clamp WITHOUT escaping (same cut shape as textClamp), for labels handed to
 * a link helper: navLink/pspLink/handleLink/acqLink escape their label exactly
 * once themselves, so pre-escaping with textClamp double-escapes — a creator
 * named "D'Angelo" rendered as "D&#39;Angelo". Never use this in direct text
 * position; that is textClamp's job. */
function clampPlain(s: unknown, max: number): string {
  const raw = typeof s === "string" ? s : "";
  return raw.length > max ? raw.slice(0, max - 1) + "…" : raw;
}

/**
 * THE FAIL-CLOSED COPY LAW, as two functions.
 *
 * A list that arrived is a FACT ("you have collected nothing"). A list that did
 * NOT arrive is the absence of one ("we could not read your collection"). These
 * rendered identically: `codexView({entries: null})` was byte-for-byte
 * `codexView({entries: []})` — "0 creators seen / Nothing here yet" — and the
 * same held for saves, achievements, acquisitions, claimed codes and what's-new.
 * So a fan whose session had expired, or whose request had been rate-limited, or
 * who was offline, was told AS FACT that they had collected nothing. That is the
 * one thing an error is never allowed to read as: reassurance.
 *
 * `report.ts` already had the right idiom ("We couldn't load your reports just
 * now… nothing about your cases has changed"); this is that idiom, shared.
 *
 * A NON-ARRAY IS ALSO UNREADABLE, not empty — a truncated or malformed body must
 * land here rather than being coerced by `arr()` into a confident zero.
 */
function unreadable(x: unknown): boolean {
  return !Array.isArray(x);
}

/**
 * The same card for a view whose payload is a single OBJECT rather than a list,
 * where `null` means either "the server said 404" or "the read never completed".
 * Those are different facts and the client cannot tell them apart from a `null`
 * alone, so every such read now carries its HTTP status (`api.getJsonStatus`,
 * the idiom `upgrade.ts` and `removal.ts` already use) and only a real 404 is
 * allowed to render the not-found copy. Anything else — 0 (offline/DNS/CORS),
 * 5xx, a 200 with a body we could not parse — lands here.
 */
function unreadableObject(what: string, cls: string): string {
  // #132PS-274: the list card's own sentence, never a near-copy of it.
  return `<section class="${xml(cls)}">${unreadableCard(what)}</section>`;
}

/** A read that carries its HTTP status. `status === 404` is the SERVER saying
 * "no such thing"; every other failure is our own read failing. Matches
 * `Api.getJsonStatus`'s shape so a route can hand one straight to a view. */
export interface Read<T> {
  status: number;
  data: T | null;
}

/** A navigation link that routes INSIDE the app (validated hash only). */
function navLink(hash: string, label: string, cls = ""): string {
  // hash is always renderer-built from a validated id, never raw user input.
  // I3-0413: the selected topbar entry / facet chip says so to a screen reader,
  // not only by colour — every caller marks selection with the `active` class.
  const current = /(^|\s)active(\s|$)/.test(cls) ? ` aria-current="page"` : "";
  return `<a href="${xml(hash)}" data-nav="${xml(hash)}"${cls ? ` class="${xml(cls)}"` : ""}${current}>${xml(label)}</a>`;
}

/** S84 install-CTA targets (web→extension / web→mobile), build-time https URLs. */

/** A small web→extension / web→mobile install row. Each link is https-allowlisted;
 * a missing/hostile URL renders nothing, never a junk href. */
function installRow(install?: InstallLinks): string {
  if (!install) return "";
  const ext = safeHref(install.extension);
  const app = safeHref(install.app);
  if (!ext && !app) return "";
  const parts =
    (ext ? `<a href="${xml(ext)}" rel="noopener noreferrer" class="ds-btn nav">Add the browser extension</a>` : "") +
    (app ? `<a href="${xml(app)}" rel="noopener noreferrer" class="ds-btn nav">Get the mobile app</a>` : "");
  return `<div class="install"><span class="installlabel">Check any creator, anywhere:</span>${parts}</div>`;
}

/**
 * The safety doors, rendered UNDER every check result.
 *
 * Two of these placements are load-bearing rather than convenient:
 *
 *  - CAUTION. A caution is machine-served and — per the design system —
 *    contestable: through `#/contest` when fan reports stand behind it, and
 *    through `#/remove` with "dispute" for a population-1 caution, which
 *    `caution.ts contestCaution` no-ops (QUEUE #146 C-14). (Only the report-derived population ages out;
 *    `caution.ts:178-186` binds no cutoff on an unproven listing.) A verdict that is always contestable has to
 *    carry the contest link in the same place as the verdict, not three pages
 *    away, or "contestable" is a claim about an API rather than about a person's
 *    actual options. So the caution branch leads with Dispute this caution.
 *  - EVERY BRANCH. "Is this listing me?" is offered on all of them, including
 *    "Not recognized" and the fail-closed "Couldn't check" — because someone
 *    arriving at their own listing has no idea what verdict Dhwaja is serving
 *    about them, and hiding the removal door behind a particular verdict would
 *    make the legal safety valve conditional on the thing they are complaining
 *    about.
 *
 * These are plain navigation. Nothing here reads or shows a verdict, and no
 * link carries the checked URL — the whole point of the k-anonymous check is
 * that the URL never left the device, so it must not leave through a link
 * either. The forms ask for it again instead.
 */
function safetyActions(status: string, howHref?: string, exampleHref?: string): string {
  // THE CAUTION DOOR IS NOW THE SHARED ONE. This chip existed here and nowhere
  // else: the handoff table every other client mirrors sent a caution to
  // "Report this account →", so a creator whose own unproven listing had been
  // cautioned was invited, on three of the four clients, to report themselves.
  // `SAFETY_HANDOFF.caution` is `#/contest` on all four now, and this chip is
  // the web's spelling of the same door — pinned to it by
  // tests/invariants/safety-handoff-parity.test.ts.
  // I3-0451 web limb (I16-W04D hand-off, D-0924(5)): a check that could not complete gets a retry door — the web
  // spelling of `SAFETY_HANDOFF.failed` (`#/check`, "Try again").
  const retry = status === "failed" ? navLink("#/check", "Try again", "ds-btn nav primary") : "";
  const contest = status === "caution" ? navLink("#/contest", "Dispute this caution", "ds-btn nav primary") : "";
  const report = status === "unsupported" || status === "failed" ? "" : navLink("#/report", "Report an impostor", "ds-btn nav");
  // I3-0375 (D-0992(7)): a warning gets its own dispute door — the removal intake with "dispute" preselected
  // (router.ts `#/remove/dispute`), the door removal.ts contestView already names for a published warning.
  const disputeWarning = status === "warning" ? navLink("#/remove/dispute", "Dispute this warning", "ds-btn nav") : "";
  const nominate = status === "unknown" ? navLink("#/nominate", "Suggest this creator", "ds-btn nav") : "";
  // UNCLAIMED IS THE ONE VERDICT WITH AN OBVIOUS NEXT MOVE AND IT HAD NONE.
  // "Listed from public information; not confirmed by this creator" used to
  // offer two doors for PROBLEMS — report an impostor, is this listing me — and
  // nothing at all for the actual situation, which is a creator who simply
  // hasn't claimed their Passport yet. The product has a whole surface for
  // that. The k-anonymous check deliberately returns no passport id for a
  // positive hit, so this cannot deep-link to the creator; the claim page
  // (`#/claim`, below) is the door that needs no id.
  // CHIP2 (QUEUE #101, docket #55 ruled (a) 2026-09-12): this chip used to dead-end at
  // the Most Requested board — a read-only leaderboard that answers a different question
  // and drops what the fan just typed. It now opens the request form, which the app
  // prefills from module memory (app.ts lastCheckedPair). The LINK still carries nothing:
  // the paragraph above is unchanged and still true.
  const ask = status === "unclaimed" ? navLink("#/nominate", "Ask for a creator on Dhwaja", "ds-btn nav") : "";
  // I3-0129 — THE DOOR FOR THE OPPOSITE OUTCOME, AT THE VERDICT.
  // Every other chip on an unclaimed result ends with the listing coming down,
  // being reported or being asked about by somebody else. The one person this
  // page is actually about — the creator who just found their own unclaimed
  // listing — had nothing here that KEEPS it. `removal.ts:439 claimDoor()`
  // already answers "where does an id-free claim CTA point in this app?" and
  // this is the same door at the earlier surface, not a second answer: the
  // href is `#/claim` (app.ts case "claim" -> removal.ts claimPage), the id-free
  // claim page; the k-anonymous check returns NO passport id (the paragraph
  // above), so an id-derived href would have zero callers. R2-I21-Q125
  // (D-1110(10)): it no longer points at the switched-off Most Requested board.
  const claim = status === "unclaimed" ? navLink("#/claim", "It's me — claim it instead", "ds-btn nav") : "";
  // N3 — THE DOOR THE EXPLAINER NEEDED. The text explaining what these words
  // mean lived only inside `/is/:platform/:handle`, which 404s for exactly the
  // subject a worried fan is asking about, so the answer to "what does this
  // mean?" was reachable only if you were already on some creator's page. It is
  // an absolute link because the page is served by trust-core, not by this SPA.
  // Absent base → no chip, rather than a dead relative link.
  const how = howHref ? `<a class="ds-btn nav" href="${xml(howHref)}" rel="noopener">What does this mean?</a>` : "";
  // I3-0048b (D-0993(1)): an example of a good Verified answer, on the unclaimed result only, from build config
  // (main.ts `__EXAMPLE_VERIFIED_URL__`); an absent or non-https value draws no chip.
  const exampleUrl = status === "unclaimed" && typeof exampleHref === "string" && /^https:\/\/[^\s"'<>]+$/.test(exampleHref) ? exampleHref : null;
  const example = exampleUrl ? `<a class="ds-btn nav" href="${xml(exampleUrl)}" rel="noopener">See what a Verified answer looks like</a>` : "";
  return `<div class="safebar">${retry}${contest}${report}${disputeWarning}${nominate}${ask}${claim}${example}${navLink("#/remove", "Is this listing me?", "ds-btn nav")}${how}</div>`;
}

/**
 * F19 / #4 — THE CHECK-RESULT SHARE PATH, and why it is a LINK rather than a card.
 *
 * The roadmap item is "a shareable card for a check result". It cannot be built
 * here, and the reason is one this file already states two functions down: **the
 * k-anonymous check never returns a passport id for a positive hit.** The bucket
 * entry carries a hash, a trust class, a per-account confirmation flag and a
 * subject kind (`registry.ts RegistryEntry`) — no id, no display name. So a fan
 * holding a green answer does not know WHO the creator is, and a card naming the
 * creator is not derivable from what this screen holds.
 *
 * The two ways to close that gap are to publish passport ids in the buckets, or
 * to make a second lookup that tells the server which handle was checked. Both
 * give away the property the whole check design exists for, and neither is worth
 * a share button.
 *
 * SO THE SHARE ARTIFACT LIVES ONE DELIBERATE STEP AWAY. `/is/<platform>/<handle>`
 * resolves the passport SERVER-side, renders the creator's own card, and is the
 * page a link unfurls to. Following it is the fan choosing to leave the
 * anonymous check — their act, in the open, not something done behind them.
 *
 * OFFERED ONLY FOR CLAIMED AND VERIFIED. That is the answer page's own gate, and
 * it is the decision: an image with no page around it is only safe where Dhwaja
 * has an affirmative answer. Everything weaker keeps the doors it already has.
 */
function checkShareLink(res: CheckResult, base?: string): string {
  if (res.status !== "verified" && res.status !== "claimed") return "";
  // I3-0403 b: `/is/` 404s for an account the entry did not confirm (claims.ts S101 asserted-unproven), so the
  // link is offered only where `res.confirmed === true` — the same strict read the card's sentence uses.
  if (res.confirmed !== true) return "";
  // I3-0044 a / I3-0403 a: `/is/` is served by trust-core, not by this SPA — an absolute link from the same
  // configured origin `howHref` uses; absent base -> no chip, never a dead relative link.
  if (typeof base !== "string" || !/^https?:\/\/[^\s"'<>]+$/.test(base)) return "";
  const platform = typeof res.platform === "string" ? res.platform : "";
  const handle = typeof res.handle === "string" ? res.handle : "";
  // The same two shapes the server route admits, re-asserted here so a malformed
  // pair cannot build a link at all — a dead share link is worse than none.
  if (!/^[a-z0-9_]{1,24}$/.test(platform) || !/^[A-Za-z0-9][A-Za-z0-9._-]{0,63}$/.test(handle)) return "";
  return `<p class="sharerow"><a class="ds-btn nav" href="${xml(base.replace(/\/+$/, ""))}/is/${xml(platform)}/${xml(handle.toLowerCase())}" rel="noopener">See and share this creator's Dhwaja card →</a></p>`;
}

/** I3-1374 b/web: the staleness window and sentence, the same as the extension popup's (popup.ts, INFO_MAX_STALE_MS). */
export const BUILT_AT_STALE_MS = 12 * 60 * 60 * 1000;
export const BUILT_AT_STALE_SUFFIX = " — older than expected, so it may be out of date";
export function checkResultView(input: string, res: CheckResult, opts?: { install?: InstallLinks; reason?: string; checkedAt?: string | number | Date; howHref?: string; ssrBase?: string; exampleHref?: string }): string {
  // Off-type status falls back to the FAIL-CLOSED entry, never the reassuring
  // "Not recognized" (flag 23).
  //
  // AND THE SENTENCE IS PICKED AT THE LEVEL IT IS TRUE AT. "Claimed — This
  // creator controls this account." is a claim about the ACCOUNT; `claimed` is a
  // fact about a PASSPORT. `res.confirmed` is the matched entry's own answer
  // about THIS account, and `=== true` rather than a truthy read because absence
  // must never be read as confirmation.
  const meta = checkVerdict(res.status, res.confirmed === true);
  // THE REASON THE CLASSIFIER ALREADY COMPUTED. `classifyInput` returns
  // "Type a handle, a profile link, or a Dhwaja id." for an EMPTY box and
  // "That's too long to be a handle or link." for a 2 KB paste, and the
  // controller used to throw both away and render the generic "That link isn't
  // a supported creator profile." — a sentence about a link, at somebody who
  // submitted nothing. Only the `unsupported` verdict has a reason to carry,
  // and it never replaces a real verdict's blurb.
  // I3-0122: a warning's blurb is its KIND's wording (the fixed sentence when the entry carried no known code).
  const blurb = res.status === "unsupported" && typeof opts?.reason === "string" && opts.reason.length > 0 ? opts.reason : res.status === "warning" ? warningWordingFor(res.warningKind).text : meta.blurb;
  // ONE display rule, shared with the server-rendered pages and both native
  // apps (`accountDisplay`): a Tier-2 `link` is a URL, not a platform, so it
  // reads "web link · livejasmin.com/…" and never "link · @livejasmin.com/…".
  // The `@` form is how this product spells "a handle on a platform we model",
  // and a link is by definition on a platform we do not model.
  // WHAT THE SUBJECT IS, beside the verdict and never instead of it. A character
  // carries the same verdicts a person does, so without this the card says
  // "Verified" about an AI persona and nothing else — which is the whole reason
  // the kind is on the wire at all. Neutral by construction: `KIND_STYLE` takes
  // no verdict hue and no verdict glyph.
  const kindMeta = res.kind ? KIND_STYLE[res.kind] : null;
  const kindChip = kindMeta ? `<p class="kindchip"><span class="ico" aria-hidden="true">${xml(kindMeta.glyph)}</span> ${xml(kindMeta.label)}</p><p class="kindplain">${xml(kindMeta.plain)}</p>` : "";
  const shown = res.platform && res.handle ? accountDisplay(res.platform, res.handle) : null;
  const who = shown ? `<p class="who"><span class="plat">${textClamp(shown.platform, 40)}</span> <span class="hnd">${textClamp(shown.ident, 80)}</span></p>` : "";
  // Find-the-real-one: a Reported verdict with a validated pointer also shows
  // WHO is being impersonated and routes to the real verified profile. pspLink
  // makes a hostile/malformed id inert text; the name is escaped + clamped.
  // S84 funnel: the k-anon check never returns a passport id for a positive hit,
  // so the one high-intent conversion here is the warning path — the fan was just
  // protected; offer to SAVE the real creator (we have its validated id) and open
  // their Passport. The button gates on sign-in in the controller.
  const pointered = (res.status === "warning" || res.status === "caution") && res.real;
  const saveVerb = res.status === "warning" ? "real" : "verified";
  // I3-0409: the save button carries the safety-mail disclosure beside it (SAVE_SAFETY_MAIL_HINT).
  const realSave = pointered && res.real && PSP_ID.test(res.real.p) ? `<button data-act="save" data-psp="${xml(res.real.p)}" data-name="${xml(clampPlain(res.real.n, 40))}" class="ds-btn act primary">Save the ${saveVerb} ${textClamp(res.real.n, 40)}</button>${SAVE_SAFETY_MAIL_HINT}` : "";
  // A warning asserts impersonation and redirects to safety; a caution is softer —
  // "isn't proven and resembles a verified creator; the real one MAY be here."
  const real = !pointered
    ? ""
    : res.status === "warning"
      ? `<div class="realone"><p class="blurb">This account appears to be impersonating a verified creator.</p><p class="realwho">The real ${textClamp(res.real!.n, 64)} is on Dhwaja: ${pspLink(res.real!.p, "view the verified profile →")}</p><div class="prel">${realSave}</div></div>`
      : `<div class="realone soft"><p class="blurb">This account isn't proven and resembles a verified creator. This is a caution, not a published warning — verify before trusting it.</p><p class="realwho">The verified ${textClamp(res.real!.n, 64)} may be on Dhwaja: ${pspLink(res.real!.p, "view the verified profile →")}</p><div class="prel">${realSave}</div></div>`;
  // Y6 — WHEN. No verdict in this product carried a date, so a screenshot of a
  // green answer was indistinguishable from a live one and the staleness was
  // Dhwaja's problem rather than visibly the screenshot's.
  //
  // TWO CLOCKS, and they are different facts. "Checked" is THIS device's, and it
  // needs nothing from the server because the lookup ran here. "Data from" is
  // the registry's own build clock off `/v1/info`, which is the one a fan cannot
  // infer: a registry generation starts every six hours and is live within
  // about twelve (ADR-0012, D-0436), behind a five-minute edge cache, so a
  // true answer can still be hours old — up to about twelve, not six.
  //
  // The second line is OMITTED when the client has no build clock — an older
  // registry, or a `/v1/info` that failed while a cached width carried the
  // check. Saying nothing about freshness is the one answer that cannot be
  // wrong; inventing "just now" would be the reassurance the fail-closed copy
  // law bans.
  // NOT ON A `failed` RESULT. The same doctrine the build clock already
  // follows: "Couldn't check — nothing was verified" above "Checked 14:03"
  // dates an ATTEMPT as though it were a check, and a screenshot of that reads
  // as a check that happened. Both clocks are suppressed together.
  // The classifier's own reason replaces the plain sentence too, not just the
  // blurb. `unsupported`'s shared sentence says "Dhwaja can't check this kind of
  // link" — a statement about a link, at somebody who submitted an empty box or
  // a 2 KB paste. Same fix the blurb already had, one line above.
  // I3-0122 (D-0995(3)): a warning whose kind is known carries that kind's wording (the blurb) INSTEAD of the fixed
  // sentence; the fixed plain sentence stays only as the fallback for a warning with no known kind.
  const kindWorded = res.status === "warning" && isWarningKindCode(res.warningKind);
  // D-0996(1): a kind-worded warning keeps the kind-neutral safety line in the plain slot (above the blurb).
  const plainLine = res.status === "unsupported" && typeof opts?.reason === "string" && opts.reason.length > 0 ? "" : kindWorded ? WARNING_SAFETY_LINE : meta.plain;
  const dateable = res.status !== "failed" && res.status !== "unsupported";
  const stamp = dateable ? `<span class="asof">Checked ${xml(shortWhen(opts?.checkedAt))}</span>` : "";
  // `shortWhen` returns "" for anything it cannot parse, and the LINE is dropped
  // on that, not merely the date — a "Dhwaja data from" with nothing after it is
  // a freshness claim with no freshness in it. The registry client already
  // shape-checks this field, so this is the second of two gates; the view has to
  // hold on its own because it is also called directly by tests and by the SSR
  // path, neither of which goes through that client.
  const builtWhen = typeof res.builtAt === "string" ? shortWhen(res.builtAt) : "";
  // I3-1374 b/web (I23, the extension's rule — apps/extension/src/registry-client.ts builtAtIsStale): data built
  // 12h or more before the check instant says so instead of printing its date as fresh. Never gates the verdict;
  // an unparseable stamp or a build stamped after the check is not asserted stale.
  const builtT = typeof res.builtAt === "string" ? Date.parse(res.builtAt) : Number.NaN;
  const checkT = opts?.checkedAt !== undefined ? new Date(opts.checkedAt).getTime() : Date.now();
  const builtStale = Number.isFinite(builtT) && Number.isFinite(checkT) && checkT - builtT >= BUILT_AT_STALE_MS;
  const built = dateable && builtWhen ? `<span class="asof">Dhwaja data from ${xml(builtWhen)}${builtStale ? BUILT_AT_STALE_SUFFIX : ""}</span>` : "";
  return `<section class="checkcard ${meta.cls}">${VERDICT_H1_STYLE}
<p class="crumb">Checked: <code>${textClamp(input, 80)}</code></p>
<div class="verdict ${meta.cls}">${verdictIcon(meta)}<h1 class="label">${xml(res.status === "warning" ? warningLabelFor(res.warningKind) : meta.label)}</h1></div>
${plainLine ? `<p class="plain">${xml(plainLine)}</p>` : ""}
${plainLine && res.status === "verified" ? `<p class="plain conduct">${xml(VERIFIED_CONDUCT_NOTE)}</p>` : ""}
${kindChip}
${who}
${stamp || built ? `<p class="stamps">${stamp}${built}</p>` : ""}
<p class="blurb">${xml(blurb)}</p>
${res.status === "unclaimed" ? `<p class="hint">Nobody has claimed this listing yet, so nothing on it is confirmed by the creator. If you'd like them on Dhwaja properly, fans can ask for a creator — the more who ask, the stronger the case we can put to them.</p>` : ""}
${real}
${checkShareLink(res, opts?.ssrBase)}
<form data-act="search" class="search mini"><input name="q" type="search" autocomplete="off" spellcheck="false" placeholder="Check another handle or profile link" aria-label="Check another account"><button type="submit" class="ds-btn act primary">Find</button></form>
${safetyActions(res.status, opts?.howHref, opts?.exampleHref)}
${installRow(opts?.install)}
</section>`;
}

/** Codex completion: a bounded "X of N collected" progress meter over a
 * creator's published set (the fan-private `progress` the archive read already
 * computes for a signed-in viewer). `owned >= total` with a real archive
 * (`total >= 3`, matching the server-side archive-completion badge gate) is the
 * COMPLETE state; below that it's a live progress bar. Purely a render of an
 * existing read — no set is enumerated client-side. The width is a clamped
 * integer percent (0–100), so the inline style can't be steered by data. */
const COMPLETION_MIN = 3;
/** I3-2097 — the COMPLETE state's icon: a filled check-circle drawn inline (colours on the SVG itself, so it needs no stylesheet rule), aria-hidden because the sentence beside it is the spoken form. */
const COMPLETE_ICON = `<svg class="cmico" width="16" height="16" viewBox="0 0 16 16" aria-hidden="true" focusable="false" style="vertical-align:-3px;margin-right:6px"><circle cx="8" cy="8" r="8" fill="#0a7d34"/><path d="M4.5 8.3l2.3 2.3 4.7-4.9" fill="none" stroke="#ffffff" stroke-width="1.8" stroke-linecap="round" stroke-linejoin="round"/></svg>`;
function completionMeter(p?: { owned: number; total: number }): string {
  if (!p || typeof p.total !== "number" || p.total <= 0) return "";
  const total = Math.floor(p.total);
  const owned = Math.max(0, Math.min(Math.floor(typeof p.owned === "number" ? p.owned : 0), total));
  const pct = Math.max(0, Math.min(100, Math.round((owned / total) * 100)));
  const done = owned >= total && total >= COMPLETION_MIN;
  const label = done
    ? `${COMPLETE_ICON}Archive complete — you hold all <b>${num(total)}</b>`
    : `Your collection: <b>${num(owned)}</b> / ${num(total)} · ${num(pct)}%`;
  return `<div class="cmeter${done ? " done" : ""}"><p class="progress">${label}</p><div class="cbar" role="progressbar" aria-valuenow="${num(pct)}" aria-valuemin="0" aria-valuemax="100" aria-label="Collection completion"><span style="width:${num(pct)}%"></span></div></div>`;
}

export function archiveGrid(a: ArchiveDto): string {
  const list = arr<ArchiveDto["items"][number]>(a?.items);
  if (list.length === 0) return `<div class="empty">Nothing in the public archive yet.</div>`;
  const prog = completionMeter(a.progress);
  const items = list
    .slice(0, 200)
    .map((it) => {
      // S91: prefer Dhwaja-STORED signed art (content-addressed, immutable) over the
      // creator's external cover. art_hash is shape-validated before it forms a URL.
      const storedArt = typeof it.art_hash === "string" && /^[0-9a-f]{64}$/.test(it.art_hash) ? `${apiBase()}/media/${it.art_hash}/thumb` : null; // RC-I3-0802 (I30-SAFETY): the grid draws the thumbnail variant
      // I3-0811 (D-0993(1)): no fallback to the creator's external `cover_ref` — a remote host would see which fans
      // browse which release. Only Dhwaja-stored art loads; otherwise the placeholder.
      const cover = safeHref(storedArt);
      const coverEl = cover ? `<div class="cover"><img src="${xml(cover)}" alt="" loading="lazy" referrerpolicy="no-referrer"></div>` : `<div class="cover ph" aria-hidden="true"></div>`;
      const fulfill = safeHref(it.fulfillment_url);
      const get = fulfill ? `<a class="get" href="${xml(fulfill)}" rel="nofollow noopener" target="_blank">Get ↗</a>` : "";
      const owned = it.owned ? `<span class="owned">◈ Owned</span>` : it.claimable ? `<span class="claimable">○ Claimable</span>` : "";
      const when = it.released_at ? xml(hx(it.released_at, 10)) : "";

" · " + when : ""}</span><span class="prov">prov ${xml(hx(it.manifest_hash, 10))}</span><span class="row2">${owned}${get}</span></figcaption></figure>`;

  // so with `truncated` (releases.ts publicArchive); a creator's storefront may

/**
 * flag 2 — the public signed FEED tab: the creator's public passkey-signed
 * broadcasts and published drops, newest first, readable with no account. Its
 * job is the anchor line underneath — if a message claiming to be this creator
 * isn't here, it isn't theirs.
 *
 * Hostile-input posture: the server sanitizes a message before it is SIGNED (no
 * angle brackets, no control characters) and this renderer escapes it AGAIN via
 * textClamp — two independent layers, so neither is alone load-bearing. Nothing
 * in a feed item reaches an href; the only URL is the content-addressed art
 * hash, and only after a strict 64-hex shape check plus the https allowlist.
 */
/**
 * CL:D-04 — the embeddable badge's ALTERNATIVE TEXT, from the passport's own
 * class. It was a static "Verified on Dhwaja" literal inside the template
 * string (only `badgeHref` interpolated), so a CLAIMED creator pasted a snippet
 * onto their own site whose alt text — the sentence a screen reader speaks and
 * the sentence a crawler indexes — said Dhwaja had verified them. The image it
 * labels says "Claimed on Dhwaja" (`services/trust-core/src/badge.ts:62`), so
 * the alt contradicted the picture and overstated the fact. INV-013: CLAIMED is
 * never equivalent to VERIFIED. INV-022: a label states exactly the passport's
 * current class.
 *
 * Mirrors badge.ts's own LABEL table. A badge value this client does not know
 * falls to the WEAKER word rather than the stronger one and rather than an
 * empty alt: an unknown class must never be spoken as "Verified", and an image
 * with no alt is its own accessibility defect.
 */
function badgeAltText(badge: unknown, rank?: unknown): string {
  const word = badge === "verified" || badge === "founding-verified" ? "Verified on Dhwaja" : "Claimed on Dhwaja";
  // I3-0526: the image draws the founding rank (badge.ts rankEl), so its alt says it too.
  const r = foundingRankOf(badge, rank);
  return r === null ? word : `${word}, ${foundingLabelOf(r)}`;
}

/**
 * I3-0526 (D-1078(13)) — Founding is a MODIFIER shown WITH its rank, never folded away. The server's rule
 * (`validFoundingRank`, services/trust-core/src/badge.ts): a `founding-` badge and an integer rank in
 * 1..FOUNDING_CAPACITY (founding.ts, 500). The words are the console's approved "Founding Creator #N".
 */
export const FOUNDING_CAPACITY = 500;
export function foundingRankOf(badge: unknown, rank: unknown): number | null {
  if (typeof badge !== "string" || !badge.startsWith("founding-")) return null;
  return typeof rank === "number" && Number.isInteger(rank) && rank >= 1 && rank <= FOUNDING_CAPACITY ? rank : null;
}
export function foundingLabelOf(rank: number): string {
  return `Founding Creator #${rank}`;
}

function feedList(feed: FeedDto | null, claimed: boolean): string {
  // PS-232 — THE ANCHOR IS AN ARGUMENT FROM COMPLETENESS, AND AN UNCLAIMED
  // PASSPORT HAS NO COMPLETENESS TO ARGUE FROM. A creator who has not claimed
  // their passport holds no passkey, so they CANNOT sign a broadcast or a drop:
  // their feed is empty by construction, always, and treating that absence as
  // proof of fakery turned that structural emptiness into evidence against them — a
  // negative claim about a person built out of a fact about Dhwaja's own
  // enrolment state (INV-012: a proof of ownership is not a safety claim, and
  // its ABSENCE is not an unsafety claim). The anchor is therefore gated on
  // `p.tabs.card` (the same fold the badge reads), and the unclaimed branch says
  // what is actually true instead — including, explicitly, that it proves
  // nothing either way. The unreadable branch below is untouched: it keeps
  // getting neither line.
  //
  // FU-03 (S-068) — AND A BOUNDED WINDOW HAS NO COMPLETENESS EITHER. PS-232
  // removed the anchor from the two cases where completeness was impossible
  // (unclaimed, unreadable) but left it on the ordinary successful read, where
  // it is equally unearned: `services/trust-core/src/feed.ts` serves at most
  // FEED_LIMIT = 20 combined items (`feed.ts:60,96`) with NO cursor
  // (`api.ts:270-271`) and no older-items control here, and it EXCLUDES every
  // followers-only broadcast at read time by design (`feed.ts:118-124`). So the
  // rendered list is a WINDOW, never a census — and treating an absence from it
  // as proof of fakery inferred a universal negative from it. A creator with 21 public
  // signed broadcasts, or one followers-only update, had authentic speech this
  // page called fake: the product's purpose inverted, aimed at the creator it
  // exists to protect.
  //
  // The fix is to stop making the claim, not to widen the window — an inference
  // from a bounded read is unsound at ANY bound, and the followers-only filter
  // (which must remain) breaks it even at one item. What survives is the half
  // that IS earned: everything shown here is passkey-signed, and the fan is told
  // what the list covers. This is the posture the product's own registry verdict
  // already takes for an absence (`apps/ios/App/Verdict.swift:269`,
  // `apps/android/.../Verdict.kt:292`: "That doesn't make it fake — it means we
  // can't help you here."); the Feed was the only surface in the tree
  // contradicting it. An authoritative "was this exact message signed by them"
  // lookup is the surface that would earn a categorical answer; it does not
  // exist, and inventing it is not this room's call.
  const anchor = claimed
    ? `<p class="feedanchor">Recent public signed updates. Followers-only posts never appear here and older ones may not, so something missing from this list is not proof it's fake — ask the creator, or check the account on Dhwaja.</p><p class="feednote oneway">Broadcasts are one-way and signed - creators never DM you through Dhwaja.</p>`
    : `<p class="feednote">This creator can't publish signed messages until they claim their passport — nothing here proves or disproves anything.</p>`;
  // THE ANCHOR LINE IS WHY THIS TAB CANNOT FAIL SOFT INTO ITS EMPTY STATE.
  //
  // `api.feed()` collapses offline, 5xx, 404 and an unparseable body into one
  // `null`, and `null` used to render "Nothing public from this creator yet."
  // followed by a categorical claim that absence proves fakery — so a 500 on
  // `/p/:id/feed` told the fan (a) as a FACT that this creator has published
  // nothing signed, and (b) that the signed broadcast they were holding in
  // another tab was an impersonation. An outage re-labelled a real creator's
  // real message as fake. That is the exact inverse of what this tab is for.
  //
  // A feed we could not read gets the unreadable card and NO ANCHOR: the anchor
  // is an argument from completeness, and we have no completeness to argue from.
  if (feed === null || feed === undefined || unreadable(feed.items)) return unreadableCard("this creator's signed feed");
  // A shape we don't know is not rendered at all — never an empty "✓ Signed"
  // row, which would be a trust mark with nothing behind it. Filtered BEFORE
  // the empty check, so an all-unknown payload shows the honest empty state.
  const list = arr<FeedDto["items"][number]>(feed.items).filter((it) => it && (it.kind === "broadcast" || it.kind === "drop"));
  if (list.length === 0) return `<div class="empty">Nothing public from this creator yet.</div>${anchor}`;
  const rows = list
    .slice(0, 20)
    .map((it) => {
      const when = `<span class="fwhen">${xml(hx(it.ts, 10))}</span>`;
      if (it.kind === "drop") {
        const art = typeof it.art_hash === "string" && /^[0-9a-f]{64}$/.test(it.art_hash) ? safeHref(`${apiBase()}/media/${it.art_hash}/thumb`) : null; // RC-I3-0802 (I30-SAFETY): feed thumbnail variant
        const thumb = art ? `<img class="fart" src="${xml(art)}" alt="" loading="lazy" referrerpolicy="no-referrer">` : "";
        return `<li class="fcell drop"><span class="fkind">Drop</span>${thumb}<span class="ftext">${textClamp(it.name, 120)}</span>${when}</li>`;
      }
      // A broadcast is the creator's own voice — marked signed, because "signed
      // with their passkey" is the entire reason this row counts as evidence.
      return `<li class="fcell bcast"><span class="fkind" title="Signed with the creator's passkey — Dhwaja doesn't check what it says.">From the creator</span><span class="ftext">${textClamp(it.message, 280)}</span>${when}</li>`;
    })
    .join("");
  return `<ul class="feed">${rows}</ul>${anchor}`;
}

function linksList(links: PresentationDto["tabs"]["links"]): string {
  const list = arr<PresentationDto["tabs"]["links"][number]>(links);
  if (list.length === 0) return `<div class="empty">No linked accounts yet.</div>`;
  const rows = list
    .slice(0, 100)
    .map((a) => {
      // I3-1136 (D-1082(3)): an Open link only for a url that names THIS account (@dhwaja/schema urlNamesAccount — the
      // write door's predicate, and the SSR page's) — a stored url naming another account renders no link.
      const safe = safeHref(a.url);

      const link = href ? `<a href="${xml(href)}" rel="nofollow noopener ugc" target="_blank">open ↗</a>` : "";

      // console both print "○ listed by Dhwaja" for exactly this case; matched
      // verbatim so the SPA doesn't invent a fourth wording for one fact.
      const mark =
        a.touched_by === "creator"

          : a.touched_by === "agency"
            ? `<span class="mark agency">by agency</span>`
            : `<span class="mark none">○ listed by Dhwaja</span>`;

      // field from an older worker fails closed to "unconfirmed", never to a

      const confirm = a.confirmed !== true ? `<span class="mark unconfirmed" title="Nobody has demonstrated control of this account for this passport — or the last proof has expired. It is listed, not confirmed.">! not confirmed</span>` : "";

/**
 * WHAT THE VIEWER HAS ALREADY DONE WITH THIS CREATOR — three-valued on purpose.
 *
 * `true` / `false` are answers; `null` is "we did not read it", and the row
 * below says so in words rather than rendering a button that quietly asserts
 * the negative. Signed out is also `null`: nothing to read.
 */
export interface ViewerRelation {
  saved: boolean | null;
  collected: boolean | null;
}

/**
 * `GET /account/saves` and `GET /account/collection` are capped server-side
 * (`collections.ts` LIST_LIMIT). A hit is a fact at any length; a MISS is only a
 * fact if the page was not full, because the id could be on a page we never
 * asked for. A full page therefore yields `null` — unknown — not `false`.
 */
const LIST_PAGE_CAP = 200;

function inPage(ids: readonly string[] | null, id: string, rowsSent: number): boolean | null {
  if (ids === null) return null; // the read failed: not an empty collection
  if (ids.includes(id)) return true;
  // I3-1262 — THE CAP IS A FACT ABOUT THE PAGE THE SERVER SENT, so it is measured
  // on that count and never on the subset this module could parse an id out of.
  // `ids()` below drops any row without a string `passport_id`; measuring the cap
  // on the SHORTENED list turns "unknown" into a confident "no" on a page that was
  // full, which is the one answer this function exists to refuse to give.
  return rowsSent >= LIST_PAGE_CAP ? null : false;
}

/**
 * Derive the viewer's own save/collect state for one passport from the two
 * owner-scoped list reads. Pure; the controller does the fetching.
 *
 * WHY THIS EXISTS. `data-on="0"` was a HARDCODED LITERAL on the Collect button
 * and the passport route read nothing about the viewer, so:
 *   - a fan who collected a card yesterday came back to a button that said
 *     "Collect card", which is simply false; and
 *   - the uncollect undo — added because "a collect is a VOTE on four public
 *     boards, with no undo" — was only ever rendered by the CONTROLLER after a
 *     successful collect in that page session. On a fresh load the only route
 *     to it was to press Collect again (a server-side no-op) and then press the
 *     undo that appeared. Nobody discovers that.
 * A malformed body is `null` at the call site and lands on `null` here, which
 * renders the "we couldn't check" line rather than a confident wrong answer.
 */
export function viewerRelation(
  saves: Array<{ passport_id?: unknown }> | null,
  collection: Array<{ passport_id?: unknown }> | null,
  passportId: string,
): ViewerRelation {
  const ids = (rows: Array<{ passport_id?: unknown }> | null): string[] | null =>
    Array.isArray(rows) ? rows.map((r) => (typeof r?.passport_id === "string" ? r.passport_id : "")).filter((s) => s.length > 0) : null;
  return {
    saved: inPage(ids(saves), passportId, saves?.length ?? 0),
    collected: inPage(ids(collection), passportId, collection?.length ?? 0),
  };
}

/**
 * PS-121 / PS-245 — the SAME hardcoded-zero defect `viewerRelation` above was
 * written to fix, on the Follow button instead of Save/Collect. `GET
 * /account/following` (`api.ts` — `FollowEntryDto[]`) is the owner-scoped read
 * that exists for exactly this; nothing calling `profileView` reads it, so the
 * public profile's Follow button carries `data-on="0"` unconditionally,
 * regardless of whether the viewer already follows this handle.
 *
 * Pure, mirroring `inPage`'s page-cap rule: `GET /account/following` is capped
 * server-side the same as saves/collection, so a MISS on a full page is unknown
 * (`null`), never a confident "not following" — and a failed read (`null` list)
 * is `null` too. Handle comparison is case-insensitive because the handle
 * namespace is (`[a-z0-9_]{3,20}`, always lowercased server-side).
 */
export function viewerFollowState(following: Array<{ handle?: unknown }> | null, handle: string): boolean | null {
  if (following === null) return null;
  const want = handle.toLowerCase();
  const handles = following.map((r) => (typeof r?.handle === "string" ? r.handle.toLowerCase() : "")).filter((s) => s.length > 0);
  if (handles.includes(want)) return true;
  // Same measurement as `inPage`, and it is a SECOND call site of the same rule:
  // the cap is tested against the rows `GET /account/following` sent, never
  // against the handles this module could parse out of them.
  return following.length >= LIST_PAGE_CAP ? null : false;
}

/**

 * warned, takedown and suppression; this client only renders what that projection served). Parsed FAIL-CLOSED:
 * an off-shape field is dropped, never coerced. Every link is https only (`safeHref`), opens in a new tab and

 * The avatar is the API origin's `/media/<hash>`; no configured base -> no picture, never a relative src.
 * The report door posts one field and one category (`POST /p/:id/profile/report`).
 */
export const CREATOR_PROFILE_REL = "nofollow ugc noopener noreferrer";
export const CREATOR_PROFILE_NOTE = "Written by the creator, not by Dhwaja.";
/** D-1057(4): child_safety carries report.ts CATEGORY_CHOICES' label and blurb (approved, D-1365 W52), read
 * from that one entry; creator-profile.ts REPORT_CATEGORIES accepts it after I30-FX-SAFETY's migration. */
const CPROF_CHILD_SAFETY = CATEGORY_CHOICES.find((c) => c.key === "child_safety");
const CPROF_REPORT_CATEGORIES: ReadonlyArray<readonly [string, string]> = [["scam", "Scam"], ["impersonation", "Impersonation"], ["offensive", "Offensive"], ...(CPROF_CHILD_SAFETY ? [["child_safety", CPROF_CHILD_SAFETY.label] as const] : []), ["other", "Something else"]];
export function creatorProfileHtml(psp: string, raw: unknown): string {
  if (!raw || typeof raw !== "object" || !/^psp_[0-9a-f]{16}$/.test(psp)) return "";
  const r = raw as { bio?: unknown; links?: unknown; avatar?: unknown };
  const bio = typeof r.bio === "string" && r.bio.trim() !== "" ? r.bio.slice(0, 300) : null;
  const links: Array<{ href: string; host: string }> = [];
  for (const l of Array.isArray(r.links) ? r.links.slice(0, 5) : []) {
    const u = l && typeof l === "object" ? (l as { url?: unknown; host?: unknown }) : null;
    const href = u && typeof u.url === "string" ? safeHref(u.url) : null;
    let host = "";

      host = href ? new URL(href).hostname : "";
    } catch {
      host = "";
    }
    if (href && host) links.push({ href, host });
  }

  const base = apiBase();
  const avatarSrc = typeof av === "string" && /^[0-9a-f]{64}$/.test(av) && base ? safeHref(`${base}/media/${av}`) : null;
  if (!bio && links.length === 0 && !avatarSrc) return "";
  const fields: Array<readonly [string, string]> = [...(bio ? [["bio", "Bio"] as const] : []), ...(links.length ? [["links", "Links"] as const] : []), ...(avatarSrc ? [["avatar", "Picture"] as const] : [])];

/**
 * A public count the server is ALLOWED TO WITHHOLD, rendered as text or null.
 *
 * Deliberately NOT `num()`: `num()` exists to make geometry safe and answers
 * "what number do I draw?" with a fallback, which is exactly the wrong answer
 * for a count. `null` here means "the server declined to publish this", and the
 * caller must render nothing (or an em dash) rather than a number nobody
 * asserted. Anything that is not a finite number — undefined, a string, NaN —
 * is treated as absent for the same reason.
 */
export function publicCount(n: unknown): string | null {
  return typeof n === "number" && Number.isFinite(n) ? String(Math.trunc(n)) : null;
}

/** The order the lookup screen offers platforms in — majors first, then the
 * rest of the canonicalizer's vocabulary alphabetically. Derived from
 * PLATFORMS so a new platform shows up here without an edit. */
const LOOKUP_PLATFORM_ORDER = ["x", "instagram", "tiktok", "youtube", "twitch", "onlyfans", "fansly", "reddit"] as const;

/**
 * M24: the lookup screen checks a bare HANDLE, so it must only offer platforms
 * that HAVE handles. A pid-only platform (e.g. manyvids, whose profiles are
 * `/profile/<numeric id>`) has no handle namespace, so a handle check there can
 * never match — offering it is a dead chip. A platform is handle-capable when
 * its extractor is handle-based (first-segment / subdomain / youtube / facebook)
 * or, for a prefix platform, when at least one prefix yields a `handle`.
 */
export function platformSupportsHandleLookup(p: string): boolean {
  const spec = (PLATFORMS as Record<string, { extract?: { kind: string; namespaced?: boolean; prefixes?: ReadonlyArray<{ idKind: string }> } }>)[p];
  const ex = spec?.extract;
  if (!ex) return false;
  // PU-02 (S-070): A NAMESPACED PLATFORM HAS NO BARE HANDLE TO CHECK. For
  // linkedin, rumble and pornhub the matched prefix IS part of the subject
  // (`tryNamespaced`, packages/canonical/src/index.ts), so the id this product
  // actually ISSUES for `linkedin.com/in/nova` is `in/nova` — and `in/nova`,
  // `company/nova` and bare `nova` are three different identifiers. The chip
  // submitted the unchanged slug, so a correctly listed full-profile account
  // could be answered "Not recognized" through the ADVERTISED shortcut, with
  // categorical no-listing wording, after a lookup that never identified an
  // account at all. That is the same dead chip M24 already excludes for a
  // pid-only platform, and it is worse: it is dead while looking answered.
  //
  // NOT youtube: `/c/<name>` and `/user/<name>` are namespaced there, but
  // `@handle` is the un-namespaced PRIMARY form and is exactly what a bare
  // handle means — so its chip stays. NOT reddit either: its `u`/`user`
  // prefixes are ALIASES of one namespace (no `namespaced` flag on its extract
  // rule), so a bare handle names one account.
  //
  // The full-profile URL box above the picker keeps the namespace and remains
  // the supported way to check these; it is the one this screen already points
  // at ("If you have the full profile link, pasting it in the box above is even
  // better"). Collecting the space here instead would be a new product choice —
  // a second control on the front door of the checker — and this is the
  // smallest edit that stops the wrong answer.
  if (ex.kind === "prefix") return !ex.namespaced && (ex.prefixes ?? []).some((x) => x.idKind === "handle");
  return true;
}

export function lookupPlatforms(): string[] {
  const known = Object.keys(PLATFORMS).filter(platformSupportsHandleLookup);
  const head = LOOKUP_PLATFORM_ORDER.filter((p) => known.includes(p));
  const tail = known.filter((p) => !(LOOKUP_PLATFORM_ORDER as readonly string[]).includes(p)).sort();
  return [...head, ...tail];
}

/**
 * The bare-handle DISAMBIGUATION screen (v14, external review P0). Typing
 * "@rileymoon" into the front door of an authenticity checker used to land on
 * whichever COLLECTOR registered that handle. This screen restores the
 * product's meaning: the creator check is the primary action; the Dhwaja
 * collector profile is offered second, labeled as exactly what it is.
 */
/**
 * The collector-profile offer at the bottom of the lookup screen.
 *
 * The two namespaces are not the same shape, and widening the handle this
 * screen accepts is what makes that visible: a Dhwaja collector username is
 * `[a-z0-9_]{3,20}`, so `mr.beast` — a perfectly legal platform handle — can
 * NEVER be a collector. Emitting `#/u/mr.beast` would render a link that routes
 * to the home page, which is exactly the "silently collapse to home" failure the
 * shared id shapes exist to stop. Say the shape rule instead; it discloses
 * nothing (the namespace rule is public) and it is true.
 */
function collectorLookupLink(handle: string): string {
  return HANDLE_ID.test(handle)
    ? navLink(`#/u/${handle}`, `View collector profile @${handle} (a fan — NOT a creator)`, "btnlink muted")
    : `<span class="muted">No collector profile can hold this handle — Dhwaja usernames are 3–20 letters, digits or underscores.</span>`;
}

export function lookupView(handle: string): string {
  // THE SHAPE THIS SCREEN ACCEPTS IS THE PLATFORM ONE, not Dhwaja's own
  // username namespace. `router.ts` widened the ROUTE to `LOOKUP_HANDLE_RE`
  // (Instagram/TikTok/YouTube handles carry dots and hyphens and run to 30
  // characters) and this view kept testing `HANDLE_ID`, the 3–20
  // letters-digits-underscore COLLECTOR shape — so `#/lookup/mr.beast` routed
  // correctly and then rendered "That doesn't look like a handle." at somebody
  // whose handle is exactly that.
  if (!LOOKUP_HANDLE_RE.test(handle)) return errorView("That doesn't look like a handle.");
  const buttons = lookupPlatforms()
    .map((p) => `<button class="ds-btn act" data-act="lookup-check" data-platform="${xml(p)}" data-handle="${xml(handle)}">${xml(p)}</button>`)
    .join(" ");
  return `<section class="lookup">
<h2>Who is <span class="handle">@${xml(handle)}</span>?</h2>
<h3>Check a creator</h3>
<p>Pick the platform you saw <b>@${xml(handle)}</b> on — the check runs on your device, and only a short anonymous hash prefix of the link is sent. If you have the full profile link, pasting it in the box above is even better.</p>
<div class="lookup-platforms">${buttons}</div>
<hr />
<h3>Dhwaja collector profile</h3>
<p class="muted">A <b>collector profile</b> is a fan account on Dhwaja — <b>not a creator Passport</b>, and not identity-verified. If a collector registered @${xml(handle)}, their page is here:</p>
<p>${collectorLookupLink(handle)}</p>
</section>`;
}

/** I3-2022 — the creators this fan Scouted (server-gated by the profile's visibility). Names and labels only; a withheld (`null`) demand figure is never printed. */
function publicScoutsLine(rows: unknown): string {
  const list = arr<{ label?: unknown; creator_name?: unknown; arrived?: unknown }>(rows as never).filter((r) => typeof r?.creator_name === "string" && typeof r?.label === "string").slice(0, 20);
  if (list.length === 0) return "";
  return `<p class="pubscouts"><b>Scouted</b> ${list.map((r) => `${textClamp(String(r.creator_name), 40)} <span class="muted">(${textClamp(String(r.label), 40)}${r.arrived === true ? ", arrived" : ""})</span>`).join(" \u00b7 ")}</p>`;
}

export function profileView(dto: ProfileDto | null, read?: Read<unknown>, viewerFollowing?: boolean | null): string {
  if (!dto) return saidNotFound(read) ? `<section class="profile"><div class="empty">No such profile.</div></section>` : unreadableObject("that profile", "profile unread");
  const collItems = arr<{ acquisition_id: string; passport_id: string; title: string; release_id?: string }>(dto.collection?.items);
  // M75: a fan can hold several acquisitions of the SAME release. Rendered one
  // per row they read as an accidental duplication; group them, link the first,
  // and show a "× N" multiplier when more than one is held. The header count
  // stays the true acquisition total — the grouping is visual only.
  //
  // FU-09 (S-068) — AND THE KEY WAS `${passport_id}|${title}`, WHICH IS NOT
  // RELEASE IDENTITY. `migrations/0007_releases_credentials.sql:3-9` gives a
  // release its own `release_id` beside `title`, `edition` and `version`; a
  // title is neither the identity nor a unique key, and one creator publishing
  // two editions under one display title is ordinary. The reader supplies
  // `release_id` (`social.ts:1261-1270`) and the DTO carries it (`api.ts:1500`);
  // this narrowing simply dropped it, so two DIFFERENT releases collapsed into
  // one "× 2" row linked only to the first acquisition — the second release's
  // verification receipt was unreachable from the collection.
  //
  // Group on the RELEASE, and when a row carries no release_id fall back to its
  // own acquisition id rather than to the title: a row whose identity we cannot
  // establish is never merged with another, because merging on weaker evidence
  // than release identity is the defect itself. The ordinary duplicate-copies
  // case M75 was written for is unchanged.
  const collGroups: Array<{ acquisition_id: string; passport_id: string; title: string; release_id?: string; count: number }> = [];
  const collSeen = new Map<string, number>();
  for (const c of collItems) {
    const key = typeof c.release_id === "string" && c.release_id.length > 0 ? `rel:${c.release_id}` : `acq:${c.acquisition_id}`;
    const at = collSeen.get(key);
    if (at === undefined) {
      collSeen.set(key, collGroups.length);
      collGroups.push({ ...c, count: 1 });
    } else {
      collGroups[at]!.count += 1;
    }
  }
  const coll = dto.collection_visible
    ? collGroups.length > 0
      ? `<h3>Collection (${num(dto.collection?.total)})</h3><ul class="pcoll">${collGroups
          .slice(0, 100)
          .map((c) => `<li>${acqLink(c.acquisition_id, clampPlain(c.title, 60))}${c.count > 1 ? ` <span class="mult">× ${num(c.count)}</span>` : ""} <span class="muted">${pspLink(c.passport_id, "creator")}</span></li>`)
          .join("")}</ul>`
      : `<p class="empty">This collection is empty.</p>`
    : `<p class="muted">This collection is private.</p>`;
  const achList = arr<{ achievement_id: string; rank?: number }>(dto.achievements);
  const ach = achList.length > 0 ? `<div class="ach">${achList.map((a) => `<span class="badge">${achievementBadgeHtml(a.achievement_id, a.rank)}</span>`).join("")}</div>` : "";
  const recList = arr<{ kind: string; message: string | null; granted_at: string; by_passport_id: string; by_name: string }>(dto.recognitions);
  const rec = recList.length > 0
    ? `<h3>Recognized by creators</h3><ul class="prec">${recList
        .slice(0, 50)
        .map((r) => `<li><b>${xml(recognitionLabel(r.kind))}</b> ${pspLink(r.by_passport_id, clampPlain(r.by_name || "a creator", 40))}${r.message ? ` — <span class="recmsg">“${textClamp(r.message, 200)}”</span>` : ""}</li>`)
        .join("")}</ul>`
    : "";
  // F11 / #5 — the game half of the Fandom Passport: a RECORD, behind its own
  // consent (`war_public`, default off) rather than the collection switch,
  // which was agreed under "your profile and, if you allow it, your
  // collection". A win/loss record is neither of those.
  //
  // Nothing else is here because nothing else is on the wire: no card count
  // (cards come from packs at a fixed credit price, so the integer divides into
  // dollars), no per-match rows (one `war_matches` row means a shared timestamp
  // pairs two public profiles into a result graph), no opponent.
  const w = dto.war;
  const war = w && w.matches > 0 ? `<h3>Creator War</h3><p class="wrec">${xml(String(w.wins))} won of ${xml(String(w.matches))}</p>${warLadderHtml(dto.war_ladder)}` : "";

  // S-070 SP-03 — THE CONSTELLATION, FINALLY RENDERED.
  //
  // The owner's Collection view says "On your public profile, strangers see:
  // connected on N platforms — and you have chosen to name X" (`meView`, the constellation note)
  // and labels each enabled link "shown on your profile" (`meView`, the per-link state).
  // `social.ts:1315` has been putting `constellation` in the public projection
  // since F8 and `fan-social-routes.ts` has been serving it; the DTO omitted the
  // field and this renderer never read it, so the count and the names appeared
  // on no page and the visibility toggle published nothing. The missing
  // renderer, not the TypeScript omission, is what made the promise false.
  //
  // A COUNT, AND ONLY THE NAMES THE FAN PUBLISHED. Never a provider the fan did
  // not name, and never an opaque account identifier — `named` is already
  // exactly the opted-in per-link set the server computed, and nothing here
  // widens it. `size === 0` renders nothing at all: "connected on 0 platforms"
  // is a statement about an absence that the product does not make, and it is
  // not visibility-gated, so silence is the right answer.
  const stars = dto.constellation;
  const starNames = arr<string>(stars?.named).filter((n) => typeof n === "string" && n.length > 0).slice(0, 20);
  const constellation =
    stars && typeof stars.size === "number" && Number.isFinite(stars.size) && stars.size > 0
      ? `<p class="constellation"><b>Connected on ${num(Math.trunc(stars.size))} platform${Math.trunc(stars.size) === 1 ? "" : "s"}</b>${
          starNames.length > 0 ? ` — ${starNames.map((n) => xml(n)).join(", ")}` : ""
        } <span class="muted">Independent sign-ins that vouch for this being a person. Dhwaja never publishes an account name or id from them.</span></p>`
      : "";

  // A WITHHELD count is not a zero. The server returns null for the founder (a
  // size oracle: the platform's own account is the only one with an
  // operational reason to follow broadly, so its following count is a live
  // lower bound on the user base) and `num()` would quietly turn that refusal
  // into "0 followers", which is a false statement about a real account. Both absent ⇒ the line does not exist; one absent ⇒ an em dash,
  // which reads as "not published" and can be mistaken for no quantity.
  const followers = publicCount(dto.follower_count);
  const following = publicCount(dto.following_count);
  const counts = followers === null && following === null ? "" : `<p class="counts"><b>${followers ?? "—"}</b> followers · <b>${following ?? "—"}</b> following</p>`;
  // AUDIT ITEM #25, SAME CLASS — the FOLLOW half of the edge had no button.
  // `POST /account/follow` and `api.follow()` both shipped and nothing on any
  // screen called either: `#/profile` rendered Unfollow for a list that could
  // only ever be populated by the (now deleted) founder auto-follow. A fan
  // could un-follow and never follow. Toggle shape, `data-on`, exactly as the
  // request CTA does, so the undo is one click away from the act.
  // Self-follow is refused by the server (`social.ts` — "cannot follow
  // yourself"), which is why the renderer does not need to know who is looking.
  // PS-121 / PS-245 fix: `data-on` and the label now carry the viewer's actual
  // read-back state (`viewerFollowState`, above) instead of a hardcoded "0" —
  // the same discipline `relationRow` applies to Save/Collect. `viewerFollowing
  // === true` renders the undo; `false` or `null` (signed out, or a failed
  // read) keeps the neutral "Follow" offer, never a confident wrong answer.
  const followingNow = viewerFollowing === true;
  const follow = /^[a-z0-9_]{3,20}$/.test(dto.handle)
    ? `<p class="prel"><button data-act="follow" data-handle="${xml(dto.handle)}" data-on="${followingNow ? "1" : "0"}" class="ds-btn act">${followingNow ? "Following ✓ · undo" : "Follow"}</button> ${rmsg("data-follow-msg")}</p>`
    : "";
  // M42: the featured showcase was WRITE-ONLY — a fan could set it (POST
  // /account/featured, wired) but profileView never rendered dto.featured, so the
  // showcase never appeared on the public profile. The server resolves each item
  // to a live view and drops stale ones; the client shape-checks defensively
  // (dto.featured is unknown[]) so a malformed item renders nothing, never markup.
  const featItems = arr<{ kind?: string; acquisition_id?: string; title?: string; passport_id?: string; card?: { passport_id?: string; display_name?: string } }>(dto.featured);
  const featRows = featItems
    .slice(0, 12)
    .map((f) => {
      if (f.kind === "acquisition" && typeof f.acquisition_id === "string") {
        return `<li>${acqLink(f.acquisition_id, clampPlain(typeof f.title === "string" ? f.title : "collectible", 60))}${typeof f.passport_id === "string" ? ` <span class="muted">${pspLink(f.passport_id, "creator")}</span>` : ""}</li>`;
      }
      if (f.kind === "card" && f.card && typeof f.card.passport_id === "string") {
        return `<li>${pspLink(f.card.passport_id, clampPlain(typeof f.card.display_name === "string" ? f.card.display_name : "a creator", 40))}</li>`;
      }
      return "";
    })
    .join("");
  const featured = featRows ? `<h3>Featured</h3><ul class="pfeat">${featRows}</ul>` : "";
  return `<section class="profile">
${collectorBanner(dto.handle, dto.creator_collision === true, dto.creator_name_collision === true)}
<h2>${textClamp(dto.display_name, 40)} <span class="handle">@${xml(dto.handle)}</span> <span class="badge collector-badge">collector</span></h2>
${dto.bio ? `<p class="bio">${textClamp(dto.bio, 200)}</p>` : ""}
${counts}
${constellation}
${publicScoutsLine(dto.scouts)}
${follow}
${featured}
${ach}
${rec}
${war}
${coll}
</section>`;
}

/** Human label for a recognition kind on a public fan profile (S90b). */
export function recognitionLabel(kind: string): string {
  switch (kind) {
    case "follow":
      return "Followed";
    case "like":
      return "Liked";
    case "thanks":
      return "Thanked";
    case "shout_out":
      return "Shout-out";
    case "spotlight":
      return "Spotlighted";
    default:
      return "Recognized";
  }
}

/** I3-1435: a Founding Fan badge names a rank but never the creator it's for
 * — `founding_fan:<psp>` already carries the psp to the client, so link the
 * badge to that creator's Passport rather than leaving it unclickable text.
 * Every other achievementLabel() case stays plain text: this is the only kind
 * whose id names a specific creator the fan can navigate to. */
export function achievementBadgeHtml(id: string, rank?: number): string {
  const label = achievementLabel(id, rank);
  if (typeof id === "string" && id.startsWith("founding_fan:")) {
    const psp = id.slice("founding_fan:".length);
    if (PSP_ID.test(psp)) return pspLink(psp, label);
  }
  return xml(label);
}

export function achievementLabel(id: string, rank?: number): string {
  if (typeof id !== "string") return "";
  if (id === "first_verified_acquisition") return "First acquisition";
  const collector = /^collector_(\d+)$/.exec(id);
  if (collector) return `Collector · ${collector[1]}`;
  if (id.startsWith("completion:")) return "Archive completionist";
  const milestone = /^milestone:psp_[0-9a-f]{16}:(\d+)$/.exec(id);
  if (milestone) return `Milestone · ${milestone[1]} collected`; // per-creator, psp never shown
  // AUDIT ITEM #34 — the LABEL moves, the ID does not. `early_supporter:<psp>`
  // is a stored `achievement_id` in shipped rows and it is the literal the
  // `scout` leaderboard's suppression subquery matches on (`rewards.ts`), so
  // renaming the id is a data migration plus a SQL rewrite for a word.
  //
  // The word was crossed, though: an achievement called "Early Supporter" fed a
  // board called "Scouts", while a board called "Supporters" counted something
  // else entirely (creator requests). "Supporter" also means a paying Premium
  // fan two screens away. This is the ask-that-landed — you asked for a creator
  // and they came — so it is named for the board it actually feeds.
  if (id.startsWith("early_supporter:")) return "Early Scout";
  // Founding Fan #N: render the rank when the server supplied a sane integer,
  // else the flat badge. The psp in the id is NEVER shown (only the number).
  if (id.startsWith("founding_fan:")) return typeof rank === "number" && Number.isInteger(rank) && rank >= 1 && rank <= 10000 ? `Founding Fan #${rank}` : "Founding Fan";
  const codex = /^codex_(\d+)$/.exec(id);
  if (codex) return `Codex · ${codex[1]} seen`;
  // flag 7: Reporter tiers — a generic count-only label (mirrors the server's
  // reporterLabel). The id carries ONLY the tier; no warning id/case ref/hash
  // ever reaches a badge.
  const reporter = /^reporter:(\d+)$/.exec(id);
  if (reporter) return `Reporter · ${reporter[1]} confirmed`;
  if (id === "scout") return "Scout";
  const scout = /^scout_(\d+)$/.exec(id);
  if (scout) return `Scout · ${scout[1]}`;
  // Fail SAFE: never echo a raw id that embeds a psp (a malformed milestone/
  // completion id must not surface the creator's passport in a badge label).
  if (id.includes("psp_")) return "Milestone";
  return id.slice(0, 40);
}

/**
 * Mock 36 billing states, as PURE copy. Three rules the design pins:
 *   - the welcome names EXACTLY the caps that rose — nothing vague, nothing
 *     about safety, because safety never moved;
 *   - the lapsed state is quiet grey, never red and never shaming (red is
 *     reserved for Reported), and says out loud that protection is unaffected;
 *   - free is not an error state.
 * Nothing here can gate a verdict: it renders copy from an entitlement the
 * server already computed.
 */
export function billingStateCopy(ent: EntitlementDto | null, justUpgraded = false): { label: string; note: string; tone: "quiet" | "good" } {
  const tier = ent?.tier ?? "free";
  const pro = ent?.subject_kind === "creator";
  const product = pro ? "Creator Pro" : "Premium";
  if (justUpgraded && tier !== "free") {
    const raised = pro
      ? "Likes go from 10 to 50 a day, named recognitions from 1 to 5 a day, active follows from 25 to 250, and broadcasts from weekly to daily."
      // H16: state only real Premium effects. D-1059(16): the upgrade page's sentence verbatim (upgrade.ts
      // PREMIUM_BENEFIT; kept as a literal so this verdict module never imports the billing page, and held equal
      // by apps/web/test/i30-fx-client.test.ts).
      : "You can save more creators. Your reports are seen sooner by an operator — a purchased place in the queue, never counting for more. Everything that keeps you safe is already free; Premium is a way to give back.";
    return { label: `Manage ${product}`, note: `Welcome to ${product}. What changed: ${raised} Everything protective was already yours.`, tone: "good" };
  }
  if (ent?.in_grace) {
    return {
      label: `Renew ${product}`,
      note: `Your ${product} payment didn't go through. You keep everything for ${ent.grace_days} days while it sorts out — and your protection isn't affected either way.`,
      tone: "quiet",
    };
  }
  if (tier !== "free") return { label: `Manage ${product}`, note: "", tone: "quiet" };
  if (ent?.purchased_tier) {
    return {
      label: `Restart ${product}`,
      note: `Your ${product} tools are back to the free limits. Checks, verdicts, warnings, reports and your passport are unchanged — protection is never paywalled.`,
      tone: "quiet",
    };
  }
  return { label: `Go ${product}`, note: "", tone: "quiet" };
}


/** Slice 24 — ONE row, so the first page and every appended page are rendered by
 * the same code. They used to be the same code only because there was no second
 * page; a "load more" that re-implements the row is how a chip stops appearing
 * halfway down a list. */
export function acqRows(rows: readonly AcquisitionDto[]): string {
  return rows
    .map((a) => {
      // #147 (I16-NATIVE hand-off): a creator no longer available is told
      // neutrally, without the title or status they no longer stand behind;
      // the record stays (FanView.swift / FanScreens.kt acquisitions list).
      const unavailable = a.creator_unavailable === true;
      const label = unavailable ? "Creator no longer listed" : typeof a.title === "string" && a.title.trim() !== "" ? a.title : a.release_id;
      const gone = !unavailable && a.release_status === "withdrawn" ? ` <span class="ds-chip quiet" title="Your acquisition is unaffected — it just isn't shown on your public collection.">no longer listed</span>` : "";
      // THE PAGE CURSOR RIDES ON THE ROW, and only on a row whose two halves BOTH
      // pass their shape check. The "Show older" handler reads it off the last
      // `<li>` rather than keeping a counter, so what it pages from is always
      // something the fan can see. A row the server sent with a malformed id or
      // stamp carries no cursor at all — better to end the list than to page
      // from a boundary neither side can name. Both values are shape-checked
      // above (hex id, ISO stamp) and neither can carry a quote, but they are
      // escaped anyway: the check is the reason it is safe, not the guarantee.
      const pageable = typeof a.acquisition_id === "string" && ACQ_ID.test(a.acquisition_id) && typeof a.acquired_at === "string" && ACQ_AT.test(a.acquired_at);
      const cursor = pageable ? ` data-acq-id="${xml(a.acquisition_id)}" data-acq-at="${xml(a.acquired_at)}"` : "";
      // F3: a shape-checked row gets the gift/history control (server-side the
      // acquisition must be the caller's; the control is inert on a bad id).
      const gift = pageable ? giftControl("acquisition", a.acquisition_id) : "";

    })
    .join("");
}

/** The page size the server serves when the client sends no `limit`. Mirrored
 * from `ACQUISITIONS_PAGE_DEFAULT`. The cross-check is
 * `apps/mobile-check/test/slice24-acquisitions-cross-client.test.ts`, which
 * reads the server constant and all three clients' copies out of source and
 * compares them — an earlier version of this comment named
 * `account-social.test.ts`, which does not mention either constant. */
export const ACQ_PAGE = 100;

/**
 * The "Show older" affordance, and the honest signal behind it.
 *
 * The server serves no `has_more` (knowing it costs a second read), so the only
 * signal is the one the page itself carries: a FULL page might have more behind
 * it, a short page is the end. Offering the button on a full page that turns out
 * to be the last one costs the fan one click and an empty result; NOT offering
 * it on a full page that does have more is how the old view lost purchases
 * behind a `.slice(0, 100)` and a footnote.
 */
/*
 * THE "SHOW OLDER" CARRY, stated where a reader meets it (I3-2069: this block used to say it was not wired).
 *
 *  - **Appended pages survive a re-render.** An unsave (700ms timer), a hashchange
 *    or the back button re-renders; `meView` renders `MeData.acqOlder` after page 1
 *    (I3-2096), and the controller feeds it: main.ts holds the carry (`acqCarry`,
 *    bound to the session token) and app.ts passes it in
 *    (`acqOlder: carriedAcqOlder(..., deps.acqOlder?.() ?? null)`, app.ts:522).
 *  - **The CODEX carries the same way**: main.ts `codexCarry` follows the server's
 *    `?before=` cursor (`api.codex(before)`), and app.ts's "codex" case passes it
 *    to `codexView` as `older: deps.codexOlder?.() ?? null` (app.ts:556).
 */
export function acqMore(rows: readonly AcquisitionDto[]): string {
  if (rows.length < ACQ_PAGE) return "";
  return `<p class="row"><button data-act="acq-more" class="linkbtn">Show older</button><span data-acq-msg class="muted" aria-live="polite"></span></p>`;
}

/** One claimed code per row, escaped text only (shared by meView and the "Show
 * older codes" controller in main.ts, so an appended page renders identically). */
export function rcRows(rows: readonly { title: string; code_value: string; claimed_at: string }[]): string {
  return rows.map((k) => `<li><span class="ctitle">${textClamp(k.title, 60)}</span><code class="codebox">${xml(k.code_value)}</code><span class="muted">${xml(hx(k.claimed_at, 10))}</span></li>`).join("");
}

/** I3-1195 — "Show older codes", offered only while the server says there is a
 * next page. The cursor rides on the button (escaped), read back by main.ts. */
function rcMore(next: { claimed_at: string; pool_id: string } | null | undefined): string {
  if (!next || typeof next.claimed_at !== "string" || typeof next.pool_id !== "string" || !next.claimed_at || !next.pool_id) return "";
  return `<p class="row"><button data-act="codes-more" class="linkbtn" data-codes-at="${xml(next.claimed_at)}" data-codes-pool="${xml(next.pool_id)}">Show older codes</button><span data-codes-msg class="muted" aria-live="polite"></span></p>`;
}

/** One row of the credit ledger. Every field is escaped; a sign is carried in
 * the text, not in a colour alone. */
function creditRow(e: { delta: number; reason: string; created_at: string; origin?: string; ref?: string; gross_cents?: number; rail?: string; status?: string }): string {
  const d = typeof e?.delta === "number" && Number.isFinite(e.delta) ? e.delta : 0;
  const sign = d > 0 ? "+" : d < 0 ? "−" : "";
  // I3-0441: a 'purchased' row can carry what the fan paid, gross_cents/rail
  // (credit_purchases, 0035_war_credits.sql:29-38), joined server-side on ref
  // — so a fan can reconcile this row with a statement. Absent (join not yet
  // built, or a non-purchase origin) renders exactly as before.
  const railLabel: Record<string, string> = { stripe: "Card", apple: "Apple", google: "Google", crypto: "Crypto" };
  const paid =
    e?.origin === "purchased" && typeof e?.gross_cents === "number" && Number.isFinite(e.gross_cents)
      ? `<span class="ledpaid">US$${(e.gross_cents / 100).toFixed(2)}${e?.rail && railLabel[e.rail] ? ` · ${xml(railLabel[e.rail])}` : ""}${e?.status === "refunded" ? " · refunded" : ""}</span>`
      : "";
  return `<li class="ledrow"><span class="leddelta">${sign}${num(Math.abs(d))}</span><span class="ledreason">${xml(typeof e?.reason === "string" ? e.reason : "")}</span>${paid}<span class="ledwhen">${xml(shortWhen(e?.created_at))}</span></li>`;
}

/** S72 reverse-code pool offer + claim page. A fan lands here from a creator's
 * off-platform link (`#/codes/<rcp>`); the page shows what the pool is and a
 * Claim button. Eligibility is decided SERVER-SIDE on claim — this page never
 * asserts the fan qualifies. The claimed code value is revealed only in the
 * claim RESPONSE (rendered by the controller), never pre-fetched here. */
const ELIGIBILITY_HINT: Record<string, string> = {
  save: "You need to have saved this creator to claim.",
  acquisition: "You need to have redeemed the linked collectible yourself to claim — a card someone gave you doesn't count.",
  achievement: "You need the linked achievement to claim.",
};
/** I3-2047: `describeRule` (services/trust-core/src/eligibility.ts:300) words a
 * Bond clause as "a Bond of level N or more with this creator"; the fan-safe
 * summary is the only place the offer carries the rule. */
const BOND_CLAUSE = /\bBond of level\b/;
function bondLine(offer: PoolOfferDto, bonds: ReadonlyArray<{ passport_id: string; level: number }> | null | undefined): string {
  // No Bond clause, or the fan's bonds were not read (signed out, or a failed
  // read): say nothing — a missing read is never "no Bond yet".
  if (typeof offer.eligibility_summary !== "string" || !BOND_CLAUSE.test(offer.eligibility_summary) || !Array.isArray(bonds)) return "";
  const mine = bonds.find((b) => b && b.passport_id === offer.passport_id);
  const level = mine && Number.isInteger(mine.level) && mine.level >= 0 ? mine.level : null;
  return `<p class="hint bond">Your Bond with this creator: ${level === null ? "no Bond yet" : `level ${num(level)}`}</p>`;
}
export function codesOfferView(poolId: string, offer: PoolOfferDto | null, opts: { signedIn: boolean; bonds?: ReadonlyArray<{ passport_id: string; level: number }> | null }, read?: Read<unknown>): string {
  // "The link may be closed or expired" is a statement about the OFFER. Only a
  // 404 licenses it; a read that never completed says nothing about the offer.
  if (!offer && !saidNotFound(read)) return unreadableObject("this code offer", "offer unread");
  if (!offer) return `<section class="offer"><div class="empty">This code offer couldn't be found. The link may be closed or expired.</div></section>`;
  const pid = PSP_ID.test(offer.passport_id) ? offer.passport_id : null;
  const from = pid ? `<p class="from">From ${pspLink(pid, "this creator's profile")}.</p>` : "";
  const hint = ELIGIBILITY_HINT[offer.eligibility_kind]
    ? `<p class="hint">${xml(ELIGIBILITY_HINT[offer.eligibility_kind]!)}</p>`
    : typeof offer.eligibility_summary === "string" && offer.eligibility_summary.length > 0
      ? `<p class="hint">To claim you need: ${xml(offer.eligibility_summary)}.</p>`
      : "";
  const body = !offer.available
    ? `<p class="soldout">This offer is closed or fully claimed.</p>`
    : opts.signedIn
      ? `<button data-act="claim-code" data-pool="${xml(POOL_ID.test(poolId) ? poolId : "")}" class="ds-btn act primary">Claim my code</button>${rmsg("data-claim-msg")}<div data-claim-out></div>`

  // "Supporter" was the one survivor of the #34 rename, and it now collides
  // with two live nouns a fan meets elsewhere — the Top Requesters board and
  // the Premium tier. This reward comes from a creator; the eyebrow says so.
  const bond = opts.signedIn ? bondLine(offer, opts.bonds) : "";
  return `<section class="offer"><p class="eyebrow">Creator reward</p><h2>${textClamp(offer.title, 80)}</h2>${from}${hint}${bond}${body}
<p class="fineprint">Dhwaja hands you the code — you redeem it on the creator's own platform. One code per fan.</p></section>`;

/** Rails show a badge only for the two PROVEN states; everything else is a
 * nameless row rather than an implied claim. Colours + icons come from the one
 * table (src/verdict.ts), so a rail badge cannot drift from the check card. */
const DISCO_STATE: Record<string, WebVerdictStyle> = {
  verified: ROW_VERDICT.verified!,
  claimed: ROW_VERDICT.claimed!,
};

/** One creator row in a rail/board. Name links to the Passport only when the id
 * is well-formed (pspLink); score is a plain number + an escaped unit. */
function rankedRow(c: RankedCreatorDto, unit: string): string {

/** Global protection counters (no per-creator identity) — the "Fans Protected" strip. */
function protectionStrip(intel: ImpersonationDto | null): string {
  const c = intel?.counters;
  if (!c) return "";
  const protectedN = typeof c.creators_protected === "number" ? c.creators_protected : 0;
  const activeN = typeof c.warnings_active === "number" ? c.warnings_active : 0;
  const rate = typeof c.resolution_rate === "number" ? Math.max(0, Math.min(100, Math.round(c.resolution_rate * 100))) : 0;
  const heat = typeof c.heat_recent === "number" ? c.heat_recent : 0;
  // Shown BESIDE the resolution rate, never folded into it. The rate used to BE
  // the retraction rate, so every corrected mistake made the product look
  // better; the honest split says both numbers out loud. Omitted at zero — the
  // absence of mistakes should not need a line that reads like a disclaimer.
  const corrected = typeof c.warnings_corrected_total === "number" ? c.warnings_corrected_total : 0;
  return `<div class="protstrip"><span class="stat"><b>${num(protectedN)}</b> creators protected</span><span class="stat"><b>${num(activeN)}</b> active warnings</span><span class="stat"><b>${num(rate)}%</b> of warnings resolved</span>${corrected > 0 ? `<span class="stat"><b>${num(corrected)}</b> withdrawn as mistakes</span>` : ""}<span class="stat"><b>${num(heat)}</b> flagged in the last 30 days</span></div>`;
}

/**

const DISCOVERY_BOARDS = ["newcomers", "breakout", "fastest_rising", "most_collected", "most_discovered", "by_category"] as const;
function discoveryReadable(d: DiscoveryDto | null): d is DiscoveryDto {
  if (!d || typeof d !== "object") return false;
  return DISCOVERY_BOARDS.some((k) => Array.isArray((d as unknown as Record<string, unknown>)[k]));
}

const FACET_LANG_RE = /^[a-z]{2}$/;
const FACET_REGION_RE = /^[A-Z]{2}$/;

/** A language/region facet chip row for the current board (S80). Chips are only
 * emitted for facet values actually PRESENT in the niche's board — so a fan can
 * narrow to "Spanish" only when Spanish creators are ranked here. Selecting a facet
 * keeps the niche; a chip toggles off when it's the active one. Every emitted hash
 * carries validated codes, so a junk facet never forms a link. */
/** I3-2098 — a facet code's English name ("es" -> "Spanish", "US" -> "United States"); the bare code when the runtime has no name for it. */
function facetName(type: "language" | "region", code: string): string {
  try {
    const name = new Intl.DisplayNames(["en"], { type, fallback: "none" }).of(code);
    return typeof name === "string" && name.length > 0 ? name : code.toUpperCase();
  } catch {
    return code.toUpperCase();
  }
}
function facetRow(category: string, list: RankedCreatorDto[], sel: { lang: string | null; region: string | null }): string {
  const langs = [...new Set(list.map((c) => (typeof c?.language === "string" && FACET_LANG_RE.test(c.language) ? c.language : null)).filter((x): x is string => !!x))].sort();
  const regions = [...new Set(list.map((c) => (typeof c?.region === "string" && FACET_REGION_RE.test(c.region) ? c.region : null)).filter((x): x is string => !!x))].sort();
  if (langs.length === 0 && regions.length === 0) return "";
  const href = (lang: string | null, region: string | null): string => {
    const q = new URLSearchParams();
    if (lang) q.set("lang", lang);
    if (region) q.set("region", region);
    const qs = q.toString();
    return `#/browse/${category}${qs ? `?${qs}` : ""}`;
  };
  const langChips = langs.map((l) => navLink(href(sel.lang === l ? null : l, sel.region), facetName("language", l), sel.lang === l ? "ds-btn nav active" : "ds-btn nav")).join("");
  const regionChips = regions.map((r) => navLink(href(sel.lang, sel.region === r ? null : r), facetName("region", r), sel.region === r ? "ds-btn nav active" : "ds-btn nav")).join("");
  const clearAll = sel.lang || sel.region ? navLink(href(null, null), "Clear filters", "ds-btn nav") : "";
  const langPart = langChips ? `<span class="facetlabel">Language</span>${langChips}` : "";
  const regionPart = regionChips ? `<span class="facetlabel">Region</span>${regionChips}` : "";
  return `<div class="facets">${langPart}${regionPart}${clearAll}</div>`;
}

/** Niche browse: category chips + language/region facets + the selected niche's
 * ranked board (facet-filtered). */
export function browseView(tax: TaxonomyDto | null, d: DiscoveryDto | null, category: string | null, facets: { lang: string | null; region: string | null } = { lang: null, region: null }): string {
  // The niche vocabulary and the boards are two separate reads, and either can
  // fail on its own. A vocabulary we could not read is not "no niches exist",
  // and a board we could not read is not "no creators in this niche yet".
  const taxUnread = unreadable(tax?.categories);
  const vocab = arr<{ key: string; label: string }>(tax?.categories).filter((v) => v && typeof v.key === "string" && CATEGORY_RE.test(v.key));
  const chips = vocab.map((v) => catLink(v.key, hx(v.label, 40) || v.key, category === v.key ? "ds-btn nav active" : "ds-btn nav")).join("");
  const chipRow = `<div class="cats">${navLink("#/discover", "← Discover", "ds-btn nav")}${chips}</div>`;
  const chipNote = taxUnread ? unreadableCard("the list of niches") : "";
  if (!category) {
    return `<section class="browse"><h2>Browse creators by niche</h2>${chipRow}${chipNote}${taxUnread ? "" : `<p class="hint">Pick a niche to see its top creators.</p>`}</section>`;
  }
  const board = arr<CategoryBoardDto>(d?.by_category).find((b) => b && b.category === category) ?? null;
  const heading = board ? hx(board.label, 40) || category : hx(vocab.find((v) => v.key === category)?.label, 40) || category;
  const full = arr<RankedCreatorDto>(board?.top);
  const facetChips = facetRow(category, full, facets);
  // Facets filter the already-served board (no new fetch / aggregate). Both apply.
  const sel = { lang: typeof facets.lang === "string" && FACET_LANG_RE.test(facets.lang) ? facets.lang : null, region: typeof facets.region === "string" && FACET_REGION_RE.test(facets.region) ? facets.region : null };
  const list = full.filter((c) => (!sel.lang || c?.language === sel.lang) && (!sel.region || c?.region === sel.region));
  const body = !discoveryReadable(d)
    ? unreadableCard(`the ${xml(heading)} board`)
    : list.length > 0
      ? `<ol class="rlist">${list.slice(0, 25).map((c) => rankedRow(c, EN["disc.unit_collectors"])).join("")}</ol>`
      : `<p class="empty">No creators in ${xml(heading)}${sel.lang || sel.region ? " match those filters" : " yet"}.</p>`;
  return `<section class="browse"><h2>${xml(heading)}</h2>${chipRow}${chipNote}${facetChips}${body}</section>`;
}

/** The public demand page for a (usually unclaimed) creator: how many fans have
 * asked for them, framed as a claim CTA. Honest about what Dhwaja does NOT know
 * (check counts). Every value escaped; the Passport link is id-validated. */
export function demandView(dto: DemandDto | null, read?: Read<unknown>, consoleUrl?: string | null): string {

  const n = shown === null ? NO_COUNT : String(num(shown));
  const pid = typeof dto.passport_id === "string" && PSP_ID.test(dto.passport_id) ? dto.passport_id : null;

  const recent = typeof dto.recent_requests === "number" ? dto.recent_requests : 0;

  const note = `<p class="privnote">A check made while signed out is never logged. Creators are never told who checks them; your own Codex keeps the creators you opened unless you have paused it. This page counts public requests only, never views or checks.</p>`;

const SOURCE_LABEL: Record<string, string> = { view: "viewed", check: "checked", save: "saved", collect: "collected", request: "requested", acquire: "collected" };

/**
 * flag 9: a join number, or null.
 *
 * A number is allocated when a creator CLAIMS their passport, not when Dhwaja
 * lists them — so most seeded, unclaimed listings have none, and null is the
 * COMMON case rather than a defensive edge. There is no such thing as join
 * number 0 and no placeholder for an unallocated one: absent means absent.
 */
function joinKey(v: unknown): number | null {
  return typeof v === "number" && Number.isInteger(v) && v > 0 ? v : null;
}

/** flag 9: the permanent № chip. Numbers are CLAIM order with jersey-number
 * semantics — an allocated row in `join_numbers`, not a ledger position. (They
 * were briefly the latter: a running count of `passport_created` computed at
 * read time, which handed the lowest numbers to a bulk import of harvested,
 * non-consenting listings and leaked the total listing count. Window item 9.)
 * Sorting only ever changes DISPLAY. An unclaimed listing has no number, and an
 * entry with no number renders NO chip: not "#—", not "#0", nothing. */
function joinNo(e: CodexEntryDto): string {
  const n = joinKey(e.join_no);
  return n === null ? "" : `<span class="joinno" title="Permanent registry join number — never changes">№ ${String(n).padStart(4, "0")}</span>`;
}

/** I3-2117 — data-name carries the WHOLE name (no ellipsis, no case fold), so the client A–Z sort compares exactly what the server order (`localeCompare` on display_name) compares; the cap only bounds the attribute. */
const CODEX_SORT_NAME_MAX = 256;
function codexRow(e: CodexEntryDto): string {
  const st = CODEX_STATE[typeof e.current_state === "string" ? e.current_state : ""] ?? CODEX_STATE.unclaimed!;
  const early = e.discovered_early ? `<span class="early" title="You found them before they were verified">★ early</span>` : "";
  const flags = `${e.saved ? `<span class="flag">saved</span>` : ""}${e.collected ? `<span class="flag">collected</span>` : ""}`;
  const src = SOURCE_LABEL[typeof e.first_source === "string" ? e.first_source : ""] ?? "seen";
  // data-join / data-name feed the client-side №/A–Z display sort (flag 9);
  // data-name is the raw name (see CODEX_SORT_NAME_MAX), escaped once by xml().
  // I3-1782: data-name is an ATTRIBUTE value, escaped once below by xml(); textClamp
  // already escapes for direct text position, so using it here double-escaped
  // names with &, ', ", < or > (e.g. "D'Angelo" became "D&amp;#39;Angelo"). clampPlain
  // cuts to the same length without escaping, matching every other attribute
  // built this way in this file (see clampPlain's own docblock).
  const sortName = (typeof e.display_name === "string" ? e.display_name : "").slice(0, CODEX_SORT_NAME_MAX);
  // EMPTY, not a sentinel. A stand-in like 999999999 is a fabricated join
  // number sitting in the DOM next to real ones, and the client comparator
  // would sort by it as if it meant something. "No number" is representable
  // only as no number.
  const sortJoin = joinKey(e.join_no) ?? "";
  return `<li class="cdxcell ${st.cls}" data-join="${sortJoin}" data-name="${xml(sortName)}">${joinNo(e)}<span class="badge ${st.cls}"><span class="ico" aria-hidden="true">${xml(st.glyph)}</span>${xml(st.label)}</span><span class="name">${pspLink(e.passport_id, clampPlain(e.display_name, 40))}</span>${early}<span class="meta">${xml(src)} · ${num(e.encounter_count)}×</span><span class="flags">${flags}</span></li>`;
}

/**
 * THE BUG THESE FUNCTIONS EXIST TO MAKE UNREPRESENTABLE.
 *
 * Both switches used to be five lines of controller each:
 *
 *   const cur = await deps.api.settings();
 *   const next = !(cur?.email_alerts ?? true);
 *   const r = await deps.api.setEmailAlerts(next);
 *   if (r.ok) button.textContent = `Email alerts: ${next ? "On" : "Off"}`;
 *
 * `settings()` returns null for EVERY failure — 401, 429, 5xx, offline,
 * unparseable body. So `!(null?.email_alerts ?? true)` is `false`, and a fan on
 * a flaky connection who pressed the switch to turn alerts ON silently turned
 * them OFF — a write derived from a read that never happened. The promos switch
 * was the same code with `?? false`, which silently MUTED. And `if (r.ok)` with
 * no else meant a failed write said nothing at all.
 *
 * The rule now, in three pieces that the controller cannot skip:
 *   - `settingsFlag` returns null for anything that is not a boolean. There is
 *     no default: "we could not read it" is a third value, not an assumption.
 *   - a null read REFUSES the write (`SETTINGS_UNREADABLE`); the fan is told
 *     nothing changed rather than having a direction guessed for them.
 *   - the new label comes from the server's ECHOED value where it sent one,
 *     never from what we hoped we set.
 */
export type FanToggle = "email_alerts" | "broadcasts_muted" | "email_report_outcomes";

const TOGGLE_COPY: Record<FanToggle, { name: string; whenTrue: string; whenFalse: string }> = {
  email_alerts: { name: "Email alerts", whenTrue: "On", whenFalse: "Off" },
  // `broadcasts_muted` is the INVERSE of what the label says: true means muted.
  broadcasts_muted: { name: "Creator broadcasts", whenTrue: "Muted", whenFalse: "On" },
  // I3-2072 (D-0995(5)): the report-outcome email switch, on the same email-preference door (field email_report_outcomes).
  email_report_outcomes: { name: "Report outcome emails", whenTrue: "On", whenFalse: "Off" },
};

/** The switch's label. `null` is UNREADABLE — never rendered as a state. */
export function settingsToggleLabel(which: FanToggle, value: boolean | null): string {
  const c = TOGGLE_COPY[which];
  return value === null ? `${c.name}: couldn't check` : `${c.name}: ${value ? c.whenTrue : c.whenFalse}`;
}

/** Read one flag out of a settings body, fail-closed: anything that is not a
 * boolean (absent field, null body, junk) is UNKNOWN, not a default. */
export function settingsFlag(which: FanToggle, s: { email_alerts?: unknown; broadcasts_muted?: unknown; email_report_outcomes?: unknown } | null): boolean | null {
  const v = s ? s[which] : undefined;
  return typeof v === "boolean" ? v : null;
}

/** Said when the settings READ failed, in place of flipping a switch whose
 * current position we do not know. */
export const SETTINGS_UNREADABLE = "Couldn't read your settings just now, so nothing was changed — we won't guess which way the switch was pointing. Try again in a moment.";

/** The write's answer, pure. A failure never moves the label, and always says
 * that the setting is unchanged; a success labels the switch with the value the
 * SERVER echoed back where it sent one. */
export function settingsToggleOutcome(
  which: FanToggle,
  wanted: boolean,
  r: { ok: boolean; status: number; data?: { email_alerts?: unknown; broadcasts_muted?: unknown; email_report_outcomes?: unknown } | null },
): { changed: boolean; label: string | null; text: string } {
  if (r.ok) {
    const echoed = settingsFlag(which, r.data ?? null);
    return { changed: true, label: settingsToggleLabel(which, echoed ?? wanted), text: "" };
  }
  const name = TOGGLE_COPY[which].name.toLowerCase();
  if (r.status === 401) return { changed: false, label: null, text: `You're signed out — sign in and try again. Your ${name} setting is unchanged.` };
  if (r.status === 429) return { changed: false, label: null, text: `That's a lot of changes at once — your ${name} setting is unchanged. Try again in a minute.` };
  // I3-2171 — status 0 / 5xx: the write may have landed before the response was lost.
  if (r.status === 0 || r.status >= 500) return { changed: false, label: null, text: `We couldn't confirm that — reload to see your ${name} setting before changing it again.` };
  return { changed: false, label: null, text: `That didn't save — your ${name} setting is unchanged. Try again in a moment.` };
}

/**
 * flag 9 — the №/A–Z display re-order, as a pure comparator.
 *
 * Extracted from the DOM controller so the ordering rule is under test.
 * ORDER IS A CHANNEL: it can carry a fact as surely as a badge can, so the
 * comparator is allowed to read exactly TWO things — the permanent registry
 * join number and the display name, both already on the row as data-
 * attributes. It must never read state (`verified` / `claimed` / `unclaimed`),
 * a saved/collected flag, or anything the view collapses, because a sort that
 * grouped by state would publish through position what the badge already says
 * once and would keep saying after the copy changed.
 */
export function codexSortCompare(mode: "num" | "az"): (a: { name?: string; join?: string }, b: { name?: string; join?: string }) => number {
  if (mode === "az") return (a, b) => (a.name ?? "").localeCompare(b.name ?? "");
  return (a, b) => {
    // `data-join` is EMPTY for an entry with no number, so this parses to null
    // rather than to a position. The unnumbered cohort keeps its own order —
    // alphabetical, the same rule the server-rendered order uses — instead of
    // arriving in whatever sequence the payload happened to have. Two rows
    // whose relative position means nothing must at least be STABLE, or the
    // list appears to re-rank itself on every toggle.
    const aj = joinKey(Number(a.join));
    const bj = joinKey(Number(b.join));
    if (aj === null && bj === null) return (a.name ?? "").localeCompare(b.name ?? "");
    if (aj === null) return 1;
    if (bj === null) return -1;
    return aj - bj;
  };
}

/**
 * The SERVER-side twin of codexSortCompare("num"), over the DTO rather than the
 * rendered row, so the first paint and every later toggle agree. Numbered rows
 * ascending (the collectible-book order); then the unnumbered cohort, A–Z.
 *
 * Sorting the unnumbered cohort ALPHABETICALLY is the whole point. They have to
 * go somewhere, and any ordering of them is a statement — but a name-ordered
 * block is self-evidently not a league table, whereas "whatever order the API
 * returned" silently invites a fan to read the top of it as the best of them.
 */
export function codexEntryCompare(a: CodexEntryDto, b: CodexEntryDto): number {
  const aj = joinKey(a.join_no);
  const bj = joinKey(b.join_no);
  if (aj === null && bj === null) return String(a.display_name ?? "").localeCompare(String(b.display_name ?? ""));
  if (aj === null) return 1;
  if (bj === null) return -1;
  return aj - bj;
}

/**
 * The redeem form's one line of feedback, pure.
 *
 * `dead` and `unknown` are named separately because both are the person's OWN
 * code and neither says anything about anybody else — but every other failure
 * collapses into one sentence rather than echoing a server string, so a new
 * outcome can never render raw.
 *
 * A TRANSPORT FAILURE IS NOT A VERDICT ON THE CODE. The fallback used to be
 * "Couldn't redeem that code." — one sentence whose subject is the code — for
 * 401 (session expired), 429 (rate-limited), 0 (offline) and every 5xx. This is
 * the highest-stakes action a fan takes here: the code may have been paid for,
 * and the client was telling them it was broken because a socket closed. That
 * is precisely the failure `api.ts`'s synthetic-401 note was written to
 * eliminate — "a transport/auth fact rendered as a verdict about the
 * credential". Every branch below now carries the fact the fan actually needs:
 * THE CODE WAS NOT USED — except status 0 and 5xx, where that fact is unknown
 * (I3-2171) and the line says so instead. The iOS and Android redeem screens already said it
 * ("Couldn't reach Dhwaja — your code wasn't used"); the web did not.
 *
 * `status` is optional so an older caller still compiles, but the controller
 * passes the whole `{ok, status, data}` envelope `api.redeem` already returns.

export function redeemMessage(r: { ok: boolean; status?: number; data: { ok?: boolean; outcome?: string; release_status?: string; instance_minted?: boolean | null } | null }): { text: string; redeemed: boolean } {
  if (r.ok && r.data?.ok === true) {
    // AUDIT ITEM #22 — the WITHDRAWN branch. The redemption succeeded and the
    // acquisition is permanent; what changes is that the fan is told the one
    // thing that would otherwise only reach them as an absence. `publicProfile`
    // excludes withdrawn releases from the public collection AND its count, and
    // the showcase drops them too, so "added to your collection" on its own was
    // a sentence the product then quietly contradicted.
    //
    // The wording carries the fact and NO blame: the creator unlisting their
    // own work is a normal thing to do, and this line is read by someone who
    // just paid. It says what they still hold before it says what they don't.
    if (r.data.release_status === "withdrawn") {
      return {
        text: "Redeemed ✓ — this release is no longer listed by the creator. Your acquisition is permanent proof of purchase, but it won't show on your public collection.",
        redeemed: true,
      };
    }
    // I3-1886: `instance_minted` is a fact about a SEPARATE asset (the War
    // card), and `false`/`null` are final answers, not a pending state — the
    // release itself was redeemed either way, so `redeemed` stays true. Every
    // fan client (this view, `apps/android` FanScreens.kt, `apps/ios`
    // FanView.swift — both REAL — WRONG OWNER) reads the same three-way split;
    // `undefined` (absent) falls through to today's message unchanged.
    if (r.data.instance_minted === false) {
      return { text: "Redeemed - the collectible is yours. No playable War card was added (your card shelf is full, or this creator isn't in Creator War).", redeemed: true };
    }
    if (r.data.instance_minted === null) {
      return { text: "Redeemed. We couldn't confirm your War card just now - check your collection.", redeemed: true };
    }
    return { text: "Redeemed ✓ — added to your collection.", redeemed: true };
  }
  // The two answers the SERVER gave about the code itself, unchanged.
  if (r.data?.outcome === "dead") return { text: "That code was already used.", redeemed: false };
  if (r.data?.outcome === "unknown") return { text: "That code wasn't recognized.", redeemed: false };
  // Everything below is a statement about the REQUEST, never about the code.
  if (r.status === 401) return { text: "You're signed out — sign in and redeem it again. Your code was not used.", redeemed: false };
  if (r.status === 429) return { text: "Too many tries just now. Your code was not used — wait a minute and try again.", redeemed: false };

  if (r.status === 0 || (typeof r.status === "number" && r.status >= 500)) return { text: "We couldn't confirm that. Check your collection before trying again — if the code went through it's there, and if not it's still unused.", redeemed: false };

}

/**

export function referralCodeOutcome(r: { ok: boolean; status: number; data: { code?: string } | null }): { code: string | null; text: string } {
  const code = r.ok && typeof r.data?.code === "string" && /^[A-Z2-7]{10}$/.test(r.data.code) ? r.data.code : null;
  if (code) return { code, text: "Your referral code — share it anywhere. It credits you when someone joins with it and redeems their first credential from a verified creator." };

/**
 * M13: pull a referral code out of an inbound URL query (e.g. "?ref=ABCDE23456"),
 * uppercased and shape-checked to the server's mint shape (10 base32 chars).
 * Anything else is null, so a junk query never rides into sign-in. This closes
 * the referral loop: the referrer shares a link, a new fan's client captures the
 * code here and forwards it on sign-in, and the server attributes it at birth.
 */
export function parseReferralParam(search: string): string | null {
  try {
    const s = typeof search === "string" ? search : "";
    const q = new URLSearchParams(s.startsWith("?") ? s : `?${s}`);
    const raw = (q.get("ref") ?? "").trim().toUpperCase();
    return /^[A-Z2-7]{10}$/.test(raw) ? raw : null;
  } catch {
    return null;
  }
}

/** M13: the shareable invite URL for a referral code — an origin + "?ref=CODE"
 * that `parseReferralParam` reads back. Kept pure so the share affordance and
 * the capture path are tested against one shape. */
export function referralShareUrl(origin: string, code: string): string {
  const base = typeof origin === "string" ? origin.replace(/\/+$/, "") : "";
  return `${base}/?ref=${encodeURIComponent(code)}`;
}

/**
 * I3-0095 a (D-0981(3)) — THE PUBLIC LEDGER PAGE. A read-only list of the signed checkpoints trust-core already
 * publishes (`GET /checkpoints`, checkpoint-public.ts CheckpointIndex + the served verify recipe), with how to
 * check them. No new data: every field is one the public route serves, and the per-anchor JSON links go to
 * `/checkpoints/<key>` on the same origin. The signing keys are NOT taken from that origin (the recipe says
 * why) — the page says so instead of pretending a browser check would be independent.
 */
export interface LedgerIndex {
  builtAt: string;
  count: number;
  truncated: boolean;
  checkpoints: { key: string; seq: number; takenAt: string }[];
  recipe: { alg: string; preimage: string; sigEncoding: string; keyIds: string[]; publicKeys: string } | null;
}
const CHECKPOINT_KEY_RE = /^[0-9A-Za-z:._-]{1,80}-seq[0-9]{1,15}$/;
const LEDGER_TEXT = (v: unknown, max = 600): string | null => (typeof v === "string" && v.length > 0 && v.length <= max ? v : null);
export function parseCheckpointIndex(raw: unknown): LedgerIndex | null {
  if (!raw || typeof raw !== "object") return null;
  const o = raw as Record<string, unknown>;
  if (!Array.isArray(o.checkpoints) || typeof o.count !== "number") return null;
  const checkpoints: LedgerIndex["checkpoints"] = [];
  for (const e of o.checkpoints.slice(0, 1000)) {
    const r = (e ?? {}) as Record<string, unknown>;
    if (typeof r.key !== "string" || !CHECKPOINT_KEY_RE.test(r.key) || !Number.isSafeInteger(r.seq) || (r.seq as number) < 0 || LEDGER_TEXT(r.taken_at, 40) === null) return null;
    checkpoints.push({ key: r.key, seq: r.seq as number, takenAt: r.taken_at as string });
  }
  const v = (o.verify ?? null) as Record<string, unknown> | null;
  const keyIds = v && Array.isArray(v.key_ids) ? v.key_ids.filter((k): k is string => typeof k === "string" && /^[a-z0-9][a-z0-9._-]{0,63}$/.test(k)) : [];
  const recipe =
    v && LEDGER_TEXT(v.alg, 80) && LEDGER_TEXT(v.preimage) && LEDGER_TEXT(v.sig_encoding) && LEDGER_TEXT(v.public_keys)
      ? { alg: v.alg as string, preimage: v.preimage as string, sigEncoding: v.sig_encoding as string, keyIds, publicKeys: v.public_keys as string }
      : null;
  return { builtAt: LEDGER_TEXT(o.built_at, 40) ?? "", count: o.count, truncated: o.truncated === true, checkpoints, recipe };
}

/**
 * I3-0095 a (D-0981(3) + D-0977) — the OpenTimestamps seal anchors, from the public `GET /seal-anchors` read
 * (hand-off: checkpoint-public.ts + index.ts). Fail-closed: a body that does not parse renders NO section, never an
 * empty list that would read as "nothing was anchored". Only fields the calendars already hold are shown: the
 * submitted digest, when, the ledger height it seals, and each calendar's pending/complete state.
 */
export interface SealAnchor {
  digest: string;
  createdAt: string;
  headSeq: number;
  calendars: { host: string; state: "pending" | "complete" }[];
}
export function parseSealAnchors(raw: unknown): SealAnchor[] | null {
  if (!raw || typeof raw !== "object") return null;
  const list = (raw as { anchors?: unknown }).anchors;
  if (!Array.isArray(list)) return null;
  const out: SealAnchor[] = [];
  for (const e of list.slice(0, 100)) {
    const r = (e ?? {}) as Record<string, unknown>;
    if (typeof r.digest !== "string" || !/^[0-9a-f]{64}$/.test(r.digest) || LEDGER_TEXT(r.created_at, 40) === null || !Number.isSafeInteger(r.head_seq) || (r.head_seq as number) < 0) return null;
    const cals: SealAnchor["calendars"] = [];
    for (const c of Array.isArray(r.receipts) ? r.receipts.slice(0, 8) : []) {
      const cr = (c ?? {}) as Record<string, unknown>;
      let host = "";
      try {
        host = typeof cr.calendar === "string" && /^https:\/\//.test(cr.calendar) ? new URL(cr.calendar).host : "";
      } catch {
        host = "";
      }
      if (!host || (cr.state !== "pending" && cr.state !== "complete")) return null;
      cals.push({ host, state: cr.state });
    }
    out.push({ digest: r.digest, createdAt: r.created_at as string, headSeq: r.head_seq as number, calendars: cals });
  }
  return out;
}
function sealAnchorsBlock(read: { status: number; data: unknown } | null | undefined): string {
  const anchors = read && read.status === 200 ? parseSealAnchors(read.data) : null;
  if (!anchors) return "";
  const rows = anchors
    .map((a) => {
      const done = a.calendars.filter((c) => c.state === "complete").length;
      const cal = a.calendars.length ? a.calendars.map((c) => `${xml(c.host)} (${c.state === "complete" ? "in Bitcoin" : "pending"})`).join(", ") : "no calendar answered";
      return `<li class="ledgerrow" data-seal-anchor><code>${xml(a.digest)}</code> <span>${xml(a.createdAt)}</span> <span>seals the ledger through event #${xml(String(a.headSeq))}</span> <span>${xml(String(done))} of ${xml(String(a.calendars.length))} confirmed: ${cal}</span></li>`;
    })
    .join("");
  return `<section class="ledger" data-seal-anchors><h3>External timestamps</h3><p class="wbody">Besides signing checkpoints itself, Dhwaja sends a digest of its sealed ledger heads to public OpenTimestamps calendars, which fold it into the Bitcoin blockchain. Once a calendar says "in Bitcoin", that digest provably existed by that block — a later rewrite of the ledger cannot match it.</p>${anchors.length ? `<ul class="ledgerlist">${rows}</ul>` : `<div class="empty">No digest has been sent to a calendar yet.</div>`}<p class="fine">To check one yourself, run the OpenTimestamps client on its proof file: <code>ots verify -d &lt;digest&gt; &lt;proof.ots&gt;</code>. The check needs a Bitcoin block-header source, not Dhwaja.</p></section>`;
}
