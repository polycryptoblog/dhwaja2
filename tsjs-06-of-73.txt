/** I3-0801 (hand-off from I4-02-R10, apps/web/src/main.ts's identical fix) --
 * ABANDONED READS ARE CANCELLED ON NAVIGATION. `renderEpoch` discards a stale
 * route's PAINT, but its fan-out kept running to completion (or to the
 * deadline) and held the browser's per-host connections the next route
 * needed. Every GET/HEAD `api` makes carries this signal, which the render()
 * epoch bump below aborts and replaces. A mutation (POST/DELETE) never
 * carries it: cancelling a write mid-flight would leave its outcome unknown. */
let navAbort = new AbortController();
function navScopedFetch(input: RequestInfo | URL, init?: RequestInit): Promise<Response> {
  const method = (init?.method ?? "GET").toUpperCase();
  if (method !== "GET" && method !== "HEAD") return fetch(input, init);
  const nav = navAbort.signal;
  const own = init?.signal ?? null;
  const link = new AbortController();
  const stop = (): void => link.abort();
  if (nav.aborted || own?.aborted) link.abort();
  nav.addEventListener("abort", stop, { once: true });
  own?.addEventListener("abort", stop, { once: true });
  const done = (): void => {
    nav.removeEventListener("abort", stop);
    own?.removeEventListener("abort", stop);
  };
  return fetch(input, { ...init, signal: link.signal }).finally(done);
}

 * `codesCell` (views.ts:3622) gives a listing with NO entry "Not loaded", and

async function issuedCodesFor(listed: readonly { release_id: string }[]): Promise<Map<string, CodeSummaryRead>> {
  const codes = new Map<string, CodeSummaryRead>();
  if (listed.length === 0) return codes;
  // B28 #521: ONE batch read first — the creator's own summaries, paged, at
  // most SUMMARY_PAGES_MAX requests however many listings there are. A listing
  // the batch did not return (the read failed, or the listing appeared after
  // it) falls through to the bounded per-id pool below, so the map stays
  // complete either way.
  const wanted = new Set(listed.map((l) => l.release_id));
  let cursor: string | undefined;
  for (let p = 0; p < SUMMARY_PAGES_MAX; p += 1) {
    const r = await api.credentialSummaries(cursor);
    if (!r.ok || !r.data) break;
    for (const s of r.data.summaries) if (wanted.has(s.release_id)) codes.set(s.release_id, { ok: true, status: r.status, data: s });
    if (r.data.next === null || codes.size === wanted.size) break;
    cursor = r.data.next;
  }
  const rest = listed.filter((l) => !codes.has(l.release_id));
  let next = 0;
  const worker = async (): Promise<void> => {
    while (next < rest.length) {
      const i = next;
      next += 1;
      const id = rest[i]!.release_id;
      codes.set(id, await api.credentialSummary(id));
    }
  };
  await Promise.all(Array.from({ length: Math.min(CODE_FETCH_CONCURRENCY, rest.length) }, () => worker()));
  return codes;
}

async function addPasskey(psp: string, held: number | null, box: string, done: (r: FlowResult) => Promise<void>): Promise<void> {
  const viaFlow = async (): Promise<void> => {
    const r = await registerPasskeyFlow(api, creds, psp);
    if (r.code_asked && r.code_asked.expires_at !== null) return codeArm({ options: r.code_asked.options, expires_at: r.code_asked.expires_at });
    // I34-FX3-PASSKEY (I34-JN-25 W3C-H3): a count of one or more that was stale (none held) meets the first-passkey
    // refusals here; they get the zero-count door's words (the server's sentence first), never the flow's generic one.
    const h3 = r.start_refused;

/** A first-passkey START that did not give options: each H3 answer its own words (the server's sentence first). 503
 * email_unavailable reads ERROR_COPY's own line through the last return (serverMessage is null at 5xx). */
function firstPasskeyStartRefusal(s: { status: number; error: string | null; answer: { ok: boolean; status: number; data: unknown } }): string {
  if (s.status === 409 && s.error === "contact_email_unverified") return serverMessage(s.answer) ?? FIRST_PASSKEY_VERIFY_FIRST_FALLBACK;
  if (s.status === 429) return serverMessage(s.answer) ?? errorCopy("rate_limited", "Too many tries in a short time — wait a minute, then try again.");
  if (s.status === 503 && s.error === "email_send_failed") return FIRST_PASSKEY_SEND_FAILED;
  return serverMessage(s.answer) ?? errorCopy(s.error, "could not start registration");
}
/** A first-passkey FINISH the server refused (the code arm). 401 email_code_* spends the challenge: start again. */
function firstPasskeyFinishRefusal(f: { status: number; data: { error?: string } | null }): string {
  const error = f.data?.error;
  if (f.status === 401 && error === "email_code_invalid") return FIRST_PASSKEY_CODE_SPENT;
  if (f.status === 401 && error === "email_code_required") return FIRST_PASSKEY_CODE_MISSING;
  if (f.status === 409 && error === "contact_email_unverified") return serverMessage({ ok: false, status: f.status, data: f.data }) ?? FIRST_PASSKEY_VERIFY_FIRST_FALLBACK;
  return errorCopy(error, "could not finish registration");
}

/**
 * THE ONLY WAY THE APP REPAINTS. Every existing caller kept its `render()`
 * call and got the one-time guard for free; a caller that has not been written
 * yet gets it too. Returns false when the creator refused to discard unsaved
 * voucher values, in which case NOTHING was repainted and the panel is still on
 * screen exactly as it was.
 */
let renderEpoch = 0;

async function render(): Promise<boolean> {
  if (oneTime.pending > 0 && !(await oneTime.consume())) return false; // I3-0422 a: the dialog only when codes are at stake
  navAbort.abort(); // I3-0801 -- the previous route's reads are abandoned from here
  navAbort = new AbortController();
  const epoch = ++renderEpoch;
  await paintApp(epoch);
  if (epoch === renderEpoch) resumeAfterPaint(); // I3-1206
  return true;
}

/** Fetch the wizard's server-derived state, derive steps, render the active
 * step's panel. Server state is the truth (self-heals across devices): contact
 * now reads from GET /account/contact-state; only the bio-link mark stays local
 * (no creator-readable API for where a link was placed). */
/**
 * The derived checklist, factored out so the WELCOME interstitial and the
 * onboarding page cannot disagree about what a creator's next step is. Both
 * take the app-wide accounts read; the three reads here are the same three
 * either screen would have spent on its own.
 */
async function onboardingSteps(accts: AccountStandingDto[] | null, ver: VerificationDto | null, readsLanded: boolean): Promise<StepState[]> {
  const psp = session.passportId();
  const [keys, rels, contact] = await Promise.all([api.passkeys(), api.releases(), api.contactState()]);
  // THE ACCOUNTS STEP READS THE CREATOR'S OWN LIST, NOT THE PUBLIC PAGE.
  //
  // It used to count `tabs.links` off `GET /p/:id/presentation` — the PUBLIC
  // view, which drops every asserted-but-unproven row. Manual review is the only
  // claim path for Instagram, TikTok and OnlyFans, and there is no manual
  // ACCOUNT proof for an already-claimed passport at all, so a creator on those
  // platforms could add accounts forever and this number never moved: "Link your
  // accounts" was un-completable, and with it the whole checklist — "All set —
  // you're provably you 🎉" was unreachable by construction for exactly the
  // creators this product is most for. The read is the app-wide one, so this
  // costs nothing extra.
  const activeAccounts = Array.isArray(accts) ? accts.filter((a) => a.status === "active") : [];
  // NOT SERVED = the registry would publish nothing for this handle, or would
  // publish it as the amber `caution` class. Both mean a fan checking that exact
  // handle does not get an answer that says "yes, this is them".
  const unserved = activeAccounts.filter((a) => a.serving === "not_served" || a.serving === "caution");
  const steps = deriveSteps({
    signedIn: session.isSignedIn(),
    passkeyCount: keys.data?.passkeys?.length ?? 0,
    linkedAccounts: activeAccounts.length,
    confirmedAccounts: activeAccounts.filter((a) => a.confirmed === true).length,
    releaseCount: Array.isArray(rels.data?.releases) ? rels.data.releases.filter((r) => r.status === "published").length : 0,
    // The contact step is done when the SERVER shows a verified email (self-heals
    // across devices). The local mark is used ONLY when the read failed — so a
    // truthful server "not verified" (e.g. after an email change reset it) is not
    // overridden by a stale local mark.
    // #131BR-01 (I16-W01D): "on" needs the digest stream AND its email half,
    // not just a verified address (onboarding.ts protectionAlertsOn).
    contactMarked: contact.ok ? protectionAlertsOn(contact.data) : psp ? marks.has(psp, "contact") : false,
    contactVerifiedAlertsOff: contact.ok && contact.data?.email_verified === true && !protectionAlertsOn(contact.data),
    bioLinkMarked: psp ? marks.has(psp, "biolink") : false,
    revealMarked: psp ? marks.has(psp, "reveal") : false,
    // F9's APPOINTMENT, derived entirely from reads this paint already made.
    // `serving` is the class the registry build WOULD publish for a handle, so
    // the gap between what the creator added and what a fan can see is a
    // property of the list rather than a second request.
    appointment: {
      unserved: unserved.length,
      provable: unserved.filter((a) => autoProvable(a.platform, a.id_kind ?? "handle", a.id)).length,
      verificationPending: ver?.renewal_pending === true,
      correctionPending: activeAccounts.some((a) => a.correction_pending === true),
      // Only a deadline that is actually near counts as an appointment: a date
      // six months out is not something to tell somebody about on day one.
      expiresAt: ver?.expiring_soon === true ? (ver.expires_at ?? null) : null,
    },
  });
  // I3-1086 — THE APPOINTMENT STEP IS THE ONLY ONE WHOSE ALL-CLEAR IS ALSO ITS
  // ALL-FAILED STATE. `appointmentClear` (onboarding.ts:178-181) is true when nothing
  // is unserved, nothing is pending and no deadline is near; a failed accounts read
  // gives an EMPTY list (nothing unserved) and a failed verification read gives
  // `null` (nothing pending, no deadline), so the four inputs read exactly as they do
  // for a creator who is genuinely finished. The other steps fail closed already —
  // a failed read leaves their counts at 0, which reads as NOT done — and
  // `contactMarked` above takes this same precaution explicitly. Not done is the
  // honest answer to a question whose reads did not land.
  return readsLanded ? steps : steps.map((s) => (s.id === "appointment" && s.done ? { ...s, done: false } : s));
}

async function renderOnboarding(accts: AccountStandingDto[] | null, ver: VerificationDto | null, readsLanded: boolean): Promise<string> {
  const psp = session.passportId();
  const steps = await onboardingSteps(accts, ver, readsLanded);
  // I3-2116: `obActive` is module state with no route lifetime -- once a
  // creator opens a step by hand it pins the panel forever, even after that
  // step is done, so the wizard never goes back to figuring out the next
  // incomplete step for the rest of the tab. Clear the pin the moment the
  // step it points at is done, so `nextStep` is consulted again.
  if (obActive && steps.find((s) => s.id === obActive)?.done) obActive = null;
  const active = obActive ?? nextStep(steps)?.id ?? null;
  let panel = "";
  if (active === "passkey") panel = obLinkOutPanel("#/passkeys", "Register this device on the Passkeys page");
  else if (active === "accounts") panel = obAccountsPanel();
  else if (active === "protection") panel = obProtectionPanel();
  else if (active === "reveal" && psp) panel = obRevealPanel(shareLink(PUBLIC_BASE, psp));

  // every bio they have; built from the API origin it was an api-subdomain URL

  else if (active === "biolink" && psp) panel = obBioLinkPanel(shareLink(PUBLIC_BASE, psp), shareLink(PUBLIC_BASE, psp) ? `${shareLink(PUBLIC_BASE, psp)}/card.svg` : null);

}

/**
 * Paint one result box. THE SECOND CHOKEPOINT: a partial render that would take
 * the one-time voucher panel off the screen asks first, and leaves the panel
 * untouched on a "no". Derived from the live DOM (`destroysOneTimePanel`), so
 * it holds for a handler nobody has written yet — the save-status box lives
 * INSIDE the panel and therefore never trips it. Returns false when the write
 * was refused.
 *
 * The two producers of one-time values ask BEFORE they mint as well; this is
 * the backstop, not the consent.
 */
function setResult(id: string, html: string, landed?: () => void): boolean {
  const el = document.getElementById(id);
  if (!el) return false;
  if (destroysOneTimePanel(el) && oneTime.pending > 0) {
    // I3-0422 a: the ask is the in-page dialog, so it cannot answer synchronously. The panel stays untouched and
    // this returns false NOW; the write (and `landed`) happen only on the creator's "discard", into the same box.
    void oneTime.consume().then((ok) => {
      if (!ok || !el.isConnected) return;
      el.innerHTML = html;
      landed?.();
    });
    return false;
  }
  el.innerHTML = html;
  landed?.();
  return true;
}

/**
 * Scroll a result box into view, IF the environment has that method.
 *
 * `scrollIntoView` is not implemented by jsdom, and every one of these calls
 * sits inside `guarded`, which has a `try/finally` and no `catch` — so a throw
 * here does not fail the handler visibly, it escapes as an UNHANDLED REJECTION.
 * Vitest reports those as errors and prints "This might cause false positive
 * tests", and it is right: `setResult` runs one line earlier, so the assertion
 * about the rendered outcome passes while the run exits non-zero. A cosmetic
 * scroll must never be able to do that, and it must never be able to swallow
 * the rest of a handler in a browser that lacks it either.
 */
function scrollResultIntoView(id: string): void {
  const el = document.getElementById(id);
  if (el && typeof el.scrollIntoView === "function") el.scrollIntoView({ block: "nearest" });
}

async function guarded(fn: () => Promise<void>): Promise<void> {
  if (busy) return;
  busy = true;
  try {
    await fn();
  } finally {
    busy = false;
  }
}

/**
 * I3-1206 (D-1079(4)): A LAPSED SESSION RETURNS TO ITS ROUTE WITH ITS TYPED VALUES. Set when the creator leaves a
 * screen for sign-in because the session lapsed — the `lapsed-signin` link every 401 banner carries (views.ts
 * `lapsedSigninLink`), or the focus re-check that finds this tab signed out — and held IN MEMORY only (typed-resume.ts
 * `typedSnapshot`). `signinReturn` hands its hash to the passkey sign-in (`creator-signin-form`, the re-entry door of a
 * claimed passport) when the SAME passport signs in, and
 * `resumeAfterPaint` puts the values back once that route has painted. Any other outcome drops it: another passport,
 * a sign-out, or a paint of any other route.
 */
interface LapsedResume { hash: string; passportId: string; values: Record<string, string> }
let lapsedResume: LapsedResume | null = null;

/** Stash the screen on screen now; false (nothing stashed) on the sign-in screen itself or without a passport. */
function stashLapsed(passportId: string | null): boolean {
  const hash = window.location.hash;
  if (!canStashLapsed(hash, passportId)) return false;
  lapsedResume = { hash, passportId: passportId!, values: typedSnapshot(root) };
  return true;
}

/** The hash a sign-in lands on (router.ts `signinLanding`); a stash for another passport is dropped here. */
function signinReturn(fallback: string): string {
  const land = signinLanding(lapsedResume, session.passportId(), fallback);
  if (!land.keepStash) lapsedResume = null;
  return land.hash;
}

/** After a paint (router.ts `resumeOnPaint`): sign-in keeps the stash; its own route gets the values back; any other
 * route drops it. */
function resumeAfterPaint(): void {
  const r = lapsedResume;
  if (!r) return;
  const what = resumeOnPaint(r.hash, window.location.hash, session.isSignedIn());
  if (what === "keep") return;
  lapsedResume = null;
  if (what === "restore") restoreTyped(root, r.values);
}

/** One query parameter off the current hash route (`#/questions?qa=…&qat=…`).
 * Returns null for a missing or empty value, so a caller can treat "absent" and
 * "blank" the same way — which for a paging cursor means "start at the top". */
function hashParam(key: string): string | null {
  const q = (window.location.hash.split("?")[1] ?? "").trim();
  if (!q) return null;
  try {
    const v = new URLSearchParams(q).get(key);
    return v && v.length > 0 ? v : null;
  } catch {
    return null;
  }
}

/** Parse the OAuth-callback return from the current hash: `#/signin?oauth=ok&code=…`
 * or `#/signin?oauth=err&reason=…`. The code is shape-validated (64 hex) before it's
 * ever used; anything off-shape is dropped. */
function oauthReturn(): { oauth?: "ok" | "err"; code?: string; reason?: string } {
  const q = (window.location.hash.split("?")[1] ?? "").trim();
  if (!q) return {};
  let p: URLSearchParams;
  try {
    p = new URLSearchParams(q);
  } catch {
    return {};
  }
  const oauth = p.get("oauth");
  if (oauth !== "ok" && oauth !== "err") return {};
  const code = p.get("code");
  const reason = p.get("reason");
  return { oauth, ...(code && /^[0-9a-f]{64}$/.test(code) ? { code } : {}), ...(reason ? { reason } : {}) };
}

 * I26-Q7 (b) (I30-APIS): remove one alert device through its hold (creator-contact-routes.ts pushRemovalDoor). A DELETE
 * with the session alone is answered 202 + the hold: a passkey hold carries `step_up`, which is signed at once and sent
 * back as the `assertion`; an emailed-code hold (or one no factor can confirm) is drawn by the repainted list, whose
 * `pending_removals` carry it. A passkey that does not approve sends nothing more and leaves the hold to confirm again.
 */
async function removePushDevice(id: string, btn: HTMLElement): Promise<void> {
  let r = await api.removeCreatorPushDevice(id);
  const stepUp = r.status === 202 && r.data?.confirm_by === "passkey" ? r.data.step_up : undefined;
  if (stepUp) {
    let approval: AssertionPackage;
    try {
      approval = await performAssertion(stepUp, creds);
    } catch (e) {
      await render();
      setResult("push-device-remove-result", banner("err", thrownCopy(e, "Removing this device needs approval from one of your passkeys, and that approval didn't complete. Nothing was removed."))); // DRAFT
      return;
    }
    r = await api.removeCreatorPushDevice(id, { assertion: approval });
  }
  if (r.ok) {
    await render(); // removed (200), or held (202): the repainted list shows which
    return;
  }
  btn.removeAttribute("disabled");
  setResult("push-device-remove-result", pushDeviceRemoveOutcome(r));
}

/** D-1059(1): the age boxes on the sign-in screen — the claim form's (bio code, X, manual) and the domain form's. */
const CLAIM_AGE_BOXES = ["claim-age-ack", "domain-age-ack"] as const;
/** D-1059(1) / D-1057(3): what every claim door is sent — the creator's own tick on a box and the age that box printed
 * (`data-min-age`, views.ts claimAgeAckView), so the server checks the age the creator saw. */
function claimAgeAck(boxId: (typeof CLAIM_AGE_BOXES)[number] = "claim-age-ack"): ClaimAgeAck {
  const input = document.querySelector<HTMLInputElement>(`#${boxId} input[name="age_ack"]`);
  const shown = Number(input?.dataset.minAge);
  return { age_ack: input?.checked === true, min_age: Number.isInteger(shown) ? shown : CLAIM_MIN_AGE_DEFAULT };
}
/** The manual-claim status poll is reached from either form: the first ticked box answers for it. */
function claimAgeAckAny(): ClaimAgeAck {
  const ticked = CLAIM_AGE_BOXES.map((id) => claimAgeAck(id)).find((a) => a.age_ack);
  return ticked ?? claimAgeAck();
}
/** I21-Q20 (I30-APIS): an `age_ack_required` refusal names the age the server holds (`min_age`): the boxes are redrawn,
 * unticked, at that age, and the server's sentence is returned for the banner. null for any other answer. */
function claimAgeRefused(data: { error?: string; min_age?: number; message?: string } | null): string | null {
  if (data?.error !== "age_ack_required") return null;
  for (const id of CLAIM_AGE_BOXES) {
    const box = document.getElementById(id);
    if (box && typeof data.min_age === "number") box.innerHTML = claimAgeAckView(data.min_age);
  }
  return typeof data.message === "string" && data.message !== "" ? data.message : "Tick the age confirmation above, then try again."; // approved, D-1365 W62
}
/** D-1057(3): the boxes print the server's minimum age (GET /claim/min-age) before the first submit, not the default.
 * A box already at that age is left alone (a tick survives); any other answer leaves the default and the refusal path. */
async function syncClaimMinAge(): Promise<void> {
  const r = await api.claimMinAge();
  const age = r.ok ? r.data?.min_age : undefined;
  if (typeof age !== "number") return;
  for (const id of CLAIM_AGE_BOXES) {
    const box = document.getElementById(id);
    if (box && Number(box.querySelector<HTMLInputElement>('input[name="age_ack"]')?.dataset.minAge) !== age) box.innerHTML = claimAgeAckView(age);
  }
}

/** Bind ONLY the freshly-rendered verify button inside #claim-step2. On a
 * not-yet failure the code view is re-rendered (same challenge) + a banner, so
 * the creator can retry verify without minting a new code. */
function wireClaimStep2(start?: import("./api.js").ClaimStartDto, subject?: string): void {
  const step2 = document.getElementById("claim-step2");
  step2?.querySelector<HTMLElement>('[data-act="claim-verify"]')?.addEventListener("click", (e) => {
    const challenge = (e.currentTarget as HTMLElement).dataset.challenge ?? "";
    void guarded(async () => {
      const snap = session.snapshot();
      const r = await api.claimVerify(challenge, claimAgeAck());
      if (r.ok && r.data?.ok && r.data.session && r.data.passport_id) {
        if (!session.setIfCurrent(snap, r.data.session, r.data.passport_id)) return; // I3-0986 c
        landAfterClaim(); // the welcome interstitial first, then the golden path
      } else if (r.status === 410 || r.data?.outcome === "expired" || r.data?.outcome === "invalid" || r.data?.outcome === "not_found") {
        // I3-0192 / I3-0730: the server considers this challenge DEAD. Re-showing
        // the same code with an error banner (the old behaviour) is a dead loop
        // -- the creator has no live code to place and no button that gets them
        // a new one. Drop the stash (so a later #/signin visit does not restore
        // this dead code claiming "it's still live" -- I3-0730) and point back
        // at the form above instead of re-rendering the dead code.
        if (subject) clearChallenge(subject);
        setResult(
          "claim-step2",
          banner("err", "That code is no longer live — it expired or was invalidated. Use the form above to get a new one; the old code will no longer work even if you post it."),
        );
      } else if (r.ok && r.data?.outcome === "pending_review") {
        // B27-CLAIMKIND: the proof matched and a person decides. The challenge is
        // CONSUMED, so the retry below would answer 409; the one-time claim_ref
        // is the only handle on the review — remember it, show it, poll it.
        if (subject) clearChallenge(subject);
        const ref = r.data.claim_ref;
        if (typeof ref === "string" && /^[0-9a-f]{64}$/.test(ref)) {
          // subject = `claim:<pid>|<canonical key v1|platform|kind|value>` (display fields only).
          const parts = subject ? subject.split("|") : [];
          const pid = r.data.passport_id ?? "";
          manualClaims.save({ passport_id: pid, claim_ref: ref, code: start?.code ?? "", platform: parts[2] ?? "", handle: parts.length > 1 ? (parts[parts.length - 1] ?? "") : "", started_at: new Date().toISOString() });
          setResult("claim-step2", claimReviewView(ref, manualClaims.get(pid)?.claim_ref === ref));
          wireManualStatus();
        } else {
          setResult("claim-step2", banner("err", "Your code matched and the claim went to a person to confirm, but the reference that follows it did not arrive, so this page cannot track it. Don't verify again: that code is spent."));
        }
      } else {
        const msg = banner("err", claimAgeRefused(r.data) ?? errorCopy(r.data?.error, `not verified yet (${r.data?.outcome ?? "pending"}) — check the code is live, then retry`));
        setResult("claim-step2", start ? claimCodeView(start) + msg : msg);
        if (start) wireClaimStep2(start, subject); // fresh button in fresh markup -- one binding
      }
    });
  });
}

/**
 * H9: poll a manual claim by its `claim_ref`, from wherever the ref came from —
 * the freshly-rendered panel, or the sign-in screen days later.
 *
 * `resultId` is the box to write into, so the SAME logic serves the panel
 * (`#manual-status-result`) and the returning-creator affordance
 * (`#manual-resume-result`) rather than being copied into two handlers that can
 * drift on the one path that must not.
 */
async function pollManualClaim(ref: string, resultId: string): Promise<void> {
  const snap = session.snapshot();
  const r = await api.manualClaimStatus(ref, claimAgeAckAny());
  if (r.ok && r.data?.state === "approved" && r.data.session && r.data.passport_id) {
    if (!session.setIfCurrent(snap, r.data.session, r.data.passport_id)) return; // I3-0986 c
    // The claim is finished, so the remembered ref has done its job: drop it
    // rather than leave a spent bearer in this browser's storage.
    manualClaims.clear(r.data.passport_id);
    landAfterClaim();
    await render();
    return;
  }
  if (r.ok && r.data?.state === "pending") {
    setResult(resultId, banner("ok", "Still under review — check back shortly. Make sure the code stays live on your profile until then."));
    return;
  }
  if (r.ok && r.data?.state === "declined") {
    setResult(resultId, banner("err", "This claim wasn't approved. If you think that's a mistake, start again with the code live on your profile."));
    return;
  }
  const ageRefusal = claimAgeRefused(r.data);
  if (ageRefusal !== null) {
    setResult(resultId, banner("err", ageRefusal));
    return;
  }
  if (r.status === 404) {
    setResult(resultId, banner("err", "We have no claim under that reference. It may have been for a different passport, or the reference may be mistyped — check it against the copy you saved."));
    return;
  }
  setResult(resultId, banner("err", errorCopy(r.data?.error, "Couldn't check the status just now — try again in a moment.")));
}

/** Bind the manual-claim status button inside the freshly-rendered panel. */
function wireManualStatus(boxId = "claim-step2"): void {
  const box = document.getElementById(boxId);
  const sbtn = box?.querySelector<HTMLElement>('[data-act="manual-status"]');
  sbtn?.addEventListener("click", (e) => {
    const ref = (e.currentTarget as HTMLElement).dataset.ref ?? "";
    void guarded(async () => {
      const label = sbtn.textContent ?? "";
      sbtn.setAttribute("disabled", "true");
      sbtn.textContent = "Checking…";
      await pollManualClaim(ref, "manual-status-result");
      sbtn.textContent = label;
      sbtn.removeAttribute("disabled");
    });
  });
  // THE REF, AS A VALUE THE CREATOR CAN ACTUALLY KEEP. It is reported honestly:
  // `writeText` resolving means the text is on the clipboard, and a refusal says
  // so and points at the box, which is the manual fallback (the same discipline
  // as the one-time voucher panel).
  box?.querySelector<HTMLElement>('[data-act="manual-ref-copy"]')?.addEventListener("click", () =>
    void guarded(async () => {
      const ta = document.getElementById("manual-claim-ref") as HTMLTextAreaElement | null;
      const value = ta?.value ?? "";
      if (!value) return;
      try {
        await navigator.clipboard.writeText(value);
        setResult("manual-ref-copy-result", banner("ok", "Copied. Paste it somewhere permanent now — a note, a password manager, an email to yourself. It cannot be shown again."));
      } catch {
        setResult("manual-ref-copy-result", banner("err", "Your browser refused the copy, so nothing is on your clipboard. Select the reference in the box above and copy it by hand — it cannot be shown again."));
      }
    }));
}

/**
 * THE LIVE BIO CODE, KEPT FOR AS LONG AS IT IS LIVE.
 *
 * A claim/proof challenge is ONE live slot per (account, caller) with a one-hour
 * TTL, and the panel that rendered it was its only copy. So closing the tab, or
 * clicking any nav pill, killed the claim for an hour: the code was gone, and
 * pressing the button again answered a refusal about a challenge already in
 * flight. The creator had a code live on their profile that they could no longer
 * read and could not replace.
 *
 * SESSION storage, deliberately, not local: this is a short-lived operational
 * value, not an identity, and it should die with the tab rather than linger on a
 * shared machine. Every field is shape-checked on the way out, an expired stash
 * is dropped rather than rendered, and nothing here is a bearer — the challenge
 * id is spendable only by a caller the server already keyed the challenge to.
 */
interface StashedChallenge {
  challenge_id: string;
  code: string;
  place_in: string;
  expires_at: string;
}
const CHALLENGE_LAST = "dhwaja.challenge.last";
const challengeKey = (subject: string): string => `dhwaja.challenge.${subject}`;

/** I3-1720: contact-email verify code, held for a repaint the way `heldChallenge`
 * holds a passkey ceremony's code -- except the server never gives the client
 * the code itself here (it's emailed), so there is nothing to redisplay, only
 * a flag that a live one exists. 15 minutes comfortably covers the code's own
 * TTL; an over-long flag just re-shows a box whose "wrong or expired" branch
 * already handles a truly dead code. */
const CONTACT_VERIFY_PENDING = "dhwaja.contact_verify.pending_until";
function stashContactVerifyPending(): void {
  try {
    sessionStorage.setItem(CONTACT_VERIFY_PENDING, String(Date.now() + 15 * 60 * 1000));
  } catch {
    /* private mode -- the box on screen right now is still correct */
  }
}
function clearContactVerifyPending(): void {
  try {
    sessionStorage.removeItem(CONTACT_VERIFY_PENDING);
  } catch {
    /* ignore */
  }
}
function contactVerifyPending(): boolean {
  let raw: string | null = null;
  try {
    raw = sessionStorage.getItem(CONTACT_VERIFY_PENDING);
  } catch {
    return false;
  }
  if (!raw) return false;
  const at = Number(raw);
  if (!Number.isFinite(at) || at <= Date.now()) {
    clearContactVerifyPending();
    return false;
  }
  return true;
}

/** I3-1723: `presentation()` read the PUBLIC, ~60s-cached route until I3-2104 moved it to the authenticated, uncached
 * GET /account/archive (api.ts presentation()). Before that move a delete that landed could be read back from a stale
 * cached copy for up to a minute, so the just-deleted item reappeared in the render right after `mediaDeleteOutcome`
 * said it was gone. The route move is the fix; this client-side filter, for the length of that old staleness, stays
 * as belt-and-braces (i5-fx1-recently-deleted-media.test.ts still pins it). */
const mediaDeletedUntil = new Map<string, number>();
function markMediaDeleted(hash: string): void {
  mediaDeletedUntil.set(hash, Date.now() + 90 * 1000);
}
function isRecentlyDeletedMedia(hash: string | null | undefined): boolean {
  if (!hash) return false;
  const until = mediaDeletedUntil.get(hash);
  if (until === undefined) return false;
  if (until <= Date.now()) {
    mediaDeletedUntil.delete(hash);
    return false;
  }
  return true;
}

function stashChallenge(subject: string, start: StashedChallenge): void {
  try {
    sessionStorage.setItem(challengeKey(subject), JSON.stringify(start));
    sessionStorage.setItem(CHALLENGE_LAST, subject);
  } catch {
    /* private mode / storage disabled — the panel on screen is still the copy */
  }
}

/** #132PS-211: on a stash miss after a refused start, ask the server for the SAME
 * live code (own passport only; 404 = none live) and re-stash it. Never throws. */
async function rereadProveCode(accountKey: string): Promise<StashedChallenge | null> {
  const r = await api.proveReread(accountKey).catch(() => null);
  if (!r?.ok || !r.data || typeof r.data.code !== "string") return null;
  stashChallenge(`prove:${accountKey}`, r.data);
  return r.data;
}

/** #132PS-211: the DNS-TXT sibling of `rereadProveCode`. */
async function rereadDomainRecord(subject: string, domain: string): Promise<import("./api.js").DomainClaimStartDto | null> {
  const r = await api.domainReread(domain).catch(() => null);
  if (!r?.ok || !r.data || typeof r.data.challenge_id !== "string") return null;
  stashDomainChallenge(subject, r.data);
  return r.data;
}

/** The stash for this subject, if it is still in date. Never throws. */
function heldChallenge(subject: string): StashedChallenge | null {
  let raw: string | null = null;
  try {
    raw = sessionStorage.getItem(challengeKey(subject));
  } catch {
    return null;
  }
  if (!raw) return null;
  let v: unknown;
  try {
    v = JSON.parse(raw);
  } catch {
    return null;
  }
  if (!v || typeof v !== "object") return null;
  const r = v as Record<string, unknown>;
  if (typeof r.challenge_id !== "string" || typeof r.code !== "string" || typeof r.place_in !== "string" || typeof r.expires_at !== "string") return null;
  const at = Date.parse(r.expires_at);
  // An unreadable deadline is treated as EXPIRED: rendering a code we cannot
  // date is how a creator ends up placing a dead one.
  if (!Number.isFinite(at) || at <= Date.now()) {
    clearChallenge(subject);
    return null;
  }
  return { challenge_id: r.challenge_id, code: r.code, place_in: r.place_in, expires_at: r.expires_at };
}

function clearChallenge(subject: string): void {
  try {
    sessionStorage.removeItem(challengeKey(subject));
    if (sessionStorage.getItem(CHALLENGE_LAST) === subject) sessionStorage.removeItem(CHALLENGE_LAST);
  } catch {
    /* ignore */
  }
}

/** Put a still-live sign-in claim code back on screen after a repaint, and put
 * the passport id + account key back in the form with it — a code with no
 * memory of which account it belongs to is not much use. */
function restoreClaimPanel(): void {
  let last: string | null = null;
  try {
    last = sessionStorage.getItem(CHALLENGE_LAST);
  } catch {
    return;
  }
  if (!last || !last.startsWith("claim:")) return;
  const held = heldChallenge(last);
  if (!held) return;
  // I3-0729: the account key itself contains "|" (e.g. "v1|reddit|handle|me"),
  // and the stash subject is `claim:${pid}|${key}`. pid never contains "|", so
  // split on the FIRST "|" only -- a naive .split("|") shatters the key and
  // destructuring only the first two segments truncates it to just "v1".
  const rest = last.slice("claim:".length);
  const sepIdx = rest.indexOf("|");
  const pid = sepIdx === -1 ? rest : rest.slice(0, sepIdx);
  const key = sepIdx === -1 ? "" : rest.slice(sepIdx + 1);
  const form = document.getElementById("claim-start-form") as HTMLFormElement | null;
  if (!form) return;
  const pidInput = form.querySelector<HTMLInputElement>('input[name="passport_id"]');
  const keyInput = form.querySelector<HTMLInputElement>('input[name="account_key"]');
  if (pidInput && pid) pidInput.value = pid;
  if (keyInput && key) keyInput.value = key;
  setResult("claim-step2", claimCodeView(held) + banner("ok", "This is the code you were given a moment ago — it's still live, so there's nothing to redo."));
  wireClaimStep2(held, last);
  // I3-2118: the stash is local and has no memory of what happened
  // server-side meanwhile -- if this claim finished (the code was entered
  // elsewhere, or a manual review approved it) while this tab sat unused, a
  // reload used to keep showing "still live" forever with nothing checking
  // back. `claimStart` is idempotent for a still-open challenge (I3-0664's
  // comment above), so a FRESH "already claimed" answer means the server is
  // telling us this is over, not still waiting -- swap the panel for that.
  void (async () => {
    const r = await api.claimStart(pid, key, claimAgeAck());
    if (!r.ok && typeof r.data?.error === "string" && r.data.error.includes("already claimed")) {
      clearChallenge(last);
      setResult("claim-step2", banner("ok", "This account has already been claimed. If that was you, sign in with your passkey or account key above."));
    }
  })();
}

/** Bind the mark-used button rendered into #rc-verify-result after a verify.
 * Bound only when that box is (re-)rendered so listeners never stack, and the
 * code + pool ride the button's data-attrs (the creator's own presented code). */

  const btn = document.getElementById("rc-verify-result")?.querySelector<HTMLElement>('[data-act="rc-redeem"]');
  btn?.addEventListener("click", () =>
    void guarded(async () => {
      const code = btn.dataset.code ?? "";
      const pool = btn.dataset.pool;
      if (!code) return;
      btn.setAttribute("disabled", "true");
      const r = await api.redeemReverseCode(code, pool && /^rcp_[0-9a-f]{16}$/.test(pool) ? pool : undefined);

    }));
}

/**
 * OFFER `body` to the browser as a download, and say only what we can see.
 *
 * WHAT THIS CANNOT DO, said plainly, because the previous version claimed it
 * could: THERE IS NO WAY TO OBSERVE A DOWNLOAD FROM A PAGE. Creating a Blob,
 * an object URL and a synthetic `<a>` click all succeed whether or not a single
 * byte reaches disk — "Downloads blocked", an enterprise policy, a sandboxed
 * frame and a full disk all throw nothing. The old `downloadText` returned
 * `true` on "the constructors didn't throw" and its own comment promised it
 * returned `false` for "a blocked download"; the panel then reported "downloaded
 * as dhwaja-codes-….txt" and RELEASED the leave-the-page guard on the strength
 * of it. That is a reassurance-shaped lie about the one artifact that cannot be
 * re-created.
 *
 * So the return value means exactly one thing — the browser ACCEPTED the
 * request — the copy says exactly that, and the guard is not released by it.
 * Only the creator can confirm the file exists, and "I've saved them" is where
 * they say so.
 */
function offerDownload(filename: string, body: string): boolean {
  try {
    const url = URL.createObjectURL(new Blob([body], { type: "text/plain" }));
    const a = document.createElement("a");
    a.href = url;
    a.download = filename;
    a.rel = "noopener";
    document.body.appendChild(a);
    a.click();
    a.remove();
    // Revoked on a timer, not synchronously: revoking in the same tick cancels
    // the download in some browsers, and never revoking would keep the codes
    // reachable in the page long after the screen they belong to is gone.
    setTimeout(() => URL.revokeObjectURL(url), 30_000);
    return true;
  } catch {
    return false; // no Blob/URL constructor, or the browser refused outright
  }
}

/**
 * THE ONLY WAY ONE-TIME VOUCHER VALUES REACH THE SCREEN.
 *
 * Render the panel, arm the guard and wire the panel's controls as ONE act, so
 * a producer cannot show the values without arming the guard. Both producers go
 * through here — `#/codes`'s Generate and the drop's optional reward codes —
 * which is the fix for the drop path having had no protection whatsoever: it
 * rendered `<ul class="voucherlist">` raw, with no "shown once" warning, no
 * save control, no acknowledgement and no leave-the-page prompt, and one nav
 * click threw the batch away.
 *
 * `prefixHtml` is whatever the producer wants above the panel (the drop's
 * "Dropped." confirmation). It is already-escaped view output, like every other
 * string this module hands to `setResult`.
 */
function showOneTimeCodes(resultId: string, poolId: string, codes: string[], opts: { prefixHtml?: string; afterAck?: () => void } = {}): void {
  // Arm only over markup that actually landed. Both producers take consent
  // before they mint, so the chokepoint inside setResult cannot refuse here
  // today — but a guard armed over a panel that is not on screen is the M-3
  // "the prompt keeps lying" defect rebuilt, so it fails closed instead.
  // I3-0422 a: arming and wiring ride on `landed`, so a write deferred behind the discard dialog still arms.
  setResult(resultId, (opts.prefixHtml ?? "") + rcGeneratedCodesView(codes, poolId), () => {
    oneTime.arm(codes.length);
    wireGeneratedCodes(poolId, codes, opts.afterAck ?? (() => void render()));
  });
}

/**
 * Bind the one-time generated-codes panel. ONE control does both halves of the
 * save (clipboard + file) because two buttons is two chances to do only one of
 * them — and each half is reported separately, because either can be refused
 * and "saved" over a failure is the reassurance-shaped lie that loses the batch.
 *
 * THE GUARD IS RELEASED BY THE ACKNOWLEDGEMENT AND BY NOTHING ELSE. It used to
 * be released by a save whose two halves both "worked", but a download cannot be
 * observed from a page (see `offerDownload`), so half of that success was never
 * a fact. The clipboard write IS observable — `writeText` resolving means the
 * text is on the clipboard — and it is reported as such; the file is reported as
 * handed to the browser, with the check left to the person who can actually do
 * it.
 *
 * Nothing here writes the values anywhere: they live in this closure and in the
 * read-only box on screen (the manual fallback when the browser blocks both),
 * and both die with the render.
 */
function wireGeneratedCodes(poolId: string, codes: string[], afterAck: () => void): void {
  const box = document.getElementById("rc-generated");
  box?.querySelector<HTMLElement>('[data-act="rc-codes-save"]')?.addEventListener("click", () =>
    void guarded(async () => {
      const body = generatedCodesText(codes);
      const filename = generatedCodesFilename(poolId);
      let copied = false;
      try {
        await navigator.clipboard.writeText(body);
        copied = true;
      } catch {
        copied = false; // blocked permission / insecure context / no clipboard API
      }
      const offered = offerDownload(filename, body);
      // Writes INSIDE the panel, so this never trips the setResult chokepoint.
      setResult("rc-codes-save-result", rcCodesSaveStatus({ copied, offered, filename }));
    }));
  box?.querySelector<HTMLElement>('[data-act="rc-codes-done"]')?.addEventListener("click", () => {
    oneTime.release(); // the creator says they hold them; that is the only witness there is
    afterAck(); // #/codes repaints so the pool's counts catch up; the drop keeps its confirmation
  });
}

/** PNG (`89 50 4E 47`) or JPEG (`FF D8 FF`) — the only two the media store keeps
 * (`storeSignedMedia`, services/trust-core/src/media.ts:230-239, which decides from
 * MAGIC BYTES and answers 415 on everything else). Read from the bytes, never from
 * `File.type`, which is the client's guess at what it picked up. */
function isPngOrJpeg(b: Uint8Array): boolean {
  return (b.length >= 4 && b[0] === 0x89 && b[1] === 0x50 && b[2] === 0x4e && b[3] === 0x47) || (b.length >= 3 && b[0] === 0xff && b[1] === 0xd8 && b[2] === 0xff);
}

        // which is what made "Handle" such an effective trap on a pid platform.

        // URL entry point — the one a fan's client calls — so a URL yields the
        // matching subject BY CONSTRUCTION rather than by correct transcription.

        const looksUrl = /^(?:[a-z][a-z0-9+.-]*:\/\/|[a-z0-9-]+\.[a-z]{2,}\/)/i.test(rawHandle);
        canon = looksUrl ? canonicalize(rawHandle) : canonicalAccount(rawPlatform, idKind, rawHandle);

          "ob-account-result",

          ),
        );

      }
      // The passkey signature IS the attestation: relationship_basis

        if (!(await askConfirm("Withdraw this listing from your public archive? Fans stop seeing it; anyone already holding it can still verify it. You can re-publish by minting again.", { title: "Withdraw this listing?", confirmLabel: "Withdraw" }))) return;
        const r = await api.withdrawRelease(id);
        if (r.ok) await render(); // the withdrawn row leaves the published set
        setResult("listing-withdraw-result", listingWithdrawOutcome(r));
      });
    });
  }

  // --- outstanding claim codes, by public id (I3-1259 / D-0845(4)) ---
  // List the codes this creator issued on one listing; each row revokes ONE
  // code by its public id, confirm-gated because a revoke can't be undone.
  const showOutstanding = async (releaseId: string, after?: string): Promise<void> => {
    const res = await api.outstandingCredentials(releaseId);
    setResult("codes-outstanding-result", outstandingCodesView(releaseId, res));
    const host = document.getElementById("codes-outstanding-result");
    if (after) {
      const out = host?.querySelector<HTMLElement>("#code-revoke-result");
      if (out) out.innerHTML = after;
      else if (host) host.insertAdjacentHTML("afterbegin", after);
    }
    for (const b of host?.querySelectorAll<HTMLElement>('[data-act="code-revoke"]') ?? []) {
      b.addEventListener("click", () => {
        const rel = b.dataset.id ?? "";
        const pid = b.dataset.pid ?? "";
        if (!/^rel_[0-9a-f]{16}$/.test(rel) || !/^crd_[0-9a-f]{16}$/.test(pid)) return;
        void guarded(async () => {
          if (!(await askConfirm(`Revoke claim code ${pid}? It stops redeeming immediately, and this can't be undone.`, { title: "Revoke this claim code?", confirmLabel: "Revoke code", retype: pid }))) return;
          const r = await api.revokeCredentialByPublicId(rel, pid);
          await showOutstanding(rel, codeRevokeOutcome(pid, r)); // re-list: the revoked row leaves
        });
      });
    }
    // I38-AGCRED AGC-06: one control per agency SEAT that minted an outstanding code here (`minted_by.seat`), which
    // revokes every outstanding code that seat minted on this listing in one call (api.ts revokeSeatCredentials).
    const seats = res.ok && res.data ? [...new Set(res.data.credentials.map((c) => c.minted_by?.seat).filter((x): x is string => typeof x === "string" && /^agm_[0-9a-f]{16}$/.test(x)))] : [];
    if (host && seats.length > 0) {
      const ctl = seats
        .map((sid) => `<button class="ds-btn" data-act="code-revoke-seat" data-id="${esc(releaseId)}" data-seat="${esc(sid)}">Revoke every code agency seat ${esc(sid)} minted here</button>`)
        .join(" ");
      host.querySelector<HTMLElement>("#code-revoke-result")?.insertAdjacentHTML("beforebegin", `<p class="roi-note">${ctl}</p>`);
    }
    for (const b of host?.querySelectorAll<HTMLElement>('[data-act="code-revoke-seat"]') ?? []) {
      b.addEventListener("click", () => {
        const rel = b.dataset.id ?? "";
        const sid = b.dataset.seat ?? "";
        if (!/^rel_[0-9a-f]{16}$/.test(rel) || !/^agm_[0-9a-f]{16}$/.test(sid)) return;
        void guarded(async () => {
          if (!(await askConfirm(`Revoke every outstanding claim code agency seat ${sid} minted on this listing? They stop redeeming immediately, and this can't be undone. Your own codes and other seats' codes are not touched.`, { title: "Revoke this seat's codes?", confirmLabel: "Revoke codes", retype: sid }))) return;
          const r = await api.revokeSeatCredentials(rel, sid);
          const n = r.ok && typeof r.data?.revoked_count === "number" ? r.data.revoked_count : null;
          const out =
            n === null
              ? codeRevokeOutcome(sid, r.ok ? { ok: false, status: 0, data: null } : { ok: false, status: r.status, data: typeof r.data?.error === "string" ? { error: r.data.error } : null }) // a 200 with no count is read as "could not tell"
              : n > 0
                ? `<div class="ds-banner banner banner-ok">Revoked ${esc(String(n))} code(s) agency seat <code>${esc(sid)}</code> minted — they will no longer redeem.</div>`
                : `<div class="ds-banner banner banner-warn">Nothing changed: agency seat <code>${esc(sid)}</code> has no outstanding codes on this listing.</div>`;
          await showOutstanding(rel, out); // re-list: the revoked rows leave
        });
      });
    }
  };
  for (const b of root.querySelectorAll<HTMLElement>('[data-act="codes-outstanding"]')) {
    b.addEventListener("click", () => {
      const id = b.dataset.id ?? "";
      if (!/^rel_[0-9a-f]{16}$/.test(id)) return;
      void guarded(() => showOutstanding(id));
    });
  }

  // --- signature editor (S91 · Slice 2) ---
  wireSign();
}

/**
 * Canvas glue for #/sign — the ONLY DOM/canvas code; all math is the pure,
 * unit-tested signature.ts. Art is decoded LOCALLY (createImageBitmap), fit to a
 * capped raster (planCanvasSize, mirrors the server's pixel cap), and the mark is
 * placed by computeSignatureLayout. On submit the canvas is exported to a PNG,
 * whose bytes are BOTH what the passkey signs (its sha256) and what's uploaded —
 * signAndStoreArtFlow forces those to be the same bytes.
 */
function wireSign(): void {
  const form = document.getElementById("sign-form") as HTMLFormElement | null;
  if (!form) return;
  const canvas = document.getElementById("sign-canvas") as HTMLCanvasElement | null;
  const submit = document.getElementById("sign-submit") as HTMLButtonElement | null;
  if (!canvas || !submit) return;

  let bitmap: ImageBitmap | null = null; // the decoded local art (never uploaded raw)
  // S91 follow-up: the two overlay marks. Both are LOCAL — the strokes never
  // leave the page and the signature PNG is composited into the export rather
  // than uploaded on its own, exactly as the art already is.
  let strokes: Stroke[] = [];
  let sigBitmap: ImageBitmap | null = null;

  const readSpec = (): SignatureSpec => {
    const fd = new FormData(form);
    const num = (k: string, dflt: number) => {
      const n = Number(fd.get(k));
      return Number.isFinite(n) ? n : dflt;
    };
    const raw = String(fd.get("sig_mode") ?? "text");
    const mode: SignatureMode = raw === "draw" || raw === "image" ? raw : "text";
    return {
      mode,
      text: String(fd.get("sig_text") ?? ""),
      strokes,
      xNorm: num("sig_x", 70) / 100,
      yNorm: num("sig_y", 90) / 100,
      scale: num("sig_scale", 6) / 100,
      color: String(fd.get("sig_color") ?? "#ffffff"),
      style: String(fd.get("sig_style") ?? "script") === "sans" ? "sans" : "script",
    };
  };

  // Composite at a chosen longest-side; the live preview uses the default, the
  // exporter re-composites smaller to fit the byte cap.
  const composite = (maxDim: number = MAX_OUTPUT_DIM): void => {
    if (!bitmap) return;
    const { width, height } = planCanvasSize(bitmap.width, bitmap.height, maxDim);
    canvas.width = width;
    canvas.height = height;
    const ctx = canvas.getContext("2d");
    if (!ctx) return;
    ctx.clearRect(0, 0, width, height);
    ctx.drawImage(bitmap, 0, 0, width, height);
    const spec = readSpec();
    // ONE SHADOW RULE FOR ALL THREE MARKS. It is what keeps a light mark legible
    // on light art, and a freehand or uploaded signature needs it for the same
    // reason a typed one does — omitting it on two of three would make "draw it"
    // quietly worse than "type it" on exactly the art where it matters.
    const shadow = (px: number): void => {
      ctx.shadowColor = "rgba(0,0,0,0.5)";
      ctx.shadowBlur = Math.max(1, Math.round(px / 12));
    };
    if (spec.mode === "draw") {
      const l = computeStrokeLayout(width, height, spec);
      if (l.paths.length > 0) {
        shadow(l.lineWidth * 6);
        ctx.strokeStyle = l.color;
        ctx.lineWidth = l.lineWidth;
        ctx.lineCap = "round";
        ctx.lineJoin = "round";
        for (const path of l.paths) {
          ctx.beginPath();
          path.forEach((pt, i) => { i === 0 ? ctx.moveTo(pt.x, pt.y) : ctx.lineTo(pt.x, pt.y); });
          ctx.stroke();
        }
        ctx.shadowBlur = 0;
      }
    } else if (spec.mode === "image") {
      const sig = sigBitmap;
      const l = sig ? computeImageLayout(width, height, sig.width, sig.height, spec) : null;
      if (sig && l) {
        shadow(l.height);
        ctx.drawImage(sig, l.x, l.y, l.width, l.height);
        ctx.shadowBlur = 0;
      }
    } else {
      const layout = computeSignatureLayout(width, height, spec);
      if (layout.text) {
        ctx.font = canvasFont(layout);
        ctx.fillStyle = layout.color;
        ctx.textBaseline = "alphabetic";
        shadow(layout.fontPx);
        ctx.fillText(layout.text, layout.x, layout.y);
        ctx.shadowBlur = 0;
      }
    }
  };
  const redraw = (): void => composite();

  const toPng = (): Promise<Blob | null> => new Promise((resolve) => canvas.toBlob((b) => resolve(b), "image/png"));

  /** Export a PNG that fits the server's byte cap, shrinking the raster down the
   * ladder (PNG has no quality knob). Returns null if even the floor is too big —
   * so we refuse BEFORE signing a drop whose bytes the upload would 413. */
  const exportWithinCap = async (): Promise<Uint8Array | null> => {
    // THE PLACEMENT IS FROZEN FOR THE DURATION. Each rung awaits `toBlob`, and
    // `redraw` is not guarded — a slider nudged between rungs changed the spec
    // mid-ladder, so a review measured an accepted blob carrying one placement
    // while the restored preview showed another. Disabling the controls is the
    // smallest honest fix: the creator cannot move a mark that is already being
    // exported, and the preview they are looking at is the bytes they will sign.
    const controls = [...form.querySelectorAll<HTMLInputElement | HTMLSelectElement>('[name^="sig_"]')];
    const wasDisabled = controls.map((c) => c.disabled);
    for (const c of controls) c.disabled = true;
    try {
      for (const dim of OUTPUT_DIM_LADDER) {
        composite(dim);
        const blob = await toPng();
        if (!blob) continue;
        if (blob.size <= MAX_OUTPUT_BYTES) return new Uint8Array(await blob.arrayBuffer());
      }
      return null;
    } finally {
      controls.forEach((c, i) => { c.disabled = wasDisabled[i] ?? false; });
    }
  };

  /**
   * PER-INPUT GENERATION TOKENS, NOT `guarded`.
   *
   * `guarded` is one module-global `busy` flag shared by 77 call sites, and it
   * DROPS rather than queues. A review measured the consequence: pick art A,
   * then pick art B while anything at all is in flight, and B's decode never
   * runs — the file input shows B's name, the preview shows A, and the passkey
   * signs A. `exact_hash` then binds a signature to an image the creator did not
   * select. Neither file handler touches the network, so there is nothing for a
   * mutex to protect; what they need is last-write-wins, which is what a
   * generation token is. A superseded decode closes its own bitmap and returns.
   */
  let artSeq = 0;
  let sigSeq = 0;

  /** RELEASE THE DECODES WHEN THE SCREEN GOES. Both bitmaps were closed only
   * when the SAME input was re-picked, so leaving `#/sign` left them unreachable
   * and unclosed — `wireSign` builds a fresh closure on every render, and a
   * review measured two full-resolution 8000x6000 decodes (~192 MB of native
   * memory) retained after two navigations, unbounded across a session. */
  window.addEventListener(
    "hashchange",
    () => {
      bitmap?.close?.();
      sigBitmap?.close?.();
      bitmap = null;
      sigBitmap = null;
    },
    { once: true },
  );

  const fileInput = form.querySelector<HTMLInputElement>('input[name="art"]');
  fileInput?.addEventListener("change", () => {
    void (async () => {
      const seq = ++artSeq;
      const f = fileInput.files?.[0];
      bitmap?.close?.();
      bitmap = null;
      submit.disabled = true;
      const hint = document.getElementById("sign-hint");
      if (!f) return;
      if (f.size > 32 * 1024 * 1024) {
        if (hint) hint.textContent = "That image is too large to edit here (max 32 MB).";
        return;
      }
      let bm: ImageBitmap;
      try {
        bm = await createImageBitmap(f);
      } catch {
        if (seq === artSeq && hint) hint.textContent = "Could not read that image — use a PNG or JPEG.";
        return;
      }
      if (seq !== artSeq) {
        bm.close?.(); // a newer pick won; this decode is stale
        return;
      }
      bitmap = bm;
      if (hint) hint.textContent = "Drag the sliders to place your signature.";
      submit.disabled = false;
      redraw();
    })();
  });

  // --- the two overlay marks (S91 follow-up / DIRECTION §3) -----------------

  /** Show exactly the block for the selected mode. The radios are inside the
   * form, so `readSpec` already sees the change; this is only the disclosure. */
  const syncMode = (): void => {
    const mode = readSpec().mode ?? "text";
    for (const el of form.querySelectorAll<HTMLElement>("[data-sigmode]")) {
      (el as HTMLElement).hidden = el.dataset.sigmode !== mode;
    }
  };

  /**
   * THE DRAWING PAD. Points are captured in the pad's OWN normalized space
   * (0..1), never in pixels: the pad's size is a viewport detail, and the same
   * mark has to composite identically onto a 2048px export and an 800px one.
   * Pointer events only — one code path for mouse, trackpad, pen and touch —
   * with capture, so a stroke that leaves the pad mid-gesture still ends
   * cleanly instead of leaving the pad stuck in a drawing state.
   */
  const pad = document.getElementById("sig-pad") as HTMLCanvasElement | null;
  if (pad) {
    let current: StrokePoint[] | null = null;
    /** WHICH pointer is drawing. Without it a second finger's `pointerdown`
     * overwrote `current` and silently discarded the first finger's ink, and a
     * `pointermove` from ANY pointer appended to the active stroke — so a
     * two-finger gesture produced one stroke made of two hands. */
    let activeId: number | null = null;
    const paint = (): void => {
      const c = pad.getContext("2d");
      if (!c) return;
      c.clearRect(0, 0, pad.width, pad.height);
      c.strokeStyle = "#111827";
      c.lineWidth = Math.max(2, Math.round(pad.height / 40));
      c.lineCap = "round";
      c.lineJoin = "round";
      for (const st of strokes) {
        c.beginPath();
        st.forEach((pt, i) => { i === 0 ? c.moveTo(pt.x * pad.width, pt.y * pad.height) : c.lineTo(pt.x * pad.width, pt.y * pad.height); });
        c.stroke();
      }
    };
    const at = (e: PointerEvent): StrokePoint => {
      const r = pad.getBoundingClientRect();
      return { x: r.width > 0 ? (e.clientX - r.left) / r.width : 0, y: r.height > 0 ? (e.clientY - r.top) / r.height : 0 };
    };
    pad.addEventListener("pointerdown", (e) => {
      if (current) return; // a stroke is already in progress — ignore the second pointer
      pad.setPointerCapture(e.pointerId);
      activeId = e.pointerId;
      current = [at(e)];
      e.preventDefault();
    });
    pad.addEventListener("pointermove", (e) => {
      if (!current || e.pointerId !== activeId) return;
      current.push(at(e));
      // Draw the in-progress stroke without committing it, so the pad feels live.
      const c = pad.getContext("2d");
      if (c && current.length > 1) {
        const a = current[current.length - 2]!;
        const b = current[current.length - 1]!;
        c.beginPath();
        c.moveTo(a.x * pad.width, a.y * pad.height);
        c.lineTo(b.x * pad.width, b.y * pad.height);
        c.stroke();
      }
    });
    const end = (e?: PointerEvent): void => {
      if (!current || (e && e.pointerId !== activeId)) return;
      activeId = null;
      // Sanitised on COMMIT, not on export: the caps then bound what is held in
      // memory as well as what is walked, and the pad shows the creator exactly
      // the ink that will be composited.
      strokes = sanitizeStrokes([...strokes, current]);
      current = null;
      // I3-1208 (I34-W2-CLIENTS): drawn, unsent ink is unsent input to the dirty-form guard (dirty-form.ts hasDirtyForm).
      if (strokes.length > 0) pad.setAttribute(UNSENT_INK_ATTR, "");
      paint();
      redraw();
    };
    pad.addEventListener("pointerup", end);
    pad.addEventListener("pointercancel", end);
    pad.addEventListener("pointerleave", end);
    document.getElementById("sig-pad-clear")?.addEventListener("click", () => {
      strokes = [];
      pad.removeAttribute(UNSENT_INK_ATTR);
      paint();
      redraw();
    });
  }

  /** The uploaded signature PNG. Decoded LOCALLY and composited into the export;
   * the original is never uploaded, exactly like the art. The byte cap is the
   * real defence (`createImageBitmap` has already allocated by the time we can
   * see dimensions); `computeImageLayout` refuses the rest. */
  const sigInput = form.querySelector<HTMLInputElement>('input[name="sig_image"]');
  sigInput?.addEventListener("change", () => {
    void (async () => {
      const seq = ++sigSeq; // same last-write-wins rule as the art input above
      const f = sigInput.files?.[0];
      sigBitmap?.close?.();
      sigBitmap = null;
      const hint = document.getElementById("sign-hint");
      if (!f) {
        redraw();
        return;
      }
      if (f.size > MAX_SIG_IMAGE_BYTES) {
        if (hint) hint.textContent = "That signature file is too large (max 4 MB).";
        redraw();
        return;
      }
      let bm: ImageBitmap;
      try {
        bm = await createImageBitmap(f);
      } catch {
        if (seq === sigSeq && hint) hint.textContent = "Could not read that signature image — use a PNG.";
        redraw();
        return;
      }
      if (seq !== sigSeq) {
        bm.close?.();
        return;
      }
      sigBitmap = bm;
      redraw();
    })();
  });

  // Live preview as controls move (no network, no ceremony — safe to run freely).
  for (const el of form.querySelectorAll<HTMLElement>('[name^="sig_"]')) {
    el.addEventListener("input", () => {
      syncMode();
      redraw();
    });
    el.addEventListener("change", () => {
      syncMode();
      redraw();
    });
  }
  syncMode();

  // Holds the last successfully-signed bytes so a soft store failure can retry the
  // UPLOAD ALONE (idempotent, content-addressed) — never re-minting a new drop.
  let pendingUpload: Uint8Array | null = null;

  const showSignResult = (r: { release_id?: string; stored?: boolean; storeError?: string }): void => {
    pendingUpload = r.stored === false ? pendingUpload : null;
    setResult("sign-result", signResultView({ ...r, retryable: r.stored === false && pendingUpload !== null }));
    document.getElementById("sign-result")?.querySelector<HTMLElement>('[data-act="retry-upload"]')?.addEventListener("click", () =>
      guarded(async () => {
        if (!pendingUpload) return;
        const up = await api.uploadMedia(pendingUpload);
        if (up.ok) {
          pendingUpload = null;
          setResult("sign-result", signResultView({ ...r, stored: true }));
        } else {
          showSignResult({ ...r, stored: false, storeError: errorCopy(up.data?.error, (up.status === 503 ? "media storage is not available yet" : "still couldn't store the art — try again shortly")) });
        }
      }),
    );
  };

  form.addEventListener("submit", (e) => {
    e.preventDefault();
    void guarded(async () => {
      if (!bitmap) {
        setResult("sign-result", banner("err", "choose an image first"));
        return;
      }
      const fd = new FormData(form);
      const title = String(fd.get("title") ?? "").trim();
      if (!title) {
        setResult("sign-result", banner("err", "a collectible title is required"));
        return;
      }
      // ONE RULE FOR ALL THREE MARKS. `sig_text` used to carry the HTML
      // `required` attribute, which is the browser's rule and only knows about
      // one of them — a creator who drew their signature and left the text box
      // empty would have been refused by the form, and one who typed a name and
      // then switched to "upload" without choosing a file would have been let
      // through to sign art with no mark on it. `hasMark` asks the mode which
      // input applies, in the same module that composites it.
      const spec = readSpec();
      if (!hasMark(spec, sigBitmap !== null)) {
        setResult("sign-result", banner("err", "add your signature first — type it, draw it, or upload one."));
        return;
      }
      // …AND THAT IT LANDS. `hasMark` asks whether there is a mark; the sliders
      // reach placements where there is one and it falls entirely off the art
      // (`sig_x` max 100 puts its left edge on the right border; `sig_y` min 0
      // puts its baseline on the top one). A review drove all three modes to a
      // completed passkey ceremony on art with nothing composited onto it, whose
      // sha256 then became the release's `exact_hash`. Refusing here is
      // recoverable — move a slider; accepting is not.
      const plan = planCanvasSize(bitmap.width, bitmap.height);
      if (!markOnCanvas(plan.width, plan.height, spec, sigBitmap?.width, sigBitmap?.height)) {
        setResult("sign-result", banner("err", "your signature is off the edge of the image — move it back onto the art before signing."));
        return;
      }
      // Export a PNG that fits the byte cap BEFORE signing anything — a drop whose
      // bytes exceed the cap would mint an un-hostable collectible (review #1).
      const bytes = await exportWithinCap();
      if (!bytes) {
        setResult("sign-result", banner("err", "This image is too detailed to store even after shrinking (over 8 MB as a PNG). Try a smaller or simpler source image."));
        composite(); // restore the full-size preview
        return;
      }
      const input: DropInputDto = {
        collectible: {
          release_id: mintReleaseId(),
          title,
          version: String(fd.get("version") ?? "1"),
          media: [{ media_kind: "image", exact_hash: "" }], // overridden to sha256(bytes) by the flow
        },
      };
      const editionSize = Number(fd.get("edition_size"));
      if (Number.isFinite(editionSize) && editionSize >= 1) input.collectible.edition_size = Math.trunc(editionSize);
      const r = await signAndStoreArtFlow(api, creds, input, bytes);
      composite(); // restore the full-size preview regardless of outcome
      if (!r.ok) {
        setResult("sign-result", banner(r.indeterminate ? "warn" : "err", r.error ?? NOT_DONE_COPY));
        return;
      }
      pendingUpload = r.stored ? null : bytes; // keep the signed bytes for an upload-only retry
      showSignResult(r);
      // I3-1208 b (I34-FX2-CLIENTS): signed, so the ink and the typed words are sent words now — and only now; every
      // refusal above returns with them still unsent (views.ts signView's form carries SENT_ON_SUCCESS_ATTR).
      markSent(form);
    });
  });
}

/**
 * M10 — the Billing page (`#/billing`). This is the ONLY function in the console
 * that reads billing state or hands off to a payment processor, and it renders
 * nothing else — no verdict, no passport, no warning — so the paywall separation
 * stays structural rather than cultural (see the header of billing.ts).
 *
 * Fail closed at every branch: an unreadable entitlement and a 503 both land on
 * the SAME designed "billing isn't set up yet" card with NO button, rather than
 * on an error state or a dead control. Nothing here can degrade anything
 * protective.
 */
function wireBillingPage(): void {
  const holder = root.querySelector<HTMLElement>("[data-plan-state]");
  if (!holder) return;
  // Stripe's success URL carries ?upgraded=1 in the SEARCH string (not the
  // hash), so the welcome moment survives hash routing.
  const justUpgraded = typeof window !== "undefined" && window.location.search.includes("upgraded=1");
  const chosenInterval = (): BillingInterval => (root.querySelector<HTMLInputElement>('input[name="pro_interval"]:checked')?.value === "month" ? "month" : "year");

  void api.creatorEntitlement().then(({ status, data }) => {
    const paint = (over?: { status?: number }): void => {
      holder.innerHTML = planStateView(planState({ status: over?.status ?? status, ent: data, interval: chosenInterval(), justUpgraded }));
      const btn = holder.querySelector<HTMLButtonElement>('[data-act="plan-action"]');
      if (!btn) return; // the 503 and expired-session cards carry no button by design
      btn.addEventListener("click", () => {
        void guarded(async () => {
          btn.disabled = true;
          const kind = btn.dataset.planKind;
          const interval = btn.dataset.planInterval === "month" ? "month" : "year";
          // Two kinds reach here: "portal" (Stripe) and "checkout". A paying
          // creator is NEVER offered checkout (`planState` withholds it on
          // active/grace), so this can't open a second live subscription.
          const r = kind === "portal" ? await api.stripePortal() : await api.startProCheckout(interval);
          const url = r.data?.url;
          // I3-1251 (BF-06): an in-flight checkout on the OTHER interval was replayed.
          // Say so and let the creator choose; never a silent redirect.
          // The second press, after the creator has read the notice, continues
          // through the one gated hand-off below.
          const switched = kind === "portal" ? null : checkoutIntervalSwitched(interval, r.data as { interval?: unknown } | null);
          if (switched && btn.dataset.planSwitchSeen !== switched && isSafeHandoffUrl(url)) {
            btn.disabled = false;
            btn.dataset.planSwitchSeen = switched;
            const note = holder.querySelector<HTMLElement>("[data-plan-msg]");

/** The export file: one address per line under a header, confirmed time beside it. A field is quoted only when it
 * holds a comma, quote or line break (RFC 4180); a leading formula character is neutralised for spreadsheet apps. */
export function newsletterCsv(list: readonly { address: string; confirmed_at: string }[]): string {
  const cell = (v: string): string => {
    const safe = /^[=+\-@\t\r]/.test(v) ? `'${v}` : v;
    return /[",\r\n]/.test(safe) ? `"${safe.replace(/"/g, '""')}"` : safe;
  };
  return ["email,confirmed_at", ...list.map((s) => `${cell(s.address)},${cell(s.confirmed_at)}`)].join("\r\n") + "\r\n";
}
