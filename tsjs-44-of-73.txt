/**
 * S-CW6c: creator payouts. Earnings are recorded per sale (S-CW6a `sales`); this
 * module reads the withdrawable balance, manages the Stripe Connect (Standard)
 * binding, and runs the weekly payout.
 *
 * MONEY DISCIPLINE:
 *  - Cleared (withdrawable) = Σ creator_net for sales past the 14-day hold
 *    (clears_at ≤ now) and not reversed, minus Σ already-paid payouts. Pending
 *    = the same sum still inside the hold. Both are folds; nothing is stored.
 *  - A creator with no ACTIVE Connect binding simply accrues — held
 *    indefinitely, never paid (DECIDED). The binding is passkey-gated to
 *    activate (passkeys.ts), so a stolen session cannot make an attacker's
 *    account the payout destination.
 *  - The weekly run is idempotent: one payout row per (passport, ISO-week),
 *    the amount RESERVED on the row, and a Stripe transfer idempotency key
 *    DERIVED from (passport, ISO-week) (`transferIdempotencyKey`, below) — a
 *    cron retry, an overrun, or a restore that loses the row cannot double-pay
 *    inside Stripe's idempotency retention window; past it (I3-0028, D-0541,
 *    D-0911(1)) EVERY send, at every door, reconciles against Stripe per
 *    creator ACCOUNT first (stripe.ts `createConnectTransfer`): a transfer in
 *    the payout's own group is recorded only when its amount, destination and
 *    reversal state equal ours, and a transfer Stripe sent to the destination
 *    that our ledger does not hold — in ANY week, from either door — holds the
 *    payout for an operator (`PAYOUT_RECONCILE_HOLD_AT`) and sends nothing.
 *    DEPLOY.md §1 "the payout half".
 */

/** $25 minimum — below it, the balance rolls forward to a later week. */

/**
 * UI-AUDIT — THE COOLING-OFF, AND WHY IT IS ON THE BIND RATHER THAN THE UNBIND.
 *
 * `payout.ts` gives a payout TAG change 24 hours before it goes live
 * (`PAYOUT_COOLING_OFF_MS`), and the Connect binding — a strictly larger
 * exposure, since it is where Dhwaja transfers a creator's whole cleared balance
 * rather than where one fan is invited to pay — had none. The open question was
 * written down as "match the ceremony", and the obvious reading of that is to
 * put friction on the disconnect. That would be exactly backwards.
 *
 * The threat is an attacker who reaches a creator's session and binds THEIR
 * Stripe account. The passkey confirm is what makes that hard, and it is already
 * there. What a cooling-off adds is a WINDOW: money does not move to a
 * newly-activated destination for 24 hours, so a creator who gets the alert, or
 * simply opens their console, can disconnect before anything leaves. Friction on
 * the DISCONNECT would hand the attacker that window instead — it is the
 * `agency_revoked` / `war_opt_out` asymmetry this codebase states in three other
 * places: exit is wider and faster than entry.
 *
 * So: the disconnect stays instant, and the payout run skips a binding younger
 * than this. Nothing is lost by waiting — an unpaid balance rolls forward by
 * design and is held indefinitely, so the only cost is that a creator's first
 * payout can be one run later than it would otherwise have been.
 *
 * The same constant as the payout tags, deliberately: two ceremonies about where
 * a creator's money goes should not disagree about how long a person gets to
 * notice.
 */
export const CONNECT_COOLING_OFF_MS = PAYOUT_COOLING_OFF_MS;

/**
 * THE STRIPE IDEMPOTENCY KEY, DERIVED FROM FACTS A RESTORE CANNOT CHANGE
 * (I3-0028). The key dispatched at :1228 and :1368 used to be the payout row's
 * own `payout_id` — random per reservation — so it lived and died with that
 * row. A D1 point-in-time restore to a pre-run snapshot (DEPLOY.md's recovery
 * path) returns a database in which the payout never happened, and the next
 * tick reserves the same money under a NEW random key that Stripe has never
 * seen: the key it deduped on was rolled back with the row.
 *
 * (passport, ISO week) is the payout's NATURAL key — `creator_payouts` carries
 * `UNIQUE (passport_id, period_key)` (migrations/0036_creator_payouts.sql:38),
 * so at most one payout per creator per week is representable and two
 * legitimately distinct payouts can never derive the same key.
 *
 * THE AMOUNT IS DELIBERATELY NOT AN INPUT. Including it would mint a fresh key
 * the moment a post-restore replay priced a different amount (sales cleared in
 * the interval) — the double pay again, one input wider. Excluded, Stripe sees
 * key reuse with different parameters and answers 400; `createConnectTransfer`
 * maps `!res.ok` to null (stripe.ts:1071) and the row stays `pending` on its
 * backoff. A refusal is the correct direction of failure here.
 *
 * The ROW ID stays random (`mintPayoutId`): identity and key are two concerns,
 * and a derived `pay_` id would encode its own creation week —
 * `tests/invariants/id-entropy.test.ts` pins exactly that.
 *
 * NOT `pay_`-shaped, and no `_`-prefix at all, so it is not an identifier the
 * id-entropy source scan can mistake for a seventh id type.
 *
 * PLAIN, NOT HASHED. A hash would buy opacity nothing needs — the key never
 * leaves the server and rides only in Stripe's `idempotency-key` header
 * (stripe.ts:1065) — and would make this module a `sha256Hex` importer, which
 * `tests/invariants/single-hasher.test.ts` allowlists by file. `psp_` ids and
 * `YYYY-Www` periods contain no `:`, so the concatenation is injective: two
 * distinct (passport, period) pairs cannot render the same key. Stripe caps a
 * key at 255 characters; this is 39.
 */
export function transferIdempotencyKey(passportId: string, periodKey: string): string {
  return `cpk:v1:${passportId}:${periodKey}`;
}

export interface Earnings {
  cleared_cents: number; // withdrawable now
  pending_cents: number; // earned but inside the 14-day hold
  paid_cents: number; // lifetime paid out
  /**
   * UI-AUDIT F-C4 — MONEY THAT WAS IN NO TILE.
   *
   * A payout row is created `pending` with its amount RESERVED, and the weekly
   * run flips it to `paid` only when Stripe confirms the transfer. A transfer
   * that keeps failing — a Connect account that got restricted, a destination
   * that stopped accepting transfers — leaves the row `pending` forever, by
   * design: that is what lets the next tick replay it under the same
   * idempotency key instead of double-paying.
   *
   * But `cleared_cents` subtracts every reserved row, `paid_cents` counts only
   * paid ones, and `pending_cents` means something else entirely (earnings still
   * inside the 14-day hold). So the amount sat in NONE of the three figures a
   * creator reads as theirs — it appeared only inside `lifetime_cents`, where a
   * drop in "Ready to pay out" reads as money that quietly disappeared. A
   * creator watching their balance fall with no payout arriving had no figure to
   * point at and nothing to ask about.
   *
   * It is its own number now, and the console gives it its own tile. Nothing
   * about the reservation changed — only that the money is visible.
   */
  in_flight_cents: number; // reserved for a payout that has not completed
  lifetime_cents: number; // lifetime gross earned (cleared+pending+paid basis)
  /** I3-1170: what refunds/chargebacks took back — the creator's net on reversed sales. */
  reversed_cents: number;
  /** D-0960(1): lifetime creator net that LOST disputes attributed to already-cleared
   * sales (`creator_dispute_debits`) — netted against payouts, never a reversed sale. */
  dispute_debit_cents: number;
  /** D-0960(1): the CARRIED negative balance — debits not yet covered by cleared
   * earnings. Taken from the next payouts before anything is paid; 0 when nothing is owed. */
  owed_cents: number;
  connect_status: "none" | "connecting" | "pending" | "active" | "revoked";
  /** When a newly-bound destination starts receiving transfers, or null when
   * nothing is waiting. See CONNECT_COOLING_OFF_MS: the window exists so a
   * binding made during a session compromise can be disconnected before money
   * moves, which is only true if the creator can see it. */
  payouts_live_at: string | null;
  min_payout_cents: number;
}

/** A creator's earnings + payout-connection state (pure reads / folds). */
export async function creatorEarnings(db: DbLike, passportId: string, nowMs: number): Promise<Earnings> {
  if (!PSP_RE.test(passportId)) throw new ConnectError("bad passport id");
  const nowIso = new Date(nowMs).toISOString();
  const cleared = await db
    .prepare("SELECT COALESCE(SUM(creator_net_cents),0) AS c FROM sales WHERE passport_id = ? AND reversed = 0 AND clears_at <= ?")
    .bind(passportId, nowIso)
    .first<{ c: number }>();
  const pending = await db
    .prepare("SELECT COALESCE(SUM(creator_net_cents),0) AS c FROM sales WHERE passport_id = ? AND reversed = 0 AND clears_at > ?")
    .bind(passportId, nowIso)
    .first<{ c: number }>();
  const paid = await db
    .prepare("SELECT COALESCE(SUM(amount_cents),0) AS c FROM creator_payouts WHERE passport_id = ? AND status = 'paid'")
    .bind(passportId)
    .first<{ c: number }>();
  // RESERVED = paid + pending payout rows. Subtracting BOTH (review HIGH) is
  // what stops a still-owed but in-flight amount from being counted as
  // withdrawable again — the same reservation the payout run relies on.
  const reserved = await db
    .prepare(`SELECT COALESCE(SUM(amount_cents),0) AS c FROM creator_payouts WHERE passport_id = ? AND status IN ${RESERVED_PAYOUT_STATUS_SQL}`)
    .bind(passportId)
    .first<{ c: number }>();
  // I3-1170: refunds.ts reverses a sale (reversed = 1) only inside its hold.
  const reversed = await db
    .prepare("SELECT COALESCE(SUM(creator_net_cents),0) AS c FROM sales WHERE passport_id = ? AND reversed = 1")
    .bind(passportId)
    .first<{ c: number }>();
  const debits = await db.prepare(`SELECT ${DISPUTE_DEBIT_CENTS("?")} AS c`).bind(passportId).first<{ c: number }>();
  const conn = await db.prepare("SELECT status, activated_at FROM creator_connect WHERE passport_id = ?").bind(passportId).first<{ status: string; activated_at: string | null }>();
  // D-0960(1): debits are netted BEFORE the floor, and what the floor would hide is
  // reported as `owed_cents` (QUEUE #490: a floored debt is an invisible over-claw).
  const netAvailable = (cleared?.c ?? 0) - (reserved?.c ?? 0) - (debits?.c ?? 0);
  const clearedNet = Math.max(0, netAvailable);
  // IN FLIGHT = the OPEN rows only (pending/failed): committed to a payout that
  // has not completed. I16-RG-MONEY:17: this was reserved MINUS paid, and
  // reserved carries `voided` (I3-0251) — a payout an operator SETTLED outside
  // Stripe then showed as still in flight forever. removal.ts heldBalance
  // restates this split (B24-518) and must move with it.
  const open = await db
    .prepare("SELECT COALESCE(SUM(amount_cents),0) AS c FROM creator_payouts WHERE passport_id = ? AND status IN ('pending','failed')")
    .bind(passportId)
    .first<{ c: number }>();
  const inFlight = open?.c ?? 0;
  return {
    cleared_cents: clearedNet,
    pending_cents: pending?.c ?? 0,
    paid_cents: paid?.c ?? 0,
    in_flight_cents: inFlight,
    lifetime_cents: clearedNet + (pending?.c ?? 0) + (reserved?.c ?? 0),
    reversed_cents: reversed?.c ?? 0,
    dispute_debit_cents: debits?.c ?? 0,
    owed_cents: Math.max(0, -netAvailable),
    connect_status: (conn?.status as Earnings["connect_status"]) ?? "none",
    // When the cooling-off on a newly-bound destination lifts, or null when
    // there is nothing waiting. The console shows it rather than leaving a
    // creator to wonder why an active binding did not pay — an unexplained
    // delay on a money screen reads as a fault.
    payouts_live_at: conn?.status === "active" && conn.activated_at && Date.parse(conn.activated_at) + CONNECT_COOLING_OFF_MS > nowMs ? new Date(Date.parse(conn.activated_at) + CONNECT_COOLING_OFF_MS).toISOString() : null,
    min_payout_cents: MIN_PAYOUT_CENTS,
  };
}

/**
 * Start a Connect OAuth flow: reserve a single-use binding nonce against the
 * passport (status 'connecting') and return the state to put in the OAuth URL.
 * Re-startable (a creator who abandons the flow can start again); an already
 * ACTIVE binding is refused (changing an established payout account is its own
 * flow, not covered here — S-CW6c ships first-bind).
 */
export async function startConnect(db: DbLike, passportId: string, mintState: () => string, now: () => string = () => new Date().toISOString()): Promise<{ state: string }> {
  if (!PSP_RE.test(passportId)) throw new ConnectError("bad passport id");
  const existing = await db.prepare("SELECT status FROM creator_connect WHERE passport_id = ?").bind(passportId).first<{ status: string }>();
  // Still refused while a binding is LIVE — but this is no longer a dead end:
  // `unbindPayoutAccount` moves the row to 'revoked' (passkey-gated), and a
  // revoked row starts a fresh flow here. UI-AUDIT F-C3: before that door
  // existed, this line was what made the first account a creator confirmed their
  // payout destination permanently.
  if (existing?.status === "active") throw new ConnectError("disconnect the current payout account first", 409);
  const state = mintState();
  if (!STATE_RE.test(state)) throw new ConnectError("bad state", 500);
  const ts = now();
  // Upsert to 'connecting' with a fresh nonce; clears any prior stale account.
  //
  // THE REFUSAL IS IN THE WRITE, NOT ONLY IN THE READ ABOVE (S-070; this is the
  // OAuth half of the F15 extension, found beside BF-14). The
  // `status === 'active'` check was a pre-read in front of an UNCONDITIONAL
  // upsert, so an activation that committed between the two was overwritten
  // here — a LIVE payout binding taken down without the signed unbind that is
  // supposed to be the only way to take one down. The `WHERE` makes the refusal
  // atomic, and a zero-row result raises the same 409 the pre-read raises, so
  // callers see no new failure mode.
  const r = await db
    .prepare(
      `INSERT INTO creator_connect (passport_id, status, stripe_account_id, oauth_state, oauth_state_at, connected_at, activated_at)
       VALUES (?, 'connecting', NULL, ?, ?, NULL, NULL)
       ON CONFLICT(passport_id) DO UPDATE SET status='connecting', stripe_account_id=NULL, oauth_state=excluded.oauth_state, oauth_state_at=excluded.oauth_state_at, connected_at=NULL, activated_at=NULL, bind_epoch=creator_connect.bind_epoch+1
         WHERE creator_connect.status <> 'active'`,
    )
    .bind(passportId, state, ts)
    .run();
  if (!r.changes) throw new ConnectError("disconnect the current payout account first", 409);
  return { state };
}

/** I21-Q52 (D-1030: "completed Connect onboarding before any transfer") and I21-Q66 / D-1037(7) ("US dollars only", at
 * send time, not only at bind — RGM-06): Stripe says the account submitted its details, can receive payouts, and its

export function connectReady(s: ConnectAccountState | null): boolean {
  return s !== null && s.details_submitted === true && s.payouts_enabled === true && s.default_currency === "usd";
}

/**
 * Handle the OAuth callback: resolve the passport from the single-use `state`,
 * exchange the `code` for the connected account id, and store it PENDING
 * (awaiting the passkey confirm). The state is consumed here so a replayed
 * callback finds nothing.
 */
export async function completeConnect(
  db: DbLike,
  cfg: ConnectConfig,
  state: string,
  code: string,
  nowMs: number,
  now: () => string = () => new Date().toISOString(),
  fetchImpl: typeof fetch = fetch,
): Promise<{ passport_id: string; stripe_account_id: string }> {
  if (!STATE_RE.test(state)) throw new ConnectError("bad state", 400);
  const row = await db.prepare("SELECT passport_id, oauth_state_at FROM creator_connect WHERE oauth_state = ? AND status = 'connecting'").bind(state).first<{ passport_id: string; oauth_state_at: string }>();
  if (!row) throw new ConnectError("no matching connect flow", 404);
  if (Date.parse(row.oauth_state_at) < nowMs - OAUTH_STATE_TTL_MS) throw new ConnectError("connect flow expired", 409);
  const accountId = await exchangeConnectCode(cfg, code, fetchImpl);
  if (!accountId) throw new ConnectError("could not connect the account", 502);
  // I21-Q52 / I21-Q66 (D-1030): onboarding finished and the account in US dollars, read before anything is bound; a
  // refusal writes nothing (the flow stays `connecting` until its state expires). Unreadable is a refusal, never a pass.
  const acct = await getConnectAccount(cfg, accountId, fetchImpl);
  if (!acct) throw new ConnectError("could not connect the account", 502);
  if (acct.default_currency !== "usd") throw new ConnectError(CONNECT_NOT_USD, 409);
  if (!connectReady(acct)) throw new ConnectError(CONNECT_UNFINISHED, 409);
  const ts = now();
  // Consume the state and store PENDING. Guarded on still-connecting so a race
  // (a second callback) cannot overwrite an already-consumed flow.
  const r = await db
    .prepare("UPDATE creator_connect SET status='pending', stripe_account_id=?, oauth_state=NULL, connected_at=? WHERE passport_id=? AND oauth_state=? AND status='connecting'")
    .bind(accountId, ts, row.passport_id, state)
    .run();
  if (r.changes !== 1) throw new ConnectError("connect flow already completed", 409);
  return { passport_id: row.passport_id, stripe_account_id: accountId };
}

/** The LIVE (or pending) account a disconnect would take down, or null. Reads
 * both states deliberately: a creator who realises mid-ceremony that they
 * connected the wrong Stripe account should be able to back out of a PENDING
 * binding too, not only a live one. */
export async function boundConnect(db: DbLike, passportId: string): Promise<{ stripe_account_id: string; status: string; bind_epoch: number } | null> {
  if (!PSP_RE.test(passportId)) return null;
  const row = await db
    // BF-14: the GENERATION rides with the destination. A creator approves a
    // specific binding generation, and both the assertion and the mutation that
    // spends it must name the one they were shown.
    .prepare("SELECT stripe_account_id, status, bind_epoch FROM creator_connect WHERE passport_id = ? AND status IN ('active','pending')")
    .bind(passportId)
    .first<{ stripe_account_id: string; status: string; bind_epoch: number }>();
  return row?.stripe_account_id ? { stripe_account_id: row.stripe_account_id, status: row.status, bind_epoch: Number(row.bind_epoch ?? 0) } : null;
}

/** The pending account awaiting a passkey confirm, or null. */
export async function pendingConnect(db: DbLike, passportId: string): Promise<{ stripe_account_id: string; bind_epoch: number } | null> {
  if (!PSP_RE.test(passportId)) return null;
  const row = await db.prepare("SELECT stripe_account_id, bind_epoch FROM creator_connect WHERE passport_id = ? AND status = 'pending'").bind(passportId).first<{ stripe_account_id: string; bind_epoch: number }>();
  return row?.stripe_account_id ? { stripe_account_id: row.stripe_account_id, bind_epoch: Number(row.bind_epoch ?? 0) } : null;
}

const ACCT_RE = /^acct_[A-Za-z0-9]{1,64}$/;

/**
 * Activate the pending binding (called AFTER the passkey assertion verified),
 * AND record it in the ledger, in ONE atomic batch.
 *
 * UI-AUDIT S-C3 — WHY THIS FUNCTION GREW A LEDGER APPEND. `activateConnect` used
 * to be the four lines of UPDATE above this comment, and that was the entire
 * durable record of the highest-value change a creator can make: the Stripe
 * account the weekly run transfers their whole cleared balance to. No
 * hash-chained row, so nothing an auditor could read to answer "when did this
 * destination change, and who signed for it" — the exact question
 * `payout_tag_added` exists to answer for the strictly SMALLER exposure of a tag
 * a fan pays into. The passkey ceremony in front of it was already right; what
 * was missing was that it wrote nothing down.
 *
 * ATOMIC, in the shape `appendCreatorSigned` uses for release manifests: the
 * activation UPDATE rides as an `extraStatement` on the append's batch, so a
 * binding can never go live with its ledger row dropped, and a row can never
 * name a binding that did not activate. The UPDATE keeps its own guard
 * (still-pending + the exact account) so a stale confirm cannot activate a
 * different account, and the batch's changes count is what tells us it applied.
 *
 * NOT `appendCreatorSigned`, deliberately. That helper verifies an assertion
 * over `actionHashForDrafts(drafts)`, and this ceremony's challenge is bound to
 * `connect-bind:v2:<op>:<acct>:<bindEpoch>` instead (passkeys.ts connect-bind challenge; op `bind` here) — a prefix that "can never equal a real
 * draft's action hash" (passkeys.ts), which is what stops a connect challenge
 * from being repurposed into a ledger append. That separation is worth keeping,
 * so the caller verifies the connect-bind assertion and this function appends
 * with the verified actor kind, the way `verification-sweep.ts` passes its own
 * governed type.
 */
export async function activateConnect(
  db: DbLike,
  passportId: string,
  stripeAccountId: string,
  now: () => string = () => new Date().toISOString(),
  /**
   * BF-14 — THE GENERATION THE CREATOR REVIEWED AND SIGNED FOR.
   *
   * The pre-read below already refuses a confirm that names a different
   * ACCOUNT than the pending one. It could not refuse one that names the same
   * account in a DIFFERENT generation: `startConnect` bumps `bind_epoch` on
   * every restart, so a creator who reviewed generation N, hesitated, and
   * restarted, could have their generation-N approval spent on generation N+1.
   * When supplied, this is checked against the stored epoch and a mismatch is a
   * refusal — never a silently-obeyed activation of a destination nobody
   * reviewed. Optional so operator/test callers that legitimately have no
   * reviewed generation keep the previous (account-guarded) behaviour.
   */
  expectedEpoch?: number,
): Promise<void> {
  if (!PSP_RE.test(passportId)) throw new ConnectError("bad passport id");
  if (!ACCT_RE.test(stripeAccountId)) throw new ConnectError("bad account id");
  // THE PRE-READ MUST CHECK THE ACCOUNT, NOT ONLY THE STATE — and this is the
  // hole the existing `activate refuses a wrong or stale account` test caught in
  // the first version of this function, which is exactly what it is for.
  //
  // The lock's primary key is (passport, account, decision, epoch), so a confirm
  // naming a DIFFERENT account than the pending one takes a lock nobody holds
  // and inserts cleanly; the activation UPDATE is guarded on the exact account,
  // matches zero rows, and — because a zero-row UPDATE does not throw — the
  // batch commits. The result would be a `payout_account_bound` row in the chain
  // for an account that was never activated. Guarding on the pair here is what
  // closes it: the lock stops the same account being bound twice, and this stops
  // a different one being recorded at all.
  const cur = await db.prepare("SELECT status, stripe_account_id, bind_epoch FROM creator_connect WHERE passport_id = ?").bind(passportId).first<{ status: string; stripe_account_id: string | null; bind_epoch: number }>();
  if (!cur || cur.status !== "pending" || cur.stripe_account_id !== stripeAccountId) throw new ConnectError("no pending binding to activate", 409);
  const storedEpoch = Number(cur.bind_epoch ?? 0);
  // BF-14: …and the GENERATION, when the caller was told which one it reviewed.
  if (expectedEpoch !== undefined && storedEpoch !== expectedEpoch) throw new ConnectError("the payout account changed while you were confirming it", 409);
  const activate = db
    .prepare(`UPDATE creator_connect SET status='active', activated_at=? WHERE passport_id=? AND status='pending' AND stripe_account_id=?${expectedEpoch === undefined ? "" : " AND bind_epoch=?"}`)
    .bind(...(expectedEpoch === undefined ? [now(), passportId, stripeAccountId] : [now(), passportId, stripeAccountId, expectedEpoch]));
  // WHAT THIS CONFIRM MOVES, not just what it sets (S10-3(b), AS-062).
  // `creator_payouts` carries no destination column (0036_creator_payouts.sql)
  // and both payout doors read `creator_connect` LIVE through ACTIVE_BINDING
  // (:615), so every row still `pending` at commit will be paid to the account
  // being confirmed here — money reserved against the old account follows the
  // new binding on the first tick after the cooling-off. That is the design;
  // leaving it unsaid is the defect.
  //
  // INV-058 (QUEUE #80/RUN-0015 C-payouts-4, was creator-payout.ts:307-310
  // pre-fix): the count used to be read HERE, by a SELECT that ran before
  // `appendPayoutAccountEvent`'s batch even opened. A payout tick settling one
  // of these rows between this read and that commit left the notice
  // (`payoutAccountNoticeStatement`) and the audit row below stating a number
  // that was never true at commit — two consumers of ONE claim disagreeing
  // under concurrency, exactly what FINDING-0015-C-payouts-4 attacked. No
  // pre-read left: `appendPayoutAccountEvent` and `payoutAccountNoticeStatement`
  // (digest.ts) each take the count from a correlated `SELECT COUNT(*)` INSIDE
  // their own statement, both riding the SAME `db.batch()` as the activate
  // UPDATE — one transaction, one snapshot, so the two served copies of this
  // claim can no longer read it at different instants.
  await appendPayoutAccountEvent(db, {
    passportId,
    stripeAccountId,
    bindEpoch: Number(cur.bind_epoch ?? 0),
    eventType: "payout_account_bound",
    actorKind: "creator_passkey",
    actorId: `creator:${passportId}`,
    guardedStatement: activate,
    guardFailure: "no pending binding to activate",
    now,
  });
}

/**
 * THE CHANGE PATH — UI-AUDIT F-C3, and the finding this one is worth restating.
 *
 * `Earnings.connect_status` has always had a `revoked` member, and
 * `apps/creator-console/src/earnings.ts` has always rendered a card for it
 * ("Your payout account was disconnected… Reconnect a Stripe account"). Nothing
 * in this repository could produce that status. `startConnect` refuses outright
 * when a binding is `active` ("a payout account is already connected", 409), and
 * no other writer moves a row off `active`. So the first Stripe account a
 * creator confirmed was their payout destination FOREVER: an account they lost
 * access to, a Stripe account that got closed, one bound during a session
 * compromise, or simply the wrong one of two — all permanent, with a console
 * card describing the way out and no door behind it.
 *
 * Unbinding never sends money anywhere, so it asks for no standing — the same
 * line `planPayoutTagChange` draws when it requires `requireVerifiedCreator` on
 * an add and not on a removal, and the same `agency_revoked` / `war_opt_out`
 * asymmetry the actor policy states: exit is wider than entry. What it DOES
 * require is the creator's own passkey (or an operator), because turning
 * payouts off is itself a denial-of-service worth a signature.
 *
 * The row goes to `revoked`, not to `none`: the console's revoked card is the
 * honest state ("payouts are paused; nothing is lost — earnings are held"), and
 * a creator can then `startConnect` again, which the existing
 * `status === 'active'` refusal no longer blocks.
 */
export async function unbindPayoutAccount(
  db: DbLike,
  passportId: string,
  opts: { actorKind: "creator_passkey" | "operator"; actorId: string; reason?: string },
  now: () => string = () => new Date().toISOString(),
  /**
   * BF-14 — WHAT THE CREATOR WAS SHOWN AND SIGNED FOR.
   *
   * The disconnect route verified its assertion against the destination IT had
   * read, then called this function, which RE-READ the current binding without
   * being told what it was supposed to remove. A binding change landing in
   * between (a concurrent confirm, an operator action) therefore revoked a
   * DIFFERENT account from the one the creator reviewed and approved — a
   * silently obeyed instruction nobody gave.
   *
   * Supplied by the creator-passkey door; omitted by the operator and
   * lockdown-injection callers (`console.ts`, `recovery.ts`), which have no
   * reviewed generation and whose whole job is "take down whatever is there".
   */
  expected?: { stripe_account_id: string; bind_epoch: number },
  /**
   * QUEUE #942(a) (D-0946(1), census X06/D20): the caller's audit row, built from the account id this
   * function reads (known only here), rides the unbind's own batch AFTER the BINDEPOCH lock, so a
   * failed audit insert rolls the unbind back and an unbind whose guarded UPDATE matched nothing throws
   * at the lock before the row can claim it happened. Called on the bound arm only.
   */
  audit?: (ctx: { stripe_account_id: string }) => DbStatement[],
): Promise<{ stripe_account_id: string }> {
  if (!PSP_RE.test(passportId)) throw new ConnectError("bad passport id");
  // ALL FOUR VALUES OF `creator_connect.status` (0036_creator_payouts.sql:21),
  // not the two this read used to name (RUN-0015 E-1). The filter was
  // `status IN ('active','pending')`, which made `connecting` — a ceremony
  // STARTED and not finished, holding a live single-use `oauth_state` and a
  // NULL account — indistinguishable here from a passport with no row at all.
  // That is the one credential a hijack lockdown has to burn, and this is the
  // only lever that reaches the table (recovery.ts:754 injects it).
  const readRow = () =>
    db
      .prepare("SELECT status, stripe_account_id, bind_epoch FROM creator_connect WHERE passport_id = ?")
      .bind(passportId)
      .first<{ status: string; stripe_account_id: string | null; bind_epoch: number }>();
  let row = await readRow();
  if (row?.status === "connecting") {
    // NOTHING IS BOUND, SO NOTHING IS UNBOUND: no ledger event, no notice, no
    // `connect_binding_decisions` row — there is no `acct_…` to name in any of
    // them (startConnect writes stripe_account_id NULL, :181). What this arm
    // owes is the END of the flow and a record of it, in ONE batch, so the
    // state and its line commit together (FAMILY-003 P1). The subject is left
    // `revoked` with no redeemable state; the next actor is the creator, via
    // `startConnect`, which the `status === 'active'` refusal does not block.
    await db.batch([
      db
        .prepare("UPDATE creator_connect SET status='revoked', oauth_state=NULL, oauth_state_at=NULL, stripe_account_id=NULL WHERE passport_id=? AND status='connecting'")
        .bind(passportId),
      // I3-2185 (I22-FX-MONEY, INV-058): the line is bound to THIS batch's UPDATE having burned the state
      // (`changes() = 1`, the row count of the statement directly above), not to the row's end state — a concurrent
      // lockdown that burned it first leaves the same end state, and this call, which burned nothing, writes no line.
      db
        .prepare("INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, ?, ?, ?, ?, ? WHERE changes() = 1")
        .bind(
          opts.actorId,
          "connect:flow_ended",
          `connecting -> revoked; oauth_state burned${opts.reason ? ` reason=${opts.reason}` : ""}`.slice(0, 300),
          now(),
          passportId,
          opts.actorKind,
        ),
    ]);
    // I3-2185 a (I22): re-read. A callback that landed in that window left a real binding (`pending`, or `active`
    // once confirmed); the lockdown's job is "take down whatever is there", so it falls through to the bound arm and
    // unbinds it instead of reporting 409 over a live binding.
    const after = await readRow();
    if ((after?.status === "pending" || after?.status === "active") && after.stripe_account_id) row = after;
    else
    // The SAME message and status the empty and the `revoked` arms throw, kept
    // deliberately: both callers branch on it (console.ts:1793 maps 409 to 404
    // on a route addressed by passport id; the injected revoker at
    // console.ts:1904 maps 409 to `null` so a lockdown converges), and the
    // sentence is true — there was no payout account to disconnect. The fact
    // that this arm ACTED is carried by `connect:flow_ended`, not by the throw.
      throw new ConnectError(NO_PAYOUT_ACCOUNT, 409);
  }
  if (!row?.stripe_account_id || row.status === "revoked") throw new ConnectError(NO_PAYOUT_ACCOUNT, 409);
  // BF-14: the row must still be the one the approval named.
  if (expected && (row.stripe_account_id !== expected.stripe_account_id || Number(row.bind_epoch ?? 0) !== expected.bind_epoch)) {
    throw new ConnectError("payout account changed while disconnecting", 409);
  }
  const acct = row.stripe_account_id;
  if (!ACCT_RE.test(acct)) throw new ConnectError("stored account id is malformed", 500);
  // Guarded on the EXACT account we read, so a binding that changed between the
  // read and the write (a concurrent confirm) is not silently revoked instead.
  const revoke = db
    .prepare(`UPDATE creator_connect SET status='revoked' WHERE passport_id=? AND stripe_account_id=? AND status IN ('active','pending')${expected ? " AND bind_epoch=?" : ""}`)
    .bind(...(expected ? [passportId, acct, expected.bind_epoch] : [passportId, acct]));
  await appendPayoutAccountEvent(db, {
    passportId,
    stripeAccountId: acct,
    bindEpoch: Number(row.bind_epoch ?? 0),
    eventType: "payout_account_unbound",
    actorKind: opts.actorKind,
    actorId: opts.actorId,
    reason: opts.reason,
    guardedStatement: revoke,
    guardFailure: "payout account changed while disconnecting",
    now,
    extraStatements: audit ? audit({ stripe_account_id: acct }) : [],
  });
  return { stripe_account_id: acct };
}

/**
 * One append + one guarded state write + one UNIQUENESS LOCK, in one atomic
 * batch. Shared by bind and unbind so the two can never drift on atomicity.
 *
 * THE LOCK IS THE PART THAT MAKES THE BATCH EXCLUSIVE, and it is here because
 * the obvious version is wrong in a way that is easy to miss: `db.batch` rolls
 * back on a thrown CONSTRAINT, and an UPDATE whose WHERE matches zero rows does
 * not throw — it succeeds having changed nothing. So the state write's own guard
 * (`status='pending' AND stripe_account_id=?`) cannot make this exclusive on its
 * own: two concurrent confirms of one pending binding would each append a
 * `payout_account_bound` row while only one UPDATE actually moved the row, and
 * the chain would say the destination was set twice when it was set once.
 *
 * `connect_binding_decisions`' primary key is what throws — the same role
 * `warning_publications`' PK plays for a duplicate warning publish. The loser's
 * whole batch, ledger row included, rolls back.
 */
async function appendPayoutAccountEvent(
  db: DbLike,
  a: {
    passportId: string;
    stripeAccountId: string;
    bindEpoch: number;
    eventType: "payout_account_bound" | "payout_account_unbound";
    actorKind: "creator_passkey" | "operator";
    actorId: string;
    reason?: string | undefined;
    guardedStatement: DbStatement;
    guardFailure: string;
    now: () => string;
    extraStatements?: DbStatement[];
  },
): Promise<void> {
  const payload: Record<string, unknown> = { stripe_account_id: a.stripeAccountId };
  if (a.eventType === "payout_account_unbound" && typeof a.reason === "string" && a.reason.length > 0) payload.reason = a.reason.slice(0, 512);
  const draft = {
    event_type: a.eventType,
    subject: a.passportId,
    actor_kind: a.actorKind,
    actor_id: a.actorId,
    payload,
  } as DraftEvent;
  const ts = a.now();
  // POST-CONDITION, NOT PRE-CONDITION (BINDEPOCH / QUEUE #82 / INV-058 sibling
  // fix). `activateConnect`'s and `unbindPayoutAccount`'s pre-reads are
  // unguarded, and a concurrent `startConnect` landing between that read and
  // `a.guardedStatement` (the activate/revoke UPDATE) can make it match ZERO
  // rows without throwing — the exact FAMILY-008 shape confirmed by
  // harness/zz-probe-bindepoch.test.ts. Rather than pull the guarded UPDATE out
  // of this batch (which would undo RETARGET's INV-058 fix: the correlated
  // `pendingRetargeted` reads below need the activate/revoke UPDATE and the
  // notice/audit rows to share ONE snapshot), this lock's own `passport_id` is
  // now sourced from a correlated re-read of `creator_connect`, run in THIS
  // SAME batch AFTER `a.guardedStatement` (see the `extra` ordering below):
  // it returns the row's id only if the row actually reached the exact
  // post-transition state this call claims, else NULL. `passport_id` is
  // NOT NULL (migration 0040) and is part of this table's PRIMARY KEY, so a
  // failed transition throws here and the WHOLE batch — ledger event, notice,
  // audit row included — rolls back instead of committing a false record.
  // Reusing an EXISTING NOT NULL column on an EXISTING table (rather than a
  // new `write_admissions` gate) needs no schema migration and keeps this
  // room's change inside creator-payout.ts.
  //
  // This does not weaken the lock's original exclusivity job: two concurrent
  // decisions in the SAME epoch still collide on the unchanged PRIMARY KEY
  // (passport_id, stripe_account_id, decision, bind_epoch) regardless of
  // which one's guarded UPDATE actually fired, and that collision is still
  // reported as `conflict` (UNIQUENESS_LOST_RE), never relabeled by
  // GUARD_TRANSITION_LOST_RE above.
  const expectedStatus = a.eventType === "payout_account_bound" ? "active" : "revoked";
  const lock = db
    .prepare(
      "INSERT INTO connect_binding_decisions (passport_id, stripe_account_id, decision, bind_epoch, decided_at, actor_kind, actor_id) " +
        "VALUES ((SELECT cbd_cc.passport_id FROM creator_connect cbd_cc WHERE cbd_cc.passport_id = ? AND cbd_cc.stripe_account_id = ? AND cbd_cc.status = ?), ?, ?, ?, ?, ?, ?)",
    )
    .bind(a.passportId, a.stripeAccountId, expectedStatus, a.stripeAccountId, a.eventType === "payout_account_bound" ? "bound" : "unbound", a.bindEpoch, ts, a.actorKind, a.actorId);
  // THE ALERT RIDES THE SAME BATCH AS THE CHANGE ITSELF, the shape
  // `setPayoutTags` uses for the payout-tag swap and for the same reason: a
  // payout destination can never move with its notice dropped. This is the part
  // that makes the cooling-off worth having — a window only helps a creator who
  // knows there is something to look at.
  //
  // Keyed on (passport, epoch, decision), which is exactly the lock's own key,
  // so a retried append is idempotent under the digest table's own UNIQUE rather
  // than sending the same alarm twice.
  const notice = payoutAccountNoticeStatement(
    db,
    a.passportId,
    {
      // THE KEY CARRIES NO ACCOUNT ID. It used to end `:${stripeAccountId}`,
      // which put the full `acct_…` into `creator_digests.period_key` — a column
      // the console renders verbatim beside every digest row. That defeats the
      // whole point of sending only the last four in the body, and it did it on
      // the surface the hint exists to keep the id off.
      //
      // (passport, epoch, action) is already unique without it: `startConnect`
      // bumps `bind_epoch` and clears the pending account, so one epoch holds at
      // most one bind and one unbind — which is exactly what
      // `connect_binding_decisions`' primary key enforces.
      eventKey: `${a.bindEpoch}:${a.eventType === "payout_account_bound" ? "bound" : "unbound"}`,
      action: a.eventType === "payout_account_bound" ? "bound" : "unbound",
      stripeAccountId: a.stripeAccountId,
      liveAt: a.eventType === "payout_account_bound" ? new Date(Date.parse(ts) + CONNECT_COOLING_OFF_MS).toISOString() : null,
      // pending_retargeted is no longer passed here (INV-058 fix): the bind arm
      // of `payoutAccountNoticeStatement` computes it itself, from a correlated
      // COUNT inside its OWN insert, so the number this notice carries is read
      // at this row's own commit instant, not at some earlier point in this
      // caller. 0 IS A VALUE, NOT AN ABSENCE either way (INV-004 clause 4's
      // shape) — the field is always present, never conditioned on being
      // non-zero.
    },
    () => ts,
  );
  // THE RECORD RIDES THE SAME BATCH AS THE FACT (INV-053/FAMILY-003): the row
  // that says how much money this confirm moved commits with the confirm or not
  // at all. Written on the bind arm only, with a correlated COUNT of its OWN
  // (INV-058) so the operator log's number is read at the SAME commit instant
  // as the notice's — not passed between them as a JS value that could have
  // gone stale in transit — and the two can never disagree. Next actor on a
  // non-zero count: the creator, inside the cooling-off window, via the unbind
  // their notice already points at.
  // `a.guardedStatement` FIRST: `lock`'s correlated subquery (above) must
  // observe the guarded UPDATE's effect, and D1/the test doubles run one
  // batch's statements in order, on one connection, in one transaction — a
  // later statement here sees an earlier one's writes (BINDEPOCH fix).
  const extra: DbStatement[] = [a.guardedStatement, lock, notice];
  if (a.eventType === "payout_account_bound") {
    extra.push(
      db
        .prepare(
          "INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) VALUES (?, ?, " +
            `'pending_retargeted=' || (SELECT COUNT(*) FROM creator_payouts WHERE passport_id = ? AND (status = 'pending' OR status = 'failed')), ?, NULL, ?)`,
        )
        .bind(a.actorId, "connect:rebind_retargets_pending", a.passportId, ts, a.actorKind),
    );
  }
  // QUEUE #942(a): the caller's rows go LAST, after `lock`, whose post-condition gates them.
  extra.push(...(a.extraStatements ?? []));
  try {
    const r = await tryAppendEvents(db, [draft], a.now, [a.eventType], extra, [a.actorKind]);
    if (r.ok) return;
    // `conflict` is the lock: somebody already made this decision in this epoch.
    // `state` is the ledger refusing the draft (retracted passport, unknown
    // subject). Both are client facts, not server faults.
    throw new ConnectError(r.refusal.kind === "conflict" ? a.guardFailure : r.refusal.message, 409);
  } catch (e) {
    // BINDEPOCH: the guarded UPDATE matched zero rows (a race moved the row
    // out from under it) — `lock`'s post-condition re-read caught it and this
    // NOT NULL failure is `tryAppendEvents`'s way of saying so; it is neither
    // `LedgerError` nor the lock's own UNIQUE collision, so it reaches here
    // raw. Translate it to the same domain refusal a guard-failed pre-check
    // would have thrown, instead of letting a bare SQL error escape.
    if (e instanceof ConnectError) throw e;
    if (e instanceof Error && GUARD_TRANSITION_LOST_RE.test(e.message)) throw new ConnectError(a.guardFailure, 409);
    throw e;
  }
}

export interface PayoutRunResult {
  paid: { passport_id: string; amount_cents: number; transfer_id: string }[];
  skipped_below_min: number;
  failed: { passport_id: string; amount_cents: number }[];
  /** Connected creators this tick actually looked at (the page, not the set). */
  examined: number;
  /** Stripe DISPATCHES this tick made — the SUBREQUEST half of the cost, at most
   * `PAYOUT_DISPATCH_SUBREQUESTS` (group lookup + account reconcile + send) per
   * dispatch (I3-0028, D-0911(1)). */
  transfers: number;
  /** Why the run stopped short, or null if it finished its page. Logged by the
   * caller: a run that stops on a ceiling every tick is a deployment that has
   * outgrown these numbers, and the alternative to saying so is a leg that
   * quietly pays a shrinking fraction of what it owes. */
  stopped: null | "transfer_budget" | "deadline";
  /** Pending rows inside THIS page's window that STEP 0 declined to retry
   * because their creator is not payable right now (binding revoked or not
   * active, passport retracted/suppressed/unverified). Their money is held, not
   * failed — but a held row is a creator who is owed and not being paid, so the
   * tick reports the count instead of leaving it to be inferred from silence.
   * UNCHANGED by docket #43 (a) — narrow ruling: still exactly this window,
   * still exactly this predicate. It does not cover a row this tick never
   * looked at at all; `retry_truncated` and `deadline_abandoned` below do. */
  held: number;
  /** I3-0062 — pending payouts this tick RE-REFUSED whose refusal count has
   * reached `PAYOUT_STALL_ATTEMPTS`. `failed` says a transfer did not land on
   * this tick; it says nothing about whether that is the first time or the
   * twelfth, and the columns that DO know (`attempts`, `next_attempt_at`) were
   * read by nothing but the selector. This is the leg's own statement that a
   * named creator is owed a named amount and the money has stopped moving,
   * carried out to the caller that logs the run. It costs NO extra statement:
   * every field is already in hand at the refusal.
   *
   * It reports only rows THIS tick re-attempted. The whole population — including
   * rows whose backoff is not yet due — is `stalledPayouts()` below, which is what
   * an operator surface reads. */
  stalled: { payout_id: string; passport_id: string; amount_cents: number; attempts: number; next_attempt_at: string }[];
  /** D-0931(3): payouts whose transfer was SENT this tick but whose `paid` flip changed no row (a void or another
   * writer moved the row first). Each is also a `PAYOUT_FLIP_MISSED` log line. Optional only for older literals. */
  paid_flip_missed?: string[];
  /** 1 when the retry read's own `LIMIT` (`retryBudget`) cut off at least one
   * further PAYABLE pending row inside THIS tick's window (`pendWindow`),
   * else 0. NOT `held`: a row counted here passes `STEP0_PAYABLE`, it is
   * simply behind the LIMIT — still `pending`, still owed, and previously
   * invisible (neither `held`, nor `failed`, nor `examined` counted it).
   * Docket #43 (a).
   * COST: a BOUND, deliberately, not the exact remainder. An exact count
   * would need a second scoped `COUNT` statement every tick; this run's own
   * pinned cost tests hold the per-tick statement count at "4 regardless of
   * N" (THE BOUNDS, :585-590), so a guaranteed-exact number is not free here.
   * Instead the retry read over-fetches by ONE (`retryBudget + 1` rows on the
   * SAME statement — free, since D1 prices a statement, not a row count) and
   * this reports whether that extra row existed. The exact remainder past
   * `retryBudget` is not reported; "at least one" is what the run can afford
   * to know for certain every tick.
   * NOT COVERED: a creator past THIS TICK's page (past `pageEnd`, :806) whom
   * the page's own `LIMIT` never fetched into `actives` at all — a different
   * LIMIT, on a different statement, and an exact (or even a cheap bound)
   * count there needs a new scan of the remainder of the connected set, which
   * is the cost THE BOUNDS paged specifically to avoid. `wrapped` (below) is
   * the free, already-existing signal for that instead: `wrapped === false`
   * already says more of the set is queued for a later tick. */
  retry_truncated: number;
  /** Rows abandoned because a clock ran out: pending retries STEP 0's loop
   * had not reached when EITHER the overall run deadline OR (CM-02) STEP 0's
   * own `PAYOUT_RETRY_DEADLINE_MS` slice ran out, plus page creators STEP 1's
   * loop had not reached when `stopped === "deadline"`. Docket #43 (a).
   * Distinct from `stopped === "transfer_budget"`, which is a designed
   * ceiling THE BOUNDS (:560-615) already names by binding; this number
   * exists because a clock overrun is the one stop reason `examined` does
   * not itself make legible — `examined` counts only what STEP 1 actually
   * looked at, never what neither loop reached. STEP 0's own slice running
   * out does NOT by itself set `out.stopped` — the run's real deadline may
   * still be far off, and STEP 1 gets whatever is left of it — so this field
   * can be non-zero even when `stopped` is `null` or `"transfer_budget"`.
   * COST: free — derived from lengths the run already holds (`pendings.length`,
   * `actives.length`, `out.examined`); no new query. */
  deadline_abandoned: number;
  /** Page creators STEP 1 priced and then declined to reserve because their
   * Connect binding was no longer the ACTIVE one the page had captured —
   * disconnected, re-pointed, or pulled back to `pending` between the page read
   * and the reservation (S-070 SJ-01). Their money is held, not failed, and
   * not reserved: no `creator_payouts` row exists for them this period, so the
   * next tick prices the same amount again. Distinct from `held`, which counts
   * ALREADY-RESERVED pending rows STEP 0 declined to retry. */
  revoked_before_reserve: number;
  /** The page reached the end of the connected set and the cursor wrapped. */
  wrapped: boolean;
  /** Q102 (D-1059(15)): `held` was counted from the cursor to the TABLE END (the page was not full), not to the page's
   * last row. Set from the same `fullPage` that picks `held`'s window, so the log line names the window actually
   * counted; `wrapped` cannot say it (a short page that stopped early counts to the table end and does not wrap).
   * `runWeeklyPayouts` (the one producer) always sets it; optional only so the hand-built results older tests cast to
   * this type still typecheck (i15-w2e-payout-log-line.test.ts:12), and absent reads as the page-end window. */
  held_to_table_end?: boolean;
  /** I3-0028 (D-0911(1)): payouts this tick HELD FOR AN OPERATOR because the
   * per-account reconcile found Stripe disagreeing with our ledger (amount,
   * destination, reversal, or a transfer the ledger does not hold). Nothing was
   * sent for them and nothing recorded paid; each has an `audit_log` row
   * (`payout:reconcile_hold`) and stays in `reconcileHolds()` until an operator
   * acts. The money stays reserved. */
  reconcile_holds: { payout_id: string; passport_id: string; period_key: string; amount_cents: number; reason: TransferHold["hold"] }[];
  /** I21-Q52: creators this tick did not pay because Stripe says their onboarding is unfinished (held, not failed). */
  connect_unfinished?: number;
}

/**
 * ───────────────────────────────────────────────────────────────────────────
 * THE BOUNDS ON THE WEEKLY RUN — why they exist and what each one binds.
 * ───────────────────────────────────────────────────────────────────────────
 *
 * This leg runs inside `scheduled()`, which has ONE ~1000-subrequest budget for
 * every leg in the invocation, and a D1 statement spends one of those exactly
 * like an outbound `fetch` does (see alerts.ts THE BUDGETS, which writes the
 * rest of the invocation's arithmetic down). The first version of this function
 * scanned EVERY active Connect binding and spent three statements per creator
 * plus one serial, un-timed Stripe transfer per creator who was owed anything.
 * Measured on `test/db-recorder.ts` + `test/sqlite-double.ts`:
 *
 *     N=10  connected ->   52 statements +  10 fetches =   62 subrequests
 *     N=100 connected ->  502 statements + 100 fetches =  602
 *     N=400 connected -> 2002 statements + 400 fetches = 2402
 *     N=400, nothing owed                              -> 1202 statements
 *
 * `alerts.ts` budgets ~585 for the alert legs alone, before the chain verify,
 * the checkpoint reads, the registry rebuild, fourteen prunes and the digests —
 * so the crossover is ~80-130 connected creators, and past it this one leg
 * exhausts the invocation and every leg after it throws on the same exhausted
 * budget into its own catch. Four bounds close that, and none is sufficient
 * alone:
 *
 *  1. THE PAGE IS ONE STATEMENT. The owed amount and "is this week already
 *     reserved?" are correlated subqueries inside the page SELECT rather than
 *     two extra round trips per creator — the shape `verification-sweep.ts`
 *     already uses for its grant payload. That is what turns the steady-state
 *     cost from `3N` into a constant: 400 connected creators with nothing owed
 *     now cost four statements instead of 1202.
 *  2. A RESUME CURSOR, in the `sweep_cursor` table `verification-sweep.ts`
 *     resumes through (migration 0026 — the table is keyed by sweep name, so
 *     this needs no migration of its own). Each tick starts strictly past where
 *     the last one stopped and wraps at the end, so a page limit cannot hide the
 *     tail of the connected set the way an un-cursored `LIMIT` would.
 *  3. AN OUTBOUND CEILING, because the transfers are the subrequests that are
 *     not statements. `PAYOUT_RETRY_BUDGET` is carved out of it for the
 *     re-attempt leg specifically, so a handful of permanently-failing pending
 *     rows can never eat the whole allowance and leave nothing for new payouts.
 *  4. A WALL-CLOCK DEADLINE, in the shape and the size of `alerts.ts`
 *     `DELIVERY_BUDGET_MS`. `createConnectTransfer` now has its own 10 s
 *     AbortController deadline (stripe.ts), but N of those awaited serially is
 *     still `N x 10 s`, and `cron-lease.ts` picked its 30-minute TTL on the
 *     STATED ASSUMPTION that a real sweep finishes well inside it. A slow Stripe
 *     pushing this run past the TTL means a successor sweep starts and both run
 *     `processNewEvents` — the duplicate the lease exists to prevent.
 *
 * WHAT STOPPING EARLY COSTS, stated rather than assumed: the run pays at most
 * `PAYOUT_TRANSFER_BUDGET - PAYOUT_RETRY_BUDGET` new creators per tick. At the
 * 6-hourly cadence that is 29 x 28 = 812 payouts a week (35 x 28 = 980 before I29-PAYOUT), against a run that
 * must clear one week's earnings per creator per week. A deployment that passes
 * that wants a queue, not a bigger number here — the knobs cannot be raised past
 * what ONE invocation can spend, and `stopped` is how an operator finds out the
 * ceiling is binding instead of discovering it in the payout backlog.
 */

/** Connected creators examined per tick. The page itself is one statement, so
 * this bounds the RESULT SET (and how fast the cursor sweeps the population),
 * not the statement count: at 400 connected creators the whole set is covered
 * in two ticks, i.e. inside twelve hours of a weekly obligation. */
export const PAYOUT_PAGE = 200;

/** I3-0028 (D-0541, D-0911(1)) — the subrequests ONE dispatch may spend: the
 * payout's `transfer_group` lookup (GET), the per-account reconcile listing
 * (GET; the extra call D-0911(1) accepted) and the send (POST). One when the
 * group lookup decides (Stripe already holds the payout's transfer, holds it on
 * a mismatch, or the answer is unreadable — stripe.ts `createConnectTransfer`
 * (A)), two when the account reconcile holds or cannot complete ((B)), three
 * when it sends ((C)). The reconcile's LEDGER rides the page and pending
 * SELECTs, so the per-dispatch STATEMENT count is unchanged. */
// I3-0028 (D-0976(1)): the account listing is one GET per destination the creator has had, at most
// MAX_RECONCILE_DESTINATIONS, so the worst dispatch is the group GET + that many listings + the POST.
// I3-0028 c (I22-FX-MONEY): plus at most RECONCILE_LOOKBACK_WEEKS group GETs (one per recent week the ledger lacks), so
// D = 2 + 5 + 5 = 12 and the run derivation below is 5 + 15 x (6 + 12) + 35 x (5 + 12) = 870.
// I29-PAYOUT (D-1030 I22-Q1, I21-Q52): + RECONCILE_RELEASE_KEYS (3) operator-release group GETs + the account read
// (getConnectAccount, CONNECT_ACCOUNT_READS) before the send: D = 2 + 5 + 5 + 3 + 1 = 16.
export const CONNECT_ACCOUNT_READS = 1;
export const PAYOUT_DISPATCH_SUBREQUESTS = 2 + MAX_RECONCILE_DESTINATIONS + RECONCILE_LOOKBACK_WEEKS + RECONCILE_RELEASE_KEYS + CONNECT_ACCOUNT_READS;

/** Outbound Stripe DISPATCHES one run may make, retries and new payouts together.
 * THE SUBREQUEST BOUND — a `fetch` is a subrequest exactly like a SELECT is, and
 * each dispatch is at most `PAYOUT_DISPATCH_SUBREQUESTS` of them, so a run's
 * Stripe subrequests are at most this x PAYOUT_DISPATCH_SUBREQUESTS = 688 (43 x 16 since I56, D-1437; 704 = 44 x 16 since I29-PAYOUT; 600 = 50 x 12 since I22-FX-MONEY;
 * 350 at D = 7, I22; 150 at D = 3, I15-PAYOUT, D-0911(1); 100 before the per-account reconcile; 65 when only
 * a replay looked up).
 * The per-tick payout count is unchanged: 50, of which at most 15 retries.
 *
 * I19-FXA (RG-MONEY-01, D-0956(1)): it is also the bound on RESERVATION ATTEMPTS. STEP 1 counts a
 * reservation INSERT that dispatched nothing (it threw and was caught, or changed 0 rows) against
 * this same ceiling (`unsent` in `runWeeklyPayouts`), so every statement the new-payout loop issues
 * takes a slot whether or not it sent. THE RUN'S WORST CASE, derived from the code (D = 3):
 *   5 fixed statements (cursor read, page read, held COUNT + pending read, cursor write)
 *   + PAYOUT_RETRY_BUDGET (15) paid retries x (6 statements: send claim, outbox, mark paid, notice,
 *     audit, claim release + D)                                                    = 135
 *   + (50 - 15) paid new payouts x (5 statements: reserve, outbox, mark paid, notice, audit + D) = 280
 *   = 420. Every other per-slot path is cheaper (a refusal 7 / 5, a reconcile hold 7 / 7, an
 *   unsent reservation 1), a STEP 0 claim the re-ask refuses (1 statement, no slot) displaces a
 *   paid retry from a pending page that holds at most PAYOUT_RETRY_BUDGET rows (1 + 8 = 9 per row,
 *   the same total), and a page row already reserved this week or below the minimum issues none.
 * Without the `unsent` count the 200-row page (PAYOUT_PAGE) bounds the uncounted INSERTs instead:
 * 420 + (200 - 35) = 585. alerts.ts `TAIL_MONEY_SUBREQUESTS` and test/payout-run-bounds.test.ts
 * carry 420.
 * I22 (I3-0028, D-0976(1)): D is now 2 + MAX_RECONCILE_DESTINATIONS = 7 (one listing per destination the creator
 * has had), so the same derivation is 5 + 15 x (6 + 7) + 35 x (5 + 7) = 620; the test carries 620 and
 * alerts.ts `TAIL_MONEY_SUBREQUESTS` must move 420 -> 620 (a hand-off: that file is not this room's).
 * I22-FX-MONEY (I3-0028 c): D = 12, so the run's worst case is 5 + 15 x (6 + 12) + 35 x (5 + 12) = 870, which the
 * test and alerts.ts `TAIL_MONEY_SUBREQUESTS` (870 + 40) carry.
 * I29-PAYOUT (D-1030 I22-Q1 + I21-Q52): D = 16, and 50 x 21 alone is past the ceiling, so this budget moves 50 -> 44:
 * 5 + 15 x (6 + 16) + 29 x (5 + 16) = 944, + 40 scaffolding = 984 <= 1,000. A tick pays at most 29 new creators.
 * I56-MONEY (D-1437, INV-088): each new reservation's batch gains its `payout:reserved` audit row, so a new payout
 * carries 6 + D like a retry; 44 would be 5 + 15 x 22 + 29 x 22 = 973, + 40 = 1,013 > 1,000, so this budget moves
 * 44 -> 43: 5 + 15 x 22 + 28 x 22 = 951, + 40 = 991 <= 1,000. A tick pays at most 28 new creators. */
export const PAYOUT_TRANSFER_BUDGET = 43;

/**
 * The slice of that ceiling reserved for re-attempting ALREADY-RESERVED
 * payouts, and the LIMIT on the pending page.
 *
 * Why it is carved out rather than shared first-come: a pending row whose
 * transfer fails stays pending forever, and a first-come split lets a handful of
 * permanently-failing rows spend the entire allowance every tick — new payouts
 * would then never happen at all, deployment-wide, with nothing in the logs
 * except a run that "completed". Reserving a minority share means the failures
 * cost at most their own slice and the new-payout leg always has 35 calls.
 */
export const PAYOUT_RETRY_BUDGET = 15;

/** Wall clock for the whole run — the same number and the same reason as
 * `alerts.ts` `DELIVERY_BUDGET_MS`. Checked BEFORE each transfer, so the worst
 * overshoot is one `TRANSFER_TIMEOUT_MS`, never `N x` it. */
export const PAYOUT_RUN_BUDGET_MS = 20_000;

/**
 * CM-02: the wall-clock slice of `PAYOUT_RUN_BUDGET_MS` reserved for STEP 0
 * (the retry door) — `PAYOUT_RETRY_BUDGET`'s carve-out, but for TIME instead
 * of call count. The two close different starvations: `PAYOUT_RETRY_BUDGET`
 * stops retries from spending a majority of the REQUEST ceiling; nothing
 * stopped a minority of slow (near-`TRANSFER_TIMEOUT_MS`) or hanging retries
 * from spending the ENTIRE run deadline before STEP 1 (new payouts) examined
 * even one page creator — the request ceiling was guaranteed to be spared,
 * cursor progress was not. Checked only inside STEP 0's own loop and only
 * when there is something to retry, so a page with no pending rows spends no
 * extra `clock()` call and STEP 1's own deadline check (unchanged) is what
 * governs everything else. */
export const PAYOUT_RETRY_DEADLINE_MS = 6_000;

/** The `sweep_cursor` row this run resumes through (table from migration 0026). */
export const PAYOUT_CURSOR = "creator_payouts";

/**
 * BF-13 — BOUNDED EXPONENTIAL BACKOFF for a pending payout whose transfer was
 * refused. The run is 6-hourly, so the first step is one tick; doubling reaches
 * the cap in eight refusals (about four days of trying) and then holds there.
 *
 * The point is not to give up — nothing is ever forfeited and the amount stays
 * reserved and owed — it is to stop a permanently broken destination from
 * occupying the retry ceiling on EVERY tick, which is what starved every later
 * pending row behind it.
 */
export const PAYOUT_RETRY_BASE_MS = 6 * 60 * 60 * 1000;
export const PAYOUT_RETRY_MAX_MS = 7 * 24 * 60 * 60 * 1000;

/**
 * I3-0062 — THE REFUSAL COUNT AT WHICH A PENDING PAYOUT IS A PERSON NOBODY HAS
 * TOLD. `attempts`/`next_attempt_at` (BF-13) were written by the refusal branch
 * and read by exactly one thing: the selector's own due-ness predicate (:1137).
 * Nothing else in the deployment read either column, so a destination that
 * refuses for ever produced a growing backoff, an amount that stays RESERVED
 * (and therefore withheld from the creator's next payout too), and no operator
 * signal and no creator notice at any point — the run's result said `failed`
 * once per tick and never said "this one has been failing since August".
 *
 * Four refusals is the threshold because `payoutBackoffMs` doubles from 6 h:
 * 6 + 12 + 24 + 48 h, so the 4th refusal is ~3.75 days after the first attempt
 * and the NEXT one is a week away. That is the point at which "a transfer
 * failed" has become "this creator is not being paid".
 *
 * I16-S5-n138C-36 (D-1030: "After 5 failed attempts, hold the payout and alert; never abandon it automatically"): 5, and
 * the refusal that reaches it PARKS the row (`recordRefusalStmt`: `next_attempt_at = PAYOUT_RECONCILE_HOLD_AT`, listed by
 * `reconcileHolds`, re-armed by `rearmHeldPayout`, returned by `returnParkedPayoutStmts`) and is reported `stalled` once.
 */
export const PAYOUT_STALL_ATTEMPTS = 5;

/** I3-1132 — how long after a payout row is reserved its re-send is still
 * deduplicated by `transferIdempotencyKey`: Stripe honours a key for 24 h; one
 * hour is left for the tick interval so the last in-window retry is not the
 * one that lands just past it. */
export const PAYOUT_KEY_WINDOW_MS = 23 * 60 * 60 * 1000;

/** The backoff after `attempts` refusals (1-based). */
export function payoutBackoffMs(attempts: number): number {
  const n = Math.max(1, Math.min(32, Math.floor(attempts)));
  return Math.min(PAYOUT_RETRY_MAX_MS, PAYOUT_RETRY_BASE_MS * 2 ** (n - 1));
}

/**
 * QUEUE #145 M-15 — WHEN A PENDING TRANSFER IS FAILED. The refusal count at
 * which the backoff reaches its cap (`payoutBackoffMs(n) === PAYOUT_RETRY_MAX_MS`,
 * n = 6: 6+12+24+48+96 h after the first refusal): from there the run tries once a
 * week, so `recordRefusalStmt` writes `status = 'failed'` in the same UPDATE.
 * `failed` is DURABLE, NOT TERMINAL: the row stays RESERVED (every reserved sum
 * here counts `pending`, `failed` and `paid`), STEP 0 keeps retrying it under
 * its ORIGINAL payout id and idempotency key, and a later success flips it to
 * `paid`. Un-reserving it would let the next week reserve the same money under a
 * NEW key while the first transfer's outcome is unknown — a double payment.
 * At go-live, migration 0235 parks every row Stripe refused before it — `failed`, and `pending` with attempts > 0
 * (refused 1 to PAYOUT_FAIL_ATTEMPTS - 1 times; D-1037(7), RGM-07) — for an operator (I4-Q1).
 */
export const PAYOUT_FAIL_ATTEMPTS = 6;
/** I3-1119 b (I22): the refusal count a dispatch's outcome writes. A PERMANENT Stripe refusal (`TransferRefusal`)
 * jumps straight to PAYOUT_FAIL_ATTEMPTS — `failed` at once, past the stall threshold (an operator line) — instead
 * of climbing the backoff ladder one retryable step at a time. */
const refusalAttempts = (prior: number, r: unknown): number => (isTransferRefusal(r) ? Math.max(PAYOUT_FAIL_ATTEMPTS, prior + 1) : prior + 1);

/* The OPEN set (reserved, not yet paid — what STEP 0 retries) is spelled

 * `${}` hole: conditional-emit-scan.test.ts finds a conditional producer by the
 * `status = 'pending'` token in its function's own literals. */
/** Reserved = open + paid + voided: payout-status-sql.ts RESERVED_PAYOUT_STATUS_SQL (I29-FX4: the billing-free leaf, so
 * removal.ts reads the one list without reaching billing), re-exported for creator-erasure.ts CREATOR_UNSETTLED_SQL
 * (I29-FX-INT, FX-ERASE H6) and this module's other importers. */
export { RESERVED_PAYOUT_STATUS_SQL };

/**
 * ONE PAGE of the connected set, priced. Three correlated subqueries instead of
 * three round trips per creator (verification-sweep.ts's shape):
 *  - `cleared_cents`  — Σ creator_net past the 14-day hold, un-reversed;
 *  - `reserved_cents` — Σ every payout row already pending OR paid, so a
 *    still-owed amount from a prior week is never counted as owed twice;
 *  - `period_rows`    — whether THIS ISO week is already reserved for them
 *    (paid, or pending and therefore the retry leg's business, not this one's).
 * Ordered by the primary key so the cursor is a strict, total, index-backed
 * resume point.
 */
/** STEP 0's settle predicate, and therefore its event's guard (S3-11): the row
 * this attempt is about to pay is STILL the one nobody else has paid. */
// The open set as `IN`, not `OR`: an `extraWhere` may carry no `OR` (conditional-emit-scan `namesAFact`).
const STILL_PENDING = "EXISTS (SELECT 1 FROM creator_payouts WHERE payout_id = ? AND status IN ('pending','failed'))";

/* STANDING IS ASKED AT THE TILL, NOT ONLY AT THE ENTRANCE (C-01). The bind
 * routes are gated by `requireVerifiedCreator` (index.ts:2180, :2263), but the
 * weekly run is a cron with no route in front of it — so a passport that was
 * verified when it bound and has since been retracted, suppressed, or dropped
 * back to `claimed` was still selected here and still paid. The `EXISTS` below
 * is `tips.ts`'s `TIPPABLE` predicate (tips.ts:73-80) correlated on
 * `c.passport_id`: the same question the rest of the system asks before it
 * moves money, asked in the SQL rather than after it, because a JS filter over
 * the page would empty a full page and wrap the cursor (:673) — starving every
 * creator past a lapsed one.
 *
 * STEP 0 (the pending retry, :765) ASKS THE SAME QUESTION IN ITS OWN SELECT
 * (`STEP0_PAYABLE`, below). It used to carry no predicate and lean on
 * `if (!dest) continue`, which is true of the ROW (that creator's earnings are
 * held, never failed, never forfeited) and false of the SET: the read is
 * LIMITed at `PAYOUT_RETRY_BUDGET`, so unpayable rows at low passport ids took
 * every slot and a STANDING creator's pending row behind them was never read at
 * all, on every tick (FINDING-0014-B-1). Correlating the predicate into the
 * SELECT means a slot is only ever spent on a row that can actually be paid;
 * the rows it declines are COUNTED into `held` (:472) so the tick says so. */
/** THE PREDICATE, WRITTEN ONCE AND ASKED AT BOTH DOORS (FINDING-0014-B-1).
 * Both fragments are correlated on the `creator_connect` alias `c`, so the page
 * read (below) and STEP 0's pending read (`STEP0_PAYABLE`) cannot drift apart:
 * there is no second copy to forget to edit. `ACTIVE_BINDING` binds the
 * cooling-off cut-off (`activated_at <= ?`) in the position it appears. */
const ACTIVE_BINDING = `c.status = 'active' AND c.stripe_account_id IS NOT NULL
          AND c.activated_at IS NOT NULL AND c.activated_at <= ?`;
/** #486 (D-0645, D-0810 (15)): a TERMINATED founder account is no longer refused
 * (its cleared money is still owed), but it is paid ONLY to its own bound
 * account — the `ACTIVE_BINDING` destination must be the passport's last
 * `creator_passkey` 'bound' decision, the D-0644 rule `payoutDestinationFor`
 * (closures.ts) reads. Correlated on `p` and `c`; asked by the standing read
 * AND re-asked by `reserveReleaseStmt`. D-0812(3): the WEEKLY run asks it too —
 * inside `STANDING_EXISTS` (page read, STEP 0 read) and, as `OWN_DEST_C`, in
 * `reservePayoutStmt` and `claimSendStmt` (the write-time re-asks). */
// I3-0699 a (D-0992(1)): a creator under a live operator suspension is paid by NO door — this predicate is asked by
// the page read and STEP 0 (STANDING_EXISTS), reservePayoutStmt and claimSendStmt (OWN_DEST_C), reserveReleaseStmt and
// the release destination read. The money stays owed and reserved; the next run after a lift pays it.
// I25-Q2 (D-1030 (a), I29-STANDING H1): a TERMINATED creator's live suspension no longer refuses once
// creator-standing.ts TERMINATED_PAYABLE_SQL holds (the 14-day hold since termination has passed and no fraud finding is
// open) — at every door this predicate is asked. It binds ONE value, the run's ISO instant (`now`), in its position.
// The fragments that interpolate a predicate from another module are built at FIRST USE (`sqlOnce`), never at load:
// creator-standing.ts imports this module, so a load-time interpolation would read a const before it exists.
const sqlOnce = (build: () => string): (() => string) => {
  let v: string | null = null;
  return () => (v ??= build());
};
const RELEASE_OWN_DEST = sqlOnce(() => `((NOT EXISTS (SELECT 1 FROM creator_suspensions cs WHERE cs.passport_id = p.passport_id AND cs.state = 'suspended') OR ${TERMINATED_PAYABLE_SQL("p.passport_id", "?")}) AND (NOT EXISTS (SELECT 1 FROM founder tf JOIN account_closures tk ON tk.account_id = tf.account_id
                             WHERE tf.passport_id = p.passport_id AND tk.state = 'terminated')
       OR c.stripe_account_id = (SELECT d.stripe_account_id FROM connect_binding_decisions d
                                  WHERE d.passport_id = p.passport_id AND d.decision = 'bound' AND d.actor_kind = 'creator_passkey'
                                  ORDER BY d.bind_epoch DESC, d.decided_at DESC LIMIT 1)))`);
/** `RELEASE_OWN_DEST` correlated on `c` alone (no `p` alias in scope). Binds `now`. */
const OWN_DEST_C = sqlOnce(() => RELEASE_OWN_DEST().replaceAll("p.passport_id", "c.passport_id"));
// I4-Q4 (D-1030: "require a current proof before a payout is sent"): a passport with no proof pair current by
// account-standing.ts proofIsCurrent is held — its money stays owed and is paid by the first run after it re-proves.
/** Binds `standingBinds(nowMs)`: the four proof binds, then `now`, then `now` again for the grant deadline.
 * I3-1768 (D-1082(1), I32-FX1-VERIFIED): a creator whose latest grant or renewal has lapsed at the run's instant is
 * not paid by the weekly page or STEP 0 even before the expiry sweep folds `claim_state` back (standing.ts
 * `grantLiveAtSql`); the money stays owed and reserved, and a renewal makes it payable on the next run. */
const STANDING_EXISTS = sqlOnce(() => `EXISTS (SELECT 1 FROM passports p
                       WHERE p.passport_id = c.passport_id
                         AND p.claim_state = 'verified'
                         AND p.retracted = 0
                         AND NOT EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = p.passport_id)
                         AND ${RELEASE_PROOF_SQL("p.passport_id")}
                         AND ${RELEASE_OWN_DEST()}
                         AND ${grantLiveAtSql("p.passport_id")})`);
const standingBinds = (nowMs: number): string[] => [...currentProofBinds(nowMs), new Date(nowMs).toISOString(), new Date(nowMs).toISOString()];

/** I3-0028 (D-0911(1)): the value an OPERATOR HOLD writes into `next_attempt_at`.
 * A held row stays `pending` (open, reserved — its money is never re-reserved
 * under another key) and is never selected again by STEP 0 (`next_attempt_at <=
 * now`) or replayed by the release door; `reconcileHolds()` lists it. No
 * migration: the column and the open status already carry it. */
export const PAYOUT_RECONCILE_HOLD_AT = "9999-12-31T23:59:59.999Z";

/** Ledger rows one reconcile carries. Past it the oldest rows drop out, which
 * can only make a transfer look UNRECORDED — a hold, never a send. */
const LEDGER_ROWS = 200;

/** The reconcile's LEDGER for one creator as a JSON array of
 * `[period_key, amount_cents, stripe_transfer_id]`: every OPEN row, and every
 * PAID row paid since the window start. Correlated on `passportCol`; binds ONE
 * parameter (the window start). It rides the SELECTs the run already makes, so
 * the reconcile costs no statement. */
// I3-0028 d (I22): a restore loses only rows NEWER than its restore point, so every transfer it can lose was
// created after the creator's newest SURVIVING paid row. The paid rows are therefore carried back to that row (two
// days of skew) whenever it is older than the window, and `reconcileFor` lists back to it — a lost payout older than
// RECONCILE_WINDOW_MS is still seen. I3-0028 / c (D-0976(1)): `dests` is every account the creator has been bound to.
const NEWEST_PAID = (passportCol: string): string =>
  `(SELECT MAX(l1.paid_at) FROM creator_payouts l1 WHERE l1.passport_id = ${passportCol} AND l1.status = 'paid')`;
const ledgerJsonSql = (passportCol: string): string => `json_object('rows', json((SELECT json_group_array(json_array(lg.period_key, lg.amount_cents, lg.stripe_transfer_id))
           FROM (SELECT l0.period_key, l0.amount_cents, l0.stripe_transfer_id, l0.created_at FROM creator_payouts l0
                  WHERE l0.passport_id = ${passportCol}
                    AND (l0.status IN ('pending','failed') OR (l0.status = 'paid' AND l0.paid_at >= min(?, COALESCE(strftime('%Y-%m-%dT%H:%M:%fZ', ${NEWEST_PAID(passportCol)}, '-2 days'), '9999'))))
                  ORDER BY l0.created_at DESC LIMIT ${LEDGER_ROWS}) lg)),
         'dests', json((SELECT json_group_array(dx.acct) FROM (SELECT dd.stripe_account_id AS acct, MAX(dd.decided_at) AS last_at FROM connect_binding_decisions dd WHERE dd.passport_id = ${passportCol} GROUP BY dd.stripe_account_id ORDER BY last_at DESC, acct) dx)),
         'newest_paid', ${NEWEST_PAID(passportCol)})`;

/** The listing window's start, and (one day earlier, for skew) the ledger's. */
function reconcileWindow(nowMs: number): { sinceMs: number; ledgerSinceIso: string } {
  const sinceMs = nowMs - RECONCILE_WINDOW_MS;
  return { sinceMs, ledgerSinceIso: new Date(sinceMs - 24 * 60 * 60 * 1000).toISOString() };
}

/** The reconcile one send carries. An unreadable ledger is a reconcile that
 * cannot complete: the returned value is one `createConnectTransfer` refuses
 * (null — nothing sent, the row stays pending), never an empty ledger, which
 * would read every transfer as unrecorded. `own` is the row being paid, added
 * when the read predates its reservation. */
function reconcileFor(passportId: string, ledgerJson: unknown, sinceMs: number, own: { periodKey: string; amountCents: number }): TransferReconcile {
  const refuse: TransferReconcile = { keyPrefix: "", ledger: [], sinceMs: Number.NaN };
  let obj: unknown;
  try {
    obj = typeof ledgerJson === "string" ? JSON.parse(ledgerJson) : null;
  } catch {
    return refuse;
  }
  if (typeof obj !== "object" || obj === null) return refuse;
  const { rows: parsed, dests, newest_paid: newestPaid } = obj as { rows?: unknown; dests?: unknown; newest_paid?: unknown };
  if (!Array.isArray(parsed) || !Array.isArray(dests) || !dests.every((d) => typeof d === "string")) return refuse;
  if (newestPaid !== null && (typeof newestPaid !== "string" || !Number.isFinite(Date.parse(newestPaid)))) return refuse;
  // I3-0028 d: back to the newest surviving paid row (one day of skew) when it is older than the window; 0 (all of
  // Stripe's history for these destinations) when the creator has no paid row at all.
  const since = newestPaid === null ? 0 : Math.min(sinceMs, Date.parse(newestPaid) - 24 * 60 * 60 * 1000);
  const ledger: PayoutLedgerRow[] = [];
  for (const r of parsed) {
    if (!Array.isArray(r) || typeof r[0] !== "string" || !Number.isInteger(r[1]) || (r[2] !== null && typeof r[2] !== "string")) return refuse;
    ledger.push({ key: transferIdempotencyKey(passportId, r[0]), amountCents: r[1] as number, transferId: r[2] as string | null });
  }
  const ownKey = transferIdempotencyKey(passportId, own.periodKey);
  if (!ledger.some((r) => r.key === ownKey)) ledger.push({ key: ownKey, amountCents: own.amountCents, transferId: null });
  // I3-0028 c (I22-FX-MONEY): `dests` is newest decision first (the listing keeps the most recent past
  // MAX_RECONCILE_DESTINATIONS), and `nowMs` anchors stripe.ts's RECONCILE_LOOKBACK_WEEKS group lookups — every caller
  // derives `sinceMs` from `reconcileWindow(nowMs)`, so this is that clock exactly.
  return { keyPrefix: transferIdempotencyKey(passportId, ""), ledger, sinceMs: since, destinations: dests as string[], nowMs: sinceMs + RECONCILE_WINDOW_MS, releaseKeys: releaseKeysAt(sinceMs + RECONCILE_WINDOW_MS) };
}

/** I22-Q1 (D-1030 (a): "accept the cost and also look up operator releases"): the operator-release periods a send asks
 * Stripe about (stripe.ts `TransferReconcile.releaseKeys`) — the RECONCILE_RELEASE_KEYS UTC days ending at the run's day,
 * newest first (a release is keyed by its day, RELEASE_PERIOD_PREFIX). A lost release older than that is still found
 * by the per-destination listing when its destination survived; one to a lost destination is not (JASON). */
function releaseKeysAt(nowMs: number): string[] {
  const out: string[] = [];
  for (let k = 0; k < RECONCILE_RELEASE_KEYS; k++) out.push(`${RELEASE_PERIOD_PREFIX}${new Date(nowMs - k * 86_400_000).toISOString().slice(0, 10)}`);
  return out;
}

/** The OPERATOR HOLD (D-0911(1)): the open row is parked (`PAYOUT_RECONCILE_HOLD_AT`)
 * and an `audit_log` row records what Stripe holds against what we hold — bound
 * to that transition (it lands only if THIS row is now open and held), so a row
 * paid by another door in the meantime is neither held nor reported held. */
function holdStmts(db: DbLike, payoutId: string, passportId: string, periodKey: string, amountCents: number, h: TransferHold, atIso: string): DbStatement[] {
  const detail = JSON.stringify({
    reason: h.hold,
    payout_id: payoutId,
    passport_id: passportId,
    period_key: periodKey,
    amount_cents: amountCents,
    stripe_transfer_id: h.transferId,
    stripe_transfer_group: h.transferGroup,
    stripe_amount_cents: h.stripeAmountCents,
    stripe_destination: h.stripeDestination,
  });
  return [
    db.prepare(`UPDATE creator_payouts SET next_attempt_at = ? WHERE payout_id = ? AND (status = 'pending' OR status = 'failed')`).bind(PAYOUT_RECONCILE_HOLD_AT, payoutId),
    db
      .prepare(
        `INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind)
              SELECT 'system', 'payout:reconcile_hold', ?, ?, ?, 'system'
               WHERE EXISTS (SELECT 1 FROM creator_payouts WHERE payout_id = ? AND next_attempt_at = ? AND status IN ('pending','failed'))`,
      )
      .bind(detail, atIso, payoutId, payoutId, PAYOUT_RECONCILE_HOLD_AT),
    // I16-RG-MONEY:16: the hold ends the attempt, so it ends the attempt's send claim (a no-op where none was taken).
    releaseSendStmt(db, payoutId),
  ];
}

/** D-0960(1): payout-status-sql.ts DISPUTE_DEBIT_CENTS (I34-W3-MONEY, QUEUE #947: moved to the billing-free leaf so

/** STEP 0's OWN standing/active predicate: `PAYOUT_PAGE_SQL`'s two EXISTS legs

const STEP0_PAYABLE = sqlOnce(() => `AND EXISTS (SELECT 1 FROM creator_connect c
            WHERE c.passport_id = creator_payouts.passport_id
              AND ${ACTIVE_BINDING}
              AND ${STANDING_EXISTS()})`);

/** Cleared minus reserved minus dispute debits for ONE passport, as a scalar:

const AVAILABLE_CENTS = `((SELECT COALESCE(SUM(sa.creator_net_cents),0) FROM sales sa
                              WHERE sa.passport_id = ? AND sa.reversed = 0 AND sa.clears_at <= ?)
                          - (SELECT COALESCE(SUM(ra.amount_cents),0) FROM creator_payouts ra
                              WHERE ra.passport_id = ? AND ra.status IN ${RESERVED_PAYOUT_STATUS_SQL})
                          - ${DISPUTE_DEBIT_CENTS("?")})`;

/** The release's standing over the passports alias `p` (D-0750 (2), D-0752):
 * every condition the weekly run's `STANDING_EXISTS` asks EXCEPT that the
 * passport is unsuppressed — here it must BE suppressed — plus the account ban.
 * One const, read by `releaseStanding` for the refusal word and interpolated
 * into `reserveReleaseStmt` so the write re-asks it. */
const RELEASE_BANNED = `EXISTS (SELECT 1 FROM founder f JOIN accounts a ON a.account_id = f.account_id
                       WHERE f.passport_id = p.passport_id
                         AND (a.banned_at IS NOT NULL
                              OR EXISTS (SELECT 1 FROM account_closures k
                                          WHERE k.account_id = a.account_id AND k.state = 'suspended')))`;
const RELEASE_SUPPRESSED = "EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = p.passport_id)";
/** R3-I29-Q7 (D-1091(4)): a TERMINATED creator (a suspension episode carrying the terminate row, RELEASE_OWN_DEST's
 * terminated limb) is exempt from the proof clause only, at EVERY paying door: the weekly run (STANDING_EXISTS: the page
 * read, STEP 0) and the release statements (RELEASE_STANDING_WHERE, releaseStanding); the 14-day hold and the
 * fraud-finding check stay in RELEASE_OWN_DEST. Binds `currentProofBinds(nowMs)`, as CURRENT_PROOF_SQL does. */
const RELEASE_PROOF_SQL = (passportExpr: string): string =>
  `(${CURRENT_PROOF_SQL(passportExpr)} OR EXISTS (SELECT 1 FROM creator_suspensions te WHERE te.passport_id = ${passportExpr} AND ${CREATOR_TERMINATED_FOR("te.suspension_id")}))`;
/** Which door pays through the release statements: the operator release (suppressed passports only) or the erasure's
 * one residual payment (R3-I29-Q9, creator-erasure.ts eraseCreator), which does not ask for a suppression. */
type ReleaseKind = "release" | "erasure";
/** I4-Q4: binds `currentProofBinds(nowMs)`, then the instant (I3-1768, I32-FX1-VERIFIED: the grant has not lapsed at it). */
const RELEASE_STANDING_WHERE = sqlOnce(() => `p.claim_state = 'verified' AND p.retracted = 0 AND ${RELEASE_SUPPRESSED} AND NOT ${RELEASE_BANNED} AND ${RELEASE_PROOF_SQL("p.passport_id")} AND ${grantLiveAtSql("p.passport_id")}`);
/** RELEASE_STANDING_WHERE without the suppression term (kind "erasure"); same binds. */
const ERASURE_STANDING_WHERE = sqlOnce(() => RELEASE_STANDING_WHERE().replace(`${RELEASE_SUPPRESSED} AND `, ""));

/* THE WEEKLY RUN'S FOUR MONEY STATEMENTS, written once (B25-635). STEP 0, STEP 1
 * and `releaseHeldPayout` build them here, so the operator release cannot pay by
 * a second spelling of reserve, re-ask, settle flip or refusal. Same SQL text, same
 * binding order, same statement count as the inline forms they replaced. */

/** SJ-01's reservation: the row is written only while the EXACT captured
 * destination is still an `ACTIVE_BINDING`, evaluated inside the INSERT. */
function reservePayoutStmt(db: DbLike, payoutId: string, passportId: string, periodKey: string, amountCents: number, reservedAt: string, destination: string, bindableBefore: string, nowIso: string): DbStatement {
  return db
    .prepare(
      `INSERT INTO creator_payouts (payout_id, passport_id, period_key, amount_cents, stripe_transfer_id, status, created_at)
              SELECT ?, ?, ?, ?, NULL, 'pending', ?
               WHERE EXISTS (SELECT 1 FROM creator_connect c
                              WHERE c.passport_id = ?
                                AND c.stripe_account_id = ?
                                AND ${ACTIVE_BINDING}
                                AND ${OWN_DEST_C()})
                 AND ? <= ${AVAILABLE_CENTS}`,
    )
    .bind(payoutId, passportId, periodKey, amountCents, reservedAt, passportId, destination, bindableBefore, nowIso, amountCents, passportId, nowIso, passportId, passportId);
}

/** B25-635 (RG-635-01/02): the release's reservation. The AMOUNT is computed
 * by this statement (`AVAILABLE_CENTS` at insert time), and the row lands only
 * while the release's whole standing still holds (`RELEASE_STANDING_WHERE`,
 * suppression included) and the captured destination is still the
 * `ACTIVE_BINDING` — so no release, weekly run or retry interleaved before it can
 * leave it reserving money that is already reserved. */
function reserveReleaseStmt(db: DbLike, payoutId: string, passportId: string, periodKey: string, reservedAt: string, destination: string, bindableBefore: string, nowIso: string, kind: ReleaseKind = "release"): DbStatement {
  return db
    .prepare(
      `INSERT INTO creator_payouts (payout_id, passport_id, period_key, amount_cents, stripe_transfer_id, status, created_at)
              SELECT ?, p.passport_id, ?, v.avail, NULL, 'pending', ?
                FROM passports p, (SELECT ${AVAILABLE_CENTS} AS avail) v
               WHERE p.passport_id = ?
                 AND v.avail > 0
                 AND ${kind === "erasure" ? ERASURE_STANDING_WHERE() : RELEASE_STANDING_WHERE()}
                 AND EXISTS (SELECT 1 FROM creator_connect c
                              WHERE c.passport_id = p.passport_id
                                AND c.stripe_account_id = ?
                                AND ${ACTIVE_BINDING}
                                AND ${RELEASE_OWN_DEST()})`,
    )
    .bind(payoutId, periodKey, reservedAt, passportId, nowIso, passportId, passportId, passportId, ...currentProofBinds(Date.parse(nowIso)), nowIso, destination, bindableBefore, nowIso);
}

/** I56-MONEY (D-1433(2), INV-088): the reservation's OWN audit row, batched directly after the reserving INSERT and
 * bound to it having landed (`changes() = 1`, the row count of the statement before it; the :578 shape). A refused
 * reservation (0 rows) writes none, and a UNIQUE throw rolls this row back with it. Ids only (INV-061). */
function reservedAuditStmt(db: DbLike, payoutId: string, periodKey: string, kind: "weekly" | ReleaseKind, at: string): DbStatement {
  return db
    .prepare(
      `INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind)
            SELECT 'system', 'payout:reserved', ?, ?, ?, 'system' WHERE changes() = 1`,
    )
    .bind(`payout ${payoutId} reserved: period ${periodKey}, ${kind}`, at, payoutId);
}
/** The rows-affected of one batch element, `meta.changes` (D1 keeps it beside the `changes` d1() lifts, db.ts:188-190;
 * the test double carries only it). Absent reads 0: the reservation is treated as refused and nothing is sent. */
const batchChanges = (r: unknown): number => (r as { meta?: { changes?: number } } | undefined)?.meta?.changes ?? 0;

/**
 * I16-RG-MONEY:16 / D-0931(3) — THE SEND CLAIM. A retry door (STEP 0, the release replay) sent a transfer and only
 * then flipped the row `paid`; `voidPayout` flips the same open row `voided`, so a void landing between `spend()`
 * and the flip left a SENT transfer on a `voided` row (no register row, a clean reconcile). The door now CLAIMS the
 * row before it sends: a `cron_leases` row named `payout_send:<id>` (the lease table's own exclusivity, cron-lease.ts),
 * live for `PAYOUT_SEND_CLAIM_MS`, released in the settle/refusal/hold batch that ends the attempt. `voidPayout`
 * refuses (409) while one is live. A STEP 1 / release row is claimed by its RESERVATION: an open row with no attempt
 * recorded (`attempts = 0`) created inside the same window is being sent by the run that just reserved it.
 * `SEND_IN_FLIGHT` is that one predicate, read by the void.
 */
export const PAYOUT_SEND_CLAIM_MS = 15 * 60 * 1000;
const sendClaimName = (payoutId: string): string => `payout_send:${payoutId}`;
/** Correlated on `creator_payouts` (unaliased); binds: (nowIso, claimCutoffIso). */
const SEND_IN_FLIGHT = `(EXISTS (SELECT 1 FROM cron_leases l WHERE l.name = 'payout_send:' || creator_payouts.payout_id AND l.expires_at >= ?)
     OR (creator_payouts.attempts = 0 AND creator_payouts.created_at > ?))`;

/** KM-017's re-ask before a retry dispatch — the row is still open AND the destination about to be paid is still the
 * active binding — now WRITTEN as the send claim (one statement, as the read it replaces): it lands only when the
 * re-ask holds and no live claim exists. `.all()` and `results.length === 1` is "claimed". */
function claimSendStmt(db: DbLike, payoutId: string, passportId: string, destination: string, bindableBefore: string, nowMs: number): DbStatement {
  const nowIso = new Date(nowMs).toISOString();
  return db
    .prepare(
      `INSERT INTO cron_leases (name, holder, acquired_at, expires_at)
         SELECT ?, 'payout_send', ?, ? WHERE EXISTS (SELECT 1 FROM creator_payouts
          WHERE payout_id = ? AND (status = 'pending' OR status = 'failed')
            AND EXISTS (SELECT 1 FROM creator_connect c
                         WHERE c.passport_id = ?
                           AND c.stripe_account_id = ?
                           AND ${ACTIVE_BINDING}
                           AND ${OWN_DEST_C()}))
       ON CONFLICT(name) DO UPDATE SET holder = excluded.holder, acquired_at = excluded.acquired_at, expires_at = excluded.expires_at
         WHERE cron_leases.expires_at < excluded.acquired_at
       RETURNING name`,
    )
    .bind(sendClaimName(payoutId), nowIso, new Date(nowMs + PAYOUT_SEND_CLAIM_MS).toISOString(), payoutId, passportId, destination, bindableBefore, nowIso);
}
/** Ends an attempt's claim; batched with the write that ends the attempt (no extra statement cost). */
function releaseSendStmt(db: DbLike, payoutId: string): DbStatement {
  return db.prepare("DELETE FROM cron_leases WHERE name = ?").bind(sendClaimName(payoutId));
}

/** The settle flip. Every caller batches it AFTER its own `creator_earning`
 * outbox statements, whose `extraWhere` is `STILL_PENDING` — this UPDATE's own
 * predicate, read before it (S3-11 / A-5). The outbox call stays at each caller
 * so each publishing function keeps its own pubEnv slot. RETURNING, not
 * `changes` (creator_payouts UPDATE is an INFLATING_WRITE, 0201's trigger):
 * `flipLanded` reads it, and D-0931(3)'s alarm fires when a SENT transfer's flip
 * changed no row. */
function markPaidStmt(db: DbLike, payoutId: string, transferId: string, paidAt: string): DbStatement {
  return db.prepare(`UPDATE creator_payouts SET status='paid', stripe_transfer_id=?, paid_at=? WHERE payout_id=? AND (status = 'pending' OR status = 'failed') RETURNING payout_id`).bind(transferId, paidAt, payoutId);
}
/** #936(c) / I3-1161b (D-0954(4)): the paid transfer's OWN audit row, in the settle batch AFTER `markPaidStmt`
 * and bound to THIS flip's values (INV-058) — the row `paid` under THIS transfer id at THIS `paid_at` — so a
 * refused, held or failed transfer (no settle batch) and a settle whose flip did not land write none. */
function paidAuditStmt(db: DbLike, payoutId: string, transferId: string, amountCents: number, periodKey: string, paidAt: string): DbStatement {
  return db
    .prepare(
      `INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind)
            SELECT 'system', 'payout:paid', ?, ?, ?, 'system'
             WHERE EXISTS (SELECT 1 FROM creator_payouts WHERE payout_id = ? AND status = 'paid' AND stripe_transfer_id = ? AND paid_at = ?)`,
    )
    .bind(`payout ${payoutId} paid: transfer ${transferId}, ${amountCents} cents, period ${periodKey}`.slice(0, 500), paidAt, payoutId, payoutId, transferId, paidAt);
}
/** True when the markPaid element of a batch result returned its row. */
function flipLanded(res: unknown): boolean {
  const rows = (res as { results?: unknown } | null | undefined)?.results;
  return Array.isArray(rows) && rows.length === 1;
}
/** D-0931(3): a transfer was SENT and its row did not flip `paid` (another writer — a void — moved it first). */
function alarmFlipMissed(payoutId: string): void {
  logError({ cid: "payout-run", hop: "cron", msg: "payout_paid_flip_missed", marker: "PAYOUT_FLIP_MISSED", detail: payoutId });
}

/** BF-13: a refused transfer is written down (attempts + backoff); amount and
 * reservation are untouched. M-15: the refusal that reaches
 * `PAYOUT_FAIL_ATTEMPTS` also writes `status = 'failed'` (still open, still
 * reserved, still retried — see the constant). */
function recordRefusalStmt(db: DbLike, payoutId: string, attempts: number, nextAttemptAt: string): DbStatement {
  return db
    .prepare(`UPDATE creator_payouts SET attempts = ?, next_attempt_at = ?, status = CASE WHEN ? >= ? THEN 'failed' ELSE status END WHERE payout_id = ? AND (status = 'pending' OR status = 'failed')`)
    .bind(attempts, stalls(attempts) ? PAYOUT_RECONCILE_HOLD_AT : nextAttemptAt, attempts, PAYOUT_FAIL_ATTEMPTS, payoutId);
}
/** I16-S5-n138C-36 (D-1030): the refusal count that parks a payout for an operator instead of retrying it. */
const stalls = (attempts: number): boolean => attempts >= PAYOUT_STALL_ATTEMPTS;
/** Every refusal's statements, at every door: the refusal write (which parks at PAYOUT_STALL_ATTEMPTS) and, when it
 * parks, the `payout:stall_hold` audit row bound to THIS park (the row now held at this attempt count). */
function refusalStmts(db: DbLike, payoutId: string, attempts: number, nextAttemptAt: string, atIso: string): DbStatement[] {
  const write = recordRefusalStmt(db, payoutId, attempts, nextAttemptAt);
  if (!stalls(attempts)) return [write];
  return [
    write,
    db
      .prepare(
        `INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind)
              SELECT 'system', 'payout:stall_hold', ?, ?, ?, 'system'
               WHERE EXISTS (SELECT 1 FROM creator_payouts WHERE payout_id = ? AND attempts = ? AND next_attempt_at = ? AND (status = 'pending' OR status = 'failed'))`,
      )
      .bind(`payout ${payoutId} held for an operator after ${attempts} refused attempts`, atIso, payoutId, payoutId, attempts, PAYOUT_RECONCILE_HOLD_AT),
  ];
}

interface PayoutPageRow {
  passport_id: string;
  stripe_account_id: string;
  cleared_cents: number;
  reserved_cents: number;
  period_rows: number;
  /** D-0960(1): Σ creator_dispute_debits for the creator, netted from the amount. */
  debit_cents: number;
  ledger_json: string | null;
}

/**
 * The weekly payout run — safe to call every cron tick (it self-limits to one
 * payout per creator per ISO week via UNIQUE(passport, period)). For each
 * creator with an ACTIVE Connect binding and cleared ≥ the minimum:
 *  1. RESERVE a payout row (pending) with the amount — a UNIQUE clash means this
 *     week is already handled, skip;
 *  2. transfer via Stripe with the payout id as the idempotency key (a retry
 *     never double-pays);
 *  3. mark the row paid. A failed transfer leaves the row `pending` and it
 *     re-attempts next tick with the SAME reserved amount + idempotency key.
 *
 * PAGED AND BOUNDED — see THE BOUNDS above for the arithmetic. Stopping early is
 * always safe here and that is not an accident: the reserve-before-transfer row
 * is the unit of work, so a creator this tick did not reach is a creator whose
 * money is still owed and still computed the same way on the next tick, and a
 * creator this tick reserved but could not pay is a `pending` row the next tick
 * replays under the SAME payout id. Neither can pay twice.
 *
 * D-1126(6): `opts.fetchImpl` is the fetch the Connect transfer is sent on (stripe.ts createConnectTransfer's third
 * argument, at `spend` below); absent, createConnectTransfer's default, the global fetch. D-1127(5): it is the Connect
 * account read's fetch too (`accountReadFor`), unless `opts.accountState` stands in for the read.
 */
export async function runWeeklyPayouts(
  db: DbLike,
  cfg: ConnectConfig,
  nowMs: number,
  mintPayoutId: () => string,
  now: () => string = () => new Date().toISOString(),
  transfer: typeof createConnectTransfer = createConnectTransfer,
  opts: PayoutRunOptions & { fetchImpl?: typeof fetch } = {},
  pubEnv?: AnalyticsTickEnv,
  pubCtx?: { waitUntil(p: Promise<unknown>): void },
): Promise<PayoutRunResult> {
  const period = isoWeek(nowMs);
  const nowIso = new Date(nowMs).toISOString();
  const page = opts.page ?? PAYOUT_PAGE;
  const transferBudget = opts.transferBudget ?? PAYOUT_TRANSFER_BUDGET;
  const retryBudget = Math.min(opts.retryBudget ?? PAYOUT_RETRY_BUDGET, transferBudget);
  const clock = opts.clock ?? (() => Date.now());
  const deadline = clock() + (opts.budgetMs ?? PAYOUT_RUN_BUDGET_MS);
  const out: PayoutRunResult = { paid: [], skipped_below_min: 0, failed: [], examined: 0, transfers: 0, stopped: null, held: 0, retry_truncated: 0, deadline_abandoned: 0, revoked_before_reserve: 0, wrapped: false, held_to_table_end: false, stalled: [], reconcile_holds: [] };
  const { sinceMs, ledgerSinceIso } = reconcileWindow(nowMs);
  const standingArgs = standingBinds(nowMs);
  const accountRead = accountReadFor(transfer, opts.accountState, opts.fetchImpl);

  /** One outbound dispatch, spent against BOTH ceilings. Answers undefined when
   * there is no budget left to spend, which every caller treats as "stop cleanly".
   * I3-0028 (D-0541, D-0911(1)): the reconcile is a REQUIRED argument of the
   * primitive (stripe.ts `createConnectTransfer`), so neither STEP 0 nor STEP 1
   * — nor the release door — can send without it. A dispatch is at most
   * `PAYOUT_DISPATCH_SUBREQUESTS`, which is what makes a dispatch count a
   * subrequest bound. A `TransferHold` answer means nothing was sent. */
  const spend = async (args: { destination: string; amountCents: number; idempotencyKey: string; reconcile: TransferReconcile }, ceiling: number): Promise<string | TransferHold | TransferRefusal | null | undefined> => {
    if (out.transfers >= ceiling) {
      out.stopped = "transfer_budget";
      return undefined;
    }
    if (clock() >= deadline) {
      out.stopped = "deadline";
      return undefined;
    }
    out.transfers += 1;
    return transfer(cfg, args, opts.fetchImpl);
  };
  const hold = async (payoutId: string, passportId: string, periodKey: string, amountCents: number, h: TransferHold): Promise<void> => {
    await db.batch(holdStmts(db, payoutId, passportId, periodKey, amountCents, h, now()));
    out.reconcile_holds.push({ payout_id: payoutId, passport_id: passportId, period_key: periodKey, amount_cents: amountCents, reason: h.hold });
  };

  // RESUME WHERE THE LAST TICK STOPPED. `cursor_id` is the last passport id
  // examined; `cursor_seq` is unused for this sweep (the connected set has no
  // sequence to order by — the primary key is the total order). An unreadable or
  // absent row starts at the beginning of the table, which is the fail-safe
  // direction: it re-examines rather than skips.
  const cur = (await db.prepare("SELECT cursor_id FROM sweep_cursor WHERE name = ?").bind(PAYOUT_CURSOR).first<{ cursor_id: string }>())?.cursor_id ?? "";

  // The cooling-off cut-off: a binding activated after this moment is skipped,
  // so a destination bound during a session compromise cannot be paid before the
  // creator has had a day to notice and disconnect it.
  const bindableBefore = new Date(nowMs - CONNECT_COOLING_OFF_MS).toISOString();
  const { results: actives } = await db.prepare(PAYOUT_PAGE_SQL()).bind(nowIso, period, ledgerSinceIso, bindableBefore, ...standingArgs, cur, page).all<PayoutPageRow>();

  const fullPage = actives.length === page;
  out.held_to_table_end = !fullPage; // Q102: the window `heldWindow` below counts over
  const pageEnd = actives.length > 0 ? actives[actives.length - 1]!.passport_id : "";
  // The window STEP 0 works in — bounded by the page only when there IS a page
  // after this one to catch the tail.
  //
  // I3-1132 — PLUS every row whose outcome may be UNKNOWN while its idempotency
  // key is still honoured, wherever the cursor is. A re-send is safe only
  // because `transferIdempotencyKey` makes Stripe dedup a transfer that already
  // landed, and Stripe keeps a key for a bounded window (24 h) — while the page
  // window above reaches a given passport again only after a full cursor cycle,
  // which on a large connected set is longer. Two shapes are in this limb:
  //  - `attempts = 0 AND next_attempt_at IS NULL`: no outcome was ever recorded
  //    (STEP 1 crashed after dispatch, or its ceiling stopped it), at any age;
  //  - any open row reserved inside the key window (`created_at > ?`): a
  //    timed-out call is recorded as a refusal (stripe.ts returns null for
  //    both), so a recorded attempt does not prove nothing landed. Its backoff
  //    still decides WHEN (`next_attempt_at <= now` below); this decides that
  //    the cursor's position does not.
  // Past the key window a re-send is no longer deduplicated by the key — every
  // dispatch therefore looks its transfer_group up first (`spend`, I3-0028).
  // `created_at < now` keeps a row this very tick reserves out of the limb
  // (STEP 1 runs after STEP 0).
  const keyWindowStart = new Date(nowMs - PAYOUT_KEY_WINDOW_MS).toISOString();
  const unknownLimb = "(created_at < ? AND ((attempts = 0 AND next_attempt_at IS NULL) OR created_at > ?))";
  // `held` stays COUNTED CURSOR-RELATIVE (docket #58 (b), pinned by
  // tcindex2-payout-held-window.test.ts): it keeps the page-only window, and
  // only STEP 0's pending read takes the widened one.
  const heldWindow = fullPage ? "AND passport_id > ? AND passport_id <= ? " : "AND passport_id > ? ";
  const heldArgs: unknown[] = fullPage ? [cur, pageEnd] : [cur];
  const pendWindow = fullPage ? `AND ((passport_id > ? AND passport_id <= ?) OR ${unknownLimb}) ` : `AND (passport_id > ? OR ${unknownLimb}) `;
  const pendArgs: unknown[] = fullPage ? [cur, pageEnd, nowIso, keyWindowStart] : [cur, nowIso, keyWindowStart];
  // `held`'s statement, prepared but not yet run: on an empty page it is the
  // ONLY statement this tick spends (below), unchanged from before. On a page
  // with rows, it is batched with STEP 0's SELECT (below) instead of run here.
  const heldStmt = db
    .prepare(`SELECT COUNT(*) AS n FROM creator_payouts WHERE (status = 'pending' OR status = 'failed') ${heldWindow}AND NOT (1 = 1 ${STEP0_PAYABLE()})`)
    .bind(...heldArgs, bindableBefore, ...standingArgs);

  if (actives.length === 0) {
    // Past the end of the connected set (or there is none). Wrap so the next
    // tick starts at the beginning; spend no TRANSFER on a pending page whose
    // rows could not be paid anyway — only an ACTIVE binding is ever paid. The
    // one statement spent here is the `held` count, which is the whole point:
    // this is the shape a revoked binding leaves behind. No STEP 0 SELECT
    // exists on this path, so there is nothing to batch `held` against.
    out.held = (await heldStmt.first<{ n: number }>())?.n ?? 0;
    if (cur !== "") await writeCursor(db, "", nowIso);
    out.wrapped = true;
    return out;
  }
  const accountOf = new Map(actives.map((a) => [a.passport_id, a.stripe_account_id]));

  // STEP 0: re-attempt EVERY still-pending payout INSIDE THIS PAGE — this week's
  // OR an orphaned prior week's — under its ORIGINAL payout_id and reserved
  // amount. The idempotency key makes this safe (Stripe dedups a transfer that
  // actually landed before a crash, so it never pays twice), and it is what
  // closes a crash-after-transfer at a week boundary instead of leaving the
  // amount to be re-paid under a fresh key next week.
  //
  // RANGE-SCOPED TO THE PAGE, deliberately: `passport_id > cur AND <= pageEnd`
  // is exactly the window the page covers, so the cursor cannot advance past a
  // pending row that was never looked at. On the LAST page (a short one) the
  // upper bound is dropped instead, because there is no page after it to catch
  // the tail. `STEP0_PAYABLE` carries the standing/active question INTO this
  // read (:643) so a row that could not be paid never spends a retry slot; the
  // rows it declines are in `out.held`. A pending row for a creator who has
  // disconnected simply waits until they reconnect.
  // `retry_truncated` (PayoutRunResult, docket #43 (a)): over-fetch by ONE —
  // `retryBudget + 1` instead of `retryBudget` — on the SAME statement, so
  // detecting truncation costs NOTHING extra (D1 prices a statement, not a
  // row count). A separate scoped COUNT would be exact but is a real added
  // statement every tick (`payout-run-bounds.test.ts` pins the per-tick
  // statement count for exactly this reason — THE BOUNDS, :585-590, hold the
  // cost at "4 statements regardless of N"), so this reports a BOUND, not the
  // exact remainder: 1 when at least one further payable row exists beyond
  // `retryBudget` this tick, 0 when the window was fully drained. The extra
  // row is never processed — only the first `retryBudget` are.
  // Batched with `held`'s COUNT (above), NOT run sequentially (docket #59
  // PAYOUT2, FINDING-0015-C-payouts-3): the two reads are un-transactioned in
  // isolation, so a binding revoked between them was excluded from `held`
  // (payable at that read) and from this SELECT (unpayable by the time it
  // ran) — owed, pending, and invisible in both. `db.batch()` gives both
  // statements one snapshot, so a row is excluded from at most one of the two
  // reads, never both / neither, closing the race. Same two statements as
  // before this change (measured, S-032 PAYOUT23
  // `payout23-transaction-cost.test.ts`: batching two statements together
  // costs the same `db.ops.length` as running them sequentially — 0 delta),
  // so THE BOUNDS' per-tick statement count (:665-722) is unchanged.
  // BF-13 — DUE-NESS IS PART OF THE SELECTION, AND THE ORDER IS BY DUE TIME.
  //
  // This was `ORDER BY passport_id ASC, created_at ASC` over every pending row
  // in the window, and `creator_payouts` recorded no attempt ever having been
  // made (the `failed` status in its CHECK is never written; a refused transfer
  // went onto an in-memory list and the row stayed `pending`). So a creator
  // whose destination fails permanently held the head of that ordering FOR
  // EVER, and once `retryBudget` such rows existed no later pending row in the
  // window was attempted again — on any tick. Their amounts stay RESERVED
  // (`creatorEarnings` subtracts pending + paid), so those creators were not
  // just unpaid: their cleared balance was withheld from new payouts too.
  //
  // `next_attempt_at IS NULL` is "never attempted, eligible now" — which is
  // every row that predates migration 0095, so nothing changes for them on the
  // first tick after deploy. A refused attempt writes a bounded exponential
  // backoff (`payoutBackoffMs`), which is what takes a broken destination OUT
  // of the selection for a while instead of leaving it at the front of it.
  // Ordering by the due time then serves the longest-waiting DUE row first,
  // which is the fair order, rather than the lowest passport id.
  const pendStmt = db
    .prepare(
      `SELECT payout_id, passport_id, period_key, amount_cents, attempts,
              (SELECT d.stripe_account_id FROM creator_connect d WHERE d.passport_id = creator_payouts.passport_id) AS dest,
              ${ledgerJsonSql("creator_payouts.passport_id")} AS ledger_json
         FROM creator_payouts WHERE (status = 'pending' OR status = 'failed') ${pendWindow}${STEP0_PAYABLE()}
         AND (next_attempt_at IS NULL OR next_attempt_at <= ?)
       ORDER BY COALESCE(next_attempt_at, created_at) ASC, passport_id ASC LIMIT ?`,
    )
    .bind(ledgerSinceIso, ...pendArgs, bindableBefore, ...standingArgs, nowIso, retryBudget + 1);
  const [heldBatchResult, pendBatchResult] = (await db.batch([heldStmt, pendStmt])) as [
    { results: { n: number }[] },
    { results: { payout_id: string; passport_id: string; period_key: string; amount_cents: number; attempts: number; dest: string | null; ledger_json: string | null }[] },
  ];
  out.held = heldBatchResult.results[0]?.n ?? 0;
  const pendingsFetched = pendBatchResult.results;
  out.retry_truncated = pendingsFetched.length > retryBudget ? 1 : 0;
  const pendings = out.retry_truncated ? pendingsFetched.slice(0, retryBudget) : pendingsFetched;

  // CM-02: STEP 0 gets its OWN wall-clock slice, the same carve-out idea as
  // `retryBudget` but for time instead of call count — otherwise a handful of
  // slow or hanging retries can spend the run's ENTIRE deadline before STEP 1
  // examines a single new creator: the request ceiling was guaranteed to be
  // spared (`retryBudget`), cursor progress was not. Skipped when there is
  // nothing to retry, so an empty pending window costs no extra `clock()`
  // call. Never wider than the run's real deadline (`Math.min`), so this can
  // only make STEP 0 stop SOONER, never later.
  const retryDeadline = pendings.length > 0 ? Math.min(deadline, clock() + (opts.retryDeadlineMs ?? PAYOUT_RETRY_DEADLINE_MS)) : deadline;

  // `deadline_abandoned`'s STEP 0 half: rows this loop had not yet reached
  // when `spend` reported the clock ran out (docket #43 (a), FINDING-0015-C-
  // payouts-5) OR (CM-02) when STEP 0's own `retryDeadline` ran out first.
  // `retryBudget` bounds both this LIMIT and `spend`'s ceiling (:772, :849)
  // and the loop starts at `out.transfers === 0`, so the ceiling itself
  // cannot be hit inside this loop — an `undefined` from `spend` here is
  // always the run deadline; the `out.stopped` check is kept anyway so this
  // number never claims a deadline that did not happen.
  let step0DeadlineLeft = 0;
  for (let i = 0; i < pendings.length; i++) {
    if (clock() >= retryDeadline) {
      // STEP 0's OWN slice ran out — not necessarily the run's. `out.stopped`
      // is deliberately left untouched in that case: the run's real deadline
      // may still be far off, and STEP 1 spends whatever is left of it rather
      // than being told the tick is over.
      //
      // W2 (S-039): `retryDeadline` is `min(deadline, clock() + slice)`, so
      // hitting it means EITHER the slice ran out OR the run itself did — and
      // CM-02's first form treated both as the slice, which silently un-set
      // the one thing FINDING-0015-C-payouts-5 exists to pin: a run that stops
      // on its deadline must SAY it stopped on its deadline. The two cases are
      // distinguishable and must be distinguished, because `stopped: null`
      // after an abandoned run is the tick hiding what it did not reach, which
      // is the whole defect docket #43 (a) was ruled on.
      if (clock() >= deadline) out.stopped = "deadline";
      step0DeadlineLeft = pendings.length - i;
      break;
    }
    const p = pendings[i]!;
    // I3-1132: an outcome-unknown row outside this page has no `accountOf`
    // entry; its destination rides the pending read (`dest`), and the
    // `claimSendStmt` re-ask below binds it exactly as it binds a page row's.
    const dest = accountOf.get(p.passport_id) ?? p.dest ?? undefined;
    if (!dest) continue; // no active binding — leave pending until they (re)connect
    // KM-017 — SJ-01's RE-ASK, AT THE RETRY DOOR TOO.
    // STEP 1 re-asks the destination INSIDE its reservation write (:1259) so a
    // revocation committing after the page read cannot be paid. STEP 0 had no
    // such write: `STEP0_PAYABLE` rides the pending SELECT (:1094), which is a
    // READ, and the destination actually dispatched is the one
    // `PAYOUT_PAGE_SQL` put in `accountOf` (above). A disconnect — or a rebind
    // to a different account — committing between that batched read and this
    // dispatch was therefore invisible here, and the transfer landed on the
    // account the creator had just successfully disconnected.
    // Same predicate (`ACTIVE_BINDING`), same binding of the EXACT captured
    // destination, re-asked here — immediately before `spend()` below — instead
    // of trusting what the batched page read put in `accountOf`.
    //
    // WHAT THIS CHECK DOES AND DOES NOT DO (D-0459; corrected I17 #317g, D-0941(2f)).
    // It NARROWS the revocation window from "page read -> dispatch", which is the
    // whole `pendings` loop, down to "this statement -> dispatch". The statement
    // below (`claimSendStmt`) DOES claim the row: it writes the `payout_send:<id>`
    // lease, so `voidPayout` refuses the row (SEND_IN_FLIGHT) and a second sender
    // cannot claim it until this attempt's batch releases it. It does NOT close the
    // REVOCATION race: a disconnect does not read that lease, so a revocation
    // committing between the claim and the transfer is still paid. Closing that
    // needs the disconnect to honour the claim or a destination-bound Stripe
    // idempotency key (#317e, Jason's); both are T3 changes to the money path.
    //
    // A row this declines is left `pending` and counted into `held` (the same
    // number the batched COUNT above uses for rows it already knew were
    // unpayable) — the money is still owed, still reserved, still this
    // creator's.
    // I16-RG-MONEY:16: the re-ask IS the send claim (claimSendStmt) — same statement count, and a void now refuses
    // this row from here until the attempt's own batch below releases it.
    // I21-Q52 (D-1030): onboarding finished before any transfer. An unfinished account is HELD, not failed: no attempt
    // is recorded, and the row waits one base step so it cannot hold the head of the due order (BF-13).
    if (!connectReady(await accountRead(cfg, dest))) {
      await db
        .prepare("UPDATE creator_payouts SET next_attempt_at = ? WHERE payout_id = ? AND (status = 'pending' OR status = 'failed') AND (next_attempt_at IS NULL OR next_attempt_at <> ?)")
        .bind(new Date(nowMs + PAYOUT_RETRY_BASE_MS).toISOString(), p.payout_id, PAYOUT_RECONCILE_HOLD_AT)
        .run();
      out.held += 1;
      out.connect_unfinished = (out.connect_unfinished ?? 0) + 1;
      continue;
    }
    const claimed = (await claimSendStmt(db, p.payout_id, p.passport_id, dest, bindableBefore, nowMs).all()).results.length === 1;
    if (!claimed) {
      out.held += 1;
      continue;
    }
    // I3-1132: a replay may be past Stripe's 24 h key window — `spend` looks up first.
    const transferId = await spend(
      { destination: dest, amountCents: p.amount_cents, idempotencyKey: transferIdempotencyKey(p.passport_id, p.period_key), reconcile: reconcileFor(p.passport_id, p.ledger_json, sinceMs, { periodKey: p.period_key, amountCents: p.amount_cents }) },
      retryBudget,
    );
    if (transferId === undefined) {
      if (out.stopped === "deadline") step0DeadlineLeft = pendings.length - i;
      break; // out of retry budget or out of clock
    }
    if (isTransferHold(transferId)) {
      await hold(p.payout_id, p.passport_id, p.period_key, p.amount_cents, transferId);
    } else if (typeof transferId === "string") {
      // S3-11 (docket #17 ruling): the RETRY door emits `creator_earning` too,
      // and the event's guard is the fact's guard — `extraWhere` carries THIS
      // UPDATE's own predicate (`analytics-events.ts:400`). It is read BEFORE
      // the UPDATE in the same batch, which is the whole point: "still pending"
      // is only true of the attempt that actually makes the transition, so a
      // row a concurrent retry already paid emits nothing and the batch is not
      // failed for it. STEP 1 (:799) is the same shape on the same side of its
      // own write — one guard discipline, both doors.
      const paidAt = now();
      // #862 NOT APPLIED HERE (open, needs_decision): an outboxSpan bracket adds two statements per paid payout,
      // and runWeeklyPayouts already sits AT MAX_SUBREQUESTS_PER_RUN (test/payout-run-bounds.test.ts). Unbracketed,
      // this publish still selects every owed row (analytics-outbox.ts publishAfterCommit).
      const earn = outboxStatements(
        db,
        { kind: "creator_earning", passportId: p.passport_id, occurredDay: occurredDay(paidAt), amountCents: p.amount_cents },
        { extraWhere: { sql: STILL_PENDING, params: [p.payout_id] } },
      );
      const res = await db.batch([
        ...earn,
        markPaidStmt(db, p.payout_id, transferId, paidAt),
        // I3-0997: the creator is told, bound to the transition above.
        payoutOutcomeNoticeStatement(db, p.passport_id, { payoutId: p.payout_id, outcome: "paid", amountCents: p.amount_cents, periodKey: p.period_key }, now),
        paidAuditStmt(db, p.payout_id, transferId, p.amount_cents, p.period_key, paidAt), // #936(c)
        releaseSendStmt(db, p.payout_id),
      ]);
      if (pubEnv && pubCtx) publishAfterCommit(pubEnv, pubCtx);
      out.paid.push({ passport_id: p.passport_id, amount_cents: p.amount_cents, transfer_id: transferId });
      if (!flipLanded(res[earn.length])) {
        alarmFlipMissed(p.payout_id);
        out.paid_flip_missed = [...(out.paid_flip_missed ?? []), p.payout_id];
      }
    } else {
      // BF-13: THE REFUSAL IS WRITTEN DOWN. Without this the row is
      // indistinguishable from one that has never been tried, which is exactly
      // how it kept taking a retry slot every tick for ever. Guarded on
      // an open status (`pending` or `failed`) so a concurrent run that paid it in between is not
      // overwritten, and it never changes the amount or the reservation (the
      // status only to `failed` at PAYOUT_FAIL_ATTEMPTS, still open and still
      // retried — M-15): the money is still owed, still held, and still this
      // creator's. One extra statement, and only on a path that has already
      // spent a Stripe subrequest.
      const attempts = refusalAttempts(Number(p.attempts ?? 0), transferId);
      const nextAttemptAt = new Date(nowMs + payoutBackoffMs(attempts)).toISOString();
      // I3-0997: the notice states `failed` only if THIS write flipped it there.
      await db.batch([
        ...refusalStmts(db, p.payout_id, attempts, nextAttemptAt, nowIso),
        payoutOutcomeNoticeStatement(db, p.passport_id, { payoutId: p.payout_id, outcome: "failed", amountCents: p.amount_cents, periodKey: p.period_key }, now),
        releaseSendStmt(db, p.payout_id),
      ]);
      out.failed.push({ passport_id: p.passport_id, amount_cents: p.amount_cents });
      // I3-0062: the write above is the ONLY place this deployment learns that a
      // refusal happened, and until now it wrote two columns nothing read. Say
      // it out loud once the count crosses the stall threshold.
      if (stalls(attempts)) {
        out.stalled.push({ payout_id: p.payout_id, passport_id: p.passport_id, amount_cents: p.amount_cents, attempts, next_attempt_at: PAYOUT_RECONCILE_HOLD_AT });
      }
    }
  }

  // STEP 1: this week's NEW payouts, from the amounts the page already priced.
  // `period_rows > 0` means this week is already reserved for them (paid, or
  // pending → step 0 above), and `reserved_cents` covers every pending+paid row
  // so a still-owed amount from a prior week is never counted as owed twice.
  //
  // THE CURSOR ADVANCES WITH THIS LOOP, one creator at a time, so a run that
  // stops on a ceiling resumes at exactly the creator it did not reach — nobody
  // is skipped and nobody is re-examined for free.
  let lastExamined = cur;
  let stoppedEarly = false;
  // I19-FXA (RG-MONEY-01): reservation INSERTs that dispatched nothing (caught, or 0 rows). Each
  // is a statement, so each takes a transfer-budget slot (PAYOUT_TRANSFER_BUDGET's derivation).
  let unsent = 0;
  for (const a of actives) {
    if (a.period_rows > 0) {
      lastExamined = a.passport_id;
      out.examined += 1;
      continue;
    }
    // D-0960(1): dispute debits come off first; a debt larger than the balance
    // leaves amount 0 (no row, never a negative transfer) and stays carried in
    // `debit_cents` until later cleared earnings cover it.
    const amount = Math.max(0, a.cleared_cents - a.reserved_cents - a.debit_cents);
    if (amount < MIN_PAYOUT_CENTS) {
      out.skipped_below_min++;
      lastExamined = a.passport_id;
      out.examined += 1;
      continue;
    }
    // The ceilings are asked BEFORE the reserve, not after it: reserving a row
    // we then decline to pay would leave a `pending` row behind purely because
    // we ran out of budget, and `reserved_cents` counts pending rows — so the
    // creator's next-tick amount would be computed against money nobody has
    // sent. Stopping before the INSERT keeps the fold honest.
    if (out.transfers + unsent >= transferBudget) {
      out.stopped = "transfer_budget";
      stoppedEarly = true;
      break;
    }
    if (clock() >= deadline) {
      out.stopped = "deadline";
      stoppedEarly = true;
      break;
    }
    // I21-Q52 (D-1030): nothing is reserved for an account whose onboarding is unfinished (held, not failed); the read
    // takes this creator's slot like an unsent reservation does.
    if (!connectReady(await accountRead(cfg, a.stripe_account_id))) {
      unsent += 1;
      out.connect_unfinished = (out.connect_unfinished ?? 0) + 1;
      lastExamined = a.passport_id;
      out.examined += 1;
      continue;
    }
    const payoutId = mintPayoutId();
    if (!PAY_RE.test(payoutId)) throw new ConnectError("bad payout id", 500);
    const reservedAt = now();
    let reserved: { changes: number };
    try {
      // A-5 (docket #17 ruling): the reservation emits nothing — `creator_earning`
      // fires at TRANSFER (below), not here, so a payout that reserves but never
      // pays never counts as earned. A UNIQUE(passport, period) collision still
      // rolls this INSERT back, so a concurrent run cannot reserve it twice.
      //
      // S-070 SJ-01 — THE DESTINATION IS RE-ASKED AT THE RESERVATION, IN THE SQL.
      // `PAYOUT_PAGE_SQL` read `c.stripe_account_id` into `actives` and the
      // dispatch below spends it; the page read is not a lock, and
      // `unbindPayoutAccount` commits `status='revoked'` atomically at any
      // instant. A disconnection landing between the page read and THIS
      // reservation therefore had no effect on this tick and the money went to
      // the account the creator had just successfully disconnected — against
      // the documented behaviour that an inactive binding HOLDS earnings.
      // The predicate is the page's own `ACTIVE_BINDING`, correlated on THIS
      // row and bound to the EXACT account the page captured, so a rebind to a
      // different destination in the same window is refused here too rather
      // than silently retargeted. It is an `INSERT … SELECT … WHERE EXISTS`
      // rather than a second read because a second read leaves the same race
      // one statement narrower; this one is evaluated inside the write.
      // A WHERE that selects no row does NOT throw — `changes` is the only thing
      // that says so, read off the batch's element 0 (batchChanges). I56-MONEY
      // (D-1433(2)): the batch carries the reservation's `payout:reserved` row.
      const res = await db.batch([reservePayoutStmt(db, payoutId, a.passport_id, period, amount, reservedAt, a.stripe_account_id, bindableBefore, nowIso), reservedAuditStmt(db, payoutId, period, "weekly", reservedAt)]);
      reserved = { changes: batchChanges(res[0]) };
    } catch {
      // UNIQUE(passport, period): a concurrent run reserved it — skip.
      unsent += 1;
      lastExamined = a.passport_id;
      out.examined += 1;
      continue;
    }
    if (reserved.changes === 0) {
      // (B25-635: the INSERT also refuses when its amount is no longer
      // available at insert time — another door reserved it first; the same
      // count covers that, and nothing is reserved or paid either way.)
      // The binding moved out from under the page: revoked, re-pointed, or
      // pulled back to `pending` between the page read and this instant. NO
      // TRANSFER IS ATTEMPTED and nothing is reserved, so the earnings stay
      // cleared-and-unreserved and the next tick prices them the same way —
      // held, exactly as `held` describes for the retry door. Counted so the
      // tick SAYS it happened rather than leaving an operator to infer it from
      // a creator who is owed and never appears in `paid`, `failed` or
      // `skipped_below_min`.
      out.revoked_before_reserve += 1;
      unsent += 1;
      lastExamined = a.passport_id;
      out.examined += 1;
      continue;
    }
    // I3-0028: `spend` looks this week's transfer_group up before sending — a
    // restore that lost this row past Stripe's key window finds the transfer
    // here and records it instead of paying the week twice.
    const transferId = await spend(
      { destination: a.stripe_account_id, amountCents: amount, idempotencyKey: transferIdempotencyKey(a.passport_id, period), reconcile: reconcileFor(a.passport_id, a.ledger_json, sinceMs, { periodKey: period, amountCents: amount }) },
      transferBudget,
    );
    if (isTransferHold(transferId)) {
      await hold(payoutId, a.passport_id, period, amount, transferId);
    } else if (typeof transferId === "string") {
      const paidAt = now();
      // A-5 (docket #17 ruling): `creator_earning` rides THIS batch — the
      // transfer that actually moved money, not the reservation above — and
      // carries this UPDATE's own predicate as `extraWhere`, `STILL_PENDING`
      // (:582), read BEFORE the UPDATE exactly as STEP 0 does at :721. The
      // reservation's UNIQUE(passport_id, period_key) does not guard this
      // transition: a concurrent run's STEP 0 page (:698) selects the row THIS
      // iteration reserved, because the cursor is not written until :824.
      // #862 NOT APPLIED HERE (open, needs_decision): an outboxSpan bracket adds two statements per paid payout,
      // and runWeeklyPayouts already sits AT MAX_SUBREQUESTS_PER_RUN (test/payout-run-bounds.test.ts). Unbracketed,
      // this publish still selects every owed row (analytics-outbox.ts publishAfterCommit).
      const earn = outboxStatements(
        db,
        { kind: "creator_earning", passportId: a.passport_id, occurredDay: occurredDay(paidAt), amountCents: amount },
        { extraWhere: { sql: STILL_PENDING, params: [payoutId] } },
      );
      const res = await db.batch([
        ...earn,
        markPaidStmt(db, payoutId, transferId, paidAt),
        // I3-0997: the creator is told, bound to the transition above.
        payoutOutcomeNoticeStatement(db, a.passport_id, { payoutId, outcome: "paid", amountCents: amount, periodKey: period }, now),
        paidAuditStmt(db, payoutId, transferId, amount, period, paidAt), // #936(c)
      ]);
      if (pubEnv && pubCtx) publishAfterCommit(pubEnv, pubCtx);
      out.paid.push({ passport_id: a.passport_id, amount_cents: amount, transfer_id: transferId });
      if (!flipLanded(res[earn.length])) {
        alarmFlipMissed(payoutId);
        out.paid_flip_missed = [...(out.paid_flip_missed ?? []), payoutId];
      }
    } else {
      // Including `undefined` — the ceiling was reached between the reserve and
      // the call. The row stays `pending` with its amount and its id, which is
      // the same state a failed transfer leaves and the next tick replays.
      // I3-2131 (QUEUE #311): a DISPATCHED call that was refused (`null`) is
      // recorded exactly as STEP 0 records one, so `attempts` counts every
      // refusal and the 0095 note ("attempts = 0 means never attempted") holds.
      // `undefined` dispatched nothing, so it records nothing.
      if (transferId === null || isTransferRefusal(transferId)) {
        const n = refusalAttempts(0, transferId);
        const nextAttemptAt = new Date(nowMs + payoutBackoffMs(n)).toISOString();
        // I3-0735 b: a PERMANENT refusal flips a NEW row to `failed` at once (refusalAttempts), and STEP 0 and the
        // release door already tell the creator in the same batch; this arm did not. Bound to the flip (INV-058).
        await db.batch([
          ...refusalStmts(db, payoutId, n, nextAttemptAt, nowIso),
          payoutOutcomeNoticeStatement(db, a.passport_id, { payoutId, outcome: "failed", amountCents: amount, periodKey: period }, now),
        ]);
        if (stalls(n)) out.stalled.push({ payout_id: payoutId, passport_id: a.passport_id, amount_cents: amount, attempts: n, next_attempt_at: PAYOUT_RECONCILE_HOLD_AT });
      }
      out.failed.push({ passport_id: a.passport_id, amount_cents: amount });
    }
    lastExamined = a.passport_id;
    out.examined += 1;
  }

  // `deadline_abandoned`'s STEP 1 half: page rows this loop had not yet
  // examined when it stopped early specifically on the clock (not on
  // `transfer_budget`, which THE BOUNDS already names by binding).
  out.deadline_abandoned = step0DeadlineLeft + (stoppedEarly && out.stopped === "deadline" ? actives.length - out.examined : 0);

  // ADVANCE OR WRAP. A page that was FULL means there is more of the table, so
  // resume strictly past the last creator examined. A short page means this run
  // reached the end, so wrap to the start — unless it stopped early, in which
  // case wrapping would skip the creators it did not reach until the tick after
  // next. Best-effort: a cursor that fails to persist costs a repeated page
  // (idempotent by construction), never a missed creator.
  const nextCursor = stoppedEarly ? lastExamined : fullPage ? pageEnd : "";
  out.wrapped = nextCursor === "";
  try {
    await writeCursor(db, nextCursor, nowIso);
  } catch {
    /* a lost cursor write re-examines the same page next tick — safe, not silent */
  }
  return out;
}

/** B25-635 / D-0746, D-0750: the release's period keys. One per UTC day, so a
 * double press is refused by UNIQUE(passport_id, period_key), while money that
 * clears later can be released by the same action on a later day. */
export const RELEASE_PERIOD_PREFIX = "release-";

/** `payout_held` (I3-0028, D-0911(1)): the per-account reconcile found Stripe
 * disagreeing with our ledger; nothing was sent and the payout is held for an
 * operator (`reconcileHolds`). */
export type ReleaseRefusal = "not_configured" | "no_such_passport" | "not_suppressed" | "retracted" | "not_verified" | "account_banned" | "no_verified_destination" | "already_released_today" | "payout_held" | "connect_unfinished";

/** The Connect transfer this module spends through (`runWeeklyPayouts`'s and
 * `releaseHeldPayout`'s injectable `transfer`), exported so a caller can inject
 * one without naming anything from stripe.ts (D-0751 (1)). */
export type ConnectTransfer = typeof createConnectTransfer;

/** D-0750 (2) / D-0751 (5): ONE standing read for the release, asked when the
 * first operator files the approval (`releaseRefusal`) AND again when the
 * release executes (`releaseHeldPayout`), so both doors refuse on the same
 * predicate. `destination` is the `ACTIVE_BINDING` account, or null. */
async function releaseStanding(db: DbLike, env: Record<string, unknown>, passportId: string, nowMs: number, kind: ReleaseKind = "release"): Promise<{ refused: ReleaseRefusal | null; destination: string | null; cfg: ConnectConfig | null }> {
  const cfg = connectConfigFromEnv(env);
  if (!cfg) return { refused: "not_configured", destination: null, cfg: null };
  if (!PSP_RE.test(passportId)) return { refused: "no_such_passport", destination: null, cfg };
  const bindableBefore = new Date(nowMs - CONNECT_COOLING_OFF_MS).toISOString();
  const st = await db
    .prepare(
      `SELECT p.retracted AS retracted, p.claim_state AS claim_state,
              ${RELEASE_SUPPRESSED} AS suppressed,
              ${RELEASE_BANNED} AS banned,
              ${RELEASE_PROOF_SQL("p.passport_id")} AS proof_current,
              ${grantLiveAtSql("p.passport_id")} AS grant_live,
              (SELECT c.stripe_account_id FROM creator_connect c
                WHERE c.passport_id = p.passport_id AND ${ACTIVE_BINDING} AND ${RELEASE_OWN_DEST()}) AS destination
         FROM passports p WHERE p.passport_id = ?`,
    )
    .bind(...currentProofBinds(nowMs), new Date(nowMs).toISOString(), bindableBefore, new Date(nowMs).toISOString(), passportId)
    .first<{ retracted: number; claim_state: string; suppressed: number; banned: number; proof_current: number; grant_live: number; destination: string | null }>();
  const refused: ReleaseRefusal | null = !st
    ? "no_such_passport"
    : !st.suppressed && kind === "release"
      ? "not_suppressed"
      : st.retracted
        ? "retracted"
        : st.claim_state !== "verified" || !st.proof_current || !st.grant_live // I4-Q4: a stale proof is not a verified creator here either; I3-1768: nor a lapsed grant
          ? "not_verified"
          : st.banned
          ? "account_banned"
          : !st.destination
            ? "no_verified_destination"
            : null;
  return { refused, destination: refused ? null : (st?.destination ?? null), cfg };
}

/** The filing-time half of D-0751 (5): the refusal a release would meet now,
 * or null. The console asks it before queueing a first approval. */
export async function releaseRefusal(db: DbLike, env: Record<string, unknown>, passportId: string, nowMs: number): Promise<ReleaseRefusal | null> {
  return (await releaseStanding(db, env, passportId, nowMs)).refused;
}

export interface ReleaseResult {
  refused: ReleaseRefusal | null;
  /** Transfers that landed: replayed pending rows first, then the new release row. */
  paid: { payout_id: string; period_key: string; amount_cents: number; transfer_id: string }[];
  /** Rows reserved but not paid (transfer refused); each stays `pending` and the next press replays it under the same key. */
  failed: { payout_id: string; period_key: string; amount_cents: number }[];
}

/**
 * B25-635 — RELEASE HELD BALANCE (D-0746, details D-0750). The OPERATOR lever
 * for a creator whose suppression keeps them off every weekly payout page
 * (`STANDING_EXISTS`). It pays, once, the money the weekly run would have paid
 * had the passport been standing, through the weekly run's own statements
 * (`reservePayoutStmt`, `claimSendStmt`, `markPaidStmt`, `recordRefusalStmt`,
 * `transferIdempotencyKey`) to the same destination predicate (`ACTIVE_BINDING`).
 * The suppression is not touched.
 *
 * `env` is read for the Connect config here, as the weekly run's caller reads it,
 * so no router names a stripe.ts symbol (D-0751 (1)).
 *
 * REFUSES (D-0750 (2), via `releaseStanding`): an unsuppressed passport (the weekly run's business), a
 * retracted one, one whose account is banned — `accounts.banned_at` or a
 * suspended `account_closures` row (a TERMINATED one is paid, to its own bound
 * account only: `RELEASE_OWN_DEST`), reached through `founder`, the
 * only account-to-passport identity limb the schema carries — and one with no
 * `ACTIVE_BINDING` (active, passkey-activated, past the cooling-off).
 *
 * WHAT IT PAYS. (1) Every `pending` row for the passport, replayed under its
 * ORIGINAL id and key (STEP 0 skips a suppressed passport, so these are held
 * too); a refused replay stops the release so no new row piles behind it.
 * (2) `cleared − reserved` (the page's own arithmetic): money inside the 14-day
 * hold is not cleared and stays. NO `MIN_PAYOUT_CENTS` (D-0750 (1)). Zero -> no
 * row, no transfer: a second release pays nothing more.
 */
export async function releaseHeldPayout(
  db: DbLike,
  env: Record<string, unknown>,
  passportId: string,
  nowMs: number,
  mintPayoutId: () => string,
  now: () => string = () => new Date().toISOString(),
  transfer: ConnectTransfer = createConnectTransfer,
  pubEnv?: AnalyticsTickEnv,
  pubCtx?: { waitUntil(p: Promise<unknown>): void },
  accountState?: ConnectAccountRead,
): Promise<ReleaseResult> {
  return payThroughRelease("release", db, env, passportId, nowMs, mintPayoutId, now, transfer, pubEnv, pubCtx, accountState);
}

/**
 * R3-I29-Q9 (D-1091(4)): an erasing creator's cleared residual under MIN_PAYOUT_CENTS, paid ONCE with no minimum through
 * releaseHeldPayout's own statements and refusals (`payThroughRelease`), except that the passport need not be
 * suppressed. Called by creator-erasure.ts payResidual only (eraseCreator, and sweepCreatorWipes given `pay`); a refusal or
 * a failed transfer pays nothing and the money stays on the record. `pubEnv`/`pubCtx` publish the payout row's analytics
 * after commit, as releaseHeldPayout's do (B23-A8).
 */
export async function payErasureResidual(
  db: DbLike,
  env: Record<string, unknown>,
  passportId: string,
  nowMs: number,
  mintPayoutId: () => string,
  now: () => string = () => new Date().toISOString(),
  transfer: ConnectTransfer = createConnectTransfer,
  pubEnv?: AnalyticsTickEnv,
  pubCtx?: { waitUntil(p: Promise<unknown>): void },
  accountState?: ConnectAccountRead,
  accountFetch?: typeof fetch,
): Promise<ReleaseResult> {
  return payThroughRelease("erasure", db, env, passportId, nowMs, mintPayoutId, now, transfer, pubEnv, pubCtx, accountState, accountFetch);
}

async function payThroughRelease(
  kind: ReleaseKind,
  db: DbLike,
  env: Record<string, unknown>,
  passportId: string,
  nowMs: number,
  mintPayoutId: () => string,
  now: () => string = () => new Date().toISOString(),
  transfer: ConnectTransfer = createConnectTransfer,
  pubEnv?: AnalyticsTickEnv,
  pubCtx?: { waitUntil(p: Promise<unknown>): void },
  accountState?: ConnectAccountRead,
  /** D-1127(5): the fetch the Connect account read is sent on (`accountReadFor`); I38-TICK-3: and both transfers below are
   * handed it as `transfer`'s `fetchImpl` (createConnectTransfer's default, the global fetch, when absent). */
  accountFetch?: typeof fetch,
): Promise<ReleaseResult> {
  const out: ReleaseResult = { refused: null, paid: [], failed: [] };
  const standing = await releaseStanding(db, env, passportId, nowMs, kind);
  const cfg = standing.cfg;
  const dest = standing.destination;
  if (standing.refused || !cfg || !dest) return { ...out, refused: standing.refused ?? "no_verified_destination" };
  // I21-Q52 (D-1030): onboarding finished before any transfer, at this door too; nothing is reserved or sent.
  if (!connectReady(await accountReadFor(transfer, accountState, accountFetch)(cfg, dest))) return { ...out, refused: "connect_unfinished" };
  const nowIso = new Date(nowMs).toISOString();
  const bindableBefore = new Date(nowMs - CONNECT_COOLING_OFF_MS).toISOString();

  const settle = async (payoutId: string, amountCents: number, transferId: string, periodKey: string): Promise<void> => {
    const paidAt = now();
    const span = outboxSpan(db);
    const earn = outboxStatements(
      db,
      { kind: "creator_earning", passportId, occurredDay: occurredDay(paidAt), amountCents },
      { extraWhere: { sql: STILL_PENDING, params: [payoutId] } },
    );
    const spanRes = await db.batch([
      span.open,
      ...earn,
      markPaidStmt(db, payoutId, transferId, paidAt),
      // I3-0997: the creator is told, bound to the transition above.
      payoutOutcomeNoticeStatement(db, passportId, { payoutId, outcome: "paid", amountCents, periodKey }, now),
      paidAuditStmt(db, payoutId, transferId, amountCents, periodKey, paidAt), // #936(c)
      releaseSendStmt(db, payoutId),
      span.close,
    ]);
    if (!flipLanded(spanRes[1 + earn.length])) alarmFlipMissed(payoutId); // D-0931(3)
    const seqs = outboxSpanSeqs(spanRes[0], spanRes[spanRes.length - 1]);
    if (pubEnv && pubCtx) publishAfterCommit(pubEnv, pubCtx, seqs);
  };

  // I3-0028 (D-0911(1)): EVERY send below carries the per-account reconcile —
  // the NEW release send too, which once sent with no lookup at all under a
  // per-DAY key (I11-RG-MONEY RGM-P4). The ledger is read fresh before each
  // send (one statement; this is an operator route, not the tick's budget).
  const { sinceMs, ledgerSinceIso } = reconcileWindow(nowMs);
  const reconcile = async (periodKey: string, amountCents: number): Promise<TransferReconcile> => {
    const row = await db.prepare(`SELECT ${ledgerJsonSql("rp.pid")} AS ledger_json FROM (SELECT ? AS pid) rp`).bind(ledgerSinceIso, passportId).first<{ ledger_json: string | null }>();
    return reconcileFor(passportId, row?.ledger_json ?? null, sinceMs, { periodKey, amountCents });
  };
  const hold = async (payoutId: string, periodKey: string, amountCents: number, h: TransferHold): Promise<ReleaseResult> => {
    await db.batch(holdStmts(db, payoutId, passportId, periodKey, amountCents, h, now()));
    return { ...out, refused: "payout_held" };
  };

  // (1) Replay what is already reserved — except a row already HELD for an
  // operator (D-0911(1)), which only an operator moves.
  const { results: pendings } = await db
    .prepare(`SELECT payout_id, period_key, amount_cents, attempts FROM creator_payouts WHERE passport_id = ? AND (status = 'pending' OR status = 'failed') AND (next_attempt_at IS NULL OR next_attempt_at <> ?) ORDER BY created_at ASC, payout_id ASC`)
    .bind(passportId, PAYOUT_RECONCILE_HOLD_AT)
    .all<{ payout_id: string; period_key: string; amount_cents: number; attempts: number }>();
  for (const p of pendings) {
    // I16-RG-MONEY:16: the re-ask is the send claim (claimSendStmt); a claim another door holds reads as not sendable now.
    if ((await claimSendStmt(db, p.payout_id, passportId, dest, bindableBefore, nowMs).all()).results.length !== 1) return { ...out, refused: "no_verified_destination" };
    // I3-1132: a replay may be past Stripe's 24 h key window — the reconcile covers it.
    const transferId = await transfer(cfg, { destination: dest, amountCents: p.amount_cents, idempotencyKey: transferIdempotencyKey(passportId, p.period_key), reconcile: await reconcile(p.period_key, p.amount_cents) }, accountFetch);
    if (isTransferHold(transferId)) return hold(p.payout_id, p.period_key, p.amount_cents, transferId);
    if (typeof transferId !== "string") {
      const attempts = refusalAttempts(Number(p.attempts ?? 0), transferId);
      // I3-0997: the notice states `failed` only if THIS write flipped it there.
      await db.batch([
        ...refusalStmts(db, p.payout_id, attempts, new Date(nowMs + payoutBackoffMs(attempts)).toISOString(), nowIso),
        payoutOutcomeNoticeStatement(db, passportId, { payoutId: p.payout_id, outcome: "failed", amountCents: p.amount_cents, periodKey: p.period_key }, now),
        releaseSendStmt(db, p.payout_id),
      ]);
      out.failed.push({ payout_id: p.payout_id, period_key: p.period_key, amount_cents: p.amount_cents });
      return out;
    }
    await settle(p.payout_id, p.amount_cents, transferId, p.period_key);
    out.paid.push({ payout_id: p.payout_id, period_key: p.period_key, amount_cents: p.amount_cents, transfer_id: transferId });
  }

  // (2) The cleared, unreserved remainder — computed BY the reserving INSERT
  // (RG-635-01/02), never by a read before it. No row: either nothing is
  // available (a second release, or another door reserved it first) or the
  // standing moved; the standing is re-asked to say which.
  const payoutId = mintPayoutId();
  if (!PAY_RE.test(payoutId)) throw new ConnectError("bad payout id", 500);
  const period = `${RELEASE_PERIOD_PREFIX}${nowIso.slice(0, 10)}`;
  let reserved: { changes: number };
  try {
    const reservedAt = now();
    const res = await db.batch([reserveReleaseStmt(db, payoutId, passportId, period, reservedAt, dest, bindableBefore, nowIso, kind), reservedAuditStmt(db, payoutId, period, kind, reservedAt)]);
    reserved = { changes: batchChanges(res[0]) };
  } catch (e) {
    if (/UNIQUE/i.test(e instanceof Error ? e.message : String(e))) return { ...out, refused: "already_released_today" };
    throw e;
  }
  if (reserved.changes === 0) {
    const again = await releaseStanding(db, env, passportId, nowMs, kind);
    return again.refused ? { ...out, refused: again.refused } : out;
  }
  const amount = (await db.prepare("SELECT amount_cents FROM creator_payouts WHERE payout_id = ?").bind(payoutId).first<{ amount_cents: number }>())?.amount_cents ?? 0;
  const transferId = await transfer(cfg, { destination: dest, amountCents: amount, idempotencyKey: transferIdempotencyKey(passportId, period), reconcile: await reconcile(period, amount) }, accountFetch);
  if (isTransferHold(transferId)) return hold(payoutId, period, amount, transferId);
  if (typeof transferId !== "string") {
    const n = refusalAttempts(0, transferId);
    await db.batch(refusalStmts(db, payoutId, n, new Date(nowMs + payoutBackoffMs(n)).toISOString(), nowIso));
    out.failed.push({ payout_id: payoutId, period_key: period, amount_cents: amount });
    return out;
  }
  await settle(payoutId, amount, transferId, period);
  out.paid.push({ payout_id: payoutId, period_key: period, amount_cents: amount, transfer_id: transferId });
  return out;
}

/** The one writer of this sweep's cursor row (same upsert shape as
 * verification-sweep.ts). `cursor_seq` stays 0: this scan's total order is the
 * `creator_connect` primary key, and there is no sequence to carry. */
async function writeCursor(db: DbLike, cursorId: string, nowIso: string): Promise<void> {
  await db
    .prepare("INSERT INTO sweep_cursor (name, cursor_seq, cursor_id, updated_at) VALUES (?, 0, ?, ?) ON CONFLICT(name) DO UPDATE SET cursor_id = excluded.cursor_id, updated_at = excluded.updated_at")
    .bind(PAYOUT_CURSOR, cursorId, nowIso)
    .run();
}

/** I3-0062: rows one `stalledPayouts` read returns. Bounded like every other
 * read in this module — an operator page, never a full scan. */
export const STALLED_PAYOUT_LIMIT = 100;

/**
 * I3-0062 — THE OPERATOR READ OVER THE BACKOFF COLUMNS.
 *
 * `runWeeklyPayouts` reports the rows IT re-attempted this tick (`stalled`),
 * which is a tick-shaped view: a row whose backoff has grown to a week appears
 * in at most one run in seven days and in none of the others. This is the
 * standing population — every pending payout whose transfer has been refused at
 * least `minAttempts` times — so a paging surface or an alert leg can ask "who
 * is owed money that has stopped moving" without owning this table's shape.
 *
 * Own-file, read-only, bounded, and NOT called from the run (it would spend a
 * statement on every tick against a pinned per-tick statement count). Its
 * CALLERS — the operator console and the alert fan-out — are outside this
 * room's ownership; the exact wiring is handed back in the report.
 */
export async function stalledPayouts(
  db: DbLike,
  opts: { minAttempts?: number; limit?: number } = {},
): Promise<{ payout_id: string; passport_id: string; amount_cents: number; attempts: number; next_attempt_at: string | null; created_at: string }[]> {
  const minAttempts = Math.max(1, Math.floor(opts.minAttempts ?? PAYOUT_STALL_ATTEMPTS));
  const limit = Math.max(1, Math.min(STALLED_PAYOUT_LIMIT, Math.floor(opts.limit ?? STALLED_PAYOUT_LIMIT)));
  const { results } = await db
    .prepare(
      `SELECT payout_id, passport_id, amount_cents, attempts, next_attempt_at, created_at
         FROM creator_payouts
        WHERE (status = 'pending' OR status = 'failed') AND COALESCE(attempts, 0) >= ?
        ORDER BY attempts DESC, created_at ASC
        LIMIT ?`,
    )
    .bind(minAttempts, limit)
    .all<{ payout_id: string; passport_id: string; amount_cents: number; attempts: number; next_attempt_at: string | null; created_at: string }>();
  return results;
}

/**

/**
 * I3-0028 b (D-0993(1), the census pick: an operator clears a held payout by VOID — `voidPayout`, which exists — or
 * by RE-ARM, this). RE-ARM puts a row the per-account reconcile parked (`next_attempt_at = PAYOUT_RECONCILE_HOLD_AT`)
 * back in STEP 0's queue (`next_attempt_at = now`) after an operator has looked at Stripe. It changes NOTHING else:
 * not the amount, not the status, not the key, and it records no Stripe figure — so the next send carries the same
 * reconcile every send carries (`reconcileFor`), and a mismatch that is still there holds the row again (D-0911(1):
 * never record Stripe's amount as ours). ONE batch: the flip, guarded on the row being open AND held, then the
 * `payout:rearm` audit row bound to THIS call's instant (INV-058). A row that is not held, paid, voided or absent is
 * not re-armed and writes nothing. The caller (console.ts) is the dual-control door (hub hand-off).
 */
export async function rearmHeldPayout(
  db: DbLike,
  payoutId: string,
  operator: string,
  input: { reason: unknown },
  now: () => string = () => new Date().toISOString(),
): Promise<{ rearmed: boolean; status: string | null }> {
  if (typeof payoutId !== "string" || !PAY_RE.test(payoutId)) throw new ConnectError("no such payout", 404);
  const reason = typeof input.reason === "string" ? input.reason.trim() : "";
  if (!rearmReasonValid(reason)) throw new ConnectError(REARM_REASON_REFUSAL, 400);
  const ts = now();
  await db.batch([
    db
      .prepare(`UPDATE creator_payouts SET next_attempt_at = ? WHERE payout_id = ? AND (status = 'pending' OR status = 'failed') AND next_attempt_at = ?`)
      .bind(ts, payoutId, PAYOUT_RECONCILE_HOLD_AT),
    db
      .prepare(
        `INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind)
              SELECT ?, 'payout:rearm', ?, ?, ?, 'operator'
               WHERE EXISTS (SELECT 1 FROM creator_payouts WHERE payout_id = ? AND next_attempt_at = ? AND (status = 'pending' OR status = 'failed'))`,
      )
      .bind(operator, `payout ${payoutId} re-armed after an operator check: ${reason}`.slice(0, 500), ts, payoutId, payoutId, ts),
  ]);
  const row = await db.prepare("SELECT status, next_attempt_at FROM creator_payouts WHERE payout_id = ?").bind(payoutId).first<{ status: string; next_attempt_at: string | null }>();
  if (!row) throw new ConnectError("no such payout", 404);
  return { rearmed: (row.status === "pending" || row.status === "failed") && row.next_attempt_at === ts, status: row.status };
}

export const PAYOUT_HISTORY_LIMIT = 1000;

// transfer's success arm, `status='paid', ..., paid_at=?`) is when it was paid, null until then.
/**
 * I3-0251 (CUR:M-13) — VOID A STUCK PAYOUT. An open payout row (`pending`/`failed`) that an operator has settled
 * by hand outside Stripe is closed as `voided` (0099 added the status and its two columns), so neither STEP 0 nor
 * `releaseHeldPayout` — both select only `status = 'pending' OR status = 'failed'` — will ever send it again, and
 * the amount stays reserved (`RESERVED_PAYOUT_STATUS_SQL`), so the creator's withdrawable earnings do not rise.
 * ONE batch: the flip, guarded by the open-state predicate, then the `payout:void` audit row bound to the flip it
 * follows (INV-058: it lands only if THIS call's `voided_at` is on a voided row). A second void, or a void of a
 * paid row, changes nothing and writes nothing. The caller (console.ts) is the dual-control door.
 * I17 J5a (RG-MONEY:6, D-0938(1a)): THE VOID ENTERS THE CASH REGISTER. `amountCents` is the operator-stated amount paid
 * outside Stripe (a positive integer, at most the row's own `amount_cents`, gated in the flip's WHERE); the same batch
 * writes one `cash_outflows` row (kind 'payout', rail 'manual', external_ref = the operator's reference, source_ref = the
 * payout id) under the audit row's own THIS-call predicate, so a refused or repeated void writes none.
 * `closures.ts reconcileCashOutflows` balances a voided payout against that row.
 */
export async function voidPayout(
  db: DbLike,
  payoutId: string,
  operator: string,
  input: { reason: unknown; externalRef: unknown; amountCents: unknown },
  now: () => string = () => new Date().toISOString(),
): Promise<{ voided: boolean; status: string | null; in_flight?: true }> {
  if (typeof payoutId !== "string" || !PAY_RE.test(payoutId)) throw new ConnectError("no such payout", 404);
  const reason = typeof input.reason === "string" ? input.reason.trim() : "";
  const ref = typeof input.externalRef === "string" ? input.externalRef.trim() : "";
  if (reason.length < 3 || reason.length > 200 || !hasVisibleText(reason)) throw new ConnectError("a void needs a reason (3-200 characters)", 400);
  if (ref.length < 1 || ref.length > 120 || !hasVisibleText(ref)) throw new ConnectError("a void needs the external payment reference the creator was paid under (1-120 characters)", 400);
  const cents = input.amountCents;
  if (typeof cents !== "number" || !Number.isSafeInteger(cents) || cents <= 0) throw new ConnectError("a void needs the amount paid outside Stripe, in cents (a positive whole number)", 400);
  const ts = now();
  // I16-RG-MONEY:16 / D-0931(3): never void a row a payout door is SENDING (SEND_IN_FLIGHT) — a transfer could land
  // after the void and leave a sent payment on a `voided` row. Refused, not queued: `in_flight` tells the route (409).
  const tsMs = Date.parse(ts);
  const claimCutoff = new Date((Number.isFinite(tsMs) ? tsMs : Date.now()) - PAYOUT_SEND_CLAIM_MS).toISOString();
  await db.batch([
    db
      .prepare(`UPDATE creator_payouts SET status = 'voided', voided_at = ?, void_reason = ? WHERE payout_id = ? AND (status = 'pending' OR status = 'failed') AND amount_cents >= ? AND NOT ${SEND_IN_FLIGHT}`)
      .bind(ts, reason, payoutId, cents, ts, claimCutoff),
    db
      .prepare(
        `INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind)
              SELECT ?, 'payout:void', ?, ?, ?, 'operator'
               WHERE EXISTS (SELECT 1 FROM creator_payouts WHERE payout_id = ? AND status = 'voided' AND voided_at = ?)`,
      )
      .bind(operator, `payout ${payoutId} voided: settled outside Stripe, ref ${ref}, ${cents} cents; ${reason}`.slice(0, 500), ts, payoutId, payoutId, ts),
    // I17 J5a: the money that left, in the register (0197), bound to THIS call's flip exactly as the audit row is.
    db
      .prepare(
        `INSERT INTO cash_outflows (source_ref, kind, rail, external_ref, cents, recorded_at)
              SELECT payout_id, 'payout', 'manual', ?, ?, ? FROM creator_payouts WHERE payout_id = ? AND status = 'voided' AND voided_at = ?`,
      )
      .bind(ref, cents, ts, payoutId, ts),
  ]);
  // THIS call voided it iff the row carries THIS call's instant (a prior void carries its own).
  const row = await db.prepare("SELECT status, voided_at, amount_cents FROM creator_payouts WHERE payout_id = ?").bind(payoutId).first<{ status: string; voided_at: string | null; amount_cents: number }>();
  if (!row) throw new ConnectError("no such payout", 404);
  const voided = row.status === "voided" && row.voided_at === ts;
  // Still open after a guarded flip that did not land: the stated amount above the payout's own, else the send claim.
  if (!voided && (row.status === "pending" || row.status === "failed")) {
    if (cents > row.amount_cents) throw new ConnectError(`the stated amount (${cents} cents) is above this payout's own amount (${row.amount_cents} cents)`, 400);
    return { voided: false, status: row.status, in_flight: true };
  }
  return { voided, status: row.status };
}

export async function payoutHistory(db: DbLike, passportId: string): Promise<{ period_key: string; amount_cents: number; status: string; created_at: string; paid_at: string | null }[]> {
  if (!PSP_RE.test(passportId)) return [];
  const { results } = await db

    .bind(passportId, PAYOUT_HISTORY_LIMIT)
    .all<{ period_key: string; amount_cents: number; status: string; created_at: string; paid_at: string | null }>();

}

/** The pure half of the link screen. */
export function screenLinkShape(raw: unknown): ScreenedLink {
  if (typeof raw !== "string") throw new ProfileError("link must be a string", 400, "link_invalid");
  const trimmed = raw.trim();
  if (trimmed.length === 0 || trimmed.length > LINK_MAX_CHARS) throw new ProfileError(`link must be 1-${LINK_MAX_CHARS} characters`, 400, "link_invalid");
  let u: URL;
  try {
    u = new URL(trimmed);
  } catch {
    throw new ProfileError("link is not a URL", 400, "link_invalid");
  }
  if (u.protocol !== "https:") throw new ProfileError("links must start with https://", 400, "link_not_https");
  if (u.username !== "" || u.password !== "") throw new ProfileError("links can't carry a user name or password", 400, "link_credentials");
  if (u.port !== "") throw new ProfileError("links can't name a port", 400, "link_port");
  const host = u.hostname.toLowerCase().replace(/\.$/, "");
  if (!host || !isFetchableWebHost(host)) throw new ProfileError("links must point at a public named site", 400, "link_host");
  if (SHORTENER_HOSTS.has(host)) throw new ProfileError("short links hide where they go; use the full address", 400, "link_shortener");
  const url = u.href;
  const subjects = new Set<string>();
  let knownPlatform = false;
  try {
    const c = canonicalize(url);
    subjects.add(c.canonical);
    knownPlatform = c.platform !== "link" && c.platform !== "url";
  } catch {
    // an unclassifiable URL still gets the domain subject below
  }
  const reg = registrable(host);
  try {
    subjects.add(canonicalAccount("domain", "domain", reg?.domain ?? host).canonical);
  } catch {
    // a host canonicalAccount refuses has no domain subject
  }
  return { url, host, subjects: [...subjects], knownPlatform };
}

/** The live-warning predicate the account-added door uses (events.ts, the
 * `warned` arm): a `warning_published` of a blocking class on the subject's
 * hashed key, not followed by a `warning_retracted`. Returns the subjects that
 * are warned. */
async function warnedSubjects(db: DbLike, subjects: readonly string[]): Promise<Set<string>> {
  const out = new Set<string>();
  const ph = BLOCKS_ACCOUNT_ADDED.map(() => "?").join(", ");
  for (const s of subjects) {
    const key = `sha256:${await subjectHash(s)}`; // single-hasher: a SUBJECT hashes through subjectHash (I23-L02)
    const r = await db
      .prepare(
        `SELECT EXISTS (SELECT 1 FROM trust_events w
                         WHERE w.subject = ? AND w.event_type = 'warning_published'
                           AND json_extract(w.payload, '$.warning_class') IN (${ph})
                           AND w.seq > COALESCE((SELECT MAX(r.seq) FROM trust_events r
                                                  WHERE r.subject = ? AND r.event_type = 'warning_retracted'), -1)) AS warned`,
      )
      .bind(key, ...BLOCKS_ACCOUNT_ADDED, key)
      .first<{ warned: number }>();
    if (r?.warned) out.add(s);
  }
  return out;
}

/** The full link screen for a write. */
export async function screenLink(db: DbLike, passportId: string, raw: unknown): Promise<string> {
  const s = screenLinkShape(raw);
  if ((await warnedSubjects(db, s.subjects)).size > 0) throw new ProfileError("that site carries a Dhwaja warning and can't be linked", 409, "link_warned");
  // N2 (I23-RG-SEC, I23-FX-TC): a known-platform link is an ACCOUNT, and the ledger may register that account to
  // another passport. The ownership gate the non-platform branch gets from domainAddGate is, for an account, the
  // roster itself: an active passport_accounts row of ANOTHER passport refuses the link. subjects[0] is the
  // canonical account key (canonicalize), the key passport_accounts stores. Only a PUBLICLY listed row counts —
  // not retracted, not suppressed, not shielded, the same legs as social.ts creatorHandleCollision — so the refusal
  // is never an oracle for a hidden listing (tests/invariants/suppression-oracles.test.ts drives this door). ONE
  // statement in every world.
  if (s.knownPlatform && (await linkListedOnOther(db, passportId, s))) {
    throw new ProfileError("that account is listed on another creator's passport; it can't be linked here", 409, "link_other_creator");
  }
  if (!s.knownPlatform) {
    const gate = await domainAddGate(db, s.host, passportId);
    if (gate.blocked) {
      if (gate.reason === "resembles") throw new ProfileError("that site looks like another creator's or brand's; it can't be linked", 409, "link_lookalike");
      if (gate.reason === "candidate_cap") throw new ProfileError("we couldn't finish checking that site; try again later", 503, "link_screen_unavailable");
      throw new ProfileError("links must point at a public named site", 400, "link_host");
    }
  }
  return s.url;
}

/** Write one field (value null = clear) and its history row in ONE batch. */
async function writeField(db: DbLike, passportId: string, field: ProfileField, value: string | null, now: string): Promise<void> {

    db
      .prepare(

           ON CONFLICT(passport_id) DO UPDATE SET ${col} = excluded.${col}, updated_at = excluded.updated_at`,
      )
      .bind(passportId, stored, now),
    db
      .prepare("INSERT INTO creator_profile_history (passport_id, field, action, value, actor, reason, created_at) VALUES (?, ?, ?, ?, 'creator', NULL, ?)")
      .bind(passportId, field, action, action === "clear" ? null : stored, now),
  ]);
}

export async function setLinks(db: DbLike, passportId: string, raw: unknown, now: () => string = () => new Date().toISOString()): Promise<string[]> {
  if (!Array.isArray(raw)) throw new ProfileError("links must be a list", 400, "links_invalid");
  if (raw.length > MAX_LINKS) throw new ProfileError(`at most ${MAX_LINKS} links`, 400, "links_too_many");
  const out: string[] = [];
  for (const r of raw) {
    const url = await screenLink(db, passportId, r);
    if (out.includes(url)) throw new ProfileError("the same link is listed twice", 400, "links_duplicate");
    out.push(url);
  }
  await writeField(db, passportId, "links", JSON.stringify(out), now());
  return out;
}

/** A passport whose creator-written profile may be served: exists, not retracted, not suppressed, and still
 * CLAIMED — I3-0220 (D-0995(7)): a reversed claim (`downgrade_claim` returns the passport to `unclaimed`) hides the
 * profile its claimant wrote, the same as retraction; a later claim shows it again. One answer for every refusal
 * (no existence oracle). */
async function servablePassport(db: DbLike, passportId: string): Promise<boolean> {
  if (!PASSPORT_ID_RE.test(passportId)) return false;
  const r = await db
    .prepare(
      `SELECT 1 AS ok FROM passports p WHERE p.passport_id = ? AND p.retracted = 0 AND p.claim_state IN ('claimed','verified')
         AND NOT EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = p.passport_id)`,
    )
    .bind(passportId)
    .first<{ ok: number }>();
  return !!r;
}

/** The public projection, or null when the passport is not servable. Hidden
 * fields are absent; a link now warned is absent; a known-platform link now listed
 * on another passport is absent (RJ-02 N2, `linkListedOnOther`); an avatar the media
 * read no longer serves is absent. */
export async function publicCreatorProfile(db: DbLike, passportId: string): Promise<PublicProfile | null> {
  if (!(await servablePassport(db, passportId))) return null;
  const row = await readRow(db, passportId);
  const hidden = await hiddenFields(db, passportId);
  const out: PublicProfile = { bio: null, links: [], avatar: null };
  if (await creatorSelfClosed(db, passportId)) out.closed = true;
  if (!row) return out;
  if (!hidden.has("bio") && row.bio) out.bio = row.bio;
  if (!hidden.has("links")) {
    const links = parseLinks(row.links);
    const warned = await warnedLinkUrls(db, links);
    for (const url of links) {
      if (warned.has(url)) continue;
      const shape = screenLinkShape(url); // passed above: a shape failure is in `warned`
      if (shape.knownPlatform && (await linkListedOnOther(db, passportId, shape))) continue; // RJ-02 N2: now listed on another passport
      out.links.push({ url, host: new URL(url).hostname });
    }
  }
  if (!hidden.has("avatar") && row.avatar_hash && (await getMediaAsset(db, row.avatar_hash))) {
    out.avatar = { hash: row.avatar_hash, path: `/media/${row.avatar_hash}` };
  }
  return out;
}

function escHtml(s: string): string {

}

/** The SSR fragment for the passport page. Every value escaped; every link
 * `rel="nofollow ugc noopener noreferrer"`. The avatar is a same-origin
 * `/media/<hash>` `<img>`, which `/p/:id`'s PASSPORT_PAGE_CSP (`img-src 'self'`,

  if (!p || (!p.bio && p.links.length === 0 && !p.avatar)) return "";

/** Operator hide/restore of one field. Idempotent per state; every change
 * writes a history row in the same batch. A restore of a hidden field also sets
 * the field's keep mark (`creator_profiles.<field>_kept_at`, D-1065(4)), the
 * durable record that its child_safety reports were reviewed and kept. Returns
 * whether the state changed. */
export async function moderateProfileField(
  db: DbLike,
  passportId: string,
  field: unknown,
  action: "hide" | "restore",
  actor: string,
  reason: string | null,
  now: () => string = () => new Date().toISOString(),
): Promise<{ changed: boolean }> {
  if (!PASSPORT_ID_RE.test(passportId)) throw new ProfileError("no such passport", 404, "not_found");
  if (!isField(field)) throw new ProfileError("unknown field", 400, "field_invalid");
  if (typeof actor !== "string" || actor.length === 0) throw new ProfileError("actor required", 400, "actor_required");
  const exists = await db.prepare("SELECT 1 AS x FROM passports WHERE passport_id = ?").bind(passportId).first();
  if (!exists) throw new ProfileError("no such passport", 404, "not_found");
  const why = typeof reason === "string" ? reason.slice(0, 500) : null;
  const ts = now();
  // History first, conditioned on the state it is about to change, then the
  // change itself: one batch, so the record and the state cannot diverge, and a
  // no-op (already hidden / already shown) writes neither.
  const hiddenNow = "EXISTS (SELECT 1 FROM creator_profile_moderation WHERE passport_id = ? AND field = ?)";
  const stmts =
    action === "hide"
      ? [
          db.prepare(`INSERT INTO creator_profile_history (passport_id, field, action, value, actor, reason, created_at) SELECT ?, ?, 'hide', NULL, ?, ?, ? WHERE NOT ${hiddenNow}`).bind(passportId, field, actor, why, ts, passportId, field),
          db.prepare("INSERT INTO creator_profile_moderation (passport_id, field, hidden_by, reason, hidden_at) VALUES (?, ?, ?, ?, ?) ON CONFLICT(passport_id, field) DO NOTHING").bind(passportId, field, actor, why, ts),
        ]
      : [
          db.prepare(`INSERT INTO creator_profile_history (passport_id, field, action, value, actor, reason, created_at) SELECT ?, ?, 'restore', NULL, ?, ?, ? WHERE ${hiddenNow}`).bind(passportId, field, actor, why, ts, passportId, field),
          db.prepare(`UPDATE creator_profiles SET ${KEPT_COLUMN[field]} = ? WHERE passport_id = ? AND ${hiddenNow}`).bind(ts, passportId, passportId, field),
          db.prepare("DELETE FROM creator_profile_moderation WHERE passport_id = ? AND field = ?").bind(passportId, field),
        ];
  const res = (await db.batch(stmts)) as Array<{ changes?: number; meta?: { changes?: number } } | undefined>;
  // The state change is the LAST statement of either branch (the moderation INSERT or DELETE).
  const last = res[stmts.length - 1];
  const n = last?.changes ?? last?.meta?.changes ?? 0;
  return { changed: n > 0 };
}

export const PROFILE_CHILD_SAFETY_LIST_MAX = 50;

export interface ProfileChildSafetyGroup { passport_id: string; field: ProfileField; reports: number; first_at: string; last_at: string }

/**
 * D-1064(1) (I30-RJ-02 / I30-FX-SAFETY-06): child_safety reports filed through the profile door, routed to the
 * operator above the fan-report queue (apps/admin lists them first). One group per (passport, field) holding
 * reports that are still UNHANDLED: the field is shown (no `creator_profile_moderation` row) and the field's keep
 * mark (`creator_profiles.<field>_kept_at`, set by an operator restore of the hidden field — D-1065(4), migration
 * 0263) is older than the report or unset. A creator's own set/clear never handles it. Oldest first; `truncated`
 * when more than PROFILE_CHILD_SAFETY_LIST_MAX groups wait. The mark is not pruned with creator_profile_history, so
 * a kept report stays handled past CREATOR_PROFILE_HISTORY_RETENTION_DAYS; a report filed after it lists again.
 */
export async function profileChildSafetyQueue(db: DbLike, limit: number = PROFILE_CHILD_SAFETY_LIST_MAX): Promise<{ items: ProfileChildSafetyGroup[]; truncated: boolean }> {
  const max = Math.max(1, Math.min(PROFILE_CHILD_SAFETY_LIST_MAX, Math.trunc(limit) || PROFILE_CHILD_SAFETY_LIST_MAX));
  const { results } = await db
    .prepare(
      `SELECT r.passport_id AS passport_id, r.field AS field, COUNT(*) AS reports, MIN(r.created_at) AS first_at, MAX(r.created_at) AS last_at
         FROM creator_profile_reports r
        WHERE r.category = 'child_safety'
          AND NOT EXISTS (SELECT 1 FROM creator_profile_moderation m WHERE m.passport_id = r.passport_id AND m.field = r.field)
          AND NOT EXISTS (
                SELECT 1 FROM creator_profiles p
                 WHERE p.passport_id = r.passport_id
                   AND (CASE r.field WHEN 'bio' THEN p.bio_kept_at WHEN 'links' THEN p.links_kept_at WHEN 'avatar' THEN p.avatar_kept_at END) >= r.created_at)
        GROUP BY r.passport_id, r.field
        ORDER BY first_at ASC, r.passport_id ASC, r.field ASC
        LIMIT ?`,
    )
    .bind(max + 1)
    .all<ProfileChildSafetyGroup>();
  return { items: results.slice(0, max).map((r) => ({ passport_id: r.passport_id, field: r.field, reports: Number(r.reports), first_at: r.first_at, last_at: r.last_at })), truncated: results.length > max };
}

function fault(c: Context<{ Bindings: Env }>, route: string, e: unknown): Response {
  if (e instanceof ProfileError) return c.json({ error: e.code, message: e.message }, e.status);
  logError({ cid: crypto.randomUUID(), hop: "http", route, msg: "unexpected route fault", err: e instanceof Error ? e.name : typeof e });
  return c.json({ error: "internal" }, 500);
}

async function jsonBody(c: Context<{ Bindings: Env }>): Promise<Record<string, unknown> | null> {
  const raw = await c.req.text();
  if (raw.length > 8 * 1024) return null;
  try {
    const v = JSON.parse(raw) as unknown;
    return v && typeof v === "object" && !Array.isArray(v) ? (v as Record<string, unknown>) : null;
  } catch {
    return null;
  }
}

/** The creator-profile write handler: creator session, JSON body, then the per-IP write meter (`fanIpAllowed`,
 * "cprof"), then the one write. A module-level `function` (not a closure) so the limiter census
 * (tests/invariants/trust-core-routes.test.ts, `fnBody`) reads the meter through the call (I23-L05). Declared ABOVE the
 * factory: that census slices each handler to the next registration or the end of the file, so a meter call placed
 * after the last route would be credited to it (I23-L05 RESUME 1). */
function profileWrite(path: string, run: (db: DbLike, id: string, body: Record<string, unknown>) => Promise<unknown>) {
  return async (c: Context<{ Bindings: Env }>) => {
    const id = await creatorId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const body = await jsonBody(c);

      return c.json({ ok: true, value });
    } catch (e) {
      return fault(c, path, e);
    }
  };
}

 * ============================================================================
 *
 * The obvious fix is parity: OTP → creator session, exactly as `fanOtpSignIn`
 * does. It is the wrong fix here, and the reason is one function.
 * `startPasskeyRegistration` (passkeys.ts) asks for an existing passkey's
 * assertion only when the passport already has an active one (I3-0034). With
 * ZERO active passkeys (a first enrolment, or after the operator reset) it takes the session and a code
 * mailed to the VERIFIED contact address — the inbox this OTP proves. So a session minted here could enrol a passkey on such a
 * passport, and the passkey is the step-up gating payout destinations,
 * Connect binding, drops, broadcasts and every `CREATOR_SIGNABLE` ledger
 * write. A new sign-in factor is one hop from full control of the
 * passport, and there is nowhere to bound it: `packages/session`'s
 * `SessionSubject` is `{kind, id, ep, member?}` and the one optional claim it
 * ever had was DELETED rather than left un-stamped, because a 30-day bearer
 * claim is an unrevocable grant.
 *
 * So a proven inbox reaches the operator queue and stops. Nothing on the
 * passport changes, no credential is issued, and every remediation stays behind
 * the dual-controlled operator path it is already behind. The creator waits for
 * a human — which is slower than the fan channel, and correct, because the
 * blast radius on this side is somebody else's money.
 *
 * ============================================================================
 * NO ADDRESS-EXISTENCE ORACLE
 * ============================================================================
 *
 * `startCreatorRecovery` is unauthenticated, so "is this address a verified
 * Dhwaja creator contact?" must be unanswerable. The technique is `fan-auth.ts`'s
 * and is not re-derived: the code is generated and hashed BEFORE the limit
 * decision, the INSERT is UNCONDITIONAL with only its binding varying, the
 * per-address hourly cap and cooldown are re-asserted INSIDE that INSERT against
 * committed rows, and the row's own binding read back is the verdict. Every
 * branch — verified address, unknown address, rate-limited address — spends
 * SELECT → INSERT and returns nothing.
 *
 * The binding is the ADDRESS, not a passport. `creator_contacts.email` has no
 * uniqueness constraint, so one inbox can be the contact for several passports;
 * resolving them at redemption keeps the claim honest ("I hold this inbox") and
 * spares a locked-out creator having to remember an opaque `psp_` id.
 */

export class CreatorRecoveryError extends Error {
  override name = "CreatorRecoveryError";
  constructor(
    message: string,
    // 404 joined the set with M27: `recoveryFilingStatus` refuses an unknown or
    // malformed ticket, and it must do so with the SAME status and body the
    // route's own catch-all produces — a distinguishable refusal would be the
    // existence oracle the capability shape exists to avoid.
    public readonly status: 400 | 404 | 429 | 503 = 400,
  ) {
    super(message);
  }
}

/** The one purpose this table carries. It is also the mailer template name, so
 * the code a creator is sent and the act it can complete cannot drift apart —
 * the same coupling `FanOtpPurpose` makes. */
export const CREATOR_RECOVERY_PURPOSE = "creator-recovery" as const;

/** M27 — the dead-end mail. Sent ONLY to an address whose control was just
 * proved by an OTP redemption, and only when the filing reached no case, so it
 * discloses nothing to anyone who is not already holding that inbox. It is the
 * one place the product may say "nothing was filed"; the public status route
 * may not (see `recoveryFilingStatus`). */
export const CREATOR_RECOVERY_NO_MATCH_PURPOSE = "creator-recovery-no-match" as const;

/** Structurally unreachable binding for a blocked or unknown address: no real
 * address can be empty (the local part is required), so a sentinel row can
 * never be found by a redemption. Mirrors `NULL_ACCOUNT_ID` in fan-auth.ts. */
const SENTINEL_ADDRESS = "";

/** Same shape as `digest.ts` — one address vocabulary across the creator side. */
const EMAIL_RE = /^[^\s@]{1,64}@[^\s@]{1,255}\.[^\s@]{2,24}$/;
const HOUR_MS = 60 * 60 * 1000;
/** Swept an hour past issue, like `fan_otp`: a code is dead at 10 minutes, and
 * the window count only looks back an hour, so nothing older is load-bearing. */
const OTP_RETENTION_MS = HOUR_MS;
/** How long an unconfirmed address change may sit on the row before the sweep
 * drops it. Matches the fan side's `FAN_EMAIL_UNVERIFIED_TTL_MS`. */
const PENDING_EMAIL_TTL_MS = 7 * 24 * HOUR_MS;
/** How long a recovery tracking reference stays lookup-able (M27).
 *
 * Ninety days, and the number comes from the human process rather than from the
 * data: this is the window in which somebody who filed might still be checking
 * back on an operator queue that moves at human pace. Deliberately much longer
 * than the OTP retention above (one hour) because these rows are a receipt, not
 * a credential — and deliberately finite, because the alternative is a table an
 * inbox-holder can grow forever, which is the exact thing `pruneCreatorRecovery`
 * exists to prevent for `creator_otp`. After it, the reference reads
 * `with_an_operator` like any other unknown ticket. */
const FILING_RETENTION_MS = 90 * 24 * HOUR_MS;
/**
 * How many live passports one verified address may file for in a single
 * redemption. Ten was the shipped hard `LIMIT` and it truncated in stable-sort
 * order, permanently; this is the bound plus an audit line when it is reached.
 */
export const MAX_PASSPORTS_PER_ADDRESS = 25;


/** Six digits by rejection sampling — no modulo bias. Mirrors `newOtpCode`. */
function newOtpCode(): string {
  let out = "";
  while (out.length < 6) {
    for (const b of crypto.getRandomValues(new Uint8Array(8))) {
      if (b < 250 && out.length < 6) out += String(b % 10);
    }
  }
  return out;
}

/** Constant-time hex compare — a wrong code must not be distinguishable from a
 * nearly-right one by timing. */
function timingSafeHexEqual(a: string, b: string): boolean {
  // LENGTH-BLIND, like the fan side's. An early `a.length !== b.length` return
  // is the shape `fan-auth.ts` deliberately does NOT have — both sides here are
  // always 64 hex, so it is unexploitable today, but silently weakening a
  // hardened primitive is how a future caller inherits a hole. The length
  // difference is folded in rather than branched on.
  const n = Math.max(a.length, b.length);
  let diff = a.length ^ b.length;
  for (let i = 0; i < n; i++) diff |= (a.charCodeAt(i) | 0) ^ (b.charCodeAt(i) | 0);
  return diff === 0;
}

export function normalizeAddress(v: unknown): string | null {
  if (typeof v !== "string") return null;
  const a = v.trim().toLowerCase();
  return EMAIL_RE.test(a) ? a : null;
}

/**
 * Leg 1 — mail a recovery code to a VERIFIED creator contact address.
 *
 * Returns `void`. There is nothing to return: the answer is identical for a
 * verified address, an unknown one and a rate-limited one, and the route
 * answers `{ok:true}` unconditionally.
 */
export async function startCreatorRecovery(
  db: DbLike,
  mailer: Mailer | null,
  addressInput: unknown,
  ipBucketCount: number,
  nowMs: number,
  now: () => string = () => new Date().toISOString(),
  defer?: (p: Promise<unknown>) => void,
  /** I3-0964: the HMAC key for the stored code hash — the route passes c.env.SESSION_SECRET (otpCodeHash, digest.ts). */
  otpSecret?: unknown,
): Promise<void> {
  const address = normalizeAddress(addressInput);
  // A malformed address is refused on SHAPE, which is a property of the input
  // and never of the database — so it leaks nothing and costs nothing.
  if (!address) throw new CreatorRecoveryError("that doesn't look like an email address", 400);

  // Generated and hashed BEFORE any decision, so a blocked issue burns the same
  // hash cost as an allowed one.
  const code = newOtpCode();
  const hash = await otpCodeHash(otpSecret, "creator-recovery", code);

  // Is this a verified creator contact? The answer only ever changes the
  // BINDING of the write below — never whether it happens.
  // THE SAME PREDICATE THE REDEEM LEG USES. Without the `retracted = 0` join an
  // address verified only on passports an operator has removed still received a
  // real recovery code — Dhwaja sending auth-critical mail on behalf of a
  // passport it took down, for a filing that could then never land.
  const known = await db
    .prepare(
      `SELECT 1 AS x FROM creator_contacts c JOIN passports p ON p.passport_id = c.passport_id
        WHERE c.email = ? AND c.email_verified = 1 AND p.retracted = 0 LIMIT 1`,
    )
    .bind(address)
    .first<{ x: number }>();

  const windowStart = new Date(nowMs - HOUR_MS).toISOString();
  const { results } = await db
    // `origin = 'request'`: rotation codes (see `rotateAtCeiling`) never spend
    // this budget, so a stranger's guessing cannot use up the owner's fresh request.
    .prepare("SELECT issued_at FROM creator_otp WHERE address = ? AND origin = 'request' AND issued_at > ?")
    .bind(address, windowStart)
    .all<{ issued_at: string }>();
  const decision = canSendOtp(results.map((r) => Date.parse(r.issued_at)), ipBucketCount, nowMs);
  // D-0819(2) (B28-RECOVERY's design): the hourly COUNT cap alone no longer
  // refuses — the INSERT below admits a request past it while the address holds
  // no live request code, so a stranger's five requests cannot starve the owner.
  // The per-IP cap (checked first by canSendOtp) and the cooldown still refuse.
  const usable = known !== null && (decision.ok || decision.reason === "email_hourly_cap") && mailer !== null;

  // THE CEILING IS DECIDED IN THIS STATEMENT, not before it. Binding on a value
  // computed from the SELECT above is check-then-act, and the per-address hourly
  // cap is the ONLY per-victim ceiling in the mail-cannon guard — the per-IP
  // bucket bounds the sender, not the recipient. Fan-side this was measured:
  // twelve concurrent posts for one address mailed twelve codes against a cap of
  // five, because every racer read zero. The count and the cooldown are
  // therefore re-asserted inside the INSERT against committed rows.
  const cooldownStart = new Date(nowMs - SEND_COOLDOWN_MS).toISOString();
  const bound = await db
    .prepare(
      "INSERT INTO creator_otp (otp_id, address, purpose, code_hash, issued_at, attempts, consumed_at) VALUES (?, " +
        "CASE WHEN ? = 1 " +
        "AND ((SELECT COUNT(*) FROM creator_otp WHERE address = ? AND origin = 'request' AND issued_at > ?) < ? " +
        // Past the cap only while no request code is live (unconsumed, under its
        // attempt cap, inside its TTL): one per TTL, so an inbox takes at most
        // 5 + 6 request codes an hour and the owner always holds a live one.
        "OR NOT EXISTS (SELECT 1 FROM creator_otp WHERE address = ? AND origin = 'request' AND consumed_at IS NULL AND attempts < ? AND issued_at > ?)) " +
        "AND NOT EXISTS (SELECT 1 FROM creator_otp WHERE address = ? AND origin = 'request' AND issued_at > ?) " +
        "THEN ? ELSE ? END, ?, ?, ?, 0, NULL) RETURNING address",
    )
    .bind(
      mintId("otp"),
      usable ? 1 : 0,
      address,
      windowStart,
      SEND_HOURLY_CAP_PER_EMAIL,
      address,
      OTP_MAX_ATTEMPTS,
      // Against the clock that STAMPS issued_at (the row's own clock), so
      // "live" is judged in the same time base the row was written in.
      new Date(Date.parse(now()) - OTP_TTL_MS).toISOString(),
      address,
      cooldownStart,
      address,
      SENTINEL_ADDRESS,
      // A CONSTANT, BOUND — never branched on, so the statement text is one
      // shape for every input.
      CREATOR_RECOVERY_PURPOSE,
      hash,
      now(),
    )
    .first<{ address: string }>();

  // The row's own binding is the verdict — it is the sentinel whenever any limit
  // held, including one a racer consumed after this request read the window.
  if (bound?.address !== address || mailer === null) return;
  if (defer) {
    defer(
      mailer
        .send(CREATOR_RECOVERY_PURPOSE, address, { code })
        .then(async (ok) => {
          if (!ok) {
            // Never the address and never the code: an operator needs to know a
            // recovery mail failed, not who asked for it.
            await auditStmt(db, "system:creator-recovery", "creator_recovery:otp_send_failed", null, "a creator recovery code could not be sent (mailer disabled or provider rejected)", now()).run();
          }
        })
        .catch(() => {}),
    );
    return;
  }
  await mailer.send(CREATOR_RECOVERY_PURPOSE, address, { code });
}

/**
 * A WRONG GUESS IS CHARGED TO THE CALLER, NEVER TO THE VICTIM'S CODES (D-0648).
 *
 * How many wrong guesses one caller may make in an hour. Derived, not picked:
 * it is one person's honest worst hour on this door — every code an address can
 * be sent in an hour (`SEND_HOURLY_CAP_PER_EMAIL`), each mistyped to its own
 * per-code limit (`OTP_MAX_ATTEMPTS`). Tighter than the route's own
 * `CREATOR_RECOVERY_REDEEM_PER_IP_HOUR` (60 attempts), so this is the ceiling
 * that binds first, and it counts only MISSES — a caller who types the right
 * code has spent nothing.
 */
export const CREATOR_RECOVERY_MISSES_PER_CALLER_HOUR = OTP_MAX_ATTEMPTS * SEND_HOURLY_CAP_PER_EMAIL;

/** The `rate_limits` namespace for the above. Exported so the route cannot
 * re-spell it (FAMILY-004 P1: one predicate, one spelling, every door). */
export const CREATOR_RECOVERY_MISS_BUCKET = "crcvmiss";

/**
 * THE PER-CODE GLOBAL CEILING (QUEUE #669, D-0817(6)). The caller meter above
 * bounds ONE caller; a guesser holding P prefixes still gets 25·P guesses. So
 * every code also counts the guesses it was COMPARED against, from any caller
 * (`creator_otp.guesses`, migration 0109), and at this many it is ROTATED:
 * retired, and a new code mailed only to the address. Success per code is
 * therefore at most N / 10^6 however many prefixes guess.
 * N = TWICE one caller's hourly miss budget (2 x 25 = 50): strictly more than any
 * single caller can spend, so no lone stranger can rotate the owner's code at
 * all (the cheapest rotation needs two attributable prefixes), and the owner's
 * own mistypes are far inside it. Per-code success <= 50 / 10^6 = 5e-5.
 */
export const CREATOR_RECOVERY_GUESSES_PER_CODE = 2 * CREATOR_RECOVERY_MISSES_PER_CALLER_HOUR;
/**
 * Rotations per address per hour (`origin = 'rotation'` rows in the last hour).
 * = SEND_HOURLY_CAP_PER_EMAIL (5): the rotation mail an inbox can receive is
 * held to the same hourly ceiling the request door already holds it to, so the
 * worst case is 5 request + 5 rotation codes an hour.
 * WHEN IT IS SPENT, A CODE AT THE CEILING STAYS LIVE (it is never retired
 * without a replacement mailed): past the budget the caller meter is the only
 * bound, which is the pre-#669 state. The alternative — retire with no
 * replacement — would let a stranger lock the owner out, which D-0648 forbids.
 */
export const CREATOR_RECOVERY_ROTATIONS_PER_ADDRESS_HOUR = SEND_HOURLY_CAP_PER_EMAIL;
/** #809 EQUAL COST (the fan door's #678, fan-auth.ts FAN_OTP_COMPARE_WINDOW): the compare
 * window — request + rotation codes an address can hold live in an hour. Every guess compares
 * EXACTLY this many slots (`matchInRecoveryWindow`) and every guess writes EXACTLY this many
 * charge rows (charged code rows + `CREATOR_RECOVERY_PAD_KEY` pad rows), so a wrong guess costs
 * the same for an address with four live codes, one, or none. */
export const CREATOR_RECOVERY_COMPARE_WINDOW = SEND_HOURLY_CAP_PER_EMAIL + CREATOR_RECOVERY_ROTATIONS_PER_ADDRESS_HOUR;
/** One FIXED rate_limits key per empty slot (`crcvpad:<slot>`), never per address, so the
 * table grows by at most the window and a pad row carries nothing about anyone. */
export const CREATOR_RECOVERY_PAD_KEY = "crcvpad:";
/** The digest an empty slot is compared against; `matchInRecoveryWindow` never lets it match. */
const CREATOR_RECOVERY_DECOY_HASH = "0".repeat(64);
/** Compare `submittedHash` against every slot of the window — no early exit, the decoy digest
 * for an empty slot — and return the first real row that matched. */
function matchInRecoveryWindow<R extends { code_hash: string }>(rows: readonly R[], submittedHash: string): R | undefined {
  let found: R | undefined;
  for (let i = 0; i < CREATOR_RECOVERY_COMPARE_WINDOW; i++) {
    const r = rows[i];
    const hit = timingSafeHexEqual(r?.code_hash ?? CREATOR_RECOVERY_DECOY_HASH, submittedHash);
    if (hit && r !== undefined && found === undefined) found = r;
  }
  return found;
}
/** The edge attributed no address. A SHARED METERED bucket, never an exemption:
 * the same fail-closed shape `route-helpers.ts`'s `NO_IP_BUCKET` uses, because a
 * caller who strips a header must not thereby buy an unlimited budget. */
const NO_CALLER_PREFIX_BUCKET = "noip";
/** No route resolved a caller at all (a direct module caller, a test). Also a
 * SHARED METERED bucket, and deliberately NOT the same one as an absent header:
 * they are different failures and pooling them would hide one behind the other.
 * `removal.ts`'s `throttleKey` collapses the same case to one shared key. */
const UNATTRIBUTED_CALLER_BUCKET = "unattributed";

/**
 * WHAT THIS KEY IS SCOPED TO, AND WHY IT CANNOT BE ROTATED FOR FREE — the whole
 * fix rests on this sentence, because a key an abuser rotates for nothing bounds
 * nothing.
 *
 * `callerPrefix` is the CF edge's own connecting address, masked by
 * `accounts.ts`'s `maskIp` to a /24 (v4) or /48 (v6), read from
 * `CF-Connecting-IP` and NEVER from the body, which the attacker controls (a
 * body-supplied caller would mint a fresh counter per request and reinstate the
 * flood the keying exists to stop — `removal.ts:172` makes the same argument for
 * the same value). Malformed input cannot mint a bucket: `maskIp` fails closed to
 * one shared `invalid::/0` sentinel. Absent input cannot mint one either: it
 * falls to the shared bucket above. So the ONLY way to move to a fresh bucket is
 * to present an address the edge will attribute — which costs address space.
 *
 * THE LIMITATION W3-11-R5 FILED AND I INHERIT, stated rather than papered over:
 * that cost holds only where the header really came from the edge. A request
 * path that does not traverse the CF edge (an enabled workers.dev route, a
 * misconfigured origin, a direct-to-origin probe) could present its own
 * `CF-Connecting-IP`, and neither R5 nor I can settle from this tree whether such
 * a path exists for this Worker. Where it does not, rotation costs an address;
 * where it does, the header is forgeable and this meter is only as good as the
 * edge in front of it. It is filed, not assumed away.
 */
function callerMissBucket(callerPrefix: string | null | undefined): string {
  const who = callerPrefix === null || callerPrefix === undefined ? UNATTRIBUTED_CALLER_BUCKET : callerPrefix === "" ? NO_CALLER_PREFIX_BUCKET : callerPrefix;
  return `${CREATOR_RECOVERY_MISS_BUCKET}:${who}`;
}

/**
 * ROTATE AT THE CEILING (#669). Runs on EVERY miss, known address or not, so the
 * statement count answers nothing; it acts only when a live code has reached
 * CREATOR_RECOVERY_GUESSES_PER_CODE. One INSERT ... SELECT mints the replacement
 * — the rotation budget, the ceiling row and the verified-contact-on-a-live-
 * passport predicate (the request door's own) are all asserted INSIDE it — then
 * the ceiling rows are retired, bound to the minted id (INV-058: no mint, no
 * retirement), and the new code is mailed to the address and nowhere else.
 */
async function rotateAtCeiling(db: DbLike, mailer: Mailer | null, address: string, nowMs: number, now: () => string, defer?: (p: Promise<unknown>) => void, otpSecret?: unknown): Promise<void> {
  const code = newOtpCode();
  const hash = await otpCodeHash(otpSecret, "creator-recovery", code);
  const otpId = mintId("otp");
  const minted = await db
    .prepare(
      "INSERT INTO creator_otp (otp_id, address, purpose, code_hash, issued_at, attempts, consumed_at, guesses, origin) " +
        "SELECT ?, ?, ?, ?, ?, 0, NULL, 0, 'rotation' WHERE ? = 1 " +
        "AND EXISTS (SELECT 1 FROM creator_otp WHERE address = ? AND purpose = ? AND consumed_at IS NULL AND attempts < ? AND issued_at > ? AND guesses >= ?) " +
        "AND (SELECT COUNT(*) FROM creator_otp WHERE address = ? AND origin = 'rotation' AND issued_at > ?) < ? " +
        "AND EXISTS (SELECT 1 FROM creator_contacts c JOIN passports p ON p.passport_id = c.passport_id WHERE c.email = ? AND c.email_verified = 1 AND p.retracted = 0) " +
        "RETURNING otp_id",
    )
    .bind(
      otpId,
      address,
      CREATOR_RECOVERY_PURPOSE,
      hash,
      now(),
      mailer !== null ? 1 : 0,
      address,
      CREATOR_RECOVERY_PURPOSE,
      OTP_MAX_ATTEMPTS,
      new Date(nowMs - OTP_TTL_MS).toISOString(),
      CREATOR_RECOVERY_GUESSES_PER_CODE,
      address,
      new Date(nowMs - HOUR_MS).toISOString(),
      CREATOR_RECOVERY_ROTATIONS_PER_ADDRESS_HOUR,
      address,
    )
    .first<{ otp_id: string }>();
  await db
    .prepare(
      "UPDATE creator_otp SET consumed_at = ? WHERE address = ? AND purpose = ? AND consumed_at IS NULL AND guesses >= ? AND otp_id <> ? " +
        "AND EXISTS (SELECT 1 FROM creator_otp WHERE otp_id = ? AND origin = 'rotation')",
    )
    .bind(now(), address, CREATOR_RECOVERY_PURPOSE, CREATOR_RECOVERY_GUESSES_PER_CODE, otpId, otpId)
    .run();
  if (minted?.otp_id !== otpId || mailer === null) return;
  const sent = mailer.send(CREATOR_RECOVERY_PURPOSE, address, { code }).then(async (ok) => {
    if (!ok) {
      await auditStmt(db, "system:creator-recovery", "creator_recovery:otp_send_failed", null, "a creator recovery code could not be sent (mailer disabled or provider rejected)", now()).run();
    }
  });
  if (defer) {
    defer(sent.catch(() => {}));
    return;
  }
  await sent;
}

/**
 * Redeem a code against this address's live OTP set.
 *
 * WHOSE BUDGET A WRONG GUESS SPENDS — the one thing this function decides, and
 * the reason it was rewritten (I3-0962, ruled D-0648 "yes charge the caller").
 *
 * It used to debit `live[0]`, the victim's NEWEST live row. A stranger who knows
 * the address — it is usually in a public bio — spent `OTP_MAX_ATTEMPTS` wrong
 * guesses and the code the owner was holding was dead, with no error that
 * explained it, on the only self-service door a locked-out creator has. Moving
 * the debit to the OLDEST row was built and executed and REJECTED: it does not
 * remove the harm, it relocates it onto the other ordering, and the attacker
 * picks the ordering by choosing when to request their decoy. No choice of
 * VICTIM row can be right, because the attacker selects among them.
 *
 * So a miss debits no row at all. It spends one unit of the CALLER's hourly miss
 * budget, and when that budget is gone the caller is refused — including on a
 * code that would have matched, because a meter that never stops the hit path
 * bounds nothing and guessing would be free. The attacker cannot re-select
 * around it: they choose the ordering, the timing and the victim, and none of
 * those change which bucket their guess debits. The edge does.
 *
 * WHAT IT COSTS, AND IT IS NOT NOTHING (filed in the report, not traded away
 * here): a legitimate owner who shares a masked prefix with an active guesser
 * can find their own MISTYPES refused for the rest of the hour. Their CODE is
 * never harmed and a first-try correct entry is never refused — which is the
 * difference between this and the shape it replaces.
 *
 * DEBIT-FIRST on the matched row is unchanged, for the same reason `redeemOtp`
 * is: reading `attempts`, deciding, then burning one is check-then-act. The
 * caller meter keeps that property on the miss path — `consumeRateLimit` is ONE
 * statement, so N concurrent misses cost N, exactly as N concurrent row debits
 * used to.
 *
 * Uniform `false` on every failure — wrong code, expired, exhausted, already
 * used, no live code at all, caller out of budget — and the candidate read runs
 * identically for a known and an unknown address, so a redemption still cannot
 * answer "was a code ever sent to this address?".
 */
async function redeemCreatorOtp(
  db: DbLike,
  address: string,
  code: string,
  nowMs: number,
  now: () => string,
  callerPrefix: string | null | undefined,
  mailer: Mailer | null,
  defer?: (p: Promise<unknown>) => void,
  otpSecret?: unknown,
): Promise<boolean> {
  if (!isWellFormedOtp(code)) return false;
  const submittedHash = await otpCodeHash(otpSecret, "creator-recovery", code);
  // THE CALLER'S BUDGET, DEBITED BEFORE ANYTHING IS COMPARED (D-0821(2)). A
  // caller who has already spent their hourly misses is refused for the rest of
  // the window, and that has to include a code that WOULD have matched: a meter
  // the hit path skips bounds nothing, because a guesser simply guesses on. The
  // check IS the debit — one atomic upsert — so N concurrent guesses from one
  // caller are compared at most CREATOR_RECOVERY_MISSES_PER_CALLER_HOUR times
  // (a read here, debited after the compare, let a raced wave through on one
  // stale count). A matched code refunds its unit below: only misses cost.
  const missBucket = callerMissBucket(callerPrefix);
  if (!(await consumeRateLimit(db, missBucket, nowMs, CREATOR_RECOVERY_MISSES_PER_CALLER_HOUR, HOUR_MS))) return false;
  // THE CANDIDATE SET, NOT "THE NEWEST ROW", and the difference is a denial of
  // service on the one door this feature exists to open.
  //
  // Selecting only the newest unconsumed row lets a DEAD row shadow a live one,
  // and an outsider can create the dead row on purpose. Measured: the victim
  // requests a code; sixty-one seconds later an attacker who knows the address
  // (it is usually in a public bio) requests another — mailed to the VICTIM, so
  // the attacker never sees it — and spends five wrong guesses on it. That row
  // is now exhausted but still `consumed_at IS NULL`, so it remains the newest
  // and every later redemption is compared against ITS hash. The victim's real
  // code stops working, they get mails that never work, and no error explains
  // it. Repeat six times an hour and a chosen creator has no self-service way
  // back in at all.
  //
  // So: exclude rows that are already dead (exhausted or expired) from
  // candidacy, and compare the submitted code against the whole live set rather
  // than against one row. A code always finds its own row.
  //
  // The set is bounded by the per-address hourly ceilings — at most
  // SEND_HOURLY_CAP_PER_EMAIL request rows plus as many rotation rows — and one
  // statement.
  //
  // #669: THE CANDIDATE READ IS FOLLOWED BY A GLOBAL GUESS DEBIT, and only the
  // rows that debit CHARGED are compared. The debit is one statement with its own
  // RETURNING verdict, so N concurrent guesses cost N and a row is compared at
  // most CREATOR_RECOVERY_GUESSES_PER_CODE times (INV-058: the compare binds the
  // charged ids, not the read's shape). A row already at the ceiling is left out
  // ONLY while a rotation can replace it (mailer present, rotation budget left);
  // otherwise it stays a candidate — a code is never retired unreplaced.
  // The set is bounded by the two hourly ceilings (request + rotation).
  const ttlCutoff = new Date(nowMs - OTP_TTL_MS).toISOString();
  const { results: candidates } = await db
    .prepare(
      "SELECT otp_id, code_hash, issued_at, attempts, consumed_at FROM creator_otp " +
        "WHERE address = ? AND purpose = ? AND consumed_at IS NULL AND attempts < ? AND issued_at > ? " +
        "ORDER BY issued_at DESC, otp_id DESC LIMIT ?",
    )
    .bind(address, CREATOR_RECOVERY_PURPOSE, OTP_MAX_ATTEMPTS, ttlCutoff, CREATOR_RECOVERY_COMPARE_WINDOW)
    .all<{ otp_id: string; code_hash: string; issued_at: string; attempts: number; consumed_at: string | null }>();
  const { results: charged } = await db
    .prepare(
      "UPDATE creator_otp SET guesses = guesses + 1 " +
        "WHERE address = ? AND purpose = ? AND consumed_at IS NULL AND attempts < ? AND issued_at > ? " +
        "AND NOT (guesses >= ? AND ? = 1 AND (SELECT COUNT(*) FROM creator_otp WHERE address = ? AND origin = 'rotation' AND issued_at > ?) < ?) " +
        "RETURNING otp_id",
    )
    .bind(
      address,
      CREATOR_RECOVERY_PURPOSE,
      OTP_MAX_ATTEMPTS,
      ttlCutoff,
      CREATOR_RECOVERY_GUESSES_PER_CODE,
      mailer !== null ? 1 : 0,
      address,
      new Date(nowMs - HOUR_MS).toISOString(),
      CREATOR_RECOVERY_ROTATIONS_PER_ADDRESS_HOUR,
    )
    .all<{ otp_id: string }>();
  // #809: the charge above writes one row per live code, so its row count was an
  // address-activity oracle (4 live codes cost 4 writes, an unknown address 0). This
  // statement is ALWAYS issued and writes the remaining slots of the window as fixed pad
  // rows, so every guess writes exactly CREATOR_RECOVERY_COMPARE_WINDOW rows.
  await db
    .prepare("INSERT INTO rate_limits (rl_key, window_start, count) SELECT ? || value, ?, 1 FROM json_each(?) WHERE true ON CONFLICT(rl_key) DO UPDATE SET count = rate_limits.count + 1")
    .bind(CREATOR_RECOVERY_PAD_KEY, now(), JSON.stringify(Array.from({ length: Math.max(0, CREATOR_RECOVERY_COMPARE_WINDOW - charged.length) }, (_, i) => charged.length + i)))
    .run();
  const chargedIds = new Set(charged.map((r) => r.otp_id));
  const live = candidates.filter((r) => chargedIds.has(r.otp_id));
  // The row whose code this actually is, if any — every slot compared (#809), not "until found".
  const matched = matchInRecoveryWindow(live, submittedHash);
  // A MISS DEBITS THE CALLER, NOT THE VICTIM. This is the whole fix. It used to
  // read `matched ?? live[0]` — the victim's newest live row — which is how a
  // stranger's five wrong guesses killed the code the owner was holding. The
  // guess still costs; it costs the party making it — its unit was debited
  // before the compare, one atomic statement, so N concurrent misses cost N.
  if (!matched) {
    await rotateAtCeiling(db, mailer, address, nowMs, now, defer, otpSecret);
    return false;
  }
  await refundRateLimit(db, missBucket, nowMs, HOUR_MS);
  const row = matched;
  const otpId = row.otp_id;
  // WHY THE `NULL_OTP_ID` SENTINEL IS GONE. It existed so that a miss against an
  // address with no live code spent the SAME two statements as a miss against
  // one that had codes, so a redemption could not answer "was a code ever sent
  // here?". That uniformity is preserved, one statement earlier: the candidate
  // read and the caller debit above run identically whether `live` is empty or
  // full, and neither branches on it. The sentinel had nothing left to hide.
  const debit = await db.prepare("UPDATE creator_otp SET attempts = attempts + 1 WHERE otp_id = ? AND consumed_at IS NULL").bind(otpId).run();
  const after = await db
    .prepare("SELECT code_hash, issued_at, attempts, consumed_at FROM creator_otp WHERE otp_id = ?")
    .bind(otpId)
    .first<{ code_hash: string; issued_at: string; attempts: number; consumed_at: string | null }>();
  if (debit.changes !== 1 || !after || after.consumed_at !== null) return false;
  // `after.attempts` already counts THIS attempt; the policy is handed the state
  // as it stood immediately before it.
  const record: OtpRecord = { codeHash: after.code_hash, issuedAtMs: Date.parse(after.issued_at), attempts: after.attempts - 1, consumedAtMs: null };
  if (verifyOtp(record, timingSafeHexEqual(after.code_hash, submittedHash), nowMs).outcome !== "consumed") return false;
  const res = await db
    .prepare("UPDATE creator_otp SET consumed_at = ? WHERE otp_id = ? AND consumed_at IS NULL AND attempts = ?")
    .bind(now(), otpId, after.attempts)
    .run();
  return res.changes === 1; // lost the race → dead, single-use
}

export interface CreatorRecoveryResult {
  /** Always true when the code was good. Deliberately carries no count and no
   * passport id — see the note in `fileCreatorRecovery`. */
  received: true;
  /** M27 — THE TICKET, and the one thing this response has ever carried back.
   *
   * Minted from the REQUEST, not from the result: the same shape comes back for
   * an address behind twelve passports, one, or none at all. That is what keeps
   * it out of the oracle business — everybody who proves an inbox gets exactly
   * one, so holding one says nothing except that you redeemed a code. */
  tracking_id: string;
}

/** The rolled-up answer `POST /recovery/status` gives. ONE WORD about a filing,
 * never a per-case list: a list's LENGTH counts passports behind an address,
 * which is the number this whole path refuses to disclose. */
export type RecoveryFilingStatus = "with_an_operator" | "resolved" | "closed" | "mixed";

/**
 * Leg 2 — prove the inbox, and file into the operator queue for every live
 * passport that verified it.
 *
 * WHAT THE RESPONSE DOES NOT SAY. Not how many passports were filed for, not
 * which, not whether any existed. A creator who owns the inbox already knows
 * what is on it; a stranger who has stolen it learns nothing from the answer
 * that the mail they had to receive did not already tell them, and an operator
 * sees the whole picture in the queue. This mirrors the coarse
 * `{ok:true, received:true, deduped}` the session-gated route returns, minus the
 * dedupe flag, which here would count passports.
 *
 * A proven inbox with no passports behind it — an address whose contact rows
 * were all changed since the code was sent — returns the same body. There is
 * nothing to file and nothing to say.
 */
export async function fileCreatorRecovery(
  db: DbLike,
  input: { email: unknown; code: unknown; kind: unknown; detail: unknown; contact?: unknown },
  nowMs: number,
  now: () => string = () => new Date().toISOString(),
  // OPTIONAL, AND NULL IS A REAL ANSWER. The dead-end mail is a courtesy on a
  // path whose contract is uniformity: a caller with no mailer configured must
  // still be able to FILE, because the alternative is a locked-out creator
  // refused at the one door they reached for over a mail they never asked for.
  // The route passes one; `startCreatorRecovery` is where absent mail config
  // fails the request closed, and it already does.
  mailer: Mailer | null = null,
  defer?: (p: Promise<unknown>) => void,
  // WHO IS GUESSING — the MASKED client prefix, supplied by the route from
  // `callerPrefix(c)` (route-helpers.ts) and never from the body. A wrong code
  // is charged to THIS, not to the address's live rows; see `redeemCreatorOtp`
  // for what the key is scoped to and why it cannot be rotated for free.
  //
  // TRAILING AND OPTIONAL so every existing caller keeps compiling, and `null`
  // — the value a caller that resolved nobody passes — is a REAL answer, not an
  // exemption: it lands in a shared METERED bucket. There is no argument to this
  // function that buys an unmetered guess.
  callerPrefix: string | null = null,
  // I3-0728: the operator support address, when configured (supportEmailFromEnv,
  // env-config.ts) — TRAILING AND OPTIONAL for the same reason callerPrefix is:
  // an existing caller that has not been updated keeps compiling, and `null`/
  // `undefined` is a real "not configured" answer, not an exemption.
  supportEmail?: string | null,
  /** I3-0964: the HMAC key for the stored code hash — the route passes c.env.SESSION_SECRET (otpCodeHash, digest.ts). */
  otpSecret?: unknown,
): Promise<CreatorRecoveryResult> {
  const address = normalizeAddress(input.email);
  const code = typeof input.code === "string" ? input.code.trim() : "";
  // SHAPE FIRST, AND ALL OF IT — including the FILING, not just the address.
  //
  // These checks used to sit after the redemption, inside a loop whose refusals
  // were swallowed. A locked-out creator whose client sent a missing or
  // malformed `kind`/`detail` therefore burned their single-use code, filed
  // nothing, and was told `{ok:true, received:true}` — then hit the 60-second
  // cooldown on the retry with no error to explain any of it. Measured, on the
  // one door a person reaches for when everything else has already failed.
  //
  // Every refusal here is a function of the INPUT and never of stored state, so
  // hoisting them leaks nothing and costs nothing — the same argument the
  // address check has always made for itself.
  if (!address) throw new CreatorRecoveryError("that doesn't look like an email address", 400);
  if (typeof input.kind !== "string" || !RECOVERY_KINDS.has(input.kind)) {
    throw new CreatorRecoveryError("tell us what happened: lost_passkey, hijack, agency_dispute or competing_claim", 400);
  }
  const detail = typeof input.detail === "string" ? input.detail.trim() : "";
  if (detail.length < 1 || detail.length > 1000) {
    throw new CreatorRecoveryError("describe what happened in 1 to 1000 characters", 400);
  }
  const contact = typeof input.contact === "string" ? input.contact.trim() : "";
  if (contact.length > 200) throw new CreatorRecoveryError("contact must be 200 characters or fewer", 400);

  if (!(await redeemCreatorOtp(db, address, code, nowMs, now, callerPrefix, mailer, defer, otpSecret))) {
    throw new CreatorRecoveryError("that code is wrong or has expired", 400);
  }

  // M27 — THE TICKET IS MINTED HERE, and the position is deliberate.
  //
  // A CORRECTION TO THE FIRST VERSION OF THIS NOTE, which claimed the position
  // stops the ticket becoming a creator-contact oracle. It does not, and it does
  // not need to: `startCreatorRecovery` binds the OTP row to the SENTINEL
  // address unless the address is a verified contact on a live passport, so an
  // unknown address can never redeem a code and never reaches this line. That
  // oracle was already closed one leg upstream. Claiming a security property
  // this code does not provide is worse than not having it, because the next
  // reader stops looking for the real guard.
  //
  // What the position DOES buy is uniformity across the outcomes that ARE
  // reachable here. A redemption can still reach zero cases — the per-passport
  // daily cap swallowed every filing, or the passport was retracted between the
  // code being sent and used — and minting before the lookup means those return
  // the same shape as a filing that reached twelve. The response never varies
  // with how much is behind the address, which is the property
  // `fileCreatorRecovery` already protects by dropping the `deduped` flag.
  //
  // The row carries NO address, in any form. It held a SHA-256 of the address in
  // the first version, justified as "for the prune" — but the prune keys on
  // `created_at`, nothing ever read the column, and an unsalted digest of a
  // lowercased email is reversible by enumeration, so it reproduced the roster
  // it was meant to prevent. See migration 0049. (Named without the canonical
  // helper's identifier on purpose: `single-hasher.test.ts` reads this file as
  // TEXT and a mention would register it as an importer.)
  //
  // BEST-EFFORT, and it has to be: at this point the single-use OTP is already
  // consumed. A throw here would burn a locked-out creator's one code, file
  // nothing, hand them a 500 and a 60-second cooldown — over a row that only
  // exists so they can check back. The status read answers `with_an_operator`
  // for a ticket with no filing row, which is exactly right for this case.
  // 128 CSPRNG BITS, not the 80-bit default. This is a bearer capability on the
  // same footing as `removal_requests.request_id`, whose own comment pins it at
  // 32 hex for exactly this reason: the ONLY thing standing between a guesser
  // and somebody else's recovery state is the width of this string. `mintId`'s
  // default (20 hex) is sized for row ids that are never a secret.
  const trackingId = mintId("trk", 32);
  const filedAt = now();
  await db
    .prepare("INSERT INTO creator_recovery_filings (tracking_id, created_at) VALUES (?, ?)")
    .bind(trackingId, filedAt)
    .run()
    .catch(() => {});

  // The inbox is proven. Resolve every LIVE passport that verified it — a
  // retracted one is not a passport anybody can be locked out of, and filing
  // against it would put a case in the queue with nothing to remediate.
  // BOUNDED, BUT NOT SILENTLY TRUNCATED. `ORDER BY` is a stable sort, so a hard
  // `LIMIT 10` meant passports 11+ behind one address could never be filed for —
  // not on a retry, not tomorrow — and the uniform response is exactly what made
  // that invisible. Measured with twelve passports on one agency address: ten
  // cases, two permanently unreachable, `{received:true}`.
  //
  // The bound stays (a single request must not be able to enqueue unboundedly),
  // but it is high enough to cover any plausible fan-out, and reaching it writes
  // an operator audit line. The SIGNAL belongs in the audit log, never in the
  // body: the response has to stay uniform.
  const { results } = await db
    .prepare(
      `SELECT c.passport_id AS passport_id FROM creator_contacts c
         JOIN passports p ON p.passport_id = c.passport_id
        WHERE c.email = ? AND c.email_verified = 1 AND p.retracted = 0
        ORDER BY c.passport_id
        LIMIT ?`,
    )
    .bind(address, MAX_PASSPORTS_PER_ADDRESS + 1)
    .all<{ passport_id: string }>();
  if (results.length > MAX_PASSPORTS_PER_ADDRESS) {
    results.length = MAX_PASSPORTS_PER_ADDRESS;
    await auditStmt(db, "system:creator-recovery", "creator_recovery:fanout_truncated", null, `more than ${MAX_PASSPORTS_PER_ADDRESS} live passports share one verified contact address; only the first ${MAX_PASSPORTS_PER_ADDRESS} were filed for`, now()).run();
  }

  // How many cases this filing actually reached. Drives ONLY the dead-end mail
  // below — never the response, which stays uniform.
  let linked = 0;
  for (const r of results) {
    // `submitRecoveryRequest` carries the daily cap (`rcvreq:<kind>:<id>`, 5),
    // the dedupe into one open case and the audit line. Its own refusals are
    // SWALLOWED here on purpose: a creator who has already filed today must get
    // the same answer as one who has not, or the response becomes a probe for
    // how much noise an inbox has already made — and there is nothing useful a
    // locked-out person could do with the difference anyway.
    try {
      const filed = await submitRecoveryRequest(
        db,
        // THE REQUESTER IS THE CREATOR, and saying so is the point: the operator
        // queue shows a case filed BY the passport holder, proven by an inbox
        // Dhwaja itself verified, rather than by an anonymous stranger. It is
        // the strongest provenance this path can carry and it is still only
        // provenance — the request executes nothing.
        { kind: "creator", id: r.passport_id },
        {
          kind: input.kind as RecoveryKind,
          target_passport_id: r.passport_id,
          description: detail,
          ...(contact ? { contact } : {}),
        },
        nowMs,
        now,
        // ITS OWN CASE, never folded into a fan's. The fold keeps the FIRST
        // filer's kind, description and contact — so without this an attacker
        // could pre-file "hijack / transfer it to me" against a popular creator
        // from a throwaway fan account and have the real creator's inbox-proven
        // filing arrive as CORROBORATION of it, with the attacker's address as
        // the reply-to. Measured before this argument existed.
        "creator_inbox",
      );
      // THE LINK, and it is a link because the CASE IS SHARED. A second filing
      // for a passport that already has an open creator-inbox case folds into
      // that case (`deduped: true`, same `item_id`), so a `tracking_id` column
      // on the queue row would hold the FIRST filer's ticket forever and every
      // later ticket would resolve to nothing — this silence, wearing a ticket.
      //
      // Its own statement, outside nothing: a failure to link must not undo a
      // filing that already landed. A case with no link is invisible to the
      // status read, which answers `with_an_operator` for it — the honest and
      // safe direction, since that is what is actually true.
      await db
        .prepare("INSERT INTO recovery_filing_cases (tracking_id, item_id, created_at) VALUES (?, ?, ?) ON CONFLICT(tracking_id, item_id) DO NOTHING")
        .bind(trackingId, filed.item_id, filedAt)
        .run()
        .catch(() => {});
      linked += 1;
    } catch (e) {
      // NARROW, and it has to be. The swallow is documented for ONE case: the
      // daily cap. Everything else reaching here means the filing did not land
      // for a reason the caller could not have caused (the input was validated
      // above), and a silent `{ok:true}` over it would tell a locked-out person
      // their one code worked when it did not.
      //
      // The cap itself stays silent on purpose: a creator who has already filed
      // today must get the same answer as one who has not, or the response
      // becomes a probe for how much noise an inbox has already made.
      if (!(e instanceof RecoveryError) || e.status !== 429) throw e;
    }
  }

  // THE DEAD END GETS AN ANSWER — in the one channel where an answer is free.
  //
  // A creator who mistypes their address, or whose contact was never verified,
  // files against nothing. The PUBLIC answer cannot say so: `with_an_operator`
  // has to cover "open" and "nothing was filed" alike, or the ticket becomes the
  // oracle. But this mail goes to an inbox the caller PROVED they control one
  // statement ago, so it discloses to a third party exactly nothing — and it is
  // the difference between a dead end and a silent one.
  //
  // WHAT `linked === 0` ACTUALLY MEANS, because the first version of this
  // comment got it wrong and the mail copy inherited the error. It is NOT "this
  // address is not a creator contact": `startCreatorRecovery` binds the OTP row
  // to the SENTINEL address unless the address is a verified contact on a live
  // passport, so an unknown address can never redeem a code and can never reach
  // this line at all. The two situations that DO reach it are:
  //   - every filing hit the per-passport daily cap and was swallowed above;
  //   - the passport was retracted between the code being sent and redeemed.
  // The mail says those two things and no longer claims the first. A body that
  // told a creator "no account lists this address" after their code worked would
  // be both false and alarming.
  //
  // Best-effort AND OFF THE RESPONSE PATH, which is `mailer.ts`'s own rule in its
  // own words: "Do not call `send` inline on a path that must not leak whether
  // an address maps to an account." The first version awaited it here, so a
  // provider timeout (`MAIL_TIMEOUT_MS` = 10s) sat on the response of the one
  // door a locked-out creator reaches for. The latency difference is not itself
  // an oracle — the only party who can observe it is the one who just redeemed
  // the code and receives the mail — but breaking a convention that keeps the
  // sibling leg safe, silently, is how the sibling leg stops being safe.
  //
  // `defer` is the route's `waitUntil`; without one the send still happens, just
  // awaited, so a caller with no deferrer (tests, the console) is not broken.
  if (linked === 0 && mailer) {
    const send = mailer
      .send(CREATOR_RECOVERY_NO_MATCH_PURPOSE, address, { tracking_id: trackingId, support_email: supportEmail ?? "" }, supportEmail ?? undefined)
      .catch(() => false);
    if (defer) defer(send);
    else await send;
  }
  return { received: true, tracking_id: trackingId };
}

/**
 * M27 — what happened to a filing, in ONE WORD.
 *
 * Capability-addressed: the ticket is the whole authorization, exactly as
 * `removalStatus` treats a `request_id`. The route consumes its rate bucket
 * BEFORE calling this, so the limiter cannot become a "does this ticket exist"
 * oracle, and an unknown ticket gets the same uniform refusal a real one with no
 * cases gets.
 *
 * `with_an_operator` IS THE LOAD-BEARING ARM. It covers four worlds that this
 * function cannot and must not tell apart: an open case; a filing that reached
 * no case (cap exhausted, or the passport retracted mid-flight); a filing whose
 * cases have since been pruned; and a ticket that never existed. The last one is
 * why an unknown-but-well-formed ticket is NOT a 404 — a real zero-case filing
 * has exactly the same zero rows behind it, so a distinct answer for "no such
 * ticket" would also be the answer for "your filing reached nothing", told to
 * the person it is about, in the one channel where it must not be said. The
 * shape refusal below is a different thing: it rejects input that could not be a
 * ticket, which is a property of the string and not of stored state.
 *
 * The cost — a creator whose filing reached nothing waits on a sentence that
 * will not change — is real, and is paid by the mail in `fileCreatorRecovery`,
 * not by this response.
 *
 * `mixed` is the one arm that discloses "more than one passport is behind this
 * address", and only to someone who proved control of the inbox those passports
 * share. Stated as a residual rather than designed around.
 */
/**
 * `review_queue.state`, PARTITIONED — and it is `review_queue`'s vocabulary, not
 * `removal_requests`'.
 *
 * THIS ROUTE SHIPPED READING THE WRONG TABLE'S WORDS. It was modelled on
 * `POST /removal/status` and the state names came along with the shape, so it
 * tested for `"resolved"` — a `removal_requests` state (`removal.ts
 * REMOVAL_STATES`) that `review_queue` has NEVER written. A recovery case is
 * closed by `resolveRecoveryCase`, which writes `state = 'approved'`. So every
 * terminal outcome, a successful recovery included, fell through to the `else`
 * and was reported to the creator as "an operator closed it without action" —
 * on the one screen a person checks after being locked out and helped. Measured,
 * not reasoned about; `review_queue.state` carries no CHECK constraint
 * (migration 0019 constrained `disposition` only), so the wrong literal wrote
 * cleanly in a fixture and the tests were green against it.
 *
 * Exhaustive over `review.ts ItemState` by construction: anything not named in
 * these two sets is a decision that went against the filer, which is the safe
 * direction for an unrecognised future state — it says "a person looked", never
 * "you are fine". `slice19-defects.test.ts` pins every member of `ItemState` to
 * an arm so a new state cannot be added without landing here.
 */
const RECOVERY_OPEN_STATES: ReadonlySet<string> = new Set(["open", "queued_warning"]);
/** The states that mean an operator ACTED on the case. `published` is here for
 * completeness of the enum rather than reachability — a recovery case is not a
 * warning — and costs nothing by being right. */
const RECOVERY_DONE_STATES: ReadonlySet<string> = new Set(["approved", "published"]);

/** The shape `mintId("trk", 32)` produces — 128 CSPRNG bits, matching
 * `removal.ts REQUEST_ID_RE`. Anchored, so a longer or shorter string is refused
 * on shape before it can reach a query. */
const TRACKING_ID_RE = /^trk_[0-9a-f]{32}$/;

export async function recoveryFilingStatus(db: DbLike, trackingId: unknown): Promise<{ status: RecoveryFilingStatus }> {
  // SHAPE FIRST — a malformed ticket is a property of the input and never of
  // stored state, so refusing it here reads nothing and leaks nothing. Without
  // it, junk would reach the query and be answered by row COUNT, which is a
  // slower path than the shape refusal and therefore a timing difference.
  if (typeof trackingId !== "string" || !TRACKING_ID_RE.test(trackingId)) {
    throw new CreatorRecoveryError("not found", 404);
  }
  const { results } = await db
    .prepare(
      `SELECT q.state AS state FROM recovery_filing_cases f
         LEFT JOIN review_queue q ON q.item_id = f.item_id
        WHERE f.tracking_id = ?`,
    )
    .bind(trackingId)
    .all<{ state: string | null }>();
  // NO ROWS IS NOT AN ERROR AND NOT A DISTINCT ANSWER. It is a ticket for a
  // filing that matched nothing, a ticket whose cases have been pruned, and a
  // ticket that never existed — all three answered identically on purpose.
  if (results.length === 0) return { status: "with_an_operator" };
  let open = 0;
  let resolved = 0;
  let closed = 0;
  for (const r of results) {
    // A NULL state is a link whose queue row is gone (pruned, or deleted by an
    // operator). It counts as OPEN — the fail direction that tells a locked-out
    // person to keep waiting rather than that they were dealt with.
    const st = r.state ?? "open";
    if (RECOVERY_OPEN_STATES.has(st)) open += 1;
    else if (RECOVERY_DONE_STATES.has(st)) resolved += 1;
    else closed += 1;
  }
  if (open > 0) return { status: "with_an_operator" };
  if (resolved > 0 && closed > 0) return { status: "mixed" };
  return { status: resolved > 0 ? "resolved" : "closed" };
}

/** Hourly sweep, mirroring `pruneFanAuth`: an OTP is dead at ten minutes and the
 * window count looks back an hour, so nothing older is load-bearing — and a
 * table of consumed hashes an unauthenticated caller can grow is not a table to
 * keep forever. */
export async function pruneCreatorRecovery(db: DbLike, nowMs: number): Promise<void> {
  await db.prepare("DELETE FROM creator_otp WHERE issued_at < ?").bind(new Date(nowMs - OTP_RETENTION_MS).toISOString()).run();
  // M27 — AND THE TRACKING TABLES, which shipped with a migration header
  // borrowing this function's own argument ("a table an unauthenticated caller
  // can grow is not a table to keep forever") and no sweep behind it.
  //

  // out. `pruneFanAuth` sweeps exactly this column for fans; the creator side

  await db
    .prepare("UPDATE creator_contacts SET pending_email = NULL, pending_created_at = NULL WHERE pending_email IS NOT NULL AND pending_created_at IS NOT NULL AND pending_created_at < ?")

/** Current epoch for a passport (0 if never bumped). */
export async function currentCreatorEpoch(db: DbLike, passportId: string): Promise<number> {
  const row = await db.prepare("SELECT session_epoch FROM creator_sessions WHERE passport_id = ?").bind(passportId).first<{ session_epoch: number }>();
  return row?.session_epoch ?? 0;
}

/** Bump the epoch (logout everywhere / suspected token compromise). Returns
 * the new epoch. Upsert so the first bump creates the row.
 *
 * `involuntary` is the OPERATOR/RECOVERY half of the door (migration 0082,
 * FINDING-0014-C-1): `updated_at` is stamped by every bump and cannot tell a
 * hijack recovery from the creator's own logout-all, and one row per passport
 * cannot carry a per-bump reason — a recovery followed later by a self-service
 * logout would read as self-service. So the involuntary doors leave a
 * MONOTONIC WATERMARK instead, which `consumeManualClaimRef` compares against
 * the approval instant. `MAX(existing, excluded)` is what makes it monotonic: a
 * later voluntary bump writes nothing here and an out-of-order involuntary one
 * cannot move the mark backwards.
 *
 * It defaults to FALSE, and the default is safe only in one direction: an
 * unclassified caller leaves the ref LIVE, never falsely burned. A door that
 * means to end outstanding authority has to say so — and `test/s007-claim-ref
 * .test.ts` reds when an epoch writer appears that this file's own classified
 * set does not name.
 *
 * `extraStatements` (QUEUE #942(a), D-0946(1)): the caller's audit row rides the
 * upsert's own `db.batch`, AFTER it, so a failed audit insert rolls the bump back
 * and a statement reading `creator_sessions` there sees this bump's epoch. */
export async function bumpCreatorEpoch(
  db: DbLike,
  passportId: string,
  now: () => string = () => new Date().toISOString(),
  involuntary = false,
  extraStatements: DbStatement[] = [],
): Promise<number> {
  const ts = now();
  await db.batch([
    db
      .prepare(
        `INSERT INTO creator_sessions (passport_id, session_epoch, updated_at, last_involuntary_bump_at) VALUES (?, 1, ?, ?)
         ON CONFLICT(passport_id) DO UPDATE SET session_epoch = session_epoch + 1, updated_at = excluded.updated_at,
           last_involuntary_bump_at = CASE WHEN excluded.last_involuntary_bump_at IS NULL THEN creator_sessions.last_involuntary_bump_at
                                           ELSE MAX(COALESCE(creator_sessions.last_involuntary_bump_at, ''), excluded.last_involuntary_bump_at) END`,
      )
      .bind(passportId, ts, involuntary ? ts : null),
    ...extraStatements,
  ]);
  return currentCreatorEpoch(db, passportId);
}

/** I25-Q2: SET a fraud finding. Idempotent: an open finding already there writes nothing and returns `changed: false`. */
export async function setCreatorFraudFinding(db: DbLike, passportId: string, reason: unknown, operator: string, now: () => string = () => new Date().toISOString()): Promise<{ changed: boolean }> {

/** Closed creators one sweep call counts (each costs one `creatorEarnings` read and one batch). */
export const CREATOR_DEBT_SWEEP_LIMIT = 50;
const SWEEP_ACTOR = "system:creator-debt";

/**
 * I20-Q2 (D-1031): ONCE PER ISO WEEK, count each closed creator (creator_closures) not yet in the register: the netted debt
 * is `creatorEarnings(..).owed_cents` (debits beyond cleared earnings, creator-payout.ts, read only). A debt that is gone, or
 * lower than at the last count (the first count compares with the close snapshot), resets the count to 0; otherwise the
 * count goes up by one. At `creatorDebtUnrecoverableRuns` it writes, in one batch, the register row and — guarded on THAT
 * row being this call's — an open fraud-banner flag and one audit row, so a call racing a registration that already landed
 * (and whose flag an operator may have cleared) writes neither. A second call in the same week selects nobody it already
 * counted. `more` is true when the bound was reached. Scheduled by I29-INT after the weekly payout tail.
 */
export async function sweepUnrecoverableCreatorDebt(db: DbLike, nowMs: number = Date.now(), limit = CREATOR_DEBT_SWEEP_LIMIT): Promise<{ counted: number; registered: number; more: boolean }> {
  const lim = Math.max(1, Math.min(limit, CREATOR_DEBT_SWEEP_LIMIT));
  const week = isoWeek(nowMs);
  const ts = new Date(nowMs).toISOString();
  const threshold = await creatorDebtUnrecoverableRuns(db);
  const { results } = await db
    .prepare(
      `SELECT cc.passport_id AS passport_id, cc.owed_cents AS closed_owed, w.owed_cents AS last_owed, w.empty_runs AS empty_runs
         FROM creator_closures cc LEFT JOIN creator_debt_weeks w ON w.passport_id = cc.passport_id
        WHERE NOT EXISTS (SELECT 1 FROM creator_reserve_losses r WHERE r.passport_id = cc.passport_id) AND (w.week IS NULL OR w.week <> ?)
        ORDER BY cc.passport_id LIMIT ?`,
    )
    .bind(week, lim)
    .all<{ passport_id: string; closed_owed: number; last_owed: number | null; empty_runs: number | null }>();
  let registered = 0;
  for (const r of results) {
    const owed = (await creatorEarnings(db, r.passport_id, nowMs)).owed_cents;
    const before = r.last_owed ?? r.closed_owed;
    const runs = owed <= 0 || owed < before ? 0 : (r.empty_runs ?? 0) + 1;
    const stmts: DbStatement[] = [
      db
        .prepare(
          `INSERT INTO creator_debt_weeks (passport_id, week, owed_cents, empty_runs) VALUES (?, ?, ?, ?)
             ON CONFLICT(passport_id) DO UPDATE SET week = excluded.week, owed_cents = excluded.owed_cents, empty_runs = excluded.empty_runs`,
        )
        .bind(r.passport_id, week, Math.max(0, owed), runs),
    ];
    if (runs >= threshold) {
      const mine = "EXISTS (SELECT 1 FROM creator_reserve_losses WHERE passport_id = ? AND recorded_at = ? AND runs_observed = ?)";
      stmts.push(
        db
          .prepare("INSERT INTO creator_reserve_losses (passport_id, debt_cents, runs_observed, recorded_at) VALUES (?, ?, ?, ?) ON CONFLICT(passport_id) DO NOTHING")
          .bind(r.passport_id, owed, runs, ts),
        db
          .prepare(`INSERT INTO creator_fraud_banner_flags (passport_id, set_at) SELECT ?, ? WHERE ${mine} AND NOT EXISTS (SELECT 1 FROM creator_fraud_banner_flags WHERE passport_id = ? AND cleared_at IS NULL)`)
          .bind(r.passport_id, ts, r.passport_id, ts, runs, r.passport_id),
        guardedAudit(db, mine, [r.passport_id, ts, runs], SWEEP_ACTOR, RESERVE_LOSS_ACTION, `debt_cents=${owed};runs=${runs};banner_flag=set`, ts, r.passport_id, "system"),
      );
    }
    await db.batch(stmts);
    if (runs >= threshold && (await db.prepare("SELECT 1 AS x FROM creator_reserve_losses WHERE passport_id = ? AND recorded_at = ?").bind(r.passport_id, ts).first()) !== null) registered++;
  }
  return { counted: results.length, registered, more: results.length === lim };
}
