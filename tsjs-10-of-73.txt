export type TrustStatus = "verified" | "claimed" | "unclaimed" | "warning" | "caution" | "unknown" | "unsupported" | "failed"; // check.ts re-exports it; declared here so this module does not import check.ts back (I13-ARCH, QUEUE #902: the cycle)

/**
 * THE POST-CHECK HANDOFF — a literal mirror of
 * `apps/mobile-check/src/safety-handoff.ts` (same hashes, same labels), pinned
 * across the four clients by `tests/invariants/safety-handoff-parity.test.ts`.
 * It lives beside the verdict mirror because it is the same kind of thing: a
 * table this client cannot import from the spec module and must therefore be
 * held to it by a guard.
 *
 * WHY THE POPUP NEEDED ONE. It appended a link only when the result carried a
 * validated real-creator pointer, or when the verdict was positive. A `warning`
 * WITHOUT a pointer — the most dangerous thing this extension can say — got a
 * red line of text and no link of any kind, and so did `caution`, `unclaimed`,
 * `unknown` and `failed`.
 *
 * Every hash is a bare route fragment on the app's own origin and carries
 * nothing about the page that was checked: the k-anonymous check keeps the
 * subject on the device, so a link must not take it off.
 */
export interface SafetyHandoff {
  hash: string;
  label: string;
}

/** "Is this listing me?" — rendered on EVERY verdict, as on the web, because
 * somebody arriving at their own listing does not know which verdict is being
 * served about them and the removal valve must not be gated on it. */
export const HANDOFF_MINE: SafetyHandoff = { hash: "#/remove", label: "Is this listing me? →" };

export const SAFETY_HANDOFF: Record<TrustStatus, SafetyHandoff> = {
  verified: { hash: "#/discover", label: "Open Dhwaja →" },
  claimed: { hash: "#/discover", label: "Open Dhwaja →" },
  unclaimed: { hash: "#/nominate", label: "Ask for a creator on Dhwaja →" },
  warning: { hash: "#/report", label: "Report this account →" },
  // A caution's primary door is the DISPUTE, not the report: a caution is
  // machine-served and always contestable, and the person most likely to be
  // reading one is its subject — a creator whose own unproven listing was
  // cautioned. "Report this account" invited them to report themselves.
  caution: { hash: "#/contest", label: "Dispute this caution →" },
  unknown: { hash: "#/nominate", label: "Suggest this creator →" },
  unsupported: { hash: "#/safety", label: "What can I do? →" },
  // I3-0451 (D-0920): a check that could not complete offers the re-run, not the hub.
  failed: { hash: "#/check", label: "Try again →" },
};

/** Fail closed: a status this build does not know gets the safety hub (the
 * `unsupported` row) — never an accusation, never a reassurance. */
export function safetyHandoff(status: unknown): SafetyHandoff {
  return (typeof status === "string" ? SAFETY_HANDOFF[status as TrustStatus] : undefined) ?? SAFETY_HANDOFF.unsupported;
}

/** I3-0375 (D-0992(7)) — the warning's dispute door, painted in ADDITION to the

/**
 * Verdict presentation for the extension — the extension's mirror of
 * `apps/mobile-check/src/verdict-presentation.ts` and `apps/web/src/verdict.ts`.
 *
 * WHY IT EXISTS. The popup rendered every verdict as one line of `#333` text.
 * All seven states — Verified, Claimed, Unclaimed, Reported, Caution, Not
 * listed, Couldn't check — were the same colour, the same weight and the same
 * shape. There was no icon anywhere. Measured: rendering NO VERDICT AT ALL
 * passed the whole extension suite, because nothing asserted that the popup
 * showed anything.
 *
 * That is worse in the popup than anywhere else. The popup is what a fan opens
 * *deliberately*, on the page they are about to trust with money — it is the one
 * surface where the answer is the entire product, and it was rendering the
 * strongest warning this system can issue in the same grey as "not listed".
 *
 * TWO PALETTES, ONE VOCABULARY. The popup is our own surface, so it runs the
 * design system dark (docs/DESIGN-SYSTEM.md §3 is dark-first) and uses the reserved
 * token hexes UNCHANGED — the same bytes the web app and both native apps paint.
 * The on-page badge is not our surface: it renders on a white card over
 * somebody else's site (§2, "embeddable surfaces can't assume a dark
 * background"), where #18e6a0 on white is about 1.6:1 and unreadable. So the
 * badge uses ON_LIGHT — one documented darkened variant PER TOKEN, never a
 * second vocabulary. Every status still resolves through the same `token` field,
 * so "claimed is amber" holds on both, and `test/verdict-render.test.ts:86` asserts ON_LIGHT is
 * total over TOKEN so a status can never acquire a colour with no token behind
 * it.
 */

/** Wire → kind. ANYTHING ELSE IS A PERSON — which is what every entry meant
 * before characters existed, so a bucket without the field, or with a value this
 * build does not know, under-claims rather than inventing a disclosure. */
export function kindFromWire(k: unknown): SubjectKind | undefined {
  return k === "a" ? "ai_character" : k === "b" ? "brand" : undefined;
}

/** The consumer-side narrowing: the LONG form, which is what `CheckResult.kind`
 * carries once `kindFromWire` has done its job at the bucket-parse boundary.
 * Calling `kindFromWire` on a long-form value returns nothing and renders
 * nothing — silently, which is why these are two functions and not one. */
export function asSubjectKind(v: unknown): SubjectKind | undefined {
  return v === "ai_character" || v === "brand" ? v : undefined;
}

export const UNCONFIRMED_STYLE: Partial<Record<TrustStatus, ExtVerdictStyle>> = {
  verified: { label: EN["ext.label_verified_unconfirmed"], glyph: "✓", token: "verified", plain: EN["check.plain_verified_unconfirmed"] },

/**
 * Resolve a status to its style, failing closed. An unrecognised status — a
 * registry newer than this build, a mangled message — resolves to `failed`,
 * never to `unknown`: an extension that cannot interpret the answer has not
 * checked anything, and must not say "not listed" about it.
 *
 * `confirmed` defaults to false, so every caller that has not been told the
 * entry confirmed this account gets the weaker sentence by construction.
 *
 * I63 K1 (Jason D-1537 I62-Q5; I45-8's words, D-1522): a warning whose entry carried a known kind code (`i` / `f`,
 * CheckResult.warningKind) is headed by that kind's label; glyph, token and plain sentence stay CHECK_STYLE.warning's.
 * No code, or any other code, keeps the kind-less label.
 */
export function verdictStyle(status: unknown, confirmed = false, warningKind?: unknown): ExtVerdictStyle {
  const key = typeof status === "string" ? (status as TrustStatus) : undefined;
  if (key === "warning" && (warningKind === "i" || warningKind === "f")) return { ...CHECK_STYLE.warning, label: WARNING_KIND_LABEL[warningKind] };
  const weaker = key !== undefined && !confirmed ? UNCONFIRMED_STYLE[key] : undefined;
  return weaker ?? (key !== undefined ? CHECK_STYLE[key] : undefined) ?? CHECK_STYLE.failed;
}

/** The popup's colour (dark surface → the reserved token, unchanged). The
 * confirmation flag never moves a colour, only a sentence. */
export function popupColor(status: unknown): string {
  return TOKEN[verdictStyle(status).token];
}

/** The on-page badge's colour (white card on a host page → the light variant). */
export function badgeColor(status: unknown): string {
  return ON_LIGHT[verdictStyle(status).token];
}

/**
 * QUEUE #204 — THE TOOLBAR BADGE, THE ONE SURFACE THAT SAID IT IN COLOUR ONLY.
 *
 * `CHECK_STYLE` gives every verdict a distinct GLYPH and a distinct WORD, and
 * the four in-page clients emit both. The toolbar action badge emitted `"!"` for
 * `warning` AND for `caution` and changed only `setBadgeBackgroundColor` between
 * them — so the difference between "an operator published an impersonation
 * report about this account" and "nobody has proved this account yet" was
 * carried by hue alone. That is WCAG 1.4.1, and it is not a cosmetic failure
 * here: `watch.ts levelOf` puts these at different rungs precisely because
 * collapsing them "made the single most important transition in the product
 * silent". It was also, specifically, WRONG: `"!"` is `caution`'s glyph in the
 * table above, so the badge was labelling the stronger state with the weaker
 * state's mark.
 *
 * WHY THIS LIVES HERE. Same reason `badgeColor` does: the badge vocabulary is
 * part of the verdict vocabulary, and a second table in background.ts is how
 * two surfaces start disagreeing about what `caution` looks like. The TITLE is
 * read off `verdictStyle(status, true, warningKind).label` — `CHECK_STYLE[status].label`,
 * or for a warning with a known kind code that kind's heading (I64 J2, D-1553) — this module's docblock already
 * claims "the label is this product's vocabulary and is what the toolbar badge
 * and the verdict line carry", which was true of the verdict line and not of the
 * badge until now.
 *
 * WHAT A TOOLBAR BADGE CAN ACTUALLY RENDER, and why the glyphs are not simply
 * the table's. `chrome.action.setBadgeText` paints into a small pill in browser
 * chrome: Chrome's own documentation says only about four characters fit, the
 * text is rasterized by the browser at roughly 8px in a system font we do not
 * choose, and there is no fallback we can observe or test from the extension —
 * a character the platform lacks is tofu or an unpredictable colour-emoji
 * substitution, silently, on the most important signal the product emits.
 * `CHECK_STYLE.warning.glyph` is `⚠` (U+26A0), whose presentation is AMBIGUOUS
 * by Unicode default: the same codepoint is a thin monochrome triangle on one
 * platform and a full-colour emoji on the next, and at badge size the emoji
 * form is mush. So the badge alphabet is restricted to ASCII, the only text
 * guaranteed to rasterize identically in every browser build:
 *
 *   caution → "!"  the table's own caution glyph, unchanged. The badge now
 *                  AGREES with the other four clients about `caution` instead
 *                  of contradicting them about `warning`.
 *   warning → "X"  ASCII, one character, and a wholly different letterform —
 *                  not a repetition of "!" (`"!!"` would differ only in LENGTH,
 *                  which at 8px is close to the same failure in a new form).
 *                  `X` is deliberately NOT in the verdict glyph set
 *                  (✓ ◐ ○ ⚠ ! ? ↻), so it collides with no other verdict's
 *                  meaning, and it is the universally-read counterpart of the
 *                  `✓` this very table already uses for `verified`.
 *
 * AND THE WORD CHANNEL IS NOT OPTIONAL. Two ASCII marks at 8px satisfy 1.4.1
 * literally, but they do not NAME anything, and the badge has no room to. The
 * action surface has a second text channel that is not the pill —
 * `chrome.action.setTitle` — which is unconstrained, is what a screen reader
 * announces for the toolbar button, and is what a hover reveals. That carries
 * the full `CHECK_STYLE` label, so the state is available as real words on the
 * same surface. Glyph AND word, as on every other client; the surface just
 * splits them across two calls.
 */
export interface ToolbarBadge {
  /** ASCII only, ≤4 chars — see the docblock. Never the empty string. */
  text: string;
  /** `verdictStyle(status, true, warningKind).label`, verbatim: the `CHECK_STYLE` label, or a known kind's heading. The word channel. */
  title: string;
  color: string;
}

const TOOLBAR_TEXT: Partial<Record<TrustStatus, string>> = {
  warning: "X",
  caution: "!",
};

/**
 * The toolbar badge for a verdict, or `null` for a verdict that must not light
 * it. `null` is the answer for everything outside `BADGE_STATUSES` — staying
 * silent is correct there, and a caller that clears the text must clear the
 * TITLE too, or a stale word outlives the mark it belonged to.
 */
export function toolbarBadge(status: unknown, warningKind?: unknown): ToolbarBadge | null {
  const text = typeof status === "string" ? TOOLBAR_TEXT[status as TrustStatus] : undefined;
  if (text === undefined) return null;
  // I64 J2 (D-1553): the title goes through verdictStyle's own kind predicate (:290), never a second copy.
  return { text, title: verdictStyle(status, true, warningKind).label, color: badgeColor(status) };
}

const HASH_RE = /^[0-9a-f]{64}$/;
const MAX_LABEL = 64;
/** The top rung `levelOf` can return. Used to reject a stored level that no
 * verdict could have produced. */
export const MAX_LEVEL = 3;

/**
 * A verdict's INFORMATION LEVEL, or `null` for a result that is not one.
 *
 *   0  `unknown`      Dhwaja looked and has no record
 *   1  listed         unclaimed / claimed / verified
 *   2  `caution`      a machine signal — looks like a creator, nobody proved it
 *   3  `warning`      an operator-PUBLISHED impersonation report
 *
 * CAUTION AND WARNING ARE NOT THE SAME RUNG, and collapsing them (the first
 * version of this did) made the single most important transition in the product
 * silent: a fan watching an account sitting at `caution` — which is exactly the
 * population that later earns a warning — would never be told when the warning
 * published. `check.ts` calls `warning` "the single most important state to
 * show"; a watch that cannot show it is not worth having.
 *
 * `unsupported` and `failed` are `null`, and that is the load-bearing case.
 * `check.ts` separates "we looked and found nothing" from "we could not look"
 * (flag 23) precisely so that an unreachable registry is never read as an
 * answer. Folding `failed` into 0 would fire a false "now listed!" on the next
 * success and silently discard a real level on the way down.
 */
export function levelOf(status: TrustStatus): number | null {
  switch (status) {
    case "unknown":
      return 0;
    case "unclaimed":
    case "claimed":
    case "verified":
      return 1;
    case "caution":
      return 2;
    case "warning":
      return 3;
    default:
      return null;
  }
}

/**
 * Does this transition earn a notification?
 *
 * ONLY A STRICT INCREASE. The decision, taken before this was built: fire on
 * "not listed → listed" and on "listed → flagged", and say NOTHING when a
 * subject goes the other way.
 *
 * A removal is usually a person exercising a removal right, and `removal.ts` is
 * built end to end so that a delisting is not detectable — never re-seeded, no
 * oracle, constant work. A watch that announced removals would hand every fan
 * holding one a cheap, continuous, automated delisting detector aimed at the
 * person who asked to be removed. This is the one answer the feature refuses to
 * give, on purpose.
 */
export function shouldNotify(prev: number, next: number | null): boolean {
  return next !== null && next > prev;
}

/**
 * Mark a row as VISITED without believing anything about it.
 *
 * Used for a check that failed and for a notification that could not be shown.
 * The first version left `checkedAt` alone on a failure, reasoning that a clock
 * should not record a check that never happened — which is true and was the
 * wrong call, because `sweepSlice` orders by that clock. A row that always
 * fails then sits at the head of the queue forever, and five of them starve the
 * other fifteen permanently. `checkHash` returns `failed` for any hash the
 * registry serves under a class this build does not know, so that is a
 * per-subject freeze the SERVER can choose to apply to every fan's list.
 *
 * Rotating is the lesser evil: the row is retried on the next full cycle, and
 * nothing about the level is asserted from a check that did not happen.
 */
export function touch(row: WatchRow, now: number): WatchRow {
  return { ...row, checkedAt: now };
}

/**
 * The row after a believed check.
 *
 * A DECREASE IS APPLIED AND SILENT, and the row SURVIVES it. "Forget the watch"
 * was the obvious reading of going quiet, and it is wrong: deleting the row
 * shrinks the fan's list, and they can see which entry left. The deletion would
 * BE the disclosure. So the level drops, the row stays, and the list renders no
 * per-watch verdict at all — it shows what you are watching, never what Dhwaja
 * currently says about it. That is what makes the silence silent.
 */
export function advance(row: WatchRow, level: number, now: number): WatchRow {
  return { ...row, level, checkedAt: now };
}

/** Trim a label to something a notification and a list row can hold. Plain text
 * everywhere it lands (`chrome.notifications` takes strings, the popup uses
 * `textContent`), so this is a length bound, not an escaping step. */
function label(v: unknown): string {
  return typeof v === "string" ? v.slice(0, MAX_LABEL) : "";
}

/**
 * Rows we are willing to act on, from whatever was in storage.
 *
 * Storage is writable by anything holding this extension's context, and a
 * corrupt or hand-edited row must not reach a fetch. Anything without a
 * well-formed hash or a level in range is DROPPED rather than repaired: a
 * repaired row is a guess about what a fan asked to watch.
 */
export function sanitizeRows(raw: unknown): WatchRow[] {
  if (!Array.isArray(raw)) return [];
  const out: WatchRow[] = [];
  const seen = new Set<string>();
  for (const r of raw) {
    if (!r || typeof r !== "object") continue;
    const { hash, platform, handle, level, addedAt, checkedAt } = r as Record<string, unknown>;
    if (typeof hash !== "string" || !HASH_RE.test(hash) || seen.has(hash)) continue;
    if (typeof level !== "number" || !Number.isInteger(level) || level < 0 || level > MAX_LEVEL) continue;
    seen.add(hash);
    out.push({
      hash,
      platform: label(platform),
      handle: label(handle),
      level,
      addedAt: typeof addedAt === "number" && Number.isFinite(addedAt) ? addedAt : 0,
      checkedAt: typeof checkedAt === "number" && Number.isFinite(checkedAt) ? checkedAt : 0,
    });
    // The cap is enforced on READ as well as on write: a list that grew by some
    // other route still cannot drive more fetches than one that did not.
    if (out.length >= MAX_WATCHES) break;
  }
  return out;
}

/** Add one, or return the list unchanged. Full is not an error the caller has to
 * distinguish — `added` says what happened and the popup says it in words. */
export function addRow(rows: WatchRow[], row: WatchRow): { rows: WatchRow[]; added: boolean } {
  if (rows.some((r) => r.hash === row.hash)) return { rows, added: true }; // idempotent
  if (rows.length >= MAX_WATCHES) return { rows, added: false };
  return { rows: [...rows, row], added: true };
}

export function removeRow(rows: WatchRow[], hash: string): WatchRow[] {
  return rows.filter((r) => r.hash !== hash);
}

/**
 * Least-recently-checked first — the ordering a list-driven sweep used.
 *
 * NO LONGER ON THE SWEEP PATH. Slice 30 chooses fetches from the cover set
 * rather than from the list, precisely so the choice cannot depend on the list;
 * an ordering over watched rows is the shape of the defect. Kept, and kept
 * tested, because it is the honest answer to "which rows are most overdue" and
 * the popup will want it the day the list gets a staleness column — but if that
 * day does not come, this is dead weight and should go rather than sit here
 * looking like a sweep primitive.
 */
export function sweepSlice(rows: WatchRow[], budget = SWEEP_BUDGET): WatchRow[] {
  return [...rows].sort((a, b) => a.checkedAt - b.checkedAt).slice(0, Math.max(0, budget));
}

/** A watch row from a check the fan just ran. `null` when the result carries
 * nothing to watch — an unsupported page, or a check that failed. */
export function rowFromResult(result: CheckResult, hash: string, now: number): WatchRow | null {
  const level = levelOf(result.status);
  if (level === null) return null;
  if (!HASH_RE.test(hash)) return null;
  return { hash, platform: label(result.platform), handle: label(result.handle), level, addedAt: now, checkedAt: now };
}

/** The sentence a fired watch shows. Plain text, and only the two label fields —
 * never a URL, which is the one thing this whole architecture keeps on device. */
export function notificationText(row: WatchRow, level: number, showAccount = false): { title: string; message: string } {
  // I3-1443 a (D-0993(1) pick A): a desktop banner is readable by whoever stands near the screen, and the
  // account may be a scammer or an abuser they should not learn is being watched. The DEFAULT names neither
  // platform nor handle; `showAccount` is the fan's opt-in (`readShowAccount`).
  const who = showAccount && row.handle ? `${row.platform ? `${row.platform} · ` : ""}${row.handle}` : "An account you're watching";
  if (level >= 3) return { title: levelTitle(level), message: `${who} — don't send money. Open Dhwaja.` };
  if (level === 2) return { title: levelTitle(level), message: `${who} — look more carefully before sending anything.` };
  return { title: levelTitle(level), message: `${who} — there's a record to read now.` };
}

function levelTitle(level: number): string {
  if (level >= 3) return "Dhwaja: an account you're watching was reported";
  if (level === 2) return "Dhwaja: an account you're watching was flagged";
  return "Dhwaja: an account you're watching is now listed";
}

// --- the four chrome wrappers -----------------------------------------------
//
// EVERY ONE IS GUARDED. These permissions are optional and a fan can revoke them
// in browser settings at any moment, which makes `chrome.storage` and
// `chrome.alarms` genuinely `undefined` at runtime rather than merely empty. The
// honest response to a revoked grant is to do nothing — never to half-run.

export function hasWatchApis(): boolean {
  return typeof chrome !== "undefined" && !!chrome.storage?.local && !!chrome.alarms;
}

/**
 * The watch list, or `null` when it could not be READ (grant revoked, or
 * `chrome.storage.local.get` rejected). A readable store with no list is `[]`.
 *
 * Every caller that WRITES a list derived from what it read must use this and
 * refuse the write on `null`: `saveWatches` replaces the whole key, so a failed
 * read treated as empty erases the fan's list (#173).
 */
export async function readWatches(): Promise<WatchRow[] | null> {
  if (!hasWatchApis()) return null;
  try {
    const got = await chrome.storage.local.get(STORAGE_KEY);
    return sanitizeRows((got as Record<string, unknown>)[STORAGE_KEY]);
  } catch {
    return null;
  }
}

/** Read-only callers only: an unreadable list is treated as empty, which skips
 * one sweep or re-arm and writes nothing. Its result must never reach
 * `saveWatches`. background.ts's sweep write-back and `handleWatch` add/remove
 * call `readWatches` and skip the write on `null` (#173, fixed in B23-B2). */
export async function loadWatches(): Promise<WatchRow[]> {
  return (await readWatches()) ?? [];
}

export async function saveWatches(rows: WatchRow[]): Promise<boolean> {
  if (!hasWatchApis()) return false;
  try {
    await chrome.storage.local.set({ [STORAGE_KEY]: rows.slice(0, MAX_WATCHES) });
    return true;
  } catch {
    return false;
  }
}

/**
 * The persisted cover set (slice 30). Same guarded shape as `loadWatches`, for
 * the same reason: these are optional permissions a fan can revoke at any
 * moment, and the honest response to a revoked grant is to do nothing.
 *
 * A cover that cannot be READ is `null`, which `coverIsStale` treats as stale
 * and the sweep rebuilds. A cover that cannot be WRITTEN is the interesting
 * case: the sweep uses the one it just built anyway, so the wake is still
 * covered, and the next wake rebuilds. That is a rebuild more often than
 * designed — the residual is defect (2) reappearing for as long as storage is
 * broken — and it is strictly better than sweeping the watch list bare.
 */
export async function loadCover(width: number): Promise<CoverState | null> {
  if (!hasWatchApis()) return null;
  try {
    const got = await chrome.storage.local.get(COVER_KEY);
    return sanitizeCover((got as Record<string, unknown>)[COVER_KEY], width);
  } catch {
    return null;
  }
}

export async function saveCover(state: CoverState): Promise<boolean> {
  if (!hasWatchApis()) return false;
  try {
    await chrome.storage.local.set({ [COVER_KEY]: state });
    return true;
  } catch {
    return false;
  }
}

/** Minutes until the next sweep: the period, plus up to a quarter of it. */
export function nextSweepDelay(random: () => number = Math.random): number {
  return SWEEP_PERIOD_MINUTES + random() * (SWEEP_PERIOD_MINUTES / 4);
}

/**
 * Arm the next sweep — ONE SHOT, re-armed after every fire.
 *
 * Not `periodInMinutes`. A fixed period jitters only the FIRST fire and then
 * beats on an exact 360-minute clock forever, which is a stable per-install
 * phase offset and a traffic shape nothing else in this extension produces. A
 * one-shot alarm re-armed with fresh jitter each time has neither.
 *
 * `force: false` (the default) is the other half: `chrome.alarms.create` with an
 * existing name REPLACES it, so re-arming on every add let a fan who watched
 * several accounts in an afternoon push the first fire out indefinitely and
 * never sweep at all. Adding a watch must not reset the clock.
 */
export async function armSweep(random: () => number = Math.random, force = false): Promise<void> {
  if (!hasWatchApis()) return;
  try {
    if (!force && (await chrome.alarms.get(WATCH_ALARM))) return;
    await chrome.alarms.create(WATCH_ALARM, { delayInMinutes: nextSweepDelay(random) });
  } catch {
    /* revoked between the check and the call — nothing to arm */
  }
}


/**
 * What `watch:list` is allowed to say.
 *
 * The LEVEL never crosses the message boundary. The safety argument for this
 * whole feature is that the list renders no per-watch verdict — and a `level`
 * sitting in the reply makes that a promise about restraint at the render site
 * rather than a property of the data. `level` is also exactly the delisting bit
 * the design refuses to disclose, so it stays in the worker.
 */
export interface WatchView {
  hash: string;
  platform: string;
  handle: string;
  addedAt: number;
}

export function toView(rows: WatchRow[]): WatchView[] {
  return rows.map((r) => ({ hash: r.hash, platform: r.platform, handle: r.handle, addedAt: r.addedAt }));
}

import { bucketPrefix, canonicalize, type CanonicalSubject } from "@dhwaja/canonical";
import { kindFromWire, type SubjectKind, type TrustClass, type TrustStatus, type WarningKindCode, warningKindFromWire } from "./verdict-presentation.js";
import { inclusionApply, inclusionPlan, MAX_CLIENT_WIDTH, verificationAction, versionAccepted, type InclusionMaterial, type InclusionResult, type PolicyAction, type SignatureOutcome } from "@dhwaja/registry-proof/registry-proof";

/**
 * Mobile check core — the SAME canonicalizer as every other surface, exposed as
 * three PURE, SYNC functions for an on-device JS engine. BOTH native mobile
 * clients run this one bundle: iOS in JavaScriptCore (a JSContext) and Android
 * in the androidx.javascriptengine sandbox (an isolated JS process). Each app
 * provides the two platform primitives the core deliberately does NOT contain:
 * SHA-256 (CryptoKit / java.security.MessageDigest) and the network fetch
 * (URLSession / OkHttp, registry-host only, cert-pinned). So the canonical
 * logic is shared JS; neither iOS nor Android re-implements it.
 *
 *   native: canonical = DhwajaCheck.canonicalize(url)      // pure JS
 *           if null → unsupported
 *           hash    = SHA256(canonical) as hex             // platform primitive
 *           width   = GET /v1/info                          // pinned fetch
 *           prefix  = DhwajaCheck.bucketPrefix(hash,width)  // pure JS (capped)
 *           bucket  = GET /v1/bucket/<prefix>               // pinned fetch
 *           status  = DhwajaCheck.matchBucket(bucket,hash)  // pure JS
 *
 * Privacy: only `prefix` ever leaves the device. The full hash/handle/URL do
 * not. Width is client-capped (rogue-registry defense) and under-padded
 * buckets are distrusted — identical rules to the browser extension.
 */

export { MAX_CLIENT_WIDTH, type TrustClass, type TrustStatus };

export interface CanonResult {
  canonical: string | null; // null → not a checkable profile
  platform?: string;
  handle?: string;
}

/** Canonicalize a shared URL. Returns null for non-profile / URL-fallback. */
export function canonicalizeForCheck(url: string): CanonResult {
  let subject: CanonicalSubject;
  try {
    subject = canonicalize(url);
  } catch {
    return { canonical: null };
  }
  if (subject.platform === "url" || subject.idKind === "norm") return { canonical: null };
  return { canonical: subject.canonical, platform: subject.platform, handle: subject.id };
}

/**
 * Prefix for the served width, capped. Returns null if the width is out of the
 * safe range (a rogue registry serving an over-wide width → the caller treats
 * it as unknown and sends NO wide prefix). `hashHex` is the CryptoKit digest.
 */
export function bucketPrefixFor(hashHex: string, servedWidth: number): string | null {
  if (!Number.isInteger(servedWidth) || servedWidth < 1 || servedWidth > MAX_CLIENT_WIDTH) return null;
  if (!/^[0-9a-f]{64}$/.test(hashHex)) return null;
  return bucketPrefix(hashHex, servedWidth);
}

/** Find-the-real-one: the verified creator an impersonation warning protects.
 * p = passport id (strict shape), n = display name (clamped; native renders it
 * as TEXT — never markup/format-string input). */
export interface RealOne {
  p: string;
  n: string;
}

/**
 * THE passport-id shape, shared by every client and by the server's own pointer
 * validator (`PASSPORT_ID_RE` in services/trust-core/src/passport-view.ts).
 *
 * It used to be two shapes on one code path: this client accepted 8–64
 * alphanumerics, the SPA's router required exactly 16 hex, and the pointer they
 * pass between them is the redirect a frightened fan is offered on a Reported
 * verdict. A widened id would have been accepted here, rendered as "See the real
 * X on Dhwaja →", and then silently collapsed to the home page on arrival. A
 * shape mismatch must fail at the VALIDATOR, where the answer is "no pointer"
 * and the warning still stands.
 */
export const PASSPORT_ID_RE = /^psp_[0-9a-f]{16}$/;
const PSP_RE = PASSPORT_ID_RE;

/** The soft pointer on a warning OR caution entry, strict shapes; anything
 * off-shape → no pointer, never junk. Strip/clamp FIRST, then require a non-empty
 * name — a name that is all markup (e.g. "<><>") must yield NO pointer, not an
 * empty label. */
function realFromEntry(e: RawEntry): RealOne | undefined {
  const rr = e.r as { p?: unknown; n?: unknown } | null | undefined;
  if (!rr || typeof rr !== "object" || typeof rr.p !== "string" || !PSP_RE.test(rr.p) || typeof rr.n !== "string") return undefined;
  const name = rr.n.replace(/[<>]/g, "").slice(0, 64).trim();
  return name.length > 0 ? { p: rr.p, n: name } : undefined;
}

/**
 * Match the full hash against a bucket. A match REQUIRES full-hash equality, so
 * a hostile/garbled bucket can only fail to match — never forge a positive.
 * `minEntries` distrusts an under-padded bucket — a registry not padding to its
 * k-floor is not following the protocol, so its answers are not believed. (A
 * compliance check, not an anonymity lever: padding buys nothing against the
 * server. See docs/REGISTRY-PRIVACY-CLAIMS.md.) Input is the bucket JSON text
 * (as URLSession returns it) to keep the Swift bridge simple.
 */
/** Safe-by-default k-floor: a host that forgets to set one still distrusts a
 * thin/non-padded bucket. Tests that use small fixtures pass 0 explicitly.
 * I3-0241 / D-0882(2): the server's published pad floor (`K_FLOOR`,
 * services/trust-core/src/registry-build.ts) — no honest bucket is smaller. */
export const DEFAULT_MIN_BUCKET = 1000;

/** D-0708 — the `/v1/info` manifest formats this reader understands (see
 * `versionAccepted`). The builder writes `MANIFEST_FORMAT_VERSION` in
 * services/trust-core/src/registry-build.ts; test/b27-vergate.test.ts couples them. */
export const MANIFEST_VERSIONS_UNDERSTOOD: readonly number[] = [3];
/** D-0708 — the bucket-body formats this reader understands (`BUCKET_FORMAT_VERSION`). */
export const BUCKET_VERSIONS_UNDERSTOOD: readonly number[] = [1];

/**
 * D-0708 — the served width out of a `/v1/info` text, or null when the text is
 * unreadable, the width is out of range, or the manifest's version is one this
 * reader does not understand (`versionAccepted`). Null is the natives' existing
 * "response unreadable" path, which renders `failed`.
 */
export function widthFromInfo(infoText: string): number | null {
  let body: unknown;
  try {
    body = JSON.parse(infoText) as unknown;
  } catch {
    return null;
  }
  if (!versionAccepted(body, MANIFEST_VERSIONS_UNDERSTOOD)) return null;
  const width = (body as { width?: unknown }).width;
  return Number.isInteger(width) && (width as number) >= 1 && (width as number) <= MAX_CLIENT_WIDTH ? (width as number) : null;
}

export function matchBucket(bucketJsonText: string, hashHex: string, minEntries: number = DEFAULT_MIN_BUCKET): TrustStatus {
  return matchBucketDetail(bucketJsonText, hashHex, minEntries).status;
}

export interface MatchDetail {
  status: TrustStatus;
  /**
   * Did the matched entry say that THIS account's own control proof is current
   * (`p` on the bucket entry)?
   *
   * Only meaningful on `verified`/`claimed`, and ABSENT means NOT confirmed —
   * an older bucket, a registry that stops sending the flag, or an account
   * nobody has ever proved all land on the weaker, passport-level sentence. A
   * client must never infer confirmation from silence.
   */
  confirmed?: boolean;
  /**
   * WHAT THE SUBJECT IS, when the matched entry said so. Absent means a person —
   * which is what every entry meant before characters existed, so an older
   * bucket, a registry that stops sending it, and a real human creator all land
   * on the same (weaker) rendering.
   */
  kind?: SubjectKind;
  /** Present only on a "warning" or "caution" match whose entry carried a valid
   * pointer. */
  real?: RealOne;
  /** I3-0122: the warning's kind code (`w` on the entry), present only on a
   * `warning` match and only as `"i"`/`"f"`. Absent = the fixed sentence
   * (`warningWordingText` in verdict-presentation.ts). */
  warningKind?: WarningKindCode;
  /** I3-0526 (D-1078(13)): the Founding rank (`f` on the entry), present only on a `verified` match and only as an
   * integer 1..500 (trust-core founding.ts FOUNDING_CAPACITY). Absent = no Founding line, which under-claims. */
  founding?: number;
}

/** matchBucket plus the find-the-real-one pointer. Separate function so the
 * long-standing string-out matchBucket contract (and its native callers) stays
 * byte-identical; new native code uses the matchBucketInfo shim instead. */
export function matchBucketDetail(bucketJsonText: string, hashHex: string, minEntries: number = DEFAULT_MIN_BUCKET): MatchDetail {
  // flag 23: anything that prevents a trustworthy look at the bucket is
  // "failed" (asserts nothing), never "unknown" (asserts "no record").
  if (!/^[0-9a-f]{64}$/.test(hashHex)) return { status: "failed" };
  let parsed: { entries?: unknown };
  try {
    parsed = JSON.parse(bucketJsonText) as { entries?: unknown };
  } catch {
    return { status: "failed" };
  }
  // D-0708: a bucket format this reader does not understand — or one carrying
  // no version at all — is not read. `failed`, never `unknown`.
  if (!versionAccepted(parsed, BUCKET_VERSIONS_UNDERSTOOD)) return { status: "failed" };
  const raw = parsed.entries;
  if (!Array.isArray(raw)) return { status: "failed" };
  // Non-padding registry → distrust. I3-0236: counted over DISTINCT well-formed
  // hashes (case-folded, any class), never raw rows — junk or a repeated hash
  // must not buy compliance; the builder never serves a duplicate (registry-build.ts:1800).
  // I12-RG-PRIVACY-03: a non-finite or negative floor is a host bug, never a way
  // to turn the floor off — it falls back to the default. 0 stays the documented
  // off switch for JS callers with small fixtures; the NATIVE door cannot reach it
  // (`nativeBridge` clamps to at least DEFAULT_MIN_BUCKET, D-0883(6)).
  const floor = Number.isFinite(minEntries) && minEntries >= 0 ? minEntries : DEFAULT_MIN_BUCKET;
  if (floor > 0) {
    const seen = new Set<string>();
    for (const e of raw as RawEntry[]) {
      const h = e && typeof e.h === "string" ? e.h.toLowerCase() : "";
      if (/^[0-9a-f]{64}$/.test(h)) seen.add(h);
    }
    if (seen.size < floor) return { status: "failed" };
  }
  for (const e of raw as RawEntry[]) {
    if (e && typeof e.h === "string" && e.h.toLowerCase() === hashHex) {
      const c = e.c;
      if (c === "verified" || c === "claimed" || c === "unclaimed") {
        // Strictly `1`: anything else — `true`, "1", 0, a missing field — is not
        // a confirmation this client will repeat to a fan.
        // The KIND rides the same entry — no second fetch, for the reason the
        // confirmation flag does not have one either: asking the server about a
        // specific subject after a local match is exactly what the prefix design
        // exists to prevent. Anything the wire vocabulary does not name reads as
        // a person, which under-claims.
        const kind = kindFromWire(e.k);
        const founding = c === "verified" && typeof e.f === "number" && Number.isInteger(e.f) && e.f >= 1 && e.f <= 500 ? e.f : undefined;
        return { status: c, confirmed: e.p === 1, ...(kind ? { kind } : {}), ...(founding !== undefined ? { founding } : {}) };
      }
      if (c === "warning") {
        const real = realFromEntry(e);
        const warningKind = warningKindFromWire(e.w);
        return { status: "warning", ...(real ? { real } : {}), ...(warningKind ? { warningKind } : {}) };
      }
      // "caution" is the soft class: amber, weaker than a warning, carrying the
      // same optional soft real-creator pointer.
      if (c === "caution") {
        const real = realFromEntry(e);
        return real ? { status: "caution", real } : { status: "caution" };
      }
      // A hit whose class this client doesn't know: the registry is newer than
      // this build — the answer can't be interpreted, so the check FAILED
      // (asserting "not listed" about a listed subject is the worst lie here).
      return { status: "failed" };
    }
  }
  return { status: "unknown" };
}

/** Host interface for a full resolve (tests + any JS host). Swift orchestrates
 * these steps itself; this convenience mirrors that orchestration exactly. */
export interface CheckHost {
  hash(canonical: string): Promise<string>; // hex sha256 (CryptoKit on iOS)
  getWidth(): Promise<number | null>; // GET /v1/info → width
  getBucketText(prefix: string): Promise<string | null>; // GET /v1/bucket/<prefix>
  minEntries?: number;
}

export interface CheckResult {
  status: TrustStatus;
  platform?: string;
  handle?: string;
  /** Whether the matched entry confirmed THIS account — see `MatchDetail`.
   * Absent/false => the passport-level sentence. */
  confirmed?: boolean;
  /** What the subject is — see `MatchDetail.kind`. Absent = a person. */
  kind?: SubjectKind;
  /** Find-the-real-one (warning or caution hits): who is really behind the name. */
  real?: RealOne;
  /** I3-0122: the warning's kind code — see `MatchDetail.warningKind`. */
  warningKind?: WarningKindCode;
  /** I3-0526: the Founding rank — see `MatchDetail.founding`; shown as "Founding Creator #N" (verdict-presentation.ts `foundingLabel`). */
  founding?: number;
}

function withMeta(status: TrustStatus, canon: CanonResult, real?: RealOne, confirmed?: boolean, kind?: SubjectKind, warningKind?: WarningKindCode, founding?: number): CheckResult {
  return {
    status,
    ...(canon.platform ? { platform: canon.platform } : {}),
    ...(canon.handle ? { handle: canon.handle } : {}),
    ...(confirmed ? { confirmed: true } : {}),
    ...(kind ? { kind } : {}),
    ...(real ? { real } : {}),
    ...(warningKind ? { warningKind } : {}),
    ...(founding !== undefined ? { founding } : {}),
  };
}

export async function resolveStatus(url: string, host: CheckHost): Promise<CheckResult> {
  const canon = canonicalizeForCheck(url);
  if (!canon.canonical) return { status: "unsupported" };
  const hashHex = await host.hash(canon.canonical);
  const width = await host.getWidth();
  // flag 23: an unreachable registry / unbuildable prefix / missing bucket is
  // a FAILED check ("couldn't look"), never "unknown" ("looked; no record").
  if (width === null) return withMeta("failed", canon);
  const prefix = bucketPrefixFor(hashHex, width);
  if (prefix === null) return withMeta("failed", canon);
  const text = await host.getBucketText(prefix);
  if (text === null) return withMeta("failed", canon);
  const detail = matchBucketDetail(text, hashHex, host.minEntries ?? DEFAULT_MIN_BUCKET);
  return withMeta(detail.status, canon, detail.real, detail.confirmed === true, detail.kind, detail.warningKind, detail.founding);
}

export interface CheckFunnel {
  /** Nudge the fan into `#/discover` (a positive verdict has no id to link). */
  discover: boolean;
  /** `#/p/<psp>` when a warning/caution carried a valid pointer; else null. */
  passportHash: string | null;
}

/**
 * The post-check funnel — IDENTICAL rules to the browser extension's
 * `checkFunnel` (`apps/extension/src/check.ts`), so the native share-sheet leads
 * the fan to the same next step as the extension badge:
 *  - a `warning`/`caution` hit carrying a valid real-creator pointer → that
 *    verified creator's profile hash (`#/p/<psp>`);
 *  - a positive verdict (`verified`/`claimed`) → discovery (`#/discover`).
 * Pure + string-safe: a hash is returned ONLY for a `PSP_RE`-validated id, so the
 * native side can join it to its app base and never render a junk deep link. */
export function checkFunnel(result: CheckResult): CheckFunnel {
  const real = result.real;
  const pointered = (result.status === "warning" || result.status === "caution") && real !== undefined && PSP_RE.test(real.p);
  return {
    discover: result.status === "verified" || result.status === "claimed",
    passportHash: pointered ? `#/p/${real!.p}` : null,
  };
}

/** Parse a MatchDetail (`{status, real?}`) JSON string — as produced by
 * `matchBucketInfo` — and compute the funnel. Tolerant of junk: an unparseable
 * or off-shape input yields the inert `{discover:false, passportHash:null}`
 * (never throws into the native caller). */
function funnelFromDetailJson(detailJsonText: string): CheckFunnel {
  let parsed: unknown;
  try {
    parsed = JSON.parse(detailJsonText);
  } catch {
    return { discover: false, passportHash: null };
  }
  // JSON.parse can yield null / an array / a scalar — anything non-object is inert.
  if (!parsed || typeof parsed !== "object") return { discover: false, passportHash: null };
  const d = parsed as { status?: unknown; real?: unknown };
  const status = (typeof d.status === "string" ? d.status : "unknown") as TrustStatus;
  // Reconstruct `real` fail-closed: require BOTH string fields (matchBucketInfo,
  // the only real producer, always emits a non-empty `n` or omits `real`), so a
  // partial `{p}` yields no pointer. `checkFunnel` then re-tests PSP_RE on `p`.
  const rr = d.real as { p?: unknown; n?: unknown } | null | undefined;
  const real = rr && typeof rr === "object" && typeof rr.p === "string" && typeof rr.n === "string" ? { p: rr.p, n: rr.n } : undefined;
  return checkFunnel({ status, ...(real ? { real } : {}) });
}

/**
 * The EXACT surface the native bridges call, as string-in/string-out shims so
 * the Swift (JSContext) and Kotlin (JavaScriptSandbox) bridges are trivial and
 * identical. `canonicalInfo` returns a JSON string (the native side JSON-parses
 * it); `bucketPrefix` returns a string or null; `matchBucket` coerces an
 * optional numeric `minEntries`. `build.mjs` assigns this to
 * `globalThis.DhwajaCheck`; the bridge test asserts this contract so a change
 * to a shim can't silently break the native callers (which aren't in CI).
 */
export interface NativeBridge {
  canonicalize(url: string): string | null;
  canonicalInfo(url: string): string;
  bucketPrefix(hashHex: string, width: number): string | null;
  /** D-0708: the served width from a `/v1/info` text, or null when the text is
   * unreadable, out of range, or of a manifest version this build does not
   * understand. The natives call this instead of reading `width` themselves. */
  infoWidth(infoText: string): number | null;
  matchBucket(bucketJsonText: string, hashHex: string, minEntries?: unknown): TrustStatus;
  /** matchBucket + find-the-real-one, as a JSON string {status, real?} the
   * native side parses — string-in/string-out like every other shim. */
  matchBucketInfo(bucketJsonText: string, hashHex: string, minEntries?: unknown): string;
  /** The post-check funnel, as a JSON string `{discover, passportHash}`. Takes
   * the MatchDetail JSON `matchBucketInfo` returns and tells the share-sheet the
   * next step (same rails as the extension). String-in/string-out; a hostile or
   * unparseable input yields the inert `{discover:false, passportHash:null}`. */
  checkFunnelInfo(detailJsonText: string): string;
  /**
   * SLICE 28, part one of the publication check. Takes the three served bodies
   * plus the prefix and this build's arming state, and returns a JSON string:
   * either `{"done":{verdict,…}}` when the answer needs no platform work, or
   * `{"material":{preimage,sig,leafInput,path,index,…}}` when it does.
   *
   * TWO CALLS AND NOT A CALLBACK, because `androidx.javascriptengine` runs this
   * bundle in a separate, isolated process whose entire API is
   * `evaluateJavaScriptAsync(String) -> String`. There is no host-function
   * injection, so a callback cannot exist there; and the fold is a chain, so it
   * cannot be batched into one round trip either. The native does the two
   * pieces of arithmetic — verify the signature over `preimage`, then hash
   * `leafInput` and fold `path` — and hands the answers to
   * `verifyApply`. Every RULE stays here.
   *
   * `minSeq` is the highest publication `seq` this install has seen, or a
   * negative number for "no memory". See `InclusionInput.minSeq`.
   *
   * I15-NATIVE-34 (D-0913(3)), both OPTIONAL so a six-argument call keeps
   * today's behaviour: `everSigned` is the install's ever-signed bit for this
   * key set (only `true` arms it; `InclusionInput.everSigned`), and `nowMs` is
   * the device clock in epoch ms for the 14-day age (a negative or non-finite
   * number means "no clock", the same encoding as `minSeq`; `InclusionInput.nowMs`).
   */
  verifyPlan(infoText: string, proofText: string, bucketText: string, prefix: string, armed: boolean, minSeq: number, everSigned?: boolean, nowMs?: number): string;
  /**
   * SLICE 28, part two. Takes the `material` object `verifyPlan` returned (as
   * JSON), the signature outcomes — a JSON array string with one of `"ok"` /
   * `"bad"` / `"unknown-key"` per `material.candidates` entry (absent
   * `candidates` = one entry; a bare word is still read as a one-entry array,
   * QUEUE #165 stage 1) — and the folded root — or an empty string when the native did not fold, which is
   * correct and expected whenever the signature did not verify.
   *
   * Returns `{"verdict":…,"action":"accept"|"retry"|"fail",…}`. The `action` is
   * the policy, computed here so that no client decides it for itself.
   */
  verifyApply(materialJson: string, signature: string, rootHex: string): string;
}

/** D-0883(6), I12-RG-PRIVACY-03: the floor a NATIVE host may set is clamped to at
 * least the server's published pad floor, whatever it passes — 500 (the old
 * native constant), 0, a negative, NaN or a non-number all become
 * DEFAULT_MIN_BUCKET. A host may only ask for MORE padding, never less. */
export function nativeFloor(minEntries: unknown): number {
  return typeof minEntries === "number" && Number.isFinite(minEntries) ? Math.max(DEFAULT_MIN_BUCKET, minEntries) : DEFAULT_MIN_BUCKET;
}

export function nativeBridge(): NativeBridge {
  return {
    canonicalize: (url) => canonicalizeForCheck(url).canonical,
    canonicalInfo: (url) => JSON.stringify(canonicalizeForCheck(url)),
    bucketPrefix: (hash, width) => bucketPrefixFor(hash, width),
    infoWidth: (infoText) => widthFromInfo(typeof infoText === "string" ? infoText : ""),
    matchBucket: (json, hash, minEntries) => matchBucket(json, hash, nativeFloor(minEntries)),
    matchBucketInfo: (json, hash, minEntries) => JSON.stringify(matchBucketDetail(json, hash, nativeFloor(minEntries))),
    checkFunnelInfo: (detailJson) => JSON.stringify(funnelFromDetailJson(detailJson)),
    verifyPlan: (infoText, proofText, bucketText, prefix, isArmed, minSeq, everSigned, nowMs) =>
      JSON.stringify(
        withDoneAction(
          inclusionPlan({
          infoText: typeof infoText === "string" ? infoText : "",
          proofText: typeof proofText === "string" ? proofText : "",
          bucketText: typeof bucketText === "string" ? bucketText : "",
          prefix: typeof prefix === "string" ? prefix : "",
          armed: isArmed === true,
          // A NEGATIVE minSeq means "this install has no memory". Encoded that
          // way rather than as null because both bridges carry numbers across
          // as numbers and null as something else on each platform, and a
          // rollback check that silently stopped running would be invisible.
            ...(typeof minSeq === "number" && Number.isSafeInteger(minSeq) && minSeq >= 0 ? { minSeq } : {}),
            ...(everSigned === true ? { everSigned: true } : {}),
            ...(typeof nowMs === "number" && Number.isFinite(nowMs) && nowMs >= 0 ? { nowMs } : {}),
          }),
        ),
      ),
    verifyApply: (materialJson, signature, rootHex) => JSON.stringify(applyFromNative(materialJson, signature, rootHex)),
  };
}

/**
 * The native side of `verifyApply`: parse what it was handed, refuse anything
 * off-shape, and never let a malformed input become a pass.
 *
 * The material came from `verifyPlan` in this same isolate moments earlier, so
 * it is not hostile input in the usual sense — but it crossed a process
 * boundary as text on Android, and a check that answered `verified` because a
 * field went missing would be the worst possible failure here. So it is parsed
 * as strictly as anything off the wire.
 */
/**
 * Stamp the policy onto an early-out plan, so a native that short-circuits
 * reads an ACTION rather than mapping a verdict itself.
 *
 * Without this each native would carry its own copy of `verificationAction`,
 * which is a policy — the one thing that must not have four implementations.
 */
function withDoneAction(plan: ReturnType<typeof inclusionPlan>): unknown {
  return "done" in plan ? { done: { ...plan.done, action: verificationAction(plan.done.verdict) } } : plan;
}

function applyFromNative(materialJson: string, signature: string, rootHex: string): InclusionResult & { action: PolicyAction } {
  const withAction = (r: InclusionResult): InclusionResult & { action: PolicyAction } => ({ ...r, action: verificationAction(r.verdict) });
  let parsed: unknown;
  try {
    parsed = JSON.parse(typeof materialJson === "string" ? materialJson : "");
  } catch {
    return withAction({ verdict: "unverifiable", reason: "material unreadable" });
  }
  const m = parsed as Partial<InclusionMaterial> | null;
  if (
    !m ||
    typeof m !== "object" ||
    typeof m.preimage !== "string" ||
    typeof m.sig !== "object" ||
    m.sig === null ||
    (m.candidates !== undefined && !Array.isArray(m.candidates)) ||
    typeof m.seq !== "number" ||
    typeof m.width !== "number" ||
    typeof m.leafInput !== "string" ||
    !Array.isArray(m.path) ||
    typeof m.index !== "number"
  ) {
    return withAction({ verdict: "unverifiable", reason: "material off-shape" });
  }
  // `deferred` IS RE-VALIDATED, and this is the sharp edge of the whole bridge.
  // `inclusionApply` returns it verbatim ahead of every root check, so on
  // Android — where this object crosses a PROCESS BOUNDARY as JSON and comes
  // back — a tampered `deferred: {verdict:"verified"}` bought a clean pass with
  // no root ever compared. Adversarial review demonstrated it against a real
  // injected entry. A deferred verdict may only ever be one of the three the
  // planner actually defers, and `verified` is not among them: a deferral is by
  // construction a REASON NOT TO PASS.
  const DEFERRABLE = ["rolled-back", "stale", "unverifiable"];
  if (m.deferred !== undefined) {
    const d = m.deferred as { verdict?: unknown } | null;
    if (!d || typeof d !== "object" || typeof d.verdict !== "string" || !DEFERRABLE.includes(d.verdict)) {
      return withAction({ verdict: "unverifiable", reason: "deferred verdict is not one the planner defers" });
    }
  }
  // A JSON ARRAY is one outcome per candidate (QUEUE #165 stage 1); its length
  // and words are refused in `inclusionApply` (B4'(d)). A bare word keeps the
  // pre-#165 reading, an unknown one being "unknown-key".
  let sig: SignatureOutcome | SignatureOutcome[];
  if (typeof signature === "string" && signature.startsWith("[")) {
    let list: unknown;
    try {
      list = JSON.parse(signature);
    } catch {
      list = null;
    }
    if (!Array.isArray(list)) return withAction({ verdict: "unverifiable", reason: "signature outcomes unreadable" });
    sig = list as SignatureOutcome[];
  } else {
    sig = signature === "ok" || signature === "bad" || signature === "unknown-key" ? signature : "unknown-key";
  }
  // An empty root string is "the native did not fold", which is what happens
  // whenever the signature failed. `inclusionApply` refuses it as a pass.
  return withAction(inclusionApply(m as InclusionMaterial, sig, typeof rootHex === "string" && rootHex.length > 0 ? rootHex : null));
}

import type { TrustStatus } from "./index.js";

/**
 * THE POST-CHECK HANDOFF — the ONE table the non-web clients mirror.
 *
 * WHAT THIS FIXES. On the web, every check result carries a row of safety doors
 * (`safetyActions` in `apps/web/src/views.ts`): report an impostor, dispute a
 * caution, suggest a creator, and — on every verdict without exception — "Is
 * this listing me?". On iOS, Android and the extension popup there was nothing.
 * `ResultView.swift` and `ResultView.kt` ended in a **Done** button; the popup
 * appended a link only when a validated real-creator pointer existed
 * (`warning`/`caution`) or the verdict was positive, which left the single most
 * dangerous screen in the product — a Flagged account with no pointer — showing
 * a red line of text and no way to do anything at all.
 *
 * `apps/web/src/safety.ts` states the principle these clients were failing:
 * *"A removal path that is only reachable by typing a URL fragment is, for
 * legal purposes, not a removal path."* Three of the four clients were in that
 * position for report, removal, contest and nominate alike.
 *
 * THIS IS THE STAGED FIX, and it is deliberately the small half: a HANDOFF to
 * the web surfaces, not a native reimplementation of them. The fan gets a door;
 * the door opens in the browser, where the intake forms, their copy law and
 * their tests already live. Native report/removal intake is the larger, separate
 * piece of work.
 *
 * TWO LINKS ON EVERY VERDICT, no exceptions and no conditions:
 *   1. `primary` — the door that fits THIS verdict.
 *   2. `mine` — "Is this listing me?", the removal valve, which the web offers
 *      on every branch for the reason its own comment gives: somebody arriving
 *      at their own listing has no idea which verdict Dhwaja is serving about
 *      them, so gating the removal door on a verdict makes the safety valve
 *      conditional on the very thing they are complaining about.
 *
 * WHAT MUST NOT CHANGE WITHOUT THE OTHERS. `apps/ios/App/ResultView.swift`,
 * `apps/android/.../ResultView.kt` and `apps/extension/src/popup.ts` are
 * literal, case-for-case mirrors of the table below (same hashes, same labels),
 * and `tests/invariants/safety-handoff-parity.test.ts` parses all four and
 * fails if any one drifts. Nothing in this repository compiles Swift or Kotlin,
 * so that guard is the only thing that will notice.
 *
 * PRIVACY: every hash below is a bare fragment on the app's own origin, and
 * carries NOTHING about the checked account — no handle, no URL, no hash of
 * one. The whole point of the k-anonymous check is that the subject never left
 * the device, so it must not leave through a link either. The web's own safety
 * bar takes the same care, and the forms ask for the subject again instead.
 */

export interface SafetyHandoff {
  /** The primary door for this verdict, as a route hash on the web app. */
  hash: string;
  /** Its label. Byte-identical across iOS, Android and the extension. */
  label: string;
}

/** "Is this listing me?" — offered on EVERY verdict, on every client. */
export const HANDOFF_MINE: SafetyHandoff = { hash: "#/remove", label: "Is this listing me? →" };

/**
 * Per-verdict primary door.
 *
 *  - `warning` → report. The fan is looking at an account an operator has
 *    published a warning about; the report intake is the act that helps the next
 *    person, and it is free, permanently.
 *  - `caution` → CONTEST. A caution is machine-served and — per the design
 *    system — "always contestable" (only the report-derived population ages
 *    out; `caution.ts:178-186` binds no cutoff on an unproven listing), and
 *    the person most likely to be
 *    looking at one is its SUBJECT: a creator whose own unproven listing has
 *    been cautioned, checking what Dhwaja says about them. This table used to
 *    send them to "Report this account →", i.e. invited them to report
 *    themselves, on three of the four clients. The web result card had already
 *    worked this out and put "Dispute this caution" first; the shared table had
 *    not, so it was one client's local fix. A caution's primary door is the
 *    dispute; report belongs on `warning`, where an operator has already looked.
 *  - `unclaimed` → the request form, `#/nominate`, with the web result card's
 *    ask chip's words (views.ts). D-1120(7): the phones give each verdict the
 *    web's doors in the web's order, and the web's unclaimed card shows the
 *    report chip, the ask chip, the claim chip, the example chip (when
 *    configured) and the removal valve, in that order (views.ts safetyActions):
 *    this row is the ask chip, `SAFETY_REPORT` below the report chip (painted first) and `SAFETY_CLAIM` the claim chip. The
 *    Most Requested board this door opened until R2-I21-Q125 is switched off by
 *    default (Jason's answer (a); requests.ts REQUESTS_TOP_BOARD_KEY). The
 *    k-anonymous check returns no passport id for a positive hit, so there is
 *    nothing to deep-link; neither form needs one.
 *  - `unknown` → suggest a creator. Dhwaja has no record; the fan can supply

/**
 * Minimal URL polyfill for the two raw JS engines that host dhwaja-check.js:
 * iOS JSContext (JavaScriptCore) and the Android androidx.javascriptengine
 * sandbox. Neither engine exposes Web-platform globals — no `URL`, no `fetch`
 * — because they're bare ECMAScript isolates, not a DOM/WebView (see
 * CheckEngine.swift / CheckEngine.kt). @dhwaja/canonical calls `new URL(...)`
 * unconditionally (index.ts:297,773,872), so on-device this always threw
 * `ReferenceError: Can't find variable: URL` (iOS) / a JS eval failure
 * (Android) — caught by `canonicalizeForCheck`'s try/catch and turned into
 * `{canonical:null}` EVERY time, silently. The mobile check has never
 * detected a single real profile on-device (A-017).
 *
 * Scope: NOT a general WHATWG URL implementation. @dhwaja/canonical only ever
 * constructs absolute http(s) URLs (`withScheme` guarantees a scheme; the
 * very next line rejects anything but http/https) and only reads
 * `.protocol .hostname .port .pathname .search .searchParams`. This covers
 * exactly that surface. No IDNA mapping of a Unicode HOST (a non-ASCII host is
 * refused; an `xn--` label IS validated, QUEUE #383 below) and no IPv6-literal
 * hosts; canonical's own `EMBEDDED_QUAD` / `INTERNAL_TLD` checks already
 * reject IP-literal and private hosts outright. A non-ASCII PATH is accepted
 * and percent-encoded the way the real URL encodes it (`percentEncodePath`,
 * I16 #131IS-01).
 * Installed ONLY when no native URL exists — Node and every browser keep
 * their real, fully spec-compliant URL untouched.
 *
 * WHATWG PARITY, MEASURED (I3-0016/I3-0084/I3-0170/I3-1375): four divergences
 * against the real URL global @dhwaja/canonical hashes against on web/
 * extension, each verified empirically against Node's spec-compliant URL:
 *  - backslash is a path/authority delimiter for a special scheme (I3-0016,
 *    critical): `https://evil.com\@instagram.com/x` real-parses to
 *    hostname "evil.com" (the page a browser actually navigates to), but the
 *    old authority regex read straight through `\`, landing on hostname
 *    "instagram.com" — a forged Verified badge on the attacker's own page.
 *  - a default port (`:443` on https, `:80` on http) is dropped by the real
 *    URL (`url.port === ""`) but was kept verbatim here, and
 *    `packages/canonical/src/index.ts:880` trusts `url.port` to already be
 *    empty for the default — so the SAME subject hashed two different ways.
 *  - a percent-encoded ASCII host (`insta%67ram.com`) is decoded by the real
 *    URL before canonical ever sees it; kept encoded here, it hashed as a
 *    different, unlisted subject (a suppression divergence).
 *  - dot-segments (`/a/../b`) are removed by the real URL's path parser
 *    (RFC 3986 §5.2.4 "remove_dot_segments", the algorithm below); left
 *    alone here, the same profile URL canonicalized to two different paths.
 * All four are fixed by `preprocessAbsoluteUrl` (backslash + control chars,
 * run before the grammar match) and by post-processing the matched host/
 * port/path below — never by widening the regex to accept more inputs.
 */

function decodeComponent(s: string): string {
  try {
    return decodeURIComponent(s.replace(/\+/g, " "));
  } catch {
    return s;
  }
}

/** Percent-decode WITHOUT the `+` → space rewrite `decodeComponent` applies
 * for query strings: a literal `+` in a hostname is not form-encoding and
 * must not become a space (there is no such host in the wild, but silently
 * mangling one would be worse than leaving it alone). ASCII-safe percent
 * escapes only, matching this file's declared non-goal of IDNA/exotic
 * percent-encoding repair; an escape that isn't valid UTF-8 is left as-is,
 * same fail-open shape as `decodeComponent`. */
function decodeHostComponent(s: string): string {
  try {
    return decodeURIComponent(s);
  } catch {
    return s;
  }
}

/**
 * RFC 3986 §5.2.4, the same algorithm the real URL parser's path state runs.
 * A segment is a dot-segment when it is `.`/`..` with any `.` spelled `%2e`
 * (either case) — WHATWG's single-dot / double-dot path segment (I16
 * #131IS-01: `/a/%2e%2e/impostor` is `/impostor` on web). `path` always
 * starts with `/` here (`MinimalURL` defaults an empty path to `/` first).
 *
 * A trailing bare `.` or `..` segment resolves to a trailing `/`, matching
 * the real URL (`/a/.` → `/a/`, `/a/b/..` → `/a/`) rather than dropping the
 * slash — verified against Node's URL for every case below.
 */
function removeDotSegments(path: string): string {
  const segs = path.split("/");
  const out: string[] = [];
  for (let i = 0; i < segs.length; i++) {
    const seg = segs[i];
    const isLast = i === segs.length - 1;
    const dots = (seg as string).replace(/%2e/gi, ".");
    if (dots === ".") {
      if (isLast) out.push("");
      continue;
    }
    if (dots === "..") {
      if (out.length > 1) out.pop();
      if (isLast) out.push("");
      continue;
    }
    out.push(seg as string);
  }
  const joined = out.join("/");
  return joined === "" ? "/" : joined;
}

/**
 * WHATWG path percent-encode set, per code point, UTF-8 with uppercase hex —
 * what the real URL does to a path (`/café` -> `/caf%C3%A9`): C0 controls,
 * space, `"`, `<`, `>`, backtick, `{`, `}`, and every code point > U+007E. A
 * lone surrogate encodes as U+FFFD. An existing `%` escape is left alone.
 * (`#`/`?` never reach here: the grammar ends the path at them.)
 */
function percentEncodePath(path: string): string {
  let out = "";
  for (const ch of path) {
    const c = ch.codePointAt(0) as number;
    if (c > 0x20 && c < 0x7f && !'"<>`{}'.includes(ch)) out += ch;
    else out += encodeURIComponent(c >= 0xd800 && c <= 0xdfff ? "\ufffd" : ch);
  }
  return out;
}

export class MinimalURLSearchParams {
  private readonly pairs: [string, string][];
  constructor(query: string) {
    const q = query.startsWith("?") ? query.slice(1) : query;
    this.pairs = q.length === 0
      ? []
      : q.split("&").filter((kv) => kv.length > 0).map((kv) => {
          const i = kv.indexOf("=");
          const k = i === -1 ? kv : kv.slice(0, i);
          const v = i === -1 ? "" : kv.slice(i + 1);
          return [decodeComponent(k), decodeComponent(v)] as [string, string];
        });
  }
  get(key: string): string | null {
    const hit = this.pairs.find(([k]) => k === key);
    return hit ? hit[1] : null;
  }
  keys(): string[] {
    return this.pairs.map(([k]) => k);
  }
  entries(): [string, string][] {
    return this.pairs.map(([k, v]) => [k, v]);
  }
}

function unpackRanges(s: string): number[] {
  const out: number[] = [];
  const nums = s.split(",").map((x) => Number.parseInt(x, 36));
  let prev = -1;
  for (let i = 0; i + 1 < nums.length; i += 2) {
    const start = prev + 1 + (nums[i] as number);
    prev = start + (nums[i + 1] as number);
    out.push(start, prev);
  }
  return out;
}

let validRanges: number[] | null = null;
let markRanges: number[] | null = null;

function inRanges(r: number[], c: number): boolean {
  let lo = 0;
  let hi = r.length / 2 - 1;
  while (lo <= hi) {
    const mid = (lo + hi) >> 1;
    if (c < (r[2 * mid] as number)) hi = mid - 1;
    else if (c > (r[2 * mid + 1] as number)) lo = mid + 1;
    else return true;
  }
  return false;
}

/** RFC 3492 §6.2 decode of the part after `xn--`; null on any invalid digit,
 * truncated delta, overflow or code point past U+10FFFF. */
function punycodeDecode(input: string): number[] | null {
  const MAXINT = 0x7fffffff;
  const d = input.lastIndexOf("-");
  const b = Math.max(d, 0);
  const out: number[] = [];
  for (let j = 0; j < b; j++) out.push(input.charCodeAt(j));
  let n = 128;
  let i = 0;
  let bias = 72;
  for (let pos = d + 1; pos < input.length; ) {
    const oldi = i;
    let w = 1;
    for (let k = 36; ; k += 36) {
      if (pos >= input.length) return null;
      const ch = input.charCodeAt(pos++);
      const digit = ch >= 0x61 && ch <= 0x7a ? ch - 0x61 : ch >= 0x30 && ch <= 0x39 ? ch - 22 : 36;
      if (digit >= 36 || digit > Math.floor((MAXINT - i) / w)) return null;
      i += digit * w;
      const t = k <= bias ? 1 : k >= bias + 26 ? 26 : k - bias;
      if (digit < t) break;
      if (w > Math.floor(MAXINT / (36 - t))) return null;
      w *= 36 - t;
    }
    const len = out.length + 1;
    let delta = oldi === 0 ? Math.floor((i - oldi) / 700) : (i - oldi) >> 1;
    delta += Math.floor(delta / len);
    let k = 0;
    while (delta > 455) {
      delta = Math.floor(delta / 35);
      k += 36;
    }
    bias = k + Math.floor((36 * delta) / (delta + 38));
    if (Math.floor(i / len) > MAXINT - n) return null;
    n += Math.floor(i / len);
    i %= len;
    if (n > 0x10ffff) return null;
    out.splice(i, 0, n);
    i++;
  }
  return out;
}

/** True iff `label` (lowercased, starting `xn--`) is an A-label `new URL()` accepts. */
function isValidALabel(label: string): boolean {
  for (let j = 0; j < label.length; j++) if (label.charCodeAt(j) > 0x7f) return false;
  const cps = punycodeDecode(label.slice(4));
  if (cps === null || cps.length === 0) return false;
  validRanges ??= unpackRanges(IDNA_VALID);
  markRanges ??= unpackRanges(IDNA_MARK);
  for (const c of cps) if (c >= 0x80 && !inRanges(validRanges, c)) return false;
  if (inRanges(markRanges, cps[0] as number)) return false;
  const u = String.fromCodePoint(...cps);
  return u.normalize("NFC") === u;
}

/**
 * I3-0016: for a special scheme (http/https, the only kind this polyfill
 * ever sees), the real URL parser treats `\` exactly like `/` everywhere up
 * to the first `?`/`#` — inside the authority AND the path — because that is
 * what every real browser's address bar does with it. `evil.com\@insta
 * gram.com/x` is a page served from evil.com (backslash ends the host) with
 * path `/@instagram.com/x`; a parser that reads `\` as an ordinary character
 * lands on hostname `instagram.com` instead — the forged-badge divergence.
 * Rewriting `\` → `/` ONLY before the first `?`/`#` (verified empirically:
 * a `\` inside a query or fragment is NOT converted by the real URL) keeps
 * the existing grammar regex correct with no widening of what it accepts.
 */
function preprocessAbsoluteUrl(input: string): string {
  // WHATWG "remove all ASCII tab or newline" + trim C0-control-or-space —
  // done before the grammar runs, same as the real parser's first step.
  // eslint-disable-next-line no-control-regex
  const noTabNewline = input.replace(/[\t\n\r]/g, "");
  // eslint-disable-next-line no-control-regex
  const trimmed = noTabNewline.replace(/^[\u0000- ]+|[\u0000- ]+$/g, "");

  const schemeSep = trimmed.indexOf("://");
  if (schemeSep === -1) return trimmed;
  const authorityStart = schemeSep + 3;
  let cut = trimmed.length;
  for (let i = authorityStart; i < trimmed.length; i++) {
    const c = trimmed[i];
    if (c === "?" || c === "#") {
      cut = i;
      break;
    }
  }
  const head = trimmed.slice(0, cut).replace(/\\/g, "/");
  const tail = trimmed.slice(cut);
  return head + tail;
}

export class MinimalURL {
  readonly protocol: string;
  readonly hostname: string;
  readonly port: string;
  readonly pathname: string;
  readonly search: string;
  readonly searchParams: MinimalURLSearchParams;
  readonly hash: string;

  constructor(input: string) {
    const m = ABSOLUTE_URL.exec(preprocessAbsoluteUrl(String(input).trim()));
    // Both the scheme and the host are required for an ABSOLUTE url, and the
    // repo compiles with `noUncheckedIndexedAccess`, so BOTH are narrowed here
    // rather than only the host: `m[1]` is the scheme and was previously read
    // as possibly-undefined one line later.
    if (!m || !m[1] || !m[2]) throw new TypeError(`invalid URL: ${input}`);
    const [, scheme, host, port, path, query, hash] = m;
    this.protocol = `${scheme.toLowerCase()}:`;
    // W3-03 FIX2 item 2b: FAIL CLOSED ON THE DECODE. The percent-decode that
    // closed `insta%67ram.com` (I3-0084) runs AFTER the grammar match, so the
    // grammar's own `[^/?#:]` exclusion no longer holds over the result:
    // `%40 %2F %3A %09 %00` are forbidden host code points that the real URL
    // parser REFUSES outright, and this decoded them into a minted host and
    // returned a subject. Where the two parsers could disagree the polyfill
    // must refuse, not guess — so the decoded host is re-checked against the
    // same set the real parser applies, and a violation throws the TypeError
    // `canonicalizeForCheck`'s catch already turns into `{canonical:null}`.
    const decodedHost = decodeHostComponent(host);
    if (FORBIDDEN_HOST.test(decodedHost)) throw new TypeError(`invalid URL: ${input}`);
    this.hostname = decodedHost.toLowerCase();
    // QUEUE #383: an `xn--` label must be an A-label the real URL accepts.
    for (const label of this.hostname.split(".")) {
      if (label.startsWith("xn--") && !isValidALabel(label)) throw new TypeError(`invalid URL: ${input}`);
    }
    // D-0784 / D-0813(4) — the two dangerous-direction divergences B27-URL found.
    // URL:6: a non-ASCII host. The real parser runs UTS46 (`bücher.ch` ->
    // `xn--bcher-kva.ch`; `a⒈.com` refused); the device ships no mapping table,
    // so it would mint the raw Unicode host. Refuse: the safe direction.
    if (NON_ASCII.test(this.hostname)) throw new TypeError(`invalid URL: ${input}`);
    // URL:5: a host that "ends in a number" goes down WHATWG's IPv4 path
    // (`instagram.com.1` refused, `127.1` -> `127.0.0.1`). The device has no
    // such path; only a dotted quad the real parser serialises unchanged passes.
    if (endsInANumber(this.hostname) && !CANONICAL_DOTTED_QUAD.test(this.hostname)) throw new TypeError(`invalid URL: ${input}`);
    const rawPort = port ?? "";
    this.port = rawPort !== "" && DEFAULT_PORT[this.protocol] === rawPort ? "" : rawPort;
    this.pathname = removeDotSegments(percentEncodePath(path && path.length > 0 ? path : "/"));
    this.search = query ? `?${query}` : "";
    this.searchParams = new MinimalURLSearchParams(this.search);
    this.hash = hash ? `#${hash}` : "";
  }
}

/** Install `MinimalURL` as `globalThis.URL` iff no URL global already exists.
 * Idempotent; never overwrites a real (Node/browser) URL implementation. */
export function installURLPolyfillIfMissing(): void {
  const g = globalThis as unknown as { URL?: unknown };
  if (typeof g.URL === "undefined") {
    g.URL = MinimalURL as unknown as typeof globalThis.URL;
  }
}

   * surfaces that CAN render them (the HTML table in `seo-page.ts`) are a
   * separate table and may differ here. */
  glyph: string;
  /** The iOS mirror's SF Symbol for the same slot (SwiftUI renders symbols, not
   * glyphs). Same meaning as `glyph`, platform-idiomatic rendering. */
  sfSymbol: string;
  /** The reserved token hex. An accent only — never the sole signal. */
  color: string;
}

/** What a single row badge can say. A row shows exactly ONE badge, so the
 * highest-precedence state wins (see `rowVerdict`). */
export type RowVerdict = "reported" | "caution" | "verified" | "claimed" | "unclaimed";

/** Precedence order, strongest first. The native mirrors list their cases in
 * THIS order so the precedence is legible in the source. */
export const ROW_ORDER: readonly RowVerdict[] = ["reported", "caution", "verified", "claimed", "unclaimed"];

/** Row labels are short + uppercase (they sit at the end of a dense list row).
 * Glyphs follow the shipped table in `services/trust-core/src/seo-page.ts` for
 * the three states it already renders, extended per docs/DESIGN-SYSTEM.md §4.
 *
 * `claimed` is `◐` here and, since I3-0090, in `ssr-shell.ts` VERDICTS too (it was
 * 🔑 there). 🔑 (U+1F511) is emoji-presentation — fine in a
 * browser, wrong in a tinted Compose `Text`. `◐` keeps the ○ → ◐ → ✓ progression
 * (nothing proved → control proved → operator-vetted) in one monochrome,
 * tintable, fixed-width family. iOS is unaffected: it renders the `key.fill`
 * SF Symbol, which is the design system's "unlocked / key" icon (§4) and IS
 * monochrome and tintable — the glyph and SF-Symbol columns are allowed to
 * differ where a platform has a better native form. */
export const ROW_STYLE: Record<RowVerdict, VerdictStyle> = {
  reported: { label: "REPORTED", glyph: "⚠", sfSymbol: "exclamationmark.triangle.fill", color: TOKEN.reported },
  caution: { label: "CAUTION", glyph: "!", sfSymbol: "exclamationmark.circle", color: TOKEN.caution },
  verified: { label: "VERIFIED", glyph: "✓", sfSymbol: "checkmark.seal.fill", color: TOKEN.verified },
  claimed: { label: "CLAIMED", glyph: "◐", sfSymbol: "key.fill", color: TOKEN.claimed },
  unclaimed: { label: "UNCLAIMED", glyph: "○", sfSymbol: "circle", color: TOKEN.unclaimed },
};

/** States that mean "an active operator warning covers this row". Accepts the
 * registry's served class name as well as the plain word, so either channel the
 * server eventually uses lands on the same badge. */
const REPORTED_STATES: ReadonlySet<string> = new Set(["reported", "warning", "warned"]);

/**
 * Reduce a row's signals to ONE badge, applying the design-system precedence:
 * reported > caution > verified/claimed > unclaimed.
 *
 * Fail-closed by construction:
 *  - a state string this build doesn't recognise (registry/server newer than the
 *    app) falls back to `unclaimed` — the WEAKEST claim, so a row can never
 *    over-claim; it mirrors the fan web app's `CODEX_STATE[…] ?? unclaimed`.
 *  - `founding-*` is stripped first: Founding is a MODIFIER, never a verdict
 *    (docs/DESIGN-SYSTEM.md §4), so `founding-claimed` must not read as verified.
 */
export function rowVerdict(input: {
  /** `undefined`/`null` are explicit here: these fields come off a JSON row and
   * may simply be absent, and an absent signal must resolve, never throw. */
  state?: string | null | undefined;
  warned?: boolean | undefined;
  caution?: boolean | undefined;
}): RowVerdict {
  const state = typeof input.state === "string" ? input.state.trim().toLowerCase() : "";
  if (input.warned === true || REPORTED_STATES.has(state)) return "reported";
  if (input.caution === true || state === "caution") return "caution";
  const base = state.startsWith("founding-") ? state.slice("founding-".length) : state;
  if (base === "verified") return "verified";
  if (base === "claimed") return "claimed";
  return "unclaimed";
}

/**
 * I3-0526 (D-1078(13)) — FOUNDING IS A MODIFIER, SHOWN WITH ITS RANK. `rowVerdict` folds `founding-*` into the
 * base verdict (above); this is the other half: the rank a row shows BESIDE that verdict, or null. The same rule as
 * the server's `validFoundingRank` (services/trust-core/src/badge.ts) — a `founding-` state and an integer rank in
 * 1..FOUNDING_CAPACITY — and shown only beside VERIFIED or CLAIMED (docs/DESIGN-SYSTEM.md §4: "layered on
 * Verified/Claimed"), never beside a REPORTED or CAUTION row. Mirrored by `Founding.rank` (Verdict.swift) and
 * `foundingRank` (Verdict.kt).
 */
export const FOUNDING_CAPACITY = 500; // services/trust-core/src/founding.ts FOUNDING_CAPACITY
export function foundingRank(input: { state?: string | null | undefined; rank?: unknown; warned?: boolean | undefined; caution?: boolean | undefined }): number | null {
  const state = typeof input.state === "string" ? input.state.trim().toLowerCase() : "";
  if (!state.startsWith("founding-")) return null;
  const rank = input.rank;
  if (typeof rank !== "number" || !Number.isInteger(rank) || rank < 1 || rank > FOUNDING_CAPACITY) return null;
  const v = rowVerdict(input);
  return v === "verified" || v === "claimed" ? rank : null;
}
/** The modifier's words — the creator console's approved "Founding Creator #N" (apps/creator-console/src/views.ts). */
export function foundingLabel(rank: number): string {
  return `Founding Creator #${rank}`;
}

/**
 * Write `text` into the page-level live region.
 *
 * Returns `true` when the shell's own region took it. When the region is absent
 * — a shell-less paint, a view rendered outside the router — the outcome is NOT
 * dropped: a region is created on `<body>` and the text written there, and the
 * `false` return is the record that the shell regressed (FAMILY-003: no loss
 * without a record; the record is the return value a caller and a test can both
 * see, never a silent success).
 */
export function announce(doc: Document, text: string): boolean {
  const t = text.trim();
  if (!t) return true; // clearing is not an outcome
  const existing = doc.querySelector<HTMLElement>(LIVE_REGION_SELECTOR);
  if (existing) {
    // Re-set: an identical string assigned twice is not re-announced by AT, so
    // blank first and write on a REAL next tick — not the same synchronous
    // call, which is what shipped here (I3-1378): both assignments landed in
    // one turn, so a reader diffing the accessibility tree asynchronously
    // never observed the intermediate empty state and the repeat announced
    // nothing. `setTimeout(…, 0)` is the tick; `ownerDocument`'s window (never
    // the global) keeps this testable in a realm that is not the test runner's.
    if (existing.textContent === t) {
      existing.textContent = "";
      const win = doc.defaultView;
      const schedule = win ? win.setTimeout.bind(win) : setTimeout;
      schedule(() => {
        existing.textContent = t;
      }, 0);
      return true;
    }
    existing.textContent = t;
    return true;
  }
  const made = doc.createElement("div");
  made.setAttribute("data-live-region", "");
  made.setAttribute("role", "status");
  made.setAttribute("aria-live", "polite");
  made.setAttribute("data-live-region-fallback", "");
  made.textContent = t;
  doc.body.appendChild(made);
  return false;
}

const FOCUSABLE = 'a[href],button:not([disabled]),input:not([disabled]),select:not([disabled]),textarea:not([disabled]),[tabindex]:not([tabindex="-1"])';

function focusables(holder: HTMLElement): HTMLElement[] {
  return [...holder.querySelectorAll<HTMLElement>(FOCUSABLE)];
}

export interface TrapOptions {
  /** Where focus goes on open. Default: the first focusable in the holder. */
  initial?: HTMLElement | null;
  /** Escape was pressed. The caller closes — the trap never removes the DOM. */
  onEscape?: () => void;
  /** Where focus returns on `release()`: the control that opened the dialog. */
  restoreTo?: HTMLElement | null;
}

/**
 * Make `holder` the only reachable part of the page until `release()`.
 *
 * The main region (`#app-root`) is marked `inert` and `aria-hidden` so AT and
 * Tab agree with what `aria-modal="true"` already told AT. `release()` is
 * idempotent — calling it twice must not leave the page inert or steal focus a
 * second time.
 */
export function trapFocus(holder: HTMLElement, opts: TrapOptions = {}): () => void {
  const doc = holder.ownerDocument;
  const main = doc.getElementById("app-root");
  if (main && !main.contains(holder)) {
    main.setAttribute("inert", "");
    main.setAttribute("aria-hidden", "true");
  }
  const onKeydown = (ev: Event): void => {
    const e = ev as KeyboardEvent;
    if (e.key === "Escape") {
      e.preventDefault();
      opts.onEscape?.();
      return;
    }
    if (e.key !== "Tab") return;
    const items = focusables(holder);
    if (items.length === 0) return;
    const first = items[0]!;
    const last = items[items.length - 1]!;
    const active = doc.activeElement as HTMLElement | null;
    if (e.shiftKey && (active === first || !holder.contains(active))) {
      e.preventDefault();
      last.focus();
    } else if (!e.shiftKey && (active === last || !holder.contains(active))) {
      e.preventDefault();
      first.focus();
    }
  };
  holder.addEventListener("keydown", onKeydown);
  (opts.initial ?? focusables(holder)[0] ?? holder).focus();
  let released = false;
  return (): void => {
    if (released) return;
    released = true;
    holder.removeEventListener("keydown", onKeydown);
    if (main) {
      main.removeAttribute("inert");
      main.removeAttribute("aria-hidden");
    }
    opts.restoreTo?.focus();
  };
}
