/** Closures one `oweClosedCreatorFaceErasures` call owes (one D1 batch of two statements each). */
export const CLOSED_CREATOR_FACE_ERASURE_LIMIT = 100;

/**
 * D-1025(3) (I28-FX-ERASE, RG-ER-01): THE ONE-OFF SWEEP — a creator who closed (or was terminated) before D-1024(4)
 * landed owes the same face-enrolment erasure a closure owes now; so does a passport whose FOUNDING account was
 * terminated (closures.ts terminateAccount) before D-1025(3) (I28-FX-INT). Selects up to `limit` passports with no
 * owed-mark — from `creator_closures` and from `account_closures` rows in state 'terminated' joined to `founder`, one row
 * per passport whichever set(s) it is in (oldest first) — and, per passport, owes the erasure (likeness.ts imageConsentOwedOnRemovalStatement, drained
 * by drainImageConsentRevokeOutbox) and writes the mark, both guarded on "no mark yet", in ONE batch. IDEMPOTENT: a
 * second call selects nothing it already owed; a concurrent call's batch is a no-op by the guard. `owed` counts marks
 * actually written (RETURNING); `more` is true when the bound was reached, so rows beyond it may remain for the next
 * call. Scheduled from index.ts's hygiene rotation (step "oweClosedCreatorFaceErasures", I28-FX-INT).
 */
export async function oweClosedCreatorFaceErasures(db: DbLike, now: () => string = () => new Date().toISOString(), limit = CLOSED_CREATOR_FACE_ERASURE_LIMIT): Promise<{ owed: number; more: boolean }> {
  const lim = Math.max(1, Math.min(limit, CLOSED_CREATOR_FACE_ERASURE_LIMIT));
  const { results } = await db
    .prepare(
      `SELECT u.pid AS passport_id FROM (SELECT cc.passport_id AS pid, cc.closed_at AS at FROM creator_closures cc ` +
        `UNION ALL SELECT f.passport_id, ac.terminated_at FROM account_closures ac JOIN founder f ON f.account_id = ac.account_id WHERE ac.state = 'terminated') u ` +
        `WHERE NOT ${FACE_ERASURE_MARKED("u.pid")} GROUP BY u.pid ORDER BY MIN(u.at), u.pid LIMIT ?`,
    )
    .bind(lim)
    .all<{ passport_id: string }>();
  if (results.length === 0) return { owed: 0, more: false };
  const ts = now();
  const unmarked = `NOT ${FACE_ERASURE_MARKED("?")}`;
  const out = await db.batch(
    results.flatMap((r) => [
      imageConsentOwedOnRemovalStatement(db, r.passport_id, ts, unmarked, [r.passport_id]), faceRevokeOwedStatement(db, r.passport_id, ts, "closure backfill", unmarked, [r.passport_id]), // I65-HOLD F1: before the mark, which `unmarked` reads
      faceErasureOwedMark(db, r.passport_id, ts, "system:creator-face-erasure-sweep", "system", unmarked, [r.passport_id]),
    ]),
  );
  const owed = out.reduce<number>((n, r, i) => n + (i % 3 === 2 ? ((r as { results?: unknown[] }).results?.length ?? 0) : 0), 0);
  return { owed, more: results.length === lim };
}

/**
 * Close the creator account of `passportId` (the caller has already admitted a
 * creator session for it). Writes the closure row and its audit row in ONE batch;
 * a passport with no `passports` row is refused (404). Idempotent: a second close
 * reports the first one's row and writes nothing. `extra` rides the batch (the restore replay's audit row).
 * D-1041(1): the closure is journaled (`creator_close`, at its own closed_at) for a restore replay, after the commit, on
 * every call, so a repeat close re-writes the same journal object a failed first write never landed; a closure a
 * termination wrote is not journaled here (terminateCreator journals it).
 */
export async function closeCreatorAccount(db: DbLike, passportId: string, now: () => string = () => new Date().toISOString(), extra: DbStatement[] = []): Promise<CreatorCloseResult> {
  if (!PSP_RE.test(passportId)) throw new CreatorCloseError("bad passport id");
  const exists = await db.prepare("SELECT 1 AS x FROM passports WHERE passport_id = ?").bind(passportId).first<{ x: number }>();
  if (!exists) throw new CreatorCloseError("no such creator", 404);
  const ts = now();
  const e = await creatorEarnings(db, passportId, Date.parse(ts));
  await db.batch([
    db
      .prepare("INSERT INTO creator_closures (passport_id, closed_at, cleared_cents, held_cents, owed_cents) VALUES (?, ?, ?, ?, ?) ON CONFLICT(passport_id) DO NOTHING")
      .bind(passportId, ts, e.cleared_cents, e.pending_cents, e.owed_cents),
    // D-1024(4) (I28-ERASURE, I26 JA-02): a self-close OWES the face-enrolment erasure exactly as terminateCreator does
    // (below), guarded on "this call wrote the closure" so a repeat close owes nothing; drained by likeness.ts.
    imageConsentOwedOnRemovalStatement(db, passportId, ts, "EXISTS (SELECT 1 FROM creator_closures WHERE passport_id = ? AND closed_at = ?)", [passportId, ts]), faceRevokeOwedStatement(db, passportId, ts, "creator self-close", "EXISTS (SELECT 1 FROM creator_closures WHERE passport_id = ? AND closed_at = ?)", [passportId, ts]), // I65-HOLD F1
    // D-1025(3): the owed-mark, same guard, so the one-off sweep (oweClosedCreatorFaceErasures) skips this closure.
    faceErasureOwedMark(db, passportId, ts, `creator:${passportId}`, "creator", "EXISTS (SELECT 1 FROM creator_closures WHERE passport_id = ? AND closed_at = ?)", [passportId, ts]),
    // ONE audit row per closure, ever (the closure is permanent): a repeat close writes neither row.
    db
      .prepare(
        "INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, 'creator:account_closed', ?, ?, ?, 'creator' " +
          "WHERE NOT EXISTS (SELECT 1 FROM audit_log WHERE action = 'creator:account_closed' AND item_id = ?)",
      )
      .bind(`creator:${passportId}`, JSON.stringify({ cleared_cents: e.cleared_cents, held_cents: e.pending_cents, owed_cents: e.owed_cents }), ts, passportId, passportId),
    ...extra,
  ]);
  const row = await db
    .prepare("SELECT closed_at, cleared_cents, held_cents, owed_cents FROM creator_closures WHERE passport_id = ?")
    .bind(passportId)
    .first<{ closed_at: string; cleared_cents: number; held_cents: number; owed_cents: number }>();
  if (!row) throw new Error("creator_closures row missing after close");
  // I46-FX2 (I29-JUDGE-AUDIT JA-13): journaled under the ROW's closed_at on every close, not only the one that wrote it,
  // so a retry after a journal write that failed both ways still journals the close; the key and body are the same.
  // I46-FXR-CODE F2 (D-1307; I46-RG-CODE RGC-06): guarded by the entry's kind. A closure a termination wrote (its
  // terminate row at the closure's instant) keeps the termination's entry, which a plain close never rewrites.
  const terminated = await db
    .prepare(`SELECT 1 AS x FROM audit_log WHERE action = '${CREATOR_TERMINATE_ACTION}' AND item_id = ? AND created_at = ?`)
    .bind(passportId, row.closed_at)
    .first<{ x: number }>();
  if (!terminated) await journalRestoreAct("creator_close", passportId, row.closed_at);
  return { closed: true, closed_now: row.closed_at === ts, closed_at: row.closed_at, cleared_cents: row.cleared_cents, held_cents: row.held_cents, owed_cents: row.owed_cents };
}

/** D-1041(1) (I29-RJ-02 FXE-RG02-JOURNAL): the restore replay of a journaled close, through closeCreatorAccount at the
 * journaled instant (`at`), so the closure's closed_at, and the self-close wipe due from it, are the original ones. */
registerRestoreReplayer("creator_close", async (db, id, reason, _now, extra, at, operator) => {
  // D-1042(1): a journaled termination re-terminates the restored live episode at the journaled instant; the door's own
  // refusals (no live suspension, an undecided appeal) throw, a counted failure.
  if (reason === CREATOR_TERMINATE_JOURNAL_REASON) {
    if (await creatorTerminated(db, id)) return "already";
    if (!(await db.prepare("SELECT 1 AS x FROM passports WHERE passport_id = ?").bind(id).first())) return "absent";
    await terminateCreator(db, id, operator, () => at, Date.parse(at), undefined, extra);
    return "applied";
  }
  if (reason !== null) throw new Error("creator_close journal entry with an unknown reason");
  if (await db.prepare("SELECT 1 AS x FROM creator_closures WHERE passport_id = ?").bind(id).first()) return "already";
  if (!(await db.prepare("SELECT 1 AS x FROM passports WHERE passport_id = ?").bind(id).first())) return "absent";
  await closeCreatorAccount(db, id, () => at, extra);
  return "applied";
});

/**
 * D-1043(1) (I30-CLOSE; FX7-04): THE SUSPENSION CHAIN IS JOURNALED, so a restore to before a suspension re-applies it
 * before a journaled termination. Each act (suspend, appeal opened, appeal decided, lift; since the I54 fix round also the
 * accepted notice that started the episode's window, `window_start` at the send instant) is one `creator_case` entry,
 * reason `<act>[:<category|outcome>]`, keyed `<HH:MM:SS.mmmZ>-<act>` so one creator's same-day acts replay in time order
 * (restore-journal-store.ts RESTORE_JOURNAL_KINDS). Only enumerated codes are journaled, never the note or a reason text;
 * the replay gives the doors CREATOR_CASE_REPLAY_REASON where they need one.
 */
async function journalCreatorCase(passportId: string, ts: string, reason: string): Promise<void> {
  await journalRestoreAct("creator_case", passportId, ts, reason, `${ts.slice(11)}-${reason.split(":")[0]}`);
}
export const CREATOR_CASE_REPLAY_REASON = "re-applied from the restore journal after a database restore";
const CASE_COLUMN: Record<string, string> = { suspend: "suspended_at", appeal_open: "appeal_opened_at", appeal_decide: "appeal_decided_at", lift: "lifted_at" };

/** The replay of one journaled chain act, through its own door at the journaled instant with the replaying operator and
 * no billing (Stripe is not asked again, D-1043(2)). An act whose instant the restored episode already carries is
 * `already`; the doors' own refusals throw (a counted failure). `extra` (the replay audit row) lands after the door. */
registerRestoreReplayer("creator_case", async (db, id, reason, _now, extra, at, operator) => {
  const [act = "", arg = ""] = (reason ?? "").split(":");
  if (act === "window_start") {
    // I54 fix round (SFX-1): the notice was accepted at `at`; the live episode suspended at or before it, still not started,
    // starts its window there, as the original send did. Any other state: the send is not this episode's to replay.
    if (!(await db.prepare("SELECT 1 AS x FROM passports WHERE passport_id = ?").bind(id).first())) return "absent";
    const live = await creatorSuspension(db, id);
    if (!live || live.appeal_deadline !== APPEAL_WINDOW_NOT_STARTED || live.suspended_at > at) return "already";
    const deadline = new Date(Date.parse(at) + APPEAL_WINDOW_DAYS * 86_400_000).toISOString();
    await db.prepare("UPDATE creator_suspensions SET appeal_deadline = ? WHERE suspension_id = ?").bind(deadline, live.suspension_id).run();
    if (extra.length > 0) await db.batch(extra);
    return "applied";
  }
  const col = CASE_COLUMN[act];
  if (!col) throw new Error("creator_case journal entry with an unknown act");
  if (await db.prepare(`SELECT 1 AS x FROM creator_suspensions WHERE passport_id = ? AND ${col} = ?`).bind(id, at).first()) return "already";
  if (!(await db.prepare("SELECT 1 AS x FROM passports WHERE passport_id = ?").bind(id).first())) return "absent";
  const clock = () => at;
  if (act === "suspend") await suspendCreator(db, id, arg as ClosureCategory, operator, "", clock, Date.parse(at));
  else if (act === "appeal_open") await openCreatorAppeal(db, id, operator, clock);
  else if (act === "appeal_decide") await decideCreatorAppeal(db, id, arg as "upheld" | "overturned", operator, CREATOR_CASE_REPLAY_REASON, clock);
  else await liftCreatorSuspension(db, id, operator, clock, CREATOR_CASE_REPLAY_REASON);
  if (extra.length > 0) await db.batch(extra);
  return "applied";
});

/** The live suspension of `passportId`, or null. */
export async function creatorSuspension(db: DbLike, passportId: string): Promise<CreatorSuspensionRow | null> {
  if (!PSP_RE.test(passportId)) return null;
  return (await db.prepare("SELECT * FROM creator_suspensions WHERE passport_id = ? AND state = 'suspended'").bind(passportId).first<CreatorSuspensionRow>()) ?? null;
}

/** Is this creator suspended by staff right now? */
export async function creatorSuspended(db: DbLike, passportId: string): Promise<boolean> {
  if (!PSP_RE.test(passportId)) return false;
  const row = await db.prepare(`SELECT ${CREATOR_SUSPENDED_SQL("?")} AS c`).bind(passportId).first<{ c: number }>();
  return !!row && row.c === 1;
}

 * throws. A suspension notice the provider took moves the episode's (suspended_at = `ts`) appeal deadline to the one the mail
 * named, if later (I21-Q24). I54 fix round (RG-NOTICE-01, SFX-1): the move is read back; landed, it is journaled
 * (`window_start` at the send instant, the restore replay's handle); `unstarted` is true when the provider took the notice
 * and this episode's window is still not started (the write or its read-back failed), which the caller alarms as undelivered. */
async function attemptCreatorClosureNotice(
  db: DbLike,
  passportId: string,
  rail: CreatorNoticeRail | undefined,
  n: CreatorClosureNotice,
  ts: string,
  sendAtMs: number,
): Promise<{ outcome: string; unstarted: boolean }> {
  let outcome = "no_rail";
  let unstarted = false;
  sendAtMs = Math.max(sendAtMs, Date.parse(ts));
  try {
    const env = rail?.env ?? null;
    const appeal_to = supportAddress(env);
    const stored =
      n.state === "terminated"
        ? closureNoticeSummary({ state: "terminated", account_id: passportId, category: n.category, appeal_to })
        : closureNoticeSummary({ state: "suspended", account_id: passportId, category: n.category, appeal_deadline: n.appeal_deadline, appeal_to });
    const atSend = n.state === "suspended" ? closureNoticeAtSend(stored, sendAtMs) : null;
    const to = (await db.prepare("SELECT email FROM creator_contacts WHERE passport_id = ? AND email_verified = 1 AND email IS NOT NULL").bind(passportId).first<{ email: string }>())?.email ?? null;
    const sender = rail?.sender ?? (env !== null && emailConfigFromEnv(env) !== null ? httpEmailFromEnv(env, rail?.fetchImpl) : null);
    if (to === null) outcome = "no_address";
    else if (sender !== null) outcome = await sendOnce(sender, closureNoticeEmail(to, atSend?.summary ?? stored)).catch((): string => "transient");
    if (atSend && outcome === "accepted") {
      unstarted = true;
      await db
        .prepare("UPDATE creator_suspensions SET appeal_deadline = ? WHERE passport_id = ? AND suspended_at = ? AND state = 'suspended' AND (appeal_deadline = ? OR appeal_deadline < ?)")
        .bind(atSend.deadline, passportId, ts, APPEAL_WINDOW_NOT_STARTED, atSend.deadline)
        .run();
      const live = await creatorSuspension(db, passportId);
      unstarted = live?.suspended_at === ts && live.appeal_deadline === APPEAL_WINDOW_NOT_STARTED;
      if (live?.appeal_deadline === atSend.deadline) await journalCreatorCase(passportId, new Date(sendAtMs).toISOString(), "window_start");
    }
  } catch {
    outcome = outcome === "no_rail" ? "unread" : outcome;
  }
  return { outcome, unstarted };
}
/** The one `creator:closure_notice` row; `ts` is the act's instant, which is how a retry finds the episode. */
async function recordCreatorClosureNotice(db: DbLike, passportId: string, state: CreatorClosureNotice["state"], outcome: string, ts: string): Promise<void> {
  try {
    await db
      .prepare(`INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) VALUES (?, '${CREATOR_CLOSURE_NOTICE_ACTION}', ?, ?, ?, 'system')`)
      .bind(passportId, `${state} ${outcome}`, ts, passportId)
      .run();
  } catch (e) {
    logError({ cid: "unknown", hop: "http", msg: "creator_closure_notice_unrecorded", err: e instanceof Error ? e.name : typeof e });
  }
}

/**
 * I32 (I3-0040, D-1082(10)): A CLOSURE NOTICE WHOSE MAIL FAILED IS RETRIED UNTIL IT IS SENT OR DEAD-LETTERED.
 * Work owed: an episode whose LATEST `creator:closure_notice` row (same passport, same act instant) is
 * `<state> transient`, `<state> unread` or `<state> permanent`, and whose episode still stands — a `suspended` notice while
 * the suspension of that instant is live and not terminated, a `terminated` notice while the live episode is terminated
 * (CREATOR_TERMINATED_FOR, the one predicate). Up to
 * `limit` episodes, oldest act first. Per episode: `permanent`, or CREATOR_CLOSURE_NOTICE_RETRY_MAX failed rows already,
 * closes it with `<state> dead` (the dead letter; the operator reads it on the creator's audit trail) and counts toward one
 * CREATOR_CLOSURE_NOTICE_DEAD_MARKER line (counts only — the subject never goes to the log); otherwise the notice is sent
 * again and its outcome recorded at the act's instant, so the next pass sees it. `accepted` and `ambiguous` (settled as
 * sent, email.ts EmailOutcome) close it; `no_address` and `no_rail` are not owed (no address to reach; a replay or an
 * unwired door). An outcome in CREATOR_CLOSURE_NOTICE_UNDELIVERED, or an `accepted` send that left its episode's window
 * not started (I54 fix round), counts toward one CREATOR_CLOSURE_NOTICE_UNDELIVERED_MARKER line (I54). No mail rail on this call: nothing is attempted or recorded. Never throws per episode.
 */
export const CREATOR_CLOSURE_NOTICE_RETRY_MAX = STRIPE_PAUSE_RETRY_MAX;
export const CREATOR_CLOSURE_NOTICE_DEAD_MARKER = "CREATOR_CLOSURE_NOTICE_DEAD";
export async function retryCreatorClosureNotices(
  db: DbLike,
  rail: CreatorNoticeRail | undefined,
  opts: { limit?: number; nowMs?: number } = {},
): Promise<{ scanned: number; sent: number; failed: number; dead: number; skipped: number }> {
  const limit = Math.max(1, Math.min(100, opts.limit ?? STRIPE_PAUSE_RETRY_ACCOUNTS));
  const out = { scanned: 0, sent: 0, failed: 0, dead: 0, skipped: 0 };
  let undelivered = 0;
  const owed = ["transient", "unread", "permanent"];
  const { results } = await db
    .prepare(
      `SELECT a.item_id AS passport_id, a.created_at AS ts, a.detail AS detail, s.category AS category, s.appeal_deadline AS appeal_deadline
         FROM audit_log a JOIN creator_suspensions s ON s.passport_id = a.item_id AND s.state = 'suspended'
        WHERE a.action = '${CREATOR_CLOSURE_NOTICE_ACTION}'
          AND a.rowid = (SELECT MAX(l.rowid) FROM audit_log l WHERE l.item_id = a.item_id AND l.action = a.action AND l.created_at = a.created_at)
          AND ((a.detail IN (SELECT 'suspended ' || value FROM json_each(?)) AND s.suspended_at = a.created_at AND NOT ${CREATOR_TERMINATED_FOR("s.suspension_id")})
            OR (a.detail IN (SELECT 'terminated ' || value FROM json_each(?)) AND ${CREATOR_TERMINATED_FOR("s.suspension_id")}))
        ORDER BY a.created_at ASC LIMIT ?`,
    )
    .bind(JSON.stringify(owed), JSON.stringify(owed), limit)
    .all<{ passport_id: string; ts: string; detail: string; category: ClosureCategory; appeal_deadline: string }>();
  const env = rail?.env ?? null;
  const hasRail = rail?.sender !== undefined || (env !== null && emailConfigFromEnv(env) !== null);
  for (const r of results) {
    out.scanned += 1;
    const state: CreatorClosureNotice["state"] = r.detail.startsWith("terminated ") ? "terminated" : "suspended";
    try {
      const failures = await db
        .prepare(`SELECT COUNT(*) AS n FROM audit_log WHERE item_id = ? AND action = '${CREATOR_CLOSURE_NOTICE_ACTION}' AND created_at = ? AND detail IN (?, ?)`)
        .bind(r.passport_id, r.ts, `${state} transient`, `${state} unread`)
        .first<{ n: number }>();
      if (r.detail === `${state} permanent` || Number(failures?.n ?? 0) >= CREATOR_CLOSURE_NOTICE_RETRY_MAX) {
        await recordCreatorClosureNotice(db, r.passport_id, state, "dead", r.ts);
        out.dead += 1;
        continue;
      }
      if (!hasRail) {
        out.skipped += 1;
        continue;
      }
      const n: CreatorClosureNotice = state === "terminated" ? { state, category: r.category } : { state, category: r.category, appeal_deadline: r.appeal_deadline };
      const { outcome, unstarted } = await attemptCreatorClosureNotice(db, r.passport_id, rail, n, r.ts, opts.nowMs ?? Date.now());
      await recordCreatorClosureNotice(db, r.passport_id, state, outcome, r.ts);
      if (CREATOR_CLOSURE_NOTICE_UNDELIVERED.includes(outcome) || unstarted) undelivered += 1;
      if (outcome === "accepted" || outcome === "ambiguous") out.sent += 1;
      else out.failed += 1;
    } catch (e) {
      out.failed += 1;
      logError({ cid: "unknown", hop: "cron", msg: "creator_closure_notice_retry_failed", err: e instanceof Error ? e.name : typeof e });
    }
  }
  if (out.dead > 0) logError({ cid: "unknown", hop: "cron", msg: CREATOR_CLOSURE_NOTICE_DEAD_MARKER, marker: CREATOR_CLOSURE_NOTICE_DEAD_MARKER, count: out.dead });
  alarmUndelivered("cron", undelivered);
  return out;
}

/** SUSPEND. Refuses a second live suspension (409) rather than rewriting the ground the creator was told. The row
 * and its `creator:suspend` audit row are one batch; the audit row is bound to THIS call's row (INV-058). */
export async function suspendCreator(
  db: DbLike,
  passportId: string,
  category: ClosureCategory,
  operator: string,
  note = "",
  now: () => string = () => new Date().toISOString(),
  nowMs: number = Date.now(),
  billing?: TerminationBilling,
  /** D-1079(1): the rail the closure notice is sent on (`sendCreatorClosureNotice`); absent, it is recorded `no_rail`. */
  notice?: CreatorNoticeRail,
): Promise<CreatorSuspensionRow> {
  if (!PSP_RE.test(passportId)) throw new CreatorCloseError("bad passport id");
  if (!isClosureCategory(category)) throw new CreatorCloseError("category must be one of the enumerated grounds");
  const exists = await db.prepare("SELECT 1 AS x FROM passports WHERE passport_id = ?").bind(passportId).first<{ x: number }>();
  if (!exists) throw new CreatorCloseError("no such creator", 404);
  const ts = now();
  const deadline = APPEAL_WINDOW_NOT_STARTED; // I54 (P31 a): the window starts when the notice is accepted (attemptCreatorClosureNotice)
  const mine = "EXISTS (SELECT 1 FROM creator_suspensions WHERE passport_id = ? AND state = 'suspended' AND suspended_at = ? AND suspended_by = ?)";
  await db.batch([
    db
      .prepare(
        `INSERT INTO creator_suspensions (suspension_id, passport_id, state, category, suspended_at, suspended_by, appeal_deadline, note)
         SELECT 'csu_' || lower(hex(randomblob(16))), ?, 'suspended', ?, ?, ?, ?, ?
          WHERE NOT ${CREATOR_SUSPENDED_SQL("?")}`,
      )
      .bind(passportId, category, ts, operator, deadline, note.slice(0, 2000), passportId),
    db
      .prepare(`INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, 'creator:suspend', ?, ?, ?, 'operator' WHERE ${mine}`)
      .bind(operator, category, ts, passportId, passportId, ts, operator),
  ]);
  const row = await creatorSuspension(db, passportId);
  if (!row || row.suspended_at !== ts || row.suspended_by !== operator) throw new CreatorCloseError("this creator is already suspended", 409);
  await journalCreatorCase(passportId, ts, `suspend:${category}`); // D-1043(1)
  // I25-Q3: pause Creator Pro billing for THIS episode, after it has landed (never throws; the suspension stands).
  await creatorStripeStep(db, passportId, operator, billing, ts, "pause", () => readCreatorStripeSubscriptionIds(db, passportId), {
    sql: "EXISTS (SELECT 1 FROM creator_suspensions WHERE suspension_id = ? AND state = 'suspended')",
    binds: [row.suspension_id],
  });
  // D-1079(1): the notice, after the suspension has landed (never throws; recorded either way).
  await sendCreatorClosureNotice(db, passportId, notice, { state: "suspended", category: row.category, appeal_deadline: row.appeal_deadline }, ts, nowMs);
  return row;
}

/** LIFT. The operator's reversal; also the act an overturned appeal performs. A lift while an appeal is open and
 * undecided IS that appeal's decision (`overturned`), as closures.ts liftClosure records it — and (D-1004(4)) it then
 * REQUIRES `reason` (closures.ts appealReason), stored in appeal_decision_reason. Idempotent on a creator with no live
 * suspension only in the sense that it refuses (404) and writes nothing. */
export async function liftCreatorSuspension(
  db: DbLike,
  passportId: string,
  operator: string,
  now: () => string = () => new Date().toISOString(),
  reason?: string,
  billing?: TerminationBilling,
): Promise<void> {
  const row = await creatorSuspension(db, passportId);
  if (!row) throw new CreatorCloseError("this creator is not suspended", 404);
  if (await episodeTerminated(db, row.suspension_id)) throw new CreatorCloseError(TERMINATED_REFUSAL, 409); // I3-0400: terminate is the end
  const why = appealReason(reason);
  if (row.appeal_opened_at && !row.appeal_decided_at && why === null) throw new CreatorCloseError(APPEAL_REASON_REFUSAL, 400);
  const ts = now();
  await db.batch([
    db
      .prepare(
        `UPDATE creator_suspensions SET state = 'lifted', lifted_at = ?, lifted_by = ?,
           appeal_outcome = CASE WHEN appeal_opened_at IS NOT NULL AND appeal_decided_at IS NULL THEN 'overturned' ELSE appeal_outcome END,
           appeal_decision_reason = CASE WHEN appeal_opened_at IS NOT NULL AND appeal_decided_at IS NULL THEN ? ELSE appeal_decision_reason END,
           appeal_decided_at = CASE WHEN appeal_opened_at IS NOT NULL AND appeal_decided_at IS NULL THEN ? ELSE appeal_decided_at END
         WHERE suspension_id = ? AND state = 'suspended'
           AND (appeal_opened_at IS NULL OR appeal_decided_at IS NOT NULL OR ? IS NOT NULL)
           AND NOT ${CREATOR_TERMINATED_FOR("creator_suspensions.suspension_id")}`,
      )
      // D-1004(4): the clause before last refuses the flip when an appeal opened after the read and no reason was
      // given; the last (I3-0400) refuses it when a termination landed after the read.
      .bind(ts, operator, why, ts, row.suspension_id, why),
    db
      .prepare(
        `INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, 'creator:suspension_lift', ?, ?, ?, 'operator'
          WHERE EXISTS (SELECT 1 FROM creator_suspensions WHERE suspension_id = ? AND state = 'lifted' AND lifted_at = ? AND lifted_by = ?)`,
      )
      .bind(operator, row.category, ts, passportId, row.suspension_id, ts, operator),
  ]);
  const after = await db.prepare("SELECT state, lifted_at, lifted_by FROM creator_suspensions WHERE suspension_id = ?").bind(row.suspension_id).first<{ state: string; lifted_at: string | null; lifted_by: string | null }>();
  if (after?.state !== "lifted" || after.lifted_at !== ts || after.lifted_by !== operator) throw new CreatorCloseError("this suspension changed while lifting — re-read the case", 409);
  await journalCreatorCase(passportId, ts, "lift"); // D-1043(1)
  await resumeCreatorStripe(db, passportId, operator, billing, row, ts); // I25-Q3
}

/** Record that the creator's appeal arrived (single operator, as closures.ts openAppeal). I23-RG-SEC N1: when the
 * actor is the creator itself (`creator:<passport>`, the creator door's spelling) the audit row's actor_kind is
 * 'creator'; any other actor is an operator recording an appeal that arrived by mail. */
export async function openCreatorAppeal(db: DbLike, passportId: string, operator: string, now: () => string = () => new Date().toISOString()): Promise<void> {
  const actorKind = operator === `creator:${passportId}` ? "creator" : "operator";
  const row = await creatorSuspension(db, passportId);
  if (!row) throw new CreatorCloseError("this creator is not suspended", 404);
  if (row.appeal_opened_at) throw new CreatorCloseError("an appeal is already open on this suspension", 409);
  if (await episodeTerminated(db, row.suspension_id)) throw new CreatorCloseError(TERMINATED_REFUSAL, 409); // I3-0400
  const ts = now();
  await db.batch([
    db
      .prepare(`UPDATE creator_suspensions SET appeal_opened_at = ? WHERE suspension_id = ? AND state = 'suspended' AND appeal_opened_at IS NULL AND NOT ${CREATOR_TERMINATED_FOR("creator_suspensions.suspension_id")}`)
      .bind(ts, row.suspension_id),
    db
      .prepare(
        `INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, 'creator:appeal_opened', ?, ?, ?, ?
          WHERE EXISTS (SELECT 1 FROM creator_suspensions WHERE suspension_id = ? AND appeal_opened_at = ?)`,
      )
      .bind(operator, row.category, ts, passportId, actorKind, row.suspension_id, ts),
  ]);
  // D-1043(1): journaled only when THIS call's flip landed.
  if (await db.prepare("SELECT 1 AS x FROM creator_suspensions WHERE suspension_id = ? AND appeal_opened_at = ?").bind(row.suspension_id, ts).first()) await journalCreatorCase(passportId, ts, "appeal_open");
}

/** Decide the open appeal. `overturned` lifts the suspension in the same batch, bound to this decision; `upheld`
 * changes no state. Every consequence is predicated on the decision THIS call made (closures.ts decideAppeal CO-09).
 * D-0995(1): `reason` (trimmed, 8-500 characters, as the account appeal) is stored in appeal_decision_reason and
 * appended to the audit detail; anything else is refused before any read. */
export async function decideCreatorAppeal(
  db: DbLike,
  passportId: string,
  outcome: "upheld" | "overturned",
  operator: string,
  reason: string,
  now: () => string = () => new Date().toISOString(),
  billing?: TerminationBilling,
): Promise<void> {
  if (outcome !== "upheld" && outcome !== "overturned") throw new CreatorCloseError("outcome must be upheld or overturned");
  const why = appealReason(reason); // D-1004(4): closures.ts's one rule
  if (why === null) throw new CreatorCloseError(APPEAL_REASON_REFUSAL);
  const row = await creatorSuspension(db, passportId);
  if (!row) throw new CreatorCloseError("this creator is not suspended", 404);
  if (!row.appeal_opened_at) throw new CreatorCloseError("no appeal is open on this suspension", 409);
  if (row.appeal_decided_at) throw new CreatorCloseError("this appeal is already decided", 409);
  if (await episodeTerminated(db, row.suspension_id)) throw new CreatorCloseError(TERMINATED_REFUSAL, 409); // I3-0400 (unreachable while terminate refuses an undecided appeal; kept per door)
  const ts = now();
  const decided = "EXISTS (SELECT 1 FROM creator_suspensions WHERE suspension_id = ? AND appeal_decided_at = ? AND appeal_outcome = ?)";
  const stmts = [
    db.prepare("UPDATE creator_suspensions SET appeal_decided_at = ?, appeal_outcome = ?, appeal_decision_reason = ? WHERE suspension_id = ? AND state = 'suspended' AND appeal_decided_at IS NULL").bind(ts, outcome, why, row.suspension_id),
    db
      .prepare(`INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, 'creator:appeal_decided', ?, ?, ?, 'operator' WHERE ${decided}`)
      .bind(operator, `${row.category}|${outcome}|${why}`, ts, passportId, row.suspension_id, ts, outcome),
  ];
  if (outcome === "overturned") {
    stmts.push(
      db
        .prepare(`UPDATE creator_suspensions SET state = 'lifted', lifted_at = ?, lifted_by = ? WHERE suspension_id = ? AND state = 'suspended' AND ${decided}`)
        .bind(ts, operator, row.suspension_id, row.suspension_id, ts, outcome),
    );
  }
  await db.batch(stmts);
  const final = await db.prepare("SELECT appeal_decided_at, appeal_outcome FROM creator_suspensions WHERE suspension_id = ?").bind(row.suspension_id).first<{ appeal_decided_at: string | null; appeal_outcome: string | null }>();
  if (!final || final.appeal_decided_at !== ts || final.appeal_outcome !== outcome) throw new CreatorCloseError("this appeal is already decided", 409);
  await journalCreatorCase(passportId, ts, `appeal_decide:${outcome}`); // D-1043(1); an overturn's lift rides this entry
  if (outcome === "overturned") await resumeCreatorStripe(db, passportId, operator, billing, row, ts); // I25-Q3: an overturn is a lift
}

/**
 * I3-0400 (D-0533, Jason 2026-09-19: "creators get an equivalent of the fan two-stage suspend/appeal/terminate path")
 * — THE CREATOR TERMINATE STAGE, the second stage after `suspendCreator`, with the fan's appeal-window rules
 * (closures.ts `terminateAccount`): it refuses while an appeal is OPEN AND UNDECIDED and before `appeal_deadline`
 * (APPEAL_WINDOW_NOT_STARTED at suspension, send + APPEAL_WINDOW_DAYS once a notice is accepted; I54), and each refusal is re-asked in the committing statement's own

 *
 * WHAT IT WRITES, in one batch: the `creator:terminate` audit row (detail = the suspension_id: the episode it ends —
 * this row IS the terminated state, `CREATOR_TERMINATED_FOR`), and, predicated on THAT row, the permanent
 * `creator_closures` row (0205, append-only) with the balance snapshot, so the creator is retired on the same plane a
 * creator's own close uses (a creator who had already closed keeps their own earlier row: ON CONFLICT DO NOTHING).
 *
 * AFTER THE BATCH: cancelTerminatedCreatorStripe cancels the creator's Creator Pro subscriptions (I25-Q3).
 *
 * WHAT IT DOES NOT DO: the suspension stays live and can no longer be lifted or appealed (the three doors below refuse
 * a terminated episode). Nothing is forfeited: the payout doors admit a terminated creator once TERMINATED_HOLD_DAYS
 * (14) have passed since the terminate row and while no fraud finding is open (creator-terminated-sql.ts
 * TERMINATED_PAYABLE_SQL, read by creator-payout.ts RELEASE_OWN_DEST; I25-Q2). The public page is untouched (`request_suppressions`/`passports` are not written). The
 * operator-facing refusals reuse terminateAccount's sentences; the creator is mailed the closure notice in the fan's

// The terminate row's action and the terminated-episode predicate live in the leaf (i13-arch-cycles, I29-REG3 H4) and

export { CREATOR_TERMINATE_ACTION, CREATOR_TERMINATED_FOR };

/** Has staff terminated this creator (a live suspension whose episode carries the terminate row)? */
export async function creatorTerminated(db: DbLike, passportId: string): Promise<boolean> {
  const row = await creatorSuspension(db, passportId);
  return !!row && (await episodeTerminated(db, row.suspension_id));
}
async function episodeTerminated(db: DbLike, suspensionId: string): Promise<boolean> {
  const r = await db.prepare(`SELECT ${CREATOR_TERMINATED_FOR("?")} AS t`).bind(suspensionId).first<{ t: number }>();
  return !!r && r.t === 1;
}

/** TERMINATE (single-operator primitive; the console door wraps it in withDualControl, a hub hand-off). D-1042(1): a landed
 * termination is journaled (`creator_close`, reason CREATOR_TERMINATE_JOURNAL_REASON) for a restore replay; `extra` (the
 * replay's audit row) is written only once the termination has landed. */
export async function terminateCreator(
  db: DbLike,
  passportId: string,
  operator: string,
  now: () => string = () => new Date().toISOString(),
  nowMs: number = Date.now(),
  billing?: TerminationBilling,
  extra: DbStatement[] = [],
  /** D-1079(1): the rail the closure notice is sent on (`sendCreatorClosureNotice`); absent, it is recorded `no_rail`. */
  notice?: CreatorNoticeRail,
): Promise<CreatorTerminateResult> {
  if (!PSP_RE.test(passportId)) throw new CreatorCloseError("bad passport id");
  const row = await creatorSuspension(db, passportId);
  if (!row) throw new CreatorCloseError("this creator is not suspended", 404);
  // D-1043(1) FX7-05: a self-close earlier the SAME UTC day holds the plain `creator_close` key, so this termination's key
  // carries its reason and neither entry overwrites the other (self-close sorts first, so it replays first).
  const journalTermination = async (at: string): Promise<void> => {
    const closure = await db.prepare("SELECT closed_at FROM creator_closures WHERE passport_id = ?").bind(passportId).first<{ closed_at: string }>();
    const sameDayClose = !!closure && closure.closed_at !== at && closure.closed_at.slice(0, 10) === at.slice(0, 10);
    await journalRestoreAct("creator_close", passportId, at, CREATOR_TERMINATE_JOURNAL_REASON, sameDayClose ? CREATOR_TERMINATE_JOURNAL_REASON : undefined);
  };
  if (await episodeTerminated(db, row.suspension_id)) {
    // I46-FXR-CODE F2 (D-1307; I46-RG-CODE RGC-03): a repeat first re-writes the SAME journal object the landed
    // termination wrote (its terminate row's instant, the same tag), so a put that failed both ways is retried; then 409.
    // The row exists on this branch (episodeTerminated reads the same action and detail); the guard only narrows the type.
    const landedAt = await db.prepare(`SELECT created_at FROM audit_log WHERE action = '${CREATOR_TERMINATE_ACTION}' AND detail = ?`).bind(row.suspension_id).first<{ created_at: string }>();
    if (landedAt) await journalTermination(landedAt.created_at);
    throw new CreatorCloseError("this creator is already terminated", 409);
  }
  if (row.appeal_opened_at && !row.appeal_decided_at) throw new CreatorCloseError("an appeal is open and undecided on this creator", 409);
  if (row.appeal_deadline === APPEAL_WINDOW_NOT_STARTED) throw new CreatorCloseError(CREATOR_WINDOW_NOT_STARTED_REFUSAL, 409);
  if (Date.parse(row.appeal_deadline) > nowMs) throw new CreatorCloseError(`the appeal window runs until ${row.appeal_deadline}`, 409);
  const ts = now();
  const e = await creatorEarnings(db, passportId, Date.parse(ts));
  // EVERY REFUSAL ABOVE, RE-ASKED: still this live episode, no undecided appeal, the window closed at nowMs, not yet terminated.
  const stillTerminable = `EXISTS (SELECT 1 FROM creator_suspensions WHERE suspension_id = ? AND passport_id = ? AND state = 'suspended'
      AND (appeal_opened_at IS NULL OR appeal_decided_at IS NOT NULL) AND appeal_deadline <= ?) AND NOT ${CREATOR_TERMINATED_FOR("?")}`;
  const mine = `EXISTS (SELECT 1 FROM audit_log WHERE action = '${CREATOR_TERMINATE_ACTION}' AND detail = ? AND created_at = ? AND actor_id = ?)`;
  await db.batch([
    db
      .prepare(`INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, '${CREATOR_TERMINATE_ACTION}', ?, ?, ?, 'operator' WHERE ${stillTerminable}`)
      .bind(operator, row.suspension_id, ts, passportId, row.suspension_id, passportId, new Date(nowMs).toISOString(), row.suspension_id),
    db
      .prepare(`INSERT INTO creator_closures (passport_id, closed_at, cleared_cents, held_cents, owed_cents) SELECT ?, ?, ?, ?, ? WHERE ${mine} ON CONFLICT(passport_id) DO NOTHING`)
      .bind(passportId, ts, e.cleared_cents, e.pending_cents, e.owed_cents, row.suspension_id, ts, operator),
    // D-0524 (I26-W1-IMAGE): the face-enrolment erasure is OWED in this batch, guarded on `mine`, never called inline —
    // an unreachable image service cannot fail a termination. The cron drain (likeness.ts drainImageConsentRevokeOutbox)
    // retries it until the image service answers 2xx; its `/internal/likeness-revoked` erases enrolments, vectors and
    // review rows as deleteCreatorImages does, and answers 503 while vectors are still pending.
    imageConsentOwedOnRemovalStatement(db, passportId, ts, mine, [row.suspension_id, ts, operator]), faceRevokeOwedStatement(db, passportId, ts, "creator termination", mine, [row.suspension_id, ts, operator]), // I65-HOLD F1
    // D-1025(3): the owed-mark, same guard (see CREATOR_FACE_ERASURE_OWED_ACTION).
    faceErasureOwedMark(db, passportId, ts, operator, "operator", mine, [row.suspension_id, ts, operator]),
  ]);
  const landed = await db.prepare(`SELECT ${mine} AS m`).bind(row.suspension_id, ts, operator).first<{ m: number }>();
  if (landed?.m !== 1) throw new CreatorCloseError("this creator's case changed while terminating — nothing was done; re-read the case", 409);
  if (extra.length > 0) await db.batch(extra);
  await journalTermination(ts);
  await cancelTerminatedCreatorStripe(db, passportId, operator, billing, row.suspension_id, ts); // I25-Q3
  // D-1079(1): the notice, after the termination has landed (never throws; recorded either way).
  await sendCreatorClosureNotice(db, passportId, notice, { state: "terminated", category: row.category }, ts, nowMs);
  return { terminated: true, terminated_at: ts, suspension_id: row.suspension_id, cleared_cents: e.cleared_cents, held_cents: e.pending_cents, owed_cents: e.owed_cents };
}

/**
 * I25-Q3 (D-1030 I25-Q3: "pause billing on suspension, cancel on termination") — CREATOR PRO BILLING follows the
 * creator's closure state, the creator twin of closures.ts stripeCollectionStep / cancelTerminatedStripe (account
 * scoped there, passport scoped here; that file is not edited). `suspendCreator` pauses (`pause_collection` void) every
 * Creator Pro subscription (entitlements-store.ts readCreatorStripeSubscriptionIds), `liftCreatorSuspension` and an
 * overturned appeal resume the ones THIS episode's pause attempted, `terminateCreator` cancels them.
 *
 * NEVER THROWS: each runs after its state change has landed, and the state stands whatever Stripe does. Every outcome
 * is one audit row bound to the state THIS call produced — `ok sub=<id>`, `failed sub=<id>`, or `skipped n=<k>
 * reason=stripe_not_wired|stripe_not_configured` (no `billing` passed / Stripe not configured). A pause is CLAIMED
 * (`attempt sub=<id>`) before Stripe is asked, so a lift landing while the call is in flight still resumes it, and a
 * pause whose outcome row the episode refuses (lifted meanwhile) is undone (`reason=pause_after_lift`) — D-1003(1) and
 * D-1004(3) as closures.ts applies them. Pause rows carry created_at = the episode's suspended_at, which is how the
 * resume finds them.
 */
export const CREATOR_STRIPE_PAUSE_ACTION = "creator:suspend_stripe_pause";
export const CREATOR_STRIPE_PAUSE_ATTEMPT_ACTION = "creator:suspend_stripe_pause_attempt";
export const CREATOR_STRIPE_RESUME_ACTION = "creator:lift_stripe_resume";
/** I31 RG-01 (D-1073(11)): the RESUME claim, `attempt sub=<id>`, written under the resume's bound (the door's own, AND no
 * suspended episode of the passport) BEFORE Stripe is asked; a subscription whose claim is refused is not asked. Its own
 * action, as the pause claim's: the outcome rows keep their meaning. */
export const CREATOR_STRIPE_RESUME_ATTEMPT_ACTION = "creator:lift_stripe_resume_attempt";
export const CREATOR_STRIPE_CANCEL_ACTION = "creator:terminate_stripe_cancel";
/** D-1059(12) (I30-CLOSE-PRO-CANCEL-RETRY): a terminated creator's Creator Pro cancel the retry gave up on — `cancel_by_hand
 * sub=<id>` per subscription that failed STRIPE_PAUSE_RETRY_MAX times, or `cancel_by_hand reason=subscriptions_unread`
 * when the subscriptions could not be read that often. Stripe may still be billing it: an operator cancels it by hand.
 * Listed on the console's #/levers from GET /ops/audit?action=<this> (apps/admin/src/main.ts). */
export const CREATOR_STRIPE_CANCEL_BY_HAND_ACTION = "creator:terminate_stripe_cancel_by_hand";
/** D-1060(4) (the pause sibling of D-1059(12)): a suspended creator's Creator Pro PAUSE the retry gave up on — `pause_by_hand
 * sub=<id>` per subscription that failed STRIPE_PAUSE_RETRY_MAX times, or `pause_by_hand reason=subscriptions_unread` when
 * the subscriptions could not be read that often — at the episode's `suspended_at`. Stripe may still be billing it: an
 * operator pauses it by hand. Never STRIPE_PAUSE_NOTHING_OWED. */
export const CREATOR_STRIPE_PAUSE_BY_HAND_ACTION = "creator:suspend_stripe_pause_by_hand";
/** OPEN-ROWS B3 (I31; the creator twin of closures.ts STRIPE_RESUME_BY_HAND_ACTION, D-1062(1)): a lifted creator's Creator Pro
 * RESUME the retry gave up on — `resume_by_hand sub=<id>` per subscription that failed STRIPE_PAUSE_RETRY_MAX times, or
 * `resume_by_hand reason=subscriptions_unread` when the paused set could not be read that often — at the episode's
 * `lifted_at`. The subscription may still be paused at Stripe: an operator resumes it by hand. Never STRIPE_PAUSE_NOTHING_OWED. */
export const CREATOR_STRIPE_RESUME_BY_HAND_ACTION = "creator:lift_stripe_resume_by_hand";

async function creatorStripeStep(
  db: DbLike,
  passportId: string,
  operator: string,
  billing: TerminationBilling | undefined,
  ts: string,
  mode: "pause" | "resume" | "cancel",
  subs: () => Promise<string[]>,
  bound: { sql: string; binds: unknown[] },
): Promise<TerminationCancelOutcome> {
  const action = mode === "pause" ? CREATOR_STRIPE_PAUSE_ACTION : mode === "resume" ? CREATOR_STRIPE_RESUME_ACTION : CREATOR_STRIPE_CANCEL_ACTION;
  const out: TerminationCancelOutcome = { cancelled: 0, failed: 0, skipped: 0 };
  const insert = async (act: string, detail: string, where: string, binds: unknown[]): Promise<"written" | "refused" | "error"> => {
    try {
      const r = await db
        .prepare(`INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, '${act}', ?, ?, ?, ? WHERE ${where}`)
        .bind(operator, detail, ts, passportId, operator.startsWith("system:") ? "system" : "operator", ...binds)
        .run();
      return (r.changes ?? 0) > 0 ? "written" : "refused";
    } catch {
      return "error"; // the closure stands; an unwritable audit row cannot undo it
    }
  };
  // I31 RG-01 (D-1073(11)): a RESUME is also bound to the passport having no suspended episode — a re-suspension is a
  // NEW creator_suspensions row, which the lift door's own bound (this episode lifted at `ts`) cannot see.
  const where = mode === "resume" ? { sql: `${bound.sql} AND NOT ${CREATOR_SUSPENDED_SQL("?")}`, binds: [...bound.binds, passportId] } : bound;
  const record = (detail: string) => insert(action, detail, where.sql, where.binds);
  let pending: string[];
  try {
    pending = await subs();
  } catch {
    await record("failed reason=subscriptions_unread");
    out.failed = 1;
    return out;
  }
  if (pending.length === 0) return out;
  const cfg = billing?.stripe ?? null;
  if (!cfg) {
    await record(`skipped n=${pending.length} reason=${billing ? "stripe_not_configured" : "stripe_not_wired"}`);
    out.skipped = pending.length;
    return out;
  }
  if (mode === "pause" || mode === "resume") {
    const claimed: string[] = [];
    const claimAction = mode === "pause" ? CREATOR_STRIPE_PAUSE_ATTEMPT_ACTION : CREATOR_STRIPE_RESUME_ATTEMPT_ACTION;
    for (const sub of pending) if ((await insert(claimAction, `attempt sub=${sub}`, where.sql, where.binds)) === "written") claimed.push(sub);
    out.skipped = pending.length - claimed.length;
    pending = claimed;
    if (pending.length === 0) return out;
  }
  const fetchImpl = billing?.fetchImpl ?? fetch;
  const call = mode === "pause" ? pauseStripeSubscription : mode === "resume" ? resumeStripeSubscription : cancelStripeSubscription;
  const lifted: string[] = [];
  const resuspended: { sub: string; ok: boolean }[] = [];
  const dl = opsCancelDeadline();
  try {
    const settled = await Promise.allSettled(pending.map((sub) => call(cfg, sub, fetchImpl, dl)));
    for (const [i, sub] of pending.entries()) {
      const r = settled[i];
      const ok = r?.status === "fulfilled" && r.value === true;
      const outcome = await record(`${ok ? "ok" : "failed"} sub=${sub}`);
      if (outcome === "refused" && mode === "pause") lifted.push(sub);
      // I31 RG-01: a RESUME outcome the bound refuses means the creator was re-suspended while Stripe was being asked; the
      // re-suspension's pause may have reached Stripe first, so this resume (applied, or of unknown outcome) is undone below.
      if (outcome === "refused" && mode === "resume") resuspended.push({ sub, ok });
      if (ok) out.cancelled += 1;
      else out.failed += 1;
    }
  } finally {
    dl.clear();
  }
  if (lifted.length > 0) {
    const dl2 = opsCancelDeadline();
    try {
      for (const sub of lifted) {
        const ok = await resumeStripeSubscription(cfg, sub, fetchImpl, dl2);
        await insert(CREATOR_STRIPE_RESUME_ACTION, `${ok ? "ok" : "failed"} sub=${sub} reason=pause_after_lift`, "1 = 1", []);
      }
    } finally {
      dl2.clear();
    }
  }
  // I31 RG-01 (D-1073(11)): a resume this step sent whose bound no longer held when Stripe answered — a refused resume
  // outcome (recorded first, `reason=lift_superseded`), or a pause_after_lift undo above — may have landed after a NEWER
  // suspension's pause. So the passport is re-checked AFTER Stripe answered: a live suspended episode gets the pause again
  // through this same step, at its `suspended_at` and bound to it (claimed, outcome rows; a `failed` row is the pause pass's
  // work owed, and a lift landing meanwhile undoes it as above). No live episode: the later episode was lifted too, and its
  // lift owns the resume.
  for (const { sub, ok } of resuspended) await insert(CREATOR_STRIPE_RESUME_ACTION, `${ok ? "ok" : "failed"} sub=${sub} reason=lift_superseded`, "1 = 1", []);
  const repause = [...resuspended.map((r) => r.sub), ...lifted];
  if (repause.length > 0) {
    try {
      const live = await creatorSuspension(db, passportId);
      if (live)
        await creatorStripeStep(db, passportId, operator, billing, live.suspended_at, "pause", async () => repause, {
          sql: "EXISTS (SELECT 1 FROM creator_suspensions WHERE suspension_id = ? AND state = 'suspended')",
          binds: [live.suspension_id],
        });
    } catch {
      // never throws. An episode read that fails here re-pauses nothing, and no pass reads the rows above for it.
    }
  }
  return out;
}

/**
 * BILLING-05 (I29-FX-BILLING; the creator twin of closures.ts retryCompromiseStripePauses' PAUSE pass) — A FAILED CREATOR
 * PRO PAUSE IS RETRIED. The pause's audit rows are its outbox: a `failed sub=<id>` row with no `ok sub=<id>` beside it,
 * or a latest row `failed reason=subscriptions_unread`, at a still-suspended, unterminated episode's `suspended_at`, is
 * work owed. Each pass takes up to STRIPE_PAUSE_RETRY_ACCOUNTS such episodes, oldest first, and re-runs the pause step
 * (same binding: the episode still suspended) for every subscription with no `ok` row and fewer than
 * STRIPE_PAUSE_RETRY_MAX `failed` rows; its rows carry the episode's `suspended_at`, so a lift resumes them. An episode
 * left owing nothing (every failed subscription paused or exhausted) or whose subscription read failed
 * STRIPE_PAUSE_RETRY_MAX times gets ONE terminal row (STRIPE_PAUSE_NOTHING_OWED / STRIPE_PAUSE_READ_EXHAUSTED) and is
 * never selected again — D-1060(4): unless the bound DROPPED a pause (a subscription at STRIPE_PAUSE_RETRY_MAX failures with
 * no `ok`, or the exhausted read): then CREATOR_STRIPE_PAUSE_BY_HAND_ACTION rows close it instead, never nothing_owed. No Stripe config: counted, nothing written. NEVER THROWS per creator. Scheduling it (index.ts
 * scheduled()) is the hub's hand-off.
 *
 * D-1051(3) (I30-CLOSE, BILLING-05 cancel limb): A FAILED CREATOR PRO CANCEL AT TERMINATION IS RETRIED by a second pass in
 * the same call. A terminated creator (its CREATOR_TERMINATE_ACTION row) with a cancel row `failed sub=<id>` and no `ok
 * sub=<id>`, or whose latest cancel row is `failed reason=subscriptions_unread`, and no terminal cancel row, is work owed;
 * up to `limit` such creators, oldest termination first, are re-run through cancelTerminatedCreatorStripe (bound: the
 * episode is terminated) for every subscription with no `ok` row and fewer than STRIPE_PAUSE_RETRY_MAX `failed` rows, at the
 * termination instant. The terminal rows close it exactly as for a pause, except that a cancel the bounds dropped (or a
 * read that failed that often) closes it with CREATOR_STRIPE_CANCEL_BY_HAND_ACTION rows, never nothing_owed (D-1059(12)).
 * A `skipped` cancel (a replayed termination, D-1043(2)) is not owed.
 *
 * OPEN-ROWS B3 (I31): A FAILED CREATOR PRO RESUME AFTER A LIFT IS RETRIED by a third pass in the same call (see the pass
 * below); exhausted, it closes the lifted episode with CREATOR_STRIPE_RESUME_BY_HAND_ACTION rows, never nothing_owed.
 */
export const CREATOR_STRIPE_PAUSE_RETRY_ACTOR = "system:creator_stripe_pause_retry";
export const CREATOR_STRIPE_CANCEL_RETRY_ACTOR = "system:creator_stripe_cancel_retry";
/** OPEN-ROWS B3 (I31): the actor of the resume pass's rows. */
export const CREATOR_STRIPE_RESUME_RETRY_ACTOR = "system:creator_stripe_resume_retry";
export async function retryCreatorStripePauses(
  db: DbLike,
  billing: TerminationBilling | undefined,
  opts: { limit?: number } = {},
): Promise<TerminationCancelOutcome & { creators: number }> {
  const limit = Math.max(1, Math.min(100, opts.limit ?? STRIPE_PAUSE_RETRY_ACCOUNTS));
  const out = { cancelled: 0, failed: 0, skipped: 0, creators: 0 };
  const { results } = await db
    .prepare(
      `SELECT s.suspension_id AS suspension_id, s.passport_id AS passport_id, s.suspended_at AS suspended_at FROM creator_suspensions s
        WHERE s.state = 'suspended' AND NOT ${CREATOR_TERMINATED_FOR("s.suspension_id")}
          AND EXISTS (SELECT 1 FROM audit_log a WHERE a.item_id = s.passport_id AND a.action = '${CREATOR_STRIPE_PAUSE_ACTION}' AND a.created_at = s.suspended_at
                        AND ((a.detail LIKE 'failed sub=%'
                              AND NOT EXISTS (SELECT 1 FROM audit_log o WHERE o.item_id = a.item_id AND o.action = a.action AND o.created_at = a.created_at AND o.detail = 'ok' || substr(a.detail, 7) AND o.rowid > a.rowid))
                          OR (a.detail = 'failed reason=subscriptions_unread'
                              AND a.rowid = (SELECT MAX(l.rowid) FROM audit_log l WHERE l.item_id = a.item_id AND l.action = a.action AND l.created_at = a.created_at))))
          AND NOT EXISTS (SELECT 1 FROM audit_log d WHERE d.item_id = s.passport_id AND d.action = '${CREATOR_STRIPE_PAUSE_ACTION}'
                            AND d.created_at = s.suspended_at AND d.detail IN (?, ?))
          -- D-1126(4) (I34-RJ-16 SEAM-06): eraseCreator moves the by-hand row, and every row whose actor is the creator, to one
          -- del_ id in one UPDATE (creator-erasure.ts); the creator's rows name that id, so a hand close is found after erasure too.
          AND NOT EXISTS (SELECT 1 FROM audit_log h WHERE h.action = '${CREATOR_STRIPE_PAUSE_BY_HAND_ACTION}' AND h.created_at = s.suspended_at
                            AND (h.item_id = s.passport_id OR h.item_id IN (SELECT c.item_id FROM audit_log c WHERE c.actor_id = 'creator:' || s.passport_id)))
        ORDER BY s.suspended_at ASC LIMIT ?`,
    )
    .bind(STRIPE_PAUSE_NOTHING_OWED, STRIPE_PAUSE_READ_EXHAUSTED, limit)
    .all<{ suspension_id: string; passport_id: string; suspended_at: string }>();
  for (const r of results) {
    if (!billing?.stripe) {
      out.skipped += 1;
      continue;
    }
    try {
      const { results: rows } = await db
        .prepare(`SELECT detail FROM audit_log WHERE item_id = ? AND action = '${CREATOR_STRIPE_PAUSE_ACTION}' AND created_at = ? ORDER BY rowid`)
        .bind(r.passport_id, r.suspended_at)
        .all<{ detail: string }>();
      // I31 RG-01: `ok` holds the subscriptions whose LAST pause row is `ok` — a re-pause after a resume that landed late
      // (creatorStripeStep, reason=lift_superseded) can fail after the episode's first `ok`, and is owed again.
      const ok = new Set<string>();
      const failures = new Map<string, number>();
      let unread = 0;
      for (const { detail } of rows) {
        if (detail === "failed reason=subscriptions_unread") unread += 1;
        else if (detail.startsWith("ok sub=")) ok.add(detail.slice("ok sub=".length));
        else if (detail.startsWith("failed sub=")) {
          ok.delete(detail.slice("failed sub=".length));
          failures.set(detail.slice("failed sub=".length), (failures.get(detail.slice("failed sub=".length)) ?? 0) + 1);
        }
      }
      const readExhausted = unread >= STRIPE_PAUSE_RETRY_MAX;
      let owesNothing = false;
      let gaveUp: string[] = [];
      const owed = async (): Promise<string[]> => {
        const open = readExhausted ? [] : (await readCreatorStripeSubscriptionIds(db, r.passport_id)).filter((sub) => !ok.has(sub));
        const subs = open.filter((sub) => (failures.get(sub) ?? 0) < STRIPE_PAUSE_RETRY_MAX);
        // D-1060(4): what the bound dropped is not "nothing owed" — it is a pause nobody made.
        gaveUp = readExhausted ? ["reason=subscriptions_unread"] : open.filter((sub) => !subs.includes(sub)).map((sub) => `sub=${sub}`);
        owesNothing = subs.length === 0;
        return subs;
      };
      const bound = { sql: "EXISTS (SELECT 1 FROM creator_suspensions WHERE suspension_id = ? AND state = 'suspended')", binds: [r.suspension_id] };
      const step = await creatorStripeStep(db, r.passport_id, CREATOR_STRIPE_PAUSE_RETRY_ACTOR, billing, r.suspended_at, "pause", owed, bound);
      if (owesNothing && gaveUp.length > 0) {
        await db.batch(
          gaveUp.map((what) =>
            db
              .prepare(`INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, '${CREATOR_STRIPE_PAUSE_BY_HAND_ACTION}', ?, ?, ?, 'system' WHERE ${bound.sql} AND ${byHandOnceSql(CREATOR_STRIPE_PAUSE_BY_HAND_ACTION, true)}`)
              .bind(CREATOR_STRIPE_PAUSE_RETRY_ACTOR, `pause_by_hand ${what}`, r.suspended_at, r.passport_id, ...bound.binds, r.passport_id, `pause_by_hand ${what}`, r.suspended_at),
          ),
        );
      } else if (owesNothing) {
        await db
          .prepare(`INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, '${CREATOR_STRIPE_PAUSE_ACTION}', ?, ?, ?, 'system' WHERE ${bound.sql}`)
          .bind(CREATOR_STRIPE_PAUSE_RETRY_ACTOR, STRIPE_PAUSE_NOTHING_OWED, r.suspended_at, r.passport_id, ...bound.binds)
          .run();
      }
      out.creators += 1;
      out.cancelled += step.cancelled;
      out.failed += step.failed;
      out.skipped += step.skipped;
    } catch {
      out.failed += 1; // one creator's fault never stops the pass
    }
  }
  const { results: terms } = await db
    .prepare(
      `SELECT a.item_id AS passport_id, MIN(a.created_at) AS terminated_at FROM audit_log a
        WHERE a.action = '${CREATOR_STRIPE_CANCEL_ACTION}' AND a.item_id IS NOT NULL
          AND ((a.detail LIKE 'failed sub=%'
                AND NOT EXISTS (SELECT 1 FROM audit_log o WHERE o.item_id = a.item_id AND o.action = a.action AND o.detail = 'ok' || substr(a.detail, 7)))
            OR (a.detail = 'failed reason=subscriptions_unread'
                AND a.rowid = (SELECT MAX(l.rowid) FROM audit_log l WHERE l.item_id = a.item_id AND l.action = a.action)))
          AND NOT EXISTS (SELECT 1 FROM audit_log d WHERE d.item_id = a.item_id AND d.action = a.action AND d.detail IN (?, ?))
          -- D-1126(4), D-1127(5): eraseCreator moves the by-hand row and the episode's terminate row to one del_ id
          -- (CREATOR_CASE_AUDIT_ACTIONS, creator-erasure.ts) while the cancel rows and creator_suspensions keep the passport, so
          -- the hand close is found through the terminate row of this passport's episode (EM-10, as the cancel step's own read).
          AND NOT EXISTS (SELECT 1 FROM audit_log h WHERE h.action = '${CREATOR_STRIPE_CANCEL_BY_HAND_ACTION}'
                            AND (h.item_id = a.item_id OR h.item_id IN (SELECT ta.item_id FROM audit_log ta JOIN creator_suspensions ts ON ts.suspension_id = ta.detail
                                                                         WHERE ta.action = '${CREATOR_TERMINATE_ACTION}' AND ts.passport_id = a.item_id)))
        GROUP BY a.item_id ORDER BY terminated_at ASC, a.item_id LIMIT ?`,
    )
    .bind(STRIPE_PAUSE_NOTHING_OWED, STRIPE_PAUSE_READ_EXHAUSTED, limit)
    .all<{ passport_id: string; terminated_at: string }>();
  for (const r of terms) {
    if (!billing?.stripe) {
      out.skipped += 1;
      continue;
    }
    try {
      const episode = await creatorSuspension(db, r.passport_id); // a terminated episode stays live (terminate is the end)
      if (!episode) continue;
      const { results: rows } = await db
        .prepare(`SELECT detail FROM audit_log WHERE item_id = ? AND action = '${CREATOR_STRIPE_CANCEL_ACTION}'`)
        .bind(r.passport_id)
        .all<{ detail: string }>();
      const failures = new Map<string, number>();
      let unread = 0;
      for (const { detail } of rows) {
        if (detail === "failed reason=subscriptions_unread") unread += 1;
        else if (detail.startsWith("failed sub=")) failures.set(detail.slice("failed sub=".length), (failures.get(detail.slice("failed sub=".length)) ?? 0) + 1);
      }
      const readExhausted = unread >= STRIPE_PAUSE_RETRY_MAX;
      const step = await cancelTerminatedCreatorStripe(db, r.passport_id, CREATOR_STRIPE_CANCEL_RETRY_ACTOR, billing, episode.suspension_id, r.terminated_at, {
        keep: (sub) => (failures.get(sub) ?? 0) < STRIPE_PAUSE_RETRY_MAX,
        readExhausted,
        terminal: STRIPE_PAUSE_NOTHING_OWED,
      });
      out.creators += 1;
      out.cancelled += step.cancelled;
      out.failed += step.failed;
      out.skipped += step.skipped;
    } catch {
      out.failed += 1; // one creator's fault never stops the pass
    }
  }
  // OPEN-ROWS B3 (I31; D-1063, D-1064(11), D-1070(2)): A FAILED CREATOR PRO RESUME AFTER A LIFT IS RETRIED — the third pass,
  // the creator twin of closures.ts retryCompromiseStripePauses' resume pass (D-1005(b), D-1062(1)). A lifted episode with a
  // CREATOR_STRIPE_RESUME_ACTION `failed sub=<id>` / `failed reason=subscriptions_unread` row — the lift's own (at
  // `lifted_at`) or the pause_after_lift undo's (at `suspended_at`) — and no terminal or by-hand row is work owed, up to
  // `limit` episodes, oldest lift first. A subscription is owed while its LAST resume row is not `ok` and it has fewer than
  // STRIPE_PAUSE_RETRY_MAX failures; the step is bound to the episode still lifted at that `lifted_at` AND superseded by no
  // later episode of the passport (a re-suspension's own pause owns the billing from then on, and its lift resumes it).
  // A `skipped` resume (a replayed lift, D-1043(2), like the cancel pass's replayed termination) is not owed.
  const { results: lifts } = await db

        binds: [r.passport_id, r.suspension_id, r.suspended_at],
      };
      const step = await creatorStripeStep(db, r.passport_id, CREATOR_STRIPE_RESUME_RETRY_ACTOR, billing, r.lifted_at, "resume", owed, bound);
      if (owesNothing && gaveUp.length > 0) {
        await db.batch(
          gaveUp.map((what) =>

              .prepare(`INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, '${CREATOR_STRIPE_RESUME_BY_HAND_ACTION}', ?, ?, ?, 'system' WHERE ${bound.sql} AND ${byHandOnceSql(CREATOR_STRIPE_RESUME_BY_HAND_ACTION, true)}`)
              .bind(CREATOR_STRIPE_RESUME_RETRY_ACTOR, `resume_by_hand ${what}`, r.lifted_at, r.passport_id, ...bound.binds, r.passport_id, `resume_by_hand ${what}`, r.lifted_at),
          ),
        );
      } else if (owesNothing) {
        await db
          .prepare(`INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, '${CREATOR_STRIPE_RESUME_ACTION}', ?, ?, ?, 'system' WHERE ${bound.sql}`)
          .bind(CREATOR_STRIPE_RESUME_RETRY_ACTOR, STRIPE_PAUSE_NOTHING_OWED, r.lifted_at, r.passport_id, ...bound.binds)
          .run();
      }
      out.creators += 1;
      out.cancelled += step.cancelled;
      out.failed += step.failed;
      out.skipped += step.skipped;
    } catch {
      out.failed += 1; // one creator's fault never stops the pass
    }
  }
  return out;
}

/** The subscriptions the episode suspended at `suspendedAt` recorded ANY pause row for (claim, ok or failed). */
async function pausedInEpisode(db: DbLike, passportId: string, suspendedAt: string): Promise<string[]> {
  const { results } = await db
    .prepare(`SELECT detail FROM audit_log WHERE item_id = ? AND action IN ('${CREATOR_STRIPE_PAUSE_ACTION}', '${CREATOR_STRIPE_PAUSE_ATTEMPT_ACTION}') AND created_at = ? ORDER BY rowid`)
    .bind(passportId, suspendedAt)
    .all<{ detail: string }>();
  const subs = results.map((r) => /\bsub=(sub_[A-Za-z0-9]+)/.exec(r.detail)?.[1]).filter((s): s is string => typeof s === "string");
  return [...new Set(subs)];
}

/** I25-Q3: cancel every Creator Pro subscription of the creator terminated on episode `suspensionId`. IDEMPOTENT: a
 * subscription with an `ok` cancel row is never asked again, so an operator or a later sweep may re-run it. */
export async function cancelTerminatedCreatorStripe(
  db: DbLike,
  passportId: string,
  operator: string,
  billing: TerminationBilling | undefined,
  suspensionId: string,
  ts: string = new Date().toISOString(),
  retry?: { keep: (sub: string) => boolean; readExhausted: boolean; terminal: string },
): Promise<TerminationCancelOutcome> {
  let owesNothing = false;
  let gaveUp: string[] = [];
  const subs = async (): Promise<string[]> => {
    // Only ever a TERMINATED creator's billing: the terminate row of THIS episode, written for THIS passport.
    // EM-10: through the episode, not the row's item_id, which an erasure moves to the del_ id (creator-erasure.ts).
    const t = await db.prepare(`SELECT 1 AS x FROM audit_log ta JOIN creator_suspensions ts ON ts.suspension_id = ta.detail WHERE ta.action = '${CREATOR_TERMINATE_ACTION}' AND ta.detail = ? AND ts.passport_id = ?`).bind(suspensionId, passportId).first();
    if (!t) return [];
    const { results } = await db
      .prepare(`SELECT detail FROM audit_log WHERE item_id = ? AND action = '${CREATOR_STRIPE_CANCEL_ACTION}' AND detail LIKE 'ok sub=%'`)
      .bind(passportId)
      .all<{ detail: string }>();
    const done = new Set(results.map((r) => r.detail.slice("ok sub=".length)));
    // D-1051(3): the retry pass's bounds — `keep` drops a subscription at STRIPE_PAUSE_RETRY_MAX failures; a read that
    // failed that often asks nothing.
    const open = retry?.readExhausted ? [] : (await readCreatorStripeSubscriptionIds(db, passportId)).filter((s) => !done.has(s));
    const owed = open.filter((s) => !retry || retry.keep(s));
    // D-1059(12): what the bound dropped is not "nothing owed" — it is a cancel nobody made.
    gaveUp = retry?.readExhausted ? ["reason=subscriptions_unread"] : open.filter((s) => !owed.includes(s)).map((s) => `sub=${s}`);
    owesNothing = owed.length === 0;
    return owed;
  };
  const bound = { sql: CREATOR_TERMINATED_FOR("?"), binds: [suspensionId] };
  const out = await creatorStripeStep(db, passportId, operator, billing, ts, "cancel", subs, bound);
  // D-1051(3): a retry that leaves nothing owed closes the termination with ONE terminal row (never selected again).
  // D-1059(12): unless the bound dropped a subscription (or the read): then one CREATOR_STRIPE_CANCEL_BY_HAND_ACTION row
  // per dropped cancel closes it instead, for an operator to cancel by hand — never nothing_owed.
  if (retry && owesNothing && gaveUp.length > 0) {
    await db.batch(
      gaveUp.map((what) =>
        db
          .prepare(`INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, '${CREATOR_STRIPE_CANCEL_BY_HAND_ACTION}', ?, ?, ?, 'system' WHERE ${bound.sql} AND ${byHandOnceSql(CREATOR_STRIPE_CANCEL_BY_HAND_ACTION, false)}`)
          .bind(operator, `cancel_by_hand ${what}`, ts, passportId, ...bound.binds, passportId, `cancel_by_hand ${what}`),
      ),
    );
  } else if (retry && owesNothing) {
    await db
      .prepare(`INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, '${CREATOR_STRIPE_CANCEL_ACTION}', ?, ?, ?, 'system' WHERE ${bound.sql}`)
      .bind(operator, retry.terminal, ts, passportId, ...bound.binds)
      .run();
  }
  return out;
}

async function contactStepUp(c: Context<{ Bindings: Env }>, db: ReturnType<typeof d1>, creator: string, binding: string, raw: unknown): Promise<Response | null> {
  const active = await db.prepare("SELECT COUNT(*) AS n FROM passkeys WHERE passport_id = ? AND status = 'active'").bind(creator).first<{ n: number }>();
  if ((active?.n ?? 0) === 0) return null;
  const cfg = passkeyConfig(c.env);
  if (!cfg) return c.json({ error: "passkeys_unavailable" }, 503);
  try {
    if (raw === undefined) {
      const stepUp = await startAssertionForAction(db, creator, [], cfg, Date.now(), undefined, undefined, binding);
      return c.json({ error: "existing_passkey_required", step_up: stepUp }, 403);
    }
    const a = raw as Record<string, unknown> | null;
    if (!a || typeof a !== "object" || typeof a.credentialId !== "string" || typeof a.clientDataJSON !== "string" || typeof a.authenticatorData !== "string" || typeof a.signature !== "string") {
      return c.json({ error: "bad_assertion" }, 400);
    }
    const assertion: AssertionResponse = { credentialId: a.credentialId, clientDataJSON: a.clientDataJSON, authenticatorData: a.authenticatorData, signature: a.signature };
    await verifyAssertion(db, creator, await actionHashForDrafts([], binding), assertion, cfg, Date.now());
    return null;
  } catch (e) {
    if (e instanceof PasskeyError) return c.json(errorBody(e), e.status);
    throw e;
  }
}

/**

const PUSH_REMOVAL_REFUSAL: Record<Exclude<CreatorPushRemovalOutcome, "removed">, readonly [string, 404 | 401 | 409 | 410]> = {

  gone: ["device_gone", 404],
  expired: ["removal_expired", 410],

};
async function pushRemovalDoor(c: Context<{ Bindings: Env }>, creator: string, deviceId: string | null, body: Record<string, unknown> | null): Promise<Response> {
  if (deviceId === null) return c.json({ ok: true, removed: false });
  const db = d1(c.env.TRUST_DB);
  const mailer = mailerFromEnv(c.env as unknown as Record<string, unknown>);
  const nowMs = Date.now();
  let proof: { kind: "passkey" } | { kind: "email_code"; code: string };
  if (body?.assertion !== undefined) {
    const cfg = passkeyConfig(c.env);
    if (!cfg) return c.json({ error: "passkeys_unavailable" }, 503);
    const assertion = parseAssertion(body.assertion);
    if (!assertion) return c.json({ error: "bad_assertion" }, 400);
    try {
      await verifyAssertion(db, creator, await actionHashForDrafts([], creatorPushRemovalBinding(deviceId)), assertion, cfg, nowMs);
    } catch (e) {
      if (e instanceof PasskeyError) return c.json(errorBody(e), e.status);
      throw e;
    }
    proof = { kind: "passkey" };
  } else if (body?.code !== undefined) {
    if (typeof body.code !== "string") return c.json({ error: "code must be a string" }, 400);
    proof = { kind: "email_code", code: body.code };
  } else {
    const hold = await holdCreatorPushRemoval(db, mailer, creator, deviceId, nowMs, undefined, c.env.SESSION_SECRET);
    if (!hold) return c.json({ error: "not_found" }, 404);
    const cfg = hold.confirm_by === "passkey" ? passkeyConfig(c.env) : null;
    const step_up = cfg ? await startAssertionForAction(db, creator, [], cfg, nowMs, undefined, undefined, creatorPushRemovalBinding(deviceId)) : undefined;

  if (out === "removed") return c.json({ ok: true, removed: true });
  const [error, status] = PUSH_REMOVAL_REFUSAL[out];
  return c.json({ error }, status);
}

/** I25-FX-DATA (RG-07): the sentence beside 409 `push_device_limit`. DRAFT — the wording is Jason's call (open question

const PUSH_DEVICE_LIMIT_MESSAGE = `This account already has ${MAX_PUSH_TOKENS} devices getting alerts. Remove one of them before adding this one.`;
/** I3-0203 b: challenges one passport may start per hour (the fan door's DEVICE_CHALLENGES_PER_ACCOUNT_HOUR). */

/** The same per-IP ceiling the anonymous filing door spends (`claim-routes.ts`

/**

export function makeCreatorContactRoutes(): Hono<{ Bindings: Env }> {
  const app = new Hono<{ Bindings: Env }>();


  // The creator's own metadata — read + set (creator-session gated). The raw
  // updated_by can be an operator actor id (`op_<hex>`, resolved through
  // `operator_identities` — see operators.ts) after an operator override; the
  // creator has no need for an internal operator's identity, so mask it to the
  // non-identifying marker "operator" here (a plain "creator" stamp is kept).
  app.get("/account/profile-meta", async (c) => {
    const id = await creatorId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const m = await getProfileMeta(d1(c.env.TRUST_DB), id);
    // Mask ANY operator actor to the bare marker. Operators are `op_<hex>` since
    // 2026-08-15 (`op:<email>` before it); the mask exists to remove the
    // cross-creator linkage signal "the same operator touched these 40
    // passports", which an opaque id carries just as well as an email did.
    const updated_by = /^op[_:]/.test(m.updated_by ?? "") ? "operator" : m.updated_by;
    return c.json({ meta: { ...m, updated_by } });
  });

  app.post("/account/profile-meta", anonBodyLimit, async (c) => {
    const id = await creatorId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const raw = await c.req.text();
    if (raw.length > 4 * 1024) return c.json({ error: "too_large" }, 413);
    const body = ((): ProfileMetaInput | null => { try { return JSON.parse(raw) as ProfileMetaInput; } catch { return null; } })();
    if (!body) return c.json({ error: "bad json" }, 400);
    try {
      const meta = await setProfileMeta(d1(c.env.TRUST_DB), id, body, "creator");
      return c.json({ ok: true, meta });
    } catch (e) {
      if (e instanceof TaxonomyError) return c.json(errorBody(e), e.status);
      return serverFault(c, "/account/profile-meta", e, "profile_meta_failed");
    }
  });

  // --- S101: per-account control-proof (re-proof of a creator-added account) ---
  // A creator_claimed account (S98) is an assertion; until the creator proves
  // control of THAT account with a bio code, it is not served in the registry and
  // does not shield from reports. Creator-session gated: you prove accounts on
  // YOUR OWN passport only. Same code/TTL/rate machinery as the original claim.
  app.post("/account/prove/start", anonBodyLimit, async (c) => {
    const id = await creatorId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const body = (await c.req.json().catch(() => null)) as { account_key?: unknown } | null;
    if (!body || typeof body.account_key !== "string") return c.json({ error: "account_key required" }, 400);
    try {
      return c.json(await startAccountProof(d1(c.env.TRUST_DB), id, body.account_key, callerPrefix(c)));
    } catch (e) {
      if (e instanceof ClaimError) return c.json(errorBody(e), e.status);
      return serverFault(c, "/account/prove/start", e, "prove_start_failed");
    }
  });

  // I3-1721 (I25): the OAuth twin of /account/prove/start, for an account added to an ALREADY-CLAIMED passport
  // whose platform proves by OAuth (X). Creator-session gated: the proof is always for the session's own
  // passport. The answer is the authorize URL; the provider returns to the shared callback
  // (`/claim/oauth/:platform/callback`), whose `completeOAuthClaim` gives this purpose its own branch.
  app.post("/account/prove/oauth/:platform/start", anonBodyLimit, async (c) => {
    const id = await creatorId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const body = (await c.req.json().catch(() => null)) as { account_key?: unknown } | null;
    if (!body || typeof body.account_key !== "string" || body.account_key.length > ACCOUNT_KEY_MAX) return c.json({ error: "account_key required" }, 400);
    try {
      return c.json(await startOAuthAccountProof(d1(c.env.TRUST_DB), c.env as unknown as Record<string, unknown>, c.req.param("platform"), id, body.account_key));
    } catch (e) {
      if (e instanceof AccountKeyFormatError) return c.json({ error: e.message }, 400);
      if (e instanceof ClaimError) return c.json(errorBody(e), e.status);
      return serverFault(c, "/account/prove/oauth/:platform/start", e, "prove_oauth_start_failed");
    }
  });

  // #132PS-211 (I16-W05B H4): the authenticated RE-READ of a live account-proof
  // code — the same live row handed back, never a new one, only to the passport
  // the session names (claims.ts rereadLiveChallenge).
  app.post("/account/prove/reread", anonBodyLimit, async (c) => {
    const id = await creatorId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const body = (await c.req.json().catch(() => null)) as { account_key?: unknown } | null;
    if (!body || typeof body.account_key !== "string") return c.json({ error: "account_key required" }, 400);

    return r ? c.json(r) : c.json({ error: "no live code" }, 404);
  });

  // #132PS-211: the domain sibling (domain-claims.ts rereadLiveDomainChallenge).
  app.post("/account/domains/reread", anonBodyLimit, async (c) => {
    const id = await creatorId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const body = (await c.req.json().catch(() => null)) as { domain?: unknown } | null;
    if (!body || typeof body.domain !== "string") return c.json({ error: "domain required" }, 400);
    const r = await rereadLiveDomainChallenge(d1(c.env.TRUST_DB), id, body.domain);
    return r ? c.json(r) : c.json({ error: "no live code" }, 404);
  });

  // RJb-T2-AS-004: dispute the amber caution on an account of YOUR OWN passport
  // (caution.ts contestOwnCaution). The anonymous `/caution/contest` keeps its
  // benign no-op for a subject with no fan report.
  app.post("/account/caution/contest", anonBodyLimit, async (c) => {
    const id = await creatorId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const body = (await c.req.json().catch(() => null)) as { account_key?: unknown; reason?: unknown } | null;
    if (!body) return c.json({ error: "account_key required" }, 400);
    try {
      return c.json(await contestOwnCaution(d1(c.env.TRUST_DB), id, body.account_key, body.reason));
    } catch (e) {
      if (e instanceof CautionError) return c.json(errorBody(e), e.status);
      return serverFault(c, "/account/caution/contest", e, "own_caution_contest_failed");
    }
  });

  app.post("/account/prove/verify", anonBodyLimit, async (c) => {
    const id = await creatorId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const body = (await c.req.json().catch(() => null)) as { challenge_id?: unknown } | null;
    if (!body || typeof body.challenge_id !== "string") return c.json({ error: "challenge_id required" }, 400);
    if (!(await fanIpAllowed(c, "crvfy", CREATOR_VERIFIES_PER_IP_HOUR))) return c.json(refusal("rate limit exceeded", RATE_LIMIT_MESSAGE), 429);
    try {
      const r = await verifyAccountProof(d1(c.env.TRUST_DB), id, body.challenge_id);
      return c.json(r, r.ok ? 200 : 409);
    } catch (e) {
      if (e instanceof ClaimError) return c.json(errorBody(e), e.status);
      return serverFault(c, "/account/prove/verify", e, "prove_verify_failed");
    }
  });

  // --- H18: per-DOMAIN control-proof on an already-claimed passport ---
  // The DNS sibling of /account/prove/*: prove a domain account on YOUR OWN
  // (already claimed/verified) passport. startDomainClaim can't do this — its
  // write is claim_proven, unclaimed→claimed only — and startAccountProof proves
  // via a bio code, not DNS. Creator-session gated; the completion writes no
  // ledger event (the used challenge row is the proof).
  app.post("/account/prove/domain/start", anonBodyLimit, async (c) => {
    const id = await creatorId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const body = (await c.req.json().catch(() => null)) as { domain?: unknown } | null;
    if (!body || typeof body.domain !== "string") return c.json({ error: "domain required" }, 400);
    // I3-0918 a / I3-2185 (I22-W1-AUTH): fail closed before a token is minted or the day budget is spent.
    if (!dohResolverFromEnv(c.env as unknown as Record<string, unknown>)) return c.json({ error: "domain_verification_unavailable" }, 503);
    try {
      const r = await startDomainProof(d1(c.env.TRUST_DB), id, domainAccountKey(body.domain), undefined, undefined, callerPrefix(c));
      return c.json(r);
    } catch (e) {
      if (e instanceof DomainClaimError) return c.json(errorBody(e), e.status);
      if (e instanceof Error && /invalid domain/i.test(e.message)) return c.json({ error: e.message }, 400);
      return serverFault(c, "/account/prove/domain/start", e, "prove_domain_start_failed");
    }
  });

  app.post("/account/prove/domain/verify", anonBodyLimit, async (c) => {
    const id = await creatorId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const body = (await c.req.json().catch(() => null)) as { challenge_id?: unknown } | null;
    if (!body || typeof body.challenge_id !== "string") return c.json({ error: "challenge_id required" }, 400);
    if (!(await fanIpAllowed(c, "crvfy", CREATOR_VERIFIES_PER_IP_HOUR))) return c.json(refusal("rate limit exceeded", RATE_LIMIT_MESSAGE), 429);
    // Fail closed: no configured DoH resolver → cannot verify DNS (mirror
    // /claim/domain/verify). No session mint here — the creator is already
    // signed in — so no SESSION_SECRET gate is needed.
    const resolver = dohResolverFromEnv(c.env as unknown as Record<string, unknown>);
    if (!resolver) return c.json({ error: "domain_verification_unavailable" }, 503);
    try {
      const r = await verifyDomainProof(d1(c.env.TRUST_DB), id, body.challenge_id, dohTxtLookup(resolver));
      return c.json(r, r.outcome === "verified" ? 200 : 409);
    } catch (e) {
      if (e instanceof DomainClaimError) return c.json(errorBody(e), e.status);
      return serverFault(c, "/account/prove/domain/verify", e, "prove_domain_verify_failed");
    }
  });

  // --- Partial #4: ADD a second proven domain to your own passport ---
  // The gap `startDomainProof`'s own refusal text pointed at and nothing filled:
  // it says "claim the passport first, then prove additional domains here", but
  // every domain door needed the `passport_accounts` row to exist already, and
  // the console's only add-account form hardcodes `id_kind: "handle"`. The row
  // could not be created — `events.ts` refuses a self-asserted domain, correctly,
  // because "control is proven separately". THIS is separately. Proof first, row
  // second, one batch.
  //
  // THE START HALF SCREENS NOTHING ABOUT THE DOMAIN — not registration, not
  // suppression, not resemblance. Every one of those moved behind the DNS proof
  // in `verifyDomainAdd`, because a start door that screens has to answer
  // "token" or "no", and a "no" the public data does not explain is the
  // delisting bit. See `startDomainAdd`'s docblock for the full argument.
  app.post("/account/domains/start", anonBodyLimit, async (c) => {
    const id = await creatorId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const body = (await c.req.json().catch(() => null)) as { domain?: unknown } | null;
    if (!body || typeof body.domain !== "string") return c.json({ error: "domain required" }, 400);
    // Fail closed BEFORE the day budget is spent: a build with no configured
    // DoH resolver can never verify the token this would mint, and handing out
    // a ceremony that cannot be completed is worse than saying so. Checked here
    // as well as at verify because the two halves are separate requests and the
    // config can change between them.
    if (!dohResolverFromEnv(c.env as unknown as Record<string, unknown>)) return c.json({ error: "domain_verification_unavailable" }, 503);
    try {
      const r = await startDomainAdd(d1(c.env.TRUST_DB), id, body.domain, undefined, undefined, callerPrefix(c));
      return c.json(r);
    } catch (e) {
      // I12-Q6: the uniform refusal (NOT_ADDABLE) says where to ask, as the verify twin below does; a typo (400), the day cap (429)
      // and the caller's own open ceremony (409) are the caller's to fix and carry no support route.
      if (e instanceof DomainClaimError && e.message === NOT_ADDABLE) return c.json({ ...errorBody(e), support_to: supportAddress(c.env as unknown as Record<string, unknown>) }, e.status);
      if (e instanceof DomainClaimError) return c.json(errorBody(e), e.status);
      return serverFault(c, "/account/domains/start", e, "domain_add_start_failed");
    }
  });

  app.post("/account/domains/verify", anonBodyLimit, async (c) => {
    const id = await creatorId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const body = (await c.req.json().catch(() => null)) as { challenge_id?: unknown } | null;
    if (!body || typeof body.challenge_id !== "string") return c.json({ error: "challenge_id required" }, 400);
    if (!(await fanIpAllowed(c, "crvfy", CREATOR_VERIFIES_PER_IP_HOUR))) return c.json(refusal("rate limit exceeded", RATE_LIMIT_MESSAGE), 429);
    // Fail closed, exactly as the sibling verify doors: no configured DoH
    // resolver means this build cannot check DNS, and a feature that cannot
    // check its own proof must not answer.
    const resolver = dohResolverFromEnv(c.env as unknown as Record<string, unknown>);
    if (!resolver) return c.json({ error: "domain_verification_unavailable" }, 503);
    try {
      const r = await verifyDomainAdd(d1(c.env.TRUST_DB), id, body.challenge_id, dohTxtLookup(resolver));
      // I3-0015 / D-0911(2): a refusal says where to ask (SUPPORT_EMAIL, else the placeholder).
      if (r.outcome === "verified") return c.json(r, 200);
      return c.json({ ...r, support_to: supportAddress(c.env as unknown as Record<string, unknown>) }, 409);
    } catch (e) {
      if (e instanceof DomainClaimError) return c.json({ ...errorBody(e), support_to: supportAddress(c.env as unknown as Record<string, unknown>) }, e.status);
      return serverFault(c, "/account/domains/verify", e, "domain_add_verify_failed");
    }
  });

  // --- S68: creator digest contact points + the in-app digest channel ---
  // (docs/DIRECTION.md Cluster A #7 — decisions: all 3 channels, daily only
  // when notable, weekly always.) All creator-session gated.

  app.post("/account/creator-contact", anonBodyLimit, async (c) => {
    const creator = await creatorId(c);
    if (!creator) return c.json({ error: "unauthorized" }, 401);
    const body = (await c.req.json().catch(() => null)) as { email?: unknown; assertion?: unknown } | null;
    if (!body || typeof body.email !== "string") return c.json({ error: "email required" }, 400);
    // Fail closed: without a configured email provider we cannot deliver the
    // verification code, so we refuse to store an unverifiable address.
    // PV-08: `httpEmailFromEnv` never returns a falsy value — on missing or
    // placeholder config it returns `nullEmailSender` itself (a function, so
    // `!sender` never fired). Test the SENTINEL, not truthiness, so the
    // disabled-config case is refused BEFORE `setCreatorEmail` below spends
    // the daily allowance and overwrites the pending verification target on
    // a send that was always going to answer false.
    const sender = httpEmailFromEnv(c.env as unknown as Record<string, unknown>);
    if (sender === nullEmailSender) return c.json({ error: "email_unavailable" }, 503);
    const db = d1(c.env.TRUST_DB);
    try {
      // I3-0221: the passkey step-up decides BEFORE `setCreatorEmail` spends the
      // daily allowance or overwrites the pending target.
      const refused = await contactChangeStepUp(c, db, creator, body.email, body.assertion);
      if (refused) return refused;
      // I3-0014 e: an address the provider has already refused PERMANENTLY gets the same 400 a fresh permanent
      // refusal gets (below), without a send and BEFORE `setCreatorEmail` spends the allowance on it.
      const key = suppressionKeyOf(sender);
      const addrHash = key ? await emailAddressHash(key, body.email) : null;
      if (addrHash && (await suppressedAddressHashes(db, [addrHash], retiredSuppressionLookup(sender, [body.email]))).size > 0) return c.json({ error: "email_rejected" }, 400);
      const { code } = await setCreatorEmail(db, creator, body.email, Date.now(), undefined, c.env.SESSION_SECRET);
      const outcome = await sendOnce(sender, {
        to: body.email.trim().toLowerCase(),
        subject: "Confirm your Dhwaja digest email",
        text: `Your Dhwaja confirmation code is: ${code}\n\nEnter it in the app to receive your creator roundup. If you didn't request this, ignore this email.`,
      });
      // I3-0014, THE THIRD CALL SITE. There is no retry LOOP on this path, so a
      // permanent rejection costs nothing here in sweeps — it costs the creator
      // a wrong answer: "try again later" over an address the provider will
      // never accept, with the daily allowance already spent by `setCreatorEmail`.
      // `transient` keeps the 503 it always had (retryable, the server's fault);
      // `permanent` is the REQUEST's fault and says so.
      if (outcome === "permanent") {
        if (addrHash) await recordEmailSuppressions(db, [addrHash], new Date().toISOString());
        return c.json({ error: "email_rejected" }, 400);
      }
      if (outcome !== "accepted") return c.json({ error: "email_send_failed" }, 503);
      return c.json({ ok: true, verification: "sent" });
    } catch (e) {
      if (e instanceof DigestError && e.message === "add_passkey_first") return c.json(refusal("add_passkey_first", CONTACT_CHANGE_NEEDS_PASSKEY), 409);
      if (e instanceof DigestError) return c.json(errorBody(e), e.status);
      return c.json({ error: "internal" }, 500);
    }
  });

  app.post("/account/creator-contact/verify", anonBodyLimit, async (c) => {
    const creator = await creatorId(c);
    if (!creator) return c.json({ error: "unauthorized" }, 401);
    const body = (await c.req.json().catch(() => null)) as { code?: unknown } | null;
    if (!body || typeof body.code !== "string") return c.json({ error: "code required" }, 400);
    try {
      const db = d1(c.env.TRUST_DB);
      // KREF A-18 — a confirmed SWAP is told to the address it displaced (the one
      // inbox that can say "that was not me"; the fan twin is verifyFanEmail's
      // fan-email-replaced). `displaced` comes from verifyCreatorEmail's claim and
      // its swap's own RETURNING (digest.ts), never from a read around it, so a
      // first-time confirm or a failed code sends nothing and a write landing
      // after the swap committed cannot suppress it (I5 RGC-02, INV-058).
      const { displaced } = await verifyCreatorEmail(db, creator, body.code, Date.now(), undefined, c.env.SESSION_SECRET);
      if (displaced) deferrer(c)(notifyContactDisplaced(c.env, displaced, "replaced"));
      return c.json({ ok: true, email_verified: true });
    } catch (e) {
      if (e instanceof DigestError && e.message === "add_passkey_first") return c.json(refusal("add_passkey_first", CONTACT_CHANGE_NEEDS_PASSKEY), 409);
      if (e instanceof DigestError) return c.json(errorBody(e), e.status);
      return c.json({ error: "internal" }, 500);
    }
  });

  // TAKE THE ADDRESS OFF — the control 0043 owed, because the address stopped
  // being a notification channel and became a way into the recovery queue. See
  // `removeCreatorEmail`: a pending change is cancelled first, and removing the
  // VERIFIED one is refused while it is the passport's only remaining door.
  app.post("/account/creator-contact/remove", anonBodyLimit, async (c) => {
    const creator = await creatorId(c);
    if (!creator) return c.json({ error: "unauthorized" }, 401);
    // An empty body is the session-only ask; `{assertion}` answers the step-up.
    const body = (await c.req.json().catch(() => null)) as { assertion?: unknown } | null;
    try {
      const db = d1(c.env.TRUST_DB);
      // D-0883(3), I12-RG-AUTH-07: a passkey holder steps up BEFORE anything is
      // removed or cancelled; 0 active passkeys keeps the session-only door.
      const refused = await contactStepUp(c, db, creator, CONTACT_REMOVE_STEP_UP_BINDING, body?.assertion);
      if (refused) return refused;
      // KREF A-18 (I5 RGC-02) — removing a VERIFIED address tells it, once; the
      // address is never echoed in the answer.
      const { removed, displaced } = await removeCreatorEmailDisplacing(db, creator);
      if (displaced) deferrer(c)(notifyContactDisplaced(c.env, displaced, "removed"));
      return c.json({ ok: true, removed });
    } catch (e) {
      if (e instanceof DigestError) return c.json(errorBody(e), e.status);
      logError({ cid: "creator-contact", hop: "http", route: "/account/creator-contact/remove", msg: "creator_contact_remove_failed", err: e instanceof Error ? e.name : typeof e });
      return c.json({ error: "internal" }, 500);
    }
  });

  // I3-0203 b (I25): PROOF OF POSSESSION, the fan device flow's twin (fan-account-routes.ts
  // `/account/device/challenge` + `/account/device`):
  //   1) POST /account/creator-push/challenge {platform, token} -> a secret code is pushed to that token
  //      and the answer is {challenge_id} only — the SAME answer whether or not the provider took it, so the
  //      door is no token-validity oracle;
  //   2) POST /account/creator-push {challenge_id, code} -> the token is stored only if the code came back.
  app.post("/account/creator-push/challenge", anonBodyLimit, async (c) => {
    const creator = await creatorId(c);
    if (!creator) return c.json({ error: "unauthorized" }, 401);
    const body = (await c.req.json().catch(() => null)) as { platform?: unknown; token?: unknown } | null;
    if (!body || typeof body.platform !== "string" || typeof body.token !== "string") return c.json({ error: "platform and token required" }, 400);
    // AFTER the shape check, before the write and the provider call — the ordering `/account/account-visibility`
    // uses below: a malformed request is the client's bug and must not spend an allowance the creator needs.
    try {
      pushTokenShape(body.platform, body.token);
    } catch (e) {
      if (e instanceof DigestError) return c.json(errorBody(e), e.status);
      throw e;
    }
    const db = d1(c.env.TRUST_DB);
    if (!(await consumeRateLimit(db, `crpushchal:${creator}`, Date.now(), CREATOR_PUSH_CHALLENGES_PER_PASSPORT_HOUR, HOUR_MS))) return c.json(refusal("rate limit exceeded"), 429);
    if (!(await fanIpAllowed(c, "crpush", CREATOR_PUSH_PER_IP_HOUR))) return c.json(refusal("rate limit exceeded", RATE_LIMIT_MESSAGE), 429);
    let chg: { challenge_id: string; code: string };
    try {
      chg = await createCreatorPushChallenge(db, creator, body.platform, body.token);
    } catch (e) {
      if (e instanceof DigestError) return c.json(errorBody(e), e.status);
      return c.json({ error: "internal" }, 500);
    }
    const delivered = await sendChallengePush(c.env as unknown as Record<string, unknown>, { device_id: "", platform: body.platform as "apns" | "fcm", push_token: body.token }, chg.code);
    if (!delivered) logError({ cid: crypto.randomUUID(), hop: "http", route: "/account/creator-push/challenge", msg: "creator push challenge not accepted" });
    return c.json({ challenge_id: chg.challenge_id });
  });

  // --- P1: PER-ACCOUNT VISIBILITY — the SHIELD door ---------------------------
  //
  // WHAT THIS DOOR IS AND IS NOT. It shields, and only shields. Taking a shield
  // OFF is the irreversible direction — a roster is public, cached and
  // screenshottable, so an account that rejoins it cannot be un-published — and
  // it therefore goes through the passkey two-step (`/account/creator-action`
  // with `account_unshielded`), not here. A request to un-shield on this door is
  // refused with the pointer rather than silently ignored.
  //
  // WHY A SESSION IS ENOUGH FOR THE SHIELDING HALF. The question "which of these
  // should be public?" lands the moment a creator proves their first account,
  // which is exactly when they have no passkey yet. Gating the safe direction
  // behind a device key would mean the only people who could compartmentalize
  // are the ones who already finished onboarding — and the harm of a stolen
  // session shielding an account is that a roster gets shorter, which the real
  // creator can undo.
  //
  // The append is `actor_kind: "system"` with actor_id `session:creator`: there
  // is no creator-SESSION actor kind and deliberately still isn't (see
  // ACTOR_POLICY in @dhwaja/schema). Ownership is enforced at the LEDGER
  // (`requireOwnedAccount`), not only here, so a client naming somebody else's
  // account key gets a refusal rather than a row.
  app.post("/account/account-visibility", anonBodyLimit, async (c) => {
    const creator = await creatorId(c);
    if (!creator) return c.json({ error: "unauthorized" }, 401);
    const raw = await c.req.text();
    // Bounded BEFORE parse, like every sibling: a JSON body is buffered and
    // parsed by the runtime, so a route that reads it first has already spent
    // the work an unbounded body was sent to make it spend.
    if (raw.length > 8 * 1024) return c.json({ error: "too_large" }, 413);
    const body = ((): { account_key?: unknown; visibility?: unknown } | null => {
      try {
        const o: unknown = JSON.parse(raw);
        return o !== null && typeof o === "object" ? (o as { account_key?: unknown; visibility?: unknown }) : null;
      } catch {
        return null;
      }
    })();
    if (!body || typeof body.account_key !== "string" || body.account_key.length === 0 || body.account_key.length > ACCOUNT_KEY_MAX) {
      return c.json({ error: "account_key required" }, 400);
    }
    // `visibility` is optional. "public" gets the POINTER, because a client that
    // tried to un-shield on the cheap door needs to learn where the real one is;
    // anything else is a malformed value and gets a 400, because answering it
    // with a sentence about passkeys would send a client chasing the wrong bug.
    if (body.visibility === "public") {
      return c.json({ error: "un-shielding is signed with a passkey — use /account/creator-action with account_unshielded" }, 403);
    }
    if (body.visibility !== undefined && body.visibility !== "shielded") return c.json({ error: "visibility must be \"shielded\"" }, 400);

    const db = d1(c.env.TRUST_DB);
    // ONE REFUSAL FOR BOTH "not on this passport" ANSWERS, and this is the
    // control, not a nicety. `requireOwnedAccount` distinguishes "no such
    // account anywhere" from "belongs to psp_<someone else>", and returning
    // either message verbatim turns a creator session into a global
    // "is this handle listed on Dhwaja, and on whose passport?" oracle — for
    // shielded, suppressed and retracted rows alike, against keys anybody can
    // compute offline with `canonicalAccount`. The ownership test is therefore
    // made HERE, scoped, exactly as `misattributionRefusal` makes it for
    // /account/accounts/not-mine: one 404, whichever of the two it is. The
    // ledger's own check still runs underneath and is still the authority.
    const owned = await db
      .prepare("SELECT 1 AS x FROM passport_accounts WHERE account_key = ? AND passport_id = ?")
      .bind(body.account_key, creator)
      .first<{ x: number }>();
    if (!owned) return c.json({ error: "no such account on this passport" }, 404);

    // IDEMPOTENT BEFORE RATE-LIMITED, the ordering `fileMisattribution` uses: a
    // creator who taps twice must not spend an allowance they need for their
    // OTHER accounts. It also BOUNDS THE LEDGER: the session door can add at
    // most one row per account on the passport, so the unbounded replay every
    // public page pays cannot be pumped from here (only a passkey-signed
    // un-shield re-opens a slot, and it is not a session's to spend).
    if ((await shieldedAccountKeys(db, creator)).has(body.account_key)) return c.json({ ok: true, visibility: "shielded" });
    if (!(await fanIpAllowed(c, "acctvis", ACCOUNT_VISIBILITY_PER_IP_HOUR))) return c.json(refusal("rate limit exceeded", RATE_LIMIT_MESSAGE), 429);

    try {
      await appendEvents(
        db,
        [{ event_type: "account_shielded", subject: creator, actor_kind: "system", actor_id: "session:creator", payload: { account_key: body.account_key } }],
        undefined,
        ["account_shielded"], // governed: NOT on APPEND_ALLOWED, so this is its only session-authenticated door
        [],
        ["system"], // S27: the system actor requires explicit authorization
      );
      return c.json({ ok: true, visibility: "shielded" });
    } catch (e) {
      // AE-06: contention is not a domain refusal — the comment below already
      // said so; the catch just didn't test for it. Same fix/shape as every
      // other ledger-write door (index.ts, claim-routes.ts, question-routes.ts,
      // handshake-routes.ts, recognition-routes.ts): a lost head race is OUR
      // failure, answered 503 busy+retry, never folded into the coarse refusal
      // below.
      if (e instanceof LedgerContentionError) return contentionResponse(c);
      // The ownership answer was already given above, uniformly. Anything that
      // reaches here is a ledger-state refusal about the caller's OWN passport,
      // so it is logged and answered coarsely rather than relayed — a relayed
      // `LedgerError` is how the oracle above came back.
      logError({ cid: "creator-contact", hop: "http", route: "/account/account-visibility", msg: "account_visibility_failed", err: e instanceof Error ? e.name : typeof e });
      return c.json({ error: "could not shield that account" }, 409);
    }
  });

  // --- BRAND CONTEST, FILED BY A PROVED PASSPORT (D-0617 / QUEUE #424) -------

      // at `domain-claims.ts:1141` withholds.

    } catch (e) {

type Obj = Record<string, unknown>;
const isObj = (v: unknown): v is Obj => typeof v === "object" && v !== null && !Array.isArray(v);
const isStr = (v: unknown): v is string => typeof v === "string";
const isBool = (v: unknown): v is boolean => typeof v === "boolean";
const isCount = (v: unknown): v is number => Number.isSafeInteger(v) && (v as number) >= 0;

/** Every member through `pick`, or null when the list is not an array or any member is refused. */
function rowsOf<T>(v: unknown, pick: (r: Obj) => T | null): T[] | null {
  if (!Array.isArray(v)) return null;
  const out: T[] = [];
  for (const r of v) {
    const p = isObj(r) ? pick(r) : null;
    if (p === null) return null;
    out.push(p);
  }
  return out;
}

/** The CONTRACT body for `passportId`, copied key by key (nothing else the image service sends is carried), or null. */
export function parseCreatorImageData(raw: unknown, passportId: string): CreatorImageService | null {
  if (!isObj(raw) || raw.ok !== true || raw.passport_id !== passportId) return null;
  const enrolments = rowsOf(raw.enrolments, (r) =>
    isStr(r.image_id) && isStr(r.consent_ref) && isStr(r.created_at) && isBool(r.reembed_owed)
      ? { image_id: r.image_id, consent_ref: r.consent_ref, created_at: r.created_at, reembed_owed: r.reembed_owed }
      : null,
  );
  const consents = rowsOf(raw.consents, (r) =>
    isStr(r.consent_ref) &&
    isStr(r.created_at) &&
    isBool(r.revoked) &&
    (r.revoked_at === null || isStr(r.revoked_at)) &&
    (r.over_18_declared_at === undefined || isStr(r.over_18_declared_at)) &&
    (r.wording_version === undefined || isStr(r.wording_version)) &&
    (r.wording_hash === undefined || isStr(r.wording_hash)) &&
    (r.accepted_at === undefined || isStr(r.accepted_at))
      ? {
          consent_ref: r.consent_ref,
          created_at: r.created_at,
          revoked: r.revoked,
          revoked_at: r.revoked_at,
          ...(isStr(r.over_18_declared_at) ? { over_18_declared_at: r.over_18_declared_at } : {}),
          ...(isStr(r.wording_version) ? { wording_version: r.wording_version } : {}),
          ...(isStr(r.wording_hash) ? { wording_hash: r.wording_hash } : {}),
          ...(isStr(r.accepted_at) ? { accepted_at: r.accepted_at } : {}),
        }
      : null,
  );
  const reviews = rowsOf<CreatorImageService["reviews"][number]>(raw.reviews, (r) =>
    isStr(r.review_id) && (r.side === "refused_attempt" || r.side === "photo_held") && (r.status === "open" || r.status === "resolved") && isStr(r.created_at)
      ? { review_id: r.review_id, side: r.side, status: r.status, created_at: r.created_at }
      : null,
  );
  const c = raw.counts;
  if (!enrolments || !consents || !reviews) return null;
  if (!isObj(c) || !isCount(c.enrolments) || !isCount(c.face_vectors) || !isCount(c.consents) || !isCount(c.reviews)) return null;
  if (!isBool(raw.truncated) || !isBool(raw.likeness_revoke_pending)) return null;
  return {
    enrolments,
    consents,
    reviews,
    counts: { enrolments: c.enrolments, face_vectors: c.face_vectors, consents: c.consents, reviews: c.reviews },
    truncated: raw.truncated,
    likeness_revoke_pending: raw.likeness_revoke_pending,
  };
}

/** The image service's answer, or null on: no reader, a throw, any status but 200, a body that is not JSON or not the
 * CONTRACT shape, or no answer within CREATOR_IMAGE_READ_TIMEOUT_MS (the reader's signal is aborted then too). */
export async function readCreatorImageData(read: CreatorImageReader | undefined, passportId: string): Promise<CreatorImageService | null> {
  if (!read) return null;
  const ac = new AbortController();
  let timer: ReturnType<typeof setTimeout> | undefined;
  const expired = new Promise<null>((resolve) => {
    timer = setTimeout(() => {
      ac.abort();
      resolve(null);
    }, CREATOR_IMAGE_READ_TIMEOUT_MS);
  });
  const answer = (async (): Promise<CreatorImageService | null> => {
    try {
      const r = await read(passportId, ac.signal);
      if (r.status !== 200) return null;
      return parseCreatorImageData(await r.json(), passportId);
    } catch {
      return null;
    }
  })();
  try {
    return await Promise.race([answer, expired]);
  } finally {
    clearTimeout(timer);
  }
}

  const audit = await all(db, "SELECT action, actor_kind, created_at FROM audit_log WHERE item_id = ? ORDER BY audit_id", passportId);

  const image_service = await readCreatorImageData(readImage, passportId);

  return {
    exported_at: nowIso,
    passport,
    names,
    profile: profile ?? null,
    platform_accounts,
    releases,
    credentials,
    earnings,
    payouts,
    dispute_debits,
    closure: closure ?? null,
    payout_destination,
    agency_grants,
    answered_questions,
    broadcasts,
    likeness_jobs,
    public_profile,
    settings: { contact: contact ?? null, questions: questionSettings ?? null, push_devices },
    passkeys,
    audit,
    own_records,
    held_elsewhere: image_service ? [] : [IMAGE_SERVICE_HELD],
    ...(image_service ? { image_service: { ...image_service, note: IMAGE_FINGERPRINT_NOTE } } : {}),
  };
}

/**
 * R3-I29-Q9 (D-1091(4), D-1111(1), D-1112(5)): a cleared balance under MIN_PAYOUT_CENTS settles nothing the weekly run
 * will ever pay, so it never blocks: eraseCreator (and sweepCreatorWipes, given `pay`) pays it once with no minimum when
 * a payout account exists, and whether that payment lands, is refused or fails, the money stays on the money record
 * (under the del_ id after an erasure) and the wipe goes ahead. R3-I29-Q10: a debt `writtenOff` (owed no more than the
 * creator_reserve_losses register absorbed) no longer blocks. CREATOR_UNSETTLED_SQL is the same predicate.
 */
export function unsettledMoney(e: Pick<Earnings, "cleared_cents" | "pending_cents" | "in_flight_cents" | "owed_cents"> & { connect_status?: Earnings["connect_status"] }, writtenOff = false): CreatorUnsettled[] {
  const out: CreatorUnsettled[] = [];
  if (e.cleared_cents >= MIN_PAYOUT_CENTS) out.push("cleared");
  if (e.pending_cents > 0) out.push("clearing");
  if (e.in_flight_cents > 0) out.push("in_payout");
  if (e.owed_cents > 0 && !writtenOff) out.push("owed");
  return out;
}
/** R3-I29-Q10 (I34-FX1-MONEY EM-04): the debt the register absorbed — the sum of the passport's creator_reserve_losses
 * rows (creator-standing.ts). A debt is written off only while what is owed is no more than this; a dispute that lands
 * after the write-off is owed on top of it and blocks. */
const ABSORBED_SQL = (p: string): string => `(SELECT COALESCE(SUM(rl.debt_cents), 0) FROM creator_reserve_losses rl WHERE rl.passport_id = ${p})`;
/** `unsettledMoney(creatorEarnings(...), written off)` for one passport at `nowMs`. `exclude`: the open payout rows an
 * erasure's own residual payment left (refused or failed after reserving): their money stays on the record, unsettled
 * by no one, and does not hold the wipe (D-1111(1)). */
export async function creatorUnsettledNow(db: DbLike, passportId: string, nowMs: number, exclude: readonly string[] = []): Promise<CreatorUnsettled[]> {
  const e = await creatorEarnings(db, passportId, nowMs);
  const off = e.owed_cents > 0 && e.owed_cents <= ((await db.prepare(`SELECT ${ABSORBED_SQL("?")} AS w`).bind(passportId).first<{ w: number }>())?.w ?? 0);
  const left = exclude.length === 0 ? 0 : ((await db.prepare(`SELECT COALESCE(SUM(amount_cents), 0) AS c FROM creator_payouts WHERE passport_id = ? AND status IN ('pending','failed') AND payout_id IN (SELECT value FROM json_each(?))`).bind(passportId, JSON.stringify(exclude)).first<{ c: number }>())?.c ?? 0);
  return unsettledMoney({ ...e, in_flight_cents: e.in_flight_cents - left }, off);
}

/**
 * The one wipe batch: the involuntary epoch bump first, then the personal rows, then the two gates (legal hold, unsettled
 * money at `moneyNowIso`) AHEAD of the caller's rows (the re-key moves the money rows the money gate reads), then the
 * caller's rows. A gate's abort rolls back the whole batch, the bump and the deletes included.
 */
async function runWipe(db: DbLike, passportId: string, now: () => string, moneyNowIso: string, callerTail: DbStatement[], exclude: readonly string[] = [], moneyGate = true): Promise<void> {
  const tail = [
    passportLegalHoldValidator(db, passportId),
    // I34-W3-TC (D-1119(2)), I34-W3-SEAM-P (D-1120(2)): `moneyGate` is false only for the two restore replayers below: a
    // journaled erasure (eraseCreator `replay`) and a journaled wipe (the `creator_wipe` replayer).
    ...(moneyGate
      ? [
          db
            .prepare(`INSERT INTO creator_erasures (passport_id, trigger, due_at) SELECT NULL, 'request', ? WHERE ${CREATOR_UNSETTLED_SQL}`)
            .bind(moneyNowIso, ...CREATOR_UNSETTLED_BINDS(passportId, moneyNowIso, exclude)),
        ]
      : []),
    ...callerTail,
  ];
  await bumpCreatorEpoch(db, passportId, now, true, [...wipeStatements(db, passportId), ...tail]);
}

/** A gate's abort as the door's refusal; anything else is rethrown unchanged. */
async function wipeRefusal(db: DbLike, passportId: string, nowMs: number, e: unknown, exclude: readonly string[] = []): Promise<unknown> {
  const m = e instanceof Error ? e.message : "";
  if (LEGAL_HOLD_ABORT_RE.test(m)) return new CreatorEraseError(CREATOR_LEGAL_HOLD_SENTENCE);
  if (CREATOR_UNSETTLED_ABORT_RE.test(m)) {
    const u = await creatorUnsettledNow(db, passportId, nowMs, exclude);
    return new CreatorEraseError(u.length > 0 ? unsettledSentence(u) : CREATOR_MONEY_CHANGED_SENTENCE, 409, u);
  }
  return e;
}

/**

 * Refused (CreatorEraseError 409, nothing written) under an open legal hold or with money unsettled at `now()`; a
 * committed wipe is journaled (`creator_wipe`, the trigger as its reason). `extra` rides the same batch.

export async function wipeCreatorPersonalData(
  db: DbLike,
  passportId: string,
  trigger: "self_close" | "removed" | "request",

    ], exclude);

const DUE_SQL = `SELECT pid, trig, MIN(due) AS due FROM (
    SELECT cc.passport_id AS pid, 'self_close' AS trig, ${ISO_PLUS("cc.closed_at", HOLD_DAYS)} AS due FROM creator_closures cc
     WHERE EXISTS (SELECT 1 FROM audit_log a WHERE a.item_id = cc.passport_id AND a.action = 'creator:account_closed')
    UNION ALL
    SELECT t.item_id, 'removed', ${ISO_PLUS("t.created_at", CREATOR_REMOVED_WIPE_DAYS)} FROM audit_log t WHERE t.action = '${CREATOR_TERMINATE_ACTION}' AND t.item_id GLOB 'psp_*'
    UNION ALL
    SELECT e.subject, 'removed', ${ISO_PLUS("e.created_at", CREATOR_REMOVED_WIPE_DAYS)} FROM trust_events e
     WHERE e.event_type = 'correction' AND json_extract(e.payload, '$.action') = 'retract_passport'
       AND EXISTS (SELECT 1 FROM passports p WHERE p.passport_id = e.subject AND p.retracted = 1)
  ) u WHERE NOT EXISTS (SELECT 1 FROM creator_erasures x WHERE x.passport_id = u.pid)
  GROUP BY pid ORDER BY MIN(due), pid LIMIT ?`;

export interface CreatorWipeSweepResult {
  enqueued: number;
  wiped: string[];
  /** Due but waiting on unsettled money or an open legal hold (re-read when the cursor comes round). */
  held: number;
  more: boolean;
}

/**
 * I21-Q28 (1) / I21-Q63 (a): one pass. (A) records the due instant of every newly closed or removed passport, (B) wipes
 * each due, unwiped passport whose money is settled NOW and which no open legal hold names. A second pass over a wiped
 * passport selects nothing. Scheduled by I29-INT (scheduled() step "sweepCreatorWipes").
 * D-1037(9) CE-01/CE-02: (B) reads one page of `limit` ready rows AFTER the CREATOR_WIPE_CURSOR row (due_at, passport_id)
 * and stores the page's last row as the next start, so rows it cannot wipe yet never pin the page; a short page wraps
 * the cursor to the start, so every ready row is examined within ceil(ready / limit) + 1 passes. `more` = a page was full.
 */
export async function sweepCreatorWipes(db: DbLike, nowMs: number = Date.now(), limit = CREATOR_WIPE_LIMIT, pay?: EraseCreatorPay): Promise<CreatorWipeSweepResult> {
  const lim = Math.max(1, Math.min(limit, CREATOR_WIPE_LIMIT));
  const { results: due } = await db.prepare(DUE_SQL).bind(lim).all<{ pid: string; trig: string; due: string }>();
  if (due.length > 0) {
    await db.batch(
      due.map((d) => db.prepare("INSERT INTO creator_erasures (passport_id, trigger, due_at) VALUES (?, ?, ?) ON CONFLICT(passport_id) DO NOTHING").bind(d.pid, d.trig, d.due)),
    );
  }
  const nowIso = new Date(nowMs).toISOString();
  const cur = (await db.prepare("SELECT cursor_id FROM sweep_cursor WHERE name = ?").bind(CREATOR_WIPE_CURSOR).first<{ cursor_id: string }>())?.cursor_id ?? "";
  const bar = cur.indexOf("|");
  const [afterDue, afterPid] = bar > 0 ? [cur.slice(0, bar), cur.slice(bar + 1)] : ["", ""];
  const { results: ready } = await db
    .prepare(
      "SELECT passport_id, trigger, due_at FROM creator_erasures WHERE wiped_at IS NULL AND due_at <= ? AND (due_at > ? OR (due_at = ? AND passport_id > ?)) " +
        "ORDER BY due_at, passport_id LIMIT ?",
    )
    .bind(nowIso, afterDue, afterDue, afterPid, lim)
    .all<{ passport_id: string; trigger: "self_close" | "removed" | "request"; due_at: string }>();
  const out: CreatorWipeSweepResult = { enqueued: due.length, wiped: [], held: 0, more: due.length === lim || ready.length === lim };
  for (const r of ready) {
    // D-1113(3) (D-1112(5)): a self-closed creator is ERASED by the wipe, the same as an erasure: eraseCreator pays the
    // residual once (given `pay`), wipes, and re-keys the money rows and the case audit rows to a del_ id in its one batch.
    if (r.trigger === "self_close" && PSP_RE.test(r.passport_id)) {
      try {
        await eraseCreator(db, r.passport_id, nowMs, [], pay, "self_close");
      } catch (e) {
        // Unsettled money, an open legal hold, or money that moved during the batch: held, re-read next lap.
        if (!(e instanceof CreatorEraseError) || e.status !== 409) throw e;
        out.held += 1;
        continue;
      }
      out.wiped.push(r.passport_id);
      continue;
    }
    // D-1112(5): a removed creator's residual is paid once first when the caller passed `pay`.
    const open = PSP_RE.test(r.passport_id) && pay ? await payResidual(db, r.passport_id, nowMs, pay) : [];
    if (!PSP_RE.test(r.passport_id) || (await creatorUnsettledNow(db, r.passport_id, nowMs, open)).length > 0) {
      out.held += 1;
      continue;
    }
    try {
      await wipeCreatorPersonalData(db, r.passport_id, r.trigger, () => nowIso, [], open);
    } catch (e) {
      // An open legal hold, or money that landed after the read above, aborted the batch (runWipe's gates): held, re-read next lap.
      if (!(e instanceof CreatorEraseError) || e.status !== 409) throw e;
      out.held += 1;
      continue;
    }
    out.wiped.push(r.passport_id);
  }
  const last = ready.length === lim ? ready[ready.length - 1] : undefined;
  await db
    .prepare("INSERT INTO sweep_cursor (name, cursor_seq, cursor_id, updated_at) VALUES (?, 0, ?, ?) ON CONFLICT(name) DO UPDATE SET cursor_id = excluded.cursor_id, updated_at = excluded.updated_at")
    .bind(CREATOR_WIPE_CURSOR, last ? `${last.due_at}|${last.passport_id}` : "", nowIso)
    .run();
  return out;
}

/**
 * R3-I29-Q9 (D-1111(1), D-1112(5)): a cleared residual under MIN_PAYOUT_CENTS, with nothing clearing or in flight, is paid
 * ONCE with no minimum (creator-payout.ts payErasureResidual). Never throws: a refusal, a failed transfer or a thrown
 * provider error leaves the money on the record. Returns the payout rows THIS attempt minted that are still open
 * (refused after reserving, held, or failed): the caller's money gate does not wait on them (creatorUnsettledNow `exclude`).
 */
async function payResidual(db: DbLike, passportId: string, nowMs: number, pay: EraseCreatorPay): Promise<string[]> {
  const e = await creatorEarnings(db, passportId, nowMs);
  if (!(e.cleared_cents > 0 && e.cleared_cents < MIN_PAYOUT_CENTS && e.pending_cents === 0 && e.in_flight_cents === 0)) return [];
  const minted: string[] = [];
  const mint = pay.mintPayoutId ?? (() => mintId("pay", 20));
  try {
    await payErasureResidual(db, pay.env, passportId, nowMs, () => {
      const id = mint();
      minted.push(id);
      return id;
    }, () => new Date(nowMs).toISOString(), pay.transfer, pay.pubEnv, pay.pubCtx, pay.accountState, pay.accountFetch);
  } catch (err) {
    logError({ cid: "unknown", hop: "http", msg: "creator_erasure_residual_unpaid", err: err instanceof Error ? err.name : typeof err });
  }
  if (minted.length === 0) return [];
  const { results } = await db
    .prepare("SELECT payout_id FROM creator_payouts WHERE passport_id = ? AND status IN ('pending','failed') AND payout_id IN (SELECT value FROM json_each(?))")
    .bind(passportId, JSON.stringify(minted))
    .all<{ payout_id: string }>();
  return results.map((r) => r.payout_id);
}

export async function eraseCreator(
  db: DbLike,
  passportId: string,
  nowMs: number = Date.now(),
  extra: DbStatement[] = [],
  pay?: EraseCreatorPay,
  trigger: "request" | "self_close" = "request",
  /** I34-W3-TC (D-1119(2)): true ONLY from the `creator_erasure` restore replayer below — the replay of a journaled
   * erasure always re-applies whatever money the restored database holds: no unsettled-money refusal (neither the
   * pre-check nor runWipe's commit gate), and the replayer passes no `pay` (no residual payment), so the money rows move
   * to the del_ id as the backup holds them. The close-first, already-erased and legal-hold refusals stand either way. */

  if (await passportUnderLegalHold(db, passportId)) throw new CreatorEraseError(CREATOR_LEGAL_HOLD_SENTENCE);

      .prepare("INSERT INTO creator_erasures (passport_id, trigger, due_at) SELECT ?, NULL, ? WHERE EXISTS (SELECT 1 FROM creator_erasures WHERE passport_id = ? AND rekeyed = 1)")
      .bind(passportId, ts, passportId),
    db
      .prepare("INSERT INTO creator_erasures (passport_id, trigger, due_at, rekeying_to) VALUES (?, ?, ?, ?) ON CONFLICT(passport_id) DO UPDATE SET rekeying_to = excluded.rekeying_to")

    db.prepare("UPDATE creator_payouts SET passport_id = ? WHERE passport_id = ?").bind(del, passportId),
    db.prepare("UPDATE creator_dispute_debits SET passport_id = ? WHERE passport_id = ?").bind(del, passportId),

      .prepare("UPDATE audit_log SET item_id = ?, detail = replace(detail, ?, ?) WHERE item_id = ? AND (actor_id = ? OR action IN (SELECT value FROM json_each(?))) AND action <> ?")
      .bind(del, passportId, del, passportId, `creator:${passportId}`, JSON.stringify([RESERVE_LOSS_ACTION, FRAUD_BANNER_CLEAR_ACTION, ...CREATOR_CASE_AUDIT_ACTIONS, ...CREATOR_ACCOUNT_AUDIT_ACTIONS]), CREATOR_FACE_ERASURE_OWED_ACTION),
    db.prepare("UPDATE creator_erasures SET wiped_at = COALESCE(wiped_at, ?), rekeyed = 1, rekeying_to = NULL WHERE passport_id = ?").bind(ts, passportId),

      .bind(`creator:${del}`, ts, del),

    ], open, !replay);
  } catch (e) {

  }
  await journalRestoreAct("creator_erasure", passportId, ts, trigger === "self_close" ? trigger : undefined);

}

 */
function serverFault(c: Context<{ Bindings: Env }>, route: string, e: unknown): Response {
  logError({ cid: crypto.randomUUID(), hop: "http", route, msg: "unexpected route fault", err: e instanceof Error ? e.name : typeof e });
  return c.json({ error: "internal" }, 500);
}


/** I3-0034 (D-1110(8)): what the first-enrolment door needs — the template mailer (absent when unconfigured: the
 * start then answers 503 email_unavailable), the ONE otp hash under SESSION_SECRET, and PASSKEY_FIRST_CODE_MINUTES.
 * D-1122(1): this mailer sends the first-passkey code only (FIRST_PASSKEY_CODE_TEMPLATE, passkeys.ts' one send through
 * it), and every send also carries `lifetime`, the code's configured lifetime as the creator-first-passkey body states it
 * (passkeys.ts firstPasskeyCodeLifetime), so the mail never says a fixed 10 minutes. */
function firstPasskeyCodeDeps(env: Env): FirstPasskeyCodeDeps | undefined {
  const e = env as unknown as Record<string, unknown>;
  if (!mailerConfigured(e)) return undefined;
  const ttlMs = firstPasskeyCodeTtlMs(e.PASSKEY_FIRST_CODE_MINUTES);
  const mailer = mailerFromEnv(e);
  const lifetime = firstPasskeyCodeLifetime(ttlMs);
  return { mailer: { send: (_template, to, vars) => mailer.send(FIRST_PASSKEY_CODE_TEMPLATE, to, { ...vars, lifetime }) }, hash: (raw) => otpCodeHash(env.SESSION_SECRET, "creator-email", raw), ttlMs };
}
const firstPasskeyContactRefusal = (c: Context<{ Bindings: Env }>): Response => c.json(refusal("contact_email_unverified", FIRST_PASSKEY_VERIFY_EMAIL_FIRST), 409);

/**
 * THE CREATOR'S PASSKEY — enrolment, the signed "touched-by" ceremony, device
 * self-management, and the one signed broadcast that rides the same machinery.
 *
 * BOUNDARY. `/passkey/register/*`, the generic signed creator-action two-step
 * (`/account/creator-action*`), passkey list + revoke (`/account/passkeys`,
 * `/account/passkey/revoke*`), the S71 one-way broadcast (`/account/broadcast*`)
 * and the fan-side mute for it (`/account/settings/broadcasts`).
 *
 * NOT here: `/account/payout-tags*`, which is also a passkey-signed two-step but
 * is about MONEY and is pinned as such by apps/creator-console's payout test
 * reading index.ts; and the drop/collectible mints, which are passkey-signed
 * ceremonies that belong with the releases they create.
 */
export function makeCreatorPasskeyRoutes(): Hono<{ Bindings: Env }> {
  const app = new Hono<{ Bindings: Env }>();


  // --- Passkey "touched-by" provenance (creator-session gated) ---
  // Registering a passkey and signing actions both require an authenticated
  // creator session for the passport (issued only after a proven claim), and a
  // passport that already holds an active passkey adds another only with an
  // assertion from one of them (I3-0034, `startPasskeyRegistration`); a FIRST
  // passkey also needs the code mailed to the verified contact address (Jason
  // I11-Q7, D-1110(8)): start mails it, finish takes it as `email_code`. The
  // creator mark that results is unforgeable: the ledger's general path refuses
  // actor_kind creator_passkey; only a verified assertion here can mint it.

  app.post("/passkey/register/start", anonBodyLimit, async (c) => {
    const id = await creatorId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const cfg = passkeyConfig(c.env);
    if (!cfg) return c.json({ error: "passkeys_unavailable" }, 503);
    // I3-0034: a passport that already holds a passkey adds another only with
    // an assertion from one of them (`startPasskeyRegistration`). Body is
    // optional — empty or `{}` asks; `{assertion}` answers the step-up.
    const raw = await c.req.text();
    if (raw.length > 16 * 1024) return c.json({ error: "too_large" }, 413);
    const stepUp = ((): AssertionResponse | null | undefined => {
      if (raw.trim() === "") return undefined;
      try {
        const o = JSON.parse(raw) as { assertion?: unknown } | null;
        if (!o || typeof o !== "object" || o.assertion === undefined) return undefined;
        const a = o.assertion as Record<string, unknown> | null;
        if (!a || typeof a.credentialId !== "string" || typeof a.clientDataJSON !== "string" || typeof a.authenticatorData !== "string" || typeof a.signature !== "string") return null;
        return { credentialId: a.credentialId, clientDataJSON: a.clientDataJSON, authenticatorData: a.authenticatorData, signature: a.signature };
      } catch {
        return null;
      }
    })();
    if (stepUp === null) return c.json({ error: "bad_assertion" }, 400);
    try {
      const opts = await startPasskeyRegistration(d1(c.env.TRUST_DB), id, cfg, Date.now(), undefined, undefined, stepUp, firstPasskeyCodeDeps(c.env));
      return c.json(opts);
    } catch (e) {
      if (e instanceof PasskeyStepUpRequired) return c.json({ error: e.message, step_up: e.stepUp }, 403);
      if (e instanceof FirstPasskeyContactRequired) return firstPasskeyContactRefusal(c);
      if (e instanceof PasskeyError) return c.json(errorBody(e), e.status);
      return serverFault(c, "/passkey/register/start", e);
    }
  });

  app.post("/passkey/register/finish", anonBodyLimit, async (c) => {
    // I3-0338: the admitted epoch rides to the INSERT, which re-asserts it.
    const adm = await creatorAdmission(c);
    if (!adm) return c.json({ error: "unauthorized" }, 401);
    const id = adm.id;
    const cfg = passkeyConfig(c.env);
    if (!cfg) return c.json({ error: "passkeys_unavailable" }, 503);
    const raw = await c.req.text();
    if (raw.length > 16 * 1024) return c.json({ error: "too_large" }, 413);
    // I3-0034: `email_code` is the code mailed at start (first enrolment only; ignored on the step-up arm).
    let emailCode: unknown;
    const body = ((): AttestationResponse | null => {
      try {
        const o = JSON.parse(raw) as Record<string, unknown>;
        if (typeof o.clientDataJSON !== "string" || typeof o.attestationObject !== "string") return null;
        emailCode = o.email_code;
        return { clientDataJSON: o.clientDataJSON, attestationObject: o.attestationObject };
      } catch {
        return null;
      }
    })();
    if (!body) return c.json({ error: "bad_response" }, 400);
    try {
      const r = await finishPasskeyRegistration(d1(c.env.TRUST_DB), id, body, cfg, Date.now(), undefined, adm.ep, { code: emailCode, hash: (s) => otpCodeHash(c.env.SESSION_SECRET, "creator-email", s) });
      return c.json({ ok: true, ...r });
    } catch (e) {
      if (e instanceof FirstPasskeyContactRequired) return firstPasskeyContactRefusal(c);
      if (e instanceof PasskeyError) return c.json(errorBody(e), e.status);
      return serverFault(c, "/passkey/register/finish", e);
    }
  });

  app.post("/account/creator-action/start", anonBodyLimit, async (c) => {
    const id = await creatorId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const cfg = passkeyConfig(c.env);
    if (!cfg) return c.json({ error: "passkeys_unavailable" }, 503);
    const raw = await c.req.text();
    if (raw.length > 32 * 1024) return c.json({ error: "too_large" }, 413);
    const body = ((): { actions?: unknown } | null => {
      try {
        return JSON.parse(raw) as { actions?: unknown };
      } catch {
        return null;
      }
    })();
    const actions = body ? parseActions(body.actions) : null;
    if (!actions) return c.json({ error: "actions required" }, 400);
    try {
      const drafts = buildCreatorDrafts(id, actions);
      const opts = await startAssertionForAction(d1(c.env.TRUST_DB), id, drafts, cfg, Date.now());
      return c.json(opts);
    } catch (e) {
      if (e instanceof PasskeyError) return c.json(errorBody(e), e.status);
      return serverFault(c, "/account/creator-action/start", e);
    }
  });

  app.post("/account/creator-action", anonBodyLimit, async (c) => {
    const id = await creatorId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const cfg = passkeyConfig(c.env);
    if (!cfg) return c.json({ error: "passkeys_unavailable" }, 503);
    const raw = await c.req.text();
    if (raw.length > 32 * 1024) return c.json({ error: "too_large" }, 413);
    const body = ((): { actions: CreatorAction[]; assertion: AssertionResponse } | null => {
      try {
        const o = JSON.parse(raw) as { actions?: unknown; assertion?: unknown };
        const actions = parseActions(o.actions);
        const a = o.assertion as Record<string, unknown> | undefined;
        if (!actions || !a || typeof a.credentialId !== "string" || typeof a.clientDataJSON !== "string" || typeof a.authenticatorData !== "string" || typeof a.signature !== "string") return null;
        return { actions, assertion: { credentialId: a.credentialId, clientDataJSON: a.clientDataJSON, authenticatorData: a.authenticatorData, signature: a.signature } };
      } catch {
        return null;
      }
    })();
    if (!body) return c.json({ error: "actions and assertion required" }, 400);
    const db = d1(c.env.TRUST_DB);
    const nowMs = Date.now();

    // X1 — THE ASSERTION CEILING. `account_added` on this door is the ONE place
    // a passport can claim an account on its own authority, and it had no bound
    // of any kind: 20 drafts a batch, unlimited batches, and the first party to
    // assert a key owns it against anybody who can prove control of it. See
    // assertions.ts for why the ceiling is on OUTSTANDING UNPROVEN assertions
    // rather than on accounts.
    //
    // The RATE is consumed here, before the ceremony, and only for the adds this
    // request actually carries — one atomic statement per add, so a burst cannot
    // walk past it. The CEILING rides in `extraStatements` below, inside the
    // ledger's own batch, because a route-level count is check-then-act.
    const adds = body.actions.filter((a) => a.event_type === "account_added");
    // THE RATE IS READ HERE AND CHARGED AFTER THE CEREMONY, not consumed up
    // front (review, HIGH). Consuming before `appendCreatorSigned` meant a
    // STOLEN SESSION carrying a garbage assertion could spend the passport's
    // whole daily budget — ten adds, `signature: "garbage"`, and the real
    // creator's next add answered 429 for 24 hours. The passkey exists so that a
    // session alone cannot act; a limiter charged ahead of it handed a session
    // alone the power to deny the passkey holder their own door.
    //
    // Read-then-charge is check-then-act, and here that is the right trade and a
    // bounded one: the CEILING is the atomic bound (it rides the ledger's batch
    // and cannot be raced), and the worst a burst wins is a few adds over the
    // daily rate while still stopped dead at ten outstanding.
    // #932b (I16-W05A): the read-then-charge above was check-then-act — N
    // concurrent requests at cap-1 each read room for one add and all landed.
    // The rate is now a CHECKED CLAIM (`consumeRateLimit` is one atomic
    // statement and its boolean is the verdict), taken per add before the
    // ceremony, and HANDED BACK on a refused claim and on any failure before the
    // append commits — so a garbage assertion still spends nothing once it
    // returns (agencies.ts appendAgencyAction is the same shape).
    const bucket = `assert:${id}`;
    let claimed = 0;
    let committed = false;
    const refundClaimed = async (): Promise<void> => {
      for (; claimed > 0; claimed--) await refundRateLimit(db, bucket, nowMs, DAY_MS);
    };
    for (let i = 0; i < adds.length; i += 1) {
      if (!(await consumeRateLimit(db, bucket, nowMs, ASSERTIONS_PER_DAY, DAY_MS))) {
        await refundClaimed();
        return c.json(capRefusal("too many new accounts today — prove the ones already on your passport, or try again tomorrow"), 429);
      }
      claimed += 1;
    }

    try {
      const drafts = buildCreatorDrafts(id, body.actions);
      // One guard per add, each riding the SAME atomic batch as the append. The
      // key is derived exactly as the fold derives it, so the guard and the row
      // it bounds cannot name different accounts.
      const guards = [
        ...adds.map((a) => {
          const p = a.payload;
          const key = subjectForAccount(String(p.platform), String(p.id_kind), String(p.id)).canonical;
          return assertionCeilingStatement(db, id, key, nowMs);
        }),
        // I3-0603 (D-0838(1)): a likeness revoke owes the image service its
        // mirror; the owed row commits with the revoke or not at all.
        ...imageConsentOwedStatements(db, drafts, new Date(nowMs).toISOString()), ...faceRevokeOwedForDrafts(db, drafts, new Date(nowMs).toISOString()), // I65-HOLD F1 (W0-5): under a hold, the marker instead
      ];
      const events = await appendCreatorSigned(db, id, drafts, body.assertion, cfg, nowMs, undefined, guards);
      committed = true;
      // After commit: the first call now; a failure stays owed for the five-minute sweep.
      const imagePeer = imageConsentPeerFromEnv(c.env as unknown as Record<string, unknown>);
      if (imagePeer && drafts.some((d) => d.event_type === "likeness_consent_revoked") && !(await faceDataHeld(db, id))) await propagateImageConsentRevoke(db, imagePeer, id); // I65-HOLD F1 (W0-5): no send under a hold
      return c.json({ ok: true, applied: events.length });
    } catch (e) {
      // The claim is kept only for adds that LANDED: a ceremony that failed
      // verification, or any throw before the append committed, hands it back.
      if (!committed) await refundClaimed();
      if (e instanceof PasskeyError) return c.json(errorBody(e), e.status);
      // THE CEILING'S REFUSAL IS A SENTENCE, NOT A CONSTRAINT NAME. It reaches
      // here as a raw NOT NULL failure out of D1, and a creator who has hit it
      // needs to be told what to do about it — which is to prove one.
      // NARROW: the CEILING's failure is specifically the NOT NULL on `ok` — the
      // column that exists to carry the CASE's NULL. A PRIMARY KEY collision on
      // the same table means something else entirely (a stale row for a key the
      // fold no longer has), and reporting it as "you hold too many unproven
      // accounts" would send a creator to prove accounts that would not help.
      if (e instanceof Error && /NOT NULL/i.test(e.message) && /assertion_gate/.test(e.message)) {
        return c.json({ error: `you can hold ${MAX_OUTSTANDING_ASSERTIONS} unproven accounts at a time — prove one of the accounts already on your passport to free a slot` }, 409);
      }
      // #114e (I16-W05A): a lost ledger head race is ours and transient — the one 503 every appending door gives.
      if (e instanceof LedgerContentionError) return contentionResponse(c);
      return serverFault(c, "/account/creator-action", e);
    }
  });

  // --- S94: passkey self-management (list + revoke a device) ---
  // A creator lists their registered passkeys and can revoke one, PROVEN by an
  // assertion from a current passkey (a stolen session alone can't strip devices).
  app.get("/account/passkeys", async (c) => {
    const id = await creatorId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const db = d1(c.env.TRUST_DB);
    // needs_reenroll flags a BREAK-GLASS reset (0 active passkeys + >=1 revoked) so
    // the console can prompt the creator to register a fresh device (S96 recovery).
    //
    // `recovery_aftermath` is the SEPARATE, longer-lived signal, and separating
    // them is the fix for a real defect: the console gated its explanation of
    // what ELSE a recovery clears (every payout destination, every agency
    // grant) on `needs_reenroll`, so it never appeared for a hijack resolved
    // without `reset_passkeys` — which is not a forced action — and it deleted
    // itself the moment the creator registered a device, i.e. exactly when they
    // started needing it. Keyed to the recovery, 30 days, derived from the
    // audit row (recovery.ts).
    const aftermath = await recoveryAftermath(db, id, Date.now());
    return c.json({
      passkeys: await listPasskeys(db, id),
      needs_reenroll: await needsPasskeyReenroll(db, id),
      recovery_aftermath: aftermath.recent,
      recovery_resolved_at: aftermath.resolved_at,
      // I3-2113: which clean-up the recent recovery actually ran, so the panel
      // names only those (only `hijack` forces both, recovery.ts FORCED_ACTIONS).
      recovery_cleared: aftermath.cleared,
    });
  });

  app.post("/account/passkey/revoke/start", anonBodyLimit, async (c) => {
    const id = await creatorId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const cfg = passkeyConfig(c.env);
    if (!cfg) return c.json({ error: "passkeys_unavailable" }, 503);
    const raw = await c.req.text();
    if (raw.length > 4 * 1024) return c.json({ error: "too_large" }, 413);
    const body = ((): { credential_id?: unknown } | null => {
      try {
        return JSON.parse(raw) as { credential_id?: unknown };
      } catch {
        return null;
      }
    })();
    if (!body || typeof body.credential_id !== "string") return c.json({ error: "credential_id required" }, 400);
    try {
      return c.json(await startRevokeAssertion(d1(c.env.TRUST_DB), id, body.credential_id, cfg, Date.now()));
    } catch (e) {
      if (e instanceof PasskeyError) return c.json(errorBody(e), e.status);
      return serverFault(c, "/account/passkey/revoke/start", e);
    }
  });

  app.post("/account/passkey/revoke", anonBodyLimit, async (c) => {
    // I12-RG-AUTH-01: the admitted epoch rides into revokePasskey's verdict.
    const adm = await creatorAdmission(c);
    if (!adm) return c.json({ error: "unauthorized" }, 401);
    const id = adm.id;
    const cfg = passkeyConfig(c.env);
    if (!cfg) return c.json({ error: "passkeys_unavailable" }, 503);
    // BEFORE the write, not after — the same order `POST /account/fan-passkey/revoke`
    // uses and for the same reason: a successful revoke signs every creator
    // session out (the epoch bump rides inside it), so a deploy that cannot mint
    // the caller's replacement would leave the creator holding nothing. Fail
    // closed first.
    if (!usableSessionSecret(c.env.SESSION_SECRET)) return c.json({ error: "sessions_unavailable" }, 503);
    const raw = await c.req.text();
    if (raw.length > 16 * 1024) return c.json({ error: "too_large" }, 413);
    const body = ((): { credential_id: string; assertion: AssertionResponse } | null => {
      try {
        const o = JSON.parse(raw) as { credential_id?: unknown; assertion?: unknown };
        const a = o.assertion as Record<string, unknown> | undefined;
        if (typeof o.credential_id !== "string" || !a || typeof a.credentialId !== "string" || typeof a.clientDataJSON !== "string" || typeof a.authenticatorData !== "string" || typeof a.signature !== "string") return null;
        return { credential_id: o.credential_id, assertion: { credentialId: a.credentialId, clientDataJSON: a.clientDataJSON, authenticatorData: a.authenticatorData, signature: a.signature } };
      } catch {
        return null;
      }
    })();
    if (!body) return c.json({ error: "credential_id and assertion required" }, 400);
    try {
      const nowMs = Date.now();
      const r = await revokePasskey(d1(c.env.TRUST_DB), id, body.credential_id, body.assertion, cfg, nowMs, undefined, adm.ep);
      // The caller's own token was minted below the new epoch, so it died with
      // every other device's. Re-mint it: the creator who pressed the button stays
      // in and the device they revoked — including the one they lost — is out.
      // This is a refresh of a session the caller already proved (with a passkey
      // assertion, moments ago), not a new grant.
      const session = await issueSession(c.env.SESSION_SECRET, { kind: "creator", id, ep: r.session_epoch }, Math.floor(nowMs / 1000));
      if (!session) return c.json({ error: "sessions_unavailable" }, 503);
      return c.json({ ok: true, revoked: r.revoked, session_epoch: r.session_epoch, session });
    } catch (e) {
      if (e instanceof PasskeyError) return c.json(errorBody(e), e.status);
      return serverFault(c, "/account/passkey/revoke", e);
    }
  });

  // --- S71: one-way signed creator broadcast (Cluster D) ---
  // Two legs like creator-action: /start sanitizes the message, checks the
  // Creator-Pro-gated frequency cap, and returns a passkey challenge bound to the
  // exact broadcast draft; the append leg re-checks the cap, appendCreatorSigned's
  // the message (unforgeably "by creator"), then delivery rides the alerts fold.
  //
  // flag 2: the draft also carries the creator's AUDIENCE choice, resolved
  // server-side on both legs (the client never chooses what is signed — it echoes
  // back the exact values /start returned, and a mismatch changes the action hash
  // and fails the assertion). So "this went out public" is a passkey-signed fact,
  // not a server annotation.

  app.post("/account/broadcast/start", anonBodyLimit, async (c) => {
    const id = await creatorId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const cfg = passkeyConfig(c.env);
    if (!cfg) return c.json({ error: "passkeys_unavailable" }, 503);
    const raw = await c.req.text();
    if (raw.length > 8 * 1024) return c.json({ error: "too_large" }, 413);
    const body = ((): { message?: unknown; audience?: unknown } | null => {
      try {
        return JSON.parse(raw) as { message?: unknown; audience?: unknown };
      } catch {
        return null;
      }
    })();
    if (!body) return c.json({ error: "message required" }, 400);
    const db = d1(c.env.TRUST_DB);
    const nowMs = Date.now();
    try {
      await assertBroadcastAllowed(db, id, nowMs); // fast fail before the ceremony
      const message = sanitizeMessage(body.message);
      const audience = resolveAudience(body.audience); // absent → public; unknown → 400
      const broadcast_id = newBroadcastId();
      const drafts = buildCreatorDrafts(id, [{ event_type: "creator_broadcast", payload: { broadcast_id, message, audience } }]);
      const opts = await startAssertionForAction(db, id, drafts, cfg, nowMs);
      // Echo the exact draft the client must re-send with the assertion, so both
      // legs sign the identical action hash.
      return c.json({ ...opts, broadcast_id, message, audience });
    } catch (e) {
      if (e instanceof BroadcastError) return c.json(errorBody(e), e.status);
      if (e instanceof PasskeyError) return c.json(errorBody(e), e.status);
      return serverFault(c, "/account/broadcast/start", e);
    }
  });

  app.post("/account/broadcast", anonBodyLimit, async (c) => {
    const id = await creatorId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const cfg = passkeyConfig(c.env);
    if (!cfg) return c.json({ error: "passkeys_unavailable" }, 503);
    const raw = await c.req.text();
    if (raw.length > 8 * 1024) return c.json({ error: "too_large" }, 413);
    const body = ((): { broadcast_id: string; message: string; audience: unknown; assertion: AssertionResponse } | null => {
      try {
        const o = JSON.parse(raw) as { broadcast_id?: unknown; message?: unknown; audience?: unknown; assertion?: unknown };
        const a = o.assertion as Record<string, unknown> | undefined;
        if (typeof o.broadcast_id !== "string" || typeof o.message !== "string" || !a || typeof a.credentialId !== "string" || typeof a.clientDataJSON !== "string" || typeof a.authenticatorData !== "string" || typeof a.signature !== "string") return null;
        return { broadcast_id: o.broadcast_id, message: o.message, audience: o.audience, assertion: { credentialId: a.credentialId, clientDataJSON: a.clientDataJSON, authenticatorData: a.authenticatorData, signature: a.signature } };
      } catch {
        return null;
      }
    })();
    if (!body) return c.json({ error: "broadcast_id, message, and assertion required" }, 400);
    const db = d1(c.env.TRUST_DB);
    const nowMs = Date.now();
    try {
      // The message and broadcast_id must match what was signed at /start (the
      // action hash binds them), so re-sanitize to the same canonical form the
      // challenge covered. The CAP is claimed further down, immediately before the
      // append — everything between here and there can still refuse the send, and
      // a claim taken early is a slot a creator can lose to their own typo.
      const message = sanitizeMessage(body.message);
      // Same for the audience: re-resolved here, never trusted as-is. A client
      // that echoes a DIFFERENT audience than /start bound rebuilds a different
      // draft → a different action hash → the assertion fails. Nothing downgrades
      // a followers-only send to public (or the reverse) without a new signature.
      const audience = resolveAudience(body.audience);
      if (!/^bcast_[0-9a-f]{16}$/.test(body.broadcast_id)) return c.json({ error: "bad broadcast_id" }, 400);
      const drafts = buildCreatorDrafts(id, [{ event_type: "creator_broadcast", payload: { broadcast_id: body.broadcast_id, message, audience } }]);
      // ATOMIC cap claim, then the signed append; a failed append hands the slot
      // straight back. Read-checking the cap here instead let four pre-opened
      // ceremonies land four broadcasts against a cap of one — measured. See
      // `claimBroadcastSlot`.
      await claimBroadcastSlot(db, id, nowMs);
      let events: Awaited<ReturnType<typeof appendCreatorSigned>>;
      try {
        events = await appendCreatorSigned(db, id, drafts, body.assertion, cfg, nowMs);
      } catch (e) {
        await releaseBroadcastSlot(db, id);
        throw e;
      }
      // I3-1273 (I22-W1-AUTH): the signed broadcast is ON THE LEDGER now, so the answer is `ok` whatever the
      // cap-log bookkeeping below does — a failure there is logged, never reported to the creator as a failed send.
      const sent = events[0];
      if (sent) {
        try {
          await recordBroadcast(db, id, body.broadcast_id, sent.event_id);
        } catch (e) {
          logError({ cid: crypto.randomUUID(), hop: "http", route: "/account/broadcast", msg: "broadcast cap-log write failed after a committed send", err: e instanceof Error ? e.name : typeof e });
        }
      }
      return c.json({ ok: true, broadcast_id: body.broadcast_id, audience });
    } catch (e) {
      if (e instanceof BroadcastError) return c.json(errorBody(e), e.status);
      if (e instanceof PasskeyError) return c.json(errorBody(e), e.status);
      // #114e (I16-W05A): a lost ledger head race is ours and transient — the one 503 every appending door gives.
      if (e instanceof LedgerContentionError) return contentionResponse(c);
      return serverFault(c, "/account/broadcast", e);
    }
  });

  // Fan mute for promotional broadcasts (safety alerts still delivered).
  app.post("/account/settings/broadcasts", anonBodyLimit, async (c) => {
    const id = await fanId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    const body = (await c.req.json().catch(() => null)) as { muted?: unknown } | null;
    if (!body || typeof body.muted !== "boolean") return c.json({ error: "muted (boolean) required" }, 400);
    await setBroadcastsMuted(d1(c.env.TRUST_DB), id, body.muted);
    return c.json({ ok: true, broadcasts_muted: body.muted });
  });

  return app;
}
