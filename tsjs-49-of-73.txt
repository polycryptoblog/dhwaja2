/**
 * Env predicates that MORE THAN ONE SURFACE has to agree on.
 *
 * Every rule in here was, until this module existed, written twice: once where
 * the feature decides whether it can run, and once in `configHealth` as a
 * loose `has(KEY)` presence check. The two disagreed, and the disagreement is
 * not cosmetic — `/ops/config` is the only pre-flight an operator gets, and a
 * row that reports a feature configured when the feature itself will refuse it
 * is worse than no row at all. That exact failure has already shipped: the
 * console reported `email: true` off the `EMAIL_API_*` alert rail while fan
 * sign-in answered 503 because `POSTMARK_TOKEN` was absent — two different
 * mail rails, one boolean, and an operator reading "email is fine".
 *
 * So: the predicate lives HERE, the feature calls it, and the console row calls
 * the same function. `has()` is never the right answer for a value with a
 * shape.
 *
 * A URL-shaped var is checked for its SHAPE, not its presence. `APP_ORIGIN` set
 * to `dhwaja.app` (no scheme) or `http://dhwaja.app` is not a usable origin:
 * the SSR pages will refuse to emit a canonical tag from it, so the honest
 * console row is `false`.
 */
import { logWarn } from "./log.js";

/**
 * The shape an ORIGIN-valued env var must have: an https scheme and a bare
 * host, no path, no port, no credentials. Exported as the source-of-truth
 * literal so a second copy of this rule elsewhere in the tree can be pinned
 * against it by a guard rather than left to drift.
 */
export const HTTPS_ORIGIN_RE = /^https:\/\/[a-z0-9.-]+$/i;

/**
 * The shape a RESOLVER-valued env var must have: https, a bare host, and an
 * optional path (`https://dns.google/resolve`). Query strings are appended by
 * the caller, never configured.
 */
export const HTTPS_ENDPOINT_RE = /^https:\/\/[a-z0-9.-]+(\/[^\s]*)?$/i;

/** I62 P12 (I60-10, I61-Q3 "every setting"): one env setting's value, padding trimmed; a non-string, a blank value or
 * one containing TODO in any case (a URL whose host is TODO included) is unset (null), and each caller keeps its own
 * unset meaning. It takes the value, not the name, so every caller still reads `env.<NAME>` where the censuses look. */
export function settingValue(v: unknown): string | null {
  if (typeof v !== "string") return null;
  const t = v.trim();
  return t.length > 0 && !/TODO/i.test(t) ? t : null;
}

/** I63 C2 (D-1537 Q6, I62-Q6 "One trailing slash on an origin setting is padding": "yes"): an origin setting's value
 * before its shape check — settingValue, then ONE trailing `/` taken off. A second slash is left on, so a caller's shape
 * check refuses it. route-helpers.ts redeemLinks reads APP_ORIGIN through this, then its own port-admitting check. */
export function originPadded(v: unknown): string | null {
  const s = settingValue(v);
  if (s === null) return null;
  return s.endsWith("/") ? s.slice(0, -1) : s;
}

/** I62 FX (D-1527, D-1528(a)): an ORIGIN setting's value — originPadded (settingValue, then ONE trailing `/` taken off
 * as padding, I61-Q3), then HTTPS_ORIGIN_RE. A second slash or a path after the host is left on, so the shape check
 * refuses it. */
function originSetting(v: unknown): string | null {
  const o = originPadded(v);
  return o !== null && HTTPS_ORIGIN_RE.test(o) ? o : null;
}

/** `APP_ORIGIN` — where the fan SPA is served. Used to build in-app links in
 * alert mail and the sign-out-everywhere link; absent/misshapen ⇒ those links
 * are omitted rather than guessed from a client-controlled Host header. */
export function appOriginFromEnv(env: Record<string, unknown>): string | null {
  return originSetting(env.APP_ORIGIN);
}

/** `CONSOLE_ORIGIN` — where the CREATOR CONSOLE SPA is served (a different
 * origin from the fan `APP_ORIGIN`). H8: the OAuth-claim callback hands its
 * one-time session code to the CONSOLE (it uses the console's `#/signin?oauth=…`
 * format and only the console has a route for it); redirecting to the fan app
 * dropped it on the floor and OAuth claiming — the primary path for X — could
 * never complete. Absent/misshapen ⇒ the callback shows a static success page
 * rather than guessing an origin. */
export function consoleOriginFromEnv(env: Record<string, unknown>): string | null {
  return originSetting(env.CONSOLE_ORIGIN);
}

/** `PUBLIC_BASE_URL` — where the server-rendered public pages (`/p/:id`,
 * `/discover`, `/browse/:category`, `/is/…`) are served. It is the ONLY source
 * for `og:url` and `<link rel=canonical>`; absent ⇒ the tags are omitted, which
 * costs the share unfurl and every crawlable page its canonical. Silent, and
 * invisible from every other surface — hence the console row. */
export function publicBaseUrlFromEnv(env: Record<string, unknown>): string | null {
  return originSetting(env.PUBLIC_BASE_URL);
}

/**
 * THE DoH RESOLVERS THIS SERVICE WILL ASK (QUEUE #527, I3-2123). The resolver
 * DECIDES every brand-domain control proof — whatever TXT answer it returns is
 * the proof — so a shape check alone let any https URL that parsed become the
 * oracle for who owns `netflix.com`. Exact endpoints, all JSON (`application/
 * dns-json`, `?name=&type=TXT`, which is what `dohTxtLookup` speaks): adding
 * one is a reviewed code change, never an env edit.
 */
export const DOH_RESOLVER_ALLOWLIST: readonly string[] = ["https://dns.google/resolve", "https://cloudflare-dns.com/dns-query"];

/** What a well-shaped but UNKNOWN `DOH_RESOLVER_URL` is replaced with. */
export const DOH_RESOLVER_DEFAULT = "https://dns.google/resolve";

/** Is this exact endpoint on the allowlist? Host case is ignored (DNS is
 * case-insensitive); path, query and everything else must match exactly. */
export function isAllowlistedDohResolver(url: string): boolean {
  let u: URL;
  try {
    u = new URL(url.trim());
  } catch {
    return false;
  }
  return DOH_RESOLVER_ALLOWLIST.some((a) => {
    const w = new URL(a);
    return u.protocol === w.protocol && u.host === w.host && u.pathname === w.pathname && u.search === "" && u.hash === "" && u.username === "" && u.password === "";
  });
}

/** `DOH_RESOLVER_URL` — the FIXED DNS-over-HTTPS resolver the brand-domain
 * control proof queries (S76). Absent/misshapen ⇒ `POST /claim/domain/verify`
 * answers 503 and no brand domain can be claimed at all. The value is trimmed
 * first, matching the route. Well-shaped but NOT on `DOH_RESOLVER_ALLOWLIST` ⇒
 * `DOH_RESOLVER_DEFAULT`, with a refusal logged on every read naming the value
 * refused — never a silent swap (QUEUE #527). */
export function dohResolverFromEnv(env: Record<string, unknown>): string | null {
  const v = env.DOH_RESOLVER_URL;
  if (typeof v !== "string") return null;
  const t = v.trim();
  if (!(t.length > 0 && HTTPS_ENDPOINT_RE.test(t))) return null;
  if (isAllowlistedDohResolver(t)) return t;
  logWarn({ cid: "config", hop: "http", msg: "doh_resolver_refused", event: "doh_resolver_refused", configured: t, using: DOH_RESOLVER_DEFAULT });
  return DOH_RESOLVER_DEFAULT;
}

/**
 * The shape a PASSKEY origin may have — and it is TWO shapes, because an
 * Android app's WebAuthn origin is not a URL.
 *
 *  • A WEB origin: `https://host[:port]`. The port is admitted because a
 *    staging/dev deployment is served on one; hosts are case-insensitive, so
 *    this half carries the `i`.
 *  • An ANDROID app origin: Credential Manager signs `clientDataJSON` with
 *    `android:apk-key-hash:<base64url SHA-256 of the app's signing
 *    certificate>` — a 32-byte digest, base64url, padding stripped, so exactly
 *    43 characters drawn from `A-Za-z0-9_-`. Without this half an operator
 *    could not EXPRESS that origin at all, and every Android ceremony died in
 *    `passkeys.ts` with "origin not allowed" — including the register-a-passkey
 *    step that is the whole point of account recovery.
 *
 * TWO REGEXES, NOT ONE ALTERNATION UNDER `/i`, and the split is the point. A
 * base64url digest is case-SENSITIVE and so is the literal prefix the platform
 * emits. An `i` flag over that half admits `ANDROID:APK-KEY-HASH:…` and any
 * mis-cased digest — entries an operator can CONFIGURE and that can never
 * MATCH, which is a silent misconfiguration wearing a healthy console row, not
 * a convenience. Padding is refused for exactly the same reason: Credential
 * Manager emits none, so `…=` would be a listed entry no ceremony ever equals.
 *
 * NEITHER REGEX IS A MATCHING RULE. `passkeys.ts` compares the ceremony's
 * `clientData.origin` against the configured list with an exact `includes`;
 * these decide only what an operator may EXPRESS. Widening the shape therefore
 * cannot admit an origin nobody listed.
 */
export const PASSKEY_WEB_ORIGIN_RE = /^https:\/\/[a-z0-9.-]+(?::\d+)?$/i;
export const PASSKEY_ANDROID_ORIGIN_RE = /^android:apk-key-hash:[A-Za-z0-9_-]{43}$/;

/**
 * `PASSKEY_RP_ID` — the WebAuthn relying-party id. `passkeyConfig` trims it and
 * treats an empty result as "not configured"; `configHealth` used to read it
 * with a bare `has()`, which is presence BEFORE the trim, and the comment that
 * shipped with that line — "the RP id has no shape worth checking beyond
 * presence, so it stays on `has()`" — was simply wrong. It has exactly the
 * shape `passkeyConfig` gives it. A value of `"\t"`, `"\n"`, `" "` or `" \t "`
 * is non-empty to `has()` and empty to the route, so the console reported
 * `passkeys: true` over a deployment where all four passkey routes answered
 * 503. One predicate now, read by both.
 *
 * A `TODO`-bearing value is refused for the same reason `usableSessionSecret`
 * refuses one: a placeholder that reaches production is an unconfigured
 * deployment wearing a configured row, and refusing it fails closed (503),
 * which is the direction this repo always leans.
 *
 * WHAT THIS DOES NOT CHECK, said plainly: an RP id is a registrable domain
 * suffix and this predicate does not test that. `https://dhwaja.app` or
 * `dhwaja.app/` are accepted here and by `passkeyConfig`, and every ceremony
 * would then fail `rpIdHashMatches` — a real trap, and an unfixed one. It is
 * left alone deliberately: narrowing the shape would change what the ROUTES
 * accept, `localhost` and bare hosts both have to keep working, and the finding
 * this predicate answers is the trim disagreement, not the domain grammar.
 */
export function passkeyRpIdFromEnv(env: Record<string, unknown>): string | null {
  const v = env.PASSKEY_RP_ID;
  if (typeof v !== "string") return null;
  const t = v.trim();
  return t.length > 0 && !/TODO/i.test(t) ? t : null;
}

/** What `PASSKEY_ORIGINS` parsed into: what survived, and what did not. */
export interface PasskeyOriginParse {
  kept: string[];
  dropped: string[];
}

/**
 * Warn ONCE PER DISTINCT VALUE, not once per parse. `passkeyConfig` runs this
 * on every passkey request, and a per-request warning on a hot path is a
 * log-flooding bug rather than a diagnostic — `corsAllowedOrigins` caches for
 * the same reason and this mirrors it, cap and all. The key is built from
 * config only, so nothing a caller sends can grow it.
 */
const warnedOrigins = new Set<string>();
const WARNED_MAX = 32;

/**
 * `PASSKEY_ORIGINS` — the comma-separated allowlist of origins a WebAuthn
 * ceremony may claim. Entries are trimmed and any entry that is not one of the
 * two shapes above is DROPPED, never repaired; an entirely malformed value
 * therefore leaves an empty list, which is what makes `passkeyConfig` return
 * null and every passkey route answer 503.
 *
 * A DROPPED ENTRY IS ANNOUNCED, and it is announced PER ENTRY. The whole-list
 * case was closed and the per-entry case was not: an operator who wrote
 * `https://dhwaja.app,android:apk-key-hash:…=` — one padded digest, the
 * likeliest form of that typo there is — lost the Android origin, kept the web
 * one, and got `passkeys: true` from the console. `PasskeyManager.kt` says what
 * that costs in its own words: the loss is "invisible on iOS and fatal to the
 * LAST STEP OF RECOVERY here". The value logged is an operator-set Worker var
 * read from `env` — no request header, body, path or query reaches it — so this
 * cannot write caller-chosen text into the logs. The behaviour is unchanged:
 * the entry is still dropped, never repaired, and one typo still cannot take
 * the rest of the list down with it.
 */
export function passkeyOriginsParse(env: Record<string, unknown>): PasskeyOriginParse {
  const raw = env.PASSKEY_ORIGINS;
  if (typeof raw !== "string") return { kept: [], dropped: [] };
  const kept: string[] = [];
  const dropped: string[] = [];
  for (const part of raw.split(",")) {
    const entry = part.trim();
    if (entry === "") continue; // an empty slot is a formatting artifact, not a typo
    // I63 C2 (W0-10, I62-CC-16): the setting rule per ENTRY — an entry holding TODO in any case is dropped (and announced
    // below like every dropped entry); the rest of the list is kept. Never settingValue on the whole value.
    if (settingValue(entry) !== null && (PASSKEY_WEB_ORIGIN_RE.test(entry) || PASSKEY_ANDROID_ORIGIN_RE.test(entry))) kept.push(entry);
    else dropped.push(entry);
  }
  return { kept, dropped };
}

export function passkeyOriginsFromEnv(env: Record<string, unknown>): string[] {
  const { kept, dropped } = passkeyOriginsParse(env);
  for (const entry of dropped) {
    const key = `${typeof env.PASSKEY_ORIGINS === "string" ? env.PASSKEY_ORIGINS : ""}\u0000${entry}`;
    if (warnedOrigins.has(key)) continue;
    if (warnedOrigins.size >= WARNED_MAX) warnedOrigins.clear();
    warnedOrigins.add(key);
    logWarn({ cid: "config", hop: "http", msg: "passkeys: ignoring malformed PASSKEY_ORIGINS entry (expected https://host[:port] or android:apk-key-hash:<43-char base64url, unpadded>)", env_key: "PASSKEY_ORIGINS", entry });
  }
  return kept;
}

/**
 * The shape a CLOUDFLARE ZONE ID has: exactly 32 LOWERCASE hex characters. The
 * case matters and is not pedantry — the id is pasted straight into the purge
 * URL path, and an operator who pasted a display value, a zone NAME
 * (`dhwaja.app`), or an account id gets a request that 404s at the API. That
 * failure is invisible from every other surface, and the surface it is invisible
 * FROM is a legal takedown. Refusing the value here means the takedown reports
 * "not configured" instead of "purge failed", which is the difference between an
 * operator fixing a var and an operator escalating a Cloudflare outage.
 */
export const CF_ZONE_ID_RE = /^[0-9a-f]{32}$/;

/**
 * The shape a CLOUDFLARE API TOKEN has: an opaque base64url-ish string. A real
 * token is 40 characters of `[A-Za-z0-9_-]`; the bound is left loose (20–200)
 * because the token format is Cloudflare's to change, and this is a REFUSAL
 * rule, not a validator.
 *
 * WHAT IT IS ACTUALLY FOR: the copy-paste artifact. A value carrying a wrapping
 * quote, a trailing newline, or a `Bearer ` prefix is non-empty, TODO-free, and
 * completely unusable — it becomes an `Authorization` header that can never
 * authenticate. Every such deployment would look configured to `configHealth`
 * and fail only at the moment an operator is taking illegal content down. The
 * shape check turns that into a fail-closed 503 an operator can see beforehand.
 */
export const CF_API_TOKEN_RE = /^[A-Za-z0-9_.~-]{20,200}$/;

/** The three values a zone purge-by-URL call needs, and it needs all three. */
export interface EdgePurgeConfig {
  /** Cloudflare zone id — the `:zone_id` of `POST /zones/:zone_id/purge_cache`. */
  zoneId: string;
  /** API token with the `Zone.Cache Purge` permission on that zone. */
  apiToken: string;
  /** https origin the media object is CACHED UNDER — the purge is by URL, so a
   * URL that does not match the cached one purges nothing and reports success. */
  mediaOrigin: string;
}

/**
 * `CF_PURGE_ZONE_ID` + `CF_PURGE_API_TOKEN` (+ `PUBLIC_BASE_URL`) — the edge
 * purge that makes a media takedown a takedown.
 *
 * WHY THIS EXISTS AT ALL. `GET /media/:hash` answers `Cache-Control: public,
 * max-age=31536000, immutable`, on purpose: Cloudflare's CSAM Scanning Tool
 * scans at CACHE-FILL, so a private or uncacheable response would silently
 * exclude every stored object from scanning (docs/CSAM-SCANNING.md §2). The
 * price of that decision is that deleting the R2 object removes nothing anyone
 * is actually being served: every PoP that already filled keeps the bytes for a
 * YEAR, and `immutable` tells every browser that has them never to revalidate.
 * `deleteMediaAsset` therefore purges the edge, and this is the config it needs.
 *
 * PURGE BY URL, NOT BY TAG. `POST /zones/:id/purge_cache` with `{"files":[…]}`
 * works on EVERY Cloudflare plan. Purge-by-tag (`{"tags":[…]}`) is
 * ENTERPRISE-ONLY — the media response emits a `Cache-Tag` so a bulk purge is
 * available to a deployment that has it, but nothing depends on it.
 *
 * THE ORIGIN IS `PUBLIC_BASE_URL`, NOT A NEW VAR AND NEVER THE HOST HEADER.
 * `/media/:hash` is served by THIS worker, on the same proxied custom domain as
 * the SSR public pages — `PUBLIC_BASE_URL` is definitionally the origin the
 * cached copy lives under. A Host-derived origin is client-controlled, and a
 * purge aimed at the wrong origin is a 200 from Cloudflare that purges nothing:
 * a success report over material still being served, which is the exact defect
 * this whole mechanism exists to close.
 *
 * ALL THREE OR NOTHING. A half-configured purger cannot purge, and the fail-
 * closed rule (invariant 6) says the FEATURE turns off rather than degrading:
 * null here makes `deleteMediaAsset` refuse the takedown with a 503 instead of
 * performing a removal it cannot finish.
 *
 * WHICH IS WHY `CF_PURGE_ZONE_ID` IS A `launch_var`, NOT AN `optional_var`.
 * The two classes differ in what a deployment is OBLIGED to do about the key:
 * an `optional_var` gates a feature that is off at launch and need only be
 * findable in the runbook, while a `launch_var` must be declared in every
 * `vars` block of `wrangler.jsonc` AND named in `docs/GO-LIVE-CHECKLIST.md`
 * (`tests/invariants/env-inventory.test.ts` enforces both). Returning null here
 * does not turn off a nice-to-have: it turns off `DELETE /ops/media/:hash` —
 * the DMCA/NCII/CSAM takedown lever — and the creator's own
 * `DELETE /account/media/:hash`, which then answer 503 and remove NOTHING.
 * Refusing is the correct behaviour and the refusal happens before any
 * destructive write; the danger is that fail-closed is also SILENT, so a
 * deployment with no working takedown looks exactly like one that has it until
 * a statutory notice arrives. The classification is what makes the omission
 * impossible to reach production without seeing. Do not demote it back to
 * `optional_var` to avoid a checklist edit — the checklist edit is the point.
 *
 * `CF_PURGE_API_TOKEN` stays classified `secret`. The inventory has no
 * launch/optional split for secrets — `SESSION_SECRET` and
 * `REGISTRY_DECOY_SEED` are launch-blocking and are plain `secret` too — and
 * inventing a class to express urgency would be worse than writing the urgency
 * where an operator reads it, which is the checklist.
 */
export function edgePurgeConfigFromEnv(env: Record<string, unknown>): EdgePurgeConfig | null {
  const zoneId = typeof env.CF_PURGE_ZONE_ID === "string" ? env.CF_PURGE_ZONE_ID.trim() : "";
  const apiToken = typeof env.CF_PURGE_API_TOKEN === "string" ? env.CF_PURGE_API_TOKEN.trim() : "";
  const mediaOrigin = publicBaseUrlFromEnv(env);
  if (!CF_ZONE_ID_RE.test(zoneId)) return null;
  if (!CF_API_TOKEN_RE.test(apiToken) || /TODO/i.test(apiToken)) return null;
  if (mediaOrigin === null) return null;
  return { zoneId, apiToken, mediaOrigin };
}

/**
 * QUEUE #355 / D-0787 — THE SUPPORT AND APPEALS INBOX, operator-set.
 *
 * One address serves appeals and support (D-0434 item 5). Until the operator
 * sets `SUPPORT_EMAIL` to a monitored inbox, every surface that would name it
 * names `SUPPORT_ADDRESS_PLACEHOLDER` instead. The placeholder has no `@`, so
 * no client renders it as a mailto (apps/web/src/views.ts `accountClosedView`),
 * and `supportEmailFromEnv` refuses it — and anything TODO-shaped or on a
 * reserved example domain — as a real address. docs/GO-LIVE-CHECKLIST.md
 * blocks launch on this key.
 */
export const SUPPORT_EMAIL_VAR = "SUPPORT_EMAIL";
export const SUPPORT_ADDRESS_PLACEHOLDER = "[support inbox not yet configured]";
const SUPPORT_EMAIL_RE = /^[^\s@]+@([^\s@]+\.[^\s@]+)$/;
const RESERVED_MAIL_DOMAIN_RE = /(^|\.)(example\.(com|net|org)|example|invalid|test|localhost)$/i;

export function supportEmailFromEnv(env: Record<string, unknown> | null | undefined): string | null {
  const v = env?.SUPPORT_EMAIL;
  if (typeof v !== "string") return null;

  const m = SUPPORT_EMAIL_RE.exec(s);
  if (!m) return null;
  if (/TODO|FIXME|PLACEHOLDER|XXX|CHANGEME/i.test(s)) return null;
  if (RESERVED_MAIL_DOMAIN_RE.test(m[1] ?? "")) return null;
  return s;
}

/** The address a user-facing surface names: the configured inbox, else the placeholder. */
export function supportAddress(env?: Record<string, unknown> | null): string {
  return supportEmailFromEnv(env) ?? SUPPORT_ADDRESS_PLACEHOLDER;
}

/**
 * I3-0726 (I21-Q156 (a), D-1078(30)) — GOOGLE PLAY NOTIFICATIONS OVER AUTHENTICATED PUB/SUB PUSH, operator-set.
 *
 * `GOOGLE_RTDN_PUSH_AUDIENCE` is the audience the push subscription stamps into its Google-signed token, and
 * `GOOGLE_RTDN_PUSH_SERVICE_ACCOUNT` the service-account email it signs as (DEPLOY.md, Google Play). With neither set
 * the RTDN door keeps the query-string shared secret (`off`); with both set the signed token is the door's only
 * credential (`on`); with one, a blank-after-trim value, a TODO, or an account that is not an email, the door answers
 * 503 (`misconfigured`) rather than fall back to the secret.
 */
export interface GoogleRtdnPushAuth {
  audience: string;
  serviceAccount: string;
}
export type GoogleRtdnPushSetting = { kind: "off" } | { kind: "on"; auth: GoogleRtdnPushAuth } | { kind: "misconfigured" };
export function googleRtdnPushAuthFromEnv(env: Record<string, unknown>): GoogleRtdnPushSetting {
  const set = (v: unknown): string | null => (typeof v === "string" && v.trim() !== "" ? v.trim() : null);
  const audience = set(env.GOOGLE_RTDN_PUSH_AUDIENCE);
  const serviceAccount = set(env.GOOGLE_RTDN_PUSH_SERVICE_ACCOUNT);
  if (audience === null && serviceAccount === null) return { kind: "off" };
  if (audience === null || serviceAccount === null || /TODO/i.test(audience) || /TODO/i.test(serviceAccount)) return { kind: "misconfigured" };
  if (!/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(serviceAccount)) return { kind: "misconfigured" };

}

export type Env = {

/**
 * The trust-event ledger. Everything consequential in Dhwaja is an event in
 * trust_events; passports/passport_accounts are a fold over that log.
 *
 * Integrity model (D1 has no roles, so the DB cannot refuse writes itself):
 *  - Hash chain: every event's hash covers a pinned preimage including the
 *    previous event's hash. prev_hash is UNIQUE, so the chain cannot fork:
 *    two concurrent appends racing on the same head → one batch fails on the
 *    constraint and RETRIES on the new head. That retry is `appendEvents`'
 *    own job, not the caller's (it never was: five callers translated the
 *    loss into a domain verdict instead — see LedgerContentionError).
 *  - Single genesis: the sentinel GENESIS_HASH is UNIQUE like any other
 *    prev_hash value, so a second "first event" is rejected by the schema.
 *  - verifyChain() recomputes every hash and linkage; replay determinism is
 *    tested against an independent pure reducer (fold.ts is deliberately a
 *    separate code path from the SQL effects below).
 *
 * PREIMAGE FORMAT IS PINNED FOREVER. Changing serialization is a chain
 * format change: bump the version string and re-anchor — never edit v1.
 * Preimage injectivity: every field is newline-free by validation (uuid,
 * vocab enums, SUBJECT_RE/ACTOR_ID_RE, RFC3339, stable-JSON with escaped
 * control chars, hex prev_hash) — so the newline join is unambiguous.
 *
 * AUTHORIZATION IS THE CALLER'S JOB. appendEvents validates shape, policy
 * (ACTOR_POLICY), and state-machine order — it cannot know whether the
 * caller truly is the operator or the passkey holder it claims to be.
 * Until S5 wires authentication (console behind Access, passkey ceremony),
 * NOTHING may expose this module to unauthenticated input; the trust-core
 * route allowlist invariant enforces exactly that.
 *
 * TAMPER-EVIDENCE SCOPE (S2): verifyChain detects any edit that does not
 * recompute the full downstream chain. A writer who rewrites the WHOLE
 * chain from genesis defeats it — that is what the external checkpoint
 * anchor (head hash exported to R2, S4) exists for. Until S4, chain
 * verification is necessary but not sufficient; named as a known gap.
 */

export const PREIMAGE_VERSION = "dhwaja-event-v1";
export const GENESIS_HASH = "0".repeat(64);

export function eventPreimage(
  e: Omit<TrustEventEnvelope, "prev_hash"> & { prev_hash: string },
): string {
  // I3-1616: the join below is injective only if no RAW field carries the
  // separator. `payload`/`supersedes` are JSON (a newline is escaped); the other
  // seven are checked HERE, in the primitive, so a row written outside
  // `validateEnvelope` (every caller but appendEventsOnce) cannot be hashed.
  for (const f of [e.event_id, e.event_type, e.subject, e.actor_kind, e.actor_id, e.created_at, e.prev_hash]) {
    if (typeof f !== "string" || f.includes("\n")) throw new LedgerError("preimage field must be a newline-free string");
  }
  return [
    PREIMAGE_VERSION,
    e.event_id,
    e.event_type,
    e.subject,
    e.actor_kind,
    e.actor_id,
    e.created_at,
    stableStringify(e.payload),
    stableStringify(e.supersedes),
    e.prev_hash,
  ].join("\n");
}

/**
 * `passport_created` / `passport_renamed` commit {name_ref, name_hash}; the name
 * STRING lives in the mutable `passport_names` table (`0001_ledger.sql`). Same
 * shape `warning_published` already uses for its public sentence (a `wording_id` with
 * the text in `wordings`) — the ledger proves the event, the mutable table holds
 * the words.
 *
 * WHY. The name used to BE the payload, and the fold only ever INSERTed it, so
 * there was no rename event at all: a typo, a rebrand, a legal name change or a
 * deadname was permanent, and `GET /proof/:event_id` served the BIRTH name
 * forever. Suppressing a listing is not erasure.
 *
 * THE COMMITMENT IS SALTED:
 *     name_hash = sha256Hex("dhwaja-name-v1\n" + name_salt + "\n" + display_name)
 * with `name_salt` (32 hex, CSPRNG) stored ONLY in the passport_names row. A bare
 * sha256 of a display name is not a privacy-preserving commitment — names are
 * low-entropy, so anyone holding the ledger could confirm a guessed name by
 * hashing it, and "drop a deadname" would be a lie: the deadname would stay
 * permanently CONFIRMABLE from an append-only row. Because the salt dies with the
 * string, erasing the row leaves the ledger with a commitment that cannot be
 * checked against any guess. While the row lives, the commitment still does its
 * job: an operator cannot swap a name in place without appending an event.
 *
 * This module is the ONLY writer of `passport_names`, and `mintName` is the only
 * place a display-name string is validated — see `assertDisplayName`.
 */
export const NAME_HASH_VERSION = "dhwaja-name-v1";

/** The name-string rules, unchanged from the old `passport_created` payload
 * validator: <=100 chars, no angle brackets, and no character in Unicode
 * categories Cc, Cf, Zl or Zp (the shared `hasUnsafeText` class). They are
 * enforced HERE now because this is the last place that still handles the string
 * — the ledger payload no longer carries it, so validating it there would be
 * theatre. Rejecting at the source keeps the name safe in every downstream sink
 * (passport page, cards, notification summaries, fan apps), not just the ones
 * that escape. */
export function assertDisplayName(value: unknown): string {
  if (typeof value !== "string" || value.length === 0 || !hasVisibleText(value)) {
    throw new LedgerError("display_name must be a non-empty string");
  }
  // D-46 (S-036-batch EVENTS_TRU), then I3-1614, then I3-0787: this check was a
  // hand-listed character class three times over, and each pass closed the
  // characters that pass had noticed. C0-only admitted bidi overrides and
  // zero-width; adding those admitted U+2028/U+2029 and DEL; adding THOSE still
  // admitted U+0085, U+00AD, U+061C and U+2060. The defect was never a missing
  // character, it was the enumeration — so the predicate is no longer written
  // here. `hasUnsafeText` (@dhwaja/schema) derives the class from the Unicode
  // general categories the rule actually means: Cc (control, incl. DEL and C1),
  // Cf (format — soft hyphen, ALM, word joiner, zero-width, bidi overrides,
  // U+FEFF), Zl (U+2028) and Zp (U+2029). It is IMPORTED, not restated, so this
  // door and every other door that stores display text cannot drift apart.
  const bad = value.includes("<") || value.includes(">") || hasUnsafeText(value);
  if (bad || value.length > 100) {
    throw new LedgerError("display_name must be <=100 chars with no angle brackets and no control, format or separator characters");
  }
  if (nameClaimsPlatformAuthority(value)) {
    throw new LedgerError("display_name may not carry a verification mark, the platform's name, or a mixed-script look-alike word");
  }
  return value;
}

/** I3-1716: the creator-name screen. The same glyph set as platform-voice.ts
 * `claimsPlatformVoice`, and the same fold: name-fold.ts drops every
 * Default_Ignorable code point and every \p{M} mark, as claimsPlatformVoice does
 * (QUEUE #789, D-0860(2)).
 * Minus its reserved-word list: a creator's own name legitimately contains "Official",
 * "Security" or "System" (a band, a brand), which a fan's display name does not. */
const NAME_VERIFIED_GLYPH_RE = /[✓✔☑✅]|\u{1F5F8}|\u{1F6E1}|\u{1F535}|\u{1F7E2}/u;
/** The fold is name-fold.ts's (QUEUE #789: one copy, shared with passport-view.ts
 * `namesakeKey`); re-exported so existing importers of events.js keep working. */
export { foldName };
/** Exported (I16 #523d) so brand-enrolment.ts `validDisplayName` asks THIS predicate at its
 * door instead of a hand-copy: one const, every door (FAMILY-004 P1). */
export function nameClaimsPlatformAuthority(name: string): boolean {
  const checkable = foldName(name);
  if (NAME_VERIFIED_GLYPH_RE.test(checkable)) return true;
  // #794b (D-0954(4)): the platform's name is matched on the shared confusable skeleton, so a
  // name wholly of look-alike letters (Cherokee/Cyrillic "Dhwaja", no Latin letter) is caught.
  if (confusableSkeleton(name).includes("dhwaja")) return true;
  return checkable.split(/\s+/).some((w) => /\p{Script=Latin}/u.test(w) && /[\p{Script=Cyrillic}\p{Script=Greek}]/u.test(w));
}

// The three `account_added:refused` audit rows (duplicate, delisted/warned, in-batch warned) bind

function randomHex(bytes: number): string {
  const b = crypto.getRandomValues(new Uint8Array(bytes));
  return [...b].map((x) => x.toString(16).padStart(2, "0")).join("");
}

/** The salted commitment that goes in the ledger. Hashing is THE hasher
 * (`sha256Hex` from @dhwaja/canonical) — this module is the single-hasher
 * invariant's named exemption for event-chain hashing, and this is chain-adjacent
 * commitment material, never a subject hash. */
export async function nameCommitment(salt: string, displayName: string): Promise<string> {
  return sha256Hex(`${NAME_HASH_VERSION}\n${salt}\n${displayName}`);
}

 * THE PUBLIC ACCUSATORY SENTENCE, committed.
 *
 * `warning_published` names a `wording_id`; the sentence a fan actually reads
 * lives in the mutable `wordings` table. Nothing tied the two together, so one
 * UPDATE turned "This account is impersonating a Dhwaja creator" into "This
 * creator is a convicted fraudster" against an already-published warning — no
 * event, no hash change, `verifyChain` still green. That is the single most
 * consequential string this product emits about a human being, and it was the
 * one with no commitment, while a display name — strictly less consequential —
 * had a salted one.
 *
 * NOT SALTED, and that is the point of difference. A salt exists so an ERASED
 * low-entropy personal string cannot be confirmed by guessing (see
 * nameCommitment). Wording copy is the opposite kind of string: fixed,
 * class-scoped, counsel-approved library text that is PUBLISHED to every fan who
 * checks the account. There is nothing to hide and everything to verify, so an
 * unsalted commitment is strictly better here — anyone who can read the sentence
 * can recompute it and check the ledger themselves, with no secret from us.
 *
 * `wording_id` is inside the preimage so a commitment made for one library entry
 * cannot be relabelled onto another.
 */
export const WORDING_HASH_VERSION = "dhwaja-wording-v1";

export async function wordingCommitment(wordingId: string, publicCopy: string): Promise<string> {
  return sha256Hex(`${WORDING_HASH_VERSION}\n${wordingId}\n${publicCopy}`);
}

export interface MintedName {
  name_ref: string;
  name_hash: string;
  display_name: string;
  /** The INSERT for `passport_names`. Ride it in the SAME atomic batch as the
   * event that references it, so a name row and its ledger row can never
   * half-land (the same rule the recognition/release projections follow). */
  statement: DbStatement;
}

/**
 * Mint a name row for a passport: validate the string, draw a fresh ref + salt,
 * compute the commitment, and return both the ledger payload halves and the
 * INSERT. Never executes anything itself.
 */
export async function mintName(
  db: DbLike,
  passportId: string,
  displayName: unknown,
  now: () => string = () => new Date().toISOString(),
): Promise<MintedName> {
  const name = assertDisplayName(displayName);
  const nameRef = `nam_${randomHex(8)}`;
  const salt = randomHex(16);
  const nameHash = await nameCommitment(salt, name);
  return {
    name_ref: nameRef,
    name_hash: nameHash,
    display_name: name,
    statement: db
      .prepare(
        "INSERT INTO passport_names (name_ref, passport_id, display_name, name_salt, name_hash, created_at) VALUES (?, ?, ?, ?, ?, ?)",
      )
      .bind(nameRef, passportId, name, salt, nameHash, now()),
  };
}

/**
 * THE fold rule for a name reference, shared by nothing — fold.ts reimplements
 * it deliberately (see that file). It must stay exactly:
 *
 *     display_name = passport_names[name_ref].display_name ?? ""
 *
 * An ERASED name folds to the empty string, in both reducers, so erasure can
 * never make them disagree (Invariant 2). `batchNames` covers refs minted earlier
 * in THIS batch, whose INSERT has not committed yet and therefore cannot be read
 * back with a SELECT.
 */
async function resolveName(db: DbLike, nameRef: string, batchNames: Map<string, string>): Promise<string> {
  const inBatch = batchNames.get(nameRef);
  if (inBatch !== undefined) return inBatch;
  const row = await db
    .prepare("SELECT display_name FROM passport_names WHERE name_ref = ?")
    .bind(nameRef)
    .first<{ display_name: string }>();
  // FAIL CLOSED AT APPEND TIME. Unresolvable here means the event is being
  // appended NOW against a name row that does not exist — a caller bug (a
  // hand-built {name_ref, name_hash} payload pointing at nothing), which would
  // otherwise mint a silently nameless passport. That is a different situation
  // from the replay, where an unresolvable ref is the legitimate ERASURE case;
  // fold.ts folds it to "" precisely because by then the name was deliberately
  // deleted. Only the live writer is strict, so a rebuild after an erasure still
  // succeeds.
  if (!row) throw new LedgerError(`name_ref does not resolve to a stored name: ${nameRef}`);
  return row.display_name;
}

export class LedgerError extends Error {
  override name = "LedgerError";
  constructor(
    message: string,
    public readonly details: string[] = [],
  ) {
    super(message);
  }
}

/**
 * CONTENTION, NOT A DOMAIN ANSWER. Every append reads the chain head and chains
 * onto it, and `prev_hash` is UNIQUE, so ANY two appends anywhere in the product
 * that read the same head collide — a bio-code verify racing an unrelated
 * `passport_created` is as much of a race as two verifies of the same claim.
 * `appendEvents` retries a lost head race internally; this is thrown only when
 * the bounded retry is exhausted, and it means "we could not get a turn", never
 * "your proof failed".
 *
 * IT IS A SEPARATE CLASS, AND ITS MESSAGE CONTAINS NEITHER "UNIQUE" NOR
 * "constraint" — deliberately. The bug this type exists to make unrepresentable
 * is five callers that wrote `catch (e) { if (/UNIQUE|constraint/i.test(e.message))
 * return <a domain outcome> }` around an append: a creator whose code WAS in
 * their bio was told `code_absent`, and charged a verify attempt for it. Such a
 * regex cannot catch this class, and a caller using `tryAppendEvents` cannot see
 * it at all — contention is not a member of `LedgerRefusal`.
 */
export class LedgerContentionError extends Error {
  override name = "LedgerContentionError";
  constructor(public readonly attempts: number) {
    super(`ledger head contention: lost the head race ${attempts} times in a row`);
  }
}

/**
 * Why the ledger REFUSED a batch — the two DOMAIN answers, and the complete set.
 * Contention is deliberately NOT a member: `tryAppendEvents` never returns one,
 * so no exhaustive translation of this type can turn a lost head race into
 * something a creator reads as "your proof failed".
 *
 *  - `state`    — the state machine or a validator refused (a `LedgerError`):
 *                 "claim_proven requires unclaimed", an unknown field, a bad
 *                 payload. Deterministic; retrying changes nothing.
 *  - `conflict` — a uniqueness lock the CALLER rode into the batch via
 *                 `extraStatements` was lost (`claim_consumed`'s PK, the
 *                 `warning_publications` PK). Someone else did this action.
 */
export type LedgerRefusal =
  | { kind: "state"; message: string; details: string[]; error: LedgerError }
  | { kind: "conflict"; message: string; error: Error };

export type AppendResult =
  | { ok: true; events: TrustEventEnvelope[] }
  | { ok: false; refusal: LedgerRefusal };

/** How many times one `appendEvents` call re-reads the head and rebuilds before
 * giving up. Five is well past the point where losing every time means sustained
 * overload rather than an ordinary two-writer collision. */
export const APPEND_MAX_ATTEMPTS = 5;

/** Internal marker: this batch failed because another append took the head we
 * chained onto. Carries the driver error so an exhausted retry is still
 * diagnosable from the logs. */
class HeadRace extends Error {
  constructor(readonly driverError: unknown) {
    super("head race");
  }
}

/**
 * Was this batch failure a lost head race? Two signals, either sufficient,
 * because neither alone is portable:
 *  - the driver named the column (better-sqlite3 and D1 both say
 *    "UNIQUE constraint failed: trust_events.prev_hash"); and
 *  - the head we chained onto is no longer the head — message-independent, and
 *    true exactly when somebody else appended underneath us.
 * A constraint failure that is NEITHER is a caller's own uniqueness lock and
 * stays a domain refusal: this never widens a domain conflict into "try again".
 *
 * NARROWED (audit items #28/#29). The second signal is a HEURISTIC — "somebody
 * appended, so it was probably them" — and it was reached by any error whose
 * text matched `/UNIQUE|constraint/i`. That is too wide by construction: a lost
 * head race can ONLY ever surface as the `prev_hash` UNIQUE violation, never as
 * a NOT NULL, a CHECK or a FOREIGN KEY failure, and those three now carry real
 * decisions in an extra statement — `write_admissions.subject` is the standing
 * and quota gate (standing.ts), and it fires precisely when a CONCURRENT write
 * has just moved the head, which is exactly when the heuristic misfires. The
 * cost of misfiring is not a wrong write (the batch rolled back either way) but
 * a wrong ANSWER: up to five retries and then a 503 "try again" where the honest
 * reply is a 403 or a 429 that no retry will change. So the three
 * non-uniqueness constraint classes are excluded BEFORE the heuristic runs.
 */
// `PRIMARY KEY must be unique` is SQLite's wording when a rowid-table PK
// collides rather than a named UNIQUE index, and it belongs here for the same
// reason the other three do: it is a batch WE built colliding with itself, never
// a lost race for the chain head. It became reachable when the admission gates
// dropped their `ON CONFLICT` clauses — without it a duplicate admission would
// fall through to the head-race heuristic and, if the head happened to have
// moved, be retried and answered 503 instead of refused.
const NOT_A_HEAD_RACE_RE = /NOT NULL constraint failed|CHECK constraint failed|FOREIGN KEY constraint failed|PRIMARY KEY must be unique/i;

async function isHeadRace(db: DbLike, e: unknown, chainedOnto: string): Promise<boolean> {
  if (!(e instanceof Error) || !/UNIQUE|constraint/i.test(e.message)) return false;
  if (/trust_events\.(prev_hash|seq)\b/i.test(e.message)) return true;
  if (NOT_A_HEAD_RACE_RE.test(e.message)) return false;
  // I3-2006: a UNIQUE failure that NAMES its column, and the column is not the
  // chain's, is a caller's own lock (write_admissions, an extraStatements
  // uniqueness row) — never a race, however the head has moved since. Only a
  // driver that names no column falls through to the head re-read below.
  if (/UNIQUE constraint failed: \w+\.\w+/i.test(e.message)) return false;
  const head = await chainHead(db);
  return (head?.event_hash ?? GENESIS_HASH) !== chainedOnto;
}

/**
 * ONLY a lost uniqueness constraint is a domain conflict. The blanket
 * `/UNIQUE|constraint/i` the callers used also matched CHECK, NOT NULL and
 * FOREIGN KEY failures — every one of them a bug in the batch we just built,
 * reported to a creator as "somebody already claimed this". Those must surface
 * as errors. (Same reasoning, and same shape, as `DUPLICATE_EVENT_RE` in
 * entitlements-store.ts.)
 */
const UNIQUENESS_LOST_RE = /UNIQUE constraint failed|PRIMARY KEY must be unique/i;

/** Backoff between head-race retries: full jitter over an exponential window, so
 * racers that collided once do not re-collide in lockstep. */
async function backoff(attempt: number): Promise<void> {
  const ms = Math.floor(Math.random() * 4 * 2 ** attempt);
  if (ms > 0) await new Promise<void>((r) => setTimeout(r, ms));
}

/**
 * Compute the fold-effect statements for one event, or throw LedgerError if
 * the event is invalid against current state. Reducing-only guarantee:
 * corrections can only move state toward less-public/less-trusted.
 */
async function foldEffects(
  db: DbLike,
  e: TrustEventEnvelope,
  seq: number,
  batchNames: Map<string, string> = new Map(),
): Promise<DbStatement[]> {
  const p = e.payload;
  switch (e.event_type) {
    case "passport_created": {
      // Exactly the mintPassportId shape (psp_ + 16 hex): registry decoy
      // refs replicate this shape, so a looser real-id shape would make
      // real entries distinguishable from padding.
      if (!/^psp_[0-9a-f]{16}$/.test(e.subject)) {
        throw new LedgerError("passport_created subject must be psp_ + 16 hex (mintPassportId shape)");
      }
      const existing = await getPassport(db, e.subject);
      if (existing) throw new LedgerError(`passport already exists: ${e.subject}`);
      // The fold column still holds the STRING (every reader — /p/:id, cards,
      // rankings, the registry's find-the-real-one pointer — reads it from here,
      // so none of them changed). It is now RESOLVED through the name reference
      // rather than copied out of the payload.
      // I20-W1D (#946 (a)): the stored skeleton rides the same write (social.ts creatorNameCollision reads it).
      const createdName = await resolveName(db, String(p.name_ref), batchNames);
      return [
        db
          .prepare(
            "INSERT INTO passports (passport_id, display_name, display_name_skeleton, claim_state, retracted, created_seq, updated_seq) VALUES (?, ?, ?, 'unclaimed', 0, ?, ?)",
          )
          .bind(e.subject, createdName, displayNameSkeleton(createdName), seq, seq),
      ];
    }
    case "passport_renamed": {
      // Window item 4. Point a LIVE passport at a different name row. The fold
      // updates only display_name + updated_seq: a rename is not a claim-state
      // change and must never launder one. The passport must exist and not be
      // retracted, like every other lifecycle event.
      const pass = await requireLivePassport(db, e.subject);
      const renamed = await resolveName(db, String(p.name_ref), batchNames);
      return [
        db
          .prepare("UPDATE passports SET display_name = ?, display_name_skeleton = ?, updated_seq = ? WHERE passport_id = ?")
          .bind(renamed, displayNameSkeleton(renamed), seq, pass.passport_id),
      ];
    }
    case "subject_kind_declared": {
      // A passport is declared a CHARACTER, and bound to whoever answers for it.
      //
      // Both facts in one write because for a character they are one decision.
      // Re-declaring rebinds the operator — an agency loses a contract, a brand
      // moves shop — and the fold takes the latest, which is why this is an
      // UPDATE rather than an insert-once. Every such re-declaration is an
      // operator act on the append-only ledger, so the history of who answered
      // for a character is recoverable even though the column holds only the
      // current answer.
      //
      // `claim_state` is deliberately untouched: a kind is not a claim, and an
      // event that could move both would be a door to laundering one into the
      // other. A declared character stays exactly as claimed as it already was.
      const pass = await requireLivePassport(db, e.subject);
      // A PERSON WHO HAS PROVED THEMSELVES CANNOT BE RELABELLED A CHARACTER.
      //
      // Nothing undoes a declaration — `DECLARABLE_KINDS` omits `person` on
      // purpose — so without this, one operator mistake or one compromised
      // operator credential permanently labels a real, verified creator an AI
      // persona on the fold and on every public surface, and the only remedy is
      // retracting their passport. A character is declared BEFORE anyone claims
      // it, which is the only order that is true anyway: an unclaimed listing is
      // Dhwaja's own guess, and a claim is somebody asserting what they are.
      if (pass.claim_state !== "unclaimed") {
        throw new LedgerError(`subject_kind_declared requires unclaimed, was ${pass.claim_state}`);
      }
      return [
        db
          .prepare("UPDATE passports SET subject_kind = ?, operated_by = ?, updated_seq = ? WHERE passport_id = ?")
          .bind(String(p.subject_kind), String(p.operated_by), seq, pass.passport_id),
      ];
    }
    case "subject_kind_corrected": {
      // #132W-08 (I16, D-0920): CORRECT a wrong kind on an UNCLAIMED listing.
      // The declaration above can never set `person` and nothing undoes it, so a
      // real person's unclaimed listing mislabelled a character could only be
      // retracted — destroying the listing to remove our own label. This is the
      // narrow door back: operator-only (ACTOR_POLICY), dual control at the
      // route, and UNCLAIMED only, for the declaration's own reason — once
      // somebody has claimed the passport they have asserted what they are, and
      // an operator relabelling them is the move this ledger refuses.
      // `claim_state` is untouched, as in the declaration. A person carries no
      // operator, so `operated_by` is cleared with the kind.
      const pass = await requireLivePassport(db, e.subject);
      if (pass.claim_state !== "unclaimed") {
        throw new LedgerError(`subject_kind_corrected requires unclaimed, was ${pass.claim_state}`);
      }
      const kind = String(p.subject_kind);
      return [
        db
          .prepare("UPDATE passports SET subject_kind = ?, operated_by = ?, updated_seq = ? WHERE passport_id = ?")
          .bind(kind, kind === "person" ? null : String(p.operated_by), seq, pass.passport_id),
      ];
    }
    case "character_operator_rebound": {
      // WHO ANSWERS FOR THIS CHARACTER, CHANGED — the door slice 16 recorded as
      // missing and slice 18 deferred for a product decision.
      //
      // `subject_kind_declared` is the only other writer of `operated_by` and it
      // requires `unclaimed`, for a reason that is correct and stays: nothing
      // undoes a declaration, so without it one operator mistake permanently
      // labels a real, proven person an AI persona. The cost was that once a
      // character is CLAIMED its operator is frozen — an agency that shuts down
      // or a person-operator who retires leaves a character that
      // `character-lapse-sweep.ts` will de-badge, and the only lever was a
      // `downgrade_claim` correction, which lapses every control proof so the
      // character must reprove account control from scratch purely to change who
      // stands behind it.
      //
      // THIS EVENT CANNOT TOUCH `subject_kind`. Not "does not" — the column is
      // not in the statement, so the back door into character-hood does not exist
      // to be closed. That is the whole reason it is a separate type rather than
      // a relaxed guard on the existing one: a conditional guard is a rule
      // somebody later reads the wrong way round, and this repo has paid for that
      // three times.
      const pass = await requireLivePassport(db, e.subject);
      // A PERSON HAS NO OPERATOR. Refused here rather than silently no-oped so an
      // operator who picked the wrong passport is told, and so this event can
      // never become the second way to make something a character.
      if (pass.subject_kind === "person") {
        throw new LedgerError("character_operator_rebound requires a character subject");
      }
      return [
        db
          .prepare(
            // THE CANDIDATE IS CHECKED, NOT THE INCUMBENT. During an UPDATE
            // `passports.operated_by` still holds the OLD value, so
            // ACCOUNTABLE_SUBJECT_SQL would answer about the operator being
            // replaced — which is exactly backwards, and would let a rebind onto
            // a dead agency through while refusing a rescue from one.
            // `accountableOperatorSql("?")` is the SAME rule over the incoming
            // id: one definition, evaluated about the right thing.
            //
            // THE CEILING RIDES `updated_seq`, NOT `operated_by`, AND IT HAS TO.
            //
            // The obvious form — `operated_by = CASE WHEN <guard> THEN ? END` —
            // is the idiom the other four gates in this repo use, and here it is
            // a catastrophe. `passports.operated_by` is NULLABLE (migration 0047
            // adds it as bare `TEXT`), so a refused CASE writes NULL rather than
            // throwing: the statement SUCCEEDS and strips the character of its
            // operator entirely, which is strictly worse than the frozen binding
            // this event exists to fix. Measured against better-sqlite3, not
            // reasoned about — the refusal left `operated_by = NULL` and no
            // error.
            //
            // `updated_seq` is `INTEGER NOT NULL` (migration 0001), and a NOT
            // NULL violation in ANY SET expression fails the WHOLE statement, so
            // `operated_by` is not written either. Verified the same way: after a
            // refusal the row still held its previous operator.
            //
            // FRAGILITY, STATED: `updated_seq` is written by nearly every arm in
            // this file, so the refusal matcher below is only unambiguous while
            // this is the ONLY arm using it as a ceiling. A second one would need
            // its own discriminator, and `slice20-rebind.test.ts` pins that this
            // arm is the only one, so a future author is told rather than
            // discovering it through a mis-worded error.
            // `selfId` stays the DEFAULT `passports.passport_id`, qualified. An
            // unqualified `passport_id` inside the second EXISTS would resolve
            // against the subquery's own `passports o`, making the self-bind
            // guard read `o.passport_id <> o.passport_id` — always false — and
            // silently killing the entire person-operator branch, so no rebind
            // onto a person could ever succeed. Correlation to the outer row is
            // the whole reason that clause names its table.
            `UPDATE passports SET operated_by = ?, updated_seq = CASE WHEN ${accountableOperatorSql("?", "passports.passport_id", Date.parse(e.created_at))} THEN ? END WHERE passport_id = ?`,
          )
          // The value, then the predicate's TWO mentions of the candidate (the
          // agency id and the operator passport id), then the seq and the
          // subject.
          .bind(String(p.operated_by), String(p.operated_by), String(p.operated_by), seq, pass.passport_id),
      ];
    }
    case "account_added": {
      const pass = await requireLivePassport(db, e.subject);
      const subject = subjectForAccount(String(p.platform), String(p.id_kind), String(p.id));
      // A domain (DNS-proven) or a generic LINK (M3) is control-proven
      // separately, never self-asserted: only an operator (OSINT/import) or the
      // system may LIST one; a creator/agency cannot add one. Serving still
      // requires the per-account proof — this gate just keeps unproven junk off
      // passports in the first place. (`link` subjects have idKind "url".)
      if ((subject.idKind === "domain" || subject.idKind === "url") && e.actor_kind !== "operator" && e.actor_kind !== "system") {
        throw new LedgerError("a domain or generic-link account may only be added by an operator; control is proven separately");
      }
      // THREE QUESTIONS, ONE ROUND TRIP: is this key already registered, was it
      // DELISTED AT ITS SUBJECT'S REQUEST, and does it carry an ACTIVE PUBLISHED
      // WARNING?
      //
      // The second one (migration 0028) shuts the door next to the one the
      // transfer guard below shuts. Once the "never re-seeded" promise is
      // recorded against the ACCOUNT KEY, the obvious way to void it is no
      // longer to move the row — it is to make the row again somewhere else.
      // Today the duplicate check happens to cover that (this fold never DELETES
      // a `passport_accounts` row, so a frozen key is still present and reads as
      // a duplicate), which is exactly the kind of accident that stops being
      // true the day an erasure or merge path removes one. So the promise is
      // asserted directly, against the table that holds it.
      //
      // A key added to an ALREADY-suppressed passport is not refused: it was
      // never frozen (it did not exist), the live suppression covers it anyway,
      // and the transfer guard keeps it from being moved off. Listing a NEW
      // account of a delisted subject is the operator's business; RE-LISTING one
      // they asked us to drop is not, and needs the unsuppress ceremony.
      //
      // The THIRD question (INV-025, INV-027) shuts the door the accusation was
      // always assumed to shut. "Removal is always faster than addition" is
      // empty if the ADDITION door never asks about the warning: an impersonator
      // whose account an operator publicly warned about could self-add it to a
      // passport, prove the passport, and be served as a creator AT THE WARNED
      // HASH. ACTIVE means a `warning_published` with no `warning_retracted` at
      // a LATER seq, so a retraction re-opens the door and an operator error is
      // not permanent. No actor_kind is exempt — the impersonator's own
      // self-service path is the one this is about.
      //
      // `sha256Hex(canonical)` IS `subjectHash`'s body (canonical/index.ts:914-916);
      // warnings are published against `sha256:<that>`.
      //
      // All three ride the ONE statement the duplicate check already spends, so
      // `account_added` costs what it always did (this runs once per imported
      // account, in bulk).
      //
      // THE REFUSAL IS ONE SHARED STRING for the delisted and the warned arms,
      // for every actor: a distinct message would make `account_added` an oracle
      // telling an impersonator whether they are warned about or merely
      // delisted, which is the INV-003 no-oracle shape applied to the accusatory
      // side. It names neither the warning, its class, nor the real account.
      const warnSubject = `sha256:${await sha256Hex(subject.canonical)}`;
      // The door's population is the CLOSED, declared constant
      // `BLOCKS_ACCOUNT_ADDED` (warnings.ts) -- never the served-to-fans set,
      // which is a runtime read of the `serve_confirmed_fraud` ops flag
      // (registry-build.ts warningEntries). An authorization door may not
      // swing on that flag: S-019 (a)(ii), FINDING-0014-C-5, W0 round-2.
      const blockingClassPlaceholders = BLOCKS_ACCOUNT_ADDED.map(() => "?").join(", ");
      const known = await db
        .prepare(
          `SELECT EXISTS (SELECT 1 FROM passport_accounts WHERE account_key = ?) AS dup,
                  EXISTS (SELECT 1 FROM passport_accounts ra JOIN passports rp ON rp.passport_id = ra.passport_id
                           WHERE ra.account_key = ? AND rp.retracted = 1) AS retracted_holder,
                  EXISTS (SELECT 1 FROM suppressed_account_keys WHERE account_key = ?) AS frozen,
                  EXISTS (SELECT 1 FROM trust_events w
                           WHERE w.subject = ? AND w.event_type = 'warning_published'
                             AND json_extract(w.payload, '$.warning_class') IN (${blockingClassPlaceholders})
                             AND w.seq > COALESCE((SELECT MAX(r.seq) FROM trust_events r
                                                    WHERE r.subject = ? AND r.event_type = 'warning_retracted'), -1)) AS warned`,
        )
        .bind(subject.canonical, subject.canonical, subject.canonical, warnSubject, ...BLOCKS_ACCOUNT_ADDED, warnSubject)
        .first<{ dup: number; retracted_holder: number; frozen: number; warned: number }>();
      if (known?.dup) {
        // FAMILY-003 P1 (record, then answer): this is the one refusal most
        // often benign (a re-import, a retried bulk row) and it used to leave
        // NOTHING — a legitimate repeat and a probe against a suppressed key
        // looked identical because neither wrote a line. `item_id` is
        // `warnSubject` (`sha256:<sha256Hex(subject.canonical)>`, already
        // computed above to test the `warned` arm) for the same reason the
        // other arm below uses it: see that comment for the INV-003
        // justification, which applies unchanged here.
        await db
          .prepare(
            "INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) VALUES (?, 'account_added:refused', ?, ?, ?, ?)",
          )
          .bind(e.actor_id, `${subject.canonical} refused (${retractedEarlier(known, e.actor_kind) ? "retracted earlier" : "duplicate"}) — already registered`, e.created_at, warnSubject, touchedByFromActor(e.actor_kind))
          .run();
        throw new LedgerError(duplicateAccountRefusal(subject.canonical, known, e.actor_kind));
      }
      if (known?.frozen || known?.warned) {
        // FINDING-0014-C-6. THE RESPONSE STRING IS UNCHANGED AND STAYS SHARED —
        // see the INV-003 paragraph above: a message that named the warning
        // would turn `account_added` into an oracle telling an impersonator
        // whether they are warned about or merely delisted. So the string keeps
        // naming the unsuppress ceremony for both arms.
        //
        // What was wrong is that "lift the removal (unsuppress) first" is the
        // ONLY remedy named anywhere, and it cannot clear the arm that actually
        // fired here: a warned key is not in `suppressed_account_keys`, and no
        // unsuppress retracts a warning. The one thing that re-opens this door
        // is a `warning_retracted` at a later seq — the mechanism the paragraph
        // above states in terms and the refusal never mentioned.
        //
        // The applicable remedy therefore goes on the AUDIT line, which only an
        // operator reads, and it names the arm that fired. An operator reading
        // back a refusal now learns which ceremony to run; the actor at the door
        // learns exactly what they learned before.
        // `item_id`: was NULL — every refused attempt landed as an unbounded,
        // unattributable stream an operator could not pivot on (they had to
        // grep `detail` text). It now carries `warnSubject`
        // (`sha256:${sha256Hex(subject.canonical)}`, the SAME hash already
        // computed above to test the `warned` arm, and the same shape
        // `trust_events.subject` and `caution_suppressions.subject_hash` use
        // for a subject identity elsewhere in this ledger). Justification
        // against INV-003's no-oracle shape (a row that names a subject the
        // caller could not otherwise learn is a leak): (1) `/ops/audit` is not
        // a public route — it is Access-JWT-gated to an authenticated
        // operator seat (console.ts `/ops/*` role guard), so INV-003's public,
        // unauthenticated-caller premise does not apply to who can read this
        // row; (2) the row's `detail` column already names the SAME subject in
        // the clear (`${subject.canonical}`, right below) to that same
        // operator audience, so `item_id` discloses nothing beyond what
        // `detail` already carries — it only makes the existing disclosure
        // findable; (3) to the actor AT THE DOOR — the only party INV-003
        // protects an oracle from — nothing changes: `item_id` is never
        // returned to them, and the thrown `LedgerError` string is unchanged
        // and still shared across the `frozen`/`warned` arms.
        const remedy = known.warned
          ? known.frozen
            ? "warning_retracted (and unsuppress: both arms are live at this key)"
            : "warning_retracted"
          : "unsuppress";
        await db
          .prepare(
            "INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) VALUES (?, 'account_added:refused', ?, ?, ?, ?)",
          )
          .bind(
            e.actor_id,
            `${subject.canonical} refused (${known.warned ? "warned" : "delisted"}) — applicable remedy: ${remedy}`,
            e.created_at,
            warnSubject,
            touchedByFromActor(e.actor_kind),
          )
          .run();
        throw new LedgerError(
          `account_added: ${subject.canonical} cannot be added — it was delisted at the subject's request and is never re-listed — lift the removal (unsuppress) first, so the re-listing is a declared act`,
        );
      }
      return [
        db
          .prepare(
            "INSERT INTO passport_accounts (account_key, passport_id, platform, id_kind, id, url, relationship_basis, status, observed_at, updated_seq, added_by) VALUES (?, ?, ?, ?, ?, ?, ?, 'active', ?, ?, ?)",
          )
          .bind(
            subject.canonical,
            pass.passport_id,
            subject.platform,
            subject.idKind,
            subject.id,
            typeof p.url === "string" ? p.url : null,
            String(p.relationship_basis),
            String(p.observed_at),
            seq,
            touchedByFromActor(e.actor_kind), // provenance: WHO added it (fold column)
          ),
      ];
    }
    case "account_revoked": {
      await requireLivePassport(db, e.subject);
      const key = String(p.account_key);
      await requireOwnedAccount(db, key, e.subject);
      return [
        db
          .prepare(
            "UPDATE passport_accounts SET status = 'revoked', updated_seq = ? WHERE account_key = ?",
          )
          .bind(seq, key),
      ];
    }
    case "account_reactivated": {
      // F5 (2026-09-03, Jason's sign-off): the ONE event that flips a key back
      // to `active`. Preconditions, all fail closed, in the order an operator
      // reads them on the console preview:
      //   1. the subject is live and OWNS the key (a key on another passport is
      //      `account_transferred` first — never both in one event);
      //   2. the key is `revoked` or `historical` — reactivating an `active` key
      //      is refused (an idempotent no-op would hide a double-fire);
      //   3. not delisted: neither the subject (request_suppressions) nor the
      //      key (suppressed_account_keys) — re-listing a delisted person's
      //      account is the one thing this event must never do quietly; the
      //      sentence is the one `account_added` / `account_transferred` use;
      //   3a. no ACTIVE warning of a BLOCKS_ACCOUNT_ADDED class on the key (I62-EVENTS P1, D-1519: the
      //      predicate `account_added` / `account_transferred` use, `activeBlockingWarning`); a later
      //      `warning_retracted` lifts it, and the refusal writes `account_reactivated:refused`;
      //   4. `reverses` names the LATEST revoking event for this key on any
      //      passport (the `account_revoked`, or the `correction{revoke_account}`
      //      — after a transfer it sits in the source's history), and the
      //      basis agrees with its kind: `takedown_reversed` ↔ account_revoked,
      //      `correction_reversed` ↔ correction, `control_reproven` ↔ either;
      //   5. `control_reproven` additionally needs a FRESH control proof for
      //      this key on THIS passport (≤ REACTIVATION_PROOF_WINDOW_MS old).
      // The fold effect is status alone (+ updated_seq, added_by = the actor
      // class, as a transfer does); url / platform / id / basis / observed_at
      // are untouched. fold.ts mirrors this branch byte-for-byte.
      await requireLivePassport(db, e.subject);
      const key = String(p.account_key);
      await requireOwnedAccount(db, key, e.subject);
      const acct = await db.prepare("SELECT status FROM passport_accounts WHERE account_key = ?").bind(key).first<{ status: string }>();
      if (!acct || (acct.status !== "revoked" && acct.status !== "historical")) {
        throw new LedgerError(`account_reactivated: ${key} is ${acct?.status ?? "missing"}, not revoked or historical`);
      }
      const delisted = await db
        .prepare(
          `SELECT EXISTS (SELECT 1 FROM request_suppressions WHERE passport_id = ?) AS subj,
                  EXISTS (SELECT 1 FROM suppressed_account_keys WHERE account_key = ?) AS key_frozen`,
        )
        .bind(e.subject, key)
        .first<{ subj: number; key_frozen: number }>();
      if (delisted?.subj || delisted?.key_frozen) {
        throw new LedgerError(
          `account_reactivated: ${key} was delisted at the subject's request and is never re-listed — lift the removal (unsuppress) first, so the re-listing is a declared act`,
        );
      }
      const warning = await activeBlockingWarning(db, key);
      if (warning.warned) {
        await db
          .prepare(
            "INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) VALUES (?, 'account_reactivated:refused', ?, ?, ?, ?)",
          )
          .bind(e.actor_id, `${key} refused (warned) — applicable remedy: warning_retracted`, e.created_at, warning.subject, touchedByFromActor(e.actor_kind))
          .run();
        throw new LedgerError(`account_reactivated: ${key} carries an active public warning and cannot be reactivated while the warning is active — retract the warning first`);
      }
      const latest = await latestRevokingEvent(db, key);
      if (!latest) throw new LedgerError(`account_reactivated: no revoking event on record for ${key}`);
      if (latest.event_id !== String(p.reverses)) {
        throw new LedgerError(`account_reactivated: reverses ${String(p.reverses)} but the latest revoking event for ${key} is ${latest.event_id}`);
      }
      const basis = String(p.basis);
      if (basis === "takedown_reversed" && latest.event_type !== "account_revoked") throw new LedgerError("account_reactivated: takedown_reversed must reverse an account_revoked event");
      if (basis === "correction_reversed" && latest.event_type !== "correction") throw new LedgerError("account_reactivated: correction_reversed must reverse a revoke_account correction");
      if (basis === "control_reproven") {
        const fresh = await freshControlProof(db, e.subject, key, Date.parse(e.created_at), latest.created_at);
        if (!fresh) throw new LedgerError(`account_reactivated: control_reproven needs a control proof for ${key} on ${e.subject} made after the revocation and within the last ${REACTIVATION_PROOF_WINDOW_MS / 86_400_000} days`);
      }
      return [
        db
          .prepare("UPDATE passport_accounts SET status = 'active', updated_seq = ? WHERE account_key = ?")
          .bind(seq, key),
      ];
    }
    case "account_transferred": {
      // Move an account from `from_passport` to THIS passport (the subject) —
      // operator-adjudicated recovery. Ledger-preserving: the immutable log keeps
      // the original account_added AND this transfer; the fold reflects the
      // current owner. The global account_key PK is freed to the rightful owner.
      const to = await requireLivePassport(db, e.subject);
      const key = String(p.account_key);
      const from = String(p.from_passport);
      if (from === e.subject) throw new LedgerError("account_transferred: from and to are the same passport");
      const row = await db.prepare("SELECT passport_id FROM passport_accounts WHERE account_key = ?").bind(key).first<{ passport_id: string }>();
      if (!row) throw new LedgerError(`account_transferred: no such account: ${key}`);
      if (row.passport_id !== from) throw new LedgerError(`account_transferred: ${key} does not belong to ${from}`);
      // A TRANSFER OUT OF A DELISTED PASSPORT IS A RE-LISTING (audit HIGH,
      // 2026-08-26). REFUSED HERE, FAIL CLOSED.
      //
      // "Never re-seeded" is enforced by `isSuppressedAccountKey` (seeding.ts)
      // and by the delisting shield on `POST /report` and on nominations — and
      // every one of those reads the LIVE join `passport_accounts` →
      // `request_suppressions`. The suppression is keyed on `passport_id`; this
      // event is an `UPDATE passport_accounts SET passport_id = ?`. It MOVES the
      // very row those readers traverse, so a single transfer erased the link
      // and, with it, the promise. Measured: before the transfer `importFile`
      // refuses the key with the "never re-seeded" reason; after one transfer
      // `isSuppressedAccountKey` is false, the next bulk import re-lists the
      // delisted person, and the registry serves their hash to fans as
      // `unclaimed` again. Nothing threw, and the operator saw no signal.
      //
      // The blast radius is wider than the re-listing: the same live join is the
      // shield that makes a delisted subject unreportable and un-nominatable, so
      // one transfer also re-opened the person who asked to leave to being
      // reported. And it bypassed the declared unsuppress ceremony entirely — a
      // returning hash is ALWAYS a re-listing, and re-listing is the one
      // direction this product makes slow and explicit.
      //
      // Only the FROM side is refused. Transferring INTO a suppressed passport
      // ADDS a suppression link, which delists the account — the fast, safe
      // direction that needs no ceremony — so it stays allowed.
      //
      // THIS GUARD IS NOW BELT, NOT BRACES. It is narrow by construction — it
      // protects a mutable pointer by forbidding the one statement that moves it
      // — and on its own it would have to be re-derived by every future author
      // of a row-moving path. The durable fix landed with migration 0028: the
      // account keys a suppression covers are FROZEN in
      // `suppressed_account_keys` at suppression time, and every enforcement
      // point (the importer, the report shield, the nomination shield, the
      // registry builder, `account_added` above) reads the snapshot UNION the
      // live join, so the promise survives the row moving however it moves.
      // This refusal stays anyway, because the ceremony matters as much as the
      // outcome: a transfer out of a delisted passport is a re-listing decision
      // and must be made explicitly, by lifting the removal, rather than
      // absorbed silently by a guard the operator never sees.
      const gone = await db.prepare("SELECT 1 AS x FROM request_suppressions WHERE passport_id = ?").bind(from).first<{ x: number }>();
      if (gone) {
        throw new LedgerError(
          `account_transferred: ${from} has been delisted at the subject's request — lift the removal (unsuppress) before moving ${key} off it, so the re-listing is a declared act`,
        );
      }
      // I3-0302 / I3-0302 b (I22-W1-LEDGER): a key carrying an ACTIVE public warning of a class that blocks
      // `account_added` cannot be MOVED onto a passport either -- the same predicate as the account_added
      // arm above (BLOCKS_ACCOUNT_ADDED published after the subject's latest warning_retracted, keyed
      // sha256:<sha256Hex(key)>), so neither door launders a warned key onto a passport. Both doors that
      // emit this event (the operator transfer route and recovery.ts's adjudicated transfer) reach this arm
      // through appendEvents. The remedy (retract the warning first) is named on the operator-only audit line.
      // I62-EVENTS P1 (W0-9): the predicate is ONE helper, `activeBlockingWarning` (below), shared with the
      // account_reactivated arm and reactivation.ts `reactivationPreview`.
      const { warned, subject: warnSubject } = await activeBlockingWarning(db, key);
      if (warned) {
        await db
          .prepare(
            "INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) VALUES (?, 'account_transferred:refused', ?, ?, ?, ?)",
          )
          .bind(e.actor_id, `${key} refused (warned) — applicable remedy: warning_retracted`, e.created_at, warnSubject, touchedByFromActor(e.actor_kind))
          .run();
        throw new LedgerError(`account_transferred: ${key} carries an active public warning and cannot be moved onto a passport — retract the warning first, as adding it would require`);
      }
      // Reassign ownership ONLY — status is preserved, never forced to 'active'.
      // A transfer must not silently resurrect an account that a takedown
      // (account_revoked) or a correction (revoke_account → 'historical') has
      // suppressed; reactivation is a separate, deliberate operator action.
      return [
        // I3-0913: added_by is PRESERVED — it records who ASSERTED the account
        // (account-standing.ts servingClassFor reads it as "must prove itself"),
        // and moving the row does not vet it. Reactivation preserves it too.
        // fold.ts must apply the same rule (determinism).
        db.prepare("UPDATE passport_accounts SET passport_id = ?, updated_seq = ? WHERE account_key = ?").bind(to.passport_id, seq, key),
      ];
    }
    case "claim_started":
      // Recorded in the log; no fold effect — but the passport must exist
      // and be live (nothing referencing a nonexistent subject enters the chain).
      await requireLivePassport(db, e.subject);
      return [];
    case "claim_proven": {
      const pass = await requireLivePassport(db, e.subject);
      if (pass.claim_state !== "unclaimed") {
        throw new LedgerError(`claim_proven requires unclaimed, was ${pass.claim_state}`);
      }
      // ANALYTICS (ADR-0006 §2). THE ONE unclaimed → claimed transition: the
      // three other statements that write `claim_state = 'claimed'`
      // (`downgrade_verification`, `verification_expired`, the operator
      // downgrade) move a passport BACK from verified, and counting those as
      // claims would make the day counter say a creator arrived every time a
      // badge lapsed. Creator-scoped, so no account takes part; the day is the
      // envelope's own `created_at`, which is the instant this batch commits.
      return [
        accountablePassport(db, e.subject, "claimed", seq, e.created_at),
        ...outboxStatements(db, { kind: "creator_claimed", passportId: e.subject, occurredDay: occurredDay(e.created_at) }),
      ];
    }
    case "verification_granted": {
      const pass = await requireLivePassport(db, e.subject);
      if (pass.claim_state !== "claimed") {
        throw new LedgerError(`verification_granted requires claimed, was ${pass.claim_state}`);
      }
      // THE SAME CONDITION AT THE STATE A FAN ACTUALLY TRUSTS. Gating only the
      // claim let a character be VERIFIED after its operator was suspended —
      // and "verified" is the word the whole product is about. What this does
      // not do is walk a verified character back when its operator lapses
      // LATER; that is continuous enforcement and is written up as a known gap.
      return [accountablePassport(db, e.subject, "verified", seq, e.created_at)];
    }
    case "verification_renewed": {
      // The grant's clock moved forward. You cannot renew what you do not hold,
      // so the passport must already be verified — and the state does NOT move.
      // `updated_seq` does: a renewal is a real change to the listing, and both
      // reducers have to agree on that byte or /ops/fold/rebuild rewrites it.
      // Mirrored in fold.ts.
      const pass = await requireLivePassport(db, e.subject);
      if (pass.claim_state !== "verified") {
        throw new LedgerError(`verification_renewed requires verified, was ${pass.claim_state}`);
      }
      // ...AND SOMEBODY STILL ANSWERS FOR IT, if it is a character.
      //
      // This arm used `statePassport`, which writes the state unconditionally —
      // so a character whose operator had lapsed could have its badge renewed
      // through a door the accountability gate does not cover. That is exactly
      // the failure `accountablePassport`'s docblock names: "a rule that lives at
      // a door is a rule somebody adds a fourth door past", and this was the
      // fourth door.
      //
      // THIS IS A LIVE DOOR, and the first version of this comment said it was
      // not. `verification_renewed` is absent from `APPEND_ALLOWED`, which stops
      // GENERIC callers only — the governed-type mechanism is the door, and
      // `review.ts approveItem` is wired to it: approving a verification item
      // for an already-verified passport appends a renewal, and
      // `test/verification-renewal.test.ts` drives that end to end.
      //
      // So this is a behaviour change rather than free hardening: an operator
      // approving a renewal for a character whose operator has lapsed is now
      // refused. That is the right outcome — the badge rests on somebody being
      // accountable, and renewal is the one remaining way to keep it alive — but
      // it had to be weighed, not assumed away. `approveItem` already translates
      // the refusal into a 409 naming the fix, so the operator sees a domain
      // answer rather than a raw constraint error.
      //
      // A person short-circuits on `subject_kind = 'person'` before either
      // EXISTS runs, so an ordinary creator's renewal is unchanged.
      return [accountablePassport(db, e.subject, "verified", seq, e.created_at)];
    }
    case "operator_lapsed": {
      // NOBODY ANSWERS FOR THIS CHARACTER ANY MORE, so the sentence that says we
      // checked the people who do is withdrawn.
      //
      // WHAT THIS CLOSES. `accountablePassport` is evaluated AT THE WRITE that
      // depends on it — the claim and the verification — which is what makes it
      // atomic and is why it belongs there. The cost is that it is evaluated
      // NOWHERE ELSE: a character verified while its agency was active kept its
      // badge forever afterwards, and slice 14's advertised lever ("suspend the
      // agency and every character it runs stops being claimable") was true only
      // of characters that had not been claimed yet. Written into that slice's
      // gaps and again into slice 15's; `character-lapse-sweep.ts` is what reads
      // the predicate on a clock.
      //
      // VERIFIED → CLAIMED, AND NO FURTHER — the same walk-back
      // `verification_expired` does, for a reason that has to be stated because
      // the claim gate is deliberately stricter. `claimed` says *whoever runs
      // this proved they control the account*: that proof happened, and an agency
      // losing a contract does not make it retroactively false. `verified` says
      // *we checked the people behind it*, which IS the sentence a lapse
      // falsifies. Accountability gates an ACTION (taking possession of a
      // listing), not a state.
      //
      // Un-claiming would also mean `downgrade_claim`, which lapses every control
      // proof on the passport (`lapseControlProofs` below). A scheduled sweep
      // must never do that: verified → claimed is repaired by re-verifying, while
      // unclaimed is a decision somebody has to take.
      const pass = await requireLivePassport(db, e.subject);
      if (pass.claim_state !== "verified") {
        throw new LedgerError(`operator_lapsed requires verified, was ${pass.claim_state}`);
      }
      // A PERSON CAN NEVER LAPSE THIS WAY. The gate short-circuits on
      // `subject_kind = 'person'`, so there is no state in which this event is
      // meaningful for one — and a bug (or a compromised system actor) that
      // aimed it at a human creator would de-badge them with a reason that
      // cannot apply. Refused at the ledger, not at the sweep, because the sweep
      // is not the only thing that could ever append it.
      if (pass.subject_kind === "person") {
        throw new LedgerError("operator_lapsed applies only to a character subject");
      }
      // THE PREDICATE IS RE-EVALUATED HERE, AT THE INSTANT OF THE WRITE — the
      // same mechanism `accountablePassport` uses, inverted.
      //
      // The first version of this arm trusted the caller: the sweep reads
      // accountability in one page `SELECT` and then appends up to a hundred
      // rows afterwards, so an agency reinstated in that gap still lost every
      // badge it ran. Adversarial review made the point sharper than that —
      // appending this event against a character whose agency was never
      // suspended at all de-badged it, because nothing downstream of the sweep
      // checked. That is exactly the asymmetry slice 14 rejected, in reverse:
      // `accountablePassport`'s own docblock argues the gate belongs in the fold
      // "rather than a check the caller performed a moment earlier ... so a
      // suspension that lands mid-flight wins". The write that REMOVES a verdict
      // had the caller-side check.
      //
      // `NOT (...)` in a `CASE` with no `ELSE` against the NOT NULL `claim_state`
      // column: a character whose operator is live yields NULL, the constraint
      // fires, and the whole batch — the event row included — rolls back. The
      // sweep already counts that as `refused`, which is now a meaningful number
      // rather than only a race.
      //
      // WITHDRAWING THE BADGE WITHDRAWS THE DESTINATIONS, in the same batch —
      // identical to `verification_expired` below and for the identical reason:
      // the serve gate already stops payout tags RENDERING for a non-verified
      // passport, and without this the stored rows survive and silently promote
      // again on re-verification. It rides the SAME batch as the guarded UPDATE,
      // so a refusal takes the retirement with it.
      return [
        db
          .prepare(`UPDATE passports SET claim_state = CASE WHEN NOT (${accountableSubjectSqlAt(Date.parse(e.created_at))}) THEN ? END, updated_seq = ? WHERE passport_id = ?`)
          .bind("claimed", seq, e.subject),
        db.prepare("UPDATE payout_tags SET status = 'retired' WHERE passport_id = ? AND status IN ('live','staged')").bind(e.subject),
      ];
    }
    case "verification_expired": {
      // THE ENFORCEMENT OF `expires_at`, and the reason it is HERE rather than
      // at the ~180 places that read `claim_state`. Modelled EXACTLY on
      // correction/`downgrade_verification` below — the existing, correct
      // precedent for "this passport stops being verified": require verified,
      // one `statePassport` back to `claimed`, nothing else touched (`retracted`
      // stays 0 — an expiry is not a retraction, the listing is still real and
      // the creator can be re-verified).
      //
      // Because the fold moves, every standing reader moves with it and at the
      // same instant: the registry verdict, the passport badge and card, payout
      // tag presentation, `requireVerifiedCreator` and `requireVerified`. That
      // simultaneity is the property — a creator publicly badged Verified while
      // silently unable to mint is the asymmetry docs/VERIFIED-MINT.md exists to
      // reject. Mirrored in fold.ts.
      const pass = await requireLivePassport(db, e.subject);
      if (pass.claim_state !== "verified") {
        throw new LedgerError(`verification_expired requires verified, was ${pass.claim_state}`);
      }
      // S-070 BF-15 — THE DEADLINE THIS WORK ENFORCES MUST STILL BE THE CURRENT
      // ONE AT COMMIT.
      //
      // "Verified" was the only precondition, and a RENEWAL leaves a passport
      // verified — that is the whole point of one. `verification-sweep.ts` reads
      // each candidate's latest grant ONCE and submits expiry work carrying the
      // deadline it read; a renewal approved after that read (and its ledger
      // contention retries do not refresh the premise) therefore committed
      // against a deadline that no longer existed. The passport was walked back
      // to `claimed` and, in the same batch below, every `live`/`staged` payout
      // destination was RETIRED — undoing an approval the operator had just
      // granted, blocking verified-only operations, and removing payout
      // destinations that re-verification does not bring back.
      //
      // The check is the event's own evidence against the ledger's: the payload
      // copies `expired_at` verbatim off the grant it enforces, so if the
      // CURRENT latest grant-or-renewal states a LATER deadline, this work is
      // stale by construction and must produce no downgrade and no retirement.
      // Refusing is safe and self-correcting: the sweep counts it as `refused`
      // (a race, which it already knows how to report), and if the new deadline
      // later lapses the next tick expires it on the new, correct evidence.
      //
      // Read here rather than in the sweep because HERE is where the batch
      // commits; a check in the sweep would be another read with another
      // interval after it. An unreadable or absent payload leaves the guard
      // silent, which preserves the existing intentional handling of grants
      // with unreadable deadlines.
      {
        // I34-RJ-02 (I34-W2-EXPIRY-B, D-1112(8)): both deadlines by the one parse every door applies (standing.ts
        // grantDeadlineMs), not `Date.parse`.
        const enforced = grantDeadlineMs((e.payload as Record<string, unknown> | undefined)?.expired_at);
        const latest = await db
          .prepare(
            // ND-1 (I34-RJ-08, D-1119): the NEWEST grant or renewal row (grantInForceOrderSql with no instant, NULL, is
            // `seq DESC`), not the one in force at this event's instant: a renewal approved between the sweep's read and this
            // append is created after the sweep's instant, and it wins.
            `SELECT payload FROM trust_events WHERE subject = ? AND event_type IN ('verification_granted','verification_renewed') ${grantInForceOrderSql("trust_events", "NULL")} LIMIT 1`,
          )
          .bind(e.subject)
          .first<{ payload: string }>();
        if (enforced !== null && latest?.payload) {
          let current: number | null = null;
          try {
            current = grantDeadlineMs((JSON.parse(latest.payload) as Record<string, unknown>).expires_at);
          } catch {
            current = null;
          }
          if (current !== null && current > enforced) {
            throw new LedgerError("verification_expired is stale: the grant it enforces has since been renewed");
          }
        }
      }
      // D9: WITHDRAWING THE BADGE WITHDRAWS THE DESTINATIONS, IN THE SAME BATCH —
      // exactly as `downgrade_verification` does below. An expiry means the same
      // thing an operator downgrade does ("this passport stops being verified"),
      // so it must have the same side effect: a staged/live payout destination
      // must not survive the expiry and silently promote on re-verification. The
      // serve gate (payout.ts CURRENTLY_VERIFIED) already stops them RENDERING;
      // this makes the STORED state agree, so "the badge lapsed" and "the money
      // destinations are gone" are one atomic fact rather than a leftover row.
      return [
        statePassport(db, e.subject, "claimed", seq),
        db.prepare("UPDATE payout_tags SET status = 'retired' WHERE passport_id = ? AND status IN ('live','staged')").bind(e.subject),
      ];
    }
    case "warning_published": {
      // Recorded in the ledger under governance (warnings.ts gates it). No
      // passport/account fold effect in S5. Defense in depth: a warning may
      // NEVER target one of our own passports — impersonation warnings are
      // about external accounts (sha256: subjects). This holds even if a
      // future caller bypasses the warnings.ts guard.
      if (e.subject.startsWith("psp_")) {
        throw new LedgerError("warning_published cannot target a psp_ passport");
      }
      // ANALYTICS (ADR-0006 §2): `impersonation_report_confirmed` is counted
      // for THE PROTECTED CREATOR, not for the warning's subject — the subject
      // is the impersonator's external account hash and belongs to no passport.
      // The pointer is optional (`warnings.ts` only attaches it for the
      // impersonation class and only when it is a well-formed passport id), so
      // a warning without one is a warning no creator's day object hears about,
      // which is the true statement.
      const protectedPsp = (p as { protected_passport_id?: unknown }).protected_passport_id;
      // Shape-checked HERE and not trusted from the payload: `assertEvent`
      // throws on a malformed passport id, and analytics never gets to veto the
      // fact it describes (0076's "no foreign keys" is the same rule).
      if ((p as { warning_class?: unknown }).warning_class !== "impersonation" || typeof protectedPsp !== "string" || !/^psp_[0-9a-f]{16}$/.test(protectedPsp)) return [];
      return outboxStatements(db, { kind: "impersonation_report_confirmed", passportId: protectedPsp, occurredDay: occurredDay(e.created_at) });
    }
    case "warning_retracted":
      // Safe-direction removal of a public warning. Like warning_published it
      // has no passport/account fold effect (the registry derives active
      // warnings from the log); the same psp guard applies.
      if (e.subject.startsWith("psp_")) {
        throw new LedgerError("warning_retracted cannot target a psp_ passport");
      }
      return [];
    case "agency_authorized":
    case "agency_revoked": {
      // Same standing checks as the family below. I3-2167 (D-0840(5), table
      // 0139): this statement keeps the bounded projection agency_report_grants
      // (read by agencies.ts agencyGrantedCreators and analytics-build.ts's
      // sweep) in the append's own batch. Forward-only on updated_seq (the
      // 0139 backfill writes the same rows); scope NULL once revoked.
      if (!e.subject.startsWith("psp_")) throw new LedgerError(`${e.event_type} must target a psp_ passport`);
      const pass = await getPassport(db, e.subject);
      if (!pass) throw new LedgerError(`no such passport: ${e.subject}`);
      if (pass.retracted === 1) throw new LedgerError(`passport retracted: ${e.subject}`);
      // I3-0738 (I22-FX-HUB): a GRANT names an agency that exists and is active now; a revoke never asks, so a
      // grant to an agency suspended since can always be ended.
      if (e.event_type === "agency_authorized") {
        const agency = await db.prepare("SELECT status FROM agencies WHERE agency_id = ?").bind(String(p.agency_id)).first<{ status: string }>();
        if (agency?.status !== "active") throw new LedgerError(`agency_authorized: no active agency ${String(p.agency_id)}`);
      }
      return [
        db
          .prepare(
            "INSERT INTO agency_report_grants (passport_id, agency_id, scope, updated_seq) VALUES (?, ?, ?, ?) " +
              "ON CONFLICT (passport_id, agency_id) DO UPDATE SET scope = excluded.scope, updated_seq = excluded.updated_seq " +
              "WHERE excluded.updated_seq > agency_report_grants.updated_seq",
          )
          .bind(e.subject, String(p.agency_id), e.event_type === "agency_authorized" ? String(p.scope) : null, seq),
      ];
    }
    case "likeness_consent_granted":
    case "likeness_consent_revoked":
    case "release_registered":
    case "release_versioned":
    case "release_withdrawn":
    // Payout tags join the same derive-from-log family. The `payout_tags` table
    // is a staging/cooling-off projection with its own lifecycle
    // (staged→live→retired), NOT a fold of the log, so there is no fold branch
    // and fold.ts stays byte-identical. What the ledger holds is the CREATOR'S
    // SIGNED INTENT: at seq N, this passkey said "route money here" / "stop
    // routing money here". That is the fact a defrauded fan and an operator both
    // need, and it is exactly what no row recorded before.
    case "payout_tag_added":
    case "payout_tag_removed": {
      // Creator-signed authorization/consent. Derived state (agency grants,
      // likeness consent) is READ from the log, not folded — so no new fold
      // table and no determinism-reducer change. Must target the creator's own
      // passport, which must exist and not be retracted.
      if (!e.subject.startsWith("psp_")) throw new LedgerError(`${e.event_type} must target a psp_ passport`);
      const pass = await getPassport(db, e.subject);
      if (!pass) throw new LedgerError(`no such passport: ${e.subject}`);
      if (pass.retracted === 1) throw new LedgerError(`passport retracted: ${e.subject}`);
      return [];
    }
    // UI-AUDIT S-C3 — the Stripe Connect binding, in the same derive-from-log
    // family and for the same reason. `creator_connect` is a state row with its
    // own lifecycle (none→connecting→pending→active→revoked), NOT a fold of the
    // log, so there is no fold branch and fold.ts stays byte-identical. What the
    // ledger holds is the CREATOR'S SIGNED INTENT: at seq N, this passkey said
    // "send my earnings to this Stripe account" / "stop sending them there".
    //
    // The binding is where DHWAJA sends money (the weekly run transfers the
    // creator's whole cleared balance to it), which is a strictly larger
    // exposure than a payout TAG — a tag is where a fan is invited to send
    // money, one fan at a time. It nonetheless shipped with no ledger row at
    // all, no alert, and no way to change it. This is the row.
    case "payout_account_bound":
    case "payout_account_unbound": {
      if (!e.subject.startsWith("psp_")) throw new LedgerError(`${e.event_type} must target a psp_ passport`);
      const pass = await getPassport(db, e.subject);
      if (!pass) throw new LedgerError(`no such passport: ${e.subject}`);
      // A retracted passport may still UNBIND — stopping money leaving is never
      // refused on standing, the same asymmetry the actor policy draws (exit is
      // wider than entry). Binding a NEW destination onto a retracted passport
      // is refused here, on top of the route's `requireVerifiedCreator`.
      if (pass.retracted === 1 && e.event_type === "payout_account_bound") throw new LedgerError(`passport retracted: ${e.subject}`);
      return [];
    }
    case "credentials_revoked": {
      // D-0762 (QUEUE #485) — derive-from-log, no fold table: `claim_credentials`
      // is the state and this row is the provenance (release + count, never a
      // hash). Its only writer is credentials.ts, which proves ownership of the
      // release before building the draft. The fold does not re-check
      // standing: the door does (`requireVerifiedCreator`, then the
      // `write_admissions` gate in the same batch — credentials.ts).
      if (!e.subject.startsWith("psp_")) throw new LedgerError("credentials_revoked must target a psp_ passport");
      const pass = await getPassport(db, e.subject);
      if (!pass) throw new LedgerError(`no such passport: ${e.subject}`);
      return [];
    }
    case "edition_set_anchored":
      // S27 tamper-evidence anchor: the subject IS the anchored set hash
      // (sha256:<hex>) — pure ledger fact, no fold table, no reducer change.
      return [];
    case "creator_broadcast": {
      // S71 one-way signed broadcast (derive-from-log): the message is a pure
      // ledger fact + delivery signal, no fold table, reducer untouched. Must
      // target the creator's own live passport (the passkey path already binds
      // actor to subject; this is defense in depth).
      if (!e.subject.startsWith("psp_")) throw new LedgerError("creator_broadcast must target a psp_ passport");
      const pass = await getPassport(db, e.subject);
      if (!pass) throw new LedgerError(`no such passport: ${e.subject}`);
      if (pass.retracted === 1) throw new LedgerError(`passport retracted: ${e.subject}`);
      return [];
    }
    case "recognition_granted":
    case "recognition_retracted": {
      // S90 creator→fan recognition (derive-from-log): a pure ledger fact — the
      // read model lives in the non-fold `recognitions` projection, so there is no
      // fold table and the determinism reducer is untouched. Subject is the
      // GRANTING creator's own live passport (the passkey path binds actor→subject;
      // this is defense in depth). The recognized fan is carried in the payload,
      // never as the subject, so recognition never touches passport/account state.
      if (!e.subject.startsWith("psp_")) throw new LedgerError(`${e.event_type} must target a psp_ passport`);
      const pass = await getPassport(db, e.subject);
      if (!pass) throw new LedgerError(`no such passport: ${e.subject}`);
      if (pass.retracted === 1) throw new LedgerError(`passport retracted: ${e.subject}`);
      return [];
    }
    case "question_answered": {
      // F9, derive-from-log — the `questions` row is the state and this event is
      // the provenance. VERIFIED-ONLY at the ledger, `handshake_answered`'s
      // reason with the stakes raised: this one PUBLISHES on a public trust
      // page, so a creator downgraded, retracted or suppressed between the
      // ceremony's two legs must not land a signed public statement.
      if (!e.subject.startsWith("psp_")) throw new LedgerError(`${e.event_type} must target a psp_ passport`);
      const qpass = await getPassport(db, e.subject);
      if (!qpass) throw new LedgerError(`no such passport: ${e.subject}`);
      if (qpass.retracted === 1) throw new LedgerError(`passport retracted: ${e.subject}`);
      if (qpass.claim_state !== "verified") throw new LedgerError(`question_answered: ${e.subject} is ${qpass.claim_state}, not verified — a public answer is published by a verified creator or not at all`);
      // I34-W2-EXPIRY-B (EXPIRY-A census: the question_answered guard): verified AT THE EVENT'S OWN INSTANT — a grant lapsed by then
      // (standing.ts lapsedGrantIds) is not verified, before the expiry sweep folds the column back.
      if ((await lapsedGrantIds(db, [e.subject], Date.parse(e.created_at))).has(e.subject)) throw new LedgerError(`question_answered: ${e.subject}'s verification has lapsed, not verified — a public answer is published by a verified creator or not at all`);
      return [];
    }
    case "question_answer_retracted": {
      // F9, the way back. Deliberately WEAKER than its sibling above: a live
      // passport, not a verified one.
      //
      // The asymmetry is the point. `question_answered` demands verification
      // because publishing under a downgraded passport would tell a fan
      // something false. Retracting says nothing to anybody — it TAKES DOWN a
      // statement — so gating it on verification would mean a creator who lost
      // their verification is stuck with a stranger's sentence on their page and
      // no lever to remove it. The remedy must not evaporate exactly when the
      // creator is in the worst position.
      //
      // A RETRACTED passport is still refused: that row is not published at all,
      // so there is nothing to take down.
      if (!e.subject.startsWith("psp_")) throw new LedgerError(`${e.event_type} must target a psp_ passport`);
      const rpass = await getPassport(db, e.subject);
      if (!rpass) throw new LedgerError(`no such passport: ${e.subject}`);
      if (rpass.retracted === 1) throw new LedgerError(`passport retracted: ${e.subject}`);
      return [];
    }
    case "handshake_answered": {
      // F7 HANDSHAKE (#43), derive-from-log — the recognition family exactly: no
      // fold table, reducer untouched, the `handshakes` row is the state and
      // this event is the provenance.
      //
      // AND THE STANDING CHECK IS AT THE LEDGER, not only at the door. A
      // Handshake answer is an attestation a fan will act on ("yes, that account
      // is mine", "yes, that is where to send money"), so it must be signed by a
      // creator who is VERIFIED at the moment the row lands — not merely one who
      // was verified when the challenge was minted. That closes the same window
      // `verifiedMintGate` exists for, on the one path where a stale signature
      // would tell a fan it is safe to send money. A downgrade, a retraction or
      // a suppression between the ceremony's two legs therefore fails the
      // append, not just the pre-read.
      if (!e.subject.startsWith("psp_")) throw new LedgerError(`${e.event_type} must target a psp_ passport`);
      const pass = await getPassport(db, e.subject);
      if (!pass) throw new LedgerError(`no such passport: ${e.subject}`);
      if (pass.retracted === 1) throw new LedgerError(`passport retracted: ${e.subject}`);
      if (pass.claim_state !== "verified") throw new LedgerError(`handshake_answered: ${e.subject} is ${pass.claim_state}, not verified — an attestation a fan acts on is signed by a verified creator or not at all`);
      // I34-W2-EXPIRY-B (EXPIRY-A census: the handshake_answered guard): verified at the event's own instant, as question_answered above.
      if ((await lapsedGrantIds(db, [e.subject], Date.parse(e.created_at))).has(e.subject)) throw new LedgerError(`handshake_answered: ${e.subject}'s verification has lapsed, not verified — an attestation a fan acts on is signed by a verified creator or not at all`);
      return [];
    }
    case "war_opt_in": {
      // S-CW4 CREATOR WAR consent (derive-from-log): no fold table, reducer
      // untouched — war-consent.ts replays these rows. On top of the family's
      // live-passport checks, ENTERING the game requires CLAIMED+ (the plan's
      // "claimed+ required"): an unclaimed listing has no proven owner, so
      // nobody exists whose consent an opt-in could record. Enforced at the
      // LEDGER, not just the doors, so no future caller can enter an unclaimed
      // creator into the game.
      if (!e.subject.startsWith("psp_")) throw new LedgerError(`${e.event_type} must target a psp_ passport`);
      const pass = await getPassport(db, e.subject);
      if (!pass) throw new LedgerError(`no such passport: ${e.subject}`);
      if (pass.retracted === 1) throw new LedgerError(`passport retracted: ${e.subject}`);
      if (pass.claim_state !== "claimed" && pass.claim_state !== "verified") {
        throw new LedgerError(`${e.event_type} requires a claimed or verified passport, was ${pass.claim_state}`);
      }
      return [];
    }
    case "account_shielded": {
      // P1 PER-ACCOUNT VISIBILITY (derive-from-log): no fold table, reducer
      // untouched — account-visibility.ts replays these rows and the two public
      // readers (passport-view.ts, cards.ts) drop what they name.
      //
      // HIDING IS NEVER BLOCKED. Unlike `war_opt_in`, this requires no claim
      // state and does not refuse a retracted passport: every state a passport
      // can be in is a state a creator might want an account out of the roster
      // for, and the failure mode of refusing is that somebody stays exposed.
      // Only two things are required, and both are about the ACCOUNT: the
      // subject is a real passport, and the account is that passport's own.
      if (!e.subject.startsWith("psp_")) throw new LedgerError(`${e.event_type} must target a psp_ passport`);
      const pass = await getPassport(db, e.subject);
      if (!pass) throw new LedgerError(`no such passport: ${e.subject}`);
      await requireOwnedAccount(db, String(e.payload.account_key), e.subject);
      return [];
    }
    case "account_unshielded": {
      // THE EXPOSING DIRECTION, and the only one of the pair that is one-way in
      // practice: a roster is public, cached and screenshottable, so a shield
      // taken off cannot be meaningfully put back. It gets the full live+claimed
      // gate (the `war_opt_in` shape) on top of the ownership check — an
      // unclaimed or retracted passport has nobody whose decision to publish
      // this could be, and the ACTOR_POLICY has already narrowed the actor to
      // the creator's own device key.
      if (!e.subject.startsWith("psp_")) throw new LedgerError(`${e.event_type} must target a psp_ passport`);
      const pass = await getPassport(db, e.subject);
      if (!pass) throw new LedgerError(`no such passport: ${e.subject}`);
      if (pass.retracted === 1) throw new LedgerError(`passport retracted: ${e.subject}`);
      if (pass.claim_state !== "claimed" && pass.claim_state !== "verified") {
        throw new LedgerError(`${e.event_type} requires a claimed or verified passport, was ${pass.claim_state}`);
      }
      await requireOwnedAccount(db, String(e.payload.account_key), e.subject);
      return [];
    }
    case "war_opt_out": {
      // EXIT IS WIDER THAN ENTRY, and that asymmetry is deliberate (review
      // S-CW4/F1; same lesson as agency_revoked's "revocation is wider than
      // authorization"). Requiring claimed+ here made consent IRREVOCABLE in
      // exactly the failure states — a passport downgraded by adjudication or
      // retracted by a takedown could never leave the game, while its
      // outstanding codes kept minting cards of a delisted person. Opting OUT
      // is the safe direction: any claim state, retracted included, only the
      // passport's existence is required.
      if (!e.subject.startsWith("psp_")) throw new LedgerError(`${e.event_type} must target a psp_ passport`);
      const pass = await getPassport(db, e.subject);
      if (!pass) throw new LedgerError(`no such passport: ${e.subject}`);
      return [];
    }
    case "correction":
      return correctionEffects(db, e, seq);
    default:
      // validateDrafts already restricts to APPEND_ALLOWED; this is defense in depth.
      throw new LedgerError(`no fold rule for event_type ${e.event_type}`);
  }
}

function statePassport(db: DbLike, id: string, state: string, seq: number): DbStatement {
  return db
    .prepare("UPDATE passports SET claim_state = ?, updated_seq = ? WHERE passport_id = ?")
    .bind(state, seq, id);
}

/**
 * SOMEBODY MUST ANSWER FOR A CHARACTER — enforced HERE, in the fold, because the
 * fold is the one place every door passes through.
 *
 * The first version of this slice put the rule in `claims.ts claimConsumedGateStmt`
 * and adversarial review found it gated ONE door of three: `oauth-claims.ts`
 * appends `claim_proven` with its own extra statements and never called it, and
 * the manual-claim path reaches `claim_proven` through `review.ts approveItem`.
 * So a character whose agency had been suspended was still claimable through the
 * OAuth callback, and the advertised lever — suspend the agency and every
 * character it runs stops being claimable — was false. A rule that lives at a
 * door is a rule somebody adds a fourth door past.
 *
 * `claim_state` is NOT NULL, so a `CASE` with no `ELSE` is the enforcement: it
 * yields NULL, the constraint throws, and the whole append rolls back — the
 * event row included. That is what makes this ATOMIC rather than a check the
 * caller performed a moment earlier: `agencies.status` and `operated_by` are read
 * by the database at the instant of the write, so a suspension that lands mid-
 * flight wins.
 *
 * A person short-circuits on `subject_kind = 'person'` before either EXISTS runs,
 * so nothing about an ordinary creator's claim changes.
 */
/**
 * THE PREDICATE ITSELF, as one exported string.
 *
 * It is a constant rather than inline SQL because a SECOND reader needs it and
 * must not paraphrase it: the operator console (`characters.ts`) tells a human
 * whether a character is claimable right now, and a console that answers from a
 * hand-copied version of this expression is a console that eventually says
 * "claimable" about a passport this gate refuses — or worse, the reverse, and an
 * operator spends an afternoon on a suspension that is not there. Sharing the
 * string makes the two answers the same answer by construction.
 *
 * It is correlated to the `passports` row it is evaluated over (`passports.` /
 * unqualified columns), so it drops into the UPDATE's CASE and into a plain
 * `SELECT ... FROM passports` alike, with no rewriting at either site.
 */
/**
 * IS THIS OPERATOR ONE SOMEBODY CAN BE HELD TO? — the rule, over an ARBITRARY
 * operator expression.
 *
 * Factored out in slice 20 so the rebind can ask the question about a CANDIDATE
 * operator without writing the rule out a second time. During an `UPDATE`,
 * `passports.operated_by` still holds the OLD value, so a rebind cannot reuse
 * `ACCOUNTABLE_SUBJECT_SQL` directly — and copying it with the candidate
 * substituted is precisely the slice-15 defect, where this predicate had drifted
 * into a third hand-written copy and the copies disagreed about a delisted
 * operator, producing an oracle. One definition, N operands.
 *
 * `operand` is interpolated as SQL, not bound, because a bind parameter cannot
 * name a column: `ACCOUNTABLE_SUBJECT_SQL` passes `passports.operated_by` and the
 * rebind passes a literal `?`. Every call site in this repo is one of those two
 * — a compile-time constant either way — and there is no path by which a request
 * value reaches this argument.
 *
 * `selfId` is the expression the operator must NOT equal. A character that
 * answers for itself answers for nothing, and the fold's own guard is the only
 * thing stopping such a row from being created and then being permanently stuck.
 */
const SQL_OPERAND_RE = /^(\?|[A-Za-z_][A-Za-z0-9_]*(\.[A-Za-z_][A-Za-z0-9_]*)?)$/;
/* D-1116(2) (EXPB-25, I34-FX2-EXPIRY): given `atMs`, the action's instant, a person operator counts only while their own
 * Verified grant is live at that instant (standing.ts `grantLiveAtSql`, inlined at `atMs` so no bind list moves), as at
 * every other verified door. Without `atMs` (the `ACCOUNTABLE_SUBJECT_SQL` const other modules read) the term is absent. */
export function accountableOperatorSql(operand: string, selfId = "passports.passport_id", atMs?: number): string {
  // I3-1064: interpolated, so admitted only as a bind marker or a (qualified)
  // column name — the two shapes every caller passes. Anything else is refused.
  if (!SQL_OPERAND_RE.test(operand) || !SQL_OPERAND_RE.test(selfId)) {
    throw new LedgerError("accountableOperatorSql: operand and selfId must be `?` or a column name");
  }
  return `EXISTS (
             SELECT 1 FROM agencies g WHERE g.agency_id = ${operand} AND g.status = 'active')
           OR EXISTS (
             SELECT 1 FROM passports o
              WHERE o.passport_id = ${operand}
                AND o.passport_id <> ${selfId}
                AND o.subject_kind = 'person'
                AND o.claim_state = 'verified'
                AND o.retracted = 0${atMs === undefined ? "" : `
                AND ${withInstantLiteral(grantLiveAtSql("o.passport_id"), atMs)}`}
                AND NOT EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = o.passport_id))`;
}

export const ACCOUNTABLE_SUBJECT_SQL = `subject_kind = 'person' OR ${accountableOperatorSql("passports.operated_by")}`;
/** `ACCOUNTABLE_SUBJECT_SQL` at the action's instant `atMs` (D-1116(2)): the fold's gates use this one. */
export const accountableSubjectSqlAt = (atMs: number): string => `subject_kind = 'person' OR ${accountableOperatorSql("passports.operated_by", "passports.passport_id", atMs)}`;

function accountablePassport(db: DbLike, id: string, state: string, seq: number, atIso: string): DbStatement {
  return db
    .prepare(`UPDATE passports SET claim_state = CASE WHEN ${accountableSubjectSqlAt(Date.parse(atIso))} THEN ? END, updated_seq = ? WHERE passport_id = ?`)
    .bind(state, seq, id);
}

/** True iff the accountability CASE above refused this append. Pinned to the
 * exact column, the way every other gate in this repo pins its own. */
export function isUnaccountableCharacter(e: unknown): boolean {
  return e instanceof Error && /NOT NULL constraint failed:\s*passports\.claim_state/i.test(e.message);
}

/** True iff the REBIND's ceiling refused — a candidate operator who is not
 * accountable, or a self-bind.
 *
 * A DIFFERENT COLUMN from `isUnaccountableCharacter`, and that is the point: the
 * two gates raise on `passports.updated_seq` and `passports.claim_state`
 * respectively, so neither can be mistaken for the other. A shared or loosened
 * matcher would let a rebind refusal be reported as "this character has no live
 * operator" — true of the OLD binding, and the exact thing the caller was trying
 * to fix.
 *
 * WHY `updated_seq` AND NOT `operated_by`: that column is nullable, so a CASE
 * over it does not throw, it CLEARS the operator. See the arm for the
 * measurement. The cost is that `updated_seq` is not a distinctive column — it
 * is written by nearly every arm — so this matcher is only unambiguous while the
 * rebind is the sole arm using it as a ceiling, which is pinned by test. */
export function isUnaccountableRebind(e: unknown): boolean {
  return e instanceof Error && /NOT NULL constraint failed:\s*passports\.updated_seq/i.test(e.message);
}

/**
 * A CORRECTION THAT SAYS "THIS CLAIM WAS WRONG" LAPSES THE CONTROL PROOFS IT
 * STOOD ON — in the SAME batch as the correction.
 *
 * WHAT THIS REPAIRS (adversarial round, 2026-08-27). `downgrade_claim` and
 * `retract_passport` are the operator's only two levers for "this claim was
 * wrong" — an impersonator who talked their way through review, a bio code that
 * turned out to be on a page the subject does not control — and both moved
 * `claim_state` and NOTHING ELSE. `claim_state` is not what any freshness reader
 * looks at: `accountProofStanding` / `freshProofPairs` (account-standing.ts) read
 * `claim_challenges` with `status = 'used'` and `oauth_challenges` with
 * `status = 'proven'`, and the revoked claimant's rows sat there untouched.
 * Measured:
 *
 *   after downgrade_claim  — claim_state: unclaimed | standing: proven_current
 *                            isAccountProven: true | creator view: confirmed:true
 *   after retract_passport — {"claim_state":"claimed","retracted":1}
 *                            standing: proven_current
 *
 * So every reader of `standing === 'proven_current'` went on believing a proof an
 * operator had just revoked as fraudulent: the registry serving class
 * (`servingClassFor`), the creator's own `confirmed` column, the removal fast
 * path, and `reports.ts`'s impersonation shield — which is the sharp one, because
 * it SILENTLY REFUSES a fan's report. It bites hardest exactly where the shield
 * matters most: `domain` / `link` and creator-asserted accounts, which shield on
 * their own per-account proof and nothing else, and every account once
 * per-account standing is on.
 *
 * A LAPSE, NOT A DELETE. The row keeps the FACT — this passport did once pass a
 * control proof, and that is history, not a claim about now — and gains an
 * explicit `valid_until` at the correction's own instant, which makes
 * `deriveStanding` answer `proven_stale`: "proved once, does not count now". That
 * is the honest sentence, it is a state with its own name and its own screens,
 * and it still lets whoever really holds the account prove it again (which a
 * delete would also allow but a `status` change to something un-provable would
 * not). `valid_until` is only ever moved BACKWARDS — a correction must never
 * extend a proof that had already lapsed on an earlier explicit deadline.
 *
 * A NON-FOLD SIDE EFFECT, and `fold.ts` deliberately does not mirror it. The
 * precedent is `downgrade_verification`, which retires `payout_tags` in the same
 * batch: neither `claim_challenges` nor `oauth_challenges` is a fold of the
 * ledger (they are proof projections with their own lifecycles, written by the
 * claim ceremonies), so they are not part of `StateSnapshot`, `/ops/fold/rebuild`
 * neither rebuilds them nor may overwrite them from the log, and the two reducers
 * stay byte-for-byte identical — INVARIANT 2 intact. If either table ever DOES
 * become a fold of the ledger, fold.ts and StateSnapshot must both grow, which is
 * exactly what the determinism guard is watching for.
 *
 * NOT applied to `downgrade_verification`: withdrawing the badge says Dhwaja
 * stopped vouching for the PERSON, and says nothing about whether the handle is
 * theirs — which is the only thing a control proof asserts. Conflating the two
 * would strip a creator's account proofs over an expired KYC file.
 */
export function lapseControlProofs(db: DbLike, passportId: string, at: string, existedBy = false): DbStatement[] {
  // EXPB-12 (I34-FX2-EXPIRY): a REPLAY (fold-rebuild.ts) of a past lapse passes `existedBy`, so it reaches only the proofs
  // that existed at `at` (proved at or before it); a proof proved after that correction was never lapsed by it.
  const by = existedBy ? " AND julianday(COALESCE(proved_at, created_at)) <= julianday(?)" : "";
  const binds = existedBy ? [at, passportId, at, at] : [at, passportId, at];
  return [
    db
      .prepare(`UPDATE claim_challenges SET valid_until = ? WHERE passport_id = ? AND status = 'used' AND (valid_until IS NULL OR valid_until > ?)${by}`)
      .bind(...binds),
    db
      .prepare(`UPDATE oauth_challenges SET valid_until = ? WHERE passport_id = ? AND status = 'proven' AND (valid_until IS NULL OR valid_until > ?)${by}`)
      .bind(...binds),
  ];
}

async function requireLivePassport(db: DbLike, id: string): Promise<PassportRow> {
  const pass = await getPassport(db, id);
  if (!pass) throw new LedgerError(`no such passport: ${id}`);
  if (pass.retracted === 1) throw new LedgerError(`passport retracted: ${id}`);
  return pass;
}

/** F5: how fresh a control proof must be for `control_reproven` (Jason: 30 days). */
export const REACTIVATION_PROOF_WINDOW_MS = 30 * 24 * 60 * 60 * 1000;

export interface RevokingEvent {
  event_id: string;
  seq: number;
  event_type: "account_revoked" | "correction";
  created_at: string;
  reason: string | null;
}

/**
 * The LATEST event that took `key` out of `active`, on WHATEVER passport owned
 * it then: an `account_revoked` naming the key, or a `correction{revoke_account}`
 * whose target `account_added` resolves to the key. Searched by KEY, not by the
 * current owner — after `account_transferred` the revocation sits in the source
 * passport's history while the key is the destination's, and that is exactly
 * the hijack case reactivation exists for. Read from the log, not from
 * `passport_accounts` (whose `updated_seq` also moves on a transfer).
 */
export async function latestRevokingEvent(db: DbLike, key: string): Promise<RevokingEvent | null> {
  const revoked = await db
    .prepare("SELECT event_id, seq, event_type, created_at, payload FROM trust_events WHERE event_type = 'account_revoked' AND json_extract(payload, '$.account_key') = ? ORDER BY seq DESC LIMIT 1")
    .bind(key)
    .first<{ event_id: string; seq: number; event_type: "account_revoked"; created_at: string; payload: string }>();
  let best: RevokingEvent | null = null;
  if (revoked) {
    let reason: string | null = null;
    try {
      const pl = JSON.parse(revoked.payload) as Record<string, unknown>;
      reason = typeof pl.reason === "string" ? pl.reason : null;
    } catch {
      /* an unparseable row still revoked */
    }
    best = { event_id: revoked.event_id, seq: revoked.seq, event_type: "account_revoked", created_at: revoked.created_at, reason };
  }
  // Corrections revoke through their TARGET (the key's `account_added` event),
  // so the key is resolved to its `account_added` event ids first and the
  // corrections are then matched by target with json_each — exact and
  // unbounded, never a "newest N" scan that a bulk correction run could push
  // the key's own correction out of (F5 review, all four personas).
  const addedIds = await accountAddedEventIds(db, key);
  // One statement per chunk of target ids (D1's bound-parameter ceiling); the
  // newest match across chunks wins.
  type CorrRow = { event_id: string; seq: number; created_at: string; payload: string };
  let corrBest: CorrRow | null = null;
  for (const chunk of inChunks(addedIds, 1)) {
    const floor: number = corrBest ? corrBest.seq : (best?.seq ?? 0);
    const corr: CorrRow | null = await db
      .prepare(
        `SELECT event_id, seq, created_at, payload FROM trust_events
          WHERE event_type = 'correction' AND json_extract(payload, '$.action') = 'revoke_account'
            AND EXISTS (SELECT 1 FROM json_each(trust_events.payload, '$.target') WHERE json_each.value IN (${chunk.map(() => "?").join(",")}))
            AND seq > ?
          ORDER BY seq DESC LIMIT 1`,
      )
      .bind(...chunk, floor)
      .first<CorrRow>();
    if (corr && (corrBest === null || corr.seq > corrBest.seq)) corrBest = corr;
  }
  if (corrBest) {
    let reason: string | null = null;
    try {
      const pl = JSON.parse(corrBest.payload) as Record<string, unknown>;
      reason = typeof pl.reason_code === "string" ? pl.reason_code : null;
    } catch {
      /* still a revoking correction */
    }
    return { event_id: corrBest.event_id, seq: corrBest.seq, event_type: "correction", created_at: corrBest.created_at, reason };
  }
  return best;
}

/**
 * The `account_added` event ids whose payload canonicalises to `key`. The
 * candidates are narrowed in SQL by the payload's id (case-insensitively — a
 * handle is lower-cased by the canonicaliser, a pid is not, a domain is a
 * host) and confirmed in JS through the ONE canonicaliser; a `link` key's id
 * is a normalised URL the raw payload does not contain, so those are narrowed
 * by id_kind instead. The RESULT is one account's listings; the READ is not
 * bounded by them: a `url` key pages through EVERY link-kind `account_added`
 * in the ledger, and an id-keyed read every event whose payload id matches
 * case-insensitively on any platform (`VERIFY_CHAIN_PAGE` rows per round trip).
 */
async function accountAddedEventIds(db: DbLike, key: string): Promise<string[]> {
  const parts = key.split("|");
  if (parts.length < 4 || parts[0] !== "v1") return [];
  const kind = parts[2];
  const id = parts.slice(3).join("|");
  const rows: { event_id: string; payload: string }[] = [];
  // I3-0244 (FIX1 — both arms): was a single unordered `LIMIT 5000` scan of
  // every matching `account_added` event in the WHOLE ledger, silently
  // dropping rows past the cap with no ordering guarantee on which 5000
  // SQLite happened to return. A `revoke_account` correction targeting a
  // dropped row's event id became invisible to `latestRevokingEvent` (called
  // above), which fails OPEN — a revoked account read back as active. R5
  // paginated only the `url`/'link' arm; the `handle`/`pid`/`domain` arm
  // (the `else` this replaced) still ran the original unbounded scan — same
  // defect, different predicate. Both arms now share ONE paginated loop, the
  // same way `verifyChain` already pages the whole table (cursor on `seq`,
  // `VERIFY_CHAIN_PAGE` rows per round trip), so no row is ever skipped,
  // however many accounts of either kind exist. The two arms differ only in
  // the WHERE clause's second condition, so that's the only thing that
  // varies below.
  const predicate = kind === "url" ? "json_extract(payload, '$.id_kind') = 'link'" : "lower(json_extract(payload, '$.id')) = lower(?)";
  const predicateArgs: unknown[] = kind === "url" ? [] : [id];
  let cursor = 0;
  for (;;) {
    const { results } = await db
      .prepare(
        `SELECT event_id, payload, seq FROM trust_events WHERE event_type = 'account_added' AND ${predicate} AND seq > ? ORDER BY seq ASC LIMIT ?`,
      )
      .bind(...predicateArgs, cursor, VERIFY_CHAIN_PAGE)
      .all<{ event_id: string; payload: string; seq: number }>();
    if (results.length === 0) break;
    for (const r of results) rows.push({ event_id: r.event_id, payload: r.payload });
    cursor = results[results.length - 1]!.seq;
    if (results.length < VERIFY_CHAIN_PAGE) break;
  }
  const out: string[] = [];
  for (const r of rows) {
    try {
      if (accountKey(JSON.parse(r.payload) as Record<string, unknown>) === key) out.push(r.event_id);
    } catch {
      /* a payload that does not canonicalise names no key */
    }
  }
  return out;
}

/**
 * A control proof of `key` on `passportId` that (a) post-dates the revocation
 * being reversed (`afterIso` — "re-proven" means proved AGAIN, the plan's
 * "≥ the revocation's seq"), (b) is no older than the reactivation window,
 * and (c) has not been LAPSED — the same `valid_until` rule `account-standing.ts`
 * applies (`downgrade_claim` stamps `valid_until` on every proof it invalidates;
 * a proof the ledger declared wrong must never re-list a key — F5 review).
 *
 * The same two tables `account-standing.ts` reads, with ONE deliberate
 * difference: a `contest` proof counts here when the contesting passport is the
 * key's CURRENT owner — that is the hijack case (the rightful owner proved
 * control while the key sat on the hijacker's passport, the operator then
 * transferred it), and `startContestProof` writes that row precisely so "the
 * proof is already on file for its new owner". `account-standing.ts` still
 * excludes contest proofs from serving standing; nothing here changes that.
 */
export async function freshControlProof(db: DbLike, passportId: string, key: string, nowMs: number, afterIso: string | null = null): Promise<boolean> {
  const nowIso = new Date(nowMs).toISOString();
  const windowStart = new Date(nowMs - REACTIVATION_PROOF_WINDOW_MS).toISOString();
  const since = afterIso !== null && afterIso > windowStart ? afterIso : windowStart;
  const row = await db
    .prepare(
      `SELECT 1 AS x FROM claim_challenges c
        WHERE c.passport_id = ? AND c.account_key = ? AND c.status = 'used'
          AND COALESCE(c.proved_at, c.created_at) > ?
          AND (c.valid_until IS NULL OR c.valid_until > ?)
          AND EXISTS (SELECT 1 FROM passport_accounts a WHERE a.account_key = c.account_key AND a.passport_id = c.passport_id)
       UNION ALL
       SELECT 1 AS x FROM oauth_challenges o
        WHERE o.passport_id = ? AND o.account_key = ? AND o.status = 'proven'
          AND COALESCE(o.proved_at, o.created_at) > ?
          AND (o.valid_until IS NULL OR o.valid_until > ?)
       LIMIT 1`,
    )
    .bind(passportId, key, since, nowIso, passportId, key, since, nowIso)
    .first();
  return row !== null;
}

/**
 * I62-EVENTS P1 (D-1519; W0-9): does `key` carry an ACTIVE public warning of a class that blocks `account_added`
 * (BLOCKS_ACCOUNT_ADDED), i.e. a `warning_published` of such a class after the subject's latest `warning_retracted`?
 * The subject is `sha256:<sha256Hex(key)>`, which is returned for the refusal's audit item_id. Asked by the
 * `account_transferred` and `account_reactivated` arms and by reactivation.ts `reactivationPreview`, so the preview
 * refuses what the ledger refuses.
 */
export async function activeBlockingWarning(db: DbLike, key: string): Promise<{ warned: boolean; subject: string }> {
  const subject = `sha256:${await sha256Hex(key)}`;
  const blockingClassPlaceholders = BLOCKS_ACCOUNT_ADDED.map(() => "?").join(", ");
  const row = await db
    .prepare(
      `SELECT 1 AS x FROM trust_events w
        WHERE w.subject = ? AND w.event_type = 'warning_published'
          AND json_extract(w.payload, '$.warning_class') IN (${blockingClassPlaceholders})
          AND w.seq > COALESCE((SELECT MAX(r.seq) FROM trust_events r
                                 WHERE r.subject = ? AND r.event_type = 'warning_retracted'), -1)
        LIMIT 1`,
    )
    .bind(subject, ...BLOCKS_ACCOUNT_ADDED, subject)
    .first<{ x: number }>();
  return { warned: row !== null, subject };
}

async function requireOwnedAccount(db: DbLike, key: string, passportId: string): Promise<void> {
  const row = await db
    .prepare("SELECT passport_id FROM passport_accounts WHERE account_key = ?")
    .bind(key)
    .first<{ passport_id: string }>();
  if (!row) throw new LedgerError(`no such account: ${key}`);
  if (row.passport_id !== passportId) {
    throw new LedgerError(`account ${key} does not belong to ${passportId}`);
  }
}

/** Corrections REDUCE state, never add it. Every action's target types are pinned. */
async function correctionEffects(
  db: DbLike,
  e: TrustEventEnvelope,
  seq: number,
): Promise<DbStatement[]> {
  const action = String(e.payload.action);
  const targets = e.payload.target as string[];
  const statements: DbStatement[] = [];

  for (const targetId of targets) {
    const target = await getEventByEventId(db, targetId);
    if (!target) throw new LedgerError(`correction target not found: ${targetId}`);
    if (target.subject !== e.subject) {
      throw new LedgerError(
        `correction subject ${e.subject} does not match target event subject ${target.subject}`,
      );
    }
    switch (action) {
      case "revoke_account": {
        if (target.event_type !== "account_added") {
          throw new LedgerError("revoke_account targets must be account_added events");
        }
        const key = accountKey(JSON.parse(target.payload) as Record<string, unknown>);
        await requireOwnedAccount(db, key, e.subject);
        statements.push(
          db
            .prepare(
              "UPDATE passport_accounts SET status = 'historical', updated_seq = ? WHERE account_key = ?",
            )
            .bind(seq, key),
        );
        break;
      }
      case "downgrade_verification": {
        if (target.event_type !== "verification_granted") {
          throw new LedgerError("downgrade_verification targets must be verification_granted events");
        }
        const pass = await requireLivePassport(db, e.subject);
        if (pass.claim_state !== "verified") {
          throw new LedgerError("downgrade_verification requires verified state");
        }
        statements.push(statePassport(db, e.subject, "claimed", seq));
        // WITHDRAWING THE BADGE WITHDRAWS THE DESTINATIONS, IN THE SAME BATCH.
        //
        // `payout_tags` is NOT a fold table — it is a staging projection with
        // its own staged→live→retired lifecycle, which is why there is no
        // `payout_tag_*` branch in this switch and why fold.ts carries no
        // mirror (a comment there records the asymmetry deliberately). This one
        // statement is a SIDE EFFECT of the downgrade, not a fold rule: the
        // serve path (payout.ts CURRENTLY_VERIFIED) is what actually stops the
        // addresses rendering, and this makes the STORED state agree with the
        // served state rather than leaving live rows behind a gate.
        //
        // Belt and braces on purpose. The read gate is load-bearing and would
        // hold alone; this exists so that "the operator withdrew the badge" and
        // "the money destinations are gone" are one atomic fact in the same
        // batch as the ledger row, instead of a sweep someone has to run.
        statements.push(
          db
            .prepare("UPDATE payout_tags SET status = 'retired' WHERE passport_id = ? AND status IN ('live','staged')")
            .bind(e.subject),
        );
        break;
      }
      case "downgrade_claim": {
        if (target.event_type !== "claim_proven") {
          throw new LedgerError("downgrade_claim targets must be claim_proven events");
        }
        const pass = await requireLivePassport(db, e.subject);
        if (pass.claim_state === "verified") {
          throw new LedgerError("downgrade verification before downgrading the claim");
        }
        if (pass.claim_state !== "claimed") {
          throw new LedgerError("downgrade_claim requires claimed state");
        }
        statements.push(statePassport(db, e.subject, "unclaimed", seq));
        // A CORRECTION THAT SAYS "THIS CLAIM WAS WRONG" HAS TO REACH THE PROOF.
        statements.push(...lapseControlProofs(db, e.subject, e.created_at));
        break;
      }
      case "retract_passport": {
        if (target.event_type !== "passport_created") {
          throw new LedgerError("retract_passport targets must be passport_created events");
        }
        await requireLivePassport(db, e.subject);
        // M30: a retraction is a takedown, so it also ERASES the creator's
        // plaintext name (the "right to be forgotten" the ledger was designed
        // to allow). The name lives in the mutable `passport_names` table keyed
        // by name_ref — the ledger keeps only {name_ref, name_hash}, so deleting
        // the rows removes the readable PII while the append-only proof stands.
        // These ride the SAME atomic batch as the retraction, and both reducers
        // stay in agreement by construction: the replay reducer resolves
        // display_name from `passport_names` (fold.ts nameOf), so once the rows
        // are gone it also produces "" — exactly what erasePassportNames does.
        statements.push(
          // I64-HOLD H2 (W0 am.3): a hold that keeps the names below owes their erase after the release (requests.ts
          // sweepRetractionNamesOwed, state-driven); first, while the names are still there to be counted.
          db
            .prepare(
              `INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT 'system:retraction', 'retraction:names_erase_owed', 'retraction: names kept under an open legal hold; erasure owed on release', ?, ?, 'system'
                WHERE ${PASSPORT_LEGAL_HOLD_SQL} AND EXISTS (SELECT 1 FROM passport_names n WHERE n.passport_id = ?)`,
            )
            .bind(e.created_at, e.subject, e.subject, e.subject),
          db
            .prepare("UPDATE passports SET retracted = 1, display_name = '', display_name_skeleton = NULL, updated_seq = ? WHERE passport_id = ?")
            .bind(seq, e.subject),
          db.prepare(`DELETE FROM passport_names WHERE passport_id = ? AND NOT ${PASSPORT_LEGAL_HOLD_SQL}`).bind(e.subject, e.subject), // I63-FX-ERASE: a hold keeps the names
          // I3-1494: the passport's brand_index name/domain entries leave in this
          // batch too, not one sweep later (brand-index.ts brandIndexErasureStmt).
          brandIndexErasureStmt(db, e.subject),
          // QUEUE #237 (D-0384; I41-NEWSLETTER H-RETRACT): the creator's newsletter list (0283) leaves in this batch too,
          // not at the removal wipe 30 days later (creator-erasure.ts wipeStatements covers that path as well).
          db.prepare("DELETE FROM newsletter_subscriptions WHERE passport_id = ?").bind(e.subject),
          // A takedown cannot leave a live control proof standing either — see
          // `lapseControlProofs`. Note `claim_state` is NOT moved by a
          // retraction (a retracted passport keeps whatever state it had, with
          // `retracted = 1` beside it), so without this the proof rows of a
          // taken-down passport went on reading `proven_current` forever.
          ...lapseControlProofs(db, e.subject, e.created_at),
          // #166 (D-0925(4), D-0933(3); I16-FXD H1): the per-creator analytics purge work order, in THIS batch, so a
          // committed retraction always has one. Drained by the analytics tick (analytics-build.ts drainCreatorErasureOrder).
          // A plain INSERT: a passport is retracted once (requireLivePassport above), and the only other writer
          // (analytics-read.ts purgeIfRetracted) writes only for an already-retracted passport.
          db
            .prepare("INSERT INTO analytics_creator_erasures (passport_id, requested_at) VALUES (?, ?)")
            .bind(e.subject, e.created_at),
          // D-0960(12) (I20-W1C H1, landed by I20-L02): a retraction owes the image service's biometric revoke in THIS
          // batch — the upsert likeness.ts imageConsentOwedStatements writes (inlined: likeness.ts imports this module),
          // drained by likeness.ts drainImageConsentRevokeOutbox. I65-HOLD F1 (D-1566, D-1569): not while a hold names the
          // passport or its founder account; the OWED marker is written instead and requests.ts sweepFaceRevokeOwed queues it.
          db.prepare(`INSERT INTO image_consent_revoke_outbox (passport_id, queued_at) SELECT ?, ? WHERE NOT ${FACE_DATA_HOLD_SQL} ON CONFLICT(passport_id) DO UPDATE SET gen = gen + 1, queued_at = MAX(queued_at, excluded.queued_at)`).bind(e.subject, e.created_at, e.subject, e.subject),
          db.prepare(`INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT 'system:retraction', 'face:revoke_owed', 'retraction: face-data revoke kept under an open legal hold; owed on release', ?, ?, 'system' WHERE ${FACE_DATA_HOLD_SQL}`).bind(e.created_at, e.subject, e.subject, e.subject),
          // I3-0680 a (D-0993(1), I23-W1-ANALYTICS hand-off 6): a retracted creator's push devices leave in THIS batch.
          db.prepare("DELETE FROM creator_push_tokens WHERE passport_id = ?").bind(e.subject),
          // I3-0203 b (I25-W1-IDENTITY H5): and any pending push-possession challenge (it carries a push token).
          db.prepare("DELETE FROM creator_push_challenges WHERE passport_id = ?").bind(e.subject),
          // I30-OWNER (0265): and any held push-device removal (it names one of the devices this batch deletes).
          db.prepare("DELETE FROM creator_push_pending_removals WHERE passport_id = ?").bind(e.subject),
        );
        break;
      }
      default:
        throw new LedgerError(`unknown correction action: ${action}`);
    }
  }
  return statements;
}

export interface ChainHead {

  event_hash: string;

  const row = await db
    .prepare("SELECT seq, event_hash FROM trust_events ORDER BY seq DESC LIMIT 1")

    .first<{ seq: number; event_hash: string }>();

/**
 * Validate, complete, chain, and atomically apply a batch of draft events.
 * All drafts succeed or none do. Returns the completed envelopes in order.
 *
 * RETRIES A LOST HEAD RACE, HERE, ONCE AND FOR ALL. `prev_hash` is UNIQUE, so
 * this is a global optimistic lock: two appends anywhere in the product that
 * read the same head collide, and at concurrency above one writer that is the
 * steady state, not the exception. Losing re-reads the head and REBUILDS from
 * the drafts — new `created_at`, new event ids, new hashes, freshly minted names
 * (the failed batch rolled its `passport_names` INSERT back, so reusing a
 * name_ref would fold the passport's name to ""). Nothing is mutated and no
 * committed row is touched: `trust_events` stays append-only and the pinned v1
 * preimage is unchanged.
 *
 * Exhausting the retries throws `LedgerContentionError` — a class no caller can
 * mistake for a domain outcome. See that type for the five callers that did.
 *
 * REQUEST-PATH PUBLISH (QUEUE #19, D-0441(4)). `afterCommit` — built by
 * `publishHook(pubEnv, pubCtx)` in `analytics-outbox.ts`, passed in rather than
 * imported because `analytics-outbox.ts` reaches back to this module (a type
 * import alone closes a cycle under `.dependency-cruiser.cjs` no-circular) —
 * runs ONCE, after the retry loop below has a committed batch, and is handed
 * the seqs of the analytics_outbox rows THAT batch wrote (`outboxSpan`, below):
 * `foldEffects`' creator_claimed / impersonation_report_confirmed and caller
 * `extraStatements` such as the answer statements publish at request time, and
 * no other request's row is sent or charged an attempt (D-0871(7)). A rejected
 * batch or a lost head race never reaches it. A hook that THROWS is caught and
 * logged through `logError` with a static message — the write committed, so the
 * call still resolves, and the write is not retried (D-0871(7)). Absent, nothing
 * is published here and the cron drain owns the rows.
 */
export async function appendEvents(
  db: DbLike,
  drafts: DraftEvent[],
  now: () => string = () => new Date().toISOString(),
  governedTypes: readonly string[] = [],
  extraStatements: DbStatement[] = [],
  authorizedPrivilegedActors: readonly string[] = [],
  afterCommit?: (ownSeqs: readonly number[]) => void,
): Promise<TrustEventEnvelope[]> {
  const ownSeqs: number[] = [];
  let committed: TrustEventEnvelope[];
  for (let attempt = 1; ; attempt++) {
    try {
      ownSeqs.length = 0;
      committed = await appendEventsOnce(db, drafts, now, governedTypes, extraStatements, authorizedPrivilegedActors, afterCommit ? ownSeqs : undefined);
      break;
    } catch (e) {
      if (!(e instanceof HeadRace)) throw e;
      if (attempt >= APPEND_MAX_ATTEMPTS) {
        // I3-1419: the one place a refused write is counted — every door (HTTP 503, cron) passes through here.
        logError({ cid: "unknown", hop: "http", msg: "ledger_contention", count: attempt });
        throw new LedgerContentionError(attempt);
      }
      await backoff(attempt);
    }
  }
  if (afterCommit) {
    try {
      afterCommit(ownSeqs);
    } catch {
      logError({ cid: "unknown", hop: "http", msg: "appendEvents: afterCommit publish hook threw after commit; cron drain owns the rows" });
    }
  }
  // I25-Q8: every committed `retract_passport` correction is journaled for a restore — here, because every retraction
  // door appends through this function (tryAppendEvents delegates). journalRestoreAct never throws.
  for (const e of committed)
    if (e.event_type === "correction" && e.payload.action === "retract_passport")
      await journalRestoreAct("retraction", e.subject, e.created_at, typeof e.payload.reason_code === "string" ? e.payload.reason_code : undefined);
  return committed;
}

/**
 * OUTBOX SPAN (QUEUE #19, D-0871(7)): two reads that bracket a batch. `open`
 * reads the outbox's highest seq BEFORE the batch's writes, `close` the newest
 * `OUTBOX_SPAN_LIMIT` seqs AFTER them. A D1 batch is one transaction, so a seq in
 * `close` above `open`'s value was written by THIS batch and by nothing else —
 * `outboxSpanSeqs` returns exactly those, ascending. A batch that wrote more than
 * `OUTBOX_SPAN_LIMIT` rows hands back only its newest `OUTBOX_SPAN_LIMIT`; the rest
 * are left to the cron drain. A result in any other shape yields [] (send nothing
 * here; the drain owns the rows).
 */
export const OUTBOX_SPAN_LIMIT = 500;

export function outboxSpan(db: DbLike): { open: DbStatement; close: DbStatement } {
  return {
    open: db.prepare("SELECT COALESCE(MAX(seq), 0) AS hi FROM analytics_outbox").bind(),
    close: db.prepare("SELECT seq FROM analytics_outbox ORDER BY seq DESC LIMIT ?").bind(OUTBOX_SPAN_LIMIT),
  };
}

export function outboxSpanSeqs(open: unknown, close: unknown): number[] {
  const rows = (r: unknown): Record<string, unknown>[] | null => {
    const x = (r as { results?: unknown } | null | undefined)?.results;
    return Array.isArray(x) ? (x as Record<string, unknown>[]) : null;
  };
  const before = rows(open);
  const after = rows(close);
  if (!before || !after || before.length !== 1) return [];
  const hi = Number(before[0]!.hi);
  if (!Number.isSafeInteger(hi)) return [];
  return after
    .map((r) => Number(r.seq))
    .filter((seq) => Number.isSafeInteger(seq) && seq > hi)
    .sort((a, b) => a - b);
}

/**
 * The DOMAIN-SAFE door onto `appendEvents`. Returns the two refusals a caller is
 * entitled to translate and lets everything else — contention above all —
 * propagate. Callers that must map an append failure onto a user-visible outcome
 * use this instead of `try/catch` + a message regex, because the regex is
 * exactly how a contention loss became `code_absent`.
 *
 * A refusal carries its original `error`, so a caller that translates only SOME
 * refusals can rethrow the rest unchanged: this door narrows what may be
 * translated, it does not flatten every ledger error into one outcome.
 */
export async function tryAppendEvents(
  db: DbLike,
  drafts: DraftEvent[],
  now: () => string = () => new Date().toISOString(),
  governedTypes: readonly string[] = [],
  extraStatements: DbStatement[] = [],
  authorizedPrivilegedActors: readonly string[] = [],
  afterCommit?: (ownSeqs: readonly number[]) => void,
): Promise<AppendResult> {
  try {
    return { ok: true, events: await appendEvents(db, drafts, now, governedTypes, extraStatements, authorizedPrivilegedActors, afterCommit) };
  } catch (e) {
    if (e instanceof LedgerError) return { ok: false, refusal: { kind: "state", message: e.message, details: e.details, error: e } };
    // A UNIQUENESS failure that survived `appendEvents` is NOT a head race (that
    // one either retried to success or threw LedgerContentionError) — it is a
    // uniqueness lock the caller rode in on `extraStatements`, and losing it IS
    // a domain answer ("somebody already did this").
    if (e instanceof Error && !(e instanceof LedgerContentionError) && UNIQUENESS_LOST_RE.test(e.message)) {
      return { ok: false, refusal: { kind: "conflict", message: e.message, error: e } };
    }
    throw e;
  }
}

async function appendEventsOnce(
  db: DbLike,
  drafts: DraftEvent[],
  now: () => string,
  governedTypes: readonly string[],
  extraStatements: DbStatement[],
  authorizedPrivilegedActors: readonly string[],
  ownSeqs?: number[],
): Promise<TrustEventEnvelope[]> {
  if (drafts.length === 0) throw new LedgerError("empty draft batch");
  if (drafts.length > 500) throw new LedgerError("draft batch too large (max 500)");

  // 0. NAME INDIRECTION (window item 4). A passport_created/passport_renamed
  //    draft may carry the NAME (`payload.display_name`); the LEDGER never does.
  //    Here — once, in the ledger's only write door, BEFORE validation and
  //    hashing — the string is validated, minted into `passport_names`, and
  //    replaced by {name_ref, name_hash}. Putting it here rather than in ~90
  //    call sites is what makes the guarantee STRUCTURAL: there is no path by
  //    which a caller can land a name string in trust_events, and validatePayload
  //    (which runs on the substituted payload) rejects `display_name` outright as
  //    an unknown field, so this step cannot silently stop working.
  //
  //    REFUSED ON THE SIGNED PATH. A privileged (passkey/agency) draft is bound
  //    by an assertion over the draft itself, so rewriting its payload here would
  //    mean the stored event is not the event that was signed. Such a caller must
  //    mint the name at the /start leg and echo {name_ref, name_hash} into both
  //    legs — the same pattern drop.ts uses for broadcast_id.
  //    The caller's array is NEVER mutated: substitution builds a local copy.
  //    appendEvents is retried by design (a racing append loses the prev_hash
  //    UNIQUE and the caller re-appends on the new head), and an in-place rewrite
  //    would make the retry reuse a name_ref whose INSERT rolled back with the
  //    failed batch — folding the passport's name to "" — instead of minting a
  //    fresh one.
  const nameStatements: DbStatement[] = [];
  const batchNames = new Map<string, string>();
  const prepared: DraftEvent[] = [...drafts];
  for (const [i, d] of prepared.entries()) {
    if (d.event_type !== "passport_created" && d.event_type !== "passport_renamed") continue;
    if (!d.payload || typeof d.payload !== "object" || !("display_name" in d.payload)) continue;
    if (authorizedPrivilegedActors.includes(d.actor_kind)) {
      throw new LedgerError(
        `draft[${i}] ${d.event_type}: a signed draft must carry {name_ref, name_hash} (mint the name at /start and echo it), never display_name — rewriting it here would break the assertion binding`,
      );
    }
    const minted = await mintName(db, d.subject, d.payload.display_name, now);
    nameStatements.push(minted.statement);
    batchNames.set(minted.name_ref, minted.display_name);
    const { display_name: _replaced, ...rest } = d.payload;
    prepared[i] = { ...d, payload: { ...rest, name_ref: minted.name_ref, name_hash: minted.name_hash } };
  }

  // 1. Shape + vocabulary + payload validation (fail closed, before any I/O).
  for (const [i, d] of prepared.entries()) {
    for (const k of Object.keys(d)) {
      if (!DRAFT_FIELDS.has(k)) throw new LedgerError(`draft[${i}] has unknown field: ${k}`);
    }
    // Governed types (e.g. warning_published) are refused by the general path
    // and only accepted when the caller — the warnings governance code — has
    // passed them in explicitly, having already checked cooling-off, second
    // look, and an approved wording. General callers cannot mint warnings.
    const allowed =
      (APPEND_ALLOWED as readonly string[]).includes(d.event_type) ||
      governedTypes.includes(d.event_type);
    if (!allowed) {
      throw new LedgerError(
        `draft[${i}] event_type ${String(d.event_type)} is not accepted by the ledger yet`,
      );
    }
    const allowedActors = ACTOR_POLICY[d.event_type];
    if (!allowedActors || !allowedActors.includes(d.actor_kind)) {
      throw new LedgerError(
        `draft[${i}] actor_kind ${String(d.actor_kind)} may not emit ${d.event_type}`,
      );
    }
    // Privileged actor kinds are the "touched-by" marks that assert a specific
    // human authorized this action: 'creator_passkey' ("by creator") and
    // 'agency' ("by authorized agency"). The general path CANNOT set them — they
    // are minted only by a caller that first verified the authorization
    // (appendCreatorSigned for a passkey assertion; the agency layer for an
    // agency session), which passes the actor kind in authorizedPrivilegedActors.
    // This is what makes the creator mark unforgeable by anyone (an agency
    // included) who lacks the creator's device key.
    if ((d.actor_kind === "creator_passkey" || d.actor_kind === "agency" || d.actor_kind === "system") && !authorizedPrivilegedActors.includes(d.actor_kind)) {
      // 'system' joined this gate in S27 (defense in depth): system events are
      // emitted only by server code that proved the triggering condition (a
      // bio-code/OAuth claim proof, the editions cron) — a future handler that
      // naively forwarded caller-supplied actor kinds must not be able to
      // forge system events (anchors included).
      throw new LedgerError(
        `draft[${i}] actor_kind ${d.actor_kind} requires a verified authorization (not accepted on the general path)`,
      );
    }
    const pv = validatePayload(d.event_type, d.payload, d.actor_kind);
    if (!pv.ok) throw new LedgerError(`draft[${i}] payload invalid`, pv.errors);
    // I3-1136 (D-0858(22)): the url a fan sees as this account's Open link must
    // point AT the account. Checked HERE, on new drafts only — never in
    // validatePayload or the fold — so events already on file still replay.
    // The canonical subject is the platform/id the fold stores and the page
    // reads; an unmappable account is left to the fold's own refusal.
    if (d.event_type === "account_added" && typeof d.payload?.url === "string") {
      let subj: { platform: string; id: string } | null = null;
      try {
        subj = subjectForAccount(String(d.payload.platform), String(d.payload.id_kind), String(d.payload.id));
      } catch {
        subj = null;
      }
      if (subj && !urlNamesAccount(subj.platform, subj.id, d.payload.url)) {
        throw new LedgerError(`draft[${i}] ${accountUrlRefusal(subj.id)}`);
      }
    }
  }

  // 2. Chain + fold. Statements accumulate across the batch; fold guards see
  // committed state only (drafts within one batch that depend on each other's
  // fold effects are handled for the common create-then-add case below).
  const head = await chainHead(db);
  let prevHash = head?.event_hash ?? GENESIS_HASH;
  let seq = head?.seq ?? 0;

  const statements: DbStatement[] = [];
  const completed: TrustEventEnvelope[] = [];
  const createdInBatch = new Map<string, PassportRow>();
  const accountsInBatch = new Set<string>();
  const lifecycleInBatch = new Set<string>();

  for (const d of prepared) {
    seq += 1;
    // I3-2035: ONE clock per envelope — event_id's embedded time is created_at's.
    const createdAt = now();
    const envelope: TrustEventEnvelope = {
      event_id: uuid7(Date.parse(createdAt)),
      event_type: d.event_type,
      subject: d.subject,
      actor_kind: d.actor_kind,
      actor_id: d.actor_id,
      created_at: createdAt,
      payload: d.payload,
      supersedes: d.supersedes ?? [],
      prev_hash: prevHash,
    };
    const ev = validateEnvelope(envelope);
    if (!ev.ok) throw new LedgerError("completed envelope invalid", ev.errors);

    for (const superseded of envelope.supersedes) {
      const t = await getEventByEventId(db, superseded);
      if (!t) throw new LedgerError(`supersedes target not found: ${superseded}`);
      // A-42 (S-036-batch EVENTS_TRU): "target exists" was the ONLY check —
      // no subject constraint — so an actor authorized to append for their OWN
      // subject could name ANY event_id in the ledger, including another
      // subject's, as superseded. `parseActions` (route-helpers.ts) accepts
      // `supersedes` straight from client-supplied creator/agency actions, so
      // this was reachable from the door, not just from a privileged caller.
      // supersedes only ever means "this event replaces an earlier one for the
      // SAME subject" (every real caller — passkeys.ts, agencies.ts,
      // releases.ts — supersedes its own prior event); bind that here, once,
      // so a future reader of the column can rely on it without re-deriving it
      // from every writer.
      if (t.subject !== envelope.subject) {
        throw new LedgerError(`supersedes target ${superseded} belongs to a different subject`);
      }
    }

    // I3-0318: foldEffects checks every precondition against COMMITTED state,
    // while fold.ts replayState applies the batch in seq order. So a draft whose
    // passport or account an EARLIER lifecycle draft of this batch changed would
    // be judged on stale state (claim_proven then subject_kind_declared: the
    // writer applies the declaration, the reducer declines it). Refused, like a
    // lifecycle event on a passport born in-batch.
    const scope = await batchScope(db, envelope);
    if (scope.reads.some((x) => lifecycleInBatch.has(x))) {
      throw new LedgerError(`${envelope.event_type} on ${envelope.subject}: an earlier draft in the same batch changed this passport or account — append it in a subsequent batch`);
    }
    const effects = await foldEffectsWithBatchState(
      db,
      envelope,
      seq,
      createdInBatch,
      accountsInBatch,
      batchNames,
    );
    for (const x of scope.writes) lifecycleInBatch.add(x);
    const eventHash = await sha256Hex(
      eventPreimage({ ...envelope, prev_hash: prevHash }),
    );

    statements.push(
      db
        .prepare(
          // I3-1623: seq is BOUND, so the row's seq is the one the fold effects
          // above were written with; a racing append collides on it (a head race).
          "INSERT INTO trust_events (seq, event_id, event_type, subject, actor_kind, actor_id, created_at, payload, supersedes, prev_hash, event_hash) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)",
        )
        .bind(
          seq,
          envelope.event_id,
          envelope.event_type,
          envelope.subject,
          envelope.actor_kind,
          envelope.actor_id,
          envelope.created_at,
          stableStringify(envelope.payload),
          stableStringify(envelope.supersedes),
          prevHash,
          eventHash,
        ),
      ...effects,
    );
    completed.push(envelope);
    prevHash = eventHash;
  }

  // Extra statements ride in the SAME atomic batch as the events — used by
  // governed writers (warnings) to include a uniqueness lock + state flip +
  // audit, so a duplicate publish fails the whole batch on a constraint.
  // `ownSeqs` (a hooked append only): the batch is bracketed by `outboxSpan`, so
  // the hook learns exactly the outbox rows THIS batch wrote. Unhooked, the
  // batch is unchanged.
  const span = ownSeqs ? outboxSpan(db) : null;
  let results: unknown[];
  try {
    results = await db.batch(span ? [span.open, ...nameStatements, ...statements, ...extraStatements, span.close] : [...nameStatements, ...statements, ...extraStatements]);
  } catch (e) {
    if (await isHeadRace(db, e, head?.event_hash ?? GENESIS_HASH)) throw new HeadRace(e);
    throw e;
  }
  if (span && ownSeqs) ownSeqs.push(...outboxSpanSeqs(results[0], results[results.length - 1]));
  return completed;
}

/**
 * #132W-08: append the operator's kind CORRECTION (the route in console.ts runs
 * it under dual control). Returns the event, or the ledger's refusal message
 * (no such passport, retracted, claimed) as a state answer.
 */
export async function correctSubjectKind(
  db: DbLike,
  operator: string,
  passportId: string,
  subjectKind: string,
  operatedBy: string | null,
  now: () => string = () => new Date().toISOString(),
  // QUEUE #942(a) D17 (D-0946(1)): the caller's audit row, joined to the correction's own batch; a failed audit
  // insert rolls the correction back, and a ledger refusal writes no row.
  extraStatements: DbStatement[] = [],
): Promise<{ ok: true; event_id: string } | { ok: false; message: string }> {
  const payload: Record<string, unknown> = subjectKind === "person" ? { subject_kind: subjectKind } : { subject_kind: subjectKind, operated_by: operatedBy };
  const r = await tryAppendEvents(db, [{ event_type: "subject_kind_corrected", subject: passportId, actor_kind: "operator", actor_id: operator, payload } as DraftEvent], now, ["subject_kind_corrected"], extraStatements);
  if (!r.ok) return { ok: false, message: r.refusal.message };
  return { ok: true, event_id: r.events[0]!.event_id };
}

/**
 * #799c: THE refusal sentence for an `account_added` url that does not point at
 * the account it lists — ONE copy, used by this door and by the importer's
 * dry-run/apply pre-check (importer.ts profileUrlProblem). It states the rule
 * `urlNamesAccount` (packages/schema) actually enforces: the host must be the
 * platform's own (or the domain itself, for a domain account) AND the id must
 * sit at that platform's profile position. The old wording ("its host or path
 * must name <id>") described a looser OR rule than the predicate applies, so an
 * operator fixing a refused line was told the wrong thing to change.
 */
/**
 * I3-0213 (D-0920, option C): a key held by a RETRACTED passport is not a
 * duplicate of a live listing — it is a listing an operator took down, and the
 * remedy is to restore that passport (keys stay with their original passport;
 * no key reuse, option A refused), not to re-list the subject. Both
 * `account_added` dup arms say so, and the importer maps the sentence to its
 * own `retracted_earlier` result instead of a duplicate. OPERATOR AND SYSTEM
 * doors only: a creator or agency adding an account keeps the unchanged
 * "already registered" sentence, because telling a self-serve caller that a
 * key sat on an operator-withdrawn listing would publish the withdrawal.
 */
export const RETRACTED_EARLIER_RE = /^account retracted earlier: /;
function retractedEarlier(known: { retracted_holder?: number } | null | undefined, actorKind: string): boolean {
  return !!known?.retracted_holder && (actorKind === "operator" || actorKind === "system");
}
function duplicateAccountRefusal(canonical: string, known: { retracted_holder?: number } | null | undefined, actorKind: string): string {
  return retractedEarlier(known, actorKind)
    ? `account retracted earlier: ${canonical} — its passport was retracted; restore that passport instead of re-listing the account`
    : `account already registered: ${canonical}`;
}

export function accountUrlRefusal(id: string): string {
  return `account_added: url must point at the account it lists (its host must be the platform's own, and ${id} must sit where that platform puts its profile)`;
}

/** I3-0318: the fold.ts arms that change a PASSPORT row (`s:`) — every other
 * fold-changing arm other than create/add changes an ACCOUNT row (`k:`). */
const BATCH_PASSPORT_LIFECYCLE: ReadonlySet<string> = new Set([
  "claim_proven",
  "subject_kind_declared",
  "subject_kind_corrected",
  "character_operator_rebound",
  "verification_granted",
  "verification_renewed",
  "operator_lapsed",
  "verification_expired",
  "passport_renamed",
]);
const BATCH_ACCOUNT_LIFECYCLE: ReadonlySet<string> = new Set(["account_revoked", "account_reactivated", "account_transferred"]);

/**
 * What a draft READS (its passport, the passport it moves an account off, the
 * accounts it names) and what it WRITES (the passport or accounts its fold arm
 * changes). A `revoke_account` correction names its accounts through its target
 * `account_added` events, so those are resolved here; a target that does not
 * resolve is left to the arm's own refusal.
 */
async function batchScope(db: DbLike, e: TrustEventEnvelope): Promise<{ reads: string[]; writes: string[] }> {
  const p = e.payload as Record<string, unknown>;
  const passports = [`s:${e.subject}`];
  if (typeof p.from_passport === "string") passports.push(`s:${p.from_passport}`);
  const accounts: string[] = [];
  if (typeof p.account_key === "string") accounts.push(`k:${p.account_key}`);
  const revokes = e.event_type === "correction" && p.action === "revoke_account" && Array.isArray(p.target);
  if (revokes) {
    for (const id of p.target as unknown[]) {
      const t = typeof id === "string" ? await getEventByEventId(db, id) : null;
      if (t?.event_type !== "account_added") continue;
      try {
        accounts.push(`k:${accountKey(JSON.parse(t.payload) as Record<string, unknown>)}`);
      } catch {
        /* left to correctionEffects */
      }
    }
  }
  const writes = BATCH_PASSPORT_LIFECYCLE.has(e.event_type) || (e.event_type === "correction" && !revokes)
    ? [`s:${e.subject}`]
    : BATCH_ACCOUNT_LIFECYCLE.has(e.event_type) || revokes
      ? accounts
      : [];
  return { reads: [...passports, ...accounts], writes };
}

/** foldEffects, aware of passports/accounts created earlier in this same batch. */
async function foldEffectsWithBatchState(
  db: DbLike,
  e: TrustEventEnvelope,
  seq: number,
  createdInBatch: Map<string, PassportRow>,
  accountsInBatch: Set<string>,
  batchNames: Map<string, string> = new Map(),
): Promise<DbStatement[]> {
  if (e.event_type === "passport_created") {
    if (createdInBatch.has(e.subject)) {
      throw new LedgerError(`passport created twice in batch: ${e.subject}`);
    }
    const effects = await foldEffects(db, e, seq, batchNames);
    createdInBatch.set(e.subject, {
      passport_id: e.subject,
      claim_state: "unclaimed",
      retracted: 0,
      // Every passport is minted a PERSON — migration 0047's column default, and
      // the only value `passport_created` can produce, since `subject_kind` is
      // set exclusively by `subject_kind_declared` and that event requires an
      // already-unclaimed passport. Stated rather than left out so the in-batch
      // row is the same shape as the one a re-read would return.
      subject_kind: "person",
    });
    return effects;
  }
  if (e.event_type === "account_added" && createdInBatch.has(e.subject)) {
    const p = e.payload;
    const subject = subjectForAccount(String(p.platform), String(p.id_kind), String(p.id));
    if ((subject.idKind === "domain" || subject.idKind === "url") && e.actor_kind !== "operator" && e.actor_kind !== "system") {
      throw new LedgerError("a domain or generic-link account may only be added by an operator; control is proven separately");
    }
    if (accountsInBatch.has(subject.canonical)) {
      throw new LedgerError(`account added twice in batch: ${subject.canonical}`);
    }
    // U-08/PS-005 (S-036-batch EVENTS_TRU): this in-batch twin used to run only
    // the dup check above and skip the two-part gate its out-of-batch sibling
    // (below, same function's non-batch account_added arm) always runs —
    // suppressed_account_keys (a delisted subject's key, frozen so it is never
    // re-listed) and an ACTIVE warning_published (INV-003's addition-side
    // door). A bulk import that creates the passport and adds the account in
    // ONE batch bypassed both. Same query, same refusal string, same audit
    // trail as the sibling — see the comment above that arm for the full
    // INV-003 reasoning; it applies unchanged here.
    const warnSubject = `sha256:${await sha256Hex(subject.canonical)}`;
    const blockingClassPlaceholders = BLOCKS_ACCOUNT_ADDED.map(() => "?").join(", ");
    const known = await db
      .prepare(
        `SELECT EXISTS (SELECT 1 FROM passport_accounts WHERE account_key = ?) AS dup,
                EXISTS (SELECT 1 FROM passport_accounts ra JOIN passports rp ON rp.passport_id = ra.passport_id
                         WHERE ra.account_key = ? AND rp.retracted = 1) AS retracted_holder,
                EXISTS (SELECT 1 FROM suppressed_account_keys WHERE account_key = ?) AS frozen,
                EXISTS (SELECT 1 FROM trust_events w
                         WHERE w.subject = ? AND w.event_type = 'warning_published'
                           AND json_extract(w.payload, '$.warning_class') IN (${blockingClassPlaceholders})
                           AND w.seq > COALESCE((SELECT MAX(r.seq) FROM trust_events r
                                                  WHERE r.subject = ? AND r.event_type = 'warning_retracted'), -1)) AS warned`,
      )
      .bind(subject.canonical, subject.canonical, subject.canonical, warnSubject, ...BLOCKS_ACCOUNT_ADDED, warnSubject)
      .first<{ dup: number; retracted_holder: number; frozen: number; warned: number }>();
    if (known?.dup) {
      throw new LedgerError(duplicateAccountRefusal(subject.canonical, known, e.actor_kind));
    }
    if (known?.frozen || known?.warned) {
      await db
        .prepare(
          "INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) VALUES (?, 'account_added:refused', ?, ?, ?, ?)",
        )
        .bind(
          e.actor_id,
          `${subject.canonical} refused (${known.warned ? "warned" : "delisted"}, in-batch) — applicable remedy: ${known.warned ? "warning_retracted" : "unsuppress"}`,
          e.created_at,
          warnSubject,
          touchedByFromActor(e.actor_kind),
        )
        .run();
      throw new LedgerError(
        `account_added: ${subject.canonical} cannot be added — it was delisted at the subject's request and is never re-listed — lift the removal (unsuppress) first, so the re-listing is a declared act`,
      );
    }
    accountsInBatch.add(subject.canonical);
    return [
      db
        .prepare(
          "INSERT INTO passport_accounts (account_key, passport_id, platform, id_kind, id, url, relationship_basis, status, observed_at, updated_seq, added_by) VALUES (?, ?, ?, ?, ?, ?, ?, 'active', ?, ?, ?)",
        )
        .bind(
          subject.canonical,
          e.subject,
          subject.platform,
          subject.idKind,
          subject.id,
          typeof p.url === "string" ? p.url : null,
          String(p.relationship_basis),
          String(p.observed_at),
          seq,
          touchedByFromActor(e.actor_kind), // provenance: WHO added it (fold column)
        ),
    ];
  }
  if (e.event_type === "account_added") {
    const key = accountKey(e.payload);
    if (accountsInBatch.has(key)) {
      throw new LedgerError(`account added twice in batch: ${key}`);
    }
    accountsInBatch.add(key);
  }
  // Deliberate limitation: only create + account_adds may reference a
  // passport born in the same batch (the importer's shape). Lifecycle events
  // (claims, verifications, corrections) read committed state only — a full
  // in-batch state overlay would be more machinery to get wrong than the
  // one-extra-batch it saves. Fail closed with a clear instruction instead.
  if (createdInBatch.has(e.subject)) {
    throw new LedgerError(
      `${e.event_type} on a passport created in the same batch — append it in a subsequent batch`,
    );
  }
  return foldEffects(db, e, seq, batchNames);
}

export interface ChainVerification {
  ok: boolean;
  events: number;
  brokenAtSeq?: number;
  reason?: string;
}

/** How many ledger rows verifyChain holds in memory at once. The walk is a
 * strict left fold (each row needs only the previous row's hash), so page size
 * is purely a memory/round-trip trade — 1,000 rows is well under a megabyte
 * against the Worker's 128 MB ceiling and far below D1's result-set caps. */
const VERIFY_CHAIN_PAGE = 1000;

/** Recompute every hash and linkage. Any mismatch names the seq.
 *
 * PAGED, deliberately (v12 known gap #5): this used to read the WHOLE ledger
 * into Worker memory in one `SELECT *` — harmless while the chain was small,
 * and a guaranteed 6-hourly OOM the day it was not, on exactly the cron whose
 * other half (checkpoint export, alarms) must keep running. The chain walk is
 * a left fold over `seq ASC`, so it pages by seq cursor with identical
 * verdict semantics: same ok/brokenAtSeq/reason, and `events` still reports
 * the total row count on every path (an upfront COUNT — one integer, so the
 * failure report reads the same as before). Rows appended BETWEEN pages have
 * seq > the cursor and simply join the walk — the linkage itself proves
 * continuity, so pagination cannot manufacture a false verdict.
 *
 * THE WALK STARTS AT seq > 0 AND MUST ACCOUNT FOR EVERY ROW. The cursor begins
 * at 0, so a forged row planted at `seq <= 0` (the schema's AUTOINCREMENT PK
 * does not itself forbid an explicit non-positive value) would never be read —
 * the single unbounded `SELECT *` this replaced read it first and returned a
 * seq gap. The upfront COUNT closes that: after a clean walk, the number of
 * rows we actually verified (`expectedSeq`, which is also the largest seq seen
 * because the walk requires `seq === expectedSeq` at every step) must equal the
 * table's row count. Any row outside `1..expectedSeq` — a seq<=0 plant, or a
 * duplicate seq the strict-increment walk skipped — makes `total > expectedSeq`
 * and fails closed. This is concurrency-safe: a row appended mid-walk only
 * makes `expectedSeq >= total`, never the reverse, so an honest late append is
 * never mistaken for a plant. */
export async function verifyChain(db: DbLike): Promise<ChainVerification> {
  const { head: _head, ...verdict } = await verifyChainFrom(db);
  return verdict;
}

/** Where a previous walk stopped: the last verified seq and the event_hash it had then. */
export interface ChainPoint {
  seq: number;
  hash: string;
}

/** One row of the walk: seq continuity, linkage, recomputed hash. Returns the failure reason, or null. */
async function checkChainRow(row: EventRow, expectedSeq: number, expectedPrev: string): Promise<string | null> {
  if (row.seq !== expectedSeq) return "seq gap";
  if (row.prev_hash !== expectedPrev) return "prev_hash mismatch";
  let preimage: string;
  try {
    const payload = JSON.parse(row.payload) as Record<string, unknown>;
    const supersedes = JSON.parse(row.supersedes) as string[];
    // I3-0298: the hash commits to the PARSE, so the stored bytes are pinned
    // only by requiring them to be exactly what the append wrote
    // (`stableStringify`, appendEventsOnce's INSERT). Without this a
    // duplicate-key edit — JSON.parse keeps the LAST value, SQLite json_extract
    // the FIRST — changes what every SQL reader sees and still verifies.
    if (stableStringify(payload) !== row.payload) return "payload bytes not canonical";
    if (stableStringify(supersedes) !== row.supersedes) return "supersedes bytes not canonical";
    preimage = eventPreimage({
      event_id: row.event_id,
      event_type: row.event_type as TrustEventEnvelope["event_type"],
      subject: row.subject,
      actor_kind: row.actor_kind as TrustEventEnvelope["actor_kind"],
      actor_id: row.actor_id,
      created_at: row.created_at,
      payload,
      supersedes,
      prev_hash: row.prev_hash,
    });
  } catch (err) {
    if (err instanceof LedgerError) return "preimage field not newline-free";
    // I3-0585: a stored payload/supersedes that is not JSON is a break AT THIS ROW, reported like any other.
    if (err instanceof SyntaxError) return "payload or supersedes not valid JSON";
    throw err;
  }
  return (await sha256Hex(preimage)) === row.event_hash ? null : "event_hash mismatch";
}

/**
 * I3-2157: `verifyChain` with an optional START POINT, returning the verified head.
 *
 * Without `from` it is the full walk above, unchanged. With `from` (a point an
 * earlier walk verified) it first RE-CHECKS that row — present, recomputing to
 * its own event_hash, and that hash still `from.hash` — then walks only
 * `seq > from.seq`, linked to `from.hash`. A tamper at or after the point is
 * caught; a tamper BELOW it is not, which is why the scheduled caller
 * (`checkpoint.ts verifyChainTick`) still runs a rolling full walk (`VERIFY_CHAIN_WINDOW_TICKS`).
 * The full-table COUNT becomes one MIN/MAX rowid lookup: `MIN(seq) <= 0` is
 * the seq<=0 plant check (seq is the INTEGER PRIMARY KEY, so no duplicate).
 *
 * QUEUE #903 (D-0888(2)): `limit` is a HARD CAP on the rows walked after the
 * start point (the re-checked start row and one existence probe are the only
 * other row reads). With it the walk stops at the cap and says whether it
 * reached the end (`done`); the range check's MAX half applies only when
 * `done` (rows past the cap are simply not walked yet), the MIN half always.
 * `head` is then the last row verified, so a caller can resume from it
 * (checkpoint.ts verifyChainTick). Without `limit` nothing here changes.
 */
export async function verifyChainFrom(
  db: DbLike,
  from?: ChainPoint,
  limit?: number,
): Promise<ChainVerification & { head?: ChainPoint; done?: boolean }> {
  const start = from !== undefined && from.seq > 0 ? from : undefined;
  const incremental = start !== undefined;
  const cap = limit === undefined ? undefined : Math.max(1, Math.floor(limit));
  let total: number;
  let lowest = 1;
  if (incremental || cap !== undefined) {
    const r = await db.prepare("SELECT MIN(seq) AS lo, MAX(seq) AS hi FROM trust_events").bind().first<{ lo: number | null; hi: number | null }>();
    total = r?.hi ?? 0;
    lowest = r?.lo ?? 1;
  } else {
    total = (await db.prepare("SELECT COUNT(*) AS n FROM trust_events").bind().first<{ n: number }>())?.n ?? 0;
  }

  let expectedPrev = GENESIS_HASH;
  let expectedSeq = 0;
  let cursor = 0;
  if (start !== undefined) {
    const at = await db.prepare("SELECT * FROM trust_events WHERE seq = ?").bind(start.seq).first<EventRow>();
    if (!at) return { ok: false, events: total, brokenAtSeq: start.seq, reason: "verified row missing" };
    const bad = await checkChainRow(at, start.seq, at.prev_hash);
    if (bad !== null) return { ok: false, events: total, brokenAtSeq: start.seq, reason: bad };
    if (at.event_hash !== start.hash) return { ok: false, events: total, brokenAtSeq: start.seq, reason: "verified row rewritten" };
    expectedPrev = start.hash;
    expectedSeq = start.seq;
    cursor = start.seq;
  }
  let walked = 0;
  let exhausted = false;
  for (;;) {
    const page = cap === undefined ? VERIFY_CHAIN_PAGE : Math.min(VERIFY_CHAIN_PAGE, cap - walked);
    if (page <= 0) break;
    const { results } = await db
      .prepare("SELECT * FROM trust_events WHERE seq > ? ORDER BY seq ASC LIMIT ?")
      .bind(cursor, page)
      .all<EventRow>();
    for (const row of results) {
      expectedSeq += 1;
      const bad = await checkChainRow(row, expectedSeq, expectedPrev);
      if (bad !== null) return { ok: false, events: total, brokenAtSeq: row.seq, reason: bad };
      expectedPrev = row.event_hash;
      cursor = row.seq;
    }
    walked += results.length;
    if (results.length < page) {
      exhausted = true;
      break;
    }
  }
  // Stopped AT the cap: one row probe decides whether anything is left to walk.
  const done = exhausted || !(await db.prepare("SELECT seq FROM trust_events WHERE seq > ? ORDER BY seq ASC LIMIT 1").bind(cursor).first<{ seq: number }>());
  // Every row must have been on the walk. `expectedSeq` is the count of rows
  // verified AND the largest seq seen; if the table holds more rows than that,
  // some row sits outside 1..expectedSeq (a seq<=0 plant, a duplicate seq) and
  // was never checked — fail closed. `> ` not `!==` so an honest append that
  // landed after the COUNT (total < the rows now present) is never a false alarm.
  // Incremental: `total` is MAX(seq) and `lowest` MIN(seq), read before the walk.
  if ((done && total > expectedSeq) || lowest <= 0) {
    return { ok: false, events: total, brokenAtSeq: 0, reason: "rows outside the verified range" };
  }
  return {
    ok: true,
    events: expectedSeq,
    ...(expectedSeq > 0 ? { head: { seq: expectedSeq, hash: expectedPrev } } : {}),
    ...(cap !== undefined ? { done } : {}),
  };
}

/**
 * I3-1096: THE import-rollback door for one passport. "Retract only a still-
 * unclaimed passport" is decided AT COMMIT: an extra statement in the same batch
 * nulls `created_seq` (NOT NULL fails, the whole batch — event included — rolls
 * back) unless `claim_state` is still 'unclaimed'. `created_seq`, not
 * `updated_seq`: the rebind arm owns the `updated_seq` ceiling and
 * `isUnaccountableRebind` keys on it (slice20-rebind.test.ts), and `claim_state`
 * is the accountability ceiling's column — this refusal gets its own. A read-then-append caller
 * could not bind that: a claim landing between its read and the batch was
 * retracted anyway. Returns "claimed" when a claim won, "absent" when there is no
 * `passport_created` to target.
 */
export async function retractImportedPassport(
  db: DbLike,
  actor: string,
  passportId: string,
  now: () => string = () => new Date().toISOString(),
): Promise<"retracted" | "claimed" | "absent"> {
  const ev = await db
    .prepare("SELECT event_id FROM trust_events WHERE event_type = 'passport_created' AND subject = ? ORDER BY seq ASC LIMIT 1")
    .bind(passportId)
    .first<{ event_id: string }>();
  if (!ev) return "absent";
  const stillUnclaimed = db
    .prepare("UPDATE passports SET created_seq = CASE WHEN claim_state = 'unclaimed' THEN created_seq END WHERE passport_id = ?")
    .bind(passportId);
  try {
    await appendEvents(
      db,
      [{ event_type: "correction", subject: passportId, actor_kind: "operator", actor_id: actor, payload: { action: "retract_passport", target: [ev.event_id], reason_code: "erroneous" } }],
      now,
      [],
      [stillUnclaimed],
    );
    return "retracted";
  } catch (e) {
    if (e instanceof Error && /NOT NULL constraint failed: passports\.created_seq/i.test(e.message)) {
      const live = await getPassport(db, passportId);
      if (live && live.claim_state !== "unclaimed") return "claimed";
    }
    throw e;
  }
}

/**
 * Rename a passport: append a `passport_renamed` event pointing at a freshly
 * minted name row, in ONE atomic batch with the fold update.
 *
 * `erasePrevious` is the second half of the decision, and it is opt-in on
 * purpose. A rebrand is a legitimate part of a creator's public history and
 * destroying the old string by default would be an odd thing for a trust product
 * to do silently; a deadname is the opposite, and for that case the old strings
 * must actually go. The superseded refs are read BEFORE the append and deleted by
 * exact id, so the DELETE can never catch the row this rename just inserted (it
 * did not exist when the list was taken) — and it rides the same batch, so a
 * rename that erases either fully happens or does not happen at all.
 *
 * After an erase the superseded `passport_created` / `passport_renamed` events
 * still replay — to "" — but a later rename overwrites that, so the fold lands on
 * the current name in BOTH reducers (Invariant 2 holds; see fold.ts `nameOf`).
 */
/** I18-FXA D18: inserts nothing when the DELETE directly before it removed exactly `n` rows; otherwise violates
 * `audit_log.detail NOT NULL` (the review.ts `changeGateStmt` idiom), aborting the rename's batch. */
function erasedGateStmt(db: DbLike, n: number): DbStatement {
  return db
    .prepare("INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT 'system:erase-gate', 'erase-gate', NULL, '', NULL, 'system' WHERE changes() <> ?")
    .bind(n);
}
const ERASED_GATE_REFUSED_RE = /NOT NULL constraint failed:\s*audit_log\.detail\b/i;

export async function renamePassport(
  db: DbLike,
  actor: string,
  passportId: string,
  displayName: string,
  opts: { erasePrevious?: boolean } = {},
  now: () => string = () => new Date().toISOString(),
  // QUEUE #942(a) D18 (D-0946(1)): the caller's audit rows, built from `erased` (known only here, before the
  // batch) and joined to the rename's own batch after the erase DELETEs; a failed audit insert rolls back both.
  audit?: (ctx: { erased: number }) => DbStatement[],
): Promise<{ event_id: string; erased: number }> {
  assertDisplayName(displayName); // fail before any I/O
  // I63-FX-ERASE: held -> no prior names, erased 0. I64-HOLD H2 (W0 am.6): the held refs are KEPT and owed instead.
  const read = opts.erasePrevious
    ? (
        await db
          .prepare(`SELECT name_ref, ${PASSPORT_LEGAL_HOLD_SQL} AS held FROM passport_names WHERE passport_id = ?`)
          .bind(passportId, passportId)
          .all<{ name_ref: string; held: number }>()
      ).results
    : [];
  const prior = read.filter((r) => r.held !== 1).map((r) => r.name_ref);
  const kept = read.filter((r) => r.held === 1).map((r) => r.name_ref);

  const extra: DbStatement[] = [];
  // I64-HOLD H2 (W0 am.6): the OWED row names the kept refs (ids only); requests.ts sweepRenameNamesOwed deletes exactly
  // those once no hold is open. Guarded on the hold still open and a kept ref still there, and gated like each chunk below:
  // a hold released (or the refs taken) since the read inserts nothing, the gate aborts the batch, and the retry erases.
  if (kept.length > 0) {
    extra.push(
      db
        .prepare(
          `INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT 'system:rename', 'rename:names_erase_owed', ?, ?, ?, 'system'
            WHERE ${PASSPORT_LEGAL_HOLD_SQL} AND EXISTS (SELECT 1 FROM passport_names n WHERE n.passport_id = ? AND n.name_ref IN (SELECT value FROM json_each(?)))`,
        )
        .bind(JSON.stringify(kept), now(), passportId, passportId, passportId, JSON.stringify(kept)),
      erasedGateStmt(db, 1),
    );
  }
  // CHUNKED: `prior` comes from an unbounded SELECT, so a passport renamed more
  // than ~90 times crossed D1's 100-parameter ceiling and failed the rename
  // (audit v10). Each chunk is its own statement in the SAME atomic batch, so
  // the erase is still all-or-nothing with the appended event.
  //
  // I18-FXA D18 (RGA-D18 count limb, D-0949(1)): each chunk's DELETE is followed DIRECTLY by a gate that aborts
  // the batch unless that DELETE removed every ref it named (`changes()`), so `erased` — returned and handed to
  // `audit` — is what THIS batch deleted: a concurrent erase that took the refs first fails this rename (409)
  // instead of both reporting the same rows.
  //
  // I62-FX-EVENTS (I62-RG-W1-05): BEFORE each chunk's passport_names DELETE (never between it and its gate), that
  // chunk's brand_index name entries (key `<passport_id>|<name>`), read by the chunk's own name_refs so the new name's
  // row is not read; kept: the new name's key, a handle entry of the passport, and every entry while an open legal
  // hold names the passport (sweepBrandIndex takes those after release). 6 binds besides the chunk.
  for (const chunk of inChunks(prior, 6)) {
    const holes = chunk.map(() => "?").join(",");
    extra.push(
      db
        .prepare(
          `DELETE FROM brand_index
            WHERE kind = 'name'
              AND key IN (SELECT ? || '|' || n.display_name FROM passport_names n WHERE n.name_ref IN (${holes}))
              AND key <> ? || '|' || ?
              AND NOT EXISTS (SELECT 1 FROM passport_accounts h WHERE h.passport_id = ? AND h.id_kind = 'handle' AND ? || '|' || h.id = brand_index.key)
              AND NOT ${PASSPORT_LEGAL_HOLD_SQL}`,
        )
        .bind(passportId, ...chunk, passportId, displayName, passportId, passportId, passportId),
    );
    extra.push(db.prepare(`DELETE FROM passport_names WHERE name_ref IN (${holes}) AND NOT ${PASSPORT_LEGAL_HOLD_SQL}`).bind(...chunk, passportId)); // I63-FX-ERASE: a hold opened since the SELECT -> 0 rows -> the gate aborts (retry)
    extra.push(erasedGateStmt(db, chunk.length));
  }
  if (audit) extra.push(...audit({ erased: prior.length }));

  let event: TrustEventEnvelope | undefined;
  try {
    [event] = await appendEvents(
      db,
      [{ event_type: "passport_renamed", subject: passportId, actor_kind: "operator", actor_id: actor, payload: { display_name: displayName } }],
      now,
      [],
      extra,
    );
  } catch (e) {
    if (e instanceof Error && ERASED_GATE_REFUSED_RE.test(e.message)) {
      throw new LedgerError("the passport's previous names changed while this rename ran — retry the rename");
    }
    throw e;
  }
  return { event_id: event!.event_id, erased: prior.length };
}

/**
 * The erase pair as STATEMENTS, for a door that must erase in the SAME batch as
 * the act that grants it (I16 #601a, D-0920: the two final granted-removal doors
 * in removal.ts). Both statements carry the caller's `guard` — an SQL boolean
 * expression evaluated inside the batch, e.g. "this request is the one that
 * flipped" — so a lost race erases nothing. The pair is the one
 * `erasePassportNames` runs; `updated_seq` is not touched, for the reason given
 * there.
 *
 * I62-EVENTS P4 (W0-11, 22A-07): FIRST, before the pair, the brand_index copy of every name being erased — kind
 * 'name', key `<passport_id>|<name>` (brand-index.ts buildBrandIndex), the names read inside the statement from
 * passport_names and passports.display_name, so it must run before the pair empties them. Under the caller's guard,
 * and kept while an open legal hold names the passport (sweepBrandIndex takes it after release). A row whose key is
 * also one of the passport's handle entries (`<passport_id>|<handle id>`) is that handle's entry and stays, as do
 * domain entries.
 */
export function erasePassportNamesStatements(db: DbLike, passportId: string, guard: string, guardBinds: readonly unknown[]): DbStatement[] {
  return [
    db
      .prepare(
        `DELETE FROM brand_index
          WHERE kind = 'name'
            AND key IN (SELECT ? || '|' || n.display_name FROM passport_names n WHERE n.passport_id = ?
                        UNION SELECT ? || '|' || p.display_name FROM passports p WHERE p.passport_id = ?)
            AND NOT EXISTS (SELECT 1 FROM passport_accounts h WHERE h.passport_id = ? AND h.id_kind = 'handle' AND ? || '|' || h.id = brand_index.key)
            AND (${guard})
            AND NOT ${PASSPORT_LEGAL_HOLD_SQL}`,
      )
      .bind(passportId, passportId, passportId, passportId, passportId, passportId, ...guardBinds, passportId),
    // I64-HOLD H1 (W0-7, Jason P19 (a) J5): an open passport hold keeps the names AND the fold name, the pair asking it together.
    db.prepare(`DELETE FROM passport_names WHERE passport_id = ? AND (${guard}) AND NOT ${PASSPORT_LEGAL_HOLD_SQL}`).bind(passportId, ...guardBinds, passportId),
    db.prepare(`UPDATE passports SET display_name = '', display_name_skeleton = NULL WHERE passport_id = ? AND display_name <> '' AND (${guard}) AND NOT ${PASSPORT_LEGAL_HOLD_SQL}`).bind(passportId, ...guardBinds, passportId),
  ];
}

/** The standalone erase's guard: no open hold names the passport (PASSPORT_LEGAL_HOLD_SQL). Binds the passport id once. */
const ERASE_NAMES_UNHELD_GUARD = `NOT ${PASSPORT_LEGAL_HOLD_SQL}`;

/**
 * ERASE every display-name string a passport has ever carried, without renaming
 * it — the standalone erasure right (a takedown, a legal demand, a creator who
 * wants the name gone and nothing put in its place).
 *
 * The fold column is set to "" in the SAME batch as the deletes. That second
 * statement is not cosmetic and must never be dropped: `passports.display_name`
 * is a projection, and the replay reducer will resolve every one of this
 * passport's name refs to "" the instant the rows are gone. Leaving the live
 * column holding the erased string would put the two reducers into permanent
 * disagreement — Invariant 2 — and `verifyFold` would report drift forever.
 *
 * `updated_seq` is deliberately NOT touched: erasure is not a ledger event, and
 * the replay derives updated_seq purely from the log. Bumping it here would
 * itself create the drift this is avoiding.
 *
 * I63-ERASE (I62-FX-EVENTS-03, D-1538 ruling 2, W0-12): ONE batch, `erasePassportNamesStatements` (brand_index first,
 * then the pair) plus the skeleton clear the pair's `display_name <> ''` UPDATE skips once the names are already gone
 * (i20-w1d-name-skeleton.test.ts), every statement under ERASE_NAMES_UNHELD_GUARD: while an
 * open hold names the passport nothing is erased and `erased` is 0. `erased` is the passport_names DELETE's own row
 * count, D1's `meta.changes` on batch element 1 (passport_names is not in db.ts INFLATING_WRITES).
 */
export async function erasePassportNames(db: DbLike, passportId: string): Promise<{ erased: number }> {
  const res = (await db.batch([
    ...erasePassportNamesStatements(db, passportId, ERASE_NAMES_UNHELD_GUARD, [passportId]),
    db
      .prepare(`UPDATE passports SET display_name_skeleton = NULL WHERE passport_id = ? AND NOT ${PASSPORT_LEGAL_HOLD_SQL}`) // = ERASE_NAMES_UNHELD_GUARD, spelled out for the hold census (I64-HOLD)
      .bind(passportId, passportId),
  ])) as { meta: { changes: number } }[];
  return { erased: res[1]!.meta.changes };
}
