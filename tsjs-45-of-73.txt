/** The creator code-revoke door's 429 sentence: the creator console's own words (apps/creator-console views.ts
 * codeRevokeOutcome), sent so the console can print the server's sentence (I25; I34-FX2-CLIENTS). */
export const CODE_REVOKE_LIMIT_SENTENCE = "You've hit the hour's revoke limit for this listing. Nothing changed — try again within the hour.";

/**
 * ONE-TIME CLAIM CREDENTIALS — the agency mints them against a release it has a
 * sales grant on, a fan redeems one for a durable acquisition.
 *
 * BOUNDARY. TWO mint doors — `POST /account/releases/:id/credentials` (the
 * creator's own, restored by Y-01/D-0560) and `POST
 * /agency/releases/:id/credentials` (the sales-scoped mirror) — their reconcile
 * reads, `POST /account/redeem`, and `GET /account/acquisitions`. The raw code
 * generator lives here too: it is shown once and only its hash is stored.
 *
 * BOTH mint doors are gated on the CREATOR'S standing, and on the agency door
 * that is the creator the GRANT resolves to, not the seat — an agency acts on
 * behalf of a creator and cannot hold more authority than the creator it
 * represents. That verdict is written down in `MINT_VERDICT` and executed by the
 * unverified/delisted sweeps in `tests/invariants/route-authorization.test.ts`.
 */
/**
 * I3-0764 (D-0833(4)) — the fallback for a throw no domain class claimed: logged,
 * and answered 500 `{error:"internal"}`, never a 400 blaming the caller.
 */
function routeFault(c: Context<{ Bindings: Env }>, route: string, e: unknown): Response {
  logError({ cid: crypto.randomUUID(), hop: "http", route, msg: "unexpected route fault", err: e instanceof Error ? e.name : typeof e });
  return c.json({ error: "internal" }, 500);
}

export function makeCredentialsRoutes(): Hono<{ Bindings: Env }> {

  const app = new Hono<{ Bindings: Env }>();

  /** Failed (non-ok) `/account/redeem` outcomes tolerated per fan account per
   * day. Generous for honest typos; a bound on `redemption_attempts` rows and on
   * the attacker-writable half of `redemptionAbuseSignal`. Successful
   * redemptions never consume from it. */
  const DEAD_REDEMPTIONS_PER_ACCOUNT_DAY = 50;

  /**
   * Credential-MINT calls one agency seat may make per hour.
   *
   * The door was authorized and unmetered, which are different things. One call
   * mints up to `count` one-time codes (capped at 1000 per call inside
   * `issueCredentials`) and every code is a row plus a hash; nothing bounded how
   * many CALLS a seat made, so a sales-capable seat — or a stolen agency session
   * — could write unbounded rows against a creator's release and flood the
   * reconcile view and the redeem rail with codes the creator never agreed to.
   * The cap is per (agency, release), so a busy agency working several releases
   * is not throttled by its own activity on another, and the bucket is the same
   * shared windowed counter every other cap in the codebase uses.
   */
  const MINTS_PER_AGENCY_RELEASE_HOUR = 20;

  /**
   * Credential-MINT calls one CREATOR may make per hour, per release.
   *
   * D-0597. The agency path is bounded by the grant it acts under; a
   * creator-direct path has no grant, so the only bound left is the route's.
   * The three bounds inside `issueCredentials` — count 1..MAX_ISSUE, the
   * withdrawn refusal, the atomic `codes_issued` reservation — bound what ONE
   * call does and say nothing about how many calls a session makes: at
   * `MAX_ISSUE` this cap is 20,000 codes an hour against one release, which is
   * the number being chosen here.
   *
   * 20 is deliberately conservative and deliberately a named constant a later
   * ruling can raise. Being wrong low is a support ticket; being wrong high is
   * an abuse channel — a stolen creator session that can mint without bound
   * writes rows plus hashes against the creator's own release and floods their
   * reconcile view and the redeem rail with codes they never agreed to.
   *
   * IT IS A SEPARATE CONSTANT FROM THE AGENCY FIGURE ABOVE, at the same value
   * today, because the two populations are not the same one: that figure was
   * chosen for a busy agency working several releases at once, which is not a
   * creator's usage profile, and the two have to be able to move without each
   * other. The bucket is per (creator, release) for the same reason the agency
   * bucket is per (agency, release), and uses the same shared windowed counter
   * every other cap in the codebase uses. The key cannot collide with the
   * agency door's: a passport is `psp_<16 hex>` and an agency is `agc_<16 hex>`
   * (packages/schema/src/index.ts:630).
   */
  const MINTS_PER_CREATOR_RELEASE_HOUR = 20;

  /**
   * Credential-REVOKE calls one CREATOR may make per hour, per release.
   *
   * A SEPARATE CONSTANT FROM THE TWO MINT FIGURES ABOVE, at the same value
   * today, for the reason `MINTS_PER_CREATOR_RELEASE_HOUR` already states about
   * its own twin: the populations are not the same one and the two have to be
   * able to move without each other. The bucket is per (creator, release) and
   * uses the same shared windowed counter every other cap in this file uses.
   *
   * IT CANNOT COLLIDE WITH A MINT BUCKET: the prefix is `credrevoke:`, not
   * `credmint:`, so exhausting one never refuses the other. That matters in the
   * direction this door is destructive — a creator who has just discovered a
   * leak must not find their revokes spent by their own minting.
   */
  const REVOKES_PER_CREATOR_RELEASE_HOUR = 20;

  /**
   * I38-AGCRED: ONE-SEAT revoke calls (`{seat}` on the creator's revoke door) one CREATOR may make per hour on ONE
   * release (I40-CLAIMS IA-02: metered per listing, so CODE_REVOKE_LIMIT_SENTENCE's "for this listing" is true for this
   * arm too). Its own bucket (`credrevokeseat:<creator>:<release bucket>`), so a creator cleaning up a removed seat
   * neither spends nor is refused by the 20-per-release single-code cap above; one call reaches every code that seat
   * minted on the release.
   */
  const REVOKE_SEAT_CALLS_PER_CREATOR_HOUR = 10;

  /**
   * THE SHAPE, BEFORE THE ID BECOMES A BUCKET KEY (W3-09 row 5, Y-01).
   *
   * Both mint doors meter on a bucket whose key interpolates `:id`, and
   * `RELEASE_ID_RE` was not applied until `credentials.ts:117` — INSIDE the
   * primitive, which the creator door reaches only AFTER its meter has already
   * written a row. Measured: 200 calls with 200 distinct junk ids were all
   * admitted, each consumed a slot of a bucket nothing else would ever touch
   * again, and each left a permanent `rate_limits` row (the table gives rows
   * back only via `pruneRateLimits`' 2-week floor, `ratelimit.ts`). The cap was
   * not raised or lowered: it was never reached, because the caller chose the
   * bucket.
   *
   * ONE PREDICATE, INTERPOLATED AT EVERY DOOR (FAMILY-004 P1), not re-typed at
   * each: the population is "doors that write a `rate_limits` row keyed on a
   * release id", and it has exactly two members, `:id` on
   * `POST /account/releases/:id/credentials` and on
   * `POST /agency/releases/:id/credentials`. The two `…/credentials/summary`
   * GETs meter nothing and already fold a malformed id to the same 404 an
   * unknown one gets, inside `credentialSummary` (`credentials.ts:376`).
   *
   * IT IS NOT AN ORACLE. A malformed id is not a release id under ANY
   * suppression state, so answering on its SHAPE discloses nothing about who
   * exists or who is delisted — the byte-identity the fold protects is between
   * two WELL-FORMED ids (`rel_<16 hex>`), which still reach the meter and the
   * resolve exactly as before.
   */
  const badReleaseId = (id: string): boolean => !RELEASE_ID_RE.test(id);

  /**
   * The bucket a MALFORMED release id is metered into. `rel_<16 hex>` can never
   * equal this string, so it cannot collide with a real release's bucket and a
   * junk call cannot deny service to a real one.
   */
  const INVALID_RELEASE_BUCKET = "__invalid__";


  // --- Claim credentials + acquisitions ---
  // A creator issues one-time credentials for a release they own (raw shown once,
  // only the hash is stored); a fan redeems one for a durable acquisition.
  const B32 = "ABCDEFGHIJKLMNOPQRSTUVWXYZ234567";
  /** RFC4648-style base32 of 15 random bytes = 120 bits → exactly 24 chars (no
   * padding, full 120-bit entropy). Matches RAW_RE ([A-Z2-7]{16,40}). */
  function mintRawCredential(): string {
    const bytes = crypto.getRandomValues(new Uint8Array(15));
    let bits = 0;
    let val = 0;
    let out = "";
    for (const b of bytes) {
      val = (val << 8) | b;
      bits += 8;
      while (bits >= 5) {
        bits -= 5;
        out += B32[(val >>> bits) & 31];
      }
    }
    return out; // 24 chars
  }

  // THE CREATOR'S OWN MINT DOOR (Y-01, D-0560). The twin v13 retired, restored:
  // a creator issues one-time codes against a release they own without first
  // standing up an agency and granting it `manage_sales`.
  //
  // IT REUSES THE AGENCY DOOR'S PRIMITIVE UNCHANGED. `issueCredentials` takes a
  // `VerifiedCreator` witness and re-checks the release belongs to THAT passport
  // (credentials.ts:134), so passing the caller's own witness IS the ownership
  // check; its `issuedActor` default is `"creator"` (credentials.ts:113), which
  // is what keeps these rows out of an agency's issuer-scoped count
  // (credentials.ts:414 — `issuerScope` is passed on the agency summary only,
  // :275). The creator's own summary read passes no `issuerScope` and so keeps
  // the union across every issuer.
  //
  // THE ONE THING THE PRIMITIVE CANNOT CARRY IS THE CALL CAP, so it is here.
  //
  // METERED BEFORE THE RELEASE IS TOUCHED, deliberately the OPPOSITE order from
  // the agency door below. That bucket is keyed on a release id a grantless seat
  // has no business knowing about, so it must not be consumable — or observable
  // — before the grant check. THIS bucket is keyed on the CALLER'S OWN passport:
  // the only session that can consume it is the one it bounds, and a 429
  // discloses nothing but the caller's own call count against an id they chose.
  // So it runs first, where it also bounds the work the door does — the release
  // read, the standing read and up to MAX_ISSUE hashes — rather than after it.
  app.post("/account/releases/:id/credentials", anonBodyLimit, async (c) => {
    const creator = await creatorId(c);
    if (!creator) return c.json({ error: "unauthorized" }, 401);
    const db = d1(c.env.TRUST_DB);
    const releaseId = c.req.param("id");
    // THE SHAPE IS CHECKED BEFORE THE ID BECOMES A KEY, AND THE ANSWER IS
    // UNCHANGED — the fold, not a refusal, and the difference is the whole of
    // W3-09-FX1's second pass at this row.
    //
    // A 400 here refuses EARLIER than the standing gate, and
    // `tests/invariants/route-authorization.test.ts:1914` sweeps every minting
    // route for the standing refusal by driving `:id` with one shared
    // placeholder (`PARAM_VALUES.id`, a `psp_` id) against a stubbed db. A
    // route that answers on its id's SHAPE first makes that sweep's standing
    // assertion unreachable — measured: both mint doors answered
    // `400 bad release id` where it expects `403 verified creators only`. The
    // badge promise that sweep defends is not this row's to move, so nothing is
    // refused earlier than it was.
    //
    // FOLDING IS THE ACTUAL FIX ANYWAY. The defect was never the answer a junk
    // id got — `credentials.ts:117` already answered it — it was that the
    // CALLER CHOSE THE BUCKET: 200 distinct junk ids were 200 distinct buckets,
    // so 200 calls consumed one slot each of a counter nothing would ever read
    // again, and each left a permanent `rate_limits` row (the table gives rows
    // back only through `pruneRateLimits`' two-week floor). Every malformed id
    // now meters into ONE bucket per creator, so the 21st junk call in the hour
    // is the 429 the cap always meant, the table takes one row instead of 200,
    // and the cap's own value is untouched.
    const bucketId = badReleaseId(releaseId) ? INVALID_RELEASE_BUCKET : releaseId;
    if (!(await consumeRateLimit(db, `credmint:${creator}:${bucketId}`, Date.now(), MINTS_PER_CREATOR_RELEASE_HOUR, 60 * 60 * 1000))) {
      return c.json(refusal("rate limit exceeded"), 429);
    }
    const raw = await c.req.text();
    if (raw.length > 4 * 1024) return c.json({ error: "too_large" }, 413);
    const body = ((): { count?: unknown } | null => {
      try {
        return JSON.parse(raw) as { count?: unknown };
      } catch {
        return null;
      }
    })();
    const count = body && typeof body.count === "number" ? body.count : 0;
    try {
      // OWNERSHIP, EXISTENCE AND THE SUPPRESSION FOLD ARE ONE STATEMENT INSIDE
      // THE PRIMITIVE (credentials.ts:126-:138), taken against the caller's own
      // passport — so there is no pre-resolve here to keep in step with it, and
      // no second answer for a probe to compare. The witness is the caller's
      // own standing, not a resolved owner's: on this door they are the same
      // passport, which is why the door needs no `issuedActor` override.
      const r = await issueCredentials(db, await requireVerifiedCreator(db, creator), releaseId, count, mintRawCredential);
      const links = redeemLinks(c.env, r.credentials);
      return c.json({ ok: true, ...r, ...(links ? { redeem_links: links } : {}) });
    } catch (e) {
      if (e instanceof VerifiedCreatorError) return c.json({ error: e.message }, e.status);
      if (e instanceof CredentialError) return c.json({ error: e.message }, e.status);
      return routeFault(c, "POST /account/releases/:id/credentials", e);
    }
  });

  /**
   * THE CREATOR'S OWN REVOKE DOOR (QUEUE #436; D-0618 refused RETIRE because it
   * removes the creator's only path to reclaim a stranded slot — this is that
   * path). `revokeCredential` (credentials.ts:288) shipped built and tested with
   * DOOR COUNT ZERO: nothing outside its own tests named it, so a leaked code
   * could not be killed by the person it was leaked from. This is its ONE door,
   * and the door count for this primitive is now ONE (FAMILY-019: stated in
   * those words). There is deliberately no agency twin — see below.
   *
   * KEYED ON THE RAW CREDENTIAL, NOT ON ITS HASH, AND THAT IS THE WHOLE
   * DECISION. The primitive takes a `credential_hash`; `claim_credentials`
   * (migrations/0007_releases_credentials.sql:70) stores ONLY the hash and
   * carries no public id, no label, no prefix and no last-N, and `issued_at` is
   * one `ts` for an entire mint call (credentials.ts:141), so it does not single
   * out a code either. A creator therefore NEVER sees a hash and there is no
   * surface that could ever show them one. The one per-credential thing a
   * creator can physically hold is the RAW code the mint response returned, so
   * that is what this door takes, and the hashing is done HERE by `credHash` —
   * the module's single exported hasher (credentials.ts:75), never a second
   * hashing path, which is the same rule its own docblock exists to state.
   *
   * THE SECOND FORM IS RETIRED (QUEUE #484, D-0712/D-0848(1)). A bulk
   * `{"all_unredeemed": true}` arm (`revokeUnredeemedCredentials`, D-0775) shipped
   * here and is removed: it killed codes already handed to real fans to recover
   * from a creator's lost page load. The creator who lost the mint response is
   * covered instead by `GET .../outstanding` (by public id) + the `public_id`
   * form below. Sending `all_unredeemed` now is refused with the same 400 as
   * sending both `credential` and `public_id`.
   *
   * EVERY REVOCATION IS A LEDGER ROW (D-0762): both forms append
   * `credentials_revoked {release_id, count}` in the same batch as the flip.
   *
   * NO AGENCY TWIN. `revokeCredential`'s statements bind `issued_by =
   * creator.passport_id` (credentials.ts:317, :331, :336) and it takes no
   * `issuerScope`, so an agency door built on it would let one `manage_sales`
   * seat revoke codes the CREATOR issued directly and codes another live grant
   * issued. The mint door partitions those by `issuedActor`/`issuerScope`; this
   * primitive cannot, so the door is not built rather than built wrong.
   *
   * AUTHORIZATION IS THE CALLER'S OWN SESSION, TWICE OVER, AND NEVER THE BODY.
   * `creatorId(c)` is the only source of the passport; it is handed to
   * `requireVerifiedCreator`, and the witness that comes back is what
   * `revokeCredential` re-checks the release against (403 `not your release`,
   * or the suppression fold's 404). An id swap therefore cannot work in either
   * direction: another creator's release id is refused by the release check,
   * and another creator's RAW code against the caller's OWN release finds no
   * `active` row for (hash, release, issuer) and revokes nothing.
   *
   * `revoked: false` IS A 200, NOT A 404. The primitive returns false — never
   * throws — for a code that is unknown, already redeemed, already revoked, or
   * lost a race, precisely so a caller cannot read more out of the exception
   * shape than the boolean says. Mapping false onto a 404 here would rebuild
   * that oracle one layer up, so it is not done.
   *
   * METERED BEFORE THE RELEASE IS TOUCHED, and the bucket is keyed on the
   * CALLER'S OWN passport with the same `badReleaseId` fold the creator mint
   * door uses: only the session it bounds can consume it, so the meter cannot
   * be turned into a denial of service against the creator this door protects,
   * and a malformed id meters into ONE bucket per creator instead of letting
   * the caller choose an unbounded number of them (W3-09 row 5).
   *
   * THE SHAPE FOLD, NOT A 400, FOR THE SAME REASON THE MINT DOOR GIVES: a route
   * that answers on its id's shape first refuses EARLIER than the standing gate
   * and makes `tests/invariants/route-authorization.test.ts`'s standing sweep
   * unreachable on this door. Nothing here is refused earlier than it was.
   */
  app.post("/account/releases/:id/credentials/revoke", anonBodyLimit, async (c) => {
    const creator = await creatorId(c);
    if (!creator) return c.json({ error: "unauthorized" }, 401);
    const db = d1(c.env.TRUST_DB);
    const releaseId = c.req.param("id");
    const bucketId = badReleaseId(releaseId) ? INVALID_RELEASE_BUCKET : releaseId;
    // The body is read before the meter so the meter can tell the one-seat arm (I38-AGCRED) from every other body; the
    // 413 still answers after the meter, as before, and every body that is not exactly `{seat}` spends `credrevoke:`.
    const raw = await c.req.text();
    const body = ((): { credential?: unknown; all_unredeemed?: unknown; public_id?: unknown; seat?: unknown } | null => {
      try {
        const v = JSON.parse(raw) as unknown;
        return v !== null && typeof v === "object" ? (v as { credential?: unknown; all_unredeemed?: unknown; public_id?: unknown; seat?: unknown }) : null;
      } catch {
        return null;
      }
    })();
    const seatArm = body !== null && body.seat !== undefined && Object.keys(body).length === 1;
    if (
      seatArm
        ? !(await consumeRateLimit(db, `credrevokeseat:${creator}:${bucketId}`, Date.now(), REVOKE_SEAT_CALLS_PER_CREATOR_HOUR, 60 * 60 * 1000))
        : !(await consumeRateLimit(db, `credrevoke:${creator}:${bucketId}`, Date.now(), REVOKES_PER_CREATOR_RELEASE_HOUR, 60 * 60 * 1000))
    ) {
      // I25 / I34-W2-TC hand-off (I34-FX2-CLIENTS): the cap answers the creator console's own sentence (capRefusal).
      return c.json(capRefusal(CODE_REVOKE_LIMIT_SENTENCE), 429);
    }
    if (raw.length > 4 * 1024) return c.json({ error: "too_large" }, 413);
    // NO SHAPE CHECK ON THE CREDENTIAL, deliberately. A string that is not a
    // credential hashes to a hash that matches no row, which the primitive
    // already answers `false` — the same answer an unknown code gets. A 400 on
    // shape would be a THIRD answer for a prober to compare against those two,
    // and it would buy nothing: the work it would save is bounded already, by
    // the 4 KiB body cap above and by the meter above that.
    const credential = body && typeof body.credential === "string" ? body.credential : "";
    try {
      // D-0838(5) / D-0690: revoke is open to a de-verified or suppressed owner,
      // so the session's passport is the revoker; ownership is the primitive's
      // release fold (credentials.ts requireRevocableRelease).
      const verified = creator;
      // QUEUE #484 (D-0712/D-0848(1)): `all_unredeemed` is retired — the bulk
      // arm it selected is gone. Refused before either remaining form runs, and
      // before any release is touched, so it never revokes anything.
      if (body?.all_unredeemed !== undefined) return c.json({ error: "send one of credential or public_id" }, 400);
      // I38-AGCRED: `{seat}` alone revokes every outstanding code that one agency seat minted on this release
      // (credentials.ts revokeSeatCredentials); combined with any other field it is refused before anything runs.
      if (body?.seat !== undefined) {
        if (!seatArm) return c.json({ error: "send one of credential, public_id or seat" }, 400);
        const r = await revokeSeatCredentials(db, verified, releaseId, body.seat);
        return c.json({ ok: true, revoked_count: r.revoked, agency_id: r.agency_id });
      }
      // I3-1259 (I4-13-R4): revoke ONE code by the public id the creator can
      // hold (GET …/credentials/outstanding lists them). Mutually exclusive with
      // `credential`. Same no-shape-check rule as `credential`: a string that
      // is not a public id revokes nothing and answers `false`.
      if (body?.public_id !== undefined) {
        if (body.credential !== undefined) return c.json({ error: "send one of credential or public_id" }, 400);
        const publicId = typeof body.public_id === "string" ? body.public_id : "";
        return c.json({ ok: true, revoked: await revokeCredentialByPublicId(db, verified, releaseId, publicId) });
      }
      const revoked = await revokeCredential(db, verified, releaseId, await credHash(credential));
      return c.json({ ok: true, revoked });
    } catch (e) {
      if (e instanceof VerifiedCreatorError) return c.json({ error: e.message }, e.status);
      if (e instanceof CredentialError) return c.json({ error: e.message }, e.status);
      if (e instanceof LedgerContentionError) return contentionResponse(c);
      // I23-L05 (I23-L01 hand-off): the ledger refusing the revocation event is a domain refusal (409), never a 500 —
      // the same mapping on BOTH revoke doors (the agency twin below).
      if (e instanceof LedgerError) return c.json({ error: REVOKE_LEDGER_REFUSAL }, 409);
      return routeFault(c, "POST /account/releases/:id/credentials/revoke", e);
    }
  });

  // THE CREATOR'S OWN DOOR ONTO THEIR RELEASE'S CODE TRACKER (S9-1). Minting
  // left, but the tracker had to come back as a READ: the agency door below is
  // gated on a LIVE sales grant, so the moment a creator revokes the grant the
  // codes that agency already issued keep redeeming (by design — nothing already
  // done is undone) while NOBODY can see what is outstanding against the
  // release. The creator can always read their own; the agency reads only while
  // granted.
  //
  // NOT-MINE AND NO-SUCH ARE THE SAME ANSWER. `credentialSummary` returns null
  // for an unknown id AND for a suppressed owner's release seen by anyone who is
  // not that owner; the ownership check below returns the IDENTICAL body and
  // status, so this door cannot be used to tell an existing release from one
  // that never was, nor a delisted creator from an absent one (INV-003). The
  // owner keeps the read even while suppressed — `viewer` is the creator.

        // redemption), never on signups — and never on a merely-claimed

   * Slice 24 — the fan's own purchase record, now PAGEABLE.
   *
   * It served a fixed 100 rows and three clients printed "older redemptions
   * aren't listed here" underneath. `listAcquisitions` had taken a `limit` since
   * it was written; nothing ever passed one.
   *
   * THE CURSOR IS A TUPLE, and it is two query params rather than one opaque
   * token because this repo has no opaque cursors and inventing an encoding here
   * would be the only one. `?before=<acquired_at>&before_id=<acquisition_id>` —
   * both halves or neither.
   *
   * A MALFORMED CURSOR IS A 400, which diverges from `auditList`'s precedent of
   * folding a junk cursor back into the un-cursored branch. That precedent is
   * fine for a bare integer, where the worst case is re-serving page 1. It is
   * not fine for a tuple: half a cursor, or a stamp in the wrong shape, yields a
   * page whose boundary is somewhere neither the caller nor this route can name,
   * and the failure mode is MISSING PURCHASES rather than repeated ones. Fail
   * closed (invariant 6) and say which param was wrong.
   */
  app.get("/account/acquisitions", async (c) => {
    const id = await fanId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    // `Number`, then INTEGER. A fractional `?limit=2.5` survives the clamp
    // (`Math.max(1, Math.min(200, 2.5))` is 2.5), reaches `LIMIT ?` as a REAL,
    // and SQLite refuses it with `datatype mismatch` — an unhandled 500 on a
    // read of the caller's own rows. Found by review; the junk-limit test had
    // enumerated `abc`, `-1`, `0` and `` and missed the one that breaks.
    const rawLimit = Number(c.req.query("limit"));
    const limit = Number.isInteger(rawLimit) && rawLimit > 0 ? rawLimit : undefined;
    // AND A LIMIT ABOVE THE MAX IS A 400, not a silent clamp. `has_more` is
    // deliberately not served, so the endpoint's only end-of-list signal is "a
    // short page is the last page" — and that signal LIES the moment the server
    // returns fewer rows than the client asked for because it clamped. A client
    // asking for 500, getting 200, and concluding it has everything loses 300
    // purchases: the exact truncation this slice removes, one query param over.
    // Fail closed and say the number, same stance as `bad_cursor` below.
    if (limit !== undefined && limit > ACQUISITIONS_PAGE_MAX) {
      return c.json({ error: "bad_limit", detail: `limit must be at most ${ACQUISITIONS_PAGE_MAX}` }, 400);
    }
    const before = c.req.query("before");
    const beforeId = c.req.query("before_id");
    // Both halves or neither. One half alone is a client bug, and serving it as
    // an un-cursored first page would loop that client forever.
    if ((before === undefined) !== (beforeId === undefined)) {
      return c.json({ error: "bad_cursor", detail: "before and before_id must be sent together" }, 400);
    }
    const cursor = before !== undefined && beforeId !== undefined ? { acquired_at: before, acquisition_id: beforeId } : undefined;
    if (cursor && !isAcquisitionCursor(cursor)) return c.json({ error: "bad_cursor", detail: "before must be an ISO timestamp and before_id an acq_ id, both from a row this endpoint served" }, 400);
    const acquisitions = await listAcquisitions(d1(c.env.TRUST_DB), id, {
      ...(limit !== undefined ? { limit } : {}),
      ...(cursor ? { before: cursor } : {}),
    });
    // `has_more` is NOT served, deliberately: knowing it costs a second read or
    // an over-fetch, and the client already has the honest signal — a short page
    // is the last page. The clients below use exactly that.
    return c.json({ acquisitions });
  });

  // S-CW2: the fan's owned card instances (the Creator War property layer —
  // copies/editions/bound, distinct from `card_collections`' public signal).
  // Session-gated, own-scoped, living instances only, bounded list.
  app.get("/account/instances", async (c) => {
    const id = await fanId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    return c.json({ instances: await listInstances(d1(c.env.TRUST_DB), id) });
  });

  // PS-238 (docket #36 (d), S-022 ODDS): the fan's OWN pack spend history.
  // Session-gated and own-scoped — `packSpendHistory` binds the account id and
  // throws on a malformed one rather than scanning, so this route cannot become
  // a way to read someone else's spending. Same shape as `/account/instances`.
  app.get("/account/war/packs/spend", async (c) => {
    const id = await fanId(c);
    if (!id) return c.json({ error: "unauthorized" }, 401);
    return c.json({ history: await packSpendHistory(d1(c.env.TRUST_DB), id) });
  });

  return app;
}

/**
 * One-time claim credentials + acquisitions. A creator issues credentials for a
 * release they own; a buyer redeems one exactly once (marked DEAD atomically),
 * which mints a durable acquisition record. The raw credential is never stored
 * (only its sha256) and is never the long-term proof — the acquisition is.
 *
 * Credentials are SEPARATE from media hashes by construction: a credential is
 * high-entropy random, unrelated to any content hash, so a public content hash
 * can never function as a reusable purchase proof.
 *
 * Dead-token reuse is recorded as a BOUNDED behavioral signal (never a public
 * accusation): repetition, the same dead credential across many accounts, or
 * many collisions from one creator are the meaningful patterns.
 */

export class CredentialError extends Error {
  override name = "CredentialError";
  constructor(message: string, public readonly status: 400 | 401 | 403 | 404 | 409 = 400) {
    super(message);
  }
}

const MAX_ISSUE = 1000;
// Raw credential: base32-ish, high entropy. Redemption presents this; we hash it.
const RAW_RE = /^[A-Z2-7]{16,40}$/;

function hex(b: Uint8Array): string {
  let s = "";
  for (const x of b) s += x.toString(16).padStart(2, "0");
  return s;
}
/**
 * The NOT NULL violation the M32 reservation ceiling raises when it refuses.
 *
 * Pinned to the exact column, like `standing.ts GATE_REFUSED_RE`,
 * `claims.ts CLAIM_GATE_REFUSED_RE` and `warnings.ts SECOND_LOOK_REFUSED_RE` —
 * the other places in this repo using a CASE-with-no-ELSE as a gate. The looser
 * pair (`/NOT NULL/` AND `/codes_issued/`) would be the same regex here, but the
 * anchored form is what stops a future column of the same name in another table
 * from being read as this one.
 *
 * `test`, not equality: better-sqlite3 says
 * `NOT NULL constraint failed: releases.codes_issued`; D1 wraps the same
 * sentence in `D1_ERROR: … : SQLITE_CONSTRAINT`.
 *
 * EXPORTED FOR ITS TEST, deliberately. A behavioural test can only reach the
 * TRUE arm — there is no way from outside to make an unrelated NOT NULL fault
 * happen inside this batch — so loosening this to `/NOT NULL/` survived every
 * behavioural mutation. What the regex must not do is only assertable against
 * the regex. Same reasoning as `claims.ts isUniformClaimRefusal`.
 */
const RESERVATION_REFUSED_RE = /NOT NULL constraint failed:\s*releases\.codes_issued\b/i;

/**
 * I3-0363 — THE AGENCY'S AUTHORITY, RE-ASKED AT THE RESERVATION.
 *
 * The agency door (`POST /agency/releases/:id/credentials`) reads the seat's
 * role, the agency's status and the `manage_sales` grant BEFORE the hashing
 * loop; the batch that makes the codes real used to re-ask only the creator's
 * standing. A grant revoked, an agency suspended or a seat demoted in that
 * window still minted. This statement rides the reservation batch and asks all
 * three again: a NOT-NULL ceiling on `releases.updated_at` (NOT NULL,
 * migrations/0007:21), its OWN column so the refusal is told apart from the
 * edition ceiling on `codes_issued`. The grant term is the question
 * `activeGrantScope` answers (agencies.ts: the latest `agency_authorized` /
 * `agency_revoked` event for the pair wins) narrowed to the sales scopes
 * `agencyCanManageSales` accepts (agencies.ts SALES_SCOPES, imported — one set,
 * so the two cannot drift; `i12-money-agency-mint-window.test.ts` checks the
 * two agree for every scope). The acting member's role, the agency session
 * epoch and a seat session's seat are agencies.ts `agencyActorLiveSql` itself,
 * imported (agencies.ts is outside the paywall verdict closure;
 * tests/invariants/paywall.test.ts). The sales-capable role set is passed in by
 * the caller (credentials-routes.ts derives it from `roleHasCap`): importing
 * agency_members.ts here would pull billing into the verdict closure that
 * social.ts → credentials.ts sits in (tests/invariants/paywall.test.ts).
 * Both sets bind as ONE JSON parameter (`json_each`), never a placeholder
 * list, so the statement's bind count is fixed (fidelity-suite IN-builder guard).
 */
const AGENCY_AUTHORITY_REFUSED_RE = /NOT NULL constraint failed:\s*releases\.updated_at\b/i;

/** The agency door's authority, refused at the reservation (I3-0363). */
export class AgencyAuthorityRefused extends CredentialError {
  constructor() {
    super("no sales grant for this release", 403);
  }
}

/** The agency door's acting seat, and the roles that carry the `sales` cap. */
export interface AgencyMintActor {
  agencyId: string;
  memberId: string;
  salesRoles: readonly string[];
  /** I12-RG-MONEY (D-0883(4)): the agency session epoch the request was
   * admitted at, and for a seat-key session its seat and seat epoch — re-asked
   * in the batch by agencies.ts `agencyActorLiveSql` (0363-sib, I12-FX8). */
  epoch: number;
  seatId?: string;
  seatEpoch?: number;
}

function agencyAuthorityStatement(db: DbLike, agency: AgencyMintActor, passportId: string, releaseId: string): DbStatement {
  const actor = agencyActorLiveSql(agency.agencyId, {
    memberId: agency.memberId,
    epoch: agency.epoch,
    capRoles: agency.salesRoles,
    ...(agency.seatId === undefined ? {} : { seatId: agency.seatId, seatEpoch: agency.seatEpoch ?? -1 }),
  });
  return db
    .prepare(
      `UPDATE releases SET updated_at = CASE WHEN
         EXISTS (SELECT 1 FROM agencies a WHERE a.agency_id = ? AND a.status = 'active')
         AND NOT ${CREATOR_CLOSED_SQL("?")}
         AND ${actor.sql}
         AND (SELECT CASE WHEN e.event_type = 'agency_authorized' THEN json_extract(e.payload, '$.scope') END
                FROM trust_events e
               WHERE e.subject = ?
                 AND e.event_type IN ('agency_authorized','agency_revoked')
                 AND json_valid(e.payload) = 1 AND json_type(e.payload) = 'object'
                 AND json_extract(e.payload, '$.agency_id') = ?
               ORDER BY e.seq DESC LIMIT 1) IN (SELECT value FROM json_each(?))
       THEN updated_at ELSE NULL END
       WHERE release_id = ?`,
    )
    .bind(agency.agencyId, passportId, ...actor.params, passportId, agency.agencyId, JSON.stringify([...SALES_SCOPES]), releaseId); // D-0960(2) (I20-FX3A): a close inside the window refuses, as agencyCanManageSales does at the door
}

/** I38-AGCRED: the seat audit actions of the two agency credential write doors (audit_log.action, actor_kind 'agency'). */
export const AGENCY_CREDENTIALS_ISSUED_ACTION = "agency:credentials_issued";
export const AGENCY_CREDENTIALS_REVOKED_ACTION = "agency:credentials_revoked";

/** The agency seat's audit_log row (actor_kind 'agency'), the shape agency_members.ts `seatAuditStatement` writes. Written
 * here rather than imported: importing agency_members.ts pulls oidc/closures/stripe into the paywall invariant's verdict
 * closure (tests/invariants/paywall.test.ts, measured). */
function agencySeatAuditStatement(db: DbLike, seat: string, action: string, passportId: string, detail: string, at: string): DbStatement {
  return db.prepare("INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) VALUES (?, ?, ?, ?, ?, 'agency')").bind(seat, action, detail, at, passportId);
}

export function isReservationRefused(e: unknown): boolean {
  return e instanceof Error && RESERVATION_REFUSED_RE.test(e.message);
}

/** The ONE way a raw credential becomes its stored hash. Exported (S-044 W2) so
 * tests use THIS and not a hand-rolled `sha256Hex(raw)` — the single-hasher
 * invariant caught exactly that in `credentials.test.ts`, and its objection is
 * not stylistic: a test that derives the stored hash by a different path than
 * production can pass while asserting a hash production would never write, so
 * the test proves nothing about the column it is supposedly checking. */
export async function credHash(raw: string): Promise<string> {
  return hex(await sha256Bytes(new TextEncoder().encode(raw)));
}

/**
 * Issue `count` one-time credentials for a release the caller owns. Returns the
 * RAW credentials (shown once, to distribute to buyers); the DB stores only
 * their hashes.
 *
 * TAKES THE VERIFICATION WITNESS (docs/VERIFIED-MINT.md), where it used to take
 * a `passportId: string`. This function had no verification read at all, on
 * EITHER of its two doors, so a creator who was verified, registered a release
 * and was then de-verified kept minting sellable one-time codes for it
 * indefinitely — measured, on a passport downgraded verified → claimed, as three
 * codes minted without complaint. Registering the release was gated; minting
 * distributable value FOR it was not.
 *
 * THE GATE IS ON THE CREATOR'S STANDING, NOT THE ACTOR'S, and that is a decision
 * rather than an oversight. `POST /agency/releases/:id/credentials` resolves a
 * creator through a `manage_sales` grant and issues on that creator's behalf; an
 * agency cannot hold more authority than the creator it represents, so the
 * witness that door needs is one for the RESOLVED CREATOR (taken after the grant
 * check, so a grantless agency is never told anything about the owner's
 * standing). If verification is withdrawn, the agency's ability to mint on that
 * creator's behalf goes with it. The badge is what fans are trusting when they
 * buy a code off-platform; revocability is the whole reason it means anything.
 *
 * CALLER-GATING that remains the HTTP layer's: that the witness belongs to the
 * creator this request is entitled to act for. This fn re-checks the release
 * belongs to that creator.
 */
export async function issueCredentials(
  db: DbLike,
  creator: VerifiedCreator,
  releaseId: string,
  count: number,
  mintRaw: () => string,
  now: () => string = () => new Date().toISOString(),
  issuedActor = "creator",
  /** The agency door's acting seat (I3-0363): its authority is re-asked inside the reservation batch. */
  agency?: AgencyMintActor,
): Promise<{ credentials: string[] }> {
  const passportId = creator.passport_id;
  if (!/^psp_[0-9a-f]{16}$/.test(passportId)) throw new CredentialError("bad passport id");
  if (!RELEASE_ID_RE.test(releaseId)) throw new CredentialError("bad release id");
  if (!Number.isInteger(count) || count < 1 || count > MAX_ISSUE) throw new CredentialError(`count must be 1..${MAX_ISSUE}`);
  // This module resolves the release ITSELF rather than through
  // `releases.ownedRelease`, so it needs the same suppression fold: a SUPPRESSED
  // owner's release is the same "no such release" 404 a never-registered id
  // gets, for anyone who is not that owner. Without it this door answered
  // `403 not your release` for a delisted creator and `404 no such release` for
  // a nonexistent id — the delisting oracle, one module along (review HIGH 1,
  // found by the CREATOR_SCOPED audit in suppression-oracles.test.ts). The
  // filter rides the same statement, so it costs no round trip.
  const rel = await db
    .prepare(
      "SELECT r.passport_id, r.status, r.edition_size, EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = r.passport_id) AS suppressed FROM releases r WHERE r.release_id = ?",
    )
    .bind(releaseId)
    .first<{ passport_id: string; status: string; edition_size: number | null; suppressed: number }>();
  if (!rel) throw new CredentialError("no such release", 404);
  if (rel.passport_id !== passportId) {
    if (Number(rel.suppressed) !== 0) throw new CredentialError("no such release", 404);
    throw new CredentialError("not your release", 403);
  }
  if (rel.status === "withdrawn") throw new CredentialError("release is withdrawn", 409);

  const ts = now();
  // Build the raws + inserts FIRST (a bad mint throws before we reserve anything).
  const raws: string[] = [];
  const stmts = [];
  const seen = new Set<string>();
  for (let i = 0; i < count; i++) {
    const raw = mintRaw();
    if (!RAW_RE.test(raw) || seen.has(raw)) throw new CredentialError("bad credential mint");
    seen.add(raw);
    raws.push(raw);
    // I38-AGCRED (0274): the agency door's acting seat on every code it mints; NULL on the creator's own door.
    stmts.push(db.prepare("INSERT INTO claim_credentials (credential_hash, release_id, issued_by, status, issued_at, issued_actor, issued_member) VALUES (?, ?, ?, 'active', ?, ?, ?)").bind(await credHash(raw), releaseId, passportId, ts, issuedActor, agency?.memberId ?? null));
  }

  // ATOMIC limited-run gate: reserve `count` slots in a single guarded UPDATE.
  // For a limited run the WHERE clause fails (0 rows) once the run is full, so
  // concurrent issuances can never over-mint past edition_size (the check and
  // the increment are one statement — no check-then-act race). For an open run
  // (edition_size IS NULL) it always reserves and just tracks the issued total.
  //
  // AND THE STANDING RE-CHECK RIDES IT (audit item #29). This is the path
  // docs/VERIFIED-MINT.md §2 measured — 1.50 ms at count=1, 213.89 ms at
  // count=1000, because every credential is hashed between the witness read and
  // the reservation. The reservation is the moment the mint becomes real, so
  // that is where the predicate belongs.
  //
  // THE CONFLATION §2 WARNED ABOUT IS AVOIDED BY ASKING, NOT BY GUESSING. It
  // said folding standing into this UPDATE "would conflate the reservation's
  // 'limited run full' answer with 'standing lost'" — true of a bare
  // `changes !== 1`, so on the refusal path (and only there) the two are
  // separated by one read. A creator who has just been de-verified is told they
  // are, not that their edition sold out.
  // M32 — THE RESERVATION RIDES THE INSERTS. ONE BATCH, NO WINDOW.
  //
  // These were two round trips: an `UPDATE … WHERE <guard>` here, then a
  // separate `db.batch` of the credential rows, with a compensating decrement in
  // the catch. The compensation covers a JS exception and CANNOT cover a worker
  // eviction — which on this platform is routine, not a disaster — so a death
  // between the two left `codes_issued` incremented with zero credentials
  // written and a limited run permanently one short, silently. The old comment
  // never claimed atomicity; it claimed the failure direction was conservative,
  // which was true and is not the same property.
  //
  // WHY IT COULD NOT SIMPLY BE MOVED INTO THE BATCH, and how that is solved.
  // A zero-row UPDATE does not throw, so inside a batch a refused reservation
  // would be INVISIBLE and the credentials would mint anyway — the reverse of
  // the bug and far worse. The guard therefore rides INSIDE the write as a
  // `CASE … THEN x END` with no `ELSE`, feeding `releases.codes_issued`, which
  // is `INTEGER NOT NULL` (migration 0007). A violated guard writes NULL into a
  // NOT NULL column and the batch throws. This is the NOT-NULL ceiling idiom
  // already load-bearing in three places — `passports.claim_state` (claims and
  // the character gate), `review_queue.second_look` (the two-person rule), and
  // slice 16's `operator_lapsed` — and it is the fourth use, not a new pattern.
  //
  // WHY THE STANDING RE-CHECK IS IN THE CASE AS WELL AS IN `verifiedMintGate`,
  // since review measured that deleting it from here changes NO behaviour: the
  // gate rides the same batch and still throws, just on a different column, and
  // `issueCredentials` maps both to `VerifiedCreatorError`. It stays because the
  // two guard different things and only look redundant while they are adjacent.
  // The gate is an admission control on WRITING; this clause is part of the
  // reservation's own truth condition — the arithmetic that decides whether a
  // slot may be taken. A future caller that reserves without the gate (a bulk
  // path, a replay, a repair script) inherits the standing rule from the
  // statement rather than from a neighbour it may not copy. Redundancy between
  // two layers is the intended state, not an accident to be tidied.
  //
  // THE CONFLATION docs/VERIFIED-MINT.md §2 WARNED ABOUT IS STILL AVOIDED BY
  // ASKING, NOT BY GUESSING. §2 said folding standing into this UPDATE "would
  // conflate the reservation's 'limited run full' answer with 'standing lost'".
  // True, and unchanged: the two are still separated by one read, on the refusal
  // path and only there. A creator who has just been de-verified is told they
  // are, not that their edition sold out.
  try {
    await db.batch([
      db
        .prepare(
          `UPDATE releases SET codes_issued = CASE
             WHEN (edition_size IS NULL OR codes_issued + ? <= edition_size)
              AND ${VERIFIED_NOW_SUBQUERY} IS NOT NULL
              AND releases.status <> 'withdrawn'
             THEN codes_issued + ? END,
             updated_at = ?
           WHERE release_id = ?`,
        )
        .bind(count, passportId, count, ts, releaseId),
      // QUEUE #947/d-mint-closure: the agency's own authority (closure included) is asked BEFORE the gate, so a close on
      // the agency path still answers AgencyAuthorityRefused; the creator's own mint is refused closed by the gate.
      ...(agency ? [agencyAuthorityStatement(db, agency, passportId, releaseId)] : []),
      // The gate rides the same batch, as it already did. It closes the window
      // between the standing re-check above and the credentials themselves,
      // which was the wider of the two on a large issuance — and is now zero.
      verifiedMintGate(db, passportId, releaseId, () => ts),
      ...stmts,
      // I38-AGCRED (R09 root A): the agency door's mint names the acting SEAT in audit_log, in this batch, so a refused
      // reservation records nothing and a failed audit insert mints nothing.
      ...(agency ? [agencySeatAuditStatement(db, agency.memberId, AGENCY_CREDENTIALS_ISSUED_ACTION, passportId, `${count} code(s) on ${releaseId}, agency ${agency.agencyId}`, ts)] : []),
    ]);
  } catch (e) {
    // NO COMPENSATION, AND NONE IS POSSIBLE TO NEED. The batch is one
    // transaction: if it threw, the reservation rolled back with the inserts.
    // The decrement that used to live here is deleted rather than kept "just in
    // case" — a decrement that can now only ever run against a reservation that
    // never landed is not a safety net, it is a way to lose a slot.
    //
    // THE RESERVATION'S OWN REFUSAL, told apart from every other fault in the
    // batch by the exact column, the way `CLAIM_GATE_REFUSED_RE` and
    // `SECOND_LOOK_REFUSED_RE` pin theirs. A looser `/NOT NULL/` would launder
    // an unrelated constraint fault into "your edition sold out" — a sentence
    // that would send a creator to look at their edition size while the real
    // fault was somewhere else entirely.
    if (isReservationRefused(e)) {
      if (!isCurrentlyVerified(await passportStanding(db, passportId, Date.parse(ts) || Date.now()))) throw new VerifiedCreatorError("verified creators only");
      // I3-1067: the withdrawal check at the top is a read BEFORE the batch; a
      // withdrawal landing in between is refused by the reservation's own
      // `releases.status <> 'withdrawn'` term, and is told as what it is.
      const after = await db.prepare("SELECT status FROM releases WHERE release_id = ?").bind(releaseId).first<{ status: string }>();
      if (after?.status === "withdrawn") throw new CredentialError("release is withdrawn", 409);
      throw new CredentialError(`issuing ${count} would exceed the limited run of ${rel.edition_size}`, 409);
    }
    if (isWriteAdmissionRefusal(e)) throw new VerifiedCreatorError("verified creators only");
    if (agency && e instanceof Error && AGENCY_AUTHORITY_REFUSED_RE.test(e.message)) throw new AgencyAuthorityRefused();
    throw e instanceof CredentialError ? e : new CredentialError("credential issuance failed", 409);
  }
  return { credentials: raws };
}

/**
 * Revoke ONE outstanding (still `active`, never redeemed) credential and give
 * the reservation slot it holds back to the release — the M32 reservation
 * ceiling's mirror image. Fixes C-35: nothing in this file ever wrote
 * `status='revoked'` (it sits in the schema's CHECK, unused), and
 * `codes_issued` only ever went up, so a lost mint response (the raws exist
 * ONLY in that response — the DB stores hashes) or a code sent to the wrong
 * buyer permanently stranded a slot in a limited run with no way back.
 *
 * ATOMIC, not check-then-act: the statements ride ONE `db.batch`, and the
 * decrement is a CORRELATED SUBQUERY re-evaluated against `claim_credentials`'
 * CURRENT (pre-flip) row inside that same transaction — not a value read
 * earlier and trusted. It counts 1 only while the credential is still
 * `active` for THIS release and THIS issuer, else 0. That makes the whole
 * operation naturally idempotent: replaying this call against an
 * already-revoked (or already-redeemed, or wrong-owner) credential finds
 * nothing `active` left to count and decrements by zero — no double credit
 * back to the edition, ever, no matter how many times a retried request
 * lands.
 *
 * Ownership and suppression are folded in the same shape `issueCredentials`
 * and `credentialSummary` use, for the same reason (review HIGH 1): a
 * SUPPRESSED owner's release must answer identically to a release id that
 * never existed, or this door becomes the same delisting oracle those two
 * were fixed for.
 *
 * Returns false when nothing was revoked — wrong credential, already dead,
 * or a race lost to a concurrent redemption/revoke (re-read, then false) — so
 * a caller cannot use an exception shape to learn more than the boolean says.
 * It throws only the release fold's 404/403, the mint door's standing refusal
 * if standing is lost at commit, and a 409 after REVOKE_ATTEMPTS lost races.
 *
 * THE LEDGER ROW (D-0762, QUEUE #485). A revocation that changes a row
 * appends `credentials_revoked {release_id, count: 1}` in the SAME batch as the
 * flip — see `commitRevocation`. The door is credentials-routes.ts
 * `POST /account/releases/:id/credentials/revoke` (by raw code, hashed there).
 */
export async function revokeCredential(
  db: DbLike,
  creator: string | { readonly passport_id: string },
  releaseId: string,
  credentialHash: string,
  now: () => string = () => new Date().toISOString(),
  /** I3-2204 a (D-0993(1) pick A): the agency door's partition — only a code whose `issued_actor` is this agency id.
   * Absent on the creator door, which revokes every code issued on its release (today's behaviour). */
  issuerScope?: string,
  /** I38-AGCRED: the agency door's acting seat (member id), named in the revoke's audit row. */
  seat?: string,
): Promise<boolean> {
  const passportId = revokerId(creator);
  if (!RELEASE_ID_RE.test(releaseId)) throw new CredentialError("bad release id");
  if (!/^[0-9a-f]{64}$/.test(credentialHash)) throw new CredentialError("bad credential hash");
  await requireRevocableRelease(db, passportId, releaseId);
  for (let attempt = 1; ; attempt++) {
    // Eligibility is decided by a plain read BEFORE the write. A
    // `credential_hash` that doesn't exist, belongs to another release, or is
    // already redeemed/revoked is a no-op, not an error — this is a boolean
    // primitive, not an oracle beyond what the 404/403 above already gives out.
    const before = await db
      .prepare(`SELECT status FROM claim_credentials WHERE credential_hash = ? AND release_id = ? AND issued_by = ?${issuerScope === undefined ? "" : " AND issued_actor = ?"}`)
      .bind(credentialHash, releaseId, passportId, ...(issuerScope === undefined ? [] : [issuerScope]))
      .first<{ status: string }>();
    if (!before || before.status !== "active") return false;
    // A redemption that lands between the read and the batch rolls the whole
    // batch back (the gate in `commitRevocation`), so the loop re-reads and
    // answers `false` — never `true` for a flip that did not happen (INV-058).
    if (await commitRevocation(db, passportId, releaseId, credentialHash, 1, now, issuerScope, seat)) return true;
    if (attempt >= REVOKE_ATTEMPTS) throw new CredentialError("revocation contended; retry", 409);
  }
}

/**
 * I3-1259 (D-0712, D-0829) — THE CREATOR'S LIST OF OUTSTANDING CREDENTIALS, BY
 * PUBLIC ID. The DB holds only hashes and a creator who lost the mint response
 * holds no raw code, so per-code revoke needs a name the creator CAN hold: a
 * random `public_id` (`crd_` + 16 hex) minted per row by the NEEDS-MIGRATION
 * trigger (I4-13-R4 REPORT) — never derived from the hash. Lists only
 * `active` (unredeemed, unrevoked) rows of one release this creator issued,
 * newest first, with the same ownership/suppression fold revoke uses
 * (`requireRevocableRelease`), so it stays open to a suppressed owner (D-0690).
 * `truncated` says a page was cut; there is no bulk revoke-all here (D-0712).
 */
export const CREDENTIAL_PUBLIC_ID_RE = /^crd_[0-9a-f]{16}$/;
export const OUTSTANDING_CREDENTIALS_PAGE_MAX = 500;
export interface OutstandingCredential {
  public_id: string;
  issued_at: string;
  /** I38-AGCRED: present on an AGENCY-minted code only — the agency, and the seat (member id) that minted it (null for a
   * code minted before migration 0274). A code the creator minted carries no such key. */
  minted_by?: { agency_id: string; seat: string | null };
}
export async function listOutstandingCredentials(
  db: DbLike,
  passportId: string,
  releaseId: string,
): Promise<{ credentials: OutstandingCredential[]; truncated: boolean }> {
  if (!RELEASE_ID_RE.test(releaseId)) throw new CredentialError("bad release id");
  await requireRevocableRelease(db, passportId, releaseId);
  const { results } = await db
    .prepare(
      `SELECT public_id, issued_at, issued_actor, issued_member FROM claim_credentials
        WHERE release_id = ? AND issued_by = ? AND status = 'active' AND public_id IS NOT NULL
        ORDER BY issued_at DESC, public_id DESC LIMIT ?`,
    )
    .bind(releaseId, passportId, OUTSTANDING_CREDENTIALS_PAGE_MAX + 1)
    .all<{ public_id: string; issued_at: string; issued_actor: string | null; issued_member: string | null }>();
  const credentials = results.slice(0, OUTSTANDING_CREDENTIALS_PAGE_MAX).map((r): OutstandingCredential =>
    typeof r.issued_actor === "string" && r.issued_actor.startsWith("agc_")
      ? { public_id: r.public_id, issued_at: r.issued_at, minted_by: { agency_id: r.issued_actor, seat: r.issued_member ?? null } }
      : { public_id: r.public_id, issued_at: r.issued_at },
  );
  return { credentials, truncated: results.length > OUTSTANDING_CREDENTIALS_PAGE_MAX };
}

/**
 * REVOKE ONE CREDENTIAL BY ITS PUBLIC ID (I3-1259) — resolves the id to its hash
 * WITHIN this release and this issuer, then is exactly `revokeCredential` (same
 * atomic batch, same ledger row, same boolean). The release fold runs FIRST, as
 * in `revokeCredential`; an id that is malformed, unknown, or belongs to another
 * release/issuer answers `false` — the same answer as an already-dead code, so
 * the boolean stays the whole oracle.
 */
export async function revokeCredentialByPublicId(
  db: DbLike,
  creator: string | { readonly passport_id: string },
  releaseId: string,
  publicId: string,
  now: () => string = () => new Date().toISOString(),
): Promise<boolean> {
  if (!RELEASE_ID_RE.test(releaseId)) throw new CredentialError("bad release id");
  await requireRevocableRelease(db, revokerId(creator), releaseId);
  if (typeof publicId !== "string" || !CREDENTIAL_PUBLIC_ID_RE.test(publicId)) return false;
  const row = await db
    .prepare("SELECT credential_hash FROM claim_credentials WHERE public_id = ? AND release_id = ? AND issued_by = ?")
    .bind(publicId, releaseId, revokerId(creator))
    .first<{ credential_hash: string }>();
  if (!row) return false;
  return revokeCredential(db, creator, releaseId, row.credential_hash, now);
}

/** I38-AGCRED: an agency seat id (agency_members.member_id). */
export const AGENCY_SEAT_ID_RE = /^agm_[0-9a-f]{16}$/;
/** I38-AGCRED: the creator's audit action for a one-seat revoke (audit_log.action, actor_kind 'creator'). */
export const CREATOR_SEAT_CODES_REVOKED_ACTION = "creator:seat_codes_revoked";
/** The one answer for a seat the creator may not name: unknown, or of an agency that never held a grant on them. */
export const SEAT_NOT_GRANTED = "that seat is not one of an agency you authorized";

/**
 * I38-AGCRED (D-1152(2); I38-RED-R09-PURPLE roots A/B, recourse only) — THE CREATOR REVOKES EVERY OUTSTANDING CODE ONE
 * AGENCY SEAT MINTED ON ONE OF THEIR RELEASES, IN ONE CALL. Not D-0712's retired revoke-all: it reaches only the codes
 * whose `issued_member` is the named seat (migration 0274), so the creator's own codes and every other seat's stay live.
 * The release fold runs first (`requireRevocableRelease`: unknown and a suppressed owner's release are 404, another
 * creator's is 403). The seat's agency is its roster row, or — once the seat is removed and its row is gone — the agency
 * its codes on THIS creator's releases name; a seat that resolves to no agency, or to one that never held an
 * `agency_authorized` grant on this creator, is ONE 403 (no oracle between the two). Ending the grant does not end this
 * recourse. Returns how many codes it revoked (0 is a 200). ONE batch per attempt, as `commitRevocation`: the
 * `credentials_revoked {release_id, count}` row, the admission fence on the seat-scoped active count, the slot
 * give-back, the flip and the creator's audit row naming the seat. Whether held codes redeem is NOT changed by
 * anything else here (JASON-QUESTIONS.md 5).
 */
export async function revokeSeatCredentials(
  db: DbLike,
  creator: string,
  releaseId: string,
  seat: unknown,
  now: () => string = () => new Date().toISOString(),
): Promise<{ revoked: number; agency_id: string }> {
  if (!RELEASE_ID_RE.test(releaseId)) throw new CredentialError("bad release id");
  await requireRevocableRelease(db, creator, releaseId);
  if (typeof seat !== "string" || !AGENCY_SEAT_ID_RE.test(seat)) throw new CredentialError("bad seat id", 400);
  const agency = await db
    .prepare(
      `SELECT x.agency_id FROM (
         SELECT m.agency_id AS agency_id FROM agency_members m WHERE m.member_id = ?
         UNION SELECT c.issued_actor FROM claim_credentials c WHERE c.issued_by = ? AND c.issued_member = ?
       ) x
       WHERE EXISTS (SELECT 1 FROM trust_events e WHERE e.subject = ? AND e.event_type = 'agency_authorized'
                       AND json_valid(e.payload) = 1 AND json_type(e.payload) = 'object' AND json_extract(e.payload, '$.agency_id') = x.agency_id)
       LIMIT 1`,
    )
    .bind(seat, creator, seat, creator)
    .first<{ agency_id: string }>();
  if (!agency) throw new CredentialError(SEAT_NOT_GRANTED, 403);
  const agencyId = agency.agency_id;
  const scope = " AND issued_member = ? AND issued_actor = ?";
  for (let attempt = 1; ; attempt++) {
    const n = Number(
      (
        await db
          .prepare(`SELECT COUNT(*) AS n FROM claim_credentials WHERE release_id = ? AND issued_by = ? AND status = 'active'${scope}`)
          .bind(releaseId, creator, seat, agencyId)
          .first<{ n: number }>()
      )?.n ?? 0,
    );
    if (n === 0) return { revoked: 0, agency_id: agencyId };
    const ts = now();
    try {
      await appendEvents(
        db,
        [{ event_type: "credentials_revoked", subject: creator, actor_kind: "system", actor_id: "session:creator", payload: { release_id: releaseId, count: n } }],
        now,
        ["credentials_revoked"],
        [
          db
            .prepare(
              `INSERT INTO credential_revoke_admissions (subject, ref, admitted_at)
               VALUES (CASE WHEN (SELECT COUNT(*) FROM claim_credentials WHERE release_id = ? AND issued_by = ? AND status = 'active'${scope}) = ? THEN ? END, ?, ?)
               ON CONFLICT(subject, ref) DO UPDATE SET admitted_at = excluded.admitted_at`,
            )
            .bind(releaseId, creator, seat, agencyId, n, creator, releaseId, ts),
          db
            .prepare(
              `UPDATE releases SET codes_issued = codes_issued - (
                 SELECT COUNT(*) FROM claim_credentials WHERE release_id = releases.release_id AND issued_by = ? AND status = 'active'${scope}
               ), updated_at = ? WHERE release_id = ?`,
            )
            .bind(creator, seat, agencyId, ts, releaseId),
          db
            .prepare(`UPDATE claim_credentials SET status = 'revoked' WHERE release_id = ? AND issued_by = ? AND status = 'active'${scope}`)
            .bind(releaseId, creator, seat, agencyId),
          db
            .prepare("INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) VALUES (?, ?, ?, ?, ?, 'creator')")
            .bind(creator, CREATOR_SEAT_CODES_REVOKED_ACTION, `${n} code(s) on ${releaseId} minted by seat ${seat}, agency ${agencyId}`, ts, releaseId),
        ],
        ["system"],
      );
      return { revoked: n, agency_id: agencyId };
    } catch (e) {
      if (!(e instanceof Error && REVOKE_FENCE_REFUSED_RE.test(e.message))) throw e;
      if (attempt >= REVOKE_ATTEMPTS) throw new CredentialError("revocation contended; retry", 409);
    }
  }
}

/**
 * QUEUE #484 (D-0712, D-0848(1)): bulk revoke-by-release (`revokeUnredeemedCredentials`,
 * D-0775/QUEUE #436) is RETIRED. It killed codes already handed to real fans to recover
 * from a creator's lost page load; the lost-mint-response case is covered instead by
 * `GET .../outstanding` (by public id) + per-id `revokeCredentialByPublicId`, both above.
 */

/** Re-reads after a lost race before a revoke gives up with a 409. */
const REVOKE_ATTEMPTS = 3;

/** D-0838(5) / D-0690: revoke stays open to a de-verified or suppressed owner, so
 * the revoke doors take the session's passport id; an object carrying
 * `passport_id` (older callers pass the verified witness) is read for its id
 * only — standing is not what admits a revoke. */
function revokerId(creator: string | { readonly passport_id: string }): string {
  return typeof creator === "string" ? creator : creator.passport_id;
}

/** The revoke fence's refusal (0141): the admission subject was NULL because the
 * active count in scope moved since the caller read it. */
const REVOKE_FENCE_REFUSED_RE = /NOT NULL constraint failed:\s*credential_revoke_admissions\.subject/i;

/**
 * Same not-mine/no-such fold as issueCredentials (review HIGH 1): a SUPPRESSED
 * owner's release reads back exactly like an unknown release id for anyone who
 * is not that owner, so neither revoke form is a delisting oracle.
 */
async function requireRevocableRelease(db: DbLike, passportId: string, releaseId: string): Promise<void> {
  const rel = await db
    .prepare("SELECT r.passport_id, EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = r.passport_id) AS suppressed FROM releases r WHERE r.release_id = ?")
    .bind(releaseId)
    .first<{ passport_id: string; suppressed: number }>();
  if (!rel) throw new CredentialError("no such release", 404);
  if (rel.passport_id !== passportId) {
    if (Number(rel.suppressed) !== 0) throw new CredentialError("no such release", 404);
    throw new CredentialError("not your release", 403);
  }
}

/**
 * ONE ATOMIC BATCH: the `credentials_revoked` ledger row (D-0762), the slot
 * give-back and the flip, for exactly ONE credential. There is no whole-release
 * form (D-0712, D-0848(1)): a hash that is not one credential hash is refused
 * before any statement is built. Exported for that refusal's test only; the
 * one production caller is `revokeCredential`.
 *
 * THE COUNT IS BOUND AT COMMIT, NOT AT READ (INV-058 claim-then-act). The
 * payload is hashed into the chain, so `expected` must be fixed before the
 * batch; a redemption or a mint landing between the caller's read and this
 * batch would make the row state a number that never happened. The first extra
 * statement is the `credential_revoke_admissions` NOT NULL fence (0141, D-0838(5)):
 * its subject is the revoking passport ONLY IF the active count in scope still
 * equals `expected`, else NULL — and NULL rolls back the event, the decrement
 * and the flip together. Standing is NOT asked: revoke stays open to a
 * de-verified or suppressed owner (D-0690). Inside the batch the decrement and
 * the flip read the same rows the fence counted, so the row's `count` IS the
 * number flipped. Returns false on a lost race (caller re-reads).
 */
export async function commitRevocation(
  db: DbLike,
  passportId: string,
  releaseId: string,
  hash: string,
  expected: number,
  now: () => string,
  issuerScope?: string,
  seat?: string,
): Promise<boolean> {
  if (typeof hash !== "string" || !/^[0-9a-f]{64}$/.test(hash)) throw new CredentialError("revoke needs one credential");
  // I3-2204 a: the agency partition rides EVERY statement of the batch (admission count, counter, flip), so a scoped
  // revoke can neither count nor flip a code another issuer minted.
  const scoped = issuerScope === undefined ? " AND credential_hash = ?" : " AND credential_hash = ? AND issued_actor = ?";
  const scopeArgs = issuerScope === undefined ? [hash] : [hash, issuerScope];
  const ts = now();
  try {
    await appendEvents(
      db,
      [{ event_type: "credentials_revoked", subject: passportId, actor_kind: "system", actor_id: issuerScope === undefined ? "session:creator" : "session:agency", payload: { release_id: releaseId, count: expected } }],
      now,
      ["credentials_revoked"], // governed: NOT on APPEND_ALLOWED — this is its only door
      [
        db
          .prepare(
            `INSERT INTO credential_revoke_admissions (subject, ref, admitted_at)
             VALUES (CASE WHEN (SELECT COUNT(*) FROM claim_credentials WHERE release_id = ? AND issued_by = ? AND status = 'active'${scoped}) = ? THEN ? END, ?, ?)
             ON CONFLICT(subject, ref) DO UPDATE SET admitted_at = excluded.admitted_at`,
          )
          .bind(releaseId, passportId, ...scopeArgs, expected, passportId, releaseId, ts),
        db
          .prepare(
            `UPDATE releases SET codes_issued = codes_issued - (
               SELECT COUNT(*) FROM claim_credentials WHERE release_id = releases.release_id AND issued_by = ? AND status = 'active'${scoped}
             ), updated_at = ? WHERE release_id = ?`,
          )
          .bind(passportId, ...scopeArgs, ts, releaseId),
        db
          .prepare(`UPDATE claim_credentials SET status = 'revoked' WHERE release_id = ? AND issued_by = ? AND status = 'active'${scoped}`)
          .bind(releaseId, passportId, ...scopeArgs),
        // I38-AGCRED: the agency door's seat audit row rides the same batch, after the flip (the fence above rolls both back).
        // The production door always passes its seat; a caller that passes none is recorded as the agency, never as nothing.
        ...(issuerScope !== undefined ? [agencySeatAuditStatement(db, seat ?? issuerScope, AGENCY_CREDENTIALS_REVOKED_ACTION, passportId, `${expected} code(s) on ${releaseId}, agency ${issuerScope}`, ts)] : []),
      ],
      ["system"], // S27: the system actor requires explicit authorization
    );
    return true;
  } catch (e) {
    if (!(e instanceof Error && REVOKE_FENCE_REFUSED_RE.test(e.message))) throw e;
    return false;
  }
}

export interface CredentialSummary {
  release_id: string;
  passport_id: string; // the owning creator (the route authorizes the caller against this)
  edition_size: number | null; // limited run size, or null for an open run
  issued: number; // credentials that exist
  redeemed: number; // credentials a buyer has claimed
  outstanding: number; // active, still-redeemable codes
  /**
   * I3-1523. The THIRD leg of the partition, and the only number that lets a
   * reader reconcile the other two with the reservation counter.
   * `claim_credentials.status` is `CHECK (status IN ('active','redeemed','revoked'))`
   * (migrations/0007_releases_credentials.sql:74) — three values, no fourth — so
   * `issued === redeemed + outstanding + revoked` is CLOSED, and
   * `issued - revoked === releases.codes_issued`, the counter `remaining` is
   * derived from. Without it the object states `issued: 3, edition_size: 3,
   * remaining: 1` after one revocation and gives the reader no way to tell that
   * from an over-mint.
   */
  revoked: number; // codes the creator revoked (`revokeCredential` gave the slot back)
  remaining: number | null; // how many MORE can be minted (limited only); null = open run
  /**
   * I3-0739 — OWNER READS ONLY: how many of `issued` an agency minted
   * (`issued_actor <> 'creator'`; a legacy NULL actor is counted in neither part).
   * Absent on the agency door, whose counts are already its own slice.
   */
  agency_issued?: number;
}

/**
 * Issued / redeemed / outstanding for a release's codes, plus how many more may
 * be minted under a limited run. Counts read the REAL credential rows (always
 * truthful); `remaining` uses the atomic reservation counter so it agrees with
 * what issuance will actually allow. Returns null if the release doesn't exist;
 * the caller (creator or authorized agency) authorizes against `passport_id`.
 *
 * `viewer` is the passport the CALLER is entitled to act for, or null for a
 * caller (an agency) that is never the owner. IT HAS NO DEFAULT
 * (FINDING-0014-E-4): an ownership control that defaults to null is opt-IN, and
 * a caller that simply forgets the argument reads back any non-suppressed
 * release on the platform. Both of today's callers are correct; the default was
 * what made the NEXT one wrong by omission. Passing `null` is now a statement —
 * "this caller is never the owner and is authorized elsewhere" — and is the only
 * way to get the unscoped read. A non-null `viewer` is also a
 * RESTRICTION: any release owned by another passport reads back as null, at the
 * same cost as an unknown id. A SUPPRESSED owner's release is
 * the same `null` an unknown release id gets for anyone who is not that owner,
 * so the two doors above this cannot be used to tell a delisted creator from one
 * who never existed (review HIGH 1). The owner keeps the read: a delisted
 * creator still has to see what is outstanding against them.
 */
export async function credentialSummary(db: DbLike, releaseId: string, viewer: string | null, issuerScope?: string): Promise<CredentialSummary | null> {
  if (!RELEASE_ID_RE.test(releaseId)) return null;
  const rel = await db
    .prepare(
      "SELECT r.passport_id, r.edition_size, r.codes_issued, EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = r.passport_id) AS suppressed FROM releases r WHERE r.release_id = ?",
    )
    .bind(releaseId)
    .first<{ passport_id: string; edition_size: number | null; codes_issued: number; suppressed: number }>();
  if (!rel) return null;
  if (Number(rel.suppressed) !== 0 && rel.passport_id !== viewer) return null;
  // OWNER-SCOPED CALLERS ARE FILTERED HERE, IN THE PRIMITIVE, BEFORE THE COUNT.
  // A non-null `viewer` says "this caller may only ever see this ONE passport's
  // release", so a release owned by anyone else is the same `null` an unknown id
  // gets — and it must cost the same, which is why the check sits above the
  // COUNT and not in the route. Measured: with the ownership check in the route
  // instead, a non-owning creator's probe on a LIVE release ran
  // `SELECT releases, SELECT claim_credentials` while a nonexistent id ran only
  // the first, so the identical 404 body was still an existence oracle on the D1
  // sequence (suppression-oracles.test.ts, the CREATOR_SCOPED audit). `viewer`
  // is null for the agency door, which is never the owner and is authorized by
  // its grant afterwards — that path is unchanged.
  if (viewer !== null && rel.passport_id !== viewer) return null;
  // DOCKET #36 (i): partitioned by issuer when `issuerScope` is given. Two
  // agencies can each hold a LIVE `manage_sales` grant on the same creator's
  // release at once (agency_authorized is per (agency, passport) pair, not
  // exclusive), and without this filter both read the SAME union count — each
  // sees the OTHER's issuance folded into its own. `issued_actor` is the actor
  // that minted the row (the agency id, or "creator" for the retired creator
  // door — see credentials-routes.ts:144's INSERT), so filtering on it scopes
  // the summary to exactly what THIS caller issued. The owner (viewer set) is
  // unaffected — a creator is entitled to the union across every agency it
  // ever granted, so `issuerScope` is only ever passed on the agency door.
  // This is a plain filtered SELECT, not a competing writer: two calls from
  // the SAME agency each run their own independent read, so there is no
  // read-then-act window here to additionally make atomic (unlike the WRITE
  // ceiling below, which already is — see the M32 comment on the UPDATE).
  const c = await db
    .prepare(
      issuerScope
        ? "SELECT COUNT(*) AS issued, SUM(CASE WHEN status = 'redeemed' THEN 1 ELSE 0 END) AS redeemed, SUM(CASE WHEN status = 'active' THEN 1 ELSE 0 END) AS outstanding, SUM(CASE WHEN status = 'revoked' THEN 1 ELSE 0 END) AS revoked FROM claim_credentials WHERE release_id = ? AND issued_actor = ?"
        : "SELECT COUNT(*) AS issued, SUM(CASE WHEN status = 'redeemed' THEN 1 ELSE 0 END) AS redeemed, SUM(CASE WHEN status = 'active' THEN 1 ELSE 0 END) AS outstanding, SUM(CASE WHEN status = 'revoked' THEN 1 ELSE 0 END) AS revoked, SUM(CASE WHEN issued_actor <> 'creator' THEN 1 ELSE 0 END) AS agency_issued FROM claim_credentials WHERE release_id = ?",
    )
    .bind(...(issuerScope ? [releaseId, issuerScope] : [releaseId]))
    .first<{ issued: number; redeemed: number | null; outstanding: number | null; revoked: number | null; agency_issued?: number | null }>();
  const issued = c?.issued ?? 0;
  return {
    release_id: releaseId,
    passport_id: rel.passport_id,
    edition_size: rel.edition_size,
    issued,
    redeemed: c?.redeemed ?? 0,
    outstanding: c?.outstanding ?? 0,
    revoked: c?.revoked ?? 0,
    remaining: rel.edition_size != null ? Math.max(0, rel.edition_size - rel.codes_issued) : null,
    ...(issuerScope ? {} : { agency_issued: Number(c?.agency_issued ?? 0) }),
  };
}

/** Page bound of `credentialSummariesForCreator` — the console's own display cap
 * for published listings (views.ts `publishedListings`), so one page covers the
 * screen for every creator the console can render in full. */
export const CREDENTIAL_SUMMARIES_PAGE_MAX = 100;

export interface CredentialSummaryPage {
  summaries: CredentialSummary[];
  /** The last release id of this page when MORE of the caller's releases follow
   * it, else null. A caller holding `next` may conclude nothing about releases
   * beyond it except that they exist and are its own. */
  next: string | null;
}

/**
 * QUEUE #521 / I3-0248 — every one of the CALLER'S releases' summaries in one
 * read, keyed by release id, paged. The OWNERSHIP control is the WHERE clause
 * (`r.passport_id = ?` bound to `creator`, the caller's own passport): a release
 * owned by any other passport is never selected, so there is no post-filter to
 * forget and no id the caller supplies. `after` is a cursor strictly inside the
 * caller's own id space (`release_id > ?`), never an id lookup. The owner keeps
 * the read while suppressed, exactly as `credentialSummary` with `viewer` set.
 */
export async function credentialSummariesForCreator(db: DbLike, creator: string, after: string | null, limit = CREDENTIAL_SUMMARIES_PAGE_MAX): Promise<CredentialSummaryPage> {
  const n = Math.max(1, Math.min(CREDENTIAL_SUMMARIES_PAGE_MAX, Math.floor(limit)));
  const { results } = await db
    .prepare(
      "SELECT r.release_id, r.passport_id, r.edition_size, r.codes_issued, COUNT(cc.credential_hash) AS issued, SUM(CASE WHEN cc.status = 'redeemed' THEN 1 ELSE 0 END) AS redeemed, SUM(CASE WHEN cc.status = 'active' THEN 1 ELSE 0 END) AS outstanding, SUM(CASE WHEN cc.status = 'revoked' THEN 1 ELSE 0 END) AS revoked, SUM(CASE WHEN cc.issued_actor <> 'creator' THEN 1 ELSE 0 END) AS agency_issued FROM releases r LEFT JOIN claim_credentials cc ON cc.release_id = r.release_id WHERE r.passport_id = ? AND r.release_id > ? GROUP BY r.release_id ORDER BY r.release_id LIMIT ?",
    )
    .bind(creator, after ?? "", n + 1)
    .all<{ release_id: string; passport_id: string; edition_size: number | null; codes_issued: number; issued: number; redeemed: number | null; outstanding: number | null; revoked: number | null; agency_issued: number | null }>();
  const rows = results ?? [];
  const page = rows.slice(0, n);
  return {
    summaries: page.map((r) => ({
      release_id: r.release_id,
      passport_id: r.passport_id,
      edition_size: r.edition_size,
      issued: Number(r.issued ?? 0),
      redeemed: Number(r.redeemed ?? 0),
      outstanding: Number(r.outstanding ?? 0),
      revoked: Number(r.revoked ?? 0),
      remaining: r.edition_size != null ? Math.max(0, r.edition_size - r.codes_issued) : null,
      agency_issued: Number(r.agency_issued ?? 0),
    })),
    next: rows.length > n ? page[page.length - 1]!.release_id : null,
  };
}

export interface RedeemResult {
  ok: boolean;
  outcome: "redeemed" | "dead" | "unknown";
  release_id?: string;
  acquisition_id?: string;
  /**
   * AUDIT ITEM #22 — the release's status AT REDEMPTION, and its title.
   *
   * The S33 DECISION LOCK stands and is not touched by this: a code for a
   * WITHDRAWN release still redeems, because the buyer paid and voiding paid
   * codes would let a seller take the money and unlist. What was wrong is that
   * the redeemer was told NOTHING. Their acquisition is real and permanent, and
   * `social.ts publicProfile` silently excludes withdrawn releases from both
   * the public collection list AND its count, and `featuredViews` silently
   * drops them from the showcase — so a fan redeemed a code, was told
   * "added to your collection", and watched nothing appear, with no surface
   * anywhere that named the reason. `GET /release/:id` carried the fact and no
   * web client called it.
   *
   * Both fields ride the SELECT that already resolves the credential (one extra
   * join, no extra round trip), and both are OPTIONAL because a dead/unknown
   * credential resolves no release at all.
   */
  release_status?: string;
  title?: string;
  /** S-CW2 (review F8): whether this redemption ALSO minted an owned card
   * instance — false when the shelf cap no-oped the mint (the redemption
   * itself still succeeded). Present only when the caller asked for a mint.
   * CO-11: `null` when the redemption committed but the optional confirmation
   * read could not be taken — unknown, which is the honest third answer and
   * is never a reason to report the redemption itself as rejected. */
  instance_minted?: boolean | null;
  instance_id?: string;
}

/**
 * Redeem a raw credential for a fan account. Atomically flips the credential
 * active→redeemed (single-use) and records an acquisition. A dead/unknown
 * credential records a bounded abuse-signal attempt and returns without minting.
 *
 * S-CW2: when `instanceMint` is provided, the SAME atomic batch also mints an
 * owned card instance of the release's creator (the Creator War property
 * layer) — so ownership can never appear without the redemption that paid for
 * it, or vice versa. Optional, so non-game callers are untouched; the mint is
 * cap-guarded as a NO-OP, never a failure (a full shelf must not void a
 * purchase). S-CW4: the mint is additionally gated on the creator's war
 * opt-in — a non-opted (or opted-out) creator's redemption still credits the
 * ACQUISITION (the collectible the fan paid for) but mints no game instance,
 * surfaced through the same `instance_minted: false` the cap-skip uses. The
 * opt-in read happens just before the batch; an opt-out racing a redemption
 * can strand one last mint on the old state, which is the benign direction
 * (opt-out freezes NEW mints, it never claws back owned property).
 */
/**
 * THE "creator is no longer reachable" UNION, and the ONE place it is written in
 * this file. Operator-suppression (`request_suppressions`) or operator-retraction
 * (`passports.retracted`) — the same pair `social.ts publicProfile` /
 * `verifyCollectible` and `rankings.ts dropSuppressed` fold on. Assumes the
 * releases row is aliased `rel`; both readers here alias it that way.
 *
 * AS-025: `redeemCredential` resolved a credential with a bare LEFT JOIN and no
 * such gate, so a code minted while a creator was live still handed the redeemer
 * that creator's release TITLE and STATUS after the operator acted — the one
 * surface that still named a subject every other surface had stopped naming.
 */
const CREATOR_UNAVAILABLE_SQL = `CASE WHEN rel.passport_id IS NOT NULL AND (
                     EXISTS (SELECT 1 FROM request_suppressions s WHERE s.passport_id = rel.passport_id)
                     OR EXISTS (SELECT 1 FROM passports p WHERE p.passport_id = rel.passport_id AND p.retracted = 1)
                   ) THEN 1 ELSE 0 END`;

/** I5-FX2-CRED: the redeemer-exists predicate every redemption write carries (one `?` = the account id). */
const REDEEMER_EXISTS_SQL = "EXISTS (SELECT 1 FROM accounts WHERE account_id = ?)";
/** The acquisitions fence's refusal: the account was erased before the batch ran. */
const REDEEMER_GONE_RE = /NOT NULL constraint failed:\s*acquisitions\.account_id/i;

export async function redeemCredential(
  db: DbLike,
  accountId: string,
  raw: string,
  mintAcqId: () => string,
  mintAttId: () => string,
  now: () => string = () => new Date().toISOString(),
  instanceMint?: { mintInstanceId: () => string; mintLogId: () => string },
  /** `ANALYTICS_FAN_SECRET`. Undefined is refused by `fanKey` (ADR-0006 §2). */
  fanSecret?: string,
  pubEnv?: AnalyticsTickEnv,
  pubCtx?: { waitUntil(p: Promise<unknown>): void },
): Promise<RedeemResult> {
  // M9: NORMALISE before matching. Codes are displayed UPPERCASE and grouped
  // (`ABCD-EFGH-…`), and mobile keyboards + paste routinely deliver lowercase or
  // a stray space/hyphen. The alphabet is base32 (A-Z2-7) with no separators, so
  // upper-casing and dropping whitespace/hyphens/underscores is loss-free and
  // reproduces the exact canonical form the mint hashed — a real code stops being
  // rejected as "malformed" for a cosmetic difference. Anything still off-shape
  // after normalising genuinely is not a credential.
  const norm = typeof raw === "string" ? raw.replace(/[\s\-_]/g, "").toUpperCase() : "";
  if (!RAW_RE.test(norm)) {
    // Malformed input isn't even a real credential — don't record it as abuse.
    throw new CredentialError("malformed credential");
  }
  const h = await credHash(norm);
  const ts = now();
  // The release JOIN is the audit-item-#22 addition: `release_status` and
  // `title` come back on the SAME statement that was already resolving the
  // credential. `rel.status` is aliased because `claim_credentials.status`
  // (active/redeemed/revoked) and `releases.status` (published/withdrawn) are
  // two different vocabularies on the same column name — conflating them here
  // would be a very quiet bug. LEFT JOIN, not JOIN: the credential row is the
  // authority on redeemability, and an unexpectedly missing release must not
  // turn a valid code into "unknown".
  const row = await db
    .prepare(
      // CO-11: `redeemed_by` and the acquisition id ride the SAME statement.
      // They are what makes a retry by the ORIGINAL redeemer answerable (see
      // the `status !== 'active'` branch below) without a second round trip.
      `SELECT c.release_id AS release_id, c.status AS status, c.redeemed_by AS redeemed_by, rel.status AS release_status, rel.title AS title, rel.passport_id AS passport_id,
              ${CREATOR_UNAVAILABLE_SQL} AS creator_unavailable,
              (SELECT a.acquisition_id FROM acquisitions a WHERE a.credential_hash = c.credential_hash) AS acquisition_id
         FROM claim_credentials c LEFT JOIN releases rel ON rel.release_id = c.release_id
        WHERE c.credential_hash = ?`,
    )
    .bind(h)
    .first<{ release_id: string; status: string; redeemed_by: string | null; release_status: string | null; title: string | null; passport_id: string | null; creator_unavailable: number; acquisition_id: string | null }>();

  if (!row) {
    // `unknown` IS NOT EMITTED (S-002 B4b). The vocabulary's `outcome` enum is
    // live|dead, and there is no release to scope the attempt to: an attempt
    // against a credential hash that matches nothing names no creator, so there
    // is no `release_attempt` cell for it to land in.
    await recordAttempt(db, h, accountId, "unknown", mintAttId, ts, [], pubEnv, pubCtx);
    return { ok: false, outcome: "unknown" };
  }
  // The one fan pseudonym this call needs, awaited BEFORE any batch is built
  // (analytics-events.ts: no statement is ever built from a promise). Null only
  // when the LEFT JOIN found no release row — then nothing is emitted, because
  // an event with no passport has no cell.
  const psp = row.passport_id;
  /**
   * THE SUBJECT THE ANSWER IS ALLOWED TO CARRY (control: `CREATOR_UNAVAILABLE_SQL`
   * above, evaluated on the statement that resolved the credential).
   *
   * Empty for a suppressed or retracted creator, which makes that answer
   * key-for-key IDENTICAL to the one a never-listed release yields (the LEFT
   * JOIN's null tolerance, right below) — a suppressed subject and one that was
   * never there read the same, which is the oracle the rest of the surface holds.
   *
   * The redemption itself is NOT refused, deliberately: a fan who already holds
   * the code is not the public, and this file's own AS-120 lock says the buyer's
   * record of a purchase is never erased. They keep the acquisition; they are
   * just not told whose it was.
   */
  const subject: { release_status?: string; title?: string } =
    row.creator_unavailable !== 0 ? {} : { ...(row.release_status !== null ? { release_status: row.release_status } : {}), ...(row.title !== null ? { title: row.title } : {}) };
  const key = psp === null ? null : await fanKey(fanSecret, accountId, psp);
  /** `redemption_attempt` for this release, or nothing. Fan-keyed, scope
   * `release_attempt` = the integrity block's membership population (S-1). */
  const attemptEvent = (outcome: "live" | "dead"): DbStatement[] =>
    psp === null || key === null ? [] : outboxStatements(db, { kind: "redemption_attempt", passportId: psp, occurredDay: occurredDay(ts), fanKey: key, scopeId: row.release_id, dims: { outcome } }, { accountId });

  // CO-11 — THE ORIGINAL REDEEMER'S RETRY GETS THE ORIGINAL RECEIPT.
  //
  // "Reuse is the abuse signal" is true of a STRANGER presenting a burned code.
  // It was never true of the person who burned it: the credential is theirs,
  // the acquisition is theirs, and `acquisitions.credential_hash` is UNIQUE, so
  // there is exactly one right answer to give them and no way for this branch
  // to mint a second anything. Returning `dead` to them was actively harmful,
  // because the redemption route runs its achievement, referral, encounter and
  // reward-policy hooks ONLY on `ok`. A redemption that committed and then hit
  // a transient failure on the way out left the fan holding the collectible,
  // told it had failed, and with those hooks unrun — and the retry that should
  // have recovered them said `dead` and skipped them again. Every one of those
  // hooks is idempotent by design, so replaying them on a retry is the
  // recovery path, and it needs no durable pending-event table to exist.
  //
  // NARROW ON PURPOSE: same account, status `redeemed` (never `revoked`), and
  // the acquisition row actually present. Anyone else still gets `dead`, and
  // the attempt is still recorded — `redemption_attempts` keeps its abuse
  // signal, which is why the row is written here rather than skipped.
  if (row.status === "redeemed" && row.redeemed_by === accountId && row.acquisition_id !== null) {
    await recordAttempt(db, h, accountId, "dead", mintAttId, ts, attemptEvent("dead"), pubEnv, pubCtx);
    return {
      ok: true,
      outcome: "redeemed",
      release_id: row.release_id,
      acquisition_id: row.acquisition_id,
      ...subject,
    };
  }
  if (row.status !== "active") {
    // Already redeemed or revoked → DEAD. Reuse is the abuse signal.
    await recordAttempt(db, h, accountId, "dead", mintAttId, ts, attemptEvent("dead"), pubEnv, pubCtx);
    return { ok: false, outcome: "dead" };
  }

  // ATOMIC single-use AND acquisition — one batch, so the credential can never
  // be burned without minting the acquisition (no purchase-loss window). The
  // flip is guarded by WHERE status='active' and the acquisition insert is
  // guarded by acquisitions.credential_hash UNIQUE, so a concurrent redemption
  // that already won makes THIS batch's insert violate UNIQUE → the whole batch
  // rolls back → we report dead. Exactly one redemption ever credits.
  const acqId = mintAcqId();
  let ciId: string | undefined;
  // S-CW4 fail-closed game gate (see docblock). Read OUTSIDE the batch —
  // consent is ledger-derived and D1 batches can't interleave reads — and only
  // when a mint was requested, so non-game callers pay nothing.
  // ND-3 (I34-RJ-08): asked at `ts`, the instant the card is minted at (war-instances.ts asks at its own `ts` too).
  // D-1123(1) (I34-RG-W3M CLOSED-WAR-MINT): and no card of a creator closed or suspended when the batch is built
  // (credits.ts CREATOR_CLOSED_SQL, the predicate packs and standings read); the fan keeps the redemption.
  const gameOn =
    instanceMint && row.passport_id
      ? (await warOptedIn(db, row.passport_id, ts)) && (await db.prepare(`SELECT ${CREATOR_CLOSED_SQL("?")} AS c`).bind(row.passport_id).first<{ c: number }>())?.c !== 1
      : false;
  let seqs: number[] = [];
  try {
    const span = outboxSpan(db);
    const spanRes = await db.batch([
      span.open,
      // I5-FX2-CRED (QUEUE #740): EVERY statement names the redeemer only while
      // the account exists, in the same statement (the MINTGUARD shape,
      // war-instances.ts). The acquisitions insert is the FENCE: its NOT NULL
      // `account_id` is read from `accounts` in the statement, so an erasure that
      // committed first aborts the WHOLE batch (caught below as a 401) and no
      // erased id lands in `redeemed_by` or `account_id`.
      db.prepare(`UPDATE claim_credentials SET status = 'redeemed', redeemed_by = ?, redeemed_at = ? WHERE credential_hash = ? AND status = 'active' AND ${REDEEMER_EXISTS_SQL}`).bind(accountId, ts, h, accountId),
      // `redeemed_by` = the redeemer, forever (0061); `account_id` = the holder,
      // which a gift may later change. Evidence readers compare the two.
      db.prepare("INSERT INTO acquisitions (acquisition_id, account_id, release_id, credential_hash, acquired_at, redeemed_by) VALUES (?, (SELECT a.account_id FROM accounts a WHERE a.account_id = ?), ?, ?, ?, ?)").bind(acqId, accountId, row.release_id, h, ts, accountId),
      db.prepare(`INSERT INTO redemption_attempts (attempt_id, credential_hash, account_id, outcome, attempted_at) SELECT ?, ?, ?, 'redeemed', ? WHERE ${REDEEMER_EXISTS_SQL}`).bind(mintAttId(), h, accountId, ts, accountId),
      // ADR-0006 §2: both analytics rows ride THIS batch, so the acquisition
      // and the event that counts it commit together or not at all. Production
      // `redeemed` is the vocabulary's `live` outcome.
      // RJb-R1-AS-025 (D-0920 option a): NOT for an unavailable creator — the fan keeps the
      // acquisition and the card, but a suppressed/retracted creator's analytics do not count it.
      ...(psp !== null && key !== null && row.creator_unavailable === 0 ? outboxStatements(db, { kind: "collectible_acquired", passportId: psp, occurredDay: occurredDay(ts), fanKey: key, scopeId: row.release_id }, { accountId }) : []),
      ...attemptEvent("live"),
      // S-CW2: the card-instance mint rides the SAME batch (see docblock). A
      // credential whose release row is unexpectedly missing (passport null —
      // the LEFT-JOIN tolerance above) still redeems, just without a card.
      ...(instanceMint && row.passport_id && gameOn
        ? (() => {
            ciId = instanceMint.mintInstanceId();
            return mintInstanceStatements(db, {
              instanceId: ciId,
              logId: instanceMint.mintLogId(),
              ownerId: accountId,
              passportId: row.passport_id,
              source: "redeemed",
              bound: false,
              ts,
            });
          })()
        : []),
      span.close,
    ]);
    seqs = outboxSpanSeqs(spanRes[0], spanRes[spanRes.length - 1]);
  } catch (e) {
    // The redeemer's account was erased between the reads above and this
    // batch: the fence rolled everything back and nothing is recorded for an
    // erased id — the same 401 the route gives a request with no account.
    if (e instanceof Error && REDEEMER_GONE_RE.test(e.message)) throw new CredentialError("unauthorized", 401);
    // Lost the race (acquisition UNIQUE) — single-use held, no double-credit.
    // The rolled-back batch took its analytics rows with it; this attempt is a
    // dead one and says so.
    await recordAttempt(db, h, accountId, "dead", mintAttId, ts, attemptEvent("dead"), pubEnv, pubCtx);
    return { ok: false, outcome: "dead" };
  }
  if (pubEnv && pubCtx) publishAfterCommit(pubEnv, pubCtx, seqs);
  // The mint is cap-guarded as a no-op inside the batch, so "did it happen"
  // is a read, not an assumption (review F8: a paid redemption that silently
  // skipped the card must at least SAY so).
  //
  // CO-11 — AND IT CANNOT REJECT A COMMITTED REDEMPTION. This read sat outside
  // the batch's `catch`, so a transient failure HERE — after the credential was
  // burned, the acquisition inserted and the card minted, all committed —
  // threw out of `redeemCredential` and the route's outer catch answered HTTP
  // 400. The fan owned their collectible and was told the redemption failed;
  // the achievement, referral, encounter and reward hooks (which run only on
  // `ok`) never ran. A display detail must never be able to overturn a
  // committed purchase, so its failure is `null` — "committed, mint
  // unconfirmed" — and not an exception.
  let instanceMinted: boolean | null | undefined;
  if (instanceMint) {
    try {
      instanceMinted = ciId !== undefined && (await db.prepare("SELECT 1 AS x FROM card_instances WHERE instance_id = ?").bind(ciId).first()) !== null;
    } catch {
      instanceMinted = null;
    }
  }
  return {
    ok: true,
    outcome: "redeemed",
    release_id: row.release_id,
    acquisition_id: acqId,
    ...subject,
    ...(instanceMinted !== undefined ? { instance_minted: instanceMinted } : {}),
    ...(instanceMinted === true && ciId ? { instance_id: ciId } : {}),
  };
}

/** The attempt row and its analytics event in ONE batch (ADR-0006 §2) — a
 * lone `.run()` beside an outbox INSERT is two writes with no relationship. */
async function recordAttempt(db: DbLike, credHashHex: string, accountId: string, outcome: "dead" | "unknown", mintAttId: () => string, ts: string, event: DbStatement[] = [], pubEnv?: AnalyticsTickEnv, pubCtx?: { waitUntil(p: Promise<unknown>): void }): Promise<void> {
  // I5-FX2-CRED: an erased account's attempt writes no row (the outbox rows
  // carry their own post-erasure guard, analytics-events.ts).
  const span = outboxSpan(db);
  const spanRes = await db.batch([
    span.open,
    db.prepare(`INSERT INTO redemption_attempts (attempt_id, credential_hash, account_id, outcome, attempted_at) SELECT ?, ?, ?, ?, ? WHERE ${REDEEMER_EXISTS_SQL}`).bind(mintAttId(), credHashHex, accountId, outcome, ts, accountId),
    ...event,
    span.close,
  ]);
  const seqs = outboxSpanSeqs(spanRes[0], spanRes[spanRes.length - 1]);
  if (pubEnv && pubCtx) publishAfterCommit(pubEnv, pubCtx, seqs);
}

export interface Acquisition {
  acquisition_id: string;
  release_id: string;
  acquired_at: string;
  /** AUDIT ITEM #22: the release's TITLE and STATUS. Owner-scoped surface, so
   * both are the holder's own to see — and without them `#/me` rendered a raw
   * clamped `release_id` and a date, which is neither a name a human recognises
   * nor a hint that a `withdrawn` item will never show on their public profile.
   * `title` is `null` only if the release row has gone, which the schema's FK
   * does not permit; the shape allows it rather than lying with an empty
   * string. */
  title: string | null;
  release_status: string | null;
  /** AS-120: does the CREATOR behind this item's release fold into "no longer
   * reachable" (operator-suppressed under review, or operator-retracted)?
   * Every OTHER surface already hides or denies this fact — `publicProfile`
   * drops a suppressed/retracted creator's releases from the public
   * collection (social.ts), and `verifyCollectible` 404s the proof page for
   * the same two conditions (social.ts) — but this list, the holder's OWN
   * record of what they bought, carried no such fold: it LEFT JOINs
   * `releases` only, so a delisted creator's item rendered as a completely
   * ordinary acquisition here while it silently vanished everywhere else. The
   * `withdrawn` case (a normal creator-initiated wind-down, audit #22) is
   * NOT folded in here — it stays visible and gets its own `release_status`
   * label, deliberately: suppression/retraction is an operator action against
   * the CREATOR, and unlike a withdrawal it is exactly the case the other two
   * surfaces already treat as "does not exist". `true` only when the
   * underlying release resolved (a release with no matching row leaves this
   * `false`, matching the existing `title: null` tolerance above). */
  creator_unavailable: boolean;
}

/** A fan's verified acquisitions (their collection proof).
 *
 * NOT FILTERED by status, deliberately. A withdrawn release is still a thing
 * this fan bought and still a valid acquisition (the S33 lock); hiding it here
 * would erase the buyer's own record of a purchase, which is the opposite of
 * the fix. It is LABELLED instead — see `release_status` — and the featured
 * picker built from this list uses that label to refuse an item the showcase
 * would silently drop. */
/**
 * The page key, and the reason it is a TUPLE rather than the bare timestamp the
 * two older cursors in this repo use.
 *
 * `acquisitions.acquired_at` is not unique — two codes redeemed in the same
 * millisecond share a stamp, and `credentials.test.ts` already covers one fan
 * holding several acquisitions of one release. A `WHERE acquired_at < ?` cursor
 * over a non-unique column DROPS every row that shares the boundary stamp with
 * the last row of the previous page: the client asks for the next page and the
 * rows in between are simply never served, with nothing anywhere reporting a
 * loss. On a fan's own record of what they bought, that is not an acceptable
 * page break.
 *
 * `acquisition_id` is the primary key, so `(acquired_at, acquisition_id)` is
 * TOTAL, which is the whole requirement — a keyset cursor needs a strict order
 * with no ties, not a meaningful one. These ids are RANDOM, not time-ordered
 * (`ids.ts`: "an identifier must not encode when it was created"; the sole
 * exemption is `event_id`), so rows sharing a millisecond come back in an
 * arbitrary-but-stable order. That is fine and it is deliberate: the alternative
 * is a time-encoding id, which this codebase spent a slice removing.
 * `0053` indexes exactly this tuple.
 */
export interface AcquisitionCursor {
  acquired_at: string;
  acquisition_id: string;
}

/** The default and maximum page sizes. The old fixed 100 stays the DEFAULT, so a
 * client that sends nothing gets the same page SIZE it got before this change.
 * Not the same BYTES: the ORDER BY gained the tie-break, so rows sharing a
 * millisecond may swap places. That is the fix, not a side effect — an order
 * that was arbitrary between equal stamps is exactly what made the old shape
 * unpageable. */
export const ACQUISITIONS_PAGE_DEFAULT = 100;
export const ACQUISITIONS_PAGE_MAX = 200;

/**
 * The shape of an `acq_` id — the id this service mints for an acquisition, and
 * THE one home for that shape. `social.ts` used to carry a byte-identical copy;
 * it imports this now. The claim "the gate agrees with the minter" is only worth
 * anything if there is one place for it to be wrong.
 */
export const ACQ_ID_RE = /^acq_[0-9a-f]{20}$/;

/**
 * The stamp half of the cursor, and it is deliberately a SANITY BOUND rather
 * than a format.
 *
 * The first draft pinned `new Date().toISOString()`'s exact shape. Review found
 * the dead end that creates: one row whose `acquired_at` is any other shape
 * makes EVERYTHING OLDER THAN IT permanently unreachable — the client echoes the
 * row's own stamp back and gets 400 forever, with no recovery path. And the
 * strictness bought nothing, because it was defending the wrong thing:
 *
 *   `ORDER BY acquired_at DESC` is a lexicographic TEXT sort, and the keyset
 *   predicate is a comparison over that same TEXT column with that same
 *   collation. A client that echoes back a value THIS ENDPOINT SERVED gets a
 *   correct, complete, non-duplicating page whatever the format is. A client
 *   that INVENTS a stamp gets an arbitrary boundary on its own list — which is
 *   already true of any invented cursor and is harmless, because `account_id`
 *   is bound first.
 *
 * So what is left to check is that the value is a plausible cursor at all: short,
 * printable, no control characters. Nothing here is a security control (it is a
 * bound parameter on the holder's own rows); it exists so an obviously-junk
 * cursor is a named 400 instead of a silently empty page.
 */
const CURSOR_AT_MAX = 64;

/** Is this a cursor `listAcquisitions` can page from? Exported so the route can
 * answer 400 rather than serve a page whose contents it cannot vouch for. */
export function isAcquisitionCursor(c: { acquired_at?: unknown; acquisition_id?: unknown }): c is AcquisitionCursor {
  if (typeof c.acquired_at !== "string" || typeof c.acquisition_id !== "string") return false;
  if (c.acquired_at.length === 0 || c.acquired_at.length > CURSOR_AT_MAX) return false;
  if (hasUnsafeIdentifierText(c.acquired_at)) return false;
  return ACQ_ID_RE.test(c.acquisition_id);
}

export async function listAcquisitions(db: DbLike, accountId: string, opts: { limit?: number; before?: AcquisitionCursor } = {}): Promise<Acquisition[]> {
  const capped = Math.max(1, Math.min(ACQUISITIONS_PAGE_MAX, opts.limit ?? ACQUISITIONS_PAGE_DEFAULT));
  const cols = `SELECT a.acquisition_id AS acquisition_id, a.release_id AS release_id, a.acquired_at AS acquired_at,
              rel.title AS title, rel.status AS release_status,
              ${CREATOR_UNAVAILABLE_SQL} AS creator_unavailable
         FROM acquisitions a LEFT JOIN releases rel ON rel.release_id = a.release_id`;
  // The keyset predicate, and the ORDER BY that has to match it exactly or the
  // page break lands somewhere the WHERE clause is not looking. Two statements
  // rather than a sentinel cursor, matching `auditList`: a sentinel for the
  // TUPLE case would have to invent a maximal `acquisition_id` as well as a
  // maximal timestamp, and a made-up id in a comparison is the thing this
  // function is being changed to stop doing.
  //
  // A ROW VALUE, `(a, b) < (?, ?)`, and not the expanded
  // `a < ? OR (a = ? AND b < ?)`. They are logically identical; their PLANS are
  // not, and the difference is the whole justification for `0053`. Measured with
  // EXPLAIN QUERY PLAN on the expanded form: no range constraint at all on one
  // data shape (every page restarts the backward scan at the account's newest
  // row and filters — O(offset)), and on another, after `ANALYZE`, a
  // `MULTI-INDEX OR` that materialises both legs and then adds
  // `USE TEMP B-TREE FOR ORDER BY` — a sort, which is precisely what the index
  // was added to remove. The row value plans as one seek:
  //   SEARCH a USING INDEX idx_acq_account_page (account_id=? AND (acquired_at,acquisition_id)<(?,?))
  // Row values are SQLite 3.15+ (2016); `test-fidelity/param-cap.d1.test.ts`
  // pins that real D1 both accepts this and picks that plan, because "the double
  // accepts it" is not the same claim.
  const order = `ORDER BY a.acquired_at DESC, a.acquisition_id DESC LIMIT ?`;
  const q = opts.before
    ? db
        .prepare(`${cols} WHERE a.account_id = ? AND (a.acquired_at, a.acquisition_id) < (?, ?) ${order}`)
        .bind(accountId, opts.before.acquired_at, opts.before.acquisition_id, capped)
    : db.prepare(`${cols} WHERE a.account_id = ? ${order}`).bind(accountId, capped);
  const { results } = await q.all<Omit<Acquisition, "creator_unavailable"> & { creator_unavailable: number }>();
  return results.map((r) => ({ ...r, creator_unavailable: r.creator_unavailable !== 0 }));
}

export interface AbuseSignal {
  release_id: string;
  dead_attempts: number; // dead attempts on THIS creator's issued credentials (unknown hashes can't be attributed, so they're excluded)
  distinct_accounts: number; // how many accounts attempted dead credentials
  reused_credentials: number; // credentials that saw >1 dead attempt (sharing/leakage)
}

/** Cron prune: drop old redemption attempts so the abuse log can't grow
 * unbounded (the signal only needs a recent window). */
export async function pruneRedemptionAttempts(db: DbLike, olderThanIso: string): Promise<number> {
  const res = await db.prepare("DELETE FROM redemption_attempts WHERE attempted_at < ?").bind(olderThanIso).run();
  return res.changes;
}

/**
 * Bounded abuse aggregate for a creator's release — one INPUT to a broader trust
 * model, never a public score. Only counts patterns, not identities.
 */
export async function redemptionAbuseSignal(db: DbLike, passportId: string, releaseId: string, sinceIso?: string): Promise<AbuseSignal> {
  // Attempts against credentials issued for THIS release by THIS creator; `sinceIso` bounds the window (I3-0846).
  const { results } = await db
    .prepare(
      `SELECT a.credential_hash AS ch, a.account_id AS acc
         FROM redemption_attempts a
         JOIN claim_credentials c ON c.credential_hash = a.credential_hash
        WHERE c.release_id = ? AND c.issued_by = ? AND a.outcome IN ('dead','unknown')${sinceIso === undefined ? "" : " AND a.attempted_at >= ?"}`,
    )
    .bind(...(sinceIso === undefined ? [releaseId, passportId] : [releaseId, passportId, sinceIso]))
    .all<{ ch: string; acc: string }>();
  const perCred = new Map<string, number>();
  const accounts = new Set<string>();
  for (const r of results) {
    perCred.set(r.ch, (perCred.get(r.ch) ?? 0) + 1);
    accounts.add(r.acc);
  }
  let reused = 0;
  for (const n of perCred.values()) if (n > 1) reused += 1;
  return { release_id: releaseId, dead_attempts: results.length, distinct_accounts: accounts.size, reused_credentials: reused };
}

export const REDEMPTION_ABUSE_WINDOW_MS = 24 * 60 * 60 * 1000;
/** I3-0846 (D-0920 option A): dead attempts on ONE release inside the window that page operators. Above the
 * per-account DEAD_REDEMPTIONS_PER_ACCOUNT_DAY (credentials-routes.ts), so reaching it takes several accounts. */
export const REDEMPTION_ABUSE_DEAD_THRESHOLD = 60;
/** Releases read per run (one signal read each). */
export const REDEMPTION_ABUSE_RELEASE_BUDGET = 50;
/** `sweep_cursor` row: `cursor_id` = the UTC day operators were last mailed. */
export const REDEMPTION_ABUSE_MAIL_CURSOR = "redemption_abuse_mailed";

/**
 * I3-0846 / RJ550b-W314-R8-7 — THE OPERATOR READ OF `redemptionAbuseSignal`. Every per-account ceiling
 * multiplies with the number of accounts; this watches the one pattern that does not: dead attempts on a
 * single release across accounts. For each release with a dead attempt in the last REDEMPTION_ABUSE_WINDOW_MS
 * (at most REDEMPTION_ABUSE_RELEASE_BUDGET), the windowed signal is read; a release at or over
 * REDEMPTION_ABUSE_DEAD_THRESHOLD is flagged. Flags are mailed to every operator address at most once per UTC
 * day: the day cursor is written only after at least one send was accepted, so a refused send is retried next
 * run. Counts only, never an account id. No fan-facing effect.
 */
export async function redemptionAbuseWatch(
  db: DbLike,
  send: EmailSender,
  operatorEmails: readonly string[],
  nowMs: number = Date.now(),
): Promise<{ flagged: (AbuseSignal & { passport_id: string })[]; mailed: number; failed: number; unconfigured: boolean }> {
  const sinceIso = new Date(nowMs - REDEMPTION_ABUSE_WINDOW_MS).toISOString();
  const { results } = await db
    .prepare(
      `SELECT DISTINCT c.release_id AS release_id, c.issued_by AS passport_id
         FROM redemption_attempts a
         JOIN claim_credentials c ON c.credential_hash = a.credential_hash
        WHERE a.attempted_at >= ? AND a.outcome IN ('dead','unknown')
        ORDER BY c.release_id, c.issued_by LIMIT ?`,
    )
    .bind(sinceIso, REDEMPTION_ABUSE_RELEASE_BUDGET)
    .all<{ release_id: string; passport_id: string }>();
  const flagged: (AbuseSignal & { passport_id: string })[] = [];
  for (const r of results) {
    const sig = await redemptionAbuseSignal(db, r.passport_id, r.release_id, sinceIso);
    if (sig.dead_attempts >= REDEMPTION_ABUSE_DEAD_THRESHOLD) flagged.push({ ...sig, passport_id: r.passport_id });
  }
  const out = { flagged, mailed: 0, failed: 0, unconfigured: operatorEmails.length === 0 };
  if (flagged.length === 0 || out.unconfigured) return out;
  const day = new Date(nowMs).toISOString().slice(0, 10);
  const last = await db.prepare("SELECT cursor_id FROM sweep_cursor WHERE name = ?").bind(REDEMPTION_ABUSE_MAIL_CURSOR).first<{ cursor_id: string }>();
  if (last?.cursor_id === day) return out;
  const lines = flagged.map((f) => `${f.release_id} (creator ${f.passport_id}): ${f.dead_attempts} dead attempts from ${f.distinct_accounts} accounts, ${f.reused_credentials} codes tried more than once`);
  const text = `Redemption abuse watch, last 24 h (threshold ${REDEMPTION_ABUSE_DEAD_THRESHOLD} dead attempts per release):\n\n${lines.join("\n")}`;
  for (const to of operatorEmails) {
    let ok = false;
    try {
      ok = await send({ to, subject: `Dhwaja ops: redemption abuse on ${flagged.length} release(s)`, text });
    } catch {
      ok = false;
    }
    if (ok) out.mailed += 1;
    else out.failed += 1;
  }
  if (out.mailed > 0) {
    await db
      .prepare("INSERT INTO sweep_cursor (name, cursor_seq, cursor_id, updated_at) VALUES (?, 0, ?, ?) ON CONFLICT(name) DO UPDATE SET cursor_id = excluded.cursor_id, updated_at = excluded.updated_at")
      .bind(REDEMPTION_ABUSE_MAIL_CURSOR, day, new Date(nowMs).toISOString())
      .run();
  }
  return out;
}

 * the `sales` row `saleId` (place it right AFTER that INSERT; it is predicated on the row). A lot's remaining is its net

 * Binds: accountId, ts (hold), accountId, saleId, saleId, credits, ts, credits, saleId, accountId.

const LOT_EDGE_SQL =
  "INSERT INTO credit_spend_lots (sale_or_tip_id, purchase_id, credits, rail, at) " +
  "WITH lots AS (SELECT p.purchase_id, p.rail, p.at, " +
  `MAX(0, ${LOT_NET_SQL} - (SELECT COALESCE(SUM(e.credits), 0) FROM credit_spend_lots e WHERE e.purchase_id = p.purchase_id)) AS r ` +
  `FROM credit_purchases p WHERE p.account_id = ? AND NOT ${HELD_LOT_SQL}), ` +

  "AND NOT EXISTS (SELECT 1 FROM credit_spend_lots e WHERE e.sale_or_tip_id = s.sale_id)), " +
  "w AS (SELECT purchase_id, rail, at, r, SUM(r) OVER (ORDER BY at, purchase_id ROWS UNBOUNDED PRECEDING) AS cum FROM lots), " +
  "v AS (SELECT purchase_id, rail, at, MIN(r, MAX(0, cum - (SELECT u FROM pre))) AS r2 FROM w), " +
  "x AS (SELECT purchase_id, rail, r2, SUM(r2) OVER (ORDER BY at, purchase_id ROWS UNBOUNDED PRECEDING) - r2 AS before FROM v) " +
  "SELECT ?, purchase_id, MIN(r2, ? - before), rail, ? FROM x " +
  "WHERE r2 > 0 AND before < ? AND EXISTS (SELECT 1 FROM sales WHERE sale_id = ? AND account_id = ?)";
export function lotEdgeStmt(db: DbLike, saleId: string, accountId: string, credits: number, ts: string): DbStatement {

  const row = await db
    .prepare(
      "SELECT COALESCE(SUM(credits),0) AS bal, COALESCE(SUM(CASE WHEN origin IN " + CASH_ORIGINS_SQL + " THEN credits ELSE 0 END),0) AS cash, " + CRYPTO_HELD_CREDITS_SQL + " AS held FROM credit_ledger WHERE account_id = ?",
    )
    .bind(accountId, atIso, accountId)
    .first<{ bal: number; cash: number; held: number }>();
  return Math.max(0, Math.min(row?.bal ?? 0, (row?.cash ?? 0) - (row?.held ?? 0)));
}

/**
 * #138 AF-10 / D-0938(1d) option A (I17 J5b). A dispute delivered BEFORE its purchase is granted maps to no
 * `credit_purchases` row, so `refunds.ts reverseGrant` records it as an `audit_log` row with this action (actor_id
 * `rail:<rail>`, item_id = the processor's payment ref, else the external txn id; detail JSON carries the dispute
 * `kind`). `grantCredits` asks for that row INSIDE its committing batch and refuses the grant.
 */
export const UNRECONCILED_DISPUTE_ACTION = "refund:unreconciled_dispute";
/** The record a refused grant leaves: one row per (rail, payment), item_id = the payment key; detail names the account (D-0942(1)). */
export const GRANT_REFUSED_DISPUTE_ACTION = "credit:grant_refused_dispute";
/** I17-Q1 (D-1030 (a) "grant the credits"; I29-LOOSE L4): the row refunds.ts `reverseGrant` writes when a dispute is WON
 * on a payment with an open marker and no purchase yet (same actor_id/item_id as the marker it answers). */
export const DISPUTE_WON_BEFORE_GRANT_ACTION = "refund:dispute_won_before_grant";
/** A marker for this payment: `?` x3 = `rail:<rail>`, the payment ref (or the txn id), the external txn id. I29-LOOSE L4:
 * a marker answered by a LATER won row (`audit_id` order, same actor and item) no longer refuses — the won dispute
 * returned nothing, so the grant that lands after it lands once (credit_purchases UNIQUE(rail, external_txn_id)). A
 * marker newer than the won row (a fresh dispute) refuses again. */
/** The dispute id a marker or won row's detail carries (refunds.ts writes `disputeId`), NULL when none. */
const DISPUTE_ID_OF = (a: string): string => `(CASE WHEN json_valid(${a}.detail) THEN json_extract(${a}.detail, '$.disputeId') END)`;
export const DISPUTE_BEFORE_GRANT_SQL =
  `EXISTS (SELECT 1 FROM audit_log m WHERE m.action = '${UNRECONCILED_DISPUTE_ACTION}' AND m.actor_id = ? AND m.item_id IN (?, ?) ` +
  `AND NOT EXISTS (SELECT 1 FROM audit_log w WHERE w.action = '${DISPUTE_WON_BEFORE_GRANT_ACTION}' AND w.actor_id = m.actor_id AND w.item_id = m.item_id ` +
  // D-1037(2) (RGM-02): a won row naming THIS marker's dispute id answers it in either order (a won delivered before its
  // open, refunds.ts reverseGrant's no-purchase won arm); rows without an id keep the audit_id order rule.
  `AND (w.audit_id > m.audit_id OR ${DISPUTE_ID_OF("w")} = ${DISPUTE_ID_OF("m")})))`;
/** The same marker from a LOST dispute (kind 'dispute') — the only kind refunds.ts bans on (an open never bans, D-0753). */
const LOST_DISPUTE_BEFORE_GRANT_SQL =
  `EXISTS (SELECT 1 FROM audit_log WHERE action = '${UNRECONCILED_DISPUTE_ACTION}' AND actor_id = ? AND item_id IN (?, ?) ` +
  "AND CASE WHEN json_valid(detail) THEN json_extract(detail, '$.kind') END = 'dispute')";
/** This payment was already refused for a dispute (a redelivered settlement): the ban is not applied a second time. */
const GRANT_REFUSED_SQL = `EXISTS (SELECT 1 FROM audit_log WHERE action = '${GRANT_REFUSED_DISPUTE_ACTION}' AND actor_id = ? AND item_id = ?)`;

/**
 * D-0938(1d) — THE CHARGEBACK TREATMENT FOR A GRANT REFUSED ON A PRIOR DISPUTE, for the grant's own batch, placed
 * AFTER the statement it refuses. No credits were granted, so there is nothing to freeze, claw or reverse (refunds.ts's
 * open arm freezes and its lost arm claws only what a purchase granted); what remains is refunds.ts's lost-dispute ban
 * (`reverseGrant` `banning`) and its `account:chargeback_banned` row, applied once per payment, then the refusal row.
 */
function disputeRefusalStmts(db: DbLike, grant: CreditGrant, ts: string): DbStatement[] {
  const actor = `rail:${grant.rail}`;
  const key = grant.paymentRef || grant.externalTxnId;
  const marker = [actor, key, grant.externalTxnId];
  return [
    db
      .prepare(`UPDATE accounts SET banned_at = ? WHERE account_id = ? AND banned_at IS NULL AND ${LOST_DISPUTE_BEFORE_GRANT_SQL} AND NOT ${GRANT_REFUSED_SQL}`)
      .bind(ts, grant.accountId, ...marker, actor, key),
    db
      .prepare(
        "INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, 'account:chargeback_banned', ?, ?, ?, 'system' " +
          `WHERE EXISTS (SELECT 1 FROM accounts WHERE account_id = ? AND banned_at = ?) AND ${LOST_DISPUTE_BEFORE_GRANT_SQL} AND NOT ${GRANT_REFUSED_SQL}`,
      )
      .bind(actor, JSON.stringify({ purchaseId: null, paymentKey: key, refusedGrant: true }), ts, key, grant.accountId, ts, ...marker, actor, key),
    db
      .prepare(
        "INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, ?, ?, ?, ?, 'system' " +
          `WHERE ${DISPUTE_BEFORE_GRANT_SQL} AND NOT ${GRANT_REFUSED_SQL} AND NOT EXISTS (SELECT 1 FROM credit_purchases WHERE rail = ? AND external_txn_id = ?)`,
      )
      .bind(
        actor,
        GRANT_REFUSED_DISPUTE_ACTION,
        // I17-FXA RGM-06/17 (D-0942(1)): the refusal row NAMES THE ACCOUNT, so a dispute that closes LOST after this
        // refusal (refunds.ts reverseGrant, no-purchase arm) can apply the chargeback ban to it. Erasure must scrub it
        // (account_data.ts deleteAccount — the I17-FXA hand-off).
        JSON.stringify({ externalTxnId: grant.externalTxnId, paymentRef: grant.paymentRef ?? null, credits: grant.credits, grossCents: grant.grossCents, accountId: grant.accountId }),
        ts,
        key,
        ...marker,
        actor,
        key,
        grant.rail,
        grant.externalTxnId,
      ),
  ];
}

export interface CreditGrant {

/**
 * Grant credits from a VERIFIED purchase — idempotent by construction. The
 * purchase row's UNIQUE external_txn_id is the dedup marker: a webhook
 * redelivery makes the second insert violate UNIQUE, the whole batch rolls
 * back, and we report `duplicate` rather than double-granting. The ledger grant
 * and the purchase row land in ONE atomic batch, so credits can never appear
 * without their paid-for record, or vice versa.
 *
 * CALLER CONTRACT: only call this AFTER the rail has authenticated the payment
 * (Stripe signature, Apple/Google receipt, crypto processor confirmation). The
 * amounts are the rail's settled figures, never client-supplied.
 */
export type GrantRefusal = "duplicate" | "frozen" | "disputed" | "refunded" | "crypto_cap" | "over_cap";
export async function grantCredits(
  db: DbLike,
  grant: CreditGrant,
  mintPurchaseId: () => string,
  mintEntryId: () => string,
  now: () => string = () => new Date().toISOString(),
): Promise<{ granted: boolean; reason?: GrantRefusal }> {
  return grantCreditsImpl(db, grant, mintPurchaseId, mintEntryId, now, false);
}

/** `won`: I17-Q1 — the grant `grantRefusedOnDisputeWin` makes for a payment whose dispute was WON after this grant was
 * refused on it. The dispute gate is then "the refusal row exists" instead of "no dispute marker". */
async function grantCreditsImpl(
  db: DbLike,
  grant: CreditGrant,
  mintPurchaseId: () => string,
  mintEntryId: () => string,
  now: () => string,
  won: boolean,
): Promise<{ granted: boolean; reason?: GrantRefusal }> {
  // F18 / #32 — VALUE DOES NOT FLOW INTO A CLOSED ACCOUNT.
  //
  // The freeze is folded into session resolution, which stops everything a FAN
  // BEARER TOKEN can reach — and settlement is not one of those things. A review
  // drove it: a Stripe/IAP settlement landing after termination credited 500 to
  // an account that had just been terminated, because the settling caller is a
  // webhook and there is no session to refuse. The forfeiture is a one-shot
  // event (`idem` is per-account and `terminated` is absorbing), so value
  // arriving afterwards is value that can never be taken again.
  //
  // Refused rather than queued: a settlement to a frozen account is a refund
  // question, and answering it by silently crediting an account nobody can
  // spend from is the worst of the three options.
  // SHAPE BEFORE STATE. `accountFrozenForCredit`'s second limb is an
  // existence test (`NOT EXISTS (SELECT 1 FROM accounts …)`, :81), and a
  // MALFORMED id has no `accounts` row either — asked first it answered
  // "frozen" for what is a caller bug, and the shape check below became
  // unreachable: the rail was told `{ok:true, granted:false}` (index.ts:1798)
  // for a settlement we never even parsed. A malformed id and a frozen
  // account are different facts and only one of them is a refund question.
  if (!ACCOUNT_RE.test(grant.accountId)) throw new WarInstanceError("bad account id", 500);
  if (!Number.isInteger(grant.credits) || grant.credits <= 0) throw new WarInstanceError("credits must be a positive integer", 500);
  if (!Number.isInteger(grant.grossCents) || grant.grossCents < 0 || !Number.isInteger(grant.netCents) || grant.netCents < 0) throw new WarInstanceError("bad amounts", 500);
  if (typeof grant.externalTxnId !== "string" || grant.externalTxnId.length === 0 || grant.externalTxnId.length > 200) throw new WarInstanceError("bad external txn id", 500);
  const owedExcess = grant.cryptoOwedExcess ?? 0;
  if (owedExcess !== 0 && (grant.rail !== "crypto" || !Number.isInteger(owedExcess) || owedExcess < 0)) throw new WarInstanceError("bad owed excess", 500);
  const obs = grant.observed ?? null;
  const centsOrNull = (v: number | null | undefined) => (v === null || v === undefined ? null : Number.isInteger(v) && v >= 0 ? v : Number.NaN);
  const observedGross = centsOrNull(obs?.grossCents);
  const observedFee = centsOrNull(obs?.feeCents);
  const observedCurrency = obs?.currency ?? null;
  if (Number.isNaN(observedGross) || Number.isNaN(observedFee) || (observedCurrency !== null && !/^[a-z]{3}$/.test(observedCurrency))) throw new WarInstanceError("bad observed amounts", 500);
  // D-0843(3) (I4-RG-MONEY-01/-02): a settlement this rail ALREADY granted is a
  // duplicate whatever the account's standing is now — asked before the freeze
  // arm and the ban arm, both of which would otherwise answer "frozen" (and the
  // freeze arm would write a refund_owed row for money that stayed credited).
  const alreadyGranted = async () =>
    (await db.prepare("SELECT 1 AS x FROM credit_purchases WHERE rail = ? AND external_txn_id = ?").bind(grant.rail, grant.externalTxnId).first()) !== null;
  if (await alreadyGranted()) return { granted: false, reason: "duplicate" };
  // D-0938(1d): which refusal a non-grant was, read AFTER the batch that decided it (the gate itself is in the SQL).
  const refusedForDispute = async () =>
    (await db.prepare(`SELECT 1 AS x FROM audit_log WHERE action = '${GRANT_REFUSED_DISPUTE_ACTION}' AND actor_id = ? AND item_id = ?`).bind(`rail:${grant.rail}`, grant.paymentRef || grant.externalTxnId).first()) !== null;
  const disputeMarker = [`rail:${grant.rail}`, grant.paymentRef || grant.externalTxnId, grant.externalTxnId];

  const refundWentOut = async () => (await db.prepare(`SELECT 1 AS x WHERE ${REFUND_WENT_OUT_SQL}`).bind(grant.externalTxnId, grant.rail).first()) !== null;
  const isCrypto = grant.rail === "crypto";
  // I17-Q2 (D-1030): a Stripe/store payment already refunded is refused like a disputed one. Crypto is not (I26-Q5).
  const refundGated = !isCrypto;
  const refundClause = refundGated ? ` AND NOT ${REFUND_BEFORE_GRANT_SQL}` : "";
  const refundBinds = refundGated ? [...disputeMarker, grant.grossCents] : [];
  const refusedForRefund = async () =>
    refundGated &&
    (await db.prepare("SELECT 1 AS x FROM audit_log WHERE action = ? AND actor_id = ? AND item_id = ?").bind(GRANT_REFUSED_REFUNDED_ACTION, `rail:${grant.rail}`, grant.paymentRef || grant.externalTxnId).first()) !== null;
  // I17-Q2: the refusal row, once per payment, in the batch that refused (a redelivery adds nothing).
  const refundRefusalStmts = (ts: string): DbStatement[] =>
    refundGated

          db

              `rail:${grant.rail}`,
              GRANT_REFUSED_REFUNDED_ACTION,
              JSON.stringify({ externalTxnId: grant.externalTxnId, paymentRef: grant.paymentRef ?? null, credits: grant.credits, grossCents: grant.grossCents }),
              ts,
              grant.paymentRef || grant.externalTxnId,
              ...refundBinds,
              GRANT_REFUSED_REFUNDED_ACTION,

              grant.paymentRef || grant.externalTxnId,
              grant.rail,
              grant.externalTxnId,

        ]
      : [];

  // below and (I22: I3-1029 a) by the post-batch arm when the freeze landed inside the window the pre-read cannot see.
  // I54 FX R-M6 (INV-077, D-1403): a refused crypto settlement is owed back WHOLE — the intended part plus the excess
  // past the lifetime cap (`cryptoOwedExcess`) — here and in the ban arm below.

      .prepare(

        // freeze must not leave a durable row under an erased id (INV-061).

          // D-0843(3): asked IN the statement too, so a grant of this txn that

  }

  const chainBinds = isCrypto ? cryptoChainBinds(grant.externalTxnId) : [];
  const capClause = isCrypto ? ` AND ${CRYPTO_CAP_OK_SQL}` : "";
  const capBinds = isCrypto ? [grant.accountId, grant.credits * CREDIT_BACKING_CENTS] : [];
  const refusedForCap = async () =>
    isCrypto &&
    (await db.prepare("SELECT 1 AS x FROM audit_log WHERE action = ? AND actor_id = 'rail:crypto' AND item_id = ?").bind(GRANT_REFUSED_CRYPTO_CAP_ACTION, grant.externalTxnId).first()) !== null;
  // I17-Q1: in `won` mode the dispute gate is "this payment's refusal row exists" (GRANT_REFUSED_SQL, ? x2).
  const disputeGate = won ? ` AND ${GRANT_REFUSED_SQL}` : ` AND NOT ${DISPUTE_BEFORE_GRANT_SQL}`;
  const disputeGateBinds = won ? [`rail:${grant.rail}`, grant.paymentRef || grant.externalTxnId] : disputeMarker;
  // D-1037(1) (RGM-01): a PARTIAL refund marker (refunds.ts, a refund delivered before this grant) does not refuse; the
  // grant lands and the refunded share is taken back IN THIS BATCH by the rows refunds.ts reverseOneGrant writes for a
  // partial refund of a fresh purchase: the reversal row (kind refund, floor(credits * refunded / gross), the marker's
  // event id and cash figure), the claw (idem `reversal:<purchase>`, none for a share of 0) and status 'refunded', so a
  // later refund tops it up and a later dispute freezes or takes the rest. No in-hold sale is walked: the claw is covered
  // by this grant (what is not is reserve_loss_cents, as there). Each row is predicated on THIS purchase row and on the
  // marker, so a refused grant writes none.
  const partialRefundStmts = (pid: string, pts: string): DbStatement[] => {
    if (!refundGated) return [];
    const m = refundBinds;
    const share = `(credits_granted * ${PARTIAL_MARKER(MARKER_REFUNDED_SQL)}) / gross_cents`;
    const held = `(SELECT COALESCE(SUM(credits),0) FROM credit_ledger WHERE account_id = ? AND reason NOT IN ('dispute_freeze', 'dispute_unfreeze'))`;
    return [
      db
        .prepare(
          "INSERT INTO credit_reversals (reversal_id, account_id, purchase_id, rail, external_txn_id, kind, credits_clawed, sales_reversed_cents, reserve_loss_cents, processor_event_id, at, cash_refunded_cents, refunded_currency) " +
            `SELECT ?, account_id, purchase_id, rail, external_txn_id, 'refund', ${share}, 0, MAX(0, ${share} - MAX(0, ${held})) * ${CREDIT_BACKING_CENTS}, ` +
            `COALESCE(${PARTIAL_MARKER("CASE WHEN json_valid(detail) THEN json_extract(detail, '$.processorEventId') END")}, 'unreconciled_refund'), ?, ` +
            `${PARTIAL_MARKER("CASE WHEN json_valid(detail) AND typeof(json_extract(detail, '$.cashRefundedCents')) = 'integer' THEN json_extract(detail, '$.cashRefundedCents') END")}, ${PARTIAL_MARKER(MARKER_CURRENCY_SQL)} ` +
            `FROM credit_purchases WHERE purchase_id = ? AND ${PARTIAL_MARKER(MARKER_REFUNDED_SQL)} IS NOT NULL`,
        )
        .bind(mintId("crv"), ...m, ...m, grant.accountId, ...m, pts, ...m, ...m, pid, ...m),
      db
        .prepare(
          "INSERT INTO credit_ledger (entry_id, account_id, kind, origin, credits, ref, idem, reason, at) " +
            "SELECT ?, r.account_id, 'reversal', 'refund_reversal', -r.credits_clawed, r.purchase_id, 'reversal:' || r.purchase_id, 'refund', ? FROM credit_reversals r WHERE r.purchase_id = ? AND r.credits_clawed > 0",
        )
        .bind(mintId("cle"), pts, pid),
      db.prepare("UPDATE credit_purchases SET status = 'refunded' WHERE purchase_id = ? AND status = 'granted' AND EXISTS (SELECT 1 FROM credit_reversals WHERE purchase_id = ?)").bind(pid, pid),
    ];
  };
  const purchaseId = mintPurchaseId();
  const entryId = mintEntryId();
  const freezeEntryId = isCrypto ? mintEntryId() : null;
  if (!CPU_RE.test(purchaseId) || !CLE_RE.test(entryId) || (freezeEntryId !== null && !CLE_RE.test(freezeEntryId))) throw new WarInstanceError("bad id shape", 500);
  const ts = now();
  try {
    await db.batch([
      db
        .prepare(
          // I3-1697: the chargeback ban is asked IN the statement, so a ban that
          // commits after a caller's `accountBanned` pre-read still refuses.
          "INSERT INTO credit_purchases (purchase_id, account_id, rail, external_txn_id, gross_cents, net_cents, credits_granted, status, payment_ref, at) SELECT ?, ?, ?, ?, ?, ?, ?, 'granted', ?, ? " +
            // I22 (I3-0023, I3-1029 a): the freeze is asked IN the statement too. The account must EXIST unbanned (an
            // erasure deletes the row, account_data.ts) and carry no frozen closure, so an erasure or a suspension that
            // commits after the `accountFrozenForCredit` pre-read above still refuses.
            "WHERE EXISTS (SELECT 1 FROM accounts WHERE account_id = ? AND banned_at IS NULL)" +
            ` AND NOT EXISTS (SELECT 1 FROM account_closures WHERE account_id = ? AND state IN ${FROZEN_CLOSURE_STATES_SQL})` +
            // D-0938(1d) (#138 AF-10, option A): a payment that already carries an unreconciled dispute is NOT granted —
            // asked IN the statement, so a dispute recorded after any pre-read still refuses. The ledger and observed
            // rows below are predicated on this row, and `disputeRefusalStmts` (last in this batch) applies the treatment.
            disputeGate +
            refundClause +
            // I40-MONEY (D-1169 answer 1; W0-MNY-02, every rail): a settlement whose refund went out is not granted.
            ` AND NOT ${REFUND_WENT_OUT_SQL}` +
            // I54 M1 a / M6 a (W0-M-04): a settlement refused for a cap stays refused on every re-delivery; only a
            // 'frozen' refusal is lifted by a later grant (D-1158).
            " AND NOT EXISTS (SELECT 1 FROM credit_settlement_refusals x WHERE x.external_txn_id = ? AND x.rail = ? AND x.reason <> 'frozen')" +
            capClause +
            // I54 M1 a (INV-072): the store rail's monthly cap, decided here, after the duplicate pre-read (W0-M-03).
            (grant.monthlyCapGate ? ` AND NOT ${grant.monthlyCapGate.sql}` : ""),
        )
        .bind(purchaseId, grant.accountId, grant.rail, grant.externalTxnId, grant.grossCents, grant.netCents, grant.credits, grant.paymentRef || null, ts, grant.accountId, grant.accountId, ...disputeGateBinds, ...refundBinds, grant.externalTxnId, grant.rail, grant.externalTxnId, grant.rail, ...capBinds, ...(grant.monthlyCapGate?.binds ?? [])),
      // Origin `purchased`: a rail settled money for this. idem = the RAIL's
      // transaction id (not the purchase_id, which is minted fresh per call),
      // so the journal itself refuses a second grant for one payment even if
      // `credit_purchases`' UNIQUE were ever relaxed (F2 review, Idempotency).
      db
        .prepare("INSERT INTO credit_ledger (entry_id, account_id, kind, origin, credits, ref, idem, reason, at) SELECT ?, ?, 'grant', 'purchased', ?, ?, ?, ?, ? WHERE EXISTS (SELECT 1 FROM credit_purchases WHERE purchase_id = ?)")
        .bind(entryId, grant.accountId, grant.credits, purchaseId, `purchase:${grant.rail}:${grant.externalTxnId}`, "rail:" + grant.rail, ts, purchaseId),
      // I28 JA-03: an open dispute on an earlier link of this crypto charge freezes this grant too (CRYPTO_CHAIN_FREEZE_SQL).
      ...(freezeEntryId !== null ? [db.prepare(CRYPTO_CHAIN_FREEZE_SQL).bind(freezeEntryId, ts, ...chainBinds, purchaseId)] : []),
      // I3-1213 / D-0920 (0196): the processor's own figures for THIS purchase, NULL where not known, written for
      // every new purchase and predicated on the purchase row this batch wrote — a refused (banned) grant leaves none.
      db
        .prepare("INSERT INTO credit_purchase_observed (purchase_id, observed_gross_cents, fee_cents, currency, recorded_at) SELECT ?, ?, ?, ?, ? WHERE EXISTS (SELECT 1 FROM credit_purchases WHERE purchase_id = ?)")
        .bind(purchaseId, observedGross, observedFee, observedCurrency, ts, purchaseId),
      ...partialRefundStmts(purchaseId, ts),
      ...(won
        ? [
            // I17-Q1: the refusal is resolved by THIS grant (once: the purchase row is UNIQUE per (rail, txn)).
            db
              .prepare(
                "INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, ?, ?, ?, ?, 'system' " +
                  "WHERE EXISTS (SELECT 1 FROM credit_purchases WHERE purchase_id = ?)",
              )
              .bind(`rail:${grant.rail}`, GRANT_REFUSED_DISPUTE_RESOLVED_ACTION, JSON.stringify({ purchaseId, externalTxnId: grant.externalTxnId, credits: grant.credits }), ts, grant.paymentRef || grant.externalTxnId, purchaseId),
          ]
        : disputeRefusalStmts(db, grant, ts)),
      ...refundRefusalStmts(ts),
      // I40-MONEY (D-1169 answer 1): a `refunded` row refuses the purchase INSERT above (REFUND_WENT_OUT_SQL), so the
      // I38-REFUNDSIG-4 mark of a refunded row as credited_after_lift is gone with it (it could no longer fire).
      // I39-CREDITS (D-1158 answer 1): this settlement's row is still refund_owed and the payer has NOT claimed it
      // (stripe.ts refundOwedStripe sets refund_claimed_at before its POST, 0275): the refund is cancelled — status
      // credited_after_lift with refund_cancelled_at — predicated on THIS purchase row, and the audit row directly after
      // is written only when that UPDATE changed the row. A claimed row is left owed (its refund may be on the wire).
      // I54 M6 a (INV-077, W0-M-08): the excess of THIS settlement past the lifetime crypto cap, owed back, keyed by this
      // settlement id and predicated on THIS purchase row (a refused grant owes it through crypto-rail.ts instead). It is
      // written before the cancel below, which is scoped to reason 'frozen' and so leaves it owed.
      // I54 FX R-M6 (D-1403, Rule 22(a) 1): the refusal table is keyed by the settlement id alone (0284). When this
      // purchase row landed, a crypto row of the key can only be a 'frozen' one still owed and unclaimed (the purchase
      // INSERT refuses any other reason, REFUND_WENT_OUT_SQL a refund that went out or was claimed, and UNIQUE a row
      // already credited after a lift), so it becomes the excess row in place, before the
      // D-1158 cancel below (scoped to 'frozen') can cancel it. Otherwise the excess row is inserted; a row of the key

      ...(owedExcess > 0
        ? [
            db
              .prepare(
                "UPDATE credit_settlement_refusals SET reason = 'crypto_cap_excess', credits = ?, gross_cents = ? WHERE external_txn_id = ? AND EXISTS (SELECT 1 FROM credit_purchases WHERE purchase_id = ?)",

              .bind(owedExcess, owedExcess * CREDIT_BACKING_CENTS, grant.externalTxnId, purchaseId),
            db
              .prepare(
                "INSERT INTO credit_settlement_refusals (external_txn_id, account_id, rail, gross_cents, credits, reason, at) SELECT ?, account_id, 'crypto', ?, ?, 'crypto_cap_excess', ? FROM credit_purchases WHERE purchase_id = ? AND NOT EXISTS (SELECT 1 FROM credit_settlement_refusals WHERE external_txn_id = ? AND rail = 'crypto')",
              )
              .bind(grant.externalTxnId, owedExcess * CREDIT_BACKING_CENTS, owedExcess, ts, purchaseId, grant.externalTxnId),
          ]
        : []),
      db
        .prepare(
          "UPDATE credit_settlement_refusals SET status = 'credited_after_lift', refund_cancelled_at = ?, resolved_at = ? WHERE external_txn_id = ? AND rail = ? AND reason = 'frozen' AND status = 'refund_owed' AND refund_claimed_at IS NULL AND EXISTS (SELECT 1 FROM credit_purchases WHERE purchase_id = ?)",
        )
        .bind(ts, ts, grant.externalTxnId, grant.rail, purchaseId),
      db
        .prepare(
          "INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, ?, s.status || ' rail=' || s.rail || ' gross_cents=' || s.gross_cents || ' credits=' || s.credits || ' purchase=' || ?, ?, s.external_txn_id, 'system' FROM credit_settlement_refusals s WHERE s.external_txn_id = ? AND changes() = 1",
        )
        .bind(`rail:${grant.rail}`, REFUND_OWED_CANCELLED_ACTION, purchaseId, ts, grant.externalTxnId),
      ...(isCrypto
        ? [
            // I26-Q5: granted on a reversed chain -> one row naming it, predicated on THIS purchase row.
            db
              .prepare(
                "INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT 'rail:crypto', ?, ?, ?, ?, 'system' " +
                  `WHERE ${CRYPTO_CHAIN_REVERSED_SQL} AND EXISTS (SELECT 1 FROM credit_purchases WHERE purchase_id = ?)`,
              )
              .bind(
                GRANTED_AFTER_REVERSAL_ACTION,
                JSON.stringify({ chargeId: chainBinds[0], externalTxnId: grant.externalTxnId, purchaseId, credits: grant.credits, grossCents: grant.grossCents }),
                ts,
                grant.externalTxnId,
                ...chainBinds,
                purchaseId,
              ),
            // I21-Q34: the cap refusal, once per chain key, only when no grant of it exists.
            db
              .prepare(
                "INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT 'rail:crypto', ?, ?, ?, ?, 'system' " +
                  `WHERE NOT (${CRYPTO_CAP_OK_SQL}) AND NOT EXISTS (SELECT 1 FROM credit_purchases WHERE rail = 'crypto' AND external_txn_id = ?) ` +
                  "AND NOT EXISTS (SELECT 1 FROM audit_log WHERE action = ? AND actor_id = 'rail:crypto' AND item_id = ?)",
              )
              .bind(
                GRANT_REFUSED_CRYPTO_CAP_ACTION,
                JSON.stringify({ externalTxnId: grant.externalTxnId, credits: grant.credits, grossCents: grant.grossCents }),
                ts,
                grant.externalTxnId,
                ...capBinds,
                grant.externalTxnId,
                GRANT_REFUSED_CRYPTO_CAP_ACTION,
                grant.externalTxnId,
              ),
          ]
        : []),
      // D-1037(4) (RGM-04, I3-1697): a ban committing after the caller's pre-read refused the purchase INSERT above; the
      // settlement is owed back, recorded HERE, atomically with the refusal (`bannedSettlementRefundOwedStmt`: no row when
      // this batch granted, when the account is not banned, or when the processor already returned the money).
      bannedSettlementRefundOwedStmt(db, { ...grant, grossCents: wholeGross, credits: wholeCredits }, ts),
    ]);
  } catch (e) {
    // UNIQUE(rail, external_txn_id) (idx_cpu_rail_txn, 0118): this rail's payment already granted. Idempotent.
    const msg = e instanceof Error ? e.message : String(e);
    if (/unique/i.test(msg)) return { granted: false, reason: "duplicate" };
    throw e;
  }
  // I3-1697: the purchase row exists only if the in-batch ban predicate let it
  // (the ledger row is gated on it). None -> refused like a closure freeze; the
  // rail's caller logs it (index.ts logFrozenSettlement).
  // D-0843(3): no row of OURS but one for this (rail, txn) -> another delivery
  // granted it (the ban predicate filtered our insert before UNIQUE could fire).
  // D-0938(1d): none, and this payment's dispute refusal row exists -> refused for the prior dispute.
  if (!(await db.prepare("SELECT 1 AS x FROM credit_purchases WHERE purchase_id = ?").bind(purchaseId).first())) {
    if (await alreadyGranted()) return { granted: false, reason: "duplicate" };
    if (!won && (await refusedForDispute())) return { granted: false, reason: "disputed" };
    if (await refusedForRefund()) return { granted: false, reason: "refunded" };
    // I40-MONEY (D-1169 answer 1): refused because this settlement's refund went out (decided by the INSERT above). The
    // settlement row records it (`redelivery_refused_at`, 0279, first refusal only) with one audit row gated on that
    // write; a retried delivery adds nothing. A lost write throws, so the rail redelivers and the record lands then.
    if (await refundWentOut()) {
      const rts = now();
      await db.batch([
        db
          .prepare(
            `UPDATE credit_settlement_refusals SET redelivery_refused_at = ? WHERE external_txn_id = ? AND rail = ? AND redelivery_refused_at IS NULL AND ${REFUND_WENT_OUT_SQL}`,
          )
          .bind(rts, grant.externalTxnId, grant.rail, grant.externalTxnId, grant.rail),
        db
          .prepare(
            "INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, ?, s.status || ' rail=' || s.rail || ' gross_cents=' || s.gross_cents || ' credits=' || s.credits || ' claimed=' || COALESCE(s.refund_claimed_at, 'no'), ?, s.external_txn_id, 'system' FROM credit_settlement_refusals s WHERE s.external_txn_id = ? AND changes() = 1",
          )
          .bind(`rail:${grant.rail}`, REDELIVERY_REFUSED_REFUND_OUT_ACTION, rts, grant.externalTxnId),
      ]);
      return { granted: false, reason: "refunded" };
    }
    if (await refusedForCap()) return { granted: false, reason: "crypto_cap" };
    // I54 M1 a (INV-072): the store cap refused the INSERT above (or a re-delivery met this settlement's cap refusal):
    // one refund_owed row keyed by the store transaction id, INSERT OR IGNORE (a re-delivery adds none), only while the
    // account EXISTS (INV-061) and no grant of (rail, txn) landed meanwhile (D-0843(3)). A frozen account is answered
    // by the freeze arm below instead.
    if (grant.monthlyCapGate && !(await accountFrozenForCredit(db, grant.accountId))) {
      await db
        .prepare(
          "INSERT OR IGNORE INTO credit_settlement_refusals (external_txn_id, account_id, rail, gross_cents, credits, reason, at) SELECT ?, ?, ?, ?, ?, 'over_cap', ? " +
            "WHERE EXISTS (SELECT 1 FROM accounts WHERE account_id = ?) AND NOT EXISTS (SELECT 1 FROM credit_purchases WHERE rail = ? AND external_txn_id = ?)",
        )
        .bind(grant.externalTxnId, grant.accountId, grant.rail, grant.grossCents, grant.credits, now(), grant.accountId, grant.rail, grant.externalTxnId)
        .run();
      if ((await db.prepare("SELECT 1 AS x FROM credit_settlement_refusals WHERE external_txn_id = ? AND rail = ? AND reason = 'over_cap'").bind(grant.externalTxnId, grant.rail).first()) !== null) return { granted: false, reason: "over_cap" };
    }
    // I22 (I3-1029 a): a freeze that landed after the pre-read refused the INSERT above; it owes the same refund_owed
    // row the pre-read arm writes (no row for an erased or merely banned account: the statement asks EXISTS accounts,
    // and a ban is not a closure).
    if (await accountFrozenForCredit(db, grant.accountId)) await frozenRefusalStmt(now()).run();
    return { granted: false, reason: "frozen" };
  }
  return { granted: true };
}

/** Per-origin balances — derivable, never stored. What a fan's value is made
 * of (purchased vs referral vs …); the class rights live in the writers, not
 * here. Lot accounting (which grant a spend consumed) is NOT modelled — the F2
 * known gap; fans never cash out, so nothing needs it yet. */
export async function balanceByOrigin(db: DbLike, accountId: string): Promise<Record<string, number>> {
  const { results } = await db
    .prepare("SELECT origin, COALESCE(SUM(credits),0) AS n FROM credit_ledger WHERE account_id = ? GROUP BY origin ORDER BY origin")
    .bind(accountId)
    .all<{ origin: string; n: number }>();
  const out: Record<string, number> = {};
  for (const r of results) out[r.origin] = r.n;
  return out;
}

/**
 * I3-0743 — WHICH precondition refused a `buyCard`, so the remedy offered fits
 * it. The guarded debit folds all four into ONE NULL PK (atomic, no race); this
 * read runs only AFTER that refusal, costs nothing on the success path, and only
 * picks the words — the money outcome was already decided by the batch. Order:
 * the three conditions no balance can fix are asked first, because when one of
 * them holds, "top up" is wrong whatever the balance says.
 *  - creator no longer eligible (BF-20 race) -> `not_in_game`, the SAME class
 *    and wording the pre-batch gate already answers (no new oracle);
 *  - daily cap or full shelf -> UNCLASSIFIED (no `reason`), which the client
 *    reads as "say nothing about money" (war-instances.ts WarInstanceError),
 *    with a sentence naming the fan's OWN limit;
 *  - otherwise -> `refused`, the one class a top-up remedies.
 * A failed diagnostic read falls back to an unclassified generic refusal:
 * never a top-up suggestion the code could not justify. */
async function classifyBuyRefusal(db: DbLike, accountId: string, passportId: string, dayStart: string, price: number, ts: string): Promise<WarInstanceError> {
  try {
    const row = await db
      .prepare(
        `SELECT (SELECT COUNT(*) FROM sales WHERE account_id = ? AND passport_id = ? AND product = 'direct_buy' AND at >= ?) AS day_n,
                (SELECT COUNT(*) FROM card_instances WHERE owner_id = ? AND consumed_at IS NULL) AS shelf_n,
                CASE WHEN ${WAR_ELIGIBLE_SQL} THEN 1 ELSE 0 END AS eligible,
                CASE WHEN ${CASH_GUARD_SQL} THEN 1 ELSE 0 END AS affordable`,
      )
      .bind(accountId, passportId, dayStart, accountId, ...warEligibleBinds(passportId, ts), ...cashGuardBinds(accountId, price, ts))
      .first<{ day_n: number; shelf_n: number; eligible: number; affordable: number }>();
    if (row) {
      if (!row.eligible) return new WarInstanceError("purchase refused: this creator is not currently in the game", 409, "not_in_game");
      if (row.day_n >= DIRECT_BUYS_PER_CREATOR_PER_DAY) return new WarInstanceError("purchase refused: you've already bought this creator's card today — nothing was charged", 409);
      if (row.shelf_n >= MAX_INSTANCES_PER_ACCOUNT) return new WarInstanceError("purchase refused: your card shelf is full — nothing was charged", 409);
      if (!row.affordable) return new WarInstanceError("purchase refused: insufficient credits — nothing was charged", 409, "refused");
    }
  } catch {
    // fall through: unclassified
  }
  return new WarInstanceError("purchase refused: nothing was charged", 409);
}

/** The guarded debit's refusal, by its own column (see the catch in `buyCard`). */
const BUY_GUARD_REFUSAL_RE = /NOT NULL constraint failed: credit_ledger\.entry_id\b/i;

/**
 * Buy ONE card of a creator with credits — the atomic spend. In ONE batch:
 *  - a guarded credit debit (a ledger 'spend' row whose PK is NULL unless the
 *    balance covers the price AND the daily cap is free → force-rollback);
 *  - the card instance mint (rides the same batch, cap-guarded no-op like every
 *    other mint);
 *  - the `sales` row attributing the creator's 80% with a 14-day clears_at.
 *
 * Fail-closed gates BEFORE the batch: the creator must be in the game
 * (warOptedIn) and rated (serveRatings) — an opted-out / unrated / suppressed
 * creator refuses with ONE generic message, no existence oracle. The 1/day/
 * creator cap is enforced INSIDE the batch (a pre-check would race) — and so,
 * as of BF-20, is standing: an opt-out, retraction or suppression that
 * commits between this pre-check and the batch is caught by `WAR_ELIGIBLE_SQL`
 * inside the same guarded debit, for the same reason the cap moved there
 * first. The pre-batch read still matters (it's what makes the refusal fast
 * and generic instead of a raw constraint failure) — it just isn't trusted
 * alone any more.
 */
export async function buyCard(
  db: DbLike,
  store: ArtifactReader,
  accountId: string,
  passportId: string,
  minters: { entryId: () => string; saleId: () => string; instanceId: () => string; logId: () => string },
  nowMs: number,
  now: () => string = () => new Date().toISOString(),
  pubEnv?: AnalyticsTickEnv,
  pubCtx?: { waitUntil(p: Promise<unknown>): void },
): Promise<{ instance_id: string; sale_id: string; balance: number }> {
  if (!ACCOUNT_RE.test(accountId)) throw new WarInstanceError("bad account id", 500);
  if (!PSP_RE.test(passportId)) throw new WarInstanceError("not for sale", 409);
  // Fail-closed game gate — suppressed folds into serveRatings; one refusal.
  // JN-02 (I34-W2-EXPIRY-B): both gates read the claim state at the buy's own instant `ts`, the instant the batch stamps.
  const ts = now();
  if (!(await warOptedIn(db, passportId, ts)) || !(await serveRatings(store, db, passportId, ts))) {
    // BF-11: a REASON, not only a sentence. The client used to append "You can
    // top up in your wallet" to every non-403 refusal, so an ineligible creator
    // was answered with a recommendation to spend money on a condition no
    // balance can fix. The two refusals are economically opposite and must be
    // distinguishable without parsing English.
    throw new WarInstanceError("not for sale: this creator is not currently in the game", 409, "not_in_game");
  }
  const price = CARD_PRICE_CREDITS;
  const entryId = minters.entryId();
  const saleId = minters.saleId();
  const instanceId = minters.instanceId();
  const logId = minters.logId();
  if (!CLE_RE.test(entryId) || !SAL_RE.test(saleId) || !CI_RE.test(instanceId) || !CIL_RE.test(logId)) throw new WarInstanceError("bad id shape", 500);
  const clearsAt = new Date(Date.parse(ts) + PAYOUT_HOLD_MS).toISOString();
  const netCents = creatorNetCents(price);
  const grossBacking = price * CREDIT_BACKING_CENTS;
  const dayStart = new Date(nowMs - (nowMs % (24 * 60 * 60 * 1000))).toISOString();

  const stmts: DbStatement[] = [
    // 1. The guarded debit — encodes EVERY precondition, so if any fails the
    //    NULL PK rolls the whole batch back (mint and sale included) and no
    //    money is taken. Three conditions, all atomic (no overspend, no
    //    cap-bypass by racing two buys, no money-for-no-card at the shelf cap):
    //      (a) the balance covers the price — the TOTAL and the CASH-BACKED
    //          sub-balance both (a store sale is creator cash; see CASH_ORIGINS),
    //      (b) the fan is under the 1/day/creator cap for this creator today,
    //      (c) the shelf has room — the SAME `< MAX` the mint's own guard uses,
    //          so a full shelf refuses the PURCHASE rather than silently
    //          no-oping the mint and pocketing the credits.
    //      (d) the creator is STILL eligible (BF-20): not retracted, not
    //          suppressed, and the war ledger's latest event is still an
    //          opt-in — re-asked here so a flip landing after the pre-batch
    //          `warOptedIn`/`serveRatings` read cannot buy its way through.
    db
      .prepare(
        `INSERT INTO credit_ledger (entry_id, account_id, kind, origin, credits, ref, idem, reason, at)
         SELECT CASE
                  WHEN ${CASH_GUARD_SQL}
                   AND (SELECT COUNT(*) FROM sales WHERE account_id = ? AND passport_id = ? AND product = 'direct_buy' AND at >= ?) < ?
                   AND (SELECT COUNT(*) FROM card_instances WHERE owner_id = ? AND consumed_at IS NULL) < ?
                   AND ${WAR_ELIGIBLE_SQL}
                  THEN ? ELSE NULL END,
                ?, 'spend', 'store_spend', ?, ?, NULL, 'direct_buy', ?`,
      )
      .bind(...cashGuardBinds(accountId, price, ts), accountId, passportId, dayStart, DIRECT_BUYS_PER_CREATOR_PER_DAY, accountId, MAX_INSTANCES_PER_ACCOUNT, ...warEligibleBinds(passportId, ts), entryId, accountId, -price, saleId, ts),
    // 2. The sale row — predicated on the debit having landed (its entry
    //    exists), so it can never record an unpaid sale.
    db
      .prepare(
        `INSERT INTO sales (sale_id, account_id, passport_id, product, credits_spent, gross_backing_cents, platform_cut_cents, creator_net_cents, instance_ids, clears_at, reversed, at)
         SELECT ?, ?, ?, 'direct_buy', ?, ?, ?, ?, ?, ?, 0, ?
          WHERE EXISTS (SELECT 1 FROM credit_ledger WHERE entry_id = ? AND ref = ?)`,
      )
      .bind(saleId, accountId, passportId, price, grossBacking, grossBacking - netCents, netCents, JSON.stringify([instanceId]), clearsAt, ts, entryId, saleId),
    // I20-Q1: which purchase(s) paid for this card, FIFO, predicated on the sale row (credit-lots.ts).
    lotEdgeStmt(db, saleId, accountId, price, ts),
    // 3. The card mint — also predicated on the debit, and cap-guarded as a
    //    no-op inside mintInstanceStatements (a full shelf is a read below, not
    //    a purchase-loss). source 'direct' marks a bought card.
    ...mintInstanceStatements(db, { instanceId, logId, ownerId: accountId, passportId, source: "direct", bound: false, ts }),
    // 4. The analytics row, in THIS batch (ADR-0006 §2). Creator-scoped and
    //    AMOUNT-BEARING, so it carries no fan pseudonym (INV-051: an amount or
    //    a pseudonym, never both — a row with both would be a per-person
    //    spending record).
    //
    //    THE AMOUNT IS THE GROSS (S-12): `gross_backing_cents`, what the fan
    //    paid, which is the sale row's own top line and the one figure the other
    //    two are a split OF. The split rides as dimensions — INV-051 refuses a
    //    second amount COLUMN, and allows an amount "by reference to a journal
    //    row"; `net_cents` and `platform_cut_cents` are that same row's numbers,
    //    not a second, fan-dimensioned amount. Sending only the net would leave
    //    `gross_credits` and `platform_cut_cents` with no writer at all, which
    //    is what B4b shipped and what this closes.
    //
    //    `clears_day` is the DAY of `clears_at`: the scheduled-clear cell is
    //    keyed by it, and the instant itself does not cross the queue.
    ...outboxStatements(db, {
      kind: "market_sale",
      passportId,
      occurredDay: occurredDay(ts),
      amountCents: grossBacking,
      dims: { clears_day: clearsAt.slice(0, 10), net_cents: netCents, platform_cut_cents: grossBacking - netCents },
    }),
  ];

  // The debit's CASE yields a NULL PK when any precondition fails (insufficient
  // balance, the daily cap, or a full shelf), and a NULL into the NOT-NULL
  // primary key throws and rolls the WHOLE batch back — sale and mint included,
  // so no credits move and no card mints. ONE refusal for all three (no oracle
  // about which limit, and none about the fan's exact balance).
  let seqs: number[] = [];
  try {
    const span = outboxSpan(db);
    const spanRes = await db.batch([span.open, ...stmts, span.close]);
    seqs = outboxSpanSeqs(spanRes[0], spanRes[spanRes.length - 1]);
  } catch (e) {
    // PS-070: don't flatten every batch failure into the domain refusal. The
    // ONLY expected failure mode is the guarded debit's NULL PK (insufficient
    // balance / daily cap / full shelf / now-ineligible creator, all folded
    // into ONE NOT NULL violation by design — see the CASE above). Anything
    // else — a timeout, a connection drop, an actual infra error — is a
    // 5xx-shaped incident, not a fan's "you can't afford this", and reporting
    // it as a 409 would tell an on-call dashboard "normal refusal traffic"
    // about an outage. `tips.ts`'s catch (tips.ts:166-175) inspects the
    // message the same way; this mirrors it rather than inventing a second
    // spelling of the same discipline.
    const msg = e instanceof Error ? e.message : String(e);
    // FX-I16-RG-MONEY:10 (I16-W13S): the refusal is the GUARDED DEBIT's own column (statement 1's CASE NULLs
    // credit_ledger.entry_id), not any NOT NULL — a NOT NULL fault on another table (sales, card_instances) is a
    // defect, not the fan's "you can't afford this". Same constraint packs.ts GUARD_REFUSAL_RE names.
    if (BUY_GUARD_REFUSAL_RE.test(msg)) throw await classifyBuyRefusal(db, accountId, passportId, dayStart, price, ts);
    throw e;
  }
  if (pubEnv && pubCtx) publishAfterCommit(pubEnv, pubCtx, seqs);
  return { instance_id: instanceId, sale_id: saleId, balance: await creditBalance(db, accountId) };
}

/**
 * I17-Q1 (D-1030 (a), from I29-REVERSALS): a dispute WON on a payment whose grant was refused for that dispute
 * (`GRANT_REFUSED_DISPUTE_ACTION`, written by `disputeRefusalStmts`) — grant the refused credits now, once. The figures
 * come off the refusal row's own detail (credits, grossCents, accountId, externalTxnId). The grant is the ordinary
 * committing batch (ban, freeze, refund, idempotency all asked in it) with the dispute gate replaced by "the refusal row
 * exists", and it writes one `GRANT_REFUSED_DISPUTE_RESOLVED_ACTION` row. A redelivered win is `duplicate`
 * (credit_purchases UNIQUE(rail, external_txn_id)). `paymentRef` is the refusal's item_id (the payment ref, else the txn).
 */
export async function grantRefusedOnDisputeWin(
  db: DbLike,
  rail: CreditGrant["rail"],
  paymentRef: string,
  mintPurchaseId: () => string = () => mintId("cpu"),
  mintEntryId: () => string = () => mintId("cle"),
  now: () => string = () => new Date().toISOString(),
): Promise<{ granted: boolean; reason?: GrantRefusal | "no_refusal" }> {
  if (typeof paymentRef !== "string" || paymentRef.length === 0 || paymentRef.length > 200) throw new WarInstanceError("bad payment ref", 500);
  const row = await db
    .prepare("SELECT detail FROM audit_log WHERE action = ? AND actor_id = ? AND item_id = ? ORDER BY rowid LIMIT 1")
    .bind(GRANT_REFUSED_DISPUTE_ACTION, `rail:${rail}`, paymentRef)
    .first<{ detail: string }>();
  if (!row) return { granted: false, reason: "no_refusal" };
  let d: { externalTxnId?: unknown; paymentRef?: unknown; credits?: unknown; grossCents?: unknown; accountId?: unknown };
  try {
    d = JSON.parse(row.detail);
  } catch {
    throw new WarInstanceError("bad refusal detail", 500);
  }
  // An erasure scrubs the account from the row (account_data.ts): no account, nothing to credit — refused as frozen.
  if (typeof d.accountId !== "string") return { granted: false, reason: "frozen" };
  if (typeof d.externalTxnId !== "string" || typeof d.credits !== "number" || typeof d.grossCents !== "number") throw new WarInstanceError("bad refusal detail", 500);
  const ref = typeof d.paymentRef === "string" && d.paymentRef.length > 0 ? d.paymentRef : undefined;
  return grantCreditsImpl(
    db,
    { accountId: d.accountId, rail, externalTxnId: d.externalTxnId, grossCents: d.grossCents, netCents: d.credits * CREDIT_BACKING_CENTS, credits: d.credits, ...(ref ? { paymentRef: ref } : {}) },
    mintPurchaseId,
    mintEntryId,
    now,
    true,
  );
}

/** I4-Q5: the correction's audit action (`audit_log.action`). */
export const CREDIT_CORRECTION_ACTION = "credit:correct";
/**
 * I4-Q5 (from I29-OPS): an operator's correction of a fan's credit balance, as statements for the caller's batch (the
 * dual-control executor). Positive: one `admin_grant` grant — NON-CASH, so it never becomes repayable cash or creator
 * money (D-0759) — refused on a terminated account (`NOT_TERMINATED_SQL`, D-0759(a)). Negative: one `admin_grant`
 * reversal, guarded on the total balance (it never drives the balance below zero) and ALSO refused on a terminated
 * account: lowering `total` there lowers the retained `MIN(total, cash)` owed back, the confiscating direction, which
 * needs a ruling (REPORT ## JASON). Either writes nothing when its guard
 * refuses; the `CREDIT_CORRECTION_ACTION` audit row is predicated on the ledger entry, so the two land together or not
 * at all — read `entryId` back to learn which. `opts.idem` (e.g. the pending-action id) makes a re-execution a UNIQUE
 * failure instead of a second correction. The per-correction cap is the caller's (I29-OPS).
 */
export function creditCorrectionStmts(
  db: DbLike,
  accountId: string,
  credits: number,
  operator: string,
  reason: string,
  opts: { entryId?: string; idem?: string | null; now?: () => string } = {},
): { stmts: DbStatement[]; entryId: string } {
  if (!ACCOUNT_RE.test(accountId)) throw new WarInstanceError("bad account id", 500);
  if (!Number.isInteger(credits) || credits === 0 || Math.abs(credits) > 1_000_000) throw new WarInstanceError("credits must be a non-zero integer", 500);
  if (typeof operator !== "string" || operator.length === 0 || operator.length > 200) throw new WarInstanceError("bad operator", 500);
  if (typeof reason !== "string" || reason.trim().length === 0 || reason.length > 500) throw new WarInstanceError("bad reason", 500);
  const entryId = opts.entryId ?? mintId("cle");
  if (!CLE_RE.test(entryId)) throw new WarInstanceError("bad id shape", 500);
  const ts = (opts.now ?? (() => new Date().toISOString()))();
  const idem = opts.idem ?? null;
  const ref = `correction:${entryId}`;
  const ledger =
    credits > 0
      ? db
          .prepare(
            `INSERT INTO credit_ledger (entry_id, account_id, kind, origin, credits, ref, idem, reason, at)
             SELECT ?, ?, 'grant', 'admin_grant', ?, ?, ?, 'credit_correction', ?
              WHERE EXISTS (SELECT 1 FROM accounts WHERE account_id = ?) AND ${NOT_TERMINATED_SQL}`,
          )
          .bind(entryId, accountId, credits, ref, idem, ts, accountId, accountId)
      : db
          .prepare(
            `INSERT INTO credit_ledger (entry_id, account_id, kind, origin, credits, ref, idem, reason, at)
             SELECT ?, ?, 'reversal', 'admin_grant', ?, ?, ?, 'credit_correction', ?
              WHERE EXISTS (SELECT 1 FROM accounts WHERE account_id = ?) AND ${TOTAL_FLOOR_SQL} AND ${NOT_TERMINATED_SQL}`,
          )
          .bind(entryId, accountId, credits, ref, idem, ts, accountId, accountId, -credits, accountId);
  const audit = db
    .prepare(
      `INSERT INTO audit_log (actor_id, action, detail, created_at, item_id, actor_kind) SELECT ?, '${CREDIT_CORRECTION_ACTION}', ?, ?, ?, 'operator' WHERE EXISTS (SELECT 1 FROM credit_ledger WHERE entry_id = ?)`,
    )
    .bind(operator, JSON.stringify({ entryId, credits, reason: reason.slice(0, 500) }), ts, accountId, entryId);
  return { stmts: [ledger, audit], entryId };
}
