// I42-COMPILE (I40-C1, D-1168(3), I42W0-01): a room dir that its batch's `_plan/INFLIGHT.jsonl` records as dispatched is
// never recompiled in place -- its PROMPT.md is the record of what the room was sent. Every door that writes into
// `org/07-findings/<batch>/<dir>/` is checked: the PROMPT path (default or --out) and --plan's --scope-out. INFLIGHT is read
// in the tree the file would be written to. A `dispatched` row protects its `room`, or the dir of its `prompt` when present;
// a later `retracted`/`withdrawn` row for the same room and attempt lifts it; `returned`/`resumed` rows do not. Refused
// (exit 2) BEFORE any write: no packet-log row, no PROMPT.md, no SCOPE.md, no audit snapshot. Runs before the QUEUE #864
// --out refusal below so an --out into a dispatched room's dir is refused under this rule's own message.
// --warm, --resume, --preflight and --contract-preflight write no room-dir file and are not checked here.
if (!a.includes("--warm") && !a.includes("--preflight") && !a.includes("--contract-preflight") && opt("resume") == null && opt("room") && opt("run")) {
  const runI = opt("run"); const dirI = opt("dir", opt("room"));
  const doors = [["PROMPT", opt("out", join(ROOT, "org/07-findings", runI, dirI, "PROMPT.md"))]];
  if (opt("plan")) doors.push(["--scope-out", opt("scope-out", join(ROOT, "org/07-findings", runI, dirI, "regression/SCOPE.md"))]);
  for (const [door, target] of doors) {
    const parts = resolve(target).split(sep);
    let at = -1;
    for (let k = parts.length - 4; k >= 1; k--) if (parts[k] === "07-findings" && parts[k - 1] === "org") { at = k; break; }
    if (at < 0) continue; // not inside any org/07-findings/<batch>/<dir>/ -- a scratch rebuild (D-1168(2))
    const roomDir = parts[at + 2];
    const inflight = [...parts.slice(0, at + 2), "_plan", "INFLIGHT.jsonl"].join(sep);
    if (!existsSync(inflight)) {
      console.warn(`warn I40-C1 (D-1168(3)): ${inflight} does not exist -- no dispatch record for ${roomDir}; compiled as never dispatched.`);
      continue;
    }
    const live = new Map();
    const rows = readFileSync(inflight, "utf8").split("\n");
    for (let n = 0; n < rows.length; n++) {
      if (!rows[n].trim()) continue;
      let r = null;
      try { r = JSON.parse(rows[n]); } catch { r = null; }
      if (!r || typeof r !== "object" || typeof r.room !== "string") {
        console.error(`compile-room-prompt: REFUSED (D-1168(3), I40-C1): ${inflight}:${n + 1} is not a JSON row with a string \`room\`, so whether ${roomDir} was dispatched cannot be read. Repair the row. NOTHING was written.`);
        process.exit(2);
      }
      const key = JSON.stringify([r.room, r.attempt ?? 1]);
      if ("dispatched" in r) {
        const promptDir = typeof r.prompt === "string" ? r.prompt.split(/[\\/]/).slice(-2, -1)[0] : undefined;
        if (!live.has(key)) live.set(key, []);
        live.get(key).push({ dir: promptDir || r.room, line: n + 1 });
      }
      if ("retracted" in r || "withdrawn" in r) live.delete(key);
    }
    const hit = [...live.values()].flat().find((e) => e.dir.toLowerCase() === roomDir.toLowerCase());
    if (hit) {
      console.error(`compile-room-prompt: REFUSED (D-1168(3), I40-C1): the ${door} write ${resolve(target)} is inside ${roomDir}/, which ${inflight}:${hit.line} records as dispatched (no later retracted or withdrawn row for that room and attempt). A dispatched room's PROMPT.md is the record of what it was sent and is never recompiled in place: compile a new attempt dir (--dir ${roomDir}-A<n>, recorded with "attempt" and "prompt"), continue it with --resume, or rebuild with --out outside the room dir. NOTHING was written.`);
      process.exit(2);
    }
  }
}

// #946(o), I19 INCIDENTS 19: THE ONE PACKET-LOG WRITE RESOLVER (the compile's append, --resume, --preflight). A compile
// whose --out lies OUTSIDE ROOT logs into the tree it was given, never into ROOT's real `org/07-findings/<run>/_packets.jsonl`:
// I19-TEACHBACK's replay ran `--root $HOME/mnt/dhwaja --out $TMP/replay/<room>.PROMPT.md --run I19-batch` and appended 21
// rows to I19-batch's real log. The row goes where the default layout puts it relative to the prompt — `<run>/_packets.jsonl`
// one level above `<run>/<dir>/PROMPT.md` — so a room directory still holds only its PROMPT.md (QUEUE #398). No --out is
// unchanged (an --out inside ROOT: I20-FX2 below); --packet-log names any file; --blind-model keeps its control log.
// I20-FXD (I20-RG-TOOLS-04, JN-B03 W1F-replay-control-logs): the --blind-model log and the --defer-packet stash follow
// the SAME --out rule — a compile whose --out lies outside ROOT writes neither into ROOT's org/08-control/blind-packets.jsonl,
// but `blind-packets.jsonl` beside that tree's `_packets.jsonl`. `outTree()` is the one place the rule is decided.
// I20-FX2 (I20-RG-TOOLS-03): an --out INSIDE ROOT is a replay too (`ROOT/.scratch/replay/<room>/PROMPT.md` appended to
// the real log before this): only a REAL dispatch — no --out, or an --out that IS the default prompt path
// `ROOT/org/07-findings/<run>/<dir>/PROMPT.md` — writes ROOT's logs. Every other --out, inside ROOT or not, logs one
// level above its prompt.
function outTree() {
  if (!opt("out", null)) return null;
  const out = resolve(opt("out"));
  return out === resolve(join(ROOT, "org/07-findings", run, dir, "PROMPT.md")) ? null : dirname(dirname(out));
}
function packetLogFor(blind) {
  if (opt("packet-log", null)) return resolve(opt("packet-log"));
  const tree = outTree();
  if (blind) return tree ? join(tree, "blind-packets.jsonl") : join(ROOT, "org/08-control/blind-packets.jsonl");
  if (tree) return join(tree, "_packets.jsonl");
  return join(ROOT, "org/07-findings", run, "_packets.jsonl");
}

  const section = (name) => { const parts = text.split(new RegExp(`^## ${name}\\b[^\\n]*\\n`, "m")); return parts.length > 1 ? parts[1].split(/^## /m)[0] : null; };
  const lesson = section("Lesson") ?? "";
  const version = (lesson.match(/^version:[ \t]*(\S+)/m) || [])[1];
  const head = version ? [`version: ${version}`] : [];
  const bullets = (sec) => sec.split("\n").map((l) => l.trimEnd()).filter((l) => /^- \S/.test(l));
  const checklist = bullets(section("Checklist") ?? "");
  if (checklist.length) {
    const pointers = lesson.split("\n").map((l) => l.trim());
    const body = checklist.slice(0, FAMILY_BODY_LINES).flatMap((c, i) => [c, ...pointers.filter((l) => Number((CHECKLIST_POINTER_RE.exec(l) || [])[1]) === i + 1)]);
    return { source: "CHECKLIST", body: [...head, ...body].join("\n") };
  }
  const lessonBullets = bullets(lesson);
  if (lessonBullets.length) return { source: `LESSON (first ${FAMILY_BODY_LINES} bullet lines; no ## Checklist)`, body: [...head, ...lessonBullets.slice(0, FAMILY_BODY_LINES)].join("\n") };
  const lines = lesson.split("\n").map((l) => l.trimEnd()).filter((l) => l.trim() && !/^version:/.test(l));
  return { source: `LESSON (first ${FAMILY_BODY_LINES} lines; no ## Checklist)`, body: [...head, ...lines.slice(0, FAMILY_BODY_LINES)].join("\n") };
}

// ---- ADR-0009 D4: render a scope packet from a PLAN.md sub-run's YAML block ----
function parseMiniYaml(text) {
  // Minimal YAML subset: "key: value", "key:" + indented "- item" lists, and
  // "key: [a, b]" flow lists. Enough for the fixed PLAN-SUBRUN.yaml.md shape;
  // NOT a general YAML parser.
  const out = {}; const lines = text.split("\n");
  let curKey = null;
  for (let i = 0; i < lines.length; i++) {
    const line = lines[i];
    if (!line.trim() || line.trim().startsWith("#")) continue;
    const listItem = line.match(/^\s+-\s*(.+)$/);
    if (listItem && curKey) { out[curKey] ??= []; out[curKey].push(stripQuotes(listItem[1].trim())); continue; }
    const kv = line.match(/^(\S[^:]*):\s*(.*)$/);
    if (!kv) continue;
    const key = kv[1].trim(); let val = kv[2].trim();
    curKey = key;
    // RUN-0015 G-11: `objective: >` / `harness_shape: |` are BLOCK SCALARS, which every sub-run in
    // scout/PLAN.md uses for exactly the two keys the room needs most. The value was read as ">"
    // and the indented body — matching neither the list nor the key:value shape — fell through
    // the loop and was lost. Folded (>) joins with a space, literal (|) keeps the newlines.
    if (/^[>|][-+]?$/.test(val)) {
      const indent = (line.match(/^\s*/) || [""])[0].length;
      const buf = [];
      while (i + 1 < lines.length) {
        const nxt = lines[i + 1];
        if (nxt.trim() && (nxt.match(/^\s*/) || [""])[0].length <= indent) break;
        buf.push(nxt.trim()); i++;
      }
      while (buf.length && !buf[buf.length - 1]) buf.pop();
      out[key] = val[0] === "|" ? buf.join("\n") : buf.join(" ").replace(/\s+/g, " ").trim();
      curKey = null;
      continue;
    }
    if (val === "") { out[key] = []; continue; } // list follows on subsequent lines
    if (val.startsWith("[") && val.endsWith("]")) { out[key] = val.slice(1, -1).split(",").map((s) => stripQuotes(s.trim())).filter(Boolean); continue; }
    if (val === "true") { out[key] = true; continue; }
    if (val === "false") { out[key] = false; continue; }
    out[key] = stripQuotes(val);
  }
  return out;
}
function stripQuotes(s) { return s.replace(/^["']|["']$/g, ""); }

function extractSubrun(planText, n) {
  const headingRe = new RegExp(`^###\\s*Sub-run\\s+${n}\\b.*$`, "m");
  const hm = headingRe.exec(planText);
  if (!hm) return null;
  const rest = planText.slice(hm.index);
  const nextHeadingIdx = rest.slice(hm[0].length).search(/^###\s|^##\s/m);
  const block = nextHeadingIdx === -1 ? rest : rest.slice(0, hm[0].length + nextHeadingIdx);
  const yamlMatch = block.match(/```yaml\n([\s\S]*?)```/);
  // RUN-0015 G-11's sibling, MEASURED in ORG-02. `prose` used to be `block.slice(0,
  // indexOf("```yaml"))`, so EVERYTHING BELOW THE FENCE WAS DISCARDED IN SILENCE. `SCOUT/PLAN.md`
  // puts a sub-run's `FILES (OWNS): ...` line BELOW the fence for sub-runs 1/2/3 (:28, :60, :116)
  // and ABOVE it for 4/5/6 (:165, :198, :229) -- so a fence-placement accident, not a spelling,
  // decided which rooms reached the write audit with a readable ownership list and which resolved
  // to `[]` and were charged with their siblings' writes. The fence is EXCISED and both sides are
  // kept: INV-060 PACKET COMPLETENESS / FAMILY-012 P1, every key or a named drop.
  const fenceAt = yamlMatch ? block.indexOf("```yaml") : -1;
  const prose = (fenceAt < 0
    ? block
    : block.slice(0, fenceAt) + block.slice(fenceAt + yamlMatch[0].length)
  ).replace(/\n{3,}/g, "\n\n").trimEnd();
  const yaml = yamlMatch ? parseMiniYaml(yamlMatch[1]) : null;
  return { heading: hm[0], prose, yaml };
}

  if (tests.length) out.push("**Driving tests** (the scout's list, verbatim):\n" + tests.map((t) => `- \`${t}\``).join("\n"));
  if (y.tier) out.push(`**Tier:** ${y.tier}`);
  if (tags.length) out.push(`**Scope tags:** ${tags.join(", ")}`);

  }
  const model = typeof y.model === "string" && y.model.trim() ? y.model.trim().toLowerCase() : null;
  return { text: out.join("\n\n"), invariants, dropped, model };
}

function verdictKey(outputText) {
  const lines = outputText.split("\n");
  const start = lines.findIndex((l) => /KEYED BY PERSONA FILE/.test(l));
  if (start < 0) return null;
  const entries = [];
  for (let i = start + 1; i < lines.length && lines[i].trim(); i++) {
    const m = lines[i].match(/^\s*((?:[a-z0-9-]+\.md,\s*)*[a-z0-9-]+\.md)(?::?\s+(.*))?$/);
    if (m) entries.push({ personas: m[1].split(",").map((f) => f.trim().replace(/\.md$/, "")), line: i + 1, text: m[2] ?? "" });
    else if (entries.length) entries[entries.length - 1].text += " " + lines[i].trim();
  }
  const key = new Map();
  for (const e of entries) {
    // The key's own negations ("It is NOT the judge's vocabulary", "— NOT PASS|FAIL") name OTHER
    // rooms' words; everything from the first upper-case NOT on is not this persona's vocabulary.
    const own = e.text.split(/\bNOT\b/)[0];
    const words = new Set();
    for (const g of own.matchAll(/`([^`]*)`/g)) for (const w of g[1].match(/\b[A-Z][A-Z_-]{2,}\b/g) ?? []) words.add(w);
    for (const g of own.matchAll(/[A-Z][A-Z_-]{2,}(?:\s*\\?\|\s*[A-Z][A-Z_-]{2,})+/g)) for (const w of g[0].split(/\s*\\?\|\s*/)) words.add(w);
    words.delete("VERDICT");
    for (const p of e.personas) key.set(p, { words: [...words], source: `org/rooms/_OUTPUT-CONTRACT.md:${e.line}` });
  }
  return key.size ? key : null;
}

}

function registerIdsIn(text, canonicalToIid = new Map()) {
  const src = String(text ?? "");
  const out = new Set();
  for (let i = 0; i < src.length; i++) {
    I3_HEAD.lastIndex = i;
    const m = I3_HEAD.exec(src);
    if (!m) continue;
    out.add(iidOf(m[1]));
    let at = I3_HEAD.lastIndex;
    for (;;) {
      I3_CONT.lastIndex = at;
      const c = I3_CONT.exec(src);
      if (!c) break;
      out.add(iidOf(c[1]));
      at = I3_CONT.lastIndex;
    }
    i = at - 1;
  }
  for (const id of extractIds(src)) if (/^I3-\d{3,8}$/.test(id)) out.add(iidOf(id.slice(3)));
  if (canonicalToIid.size) {
    const prefixes = new Set([...src.matchAll(/(?<![A-Za-z0-9])([A-Z][A-Z0-9]*):/g)].map((x) => x[1]));

    }
  }
  return [...out].sort();
}

// ADR-0011 D3 item 3: filter a SURFACE.md/GRAPH.md's own "### <file>" blocks (and, for GRAPH.md,
// its adjacency table rows) down to the files this room's scope packet actually names — a
// room reads only the surface/graph rows that touch its own files, never the whole sweep's.
function filterSurfaceOrGraph(text, packetText) {
  const files = [...new Set([...packetText.matchAll(/`([\w./-]+\.(?:ts|tsx|mjs|js))`/g)].map((m) => m[1]))];
  if (!files.length) return "";
  const blocks = text.split(/(?=^### `)/m).filter((b) => files.some((f) => b.startsWith(`### \`${f}\``)));
  const tableRows = text.split("\n").filter((line) => line.startsWith("|") && files.some((f) => line.includes(`\`${f}\``)));
  const parts = [];
  if (blocks.length) parts.push(blocks.join("").trim());
  if (tableRows.length) parts.push(tableRows.join("\n"));
  return parts.join("\n\n").trim();
}

// I33-FX2-COMPILE (I33-RG-ORG:14): a `--shadow` block of a family that is not active is headed with NOT_DELIVERED_MARK.
// It is a record (ADR-0008 D7), not a delivery: it takes no cap slot, satisfies no zero-family refusal and is never in
// `families`. The mark and deliveredFamilyIds live in scripts/lib/family-delivery.mjs, which family-outcomes.mjs also
// reads (I33-LAND1).
function lessonBlocks() {
  if (!lessonTags.length) return "";
  const dir = join(ROOT, "org/07-findings/families");
  // QUEUE #230: lessons were asked for (--lessons); an unreadable index delivered none, silently.
  if (!existsSync(join(dir, "INDEX.md"))) { console.error(`compile-room-prompt: ${join(dir, "INDEX.md")} cannot be read — --lessons ${lessonTags.join(",")} cannot be matched`); process.exit(1); }
  const idx = readFileSync(join(dir, "INDEX.md"), "utf8");
  const trig = idx.split("## Trigger")[1] ?? "";
  const fams = new Set();
  for (const line of trig.split("\n")) {
    const m = line.match(/^\|\s*([^|]+?)\s*\|\s*([0-9, ]+)\s*\|$/);
    if (!m) continue;
    const tags = m[1].toLowerCase();
    if (lessonTags.some((t) => tags.includes(t))) for (const n of m[2].split(",")) fams.add(`FAMILY-${n.trim().padStart(3, "0")}`);
  }
  const out = [];
  for (const f of [...fams].sort()) {
    const file = readdirSync(dir).find((x) => x === f + ".md" || x.startsWith(f + "-"));
    if (!file) { console.error(`compile-room-prompt: families/INDEX.md triggers ${f} but no ${f}.md exists in ${dir}`); process.exit(1); }
    const txt = readFileSync(join(dir, file), "utf8");
    const status = (txt.match(/^status:\s*(.+)$/m) || [])[1] ?? "";
    const active = /^active\b/.test(status);
    if (!active && !shadow) continue;
    const sec = txt.split(/^## Lesson\b[^\n]*\n/m)[1]?.split(/^## /m)[0]?.trim();
    if (!sec) { console.error(`compile-room-prompt: ${file} (${active ? "ACTIVE" : "shadow"}) has no "## Lesson" section — the matched lesson cannot be delivered`); process.exit(1); }
    const short = familyShortBody(txt);
    out.push({ id: f, delivered: active, block: `### ${f} (${file}) — ${active ? "ACTIVE" : "SHADOW — status: " + status + " — " + NOT_DELIVERED_MARK} — ${short.source}\n${short.body}` });
  }
  // I33 D-1089(2): the --lessons blocks share the room's cap with the matcher's (familyMatchBlock is handed these ids).
  // Only delivered blocks take a slot; NOT-delivered shadow records follow them, uncapped (I33-RG-ORG:14).
  const live = out.filter((x) => x.delivered);
  const kept = [...live.slice(0, maxFamilies), ...out.filter((x) => !x.delivered)];
  const over = live.slice(maxFamilies).map((x) => x.id);
  return kept.length ? "## LESSONS (ADR-0008 D3 — obligations and traps your scope triggers; report each as `lesson: FAMILY-NNN/vN — applied|adapted|not_applicable|conflicting|insufficient_context — <clause>`)\n" + kept.map((x) => x.block).join("\n\n") + "\n" +
    (over.length ? `MATCHED, NOT DELIVERED (over the cap of ${maxFamilies} families per room, D-1089(2)): ${over.join(", ")}\n` : "") : "";
}
// QUEUE #47: paste verbatim every docs/INVARIANTS.md row whose id the scope packet or the delivered lessons cite,
// so a room never opens the 83 KB registry. Rows are the `| INV-NNN | ... |` table lines.
// S-093 item 3 (QUEUE #368, D-0443): a cited row used to be pasted WHOLE. INV-004 is 14,219

// QUEUE #414 (W3-11a) — A CITATION IS CHECKED FOR RESOLUTION, NOT JUST FOR NOVELTY.
// Six stale citations reached five of nine builder rooms in ONE batch, and every one of them was
// verified dead BY THE ROOM THAT PAID TO CHASE IT: `index.ts:3323-3329` (`grep -c console.error`
// -> 0), `rewards.ts:281-283` (unrelated referral code), `registry-build.ts:1846` and `:1841-1849`

// The QUEUE #201 closed-row gate (closedRegisterRows, above) compares iids against the closure ledger; it never asks whether the
// `path:line` still RESOLVES. That question is mechanically decidable at bind time, it costs one
// stat and one line count per citation, and the build tree is now committed and clean (HEAD
// c4390a4) so "resolves at HEAD" is finally a meaningful question to ask of it.
//
// DECIDABLE here, and only this: the file exists; the line range is within it; the path is not one
// that resolves only inside vendored node_modules/. A citation may also carry `#token`, and then
// the token must still occur inside the cited range -- that is the `grep -c console.error -> 0`
// case, and it is checkable only when the packet says which token it meant.
// NOT decidable here: whether the lines still mean what the finding said. A citation pinned to a
// commit is reported as its own class rather than guessed at.
// MEASURED BEFORE THIS EXISTED, on W3-07-batch -- the very batch QUEUE #414 was filed from, nine
// builder rooms: the path-rooted regex saw 16 citations, ALL rangeless, and ZERO dead, while the
// same packets carried `analytics-consumer.ts:19-28`, `analytics-outbox.ts:787-794`,
// `events.ts:1563` and `main.ts:1347-1362` -- BASENAME form, with the line ranges, invisible to a
// regex that requires a `services/`-rooted path. A check whose population is zero on the batch
// that motivated it is decoration. So a basename is resolved against the tree, which is also the
// only way to decide #414's `cli.js`: a name that exists NOWHERE outside vendored node_modules.
let basenameIndex = null;
function indexBasenames(root) {
  if (basenameIndex) return basenameIndex;
  basenameIndex = new Map();
  const SKIP = new Set(["node_modules", ".git", "dist", "build", ".wrangler", "coverage", ".next"]);
  const walk = (d, depth) => {
    if (depth > 12) return;
    let entries;
    try { entries = readdirSync(d, { withFileTypes: true }); } catch { return; }
    for (const e of entries) {
      if (e.name.startsWith(".") && e.name !== ".github") continue;
      const full = join(d, e.name);
      // A SYMLINKED DIRECTORY IS A DIRECTORY. withFileTypes reports a symlink as neither
      // isDirectory() nor isFile(), so a tree that symlinks a package (or a test root that
      // symlinks the inputs) would be walked as empty and every basename in it would resolve
      // as "no file with this name at HEAD" — a check that invents dead citations is worse
      // than one that misses them. The depth cap and SKIP set bound the follow.
      let kind = e;
      if (e.isSymbolicLink()) {
        try { const st = statSync(full); kind = { isDirectory: () => st.isDirectory(), isFile: () => st.isFile() }; } catch { continue; }
      }
      if (kind.isDirectory()) { if (!SKIP.has(e.name)) walk(full, depth + 1); continue; }
      if (!kind.isFile()) continue;
      if (!basenameIndex.has(e.name)) basenameIndex.set(e.name, []);
      basenameIndex.get(e.name).push(full);
    }
  };
  walk(root, 0);
  return basenameIndex;
}
// QUEUE #683 (I7): fenced code blocks are quoted examples, not citations -- blank them, keeping line structure.
function blankFences(text) {
  let fence = null;
  return String(text).split("\n").map((line) => {
    // i7rg-05 (I7-FX2): only a backtick fence is blanked; a ~~~ line opens nothing, so its body is still scanned.
    // I42-COMPILE (I40-C7, I40TT-H1): a backtick line whose info string holds a backtick is inline code, not a fence
    // (CommonMark 4.5) -- it opens one only when fenceStep (findings-vocab.mjs) agrees, so later citations stay scanned.
    const m = /^ {0,3}(`{3,})/.exec(line);
    // I42-FX-COMPILE (P17-CMP02): the close is fenceStep's -- same character, at least the opener's length, nothing after but what trim() strips.
    if (fence === null) { if (m && fenceStep(null, line).fenceLine) { fence = m[1]; return ""; } return line; }
    if (fenceStep(fence, line).fence === null) fence = null;
    return "";
  }).join("\n");
}
function citationResolution(cites, root, skippedOut = []) {
  const dead = [];
  // A CHECK THAT INVENTS DEAD CITATIONS IS WORSE THAN ONE THAT MISSES THEM. `--root` is not
  // always the repo: committed fixtures compile against a synthetic root holding org/rooms and a
  // family corpus and nothing else (tests/tooling/compile-room-prompt-candidate-delivery.test.ts
  // builds seven of them), and against such a root EVERY `services/...` citation resolves as "no
  // such file" -- seven committed tests reddened on exactly that before this gate existed. So the
  // question is asked only where it can be answered: the citation's own top-level directory must
  // be present. Unanswerable is reported as unanswerable, never as dead.
  let top;
  try { top = new Set(readdirSync(root, { withFileTypes: true }).filter((e) => e.isDirectory() || e.isSymbolicLink()).map((e) => e.name)); } catch { top = new Set(); }
  const treePresent = top.has("services") || top.has("apps") || top.has("packages");
  const skipped = [];
  for (const { file, range, token } of cites) {
    const seg = file.includes("/") ? file.split("/")[0] : null;
    if (seg ? !top.has(seg) : !treePresent) { skipped.push(file); skippedOut.push({ file, range, token }); continue; }
    // BASENAME form (`events.ts:1563`) -- resolved against the tree, node_modules excluded.
    if (!file.includes("/")) {
      const hits = indexBasenames(root).get(file) ?? [];
      if (!hits.length) { dead.push({ file, range, why: "no file with this name at HEAD outside vendored node_modules/ (the QUEUE #414 `cli.js` case)" }); continue; }
      if (!range) continue;
      const [a0, b0] = range.replace(/^:/, "").split("-").map(Number);
      const hi0 = Number.isFinite(b0) ? b0 : a0;
      const fits = hits.filter((h) => { try { return readFileSync(h, "utf8").split("\n").length >= hi0; } catch { return false; } });
      if (!fits.length) {
        const sizes = hits.map((h) => { try { return readFileSync(h, "utf8").split("\n").length; } catch { return 0; } });
        dead.push({ file, range, why: `out of range in every file of that name at HEAD (${hits.length} candidate(s), ${Math.max(...sizes)} lines at most)` });
      }
      continue;
    }
    if (/(^|\/)node_modules\//.test(file)) { dead.push({ file, range, why: "resolves only inside vendored node_modules/ — not a reviewable path" }); continue; }
    const abs = join(root, file);
    if (!existsSync(abs)) { dead.push({ file, range, why: "no such file at HEAD" }); continue; }
    let lines;
    try { lines = readFileSync(abs, "utf8").split("\n"); } catch (e) { dead.push({ file, range, why: `unreadable at HEAD (${e.message})` }); continue; }
    const n = lines.length;
    if (range) {
      const [a, b] = range.replace(/^:/, "").split("-").map(Number);
      const hi = Number.isFinite(b) ? b : a;
      if (!Number.isFinite(a) || a < 1) { dead.push({ file, range, why: "malformed line range" }); continue; }
      if (a > n || hi > n) { dead.push({ file, range, why: `out of range — the file has ${n} lines at HEAD` }); continue; }
      if (token) {
        const body = lines.slice(a - 1, hi).join("\n");
        if (!body.includes(token)) { dead.push({ file, range, why: `\`${token}\` does not occur in those lines at HEAD (the QUEUE #414 \`grep -c\` case)` }); continue; }
      }
    } else if (token && !lines.join("\n").includes(token)) {
      dead.push({ file, range, why: `\`${token}\` does not occur in the file at HEAD` });
    }
  }
  if (skipped.length) {
    console.warn(`warn QUEUE #414: ${skipped.length} citation(s) NOT CHECKED — their tree is not under --root (${root}), so "does it still resolve" has no answer here: ${[...new Set(skipped)].slice(0, 6).join(", ")}${skipped.length > 6 ? " …" : ""}`);
  }
  return dead;
}
// W3-11a-R1 item 3 (QUEUE #412, open since W3-07) — THE ELISION THAT COST A CHILD-SAFETY RULING.
// An invariant auditor could not rule on whether a raw `passportId` reaches `audit_log.detail` on
// the MINOR-PREFILTER path because INV-061 arrived as 396 of 773 bytes and the CORE forbids
// opening `docs/INVARIANTS.md`. Two rooms stopped there independently; both were right to.
//
// MEASURED, not argued (node over all 61 rows of docs/INVARIANTS.md, 2026-09-20). The short form
// appends ~260 bytes of boilerplate to a <=400-byte head, so for a row under ~660 bytes IT IS
// LONGER THAN THE ROW IT REPLACES — and it was emitted anyway: `rows.push(short)` below was
// unconditional while the savings counter was guarded by `if (short.length < row.length)`, so the
// code could SEE the elision was costing bytes and shortened regardless.
//   16 of 61 rows: short form LONGER than the row, by up to 253 bytes (INV-019: 409 -> 662).
//   INV-061 itself: 773 -> 660, a 113-byte saving (14.6%) bought by deleting the operative clause.
// So the elision is now required to EARN its cut: it must save at least one head's worth, which
// is the least that can be called reading economy rather than truncation. Rows that cannot clear
// it are pasted WHOLE — the prompt gets SMALLER for 16 rows and grows by 27-113 bytes for the 5
// (INV-026/034/038/056/061) that were being truncated for pocket change. The genuinely large rows
// the economy was built for are untouched: INV-004 (14,219 B) and INV-051 (11,510 B) still elide.
// Rooms whose verdict IS a ruling about invariant text get the full row for the ids they were
// given, and get rows for their area even when the scope text names no id at all (QUEUE #362:
// four rooms across three batches filed `insufficient_context: INV-xxx` because no INVARIANT
// ROWS block was compiled unless the scope happened to name an id).
// QUEUE #412: the rooms whose VERDICT IS A RULING ABOUT INVARIANT TEXT, which is a strictly
// smaller set than the rooms that get rows at all. A judge rules on FINDINGS and reaches an
// invariant row only when a finding turns on one -- `rules-on:` is its channel, and
// `tests/tooling/compile-room-prompt-s093.test.ts:75` is the committed decision that says so.
// An invariant-auditor has no other subject: the row IS the thing it rules on, so eliding it
// does not economise the room, it ends it. See the R09 report for the judge limb, which is a
// decision this file is not the owner of.
function rulesOnIds(texts) {
  const ids = new Set();
  for (const t of texts) {
    // `rules-on: INV-051, INV-004` — one or more ids on a single line, anywhere in the packet.
    for (const m of t.matchAll(/^\s*(?:[-*]\s*)?rules-on:\s*([^\n]+)$/gim)) {
      for (const id of m[1].matchAll(/\bINV-(\d{3})\b/g)) ids.add(`INV-${id[1]}`);
    }
  }
  return ids;
}
function shortRow(row, id) {
  if (row.length <= ROW_HEAD_CHARS) return row;
  // Cut at the last cell boundary (` | `) inside the budget, else the last sentence end, else
  // the last space — never mid-word, and never inside a markdown table cell we opened.
  const head = row.slice(0, ROW_HEAD_CHARS);
  const cut = Math.max(head.lastIndexOf(" | "), head.lastIndexOf(". "), head.lastIndexOf(" "));
  const kept = (cut > ROW_HEAD_CHARS * 0.5 ? head.slice(0, cut) : head).trimEnd();
  // THE SUFFIX USED TO PRESCRIBE A READ THE CORE FORBIDS (QUEUE #412). It said "open it AT THAT
  // ID", while the CORE's reading discipline bans `docs/INVARIANTS.md` outright and its next
  // bullet says an invariant you lack is an `insufficient_context:` line, never a read. A room
  // caught between the two stops, and two did. The legal move now comes FIRST, and the line names
  // the NEXT ACTOR (FAMILY-003: an early exit states who can act next) — a room cannot lift its
  // own elision, but the orchestrator can re-compile with `rules-on:` and that is a request the
  // room can actually make.
  return `${kept} … [SHORT FORM — ${row.length - kept.length} of ${row.length} bytes elided. YOU MAY NOT OPEN docs/INVARIANTS.md (CORE reading discipline bans it). If your verdict turns on the elided text, write \`insufficient_context: ${id} — ruling on the row, need it whole\` and STOP on that point; the orchestrator re-compiles with \`rules-on: ${id}\` in the scope packet, which pastes this row whole. Do not infer the missing clause from the head.] |`;
}
// QUEUE #362 / S-093 item 3: the ruling-room INDEX. Derived from the registry's own table —
// `| INV-NNN | <area> | <text> |` — so it cannot go stale against a hand list. Rows whose AREA
// token appears in the scope packet are marked `<-- your area`; the rest are still listed,
// because a judge discovering that a row it was not pointed at applies is exactly the case
// QUEUE #362 was opened for.
function invariantIndex(texts, roomName) {
  const regPath = join(ROOT, "docs/INVARIANTS.md");
  let lines;
  try {
    if (!existsSync(regPath)) throw new Error("file does not exist");
    lines = readFileSync(regPath, "utf8").split("\n");
  } catch (e) {
    console.error(`compile-room-prompt: docs/INVARIANTS.md UNREADABLE at ${regPath} (${e.message}) — no invariant INDEX could be compiled for ${roomName}`);
    return "## INVARIANT ROWS — UNREADABLE: docs/INVARIANTS.md could not be read at compile time, so no index could be compiled. Report each id you need as `insufficient_context: INV-NNN — <why>`; do not treat their absence as \"no invariant applies\".\n";
  }
  const packet = texts.join("\n").toLowerCase();
  // Two tiers, because an index of all 61 rows with a head each measured 9.8 kB — more than the
  // short form saved anywhere else in the wave (S-093 measurement). Rows in the room's OWN area
  // get a one-line head; every other row is listed as id + area only (~35 bytes), which is still
  // the thing QUEUE #362 was opened for: a judge can SEE that a row exists and open it at its id.
  const mineRows = [];
  const restRows = [];
  for (const l of lines) {
    const m = l.match(/^\|\s*(INV-\d{3})\s*\|\s*([^|]*?)\s*\|\s*([\s\S]*)$/);
    if (!m) continue;
    const [, id, area, rest] = m;
    // An area token is matched only at >=4 characters: "money" and "privacy" discriminate,
    // "ops" and "id" would match half the English in a scope packet.
    const mine = area.split(/[^a-z0-9]+/i).some((t) => t.length >= 4 && packet.includes(t.toLowerCase()));
    // Score = how many distinct area tokens of this row the packet actually mentions, so the
    // heads go to the rows the scope is nearest, not to whichever 33 of 61 share a common word.
    const score = mine ? new Set(area.split(/[^a-z0-9]+/i).filter((t) => t.length >= 4 && packet.includes(t.toLowerCase())).map((t) => t.toLowerCase())).size : 0;
    if (score > 0) mineRows.push({ score, id, area, line: `| ${id} | ${area} | ${rest.replace(/\s+/g, " ").slice(0, INDEX_HEAD_CHARS).trim()}… |` });
    else restRows.push(`${id} (${area})`);
  }
  // MEASURED CAP (S-093): loose token matching put 33 of 61 rows in "your area" and the block
  // cost 5.9 kB — a cheaper way of being expensive. The heads are capped; everything else drops
  // to the id list, which still satisfies QUEUE #362 (the room can SEE the row and open it).
  mineRows.sort((x, y) => y.score - x.score || x.id.localeCompare(y.id));
  for (const r of mineRows.splice(INDEX_MAX_HEADS)) restRows.push(`${r.id} (${r.area})`);
  restRows.sort();
  if (!mineRows.length && !restRows.length) return "";
  console.log(`compile-room-prompt: ${roomName} scope named no INV id — compiled an INDEX instead of nothing: ${mineRows.length} row(s) in its area with heads, ${restRows.length} listed by id (QUEUE #362)`);
  return (
    `## INVARIANT ROWS — INDEX (QUEUE #362, S-093 item 3): your scope named no INV id and your verdict can turn on one, so this is the registry's index, not nothing.\n` +
    "It is NOT the rows. Open `docs/INVARIANTS.md` AT an id (a bounded read, never the 83 kB file) before ruling on it; file `insufficient_context: INV-NNN — <why>` for anything you needed and could not get.\n" +
    (mineRows.length ? `NEAREST YOUR AREA (${mineRows.length} of ${mineRows.length + restRows.length}, by area match):\n${mineRows.map((r) => r.line).join("\n")}\n` : "") +
    (restRows.length ? `EVERY OTHER ROW, by id (${restRows.length}) — open one at its id if your ruling reaches it: ${restRows.join(" · ")}\n` : "")
  );
}
function invariantRows(texts, { room: roomName = null } = {}) {
  const ids = new Set();
  // accepts INV-051 and the family files' shorthand INV-051/052/005
  for (const t of texts) for (const m of t.matchAll(/\bINV-(\d{3})((?:\/\d{3})*)\b/g)) {
    ids.add(`INV-${m[1]}`);
    for (const extra of (m[2] || "").split("/").filter(Boolean)) ids.add(`INV-${extra}`);
  }
  const rulesOn = rulesOnIds(texts);
  for (const id of rulesOn) ids.add(id);
  // QUEUE #362 (four rooms across three batches filed `insufficient_context: INV-xxx` because no
  // INVARIANT ROWS block was compiled at all unless the scope text happened to name an id): a
  // judge or auditor with no id named anywhere gets an INDEX instead of nothing — id, area and a
  // one-line head per row, so it can open exactly the row it needs at that id. An index of every
  // row is ~2.5 kB against the ~40 kB pasting them all would cost, so this is affordable enough
  // to be unconditional for those rooms.
  if (!ids.size && RULING_ROOMS.has(roomName)) return invariantIndex(texts, roomName);
  if (!ids.size) return "";
  // RUN-0015 G-11: a packet rendered from a PLAN sub-run already carries its rows verbatim;
  // emitting them again here would bill the room twice for the same 8 KB. Decided BEFORE the
  // registry is touched, so a row nobody is owed never reports the registry either way.
  const owed = [...ids].sort().filter((id) => !texts.some((t) => t.includes(`| ${id} |`)));
  if (!owed.length) return "";
  const regPath = join(ROOT, "docs/INVARIANTS.md");
  // R-2 (FINDING-0015-AUDIT-2), the treatment G-8 gave --surface/--graph: a registry that cannot
  // be read is SAID, never dropped. The CORE tells every room this block is authoritative and
  // that a row it needs but cannot see is an `insufficient_context:` line — which a room can
  // only write if it can tell "no invariant was cited" from "the registry was unreadable".
  let lines;
  try {
    if (!existsSync(regPath)) throw new Error("file does not exist");
    lines = readFileSync(regPath, "utf8").split("\n");
  } catch (e) {
    console.error(`compile-room-prompt: docs/INVARIANTS.md UNREADABLE at ${regPath} (${e.message}) — ${owed.length} owed row(s) cannot be pasted: ${owed.join(", ")}`);
    return (
      "## INVARIANT ROWS — UNREADABLE: docs/INVARIANTS.md could not be read at compile time, so the rows below are OWED and MISSING. Report each id you need as `insufficient_context: INV-NNN — <why>`; do not treat their absence as \"no invariant applies\".\n" +
      // Deliberately NOT `| INV-NNN | … |`: a row-shaped placeholder written into a rendered
      // SCOPE packet would satisfy the G-11 de-duplication above on the NEXT compile and
      // suppress the real row once the registry is readable again.
      owed.map((id) => `- ${id} — OWED, not pasted (registry unreadable at compile time)`).join("\n") +
      "\n"
    );
  }
  const rows = [];
  let shortened = 0;
  let savedBytes = 0;
  const keptWhole = []; // ids the elision could not earn its cut on — pasted whole, and SAID so
  for (const id of owed) {
    const row = lines.find((l) => l.startsWith(`| ${id} |`));
    if (!row) { rows.push(`| ${id} | (no row with this id in docs/INVARIANTS.md at compile time) |`); continue; }
    // FULL when the SCOPE names the row as one this room must RULE on -- OR when the ROOM's own
    // verdict is a ruling about invariant text (QUEUE #412, and #412's real limb).
    //
    // The head of this block already earned INV-061 its 773 bytes back by making the elision pay
    // for itself. That was necessary and not sufficient. `rulesOn` is populated only if whoever
    // wrote the scope packet REMEMBERED to type `rules-on: INV-061`, so the row an auditor gets
    // depended on the orchestrator's memory, and W3-11a-R1 measured what survives that: INV-004
    // (14,219 B) and INV-051 (11,510 B) still arrived elided TO THE AUDITORS RULING ON THEM --
    // large enough to clear MIN_ELISION_SAVING, which is exactly the size at which the operative
    // clause is most likely to be the part cut.
    //
    // The tension with room-economy rule 24 (findings are data, the digest is the channel) is
    // real and it resolves ONE way: the digest is PROSE relief, never EVIDENCE relief. A room
    // ruling on X is billed for X. The economy argument this comment used to make -- "not a
    // licence to be billed 14 kB for INV-004 on every turn" -- prices the wrong thing: 14 kB is
    // paid once, in a block the prompt layout (line ~1008) calls stable within a batch, and the
    // alternative is not 14 kB saved. It is an `insufficient_context:` line, a room that stops,
    // a judge that inherits the block and stops too (both did, on a child-safety audit path) and
    // a WHOLE RE-DISPATCH to recover the 396 bytes. That is the trade, and it is not close.
    //
    // Bounded, not open-ended: `owed` is already only the ids this room's own scope named and
    // does not already hold, so this pastes the rows the room was POINTED AT -- not the registry.
    // A ruling room pointed at nothing gets invariantIndex(), which this does not touch.
    const full = rulesOn.has(id) || ROWS_WHOLE_ROOMS.has(roomName);
    if (full) { rows.push(row); continue; }
    const short = shortRow(row, id);
    const saving = row.length - short.length;
    // THE ELISION MUST EARN ITS CUT. `rows.push(short)` used to run whatever this arithmetic
    // said, so a row the short form made LONGER was still truncated: the room paid more bytes
    // AND lost the clause (16 of 61 rows; INV-061, the child-safety row, among them).
    // A row under the head budget was never an elision CANDIDATE; reporting it beside a row whose
    // elision was REFUSED would make two different facts byte-identical in the log (INV-060).
    if (saving < MIN_ELISION_SAVING) { if (row.length > ROW_HEAD_CHARS) keptWhole.push(id); rows.push(row); continue; }
    shortened++; savedBytes += saving;
    rows.push(short);
  }
  if (!rows.length) return "";
  // BOTH POPULATIONS ARE PRINTED. A silent "7 rows shortened" could not be told apart from
  // "7 rows shortened and 5 more truncated for less than they cost" (INV-060 — an empty and an
  // absent are never byte-identical; here, an elision that fired and one that was refused).
  if (shortened) console.log(`compile-room-prompt: ${shortened} invariant row(s) in SHORT FORM, ${savedBytes} bytes elided (S-093 item 3; full rows go to \`rules-on:\` ids and ruling rooms)`);
  if (keptWhole.length) console.log(`compile-room-prompt: ${keptWhole.length} invariant row(s) pasted WHOLE — the elision would have saved < ${MIN_ELISION_SAVING} bytes (QUEUE #412): ${keptWhole.join(", ")}`);
  // The note is deliberately ONE line: prose in the volatile part of the prompt is billed on
  // every turn of every room, which is the defect this item exists to cut (S-093 measurement —
  // a 1.2 kB explanation here cost more across a wave than the short form saved).
  const note = rulesOn.size
    ? ` Whole rows (you RULE on these): ${[...rulesOn].sort().join(", ")}.`
    : ROWS_WHOLE_ROOMS.has(roomName)
      ? " Every row below is WHOLE: your verdict is a ruling about invariant text, so none of it is elided (QUEUE #412)."
      : "";
  const header =
    "## INVARIANT ROWS (docs/INVARIANTS.md at compile time — QUEUE #47; cite these, do not open the registry)\n" +
    `SHORT FORM over ${ROW_HEAD_CHARS} chars (S-093/#368): open docs/INVARIANTS.md AT the id if you must rule on one, and say so as \`insufficient_context:\`.${note}\n`;
  return header + rows.join("\n") + "\n";
}

function familyMatchBlock(packetText, persona, ownsPaths, roomFiles = ownsPaths, reserved = []) {

// ---- W3-11a-R1 item 2: THE ANTIPATTERN CORPUS GETS THE CHANNEL THE FAMILIES HAVE ----
//
// THE MEASUREMENT THIS EXISTS FOR. 24 antipatterns exist; THREE had a delivery channel —
// ANTI-024 hardcoded into `_PROMPT-CORE.md`, ANTI-004 into `_OUTPUT-CONTRACT.md`, ANTI-021 into
// `org/rooms/blue-builder.md`. A recent batch cited exactly those three (58/36/20 citations over
// 19 working rooms) and nothing else, while SIX others were instantiated in its actual defects
// and none was cited. The citation set is a pure function of the delivery set. The other 21 had a
// channel on paper — `scripts/stage-lessons.mjs <stage>` — but it is ORCHESTRATOR-side, and a
// room never runs it, which is why a room-side corpus reached rooms only where someone had
// pasted it into a prose block by hand.
//
// RANKED, NOT FILTERED, AND THAT IS NOT A STYLE CHOICE. The obvious build here is a selector:
// match `triggers:`/`applies_to:` and deliver the hits. This org has shipped that four times and
// it has delivered nothing four times — FAMILY-014 reached 1 of 25 rooms while OUT-SCORING a
// family that reached 24. The families' own fix (S-094-R1) was to stop letting a score decide
// MEMBERSHIP: every family is delivered, the channels set ORDER and EMPHASIS. This mirrors that,
// in two tiers so the corpus stays affordable:
//   BLOCK  — a DIRECT channel hit (persona / owned path / trigger tag): `## Trap` + `## Correction`.
//   ROSTER — everyone else: one line carrying id, stage, audience and the mechanism's head, so an
//            antipattern the room was not routed is still VISIBLE and nameable, never absent.
// An absent antipattern and an unrouted one are therefore never byte-identical (INV-060).
const ANTI_BLOCK_CHARS = 700;   // per section in the BLOCK tier
const ANTI_ROSTER_CHARS = 150;  // mechanism head in the ROSTER tier
function antipatternBlock(packetText, persona, ownsPaths, alreadyDelivered) {
  if (!doMatch) return "";
  const dir = join(ROOT, "org/07-findings/antipatterns");
  // Same posture as familyMatchBlock: no corpus directory is an ENVIRONMENT fact (a synthetic
  // root, a fresh checkout), not an operator error — and it is SAID, not silently skipped.
  if (!existsSync(dir)) {
    console.error("compile-room-prompt: org/07-findings/antipatterns/ not found — NO antipattern block compiled (environment fact, not a filter result)");
    return "";
  }
  const files = readdirSync(dir).filter((f) => /^ANTI-\d{3}\.md$/.test(f)).sort();
  if (!files.length) {
    console.error(`compile-room-prompt: ${dir} contains no ANTI-NNN.md — NO antipattern block compiled`);
    return "";
  }
  const fmOf = (t) => { const m = t.match(/^---\n([\s\S]*?)\n---\n/); const fm = {}; if (!m) return fm; for (const l of m[1].split("\n")) { const mm = l.match(/^([a-zA-Z_]+):\s*(.*)$/); if (mm) fm[mm[1]] = mm[2]; } return fm; };
  // First `## <name>...` section, whatever the heading's parenthetical says. ANTI-024 spells it
  // `## Correction (v2, 2026-09-18 ...)` and ANTI-019 `## Correction`; a `^## Correction$` anchor
  // would read the first as ABSENT and deliver an antipattern with no correction in it.
  // NO `m` FLAG, AND THAT IS THE WHOLE POINT. Under `m`, `$` matches an END OF LINE, so the lazy
  // `[\s\S]*?` stopped at the FIRST newline and every section the file WRAPS was delivered as its
  // first line — ANTI-003's trap arrived as 91 of 138 chars, cut mid-clause, and read as a
  // complete trap. Caught by diffing the two regexes over the corpus, not by reading the code.
  // (FAMILY-006's trap, exactly: a scan whose unit is the line while the subject wraps.)
  const sectionOf = (t, name) => { const m = t.match(new RegExp(`(?:^|\\n)## ${name}[^\\n]*\\n([\\s\\S]*?)(?=\\n## |$)`)); return m ? m[1].trim().replace(/\s+/g, " ") : ""; };
  const clip = (s, n) => (s.length <= n ? s : s.slice(0, s.lastIndexOf(" ", n) > n * 0.6 ? s.lastIndexOf(" ", n) : n).trimEnd() + " …");

  const rows = [];
  const excluded = {};
  for (const f of files) {
    const id = f.replace(".md", "");
    const text = readFileSync(join(dir, f), "utf8");
    const fm = fmOf(text);
    const status = (fm.status || "candidate").trim();
    const triggers = parseListField(fm.triggers);
    const appliesTo = parseListField(fm.applies_to);
    const paths = parseListField(fm.paths);
    // A CLOSED/retired antipattern would teach a withdrawn obligation. Named with its reason
    // rather than silently skipped — same rule the family branch follows.
    if (/^(closed|retired|superseded|withdrawn)\b/i.test(status)) { excluded[id] = `status: ${status}`; continue; }
    const trap = sectionOf(text, "Trap");
    const correction = sectionOf(text, "Correction");
    const mech = sectionOf(text, "Mechanism");
    if (!trap && !correction) { excluded[id] = 'no "## Trap" or "## Correction" section to deliver'; continue; }
    rows.push({
      id, status, stage: (fm.stage || "—").trim(), appliesTo, trap, correction, mech,
      tagHits: triggerHits(triggers, packetText),
      personaHitsOn: personaHits(appliesTo, persona),
      pathHitsOn: pathHits(paths, ownsPaths),
      // DERIVED, NEVER HAND-LISTED (FAMILY-012's trap is "a key list chosen by hand"). The three
      // antipatterns with a prose channel are found by asking whether the assembled CORE / output
      // contract / room contract ALREADY names this id — so the day a fourth is pasted into a
      // contract, or ANTI-024 is taken out of the CORE, this answer moves with it.
      alreadyIn: alreadyDelivered.includes(id),
    });
  }
  for (const r of rows) r.direct = r.tagHits.length + r.personaHitsOn.length + r.pathHitsOn.length;
  rows.sort((x, y) => (y.direct > 0) - (x.direct > 0) || y.direct - x.direct || x.id.localeCompare(y.id));

  const blocks = [];
  const roster = [];
  const channels = {};
  for (const r of rows) {
    const why = [];
    if (r.personaHitsOn.length) why.push(`applies_to: ${r.personaHitsOn.join(", ")}`);
    if (r.pathHitsOn.length) why.push(`paths: ${r.pathHitsOn.join("; ")}`);
    if (r.tagHits.length) why.push(`tag: ${r.tagHits.join(", ")}`);
    channels[r.id] = why.slice();
    // An id already carried VERBATIM by a stable block above is not pasted twice — the room has
    // it, and ANTI-024's own block is 4.6 kB. It still gets a roster line saying WHERE it is, so

    if (r.alreadyIn) { roster.push(`- ${r.id} (${r.stage}) — ALREADY IN YOUR PROMPT above (CORE / output contract / room contract); not repeated here.`); continue; }
    if (!r.direct) {
      roster.push(`- ${r.id} (${r.stage}${r.appliesTo.length ? `; for: ${r.appliesTo.join(", ")}` : ""}) — ${clip(r.mech || r.trap, ANTI_ROSTER_CHARS)}`);
      continue;
    }
    blocks.push(
      `### ${r.id} — ${r.status} (stage: ${r.stage}) — DELIVERY: DIRECT (${why.join("; ")})\n` +
      (r.trap ? `trap: ${clip(r.trap, ANTI_BLOCK_CHARS)}\n` : "") +
      (r.correction ? `correction: ${clip(r.correction, ANTI_BLOCK_CHARS)}\n` : ""),
    );
  }
  const header =
    "## ANTIPATTERNS (org/07-findings/antipatterns/, routed by this compiler — W3-11a-R1)\n" +
    "RANKED, NOT FILTERED, exactly as the families are: the whole corpus is below. A `applies_to`/`paths`/`tag` hit is a DIRECT channel and gets the trap and the correction; everything else is a ROSTER line you can still name and open at its id. These are ORG antipatterns — defects in how a room reasons and reports, not in the product. `not_applicable` is a complete answer to any of them.\n" +
    `DELIVERED: ${blocks.length} direct, ${roster.length} roster, of ${rows.length} deliverable.\n` +
    (Object.keys(excluded).length
      ? "EXCLUDED: " + Object.entries(excluded).map(([id, w]) => `${id} (${w})`).join("; ") + "\n"
      : "EXCLUDED: none — every antipattern in the corpus was delivered.\n");
  const body = (blocks.length ? blocks.join("\n") + "\n" : "") + (roster.length ? "ROSTER (no direct channel to this room):\n" + roster.join("\n") + "\n" : "");
  console.log(`compile-room-prompt: ANTIPATTERNS ${blocks.length} direct + ${roster.length} roster of ${rows.length} deliverable (${Object.keys(excluded).length} excluded), ${(header + body).length} bytes`);
  return header + body;
}

function findPersonaFile(id) {
  const dir = join(ROOT, "org/personas");
  if (!existsSync(dir)) return null;
  const walk = (d) => {
    let out = [];
    for (const e of readdirSync(d, { withFileTypes: true })) {
      const p = join(d, e.name);
      if (e.isDirectory()) out = out.concat(walk(p));
      else if (e.name.endsWith(".md") && e.name !== "_COMMON.md") out.push(p);
    }
    return out;
  };
  for (const f of walk(dir)) {
    const text = readFileSync(f, "utf8");
    const m = readFrontmatter(text); // I38-T4 (Q1000-B7): the shared LF/CRLF reader
    if (!m) continue;
    const idLine = m.fm.split("\n").find((l) => /^id:\s*/.test(l));
    const val = idLine ? idLine.replace(/^id:\s*/, "").replace(/^["']|["']$/g, "") : undefined;
    if (val === id) return { file: f, text };
  }
  return null;
}
function personaBlocks(id) {
  const found = findPersonaFile(id);
  if (!found) { console.error(`no persona with id "${id}" under org/personas`); process.exit(1); }
  const { text } = found;
  const fmMatch = readFrontmatter(text); // I38-T4 (Q1000-B7)
  const fm = fmMatch ? fmMatch.fm : "";
  const body = fmMatch ? fmMatch.body : text;
  const parts = body.split(/(^## .*$)/m);
  const uniqueSections = [];
  for (let i = 1; i < parts.length; i += 2) {
    const heading = parts[i].trim();
    const sectionBody = (parts[i + 1] ?? "").trim();
    if (/^<!-- COMMON: /.test(sectionBody)) continue; // dropped -- lives in PERSONA COMMON
    uniqueSections.push(`${heading}\n${sectionBody}`);
  }
  const fmList = fm.split("\n").filter(Boolean).map((l) => `- ${l}`).join("\n");
  if (!existsSync(join(ROOT, "org/personas/_COMMON.md"))) { console.error("compile-room-prompt: org/personas/_COMMON.md cannot be read — the PERSONA COMMON block would be empty"); process.exit(1); }
  const common = readFileSync(join(ROOT, "org/personas/_COMMON.md"), "utf8").replace(/^<!--[\s\S]*?-->\n\n/, "");
  const commonBlock = "## PERSONA COMMON (org/personas/_COMMON.md — identical for every persona room; cached)\n" + common.trim() + "\n";
  const idBlock = `## PERSONA ${id}\n${fmList}\n\n${uniqueSections.join("\n\n")}\n`;
  return { commonBlock, idBlock };
}

function ownsEntries(text) {
  const out = [];
  const secs = ownsSections(text);
  const bodies = ["heading", "label", "files"].flatMap((form) => secs.filter((s) => s.form === form).map((s) => s.body));

  }

}

function auditOwnsScope(text, derived, { exempt = () => false, tracked = trackedFiles } = {}) {
  const paths = [];
  const dropped = [];
  for (const p of derived) {
    if (String(p).includes("/")) { paths.push(p); continue; }
    const all = tracked();
    const hits = all ? all.filter((f) => f.split("/").pop() === p) : null;
    if (hits && hits.length === 1) paths.push(hits[0]);
    else dropped.push(`\`${p}\` (bare basename: ${hits === null ? "git ls-files unavailable under --root" : `names ${hits.length} tracked files`})`);
  }
  const derivedSet = new Set(derived.map(String));
  for (const e of ownsEntries(text)) {
    const toks = [...e.matchAll(/`([^`\n]+)`/g)].map((m) => m[1].trim().replace(/^\.\//, "").replace(/\/+$/, ""));
    if (!toks.length) { dropped.push(`"${e}" (no path token)`); continue; }
    // a glob token is ownsGlobGrants' to keep or name (I33-COMPILE-06)
    for (const t of toks) if (!derivedSet.has(t) && !exempt(t) && !GLOB_META_RE.test(t)) dropped.push(`\`${t}\` (not a repo path token; OWNS entry "${e}")`);
  }
  return { paths: [...new Set(paths)], dropped };
}
// I33-COMPILE-06 (I33-FX1-TOOLS): A GLOB OWNS ENTRY IS A GRANT. It used to land in `Scope (dropped)`, so the write audit
// resolved no grant for a room owning `org/07-findings/families/FAMILY-*.md` and charged its writes to every concurrent
// room. A glob that matches at least one file listed by listedFiles() — tracked or untracked, the write audit's listing —
// (globGrantRe, the validator's own reading) is kept, as the glob,
// on a `Scope (glob grants ...)` line that validate-room-report.mjs reads (globGrantsInPrompt). A glob that matches
// nothing, or is not a grant (no literal leading directory), is refused as an unresolved path is: named in
// `Scope (dropped)` and on stderr. Its reason never reads `(glob; OWNS entry`, the pre-I33-FX1 form the audit credits.
// I34-TOOLS-B (I33 B11): the entries are ownsLineEntries -- every line of the OWNS sections -- because parseOwns grants an
// exact path on a continuation line or a numbered item of a heading section, and a glob there used to be neither kept nor
// named (ownsEntries reads one line per `-`/`*`/`+` bullet).
function ownsGlobGrants(text, { exempt = () => false, listed = listedFiles } = {}) {
  const globs = [];
  const dropped = [];
  const seen = new Set();
  for (const e of ownsLineEntries(text)) {
    for (const m of e.matchAll(/`([^`\n]+)`/g)) {
      const t = m[1].trim().replace(/^\.\//, "").replace(/\/+$/, "");
      if (!GLOB_META_RE.test(t) || exempt(t) || seen.has(t)) continue;
      seen.add(t);
      const re = globGrantRe(t);
      const all = re ? listed() : null;
      const n = re && all ? all.filter((f) => re.test(f)).length : 0;
      if (n > 0) globs.push({ glob: t, n });
      else dropped.push(`\`${t}\` (glob; ${!re ? "not a grant: no literal leading directory, or malformed" : all === null ? "not matched: git ls-files unavailable under --root" : "matches no file under --root, tracked or untracked"}; OWNS entry "${e}")`);
    }
  }
  return { globs, dropped };
}

// I46-FXR-TOOLS (D-1307 F9; I46-JN-04 TO-028, I46-RG-TOOLS TOOLS:16): a token may also end before a dash that ends the word

/** The distinct tag tokens in `text` that `git rev-parse --verify` does not resolve to a commit in `repo`, in first-seen order. */
function deadTagTokens(text, repo) {
  const dead = [];
  for (const tok of new Set(tagTokens(text))) {
    const r = spawnSync("git", ["--no-optional-locks", "-C", repo, "rev-parse", "--verify", "-q", `${tok}^{commit}`], { encoding: "utf8" });
    if (r.status !== 0) dead.push(tok);
  }
  return dead;
}

  // relying on one.

  // stdout, and the orchestrator sends it as the second message -- the content in the message

  //   1. a --scope file INSIDE the room's own directory is REFUSED outright, because deferring
  //      the packet while leaving its source next door defers nothing;

  //      refused if it points inside the room's directory.

  // #393 left SCOPE.md in the room's directory. #408 closed that, and the three guards below were
  // written against ONE envelope: "inside the ROOM's own directory". #398 then found the same
  // material one `ls ..` up at the batch root, and #415 watched it GROW there during a judge's run.
  // MEASURED at HEAD c4390a4, BEFORE this change, with the #408 guards intact:

  // is what `ls .` and `ls ..` return: its own directory AND the batch directory holding it.

    const r = resolve(p);
    return r === batchDir || r.startsWith(batchDir + sep) || r === roomDir || r.startsWith(roomDir + sep);
  };

  // checks above test the raw argv `--scope`/`--packet-out`. The `--plan` path never sets either:

  // `org/07-findings/<run>/<dir>/regression/SCOPE.md` -- inside the room's own directory -- writes
  // it, and only then assigns the `scope` VARIABLE. So `opt("scope")` was null and the compiler
  // announced "deliberately not written to disk" in the same breath in which it reported writing
  // it. Measured: the stdout packet body and that SCOPE.md were both md5 17d895a48ce5....

  // directory -- so it is checked as one, over what is actually on disk, after every writer has
  // run. A predicate over flags can always be reached through a door nobody listed.
  // The scan is over the REACH, not the room directory: `ls .` (the room dir, recursively --
  // `--plan` renders a level down into regression/) and `ls ..` (the batch root's own files).
  // The batch root is scanned NON-recursively on purpose: its subdirectories are the SIBLING
  // rooms, whose own PROMPT.md legitimately carries their own packet, and a sibling sharing this
  // scope would otherwise refuse a dispatch that is not this flag's business to refuse. That
  // residual is stated in the R09 report rather than closed here.
  const scanRoots = [
    { root: roomDir, recursive: true },
    { root: batchDir, recursive: false },
  ];
  const needle = packet.trim().slice(0, 400);
  const leaked = [
    ...new Set(
      scanRoots.flatMap(({ root, recursive }) => {
        if (!existsSync(root)) return [];
        let entries;
        try {
          entries = readdirSync(root, { recursive, withFileTypes: true });
        } catch {
          return [];
        }
        return entries
          .filter((e) => e.isFile())
          .map((e) => join(e.parentPath ?? e.path ?? root, e.name))
          .filter((f) => f !== resolve(out))
          .filter((f) => {
            try {
              return readFileSync(f, "utf8").includes(needle);
            } catch {
              return false;
            }
          });
      }),
    ),
    // I33-COMPILE-05: --plan's SCOPE.md is written at commit, after this scan; a --scope-out in reach is a leak now.
    ...(planScopeBody !== null && inReach(resolve(scope)) ? [`${resolve(scope)} (--plan writes it at commit)`] : []),
  ];
  if (leaked.length) {
    console.error(
      `compile-room-prompt: --defer-packet, but the packet body is READABLE from the room's own directory or from \`ls ..\` at the batch root:\n` +
        leaked.map((f) => `    ${f}`).join("\n") +
        "\nA deferral the room can defeat with `ls` is not a deferral (QUEUE #393/#398/#408, W3-06 JUDGE J-01 and JUDGE2 J2-01).\n" +
        "If a renderer wrote it there (--plan renders to --scope-out, which DEFAULTS inside the room dir), point that\n" +
        "renderer outside org/07-findings/<run>/<dir>/ and re-run.",
    );
    process.exit(2);
  }
  const body = ["## SCOPE PACKET (message 2 of 2)", "", scopePathsLine + packet.trim(), "", rulingsSection, ...(censusSection ? [censusSection] : [])].join("\n");
  commitDeferred = () => {
    if (packetOut) {
      mkdirSync(dirname(resolve(packetOut)), { recursive: true });
      writeFileSync(packetOut, body);
      console.log(`OK: --defer-packet -- PROMPT.md carries no packet; send ${packetOut} as a SECOND message once the room reports \`## HYPOTHESES\` is on disk.`);
    } else {
      console.log("OK: --defer-packet -- PROMPT.md carries no packet. The packet follows on stdout; send it as a SECOND message once the room reports `## HYPOTHESES` is on disk. It is deliberately not written to disk (W3-06 JUDGE J-01).");
      console.log("----8<---- PACKET (message 2 of 2) ----8<----");
      console.log(body);
      console.log("----8<---- END PACKET ----8<----");
    }
  };
}

    } else {

      );
      refuse = true;

  if (verdictOverrideBlank) {
    console.error(`compile-room-prompt: --allow-verdict-mismatch needs a non-blank reason (QUEUE #505); ${WROTE_ON_REFUSAL}.`);
    refuse = true;

          `Remove the vocabulary from the packet, compile with the --room it was written for, or dispatch anyway with --allow-verdict-mismatch "<reason>", which is recorded in the packet log.`,
      );
      refuse = true;
    }
  }
  if (modelOverrideBlank) {
    console.error(`compile-room-prompt: --allow-model-mismatch needs a non-blank reason (QUEUE #866); ${WROTE_ON_REFUSAL}.`);
    refuse = true;
  }

      console.warn(`warn QUEUE #866: --dispatch-model ${dispatchModel} differs from ${modelMismatches.length} declared model(s), dispatched anyway (--allow-model-mismatch "${modelOverride}"):\n${lines}`);

          `Dispatch on the declared model, correct the declaration, or dispatch anyway with --allow-model-mismatch "<reason>" (a SPLIT PASS builder, a cost call), which is recorded in the packet log.`,
      );
      refuse = true;
    }
  }
  if (refuse) process.exit(2);
  commitPrompt();
  commitDeferred();
}
